def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, top_k=10, smooth_eps=0.0):
    """Compute weighted subgradient of max_k (a*a)[k].
    
    Returns (subgradient, max_value, active_lags)
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    if max_b < 1e-18:
        return np.zeros(n), 0.0, []
    
    # Get top-K lags
    k_eff = min(top_k, len(b))
    top_idx = np.argsort(b)[::-1][:k_eff]
    top_vals = b[top_idx]
    
    # Weight by value (exponential weighting)
    if smooth_eps > 0:
        shifted = top_vals - top_vals[0]
        weights = np.exp(shifted / smooth_eps)
    else:
        # Uniform weight over top-K, but boost the max
        weights = np.ones(k_eff)
        weights[0] = 2.0  # Boost the primary peak
    
    weights = weights / weights.sum()
    
    # Compute subgradient: sg[i] = sum_k w_k * 2 * a[k-i]
    sg = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        # a[k-i] is valid when 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        # sg[i] += 2*w * a[k-i] for i in [i_start, i_end)
        # a[k-i] for i in [i_start, i_end) is a[k-i_start : k-i_end+1] reversed
        # When i = i_start: a[k - i_start] = a[k - max(0, k-n+1)]
        # When i = i_end-1: a[k - (i_end-1)] = a[k - min(n-1, k)]
        
        # Simpler: just index directly
        i_range = np.arange(i_start, i_end)
        j_range = k - i_range  # = k-i for each i
        sg[i_range] += 2.0 * w * a[j_range]
    
    return sg, max_b, top_idx.tolist()


def nesterov_subgradient(a0, time_limit, top_k_start=5, top_k_end=30, 
                          c=1.0, smooth_eps=0.0, verbose=True, seed=42,
                          max_iters=200000):
    """Nesterov-accelerated subgradient method for minimizing max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    cur_top_k = top_k_start
    
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive top-K
        frac = min(1.0, t / 20000.0)
        cur_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)
        
        # Subgradient
        sg, M_curr, _ = compute_subgradient(y, top_k=cur_k, smooth_eps=smooth_eps)
        
        # Step size
        eta = c / (np.sqrt(t) + 1.0)
        
        # Projected subgradient step
        x_new = project_simplex(y - eta * sg)
        
        x_prev = x
        x = np.maximum(x_new, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            x = best_a.copy()
        else:
            x = x / sx
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            # Perturb
            noise_scale = 0.01 * (1.0 + 0.5 * rng.random())
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      NSG restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_k}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      NSG t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_k}, M={best_m:.2e}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"      NSG done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent polish: move mass from peak contributors to safe positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peak lags
        top_peaks = np.argsort(ac)[::-1][:min(10, len(ac))]
        
        # Compute contribution of each position to the peaks
        peak_contrib = np.zeros(n)
        for k in top_peaks[:5]:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            i_range = np.arange(i_start, i_end)
            j_range = k - i_range
            # Contribution of a[i] to b[k] is a[i] * a[j]
            contrib = a[i_range] * a[j_range]
            np.add.at(peak_contrib, i_range, contrib)
        
        # Find high-contribution sources
        sources = np.argsort(peak_contrib)[::-1][:20]
        improved = False
        
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Find safe destinations (low contribution to all peaks)
            # Score: how much would adding to position d increase the peaks?
            impact = np.zeros(n)
            for k in top_peaks[:5]:
                k = int(k)
                d = k - np.arange(n)  # For position d, the partner is k-d
                valid = (d >= 0) & (d < n)
                partner = np.clip(d, 0, n-1)
                impact[valid] += a[partner[valid]]
            
            # We want positions with low impact
            # Also prefer positions that are currently small
            score = impact + 0.1 * a  # Combined score (lower is better)
            
            # Try top-50 lowest-score destinations
            candidates = np.argsort(score)[:min(50, n)]
            best_dst = -1
            best_M = best_m
            
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.01, 0.005, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M:
                        best_M = M_trial
                        best_dst = dst
                        best_amt = amt
                
                if best_dst >= 0:
                    break
            
            if best_dst >= 0:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_M
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def normalize_seq(a):
    """Normalize to simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize_seq(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize_seq(f_interp)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated exact subgradient with adaptive top-K lag weighting
2. Multiple diverse initializations (parent, structured, random)
3. Adaptive length exploration (try longer sequences)
4. Coordinate descent polish with peak-aware mass redistribution
5. Multi-restart with perturbation strategies for escaping local minima
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, top_k=10, smooth_eps=0.0):
    """Compute weighted subgradient of max_k (a*a)[k] and return (sg, max_val)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()

    if max_b < 1e-18:
        return np.zeros(n), 0.0

    k_eff = min(top_k, len(b))
    top_idx = np.argsort(b)[::-1][:k_eff]
    top_vals = b[top_idx]

    if smooth_eps > 0:
        shifted = top_vals - top_vals[0]
        weights = np.exp(shifted / smooth_eps)
    else:
        weights = np.ones(k_eff)
        weights[0] = 2.0

    weights = weights / weights.sum()

    sg = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        i_range = np.arange(i_start, i_end)
        j_range = k - i_range
        sg[i_range] += 2.0 * w * a[j_range]

    return sg, max_b


def nesterov_subgradient(a0, time_limit, top_k_start=5, top_k_end=30,
                          c=1.5, smooth_eps=0.0, verbose=True, seed=42,
                          max_iters=500000):
    """Nesterov-accelerated subgradient method for minimizing max autocorrelation peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 1500

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 15000.0)
        cur_k = int(top_k_start + (top_k_end - top_k_start) * frac)

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        # Subgradient
        sg, M_curr = compute_subgradient(y, top_k=cur_k, smooth_eps=smooth_eps)

        # Adaptive step size: larger early, smaller late
        eta = c / (np.sqrt(t) + 1.0)

        # Projected subgradient step
        x_new = project_simplex(y - eta * sg)

        x_prev = x
        x = np.maximum(x_new, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            x = best_a.copy()
            x_prev = best_a.copy()
        else:
            x = x / sx

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise_scale = 0.005 + 0.01 * rng.random()
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      NSG restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_k}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      NSG t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_k}, M={best_m:.2e}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      NSG done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent: move mass from peak contributors to safe positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 40

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:min(8, len(ac))]

        # Compute per-position peak contribution
        peak_contrib = np.zeros(n)
        for k in top_peaks[:5]:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            i_range = np.arange(i_start, i_end)
            j_range = k - i_range
            contrib = a[i_range] * a[j_range]
            np.add.at(peak_contrib, i_range, contrib)

        # Find high-contribution sources
        sources = np.argsort(peak_contrib)[::-1][:15]
        improved = False

        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Score destinations by impact on peaks
            impact = np.zeros(n)
            for k in top_peaks[:4]:
                k = int(k)
                d = k - np.arange(n)
                valid = (d >= 0) & (d < n)
                partner = np.clip(d, 0, n - 1)
                impact[valid] += a[partner[valid]]

            score = impact + 0.05 * a
            candidates = np.argsort(score)[:min(40, n)]

            best_dst = -1
            best_M = best_m
            best_amt = 0.0

            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.01, 0.005, 0.015]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M:
                        best_M = M_trial
                        best_dst = dst
                        best_amt = amt
                if best_dst >= 0:
                    break

            if best_dst >= 0:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_M
                best_a = a.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def normalize_seq(a):
    """Normalize to simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving simplex normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize_seq(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize_seq(f_interp)

        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
