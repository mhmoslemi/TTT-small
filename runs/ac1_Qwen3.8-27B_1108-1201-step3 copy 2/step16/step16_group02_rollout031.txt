def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def logsumexp_objective_and_grad(z, n, tau):
    """
    Compute smooth objective and gradient w.r.t. z (log-domain parameterization).
    a = softmax(z), objective = (2n/tau) * logsumexp(tau * conv(a,a))
    """
    # Compute softmax
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    a = a / s
    
    # Compute convolution
    ac = fast_autocorr(a)
    
    # Compute logsumexp smoothly
    tau_ac = tau * ac
    max_val = tau_ac.max()
    lse = max_val + np.log(np.sum(np.exp(tau_ac - max_val)))
    
    obj = (2.0 * n / tau) * lse
    
    # Gradient w.r.t. a: (2n/tau) * 2 * sum_k w_k * a[k-i]
    w = np.exp(tau_ac - max_val)
    w_sum = w.sum()
    w = w / w_sum
    
    # grad_a[i] = (4n/tau) * sum_k w_k * a[k-i] (valid k)
    grad_a = np.zeros(n)
    for k in range(len(ac)):
        wk = w[k]
        if wk < 1e-15:
            continue
        k_int = k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad_a[idx] += 2.0 * wk * a[j_idx]
    
    grad_a *= (4.0 * n / tau)
    
    # Transform to z-space: grad_z = diag(a) @ (grad_a - a @ grad_a)
    # Actually: grad_z_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    inner = np.dot(grad_a, a)
    grad_z = a * (grad_a - inner)
    
    return obj, grad_z.copy()

            break
        
        if verbose:
            print(f"    LSE phase {phase+1}/{n_phases}, tau={tau:.1f}, time_budget={remaining_in_phase:.1f}s")
        
        def fun(z_val):
            obj, _ = logsumexp_objective_and_grad(z_val, n, tau)
            return obj
        
        def fun_grad(z_val):
            obj, grad = logsumexp_objective_and_grad(z_val, n, tau)
            return obj, grad
        
        try:
            result = minimize(fun_grad, z, method='L-BFGS-B', 
                            options={'maxiter': 2000, 'maxfun': 5000,
                                   'iprint': -1, 'maxls': 50})
            z_new = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B error: {e}")
            z_new = z
        
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        
        # Evaluate true objective
        ac = fast_autocorr(a_new)
        score = 2.0 * n * ac.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        z = z_new
        a = a_new
        
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
    
    # Final: also try the current a
    ac = fast_autocorr(a)
    score = 2.0 * n * ac.max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score / (2.0 * n)


def exact_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """
    Exact coordinate descent: for each coordinate, find the exact minimizer
    of J(a_i) = 2n * max_k(α_k * a_i + β_k) / (a_i + S')^2
    by checking all candidate k values.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    # Work with scale-invariant version (don't normalize)
    # J(a) = 2n * max_conv / (sum)^2
    
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 3
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep_improved = False
        coord_order = np.arange(n)
        rng.shuffle(coord_order)
        
        for i in coord_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute (a*a)[k] = α_k * a_i + β_k for all k
            # α_k = 2*a[k-i] if 0 <= k-i < n and k != 2i, else a[k-i] if k=2i (but then it's a_i^2)
            # Actually: (a*a)[k] = 2*a_i*a[k-i] + C_k for k≠2i, a_i^2 + C_{2i} for k=2i
            
            # Current convolution
            curr_ac = fast_autocorr(a)
            
            # For each k, the coefficient of a_i (linear part) and constant part
            # (a*a)[k] at current a_i: we know this from curr_ac
            # If we change a_i by δ: (a*a)[k] changes by 2*δ*a[k-i] (for k≠2i)
            
            # So: (a*a)[k](a_i) = curr_ac[k] + 2*(a_i_new - a_i)*a[k-i] for k≠2i
            #     (a*a)[2i](a_i) = curr_ac[2i] + 2*(a_i_new - a_i)*a[i] + (a_i_new-a_i)^2
            
            # Let's define: new (a*a)[k] = A_k * a_i_new + B_k for k≠2i
            # where A_k = 2*a[k-i] (if 0<=k-i<n) else 0
            # and B_k = curr_ac[k] - A_k * a[i]
            
            a_i_curr = a[i]
            
            # Compute A_k and B_k
            A = np.zeros(2*n - 1)
            B = curr_ac.copy()
            
            for k in range(2*n - 1):
                j = k - i
                if 0 <= j < n and k != 2*i:
                    A[k] = 2.0 * a[j]
                elif k == 2*i:
                    # Quadratic: (a*a)[2i] = a_i^2 + C_{2i}
                    # At current: curr_ac[2i] = a_i_curr^2 + C_{2i}
                    # So C_{2i} = curr_ac[2i] - a_i_curr^2
                    # New: a_i_new^2 + C_{2i} = a_i_new^2 + curr_ac[2i] - a_i_curr^2
                    pass  # Handle separately
                else:
                    A[k] = 0.0
            
            B = curr_ac - A * a_i_curr
            # For k = 2i:
            k_2i = 2 * i
            if 0 <= k_2i < 2*n - 1:
                C_2i = curr_ac[k_2i] - a_i_curr * a_i_curr
                B[k_2i] = C_2i  # (a*a)[2i] = a_i^2 + C_2i, so B=0 for linear, handle quadratic separately
                A[k_2i] = 0.0  # No linear term for the quadratic part
            
            S_prime = s - a_i_curr  # Sum of all other coordinates
            
            # Now minimize J(a_i_new) = 2n * max_k f_k(a_i_new) / (a_i_new + S_prime)^2
            # where f_k(a_i_new) = A[k]*a_i_new + B[k] for k≠2i
            # and f_{2i}(a_i_new) = a_i_new^2 + C_2i
            
            # Candidate optimal values:
            # For each k (linear): a_i* = S_prime - 2*B[k]/A[k] (if A[k] > 0)
            # For k=2i (quadratic): a_i* = C_2i / S_prime (if S_prime > 0)
            
            candidates = [a_i_curr, 0.0]
            
            for k in range(2*n - 1):
                if k == 2*i:
                    if S_prime > 1e-15:
                        cand = C_2i / S_prime
                        if cand >= 0:
                            candidates.append(cand)
                else:
                    if A[k] > 1e-15:
                        cand = S_prime - 2.0 * B[k] / A[k]
                        if cand >= 0:
                            candidates.append(cand)
            
            # Also add the current value and 0
            candidates.append(a_i_curr)
            candidates.append(0.0)
            
            # Evaluate J at each candidate and pick the best
            best_val = float('inf')
            best_ai = a_i_curr
            
            for cand in candidates:
                cand = max(0.0, cand)
                a[i] = cand
                new_s = s  # sum doesn't change if we're working with the ratio
                # Actually wait - we're changing a_i but not renormalizing
                # J = 2n * max_conv / (sum)^2
                # sum changes! new_sum = S_prime + cand
                new_sum = S_prime + cand
                if new_sum < 1e-15:
                    continue
                
                # Compute new max conv
                # (a*a)[k] = A[k]*cand + B[k] for k≠2i
                # (a*a)[2i] = cand^2 + C_2i
                new_ac = A * cand + B
                if 0 <= 2*i < 2*n - 1:
                    new_ac[2*i] = cand * cand + C_2i
                new_ac = np.maximum(new_ac, 0.0)  # Should be non-negative for non-negative a
                
                new_score = 2.0 * n * new_ac.max() / (new_sum * new_sum)
                
                if new_score < best_val:
                    best_val = new_score
                    best_ai = cand
            
            # Update a_i
            if best_ai != a_i_curr:
                a[i] = best_ai
                s = S_prime + best_ai
                a[best_ai if False else i] = best_ai  # already set
                
                # Check global score
                if best_val < best_score - 1e-12:
                    best_score = best_val
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if n > 2:
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    
                    ac = fast_autocorr(a)
                    sc = 2.0 * n * ac.max() / (a.sum()**2)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD stall={stall} ({elapsed:.1f}s): score={best_score:.8f}")
    
    return best_a, best_score


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP Frank-Wolfe with full constraint set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Use ALL constraints (or top K if too many)
        # For n=2400, we have 4799 constraints - too many for LP
        # Use constraints within 99% of max
        threshold = M_curr * 0.99
        active_lags = np.where(ac >= threshold)[0]
        
        if len(active_lags) > 500:
            active_lags = np.argsort(ac)[::-1][:500]
        
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(ac))])
        
        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={2*n*best_m:.8f}")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Phase 1: LogSumExp L-BFGS-B (40% time)
    Phase 2: Exact Coordinate Descent (30% time)
    Phase 3: LP Frank-Wolfe (20% time)
    Phase 4: ECD polish + multi-length (10% time)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        # Triangle initialization
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    # === Phase 1: LogSumExp L-BFGS-B (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: LogSumExp-LBFGS n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = logsumexp_optimize(best_seq, t1, tau_start=100.0, tau_end=5000.0, 
                                       n_phases=5, verbose=True)
        try_update(a_out, "LSE")
    
    # === Phase 2: Exact Coordinate Descent (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: ECD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = exact_coordinate_descent(best_seq, t2, verbose=True, rng_seed=777)
        try_update(a_out, "ECD")
    
    # === Phase 3: LP Frank-Wolfe (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP-FW n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = lp_frank_wolfe(best_seq, t3, solver, verbose=True)
        try_update(a_out, "LP")
    
    #
