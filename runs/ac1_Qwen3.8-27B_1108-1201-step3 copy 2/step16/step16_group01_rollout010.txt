            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SMD ({elapsed:.1f}s): J={J:.6f}, p={p:.1f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_M

def coordinate_descent_fast(a0, time_limit, rng_seed=42, verbose=True):
    """Fast coordinate descent with peak prediction."""
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_M = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Get top peak lags
        top_k = np.argsort(ac)[::-1][:10]
        
        # Precompute sensitivity for top peaks
        # sens[i] = how much a[i] contributes to the peaks
        peak_sens = np.zeros(n)
        for k in top_k[:5]:
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    peak_sens[i] += a[j]
        
        improved = False
        n_trials = min(200, max(50, n // 50))
        
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick source: prefer high-sensitivity, high-value positions
            weights = peak_sens * a
            total_w = weights.sum()
            if total_w < 1e-15:
                i = int(rng.integers(n))
            else:
                i = int(rng.choice(n, p=weights / total_w))
            
            if a[i] < 1e-10:
                continue
            
            # Pick destination: prefer low-sensitivity positions
            # Sample from inverse sensitivity
            inv_sens = 1.0 / (peak_sens + 1e-10)
            total_inv = inv_sens.sum()
            if total_inv < 1e-15:
                j = int(rng.integers(n))
            else:
                j = int(rng.choice(n, p=inv_sens / total_inv))
            
            if i == j:
                continue
            
            # Try different transfer fractions
            for frac in [0.001, 0.003, 0.01, 0.03, 0.05]:
                amt = a[i] * frac
                if amt < 1e-12:
                    continue
                
                # Predict new peak (first order)
                max_pred = 0.0
                for k in top_k[:5]:
                    ki = int(k)
                    delta = 0.0
                    if 0 <= ki - i < n:
                        delta -= 2.0 * amt * a[ki - i]
                    if 0 <= ki - j < n:
                        delta += 2.0 * amt * a[ki - j]
                    pred = ac[k] + delta
                    if pred > max_pred:
                        max_pred = pred
                
                if max_pred < best_M - 1e-12:
                    # Accept (verify with exact computation)
                    old_ai = a[i]
                    old_aj = a[j]
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    ac_new = fast_autocorr(a)
                    M_new = ac_new.max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        ac = ac_new
                        improved = True
                        stall = 0
                        break
                    else:
                        # Reject
                        a[i] = old_ai
                        a[j] = old_aj
                
                break  # Move to next trial
        
        if not improved:
            stall += 1
            # Escape: random perturbation
            n_perturb = max(5, n // 100)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    ac = fast_autocorr(a)
                    M_new = ac.max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    CoordDesc ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
