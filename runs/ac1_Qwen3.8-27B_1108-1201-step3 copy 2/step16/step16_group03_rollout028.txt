                      lr=0.05, verbose=True, seed=42):
    """Adam optimization of log-sum-exp proxy."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    t_adam = 0
    stall = 0
    max_stall = 300
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / (time_limit * 0.8), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad = lse_gradient(a, beta)
        
        t_adam += 1
        b1, b2, eps = 0.85, 0.999, 1e-8
        m_s = b1 * m_s + (1 - b1) * grad
        v_s = b2 * v_s + (1 - b2) * grad**2
        m_hat = m_s / (1 - b1**t_adam)
        v_hat = v_s / (1 - b2**t_adam)
        
        # Adaptive learning rate
        lr_t = lr * (0.5 ** (elapsed / (time_limit * 0.5)))
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        iters += 1
        
        M = fast_conv(a).max()
        if M < best_m - 1e-12:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            scale = 0.005 if beta > 50 else 0.02
            a = best_a + rng.normal(0, scale, n) * (best_a + 0.01)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m_s = np.zeros(n)
            v_s = np.zeros(n)
            t_adam = 0
            stall = 0
        
        if verbose and iters % 500 == 0:
            J = 2.0 * n * best_m
            print(f"    Adam-LSE iter={iters} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m

def lp_step(a, M, solver):
    """Single LP Frank-Wolfe step: find b maximizing sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Build constraint: for each k, sum_j a[j]*b[k-j] <= M (where 0 <= k-j < n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) == 1]
    
    # conv(a, b)[k] = sum_j a[j] * b[k-j] for valid j, k-j
    # For each k from 0 to 2n-2:
    #   sum over j where 0<=j<n and 0<=k-j<n of a[j]*b[k-j] <= M
    # This is a linear constraint in b
    
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints.append(T @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(a)
        M_curr = ac.max()
        
        b_dir = lp_step(a, M_curr, solver)
        if b_dir is None:
            break
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_m = fast_conv((1-best_alpha)*a + best_alpha*b_dir).max()
        
        for alpha in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05]:
            trial = np.maximum((1-alpha)*a + alpha*b_dir, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            tm = fast_conv(trial).max()
            if tm < best_trial_m:
                best_trial_m = tm
                best_alpha = alpha
        
        a_new = np.maximum((1-best_alpha)*a + best_alpha*b_dir, 0.0)
        a_new = a_new / a_new.sum()
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        if verbose and t % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def peak_reduction_descent(a0, time_limit, verbose=True, seed=42):
    """
    Direct peak reduction: identify top peaks, find elements contributing to them,
    and move mass to reduce the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_conv(a)
        M = ac.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:5]
        improved = False
        
        for k in top_k:
            k_int = int(k)
            # Elements contributing to peak k
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            total = contrib.sum()
            
            if total < 1e-20:
                continue
            
            # For each contributing pair, try to reduce it
            src = idx[np.argmax(contrib)]
            partner = k_int - src
            
            # Find a good destination: an index that doesn't contribute to top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:3]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True
            
            # Try moving a small amount from src to a random location not in peaks
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            found = False
            for _ in range(20):
                dst = candidates[rng.integers(len(candidates))]
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            
            if found:
                break
        
        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Peak reduction ({time.monotonic()-t0:.1f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""

1. Adam optimizer with log-sum-exp smooth approximation and temperature annealing
2. Mirror descent with adaptive soft subgradient (weighted peak normalization)
3. LP-based Frank-Wolfe with exact line search

5. Optimal peak-weight computation via variance minimization

import numpy as np
import cvxpy as cp
import time


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Project onto the probability simplex (non-negative, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def lse_gradient(a, beta):
    """
    Compute gradient of log(sum(exp(beta * conv(a,a)))) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a)
    r = fast_conv(a)
    r_max = r.max()
    exp_r = np.exp(beta * (r - r_max))
    w = exp_r / exp_r.sum()

    # grad[i] = 2*beta * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Compute via correlation using FFT
    a_flip = a[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    conv_res = np.fft.irfft(W * AF, fft_size)
    grad = 2.0 * beta * conv_res[n - 1:2 * n - 1]
    return grad


def hard_subgradient(a, k_star):
    """Exact subgradient of conv(a,a)[k_star] w.r.t. a."""
    n = len(a)
    k = int(k_star)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return grad


def soft_subgradient(a, r, top_k=3, beta=2.0):
    """
    Soft subgradient: weighted sum of top-k peak subgradients.
    Weights are proportional to peak_value^beta.
    """
    n = len(a)
    k_max = min(top_k, len(r))
    top_idx = np.argsort(r)[::-1][:k_max]
    top_vals = np.maximum(r[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * hard_subgradient(a, k)
    return grad


def compute_optimal_peak_weights(a, r, top_k=5):
    """
    Compute optimal weights on top peaks to minimize variance of the
    weighted subgradient (making it as 'flat' as possible over the support).
    """
    n = len(a)
    k_max = min(top_k, len(r))
    top_idx = np.argsort(r)[::-1][:k_max]

    # Build subgradient matrix: G[i, j] = hard_subgradient(a, top_idx[j])[i]
    G = np.zeros((n, k_max))
    for j, k in enumerate(top_idx):
        G[:, j] = hard_subgradient(a, k)

    # Find support of a
    support = a > 1e-10
    if support.sum() < 2:
        return np.ones(k_max) / k_max
    G_sup = G[support, :]

    # Minimize Var(G @ lambda) = lambda^T G^T G lambda - (G^T 1 @ lambda)^2 / |S|
    # subject to lambda >= 0, sum(lambda) = 1
    # This is a small QP. For simplicity, use a projected gradient approach.
    m = k_max
    lam = np.ones(m) / m
    GtG = G_sup.T @ G_sup
    Gt1 = G_sup.T @ np.ones(len(G_sup))
    n_s = len(G_sup)

    for _ in range(50):
        # Gradient of lambda^T G^T G lambda - (G^T 1 @ lambda)^2 / n_s
        grad_lam = 2.0 * GtG @ lam - 2.0 * Gt1 * (Gt1 @ lam) / n_s
        # Projected gradient step
        lr = 0.1 / (np.max(np.abs(GtG)) + 1e-8)
        lam = lam - lr * grad_lam
        lam = np.maximum(lam, 0)
        lam = lam / lam.sum()

    return lam


def adam_lse_optimize(a0, time_limit, beta_start=5.0, beta_end=300.0,
                      lr=0.05, verbose=True, seed=42):
    """Adam optimization of log-sum-exp proxy with temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    m_s = np.zeros(n)
    v_s = np.zeros(n)
    t_adam = 0
    stall = 0
    max_stall = 400
    iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break

        # Anneal beta
        progress = min(elapsed / (time_limit * 0.85), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad = lse_gradient(a, beta)

        # Adam with beta1=0.85 for faster initial convergence
        t_adam += 1
        b1, b2, eps = 0.85, 0.999, 1e-8
        m_s = b1 * m_s + (1 - b1) * grad
        v_s = b2 * v_s + (1 - b2) * grad ** 2
        m_hat = m_s / (1 - b1 ** t_adam)
        v_hat = v_s / (1 - b2 ** t_adam)

        # Decaying learning rate
        lr_t = lr * (0.3 ** (elapsed / (time_limit * 0.6)))
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        iters += 1

        M = fast_conv(a).max()
        if M < best_m - 1e-12:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation
        if stall > max_stall:
            scale = 0.003 if beta > 50 else 0.015
            a = best_a + rng.normal(0, scale, n) * (best_a + 0.005)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m_s = np.zeros(n)
            v_s = np.zeros(n)
            t_adam = 0
            stall = 0

        if verbose and iters % 500 == 0:
            J = 2.0 * n * best_m
            print(f"    Adam-LSE iter={iters} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

    return best_a, best_m


def mirror_descent_optimize(a0, time_limit, verbose=True, top_k=3, c0=3.0, seed=42):
    """
    Mirror descent (multiplicative) with adaptive soft subgradient.
    Periodically recomputes optimal peak weights.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    gamma = c0
    stall = 0
    max_stall = 5000
    current_top_k = top_k
    iters = 0
    last_weight_update = 0

    # For Nesterov: maintain momentum
    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)

        # Periodically recompute optimal weights
        if t - last_weight_update > 2000 or t == 1:
            last_weight_update = t
            # Use optimal weights if we have enough peaks
            if current_top_k >= 3:
                optimal_w = compute_optimal_peak_weights(y, ac, top_k=current_top_k)
                # Use these weights directly
                k_max = min(current_top_k, len(ac))
                top_idx = np.argsort(ac)[::-1][:k_max]
                g = np.zeros(n)
                for wi, k in zip(optimal_w, top_idx):
                    g += wi * hard_subgradient(y, k)
            else:
                g = soft_subgradient(y, ac, top_k=current_top_k, beta=1.5)
        else:
            g = soft_subgradient(y, ac, top_k=current_top_k, beta=1.5)

        # Mirror descent step: a_new[i] = a[i] * exp(-eta * g[i])
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        iters += 1

        M = fast_conv(x).max()
        if M < best_m - 1e-13:
            best_m = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive top_k increase when stuck
        if stall > max_stall // 2 and current_top_k < 8:
            current_top_k += 1

        # Restart
        if stall > max_stall:
            scale = 0.004
            x_restarted = best_a + rng.normal(0, scale, n) * (best_a + 0.003)
            x_restarted = np.maximum(x_restarted, 1e-12)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            current_top_k = min(current_top_k + 2, 10)

        # Periodic larger jump
        if t % 20000 == 0 and t > 0:
            x_jumped = best_a + rng.normal(0, 0.01, n) * (best_a + 0.002)
            x_jumped = np.maximum(x_jumped, 1e-10)
            x_jumped = x_jumped / x_jumped.sum()
            M_jump = fast_conv(x_jumped).max()
            if M_jump < best_m:
                best_m = M_jump
                best_a = x_jumped.copy()
                x = x_jumped
                x_prev = x_jumped

        if verbose and iters % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MirrorDescent t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")

    return best_a, best_m


def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    # Precompute index structure for T matrix
    # T[k, j] = 1 if we're computing conv(a,b)[k] = sum_j a[j]*b[k-j]
    # T depends on a, so we rebuild each iteration but use efficient construction
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(a)
        M_curr = ac.max()

        # Build T matrix: T[k,j] = a[j] if 0 <= k-j < n
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_eval = fast_conv((1 - best_alpha) * a + best_alpha * direction).max()

        for alpha in [0.5, 0.4, 0.35, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05]:
            trial = np.maximum((1.0 - alpha) * a + alpha * direction, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        y_new = np.maximum((1.0 - best_alpha) * a + best_alpha * direction, 0.0)
        y_new = y_new / y_new.sum()
        a = y_new

        M = fast_conv(a).max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def peak_reduction_descent(a0, time_limit, verbose=True, seed=42):
    """
    Targeted peak reduction: identify top peaks and systematically
    move mass to reduce them.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M = ac.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks
        n_peaks = min(8, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find destination: minimize impact on all top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_score_dst = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                # Sensitivity: how much would this dst contribute to top peaks?
                sens = 0.0
                for k2 in top_peaks[:5]:
                    k2_int = int(k2)
                    if 0 <= k2_int - dst < n:
                        sens += a[k2_int - dst]
                    if 0 <= dst - (k2_int - dst) < n:
                        sens += 1.0  # crude
                if sens < best_score_dst:
                    best_score_dst = sens
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            for frac in [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            # Random perturbation when stuck
            if stall > 50:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.002, 0.015)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        J = 2.0 * n * best_m
        print(f"    PeakReduction ({time.monotonic() - t0:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            print(f"  *** NEW BEST [mirror]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP n={n_cur} ===")
    t3 = min(remaining * 0.22, 350.0)
    if t3 > 60 and n_cur <= 3000:
        a_out, _ = lp_frank_wolfe_optimize(best_seq, t3, solver, verbose=True)
        a_out = norm_sum1(a_out)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [lp]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 4: Adam-LSE with high beta (fine tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Adam-LSE high-beta n={len(best_seq)} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = adam_lse_optimize(best_seq, t4, beta_start=80.0, beta_end=500.0,
                                     lr=0.02, verbose=True, seed=99)
        a_out = norm_sum1(a_out)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [adam_high]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 5: Peak reduction coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: PeakReduction n={len(best_seq)} ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_out, _ = peak_reduction_descent(best_seq, t5, verbose=True, seed=42)
        a_out = norm_sum1(a_out)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [peak]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        for delta in [200, -200, 400, -400]:
            n_try = n_best + delta
            if n_try < 1200 or n_try > 3500:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 6: Length n={n_try} ===")
            a_long = norm_sum1(interpolate_sequence(best_seq, n_try))
            t6 = min(remaining * 0.35, 150.0)
            if t6 > 20:
                a_out, _ = adam_lse_optimize(a_long, t6, beta_start=10.0, beta_end=200.0,
                                             lr=0.04, verbose=False, seed=100 + n_try % 50)
                a_out = norm_sum1(a_out)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [lens_{n_try}]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s}")

    # === Phase 7: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2800 and remaining > 60:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 30:
            a_out, _ = lp_frank_wolfe_optimize(best_seq, t7, solver, verbose=True)
            a_out = norm_sum1(a_out)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [lp_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 8: Final peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final PeakReduction n={len(best_seq)} ===")
    if remaining > 20:
        t8 = min(remaining * 0.8, 100.0)
        if t8 > 10:
            a_out, _ = peak_reduction_descent(best_seq, t8, verbose=True, seed=314)
            a_out = norm_sum1(a_out)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [peak_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 9: One more brief mirror descent pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 9: Final Mirror n={len(best_seq)} ===")
        t9 = min(remaining * 0.7, 80.0)
        if t9 > 10:
            a_out, _ = mirror_descent_optimize(best_seq, t9, verbose=False, top_k=4, c0=4.0, seed=555)
            a_out = norm_sum1(a_out)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [mirror_final]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
