"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth optimization (log-sum-exp approximation of max) with Adam optimizer
   in softmax parameterization, with beta annealing
2. Coordinate descent polishing with exact max
3. Frank-Wolfe with LP oracle
4. Multi-length exploration
"""

import numpy as np
import time
import math

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta * conv(a,a)_k)) w.r.t. a.
    Returns (gradient, value).
    """
    n = len(a)
    conv_a = fast_conv(a)
    max_c = conv_a.max()
    log_sum_exp = max_c + np.log(np.sum(np.exp(beta * (conv_a - max_c))))
    value = log_sum_exp / beta
    
    exp_bc = np.exp(beta * (conv_a - max_c))
    w = exp_bc / exp_bc.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # = 2 * cross_corr(w, a)[i] where cross_corr[w,a][i] = sum_k w[k]*a[k-i]
    # = 2 * conv(w, a[::-1])[n-1+i]
    a_rev = a[::-1]
    cross_size = (2*n - 1) + n - 1  # = 3n-2
    fft_size = 1 << (cross_size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Arev = np.fft.rfft(a_rev, fft_size)
    cross = np.fft.irfft(W * Arev, fft_size)[:cross_size]
    grad = 2.0 * cross[n-1 : 2*n-1]
    return grad, value

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_param(x):
    """Convert x to a probability vector via softmax."""
    x = x - x.max()
    exp_x = np.exp(x)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s

def softmax_jacobian(grad_a, a):
    """
    Given gradient of f w.r.t. a, compute gradient w.r.t. x where a = softmax(x).
    df/dx_i = df/da_i - a_i * sum_j a_j * df/da_j
    """
    return grad_a - a * np.dot(a, grad_a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def adam_smooth_optimize(a0, time_limit, beta_start=1.0, beta_end=80.0, 
                          lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                          verbose=True, seed=42):
    """
    Adam optimization in softmax parameterization with log-sum-exp smooth max.
    Beta is annealed from beta_start to beta_end.
    """
    n = len(a0)
    a0 = norm_sum1(a0)
    
    # Initialize x from a0: x_i = log(a_i)
    x = np.log(np.maximum(a0, 1e-30)).copy()
    x = x - x.mean()  # center
    
    best_a = a0.copy()
    best_val = fast_conv(a0).max()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    t0 = time.monotonic()
    
    # Total iterations estimate: we'll run until time limit
    # Estimate: each iteration takes ~0.5ms for n=2400
    # In 600s, we can do ~1.2M iterations
    # For beta annealing, we want to anneal over the whole period
    
    # Use a fixed number of iterations for annealing
    total_iters = 500000
    # Actually, let's just anneal based on time fraction
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        frac = min(1.0, t / total_iters)
        # Geometric annealing of beta
        beta = beta_start * (beta_end / beta_start) ** frac
        
        a = softmax_param(x)
        
        # Compute gradient
        grad_a, _ = smooth_grad(a, beta)
        grad_x = softmax_jacobian(grad_a, a)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr * (0.5 ** (frac * 3))  # decay learning rate
        x -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if t % 1000 == 0:
            a_check = softmax_param(x)
            val = fast_conv(a_check).max()
            if val < best_val:
                best_val = val
                best_a = a_check.copy()
            
            if verbose and t % 10000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_val
                print(f"      Adam t={t} ({elapsed:.1f}s): beta={beta:.2f}, J={J:.6f}")
    
    return best_a, best_val


def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with mass transfer moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find good destination: low contribution to peaks
            # Compute total contribution of each position to top peaks
            peak_sens = np.zeros(n)
            for k2 in top_peaks[:10]:
                k2_int = int(k2)
                s2 = max(0, k2_int - n + 1)
                e2 = min(n, k2_int + 1)
                for j in range(s2, e2):
                    peak_sens[j] += a[k2_int - j] * ac[k2_int]
            
            # Move mass from src to position with lowest sensitivity
            candidates = np.where(peak_sens < 0.5 * peak_sens[src])[0]
            if len(candidates) < 10:
                candidates = np.argsort(peak_sens)[:n//4]
            
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            n_cand = min(len(candidates), 100)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]
            
            best_dst = -1
            best_trial_m = M_curr
            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_dst = dst
                        best_amt = amt
                        break
                
            if best_dst >= 0 and best_trial_m < M_curr - 1e-15:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = fast_conv(a).max()
                best_a = a.copy()
                improved = True
                stall = 0
                break  # restart peak loop
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def gradient_descent_exact(a0, time_limit, verbose=True, seed=42):
    """
    Gradient descent on the simplex using the exact (nonsmooth) max.
    Uses subgradient from all active peaks.
    This is a fallback/refinement method.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = 0.01
    decay = 0.9999
    stall = 0
    
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(a)
        M_curr = ac.max()
        
        # Get subgradient: average over all peaks within 1% of max
        threshold = M_curr * 0.99
        active = np.where(ac >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(ac)])
        
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j] / len(active)
        
        # Projected gradient step on simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        lr *= decay
        
        if stall > 3000:
            # Restart with perturbation
            a = best_a + rng.normal(0, 0.003, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            lr = 0.01
            stall = 0
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      GD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_grad(a, beta):
    """
    Compute gradient and value of (1/beta)*log(sum_k exp(beta * conv(a,a)_k)) w.r.t. a.
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a)
    conv_a = fast_conv(a)
    max_c = conv_a.max()
    exp_bc = np.exp(beta * (conv_a - max_c))
    sum_exp = exp_bc.sum()
    if sum_exp < 1e-300:
        sum_exp = 1.0
    log_sum_exp = max_c + np.log(sum_exp)
    value = log_sum_exp / beta
    w = exp_bc / sum_exp

    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # = 2 * conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    cross_size = (2 * n - 1) + n - 1
    fft_size = 1 << (cross_size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Arev = np.fft.rfft(a_rev, fft_size)
    cross = np.fft.irfft(W * Arev, fft_size)[:cross_size]
    grad = 2.0 * cross[n - 1: 2 * n - 1]
    return grad, value


def softmax_param(x):
    """Convert x to probability vector via numerically stable softmax."""
    x = x - x.max()
    exp_x = np.exp(x)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def softmax_jacobian(grad_a, a):
    """
    Given gradient of f w.r.t. a (on simplex), compute gradient w.r.t.
    x where a = softmax(x). df/dx = diag(a) - a*a^T applied to grad_a.
    """
    return grad_a - a * np.dot(a, grad_a)


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def adam_smooth_optimize(a0, time_limit, beta_start=1.0, beta_end=80.0,
                         lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                         total_iters=400000, verbose=True, seed=42):
    """
    Adam optimization in softmax parameterization with log-sum-exp smooth max.
    Beta is annealed geometrically from beta_start to beta_end.
    """
    n = len(a0)
    a0 = norm_sum1(a0)

    # Initialize x from a0
    x = np.log(np.maximum(a0, 1e-30)).copy()
    x = x - x.mean()

    best_a = a0.copy()
    best_m = fast_conv(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    eval_interval = 500 if n > 1000 else 200

    while time.monotonic() - t0 < time_limit:
        t += 1
        frac = min(1.0, t / float(total_iters))
        # Geometric annealing
        beta = beta_start * (beta_end / beta_start) ** frac

        a = softmax_param(x)
        grad_a, _ = smooth_grad(a, beta)
        grad_x = softmax_jacobian(grad_a, a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning rate schedule: start high, decay
        lr_t = lr * (1.0 + 0.5 * frac) ** (-2.0)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Keep x centered to avoid overflow
        if t % 1000 == 0:
            x = x - x.mean()

        # Periodic evaluation
        if t % eval_interval == 0:
            a_check = softmax_param(x)
            val = fast_conv(a_check).max()
            if val < best_m:
                best_m = val
                best_a = a_check.copy()

            if verbose and t % (eval_interval * 20) == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      Adam t={t} ({elapsed:.1f}s): beta={beta:.2f}, J={J:.6f}")

    # Final check
    a_final = softmax_param(x)
    val = fast_conv(a_final).max()
    if val < best_m:
        best_m = val
        best_a = a_final.copy()

    return best_a, best_m

            break

        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # To keep LP small, only use top constraints

            break

        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05]:

        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        iter_count += 1

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

                for j in range(s2, e2):
                    peak_sens[j] += a[k2_int - j] * ac[k2_int]

            # Find low-sensitivity destinations
            threshold = 0.3 * peak_sens[src]
            candidates = np.where(peak_sens < threshold)[0]
            if len(candidates) < 5:
                candidates = np.argsort(peak_sens)[:max(5, n // 10)]

            if len(candidates) == 0:
                continue

            n_cand = min(len(candidates), 80)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_trial_m = M_curr
            best_amt = 0.0
            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_dst = dst
                        best_amt = amt
                        break

            if best_dst >= 0 and best_trial_m < M_curr - 1e-15:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = fast_conv(a).max()
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def mirror_descent_refine(a0, time_limit, verbose=False, seed=42):
    """
    Mirror descent (multiplicative weights) with adaptive subgradient.
    Cycles through top peaks to avoid getting stuck.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    c0 = 3.0
    stall = 0
    max_stall = 5000
    top_k = 1
    cycle_idx = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(a)
        M_curr = ac.max()

        # Get top-k peaks
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]

        # Rotate which peaks to focus on
        if top_k > 1:
            offset = (t // 100) % top_k
            use_idx = top_idx[offset:offset + 1] if top_k > 1 else top_idx[:1]
            if len(use_idx) == 0:
                use_idx = top_idx[:1]
        else:
            use_idx = top_idx[:1]

        # Subgradient from selected peaks
        grad = np.zeros(n)
        for k in use_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        M_new = fast_conv(a).max()

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive top_k
        if stall > max_stall // 2 and top_k < 6:
            top_k += 1
        elif stall > max_stall:
            # Restart
            a = best_a + rng.normal(0, 0.005, n) * best_a
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            top_k = 1
            c0 = max(c0 * 1.2, 10.0)

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={top_k}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with adaptive time allocation:
    Phase 1: Adam smooth optimization (main workhorse)
    Phase 2: Mirror descent refinement
    Phase 3: LP Frank-Wolfe (if solver available)
    Phase 4: Coordinate descent polish
    Phase 5: Multi-length exploration
    Phase 6: Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")

    solver_info = get_solver()

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, J={best_score:.6f}")

    # === Phase 1: Adam smooth optimization (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: Adam smooth n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = adam_smooth_optimize(
            a_in, t1, beta_start=1.0, beta_end=50.0,
            lr=0.006, beta1=0.9, beta2=0.999,
            total_iters=300000, verbose=True, seed=42
        )
        try_update(a_out, f"adam1_n{n0}")

    # === Phase 2: Adam smooth v2 (different params, continue from best) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 2: Adam smooth v2 n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = adam_smooth_optimize(
            a_in, t2, beta_start=10.0, beta_end=120.0,
            lr=0.003, beta1=0.85, beta2=0.998,
            total_iters=200000, verbose=True, seed=7
        )
        try_update(a_out, f"adam2_n{len(best_seq)}")

    # === Phase 3: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.13, 200.0)
    print(f"\n=== Phase 3: Mirror descent n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_refine(a_in, t3, verbose=True, seed=99)
        try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if solver_info is not None and n_cur <= 3000:
        t4 = min(remaining * 0.12, 180.0)
        print(f"\n=== Phase 4: LP Frank-Wolfe n={n_cur} (budget={t4:.0f}s) ===")
        if t4 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t4, solver_info, verbose=True)
            try_update(a_out, f"lp_n{n_cur}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Coord polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        deltas = [-400, 400, -800, 800]
        for delta in deltas:
            n_try = n_best + delta
            if n_try < 800 or n_try > 6000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 6: n={n_try} (remaining={remaining:.0f}s) ===")
            a_long = interpolate_sequence(best_seq, n_try)
            t6 = min(remaining * 0.25, 120.0)
            if t6 > 30:
                a_out, _ = adam_smooth_optimize(
                    a_long, t6, beta_start=2.0, beta_end=60.0,
                    lr=0.004, total_iters=100000,
                    verbose=False, seed=55 + n_try % 200
                )
                try_update(a_out, f"adam_n{n_try}")

                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    t6b = min(remaining * 0.3, 60.0)
                    if t6b > 15:
                        a_out2, _ = coordinate_polish(a_out, t6b, verbose=False, seed=77)
                        try_update(a_out2, f"coord_n{n_try}")

    # === Phase 7: Final Adam polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50:
        t7 = min(remaining * 0.4, 100.0)
        print(f"\n=== Phase 7: Final Adam n={n_best} (budget={t7:.0f}s) ===")
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = adam_smooth_optimize(
                a_in, t7, beta_start=30.0, beta_end=200.0,
                lr=0.002, beta1=0.9, beta2=0.999,
                total_iters=100000, verbose=False, seed=314
            )
            try_update(a_out, f"adam_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.6, 60.0)
        print(f"\n=== Phase 8: Final coord n={n_best} (budget={t8:.0f}s) ===")
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
