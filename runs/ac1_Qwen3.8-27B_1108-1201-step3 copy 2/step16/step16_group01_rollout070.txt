_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient_smoothed(a, c, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def compute_subgradient_exact(a, c, threshold_frac=0.995):
    n = len(a)
    M = c.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(c >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(c.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    return g


def mirror_update(y, g, eta):
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def amd_optimize(a0, time_limit, mode='smooth', c0=3.0, threshold_frac=0.99,

        if time.monotonic() - t0 > time_limit:
            break

            best_m = M_new
            best_a = x.copy()
            stall_count = 0

            prob.solve(solver=solver, verbose=False)

            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}, active={n_active}, frac={current_frac:.2f}")

    return best_a, best_m

    if t1 > 15:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t1, mode='smooth', c0=3.0,
                                top_k=60, beta=4.0, verbose=True, rng_seed=1)
        try_update(a_out, f"amd_s1_n{n0}")

    # Phase 2: LP Frank-Wolfe (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP FW n={n_best} ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, solver, verbose=True,
                                  max_iters=200, min_lag_frac=0.85)
        try_update(a_out, f"lpfw2_n{n_best}")

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t3, verbose=True, rng_seed=4)
        try_update(a_out, f"coord3_n{n_best}")

    # Phase 4: AMD exact
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AMD exact n={n_best} ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t4, mode='exact', c0=2.5,
                                threshold_frac=0.995, verbose=True, rng_seed=5)
        try_update(a_out, f"amd_e4_n{n_best}")

    # Phase 5: LP FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP FW r2 n={n_best} ===")
    t5 = min(remaining * 0.22, 350.0)
    if t5 > 30:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5, solver, verbose=True,
                                  max_iters=200, min_lag_frac=0.80)
        try_update(a_out, f"lpfw5_n{n_best}")

    # Phase 6: Pairwise optimal transfer
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Pairwise opt n={n_best} ===")
    t6 = min(remaining * 0.10, 130.0)
    if t6 > 15:
        a_out, _ = pairwise_optimal_transfer(norm_sum1(best_seq), t6, verbose=True, rng_seed=6)
        try_update(a_out, f"pw6_n{n_best}")

    # Phase 7: MWU
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: MWU n={n_best} ===")
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 15:
        a_out, _ = multiplicative_weight_update(norm_sum1(best_seq), t7, verbose=True, rng_seed=7)
        try_update(a_out, f"mwu7_n{n_best}")

    # Phase 8: Final AMD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final AMD n={n_best} ===")
    t8 = min(remaining * 0.18, 180.0)
    if t8 > 15:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t8, mode='smooth', c0=1.5,
                                top_k=80, beta=6.0, verbose=True, rng_seed=8)
        try_update(a_out, f"amd8_n{n_best}")

    # Phase 9: Final coord
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord n={n_best} ===")
    t9 = min(remaining * 0.15, 120.0)
    if t9 > 10:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t9, verbose=True, rng_seed=9)
        try_update(a_out, f"coord9_n{n_best}")

    # Phase 10: Multi-length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 10: Multi-length ===")
        n_try = min(n_best + 300, 3200)
        if n_try > n_best:
            print(f"  Trying n={n_try}")
            a_long = interpolate_sequence(best_seq, n_try)
            t10a = min(remaining * 0.35, 150.0)
            if t10a > 30:
                a_out, _ = amd_optimize(norm_sum1(a_long), t10a, mode='smooth',
                                        c0=3.0, top_k=50, beta=5.0, verbose=True, rng_seed=10)
                try_update(a_out, f"amd10_n{n_try}")

        remaining = time_budget - (time.monotonic() - t_global)
        n_best_now = len(best_seq)
        if remaining > 60:
            n_short = max(n_best_now - 300, 1000)
            if n_short < n_best_now:
                print(f"  Trying n={n_short}")
                a_short = interpolate_sequence(best_seq, n_short)
                t10b = min(remaining * 0.30, 120.0)
                if t10b > 20:
                    a_out, _ = amd_optimize(norm_sum1(a_short), t10b, mode='smooth',
                                            c0=3.0, top_k=50, beta=5.0, verbose=True, rng_seed=11)
                    try_update(a_out, f"amd10s_n{n_short}")

    # Phase 11: Final LP if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 11: Final LP n={n_best} ===")
        t11 = min(remaining * 0.50, 100.0)
        if t11 > 15:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t11, solver, verbose=True,
                                      max_iters=50, min_lag_frac=0.75)
            try_update(a_out, f"lp11_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
