```python
def softmax_gradient(a, conv_vals, temperature, top_k=None):
    """
    Compute soft-max weighted gradient.
    gradient of max_k q_k(a) ≈ sum_k w_k * grad q_k(a)

    n = len(a)
    
    if top_k is not None and top_k < len(conv_vals):

        vals = conv_vals[top_idx]

        top_idx = np.arange(len(conv_vals))
        vals = conv_vals

    scaled = vals / temperature
    scaled -= scaled.max()
    weights = np.exp(scaled)

    if w_sum < 1e-30:

    else:
        weights /= w_sum
    
    # Compute gradient: grad[i] = 2 * sum_k w_k * a[k-i] (with zero extension)
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
```

```python
def compute_soft_gradient(a, conv_vals, temperature, top_k=None):
    """
    Compute the soft-max weighted gradient using FFT.
    grad[i] = 2 * sum_k w_k * a[k-i] (with zero extension)
    = 2 * (a_rev * w)[i + n - 1]
    """
    n = len(a)
    
    if top_k is not None and top_k < len(conv_vals):
        top_idx = np.argsort(conv_vals)[::-1][:top_k]
        vals = conv_vals[top_idx]
    else:
        top_idx = None
        vals = conv_vals
    
    # Soft-max weights
    scaled = vals.astype(np.float64) / temperature
    scaled -= scaled.max()
    weights_k = np.exp(scaled)
    w_sum = weights_k.sum()
    if w_sum < 1e-30:
        weights_k = np.ones_like(vals) / len(vals)
    else:
        weights_k = weights_k / w_sum
    
    # Build full weight vector (length 2n-1)
    w = np.zeros(2 * n - 1, dtype=np.float64)
    if top_idx is not None:
        w[top_idx] = weights_k
    else:
        w[:] = weights_k
    
    # Compute grad via FFT
    a_rev = a[::-1].copy()
    N_fft = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a_rev, N_fft)
    W = np.fft.rfft(w, N_fft)
    conv_result = np.fft.irfft(A * W, N_fft)
    
    # grad[i] = 2 * conv_result[i + n - 1]
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    
    return grad
```

The projection onto the simplex {a: a >= 0, sum(a) = 1} is:
For a point v, find μ such that:
projected[i] = max(0, v[i] - μ)
and sum_i projected[i] = 1

This can be done by sorting v and finding the right μ via a threshold.

```python
def project_simplex(v):
    """Project v onto {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is the standard algorithm for simplex projection (O(n log n) due to sorting).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated Nesterov subgradient with adaptive soft-max over active lags
2. LP-based Frank-Wolfe (conditional gradient with exact LP solve) for precise polishing
3. Adaptive coordinate descent with bisection
4. Multi-start with diverse initializations and multiple sequence lengths

Key insight: The gradient of (a*a)[k] w.r.t. a[i] is 2*a[k-i] (zero-extended),
enabling O(n log n) gradient computation via FFT. The problem is non-convex
(quadratic forms can be indefinite), so multi-start is essential.
"""

import numpy as np
import time
import cvxpy as cp

_lp_solver = None

        try:
            x = cp.Variable(8, nonneg=True)

                _lp_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:

    _lp_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _lp_solver

def fast_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2*n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the score."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_soft_gradient(a, conv_vals, temperature, top_k=None):
    """
    Compute soft-max weighted gradient using FFT.
    grad[i] = 2 * sum_k w_k * a[k-i] (with zero extension)
    Computed as 2 * (a_rev * w)[i + n - 1] via FFT convolution.
    """
    n = len(a)
    conv_len = 2 * n - 1

    if top_k is not None and top_k < conv_len:
        top_idx = np.argsort(conv_vals)[::-1][:top_k]
        vals = conv_vals[top_idx]
    else:
        top_idx = None
        vals = conv_vals.astype(np.float64)

    # Soft-max weights
    scaled = vals / max(temperature, 1e-10)
    scaled -= scaled.max()
    weights_k = np.exp(scaled)
    w_sum = weights_k.sum()
    if w_sum < 1e-30:
        weights_k = np.ones_like(vals) / len(vals)
    else:
        weights_k = weights_k / w_sum

    # Build full weight vector (length 2n-1)
    w = np.zeros(conv_len, dtype=np.float64)
    if top_idx is not None:
        w[top_idx] = weights_k
    else:
        w[:] = weights_k

    # FFT-based computation of grad[i] = 2 * (a_rev * w)[i + n - 1]
    a_rev = a[::-1].copy()
    conv_len_result = n + conv_len - 1  # = 3n - 2
    N_fft = 1 << (conv_len_result - 1).bit_length()
    A = np.fft.rfft(a_rev, N_fft)
    W = np.fft.rfft(w, N_fft)
    conv_result = np.fft.irfft(A * W, N_fft)

    grad = 2.0 * conv_result[n - 1: 2 * n - 1].copy()
    return grad

def nesterov_subgradient(a0, time_limit, initial_temp=0.05, final_temp=0.002,
                          c0=3.0, restart_interval=2000, verbose=True):
    """
    Nesterov-accelerated projected subgradient with adaptive soft-max.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42 + n)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = restart_interval
    eta_0 = c0

    for t in range(1, 200000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Adaptive temperature
        frac = min(1.0, t / float(max(1, t // 2)))
        temp = max(initial_temp * (1.0 - frac) + final_temp * frac, 1e-6)

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        # Compute gradient
        conv_y = fast_conv(y)
        # Use top-k for efficiency when temp is small
        if temp > 0.01:
            top_k = min(200, 2 * n - 1)
        else:
            top_k = min(80, 2 * n - 1)
        g = compute_soft_gradient(y, conv_y, temp, top_k=top_k)

        # Step size
        eta = eta_0 / (np.sqrt(t) + 1.0)

        # Update
        a_new = y - eta * g
        a_new = project_simplex(a_new)

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Ergodic average check
        if t % 1000 == 0:
            # Just check current x for now (maintaining a running average is expensive)
            pass

        # Periodic restart
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            # Perturb best and restart
            noise_scale = max(0.01, 0.1 / np.sqrt(t / 1000.0 + 1))
            pert = rng.normal(0, noise_scale, n)
            x_r = best_a * (1.0 + pert)
            x_r = project_simplex(x_r)
            x = x_r
            x_prev = x_r
            eta_0 = max(1.0, eta_0 * 0.7)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NS t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, stall={stall_count}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=300, verbose=True):
    """
    LP-based Frank-Wolfe with focused constraint set.
    Solves: max sum(b) s.t. (a*b)[k] <= M for active k, b >= 0
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    for t in range(1, 20000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv_a = fast_conv(a)
        M_curr = conv_a.max()

        # Select active constraints (near-max lags)
        threshold = M_curr * 0.995
        active_lags = np.where(conv_a >= threshold)[0]
        if len(active_lags) > max_constraints:
            # Keep top ones
            top_idx = np.argsort(conv_a)[::-1][:max_constraints]
            active_lags = top_idx

        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(conv_a))])

        # Build constraint matrix: T[row, col] = a[k - col] if valid
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(k_int, n - 1)
            if i_start <= i_end:
                cols = np.arange(i_start, i_end + 1)
                T[row_idx, cols] = a[k_int - cols]

        # Solve LP: max sum(b) s.t. T @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g_dir
        a_new = project_simplex(a_new)

        a = a_new
        M_new = fast_conv(a).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, active={len(active_lags)}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent: for each coordinate, try moving mass to other coordinates.
    Uses the fact that changing a[i] to a[i]+delta (compensated at a[j]) changes
    the convolution in a predictable way.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    sweep = 0
    stall = 0
    max_stall = 4

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        improved = False
        coord_order = np.arange(n)
        rng.shuffle(coord_order)

        for i in coord_order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_m
            if a[i] < 1e-14:
                continue

            best_trial_m = best_m
            best_move = None

            # Try a few random destination coordinates
            n_tries = min(6, n - 1)
            candidates = rng.integers(0, n, size=n_tries)

            for dst in candidates:
                if dst == i:
                    continue
                for frac in [0.003, 0.01, 0.03, 0.08, 0.15]:
                    amt = a[i] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[dst] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_trial_m - 1e-16:
                        best_trial_m = m_trial
                        best_move = (dst, amt)

            if best_move is not None:
                dst, amt = best_move
                a[i] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            if n > 1:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_conv(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()

        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def paired_polish(a0, time_limit, verbose=True, rng_seed=789):
    """Paired coordinate optimization: simultaneously adjust two coordinates."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(30, n // 8)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_m
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                continue

            for direction in [1, -1]:
                src, dst = (i, j) if direction == 1 else (j, i)
                if a[src] < 1e-12:
                    continue
                for frac in [0.005, 0.015, 0.04, 0.08]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1

        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PP ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def make_gaussian_init(n, alpha=2.0):
    """Gaussian-shaped initialization centered at n/2."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-alpha * (x - center) ** 2 / n)
    return normalize(a)

def make_polynomial_init(n, p=1.0):
    """Polynomial-shaped initialization: a[i] ~ (i+1)^p * (n-i)^p."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p * (n + 1 - x) ** p
    return normalize(a)

def make_random_init(n, rng, concentration=2.0):
    """Random Dirichlet initialization."""
    a = rng.dirichlet(np.ones(n) * concentration)
    return a

def make_stripe_init(n, period=None):
    """Striped initialization to break symmetry."""
    if period is None:
        period = max(2, n // 50)
    a = np.ones(n)
    # Add periodic modulation
    x = np.arange(n)
    a = 1.0 + 0.3 * np.sin(2.0 * np.pi * x / period)
    a = np.maximum(a, 0.01)
    return normalize(a)

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s

            try_update(a_out, f"Gauss_NS_n{n_base}")

        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining > 60:
            t8b = min(remaining * 0.5, 80.0)
            if t8b > 20:
                a_out, _ = lp_frank_wolfe(best_seq, t8b, solver, max_constraints=300, verbose=False)
                try_update(a_out, f"LP_final_{len(best_seq)}")

    # === Phase 9: Final LP polish ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 40:
        print(f"\n=== Phase 9: Final LP n={len(best_seq)} (budget={remaining:.0f}s) ===")
        t9 = min(remaining * 0.7, 100.0)
        if t9 > 20:
            a_out, _ = lp_frank_wolfe(best_seq, t9, solver, max_constraints=500, verbose=False)
            try_update(a_out, f"LPFINAL_{len(best_seq)}")

    # === Phase 10: Final coordinate polish ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 20:
        t10 = min(remaining * 0.6, 40.0)
        if t10 > 10:
            a_out, _ = coordinate_polish(best_seq, t10, verbose=False, rng_seed=321)
            try_update(a_out, f"CDFINAL_{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
