import numpy as np

from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_objective_and_grad(params, state):
    """
    Objective and gradient for softmax-parameterized optimization.
    params: log-weights (unconstrained)
    state: dict with beta, top_k
    """
    a = np.exp(params - params.max())
    a = a / a.sum()
    beta = state['beta']
    top_k = state['top_k']
    n = len(a)
    
    ac = fast_autocorr(a)
    # Get top-k values
    top_idx = np.argpartition(ac, -top_k)[-top_k:]
    top_vals = ac[top_idx]
    
    # Log-sum-exp of top values
    max_v = top_vals.max()
    log_sum_exp = max_v + np.log(np.sum(np.exp(top_vals - max_v)))
    
    # Objective: 2n * (1/beta) * log_sum_exp
    obj = 2.0 * n * log_sum_exp / beta
    
    # Gradient w.r.t. a
    weights = np.exp(top_vals - max_v)
    weights = weights / weights.sum()
    
    grad_a = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(k_int, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad_a[idx] += 2.0 * wi * a[k_int - idx]
    
    # Chain rule through softmax: a = softmax(params)
    # da[i]/dp[j] = a[i] * (delta(i,j) - a[j])

    #                = grad_a[j] * a[j] - a[j] * sum_i grad_a[i] * a[i]
    
    grad_params = grad_a * a - a * np.dot(grad_a, a)
    
    # Scale by 2n/beta
    grad_params *= 2.0 * n / beta
    
    return obj, grad_params

def lbfgs_smooth(a_init, beta_schedule, time_limit):
    """
    Optimize using L-BFGS with log-sum-exp smoothing.
    a_init: initial point in simplex
    beta_schedule: list of beta values to try (increasing)
    """
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Setup softmax params from current a
        # a = exp(p) / sum(exp(p)) => p = log(a) (up to constant)
        a_safe = np.maximum(a, 1e-30)
        p0 = np.log(a_safe)
        
        state = {'beta': beta, 'top_k': min(50, 2*n-1), 'n': n}
        
        # Define objective and gradient (for scipy)
        def fun_grad(p):
            a_s = np.exp(p - p.max())
            a_s = a_s / a_s.sum()
            
            ac = fast_autocorr(a_s)
            top_k = state['top_k']
            top_idx = np.argpartition(ac, -top_k)[-top_k:]
            top_vals = ac[top_idx]
            
            max_v = top_vals.max()

            obj = 2.0 * n * lse / beta

            # Chain rule through softmax

            grad_p *= 2.0 * n / beta
            
            # For scipy, return (fun, grad)
            return obj, grad_p
        
        # Run L-BFGS-B

"""
Hybrid optimizer combining L-BFGS with adaptive log-sum-exp smoothing,
LP-based Frank-Wolfe, and coordinate descent to minimize
2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.

Key innovations over previous approach:
1. L-BFGS with softmax parameterization and beta annealing for superlinear convergence
2. Multi-scale beta schedule for global-to-local optimization
3. Adaptive active set selection for LP-FW
4. Vectorized gradient computation for speed
5. Multiple diverse initializations with spectral shaping
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return s


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def spectral_equalize(a, n_iter=30, eps=1e-8):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def lbfgs_smoothed(a_init, beta_list, top_k, time_limit, verbose=True):
    """
    L-BFGS optimization with log-sum-exp smoothing and softmax parameterization.
    Beta is annealed through beta_list (should be increasing).
    """
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()

    for beta in beta_list:
        if time.monotonic() - t0 > time_limit:
            break

        a_safe = np.maximum(a, 1e-30)
        p0 = np.log(a_safe)
        n_lags = 2 * n - 1
        tk = min(top_k, n_lags)

        def fun_and_grad(p):
            a_s = np.exp(p - p.max())
            a_s = a_s / a_s.sum()

            ac = fast_autocorr(a_s)
            top_idx = np.argpartition(ac, -tk)[-tk:]
            top_vals = ac[top_idx]

            max_v = top_vals.max()
            lse = max_v + np.log(np.sum(np.exp(top_vals - max_v)))
            obj = 2.0 * n * lse / beta

            weights = np.exp(top_vals - max_v)
            weights = weights / weights.sum()

            grad_a = np.zeros(n)
            for wi, k in zip(weights, top_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(k_int, n - 1)
                if i_start <= i_end:
                    idx = np.arange(i_start, i_end + 1)
                    grad_a[idx] += 2.0 * wi * a_s[k_int - idx]

            grad_p = grad_a * a_s - a_s * np.dot(grad_a, a_s)
            grad_p *= 2.0 * n / beta

            return obj, grad_p

        max_iter = min(800, max(100, int((time_limit - (time.monotonic() - t0)) * 30)))
        result = minimize(fun_and_grad, p0, method='L-BFGS-B',
                          jac=True,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                   'ftol': 1e-14, 'gtol': 1e-10,
                                   'gtol_rel': 1e-10})

        p_final = result.x
        a_final = np.exp(p_final - p_final.max())
        a_final = a_final / a_final.sum()

        score = evaluate_safe(a_final)
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS beta={beta:.0f} ({elapsed:.1f}s): "
                  f"smooth_obj={result.fun:.8f}, true_score={score:.6f}, "
                  f"iters={result.nit}")

        if score < best_score:
            best_score = score
            best_a = a_final.copy()
            a = a_final.copy()

    return best_a, best_score


def lp_fw_step(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized direction or None.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)

    if not constraints:
        return None

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def accelerated_lp_fw(a0, time_limit, solver, top_k_init=50, top_k_max=200,
                      verbose=True):
    """Accelerated LP Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        active_lags = np.argsort(ac)[::-1][:current_top_k]
        direction = lp_fw_step(y, M_curr, active_lags, solver)
        if direction is None:
            continue

        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if t % 100 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 10, top_k_max)

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, rng_seed=42, verbose=True):
    """Peak-targeted coordinate descent with line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    move_fracs = [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.06, 0.1]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(300, len(candidates))]]

            best_dst = -1
            best_score_val = float('inf')

            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    if 0 <= d1 < n:
                        sens += a[d1]
                    d2 = dst - k2i
                    if 0 <= d2 < n and d2 != d1:
                        sens += a[d2]
                if sens < best_score_val:
                    best_score_val = sens
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in move_fracs:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and not improved and rng.random() < 0.3:
                # Small random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0) / max(a.sum(), 1e-15)

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def mirrored_gradient_descent(a0, time_limit, top_k=50, beta=4.0,
                              c0=3.0, verbose=True):
    """Nesterov-accelerated mirror descent (exponential family)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = y
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 2000:
            stall_count = 0
            pert = rng.normal(0, 0.004, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            try_update(a_out, f"lbfgs_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break

    # === Phase 9: Final L-BFGS + Coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final refine, n={n_best}, budget={remaining:.0f}s ===")
    if remaining > 60:
        t9a = min(remaining * 0.5, 120.0)
        if t9a > 20:
            beta_final = [100, 500, 2000]
            a_in = normalize(best_seq)
            a_out, _ = lbfgs_smoothed(a_in, beta_final, top_k=15,
                                      time_limit=t9a, verbose=True)
            try_update(a_out, "lbfgs_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9b = min(remaining * 0.7, 60.0)
            if t9b > 10:
                a_in = normalize(best_seq)
                a_out, _ = coordinate_descent_polish(a_in, t9b,
                                                     rng_seed=777, verbose=True)
                try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
