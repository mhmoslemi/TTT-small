import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:

    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_val(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    C = fast_conv(a)
    return float(2.0 * n * C.max() / (s * s))

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_subgradient(a, tau):
    """Compute smooth subgradient of tau*logsumexp(C/tau) w.r.t. a, and return C."""
    n = len(a)
    C = fast_conv(a)
    C_max = C.max()
    exp_C = np.exp((C - C_max) / tau)
    w = exp_C / exp_C.sum()
    # grad[i] = 2 * sum_j w[i+j] * a[j] via cross-correlation
    L = 1
    while L < 3 * n:
        L *= 2
    W = np.zeros(L)
    A = np.zeros(L)
    W[:2 * n - 1] = w
    A[:n] = a
    corr = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
    return 2.0 * corr[:n], C

def hard_subgradient(a, k_star):
    """Exact subgradient at the peak location."""
    n = len(a)
    k = int(k_star)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return grad

def nesterov_smooth(a0, time_limit, verbose=True, seed=42, tau_init=0.01, tau_end=0.001, c0=4.0):
    """Nesterov mirror descent with smoothed logsumexp subgradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = objective_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 4000
    tau = tau_init
    log_tau_ratio = np.log(tau_end / tau_init)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        C = fast_conv(y)
        M_curr = C.max()

        # Current objective (normalized)
        curr_val = 2.0 * n * M_curr  # since sum(y)=1

        # Decay tau geometrically towards tau_end over expected iterations
        tau_t = tau_init * np.exp(log_tau_ratio * min(t, 50000.0) / 50000.0)
        tau_t = max(tau_t, tau_end)

        g, _ = smooth_subgradient(y, tau_t)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - eta * g
        mx = log_new.max()
        a_new = np.exp(log_new - mx)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        val_new = 2.0 * n * M_new

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            x_restarted = best_a + rng.normal(0, 0.003, n) * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            tau = max(tau * 2.0, 0.01)

        if t > 0 and t % 20000 == 0:
            x_jumped = best_a + rng.normal(0, 0.01, n) * best_a
            x_jumped = np.maximum(x_jumped, 1e-8)
            x_jumped = x_jumped / x_jumped.sum()
            M_jump = fast_conv(x_jumped).max()

                x = x_jumped
                x_prev = x_jumped

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAMD t={t} ({elapsed:.1f}s): obj={best_val:.6f}, tau={tau_t:.5f}, stall={stall}")

    return best_a, best_val

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=200):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        alphas = [2.0 / (t + 2.0), 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05]
        best_alpha = alphas[0]
        best_eval = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()

        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        if t == 1:
            x_new = y_new.copy()
        else:
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_conv(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with targeted mass transfer from peak-contributing positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:8]
        improved = False

        for kp in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(kp)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src_candidates = idx[np.argsort(contrib)[::-1][:5]]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            for src in src_candidates:
                if a[src] < 1e-12:
                    continue
                best_dst = -1
                best_score = float('inf')
                for dst in chosen:
                    if dst == src:
                        continue
                    sens = a[dst] * a[k_int - dst] if 0 <= k_int - dst < n else 0.0
                    if sens < best_score:
                        best_score = sens
                        best_dst = dst

                if best_dst < 0:
                    continue
                dst = best_dst

                found = False
                for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.15]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.8, 80.0)
        if t8 > 5:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
