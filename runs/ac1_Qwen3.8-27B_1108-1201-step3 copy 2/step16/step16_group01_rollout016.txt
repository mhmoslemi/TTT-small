import numpy as np
import time

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} in Euclidean distance."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, C, beta):
    """
    Compute gradient of f(a) = (1/beta) * log(sum_l exp(beta * C[l]))
    where C = conv(a, a).
    
    grad[k] = (2/beta) * sum_j w[k+j] * a[j]
    where w[l] = exp(beta*(C[l]-Cmax)) / sum(exp(beta*(C[Cmax)]))
    
    This is a cross-correlation, computable via FFT.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Compute softmax weights (numerically stable)
    C_shifted = C - C.max()
    w = np.exp(beta * C_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform weights
        w = np.ones(size) / size
    else:
        w = w / w_sum
    
    # Cross-correlation: g[k] = sum_j w[k+j] * a[j]
    # This is (w reversed) convolved with a, or equivalently:
    # g = ifft(fft(w_padded) * fft(a_padded)) with appropriate indexing
    
    # Method: use the fact that g[k] = sum_j w[k+j] * a[j]
    # Let's define w_rev[m] = w[size-1-m] (reverse w)
    # Then g[k] = sum_j w_rev[size-1-k-j] * a[j] = (w_rev * a)[size-1-k]
    # So g = ifft(fft(w_rev) * fft(a))[size-1-k] for k = 0..n-1
    
    fft_size = 1 << (size - 1).bit_length()
    w_rev = w[::-1]  # w_rev[m] = w[size-1-m]
    
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_full = np.fft.irfft(W * A, fft_size)
    
    # g[k] = conv_full[size-1-k] for k=0..n-1
    g = conv_full[size-1 : size-1 - n : -1]
    g = (2.0 / beta) * g
    
    return g

def estimate_lipschitz(w, beta):
    """Estimate Lipschitz constant of the smooth gradient."""
    n = len(w)
    fft_size = 1 << (n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    W_rev = np.fft.rfft(w[::-1], fft_size)
    # The operator norm of cross-correlation is max |W[m]|
    max_W = np.max(np.abs(W))
    return (2.0 / beta) * max_W

def compute_smooth_gradient_fft(a, C, beta):

    size = 2 * n - 1
    
    C_shifted = C - C.max()
    w = np.exp(beta * C_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(size) / size
    else:
        w = w / w_sum
    
    w_rev = w[::-1]
    
    # Need fft_size >= len(w_rev) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 3).bit_length()
    
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(W * A, fft_size)
    
    # g[k] = (2/beta) * conv[size-1-k] for k=0..n-1
    # indices: size-1, size-2, ..., size-n = 2n-2, 2n-3, ..., n-1
    g = conv[size-1 : size-1-n : -1].copy()
    g = (2.0 / beta) * g
    
    return g

import numpy as np
import time

def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(a, C, beta, fft_size=None):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*C_l)) w.r.t. a.
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    size = 2 * n - 1
    
    if fft_size is None:
        fft_size = 1 << (3 * n - 3).bit_length()
    
    # Softmax weights (numerically stable)
    C_shift = C - C.max()
    w = np.exp(beta * C_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(size) / size
    else:
        w = w / w_sum
    
    # Cross-correlation via FFT
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(W * A, fft_size)
    
    # g[k] = (2/beta) * conv[2n-2-k] for k=0..n-1
    g = conv[2*n-2 : n-2 : -1].copy()
    g = (2.0 / beta) * g
    
    return g, w

    """Upper bound on Lipschitz constant of the smooth gradient."""
    size = len(w)
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    return (2.0 / beta) * np.max(np.abs(W))

                       verbose=True, rng_seed=42):

    """

    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    C = fast_conv(a)
    best_a = a.copy()
    best_Cmax = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    fft_size = 1 << (3 * n - 3).bit_length()
    
    stall = 0
    max_stall = 800
    L_ema = 1.0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        t = iter_count
        
        # Adaptive beta: start smooth, get sharper
        frac = min(1.0, t / 3000.0)
        beta = beta_start + (beta_end - beta_start) * frac * frac  # quadratic annealing
        
        # Nesterov momentum

        y = a + mu * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:

        else:
            y = y / sy

        a = a_new

        Cmax = Ca.max()
        if Cmax < best_Cmax - 1e-14:

            best_a = a.copy()
            stall = 0
        else:

            # Reset EMA
