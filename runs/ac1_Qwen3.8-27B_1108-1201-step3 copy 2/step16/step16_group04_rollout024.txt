  = (1/S²) [n*S - sum_{j=1}^{n} j(j-1)]
  = (1/S²) [n*S - (sum j² - sum j)]
  = (1/S²) [n*S - (n(n+1)(2n+1)/6 - n(n+1)/2)]
  = (1/S²) [n*S - n(n+1)(2n+1-3)/6]

  = (1/S²) [n*S - n(n+1)(n-1)/3]

  = (1/S²) * n * [n(n+1)/2 - (n²-1)/3]
  = (1/S²) * n * [3n(n+1) - 2(n²-1)] / 6
  = (1/S²) * n * [3n²+3n-2n²+2] / 6
  = (1/S²) * n * [n²+3n+2] / 6
  = (1/S²) * n * (n+1)(n+2) / 6

S = n(n+1)/2, so S² = n²(n+1)²/4.

conv[n-1] = n(n+1)(n+2)/6 / (n²(n+1)²/4) = 4n(n+1)(n+2) / (6n²(n+1)²) = 4(n+2) / (6n(n+1)) = 2(n+2)/(3n(n+1))

```python
def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_simplex(a):
    """Project/normalize to simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def make_ramp(n, alpha=1.0):
    """Create a ramp-like initialization: a_j ∝ (j+1)^alpha."""
    j = np.arange(1, n+1, dtype=np.float64)
    a = j ** alpha
    return a / a.sum()

def make_exp_ramp(n, rate=1.0):
    """Create an exponential ramp: a_j ∝ exp(rate * j / n)."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(rate * j / n)

def spectral_equalize(a, n_iter=50, temperature=0.5):
    """Flatten the power spectrum iteratively."""
    a = normalize_simplex(a).copy()
    n = len(a)
    for it in range(n_iter):

        P = np.abs(A) ** 2
        # Target: flat spectrum

        A_mod = A / (np.abs(A) ** temperature + 1e-15)

    grad = np.zeros(n)
    # For each i, grad[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n and k-i != i)
    # = 2*a[k-i] if 0<=k-i<n, with special case when k-i==i giving 2*a[i] (same thing)
    # Actually: grad[i] = a[k-i] + a[i-k+2k-i]... no let me think again.
    # f_k = sum_j a[j]*a[k-j]
    # d f_k / d a[i] = sum_j [1(i==j)*a[k-j] + a[j]*1(k-j==i)]
```

Hmm, I realize the `fw_backtracking` function above is not really doing a proper Frank-Wolfe because I'm not solving the LP. Let me reconsider.

For a proper FW, I need to solve:
min_{b in simplex} ⟨∇f(x), b⟩  (for smooth f)
or for the non-smooth case: min_{b in simplex} max_k (b^T C_k b + 2*linear terms) ... this gets complicated.

The standard FW for non-smooth problems uses a "linearized" oracle. For our problem, at the current point y, we want to find b that minimizes the "worst-case" increase:
max_k (conv(y+α(b-y), y+α(b-y))[k] - conv(y,y)[k])

For small α, this is approximately:
max_k 2α * (b-y)^T C_k y ... hmm this is getting complicated.

Let me go back to the approach used in the previous code: solve the LP
max sum(b) s.t. conv(y, b)[k] ≤ M for k in active set, b ≥ 0.

This works because if we move from y towards b (i.e., x = (1-α)y + αb), then for small α:
conv(x, x)[k] ≈ conv(y,y)[k] + 2α(conv(y,b)[k] - conv(y,y)[k])

So if conv(y,b)[k] ≤ M = max_k conv(y,y)[k], then for the peak lags, conv(x,x)[k] ≤ M + 2α(M - M) = M (no increase). And for non-peak lags, it might decrease.

The LP solution gives a direction that doesn't increase the peak (for active constraints) while maximizing the "spread" of b.

Let me use scipy's linprog for this instead of cvxpy, which should be faster for this size.

Actually, for n=2400 and ~500 constraints, the LP has 2400 variables and 500 constraints. scipy's HiGHS solver should handle this in a fraction of a second.

```python
from scipy.optimize import linprog

def solve_lp_direction(y, M, active_lags):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(y)
    # c = -1 (minimize -sum(b))
    c = -np.ones(n)
    
    # A_ub: each row is a constraint (y*b)[k] <= M
    # (y*b)[k] = sum_j y[j]*b[k-j] where 0<=j<n and 0<=k-j<n
    A_rows = []
    b_vals = []
    for k in active_lags:
        k = int(k)
        # b indices: j where 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        # row: for b[j], coefficient is y[k-j]
        row = np.zeros(n)
        for j in range(j_start, j_end + 1):
            row[j] = y[k - j]
        A_rows.append(row)
        b_vals.append(M)
    
    if not A_rows:
        return None
    
    A_ub = np.array(A_rows)
    
    # b >= 0: bounds = [(0, None)] * n
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_vals, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    b_val = np.maximum(res.x, 0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum
```
