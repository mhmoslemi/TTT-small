def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_weights(ac, temperature):
    """Compute softmax weights over autocorrelation lags."""
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    w = np.exp(log_ac)
    s = w.sum()
    if s < 1e-30:
        idx = np.argmax(ac)
        w = np.zeros_like(ac)
        w[idx] = 1.0
    else:
        w /= s
    return w

def topk_weights(ac, top_k, beta=3.0):
    """Compute weights concentrated on top-k lags."""
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    w_full = np.zeros(len(ac))
    vals = np.maximum(ac[top_idx], 1e-30)
    w_top = vals ** beta
    s = w_top.sum()
    if s < 1e-30:
        w_top = np.ones(k_max) / k_max
    else:
        w_top /= s
    w_full[top_idx] = w_top
    return w_full

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1
    total_iters = 0

    while total_iters < 10000000:
        total_iters += 1
        t = total_iters
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive parameters based on progress
        frac = min(1.0, t / 3000.0)
        temp = initial_temp * (1 - frac)**2 + final_temp * frac**2
        temp = max(temp, 1e-5)
        top_k = max(top_k_end, int(top_k_start + (top_k_end - top_k_start) * frac))

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Choose subgradient method based on temperature
        if temp > 0.01:
            weights = softmax_weights(y * ac, temp)
        else:
            weights = topk_weights(ac, top_k, beta=5.0)

        g = compute_grad_with_weights(y, ac, weights)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart strategy
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.7)

        # Ergodic average
        running_avg += x
        avg_count += 1
        if avg_count % 1000 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD2 t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, topk={top_k}, stall={stall_count}")

    return best_a, best_m

def fista_proximal(a0, time_limit, initial_temp=0.01, final_temp=0.001, verbose=True):
    """FISTA (accelerated proximal gradient) with Euclidean geometry."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    x = a.copy()
    z = a.copy()  # Nesterov momentum variable
    t_curr = 1.0
    gamma = 3.0
    stall_count = 0
    max_stall = 3000

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, it / 5000.0)
        temp = initial_temp * (1 - frac) + final_temp * frac
        temp = max(temp, 1e-5)

        ac = fast_autocorr(z)
        weights = softmax_weights(ac, temp)
        g = compute_grad_with_weights(z, ac, weights)

        eta = gamma / (t_curr + 1.0)
        x_trial = z - eta * g
        x_new = project_simplex(x_trial)

        # Nesterov update
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr))
        z = x_new + (t_curr - 1.0) / t_next * (x_new - x)
        z = np.maximum(z, 0.0)
        sz = z.sum()
        if sz < 1e-15:
            z = x_new.copy()
        else:
            z = z / sz  # reproject

        t_curr = t_next
        x = x_new

        M_x = fast_autocorr(x).max()
        M_z = ac.max()
        if M_x < best_m:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0
        elif M_z < best_m:
            best_m = M_z
            best_a = z.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.005, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 0.0)
            x_r = x_r / x_r.sum()
            x = x_r
            z = x_r.copy()
            t_curr = 1.0
            gamma = max(1.0, gamma * 0.6)

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA it={it} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, stall={stall_count}")

    return best_a, best_m

def lp_frank_wolfe_v2(a0, time_limit, solver, max_constraints=400, verbose=True):
    """LP Frank-Wolfe with structural constraints and adaptive constraint set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Select constraints: top lags + structural lags (0, n-1, 2n-2)
        structural = [0, n - 1, 2 * n - 2]
        threshold = M_curr * 0.95
        active_lags = list(set(np.where(ac >= threshold)[0].tolist() + [s for s in structural if 0 <= s < len(ac)]))
        if len(active_lags) > max_constraints:
            # Keep top by value
            active_lags = sorted(active_lags, key=lambda k: -ac[k])[:max_constraints]

        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={t} ({elapsed:.1f}s): J={J:.6f}, active={len(active_lags)}")

    return best_a, best_m

def subgradient_local_search(a0, time_limit, verbose=True, rng_seed=42):
    """Local search guided by subgradient information with exact 2D optimization."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 5

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find the top lags
        top_lags = np.argsort(ac)[::-1][:20]
        
        # For each top lag, find the coordinate that contributes most
        # and try to reduce it by transferring mass to a "safe" coordinate
        
        improved = False
        for k in top_lags[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            contribs = a[idx] * a[k_int - idx]
            
            # Find the coordinate with highest contribution
            max_contrib_idx = idx[np.argmax(contribs)]
            max_contrib_partner = k_int - max_contrib_idx
            
            # Try transferring mass from max_contrib_idx to a random "safe" position
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == max_contrib_idx or dst == max_contrib_partner:
                    continue
                
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[max_contrib_idx] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[max_contrib_idx] -= amt
                    a_trial[dst] += amt
                    if a_trial[max_contrib_idx] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        a = a / a.sum()
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGLS ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m
