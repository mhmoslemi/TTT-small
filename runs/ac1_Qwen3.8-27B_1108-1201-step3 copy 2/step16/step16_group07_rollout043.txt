            if s < 1e-15:

                a[j] += delta
                continue

            if rng.random() > np.exp(-delta_score / temp):

def efficient_coord_descent(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = conv.max() / (a.sum()**2) * 2 * n
    # Actually let's just track max_conv since sum is fixed
    best_max_conv = conv.max()
    t0 = time.monotonic()
    
    n_passes = 0
    no_improve = 0
    max_no_improve = 5
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        # Randomize order
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            c = a[j]
            s_others = a.sum() - c
            
            # Try candidate x values
            # The optimal x should be near c (small perturbation) or near 0
            candidates = [0.0, c * 0.1, c * 0.3, c * 0.5, c * 0.7, c, c * 1.3, c * 1.5, c * 2.0]
            # Add some values based on neighbors
            neighbors = [a[max(0,j-1)], a[min(n-1,j+1)]] if n > 1 else [c]
            candidates.extend([0.5 * x for x in neighbors])
            candidates.extend([x for x in neighbors])
            
            best_x = c
            best_val = conv.max()
            
            for x in sorted(set(max(0.0, min(v, norm)) for v in candidates)):
                if abs(x - c) < 1e-12:
                    continue
                
                # Update convolution
                delta = x - c
                new_conv = conv.copy()
                # (a*a)[k] changes by 2*delta*a[k-j] for k in [j, j+n-1] where 0<=k-j<n
                # And for k=2j, additional delta^2
                lo = j
                hi = min(2*n-1, j + n)
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a[lo-j:hi-j]
                if 2*j < 2*n-1:
                    new_conv[2*j] += delta * delta  # Correction for the x^2 term
                
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                
                # Scale
                scale = norm / new_sum
                new_max = (new_conv * scale * scale).max()
                
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_x = x
            
            if abs(best_x - c) > 1e-12:
                # Apply the change
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    conv = fast_autocorr(a)
                    max_conv = conv.max()
                    if max_conv < best_max_conv - 1e-14:
                        best_max_conv = max_conv
                        best_a = a.copy()
                        best_conv = conv.copy()

**Peak-targeted local search:**
1. Compute conv(a,a) and find the maximum peak at index k*.
2. The gradient at k* tells us which a[i] values contribute most to this peak.
3. Specifically, d(conv[a,a][k*])/a[i] = a[k*-i] + a[i-k*] (for valid indices).
4. We want to reduce a[i] where this gradient is large (these contribute most to the peak).

            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 1.5)
        else:
            patience += 1
            step *= 0.993
            if step < 1e-12:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PTS t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score

def efficient_coord_descent(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_max = conv.max()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    max_no_improve = 8
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            c = a[j]
            lo = j
            hi = min(size, j + n)
            if lo >= hi:
                continue

            typical = norm_val / n
            candidate_list = [0.0, c * 0.3, c * 0.6, c * 1.4, c * 2.0, typical * 0.5, typical * 1.5]
            if j > 0:
                candidate_list.append((c + a[j - 1]) * 0.5)
            if j < n - 1:
                candidate_list.append((c + a[j + 1]) * 0.5)

            best_x = c
            best_val = conv.max()

            for x in sorted(set(max(0.0, min(v, norm_val * 0.5)) for v in candidate_list)):
                if abs(x - c) < 1e-12:
                    continue

                delta = x - c
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                new_max_local = new_conv.max()
                if lo > 0:
                    new_max_local = max(new_max_local, conv[:lo].max())
                if hi < size:
                    new_max_local = max(new_max_local, conv[hi:].max())

                if new_max_local < best_val - 1e-14:
                    best_val = new_max_local
                    best_x = x

            if abs(best_x - c) > 1e-12:
                old_slice = a[lo - j:hi - j].copy()
                a[j] = best_x
                delta = best_x - c
                conv[lo:hi] += 2.0 * delta * old_slice
                if 2 * j < size:
                    conv[2 * j] += delta * delta

                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
                    conv = fast_autocorr(a)
                    if conv.max() < best_max - 1e-14:
                        best_max = conv.max()
                        best_a = a.copy()
                        best_conv = conv.copy()
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = best_conv.copy()

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD pass={n_passes} ({elapsed:.0f}s): best={best_max:.10f}")

    return best_a, best_max

def local_search_sa(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_max = conv.max()
    t0 = time.monotonic()

    temp = 0.005 * best_max
    cooldown = 0.9997
    min_temp = 1e-10
    n_evals = 0

    while time.monotonic() - t0 < time_limit and temp > min_temp:
        move_type = rng.random()

        if move_type < 0.4:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            scale = max(a[i], norm_val / n * 0.05)
            delta = scale * rng.standard_normal() * 0.3
            if a[i] - delta < 0 or a[j] + delta < 0:
                temp *= cooldown
                continue
            old_i = a[i]
            old_j = a[j]
            a[i] -= delta
            a[j] += delta
            s = a.sum()
            if s < 1e-15:
                a[i] = old_i
                a[j] = old_j
                temp *= cooldown
                continue
            a = a * norm_val / s

        elif move_type < 0.6:
            block_size = max(1, n // 60)
            i = rng.integers(0, max(1, n - block_size))
            shift = int(rng.integers(-block_size * 4, block_size * 4 + 1))
            if shift == 0:
                temp *= cooldown
                continue
            j = i + shift
            if j < 0 or j + block_size > n:
                temp *= cooldown
                continue
            alpha = 0.05 + 0.15 * rng.random()
            old_a = a.copy()
            diff = old_a[i:i + block_size] - old_a[j:j + block_size]
            a[i:i + block_size] -= alpha * diff
            a[j:j + block_size] += alpha * diff
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = old_a
                temp *= cooldown
                continue
            a = a * norm_val / s

        elif move_type < 0.8:
            i = rng.integers(n)
            c = a[i]
            if c < 1e-15:
                temp *= cooldown
                continue
            new_val = c * (1.0 + 0.5 * rng.standard_normal())
            new_val = max(0.0, min(new_val, norm_val * 0.5))
            if abs(new_val - c) < 1e-12:
                temp *= cooldown
                continue
            a[i] = new_val
            s = a.sum()
            if s < 1e-15:
                a[i] = c
                temp *= cooldown
                continue
            a = a * norm_val / s

        else:
            sigma = 0.5 + 3.0 * rng.random()
            k_size = int(6 * sigma) | 1
            k_half = k_size // 2
            x_k = np.arange(-k_half, k_half + 1)
            kernel = np.exp(-x_k ** 2 / (2 * sigma ** 2))
            kernel /= kernel.sum()
            alpha = 0.03 + 0.1 * rng.random()
            smoothed = np.convolve(a, kernel, mode='same')
            a_new = a - alpha * (a - smoothed)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                temp *= cooldown
                continue
            a = a_new * norm_val / s

        conv = fast_autocorr(a)
        cur_max = conv.max()
        n_evals += 1

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_conv = conv.copy()
        else:
            delta_score = cur_max - best_max
            if rng.random() > np.exp(-delta_score / temp):
                a = best_a.copy()
                conv = best_conv.copy()

        temp *= cooldown

        if verbose and n_evals % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA evals={n_evals} ({elapsed:.0f}s): best={best_max:.10f}, temp={temp:.8f}")

    return best_a, best_max

def generate_beta_start(n, alpha, beta):
    x = np.linspace(0.01, 0.99, n)
    a = x ** (alpha - 1) * (1.0 - x) ** (beta - 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a

def generate_arcsine_start(n):
    x = np.linspace(0.01, 0.99, n)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a

def generate_w_shaped_start(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a

def generate_cosine_start(n, freq, amp):
    x = np.linspace(0, 1, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a

def diverse_start(n, idx):
    if idx == 0:
        return generate_beta_start(n, 0.5, 2.0)
    elif idx == 1:
        return generate_beta_start(n, 2.0, 0.5)
    elif idx == 2:
        return generate_arcsine_start(n)
    elif idx == 3:
        return generate_w_shaped_start(n)
    elif idx == 4:
        return generate_cosine_start(n, 1.0, 0.5)
    elif idx == 5:
        return generate_cosine_start(n, 2.0, 0.4)
    elif idx == 6:
        return generate_cosine_start(n, 3.0, 0.3)
    elif idx == 7:
        return np.ones(n) / n
    elif idx == 8:
        return generate_beta_start(n, 0.7, 1.3)
    elif idx == 9:
        return generate_cosine_start(n, 1.5, 0.6)
    else:
        return np.ones(n) / n
