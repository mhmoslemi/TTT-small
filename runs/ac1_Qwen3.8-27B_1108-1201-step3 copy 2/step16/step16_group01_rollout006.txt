def fast_autocorr(a):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_lag_gradient(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * a[j]
    return g

def temperature_subgradient(a, ac, temperature=1.0, top_k=None):
    """
    Compute subgradient with temperature-based weighting.
    temperature=0: only the max lag (exact)
    temperature=inf: all lags weighted equally
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    if top_k is None:
        top_k = min(50, len(ac))
    
    # Get top-k lags
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]
    
    # Temperature-based weights
    if temperature < 1e-10:
        # Only use the max
        g = compute_lag_gradient(a, int(top_idx[0]))
        return g
    
    # Softmax-like weights with temperature
    shifted = top_vals - top_vals[0]
    if temperature < 1.0:
        weights = np.exp(shifted / (temperature * M + 1e-30))
    else:
        weights = top_vals / (top_vals.sum() + 1e-30)
    
    weights = weights / (weights.sum() + 1e-30)
    
    g = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        if wi < 1e-10:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                g[i] += 2.0 * wi * a[j]
    
    return g

    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_gradient_active(a, ac, threshold_frac):
    """Compute subgradient from active set (lags above threshold_frac * max)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    active_mask = ac >= threshold_frac * M
    g = np.zeros(n)
    for k in range(len(ac)):
        if not active_mask[k]:
            continue
        k_int = k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                g[i] += 2.0 * a[j]
    return g

def compute_gradient_topk(a, ac, top_k, beta):
    """Compute soft subgradient from top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                g[i] += 2.0 * wi * a[j]
    return g

def amd_phase(a0, time_limit, mode, verbose=True, rng_seed=42):
    """
    AMD with a specific mode:
    - mode='wide': use top-k soft gradient (broad view)
    - mode='narrow': use exact active-set gradient (focused)
    - mode='adaptive': switch from wide to narrow over time
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    
    c0 = 3.0
    if mode == 'wide':
        top_k = 50
        beta = 3.0
    elif mode == 'narrow':
        top_k = 5
        beta = 20.0
    else:  # adaptive
        top_k = 30
        beta = 5.0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        
        # Compute subgradient based on mode
        if mode == 'wide':
            g = compute_gradient_topk(y, ac, top_k=top_k, beta=beta)
        elif mode == 'narrow':
            g = compute_gradient_topk(y, ac, top_k=top_k, beta=beta)
        else:  # adaptive: interpolate between wide and narrow
            frac = min(t / 50000.0, 1.0)  # transition over 50k iters
            effective_k = int(50 - 45 * frac)  # 50 -> 5
            effective_beta = 3.0 + 17.0 * frac  # 3 -> 20
            g = compute_gradient_topk(y, ac, top_k=max(3, effective_k), beta=effective_beta)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
        
        # Mirror update
        log_y = np.log(np.maximum(y, 1e-15))
        log_new = log_y - eta * g
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
