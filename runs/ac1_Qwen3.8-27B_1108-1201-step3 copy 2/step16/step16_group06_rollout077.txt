```
    for wi, k in zip(w, idx):
       k = int(k); wi=float(wi)
       if wi == 0: continue
       js = np.arange(max(0, k-n+1), min(n-1, k)+1)
       if js.size:
           sens[js] += wi * a[k - js]
       js2 = np.arange(max(0, k), min(n-1, k+n-1)+1)
       if js2.size:
           sens[js2] += wi * a[js2 - k]
    return sens
```

```python
"""
Search for low-score nonnegative step-function heights.

Main method:
- minimize the p-norm surrogate of max(conv(a, a)) under sum(a)=1, a>=0
- use a fixed-coordinate softmax reparameterization for the simplex constraint
- increasing p anneals the surrogate toward the true max objective
- FFT-based derivatives and scores
- local restarts, length exploration, and peak-sensitive coordinate polish

The final sequence is validated with the provided evaluate_sequence function.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        return arr
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1000.0)
    s = float(arr.sum())
    if s <= 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _free_indices(n, fix):
    n = int(n)
    fix = int(fix)
    if n <= 1:
        return np.empty(0, dtype=np.int64)
    if fix < 0:
        fix = 0
    if fix >= n:
        fix = n - 1
    return np.concatenate((np.arange(0, fix, dtype=np.int64), np.arange(fix + 1, n, dtype=np.int64)))


def _softmax_with_fix(z, n, fix, free_indices):
    n = int(n)
    fix = int(fix)
    out = np.empty(max(n, 0), dtype=np.float64)
    if n == 0:
        return out
    if fix < 0:
        fix = 0
    if fix >= n:
        fix = n - 1
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size == 0:
        out[fix] = 1.0
        return out
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.0)
    m = float(np.max(z))
    if m < 0.0:
        m = 0.0
    ez = np.exp(z - m)
    efix = np.exp(-m)
    den = float(ez.sum() + efix)
    if den <= 1e-300 or not np.isfinite(den):
        return np.ones(n, dtype=np.float64) / float(n)
    if free_indices.size:
        out[free_indices] = ez / den
    out[fix] = efix / den
    return out


def _init_z_free_from_a(a, fix=None):
    a = _norm_sum1(a)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64), a, 0
    if fix is None:
        fix = int(np.argmax(a))
    fix = int(fix)
    if fix < 0 or fix >= n:
        fix = 0
    if a[fix] <= 1e-300:
        a = np.ones(n, dtype=np.float64) / float(n)
        fix = n // 2
    free = _free_indices(n, fix)
    z = np.empty(max(n - 1, 0), dtype=np.float64)
    if z.size:
        denom = max(float(a[fix]), 1e-300)
        vals = np.maximum(a[free], 1e-300) / denom
        z = np.log(vals)
    return z, a, fix


def _fast_convolution(a, fsize):
    n = int(a.size)
    A = np.fft.rfft(a, int(fsize))
    out = np.fft.irfft(A * A, int(fsize))
    return out[:max(1, 2 * n - 1)]


def _score_fft(a):
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        return float("inf")
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 0.01:
        return float("inf")
    arr = arr / s
    n = int(arr.size)
    fsize = _next_pow2(2 * n - 1)
    conv = _fast_convolution(arr, fsize)
    conv = np.maximum(conv, 0.0)
    if conv.size == 0:
        return float("inf")
    M = float(np.max(conv))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M)


def _actual_score(a):
    try:
        arr = np.asarray(a, dtype=np.float64).reshape(-1)
        if arr.size == 0:
            return float("inf")
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if float(arr.sum()) <= 0.01:
            return float("inf")
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def _weighted_deriv(weights, A, n, fsize):
    """
    Derivative contribution of a weighted sum of self-convolution lags.

    For C_k = sum_i a_i a_{k-i},
    dC_k/da_j = 2 a_{k-j} if 0 <= k-j < n, else 0.
    This routine returns sum_k weights[k] * dC_k/d a.
    """
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    rweights = weights[::-1]
    Rw = np.fft.rfft(rweights, int(fsize))
    conv = np.fft.irfft(Rw * A, int(fsize))
    part = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * part


def _schedule_total(schedule):
    s = 0.0
    for _, fr in schedule:
        s += float(fr)
    return s


def _make_lbfgs_state(n, fix, p, deadline):
    n = int(n)
    fix = int(fix)
    free = _free_indices(n, fix)
    return {
        "n": n,
        "fix": fix,
        "free": free,
        "p": float(p),
        "fsize": _next_pow2(3 * n - 2),
        "deadline": float(deadline),
        "stop": False,
        "last_f": 0.0,
    }


def _smooth_obj_grad(z, state):
    n = int(state["n"])
    fsize = int(state["fsize"])
    fix = int(state["fix"])
    free = state["free"]
    p = float(state["p"])

    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if n <= 1:
        return 0.0, np.empty(0, dtype=np.float64)
    if z.size != n - 1:
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    if state.get("stop", False) or time.monotonic() > state["deadline"]:
        state["stop"] = True
        return state.get("last_f", 0.0), np.zeros(n - 1, dtype=np.float64)

    a = _softmax_with_fix(z, n, fix, free)
    if not np.all(np.isfinite(a)):
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    A = np.fft.rfft(a, fsize)
    conv = np.fft.irfft(A * A, fsize)
    conv = conv[:2 * n - 1]
    conv = np.maximum(conv, 0.0)

    Cmax = float(np.max(conv)) if conv.size else 0.0
    if Cmax <= 1e-300 or not np.isfinite(Cmax):
        state["last_f"] = 0.0
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    u = np.clip(conv / Cmax, 0.0, 1.0)
    if p < 2.0:
        p = 2.0

    up = np.power(u, p)
    sum_up = float(np.sum(up))

    if sum_up <= 1e-300 or not np.isfinite(sum_up):
        weights = np.zeros_like(conv)
        weights[int(np.argmax(conv))] = 1.0
        f = 2.0 * n * Cmax
    else:
        upp = np.power(u, p - 1.0)
        denom = sum_up ** ((p - 1.0) / p)
        if denom <= 1e-300 or not np.isfinite(denom):
            weights = np.zeros_like(conv)
            weights[int(np.argmax(conv))] = 1.0
            f = 2.0 * n * Cmax
        else:
            weights = upp / denom
            if not np.all(np.isfinite(weights)):
                weights = np.where(np.isfinite(weights), weights, 0.0)
                sw = float(weights.sum())
                if sw <= 1e-300:
                    weights = np.zeros_like(conv)
                    weights[int(np.argmax(conv))] = 1.0
                else:
                    weights = weights / sw
            f = 2.0 * n * Cmax * (sum_up ** (1.0 / p))

    if not np.isfinite(f):
        f = 1e100

    gw = _weighted_deriv(weights, A, n, fsize)
    gw = np.where(np.isfinite(gw), gw, 0.0)

    # Objective is 2n * N, and dC/d a already contains factor 2.
    grad_a = (2.0 * n) * gw
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)

    mean = float(np.dot(a, grad_a))
    if not np.isfinite(mean):
        mean = 0.0

    g = a[free] * (grad_a[free] - mean)
    g = np.where(np.isfinite(g), g, 0.0)

    if np.isfinite(f):
        state["last_f"] = f
    return f, g


def _smooth_obj_f(z, state):
    return _smooth_obj_grad(z, state)[0]


def _smooth_obj_g(z, state):
    return _smooth_obj_grad(z, state)[1]


def _run_lbfgsb(z0, n, fix, p, deadline):
    z0 = np.asarray(z0, dtype=np.float64).reshape(-1).copy()
    n = int(n)
    if n <= 1:
        return z0
    if time.monotonic() >= deadline:
        return z0

    options = {
        "maxfun": 200000,
        "maxiter": 200000,
        "ftol": 1e-12,
        "gtol": 1e-9,
        "maxls": 50,
        "maxcor": 20,
        "iprint": -1,
    }

    state = _make_lbfgs_state(n, fix, p, deadline)
    try:
        res = minimize(_smooth_obj_grad, z0, method="L-BFGS-B", jac=True, args=(state,), options=options)
        if hasattr(res, "x"):
            x = np.asarray(res.x, dtype=np.float64)
            if x.size == n - 1 and np.all(np.isfinite(x)):
                return x.copy()
    except Exception:
        pass

    # Fallback for scipy builds where jac=True tuple interface is inconvenient.
    state = _make_lbfgs_state(n, fix, p, min(deadline, time.monotonic() + max(5.0, 0.2 * max(1.0, deadline - time.monotonic()))))
    try:
        res = minimize(_smooth_obj_f, z0, method="L-BFGS-B", jac=_smooth_obj_g, args=(state,), options=options)
        if hasattr(res, "x"):
            x = np.asarray(res.x, dtype=np.float64)
            if x.size == n - 1 and np.all(np.isfinite(x)):
                return x.copy()
    except Exception:
        pass

    return z0


def _run_schedule(z, n, fix, schedule, deadline, a_start=None, score_start=None):
    n = int(n)
    fix = int(fix)
    if n <= 0:
        return a_start, score_start, z if z is not None else np.empty(0), z if z is not None else np.empty(0), fix

    free = _free_indices(n, fix)
    z = np.asarray(z, dtype=np.float64).reshape(-1).copy()
    if z.size != n - 1:
        if a_start is None:
            a_start = np.ones(n, dtype=np.float64) / float(n)
        z, a_start, fix = _init_z_free_from_a(a_start, fix)
        free = _free_indices(n, fix)

    if a_start is None:
        a_start = _softmax_with_fix(z, n, fix, free)
    a_start = _norm_sum1(a_start)

    if score_start is None:
        score_start = _score_fft(a_start)
    score_start = float(score_start)

    best_a = a_start.copy()
    best_score = score_start
    best_z = z.copy()
    z_cur = z.copy()

    total_fr = _schedule_total(schedule)
    if total_fr <= 0.0:
        return best_a, best_score, best_z, z_cur, fix

    for idx, item in enumerate(schedule):
        if time.monotonic() >= deadline:
            break
        p, fr = item
        fr = float(fr)
        if fr <= 0.0:
            continue

        rem_fr = 0.0
        for _, q in schedule[idx:]:
            rem_fr += float(q)
        if rem_fr <= 0.0:
            break

        rem_time = float(deadline - time.monotonic())
        if rem_time <= 1.0:
            break

        alloc = rem_time * (fr / rem_fr)
        run_deadline = time.monotonic() + max(0.5, alloc)
        run_deadline = min(run_deadline, float(deadline))

        z_out = _run_lbfgsb(z_cur, n, fix, p, run_deadline)
        if z_out is not None and z_out.size == n - 1 and np.all(np.isfinite(z_out)):
            z_cur = z_out.copy()

        a = _softmax_with_fix(z_cur, n, fix, free)
        a = _norm_sum1(a)
        sc = _score_fft(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = float(sc)
            best_a = a.copy()
            best_z = z_cur.copy()

    return best_a, best_score, best_z, z_cur, fix


def _interp_from_a(a, new_n):
    a = _norm_sum1(a)
    old_n = int(a.size)
    new_n = int(new_n)
    if old_n <= 0:
        return np.ones(new_n, dtype=np.float64) / float(max(1, new_n))
    if new_n <= 0:
        return np.empty(0, dtype=np.float64)
    if old_n == new_n:
        return a
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    return _norm_sum1(y)


def _shift_zero(a, s):
    a = _norm_sum1(a)
    n = int(a.size)
    s = int(s)
    if n <= 0:
        return a
    if s == 0:
        return a
    out = np.zeros(n, dtype=np.float64)
    if s > 0:
        m = min(s, n)
        out[m:n] = a[:n - m]
    else:
        m = min(-s, n)
        out[:n - m] = a[m:]
    return _norm_sum1(out)


def _make_default_a(n):
    n = max(1, int(n))
    x = np.linspace(0.0, 1.0, n)
    a = 0.25 + 0.5 * np.sin(np.pi * x) ** 2 + 0.25 * (x ** 3)
    return _norm_sum1(a)


def _local_perturb_z(a, fix, sigma, rng):
    a = _norm_sum1(a)
    n = int(a.size)
    if n <= 0:
        return np.empty(0, dtype=np.float64), 0
    noise = rng.normal(0.0, float(sigma), n)
    a2 = np.maximum(a, 1e-300) * np.exp(np.clip(noise, -12.0, 12.0))
    a2 = _norm_sum1(a2)
    z, a2, fix2 = _init_z_free_from_a(a2, fix)
    return z, fix2


def _sensitivity_top(a, idx, w):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    sens = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, idx):
        wi = float(wi)
        if wi <= 0.0:
            continue
        k = int(k)
        js = np.arange(max(0, k - n + 1), min(n - 1, k) + 1, dtype=np.int64)
        if js.size:
            sens[js] += 2.0 * wi * a[k - js]
    return sens


def _coordinate_polish(a, deadline, top_k=10, beta=30.0):
    a = _norm_sum1(a).copy()
    n = int(a.size)
    if n <= 0:
        return a, float("inf")
    if time.monotonic() >= deadline:
        return a, _score_fft(a)

    best_a = a.copy()
    best_score = _score_fft(a)
    fsize = _next_pow2(2 * n - 1)
    n_lags = 2 * n - 1
    top_k = int(max(2, min(top_k, n_lags)))
    rng = np.random.default_rng(987654321)
    stall = 0
    moves = (0.002, 0.005, 0.01, 0.02, 0.05, 0.10, 0.20)

    while time.monotonic() < deadline:
        conv = _fast_convolution(a, fsize)
        conv = np.maximum(conv, 0.0)
        if conv.size == 0:
            break

        order = np.argsort(conv)
        top = order[-top_k:]
        vals = conv[top]
        vmax = float(np.max(vals)) if vals.size else 0.0
        shifted = beta * (vals - vmax)
        ew = np.exp(shifted)
        sw = float(ew.sum())
        if sw <= 0.0 or not np.isfinite(sw):
            w = np.ones(top_k, dtype=np.float64) / float(top_k)
        else:
            w = ew / sw

        sens = _sensitivity_top(a, top, w)
        mass_thresh = max(1e-8, float(a.mean()) * 1e-3)
        candidates = np.where(a > mass_thresh)[0]
        if candidates.size < 2:
            candidates = np.arange(n)

        src_order = np.argsort(sens[candidates])[::-1][:12]
        srcs = candidates[src_order]
        dst_order = np.argsort(sens)
        dsts = dst_order[:12]

        improved = False
        for src in srcs:
            if time.monotonic() >= deadline:
                break
            src = int(src)
            for dst in dsts:
                dst = int(dst)
                if dst == src:
                    continue
                base = float(a[src])
                if base <= 1e-14:
                    continue
                for frac in moves:
                    amt = base * float(frac)
                    if amt < 1e-14:
                        continue
                    trial = a.copy()
                    trial[src] = max(0.0, trial[src] - amt)
                    trial[dst] += amt
                    s = float(trial.sum())
                    if s <= 0.01:
                        continue
                    trial = trial / s
                    sc = _score_fft(trial)
                    if sc < best_score - 1e-12:
                        a = trial
                        best_a = trial.copy()
                        best_score = float(sc)
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            improved_random = False
            for _ in range(24):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                if a[i] < a[j]:
                    continue
                frac = float(rng.uniform(0.001, 0.02))
                amt = float(a[i]) * frac
                if amt < 1e-14:
                    continue
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                if trial.min() < 0.0:
                    continue
                s = float(trial.sum())
                if s <= 0.01:
                    continue
                trial = trial / s
                sc = _score_fft(trial)
                if sc < best_score - 1e-12:
                    a = trial
                    best_a = trial.copy()
                    best_score = float(sc)
                    improved_random = True
                    stall = 0
                    break
            if not improved_random:
                stall += 1
                if stall > 90:
                    break

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1800.0
    if not np.isfinite(total_budget) or total_budget <= 0.0:
        total_budget = 1800.0

    reserve = 30.0
    global_deadline = t0 + max(60.0, total_budget - reserve)

    parent = None
    try:
        h = height_sequence_1
        arr = np.asarray(h, dtype=np.float64).reshape(-1)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size > 0 and float(arr.sum()) > 0.01:
            parent = arr
    except Exception:
        parent = None

    if parent is not None:
        n0 = int(parent.size)
        if n0 < 200 or n0 > 5000:
            n0 = 2400
            a0 = _interp_from_a(parent, n0)
        else:
            a0 = _norm_sum1(parent)
    else:
        n0 = 2400
        a0 = _make_default_a(n0)

    best_a = _norm_sum1(a0).copy()
    best_score = _score_fft(best_a)

    # A few cheap deterministic variants to avoid a bad parent boundary effect.
    for s in (n0 // 100, n0 // 50, -n0 // 50):
        if s == 0:
            continue
        cand = _shift_zero(best_a, int(s))
        sc = _score_fft(cand)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_a = cand.copy()
            best_score = float(sc)

    best_z, best_a, fix = _init_z_free_from_a(best_a)
    cur_n = int(best_a.size)

    main_schedule = [
        (8, 0.04),
        (16, 0.06),
        (32, 0.10),
        (64, 0.14),
        (128, 0.18),
        (256, 0.22),
        (512, 0.16),
        (1024, 0.10),
    ]
    main_deadline = t0 + 0.72 * (global_deadline - t0)
    best_a, best_score, best_z, z_cur, fix = _run_schedule(
        best_z, cur_n, fix, main_schedule, main_deadline, a_start=best_a, score_start=best_score
    )
    cur_n = int(best_a.size)

    # Length exploration: interpolate the current champion and polish.
    length_candidates = []
    for d in (400, 800, 200, 1200, -200, -400):
        nt = cur_n + int(d)
        if 800 <= nt <= 5000 and nt not in length_candidates:
            length_candidates.append(nt)

    length_schedule = [
        (16, 0.15),
        (64, 0.30),
        (256, 0.35),
        (1024, 0.20),
    ]

    for nt in length_candidates[:4]:
        if time.monotonic() >= global_deadline - 150.0:
            break
        a_int = _interp_from_a(best_a, int(nt))
        sc_int = _score_fft(a_int)
        z_int, a_int, fix_int = _init_z_free_from_a(a_int)
        run_deadline = min(global_deadline - 120.0, time.monotonic() + 90.0)
        a_out, sc_out, z_b_out, _, fix_out = _run_schedule(
            z_int, int(nt), fix_int, length_schedule, run_deadline, a_start=a_int, score_start=sc_int
        )
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_a = a_out.copy()
            best_score = float(sc_out)
            best_z = z_b_out.copy()
            fix = fix_out
            cur_n = int(best_a.size)
            break

    # Local annealing restarts from the current champion.
    restart_deadline = global_deadline - 70.0
    rng = np.random.default_rng(1234567)
    sigma = 0.08
    trial_time = 45.0
    last_improve = time.monotonic()
    trial_count = 0

    local_schedule = [
        (8, 0.15),
        (32, 0.30),
        (128, 0.35),
        (512, 0.20),
    ]

    while time.monotonic() < restart_deadline:
        if time.monotonic() - last_improve > 120.0:
            sigma = max(0.015, sigma * 0.5)
            trial_time = max(20.0, trial_time * 0.8)

        run_deadline = min(restart_deadline, time.monotonic() + trial_time)
        z_try, fix_try = _local_perturb_z(best_a, fix, sigma, rng)
        if int(best_a.size) != cur_n:
            z_try, best_a, fix_try = _init_z_free_from_a(best_a)
            cur_n = int(best_a.size)

        a_out, sc_out, z_b_out, _, fix_out = _run_schedule(
            z_try, cur_n, fix_try, local_schedule, run_deadline, a_start=best_a, score_start=best_score
        )
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_a = a_out.copy()
            best_score = float(sc_out)
            best_z = z_b_out.copy()
            fix = fix_out
            cur_n = int(best_a.size)
            last_improve = time.monotonic()
            sigma = max(0.008, sigma * 0.7)
            trial_time = min(85.0, trial_time * 1.2)
        else:
            if trial_count % 3 == 2:
                sigma = min(0.25, sigma * 1.6)
            else:
                sigma = min(0.20, sigma * 1.2)
            trial_time = max(20.0, trial_time * 0.9)
        trial_count += 1

    # High-p final polish on the best known sequence.
    if time.monotonic() < global_deadline - 50.0:
        z_full, best_a, fix_full = _init_z_free_from_a(best_a)
        final_deadline = global_deadline - 25.0
        final_schedule = [
            (128, 0.25),
            (512, 0.45),
            (1024, 0.30),
        ]
        a_out, sc_out, z_b_out, _, fix_out = _run_schedule(
            z_full, int(best_a.size), fix_full, final_schedule, final_deadline, a_start=best_a, score_start=best_score
        )
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_a = a_out.copy()
            best_score = float(sc_out)
            best_z = z_b_out.copy()
            fix = fix_out

    # Peak-sensitive coordinate polish.
    if time.monotonic() < global_deadline - 8.0:
        coord_deadline = global_deadline - 3.0
        a_pol, sc_pol = _coordinate_polish(best_a, coord_deadline)
        if np.isfinite(sc_pol) and sc_pol < best_score - 1e-12:
            best_a = a_pol.copy()
            best_score = float(sc_pol)

    # Very short final L-BFGS polish if coordinate polish improved things.
    if time.monotonic() < global_deadline - 8.0:
        z_final, best_a, fix_final = _init_z_free_from_a(best_a)
        final2_deadline = global_deadline - 2.0
        final2_schedule = [
            (256, 0.45),
            (1024, 0.55),
        ]
        a_out, sc_out, z_b_out, _, fix_out = _run_schedule(
            z_final, int(best_a.size), fix_final, final2_schedule, final2_deadline, a_start=best_a, score_start=best_score
        )
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_a = a_out.copy()
            best_score = float(sc_out)
            best_z = z_b_out.copy()
            fix = fix_out

    best_a = _norm_sum1(best_a)

    # Validate against the actual evaluator once before returning.
    final_check = _actual_score(best_a)
    if not np.isfinite(final_check):
        fallback = np.ones(max(1, int(best_a.size)), dtype=np.float64) / float(max(1, int(best_a.size)))
        fb_score = _actual_score(fallback)
        if np.isfinite(fb_score):
            best_a = fallback
    else:
        best_score = float(final_check)

    if best_a.size == 0:
        best_a = np.ones(1, dtype=np.float64)

    return [float(x) for x in best_a]
```
