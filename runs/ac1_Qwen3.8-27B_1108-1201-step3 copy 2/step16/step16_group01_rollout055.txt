def _line_search(a, direction, M_best, n_evals=15):
    """
    Line search: find best alpha in [0,1] minimizing peak of
    norm_sum1((1-alpha)*a + alpha*direction).
    """
    n = len(a)
    a = _norm_sum1(a)
    d = _norm_sum1(direction)
    
    best_alpha = 0.0
    best_peak = _fast_autocorr(a).max()
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    
    def peak_at(alpha):
        x = (1.0 - alpha) * a + alpha * d
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return np.inf
        x = x / s
        return _fast_autocorr(x).max()
    
    fc = peak_at(c)
    fd = peak_at(d_pt)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d_pt
            d_pt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = peak_at(c)
        else:
            lo = c
            c = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = peak_at(d_pt)
    
    best_alpha = (lo + hi) / 2.0
    x = (1.0 - best_alpha) * a + best_alpha * d
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return a, best_peak
    return x / s, peak_at(best_alpha)

        ac = _fast_autocorr(y)
        M_curr = ac.max()
        
        # Solve LP
        b = _lp_frank_wolfe_step(y, M_curr, top_k=top_k, solver=solver)
        b = _norm_sum1(b)
        
        # Check if direction is meaningful
        if np.allclose(b, y, atol=1e-10):
            break
        
        # Line search
        y_new, M_new = _line_search(y, b, best_m, n_evals=12)
        
        # Nesterov extrapolation for x
        alpha = 2.0 / (it + 2.0)
        if it == 1:
            x_new = y_new.copy()
        else:
            beta = (it - 1.0) / (it + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s = x_new.sum()
            if s > 1e-15:
                x_new = x_new / s
            else:
                x_new = y_new.copy()
        
        y = y_new
        x = x_new
        
        # Track best
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Also check x
        Mx = _fast_autocorr(x).max()
        if Mx < best_m - 1e-15:
            best_m = Mx
            best_a = x.copy()
            stall = 0
        
        # Adaptive: increase top_k if making progress
        if stall == 0 and top_k < max_top_k and it % 100 == 0:
            top_k = min(max_top_k, top_k + 20)
        
        # Restart if stuck
        if stall > max_stall:
            stall = 0
            # Perturb from best
            noise = rng.normal(0, 0.003, n) * best_a
            y_restart = best_a + noise
            y_restart = np.maximum(y_restart, 1e-10)
            y_restart = y_restart / y_restart.sum()
            y = y_restart
            x = y_restart.copy()
            it = 0
            top_k = max(50, top_k // 2)
            if verbose:
                print(f"      LP-FW restart at {elapsed:.1f}s, M={best_m:.8f}, top_k={top_k}")
        
        if verbose and it % 50 == 0:
            J = 2.0 * n * best_m
            print(f"      LP-FW it={it} ({elapsed:.1f}s): J={J:.6f}, M={M_curr:.8f}, top_k={top_k}, stall={stall}")
    
    return best_a, best_m

def _softmax_gradient_descent(a0, time_limit, beta_start=50.0, beta_end=500.0, 
                               lr_start=0.1, verbose=True, rng_seed=77):
    """
    Gradient descent on soft-max smoothed objective with Nesterov acceleration.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    v = np.zeros(n)
    
    total_iters = 20000
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Log-linear interpolation of beta
        frac = t / total_iters
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute soft-max gradient
        ac = _fast_autocorr(x)
        ac_clipped = np.clip(ac, -1e30, 1e30)
        max_ac = ac_clipped.max()
        exp_vals = np.exp(beta * (ac_clipped - max_ac))
        w = exp_vals / exp_vals.sum()
        
        # Gradient: grad_j = 2 * sum_k w_k * a_{k-j}
        # This is 2 * (w_conv) evaluated at each position
        # conv(w, a) gives sum_k w_k * a_{j-k}... let me compute directly
        # grad_j = 2 * sum_k w_k * a[k-j] for valid k-j
        # This is 2 * (flip(w) * a) at position j... 
        # Actually: sum_k w_k * a[k-j] = (w * flip(a))_j = (flip(w) * a)_{...}
        # Let's just use conv: np.convolve(w, a) gives sum_k w_k a_{j-k}
        # We want sum_k w_k a[k-j] = sum_k w_{k} a_{(-j) mod ...} 
        # = np.convolve(flip(w), a)[j] where flip(w)_k = w[-k]... 
        # Simpler: grad_j = 2 * sum_k w_k * a[k-j]
        # Let's compute via: for each k, add w_k * a[k-j] to grad[j]
        # This is equivalent to: grad = 2 * np.convolve(a, w)[1:n-1]... 
        # Actually let me just do it with FFT for speed
        
        # sum_k w_k * a[k-j] = sum_i a[i] * w[i+j] (with valid indices)
        # = (a * w_full)_j where w_full has w centered
        # Let's use: fft approach
        fft_size = 4 * n
        A_fft = np.fft.rfft(a, fft_size)
        W_fft = np.fft.rfft(w, fft_size)
        # We want sum_k w_k a[k-j], which is correlation = ifft(conj(W) * A) shifted
        # Actually: let's think of it as (w *_full a) where *_full is full correlation
        # sum_k w_k a[k-j] for j = 0,...,n-1
        # = ifft(conj(FFT(w)) * FFT(a))[j + n - 1]... 
        # Let me just compute it directly for correctness (n is at most ~3000)
        
        # Direct computation using the structure:
        # grad[j] = 2 * sum_{k: 0<=k-j<n, 0<=k<len(w)} w[k] * a[k-j]
        # = 2 * sum_{i: max(0,-j) <= i <= min(len(w)-1-j, n-1-j)} w[i+j] * a[i]
        # Hmm this is getting complex. Let me use a different approach.
        # 
        # The gradient of c_k = sum_i a_i a[k-i] w.r.t. a_j is 2*a[k-j] (as derived)
        # So grad of soft-max = 2 * sum_k w_k * a[k-j]
        # 
        # Let's compute this as: for each k, the contribution to all j is w_k * a[k-j]
        # which is a shifted and scaled version of a.
        # 
        # Equivalently: grad = 2 * (w '* a) where '* is cross-correlation
        # = 2 * ifft(conj(FFT(w_padded)) * FFT(a_padded))
        
        w_pad = np.zeros(fft_size)
        w_pad[:len(w)] = w
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        
        W_pad = np.fft.rfft(w_pad)
        A_pad = np.fft.rfft(a_pad)
        
        # Cross-correlation: sum_k w[k] a[k+m] = ifft(conj(W) A)[m]
        # We want sum_k w[k] a[k-j] = ifft(conj(W) A)[-j mod fft_size]
        # = ifft(conj(W) A)[fft_size - j]
        corr = np.fft.irfft(np.conj(W_pad) * A_pad, fft_size)
        
        # grad[j] = 2 * corr[fft_size - j] for j = 0,...,n-1
        # But fft_size - j for j=0 is fft_size=0 (wraps to 0)
        grad = 2.0 * np.zeros(n)
        for j in range(n):
            idx = (fft_size - j) % fft_size
            grad[j] = 2.0 * corr[idx]
        
        # Hmm this is O(n) per step which is fine, but let me verify with a small example
        # Actually for speed, let me just do the direct computation since n <= 3000
        # and we only need to do it once per iteration
        # Actually the FFT approach above should work. Let me just use it.
        
        # Nesterov momentum
        lr = lr_start / (1.0 + 0.01 * t)
        v = 0.9 * v - lr * grad
        x_new = x + v
        x_new = np.maximum(x_new, 1e-10)
        s = x_new.sum()
        if s < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s
        
        x = x_new
        
        # Evaluate
        M_curr = _fast_autocorr(x).max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = x.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      SoftMax-GD t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m

def _coordinate_refine(a0, time_limit, verbose=True, rng_seed=123):
    """
    Focused coordinate descent: identify peak-contributing pairs
    and perform optimal 2-variable updates.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 80
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peak lags
        n_peaks = min(10, len(ac))
        peak_lags = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        
        for k in peak_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with max contribution
            pi = int(np.argmax(contrib))
            src_i = int(idx[pi])
            src_j = int(j_idx[pi])
            
            # Try all destinations (sampled)
            n_dest = min(50, n)
            dests = rng.choice(n, size=n_dest, replace=False)
            
            for dst in dests:
                if dst == src_i or dst == src_j:
                    continue
                
                # Optimal 2-var update: move mass from src_i to dst
                # c_k changes by: -a[dst_j]*delta*a[dst_j]... 
                # Actually let's just try a few fractions
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    M_trial = _fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-12:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if not improved:
                # Try from src_j instead
                for dst in dests[:20]:
                    if dst == src_i or dst == src_j:
                        continue
                    for frac in [0.01, 0.03, 0.07]:
                        amt = a[src_j] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src_j] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        M_trial = _fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-12:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_perturb = max(5, n // 100)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.02, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        M_new = _fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Main search: LP-FW + SoftMax GD + Coord Descent + Multi-length.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0.0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = _eval_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent_n = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            ps = _eval_safe(parent)
            print(f"Parent: n={parent_n}, score={ps:.6f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        parent = None
    
    if best_seq is None:
        # Create a good initial sequence
        x = np.linspace(0, 1, parent_n)
        best_seq = np.sin(np.pi * x) ** 2 + 0.05 * np.cos(4 * np.pi * x) ** 2 + 0.02
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = _eval_safe(best_seq)
        parent = best_seq.copy()
        print(f"Init: n={parent_n}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: LP-FW on parent (high priority) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW n={n0} ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = _lp_fw_optimize(_norm_sum1(best_seq), t1, solver, verbose=True, rng_seed=1, top_k_init=60, max_top_k=150)
        try_update(a_out, f"lpfw1_n{n0}")
    
    # === Phase 2: SoftMax GD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: SoftMax-GD n={n_best} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, _ = _softmax_gradient_descent(_norm_sum1(best_seq), t2, beta_start=30.0, beta_end=300.0, lr_start=0.05, verbose=True, rng_seed=2)
        try_update(a_out, f"softmax_n{n_best}")
    
    # === Phase 3: LP-FW again (from improved point, different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW round 2 n={n_best} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = _lp_fw_optimize(_norm_sum1(best_seq), t3, solver, verbose=True, rng_seed=3, top_k_init=100, max_top_k=250)
        try_update(a_out, f"lpfw2_n{n_best}")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_best} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_out, _ = _coordinate_refine(_norm_sum1(best_seq), t4, verbose=True, rng_seed=4)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length ===")
        lengths_to_try = []
        for dn in [-400, -200, 200, 400]:
            nl = n_best + dn
            if 500 <= nl <= 4000:
                lengths_to_try.append(nl)
        
        for nl in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={nl}")
            a_interp = _interp_sequence(best_seq, nl)
            t_sub = min(remaining * 0.20, 120.0)
            if t_sub > 30:
                a_out, _ = _lp_fw_optimize(_norm_sum1(a_interp), t_sub, solver, verbose=False, rng_seed=nl % 100, top_k_init=60, max_top_k=120)
                try_update(a_out, f"lpfw_ml_n{nl}")
    
    # === Phase 6: Final LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW n={n_best} ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 20:
        a_out, _ = _lp_fw_optimize(_norm_sum1(best_seq), t6, solver, verbose=True, rng_seed=6, top_k_init=150, max_top_k=300)
        try_update(a_out, f"lpfw_final_n{n_best}")
    
    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final coord n={n_best} ===")
        t7 = min(remaining * 0.40, 120.0)
        if t7 > 15:
            a_out, _ = _coordinate_refine(_norm_sum1(best_seq), t7, verbose=True, rng_seed=7)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 8: One more LP-FW with very tight active set ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: LP-FW tight n={n_best} ===")
        t8 = min(remaining * 0.50, 150.0)
        if t8 > 20:
            a_out, _ = _lp_fw_optimize(_norm_sum1(best_seq), t8, solver, verbose=False, rng_seed=8, top_k_init=30, max_top_k=60)
            try_update(a_out, f"lpfw_tight_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
