def lp_min_peak(y, solver):
    """Find b that minimizes max(conv(y, b)) subject to sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(y)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build T @ b
    # T[k, j] = y[j] if j <= k <= j+n-1
    # So (T @ b)[k] = sum_j y[j] * b[k-j] for valid j
    # This is conv(y, b)[k]
    
    # We need: conv(y, b)[k] <= M for all k
    # And: sum(b) = 1
    # And: b >= 0
    
    # Build the constraint matrix efficiently
    # For each k, the constraint is: sum_j y[j] * b[k-j] <= M
    # where the sum is over j such that 0 <= j < n and 0 <= k-j < n
    
    # This is a circulant-like matrix (banded)
    # Let's build it as a sparse matrix
    import scipy.sparse as sp
    
    T_rows = []
    T_cols = []
    T_vals = []
    
    for j in range(n):
        if y[j] < 1e-15:
            continue
        for k in range(j, j + n):
            T_rows.append(k)
            T_cols.append(j)
            T_vals.append(y[j])
    
    T = sp.csr_matrix((T_vals, (T_rows, T_cols)), shape=(2*n-1, n))
    
    constraints = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status in ('optimal', 'optimal_inaccurate'):
        b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0)
        b_val = b_val / b_val.sum()
        return b_val, M.value
    return None, np.inf

**Approach: Spectral methods / Zakharov-type iteration**

The problem is equivalent to: minimize ||a*a||_∞ subject to a >= 0, sum(a) = 1.

1. In the frequency domain, minimize the peak of |F(a)|^2.
2. The "steepest descent" direction in the frequency domain is to reduce the peak frequency components.
3. Transform back to time domain and project onto a >= 0.

def build_conv_matrix_csc(n):
    """Build CSC matrix structure for T @ b = conv(y, b)."""
    # Column j has entries in rows j to j+n-1, all with value y[j]
    col_ptr = np.arange(0, (n + 1) * n, n, dtype=np.int32)
    row_ind = np.concatenate([np.arange(j, j + n) for j in range(n)])
    data = np.zeros(n * n, dtype=np.float64)
    T = sp.csc_matrix((data, row_ind, col_ptr), shape=(2 * n - 1, n))

    T.data = np.repeat(y, n)

    M = cp.Variable()

        prob.solve(solver=solver, verbose=False)

        return None, np.inf

    s = b_val.sum()
    if s < 1e-15:
        return None, np.inf

    return b_val, float(M.value)
