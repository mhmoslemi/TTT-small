Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive Nesterov acceleration and progressive active-set
2. 1D coordinate descent with exact line search (golden section)
3. Entropy-based mirror descent for diversity
4. Multi-scale restart with structured perturbations
Key insight: score = max_autocorr when normalized to sum=sqrt(2n)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None
    
    # Build constraint matrix T: T[idx, i] = a_fixed[k - i] if 0 <= k-i < n
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with adaptive active-set management."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    n_iters = 0
    conv_size = 2 * n - 1
    
    # Adaptive parameters
    base_stall_threshold = 80
    anneal_period = 300  # Every this many iters, relax the active set
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set management
        # "Anneal" periodically: temporarily use a looser active set
        is_anneal = (t % anneal_period < 20)  # First 20 iters of each period
        
        if active_set is None or t % 30 == 0 or is_anneal:
            if is_anneal:
                # Loose active set during annealing
                tol = 0.15
            else:
                # Progressive tightening
                progress = min(1.0, t / 2000.0)
                tol = 0.06 * (1.0 - progress) + 0.003 * progress
            
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            # Always include top peaks
            n_top = max(10, n // 50)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap for efficiency
            max_active = int(2.0 * n)
            if len(active_set) > max_active:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_active:]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            # Fallback: try with just the top peaks
            top_k = np.argsort(conv_y)[-max(20, n//20):]
            b, duals = lp_best_response(y, M_curr, top_k, solver)
            if b is None:
                # Last resort: use just the global max
                top_k = np.argsort(conv_y)[-5:]
                b, duals = lp_best_response(y, M_curr, top_k, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive Nesterov update
        alpha = 2.0 / (t + 2.0)
        
        # During annealing, use a larger step
        if is_anneal:
            alpha = min(alpha * 2.0, 0.3)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            # Adaptive momentum
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            # During annealing, reduce momentum to be more responsive
            if is_anneal:
                momentum = momentum * 0.7
            
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average (every 100 iters)
        if t % 100 == 0:
            a_damped = 0.6 * best_a + 0.4 * a
            a_damped = np.maximum(a_damped, 0.0)
            s_d = a_damped.sum()
            if s_d > 1e-15:
                a_damped = a_damped * norm / s_d
                score_d = fast_autocorr(a_damped).max()
                if score_d < best_score:
                    best_score = score_d
                    best_a = a_damped.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > base_stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            # Add a small random perturbation on restart
            pert = 1.0 + 0.01 * rng.standard_normal(n)
            pert = np.maximum(pert, 0.5)
            a = a * pert
            a = np.maximum(a, 0.0)
            a = a * norm / a.sum()
            y = a.copy()
            y_prev = a.copy()

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, M={M_curr:.10f}")

    return best_a, best_score

2. Direct subgradient with Polyak step size and Nesterov momentum

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Bisection on threshold tau: x_i = max(0, v_i - tau), sum = target_sum
    lo = 0.0
    hi = v.max() + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        x = np.maximum(v - tau, 0.0)
        s = x.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    x = np.maximum(v - tau, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) * target_sum / n
    return x * target_sum / s


def solve_lp_best_response(a_fixed, M, active_set):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0 using scipy linprog (HiGHS)."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None

    # Build constraint matrix T: T[idx, i] = a_fixed[k - i] if 0 <= k-i < n
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    # LP: min -sum(b) s.t. T@b <= M, -b <= 0
    c = -np.ones(n)
    A_ub = np.vstack([T, -np.eye(n)])
    b_ub = np.concatenate([np.full(size, M), np.zeros(n)])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.status != 0:
        return None, None
    b_val = np.maximum(res.x, 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    # Dual variables (from inequality constraints)
    duals = None
    try:
        duals = np.asarray(res.ineqlin.marginals[:size], dtype=np.float64)
        duals = np.maximum(duals, 0.0)
    except Exception:
        pass
    return b_val, duals

            top_k = np.argsort(conv_y)[-10:]

        n_iters += 1

        y = y_new
        a = a_new

            a_damped = np.maximum(a_damped, 0.0)
            s_d = a_damped.sum()
            if s_d > 1e-15:

            y = a.copy()

        if verbose and n_iters % 300 == 0:

def subgradient_method(a0, time_limit, verbose=False, seed=0):
    """
    Direct subgradient method with Polyak step size and Nesterov momentum.
    Minimizes max_k conv(a,a)[k] on the simplex {a>=0, sum(a)=sqrt(2n)}.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    conv_size = 2 * n - 1

    y = a.copy()
    y_prev = a.copy()
    patience = 0
    max_patience = 800
    step_base = 2.0 / (n ** 0.3)

    # Lower bound estimate (for Polyak step)
    # Average of conv is (sum)^2/(2n-1) = 2n/(2n-1) ~ 1
    lb = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()

        # Select active peaks with weights
        n_peaks = max(5, min(20, n // 60))
        threshold = M_curr * 0.85
        active = np.where(conv_y >= threshold)[0]
        if len(active) < n_peaks:
            active = np.argsort(conv_y)[-n_peaks:]
        elif len(active) > 3 * n_peaks:
            # Keep only the most important
            order = np.argsort(conv_y[active])[::-1]
            active = active[order[:n_peaks]]

        # Exponential weights (sharper = more focused)
        weights = np.exp(-10.0 * (M_curr - conv_y[active.astype(int)]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Compute weighted subgradient
        # subgrad[i] = sum_k w_k * 2 * a[k-i] (if 0 <= k-i < n)
        grad = np.zeros(n)
        for w, k in zip(weights, active.astype(int)):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * 2.0 * a[k - np.arange(lo, hi)]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Polyak step size
        gap = M_curr - lb
        if gap < 1e-6:
            eta = step_base / n
        else:
            eta = gap / (gn * gn) * step_base
        eta = min(eta, 2.0 / n ** 0.2)
        eta = max(eta, 1e-8)

        # Nesterov: step from y (extrapolated point)
        alpha = 2.0 / (t + 2.0)
        if t == 1:
            a_new = y - eta * grad
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y - eta * grad + momentum * (y - y_prev)

        # Project onto simplex
        a_new = project_simplex(a_new, norm)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = project_simplex(y_new, norm)

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, eta={eta:.6f}")

    return best_a, best_score

                #           + di*dj*I(k==i+j)
                # where a[k-i] means a[k-i] if 0<=k-i<n else 0, similarly for a[k-j].
                # And when k-i == j (k = i+j), the term a[k-i]*di should be a'[j]*di = (aj+dj)*di,
                # but we've already added di*a[j] from the first term. So the "a[k-i]*di" term
                # should use a'[k-i] not a[k-i] when k-i is one of the changed positions.
                #
                # This is getting too complicated for an inline computation. Let me just
                # do a full recompute using FFT for each candidate. It's O(n) per candidate
                # and we have ~10 candidates per pair. With n=1600, that's ~16000 ops per pair.
                # With 2000 pairs per pass, that's 32 million ops per pass. At ~1 GHz effective,
                # that's 32ms per pass. We can do ~30 passes per second. OK.

                pass  # Skip this candidate for now, use the simple approach below

            # Simple approach: just evaluate a few candidates with full FFT
            for frac in candidates[:6]:  # Limit to 6 candidates
                if abs(frac - cur_frac) < 1e-10:
                    continue
                a_pass[i] = s_pair * frac
                a_pass[j] = s_pair * (1.0 - frac)
                test_conv = fast_autocorr(a_pass)
                test_max = test_conv.max()
                if test_max < best_local:
                    best_local = test_max
                    best_frac = frac

            # Restore
            a_pass[i] = ai
            a_pass[j] = aj

            if best_frac != cur_frac:
                a_pass[i] = s_pair * best_frac
                a_pass[j] = s_pair * (1.0 - best_frac)
                pass_improved = True
                n_updates += 1

        if pass_improved:
            # Check if the pass improved the actual score
            final_conv = fast_autocorr(a_pass)
            final_max = final_conv.max()
            if final_max < best_score - 1e-12:
                best_score = final_max
                best_a = a_pass.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and n_updates % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD updates={n_updates} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def single_cd_with_line_search(a0, time_limit, verbose=False, seed=0):
    """
    Single-coordinate descent: for each coordinate j, find optimal x that
    minimizes max(conv) given sum constraint (via redistribution).
    Since score is scale-invariant, we just minimize max(conv) directly.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    conv_size = 2 * n - 1
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break

        a_pass = a.copy()
        base_conv = fast_autocorr(a_pass)
        pass_improved = False
        n_upds = 0

        order = rng.permutation(n)
        typical = norm / n

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            aj = a_pass[j]
            # Try: reduce, keep, increase
            candidates = [0.0, aj * 0.1, aj * 0.5, aj, aj * 1.5, aj * 2.0,
                          typical * 0.5, typical, typical * 2.0]
            candidates = sorted(set([max(0.0, min(c, 5.0 * typical)) for c in candidates]))

            best_x = aj
            best_local = base_conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                # Compute new conv efficiently
                delta = x - aj
                # new_conv[k] = base_conv[k] + 2*delta*a[k-j] + delta^2*I(k=2j)
                # But only for k where the change matters
                new_conv = base_conv.copy()
                lo = max(0, j)
                hi = min(conv_size, j + n)
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a_pass[lo - j:hi - j]
                if 2 * j < conv_size:
                    new_conv[2 * j] += delta * delta
                local_max = new_conv.max()
                if local_max < best_local:
                    best_local = local_max
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a_pass[j] = best_x
                pass_improved = True
                n_upds += 1

        if pass_improved:
            s_pass = a_pass.sum()
            if s_pass > 1e-15:
                a_norm = a_pass * norm / s_pass
                test_score = fast_autocorr(a_norm).max()
                if test_score < best_score - 1e-12:
                    best_score = test_score
                    best_a = a_norm.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and n_upds % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD updates={n_upds} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score
