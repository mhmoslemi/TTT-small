def compute_recursive(n, c):
    a = np.zeros(n)
    a[0] = np.sqrt(c)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k-i]
        a[k] = (c - s) / (2 * a[0])
