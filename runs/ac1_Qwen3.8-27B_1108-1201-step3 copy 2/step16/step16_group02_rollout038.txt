Now, let me think about the overall algorithm more carefully.

The key insight is: we want min max_k (a*a)[k] subject to sum a = 1, a >= 0.

This is a non-convex problem (the constraints are quadratic). But we can use:

2. Frank-Wolfe with LP subproblems for the active constraints
3. Coordinate descent for local polish

Let me also think about a different approach: **parametric methods**.

**Approach: Truncated power series / polynomial approximation**

We want A(x)^2 to be approximately flat. If A(x) = sum a[i] x^i, then A(x)^2 = t * (1 + x + ... + x^{2n-2}) + R(x) where R is the "error" polynomial.

If we could make R(x) = 0, we'd have A(x) = sqrt(t * (1-x^{2n-1})/(1-x)) = sqrt(t) * sqrt(1+x+...+x^{2n-2}).

The coefficients of sqrt(1+x+...+x^m) are known and related to the central binomial coefficients. For m = 2n-2:
sqrt(1+x+...+x^{2n-2}) has coefficients that are positive and sum to sqrt((2n-1) * (2n-2 + 1)/...) hmm, this is getting complicated.

Actually, let's think about it differently. If we want A(x)^2 = c * (1-x^{2n-1})/(1-x), then:
A(x) = sqrt(c) * ((1-x^{2n-1})/(1-x))^{1/2}

The polynomial (1-x^{2n-1})/(1-x) = 1 + x + x^2 + ... + x^{2n-2} has all coefficients equal to 1. Its square root has coefficients related to the "Ramanujan sums" or similar.

For large n, A(x) ≈ sqrt(c/n) * n / (1-x)^{1/2} * (1 + correction terms)
The coefficients of (1-x)^{-1/2} are binomial(2k, k)/4^k ≈ 1/sqrt(pi*k).

So a[i] ≈ c^{1/2} * n^{1/2} * 1/sqrt(pi*(n/2 - i)) approximately... this is getting complicated.

Let me just focus on the practical optimization approach.

**My plan:**

1. **Efficient Nesterov Mirror Descent with FFT-based gradient** (main workhorse, ~50% time)
   - Use adaptive softmax temperature
   - Use FFT to compute the gradient in O(n log n)

   - Track best solution

2. **Active-set Frank-Wolfe with LP** (~20% time)

4. **Final polish** (~10% time)

Key improvements over the previous code:
- **FFT-based gradient**: Much faster for large n, enabling more iterations

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov Mirror Descent with FFT-based soft-max gradient and adaptive temperature
2. Active-set LP Frank-Wolfe for focused refinement
3. Multi-length exploration
4. Coordinate descent polish
Key improvement: O(n log n) gradient via FFT enables many more iterations.

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient_fft(a, w):
    """
    Compute g[i] = sum_j a[j] * w[i+j] for i=0,...,n-1.
    w is defined on indices 0,...,2n-2 (length 2n-1).
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    w_rev = w[::-1]  # length 2n-1
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:conv_len]
    # g[k] = conv[2n-2-k] for k=0,...,n-1
    g = conv[2 * n - 2 - np.arange(n)]
    return g

def softmax_weights(c, temperature):
    """Compute softmax weights for the convolution coefficients."""
    c_clipped = np.maximum(c, 1e-30)
    log_c = np.log(c_clipped / temperature)
    log_c -= log_c.max()
    weights = np.exp(log_c)
    s = weights.sum()
    if s < 1e-30:
        # Fallback: uniform over top 50
        top_idx = np.argsort(c)[::-1][:min(50, len(c))]
        weights = np.zeros_like(c)
        weights[top_idx] = 1.0
        s = weights.sum()
    return weights / s

def nesterov_mirror_descent(a0, time_limit, initial_temp=0.01, final_temp=0.001,
                             c0=3.0, verbose=True, rng_seed=42):
    """Nesterov mirror descent with FFT-based gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1
    
    max_iter = 2000000
    
    for t in range(1, max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature (log-annealing)
        frac = min(1.0, t / 10000.0)
        temp = initial_temp * (1 - frac) ** 2 + final_temp * frac
        temp = max(temp, 1e-6)
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute convolution and gradient
        ac = fast_autocorr(y)
        weights = softmax_weights(ac, temp)
        g = compute_gradient_fft(y, weights)
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_x = np.log(y) - eta * g
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = x
        x = x_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Periodic restart
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.7)
        
        # Ergodic average
        running_avg += x
        avg_count += 1
        if avg_count % 200 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.6f}, stall={stall_count}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=200, verbose=True):
    """LP Frank-Wolfe with active constraint selection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
        
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select active constraints
        threshold = M_curr * 0.95
        active = np.where(ac >= threshold)[0]
        if len(active) > max_constraints:
            active = np.argsort(ac)[::-1][:max_constraints]
        
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active), n))
        for row_idx, k in enumerate(active):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, active={len(active)}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with random target selection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
        
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 10
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        improved = False
        
        # Try random pairs of coordinates
        n_tries = min(200, n)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    break
                a = a / s
                m = fast_autocorr(a).max()
                if m < best_m - 1e-16:
                    best_m = m
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    # Revert
                    a[j] -= amt
                    a[i] += amt
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m = fast_autocorr(a).max()
                if m < best_m:
                    best_m = m
                    best_a = a.copy()
                    stall = 0
                else:
                    a[j] -= amt
                    a[i] += amt
                    a = a / a.sum()
        
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

import cvxpy as cp
import time

_global_solver = None

def fast_conv(a):
    """Compute full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient_fft(a, w):
    """
    Compute g[i] = sum_j a[j] * w[i+j] for i=0,...,n-1 via FFT.
    w has length 2n-1 (indices 0..2n-2).
    """
    n = len(a)
    w_rev = w[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    return conv[idx]

def softmax_weights(c, temperature):
    """Compute softmax weights over convolution coefficients."""
    c_clipped = np.maximum(c, 1e-30)
    log_c = np.log(c_clipped) / temperature
    log_c -= log_c.max()
    weights = np.exp(log_c)
    s = weights.sum()
    if s < 1e-30:
        top_k = min(100, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros_like(c)
        weights[top_idx] = 1.0
        s = weights.sum()
    return weights / s

def nesterov_mirror_descent(a0, time_limit, initial_temp=1.0, final_temp=0.1,
                             c0=3.0, verbose=True, rng_seed=42):
    """
    Nesterov mirror descent with FFT-based soft-max gradient.
    a0 should be normalized to sum=1. Temperatures are relative to 1/n scale.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Scale temperature to be relative to 1/n
    initial_temp_abs = initial_temp / n
    final_temp_abs = final_temp / n

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 3000
    running_avg = a.copy()
    avg_count = 1
    total_iters = 0

    while True:
        if time.monotonic() - t0 > time_limit:
            break

        for _ in range(500):
            if time.monotonic() - t0 > time_limit:
                break

            total_iters += 1
            t = total_iters

            # Temperature schedule: fast initial drop then slow
            frac = min(1.0, t / 8000.0)
            temp = initial_temp_abs * (1.0 - frac) ** 1.5 + final_temp_abs * frac
            temp = max(temp, final_temp_abs)

            # Nesterov extrapolation
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            # Compute convolution and gradient via FFT
            ac = fast_conv(y)
            weights = softmax_weights(ac, temp)
            g = compute_gradient_fft(y, weights)

            # Mirror descent step
            eta = gamma / (np.sqrt(t) + 1.0)
            log_x = np.log(y) - eta * g
            max_log = log_x.max()
            x_new = np.exp(log_x - max_log)
            x_new = np.maximum(x_new, 1e-30)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y.copy()
            else:
                x_new = x_new / sx

            x_prev = x
            x = x_new

            M_new = fast_conv(x).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = x.copy()
                stall_count = 0
            else:
                stall_count += 1

            # Restart if stalled
            if stall_count > max_stall:
                stall_count = 0
                max_stall = max(500, max_stall // 2)
                noise_scale = 0.002 + 0.005 * (max_stall / 3000.0)
                pert = rng.normal(0, noise_scale, n)
                x_r = best_a * (1.0 + pert)
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r.copy()
                x_prev = x_r.copy()
                gamma = max(1.0, gamma * 0.8)
                max_stall = min(max_stall, 3000)

            # Ergodic average every 100 iters
            running_avg += x
            avg_count += 1
            if avg_count % 100 == 0:
                avg = running_avg / avg_count
                avg = np.maximum(avg, 0.0)
                s_avg = avg.sum()
                if s_avg > 1e-15:
                    avg = avg / s_avg
                    M_avg = fast_conv(avg).max()
                    if M_avg < best_m:
                        best_m = M_avg
                        best_a = avg.copy()

            if verbose and total_iters % 5000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    NMD t={total_iters} ({elapsed:.1f}s): J={J:.6f}, temp={temp*1.0/n*1.0:.2e}, stall={stall_count}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=250, verbose=True):
    """LP Frank-Wolfe with active constraint selection and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iters = 0

    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        iters = t

        ac = fast_conv(y)
        M_curr = ac.max()

        # Select active constraints: lags within 3% of max
        threshold = M_curr * 0.97
        active = np.where(ac >= threshold)[0]
        if len(active) > max_constraints:
            active = np.argsort(ac)[::-1][:max_constraints]

        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active), n))
        for row_idx, k in enumerate(active):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, active={len(active)}")

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent: transfer mass between pairs of coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 15
    sweep = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        improved = False

        n_tries = min(300, n)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10:
                continue
            for frac in [0.003, 0.008, 0.015, 0.03]:
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s2 = a.sum()
                if s2 < 1e-15:
                    break
                a = a / s2
                m = fast_conv(a).max()
                if m < best_m - 1e-17:
                    best_m = m
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[j] -= amt
                    a[i] += amt
                    a = a / a.sum()

        if improved:
            stall = 0
        else:
            stall += 1
            # Perturbation to escape
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m = fast_conv(a).max()
                    if m < best_m:
                        best_m = m
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[j] -= amt
                        a[i] += amt
                        a = a / a.sum()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def generate_initializations(n, rng):
    """Generate diverse initializations for the optimization."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-heavy (bimodal)
    edge = np.exp(-2.0 * np.minimum(x, 1.0 - x) * 3.0)
    edge = edge / edge.sum()
    inits.append(edge)

    # Power law at edges
    d = np.minimum(x, 1.0 - x) + 1e-6
    power = d ** (-0.5)
    power = power / power.sum()
    inits.append(power)

    # Sine wave
    sine = np.abs(np.sin(np.pi * x)) ** 0.5 + 0.1
    sine = sine / sine.sum()
    inits.append(sine)

    # Two Gaussian bumps
    g1 = np.exp(-20.0 * (x - 0.2) ** 2)
    g2 = np.exp(-20.0 * (x - 0.8) ** 2)
    bump = (g1 + g2) / (g1 + g2).sum()
    inits.append(bump)

    # Slightly perturbed uniform
    pert = np.ones(n) + rng.normal(0, 0.1, n)
    pert = np.maximum(pert, 0.01)
    pert = pert / pert.sum()
    inits.append(pert)

    return inits

def propose_candidate(budget_s=1800.0):
    """
    Main search: FFT-accelerated Nesterov Mirror Descent + LP Frank-Wolfe
    + multi-length exploration + coordinate descent polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default uniform init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: NMD at base length with best init (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 650.0)
    print(f"\n=== Phase 1: NMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, _ = nesterov_mirror_descent(best_seq, t1, initial_temp=1.5, final_temp=0.1,
                                           c0=5.0, verbose=True, rng_seed=42)
        try_update(a_out, f"NMD1_n{n0}")

    # === Phase 2: LP-FW (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP-FW n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, max_constraints=250, verbose=True)
        try_update(a_out, f"LP_n{len(best_seq)}")

    # === Phase 3: NMD with different seed/params (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: NMD v2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        # Perturb best and re-optimize
        pert = rng.normal(0, 0.01, len(best_seq))
        a_pert = np.maximum(best_seq * (1 + pert), 1e-10)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = nesterov_mirror_descent(a_pert, t3, initial_temp=0.8, final_temp=0.05,
                                           c0=3.0, verbose=True, rng_seed=777)
        try_update(a_out, f"NMD2_n{len(best_seq)}")

    # === Phase 4: Multi-length exploration (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length (budget={remaining:.0f}s) ===")
    n_candidates = [n0 + 400, n0 + 800]
    for n_alt in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100 or n_alt > 5000:
            break
        a_long = interpolate_sequence(best_seq, n_alt)
        t4 = min(remaining * 0.40, 180.0)
        a_out, _ = nesterov_mirror_descent(a_long, t4, initial_temp=1.0, final_temp=0.1,
                                           c0=4.0, verbose=True, rng_seed=n_alt % 10000)
        try_update(a_out, f"NMD_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) <= 4000:
            t4b = min(remaining * 0.35, 120.0)
            if t4b > 30:
                a_out, _ = lp_frank_wolfe(a_long, t4b, solver, max_constraints=200, verbose=False)
                try_update(a_out, f"LP_n{n_alt}")

    # === Phase 5: Different initialization at base length (8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 5: Alt init n={n0} (budget={remaining:.0f}s) ===")
        inits = generate_initializations(n0, rng)
        t5 = min(remaining * 0.5, 120.0)
        per_init_time = t5 / len(inits)
        for idx, init in enumerate(inits):
            if time.monotonic() - t_global > time_budget - 60:
                break
            if per_init_time < 10:
                continue
            a_out, _ = nesterov_mirror_descent(init, per_init_time, initial_temp=1.0,
                                               final_temp=0.05, c0=3.0, verbose=False,
                                               rng_seed=1000 + idx)
            try_update(a_out, f"Init{idx}_n{n0}")

    # === Phase 6: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 6: CD polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.7, 80.0)
        a_out, _ = coordinate_descent_polish(best_seq, t6, verbose=True, rng_seed=555)
        try_update(a_out, f"CD_{len(best_seq)}")

    # === Phase 7: Quick LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 7: Final LP n={len(best_seq)} (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.8, 60.0)
        a_out, _ = lp_frank_wolfe(best_seq, t7, solver, max_constraints=150, verbose=False)
        try_update(a_out, f"LP_final_{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
