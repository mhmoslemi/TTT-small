import numpy as np

def fast_convolution(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def mirror_descent(a0, time_limit, eta_start=0.1, eta_end=0.001, 
                   verbose=True, seed=42):
    """Mirror descent with multiplicative updates for minimizing max convolution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_conv = fast_convolution(a)
    best_max = best_conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    total_iters = 100000
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive learning rate
        frac = it / total_iters
        eta = eta_start * (eta_end / eta_start) ** frac
        
        # Compute active set (lags achieving the max within tolerance)
        conv = fast_convolution(a)
        max_val = conv.max()
        tol = 1e-10 + 1e-6 * max_val
        active = np.where(conv >= max_val - tol)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            # g_i += 2 * a[k-i] for valid i
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                g[idx] += 2.0 * a[k_int - idx]
        g /= max(1, len(active))
        
        # Center the gradient (for invariance to constant shifts)
        g = g - g.sum() / n

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * g
        # Normalize
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / a_new.sum()

        conv_new = fast_convolution(a_new)
        max_new = conv_new.max()

            best_max = max_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:

            a = a  # stay

            print(f"    MD iter={it} ({elapsed:.0f}s): score={score:.6f}, eta={eta:.4f}, |active|={len(active)}")

                best_max = best_score

                improved = True

        else:

                a[j] += t

                    best_a = a.copy()
                    stall = max(0, stall - 5)

            score = 2.0 * n * best_max
            print(f"    CD ({elapsed:.0f}s): score={score:.6f}, stall={stall}")
    
    return best_a, best_max
