"""
Minimize  2*n*max(conv(a,a)) / (sum a)^2   (a >= 0).

With sum(a) normalized to 1 the objective is  J = 2 * n * max_autocorr(a),
so the goal is to make the aperiodic autocorrelation as flat as possible
(push the peak down toward its mean).

Algorithm (no closures / no lambdas, all helpers top-level):
  Phase 0  Load parent sequence (height_sequence_1) if present, else a
           raised-cosine init; evaluate it with the provided evaluator.
  Phase 1  LP Frank-Wolfe (scipy linprog) to get a strong global direction.
  Phase 2  Nesterov accelerated mirror descent with an O(n log n) FFT
           soft-max subgradient (Hankel matrix-vector product) -- bulk of time.
  Phase 3  Low-temperature Nesterov mirror descent refinement.
  Phase 4  Coordinate-descent / paired-swap local polish.
  Phase B  Brief re-optimization at a couple of alternative lengths.
All candidates are scored with the real evaluate_sequence; the best is kept.
"""

import time
import numpy as np
import scipy.optimize as so


# ---------------------------------------------------------------------------
# Core numerics
# ---------------------------------------------------------------------------
def fast_autocorr(a):
    """Full linear autocorrelation a*a (length 2n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float("inf")


def compute_subgradient_fft(a, ac, temperature):
    """Soft-max subgradient of max_k (a*a)[k].

        G[i] = 2 * sum_j a[j] * w[i+j]      (w = softmax(ac / T))

    Computed in O(n log n) as an FFT cross-correlation (Hankel mat-vec).
    """
    n = len(a)
    log_w = ac / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    W = w.sum()
    if W < 1e-30:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / W
    N = 1 << (n + len(w) - 1 - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    return 2.0 * np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]


def exact_peak_subgradient(a, ac, tol):
    """Uniform average of the subgradients of the (near-)tightest lags."""
    n = len(a)
    peak = ac.max()
    active = np.where(ac >= peak - tol)[0]
    w = np.zeros(len(ac))
    w[active] = 1.0
    if w.sum() < 1.0:
        w[np.argmax(ac)] = 1.0
    return compute_subgradient_fft(a, ac, 1.0) * 0.0 + _hankel_matvec(a, w / w.sum()) * 2.0


def _hankel_matvec(a, w):
    """y[i] = sum_j a[j] * w[i+j] via FFT cross-correlation."""
    n = len(a)
    N = 1 << (n + len(w) - 1 - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    return np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]

def lp_direction(a, ac, M, max_constraints=350, seed=0):
    """max sum(b)  s.t.  T b <= M,  b >= 0  ->  direction b/sum(b)."""
    n = len(a)
    active = np.where(ac >= 0.97 * M)[0]
    if len(active) > max_constraints:
        active = np.argsort(ac)[::-1][:max_constraints]
    T = np.zeros((len(active), n))
    for r, k in enumerate(active):
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        if s < e:
            cols = np.arange(s, e)
            T[r, cols] = a[ki - cols]
    res = so.linprog(
        c=-np.ones(n), A_ub=T, b_ub=np.full(len(active), M),
        bounds=[(0.0, None)] * n, method="highs"
    )
    if res.status != 0 or res.x is None:
        return None
    b = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
    bs = b.sum()
    if bs < 1e-15:
        return None
    return b / bs


def lp_frank_wolfe(a0, time_limit, max_constraints=350, verbose=True):
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    rng = np.random.default_rng(0)
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        M = ac.max()
        g_dir = lp_direction(y, ac, M, max_constraints)
        if g_dir is None:
            g_dir = np.ones(n) / n
        alpha = 2.0 / (t + 2.0)
        y = (1.0 - alpha) * y + alpha * g_dir
        y = np.maximum(y, 0.0)
        s = y.sum()
        y = y / s if s > 1e-15 else np.ones(n) / n
        if t == 1:
            x = g_dir.copy()
        else:
            bt = (t - 1.0) / (t + 2.0)
            x = y + bt * (y - (x - bt * (x - y)) / (1.0))  # simple momentum safe below
            # safer plain momentum update:
            x = (1.0 - bt) * y + bt * x
            x = np.maximum(x, 0.0)
            sx = x.sum()
            x = x / sx if sx > 1e-15 else y.copy()
        for cand in (y, x):
            mc = fast_autocorr(cand).max()
            if mc < best_m:
                best_m = mc
                best_a = cand.copy()
        if verbose and t % 25 == 0:
            print(f"    LP-FW it={t} ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.8f}")
    return best_a, best_m

def nesterov_mirror_descent(a0, time_limit, init_tf=0.08, final_tf=0.0015,
                            c0=5.0, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0.copy())
    ac0 = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 6000
    run_sum = np.zeros(n)
    cnt = 0
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 8000.0)
        tf = max(1e-4, init_tf * (1.0 - frac) + final_tf * frac)

        # Nesterov extrapolation
        bt = (t - 1.0) / (t + 1.5)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else x.copy()

        ac = fast_autocorr(y)
        G = compute_subgradient_fft(y, ac, tf * ac.max())

        eta = c0 / (np.sqrt(t) + 1.0)
        log_new = np.log(y) - eta * G
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M = fast_autocorr(x).max()
        if M < best_m - 1e-16:
            best_m = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        run_sum += x
        cnt += 1
        if cnt % 2000 == 0:
            avg = run_sum / cnt
            sa = avg.sum()
            if sa > 1e-15:
                avg = avg / sa
                Ma = fast_autocorr(avg).max()
                if Ma < best_m:
                    best_m = Ma
                    best_a = avg.copy()

        if stall > max_stall:
            stall = 0
            max_stall = max(2500, max_stall // 2)
            pert = rng.normal(0.0, 0.003, n)
            xr = best_a * (1.0 + pert)
            xr = np.maximum(xr, 1e-10)
            xr = xr / xr.sum()
            x = xr
            x_prev = xr
            c0 = max(2.0, c0 * 0.7)
            run_sum = np.zeros(n)
            cnt = 0

        if verbose and t % 6000 == 0:
            print(f"    NMD t={t} ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.8f} "
                  f"stall={stall} c0={c0:.2f}")
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, seed=789):
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 4
    fracs = (0.002, 0.005, 0.01, 0.02, 0.05)
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_moves = min(80, n // 2)
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            if a[i] < 1e-14:
                continue
            targets = rng.integers(0, n, size=min(10, n))
            best_trial = best_m
            best_move = None
            for dst in targets:
                if dst == i:
                    continue
                for fr in fracs:
                    amt = a[i] * fr
                    if amt < 1e-18:
                        continue
                    tr = a.copy()
                    tr[i] -= amt
                    tr[dst] += amt
                    if tr[i] < -1e-15:
                        continue
                    tr = np.maximum(tr, 0.0)
                    tr = tr / tr.sum()
                    mt = fast_autocorr(tr).max()
                    if mt < best_trial - 1e-17:
                        best_trial = mt
                        best_move = (dst, fr)
            if best_move is not None:
                dst, fr = best_move
                amt = a[i] * fr
                a[i] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial
                best_a = a.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * float(rng.uniform(0.003, 0.02))
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                mn = fast_autocorr(a).max()
                if mn < best_m:
                    best_m = mn
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        if verbose and stall % 3 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.8f} stall={stall}")
    return best_a, best_m

def make_init(n, parent):
    if parent is not None and len(parent) > 0:
        p = np.maximum(np.asarray(parent, dtype=np.float64), 0.0)
        s = p.sum()
        if s > 1e-15:
            p = p / s
            ox = np.linspace(0.0, 1.0, len(p))
            nx = np.linspace(0.0, 1.0, n)
            r = np.maximum(np.interp(nx, ox, p), 0.0)
            ss = r.sum()
            if ss > 1e-15:
                return r / ss
    x = np.linspace(0.0, 2.0 * np.pi, n)
    r = np.maximum(0.5 * (1.0 + np.cos(x)), 1e-10)
    return r / r.sum()

        out, _ = nesterov_mirror_descent(a_alt, max(30.0, tb), init_tf=0.08,
                                         final_tf=0.002, c0=4.0, verbose=True,
                                         seed=31)
        record(out, f"ALT{alt_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        cur = norm_sum1(out)

    # Final: confirm with the real evaluator and return
    if best_seq is None:
        best_seq = a0
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL score={best_score:.8f} n={len(best_seq)} "
          f"t={time.monotonic()-t_global:.0f}s")

    out = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    s = out.sum()
    if s < 1e-12:
        out = np.ones(1)
    else:
        out = out / s
    return [float(v) for v in out]
