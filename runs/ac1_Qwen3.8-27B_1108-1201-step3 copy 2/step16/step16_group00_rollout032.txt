"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient soft-subgradient mirror descent with Nesterov acceleration
2. LP-based Frank-Wolfe with adaptive constraint selection
3. Coordinate descent polish with peak-targeting moves
4. Multi-length exploration and diverse initializations
Key insight: optimal sequences exploit sparsity (zeros) to reduce peak autocorrelation

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def soft_subgradient(a, ac, top_k, temperature):
    """Efficient soft subgradient: grad[i] = sum_k w_k * 2*a[k-i] (valid range)."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    w = np.exp(temperature * shifted)
    # Kill underflow
    w[w < 1e-300] = 0.0
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        # grad[i] += 2*wi*a[ki-i] for valid i (0 <= ki-i < n, i.e., ki-n+1 <= i <= ki)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo > i_hi:
            continue
        idx = i_lo + np.arange(i_hi - i_lo + 1)
        a_idx = ki - idx
        grad[idx] += 2.0 * wi * a[a_idx]
    return grad


def make_init_sine(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
    return norm_sum1(a)


def make_init_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_init_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5 * width) ** 2) / (2 * width ** 2)) + \
        np.exp(-((x - 1.0 + 0.5 * width) ** 2) / (2 * width ** 2)) + 0.005
    return norm_sum1(a)


def make_init_sparse(n, density=0.5, seed=0):
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    mask = rng.random(n) < density
    a[mask] = rng.uniform(0.5, 1.5, mask.sum())
    return norm_sum1(a)


def make_init_chebyshev(n, order=3):
    x = np.linspace(-1, 1, n)
    from numpy.polynomial import chebyshev as C
    a = np.abs(C.chebval(x, np.ones(order + 1))) + 0.001
    return norm_sum1(a)


def amd_optimize(a0, time_limit, top_k=50, temperature=6.0, c0=5.0,
                 restart_every=6000, verbose=False):
    """Mirror descent with Nesterov acceleration and adaptive restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_temp = temperature

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-14)
        sy = y.sum()
        if sy < 1e-14:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = soft_subgradient(y, ac, top_k=cur_top_k, temperature=cur_temp)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-14:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert_scale = min(0.03, 0.01 + 0.02 * (stall_count / restart_every))
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_temp = max(cur_temp - 0.5, 2.0)
            gamma = max(gamma * 0.9, 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, temp={cur_temp:.1f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix: T[k, i] = y[k-i] if 0<=k-i<n else 0
        # Use top-K binding constraints
        K_sel = min(top_K, 2 * n - 1)
        # Find most binding constraints (smallest margin to M_curr)
        # Compute all constraint values: T @ y = ac (the autocorrelation of y with itself is not quite right)
        # Actually T @ b where T is built from y gives the cross-correlation y * b
        # We need: (y * b)[k] = sum_i T[k,i] * b[i] where T[k,i] = y[k-i] (valid)
        # The constraint is (y * b)[k] <= M_curr for all k
        # The binding ones are where (y*y)[k] is close to M_curr, i.e., where ac[k] is largest
        top_idx = np.argsort(ac)[::-1][:K_sel]

        T_sel = np.zeros((K_sel, n))
        for row, k in enumerate(top_idx):
            ki = int(k)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            if i_lo <= i_hi:
                T_sel[row, i_lo:i_hi + 1] = y[ki - np.arange(i_lo, i_hi + 1)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-14:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t <= 2:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-14:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent targeting top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            # Find the pair (i, ki-i) with maximum product contributing to this lag
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-25:
                continue

            src_pos = idx[np.argmax(contrib)]
            src_val = a[src_pos]
            if src_val < 1e-14:
                continue

            # Find best destination: position that minimizes increase in top lags
            # Score each position by how much it would increase peak lags
            n_sens = min(8, n_top)
            scores = np.zeros(n)
            for k2 in top_k[:n_sens]:
                k2i = int(k2)
                for d in range(n):
                    p = k2i - d
                    if 0 <= p < n:
                        scores[d] += a[p]

            # Prefer moving mass to low-score positions
            candidates = np.argsort(scores)[:max(5, n // 10)]
            best_dst = candidates[np.argmax(a[candidates] < 0.1 * a.max())] if (a[candidates] < 0.1 * a.max()).any() else int(rng.integers(n))

            if best_dst == src_pos:
                continue

            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.1, 0.15]:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-14:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 80:
                j = int(rng.integers(n))
                if j != src_pos and a[src_pos] > 1e-12:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src_pos] * frac
                    a[src_pos] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-14:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

    t4 = min(remaining * 0.10, 100.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=False, rng_seed=77)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # Phase 5: Try longer sequences
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")
    longer_lengths = sorted(set([n_best + 400, n_best + 800, n_best + 1200]), reverse=True)
    longer_lengths = [l for l in longer_lengths if l <= 5000]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.35, 180.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=60, temperature=5.0,
                                     c0=4.0, restart_every=5000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 3500:
            t5b = min(remaining * 0.25, 120.0)
            if t5b > 20:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5b, solver,
                                           verbose=False, top_K=250)
                # Interpolate result to new_n and do a quick AM
                a_interp = interpolate_sequence(a_out, new_n)
                t_short = min(remaining * 0.1, 40.0)
                if t_short > 10:
                    a_out2, _ = amd_optimize(a_interp, t_short, top_k=60, temperature=5.0,
                                              c0=3.0, restart_every=3000, verbose=False)
                    try_update(a_out2, f"lp_interp_n{new_n}")

    # Phase 6: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 50 and n_best <= 3500:
        t6 = min(remaining * 0.30, 180.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=False, top_K=400)
            try_update(a_out, f"lp_final_n{n_best}")

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 25:
        t7 = min(remaining * 0.45, 120.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # Phase 8: One more AMD if time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t8, top_k=120, temperature=3.0,
                                     c0=2.0, restart_every=5000, verbose=False)
            try_update(a_out, f"amd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_init_sine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
