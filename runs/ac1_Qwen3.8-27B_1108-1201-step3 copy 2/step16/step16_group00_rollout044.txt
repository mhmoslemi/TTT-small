    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_j(a):
    """Compute the objective J = 2*n*max(conv(a,a))/(sum(a))^2 for normalized a."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def fw_line_search(y, g_dir, M, max_alpha=1.0):
    """
    Exact line search for FW step.
    Minimize compute_j((1-t)*y + t*g_dir) over t in [0, 1].
    Uses golden section search.
    """
    # Check endpoints
    val_0 = compute_j(y)
    val_1 = compute_j(g_dir)
    if val_1 >= val_0:
        # Direction is not improving, try smaller step
        # Bisect
        lo, hi = 0.0, 1.0
        for _ in range(20):
            mid = 0.5 * (lo + hi)
            point = (1.0 - mid) * y + mid * g_dir
            point = np.maximum(point, 0.0)
            s = point.sum()
            if s < 1e-15:
                mid = lo
                break
            point = point / s
            val_mid = compute_j(point)
            if val_mid < val_0:
                lo = mid
            else:
                hi = mid
        best_t = lo
    else:
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, max_alpha
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = compute_j(norm_sum1((1.0 - c) * y + c * g_dir))
        fd = compute_j(norm_sum1((1.0 - d) * y + d * g_dir))
        for _ in range(30):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = compute_j(norm_sum1((1.0 - c) * y + c * g_dir))
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = compute_j(norm_sum1((1.0 - d) * y + d * g_dir))
        best_t = 0.5 * (lo + hi)
    
    best_point = (1.0 - best_t) * y + best_t * g_dir
    return best_t, norm_sum1(best_point), compute_j(norm_sum1(best_point))


def lp_frank_wolfe_line_search(a0, time_limit, solver, verbose=True, top_K=300, use_line_search=True):
    """LP-based Frank-Wolfe with exact line search and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    iter_count = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select top-K most binding constraints
        K_sel = min(top_K, 2 * n - 1)
        # Find the K lags with highest autocorrelation values
        binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix: T_sel[k, i] = y[k-i] (valid indices only)
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    T_sel[ii, i] = y[j]
        
        # Build M vector
        M_vec = ac[binding_idx].copy()
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_vec]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        if use_line_search:
            # Exact line search
            alpha, y_new, val_new = fw_line_search(y, g_dir, M_curr)
        else:
            # Standard FW step
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy
            val_new = compute_j(y_new)
        
        # Nesterov update for x
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = (1.0 - beta_t) * y_new + beta_t * (y_new - y)
            # Actually, standard Nesterov: x_new = y_new + (t-1)/(t+2) * (y_new - y)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-LS iter={t} ({elapsed:.0f}s): J={J:.6f}, alpha={alpha:.4f}")
    
    return best_a, best_m


def gradient_descent_exploit(a0, time_limit, verbose=True):
    """
    Gradient descent exploiting the structure d/da[i] conv[k] = 2*a[k-i].
    Uses the exact subgradient at the argmax lag.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    lr = 0.1  # Learning rate
    decay = 0.999
    stall = 0
    stall_limit = 500
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))
        
        # Exact subgradient at k_star: g[i] = 2*a[k_star - i] (valid indices)
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Mirror descent step (exponential update)
        eta = lr / (1.0 + 0.1 * t)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            a = a_new
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            # Try smaller step
            eta_half = eta * 0.5
            log_a2 = np.log(np.maximum(a, 1e-30)) - eta_half * g
            max_log2 = log_a2.max()
            a_half = np.exp(log_a2 - max_log2)
            a_half = np.maximum(a_half, 1e-30)
            a_half = a_half / a_half.sum()
            M_half = fast_autocorr(a_half).max()
            if M_half < M_curr:
                a = a_half
                if M_half < best_m:
                    best_m = M_half
                    best_a = a_half.copy()
                stall = 0
            else:
                stall += 1
                lr *= decay
        
        if stall > stall_limit:
            # Restart with perturbation
            pert = rng.normal(0, 0.01, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            stall = 0
            lr = 0.1
            stall_limit = int(stall_limit * 1.2)
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GD t={t} ({elapsed:.0f}s): J={J:.6f}, lr={lr:.4f}, stall={stall}")
    
    return best_a, best_m
