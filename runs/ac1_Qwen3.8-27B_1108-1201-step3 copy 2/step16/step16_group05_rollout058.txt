- Term 2: i in S, i+k not in S → a_i * a_{i+k} (linear in a_S since a_{i+k} is fixed)
- Term 3: i not in S, i+k in S → a_i * a_{i+k} (linear in a_S since a_{i+k} is in S)
- Term 4: i not in S, i+k not in S → fixed

So (a*a)[k] = [Term 1] + [Term 2 + Term 3] + [Term 4]

Term 1 is quadratic in a_S for lags where both i and i+k are in S. This happens when 0 <= k < B (and i, i+k both in S).

For the LP, I'll approximate Term 1 by its value at the current a_S (i.e., treat it as a constant). This gives:

(a*a)[k] ≈ C'_k + D_k @ a_S

where C' includes Term 1 evaluated at current a_S plus Term 4, and D_k @ a_S = Term 2 + Term 3.

This is an LP! The approximation is exact at the current point and becomes less accurate as a_S moves away, but with small step sizes (or with re-linearization after each update), it should work well.

```python
from scipy.optimize import linprog

def solve_block_lp(D, C, s_sum, B_actual):
    # min t
    # s.t. D @ x - t <= C   (for each k: D[k,:] @ x - t <= C[k])
    #      x >= 0
    #      sum(x) = s_sum
    #
    # Variables: [x_0, ..., x_{B-1}, t]
    # Objective: minimize t → c = [0, ..., 0, 1]
    # Inequality constraints: [D, -I] @ [x; t] <= C
    # Equality: sum(x) = s_sum → [1, ..., 1, 0] @ [x; t] = s_sum
    # Bounds: x_i in [0, inf), t in [-inf, inf]
    
    n_vars = B_actual + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0  # minimize t
    
    # Inequality: D @ x - t <= C
    # [D, -1] @ [x; t] <= C
    A_ub = np.hstack([D, -np.ones((D.shape[0], 1))])
    b_ub = C
    
    # Equality: sum(x) = s_sum
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :B_actual] = 1.0
    b_eq = np.array([s_sum])
    
    bounds = [(0, None)] * B_actual + [(-np.inf, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    if result.success:
        return result.x[:B_actual], result.x[-1]
    return None, None
```

```python
def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.maximum(v - (v.sum() - 1.0) / n, 0.0)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    max_shift = shifted.max()
    if max_shift > 50:
        shifted = shifted / max_shift * 50
    exp_vals = np.exp(beta * (shifted - shifted.max()))
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, ac.max()


def accelerated_projected_gradient(a0, time_limit, beta_start=25.0, beta_end=6.0,
                                    top_k=200, c0=2.0, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 2000
    current_top_k = top_k

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 15000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)
        g, _ = compute_smoothed_gradient(y, cur_beta, top_k=current_top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15
        eta = gamma / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      APG restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def build_T_matrix(a):
    """Build the convolution matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_fw_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M, b >= 0."""
    n = len(a)
    T = build_T_matrix(a)
    c = -np.ones(n)
    A_ub = T
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if result.success:
        b = np.maximum(result.x, 0)
        return b, b.sum()
    return None, 0


def fw_line_search(a, b, n_points=500):
    """Find best t in [0,1] minimizing max_k ((1-t)*a+t*b)*((1-t)*a+t*b)."""
    aa = fast_autocorr(a)
    ab = np.convolve(a, b)
    bb = fast_autocorr(b)
    # f_k(t) = (1-t)^2*aa[k] + 2t(1-t)*ab[k] + t^2*bb[k]
    # = aa[k] + t*(2*ab[k]-2*aa[k]) + t^2*(aa[k]-2*ab[k]+bb[k])
    A = aa - 2 * ab + bb
    B = 2 * (ab - aa)
    C = aa
    ts = np.linspace(0, 1, n_points)
    vals = C[:, None] + ts[None, :] * B[:, None] + (ts[None, :] ** 2) * A[:, None]
    max_vals = vals.max(axis=0)
    best_idx = np.argmin(max_vals)
    best_t = ts[best_idx]
    # Refine with golden section
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(len(ts) - 1, best_idx + 2)]
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(30):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        v1 = (C + m1 * B + m1**2 * A).max()
        v2 = (C + m2 * B + m2**2 * A).max()
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    best_t = (lo + hi) / 2
    return best_t


def lp_frank_wolfe(a0, time_limit, verbose=True):
    """LP-based Frank-Wolfe with exact line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        b, b_sum = lp_fw_direction(y, M_curr)
        if b is None or b_sum < 1e-15:
            break
        b_norm = b / b_sum
        # Check LP value vs current
        if b_sum <= 1.0 + 1e-8:
            # No improvement expected
            break
        # Line search
        best_t = fw_line_search(y, b_norm)
        y_new = (1.0 - best_t) * y + best_t * b_norm
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            continue
        y_new = y_new / sy
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y_mom = y_new + beta_t * (y_new - y)
        y_mom = np.maximum(y_mom, 0.0)
        sm = y_mom.sum()
        if sm > 1e-15:
            y_mom = y_mom / sm

        for candidate in [y_new, y_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-16:
                best_m = mc
                best_a = candidate.copy()
        y = y_new

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, t={best_t:.4f}, bsum={b_sum:.4f}")

    return best_a, best_m


def build_block_D(a, block_start, block_size):
    """Build the linear coefficient matrix for a block LP."""
    n = len(a)
    end = min(block_start + block_size, n)
    B_actual = end - block_start
    num_lags = 2 * n - 1
    D = np.zeros((num_lags, B_actual))
    for j in range(B_actual):
        idx = block_start + j
        # D[k,j] = a[idx+k] (if valid, not in block) + a[idx-k] (if valid, not in block)
        for k in range(n - idx):
            other = idx + k
            if block_start <= other < end:
                continue
            D[k, j] += a[other]
        for k in range(1, idx + 1):
            other = idx - k
            if block_start <= other < end:
                continue
            D[k, j] += a[other]
    return D


def solve_block_lp_scipy(D, C, s_sum, B_actual):
    """Solve: min t s.t. D@x - t <= -C, x >= 0, sum(x) = s_sum."""
    n_vars = B_actual + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0
    A_ub = np.hstack([D, -np.ones((D.shape[0], 1))])
    b_ub = -C
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :B_actual] = 1.0
    b_eq = np.array([s_sum])
    bounds = [(0, None)] * B_actual + [(-np.inf, None)]
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x[:B_actual], result.x[-1]
    return None, None


def block_lp_coordinate_descent(a0, time_limit, block_size=50, verbose=True, seed=7):
    """Block coordinate descent where each block update solves a small LP."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    num_blocks = (n + block_size - 1) // block_size
    stall = 0
    max_stall = 200
    sweep = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        improved_this_sweep = False

        # Shuffle block order for diversity
        block_order = list(range(num_blocks))
        rng.shuffle(block_order)

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            bs = bi * block_size
            be = min(bs + block_size, n)
            B_actual = be - bs
            if B_actual < 2:
                continue

            s = a[bs:be]
            s_sum = s.sum()
            if s_sum < 1e-12:
                continue

            # Build D and C
            D = build_block_D(a, bs, block_size)
            D = D[:2 * n - 1, :B_actual]
            aa = fast_autocorr(a)
            C = aa - D @ s

            # Solve block LP
            x_new, t_val = solve_block_lp_scipy(D, C, s_sum, B_actual)
            if x_new is None:
                continue
            x_new = np.maximum(x_new, 0)
            # Renormalize to maintain exact sum
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new * (s_sum / sx)

            # Check if this actually improves
            a_trial = a.copy()
            a_trial[bs:be] = x_new
            M_trial = fast_autocorr(a_trial).max()

            if M_trial < best_m - 1e-14:
                # Line search: find best alpha in [0,1]
                a_old_block = s.copy()
                best_alpha = 1.0
                best_M = M_trial
                for alpha in [1.0, 0.8, 0.6, 0.4, 0.2]:
                    a_ls = a.copy()
                    a_ls[bs:be] = (1 - alpha) * a_old_block + alpha * x_new
                    M_ls = fast_autocorr(a_ls).max()
                    if M_ls < best_M:
                        best_M = M_ls
                        best_alpha = alpha
                a[bs:be] = (1 - best_alpha) * a_old_block + best_alpha * x_new
                best_m = best_M
                best_a = a.copy()
                improved_this_sweep = True
            elif M_trial < fast_autocorr(a).max() - 1e-12:
                # Slight improvement from current (but not from best)
                a[bs:be] = x_new
                improved_this_sweep = True

        if improved_this_sweep:
            stall = 0
        else:
            stall += 1
            # Perturbation to escape
            idx = int(rng.integers(n))
            if a[idx] > 1e-10:
                jdx = int(rng.integers(n))
                if jdx != idx:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[idx] * frac
                    a[idx] -= amt
                    a[jdx] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    M_p = fast_autocorr(a).max()
                    if M_p < best_m:
                        best_m = M_p
                        best_a = a.copy()
                        stall = 0

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      BlockLP sweep={sweep} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      BlockLP done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent with peak-aware moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        # Compute peak contribution
        peak_contrib = np.zeros(n)
        for k in top_k[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_contrib[idx] += a[idx] * a[j_idx]

        sources = np.argsort(peak_contrib)[::-1][:20]
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            # Find best destination
            best_dst = -1
            best_delta = 1e-15
            n_try = min(n, 300)
            candidates = rng.choice(n, n_try, replace=False)

            for dst in candidates:
                if dst == src:
                    continue
                # Estimate: adding to dst increases lags k where dst+k or dst-k has mass
                impact = 0.0
                for k in top_k[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d] * 0.5
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2] * 0.5
                score = -impact
                if score > best_delta:
                    best_delta = score
                    best_dst = dst

            if best_dst < 0:
                continue
            for frac in [0.003, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with block LP as the primary method."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    rng = np.random.default_rng(2024)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = get_parent_sequence()
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Quick APG warm-up ===
    rem = remaining_time()
    print(f"\n=== Phase 1: APG warm-up n={n0} ===")
    t1 = min(rem * 0.08, 120.0)
    if t1 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_projected_gradient(
            a_in, t1, beta_start=25.0, beta_end=8.0,
            top_k=200, c0=2.0, verbose=True, seed=42)
        try_update(a_out, f"apg_warm_n{n0}")

    # === Phase 2: Block LP coordinate descent (MAIN) ===
    rem = remaining_time()
    print(f"\n=== Phase 2: Block LP n={len(best_seq)} ===")
    t2 = min(rem * 0.40, 650.0)
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        # Use smaller blocks first for speed
        t2a = min(t2 * 0.4, 250.0)
        a_out, _ = block_lp_coordinate_descent(
            a_in, t2a, block_size=30, verbose=True, seed=7)
        try_update(a_out, f"block30_n{len(a_out)}")

        rem = remaining_time()
        if rem > 60:
            t2b = min(rem * 0.5, 350.0)
            a_in2 = norm_sum1(best_seq)
            a_out2, _ = block_lp_coordinate_descent(
                a_in2, t2b, block_size=60, verbose=True, seed=13)
            try_update(a_out2, f"block60_n{len(a_out2)}")

    # === Phase 3: LP Frank-Wolfe with line search ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 3000:
        print(f"\n=== Phase 3: LP-FW n={n_best} ===")
        t3 = min(rem * 0.20, 300.0)

            a_out, _ = lp_frank_wolfe(a_in, t3, verbose=True)

    rem = remaining_time()
    print(f"\n=== Phase 4: APG polish n={len(best_seq)} ===")

    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_projected_gradient(

            top_k=250, c0=1.5, verbose=False, seed=77)
        try_update(a_out, f"apg_polish_n{len(a_out)}")

    # === Phase 5: Coordinate descent polish ===
    rem = remaining_time()
    print(f"\n=== Phase 5: Coord polish n={len(best_seq)} ===")

            rem = remaining_time()
            if rem > 40:

                    norm_sum1(best_seq if len(best_seq) == n1 else a_out),

    rem = remaining_time()

    if rem > 30:
        print(f"\n=== Phase 8: Final coord n={n_best} ===")
        t8 = min(rem * 0.6, 60.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
