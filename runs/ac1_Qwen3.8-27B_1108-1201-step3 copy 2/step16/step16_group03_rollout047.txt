import numpy as np
import cvxpy as cp
import time

# Helper functions (top-level, no closures)

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_grad_logsumexp(a, beta):
    """
    Compute gradient of (2n/beta) * log(sum exp(beta * C[k])) where C = conv(a,a).
    Returns the gradient vector of length n.
    """
    n = len(a)
    C = fast_conv(a)
    max_C = C.max()
    shifted = beta * (C - max_C)
    # Clip for numerical stability
    shifted = np.minimum(shifted, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    # g[j] = 4*n * sum_{j'=0}^{n-1} p[j+j'] * a[j']
    # Using np.correlate: correlate(p, a, 'full')[n-1 : 2n-1]
    corr = np.correlate(p, a, mode='full')
    g = 4.0 * n * corr[n-1 : 2*n-1]
    return g

def hard_subgradient_topk(a, top_k=1, beta_peak=3.0):
    """
    Weighted subgradient using top-k peaks of the convolution.
    """
    n = len(a)
    C = fast_conv(a)
    k_max = min(top_k, len(C))
    top_idx = np.argsort(C)[::-1][:k_max]
    top_vals = np.maximum(C[top_idx], 1e-30)
    w = top_vals ** beta_peak
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += 2.0 * wi * a[j]
    return grad

def mirror_descent_adam(a0, time_limit, beta_schedule, verbose=True, seed=42):
    """
    Mirror descent with Adam-like adaptive learning rates and log-sum-exp smoothing.
    beta_schedule: list of (beta, n_iters) pairs
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_C = fast_conv(a).max()
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    alpha_lr = 0.01  # Base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    for beta, n_iters in beta_schedule:
        for _ in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_C
            
            # Compute smooth gradient
            y = np.maximum(a, 1e-30)
            y = y / y.sum()
            g = smooth_grad_logsumexp(y, beta)
            
            # Normalize gradient (scale-invariant objective, so gradient is on the tangent space)
            # Project gradient onto tangent space of simplex: g_tan = g - mean(g) * a
            # Actually for mirror descent, we just use g directly in the multiplicative update
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            
            # Mirror descent step: a_new[j] = a[j] * exp(-lr * m_hat[j] / sqrt(v_hat[j]))
            lr = alpha_lr / np.sqrt(t_adam + 1)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            
            log_update = -step
            max_log = log_update.max()
            a_new = a * np.exp(log_update - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            a_new = a_new / s_new
            
            # Track best
            C_max = fast_conv(a_new).max()
            if C_max < best_C:
                best_C = C_max
                best_a = a_new.copy()
            
            a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_C
            print(f"    Beta={beta}, elapsed={elapsed:.1f}s, J={J:.8f}")
    
    return best_a, best_C

def lp_direction(y, M, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for all k, b >= 0.
    Returns the optimal b (normalized to sum=1) or None.
    """
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def frank_wolfe_lp(a0, time_limit, solver, verbose=True, nesterov=True):
    """LP-based Frank-Wolfe with Nesterov acceleration and exact line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_C = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_mom = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        C_y = fast_conv(y)
        M_curr = C_y.max()
        if M_curr < 1e-14:
            break
        
        # Get LP direction
        direction = lp_direction(y, M_curr, solver)
        if direction is None:
            # Try with slightly relaxed M
            direction = lp_direction(y, M_curr * 1.01, solver)
            if direction is None:
                break
        
        # Exact line search
        alphas = np.concatenate([
            np.array([2.0 / (t + 2.0)]),
            np.linspace(0.01, 0.99, 20)
        ])
        best_alpha = alphas[0]
        best_eval = fast_conv(np.maximum(y + best_alpha * (direction - y), 0)).max()
        
        for alpha in alphas[1:]:
            trial = y + alpha * (direction - y)
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_C = fast_conv(trial).max()
            if trial_C < best_eval:
                best_eval = trial_C
                best_alpha = alpha
        
        y_new = y + best_alpha * (direction - y)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov momentum
        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            x_mom = y_new + beta_t * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sx = x_mom.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_mom / sx
        else:
            x_mom = y_new.copy()
        
        y = y_new
        
        # Track best
        C_new = fast_conv(y).max()
        if C_new < best_C:
            best_C = C_new
            best_a = y.copy()
        C_mom = fast_conv(x_mom).max()
        if C_mom < best_C:
            best_C = C_mom
            best_a = x_mom.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_C
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.8f}, alpha={best_alpha:.4f}")
    
    return best_a, best_C

def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with mass transfer moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_C = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_conv(a)
        top_peaks = np.argsort(C)[::-1][:min(5, len(C))]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_C
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            
            # Try transferring mass from src to various destinations
            # Focus on positions that don't affect the top peaks much
            n_tries = min(n, 100)
            for _ in range(n_tries):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    C_trial = fast_conv(a_trial).max()
                    if C_trial < best_C - 1e-18:
                        a = a_trial
                        best_C = C_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        C_new = fast_conv(a).max()
                        if C_new < best_C:
                            best_C = C_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            # Revert
                            a[j] -= amt
                            a[i] += amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
    
    return best_a, best_C

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():
    """Find a working LP solver."""
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS

Wait, I need to double-check my smooth gradient computation. Let me re-derive it.

We want to minimize:
J(a) = (2n/beta) * log(sum_k exp(beta * C[k]))

where C[k] = (a*a)[k] = sum_j a[j]*a[k-j] (with valid indices).

dJ/da[j0] = (2n/beta) * (1/S) * sum_k exp(beta*C[k]) * dC[k]/da[j0]

where S = sum_k exp(beta*C[k]).

And dC[k]/da[j0] = 2*a[k-j0] (if 0 <= k-j0 < n), else 0.

So:
dJ/da[j0] = (2n/beta) * (1/S) * sum_k exp(beta*C[k]) * 2*a[k-j0] * [0 <= k-j0 < n]
= (4n/beta) * (1/S) * sum_{k: 0 <= k-j0 < n} exp(beta*C[k]) * a[k-j0]
= (4n/beta) * sum_{j': 0 <= j' < n} exp(beta*C[j0+j']) * a[j'] / S
= (4n/beta) * sum_{j': 0 <= j' < n} p[j0+j'] * a[j']

where p[k] = exp(beta*C[k]) / S.

So the gradient is:
g[j0] = (4n/beta) * sum_{j'=0}^{n-1} p[j0+j'] * a[j']

And using np.correlate:
corr[m] = sum_k p[k+m] * a[k] for valid k

So g[j0] = (4n/beta) * corr[j0]

In the 'full' mode of np.correlate, the output has length len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n-2.
The index offset is: output[i] corresponds to m = i - (len(a) - 1) = i - (n-1).

So corr[j0] is at output index j0 + (n-1).

g = (4n/beta) * np.correlate(p, a, mode='full')[n-1 : n-1+n]
