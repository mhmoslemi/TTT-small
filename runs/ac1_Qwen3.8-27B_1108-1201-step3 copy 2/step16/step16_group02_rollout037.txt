However, for n=2400, a full Newton step would require an n×n Hessian, which is 5.76M entries - feasible but the Hessian computation would be expensive.

Alternative: **L-BFGS** with the smoothed objective. scipy has L-BFGS-B which handles bound constraints. We can minimize `f_β(a)` subject to `a_i >= 0` and `sum(a) = 1`.

This is probably the most straightforward improvement. Let me use:
- scipy.optimize.minimize with method='L-BFGS-B'
- Constraint: sum(a) = 1 (handled by parameterizing as a simplex, or using SLSQP)
- Bounds: a_i in [0, 1]

Actually, for the simplex constraint, I can parameterize `a_i = exp(b_i) / sum_j exp(b_j)` (softmax), which automatically satisfies the constraints. Then I minimize `f_β(softmax(b))` over `b` unconstrained.

But L-BFGS on 2400 variables with an expensive objective evaluation might be slow. Each evaluation requires computing the full convolution (O(n^2) or O(n log n) with FFT) and the log-sum-exp.

Let me think about the computational budget:
- 1800 seconds total
- Each L-BFGS iteration evaluates the objective and gradient: ~2 FFT computations (O(n log n))
- For n=2400, one FFT is ~2400 * 12 ≈ 28800 operations, so very fast
- L-BFGS might need 100-1000 iterations to converge
- Total: maybe 2000 FFT computations, which is trivial

So L-BFGS should be very fast. The question is whether it can find a good minimum.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed objective with L-BFGS (multi-beta annealing)
2. Nesterov mirror descent with exact subgradient for final convergence
3. Multiple diverse initializations
4. Adaptive length search
Key insight: The gradient of the smoothed objective w.r.t. log-space parameters
is O(n^2) to compute, making L-BFGS very fast for n ~ 2400.

import numpy as np
import time
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smoothed_obj_grad(b, beta):
    n = len(b)
    # Softmax
    b_max = b.max()
    exp_b = np.exp(b - b_max)
    a = exp_b / exp_b.sum()
    
    # Autocorrelation
    ac = fast_autocorr(a)
    
    # Log-sum-exp
    beta_ac = beta * ac
    bmax = beta_ac.max()
    exp_vals = np.exp(beta_ac - bmax)
    log_sum = bmax + np.log(exp_vals.sum())
    obj = log_sum / beta
    
    # Weights
    w = exp_vals / exp_vals.sum()
    
    # Gradient w.r.t. a: g[i] = 2 * sum_m w[i+m] * a[m]
    # Using direct computation
    # g[i] = 2 * dot(w[i:i+n], a)
    idx = np.arange(n)
    w_slice = w[idx[:, None] + idx[None, :]]  # w_slice[i,m] = w[i+m]
    g = 2.0 * (w_slice @ a)
    
    # Gradient w.r.t. b: grad_b = a * (g - dot(g,a))
    g_dot_a = np.dot(g, a)
    grad_b = a * (g - g_dot_a)
    
    return obj, grad_b

def exact_subgradient(a):
    """Compute subgradient of max_k (a*a)[k] w.r.t. a."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    
    # Find active lags (within tolerance of max)
    tol = 1e-10
    active = np.where(ac >= M - tol)[0]
    
    # For numerical stability, use soft weights over active lags
    if len(active) <= 1:
        w = np.zeros(len(ac))
        w[active[0]] = 1.0
    else:
        # Use slightly soft weighting to avoid exact ties
        ac_active = ac[active]
        w_active = np.exp(100 * (ac_active - ac_active.max()))
        w_active /= w_active.sum()
        w = np.zeros(len(ac))
        w[active] = w_active
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    g = np.zeros(n)
    for k in active:
        wi = w[k]
        if wi < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            g[i] += 2.0 * wi * a[k - i]
    
    return g, M

def nesterov_mirror_descent(a0, time_limit, c0=3.0, verbose=True):

    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        g, M_curr = exact_subgradient(y)
        
        eta = c0 / (np.sqrt(t) + 1.0)
        log_x = np.log(y) - eta * g
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:

            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            stall = 0
            c0 = max(1.0, c0 * 0.7)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def generate_initializations(n, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Parent sequence (interpolated to length n)
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.sum() > 1e-10:
            p = p / p.sum()
            old_n = len(p)
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            p_interp = np.interp(x_new, x_old, p)
            p_interp = np.maximum(p_interp, 0)
            p_interp = p_interp / p_interp.sum()
            inits.append(('parent', p_interp))
    
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp / ramp.sum()))
    
    # Reverse ramp
    inits.append(('rev_ramp', ramp[::-1] / ramp.sum()))
    
    # Sinusoidal
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2
    inits.append(('sine', sine / sine.sum()))
    
    # Quadratic ramp
    quad = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits.append(('quad', quad / quad.sum()))
    
    # Asymmetric bump (Gaussian-like)
    center = 0.3 * (n - 1)
    sigma = 0.2 * n
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(('gauss', gauss / gauss.sum()))
    
    # Random
    rng = np.random.default_rng(123)
    rand = rng.random(n)
    inits.append(('random', rand / rand.sum()))
    
    return inits

def lbfgs_optimize(a0, betas, max_iter_per_beta=200, tol=1e-10, time_limit=np.inf):
    """Run L-BFGS optimization with multiple beta values."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    
    b = np.log(np.maximum(a, 1e-30))  # Log-space initialization
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        result = minimize(
            lambda bb: smoothed_obj_grad(bb, beta),
            b.copy(),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_beta, 'ftol': tol, 'gtol': 1e-8}
        )
        
        b = np.asarray(result.x, dtype=np.float64)
        
        # Convert back
        exp_b = np.exp(b - b.max())
        a_new = exp_b / exp_b.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        M = fast_autocorr(a_new).max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_m = M
            best_a = a_new.copy()
        
        print(f"      beta={beta:.0f}: smoothed_obj={result.fun:.8f}, actual_score={score:.8f}")
    
    return best_a, best_score

def coordinate_polish(a0, time_limit, verbose=True):
    """Simple coordinate-wise polish: try adjusting each coordinate."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(789)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 20:
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            a_i = a[i]
            if a_i < 1e-15:
                continue
            
            # Try moving small amount from i to a random other coordinate
            for _ in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                for frac in [0.005, 0.01, 0.03, 0.1]:
                    amt = a_i * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_t = fast_autocorr(a_trial).max()
                    if m_t < best_m - 1e-12:
                        a = a_trial
                        best_m = m_t
                        improved = True
                        break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * 0.02
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    stall = 0
    
    return a, best_m

        a_out, score = lbfgs_optimize(a_init, betas, max_iter_per_beta=100, 
                                       time_limit=per_init_time)

        t_nmd = min(remaining * 0.3, 400.0)
