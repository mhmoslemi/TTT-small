"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a.

Method (differs from prior art in execution strategy):
 1. LP-based Frank-Wolfe using the FULL constraint set (all 2n-1 lags),
    a vectorized sparse matrix build, and a golden-section line search on the
    exact (convex-in-tau) objective along the FW chord.
 2. Multi-length exploration (parent length + a couple of longer lengths),
    each started from an interpolated parent / structural init.
 3. Accelerated Mirror Descent (AMD) with soft subgradient + random restarts
    to polish the best FW iterate.
 4. Stochastic coordinate / perturbation polish in the final time budget.
"""

import numpy as np
from scipy import sparse
import cvxpy as cp
import time

_GLOBAL_SOLVER = None


def get_solver():
    global _GLOBAL_SOLVER
    if _GLOBAL_SOLVER is not None:
        return _GLOBAL_SOLVER
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _GLOBAL_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _GLOBAL_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _GLOBAL_SOLVER


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_two(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def objective_value(a, maxb=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if maxb is None:
        maxb = fast_autocorr(a).max()
    s = a.sum()
    if s <= 0:
        return float('inf')
    return float(2.0 * n * maxb / (s * s))


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = f.size
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structural_init(n, kind):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'ramp':
        a = np.linspace(0.05, 1.0, n)
    elif kind == 'bimodal':
        a = np.exp(-40.0 * (x - 0.12) ** 2) + np.exp(-40.0 * (x - 0.88) ** 2) + 0.02
    elif kind == 'bell':
        a = np.exp(-10.0 * (x - 0.5) ** 2) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


_LP_CACHE = {}


def build_lp_indices(n):
    if n in _LP_CACHE:
        return _LP_CACHE[n]
    n_rows = 2 * n - 1
    k = np.arange(n_rows, dtype=np.int64)
    i_start = np.maximum(k - n + 1, 0)
    i_end = np.minimum(k, n - 1)
    row_lengths = i_end - i_start + 1
    total_nz = int(row_lengths.sum())
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    pos = 0
    for kk in range(n_rows):
        ls = int(i_start[kk])
        le = int(i_end[kk])
        ln = le - ls + 1
        rows[pos:pos + ln] = kk
        cols[pos:pos + ln] = np.arange(ls, le + 1, dtype=np.int32)
        pos += ln
    _LP_CACHE[n] = (rows, cols)
    return rows, cols


def solve_fw_lp(a, M, rows, cols, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    idx = (rows.astype(np.intp) - cols.astype(np.intp))
    vals = np.asarray(a)[idx]
    A_mat = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [A_mat @ b_var <= M * 1.0000001])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, 0.0
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, 0.0
    lp_opt = float(prob.value) if prob.value is not None else 0.0
    if b_var.value is None:
        return None, lp_opt
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, lp_opt
    return b_val / s, lp_opt


def golden_section_J(ac_a, ac_ab, ac_b):
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)

    def Jv(tau):
        C = (1.0 - tau) * (1.0 - tau) * ac_a + 2.0 * tau * (1.0 - tau) * ac_ab + tau * tau * ac_b
        return float(C.max())

    f1 = Jv(x1)
    f2 = Jv(x2)
    for _ in range(50):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = Jv(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = Jv(x2)
    return (lo + hi) / 2.0


def run_fw(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = objective_value(a)
    rows, cols = build_lp_indices(n)
    t0 = time.monotonic()
    iteration = 0
    print(f"    FW start n={n} J={best_J:.6f}")

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        step_start = time.monotonic()

        ac_a = fast_autocorr(a)
        M = float(ac_a.max())
        if M < 1e-18:
            break

        b, lp_opt = solve_fw_lp(a, M, rows, cols, solver)
        if b is None:
            break

        # Stationary if LP can't beat the current point (value ~ 1).
        if lp_opt <= 1.0 + 1e-7:
            break

        ac_ab = conv_two(a, b)
        ac_b = fast_autocorr(b)
        tau = golden_section_J(ac_a, ac_ab, ac_b)

        a_new = (1.0 - tau) * a + tau * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        J_new = objective_value(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

        if verbose and (iteration % 5 == 0 or time.monotonic() - step_start > 3.0):
            el = time.monotonic() - t0
            st = time.monotonic() - step_start
            print(f"      FW it={iteration} ({el:.0f}s, {st:.1f}s/it) J={best_J:.6f} tau={tau:.3f} lp={lp_opt:.4f}")

    print(f"    FW done n={n} iter={iteration} J={best_J:.6f}")
    return best_a, best_J


def compute_soft_subgradient(y, ac, top_k, beta):
    n = y.size
    k_max = min(top_k, ac.size)
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        grad[idx] += 2.0 * wi * y[k_int - idx]
    return grad


def amd_optimize(a0, time_limit, top_k=60, beta=5.0, c0=4.0, restart_every=6000, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_J = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_k = top_k
    cur_beta = beta

    for t in range(1, 50_000_000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else x.copy()

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, cur_k, cur_beta)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        a_new = a_new / sn if sn > 1e-15 else y.copy()

        x_prev = x
        x = a_new

        m_new = fast_autocorr(x).max()
        J_new = 2.0 * n * m_new
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_m = m_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_k = min(cur_k + 10, 120)
            cur_beta = max(cur_beta - 0.4, 2.5)

    return best_a, best_J


def coord_polish(a0, time_limit, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(16, ac.size)
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            sv = a[src]
            if sv < 1e-12:
                continue

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            cand = np.where(~in_peak)[0]
            if cand.size == 0:
                cand = np.arange(n)
            # score: how much mass would interacting with top lags add
            scores = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                d1 = k2i - np.arange(n)
                d2 = np.arange(n) - k2i
                m1 = (d1 >= 0) & (d1 < n)
                m2 = (d2 >= 0) & (d2 < n)
                scores[m1] += a[np.clip(d1, 0, n - 1)][m1]
                scores[m2] += a[np.clip(d2, 0, n - 1)][m2]
            cs = scores[cand]
            dst = cand[np.argmin(cs)] if cs.size else int(rng.integers(n))
            if dst == src:
                continue

            for frac in (0.02, 0.05, 0.08, 0.12, 0.15, 0.25):
                amt = sv * frac
                if amt < 1e-15:
                    continue
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at = at / st
                Jt = objective_value(at)
                if Jt < best_J - 1e-14:
                    a = at
                    best_J = Jt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            a_p = a.copy()
            if i != j and a_p[i] > 1e-10:
                amt = a_p[i] * rng.uniform(0.005, 0.03)
                a_p[i] -= amt
                a_p[j] += amt
                a_p = np.maximum(a_p, 0.0)
                sp = a_p.sum()
                if sp > 1e-15:
                    a_p = a_p / sp
                    Jp = objective_value(a_p)
                    if Jp < best_J - 1e-14:
                        a = a_p
                        best_J = Jp
                        best_a = a.copy()
                        stall = 0
                    else:
                        a = a_p  # accept walk to escape basin
                        best_a = a.copy() if objective_value(a_p) < best_J else best_a

        if stall % 100 == 0:
            print(f"      coord ({time.monotonic()-t0:.0f}s) J={best_J:.6f} stall={stall}")

    print(f"    coord final J={best_J:.6f}")
    return best_a, best_J


def propose_candidate():
    t_global = time.monotonic()
    TIME_BUDGET = 1740.0
    solver = get_solver()
    print(f"Total time budget: {TIME_BUDGET:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(aic, label):
        nonlocal best_seq, best_score
        if aic is None:
            return
        a = np.maximum(np.asarray(aic, dtype=np.float64), 0.0)
        if a.sum() < 1e-12:
            return
        a = a / a.sum()
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}] {sc:.6f} (n={a.size} t={time.monotonic()-t_global:.0f}s)")

    # ---- load parent ----
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() > 1e-10:
                parent = p.copy()
    except Exception:
        parent = None

    if parent is not None:
        ps = evaluate_safe(norm_sum1(parent))
        best_seq = norm_sum1(parent)
        best_score = ps
        print(f"Parent: n={parent.size} score={best_score:.6f}")
    else:
        best_seq = structural_init(3200, 'sin2')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={best_seq.size} score={best_score:.6f}")

    n_parent = len(best_seq)

    # ---- lengths to explore ----
    candidate_lengths = []
    for L in (n_parent, 3600, 4000, 4400):
        if L not in candidate_lengths:
            candidate_lengths.append(L)
    candidate_lengths = [L for L in candidate_lengths if 1500 <= L <= 4600]

    # ---- Phase A: FW on each length (heaviest time) ----
    print("\n=== Phase A: full-constraint FW ===")
    for L in candidate_lengths:
        if time.monotonic() - t_global > TIME_BUDGET - 240:
            break
        remaining = TIME_BUDGET - (time.monotonic() - t_global)
        t_alloc = min(remaining - 120.0, 600.0)
        if t_alloc < 90:
            break
        # init: interpolated parent if big enough else structural
        if parent is not None and parent.size >= 100:
            init = interpolate_sequence(norm_sum1(parent), L)
        else:
            init = structural_init(L, 'sin2')
        # nudge to avoid exact symmetry that stalls FW
        rng = np.random.default_rng(L)
        init = init + 1e-4 * rng.random(L)
        init = norm_sum1(init)
        print(f"  -> FW length {L} (alloc {t_alloc:.0f}s)")
        a_out, j_out = run_fw(init, t_alloc, solver, verbose=True)
        try_update(a_out, f"fw_n{L}")

    # ---- Phase B: AMD polish on overall best ----
    print("\n=== Phase B: AMD polish ===")
    n_best = len(best_seq)
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    t_b = min(remaining - 120.0, 350.0)
    if t_b > 60 and n_best <= 4600:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t_b, top_k=70, beta=4.5, c0=3.5,
                                restart_every=8000, rng_seed=7)
        try_update(a_out, f"amd_n{n_best}")

    # ---- Phase C: coordinate polish ----
    print("\n=== Phase C: coordinate polish ===")
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    t_c = min(remaining - 60.0, 220.0)
    if t_c > 40:
        n_best = len(best_seq)
        a_out, _ = coord_polish(norm_sum1(best_seq), t_c, rng_seed=999)
        try_update(a_out, f"coord_n{n_best}")

    # ---- Phase D: short extra FW on best (structural nudge) if time permits ----
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    if remaining > 120 and len(best_seq) <= 4000:
        print("\n=== Phase D: extra FW on best ===")
        rng = np.random.default_rng(2024)
        a_in = best_seq.copy() + 0.01 * rng.standard_normal(best_seq.size)
        a_in = norm_sum1(a_in)
        a_out, _ = run_fw(a_in, min(remaining - 60.0, 150.0), solver, verbose=True)
        try_update(a_out, f"fw_extra_n{len(a_in)}")

    # ---- final validation / fallback ----
    if best_seq is None:
        best_seq = structural_init(3200, 'sin2')
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    print(f"\nFINAL score={best_score:.6f} n={best_seq.size} time={time.monotonic()-t_global:.0f}s")
    return [float(x) for x in best_seq]
