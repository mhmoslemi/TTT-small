"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with Nesterov acceleration (primary method)
2. Multiplicative weight descent with adaptive soft subgradient
3. Coordinate descent polishing targeting peak lags
4. Multi-length exploration with interpolation
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_local(a):
    """Fast local evaluation (assumes sum=1, nonneg)."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgradient(a, ac, top_k=20, beta=3.0):
    """Compute soft subgradient focusing on top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        # d(conv[k])/da[i] = a[k-i] (for i in range)
        # But we need to handle the case where i = k-i (i.e., 2i=k)
        # For the convolution sum, the gradient is simply a[k-i] for each valid i
        grad[idx] += wi * a[j_idx]
    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=500, nesterov=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    prev_y = a.copy()
    iter_count = 0
    best_lp_time = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Build constraint matrix (sparse)
        # T[k, i] = y[k-i] if 0 <= k-i < n
        t_start = time.monotonic()
        
        # Use top-K constraints for efficiency
        K_sel = min(top_K, 2 * n - 1)
        # Select most binding constraints
        binding_idx = np.argsort(ac)[-K_sel:]  # top K by value
        # Actually we want constraints closest to being violated, i.e., high values
        # But for Frank-Wolfe, we want all constraints: T @ b <= M_curr
        # Using all constraints is most accurate but slowest
        
        # For speed, use a subset. Let's use the top K by value.
        T_sel = np.zeros((K_sel, n))
        for j in range(n):
            for k_offset in range(K_sel):
                k = binding_idx[k_offset]
                j2 = k - j
                if 0 <= j2 < n:
                    T_sel[k_offset, j] = y[j2]
        
        b_var = cp.Variable(n, nonneg=True)
        M_vec = M_curr * np.ones(K_sel)
        constraints = [T_sel @ b_var <= M_vec]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if nesterov:
            if t == 1:
                x_new = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
            prev_y = y
            y = y_new
        else:
            x_new = y_new
            y = y_new
        
        iter_count += 1
        lp_time = time.monotonic() - t_start
        
        for candidate in [y, x_new]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m:
                best_m = mc
                best_a = candidate.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s, lp_t={lp_time:.2f}s): J={J:.6f}")
    
    return best_a, best_m

        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += wi * a[j_idx]
    return grad


def build_toeplitz_rows(y, k_indices):
    n = len(y)
    K = len(k_indices)
    T = np.zeros((K, n))
    for row in range(K):
        k_int = int(k_indices[row])
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            T[row, j_start:j_end] = y[k_int - js]
    return T


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=800, nesterov=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    iter_count = 0
    total_lp_time = 0.0
    stall = 0
    stall_limit = 200

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(ac)[-K_sel:]

        t_start = time.monotonic()
        T_sel = build_toeplitz_rows(y, binding_idx)
        b_var = cp.Variable(n, nonneg=True)
        M_vec = M_curr * np.ones(K_sel)
        constraints = [T_sel @ b_var <= M_vec]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            total_lp_time += time.monotonic() - t_start
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            total_lp_time += time.monotonic() - t_start
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        total_lp_time += time.monotonic() - t_start
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if nesterov:
            if t == 1:
                x_new = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
            y = y_new
        else:
            x_new = y_new
            y = y_new

        iter_count += 1

        improved = False
        mc_y = fast_autocorr(y).max()
        if mc_y < best_m:
            best_m = mc_y
            best_a = y.copy()
            improved = True
        if nesterov:
            mc_x = fast_autocorr(x_new).max()
            if mc_x < best_m:
                best_m = mc_x
                best_a = x_new.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                stall = 0
                y = best_a.copy()
                top_K = min(top_K + 200, 2 * n - 1)

        if verbose and (t % 20 == 0 or t <= 3):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            avg_lt = total_lp_time / max(iter_count, 1)
            print(f"    LP iter={t} ({elapsed:.0f}s, avg_lp={avg_lt:.2f}s): J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LP done: {iter_count} iters, {elapsed:.0f}s, J={J:.6f}")
    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=30, beta=4.0, c0=3.0, restart_every=6000, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.5, 2.0)
            gamma = min(gamma * 1.2, 8.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved_this_round = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:12]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        scores[d] += a[d1]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:12]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                peak_mask[s2:e2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            best_dst = candidates[np.argmin(scores[candidates])]
            if best_dst == src:
                continue

            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and (improved_this_round or stall % 50 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_uniform(n):
    return np.ones(n) / n


def make_linear_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_bimodal(n, center_sep=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-50 * (x - 0.1) ** 2) + np.exp(-50 * (x - 0.9) ** 2) + 0.01
    return norm_sum1(a)


def make_random_phase(n, seed=0):
    rng = np.random.default_rng(seed)
    nfft = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, nfft)
    phases[0] = 0.0
    spec = np.exp(1j * phases)
    if n % 2 == 0:
        if nfft > 1:
            spec[1:nfft//2] = np.conj(spec[nfft//2:nfft-1:-1]) if nfft > 2 else spec[1:nfft//2]
    a = np.fft.irfft(spec, n)
    a = a - a.min() + 0.05
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 7]:
        coeff = rng.normal(0, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_logistic(n, seed=0):
    rng = np.random.default_rng(seed)
    x = rng.uniform(0.2, 0.8)
    a = np.zeros(n)
    for i in range(n):
        x = 4.0 * x * (1.0 - x)
        a[i] = x
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_random_phase(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP Frank-Wolfe on parent ===
    print(f"\n=== Phase 1: LP-FW (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True, top_K=1000, nesterov=True)
        try_update(a_out, f"lp1_n{n0}")

    # === Phase 2: AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: AMD polish (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t2, top_k=40, beta=4.0, c0=3.5,
                                 restart_every=8000, verbose=False)
        try_update(a_out, f"amd_n{len(a_in)}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord descent (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(a_in)}")

    # === Phase 4: Try diverse initializations at same length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Diverse inits (n={n0}) ===")
    if remaining > 120:
        inits = [
            ("rand_phase", make_random_phase(n0, seed=99)),
            ("smooth", make_smooth_random(n0, seed=13)),
            ("logistic", make_logistic(n0, seed=7)),
            ("bimodal", make_bimodal(n0)),
        ]
        if parent is not None:
            inits.append(("blend_parent", norm_sum1(0.5 * norm_sum1(parent) + 0.5 * make_random_phase(n0, seed=5))))
        
        t_per = min(30.0, max(10.0, remaining / (len(inits) * 1.5)))
        for lbl, init in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 60)
            if t_avail < 10:
                break
            print(f"  Trying init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=30, beta=4.0, c0=3.0,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"div_{lbl}")

    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 400, n_best + 800, n_best + 1200]
    longer_lengths = [l for l in longer_lengths if l <= 4500]
    
    print(f"\n=== Phase 5: Longer sequences ===")
    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Scaling to n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.25, 150.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=40, beta=4.0, c0=3.0,
                                     restart_every=6000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")
        
        # Short LP on longer sequence
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 3500:
            t5b = min(remaining * 0.20, 100.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=False, top_K=600, nesterov=True)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 3800:
        t6 = min(remaining * 0.30, 180.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=1200, nesterov=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.40, 120.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: One more round of LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 3500:
        print(f"\n=== Phase 8: Extra LP (n={n_best}) ===")
        t8 = min(remaining * 0.60, 100.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=1500, nesterov=True)
            try_update(a_out, f"lp_extra_n{n_best}")

    # === Final AMD sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t_final = min(remaining * 0.5, 60.0)
        if t_final > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t_final, top_k=50, beta=3.0, c0=2.5,
                                     restart_every=2000, verbose=False)
            try_update(a_out, "final_amd")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_random_phase(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
