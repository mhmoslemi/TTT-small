    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_exp
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            x_restarted = best_a + rng.normal(0, 0.03, n) * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def paired_coordinate_descent(a0, time_limit, verbose=True, rng_seed=123,
                              max_rounds=2000):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        peak_idx = np.argmax(ac)
        k_int = int(peak_idx)

        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        idx = np.arange(i_start, i_end + 1)
        contrib = np.zeros(len(idx))
        for ii, j in enumerate(idx):
            partner = k_int - j
            if 0 <= partner < n:
                contrib[ii] = a[j] * a[partner]

        src_candidates = idx[np.argsort(contrib)[::-1][:10]]

        top_peaks = np.argsort(ac)[::-1][:15]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            for pos in range(n):
                d1 = pk_int - pos
                d2 = pos - pk_int
                if 0 <= d1 < n:
                    sensitivity[pos] += a[d1]
                if 0 <= d2 < n:
                    sensitivity[pos] += a[d2]

        improved = False
        for src in src_candidates[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            dst_candidates = np.argsort(sensitivity)[:max(30, n // 20)]
            found_move = False
            for dst in dst_candidates:
                if dst == src or found_move:
                    continue
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[max(0, dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        stall = 0
                        break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if stall > max_stall:
            if verbose:
                J = 2.0 * n * best_m
                print(f"    PairCD stalled at round {rnd}, J={J:.6f}")
            break

        if verbose and rnd % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairCD round={rnd} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_guided_update(a0, time_limit, verbose=False, max_iters=50000):
    """Use DFT structure to guide updates towards flat autocorrelation."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    # Use a longer DFT to capture linear convolution spectrum
    N = 2 * n - 1
    N_next = 1 << (N - 1).bit_length()

    x = a.copy()
    x_prev = a.copy()
    gamma = 2.0
    stall = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute DFT and target: flatten |A|^2 away from DC
        A = np.fft.rfft(y, N_next)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target_avg = total_power / len(power)

        # Gradient: push power away from peaks
        # In time domain, this is the subgradient of max autocorrelation
        ac = fast_autocorr(y)
        max_ac = ac.max()
        # Soft max gradient
        temp = max_ac * 0.1 + 1e-10
        softmax_weights = np.exp(ac / temp)
        softmax_weights = softmax_weights / softmax_weights.sum()

        # Convert to time-domain gradient
        grad = np.zeros(n)
        for k in range(len(ac)):
            wk = softmax_weights[k]
            if wk < 1e-10:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wk * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            x = best_a + rng.normal(0, 0.02, n) * best_a
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase search: LP-FW with adaptive relaxation, spectral-guided
    subgradient descent, paired coordinate descent, and multi-n exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: LP-FW (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 1: LP-FW n={len(best_seq)} ===")
        t1 = min(remaining * 0.35, 600.0)
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t1, solver,
                                  verbose=True, base_relax=1.003,
                                  relax_decay=300.0, restart_every=500,
                                  perturb_scale=0.008)
        try_update(a_out, f"lp1_n{len(a_out)}")

    # === Phase 2: Spectral-guided subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 2: Spectral descent n={len(best_seq)} ===")
        t2 = min(remaining * 0.20, 300.0)
        a_out, _ = spectral_guided_update(normalize(best_seq), t2, verbose=False)
        try_update(a_out, f"spectral_n{len(a_out)}")

    # === Phase 3: LP-FW round 2 (with different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 3: LP-FW round 2 n={len(best_seq)} ===")
        t3 = min(remaining * 0.30, 400.0)
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t3, solver,
                                  verbose=True, base_relax=1.0,
                                  relax_decay=100.0, restart_every=600,
                                  perturb_scale=0.005)
        try_update(a_out, f"lp2_n{len(a_out)}")

    # === Phase 4: Paired coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 4: PairCD n={len(best_seq)} ===")
        t4 = min(remaining * 0.15, 200.0)
        a_out, _ = paired_coordinate_descent(normalize(best_seq), t4,
                                             verbose=True, rng_seed=42)
        try_update(a_out, f"paircd_n{len(a_out)}")

    # === Phase 5: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_target = n_cur + 400
    if remaining > 150 and n_target <= 4000:
        print(f"\n=== Phase 5: Expand to n={n_target} ===")
        a_long = interpolate_sequence(best_seq, n_target)
        t5a = min(remaining * 0.25, 250.0)
        a_out, _ = lp_frank_wolfe(normalize(a_long), t5a, solver,
                                  verbose=False, base_relax=1.005,
                                  relax_decay=200.0, restart_every=400,
                                  perturb_scale=0.01)
        try_update(a_out, f"lp_n{n_target}")

    # === Phase 6: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 6: Multi-n exploration ===")
        n_best = len(best_seq)
        test_ns = []
        for delta in [200, 400, -200, 600]:
            nt = n_best + delta
            if 1600 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.25, 150.0)
            a_out, _ = lp_frank_wolfe(normalize(a_inter), t_per, solver,
                                      verbose=False, base_relax=1.005,
                                      relax_decay=200.0, restart_every=400,
                                      perturb_scale=0.01)
            try_update(a_out, f"multin_n{nt}")

    # === Phase 7: Final subgradient + coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final polish n={n_best} ===")
        t7 = min(remaining * 0.4, 200.0)
        a_out, _ = subgradient_peak_descent(normalize(best_seq), t7,
                                            top_k=40, beta_exp=4.0,
                                            gamma=2.5, verbose=False)
        try_update(a_out, f"sgd_final_n{n_best}")

    # === Phase 8: Final LP if time remaining ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final LP n={n_best} ===")
        t8 = min(remaining * 0.7, 150.0)
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t8, solver,
                                  verbose=False, base_relax=1.0,
                                  relax_decay=50.0, restart_every=300,
                                  perturb_scale=0.003)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final coordinate polish with tiny steps
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t9 = min(remaining * 0.5, 60.0)
        a_out, _ = paired_coordinate_descent(normalize(best_seq), t9,
                                             verbose=False, rng_seed=314)
        try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
