    # Therefore corr[j] = (flip_a * p)[j+n-1]
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()  # need at least 3n-2
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    p_pad = np.zeros(fft_size)
    p_pad[:2*n-1] = p
    A_f = np.fft.rfft(a_pad)
    P_f = np.fft.rfft(p_pad)
    # cross-correlation: corr[j] = sum_m a[m]*p[j+m]
    # = full convolution of reverse(a) with p, evaluated at appropriate index
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    AR_f = np.fft.rfft(a_rev)
    conv = np.fft.irfft(AR_f * P_f, fft_size)
    # conv[k] = sum_m a_rev[m] * p[k-m] = sum_m a[n-1-m] * p[k-m]
    # Let l = n-1-m: conv[k] = sum_l a[l] * p[k-n+1+l]
    # We want corr[j] = sum_m a[m] * p[j+m], so corr[j] = conv[j+n-1]
    corr = conv[n-1 : n-1 + 2*n - 1]  # j = 0, ..., 2n-2
    # But we need it truncated: for j > 2n-2-n+1 = n-1, the sum should be truncated
    # Actually the full cross-correlation naturally handles this (p[j+m] = 0 for j+m >= 2n-1)
    
    # Second term: v[j] = sum_k p[k] * a[j-k] * 1_{0<=j-k<n}
    # For fixed j, k ranges from max(0, j-n+1) to j
    # Let m = j-k: v[j] = sum_{m=0}^{min(j, n-1)} p[j-m] * a[m]
    # = sum_m a[m] * p[j-m] = (a * p)[j] (standard convolution!)
    conv2 = np.fft.irfft(A_f * P_f, fft_size)
    v = conv2[:n]  # j = 0, ..., n-1
    
    # Wait, I need to be more careful about the bounds.
    # v[j] = sum_{m=0}^{min(j, n-1)} a[m] * p[j-m]
    # Standard convolution (a*p)[j] = sum_m a[m] * p[j-m] for all m where both are in range
    # Since a has length n and p has length 2n-1, (a*p)[j] for j=0..n-1
    # automatically has m from 0 to min(j, n-1) since a[m] = 0 for m >= n
    # and p[j-m] = 0 for j-m < 0 or j-m >= 2n-1
    # For j < n: j-m >= 0 requires m <= j, and j-m <= 2n-2 always holds since j < n < 2n-1
    # So v[j] = (a*p)[j] for j = 0, ..., n-1. 
    
    # For the first term, corr[j] = sum_m a[m] * p[j+m]
    # For j = 0, ..., n-1: j+m ranges from j to j+n-1, all < 2n-1 (since j < n), so no truncation needed
    # For j >= n: j+m can exceed 2n-2, so p[j+m] = 0 naturally. The FFT handles this.
    # But we only need grad for j = 0, ..., n-1.
    
    # So for j = 0, ..., n-1:
    # grad[j] = corr[j] + v[j]
    
    grad = corr[:n] + v[:n]
    return grad

def softmax_gradient_descent(a0, time_limit, beta_start=10.0, beta_end=200.0, 
                              max_iters=50000, verbose=True, seed=42):
    """Nesterov-accelerated gradient descent on soft-max approximation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    
    # Adaptive beta schedule
    n_betas = 5
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        print(f"    SGrad beta={beta:.1f}, current M={best_m:.8f}")
        
        # Reset momentum for each beta
        x = best_a.copy()
        x_prev = best_a.copy()
        
        for t in range(1, max_iters + 1):
            if time.monotonic() - t0 > time_limit:
                break
            
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
            
            # Compute gradient
            grad = compute_grad_softmax(y, beta)
            
            # Also add a small gradient for the sum constraint (projection)
            # We'll use a penalty: add lambda * (sum(y) - 1) to the gradient
            # Or just renormalize after the step
            
            # Step size: for convex f, eta ~ 1/sqrt(t) works, but we want it adaptive
            # The gradient magnitude gives us a scale
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-15:
                break
            eta = 2.0 / (g_norm * (np.sqrt(t) + 1.0))
            
            # Mirror descent step (exponential family / entropic regularization)
            log_y = np.log(y) - eta * grad
            max_log = log_y.max()
            a_new = np.exp(log_y - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            
            x_prev = x
            x = a_new
            
            M_new = fast_autocorr(x).max()
            if M_new < best_m:
                best_m = M_new
                best_a = x.copy()
            
            if verbose and t % 1000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, verbose=True, solver_name=None):
    """LP-based Frank-Wolfe with adaptive step size."""
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if solver_name is None:
        solver_name = _get_cvxpy_solver()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix T where (T @ b)[k] = (y*b)[k]
        # T is (2n-1) x n, T[k, j] = y[j] if 0 <= k-j < n, else 0
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = y[j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m


def _get_cvxpy_solver():
    global _solver_cache
    if '_solver_cache' not in globals():
        _solver_cache = None
    if _solver_cache is not None:
        return _solver_cache
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    return _solver_cache


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Targeted coordinate descent on peak-contributing positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = int(np.argmax(ac))
        
        # Find positions contributing to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        
        if contrib.max() < 1e-25:
            stall += 1
            break
        
        # Compute sensitivity: how much each position contributes to top peaks
        top_peaks = np.argsort(ac)[::-1][:min(20, len(ac))]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk = int(pk)
            sjs = max(0, pk - n + 1)
            sje = min(n - 1, pk)
            s_idx = np.arange(sjs, sje + 1)
            sensitivity[s_idx] += a[pk - s_idx]
            sensitivity[pk - s_idx] += a[s_idx]
        
        improved = False
        # Try moving mass from high-contrib to low-sensitivity positions
        src_order = np.argsort(contrib)[::-1][:10]
        dst_order = np.argsort(sensitivity)[:max(100, n // 5)]
        
        for si in src_order:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[si])
            if a[src] < 1e-12:
                continue
            for di in dst_order:
                dst = int(di)
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial /= st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = min(a[i] * frac, a[i] * 0.5)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a /= st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def solve_qp_direct(a0, time_limit, verbose=True):
    """Solve the direct convex QP: min M s.t. a*a[k] <= M, sum(a)=1, a>=0."""
    import cvxpy as cp
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    t0 = time.monotonic()
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Objective: minimize M
    # Constraints: (a*a)[k] <= M for all k, sum(a) = 1
    # (a*a)[k] = sum_j a[j]*a[k-j] which is a quadratic form
    
    # For the full convolution, c[k] = a^T Q_k a where Q_k[j,l] = 1/2 if l-j = k or j-l = k
    # This is a PSD quadratic form.
    
    # To keep the problem manageable, only include constraints for the top peaks
    c = fast_autocorr(a_init)
    top_k = min(50, len(c))
    active_peaks = np.argsort(c)[::-1][:top_k]
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_peaks:
        k = int(k)
        # c[k] = sum_j a[j]*a[k-j]
        # Build the quadratic form matrix (sparse)
        # Q[k][j, k-j] = 1/2 for valid j and k-j (both in [0, n-1])
        # Q[k][k-j, j] = 1/2
        # So Q[k] has nonzeros on the k-th superdiagonal and k-th subdiagonal
        
        # For cvxpy, we can express this as:
        # sum_j a[j] * a[k-j] <= M
        # This is a quadratic constraint. cvxpy can handle it with cp.quad_form
        # But we need the matrix. Let's use a different approach.
        
        # Actually, cp.quad_form(a_var, Q) where Q is the appropriate matrix
        # But building full n x n matrices for n=2400 is expensive.
        
        # Alternative: express as a sum of products
        # sum_j a[j]*a[k-j] <= M
        # This is a concave-in-M, convex-in-a constraint (since it's a^T Q a with Q PSD)
        # cvxpy should handle this.
        
        pass  # Too complex to implement efficiently for large n
    
    # Skip direct QP for now - too complex for large n
    # Instead, use the LP-FW which is more practical
    if verbose:
        print(f"    QP: skipped (too complex for n={n})")
    return a_init, fast_autocorr(a_init).max()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diverse_initializations(n, num_inits=5, seed=0):
    """Generate diverse initial sequences for restart."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # 1. Parent-like (use global)
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_interp = interpolate_sequence(p, n)
            inits.append(p_interp)
    except Exception:
        pass
    
    # 2. Sinusoidal
    x = np.linspace(0, 1, n)
    inits.append(np.sin(np.pi * x) ** 2)
    
    # 3. Gaussian
    center = n // 2
    sigma = n / 4
    g = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(g)
    
    # 4. Linear ramp
    inits.append(np.linspace(0.1, 1.0, n))
    
    # 5. Random
    r = rng.random(n) + 0.1
    inits.append(r)
    
    # 6. Cosine with different frequency
    inits.append((1 + np.cos(4 * np.pi * x)) / 2)
    
    # 7. Double-sinusoid
    inits.append(np.sin(np.pi * x) ** 2 * (0.5 + 0.5 * np.cos(2 * np.pi * x)))
    
    # Normalize all
    normalized = []
    for a in inits:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            normalized.append(a / s)
    return normalized


def propose_candidate(budget_s=1800.0):
    """
    Main search: combines soft-max gradient descent, LP-FW, and coordinate polish
    with multiple restarts and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")
    
    solver_name = _get_cvxpy_solver()
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    parent_n = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={parent_n}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"No parent available: {e}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq /= best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Strategy: Multiple rounds at the best length found, with different methods ===
    
    # Phase 1: Soft-max gradient descent (good for escaping local minima)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 1: Soft-max GD n={n0} ===")
        t1 = min(remaining * 0.2, 200.0)
        a_out, _ = softmax_gradient_descent(norm_sum1_arr(best_seq), t1, 
                                            beta_start=5.0, beta_end=100.0,
                                            max_iters=10000, verbose=True, seed=42)
        try_update(a_out, "sgrad")
    
    # Phase 2: LP-FW (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW n={n_cur} ===")
        t2 = min(remaining * 0.3, 400.0)
        a_out, _ = lp_frank_wolfe(best_seq, t2, verbose=True, solver_name=solver_name)
        try_update(a_out, f"lp_n{n_cur}")
    
    # Phase 3: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord polish n={n_cur} ===")
        t3 = min(remaining * 0.15, 150.0)
        a_out, _ = coordinate_polish(best_seq, t3, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_cur}")
    
    # Phase 4: Another round of LP-FW (warm-started from improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.25, 300.0)
        a_out, _ = lp_frank_wolfe(best_seq, t4, verbose=True, solver_name=solver_name)
        try_update(a_out, f"lp2_n{n_cur}")
    
    # Phase 5: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:

        test_ns = []
        for delta in [-200, -400, 200, 400, 600]:

            if 800 <= nt <= 4000 and nt != n_best:

        for nt in test_ns[:4]:

            if remaining < 50:
                break

            a_inter = interpolate_sequence(best_seq, nt)
            # Quick optimization at new length
            t_per = min(remaining * 0.2, 150.0)
            a_out, _ = lp_frank_wolfe(a_inter, t_per, verbose=False, solver_name=solver_name)
