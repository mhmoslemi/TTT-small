"""
Minimizes 2*n*max(np.convolve(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with softmax-weighted subgradient from top-k active lags
2. Nesterov-accelerated mirror descent with adaptive temperature
3. Multi-restart at different sequence lengths
4. Coordinate descent polish
Key insight: max of np.convolve(a,a) is NOT at center for non-palindromic a,
allowing J < 2. We optimize the full convolution profile via subgradient methods.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, conv, top_k=100, beta=4.0):
    """Compute subgradient using top-k lags with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(conv))

    if len(conv) <= k_max:
        idx = np.arange(len(conv))
        vals = conv[idx].copy()
    else:
        part = np.argpartition(conv, -k_max)[-k_max:]
        idx = np.sort(part)
        vals = conv[idx].copy()

    vals_clipped = np.maximum(vals, 1e-30)
    log_w = beta * np.log(vals_clipped)
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        a_indices = k_int - np.arange(j_start, j_end)
        g[j_start:j_end] += 2.0 * wi * a[a_indices]

    return g


def frank_wolfe_solve(a0, time_limit, top_k=100, beta=4.0, verbose=True):
    """Frank-Wolfe with softmax subgradient and Chebyshev step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        if current_max < best_max - 1e-15:
            best_max = current_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        g = compute_subgradient(a, conv, top_k=top_k, beta=beta)

        j = int(np.argmin(g))
        v = np.zeros(n)
        v[j] = 1.0

        gv = g[j]
        ga = float(g @ a)
        diff = a - v
        denom = float(diff @ diff)
        if denom < 1e-30:
            break
        alpha = (ga - gv) / denom
        alpha = min(max(alpha, 0.0), 1.0)

        a_new = (1.0 - alpha) * a + alpha * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print(f"    FW t={t} ({elapsed:.1f}s): J={J:.6f}, max_conv={best_max:.8f}")

    return best_a, best_max


def mirror_descent_solve(a0, time_limit, top_k=100, initial_lr=0.5, verbose=True):
    """Nesterov mirror descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_max = fast_convolve(a).max()

    x = a.copy()
    x_prev = a.copy()
    lr = initial_lr
    stall = 0
    max_stall = 3000
    rng = np.random.default_rng(777)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 0.5) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_convolve(y)
        g = compute_subgradient(y, conv, top_k=top_k, beta=4.0)

        eta = lr / (np.sqrt(t) + 1.0)

        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        m = fast_convolve(x).max()
        if m < best_max - 1e-15:
            best_max = m
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = max(500, max_stall // 2)
            lr = max(lr * 0.5, 0.01)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r

        # Also check ergodic average periodically
        if t % 2000 == 0:
            avg = (x + best_a) / 2.0
            m_avg = fast_convolve(avg).max()
            if m_avg < best_max:
                best_max = m_avg
                best_a = avg.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, lr={lr:.4f}")

    return best_a, best_max


def cd_polish(a0, time_limit, verbose=True):
    """Coordinate descent: try transferring mass between coordinates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 5

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        coord_order = np.arange(n)
        rng.shuffle(coord_order)

        for i in coord_order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            n_targets = min(6, n)
            targets = rng.choice(n, size=n_targets, replace=False)

            best_trial = None
            best_trial_max = best_max

            for dst in targets:
                if dst == i:
                    continue
                for frac in [0.005, 0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m = fast_convolve(a_trial).max()
                    if m < best_trial_max - 1e-14:
                        best_trial_max = m
                        best_trial = a_trial.copy()

            if best_trial is not None:
                a = best_trial
                best_max = best_trial_max
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_convolve(a).max()
                if m_new < best_max:
                    best_max = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and stall % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_max


def paired_swap_solve(a0, time_limit, verbose=True):
    """Paired swap: simultaneously adjust two coordinates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()

    rng = np.random.default_rng(456)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(100, n // 3)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if a[i] < 1e-12 and a[j] < 1e-12:
                continue

            for direction in [1, -1]:
                src = i if direction == 1 else j
                dst = j if direction == 1 else i
                if a[src] < 1e-12:
                    continue
                for frac in [0.003, 0.01, 0.03, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m = fast_convolve(a_trial).max()
                    if m < best_max - 1e-14:
                        a = a_trial
                        best_max = m
                        best_a = a.copy()
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print(f"    PS ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_max


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
