def log_param_optimize(a0, time_limit, verbose=True):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 2400
        x = np.linspace(0, 1, n_init)
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 2 + 0.3 * (1 + np.cos(4 * np.pi * x)) ** 2
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.8f}")
    
    # === Phase 1: L-BFGS on parent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS main (n={n0}) ===")
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 30:
        a_out, _ = lgbfgs_smoothed_a0(best_seq, t1, verbose=True)
        try_update(a_out, f"lgbfgs_main_n{n0}")
    
    # === Phase 2: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coord descent (n={n_best}) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        a_out, _ = coordinate_descent_polish(best_seq, t2, verbose=True, rng_seed=1)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 3: L-BFGS again from improved point ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: L-BFGS round 2 (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = lgbfgs_smoothed_a0(best_seq, t3, verbose=True)
        try_update(a_out, f"lgbfgs_r2_n{n_best}")
    
    # === Phase 4: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = coordinate_descent_polish(best_seq, t4, verbose=True, rng_seed=2)
        try_update(a_out, f"coord_r2_n{n_best}")
    
    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length exploration ===")
        try_lengths = []
        # Try nearby lengths
        for delta in [200, 400, -200, -400, 600]:
            nl = n_best + delta
            if 500 <= nl <= 4000 and nl != n_best:
                try_lengths.append(nl)
        
        for nl in try_lengths[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={nl}")
            a_interp = interpolate_sequence(best_seq, nl)
            t_ll = min(remaining * 0.30, 150.0)
            if t_ll > 20:
                a_out, _ = lgbfgs_smoothed_a0(a_interp, t_ll, verbose=False)
                try_update(a_out, f"lgbfgs_n{nl}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    t_cd = min(remaining * 0.20, 80.0)
                    if t_cd > 15:
                        a_out2, _ = coordinate_descent_polish(a_out, t_cd, verbose=False, rng_seed=nl)
                        try_update(a_out2, f"coord_n{nl}")
    
    # === Phase 6: Gradient descent with momentum (different landscape) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: GD momentum (n={n_best}) ===")
        t6 = min(remaining * 0.20, 200.0)
        if t6 > 20:
            a_out, _ = gradient_descent_momentum(best_seq, t6, lr0=0.05, 
                                                   momentum=0.9, t_smooth=0.05,
                                                   verbose=True, rng_seed=3)
            try_update(a_out, f"gd_mom_n{n_best}")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
        # Quick coordinate descent
        t7a = min(remaining * 0.40, 120.0)
        if t7a > 15:
            a_out, _ = coordinate_descent_polish(best_seq, t7a, verbose=False, rng_seed=4)
            try_update(a_out, f"coord_final_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        # Quick L-BFGS with very small t
        if remaining > 20:
            t7b = min(remaining * 0.40, 80.0)
            if t7b > 10:
                conv_scale = fast_autocorr(best_seq).max()
                a_out, _ = lgbfgs_smoothed_a0(best_seq, t7b, verbose=False,
                                             t_list=[conv_scale * 0.1, conv_scale * 0.05, conv_scale * 0.02])
                try_update(a_out, f"lgbfgs_final_n{n_best}")
    
    # === Phase 8: Try diverse initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 8: Diverse init ===")
        rng = np.random.default_rng(123)
        n_div = len(best_seq)
        
        # Try a different structured init
        x = np.linspace(0, 1, n_div)
        a_div = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
        a_div = np.maximum(a_div, 0.01)
        a_div = a_div / a_div.sum()
        
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 20:
            a_out, _ = lgbfgs_smoothed_a0(a_div, t8, verbose=False)
            try_update(a_out, f"diverse_lgbfgs_n{n_div}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                t8b = min(remaining * 0.20, 60.0)
                if t8b > 15:
                    a_out2, _ = coordinate_descent_polish(best_seq, t8b, verbose=False, rng_seed=5)
                    try_update(a_out2, f"coord_div_n{n_div}")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
