def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def full_spectrum_subgradient(y, ac, beta=5.0):
    """Compute weighted subgradient using all peaks with exponential weighting."""
    n = len(y)
    # Weight peaks by exp(beta * (ac[k] - ac.max()))
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n)
    weights = np.exp(beta * (ac - ac_max))
    total_w = weights.sum()
    if total_w < 1e-30:
        # Fallback to uniform over top peaks
        top_k = 5
        top_idx = np.argsort(ac)[::-1][:top_k]
        w = np.zeros(len(ac))
        w[top_idx] = 1.0 / top_k
        weights = w
        total_w = 1.0
    else:
        weights = weights / total_w
    
    # Compute subgradient for each active peak and combine
    grad = np.zeros(n)
    # For efficiency, only process peaks with significant weight
    threshold = 1.0 / (n * 10)
    active = np.where(weights > threshold)[0]
    
    for k in active:
        k_int = int(k)
        w = weights[k_int]
        if w < 1e-10:
            continue
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += w * 2.0 * y[j]
    
    return grad

def nesterov_full(a0, time_limit, verbose=True, seed=42, beta=5.0, c0=5.0):
    """Nesterov mirror descent with full-spectrum subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 2000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute convolution and subgradient
        ac = fast_conv(y)
        M_curr = ac.max()
        
        g = full_spectrum_subgradient(y, ac, beta=beta)
        
        # Backtracking line search for step size
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Try the step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        M_new = fast_conv(a_new).max()
        
        # If step didn't help, try smaller step
        if M_new > M_curr:
            for shrink in [0.5, 0.25, 0.1]:
                eta_s = eta * shrink
                log_a_s = np.log(y) - eta_s * g
                max_log_s = log_a_s.max()
                a_s = np.exp(log_a_s - max_log_s)
                a_s = np.maximum(a_s, 1e-30)
                a_s = a_s / a_s.sum()
                M_s = fast_conv(a_s).max()
                if M_s < M_curr:
                    a_new = a_s
                    M_new = M_s
                    break
            else:
                a_new = y.copy()
                M_new = M_curr
        
        x_prev = x
        x = a_new
        
        # Track best
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with perturbation
        if stall > max_stall:
            x_restarted = best_a * (1 + rng.normal(0, 0.01, n))
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NFS t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=True, seed=42):

        # Find the contributor pair with maximum product at k_star

        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        products = a[idx] * a[k_star - idx]
        src = idx[np.argmax(products)]
        
        # Try to move mass from src to a location that minimizes damage
        # Find peaks and avoid them
        top_peaks = np.argsort(ac)[::-1][:10]
        
        # Find best destination
        best_dst = -1
        best_delta = float('inf')
        
        # Sample candidate destinations
        candidates = np.random.choice(n, size=min(2000, n), replace=False)
        for dst in candidates:
            if dst == src:
                continue
            # Estimate: moving mass from src to dst changes ac[k] for various k
            # The change at k_star is: -2*a[src]*a[k_star-src] + a[dst]*a[k_star-dst] + a[k_star-dst]*a[dst]
            # = -2*a[src]*a[k_star-src] + 2*a[dst]*a[k_star-dst] (if dst != k_star-src)
            # But we also affect other peaks
            
            # Simple heuristic: prefer dst where a[dst] is small and k_star-dst is out of range or has small a
            kd = k_star - dst
            if 0 <= kd < n:
                new_contrib = 2.0 * a[dst] * a[kd]
            else:
                new_contrib = 0.0
            old_contrib = 2.0 * a[src] * a[k_star - src] if 0 <= k_star - src < n else 0.0
            delta = new_contrib - old_contrib
            if delta < best_delta:
                best_delta = delta
                best_dst = dst
        
        if best_dst < 0:
            stall += 1
            continue
        
        dst = best_dst

            a_trial = a.copy()

            a_trial[dst] += amt

        J = 2.0 * n * best_m
        print(f"    Peak shaving ({elapsed:.1f}s): J={J:.6f}")

def coordinate_ld(a0, time_limit, verbose=False, seed=123):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def full_spectrum_subgradient(y, ac, beta=5.0):
    """
    Compute weighted subgradient using all peaks with exponential weighting.
    Uses FFT for O(n log n) computation.
    
    g_i = 2 * sum_j weights[i+j] * y[j] for j=0..n-1
    """
    n = len(y)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n)
    
    # Compute weights: exp(beta * (ac - ac_max)) normalized
    weights = np.exp(beta * (ac - ac_max))
    total_w = weights.sum()
    if total_w < 1e-30:
        # Fallback: uniform over top 10 peaks
        top_k = min(10, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        weights = np.zeros(len(ac))
        weights[top_idx] = 1.0 / top_k
    else:
        weights = weights / total_w

    rev_w = weights[::-1]

    Rw = np.fft.rfft(rev_w, fft_size)

    conv = np.fft.irfft(Rw * Yv, fft_size)[:conv_len]

def top_peak_subgradient(y, ac, top_k=5, beta=3.0):
    """Compute subgradient from top-k peaks with power weighting."""
    n = len(y)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += wi * 2.0 * y[j]

    subgrad_mode: 'full' for FFT-based full spectrum, 'top' for top-k peaks
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute convolution and subgradient
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if subgrad_mode == 'full':
            g = full_spectrum_subgradient(y, ac, beta=beta)
        else:
            g = top_peak_subgradient(y, ac, top_k=top_k, beta=3.0)
        
        # Adaptive step size with backtracking
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror descent step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        M_new = fast_conv(a_new).max()
        
        # Backtracking if step was bad
        if M_new > M_curr * 1.001:
            found_step = False
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                eta_s = eta * shrink
                log_a_s = np.log(y) - eta_s * g
                max_log_s = log_a_s.max()
                a_s = np.exp(log_a_s - max_log_s)
                a_s = np.maximum(a_s, 1e-30)
                a_s = a_s / a_s.sum()
                M_s = fast_conv(a_s).max()
                if M_s < M_curr:
                    a_new = a_s
                    M_new = M_s
                    found_step = True
                    break
            if not found_step:
                a_new = y.copy()
                M_new = M_curr
        
        x_prev = x
        x = a_new
        
        # Track best
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with perturbation when stuck
        if stall > max_stall:
            scale = 0.01 + 0.02 * (stall // max_stall)
            x_restarted = best_a * (1.0 + rng.normal(0, scale, n))
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            gamma = max(gamma * 0.8, 1.0)
        
        # Periodic larger jump
        if t % 20000 == 0 and t > 0:
            x_jumped = best_a * (1.0 + rng.normal(0, 0.03, n))
            x_jumped = np.maximum(x_jumped, 1e-8)
            x_jumped = x_jumped / x_jumped.sum()
            M_jump = fast_conv(x_jumped).max()
            if M_jump < best_m:
                best_m = M_jump
                best_a = x_jumped.copy()
                x = x_jumped
                x_prev = x_jumped
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True, seed=42):
    """
    Iteratively identify the highest convolution peak and redistribute mass
    from the contributing indices to reduce it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        k_star = int(np.argmax(ac))
        M_curr = ac[k_star]
        
        # Find the main contributor index at k_star
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        products = a[idx] * a[k_star - idx]
        max_prod_idx = np.argmax(products)
        src = idx[max_prod_idx]
        
        # Find best destination using many random candidates
        best_dst = -1
        best_M = M_curr
        
        n_cand = min(3000, n)
        candidates = rng.integers(0, n, size=n_cand)
        
        for dst in candidates:
            dst = int(dst)
            if dst == src:
                continue
            # Try a small transfer
            for frac in [0.01, 0.005]:
                amt = a[src] * frac
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_dst = dst
                    best_frac = frac
                break  # Only try first frac per dst for speed
        
        if best_dst >= 0 and best_M < best_m - 1e-14:
            amt = a[src] * best_frac
            a[src] -= amt
            a[best_dst] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = best_M
            best_a = a.copy()
            stall = 0
        else:
            # Try random small perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = a / a.sum()
            stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Peak shaving ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_ld(a0, time_limit, verbose=False, seed=123):
    """
    Coordinate descent with binary search line search on (src, dst) pairs.
    Focuses on reducing the top convolution peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        top_peaks = np.argsort(ac)[::-1][:min(8, len(ac))]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            products = a[idx] * a[k_int - idx]
            
            if products.max() < 1e-20:
                continue
            
            src = idx[np.argmax(products)]
            
            # Try multiple random destinations with binary search
            best_move = None
            best_move_M = best_m
            
            for _ in range(30):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                
                # Binary search for optimal fraction in [0, 1]
                lo, hi = 0.0, 0.95
                best_frac = 0.0
                best_local_M = best_move_M
                
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    amt = a[src] * mid
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_local_M:
                        best_local_M = M_trial
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_frac > 0.05 and best_local_M < best_move_M:
                    best_move = (src, dst, best_frac)
                    best_move_M = best_local_M
            
            if best_move is not None:
                src_m, dst_m, frac_m = best_move
                amt = a[src_m] * frac_m
                a[src_m] -= amt
                a[dst_m] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_move_M
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Small random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:
                stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord LD ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length while preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, kind='sin'):
    """Create various structured initializations."""
    x = np.linspace(0, 1, n)
    if kind == 'sin':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'raised_cos':
        a = (1 + np.cos(2 * np.pi * x)) / 2
    elif kind == 'gaussian':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'bump':
        a = np.exp(-20 * (x - 0.5) ** 2)
    else:
        a = np.sin(np.pi * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with adaptive time allocation.
    Uses FFT-accelerated full-spectrum subgradient Nesterov mirror descent
    followed by peak shaving and coordinate line search.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p.copy()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # If no parent, create default
    if best_seq is None:
        n_base = 2400
        best_seq = make_initialization(n_base, 'sin')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Nesterov mirror descent, full spectrum subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: NMD full-spectrum n={n0} (t={t1:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror(a_in, t1, verbose=True, seed=42,
                                   subgrad_mode='full', beta=5.0, c0=5.0)
        try_update(a_out, f"nmd_full_n{n0}")
    
    # === Phase 2: NMD with higher beta (more focused on peak) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        print(f"\n=== Phase 2: NMD beta=12 n={len(best_seq)} (t={t2:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror(a_in, t2, verbose=True, seed=7,
                                   subgrad_mode='full', beta=12.0, c0=4.0)
        try_update(a_out, f"nmd_b12_n{len(best_seq)}")
    
    # === Phase 3: NMD with top-k subgradient (different flavor) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: NMD top-5 n={len(best_seq)} (t={t3:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror(a_in, t3, verbose=True, seed=111,
                                   subgrad_mode='top', top_k=5, c0=5.0)
        try_update(a_out, f"nmd_top5_n{len(best_seq)}")
    
    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.13, 220.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Peak shaving n={len(best_seq)} (t={t4:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t4, verbose=True, seed=42)
        try_update(a_out, f"peak_n{len(best_seq)}")
    
    # === Phase 5: Short NMD polish after peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 170.0)
    if t5 > 20:
        print(f"\n=== Phase 5: NMD polish n={len(best_seq)} (t={t5:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror(a_in, t5, verbose=False, seed=200,
                                   subgrad_mode='full', beta=3.0, c0=6.0)
        try_update(a_out, f"nmd_polish_n{len(best_seq)}")
    
    # === Phase 6: Coordinate line search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Coord LD n={len(best_seq)} (t={t6:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_ld(a_in, t6, verbose=True, seed=314)
        try_update(a_out, f"coordld_n{len(best_seq)}")
    
    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        for delta_n in [200, -200, 400, -400]:
            n_try = n_best + delta_n
            if n_try < 500 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 7: Length search n={n_try} ===")
            a_inter = interpolate_sequence(best_seq, n_try)
            a_in = norm_sum1(a_inter)
            t7 = min(remaining * 0.35, 150.0)
            if t7 > 20:
                a_out, _ = nesterov_mirror(a_in, t7, verbose=False, seed=55 + n_try % 100,
                                           subgrad_mode='full', beta=5.0, c0=5.0)
                try_update(a_out, f"nmd_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    a_out2, _ = coordinate_ld(a_in, min(remaining * 0.4, 60.0),
                                              verbose=False, seed=77)
                    try_update(a_out2, f"coordld_n{n_try}")
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t8 = min(remaining * 0.7, 120.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final coord polish n={n_best} (t={t8:.0f}s) ===")
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_ld(a_in, t8, verbose=True, seed=999)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 9: One more NMD pass if time allows ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 15:
            print(f"\n=== Phase 9: Final NMD n={len(best_seq)} (t={t9:.0f}s) ===")
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_mirror(a_in, t9, verbose=False, seed=777,
                                       subgrad_mode='full', beta=4.0, c0=6.0)
            try_update(a_out, f"nmd_ultimate_n{len(best_seq)}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
