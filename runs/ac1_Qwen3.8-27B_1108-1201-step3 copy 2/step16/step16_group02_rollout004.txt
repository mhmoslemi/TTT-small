def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def exact_subgradient(a, ac, top_k=20, rel_thresh=0.01):
    """Compute exact subgradient: uniform average over lags within rel_thresh of max."""
    n = len(a)
    M = ac.max()
    # Find lags within rel_thresh of max
    threshold = M * (1.0 - rel_thresh)
    active = np.where(ac >= threshold)[0]
    
    # Weight by how close to max (power weighting for smoothness)
    weights = (ac[active] / M) ** 10  # Sharp weighting
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(weights, active):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def softmax_subgradient(a, ac, temperature):
    """Compute softmax subgradient."""
    n = len(a)
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        top_idx = np.argsort(ac)[::-1][:min(50, len(ac))]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0
        w_sum = weights.sum()
    weights /= w_sum
    
    grad = np.zeros(n)
    for k in range(len(ac)):
        w = weights[k]
        if w < 1e-12:
            continue
        k_int = k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]
    return grad

def nesterov_mirror_descent(a0, time_limit, initial_temp=0.01, final_temp=0.0005,
                            c0=3.0, verbose=True):
    """Nesterov accelerated mirror descent with sharp subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: faster decay
        frac = min(1.0, t / 3000.0)
        temp = initial_temp * (1 - frac) + final_temp * frac
        temp = max(temp, 1e-6)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Use exact subgradient for low temp, softmax for high temp
        if temp < 0.002:
            g = exact_subgradient(y, ac, top_k=30, rel_thresh=0.005)
        else:
            g = softmax_subgradient(y, ac, temp)

        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm  # Normalize gradient
        
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Periodic restarting with perturbation
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.7)

        # Running average
        running_avg += x
        avg_count += 1
        if avg_count % 1000 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, stall={stall_count}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=500, verbose=True):
    """LP Frank-Wolfe with adaptive constraint set and backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = np.zeros(n)  # Nesterov momentum

    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Select constraints: lags within 2% of max, up to max_constraints
        threshold = M_curr * 0.98
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]

        # Build constraint matrix
        n_cons = len(active_lags)
        T = np.zeros((n_cons, n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        # Wait, if we constrain sum(b_var) == 1, then max sum(b_var) = 1 always.
        # The LP should be: max sum(b_var) s.t. T@b <= M, b >= 0
        # This gives the maximum total mass in b that satisfies the constraints.
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_val_raw(b_var), dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        alpha = 2.0 / (t + 2.0)
        for _ in range(10):
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                alpha *= 0.5
                continue
            y_trial = y_trial / sy
            m_trial = fast_autocorr(y_trial).max()
            if m_trial <= M_curr + 1e-12:
                break
            alpha *= 0.5

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov momentum
        if t == 1:
            x = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x = y_new + beta_t * (y_new - y)
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                x = y_new.copy()
            else:
                x = x / sx

        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, n_cons={n_cons}")

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Subgradient-guided coordinate descent with 1D line search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 8
    sweep = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        sweep_improved = False

        ac = fast_autocorr(a)
        M = ac.max()
        
        # Compute subgradient
        g = exact_subgradient(a, ac, rel_thresh=0.01)
        
        # Sort coordinates by |gradient| (most important first)
        order = np.argsort(-np.abs(g))
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            a_i = a[i]
            if a_i < 1e-15:
                continue
            
            # Find best transfer: move some mass from i to j
            # Try j's that have opposite sign gradient
            if g[i] > 0:
                # Want to decrease a[i], so transfer to j with g[j] < 0
                candidates = np.where(g < 0)[0]
            else:
                candidates = np.where(g > 0)[0]
            
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Take top candidates by |g[j]|
            if len(candidates) > 10:
                cand_scores = np.abs(g[candidates])
                top_cands = candidates[np.argsort(-cand_scores)][:10]
            else:
                top_cands = candidates
            
            best_trial_m = None
            best_move = None
            
            for j in top_cands:
                if j == i:
                    continue
                # Try several fractions
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[i] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if best_trial_m is None or m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_move = (j, frac)
            
            if best_move is not None and best_trial_m < best_m - 1e-16:
                j, frac = best_move
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                sweep_improved = True
                # Update gradient estimate
                if i % 200 == 0:  # Recompute gradient periodically
                    ac = fast_autocorr(a)
                    g = exact_subgradient(a, ac, rel_thresh=0.01)

        if sweep_improved:
            stall = 0
        else:
            stall += 1
            # Perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.003, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def spectral_init(n, rng):
    """Initialize with a flat-spectrum signal (Zadoff-Chu like)."""
    # Generate a random phase sequence, take real part, clip to non-negative
    # Or use a specific structure
    x = np.linspace(0, 1, n)
    # Try a fewstructured shapes and pick the best
    best = None
    best_m = float('inf')
    
    for trial in range(20):
        # Random smooth shape
        freqs = rng.uniform(0.1, 3.0, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        f = np.zeros(n)
        for freq, phase in zip(freqs, phases):
            f += np.sin(2*np.pi*freq*x + phase)
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-10:
            continue
        f = f / s
        m = fast_autocorr(f).max()
        if m < best_m:
            best_m = m
            best = f.copy()
    
    if best is None:
        # Fallback: triangle
        best = np.maximum(1.0 - 2.0*np.abs(x - 0.5), 0.0)
        best = best / best.sum()
    
    return best
