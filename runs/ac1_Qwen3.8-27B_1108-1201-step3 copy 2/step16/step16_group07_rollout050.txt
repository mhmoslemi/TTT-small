    """Compute gradient of log-sum-exp approximation to max autocorrelation."""
    n = len(a)
    conv = fast_autocorr(a)
    
    # For numerical stability
    conv_shifted = conv - conv.max()
    weights = np.exp(beta * conv_shifted)
    weights = weights / weights.sum()
    
    # Gradient: sum_k weights[k] * grad_k
    # grad_k[j] = 2*a[k-j] if 0 <= k-j < n
    # So gradient[j] = 2 * sum_k weights[k] * a[k-j]
    # This is 2 * (weights cross-correlate a)[j]

    FA = rfft(a, fft_size)
    corr = irfft(FW * np.conj(FA), fft_size)

2. Subgradient method with Nesterov momentum (exact max)
3. Coordinate descent with exact line search (polish)
4. Multiple restarts with perturbations
Key innovation: FFT-based gradient computation enables 1000x more iterations than LP-based methods.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_to_simplex(y, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(y)
    y = y.copy()
    # Find lambda such that sum(max(0, y_i - lambda)) = S
    lo, hi = 0.0, max(y.max(), 0.0) + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2
        s = np.maximum(y - mid, 0.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(y - (lo + hi) / 2, 0.0)

def sme_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv(a,a)[k]))) using FFT."""
    n = len(a)
    conv = fast_autocorr(a)
    # Shift for numerical stability
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(beta * conv_shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Compute gradient via cross-correlation
    size = 2 * n - 1
    W = weights
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    FW = rfft(W, fft_size)
    FA = rfft(a, fft_size)
    corr = irfft(FW * np.conj(FA), fft_size)
    grad = 2.0 * corr[:n]
    
    return grad, exp_vals.sum() ** (1.0/beta)  # (1/beta)*log(sum) + max_conv

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize(a, n):
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * S / s

def logsumexp_descent(a0, time_limit, beta_start=1.0, beta_end=50.0, verbose=False, seed=0):
    """Optimize using log-sum-exp smooth approximation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    step = 0.1
    patience = 0
    max_patience = 50
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        # Adaptive beta
        progress = min(1.0, t / 500.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute gradient
        grad, _ = sme_gradient(y, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Nesterov momentum
        if t > 1:
            mu = min((t - 1.0) / (t + 2.0), 0.9)
            y_new = a + mu * (a - y_prev)
        else:
            y_new = a.copy()
        
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, n)
        
        # Recompute gradient at y_new
        grad_new, _ = sme_gradient(y_new, beta)
        gn = np.linalg.norm(grad_new)
        if gn < 1e-15:
            break
        grad_new = grad_new / gn
        
        # Step
        a_new = y_new - step * grad_new * S / np.sqrt(n)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, n)
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            patience = 0
            step = min(step * 1.005, 5.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-6:
                break
        
        y_prev = y
        y = y_new
        a = a_new
        
        if verbose and t % 500 == 0:
            print(f"  LSE iter={t} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}, beta={beta:.1f}, step={step:.4f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with exact 1D optimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            # Try different values for a[j]
            # The change in conv when a[j] changes: conv[k] changes by 2*(new-val)*a[k-j] for k in [j, j+n-1]
            # (with correction at k=2j)
            
            candidates = [0.0, aj * 0.5, aj * 0.7, aj * 1.3, aj * 1.6, aj * 2.0]
            typical = S / n
            candidates.extend([typical * 0.3, typical, typical * 2.0, typical * 4.0])
            candidates = [max(0.0, min(c, S)) for c in candidates]
            # Remove duplicates
            candidates = list(set([round(c, 10) for c in candidates]))
            
            best_val = aj
            best_obj = best_score  # Use current best as baseline
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                # Update conv
                new_conv = conv.copy()
                delta = x - aj
                lo = j
                hi = min(2*n-1, j + n)
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2*j < 2*n-1:
                    new_conv[2*j] += delta * delta
                
                score = new_conv.max()
                # Scale: score_actual = score * (S/(S-aj))^2 (approximately)
                # Actually, if we change a[j] from aj to x, the new sum is S - aj + x
                # The actual score = 2*n*max_b_new / (new_sum)^2
                # But we're comparing unnormalized, so we need to be careful.
                # Since we normalize after, let's just compare the unnormalized max.
                # Actually for coordinate descent, we should compare the normalized score.
                new_sum = S - aj + x
                if new_sum < 1e-15:
                    continue
                scale = S / new_sum  # We'll normalize to S
                score_normalized = score * scale * scale
                
                if score_normalized < best_obj:
                    best_obj = score_normalized
                    best_val = x
            
            if abs(best_val - aj) > 1e-10:
                a[j] = best_val
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * S / s_new
                    actual_score = fast_autocorr(a).max()
                    if actual_score < best_score - 1e-12:
                        best_score = actual_score
                        best_a = a.copy()
                        improved = True
                        conv = fast_autocorr(a)
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose:
            print(f"  CD pass={no_improve+1} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score
