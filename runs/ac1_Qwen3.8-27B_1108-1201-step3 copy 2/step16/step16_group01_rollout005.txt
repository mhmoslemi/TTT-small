import numpy as np
import time
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_weights(ac, T):
    """Compute softmax weights for convolution values."""
    ac_clipped = np.maximum(ac, 1e-30)
    log_w = ac_clipped / T
    log_w -= log_w.max()
    w = np.exp(log_w)
    sw = w.sum()
    if sw < 1e-30:
        w = np.zeros_like(ac)
        w[int(np.argmax(ac))] = 1.0
    else:
        w = w / sw
    return w

def compute_ssl_grad(w, a):
    """
    Compute grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    Using FFT: grad = (w * a_flipped)[n-1 : 2n-1]
    """
    n = len(a)
    a_flipped = a[::-1].copy()
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flipped, fft_size)
    conv = np.fft.irfft(W * AF, fft_size)
    return conv[n - 1:n - 1 + n]

def smooth_objective_and_grad(x, T, fft_size):
    """
    Compute F(x) = T * log(sum_k exp(conv_k(a)/T)) and its gradient,
    where a = softmax(x).
    """
    n = len(x)
    # Compute a = softmax(x)
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    Z = exp_x.sum()
    a = exp_x / Z
    
    # Compute autocorrelation
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:2*n-1]
    ac = np.maximum(ac, 0.0)  # numerical safety
    
    # Objective
    ac_pos = np.maximum(ac, 1e-30)
    log_sum_exp = T * (np.log(np.sum(np.exp(ac_pos / T))) - ac_pos.max() / T)
    # Actually: T * log(sum exp(ac/T)) = T * (max(ac)/T + log(sum exp((ac-max)/T)))
    max_ac = ac.max()
    log_sum_exp = max_ac + T * np.log(np.sum(np.exp((ac - max_ac) / T)))
    
    # Gradient w.r.t. a
    w = np.exp((ac - max_ac) / T)
    sw = w.sum()
    if sw < 1e-30:
        w = np.zeros_like(ac)
        w[int(np.argmax(ac))] = 1.0
    else:
        w = w / sw
    grad_a = compute_ssl_grad(w, a)
    
    # Gradient w.r.t. x: dF/dx[k] = a[k] * (grad_a[k] - sum_i grad_a[i]*a[i])
    mean_grad = np.dot(grad_a, a)
    grad_x = a * (grad_a - mean_grad)
    
    return log_sum_exp, grad_x

def smoothed_mirror_descent(a0, time_limit, T_start=0.05, T_end=0.005, c0=3.0,

    Smoothed mirror descent with temperature annealing.
    """

    a = normalize(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    x_prev = a.copy()

    max_stall = 3000
    current_c = c0

        # Anneal temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, 0.001)
        
        # Nesterov extrapolation
        beta = (total_iters - 1.0) / (total_iters + 1.5) if total_iters > 1 else 0.0
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and smooth gradient
        ac = fast_autocorr(y)
        w = softmax_weights(ac, T)
        grad = compute_ssl_grad(w, y)
        
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        # Step size
        eta = current_c / (np.sqrt(total_iters + 1) + 1.0) * np.sqrt(n)
        
        # Mirror descent update
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        total_iters += 1
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive restart
        if stall > max_stall:
            stall = 0
            current_c = min(current_c * 1.25, 15.0)
            # Anti-peak perturbation
            ac_best = fast_autocorr(best_a)
            peak_k = int(np.argmax(ac_best))
            # Create structured perturbation: boost positions away from peak
            noise = rng.normal(0, 0.005, n)
            # Add anti-peak component
            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            for i in range(i_start, i_end):
                j = peak_k - i
                if 0 <= j < n:
                    # Reduce positions that contribute to the peak
                    noise[i] -= 0.01 * best_a[j]
                    noise[j] -= 0.01 * best_a[i]
            x_restarted = np.maximum(best_a + noise, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            # Reset iter counter for step size
            total_iters = max(total_iters, 10)
        
        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SMD iter={total_iters} ({elapsed:.1f}s): J={J:.6f}, T={T:.5f}, stall={stall}")
    
    return best_a, best_M

def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """
    Peak-aware coordinate descent: move mass from high-peak-contribution
    positions to low-peak-contribution positions.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        # Get top peak lags
        n_peaks = min(10, len(ac))
        peak_idx = np.argsort(ac)[::-1][:n_peaks]
        
        # Compute peak sensitivity for each position
        sens = np.zeros(n)
        for k in peak_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    sens[i] += ac[k_int] * a[j]
        
        # Find high and low sensitivity positions
        high_sens = np.argsort(sens)[::-1][:50]
        low_sens = np.argsort(sens)[:50]
        
        improved = False
        for src in high_sens:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in low_sens:
                if dst == src or a[dst] > a[src]:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = max(3, n // 200)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
            if not improved and stall > 20:
                break
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    if f_norm.max() < 1e-15 or np.sum(f_norm > 1e-10) < 3:
        # Near-uniform or degenerate: just use uniform
        return np.ones(new_n) / new_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
