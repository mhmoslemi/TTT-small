1. APG with power-smoothed max objective (larger top_k)
2. Mirror descent with multi-peak subgradient

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)

        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def power_smoothed_obj(a, top_k, p):
    """Compute (1/K * sum top-K c[k]^p)^(1/p) and its gradient."""
    c = fast_conv(a)
    n = len(a)
    K = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    # Power smoothing: (1/K * sum v^p)^(1/p)
    # Gradient: (1/K * p * v^(p-1) * grad_v) * (1/p) * (1/K * sum v^p)^(1/p - 1)
    # = (sum v^(p-1) * grad_v / K) * (obj)^(1-p)
    obj = (np.mean(top_vals ** p)) ** (1.0 / p)
    # Weights proportional to v^(p-1)
    weights = top_vals ** (p - 1)
    weights /= weights.sum()
    # Gradient
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return obj, grad, c.max()

def apg_power(a0, time_limit, top_k=500, p_start=20.0, p_end=50.0, c0=2.0, seed=42, verbose=True):
    """APG with power-smoothed max objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 2000

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 15000.0)
        cur_p = p_start + (p_end - p_start) * frac
        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)
        obj, g, m_curr = power_smoothed_obj(y, top_k, cur_p)
        eta = gamma / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:

            max_stall = int(max_stall * 1.2)
            noise = rng.normal(0, 0.003, n) * best_a

            x = x_re
            x_prev = x_re
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG-pow t={t} ({elapsed:.0f}s): J={J:.6f}, p={cur_p:.0f}")

def mirror_descent(a0, time_limit, n_peaks=20, eta0=0.5, decay=0.9995, seed=42, verbose=True):

    n = len(a0)

def coord_peak_reduce(a0, time_limit, n_peaks=10, seed=42, verbose=True):
    """Direct coordinate moves to reduce individual peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        improved = False
        for k_star in top_idx:
            k_int = int(k_star)
            # For this lag, the contribution is sum_i a[i]*a[k_int-i]
            # We want to reduce this. Move mass from high-product pairs to low-product pairs.
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            products = a[idx] * a[j_idx]
            # Find the pair with the largest product
            if len(idx) == 0:
                continue
            src_pos = idx[np.argmax(products)]
            # Find a good destination: a position where moving mass doesn't increase other peaks
            # Approximate: destination j is good if a[j]*a[k_int-j] is small
            # and j is not in the idx range
            candidates = np.setdiff1d(np.arange(n), idx)
            if len(candidates) == 0:
                continue
            # Score candidates by their product at this lag (lower is better)
            scores = np.zeros(len(candidates))
            for i, j in enumerate(candidates):
                if 0 <= k_int - j < n:
                    scores[i] = a[j] * a[k_int - j]

                    scores[i] = 0  # Doesn't contribute to this lag

                for i, j in enumerate(candidates):
                    if 0 <= k2_int - j < n:

            dst_pos = candidates[np.argmin(scores)]
            # Try moving a small amount

                amt = a[src_pos] * frac

                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst_pos] += amt
                a_trial = np.maximum(a_trial, 0.0)

                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()

                    best_m = M_trial

        else:

                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt

                a = a / a.sum()
                M_new = fast_conv(a).max()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
