Summary: d(a*a)[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.

This is clean! So the subgradient of max_k (a*a)[k] is in the convex hull of the vectors g_k where g_k[j] = 2*a[k-j] if 0<=k-j<n else 0.

def solve_lp(a_fixed, M, active_k, solver):
    """LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_k, b >= 0."""
    n = len(a_fixed)
    k_arr = np.asarray(active_k, dtype=int)
    size = len(k_arr)
    
    # Build constraint matrix
    T = np.zeros((size, n))

        lo = max(0, k - n + 1)

            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")
    
    # === Main LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 650.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov, budget={t1:.0f}s ===")
        a_norm = normalize(best_seq)
        a_out, _ = lp_nesterov(a_norm, t1, solver, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp1", t_global)
    
    # === Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Coordinate Descent, budget={t2:.0f}s ===")
        a_norm = normalize(best_seq)
        a_out, _ = coordinate_descent(a_norm, t2, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cd1", t_global)
    
    # === LP Nesterov Round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP Nesterov r2, budget={t3:.0f}s ===")
        a_norm = normalize(best_seq)
        a_out, _ = lp_nesterov(a_norm, t3, solver, verbose=True, seed=777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp2", t_global)
    
    # === Softmax Subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Softmax Subgrad, budget={t4:.0f}s ===")
        a_norm = normalize(best_seq)
        a_out, _ = softmax_subgrad(a_norm, t4, verbose=True, seed=123)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "softmax", t_global)
    
    # === Final Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Final CD, budget={t5:.0f}s ===")
        a_norm = normalize(best_seq)
        a_out, _ = coordinate_descent(a_norm, t5, verbose=True, seed=99)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cd2", t_global)
    
    # === Perturbation Restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 6: Perturbation restarts, budget={remaining:.0f}s ===")
        for r in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t_r = min(remaining * 0.3, 120.0)
            if t_r < 20:
                break
            scale = 0.02 + 0.02 * rng.random()
            noise = 1.0 + scale * rng.standard_normal(len(best_seq))
            noise = np.maximum(noise, 0.5)
            perturbed = best_seq * noise
            perturbed = normalize(perturbed)
            
            # Quick LP on perturbed
            t_lp = t_r * 0.6
            a_out, _ = lp_nesterov(perturbed, t_lp, solver, seed=500 + r * 10)
            # Quick CD
            t_cd = t_r * 0.4
            a_out2, _ = coordinate_descent(a_out, t_cd, seed=600 + r * 10)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_lp", t_global)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out2, f"pert{r}_cd", t_global)
    
    # === Final micro-polish CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t_final = min(remaining * 0.7, 60.0)
        a_norm = normalize(best_seq)
        a_out, _ = coordinate_descent(a_norm, t_final, verbose=False, seed=31415)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cdFinal", t_global)
    
    # Final evaluation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
