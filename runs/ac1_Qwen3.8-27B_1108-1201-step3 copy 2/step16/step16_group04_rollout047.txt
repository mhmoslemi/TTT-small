def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_equalize(a, n_iter=30, eps=1e-10, blend=0.5):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for i in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        # Target: flat spectrum
        target = P.mean()
        A_new = A * (target / (P + eps)) ** 0.5
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1 - blend) * a + blend * a_new
    return a

def lse_smoothed_objective(a, tau):
    """Compute log-sum-exp smoothed max of autocorrelation."""
    ac = fast_autocorr(a)
    # Shift to avoid overflow
    ac_shifted = ac - ac.max()
    lse = ac.max() + np.log(np.sum(np.exp(tau * ac_shifted))) / tau
    return lse, ac

def lse_gradient(a, tau, top_k=None):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_shifted = ac - ac.max()
    log_weights = tau * ac_shifted - np.log(np.sum(np.exp(tau * ac_shifted)))
    
    if top_k is not None:
        # Use only top-K lags for efficiency
        idx = np.argsort(ac)[::-1][:top_k]
        log_weights_full = np.full(len(ac), -100.0)
        log_weights_full[idx] = log_weights[idx]
        log_weights = log_weights_full
    
    weights = np.exp(log_weights)
    
    # Gradient: d/d a[j] of sum_k w_k * conv(a,a)[k]
    # = sum_k w_k * (a[k-j] + a[j-k]) (with boundary handling)
    grad = np.zeros(n)
    # For efficiency, use FFT-based approach
    # grad[j] = sum_k w_k * (a[k-j] if valid + a[j-k] if valid)
    # This is like a weighted cross-correlation
    
    # For each j: grad[j] = sum_{k: 0<=k-j<n} w_k * a[k-j] + sum_{k: 0<=j-k<n} w_k * a[j-k]
    # = sum_{l: 0<=l<n} w_{j+l} * a[l] + sum_{l: 0<=l<n} w_{j-l} * a[l] (with boundary)
    # = (w * a)[j] + (w * a reversed)[n-1-j] ... this isn't quite right
    
    # Let me use direct computation for the top-K approach
    if top_k is not None:
        for k_int in np.argsort(ac)[::-1][:top_k]:
            k_i = int(k_int)
            w = weights[k_i]
            if w < 1e-15:
                continue
            # d conv(a,a)[k_i] / da[j] = a[k_i-j] (if 0<=k_i-j<n) + a[j-k_i] (if 0<=j-k_i<n)
            # For j in range [max(0, k_i-n+1), min(n-1, k_i)]:
            #   a[k_i-j] is valid
            # For j in range [max(0, k_i), min(n-1, n-1+k_i)] = [k_i, n-1] (if k_i < n):
            #   a[j-k_i] is valid
            
            js1 = max(0, k_i - n + 1)
            je1 = min(n - 1, k_i)
            if js1 <= je1:
                j_idx = np.arange(js1, je1 + 1)
                grad[j_idx] += w * a[k_i - j_idx]
            
            js2 = max(0, k_i)
            je2 = min(n - 1, n - 1 + k_i)
            if js2 <= je2:
                j_idx = np.arange(js2, je2 + 1)
                # a[j-k_i]
                # But we need to avoid double counting when j-k_i == k_i-j, i.e., j = k_i/2
                a_vals = a[j_idx - k_i]
                # Check for double counting
                for ii, jj in enumerate(j_idx):
                    if jj - k_i == k_i - jj:  # same term
                        grad[jj] += 0.5 * w * a_vals[ii]
                    else:
                        grad[jj] += w * a_vals[ii]
    else:
        # Full gradient via FFT
        # grad = 2 * (w circularly convolved with a) ... 
        # Actually for LINEAR convolution, it's more complex
        # Let's use the "slice" trick
        # grad[j] = sum_k w[k] * a[k-j] (where valid) + sum_k w[k] * a[j-k] (where valid)
        # First term: for fixed j, sum over k where 0 <= k-j < n, i.e., j <= k < j+n
        # = sum_{k=j}^{j+n-1} w[k] * a[k-j] = (w_shifted * a)[j] where w_shifted[k] = w[k] for k in [0, 2n-2]
        # This is a cross-correlation
        
        # Use FFT for efficiency
        # cross_corr[j] = sum_k w[k] * a[k-j] = sum_l w[j+l] * a[l]
        # = (a * w_flipped)[j] where w_flipped[k] = w[2n-2-k]
        # Hmm, let me just use the direct approach with top_k for simplicity
    
    return grad, ac

def lse_gradient_fft(a, tau, top_k=None):
    """
    Compute gradient of log-sum-exp smoothed max autocorrelation using FFT.
    
    grad[j] = sum_k p_k * (a[k-j] + a[j-k]) where valid
    p_k = exp(tau * ac[k]) / sum_l exp(tau * ac[l])
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Compute softmax weights
    ac_max = ac.max()
    log_p = tau * (ac - ac_max)
    log_p -= np.log(np.exp(log_p).sum())
    p = np.exp(log_p)
    
    if top_k is not None:
        # Zero out non-top-K weights for efficiency
        keep = np.argsort(p)[::-1][:top_k]
        p_masked = np.zeros_like(p)
        p_masked[keep] = p[keep]
        p = p_masked
    
    # grad[j] = sum_k p[k] * a[k-j] (for valid k-j) + sum_k p[k] * a[j-k] (for valid j-k)
    # First term: cross-correlation of p and a
    # (p * a_conj)[j] in the "circular" sense, but we need linear
    # Actually: sum_k p[k] * a[k-j] where 0 <= k-j < n means j <= k < j+n
    # Let's define p_ext of length 2n-1 (the autocorrelation has this length)
    # For j in [0, n-1]: sum_{k=max(0,j)}^{min(2n-2, j+n-1)} p[k] * a[k-j]
    
    # Use FFT: create a vector of length >= 2n
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    
    # First term: f1[j] = sum_k p[k] * a[k-j] for valid indices
    # = sum_l p[j+l] * a[l] for l in [0, n-1] and j+l in [0, 2n-2]
    # This is the cross-correlation of p (length 2n-1) with a (length n), evaluated at lags 0 to n-1
    # Using FFT: ifft(FFT(p) * conj(FFT(a)))... but we need to be careful with padding
    
    # Simpler approach: use the fact that
    # f1[j] = sum_l a[l] * p[j+l]
    # This is the "asymmetric" cross-correlation
    
    # Let me just compute it directly using the slice trick with FFT
    # f1 = ifft(conj(fft(a, fft_size)) * fft(p_padded, fft_size))
    # where p_padded is p padded to length fft_size
    
    # Actually, let's use a different approach:
    # f1[j] = sum_{l=0}^{n-1} a[l] * p[j+l]
    # If I define q of length 2n-1 as q[k] = p[k] (for k in [0, 2n-2])
    # Then f1[j] = sum_l a[l] * q[j+l]
    # This is the "anti-convolution" or "cross-correlation" of a and q
    
    # Using FFT: f1 = ifft(fft(a, N) * conj(fft(q, N))) ... no
    # conv(a, q)[j] = sum_l a[l] * q[j-l]
    # We want sum_l a[l] * q[j+l] = sum_l a[l] * q_rev[l-j] where q_rev[k] = q[2n-2-k]
    # = conv(a, q_rev)[j + (2n-2)]... this is getting complicated
    
    # Let me just use a simple approach: for each j, the sum is over at most n terms
    # with n up to ~3000, and 2n-1 lags, the total work is O(n^2) which is fine for n=3000
    # (9 million operations)
    
    if top_k is not None and top_k < n:
        # Use only top-K lags
        top_idx = np.argsort(p)[::-1][:top_k]
        grad = np.zeros(n)
        for k_i in top_idx:
            k = int(k_i)
            w = p[k_i]
            if w < 1e-18:
                continue
            # d conv(a,a)[k] / da[j]:
            # For j in [max(0,k-n+1), min(n-1,k)]: contributes a[k-j]
            # For j in [max(0,k), n-1]: contributes a[j-k]
            js1 = max(0, k - n + 1)
            je1 = min(n - 1, k)
            if js1 <= je1:
                j_idx = np.arange(js1, je1 + 1)
                grad[j_idx] += w * a[k - j_idx]
            
            js2 = max(0, k)
            je2 = n - 1
            if js2 <= je2:
                j_idx = np.arange(js2, je2 + 1)
                # a[j-k], but avoid double counting
                # When j-k == k-j, i.e., j = k/2 (only if k even and j in both ranges)
                overlap_start = max(js1, js2)
                overlap_end = min(je1, je2)
                for jj in j_idx:
                    if js1 <= jj <= je1:
                        # In both ranges, count only once (the two terms are equal: a[k-j] = a[j-k])
                        grad[jj] += 0.5 * w * a[k - jj]  # a[k-j] = a[j-k]
                    else:
                        grad[jj] += w * a[jj - k]
    else:
        # Full computation
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = p[k]
            if w < 1e-18:
                continue
            js1 = max(0, k - n + 1)
            je1 = min(n - 1, k)
            if js1 <= je1:
                j_idx = np.arange(js1, je1 + 1)
                grad[j_idx] += w * a[k - j_idx]
            js2 = max(0, k)
            je2 = n - 1
            if js2 <= je2:
                j_idx = np.arange(js2, je2 + 1)
                for jj in j_idx:
                    if js1 <= jj <= je1:
                        grad[jj] += 0.5 * w * a[k - jj]
                    else:
                        grad[jj] += w * a[jj - k]
    
    return grad, ac, p

def mirror_descent_smoothed(a0, time_limit, tau_init=50.0, tau_min=5.0, 
                            tau_decay=0.9999, top_k=100, verbose=True,
                            c0=5.0, use_nesterov=True):
    """
    Mirror descent with log-sum-exp smoothed objective.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    tau = tau_init
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if use_nesterov and t > 1:
            momentum = (t - 1.0) / (t + 1.5)
            y = x + momentum * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        g, ac, p = lse_gradient_fft(y, tau, top_k=top_k)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        
        x_prev = y
        x = a_new
        
        # Decay temperature
        tau *= tau_decay
        tau = max(tau, tau_min)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, tau={tau:.1f}, M={best_m:.8f}")
    
    return best_a, best_m

def lp_refine(a0, time_limit, solver, top_k_init=50, top_k_max=200, verbose=True):
    """LP Frank-Wolfe refinement."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    current_top_k = top_k_init
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(x)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = x.copy()
        
        # Select active constraints
        active_lags = np.argsort(ac)[::-1][:current_top_k]
        
        # Solve LP: max sum(b) s.t. conv(x, b)[k] <= M_curr for k in active_lags, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active_lags:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            j_idx = np.arange(j_start, j_end + 1)
            x_coeff = x[k_int - j_idx]
            constraints.append(x_coeff @ b_var[j_idx] <= M_curr)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            current_top_k = max(10, current_top_k // 2)
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            current_top_k = max(10, current_top_k // 2)
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        b_val = b_val / b_sum
        
        # Step size (diminishing)
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * x + alpha * b_val
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            continue
        x_new = x_new / sx
        
        x = x_new
        
        # Adapt top_k
        if t % 50 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 5, top_k_max)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)
            perm = rng.permutation(len(candidates))[:min(300, len(candidates))]
            chosen = candidates[perm]
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and not improved:
                pass
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def get_solver():
    """Get an available CVXPY solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def dual_ascent_refine(a0, time_limit, n_iter_per_round=50, verbose=True):
    """
    Dual ascent method: maintain dual variables (lagrange multipliers) for each lag,
    and update the primal by solving a QP subproblem.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Dual variables for each lag
    mu = np.ones(2 * n - 1) / (2 * n - 1)
    
    for round_num in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update dual: mu_k proportional to max(0, ac[k] - t)
        ac = fast_autocorr(a)
        M_curr = ac.max()
        excess = np.maximum(ac - M_curr * 0.95, 0)
        mu = excess / max(excess.sum(), 1e-15)
        
        # Solve QP: min sum_k mu_k * conv(a,a)[k] + 0.5 * ||a - a_old||^2
        # s.t. sum(a) = 1, a >= 0
        # This is equivalent to: min (1/2) a^T (sum_k mu_k Q_k + I) a - a^T Q_old + const
        # where Q_k is the quadratic form matrix for lag k
        
        # For efficiency, use the fact that the Hessian is:
        # H[j,l] = sum_k mu_k * (delta_{l, k-j} + delta_{l, j-k})/2 + delta_{j,l}
        # This is a banded + diagonal matrix (banded because for each k, only O(1) entries)
        
        # Actually, let's just use the gradient of the weighted sum:
        # grad[j] = sum_k mu_k * (a[k-j] + a[j-k]) (with boundary)
        # The "natural" update is a <- a - eta * grad, projected to simplex
        
        # But for a proper QP solution, we'd need to solve the system.
        # For now, use gradient descent on the weighted objective with projection.
        
        for _ in range(n_iter_per_round):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute weighted gradient
            g = np.zeros(n)
            for k in range(2 * n - 1):
                w = mu[k]
                if w < 1e-12:
                    continue
                js1 = max(0, int(k) - n + 1)
                je1 = min(n - 1, int(k))
                if js1 <= je1:
                    j_idx = np.arange(js1, je1 + 1)
                    g[j_idx] += w * a[int(k) - j_idx]
                js2 = max(0, int(k))
                je2 = n - 1
                if js2 <= je2:
                    j_idx = np.arange(js2, je2 + 1)
                    for jj in j_idx:
                        if js1 <= jj <= je1:
                            g[jj] += 0.5 * w * a[int(k) - jj]
                        else:
                            g[jj] += w * a[jj - int(k)]
            
            # Gradient step with projection
            eta = 2.0 / (g @ g + 1e-10)
            a_new = a - eta * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
            a = a_new
        
        if verbose and (round_num + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Dual r={round_num+1} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
