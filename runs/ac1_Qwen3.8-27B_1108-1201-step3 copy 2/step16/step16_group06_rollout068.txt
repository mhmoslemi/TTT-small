def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def peak_to_sum_ratio(a):
    """Objective in normalized form (sum=1): 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = a / s
    return float(2.0 * n * fast_autocorr(b).max())


def build_convolution_matrix(a):
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j + n, j] = a[j]
    return T


def compute_gradient_peaks(ac, a, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent_nesterov(a0, time_limit, top_k=60, beta_start=3.0, beta_end=8.0,
                            verbose=True, seed=42):
    """Nesterov mirror descent with temperature-scheduled soft peak weighting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall_count = 0
    max_stall = 6000

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        # schedule beta from smooth (low) to sharp (high) over first ~20k iters
        sched = min(1.0, t / 20000.0)
        beta_t_soft = beta_start + (beta_end - beta_start) * sched
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        grad = compute_gradient_peaks(ac, a, top_k, beta_t_soft)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_ratio = -eta * grad
        log_ratio = np.clip(log_ratio, -50.0, 50.0)
        a_new = y * np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 1000 == 0:
                gamma *= 0.93
                if gamma < 0.3:
                    x = best_a.copy()
                    x_prev = best_a.copy()
                    gamma = 4.0
                    stall_count = 0

        if stall_count > max_stall:
            stall_count = 0
            sigma = 0.05
            perturbation = rng.normal(0.0, sigma, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            gamma = 4.0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MD t=%d (%.1fs): J=%.6f gamma=%.3f beta=%.2f"
                  % (t, elapsed, J, gamma, beta_t_soft))

def spectral_shaping(a0, strength=0.5, verbose=False):
    """Frequency-move: pull the spectral magnitude toward flat (reduces peak
    autocorrelation), keep phase, project to nonnegative, renormalize."""
    a = norm_sum1(a0)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.exp(1j * np.angle(A))
    # target magnitude ~ geometric smoothing between current and its mean
    target = (mag + 1e-12) ** (1.0 - strength) * mag.mean() ** strength
    A_new = phase * target
    a_new = np.fft.irfft(A_new, fft_size)
    a_new = a_new[:size]
    # take the only window of length n with max sum (the original alignment)
    a_new = np.maximum(a_new[:n], 0.0)
    a_new = norm_sum1(a_new)
    return a_new

            prob.solve(solver=solver_name, verbose=False)

            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_seed(n):
    """A well-spread sinusoidal seed that already beats the uniform vector."""
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    if remaining > 120:
        n_best = len(best_seq)
        print("\n=== Phase 6: multi-n exploration ===")
        test_ns = []
        for delta in (200, 400, 600, -200, -400):
            nt = n_best + delta
            if 1200 <= nt <= 3600 and nt != n_best:
                test_ns.append(nt)
        for nt in test_ns[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print("  trying n=%d" % nt)
            a_inter = interpolate_sequence(best_seq, nt)
            t_a = min(remaining * 0.15, 70.0)
            a_out, _ = mirror_descent_nesterov(a_inter, t_a,
                                               top_k=40, beta_start=3.0, beta_end=8.0,
                                               verbose=False, seed=21)
            consider(a_out, "md_n%d" % nt)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60:
                t_b = min(remaining * 0.20, 100.0)
                a_out, _ = lp_frank_wolfe_fast(a_inter, t_b, solver, verbose=False)
                consider(a_out, "lp_n%d" % nt)

    # Phase 7: final LP-FW push at best length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print("\n=== Phase 7: final LP-FW n=%d ===" % n_best)
        t7 = min(remaining * 0.45, 180.0)
        a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t7, solver, verbose=True)
        consider(a_out, "lp_final")

    # Phase 8: final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        print("\n=== Phase 8: final coord polish n=%d ===" % n_best)
        t8 = min(remaining * 0.6, 120.0)
        a_out, _ = peak_targeted_coordinate_descent(norm_sum1(best_seq), t8,
                                                    verbose=True, rng_seed=777)
        consider(a_out, "coord_final")

    # Final validation against the real evaluator
    if best_seq is not None:
        fs = score_now(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print("\nFINAL: score=%.6f, n=%d, time=%.0fs"
              % (best_score, len(best_seq), time.monotonic() - t_global))
    else:
        print("\nERROR: no valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
