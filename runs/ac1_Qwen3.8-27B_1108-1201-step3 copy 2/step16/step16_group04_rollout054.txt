    """Compute gradient of log-sum-exp smoothing using FFT."""
    n = len(f)
    ac = fast_autocorr(f)  # length 2n-1
    # Softmax weights
    ac_shifted = ac - ac.max()  # for numerical stability
    w = np.exp(ac_shifted / tau)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated full-gradient mirror descent with adaptive log-sum-exp temperature annealing
2. Nesterov acceleration with periodic symmetrization
3. Active-set LP polishing via HiGHS (scipy) for large correction steps
4. Multi-scale coordinate descent with peak-targeted mass redistribution
5. Multi-length exploration and perturbation-based restarts

import numpy as np
from scipy.optimize import linprog
import time

def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution) via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def symmetrize(a):
    """Symmetrize: (a + a[::-1]) / 2, then renormalize."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return a / max(a.sum(), 1e-15)
    return a_sym / s

def compute_gradient_fft(f, tau):
    """
    Compute gradient of log-sum-exp smoothed max-autocorrelation.
    Returns (grad, ac) where grad[i] = 2 * sum_k w[k]*f[k-i] via FFT.
    """
    n = len(f)
    ac = fast_autocorr(f)
    # Softmax weights
    ac_max = ac.max()
    w = np.exp((ac - ac_max) / tau)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / w_sum
    # Gradient: grad[i] = 2 * sum_j w[j+i] * f[j], computed via FFT
    N = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    d = np.fft.irfft(W * np.conj(F), N)
    grad = 2.0 * d[:n]
    return grad, ac

def spectral_equalize(a, n_iter=20, eps=1e-8):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a

def accelerated_mirror_descent(a0, time_limit, tau0=0.005, tau_decay=0.0005,
                               eta0=2.0, sym_every=50, verbose=True, seed=42):
    """
    Nesterov-accelerated mirror descent (exponential family) with adaptive temperature.
    O(n log n) per step via FFT gradient.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Adaptive temperature
        tau = max(tau0 * np.exp(-tau_decay * t), 1e-6)

        # Compute gradient via FFT
        grad, ac = compute_gradient_fft(y, tau)
        M_curr = ac.max()

        # Track best
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Mirror descent step (exponential family)
        eta = eta0 / (np.sqrt(t) + 1.0)
        # Ensure gradient is normalized in scale
        g_scale = np.max(np.abs(grad))
        if g_scale > 1e-30:
            grad = grad / g_scale
        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Periodic symmetrization
        if t > 200 and t % sym_every == 0:
            a_new = symmetrize(a_new)

        x_prev = y
        x = a_new

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMF t={t} ({elapsed:.1f}s): J={J:.6f}, tau={tau:.2e}")

    return best_a, best_m

def solve_lp_direction(y, M, active_lags):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0, sum(b)=1.
    Uses scipy HiGHS solver. Returns normalized b or None.
    """
    n = len(y)
    # Build constraint matrix A_ub (len(active_lags) x n)
    # (y*b)[k] = sum_j y[j] * b[k-j] for valid j
    rows = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        # Create sparse row
        row = np.zeros(n)
        row[j_idx] = y_coeff
        rows.append(row)
    if not rows:
        return None
    A_ub = np.array(rows)
    b_ub = np.full(len(rows), M)

    # Objective: max sum(b) => min -sum(b)
    c = -np.ones(n)

    # Equality constraint: sum(b) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30.0})
    except Exception:
        return None
    if res.status != 0:
        return None
    b_val = np.maximum(res.x, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def lp_polish(a0, time_limit, top_k=80, verbose=True):
    """LP-based Frank-Wolfe polishing using HiGHS."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Select active lags
        active_lags = np.argsort(ac)[::-1][:top_k]

        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags)
        if direction is None:
            continue

        # Step size
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                                                  eta0=3.0, sym_every=100, verbose=False, seed=777)
            try_update(a_out, f"amf_n{n_alt}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 3000 and remaining > 60:
        print(f"\n=== Phase 6: Final LP n={n_cur} ===")
        t6 = min(remaining * 0.25, 150.0)
        if t6 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_polish(a_in, t6, top_k=100, verbose=True)
            try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final Coord n={n_cur} ===")
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 15:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, seed=999)
            try_update(a_out, f"coord_final_n{n_cur}")

    # === Phase 8: Very final micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Micro-polish ===")
        t8 = min(remaining * 0.7, 60.0)
        if t8 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=False, seed=2024)
            try_update(a_out, f"micro_n{n_cur}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
