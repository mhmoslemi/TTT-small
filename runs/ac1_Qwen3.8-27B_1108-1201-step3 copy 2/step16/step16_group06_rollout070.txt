def fast_autocorr(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * np.conjugate(A), fft_size)[:size]

def compute_grad(a, beta):
    """Gradient of (1/beta)*log(sum_k exp(beta*c[k])) where c = a*a full conv."""
    n = len(a)
    c = fast_autocorr(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / exp_bc.sum()
    # grad[j] = 2 * (a_rev * p)[j + n - 1]
    a_rev = a[::-1]
    fft_size = 1 << (n + len(p) - 2 - 1).bit_length()
    ar_f = np.fft.rfft(a_rev, fft_size)
    p_f = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(ar_f * p_f, fft_size)
    grad = 2.0 * conv[n - 1 : 2 * n - 1]
    return grad

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 60:
        print(f"\n=== Phase 1: Soft-max GD n={n0} ===")
        t1 = min(remaining * 0.15, 150.0)
        a_out, _ = softmax_prox_grad(_best_seq, t1, beta_start=5.0, beta_end=200.0,
                                     max_iters=10000, verbose=True, seed=42)
        try_update_global(a_out, "sgrad1")

    # === Phase 2: LP-FW main workhorse ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 60:
        n_cur = len(_best_seq)
        print(f"\n=== Phase 2: LP-FW n={n_cur} ===")
        t2 = min(remaining * 0.30, 400.0)
        a_out, _ = lp_frank_wolfe(_best_seq, t2, verbose=True, solver_name=solver_name)
        try_update_global(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 40:
        n_cur = len(_best_seq)
        print(f"\n=== Phase 3: Coord n={n_cur} ===")
        t3 = min(remaining * 0.12, 120.0)
        a_out, _ = coord_polish(_best_seq, t3, verbose=True, seed=42)
        try_update_global(a_out, f"coord1_n{n_cur}")

    # === Phase 4: LP-FW round 2 (warm-started) ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 60:
        n_cur = len(_best_seq)
        print(f"\n=== Phase 4: LP-FW r2 n={n_cur} ===")
        t4 = min(remaining * 0.20, 250.0)
        a_out, _ = lp_frank_wolfe(_best_seq, t4, verbose=True, solver_name=solver_name)
        try_update_global(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 80:
        n_best = len(_best_seq)
        print(f"\n=== Phase 5: Length exploration ===")
        test_ns = []
        for delta in [-300, -100, 100, 300, 500]:
            nt = n_best + delta
            if 800 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:4]:
            remaining = time_budget - (time.monotonic() - T_GLOBAL)
            if remaining < 60:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(_best_seq, nt)
            t_per = min(remaining * 0.20, 150.0)
            a_out, _ = combined_iteration(a_inter, t_per, solver_name, verbose=False, seed=nt)
            try_update_global(a_out, f"combined_n{nt}")

    # === Phase 6: Multi-restart with diverse inits ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 60:
        n_best = len(_best_seq)
        print(f"\n=== Phase 6: Multi-restart n={n_best} ===")
        inits = diverse_inits(n_best, seed=99)
        for ii, a_init in enumerate(inits):
            remaining = time_budget - (time.monotonic() - T_GLOBAL)
            if remaining < 50:
                break
            print(f"  Restart {ii + 1}/{len(inits)}...")
            t_per = min(remaining * 0.18, 120.0)
            a_out, _ = combined_iteration(a_init, t_per, solver_name, verbose=False, seed=100 + ii)
            try_update_global(a_out, f"restart{ii}_n{n_best}")

    # === Phase 7: Final intensive LP-FW at best length ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 60:
        n_best = len(_best_seq)
        print(f"\n=== Phase 7: Final LP-FW n={n_best} ===")
        t7 = min(remaining * 0.40, 300.0)
        a_out, _ = lp_frank_wolfe(_best_seq, t7, verbose=True, solver_name=solver_name)
        try_update_global(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 30:
        n_best = len(_best_seq)
        print(f"\n=== Phase 8: Final coord n={n_best} ===")
        t8 = min(remaining * 0.5, 150.0)
        a_out, _ = coord_polish(_best_seq, t8, verbose=True, seed=777)
        try_update_global(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Last LP if time remains ===
    remaining = time_budget - (time.monotonic() - T_GLOBAL)
    if remaining > 30:
        n_best = len(_best_seq)
        print(f"\n=== Phase 9: Last LP n={n_best} ===")
        t9 = min(remaining * 0.7, 200.0)
        a_out, _ = lp_frank_wolfe(_best_seq, t9, verbose=True, solver_name=solver_name)
        try_update_global(a_out, f"lp_last_n{n_best}")

    # Final validation
    if _best_seq is not None:
        fs = evaluate_safe(_best_seq)
        if np.isfinite(fs):
            _best_score = fs
        print(f"\nFINAL: score={_best_score:.6f}, n={len(_best_seq)}, "
              f"time={time.monotonic() - T_GLOBAL:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in _best_seq]
