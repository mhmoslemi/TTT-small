```python
def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) * target_sum / n
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-lag exact subgradient with Nesterov momentum and adaptive stepping
2. Reduced-constraint LP Frank-Wolfe (top lags only for speed)
3. Peak-aware coordinate descent with contribution scoring
4. Spectral perturbation for escaping local optima
5. Adaptive multi-length exploration with interpolation
6. Iterative round-based time management for optimal resource allocation
"""

import numpy as np
import time
import math

def fast_autocorr(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_at_lag(a, lag):
    """Gradient of c[lag] = sum_i a[i]*a[lag-i] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, lag - n + 1)
    j_hi = min(n, lag + 1)
    if j_lo >= j_hi:
        return g
    idx = np.arange(j_lo, j_hi)
    src = lag - idx
    g[idx] = 2.0 * a[src]
    return g

def compute_weighted_gradient(a, lags, weights):
    """Weighted sum of gradients at multiple lags."""
    n = len(a)
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        lag = int(lag)
        j_lo = max(0, lag - n + 1)
        j_hi = min(n, lag + 1)
        if j_lo >= j_hi:
            continue
        idx = np.arange(j_lo, j_hi)
        src = lag - idx
        g[idx] += 2.0 * w * a[src]
    return g

def get_top_lags(c, k):
    """Return indices of top k values in autocorrelation, sorted descending."""
    k = min(k, len(c))
    return np.argsort(c)[::-1][:k]

def softmax_weights(values, temperature=1.0):
    """Softmax weights over values."""
    shifted = values - values[0]
    w = np.exp(shifted / temperature)
    s = w.sum()
    if s < 1e-30:
        return np.ones(len(w)) / len(w)
    return w / s

def nesterov_momentum(t, gamma=1.0):
    """Nesterov momentum coefficient."""
    m_prev = (t - 1.0 + gamma) / (t + gamma) if t > 1 else 0.0
    return m_prev

def subgradient_optimize(a0, time_limit, top_k_start=10, top_k_end=40,
                         verbose=True, seed=42, use_momentum=True):
    """Multi-lag exact subgradient with Nesterov momentum and adaptive step."""
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 800
    iteration = 0
    eta = 0.1

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(x)
        M_curr = c.max()

        # Adaptive top_k: start focused, widen over time
        frac = min(1.0, iteration / 3000.0)
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        top_idx = get_top_lags(c, cur_top_k)
        top_vals = c[top_idx]

        # Temperature decreases over time (sharper focus on peak)
        temp = 0.01 * top_vals[0] * (1.0 + 2.0 * (1.0 - frac))
        weights = softmax_weights(top_vals, temp)

        # Compute subgradient
        g = compute_weighted_gradient(x, top_idx, weights)

        # Nesterov extrapolation
        if use_momentum and iteration > 1:
            beta_t = (iteration - 1.0 + 1.0) / (iteration + 1.0)
            y = x + beta_t * (x - x_prev)
            y = project_simplex(y, 1.0)
        else:
            y = x.copy()

        # Gradient step with adaptive eta
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15

        # Backtracking line search
        a_trial = project_simplex(y - eta * g, 1.0)
        M_trial = fast_autocorr(a_trial).max()
        attempts = 0
        while M_trial >= M_curr - 1e-15 and attempts < 8:
            eta *= 0.5
            a_trial = project_simplex(y - eta * g, 1.0)
            M_trial = fast_autocorr(a_trial).max()
            attempts += 1

        if M_trial < M_curr - 1e-15:
            x_prev = x.copy()
            x = a_trial
            eta = min(eta * 1.1, 0.5)  # slightly increase if successful
        else:
            eta *= 0.7

        if M_trial < best_M:
            best_M = M_trial
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        iteration += 1

        # Periodic restart
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.15)
            noise_scale = 0.002 + 0.003 * (max_stall / 800.0)
            noise = rng.normal(0, noise_scale, n) * best_a
            x = project_simplex(best_a + noise, 1.0)
            x_prev = x.copy()
            eta = 0.1

        if verbose and iteration % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      SG iter={iteration} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, top_k={cur_top_k}")

    return best_a, best_M

def lp_frank_wolfe_reduced(a0, time_limit, max_constraints=80, verbose=True):
    """LP Frank-Wolfe using only top constraints for speed."""
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()
    iteration = 0

    # Try to use a fast LP solver
    try:
        import cvxpy as cp
        solver = None
        for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                x_test = cp.Variable(4, nonneg=True)
                prob_test = cp.Problem(cp.Minimize(cp.sum(x_test)), [cp.sum(x_test) == 1])
                prob_test.solve(solver=s)
                if prob_test.status in ('optimal', 'optimal_inaccurate'):
                    solver = s
                    break
            except Exception:
                continue
        if solver is None:
            solver = cp.ECOS
        print(f"      LP solver: {solver}")
    except ImportError:
        print("      cvxpy not available, skipping LP-FW")
        return best_a, best_M

    import cvxpy as cp

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(y)
        M_curr = c.max()
        if M_curr < 1e-14:
            break

        # Use only top constraints
        n_con = min(max_constraints, 2 * n - 1)
        top_idx = get_top_lags(c, n_con)
        M_threshold = c[top_idx[-1]]  # Use the lowest of the top constraints

        # Build LP: maximize sum(b) s.t. (y*b)[k] <= M_threshold for k in top_idx, b >= 0
        b_var = cp.Variable(n, nonneg=True)

        constraints = []
        for k in top_idx:
            k_int = int(k)
            # (y*b)[k] = sum_i y[i] * b[k-i] for valid i
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n, k_int + 1)
            if j_lo >= j_hi:
                continue
            idx = np.arange(j_lo, j_hi)
            src = k_int - idx
            # constraint: sum_{j in range} y[src] * b[j] <= M_threshold
            coeff = y[src]
            constraints.append(coeff @ b_var[idx] <= M_threshold)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov FW update
        alpha = 2.0 / (iteration + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = project_simplex(y_new, 1.0)

        beta_t = (iteration - 1.0 + 1.0) / (iteration + 1.0) if iteration > 1 else 0.0
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = project_simplex(x_mom, 1.0)

        y = y_new
        iteration += 1

        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_M - 1e-15:
                best_M = mc
                best_a = candidate.copy()

        if verbose and iteration % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      LP-FW iter={iteration} ({elapsed:.0f}s): J={J:.6f}, n_con={n_con}")

    return best_a, best_M

def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Peak-aware coordinate descent: move mass from high-contribution to low-impact positions."""
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        M_curr = c.max()
        # Top peaks
        n_peaks = min(15, len(c))
        top_idx = get_top_lags(c, n_peaks)

        # Compute peak contribution per coordinate
        peak_contrib = np.zeros(n)
        for k in top_idx[:8]:
            k_int = int(k)
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n, k_int + 1)
            if j_lo >= j_hi:
                continue
            idx = np.arange(j_lo, j_hi)
            src = k_int - idx
            contrib = a[idx] * a[src]
            np.add.at(peak_contrib, idx, contrib)

        # Also compute "safety" score: how much would adding mass at j increase peaks?
        safety = np.zeros(n)
        for k in top_idx[:8]:
            k_int = int(k)
            for j in range(n):
                d = k_int - j
                if 0 <= d < n:
                    safety[j] += a[d]
                d2 = j - k_int
                if 0 <= d2 < n and d2 != d:
                    safety[j] += a[d2]

        # Find high-contribution sources and low-safety destinations
        sources = np.argsort(peak_contrib)[::-1][:min(50, n)]
        # Sample destinations from low-safety region
        n_sample = min(n, 300)
        if n > n_sample:
            # Weighted sampling favoring low safety
            w = 1.0 / (safety + 1e-10)
            w /= w.sum()
            dsts = rng.choice(n, n_sample, replace=False, p=w)
        else:
            dsts = np.argsort(safety)[:n_sample]

        improved = False
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Try best few destinations
            best_dst = -1
            best_M_trial = M_curr
            n_try = min(20, len(dsts))
            for dst in dsts[:n_try]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = project_simplex(a_trial, 1.0)
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M_trial:
                        best_M_trial = M_trial
                        best_dst = dst
                        best_amt = amt
                        break

            if best_dst >= 0 and best_M_trial < M_curr - 1e-15:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = project_simplex(a, 1.0)
                M_curr = fast_autocorr(a).max()
                if M_curr < best_M:
                    best_M = M_curr
                    best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = project_simplex(a, 1.0)
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = project_simplex(a, 1.0)

        iteration += 1
        if stall > max_stall:
            break

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      CD iter={iteration} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_M

def spectral_perturb(a0, strength=0.1, seed=0):
    """Perturb in frequency domain to escape local minima."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    # FFT
    A = np.fft.rfft(a, 2 * n)
    # Random phase perturbation (preserves magnitude roughly)
    phase_noise = rng.normal(0, strength * 0.1, len(A))
    A_mag = np.abs(A)
    A_phase = np.angle(A) + phase_noise
    A_new = A_mag * np.exp(1j * A_phase)
    # IFFT
    a_new = np.fft.irfft(A_new, 2 * n)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0
    return project_simplex(a_new, 1.0)

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def get_parent_sequence():
    """Try to load the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

    return None

def evaluate_safe_local(a):
    """Evaluate using our fast computation (matches evaluate_sequence logic)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000.0)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = fast_autocorr(a)
    M = c.max()
    return float(2.0 * n * M / (S * S))

def structured_init(n, kind=0):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 0:
        # Sinc-like (flat spectrum)
        a = np.sinc(4 * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
    elif kind == 1:
        # Bimodal
        a = np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2)
    elif kind == 2:
        # Quartic
        a = (1.0 - (2 * x - 1) ** 2) ** 4
        a = np.maximum(a, 0)
    elif kind == 3:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 4:
        # Triangle
        a = 1.0 - np.abs(2 * x - 1)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(2024)

    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe_local(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # Load parent or create default
    parent = get_parent_sequence()
    if parent is not None:
        p_norm = parent / parent.sum()
        best_seq = p_norm.copy()
        best_score = evaluate_safe_local(p_norm)
        print(f"Parent loaded: n={len(parent)}, score={best_score:.6f}")
    else:
        n_base = 2400
        best_seq = structured_init(n_base, 0)
        best_score = evaluate_safe_local(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Round-based optimization ===
    # Each round does: subgradient -> LP-FW -> coordinate polish
    # We do multiple rounds, with decreasing time per round

    n_rounds = 4
    round_times = [0.30, 0.25, 0.20, 0.20]  # fractions of budget
    # Reserve 5% for length exploration and final polish
    round_times = [t * 0.92 for t in round_times]

    round = 0
    for round_frac in round_times:
        if remaining() < 60:
            break
        t_round = min(remaining() * 0.45, time_budget * round_frac)
        if t_round < 30:
            break

        n_cur = len(best_seq)
        print(f"\n=== Round {round+1}: n={n_cur}, time={t_round:.0f}s ===")

        # Phase A: Subgradient (50% of round)
        t_sg = t_round * 0.50
        if t_sg > 20 and remaining() > 40:
            a_in = project_simplex(best_seq, 1.0)
            a_out, _ = subgradient_optimize(a_in, min(t_sg, remaining() - 30),
                                            top_k_start=8, top_k_end=35,
                                            verbose=True, seed=100 + round)
            try_update(a_out, f"sg_r{round+1}")

        # Phase B: LP-FW (30% of round)
        t_lp = t_round * 0.30
        if t_lp > 20 and remaining() > 40 and len(best_seq) <= 3000:
            a_in = project_simplex(best_seq, 1.0)
            a_out, _ = lp_frank_wolfe_reduced(a_in, min(t_lp, remaining() - 30),
                                               max_constraints=60 + 20 * round,
                                               verbose=True)
            try_update(a_out, f"lpfw_r{round+1}")

        # Phase C: Coordinate polish (20% of round)
        t_cd = t_round * 0.20
        if t_cd > 15 and remaining() > 30:
            a_in = project_simplex(best_seq, 1.0)
            a_out, _ = coordinate_polish(a_in, min(t_cd, remaining() - 20),
                                         verbose=False, seed=200 + round)
            try_update(a_out, f"cd_r{round+1}")

        # Spectral perturbation + quick re-optimize
        if remaining() > 60:
            a_pert = spectral_perturb(best_seq, strength=0.15 + 0.1 * round,
                                      seed=500 + round)
            try_update(a_pert, f"spec_r{round+1}")
            if remaining() > 40:
                t_quick = min(30.0, remaining() * 0.1)
                a_out, _ = subgradient_optimize(project_simplex(a_pert, 1.0),
                                                 t_quick, top_k_start=10,
                                                 top_k_end=25, verbose=False,
                                                 seed=600 + round)
                try_update(a_out, f"spec_sg_r{round+1}")

        round += 1

    # === Length exploration ===
    if remaining() > 100:
        n_best = len(best_seq)
        for delta in [200, 400, 600]:
            n_try = n_best + delta
            if n_try > 4000:
                break
            if remaining() < 80:
                break
            print(f"\n=== Length exploration: n={n_try} ===")
            a_interp = interpolate_to_length(best_seq, n_try)
            t_explore = min(remaining() * 0.25, 150.0)
            if t_explore < 30:
                break
            a_out, _ = subgradient_optimize(project_simplex(a_interp, 1.0),
                                             min(t_explore, remaining() - 30),
                                             top_k_start=8, top_k_end=30,
                                             verbose=True, seed=700 + delta)
            try_update(a_out, f"interp_n{n_try}")
            if remaining() > 30:
                a_out2, _ = coordinate_polish(project_simplex(best_seq, 1.0),
                                               min(30.0, remaining() * 0.15),
                                               verbose=False, seed=800 + delta)
                try_update(a_out2, f"cd_n{n_try}")

    # === Final intensive polish ===
    if remaining() > 40:
        n_final = len(best_seq)
        print(f"\n=== Final polish: n={n_final} ===")
        t_final = remaining() * 0.5
        if t_final > 20:
            a_in = project_simplex(best_seq, 1.0)
            a_out, _ = subgradient_optimize(a_in, min(t_final, remaining() - 20),
                                            top_k_start=5, top_k_end=20,
                                            verbose=True, seed=999)
            try_update(a_out, "final_sg")

        if remaining() > 20:
            a_in = project_simplex(best_seq, 1.0)
            a_out, _ = coordinate_polish(a_in, min(remaining() * 0.4, 60.0),
                                         verbose=True, seed=1111)
            try_update(a_out, "final_cd")

    # Final validation using the actual evaluate_sequence
    if best_seq is not None:
        final_list = [float(x) for x in best_seq]
        try:
            actual_score = evaluate_sequence(final_list)
            if np.isfinite(actual_score):
                if actual_score != best_score:
                    print(f"Validation: local={best_score:.6f}, actual={actual_score:.6f}")
                best_score = actual_score
        except Exception as e:
            print(f"Validation error: {e}")

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
