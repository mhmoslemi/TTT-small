def _fast_autocorr(a):
    """Compute full linear autocorrelation (self-convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

def _norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_max_and_grad(a, beta, top_k):
    """Compute smoothed max of autocorrelation and its gradient."""
    ac = _fast_autocorr(a)
    num_lags = len(ac)
    k_eff = min(top_k, num_lags)
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(len(a))
    n = len(a)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, ac.max()


def _apg_with_backtracking(a0, time_limit, beta_start=30.0, beta_end=8.0,
                            top_k_start=100, top_k_end=300, seed=42):
    """
    Nesterov-accelerated projected gradient with backtracking line search.
    Minimizes max_k (a*a)[k] on the simplex.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    c0 = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 10000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = _project_simplex(y)

        # Compute gradient
        g, _ = _smoothed_max_and_grad(y, cur_beta, cur_top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        # Backtracking line search
        f_curr = _fast_autocorr(y).max()
        eta = c0 / np.sqrt(t + 1.0)
        improved = False
        for bt in range(20):
            x_new = _project_simplex(y - eta * g)
            f_new = _fast_autocorr(x_new).max()
            # Armijo condition (using exact max)
            if f_new <= f_curr - 0.01 * eta * gn * gn:
                improved = True
                break
            eta *= 0.5

        if not improved:
            # Just take the step anyway
            x_new = _project_simplex(y - eta * g)

        x_prev = x
        x = x_new

        M_new = _fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re

    return best_a, best_m

    return np.fft.irfft(A * A, n=fft_size)[:size]


def _eval_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_mask = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(rho_mask)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_grad(a, beta, top_k):
    ac = _fast_autocorr(a)
    num_lags = len(ac)
    k_eff = min(top_k, num_lags)
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        j_idx = np.arange(j_start, j_end)
        paired = k_int - j_idx
        grad[j_idx] += 2.0 * wi * a[paired]
    return grad, ac.max()


def _apg_backtrack(a0, time_limit, beta_start=30.0, beta_end=6.0,
                   top_k_start=100, top_k_end=400, seed=42, verbose=False):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 4000

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 8000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = _proj_simplex(y)

        g, _ = _smooth_grad(y, cur_beta, cur_top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break

        f_curr = _fast_autocorr(y).max()
        eta = 1.0 / np.sqrt(t + 1.0)
        x_new = None
        for bt in range(25):
            cand = _proj_simplex(y - eta * g)
            f_cand = _fast_autocorr(cand).max()
            if f_cand <= f_curr - 0.005 * eta * gn * gn:
                x_new = cand
                break
            eta *= 0.5
        if x_new is None:
            x_new = _proj_simplex(y - eta * g)

        x_prev = x
        x = x_new

        M_new = _fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise_scale = 0.002 * (1.0 + stall * 0.001)
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-14)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      APG restart t={t} ({elapsed:.0f}s): M={best_m:.10f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      APG t={t} ({elapsed:.0f}s): M={best_m:.10f}, beta={cur_beta:.1f}")

    return best_a, best_m


def _build_lp_matrix(y):
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def _lp_fw(a0, time_limit, solver, top_k_constraints=200, verbose=True):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()
    n_total = 2 * n - 1
    K = min(top_k_constraints, n_total)

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        top_c = np.argsort(ac)[::-1][:K]
        T = _build_lp_matrix(y)
        T_sub = T[top_c]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sub @ b_var <= M_curr, cp.sum(b_var) == 1.0]
        prob = cp.Problem(cp.Maximize(0), constraints)

        # Actually, the LP FW direction: maximize sum(b) s.t. T@b <= M, b >= 0
        # But we can normalize: maximize 0 is wrong. Let's maximize sum(b).
        # With sum(b) == 1 constraint, the direction is just b itself.
        # The standard FW: find s in feasible set that minimizes <g, s>.
        # Here g = gradient of max, and the feasible set is {b >= 0, sum(b)=1, T@b <= M}.
        # The FW direction is argmin_{b in S} <grad, b> = argmin sum grad[i]*b[i]
        # But we're using the "linearization" approach: the LP gives the next vertex.

        # Let me redo this properly:
        # The FW step: find s = argmin_{b in S} <f'(y), b> where S = {b >= 0, sum(b)=1, (y*b)[k] <= M for all k}
        # The gradient f'(y) has components: f'(y)[i] = sum over active k of 2*(y*b_pattern)
        # For the max function, f'(y)[i] = max over active k of (contribution of y[i] to that k)
        # This is complex. Instead, use the standard LP approach:
        # max sum(b) s.t. T @ b <= M, b >= 0
        # The solution gives the FW direction (normalized).

        b_var2 = cp.Variable(n, nonneg=True)
        constraints2 = [T_sub @ b_var2 <= M_curr]
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        s_dir = b_val / b_sum

        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * s_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx

        y = y_new

        for candidate in [y, x_mom]:
            mc = _fast_autocorr(candidate).max()
            if mc < best_m - 1e-16:
                best_m = mc
                best_a = candidate.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * mc
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): obj={obj:.8f}")

    return best_a, best_m


def _coord_descent(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:15]

        # Find heavy contributors to top lags
        heavy = np.zeros(n)
        for k in top_lags[:8]:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            j_idx = np.arange(j_start, j_end)
            paired = k_int - j_idx
            contrib = a[j_idx] * a[paired]
            heavy[j_idx] += contrib

        # Light positions: low autocorrelation contribution
        light_scores = np.zeros(n)
        for k in top_lags[:8]:
            k_int = int(k)
            for j in range(n):
                dj = k_int - j
                if 0 <= dj < n:
                    light_scores[j] += a[dj]
                dj2 = j - k_int
                if 0 <= dj2 < n:
                    light_scores[j] += a[dj2]

        heavy_idx = np.argsort(heavy)[::-1][:40]
        light_idx = np.argsort(light_scores)[:40]

        improved = False
        for src in heavy_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in light_idx:
                if dst == src:
                    continue
                # Try several transfer amounts
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = _fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if time.monotonic() - t0 > time_limit:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i_idx = int(rng.integers(n))
            j_idx = int(rng.integers(n))
            if i_idx != j_idx and a[i_idx] > 1e-8:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i_idx] * frac
                a[i_idx] -= amt
                a[j_idx] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_ac = _fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i_idx] += amt
                        a[j_idx] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): obj={obj:.8f}, stall={stall}")

    return best_a, best_m


def _interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    rng = np.random.default_rng(2024)

    solver = _detect_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = _get_parent()
    if parent is not None:
        p_norm = _norm1(parent)
        p_score = _eval_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.8f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    def _remaining():
        return time_budget - (time.monotonic() - t_global)

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _eval_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # === Phase 1: APG at parent length (main workhorse, 35%) ===
    rem = _remaining()
    print(f"\n=== Phase 1: APG n={n0} ===")
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        a_in = _norm1(best_seq)
        a_out, _ = _apg_backtrack(
            a_in, t1, beta_start=35.0, beta_end=6.0,
            top_k_start=100, top_k_end=400, seed=42, verbose=True)
        _try_update(a_out, f"apg_n{n0}")

    # === Phase 2: LP Frank-Wolfe (30%) ===
    rem = _remaining()
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-FW n={n_cur} ===")
    t2 = min(rem * 0.30, 450.0)
    if t2 > 30 and n_cur <= 3500:
        a_in = _norm1(best_seq)
        a_out, _ = _lp_fw(a_in, t2, solver, top_k_constraints=200, verbose=True)
        _try_update(a_out, f"lpfw_n{n_cur}")

    # === Phase 3: APG polish (10%) ===
    rem = _remaining()
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: APG polish n={n_cur} ===")
    t3 = min(rem * 0.10, 120.0)
    if t3 > 20:
        a_in = _norm1(best_seq)
        a_out, _ = _apg_backtrack(
            a_in, t3, beta_start=20.0, beta_end=8.0,
            top_k_start=200, top_k_end=500, seed=77, verbose=False)
        _try_update(a_out, f"apg_polish_n{n_cur}")

    # === Phase 4: Coordinate descent (10%) ===
    rem = _remaining()
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: CD n={n_cur} ===")
    t4 = min(rem * 0.10, 120.0)
    if t4 > 15:
        a_in = _norm1(best_seq)
        a_out, _ = _coord_descent(a_in, t4, seed=42, verbose=True)
        _try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: Try longer length (8%) ===
    rem = _remaining()
    n_cur = len(best_seq)
    n1 = min(n_cur + 400, 3200)
    if n1 > n_cur and rem > 100:
        print(f"\n=== Phase 5: Scale to n={n1} ===")
        a_long = _interpolate_to(best_seq, n1)
        a_in = _norm1(a_long)
        t5 = min(rem * 0.08, 100.0)
        if t5 > 20:
            a_out, _ = _apg_backtrack(
                a_in, t5, beta_start=25.0, beta_end=8.0,
                top_k_start=150, top_k_end=350, seed=99, verbose=False)
            _try_update(a_out, f"apg_n{n1}")
            rem = _remaining()
            if rem > 40:
                a_out2, _ = _coord_descent(a_out, min(rem * 0.5, 50.0), seed=55, verbose=False)
                _try_update(a_out2, f"coord_n{n1}")

    # === Phase 6: Try shorter length (5%) ===
    rem = _remaining()
    if rem > 80:
        n2 = max(len(best_seq) - 400, 1600)
        if n2 < len(best_seq):
            print(f"\n=== Phase 6: Try n={n2} ===")
            a_short = _interpolate_to(best_seq, n2)
            a_in = _norm1(a_short)
            t6 = min(rem * 0.05, 70.0)
            if t6 > 20:
                a_out, _ = _apg_backtrack(
                    a_in, t6, beta_start=25.0, beta_end=8.0,
                    top_k_start=100, top_k_end=300, seed=111, verbose=False)
                _try_update(a_out, f"apg_n{n2}")

    # === Phase 7: Final LP at best length (remaining) ===
    rem = _remaining()
    n_best = len(best_seq)
    if n_best <= 3000 and rem > 40:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(rem * 0.5, 100.0)
        if t7 > 20:
            a_in = _norm1(best_seq)
            a_out, _ = _lp_fw(a_in, t7, solver, top_k_constraints=300, verbose=True)
            _try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    rem = _remaining()
    n_best = len(best_seq)
    if rem > 20:
        print(f"\n=== Phase 8: Final coord n={n_best} ===")
        t8 = min(rem * 0.7, 60.0)
        if t8 > 10:
            a_in = _norm1(best_seq)
            a_out, _ = _coord_descent(a_in, t8, seed=314, verbose=True)
            _try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
