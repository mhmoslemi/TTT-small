def fast_autocorr(a):

    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_gradient(a, beta):
    """Compute smoothed gradient using logsumexp."""
    n = len(a)
    ac = fast_autocorr(a)
    # Shift for numerical stability
    ac_shifted = ac - ac.max()
    weights = np.exp(beta * ac_shifted)
    weights /= weights.sum()

    # g[j] = 2 * sum_k weights[k] * a[k-j] * I(0 <= k-j < n)
    # Let m = k-j, so k = j+m: g[j] = 2 * sum_m weights[j+m] * a[m] * I(0<=j+m<2n-1, 0<=m<n)
    # This is (2 * weights_reversed) convolved with a, then...
    
    # Actually, let me think more carefully.
    # g[j] = 2 * sum_{k: 0<=k-j<n} weights[k] * a[k-j]
    # Let m = k-j: g[j] = 2 * sum_{m: 0<=m<n, 0<=j+m<2n-1} weights[j+m] * a[m]
    
    # This is the cross-correlation of a with weights, evaluated at... 
    # Actually it's: g = 2 * Ifft(conj(fft(a)) * fft(weights_padded))
    # Let me just do it directly for clarity (or use FFT)
    
    # g[j] = 2 * sum_m a[m] * weights[j+m] for valid j+m
    # This is the cross-correlation of a and weights: (a ⋆ weights)[j]
    # In FFT terms: Ifft(conj(fft(a)) * fft(weights))
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    # Cross-correlation: Ifft(conj(A) * W)
    corr = np.fft.irfft(np.conj(A) * W, fft_size)
    g = 2.0 * corr[:n]
    
    return g, ac.max()

def smooth_mirror_descent(a0, time_limit, beta_start=5.0, beta_end=500.0,

    a = norm_sum1(a0)

    x = a.copy()
    x_prev = a.copy()

    # We'll increase beta over iterations
    beta = beta_start
    beta_growth = (beta_end / beta_start) ** (1.0 / max_iters)
    
    # Backtracking parameters
    c_bt = 0.01  # sufficient decrease
    rho = 0.5    # step size reduction
    
    # Initial step size estimate
    eta = 1.0 / np.sqrt(n)
    
    stall = 0
    max_stall = 2000
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute smoothed gradient
        g, M_curr = smooth_gradient(y, beta)
        
        # Greedy mirror descent step with backtracking
        # We want to minimize f(y * exp(-eta * g)) approximately
        # In log space: new_log = log(y) - eta * g
        # For backtracking, we try decreasing eta until we get improvement
        
        # First, compute a baseline
        f_curr = M_curr  # approximate (using smoothed value)
        
        eta_try = eta
        improved = False
        for _ in range(10):
            log_y = np.log(np.maximum(y, 1e-15))
            log_new = log_y - eta_try * g
            log_new -= log_new.max()  # normalize
            a_new = np.exp(log_new)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()

                a_new = a_new / s_new
            
            # Evaluate
            M_new = fast_autocorr(a_new).max()
            if M_new < f_curr - c_bt * np.linalg.norm(g) * eta_try:

                break

        if not improved:
            # Take a small step anyway
            log_y = np.log(np.maximum(y, 1e-15))
            log_new = log_y - eta_try * g
            log_new -= log_new.max()

                a_new = y.copy()
            else:
                a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Update best
        M_x = fast_autocorr(x).max()
        if M_x < best_obj - 1e-15:
            best_obj = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Also check y
        M_y = fast_autocorr(y).max()
        if M_y < best_obj - 1e-15:
            best_obj = M_y
            best_a = y.copy()

def gradient_coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):

    Coordinate descent guided by the smoothed gradient.
    Moves mass from positions with high gradient to positions with low gradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    stall = 0
    max_stall = 200
    beta = 20.0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        g, M_curr = smooth_gradient(a, beta)
        peak_k = np.argmax(fast_autocorr(a))
        
        # Positions with high gradient are "bad" - they contribute to peaks
        # Positions with low gradient are "good" - they don't contribute much
        # Move mass from high-g to low-g positions
        
        # Sort positions by gradient
        g_sorted = np.argsort(g)
        
        # Try moving mass from top-gradient positions to bottom-gradient positions
        n_src = min(50, n // 4)
        n_dst = min(50, n // 4)
        
        src_candidates = g_sorted[-n_src:]  # highest gradient
        dst_candidates = g_sorted[:n_dst]    # lowest gradient
        
        improved = False
        
        # Shuffle for diversity
        rng.shuffle(src_candidates)
        rng.shuffle(dst_candidates)
        
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                if a[dst] > a[src] * 1.5:
                    continue
                
                # Try several transfer fractions
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Try random perturbation
            n_perturb = max(5, n // 50)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
            
            # Increase smoothing
            beta = min(beta * 1.2, 100.0)
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GCD ({elapsed:.1f}s): J={J:.6f}, stall={stall}, beta={beta:.1f}")
    
    return best_a, best_m

def adam_mirror_descent(a0, time_limit, verbose=True, rng_seed=77):
    """
    Adam-style adaptive mirror descent with smoothed gradient.
    Uses per-coordinate learning rates based on gradient history.

    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    lr = 1.0 / np.sqrt(n)
    beta_smooth = 10.0
    
    x = a.copy()
    x_prev = a.copy()
    
    stall = 0

            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)

        else:
            y = y / sy
        
        # Smoothed gradient
        g, M_curr = smooth_gradient(y, beta_smooth)
        
        # Update Adam moments

def exact_subgradient_step(a, threshold_frac=0.999):
    """Compute exact subgradient from active set."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    # Average over active set
    g /= len(active)
    return g

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, seed=0):
    """Create structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    choices = [
        # Pure sine
        np.sin(np.pi * x) ** 2,
        # Sine + noise
        np.sin(np.pi * x) ** 2 + 0.1 * rng.random(n),
        # Bump function
        np.exp(-10 * (x - 0.5) ** 2),
        # Double bump
        np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2),
        # Cosine window
        0.5 * (1 + np.cos(2 * np.pi * x)),
        # Sine with harmonics
        np.sin(np.pi * x) ** 2 + 0.3 * np.sin(3 * np.pi * x) ** 2 + 0.1 * np.sin(5 * np.pi * x) ** 2,
        # Power law
        x ** 0.5 * (1 - x) ** 0.5,
        # Bernoulli conv inspired
        np.maximum(np.sin(2 * np.pi * x * 3) * np.sin(np.pi * x), 0),
    ]
    
    a = choices[seed % len(choices)].copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_gradient(a, beta):
    """Compute smoothed gradient using logsumexp approximation to max."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]

    ac_shifted = ac - ac.max()
    weights = np.exp(beta * ac_shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(size) / size
    else:
        weights /= wsum

    W = np.fft.rfft(weights, fft_size)
    corr = np.fft.irfft(np.conj(A) * W, fft_size)
    g = 2.0 * corr[:n]
    return g, ac.max()


def exact_subgradient(a, threshold_frac=0.999):
    """Compute exact subgradient from active set of near-max lags."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    g /= len(active)
    return g


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def adam_mirror_descent(a0, time_limit, verbose=True, rng_seed=77):
    """Adam-style adaptive mirror descent with smoothed gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 2.0 / np.sqrt(n)
    beta_smooth = 10.0

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 4000

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        g, M_curr = smooth_gradient(y, beta_smooth)

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        log_y = np.log(np.maximum(y, 1e-15))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_new = log_y - step
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if t % 1000 == 0:
            beta_smooth = min(beta_smooth * 1.02, 300.0)
            lr = lr * 0.9995

        if stall > max_stall:
            stall = 0
            noise = rng.normal(0, 0.015, n) * best_a
            x_restarted = best_a + noise
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            m = np.zeros(n)
            v = np.zeros(n)
            beta_smooth = max(5.0, beta_smooth * 0.3)
            lr = max(lr, 1.0 / np.sqrt(n))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AdamMD t={t} ({elapsed:.1f}s): J={J:.6f}, beta_s={beta_smooth:.1f}, lr={lr:.5f}, stall={stall}")

    return best_a, best_m
