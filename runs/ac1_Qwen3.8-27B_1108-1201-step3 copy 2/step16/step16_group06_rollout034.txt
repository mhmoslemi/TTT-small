def peak_mirror_descent(a0, time_limit, top_k=30, beta=4.0, verbose=True):

            break

        y = x + beta_t * (x - x_prev)

        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:

        else:
            w = w / w_sum

        grad = np.zeros(n)

            if i_start >= i_end:

            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * (y[j_idx] + 0)  # subgradient: a[k-i]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()

def coord_descent_peaks(a0, time_limit, verbose=True, rng_seed=42):

                    val += a[d2]

        improved = False

def generate_init(n, kind, alpha=1.0, seed=0):
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        f = np.ones(n)
    elif kind == 'ramp':
        f = np.linspace(0.1, 1.0, n)
    elif kind == 'rev_ramp':
        f = np.linspace(1.0, 0.1, n)
    elif kind == 'power':
        f = (np.arange(n) + 1) ** alpha
    elif kind == 'exp_decay':
        f = np.exp(-alpha * x)
    elif kind == 'sine':
        f = np.sin(np.pi * x) ** 2
    elif kind == 'asym_sine':
        f = np.sin(np.pi * x ** alpha) ** 2
        f = np.maximum(f, 1e-10)
    elif kind == 'two_peak':
        f = np.exp(-alpha * (x - 0.3) ** 2) + 0.7 * np.exp(-alpha * (x - 0.8) ** 2)
    elif kind == 'random':
        rng = np.random.default_rng(seed)
        f = rng.random(n) + 0.01
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_seq(src, new_n):
    old_n = len(src)
    if old_n == new_n:
        return src.copy()
    s = src.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = src / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    for kind, alpha in inits_to_try:
        a_init = generate_init(n_base, kind, alpha)
        sc = evaluate_safe(a_init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_init.copy()
            print(f"  Init {kind}({alpha}): score={sc:.6f}")

    if parent_norm is not None:
        try_update(parent_norm, "parent")

    print(f"Best after init: score={best_score:.6f}, n={len(best_seq)}")

    n_cur = len(best_seq)
    rows, cols = build_sparsity(n_cur)

    # Phase 2: LP-FW main optimization
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 2: LP-FW n={n_cur} (main, {remaining*0.45:.0f}s) ===")
        t2 = remaining * 0.45
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, solver, rows, cols, verbose=True)
        try_update(a_out, f"lp_main_n{n_cur}")

    # Phase 3: Mirror descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Peak mirror descent n={n_cur} ===")
        t3 = min(remaining * 0.15, 200.0)
        a_out, _ = peak_mirror_descent(norm_sum1(best_seq), t3, top_k=25, beta=4.0, verbose=True)
        try_update(a_out, f"md_n{n_cur}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: Coord descent n={n_cur} ===")
        t4 = min(remaining * 0.12, 150.0)
        a_out, _ = coord_descent_peaks(norm_sum1(best_seq), t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord1_n{n_cur}")

    # Phase 5: Second LP-FW round (warm-started)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: LP-FW round 2 n={n_cur} ===")
        rows2, cols2 = build_sparsity(n_cur)
        t5 = min(remaining * 0.25, 400.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5, solver, rows2, cols2, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # Phase 6: Coordinate descent again
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 6: Coord descent round 2 n={n_cur} ===")
        t6 = min(remaining * 0.15, 120.0)
        a_out, _ = coord_descent_peaks(norm_sum1(best_seq), t6, verbose=True, rng_seed=777)
        try_update(a_out, f"coord2_n{n_cur}")

    # Phase 7: Try larger n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 7: Try larger n ===")
        for n_target in [n_best + 400, n_best + 800]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Expanding to n={n_target}...")
            a_long = interpolate_seq(best_seq, n_target)
            rows_l, cols_l = build_sparsity(n_target)
            t7a = min(remaining * 0.3, 200.0)
            a_out, _ = lp_frank_wolfe(norm_sum1(a_long), t7a, solver, rows_l, cols_l, verbose=False)
            try_update(a_out, f"lp_long_n{n_target}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t7b = min(remaining * 0.2, 100.0)
                a_out, _ = coord_descent_peaks(norm_sum1(best_seq), t7b, verbose=False, rng_seed=999)
                try_update(a_out, f"coord_long_n{n_target}")

    # Phase 8: Try smaller n
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_smaller = max(1600, n_best - 400)
        if n_smaller != n_best:
            print(f"\n=== Phase 8: Try n={n_smaller} ===")
            a_short = interpolate_seq(best_seq, n_smaller)
            rows_s, cols_s = build_sparsity(n_smaller)
            t8 = min(remaining * 0.4, 200.0)
            a_out, _ = lp_frank_wolfe(norm_sum1(a_short), t8, solver, rows_s, cols_s, verbose=False)
            try_update(a_out, f"lp_short_n{n_smaller}")

    # Phase 9: Final LP-FW squeeze
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_final = len(best_seq)
        print(f"\n=== Phase 9: Final LP-FW n={n_final} ===")
        rows_f, cols_f = build_sparsity(n_final)
        t9 = remaining * 0.5
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t9, solver, rows_f, cols_f, verbose=True)
        try_update(a_out, f"lp_final_n{n_final}")

    # Phase 10: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_final = len(best_seq)
        print(f"\n=== Phase 10: Final coord polish n={n_final} ===")
        t10 = remaining * 0.6
        a_out, _ = coord_descent_peaks(norm_sum1(best_seq), t10, verbose=True, rng_seed=2024)
        try_update(a_out, f"coord_final_n{n_final}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
