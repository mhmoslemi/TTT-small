def fast_autocorr(a):
    """Compute full convolution conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def binomial_sequence(n):
    """Generate b[i] = binom(2i,i)/4^i for i=0..n-1, normalized to sum=1."""
    b = np.zeros(n)
    for i in range(n):
        # Use log to avoid overflow for large i
        log_bi = 0.0
        for j in range(1, i + 1):
            log_bi += math.log(2 * j) - math.log(j) - math.log(4)
        b[i] = math.exp(log_bi) if log_bi > -700 else 0.0
    s = b.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return b / s


def binomial_sequence_fast(n):
    """Faster generation using recurrence: b[i+1] = b[i] * (2i+1)/(2i+2)."""
    b = np.zeros(n)
    b[0] = 1.0
    for i in range(1, n):
        b[i] = b[i - 1] * (2 * i - 1) / (2 * i)
    s = b.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return b / s


def tapered_binomial(n, beta=1.0):
    """Binomial sequence with power taper at the end: a[i] = b[i] * (1 - i/n)^beta."""
    b = binomial_sequence_fast(n)
    x = np.arange(n, dtype=np.float64) / n
    w = (1.0 - x) ** beta
    a = b * w
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, temperature):
    """Compute soft-max subgradient of max(ac) w.r.t. a."""
    n = len(a)
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        top_idx = np.argsort(ac)[::-1][:min(20, len(ac))]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0
        w_sum = weights.sum()
    weights /= w_sum

    grad = np.zeros(n)
    for k in range(len(ac)):
        w = weights[k]
        if w < 1e-10:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]
    return grad


def compute_topk_subgradient(a, ac, top_k=50, beta_power=3.0):
    """Compute subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta_power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_mirror_descent(a0, time_limit, initial_temp=0.01, final_temp=0.001,
                             c0=3.0, max_steps=3000000, verbose=True):
    """Nesterov accelerated mirror descent with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1

    for t in range(1, max_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 3000.0)
        temp = initial_temp * (1 - frac) + final_temp * frac
        temp = max(temp, 1e-5)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        if temp > 0.003:
            g = compute_subgradient(y, ac, temp)
        else:
            g = compute_topk_subgradient(y, ac, top_k=60, beta_power=4.0)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(800, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(0.5, gamma * 0.5)

        running_avg += x
        avg_count += 1
        if avg_count % 1000 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, stall={stall_count}")

    return best_a, best_m


def deficit_correction(a0, target_frac=1.0):
    """
    Solve the linear system 2*conv(a, h) = target - conv(a,a) to get
    a first-order correction direction h. This "fills deficits" in the convolution.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    S = a.sum()
    if S < 1e-15:
        return a0.copy()
    a = a / S

    ac = fast_autocorr(a)
    total = ac.sum()  # = S^2 = 1 after normalization
    target = total / len(ac)  # uniform target

    # We want: 2 * conv(a, h) ≈ target - ac (elementwise)
    # In matrix form: 2 * T @ h = (target - ac)
    # where T is the "convolution with a" operator
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a (2n-1) x n system (overdetermined). Solve in least squares.

    # Build T as a circulant-like matrix (but truncated)
    # T has shape (2n-1, n)
    # T[k, j] = a[k-j] if 0 <= k-j < n else 0
    # For efficiency, use FFT: conv(a, h) = ifft(fft(a) * fft(h))

    # Actually, let's use the fact that T @ h = conv(a, h)
    # We can solve the least squares problem:
    # min_h ||2*T@h - (target - ac)||^2 + lambda*||h||^2
    # Using the normal equations: (T^T T + lambda*I) @ h = T^T @ (target - ac) / 2

    # T^T T is an n x n matrix. Let's compute it.
    # (T^T T)[i, j] = sum_k T[k,i] * T[k,j] = sum_k a[k-i] * a[k-j] (valid k)
    # = ac[|i-j| + (i-j)] ... actually this is autocorrelation of a
    # (T^T T)[i,j] = sum_{k: 0<=k-i<n, 0<=k-j<n} a[k-i]*a[k-j]
    # Let u = k-i, then k = u+i, and we need 0 <= u+i < 2n-1 and 0 <= u+i-j < n
    # = sum_u a[u] * a[u + i - j] (with bounds)
    # This is essentially the autocorrelation of a convolved... 

    # Actually, T^T T is the Gram matrix of the convolution operator.
    # For a circulant (periodic) case, T^T T would be diagonalizable by FFT.
    # For the truncated case, it's close to circulant.

    # Simpler approach: just use the FFT-based solution.
    # conv(a, h) = ifft(fft(a) * fft(h, 2n-1))
    # We want: ifft(FFT_a * FFT_h) * 2 ≈ target - ac
    # In frequency domain: FFT_a * FFT_h ≈ fft(target - ac) / 2
    # FFT_h ≈ fft(target - ac) / (2 * FFT_a)

    fft_size = 1 << (2 * n - 1).bit_length()
    FFT_a = np.fft.fft(a, fft_size)
    r = (target - ac)
    r = np.append(r, np.zeros(fft_size - len(r)))
    FFT_r = np.fft.fft(r)

    # Avoid division by zero
    denom = 2.0 * FFT_a
    mask = np.abs(denom) > 1e-10
    FFT_h = np.zeros(fft_size, dtype=np.complex128)
    FFT_h[mask] = FFT_r[mask] / denom[mask]

    h = np.real(np.fft.ifft(FFT_h))[:n]
    h = np.maximum(h, 0.0)

    # The correction direction is h (we add it to a)
    # But we need to normalize. Let's compute the new sequence.
    a_new = a + h
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    a_new = a_new / s_new
    return a_new


def deficit_correction_iterative(a0, n_iter=5, verbose=False):
    """Apply deficit correction iteratively with damping."""
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    for it in range(n_iter):
        a_corr = deficit_correction(a)
        # Damped update
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - alpha) * a + alpha * a_corr
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_t = fast_autocorr(a_trial).max()
            if m_t < best_m - 1e-15:
                a = a_trial
                best_m = m_t
                best_a = a.copy()
                break
        if verbose:
            print(f"      Deficit corr iter {it}: M={best_m:.8f}, score={2*len(a)*best_m:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, max_constraints=150, verbose=True):
    """LP Frank-Wolfe with active constraint selection."""
    if solver is None:
        return a0, fast_autocorr(norm_sum1(a0)).max()

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 3000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Select active constraints
        threshold = M_curr * 0.97
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]

        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m - 1e-15:
            best_m = my
            best_a = y.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with random candidate moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_moves = min(100, n // 2)

        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                continue

            for direction in [1, -1]:
                src, dst = (i, j) if direction == 1 else (j, i)
                if a[src] < 1e-12:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1
            # Escape: random perturbation
            idx = rng.integers(0, n, 3)
            for i in idx:
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def multi_start_nmd(a0, time_limit, n_starts=3, verbose=True):
    """Run NMD from multiple perturbed starting points."""
    n = len(a0)
    rng = np.random.default_rng(777)
    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()
    remaining = time_limit

    for s in range(n_starts):
        t_rem = remaining - (time.monotonic() - t0)
        if t_rem < 30:
            break
        t_alloc = t_rem / (n_starts - s) * 0.9

        # Perturb
        scale = 0.01 * (s + 1)
        pert = rng.normal(0, scale, n)
        a_start = a0 * (1 + pert)
        a_start = np.maximum(a_start, 1e-10)
        a_start = a_start / a_start.sum()

        a_out, m_out = nesterov_mirror_descent(a_start, t_alloc,
                                                initial_temp=0.01 + 0.005 * s,
                                                final_temp=0.001,
                                                c0=3.0 + s,
                                                verbose=(verbose and s == 0))
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                print(f"    Multi-start {s}: NEW BEST J={2*n*best_m:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    if solver is not None:
        print(f"LP solver: {solver}")
    else:
        print("No LP solver available")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent sequence ===
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    # === Generate candidate initializations ===
    candidates = []
    n_base = 2400

    if best_seq is not None:
        candidates.append(('parent', best_seq.copy()))
        n_base = len(best_seq)

    # Binomial-based initializations
    for beta in [0.0, 0.3, 0.5, 0.7, 1.0]:
        b = tapered_binomial(n_base, beta)
        candidates.append((f'binom_beta{beta}', b))

    # Deficit-corrected binomial
    b_plain = binomial_sequence_fast(n_base)
    b_corr, _ = deficit_correction_iterative(b_plain, n_iter=3, verbose=False)
    candidates.append(('binom_deficit', b_corr))

    # Evaluate all candidates
    for name, cand in candidates:
        sc = evaluate_safe(cand)
        if np.isfinite(sc):
            print(f"  Init [{name}]: score={sc:.8f}")
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    print(f"Best init: score={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: Deficit correction on best init (quick, 30s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 10:
        print(f"\n=== Phase 1: Deficit correction (budget={t1:.0f}s) ===")
        a_out, _ = deficit_correction_iterative(best_seq, n_iter=10, verbose=True)
        try_update(a_out, "deficit_init")

    # === Phase 2: Nesterov Mirror Descent (main, 40% time) ===
    remaining = time_budget - (time.monotonic() - t0_global() if False else time_budget - (time.monotonic() - t_global))
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 2: NMD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t2, initial_temp=0.015, final_temp=0.001,
                                            c0=4.0, verbose=True)
        try_update(a_out, f"NMD_{len(best_seq)}")

    # === Phase 3: Multi-start NMD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Multi-start NMD (budget={t3:.0f}s) ===")
    if t3 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_start_nmd(a_in, t3, n_starts=3, verbose=True)
        try_update(a_out, f"MultiNMD_{len(best_seq)}")

    # === Phase 4: LP Frank-Wolfe (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: LP-FW n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 60 and solver is not None:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, max_constraints=200, verbose=True)
        try_update(a_out, f"LP_{len(best_seq)}")

    # === Phase 5: NMD polish after LP (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 5: NMD polish (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t5, initial_temp=0.005, final_temp=0.0005,
                                            c0=2.0, verbose=False)
        try_update(a_out, f"Polish_{len(best_seq)}")

    # === Phase 6: Coordinate descent (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 6: CD polish (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t6, verbose=True, rng_seed=789)
        try_update(a_out, f"CD_{len(best_seq)}")

    # === Phase 7: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_alt = len(best_seq) + 300
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_long = norm_sum1(a_long)

        t7 = min(remaining * 0.4, 180.0)
        if t7 > 30:
            a_out, _ = nesterov_mirror_descent(a_long, t7, initial_temp=0.01, final_temp=0.002,
                                                c0=3.0, verbose=True)
            try_update(a_out, f"NMD_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and solver is not None:
            t7b = min(remaining * 0.4, 120.0)
            if t7b > 30:
                a_out, _ = lp_frank_wolfe(a_long, t7b, solver, max_constraints=150, verbose=False)
                try_update(a_out, f"LP_n{n_alt}")

    # === Phase 8: Try smaller length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100 and len(best_seq) > 1000:
        n_small = max(800, len(best_seq) // 2)
        print(f"\n=== Phase 8: Small length n={n_small} (budget={remaining:.0f}s) ===")
        a_small = interpolate_sequence(best_seq, n_small)
        a_small = norm_sum1(a_small)

        t8 = min(remaining * 0.5, 150.0)
        if t8 > 30:
            a_out, _ = nesterov_mirror_descent(a_small, t8, initial_temp=0.02, final_temp=0.002,
                                                c0=4.0, verbose=True)
            try_update(a_out, f"NMD_n{n_small}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.5, 80.0)
            if t8b > 20:
                a_out, _ = coordinate_descent(a_small, t8b, verbose=False, rng_seed=444)
                try_update(a_out, f"CD_n{n_small}")

    # === Phase 9: Final LP polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and solver is not None:
        print(f"\n=== Phase 9: Final LP polish n={n_best} ===")
        a_in = norm_sum1(best_seq)
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 20:
            a_out, _ = lp_frank_wolfe(a_in, t9, solver, max_constraints=100, verbose=False)
            try_update(a_out, f"LP_final_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        # Fallback
        best_seq = binomial_sequence_fast(1000)
        print(f"\nFALLBACK: n={len(best_seq)}")

    return [float(x) for x in best_seq]
