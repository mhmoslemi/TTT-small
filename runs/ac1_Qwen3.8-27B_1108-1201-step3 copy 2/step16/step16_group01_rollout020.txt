    valid = u > candidates
    if np.any(valid):
        rho = np.where(valid)[0][-1]
    else:
        rho = 0
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(x - theta, 0.0)


def project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    u = np.sort(x)[::-1]
    cv = np.cumsum(u)
    rho_candidates = (u * np.arange(1, n + 1) - (cv - 1.0)) > 0
    if np.any(rho_candidates):
        rho = np.where(rho_candidates)[0][-1]
    else:
        rho = 0
    theta = (cv[rho] - 1.0) / (rho + 1)
    proj = np.maximum(x - theta, 0.0)
    s = proj.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return proj / s


def clip_and_normalize(x):
    """Fast projection: clip to non-negative and normalize."""
    x = np.maximum(np.asarray(x, dtype=np.float64), 0.0)
    s = x.sum()
    if s < 1e-30:
        return np.ones_like(x) / len(x)
    return x / s


def optimize_gradient(a_init, time_limit, beta=5.0, use_smooth=True, verbose=True, seed=42):
    """
    Main gradient-based optimizer with backtracking line search.
    Uses smooth-max or exact subgradient.
    """
    n = len(a_init)
    a = normalize(a_init)
    ac = fast_conv(a)
    M = ac.max()
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta = 0.1
    eta_min = 1e-8
    stalling = 0
    stall_threshold = 2000
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        # Compute gradient
        if use_smooth:
            g = smooth_max_gradient(a, ac, beta)
        else:
            g = exact_subgradient(a, ac)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Gradient direction (we want to DECREASE max, so go against gradient)
        # But we need to stay on the simplex, so project the step
        direction = -g / g_norm
        
        # Try different step sizes (backtracking)
        accepted = False
        for trial in range(20):
            eta_try = eta * (0.7 ** trial)
            if eta_try < eta_min:
                break
            a_trial = clip_and_normalize(a + eta_try * direction)
            ac_trial = fast_conv(a_trial)
            M_trial = ac_trial.max()
            if M_trial < M - 1e-12:
                a = a_trial
                ac = ac_trial
                M = M_trial
                eta = min(eta * 1.1, 0.5)  # increase step if it worked
                accepted = True
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                    stalling = 0
                break
        
        if not accepted:
            stalling += 1
            # If stuck, try a perturbation
            if stalling > stall_threshold:
                # Random restart from best with perturbation
                noise = rng.normal(0, 0.01, n) * best_a
                a = clip_and_normalize(best_a + noise)
                ac = fast_conv(a)
                M = ac.max()
                stalling = 0
                eta = 0.05
                stall_threshold = int(stall_threshold * 1.5)
            else:
                eta *= 0.9
        
        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"      GradOpt step={step} ({elapsed:.1f}s): score={score:.6f}, eta={eta:.4f}")
    
    return best_a, 2.0 * n * best_M


def optimize_coord(a_init, time_limit, verbose=True, seed=123):
    """
    Coordinate descent: for each position, find the optimal value
    given all other positions fixed (1D optimization).
    Uses ternary search on [0, a_i + margin].
    """
    n = len(a_init)
    a = normalize(a_init)
    ac = fast_conv(a)
    best_M = ac.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M = ac.max()
        peak_k = int(ac.argmax())
        
        # Focus on positions that contribute to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        hot_positions = set(range(i_start, i_end))
        
        # Also add positions from second/third peak
        sorted_lags = np.argsort(ac)[::-1][:5]
        for k2 in sorted_lags:
            k2 = int(k2)
            s2 = max(0, k2 - n + 1)
            e2 = min(n, k2 + 1)
            hot_positions.update(range(s2, e2))
        
        hot_list = list(hot_positions)
        if len(hot_list) > 200:
            hot_list = list(rng.choice(hot_list, size=200, replace=False))
        rng.shuffle(hot_list)
        
        improved = False
        
        for pos in hot_list:
            if time.monotonic() - t0 > time_limit:
                break
            if a[pos] < 1e-12:
                continue
            
            # Try reducing this position and redistributing to others
            # Binary search on how much to reduce
            original_val = a[pos]
            
            # Try several reduction fractions
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = original_val * frac
                a[pos] = original_val - amt
                # Distribute to random cold positions
                n_dist = min(10, n // 50)
                targets = rng.choice(n, size=n_dist, replace=False)
                for t in targets:
                    if t != pos:
                        a[t] += amt / n_dist
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[pos] = original_val
                    break
                a /= s
                
                ac_new = fast_conv(a)
                M_new = ac_new.max()
                
                if M_new < best_M - 1e-12:
                    best_M = M_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[pos] += amt  # revert
                    for t in targets:
                        if t != pos:
                            a[t] -= amt / n_dist
        
        if not improved:
            # Try moving mass FROM hot TO cold positions more aggressively
            cold_positions = [i for i in range(n) if i not in hot_positions]
            if len(cold_positions) > 0:
                n_try = min(20, len(hot_list))
                for _ in range(n_try):
                    if time.monotonic() - t0 > time_limit:
                        break
                    src = int(rng.choice(hot_list))
                    dst = int(rng.choice(cold_positions))
                    if a[src] < 1e-10:
                        continue
                    for frac in [0.01, 0.03, 0.08]:
                        amt = a[src] * frac
                        a[src] -= amt
                        a[dst] += amt
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        if s < 1e-15:
                            break
                        a /= s
                        ac_new = fast_conv(a)
                        M_new = ac_new.max()
                        if M_new < best_M - 1e-12:
                            best_M = M_new
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += amt
                            a[dst] -= amt
                            a = np.maximum(a, 0.0)
                            a /= a.sum()
            
            if not improved:
                stall += 1
                # Escape: larger random perturbation
                n_pert = max(5, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.05, 0.2)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                ac_new = fast_conv(a)
                if ac_new.max() < best_M:
                    best_M = ac_new.max()
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      CoordDescent ({elapsed:.1f}s): score={2.0*n*best_M:.6f}, stall={stall}")
    
    return best_a, 2.0 * n * best_M


def optimize_anneal(a_init, time_limit, verbose=True, seed=77):
    """
    Simulated annealing with structured moves.
    Moves: (1) shift mass between two positions, (2) split/merge, (3) block shift.
    """
    n = len(a_init)
    a = normalize(a_init)
    ac = fast_conv(a)
    best_M = ac.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T_start = 0.01
    T_end = 0.0001
    n_steps_estimate = int(time_limit / 0.001)  # rough
    cooling = (T_end / T_start) ** (1.0 / max(1, n_steps_estimate))
    T = T_start
    
    current_M = best_M
    stall = 0
    
    for step in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Choose move type
        move_type = rng.integers(3)
        a_trial = a.copy()
        
        if move_type == 0:
            # Two-position transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.005, 0.05)
            a_trial[i] -= amt
            a_trial[j] += amt
            
        elif move_type == 1:
            # Block shift: shift a block of mass
            block_size = max(2, n // 100)
            start = int(rng.integers(n - block_size))
            amt = a[start:start+block_size].sum() * rng.uniform(0.01, 0.1)
            a_trial[start:start+block_size] -= amt / block_size
            dest = int(rng.integers(n - block_size))
            a_trial[dest:dest+block_size] += amt / block_size
            
        else:
            # Three-position: move from peak to two random positions
            ac_curr = fast_conv(a)
            peak_k = int(ac_curr.argmax())
            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            i = int(rng.integers(i_start, i_end))
            j1 = int(rng.integers(n))
            j2 = int(rng.integers(n))
            amt = a[i] * rng.uniform(0.01, 0.05)
            a_trial[i] -= amt
            a_trial[j1] += amt / 2
            a_trial[j2] += amt / 2
        
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial /= s
        
        ac_trial = fast_conv(a_trial)
        M_trial = ac_trial.max()
        delta = M_trial - current_M
        
        # Accept/reject
        if delta < 0 or (T > T_end and rng.random() < np.exp(-delta / T)):
            a = a_trial
            current_M = M_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        T *= cooling
        
        if stall > 5000:
            # Restart from best
            a = clip_and_normalize(best_a + rng.normal(0, 0.005, n) * best_a)
            current_M = fast_conv(a).max()
            T = max(T, 0.003)
            stall = 0
        
        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"      Anneal step={step} ({elapsed:.1f}s): score={score:.6f}, T={T:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      Anneal done ({elapsed:.1f}s): score={2.0*n*best_M:.6f}")
    
    return best_a, 2.0 * n * best_M


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_interp / ss


def optimize_two_phase(a_init, time_limit, verbose=True, seed=1):
    """
    Two-phase optimization: 
    Phase A: Smooth gradient (low beta) for exploration
    Phase B: Sharp gradient (high beta) for exploitation
    """
    n = len(a_init)
    a = normalize(a_init)
    t0 = time.monotonic()
    
    half_time = time_limit * 0.45
    a, score = optimize_gradient(a, half_time, beta=2.0, use_smooth=True, 
                                  verbose=verbose, seed=seed)
    if verbose:
        print(f"      Phase A (smooth): score={score:.6f}")
    
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 5:
        a, score = optimize_gradient(a, remaining, beta=15.0, use_smooth=True,
                                      verbose=verbose, seed=seed+1)
        if verbose:
            print(f"      Phase B (sharp): score={score:.6f}")
    
    return a, score


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        # Default initialization: use a structured sequence
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        # Try a sequence with slow variation
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    a_current = best_seq.copy()
    
    # === Phase 1: Quick smooth gradient (establish direction) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth gradient n={n0} ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        a_out, _ = optimize_gradient(a_current, t1, beta=3.0, use_smooth=True,
                                      verbose=True, seed=1)
        try_update(a_out, "smooth_grad")
    
    # === Phase 2: Sharp gradient (exploitation) ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    print(f"\n=== Phase 2: Sharp gradient n={len(a_current)} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, _ = optimize_gradient(a_current, t2, beta=20.0, use_smooth=True,
                                      verbose=True, seed=2)
        try_update(a_out, "sharp_grad")
    
    # === Phase 3: Exact subgradient (focused on peak) ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    print(f"\n=== Phase 3: Exact subgradient n={len(a_current)} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = optimize_gradient(a_current, t3, beta=5.0, use_smooth=False,
                                      verbose=True, seed=3)
        try_update(a_out, "exact_subgrad")
    
    # === Phase 4: Coordinate descent (local refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    print(f"\n=== Phase 4: Coord descent n={len(a_current)} ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        a_out, _ = optimize_coord(a_current, t4, verbose=True, seed=4)
        try_update(a_out, "coord")
    
    # === Phase 5: Simulated annealing (escape local minima) ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    print(f"\n=== Phase 5: Annealing n={len(a_current)} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_out, _ = optimize_anneal(a_current, t5, verbose=True, seed=5)
        try_update(a_out, "anneal")
    
    # === Phase 6: Two-phase refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    print(f"\n=== Phase 6: Two-phase n={len(a_current)} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_out, _ = optimize_two_phase(a_current, t6, verbose=True, seed=6)
        try_update(a_out, "two_phase")
    
    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 7: Multi-length ===")
        n_best = len(best_seq)
        try_lengths = []
        # Try slightly different lengths
        if n_best > 1000:
            try_lengths.append(n_best - 200)
        try_lengths.append(n_best + 200)
        try_lengths.append(n_best + 400)
        if n_best > 1500:
            try_lengths.append(n_best - 400)
        
        for n_try in try_lengths:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            if n_try < 500 or n_try > 4000:
                continue
            a_interp = interpolate_seq(best_seq, n_try)
            t7 = min(remaining * 0.3, 200.0)
            if t7 > 20:
                print(f"  Trying n={n_try} (t={t7:.0f}s)")
                a_out, _ = optimize_gradient(a_interp, t7 * 0.5, beta=5.0,
                                              use_smooth=True, verbose=False, seed=10)
                a_out2, _ = optimize_coord(a_out, t7 * 0.5, verbose=False, seed=11)
                try_update(a_out, f"ml_grad_n{n_try}")
                try_update(a_out2, f"ml_coord_n{n_try}")
    
    # === Phase 8: Final intensive coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    print(f"\n=== Phase 8: Final coord n={len(a_current)} ===")
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 15:
        a_out, _ = optimize_coord(a_current, t8, verbose=True, seed=20)
        try_update(a_out, "final_coord")
    
    # === Phase 9: Final gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    print(f"\n=== Phase 9: Final gradient n={len(a_current)} ===")
    t9 = min(remaining * 0.50, 150.0)
    if t9 > 10:
        a_out, _ = optimize_gradient(a_current, t9, beta=30.0, use_smooth=True,
                                      verbose=False, seed=30)
        try_update(a_out, "final_grad")
    
    # === Phase 10: Very short anneal for last-mile improvement ===
    remaining = time_budget - (time.monotonic() - t_global)
    a_current = best_seq.copy()
    if remaining > 20:
        print(f"\n=== Phase 10: Final anneal ===")
        t10 = min(remaining * 0.6, 100.0)
        if t10 > 10:
            a_out, _ = optimize_anneal(a_current, t10, verbose=False, seed=40)
            try_update(a_out, "final_anneal")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
