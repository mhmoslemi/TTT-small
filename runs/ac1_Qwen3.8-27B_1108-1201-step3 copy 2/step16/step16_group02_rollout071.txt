def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.empty(0)
    size = 2*n -1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return out

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 by optimizing a normalized nonnegative
sequence a in a fixed-reference softmax parameterization.

The objective is approximated by a log-sum-exp of the autocorrelation
coefficients.  Gradients are computed with FFT cross-correlations, making each
evaluation O(n log n).  L-BFGS-B is used as the quasi-Newton solver, with an
adaptive beta (smoothing sharpness) schedule, perturbative restarts, and
optional length re-interpolation.
"""

import numpy as np
import scipy.optimize as spopt
import time

# Globals used by top-level L-BFGS-B callbacks.
_LB_N = 0
_LB_REF = 0
_LB_BETA = 1.0
_LB_Z_TO_IDX = np.empty(0, dtype=np.int64)
_LB_START = 0.0
_LB_LIMIT = 0.0


def fast_conv(a):
    """Linear autocorrelation convolution a*a via zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return out


def sanitize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    return np.maximum(a, 0.0)


def normalize(a):
    """Project to exactly nonnegative and sum 1, with safe fallback."""
    a = sanitize_nonnegative(a)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def evaluate_safe(a):
    """Call the externally provided evaluate_sequence without exceptions."""
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        lst = [float(x) for x in arr]
        val = float(evaluate_sequence(lst))
        if np.isfinite(val):
            return val
        return float("inf")
    except Exception:
        return float("inf")


def mix_normalize(a, b, t):
    """Normalized convex combination of same-length sequences."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if a.size != b.size or a.size == 0:
        return normalize(a)
    c = (1.0 - float(t)) * a + float(t) * b
    return normalize(c)


def perturb_sequence(a, sigma, rng_seed):
    """Multiplicative lognormal perturbation in probability space."""
    a = normalize(a)
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    rng = np.random.default_rng(rng_seed)
    loga = np.log(np.maximum(a, 1e-300))
    z = loga + rng.normal(0.0, float(sigma), size=n)
    z -= z.max()
    p = np.exp(z)
    return normalize(p)


def interpolate_sequence(a, new_n):
    """Interpolate a normalized sequence to a new integer length."""
    a = normalize(a)
    if new_n <= 0:
        return a
    if a.size == 0:
        return np.ones(int(new_n), dtype=np.float64) / float(new_n)
    if a.size == int(new_n):
        return a
    if a.size == 1:
        return np.ones(int(new_n), dtype=np.float64) / float(new_n)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    g = np.interp(new_x, old_x, a)
    return normalize(g)


def make_init(n, kind="uniform"):
    """Simple deterministic initial shapes."""
    n = int(max(1, n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    if kind == "uniform":
        v = np.ones(n, dtype=np.float64)
    elif kind == "triangle":
        v = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == "gaussian":
        v = np.exp(-((x - 0.5) / 0.15) ** 2)
    elif kind == "raised_cosine":
        v = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif kind == "broad_gaussian":
        v = np.exp(-((x - 0.5) / 0.35) ** 2)
    else:
        v = np.ones(n, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return v / s


def _lse_obj(C, beta):
    """Log-sum-exp smooth approximation to max(C)."""
    C = np.asarray(C, dtype=np.float64)
    if C.size == 0:
        return 0.0
    m = float(np.max(C))
    scaled = float(beta) * (C - m)
    scaled = np.clip(scaled, -745.0, 690.0)
    s = float(np.sum(np.exp(scaled)))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return (m + float(np.log(s))) / float(beta)


def _lse_grad(C, a, beta):
    """Gradient of log-sum-exp(max conv) wrt a, using FFT cross-correlation."""
    a = np.asarray(a, dtype=np.float64)
    C = np.asarray(C, dtype=np.float64)
    n = a.size
    L = C.size
    if n == 0 or L == 0:
        return np.zeros_like(a, dtype=np.float64)

    m = float(np.max(C))
    scaled = float(beta) * (C - m)
    scaled = np.clip(scaled, -745.0, 690.0)
    e = np.exp(scaled)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(L, dtype=np.float64)
        w[0] = 1.0
    else:
        w = e / s

    need = L + n - 1
    fft_size = 1 << (need - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    S = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * S


def choose_beta(a, min_beta, max_beta):
    """Choose LSE beta from the gap near the dominant convolution peak."""
    a = normalize(a)
    C = fast_conv(a)
    if C.size == 0:
        return float(min_beta)
    M = float(np.max(C))
    if M <= 0.0 or not np.isfinite(M):
        return float(min_beta)
    cdesc = np.sort(C)[::-1]
    k = max(1, min(20, cdesc.size - 1))
    gap = M - float(cdesc[k])
    if gap <= 0.0 or not np.isfinite(gap):
        gap = max(M * 1e-6, 1e-12)
    beta = 50.0 / float(gap)
    beta = float(np.clip(beta, float(min_beta), float(max_beta)))
    return beta


def _lbfgs_prepare(a0):
    """Create fixed-reference logit parameterization a_i / a_ref."""
    a = normalize(a0)
    n = a.size
    if n == 1:
        return np.empty(0, dtype=np.float64), 0, np.empty(0, dtype=np.int64)

    ref = int(np.argmax(a))
    if ref < 0 or ref >= n or a[ref] <= 1e-300:
        ref = 0

    idxs = np.delete(np.arange(n, dtype=np.int64), ref)
    denom_ref = max(float(a[ref]), 1e-300)
    z0 = np.log(np.maximum(a[idxs], 1e-300) / denom_ref)
    return z0, ref, idxs
