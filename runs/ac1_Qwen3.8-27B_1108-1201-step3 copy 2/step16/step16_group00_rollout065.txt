def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * fast_autocorr(a).max() / (s * s)


def make_symmetric(a):
    n = len(a)
    a_sym = np.zeros(n, dtype=np.float64)
    for i in range(n):
        a_sym[i] = 0.5 * (a[i] + a[n - 1 - i])
    return a_sym


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = a
    return T

def soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def make_power_law(n, power=1.0):
    x = np.linspace(0.01, 1.0, n)
    a = x ** power
    return norm_sum1(a)


def make_gaussian(n, sigma=0.2, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return norm_sum1(a)


def make_bimodal(n, sep=0.3, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5 + sep) / width) ** 2) + \
        np.exp(-0.5 * ((x - 0.5 - sep) / width) ** 2) + 0.01
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 7]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def multi_scale_optimize(best_seq, time_limit, solver, verbose=True):

        scales = [512, 1024, 2048]

            break

            best = a_out
            best_obj = obj_out
