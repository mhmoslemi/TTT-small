def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:

def compute_softmax_subgradient(a, ac, top_k, temperature):
    n = len(a)

    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)

        i_start = max(0, k_int - n + 1)

            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_power_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 120)
            cur_beta = max(cur_beta - 0.3, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe_adaptive(a0, time_limit, solver, verbose=True,
                            K_start=100, K_max=800, K_growth=5):
    """Adaptive LP-Frank-Wolfe: grow K if LP solution violates excluded constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    K = K_start
    iter_count = 0
    solve_time_total = 0.0

    for t in range(1, 100000):
        t_start = time.monotonic()
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K most binding constraints
        K_sel = min(K, 2 * n - 1)
        binding_idx = np.argsort(ac)[:K_sel]

        # Build constraint matrix for selected lags
        # T_sel[k, i] = y[k - i] if 0 <= k-i < n, else 0
        T_sel = np.zeros((K_sel, n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                T_sel[row, i_start:i_end] = y[k_int - np.arange(i_start, i_end)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = min(K * 2, K_max)
            continue

        solve_time_total += time.monotonic() - t_start

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Check violation of excluded constraints
        if K_sel < 2 * n - 1:
            full_ac_new = T_sel full check via autocorr
            pass  # We'll check via full autocorr below

        # Nesterov FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Check if LP solution violates excluded constraints -> grow K
        if K_sel < 2 * n - 1:
            ac_check = fast_autocorr(g_dir)
            max_excluded = np.max(np.delete(ac_check, binding_idx)) if K_sel < len(ac_check) else 0
            if max_excluded > M_curr * 1.5:
                K = min(K + K_growth * 10, K_max)

        # Adaptive: if LP is too slow, reduce K
        if solve_time_total / max(iter_count, 1) > 2.0 and K > K_start * 2:
            K = max(K // 2, K_start)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            avg_solve = solve_time_total / max(iter_count, 1)
            print(f"    LP t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, avg_solve={avg_solve:.2f}s")

    return best_a, best_m


def peak_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Coordinate descent that specifically targets reducing the peak autocorrelation lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]

        improved = False
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: minimize increase in autocorrelation
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    d2 = d
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    scores[d] += add

            candidates = np.where(a > 1e-15)[0]
            if len(candidates) < 2:
                candidates = np.arange(n)
            candidates = candidates[candidates != src]

            if len(candidates) == 0:
                continue

            best_dst = candidates[np.argmin(scores[candidates])]
            dst = best_dst

            found = False
            for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.15, 0.25]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 50 and a[src] > 1e-10:
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def make_structured_init(n, kind='sine'):
    x = np.linspace(0, 1, n)
    if kind == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'gauss':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'bimodal':
        a = np.exp(-30 * (x - 0.15) ** 2) + np.exp(-30 * (x - 0.85) ** 2) + 0.05
    elif kind == 'ramp':
        a = np.linspace(0.1, 1.0, n)
    else:
        a = np.ones(n)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_structured_init(n_base, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick multi-init AMD (60s) ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.08, 80.0)

    if t_phase1 > 30:
        inits = [norm_sum1(best_seq)]
        labels = ['parent']
        inits.append(make_structured_init(n0, 'ramp'))
        labels.append('ramp')
        inits.append(make_structured_init(n0, 'bimodal'))
        labels.append('bimodal')
        inits.append(make_structured_init(n0, 'gauss'))
        labels.append('gauss')

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 5:
                break
            print(f"  AMD '{lbl}' ({t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=40, beta=4.0, c0=3.0,
                                    restart_every=4000, verbose=False)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Main LP-FW (700s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adaptive LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_adaptive(a_in, t2, solver, verbose=True,
                                           K_start=80, K_max=600, K_growth=5)
        try_update(a_out, f"lp_main_n{n_best}")
    else:
        print("  Skipping LP (time/size)")

    # === Phase 3: AMD polish (250s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD polish (n={n_best}) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t3, top_k=70, beta=4.0, c0=3.5,
                                restart_every=8000, verbose=False)
        try_update(a_out, f"amd_polish_n{n_best}")

    # === Phase 4: Coordinate descent (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_coordinate_descent(a_in, t4, verbose=True, rng_seed=77)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (base n={n_best}) ===")

    longer_lengths = []
    for delta in [300, 600, 1000, 1500]:
        nn = n_best + delta
        if nn <= 5000:
            longer_lengths.append(nn)

    for new_n in longer_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick AMD
        t5a = min(remaining * 0.25, 120.0)
        if t5a > 15:
            a_out, _ = amd_optimize(a_long, t5a, top_k=50, beta=4.5, c0=3.0,
                                    restart_every=5000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

            # LP-FW if time and size allow
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 80 and new_n <= 3800:
                t5b = min(remaining * 0.30, 150.0)
                if t5b > 20:
                    a_out, _ = lp_frank_wolfe_adaptive(a_out, t5b, solver, verbose=False,
                                                       K_start=80, K_max=400, K_growth=5)
                    try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Second LP-FW pass on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Second LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 4000:
        t6 = min(remaining * 0.40, 300.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_adaptive(a_in, t6, solver, verbose=True,
                                               K_start=150, K_max=800, K_growth=10)
            try_update(a_out, f"lp2_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 120.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_coordinate_descent(a_in, t7, verbose=False, rng_seed=1234)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: One more AMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: AMD alt (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.6, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t8, top_k=100, beta=3.0, c0=5.0,
                                    restart_every=10000, verbose=False, rng_seed=999)
            try_update(a_out, f"amd_alt_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 3200
        best_seq = make_structured_init(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
