def soft_subgrad_multi(a, b, top_k=5, power=2.0):

def smooth_mirror_descent(a0, time_limit, beta_start=20.0, beta_end=500.0,
                          eta_init=0.5, momentum=0.7, seed=42, verbose=True):
    """
    Smooth mirror descent with adaptive beta, momentum, and restarts.
    Minimizes max(conv(a,a)) on the simplex via log-sum-exp smoothing.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(n)  # momentum
    stall = 0
    max_stall = 5000
    eta = eta_init
    eta_min = 0.01
    eta_max = 5.0
    current_beta = beta_start
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1

        # Adapt beta: increase over time (annealing the smoothness)
        progress = min(n_iter / max(n_iter, 10000), 1.0)
        current_beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)

        # Compute gradient
        g = smooth_grad(a, current_beta)

        # Momentum
        v = momentum * v + (1.0 - momentum) * g

        # Line search: try the step, if no improvement, reduce eta
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125]:
            eta_try = eta * factor
            a_new = mirror_step(a, v, eta_try)
            M_new = fast_conv(a_new).max()
            if M_new < best_m - 1e-15:
                a = a_new
                best_m = M_new
                best_a = a.copy()
                stall = 0
                improved = True
                if factor < 1.0:
                    eta = max(eta_min, eta * factor)
                break
        if not improved:
            eta = max(eta_min, eta * 0.8)
            stall += 1

        # Restart if stuck
        if stall > max_stall:
            # Gaussian perturbation
            a_perturbed = best_a + rng.normal(0, 0.003, n) * np.maximum(best_a, 0.01)
            a_perturbed = np.maximum(a_perturbed, 1e-10)
            a_perturbed = a_perturbed / a_perturbed.sum()
            a = a_perturbed
            v = np.zeros(n)
            eta = eta_init
            stall = 0
            max_stall = min(max_stall + 500, 10000)

        if n_iter % 20000 == 0 and n_iter > 0:
            a_jump = best_a + rng.normal(0, 0.01, n) * np.maximum(best_a, 0.005)
            a_jump = np.maximum(a_jump, 1e-8)
            a_jump = a_jump / a_jump.sum()
            M_jump = fast_conv(a_jump).max()
            if M_jump < best_m:
                a = a_jump
                best_m = M_jump

def peak_cycling_descent(a0, time_limit, seed=123, verbose=False):
    """
    Focus on reducing the specific peak that is currently maximal.
    Cycles between top peaks to avoid getting stuck.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 2000
    eta = 0.3
    top_k = 3

    peak_history = []

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        M_curr = b.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0

        # Get top peaks
        top_idx = np.argsort(b)[::-1][:top_k * 2]
        # Filter to those near the max
        threshold = best_m * 0.98
        active_peaks = [k for k in top_idx if b[k] > threshold]
        if len(active_peaks) == 0:
            active_peaks = [int(top_idx[0])]

        # Track peak cycling
        peak_history.append(active_peaks[0])
        if len(peak_history) > 500:
            peak_history.pop(0)

        # Detect cycling: if peaks keep switching, increase top_k
        if len(peak_history) >= 100:
            recent = peak_history[-100:]
            unique_recent = len(set(recent))
            if unique_recent > 5:
                top_k = min(top_k + 1, 8)
            elif unique_recent < 3:
                top_k = max(top_k - 1, 2)

        # Compute gradient at the dominant peak (or weighted average)
        if len(active_peaks) <= 1:
            g = hard_subgrad(a, active_peaks[0])
        else:
            g = soft_subgrad_multi(a, b, top_k=min(len(active_peaks), top_k), power=3.0)

        # Mirror descent step
        a_new = mirror_step(a, g, eta)
        M_new = fast_conv(a_new).max()

        if M_new < M_curr - 1e-15:
            a = a_new
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            eta = max(0.01, eta * 0.9)

        # Restart
        if stall > max_stall:
            a_pert = best_a + rng.normal(0, 0.005, n) * np.maximum(best_a, 0.01)
            a_pert = np.maximum(a_pert, 1e-10)
            a = a_pert / a_pert.sum()
            eta = 0.3
            stall = 0

        if verbose and (int((time.monotonic() - t0) * 4) % 4 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PC iter ({elapsed:.1f}s): J={J:.6f}, stall={stall}, top_k={top_k}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, seed=42, verbose=False):
    """
    Coordinate descent: transfer small amounts of mass between positions
    to reduce the maximum convolution value.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    b = fast_conv(a)
    M_curr = b.max()
    k_star = int(np.argmax(b))

    def try_move(src, dst, amt):
        nonlocal a, best_a, best_m, b, M_curr, k_star, stall
        if src == dst or amt < 1e-18 or amt > a[src]:
            return False
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            return False
        a_trial = a_trial / s
        b_trial = fast_conv(a_trial)
        M_trial = b_trial.max()
        if M_trial < best_m - 1e-15:
            a = a_trial
            best_m = M_trial
            best_a = a_trial.copy()
            b = b_trial
            M_curr = M_trial
            k_star = int(np.argmax(b))
            stall = 0
            return True
        return False

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_conv(a)
        M_curr = b.max()
        k_star = int(np.argmax(b))

        # Find the pair (src, dst) that most reduces b[k_star]
        # b[k_star] = sum_i a[i]*a[k_star-i]
        # Reducing a[src] by amt changes b[k_star] by -amt*(a[k_star-src] + a[src - (k_star-src)])
        # approximately -2*amt*a[k_star-src] (if both indices valid)

        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)

        # For each i in the active range, the "sensitivity" of b[k_star] to a[i]
        # is 2*a[k_star-i]. We want to reduce a[i] where this is large.
        # And increase a[j] where the sensitivity is small (or negative impact).

        sens = np.zeros(n)
        for i in range(i_start, i_end):
            j = k_star - i
            if 0 <= j < n:
                sens[i] += a[j]
                sens[j] += a[i]

        # Source: high sensitivity, high value
        # Destination: low sensitivity
        src_candidates = np.where(sens > np.median(sens))[0]
        dst_candidates = np.where(sens < np.median(sens))[0]

        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            stall += 1
            continue

        # Try random pairs
        n_tries = min(50, len(src_candidates) * len(dst_candidates))
        improved = False

        for _ in range(n_tries):
            src = src_candidates[rng.integers(len(src_candidates))]
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            if src == dst:
                continue

            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if try_move(int(src), int(dst), amt):
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random perturbations
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                if try_move(i, j, amt):
                    improved = True
                    break
                # Try opposite direction
                if a[j] > 1e-10:
                    amt2 = a[j] * frac
                    if try_move(j, i, amt2):
                        improved = True
                        break

        if not improved:
            stall += 1

        # Periodic large restart
        if stall % 200 == 199:
            a_pert = best_a + rng.normal(0, 0.008, n) * np.maximum(best_a, 0.01)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            M_pert = fast_conv(a_pert).max()
            if M_pert < best_m:
                a = a_pert
                best_a = a_pert.copy()
                best_m = M_pert
                b = fast_conv(a)
            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def simulated_annealing_polish(a0, time_limit, T_init=0.01, T_end=0.001, seed=99, verbose=False):
    """
    Simulated annealing: random pairwise mass transfers with Boltzmann acceptance.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    b = fast_conv(a)
    M_curr = b.max()
    stall = 0
    max_stall = 3000
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T_init * (T_end / T_init) ** elapsed_frac

        # Pick random pair
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Random transfer amount
        max_transfer = min(a[i], a[j]) * 0.1 + 0.001
        amt = rng.uniform(0.0005, max_transfer)

        direction = 1 if rng.random() < 0.5 else -1
        if direction == 1:
            if a[i] < amt:
                continue
            a[i] -= amt
            a[j] += amt
        else:
            if a[j] < amt:
                continue
            a[j] -= amt
            a[i] += amt

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            b = fast_conv(a)
            M_curr = b.max()
            continue
        a = a / s

        b_new = fast_conv(a)
        M_new = b_new.max()
        delta = M_new - M_curr

        if delta < 0:
            # Accept (improvement)
            b = b_new
            M_curr = M_new
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Accept with probability exp(-delta/T)
            if rng.random() < np.exp(-delta / max(T, 1e-15)):
                b = b_new
                M_curr = M_new
                stall += 1
            else:
                # Reject: revert
                if direction == 1:
                    a[i] += amt
                    a[j] -= amt
                else:
                    a[j] += amt
                    a[i] -= amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall += 1

        if stall > max_stall:
            # Restart from best with perturbation
            a = best_a + rng.normal(0, 0.005, n) * np.maximum(best_a, 0.01)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            b = fast_conv(a)
            M_curr = b.max()
            stall = 0

        if verbose and total_iters % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA iter={total_iters} ({elapsed:.1f}s): J={J:.6f}, T={T:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive time allocation."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = eval_obj(parent / parent.sum())
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_obj(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = eval_obj(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Smooth mirror descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth Mirror Descent n={n0} ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_in = normalize(best_seq)
        a_out, _ = smooth_mirror_descent(a_in, t1, beta_start=20.0, beta_end=300.0,
                                         eta_init=0.4, momentum=0.7, seed=42, verbose=True)
        try_update(a_out, f"smd1_n{n0}")

    # === Phase 2: Smooth mirror descent with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SMD variant n={len(best_seq)} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        a_in = normalize(best_seq)
        a_out, _ = smooth_mirror_descent(a_in, t2, beta_start=50.0, beta_end=800.0,
                                         eta_init=0.3, momentum=0.8, seed=77, verbose=True)
        try_update(a_out, f"smd2_n{len(best_seq)}")

    # === Phase 3: Peak cycling descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Cycling n={len(best_seq)} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        a_in = normalize(best_seq)
        a_out, _ = peak_cycling_descent(a_in, t3, seed=123, verbose=False)
        try_update(a_out, f"peakcycle_n{len(best_seq)}")

    # === Phase 4: Simulated annealing polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Simulated Annealing n={len(best_seq)} ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_in = normalize(best_seq)
        a_out, _ = simulated_annealing_polish(a_in, t4, T_init=0.005, T_end=0.0005,
                                              seed=314, verbose=False)
        try_update(a_out, f"sa_n{len(best_seq)}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coordinate Descent n={len(best_seq)} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, seed=42, verbose=True)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Second round SMD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: SMD Round 2 n={len(best_seq)} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 50:
        a_in = normalize(best_seq)
        a_out, _ = smooth_mirror_descent(a_in, t6, beta_start=100.0, beta_end=1000.0,
                                         eta_init=0.2, momentum=0.85, seed=555, verbose=True)
        try_update(a_out, f"smd3_n{len(best_seq)}")

    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        for delta in [-400, 400, -800]:
            n_try = n_best + delta
            if n_try < 1000 or n_try > 4000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
            print(f"\n=== Phase 7: Length n={n_try} ===")
            a_long = interpolate_seq(best_seq, n_try)
            a_in = normalize(a_long)
            t7 = min(remaining * 0.35, 150.0)
            if t7 > 30:
                a_out, _ = smooth_mirror_descent(a_in, t7, beta_start=30.0, beta_end=400.0,
                                                 eta_init=0.3, momentum=0.75,
                                                 seed=100 + n_try % 1000, verbose=False)
                try_update(a_out, f"smd_n{n_try}")

                remaining = time_budget - (time.monotonic() - t_global)
                t7b = min(remaining * 0.2, 60.0)
                if t7b > 15:
                    a_in2 = normalize(best_seq)
                    a_out2, _ = coordinate_polish(a_in2, t7b, seed=700, verbose=False)
                    try_update(a_out2, f"coord_n{n_try}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Polish n={n_best} ===")
    if remaining > 30:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 10:
            a_in = normalize(best_seq)
            a_out, _ = smooth_mirror_descent(a_in, t8, beta_start=200.0, beta_end=1500.0,
                                             eta_init=0.1, momentum=0.9, seed=999, verbose=False)
            try_update(a_out, f"final_smd_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.6, 50.0)
            if t8b > 10:
                a_in = normalize(best_seq)
                a_out, _ = coordinate_polish(a_in, t8b, seed=2718, verbose=False)
                try_update(a_out, f"final_coord_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = eval_obj(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
