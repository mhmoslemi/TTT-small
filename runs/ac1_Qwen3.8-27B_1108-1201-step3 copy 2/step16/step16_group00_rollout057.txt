import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_super_elliptical(n, power=1.0):
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    a = np.maximum(1.0 - x**2, 0.0) ** power
    return norm_sum1(a)

def make_kaiser(n, beta=8.0):
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    a = np.i0(beta * np.sqrt(np.maximum(1.0 - x**2, 0.0))) / np.i0(beta)
    return norm_sum1(a)

def make_gaussian(n, sigma=0.4):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return norm_sum1(a)

def make_flat_top(n, w=0.3):
    x = np.linspace(0, 1, n)
    a = np.where(np.abs(x - 0.5) < w, 1.0, 0.0)
    # Smooth the edges
    from scipy.ndimage import gaussian_filter1d
    a = gaussian_filter1d(a, sigma=2)
    return norm_sum1(a)

def make_blend(a1, a2, t=0.5):
    n = len(a1)
    a2_resized = interpolate_sequence(a2, n) if len(a2) != n else a2
    return norm_sum1(t * a1 + (1 - t) * a2_resized)

def compute_grad_ac(a, ac, top_k=80, sharpness=8.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** sharpness
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def amd_optimize(a0, time_limit, top_k=80, sharpness=8.0, c0=5.0,

    x_prev = a.copy()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_grad_ac(y, ac, top_k=cur_top_k, sharpness=cur_sharp)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r

            cur_top_k = min(cur_top_k + 20, 150)
            cur_sharp = max(cur_sharp - 1.0, 3.0)
            if verbose:

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m

def lp_frank_wolfe(a0, time_limit, verbose=True, top_K_init=250, top_K_max=500):

            prob.solve(solver=s, verbose=False)

                solver = s
                print(f"  LP solver: {s}")
                break
        except Exception:
            continue
    if solver is None:

        print(f"  LP solver (fallback): ECOS")

    y = a.copy()
    x_nest = a.copy()
    prev_y = a.copy()
    iter_count = 0

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x_nest = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        mx = fast_autocorr(x_nest).max()
        if mx < best_m:
            best_m = mx
            best_a = x_nest.copy()
            no_improve = 0

        # Adapt top_K
        if no_improve > 200:
            top_K = min(top_K + 50, top_K_max)
            no_improve = 0

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={top_K}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LP final ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m

def peak_reduction(a0, time_limit, verbose=True, rng_seed=77):
    """Targeted peak reduction: identify top contributors and redistribute."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the top contributors
            top_contribs = np.argsort(contrib)[::-1][:min(5, len(contrib))]
            for tc in top_contribs:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Compute sensitivity for each destination
                n_sens = min(15, n_top)
                scores = np.zeros(n)
                for k2 in top_lags[:n_sens]:
                    k2i = int(k2)
                    for d in range(n):
                        d1 = k2i - d
                        if 0 <= d1 < n:
                            scores[d] += a[d1]
                        d2 = d - k2i
                        if 0 <= d2 < n:
                            scores[d] += a[d2]

                # Find best destination (low score, not the source)
                scores[src] = np.inf
                dst = np.argmin(scores)

                if dst == src:
                    continue

                # Try multiple transfer fractions
                found = False
                for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

                if not found and stall > 30:
                    # Random move
                    j = int(rng.integers(n))
                    if j != src and a[src] > 1e-10:
                        frac = rng.uniform(0.002, 0.01)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Escape: random perturbation
            n_pert = min(10, n // 10)
            indices = rng.integers(0, n, n_pert * 2)
            for i in range(0, n_pert * 2, 2):
                i1, i2 = int(indices[i]), int(indices[i + 1])
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakRed ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakRed final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def mirror_descent_guided(a0, time_limit, verbose=False):
    """Mirror descent using LP dual variables as guidance for the subgradient."""
    import cvxpy as cp

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    solver = None
    for s in [cp.CLARABEL, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    eta_cur = 3.0
    iter_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Occasionally solve LP to get dual-guided direction
        if t % 50 == 0:
            top_K = min(200, 2 * n - 1)
            binding_idx = np.argsort(M_curr - ac)[:top_K]
            T_sel = np.zeros((top_K, n))
            for idx_i, k in enumerate(binding_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                idx = np.arange(i_start, i_end)
                T_sel[idx_i, idx] = a[k_int - idx]

            b_var = cp.Variable(n, nonneg=True)
            constraints = [T_sel @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                    b_sum = b_val.sum()
                    if b_sum > 1e-15:
                        g_dir = b_val / b_sum
                        # Mix current a with LP direction
                        alpha = 2.0 / (t + 2.0)
                        a_new = (1 - alpha) * a + alpha * g_dir
                        a_new = np.maximum(a_new, 0.0)
                        a_new = a_new / a_new.sum()
                        M_new = fast_autocorr(a_new).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a_new.copy()
                        a = a_new
            except Exception:
                pass

        # Regular mirror descent step
        ac = fast_autocorr(a)
        g = compute_grad_ac(a, ac, top_k=100, sharpness=6.0)
        eta = eta_cur / (np.sqrt(iter_count + 1) + 1.0)
        log_a = np.log(np.maximum(a, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x = a_new
        a = x
        iter_count += 1

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD-guided t={t} ({elapsed:.0f}s): J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MD-guided final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Fast coordinate descent for final polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_lags[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Best destination
            scores = np.zeros(n)
            for k2 in top_lags[:8]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
                    d2 = d - k2i
                    if 0 <= d2 < n:
                        scores[d] += a[d2]
            scores[src] = np.inf
            dst = int(np.argmin(scores))

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Try a few random destinations
                for _ in range(3):
                    d2 = int(rng.integers(n))
                    if d2 == src or a[src] < 1e-10:
                        continue
                    amt = src_val * rng.uniform(0.002, 0.01)
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[d2] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

        if not improved:
            stall += 1
            # Perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.003, 0.02)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    
    Strategy:
    1. Generate diverse initializations (parent, super-elliptical, Kaiser, blends)
    2. Run AMD on each to get good local solutions
    3. Apply LP Frank-Wolfe to the best few for significant improvement
    4. Polish with targeted peak reduction and coordinate descent
    5. Explore longer sequences by interpolation
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_super_elliptical(n_base, power=1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-initialization AMD ===
    print(f"\n=== Phase 1: Multi-init AMD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 360.0)

    if t_phase1 > 40:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        # Super-elliptical with various powers
        for pwr in [0.5, 1.0, 2.0, 4.0]:
            inits.append(make_super_elliptical(n0, power=pwr))
            labels.append(f"super_ell_p{pwr}")

        # Kaiser window
        for beta in [5.0, 10.0, 15.0]:

            labels.append(f"kaiser_b{beta}")

        # Gaussian

        inits.append(make_gaussian(n0, sigma=0.45))
        labels.append("gauss_045")

        # Blends with parent

            pn = norm_sum1(parent)
            inits.append(norm_sum1(0.5 * pn + 0.5 * make_super_elliptical(n0, 2.0)))
            labels.append("blend_parent_super2")
            inits.append(norm_sum1(0.7 * pn + 0.3 * make_kaiser(n0, 10.0)))
            labels.append("blend_parent_kaiser")
