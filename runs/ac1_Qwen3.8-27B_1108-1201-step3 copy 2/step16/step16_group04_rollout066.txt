"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative step function
sequence using a fast, LP-free optimization stack:

1. Flexible Fourier amplitude equalization (spectral flattening) to get close
   to a flat autocorrelation quickly.
2. Full-spectrum subgradient (ALL lags above the flat-average floor) with an
   FFT/correlate-accelerated gradient, a Polyak step size, and Anderson
   acceleration for fast, stable convergence.
3. A targeted peak-shaving local search that moves mass away from the
   coordinates contributing most to the tallest autocorrelation lags.
4. A biased random pair-swap local search to polish and escape stagnation.
5. Multi-length exploration (interpolation to nearby lengths) to broaden the
   search and reduce the chance of being trapped in a poor local minimum.
6. Diverse solid initializations (sine, raised-cosine, bimodal, Dirichlet, etc.)

All heavy per-iteration work is O(n log n) (FFT) or O(n * #top_lags), so thousands
of iterations fit easily inside the time budget. The function always returns the
best valid non-negative sequence found.
"""

def fast_autocorr(a):
    """Linear autocorrelation of a via zero-padded FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Call the provided evaluator with a plain float list; inf on failure."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Clip to >=0 and rescale to sum 1 (robust)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return np.array([1.0])
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def diverse_init(n, seed=0):
    """A set of solid, structurally distinct non-negative initial vectors."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    inits = []

    s = np.sin(np.pi * x) ** 2
    inits.append(s / s.sum())

    r = np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5))), 0.0)
    inits.append(r / r.sum())

    b = np.exp(-40.0 * (x - 0.25) ** 2) + np.exp(-40.0 * (x - 0.75) ** 2)
    inits.append(b / b.sum())

    tri = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    inits.append(tri / tri.sum())

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    inits.append(rng.dirichlet(np.ones(n) * 8.0))
    sparse = rng.random(n) ** 4.0
    inits.append(sparse / sparse.sum())
    return inits


def interpolate_sequence(f, new_n):
    """Resample a normalized sequence to a new length (normal output)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    if old_n == 1:
        return np.ones(new_n) / new_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def spectral_flatten(a, n_iter=50, damp=0.3):
    """Flatten the power spectrum by blending amplitudes toward their mean."""
    a = normalize(a).copy()
    n = len(a)
    fft_size = 1 << (3 * n).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        nz = amp > 1e-12
        if nz.sum() == 0:
            break
        target = amp[nz].mean()
        new_amp = damp * amp + (1.0 - damp) * target
        new_amp[0] = amp[0]  # keep DC so the sum is preserved
        phase = A / (amp + 1e-30)
        a_new = np.real(np.fft.irfft(new_amp * phase, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a = a_new / ss
    return a


def spectral_flatten_geomean(a, n_iter=50):
    """Flatten using the geometric mean of each amplitude and the mean amplitude."""
    a = normalize(a).copy()
    n = len(a)
    fft_size = 1 << (3 * n).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        nz = amp > 1e-12
        if nz.sum() == 0:
            break
        target = amp[nz].mean()
        new_amp = np.sqrt(amp * target)
        new_amp[0] = amp[0]
        phase = A / (amp + 1e-30)
        a_new = np.real(np.fft.irfft(new_amp * phase, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a = a_new / ss
    return a


# --------------------------- subgradient methods ---------------------------

def grad_full_spectrum(a, ac):
    """
    Subgradient of max_k (a*a)[k] using every lag above the flat floor.
    grad[j] = 2 * sum_k w[k] * a[k-j], with w proportional to (ac - floor)_+.
    Uses np.correlate (FFT-backed) so it is very fast.
    Returns (grad, M, floor).
    """
    n = len(a)
    floor = 1.0 / (2.0 * n - 1.0)
    excess = np.maximum(ac - floor, 0.0)
    wsum = excess.sum()
    M = ac.max()
    if wsum < 1e-18:
        return np.zeros(n), M, floor
    w = excess / wsum
    cc = np.correlate(w, a, mode='full')   # cc[j] = sum_n w[n+j]*a[n]
    return 2.0 * cc[:n], M, floor


def subgradient_polyak_anderson(a0, time_limit, verbose=True, max_hist=6,
                                restart_period=400, rng_seed=2024):
    """Subgradient + Polyak step + Anderson acceleration + periodic restarts."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    f_star = 1.0 / (2.0 * n - 1.0)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    hist_x = []
    hist_g = []
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_period:
            a = best_a * (1.0 + rng.normal(0.0, 0.02, n))
            a = project_simplex(a)
            hist_x = []
            hist_g = []
            stall = 0
            continue

        grad, M_curr, _ = grad_full_spectrum(a, ac)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        g = grad / gn

        gap = M_curr - f_star
        if gap < 1e-14:
            break
        eta = min(gap / (gn * gn), 2.0)

        if len(hist_x) >= max_hist:
            X = np.array(hist_x[-max_hist:])
            G = np.array(hist_g[-max_hist:])
            s_vecs = np.diff(X, axis=0)
            y_vecs = np.diff(G, axis=0)
            Mmat = y_vecs @ y_vecs.T
            rhs = -(G[-1] @ y_vecs.T)
            try:
                alpha = np.linalg.solve(Mmat + 1e-12 * np.eye(len(Mmat)), rhs)
                alpha = np.clip(alpha, -1.0, 1.0)
                x_new = a + alpha @ s_vecs
            except np.linalg.LinAlgError:
                x_new = a - eta * g
        else:
            x_new = a - eta * g

        a = project_simplex(x_new)

        hist_x.append(np.array(a, copy=True))
        hist_g.append(np.array(g, copy=True))
        if len(hist_x) > max_hist + 2:

            hist_g = hist_g[-(max_hist + 2):]

            elapsed = time.monotonic() - t0
            print("    PG+AA t=%d (%.1fs): J=%.6f M=%.8e" %
                  (t, elapsed, 2.0 * n * best_m, best_m))

    return best_a, best_m


def subgradient_topk(a0, time_limit, top_k=40, beta=10.0, verbose=True,
                     rng_seed=999):
    """Subgradient of max over the top-K lags with softmax (softmax) weights."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 800:
            a = best_a * (1.0 + rng.normal(0.0, 0.01, n))
            a = project_simplex(a)
            stall = 0
            continue

        kmax = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:kmax]
        top_vals = ac[top_idx]
        w = np.exp(beta * (top_vals - top_vals[0]))
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(kmax) / kmax
        else:
            w = w / wsum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            js = max(0, ki - (n - 1))
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * wi * a[ki - idx]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        g = grad / gn
        eta = 1.0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * g)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    PG-topK t=%d (%.1fs): J=%.6f" %
                  (t, elapsed, 2.0 * n * best_m))

    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=True, rng_seed=42):
    """Move mass from the coordinate that best explains the tallest lag."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()

        if stall > 300:
            break

        peak_lag = int(np.argmax(ac))
        js = max(0, peak_lag - (n - 1))
        je = min(n - 1, peak_lag)
        if js > je:
            stall += 1
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_lag - idx]
        src = int(idx[np.argmax(contrib)])
        if a[src] < 1e-10:
            stall += 1
            continue

        top_lags = np.argsort(ac)[::-1][:min(8, len(ac))]
        in_peak = np.zeros(n, dtype=bool)
        for kl in top_lags[:5]:
            kli = int(kl)
            s2 = max(0, kli - (n - 1))

        if len(cand) < 20:
            cand = np.arange(n)
        nsamp = min(300, len(cand))
        chosen = cand[rng.choice(len(cand), nsamp, replace=False)]

        best_dst = -1
        best_sens = float('inf')
        for dst in chosen:
            di = int(dst)
            if di == src:
                continue
            sens = 0.0
            for kl in top_lags[:4]:
                kli = int(kl)
                d1 = kli - di
                d2 = di - kli
                add = 0.0
                if 0 <= d1 < n:
                    add += a[d1]
                if 0 <= d2 < n and d2 != d1:
                    add += a[d2]
                sens += add
            if sens < best_sens:
                best_sens = sens
                best_dst = di

        if best_dst < 0:
            stall += 1
            continue

        improved = False

            if amt < 1e-14:

            tr = a.copy()
            tr[src] -= amt
            tr[best_dst] += amt
            tr = np.maximum(tr, 0.0)
            st = tr.sum()
            if st < 1e-15:
                continue
            tr = tr / st
            mt = fast_autocorr(tr).max()
            if mt < best_m - 1e-14:
                a = tr
                best_m = mt
                best_a = a.copy()
                stall = 0

            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.05, 0.15)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    mn = fast_autocorr(a).max()
                    if mn < best_m:

                        best_a = a.copy()
                        stall = 0

    if verbose:
        print("    PeakShave (%.1fs): J=%.6f" %
              (time.monotonic() - t0, 2.0 * n * best_m))
    return best_a, best_m


def pair_swap_random(a0, time_limit, verbose=True, rng_seed=42,
                     n_trials=400):
    """Contribution-biased random pair-swap local polish."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 300:
            break

        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:min(6, len(ac))]
        contrib = np.zeros(n)
        for kl in top_lags:
            kli = int(kl)
            js = max(0, kli - (n - 1))
            je = min(n - 1, kli)
            if js <= je:
                idx = np.arange(js, je + 1)
                contrib[idx] += a[idx] * a[kli - idx]
        sw = contrib + 1e-10
        sw = sw / sw.sum()

        improved = False
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            if stall > 300:
                break
            src = int(rng.choice(n, p=sw))

            tr[dst] += amt

                improved = True
                stall = 0

            stall += 1

    if verbose:
        print("    PairSwap (%.1fs): J=%.6f" %

    return best_a, best_m

        remaining = time_budget - (time.monotonic() - t_global)

    # Final refinement on the global best.
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("=== Final refine n=%d (budget=%.0fs) ===" % (n_best, remaining))
    t6 = min(remaining * 0.35, 150.0)
    if t6 > 30:
        a_out, _ = subgradient_polyak_anderson(
            normalize(best_seq), t6, verbose=True, max_hist=6,
            restart_period=300, rng_seed=55)
        try_update(a_out, "pg_final")

    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.4, 120.0)
    if t7 > 20:
        a_out, _ = peak_shaving(normalize(best_seq), t7, verbose=True,
                                rng_seed=123)
        try_update(a_out, "peak_final")

    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.8, 80.0)
    if t8 > 15:
        a_out, _ = pair_swap_random(
            normalize(best_seq), t8, verbose=True, rng_seed=31415,
            n_trials=500)
        try_update(a_out, "pair_final")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("FINAL: score=%.6f n=%d time=%.0fs" %
              (best_score, len(best_seq), time.monotonic() - t_global))
        return [float(x) for x in best_seq]

    print("ERROR: fallback")
    return [1.0]
