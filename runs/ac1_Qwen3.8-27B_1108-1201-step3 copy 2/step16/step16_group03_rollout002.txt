def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        result = float(evaluate_sequence(a_list))
        return result
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective_value(a):
    """Compute J = 2n * max(conv) / (sum)^2 for normalized a."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    ac = fast_conv(a_norm)
    return 2.0 * n * ac.max()

def subgradient_at_k(y, k):
    """Compute subgradient of (y*y)(k) w.r.t. y at lag k."""
    n = len(y)
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * y[j]
    return grad

def kkt_direction(y, active_lags, num_right=0):
    """
    Compute the best direction using KKT conditions.
    We want to find d such that:
    - sum(d) = 0
    - For each active lag k: (y*d)(k) <= 0 (reduce the peaks)
    - y + d >= 0 (non-negativity)
    
    We solve: min_d max_{k in K} (y*d)(k) subject to sum(d)=0, y+d >= 0
    This is a linear program in d.
    But for efficiency, we use a simpler approach: weighted subgradient.
    """
    n = len(y)
    # Weighted average of subgradients at active lags
    # Weights proportional to how close each lag is to the max
    ac = fast_conv(y)
    M = ac.max()
    # Only consider lags within a tolerance of the max
    tol = 0.01 * M
    active_mask = ac >= (M - tol)
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(ac)])
    
    # Weights: proportional to the convolution value at each active lag
    w = ac[active_idx]
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, active_idx):
        grad += wi * subgradient_at_k(y, int(k))
    
    # Project onto sum(d) = 0
    grad -= np.mean(grad)
    
    # Ensure y + alpha*grad won't go negative (for small alpha)
    # Find max alpha such that y + alpha*(-grad) >= 0
    # Actually we want to go in the direction that reduces the peaks
    direction = -grad
    
    return direction, active_idx

def exact_line_search(y, direction, max_iter=50):
    """
    Find the best step size alpha such that max(conv((1-alpha)*y + alpha*(y+direction))) is minimized.
    Actually, we're moving from y in direction d: new = y + alpha*d, normalized.
    """
    n = len(y)
    y = np.asarray(y, dtype=np.float64)
    direction = np.asarray(direction, dtype=np.float64)
    
    # Find alpha range where y + alpha*direction >= 0
    # y_i + alpha * direction_i >= 0 => alpha >= -y_i / direction_i (if direction_i < 0)
    # and alpha <= -y_i / direction_i (if direction_i > 0)
    
    alpha_min = 0.0
    alpha_max = 1.0
    
    neg_mask = direction < -1e-15
    if np.any(neg_mask):
        # y_i + alpha * direction_i >= 0 => alpha <= y_i / (-direction_i)
        candidates = y[neg_mask] / (-direction[neg_mask])
        alpha_max = min(alpha_max, candidates.min())
    
    if alpha_max < 1e-10:
        return 0.0, y.copy()
    
    # Try several alpha values
    best_alpha = 0.0
    best_val = fast_conv(y).max()
    
    # Use golden section or bisection
    # For simplicity, try a grid
    alphas = np.concatenate([
        np.linspace(0.0, min(alpha_max, 0.1), 10),
        np.linspace(0.1, min(alpha_max, 0.5), 10),
        np.linspace(0.5, alpha_max, 10),
        [alpha_max * 0.99]
    ])
    alphas = np.unique(alphas[(alphas >= 0) & (alphas <= alpha_max)])
    
    for alpha in alphas:
        trial = y + alpha * direction
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            continue
        trial = trial / s
        val = fast_conv(trial).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Refine around best_alpha
    if best_alpha > 0:
        lo = max(0, best_alpha * 0.8)
        hi = min(alpha_max, best_alpha * 1.2)
        for _ in range(20):
            mid = (lo + hi) / 2
            trial = y + mid * direction
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-15:
                break
            trial = trial / s
            val = fast_conv(trial).max()
            trial_lo = y + lo * direction
            trial_lo = np.maximum(trial_lo, 0.0)
            trial_lo = trial_lo / trial_lo.sum()
            val_lo = fast_conv(trial_lo).max()
            
            if val < val_lo:
                lo = mid
            else:
                hi = mid
    
    trial = y + best_alpha * direction
    trial = np.maximum(trial, 0.0)
    s = trial.sum()
    if s > 1e-15:
        trial = trial / s
    return best_alpha, trial

def kkt_optimize(a0, time_limit=60.0, verbose=False):
    """
    KKT-based iterative optimization.
    At each step, identify active constraints and solve for the optimal direction.
    """
    a = norm_sum1(a0)
    n = len(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        if M < 1e-12:
            break
        
        # Find active lags (within 1% of max)
        tol = 0.005 * M
        active_idx = np.where(ac >= M - tol)[0]
        
        if verbose and it % 100 == 0:
            print(f"    KKT iter {it}: M={M:.8f}, n_active={len(active_idx)}")
        
        # Compute weighted subgradient
        w = ac[active_idx] ** 2  # Square to emphasize the peaks
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, k in zip(w, active_idx):
            grad += wi * subgradient_at_k(a, int(k))
        
        # Project gradient onto the hyperplane sum(x) = 0
        grad_proj = grad - np.mean(grad)
        
        # Direction: negative projected gradient
        direction = -grad_proj
        
        # Line search
        alpha, a_new = exact_line_search(a, direction)
        
        if alpha < 1e-10:
            # Try a random perturbation
            rng = np.random.default_rng(it)
            pert = rng.normal(0, 0.01, n)
            a_new = a + pert
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            # Only accept if it improves
            M_new = fast_conv(a_new).max()
            if M_new >= M:
                break
        
        a = a_new
    
    return best_a, best_m

def lp_frank_wolfe_fast(a0, time_limit=120.0, solver=None):
    """LP-based Frank-Wolfe with momentum and exact line search."""
    if solver is None:
        solver = cp.CLARABEL
    
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = y.copy()
        
        # Build constraint matrix
        # (T @ b)[k] = (y * b)[k] for k = 0, ..., 2n-2
        # T has shape (2n-1, n)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            start = j
            end = j + n
            T[start:end, j] = y
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        
        direction = b_val / b_sum
        
        # Line search
        alphas = [2.0 / (t + 2.0)] + list(np.linspace(0.05, 0.9, 8))
        best_alpha = alphas[0]
        best_trial_m = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()
        
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            val = fast_conv(trial).max()
            if val < best_trial_m:
                best_trial_m = val
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s > 1e-15:
            y_new = y_new / s
        else:
            y_new = y.copy()
        
        # Check momentum point
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            x_mom = y_new + beta_t * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sm = x_mom.sum()
            if sm > 1e-15:
                x_mom = x_mom / sm
                M_mom = fast_conv(x_mom).max()
                if M_mom < best_m:
                    best_m = M_mom
                    best_a = x_mom.copy()
        
        y = y_new
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter {t} ({elapsed:.1f}s): best_M={best_m:.8f}")
    
    return best_a, best_m

def coordinate_descent_advanced(a0, time_limit=60.0, verbose=False, seed=42):
    """Advanced coordinate descent with sweep over all elements."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_conv(a)
        M = ac.max()
        k_star = int(np.argmax(ac))
        
        improved = False
        # Sweep: try moving mass from each position to reduce the peak
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            # Compute the gradient contribution of a[i] to the peak
            # d/d a_i [(a*a)(k_star)] = 2 * a[k_star - i] (if valid)
            j = k_star - i
            if 0 <= j < n:
                contrib = a[j]
            else:
                contrib = 0
            
            if contrib < 1e-15:
                continue
            
            # Try moving a small fraction of a[i] to various other positions
            # Best target: position j' where the impact on all top peaks is minimal
            top_peaks = np.argsort(ac)[::-1][:5]
            
            best_target = -1
            best_score = float('inf')
            
            # Sample candidate targets
            candidates = np.random.choice(n, size=min(n, 100), replace=False)
            for dst in candidates:
                if dst == i:
                    continue
                # Score: impact on top peaks
                score = 0
                for k in top_peaks:
                    k_int = int(k)
                    # Moving mass from i to dst changes (a*a)(k) by:
                    # -2*a[k-i]*amt + 2*a[k-dst]*amt (approximately, for small amt)
                    # We want this to be negative (reduce the peak)
                    dj = k_int - i
                    dd = k_int - dst
                    delta_i = a[dj] if 0 <= dj < n else 0
                    delta_d = a[dd] if 0 <= dd < n else 0
                    score += delta_d - delta_i
                if score < best_score:
                    best_score = score
                    best_target = dst
            
            if best_target < 0:
                continue
            
            dst = best_target
            # Try fraction of mass to move
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                M_trial = fast_conv(a_trial).max()
                if M_trial < M - 1e-14:
                    a = a_trial
                    M = M_trial
                    if M < best_m:
                        best_m = M
                        best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random restart from current point
            if stall > 50:
                pert = rng.normal(0, 0.02, n)
                a = a + pert * a
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
    
    return best_a, best_m

def parametric_init(n, variant=0, seed=0):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    xc = (x - n / 2) / n  # centered in [-0.5, 0.5]
    
    if variant == 0:
        # Uniform
        a = np.ones(n)
    elif variant == 1:
        # Semicircle
        a = np.sqrt(np.maximum(n * n / 4 - (x - n / 2) ** 2, 0))
    elif variant == 2:
        # Edge-heavy (U-shaped)
        a = 1.0 / (1.0 + 10.0 * np.cos(np.pi * x / n) ** 2)
    elif variant == 3:
        # Gaussian centered
        sigma = n / 4
        a = np.exp(-x ** 2 / (2 * sigma ** 2))
    elif variant == 4:
        # Two peaks
        sigma = n / 8
        a = np.exp(-(x - n * 0.2) ** 2 / (2 * sigma ** 2)) + np.exp(-(x - n * 0.8) ** 2 / (2 * sigma ** 2))
    elif variant == 5:
        # Cosine modulation
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x / n)
    elif variant == 6:
        # Random smooth
        a = np.abs(rng.normal(1, 0.5, n))
        # Smooth
        kernel = np.exp(-np.arange(-20, 21) ** 2 / 50)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    elif variant == 7:
        # Power law at edges
        a = np.maximum(x, 1) ** 0.5 * np.maximum(n - x, 1) ** 0.5
    elif variant == 8:
        # Sinc-like
        a = np.abs(np.sinc(x / (n * 0.1)))
    elif variant == 9:
        # Triangle wave
        a = 1.0 - np.abs(2.0 * (x / n) - 1.0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
