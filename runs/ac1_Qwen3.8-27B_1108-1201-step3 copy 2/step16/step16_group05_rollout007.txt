def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # = 2 * a[k-t] (if 0 <= k-t < n)
        
        # Wait, that's only for k > 0. For k = 0:
        # (a*a)[0] = sum_i a[i]*a[i] = sum a[i]^2
        # d/d a[t] = 2*a[t]
        
        if k_int == 0:
            grad[t] += 2.0 * wi * a[t]  # This should be a vector operation
        else:
            # d/d a[t] (a*a)[k] = 2*a[k-t] if 0 <= k-t < n
            # (the factor of 2 comes from both a[i] and a[k-i] contributing)
            # Wait, let me be more careful.
            # (a*a)[k] = sum_{i} a[i]*a[k-i]
            # = a[0]*a[k] + a[1]*a[k-1] + ... + a[k]*a[0] + ... (terms where both indices are valid)
            # d/d a[t]: 
            #   From term i=t: a[k-t] (if 0 <= k-t < n)
            #   From term i=k-t: a[k-t] (if 0 <= k-t < n) -- same as above!
            #   Wait no. The term with i=k-t is a[k-t]*a[t], and d/d a[t] of this is a[k-t].
            #   The term with i=t is a[t]*a[k-t], and d/d a[t] of this is a[k-t].
            #   So total is 2*a[k-t] (if 0 <= k-t < n and t != k-t, i.e., k != 2t)
            #   If k = 2t, the term is a[t]*a[t] = a[t]^2, and d/d a[t] = 2*a[t] = 2*a[k-t]. Same formula!
            # So: d/d a[t] (a*a)[k] = 2*a[k-t] if 0 <= k-t < n, else 0.
            
            valid = np.zeros(n, dtype=bool)
            # k-t must be in [0, n-1], so t must be in [k-(n-1), k] = [k-n+1, k]
            lo = max(0, k_int - n + 1)
            hi = min(n - 1, k_int)
            valid[lo:hi+1] = True
            grad[lo:hi+1] += 2.0 * wi * a[k_int - np.arange(n)[lo:hi+1]]
    
    return grad, ac.max()

def riemannian_gradient_descent(a0, time_limit, verbose=True, seed=42):
    """
    Riemannian gradient descent on the simplex with smooth max approximation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Parameters
    top_k = 100
    beta = 20.0
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    max_stall = 500
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive parameters
        frac = min(1.0, t / 50000.0)
        cur_beta = 20.0 + (5.0 - 20.0) * frac
        cur_top_k = int(100 + 100 * frac)
        cur_lr = lr / (1.0 + 0.01 * t)
        
        # Compute gradient
        g, M_curr = compute_gradient_smoothed(a, top_k=cur_top_k, beta=cur_beta)
        
        # Momentum
        v = momentum * v + (1 - momentum) * g
        g_mom = v
        
        # Gradient step in Euclidean space, then project
        a_new = a - cur_lr * g_mom
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        a = a_new
        
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with perturbation if stalled
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            noise = rng.normal(0, 0.001 * (1 + t/10000.0), n) * best_a
            a_re = best_a + noise
            a_re = np.maximum(a_re, 0)
            a_re = a_re / a_re.sum()
            a = a_re
            v = np.zeros(n)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      RGD iter={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, top_k={cur_top_k}")
    
    return best_a, best_m

def freq_domain_smooth(a, amount=0.1):
    """
    Apply frequency-domain low-pass filtering to reduce autocorrelation peaks.
    amount: fraction of high-frequency energy to remove (0 = no smoothing, 1 = heavy smoothing)
    """
    n = len(a)
    A = np.fft.rfft(a)
    freqs = np.arange(len(A))
    # Create a smooth low-pass filter
    # Keep low frequencies, attenuate high frequencies
    cutoff = int(len(A) * (1 - amount))
    if cutoff < 1:
        cutoff = 1
    # Smooth transition
    transition = max(1, len(A) // 20)
    mask = np.ones(len(A))
    mask[cutoff:] = 0
    if cutoff + transition < len(A):
        trans_idx = np.arange(cutoff, min(cutoff + transition, len(A)))
        mask[trans_idx] = 1.0 - (trans_idx - cutoff) / transition
    A_filtered = A * mask
    a_new = np.fft.irfft(A_filtered, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def lp_direction(a, M, solver=None, max_constraints=200):
    """
    Find the Frank-Wolfe direction by solving LP:
    maximize sum(b) subject to (a*b)[k] <= M for top constraints, b >= 0.
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Select most important constraints (top lags)
    top_lags = np.argsort(ac)[::-1][:max_constraints]
    
    # Build constraint matrix for selected lags
    # (a*b)[k] = sum_i a[i]*b[k-i] for valid i
    # = sum_j b[j] * a[k-j] for valid j
    # So the constraint is: sum_j a[k-j] * b[j] <= M
    # The row for lag k is: [a[k-(n-1)], ..., a[k], ..., a[k-0]] (padded to length n)
    
    T = np.zeros((len(top_lags), n))
    for idx, k in enumerate(top_lags):
        k_int = int(k)
        for j in range(n):
            val = k_int - j
            if 0 <= val < n:
                T[idx, j] = a[val]
    
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    return None

def fw_iteration(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe with LP directions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Get LP direction
        g_dir = lp_direction(y, M_curr, solver, max_constraints=100)
        if g_dir is None:
            break
        
        # FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        y = y_new
        
        # Track best
        M_new = fast_autocorr(y).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = y.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent with peak-aware moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:10]
        improved = False
        
        # Find the worst region and try to move mass from it
        peak_idx = top_k[0]
        
        # Compute contribution to peak
        k_int = int(peak_idx)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        idx = np.arange(lo, hi + 1)
        j_idx = k_int - idx
        contrib = a[idx] * a[j_idx]
        source = idx[np.argmax(contrib)]
        
        if a[source] < 1e-10:
            break
        
        # Find best destination (low impact on peaks)
        best_dst = -1
        best_delta = 0.0
        n_try = min(n, 200)
        candidates = rng.choice(n, n_try, replace=False) if n > n_try else np.arange(n)
        
        for dst in candidates:
            if dst == source:
                continue
            # Estimate impact on top peaks
            impact = 0.0
            for k in top_k[:5]:
                k_int2 = int(k)
                d = k_int2 - dst
                if 0 <= d < n:
                    impact += a[d]
                d2 = dst - k_int2
                if 0 <= d2 < n:
                    impact += a[d2]
            score = -impact
            if score > best_delta:
                best_delta = score
                best_dst = dst
        
        if best_dst < 0:
            break
        
        # Try moving
        for frac in [0.005, 0.01, 0.02, 0.05]:
            amt = a[source] * frac
            a_trial = a.copy()
            a_trial[source] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.001, 0.005)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
