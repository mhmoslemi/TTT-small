"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with smooth max (log-sum-exp) objective via log-parameterization
2. L-BFGS-B with hard max (peak cycling subgradient)

"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def to_log(a):
    """Convert normalized non-negative vector to log-parameterization."""
    a = np.maximum(a, 1e-30)
    return np.log(a) - np.log(a.sum())


def from_log(z):
    """Convert log-parameterization back to normalized non-negative vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def soft_obj_grad(z, alpha):
    """
    Compute soft-max objective and gradient w.r.t. z.
    a = softmax(z), objective = 2n * (1/alpha) * log(sum exp(alpha * (a*a)[k]))
    """
    n = len(z)
    a = from_log(z)
    conv = fast_conv(a)
    
    conv_scaled = alpha * conv
    conv_max = conv_scaled.max()
    shifted = np.exp(conv_scaled - conv_max)
    log_sum_exp = conv_max + np.log(shifted.sum())
    obj = 2.0 * n * log_sum_exp / alpha
    
    w = shifted / shifted.sum()
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] = 2 * correlate(w, a, 'valid')[m]
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # Chain rule through softmax: grad_z = a * (grad_a - <a, grad_a>)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return obj, grad_z


def hard_obj_grad(z, peak_idx=None):
    """
    Compute hard-max objective and subgradient w.r.t. z.
    """
    n = len(z)
    a = from_log(z)
    conv = fast_conv(n, a) if False else fast_conv(a)
    
    if peak_idx is not None:
        k_star = int(peak_idx)
    else:
        k_star = int(np.argmax(conv))
    
    obj = 2.0 * n * conv[k_star]
    
    # grad_a[m] = 2 * a[k_star - m] if 0 <= k_star - m < n
    grad_a = np.zeros(n)
    m_start = max(0, k_star - n + 1)
    m_end = min(n, k_star + 1)
    for m in range(m_start, m_end):
        j = k_star - m
        if 0 <= j < n:
            grad_a[m] = 2.0 * a[j]
    
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return obj, grad_z, k_star


def get_top_peaks(a, num=8):
    """Get indices of top peaks in convolution."""
    conv = fast_conv(a)
    return np.argsort(conv)[::-1][:num]

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
