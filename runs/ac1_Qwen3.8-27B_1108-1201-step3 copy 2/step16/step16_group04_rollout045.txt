        else:
            # Revert and reduce step size
            eta *= 0.95
        
        eta = max(eta * decay, 1e-10)

        if verbose and t % 2000 == 0:

def admm_autocorr(a0, time_limit, M_init=None, n_sub=50, verbose=True):
    """
    ADMM-style approach for min max_k R(k) s.t. sum(a)=1, a>=0.
    
    idea: alternate between:
    1. Given M, find a that minimizes sum(R(k)-M)^2_+ s.t. sum(a)=1, a>=0 (hard)
    2. Update M to be the current max R(k)
    
    Since step 1 is hard, we use a gradient-based approximation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    if M_init is None:
        M = fast_autocorr(a).max() * 0.9  # Start slightly below current peak
    
    rho = 1.0  # ADMM penalty parameter
    lambda_ = np.zeros(2 * n - 1)  # Dual variables for R(k) <= M
    
    for outer in range(n_sub):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Inner: minimize sum_k max(0, R(k) - M - lambda_k/rho)^2 + (rho/2) sum_k (z_k - R(k))^2
        # Simplified: just do a few gradient steps on the penalty
        for inner in range(10):
            ac = fast_autocorr(a)
            # Gradient of penalty: sum_k rho * (R(k) - M - lambda_k/rho) * dR(k)/da
            # where (R(k)-M-lambda_k/rho) > 0
            violations = np.maximum(ac - M - lambda_ / rho, 0)
            
            # Gradient of sum violations * dR/da
            grad = np.zeros(n)
            max_v = violations.max()
            if max_v < 1e-15:
                break
            
            # Only process top violators for efficiency
            n_proc = min(50, n)
            top_v_idx = np.argsort(violations)[::-1][:n_proc]
            
            for k in top_v_idx:
                k_int = int(k)
                v = violations[k_int]
                if v < 1e-15:
                    continue
                i_start = max(0, k_int - n + 1)
                i_end = min(n - 1, k_int)
                if i_start <= i_end:
                    idx = np.arange(i_start, i_end + 1)
                    grad[idx] += 2.0 * rho * v * a[k_int - idx]
            
            # Update a: projected gradient step
            eta = 1.0 / (np.linalg.norm(grad) + 1e-10) * 0.1
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a
            else:
                a_new = a_new / s
            a = a_new
        
        # Update dual variables
        ac = fast_autocorr(a)
        lambda_ = np.maximum(lambda_ + rho * (ac - M), 0)
        
        # Update M: reduce it gradually
        M = max(M * 0.995, ac.max() * 0.95)
        
        # Track best
        M_curr = fast_autocorr(a).max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
        
        if verbose and outer % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ADMM iter={outer} ({elapsed:.1f}s): J={J:.6f}, M_target={M:.6e}")
    
    return best_a, best_m

def coordinate_polish_v2(a0, time_limit, verbose=True, rng_seed=42):
    """
    Improved coordinate descent: try all pairs of (source, destination) 
    for the top peaks, with multiple step sizes.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    step_sizes = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            # Contribution to R(k): a[j] * a[k-j]
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Source: index with largest contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find best destination using sensitivity
            # Compute sensitivity of all top peaks to each position
            sens = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                i_start2 = max(0, k2i - n + 1)
                i_end2 = min(n - 1, k2i)
                if i_start2 <= i_end2:
                    idx2 = np.arange(i_start2, i_end2 + 1)
                    sens[idx2] += 2.0 * a[k2i - idx2]
            
            # Try to find a destination with low sensitivity
            # Sample candidates
            n_candidates = min(200, n)
            candidates = rng.choice(n, size=n_candidates, replace=False)
            
            best_dst = -1
            best_add = float('inf')
            for dst in candidates:
                if dst == src:
                    continue
                if sens[dst] < best_add:
                    best_add = sens[dst]
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            found = False
            for frac in step_sizes:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found:
                # Try shifting mass between two positions within the peak
                if len(idx) > 1:
                    i1_local = src_local
                    i2_local = (src_local + 1) % len(idx)
                    if i1_local != i2_local:
                        i1 = idx[i1_local]
                        i2 = idx[i2_local]
                        # Move mass from i1 to i2 (both contribute to this peak)
                        # This redistributes within the peak
                        for frac in [0.01, 0.02, 0.05]:
                            amt = a[i1] * frac
                            a_trial = a.copy()
                            a_trial[i1] -= amt
                            a_trial[i2] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            s_t = a_trial.sum()
                            if s_t > 1e-15:
                                a_trial = a_trial / s_t
                                M_trial = fast_autocorr(a_trial).max()
                                if M_trial < best_m - 1e-16:
                                    a = a_trial
                                    best_m = M_trial
                                    best_a = a.copy()
                                    improved = True
                                    stall = 0
                                    break
    
        if not improved:
            stall += 1
            # Perturbation
            n_pert = rng.integers(1, min(5, n))
            sources = rng.choice(n, size=n_pert, replace=False)
            dests = rng.choice(n, size=n_pert, replace=False)
            for s, d in zip(sources, dests):
                if s != d and a[s] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[s] * frac
                    a[s] -= amt
                    a[d] += amt
            a = np.maximum(a, 0.0)
            s_t = a.sum()
            if s_t > 1e-15:
                a = a / s_t
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordV2 ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def random_well_spread(n, rng, method='random'):
    """Generate a random well-spread initial sequence."""
    if method == 'random':
        a = rng.random(n)
    elif method == 'sine':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 + 0.1 * rng.random(n)
    elif method == 'exponential_decay':
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.05 * rng.random(n)
    elif method == 'biperiodic':
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) ** 2 + 0.05 * rng.random(n)
    elif method == 'triangle':
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1) + 0.05 * rng.random(n)
    else:
        a = rng.random(n)
    return normalize(a)
