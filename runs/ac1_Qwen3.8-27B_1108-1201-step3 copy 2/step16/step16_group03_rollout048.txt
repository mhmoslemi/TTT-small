    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(p, a):
    """
    Compute cross-correlation R[i] = sum_k p[k]*a[k-i] for i=0..n-1.
    Uses the identity: R[i] = (flip(p) conv a)[len(p)-1-i]
    """
    n = len(a)
    n_p = len(p)
    size = n_p + n - 1
    fft_size = 1 << (size - 1).bit_length()
    p_flip = p[::-1]
    result = np.fft.irfft(np.fft.rfft(p_flip, fft_size) * np.fft.rfft(a, fft_size), fft_size)[:size]
    # R[i] = result[n_p - 1 - i]
    return result[n_p - 1 - np.arange(n)]


def logsumexp_stable(x):
    """Numerically stable log-sum-exp."""
    m = x.max()
    if m > 700:
        return m + np.log(x.size)  # overflow protection
    if m < -700:
        return -np.inf
    return m + np.log(np.sum(np.exp(x - m)))


def smoothed_gradient(a, T):
    """
    Compute gradient of logsumexp(conv(a,a)/T) w.r.t. a.
    Returns (objective_value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    lc = logsumexp_stable(c / T)
    # p[k] = softmax(c[k]/T)
    p = np.exp(c / T - lc)
    # grad[i] = (2/T) * sum_k p[k] * a[k-i] = (2/T) * cross_corr(p, a)[i]
    cc = fast_cross_corr(p, a)
    grad = (2.0 / T) * cc
    obj = 2.0 * n * T * lc
    return obj, grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a0, time_limit, T_init=0.05, T_min=0.005, verbose=True, seed=42):
    """
    FISTA with log-sum-exp smoothed objective.
    Minimizes 2n * max(conv(a,a)) subject to a>=0, sum(a)=1.
    Uses smoothed surrogate for gradient, tracks true max for best solution.
    """
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0.0).copy())
    y = x.copy()
    t_mom = 1.0
    best_a = x.copy()
    best_true = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T = T_init
    step = 0.5
    stall = 0
    max_stall = 800
    iter_count = 0

    for it in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        T = max(T_min, T * 0.9997)

        # Compute smoothed gradient at y
        _, grad_y = smoothed_gradient(y, T)

        # Adaptive step size (backtracking)
        # Try step, if objective (smoothed) increases, halve step
        obj_y, _ = smoothed_gradient(y, T)
        trial = project_simplex(y - step * grad_y)
        obj_trial, _ = smoothed_gradient(trial, T)

        backtracks = 0
        while obj_trial > obj_y + 1e-12 and step > 1e-10 and backtracks < 10:
            step *= 0.5
            trial = project_simplex(y - step * grad_y)
            obj_trial, _ = smoothed_gradient(trial, T)
            backtracks += 1

        x_new = trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x_new.copy()
        else:
            y = y / sy
        x = x_new
        t_mom = t_new
        iter_count += 1

        # Track best using TRUE max
        true_max = fast_conv(x).max()
        if true_max < best_true - 1e-15:
            best_true = true_max
            best_a = x.copy()
            stall = 0
            step = min(step * 1.05, 2.0)
        else:
            stall += 1

        # Restart with perturbation if stuck
        if stall > max_stall:
            noise_scale = 0.002 + 0.003 * (stall // (2 * max_stall))
            x = best_a + rng.normal(0, noise_scale * best_a.max(), n)
            x = project_simplex(x)
            y = x.copy()
            t_mom = 1.0
            stall = 0
            T = max(T_min, min(T_init, T * 2.0))
            step = 0.5

        # Periodic larger jump
        if it % 20000 == 0 and it > 0:
            x_jump = best_a + rng.normal(0, 0.01 * best_a.max(), n)
            x_jump = project_simplex(x_jump)
            jm = fast_conv(x_jump).max()
            if jm < best_true:
                best_true = jm
                best_a = x_jump.copy()
                x = x_jump
                y = x_jump
                t_mom = 1.0
                stall = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_true
            print(f"    FISTA it={it} ({elapsed:.1f}s): score={score:.6f}, T={T:.5f}, step={step:.4f}, stall={stall}")

    return best_a, 2.0 * n * best_true


def peak_subgradient_descent(a0, time_limit, top_k=3, verbose=True, seed=42):
    """
    Nesterov-accelerated subgradient method using top-k peak weighting.
    """
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0.0).copy())
    x_prev = x.copy()
    y = x.copy()
    best_a = x.copy()
    best_true = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    c0 = 3.0
    stall = 0
    max_stall = 2000
    iter_count = 0

    for it in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta = (it - 1.0) / (it + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        c = fast_conv(y)
        M = c.max()

        # Top-k peak subgradient
        k_max = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = np.maximum(c[top_idx], 1e-30)
        w = top_vals ** 3.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += wi * 2.0 * y[j]

        # Mirror descent (multiplicative weights)
        eta = c0 / (np.sqrt(it) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new
        iter_count += 1

        true_max = fast_conv(x).max()
        if true_max < best_true - 1e-15:
            best_true = true_max
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            noise_scale = 0.003
            x = best_a + rng.normal(0, noise_scale * best_a.max(), n)
            x = project_simplex(x)
            x_prev = x.copy()
            stall = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_true
            print(f"    gradSD it={it} ({elapsed:.1f}s): score={score:.6f}, stall={stall}")

    return best_a, 2.0 * n * best_true


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent with targeted mass transfer.
    Moves mass from high-contributing positions to low-impact positions.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0.0).copy())
    best_a = a.copy()
    best_true = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_peaks = np.argsort(c)[::-1][:min(15, len(c))]
        improved = False

        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find destination with minimal impact on top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                # Sensitivity: how much would moving to dst affect peaks?
                sens = 0.0
                for k2 in top_peaks[:3]:
                    j = int(k2) - dst
                    if 0 <= j < n:
                        sens += a[j]
                if sens < best_score:
                    best_score = sens
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            found = False
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_true - 1e-16:
                    a = a_trial
                    best_true = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 50:
                # Random perturbation to escape
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_true:
                        best_true = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if not improved:
            stall += 1
        iter_count += 1

    if verbose:
        elapsed = time.monotonic() - t0
        score = 2.0 * n * best_true
        print(f"    Coord ({elapsed:.1f}s, {iter_count} iters): score={score:.6f}")
    return best_a, 2.0 * n * best_true


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initial(n, kind='sine'):
    """Generate initial sequence of given type."""
    x = np.linspace(0, 1, n)
    if kind == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'cosine':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif kind == 'hann':
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
        a[0] = 0.0
        a[-1] = 0.0
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif kind == 'plateau':
        a = np.ones(n)
        fade = max(1, n // 8)
        for i in range(fade):
            frac = (i + 1) / fade
            a[i] = frac
            a[n - 1 - i] = frac
    else:
        a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        result = evaluate_sequence(a_list)
        return float(result)
    except Exception:
        return float('inf')
