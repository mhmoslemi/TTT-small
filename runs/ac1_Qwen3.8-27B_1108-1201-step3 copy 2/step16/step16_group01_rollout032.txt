        idx = np.array(active)
        G_sub = G[np.ix_(idx, idx)]
        ones = np.ones(len(idx))

    return a_new / s


def euclidean_update(a, g, eta):
    """Euclidean gradient step with simplex projection."""
    a_new = a - eta * g
    a_new = simplex_project(a_new)
    return a_new


def compute_objective_normalized(a):
    """Compute max of autocorrelation (assuming sum(a)=1)."""
    ac = fast_autocorr(a)
    return ac.max()
