def peak_gradient(y, k):
    """Subgradient of (y*y)[k] w.r.t. y. Returns array of same length as y."""
    n = len(y)
    k = int(k)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * y[j]
    return grad

def soft_subgradient(y, ac, top_k=5, beta=2.0):
    """Weighted average of top-k peak subgradients."""
    n = len(y)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * peak_gradient(y, k)
    return grad

        top_peaks = np.argsort(ac)[::-1][:5]

        # Compute current spectrum
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # The time-domain peak is the max of inverse FFT of power
        # To reduce the peak, we want to redistribute power
        # Strategy: reduce power at frequencies that contribute most to the peak

        # Use the peak position to identify problematic frequencies
        k_star = int(top_peaks[0])
        # The peak at position k_star in the convolution corresponds to
        # oscillating components at frequency k_star/(n-1) approximately

        # Simple approach: apply a small perturbation in the frequency domain
        # that reduces the contribution to the peak
        phase = np.angle(A)
        mag = np.abs(A)

        # Perturb the magnitude to flatten the autocorrelation
        n_freq = len(mag)
        perturbation = rng.normal(0, 0.01 / np.sqrt(t + 1), n_freq)
        # Ensure DC component is preserved (for sum constraint)
        perturbation[0] = 0

        mag_new = np.maximum(mag * (1 + perturbation), 1e-10)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            # Try smaller perturbation
            perturbation_small = rng.normal(0, 0.005 / np.sqrt(t + 1), n_freq)
            perturbation_small[0] = 0
            mag_small = np.maximum(mag * (1 + perturbation_small), 1e-10)
            A_small = mag_small * np.exp(1j * phase)
            a_small = np.fft.irfft(A_small, fft_size)[:n]
            a_small = np.maximum(a_small, 0.0)
            s_small = a_small.sum()
            if s_small > 1e-15:
                a_small = a_small / s_small
                M_small = fast_conv(a_small).max()
                if M_small < best_m - 1e-14:
                    best_m = M_small
                    best_a = a_small.copy()
                    a = a_small.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

        if stall > max_stall:
            break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FPE t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive time allocation."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror descent n={n0} ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t1, verbose=True, seed=42, top_k_init=1, c0=3.0)
        try_update(a_out, f"md1_n{n0}")

    # === Phase 2: Mirror descent with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MD soft n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t2, verbose=True, seed=13, top_k_init=3, c0=4.0)
        try_update(a_out, f"md2_n{len(best_seq)}")

    # === Phase 3: Direct peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Direct peak reduction n={len(best_seq)} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = direct_peak_reduction(a_in, t3, verbose=True, seed=55)
        try_update(a_out, f"dpr_n{len(best_seq)}")

    # === Phase 4: Fourier peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Fourier equalization n={len(best_seq)} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fourier_peak_equalization(a_in, t4, verbose=True, seed=77)
        try_update(a_out, f"fpe_n{len(best_seq)}")

    # === Phase 5: Mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: MD polish n={len(best_seq)} ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t5, verbose=False, seed=99, top_k_init=2, c0=5.0)
        try_update(a_out, f"md_polish_n{len(best_seq)}")

    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        for delta in [-200, 200, -400, 400]:
            n_try = n_best + delta
            if n_try < 500 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 6: Length n={n_try} ===")
            a_long = interpolate_sequence(best_seq, n_try)
            a_in = norm_sum1(a_long)
            t6 = min(remaining * 0.35, 150.0)
            if t6 > 20:
                a_out, _ = mirror_descent(a_in, t6, verbose=False, seed=100 + n_try % 200, c0=3.0)
                try_update(a_out, f"md_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 40:
                    a_out2, _ = coordinate_polish(a_in, min(remaining * 0.3, 60.0), verbose=False, seed=200)
                    try_update(a_out2, f"coord_n{n_try}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord polish n={n_best} ===")
    if remaining > 30:
        t7 = min(remaining * 0.5, 120.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final MD with very small steps ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final MD n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.6, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent(a_in, t8, verbose=False, seed=555, top_k_init=1, c0=2.0)
            try_update(a_out, f"md_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
