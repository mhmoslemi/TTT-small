def compute_subgradient(a, conv, n, n_active=None, temperature=None):
    """Compute weighted subgradient of max_k (a*a)[k]."""
    M = conv.max()

    if temperature is not None and temperature > 0:
        # Soft weighting: all k contribute, weighted by exp(-gap/temperature)
        gaps = M - conv
        weights = np.exp(-gaps / temperature)
        weights /= weights.sum()
        active = np.arange(len(conv))
    else:
        # Hard weighting: top n_active peaks

        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        g[lo:hi] += 2.0 * w * a[k - np.arange(lo, hi)]
    return g

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    import cvxpy as cp
    n = len(a_fixed)
    
    T = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_optimize(a0, time_limit, solver, verbose=True, seed=0):
    """LP best-response with progressive tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    s = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * s / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 100:
            break
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        if M < best_score:
            best_score = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Active set
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 500.0)
            tol = 0.1 * (1 - progress) + 0.005 * progress
            threshold = M * (1.0 - tol)
            active_set = np.where(conv >= threshold)[0]
            n_top = max(5, n // 100)
            top_k = np.argsort(conv)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > 2 * n:
                conv_at = conv[active_set]
                keep = np.argsort(conv_at)[-2*n:]
                active_set = active_set[keep]
        
        b = lp_best_response(a, M, active_set, solver)
        if b is None:
            if t % 5 == 0:
                active_set = np.arange(2*n-1)
                b = lp_best_response(a, M, active_set, solver)
            if b is None:
                continue
        
        b = b * s / b.sum()
        
        # Nesterov-like mixing
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * s / a_new.sum()
        
        new_M = fast_autocorr(a_new).max()
        if new_M < M * 1.001:
            a = a_new
        else:
            a_new = 0.9 * a + 0.1 * b
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * s / a_new.sum()
            a = a_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP t={t} ({elapsed:.0f}s): M={M:.10f}, |A|={len(active_set)}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=True, seed=0):
    """Coordinate descent with FFT-based updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    s = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * s / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 6:
            break
        
        conv = fast_autocorr(a)
        improved = False
        
        for j in rng.permutation(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            # Try to find best value for a[j]
            # (a*a)[k] changes by 2*(x-aj)*a[k-j] for k in valid range
            # plus (x-aj)^2 at k=2j
            
            lo_k = max(0, j)
            hi_k = min(size, j + n)
            
            # Binary search for optimal x
            best_obj = conv.max()
            best_x = aj
            
            # Try several candidates
            candidates = np.unique(np.clip(
                [0.0, aj*0.1, aj*0.3, aj*0.5, aj*0.7, aj, aj*1.3, aj*1.7,
                 s/n * 0.5, s/n, s/n * 2, s/n * 4],
                [0.0, s]
            ))
            
            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                new_conv[lo_k:hi_k] += 2.0 * delta * a[lo_k - j: hi_k - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                
                # Renormalize
                new_sum = a.sum() + delta
                if new_sum < 1e-15:
                    continue
                scale = s / new_sum
                new_max = (new_conv * scale * scale).max()
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                a = a * s / a.sum()
                new_score = fast_autocorr(a).max()
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    conv = fast_autocorr(a)
                    improved = True
                else:
                    a = best_a.copy()
                    conv = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
            # Perturb to escape
            a = a * (1.0 + 0.02 * rng.standard_normal(n))
            a = np.maximum(a, 1e-15)
            a = a * s / a.sum()
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  CD ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score

def get_solver():
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS
