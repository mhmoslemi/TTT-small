# Global solver cache

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:

        return float(evaluate_sequence(a_list))

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                T_cols.append(i)

        prev_y = y

        x = x_new

def lp_fw_optimize(a0, time_limit, solver, max_K=500, verbose=False):
    """LP-based Frank-Wolfe with Nesterov acceleration and adaptive K."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    
    K = 50  # Start with small K
    K_target = min(max_K, 2*n-1)
    K_growth = 5  # Grow K by this amount
    iterations_since_K_change = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break
        
        # Select top-K constraints
        K = min(K, 2*n-1)
        binding_idx = np.argsort(ac)[::-1][:K]
        
        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row in range(K):
            k = int(binding_idx[row])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            length = i_end - i_start + 1

            cp.Maximize(cp.sum(b_var)),

        )

            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        if b_val is None or np.any(np.isnan(b_val)):
            continue
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / max(sy, 1e-15)

        else:

            sx = x_new.sum()
            x_new = x_new / max(sx, 1e-15)

        x = x_new
        
        my = fast_autocorr(y).max()

            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:

            best_a = x.copy()

            iterations_since_K_change = 0

            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.8f}, K={K}")

Now let me also think about the coordinate descent. The idea is:
- For each top autocorrelation lag k, identify the pairs (i, j) with i+j=k (or i-j=k for the other side) that contribute most to the autocorrelation.
- Move mass from high-contributing positions to low-contributing positions.
- Evaluate the improvement.

This is a local search that can find small improvements that the global methods miss.

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top contributing lags
        n_top = min(30, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            
            # Find the pair with maximum contribution
            best_i = idx[np.argmax(contrib)]
            
            # Compute sensitivity: for each position d, how much would adding mass increase the peak?
            sens = np.zeros(n)
            for k2 in top_lags[:min(15, len(top_lags))]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    if 0 <= partner < n:
                        sens[d] += a[partner]
            
            # Find position with lowest sensitivity (excluding current peak positions)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_lags[:min(10, len(top_lags))]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                peak_mask[s2:e2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Pick the one with lowest sensitivity
            best_dst = candidates[np.argmin(sens[candidates])]
            
            if best_dst == best_i:
                continue
            
            # Try moving mass
            src_val = a[best_i]
            if src_val < 1e-12:

            for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.25]:

                a_trial[best_i] -= amt

                    M_new = fast_autocorr(a).max()

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
