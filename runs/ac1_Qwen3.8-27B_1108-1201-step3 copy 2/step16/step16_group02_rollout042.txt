import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def linear_ramp(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()

def power_ramp(n, p):
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = np.power(a, p)
    return a / a.sum()

def exponential_ramp(n, alpha_scale):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha_scale * i / n)
    return a / a.sum()

def subgradient_descent(a0, time_limit, verbose=True, seed=42, lr_init=0.5, active_frac=0.95):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    lr = lr_init
    stall = 0
    max_stall = 300
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        ac = fast_autocorr(a)
        M = ac.max()
        threshold = M * active_frac
        active = np.where(ac >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(ac)])
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += a[j_idx]
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            break
        step = lr / (1.0 + 0.001 * it)
        a_new = a - step * grad / grad_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            a = best_a * (1.0 + rng.normal(0, 0.02, n))
            a = norm_sum1(a)
            stall = 0
            max_stall = max(100, max_stall * 2)
            lr = min(lr * 1.3, 5.0)
        a = a_new
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD it={it} ({elapsed:.1f}s): J={J:.8f}, active={len(active)}")
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=200, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        M_curr = ac.max()
        threshold = M_curr * 0.98
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        constraints = [T @ b_var <= M_curr + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        my = fast_autocorr(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()
        y = y_new
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.8f}")
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 8
    sweep = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        improved = False
        n_tries = min(30, n // 4)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.3]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): J={J:.8f}, stall={stall}")
    return best_a, best_m

def multi_start_search(a0, time_limit, n_starts, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    for s in range(n_starts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_sub = min(remaining / (n_starts - s), 120.0)
        perturb = rng.normal(0, 0.05, n)
        a_try = a * (1.0 + perturb)
        a_try = norm_sum1(a_try)
        a_out, m_out = subgradient_descent(a_try, t_sub, verbose=False, seed=100+s, lr_init=0.5)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MultiStart [{s}] ({elapsed:.1f}s): J={2*n*best_m:.8f}")
    return best_a, best_m
