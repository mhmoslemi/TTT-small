"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_max_grad(a, beta, top_k=20, tol=1e-10):
    """
    Compute gradient of (1/beta) * log(sum_{k in top} exp(beta * conv_k / max_conv))
    which approximates max_k conv_k.
    
    Returns (gradient, smoothed_max_value, top_lags)
    """
    n = len(a)
    ac = fast_autocorr(a)
    num_lags = len(ac)
    
    # Find top lags
    k_eff = min(top_k, num_lags)
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    max_val = top_vals[0]
    
    if max_val < tol:
        return np.zeros(n), 0.0, top_idx
    
    # Softmax weights over top lags
    shifted = beta * (top_vals - max_val)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Smoothed max value
    smooth_max = max_val + (1.0 / beta) * np.log(exp_vals.sum() / k_eff)
    
    # Gradient: sum of w_k * grad(conv_k)
    # grad of conv(a,a)[k] = 2 * (Q_k @ a) where (Q_k)_ij = 1 if i+j=k
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        # (Q_k @ a)[i] = a[k-i] for valid i
        # So grad contribution from lag k: 2 * w * a[k-i] for i in valid range
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad, smooth_max, top_idx

def line_search(a, grad, beta_ls=0.95, max_iter=50):
    """Backtracking line search on the simplex."""
    n = len(a)
    ac_curr = fast_autocorr(a)
    max_curr = ac_curr.max()
    
    step = 1.0
    for _ in range(max_iter):
        a_new = project_simplex(a - step * grad)
        ac_new = fast_autocorr(a_new)
        max_new = ac_new.max()
        if max_new < max_curr:
            return a_new, max_new
        step *= beta_ls
        if step < 1e-12:
            break
    return a, max_curr

def lp_polish(a0, time_limit, top_k=20, verbose=True):
    """
    LP-based Frank-Wolfe: at each step, solve LP maximizing sum(b)
    subject to conv(a,b)[k] <= t for top-k active lags, b >= 0.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t_iter in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        if M < 1e-14:
            break
        
        # Get active lags
        active_idx = np.argsort(ac)[::-1][:top_k]
        
        # Build LP: max sum(b) s.t. a^T T_k b <= M for k in active, b >= 0
        # T_k b = conv(a, b)[k] (cross-convolution)
        # conv(a, b)[k] = sum_i a[i] * b[k-i]
        
        # This is a linear constraint: for each active k, sum_i a[i]*b[k-i] <= M
        # Variables: b[0], ..., b[n-1]
        
        # Build constraint matrix
        # For lag k: sum_i a[i]*b[k-i] <= M
        # b[j] appears in the constraint for lag k when j = k-i, i.e., i = k-j
        # So coefficient of b[j] in constraint k is a[k-j] (if 0 <= k-j < n)
        
        n_cons = len(active_idx)
        A_mat = np.zeros((n_cons, n))
        for ci, k in enumerate(active_idx):
            k_int = int(k)
            for j in range(n):
                i = k_int - j
                if 0 <= i < n:
                    A_mat[ci, j] = a[i]
        
        # Solve LP: max 1^T b s.t. A_mat @ b <= M, b >= 0
        # Use scipy or manual
        try:
            from scipy.optimize import linprog
            c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
            A_ub = A_mat
            b_ub = np.full(n_cons, M)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status == 0:
                b_val = np.maximum(res.x, 0)
                b_sum = b_val.sum()
                if b_sum > 1e-15:
                    g_dir = b_val / b_sum
                    # FW step
                    alpha = 2.0 / (t_iter + 2.0)
                    a_new = (1 - alpha) * a + alpha * g_dir
                    a_new = np.maximum(a_new, 0)
                    s = a_new.sum()
                    if s > 1e-15:
                        a_new = a_new / s
                        M_new = fast_autocorr(a_new).max()
                        if M_new < best_max - 1e-14:
                            best_max = M_new
                            best_a = a_new.copy()
                        a = a_new
        except Exception:
            pass
        
        if verbose and t_iter % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print(f"      LP t={t_iter} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_max

def coordinate_polish(a0, time_limit, verbose=True, seed=123):

        top_lags = np.argsort(ac)[::-1][:10]
        improved = False
        
        # Random coordinate sweep
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try moving mass from i to other positions
            if a[i] < 1e-12:
                continue
            
            # Find best destination
            best_j = -1
            best_val = M_curr
            # Sample some destinations
            n_try = min(100, n)
            candidates = rng.choice(n, n_try, replace=False)
            for j in candidates:
                if j == i:
                    continue
                # Try moving a small fraction

                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt

                    if s < 1e-15:

                    a_trial = a_trial / s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_val:

                        best_amt = amt
            
            if best_j >= 0 and best_val < M_curr - 1e-14:
                a[best_j] += best_amt
                a[i] -= best_amt
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                M_curr = best_val
                if M_curr < best_max - 1e-14:
                    best_max = M_curr
                    best_a = a.copy()
                improved = True

        else:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
