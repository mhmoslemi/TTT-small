    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    # Scale v to have sum target_sum first (if needed)
    # Standard simplex projection algorithm
    v = v.copy()
    if v.sum() == target_sum:
        return np.maximum(v, 0)
    
    # Project onto {x >= 0, sum(x) = target_sum}
    # This is: argmin ||x - v||^2 s.t. x >= 0, sum(x) = target_sum
    # Solution: x[i] = max(v[i] - tau, 0) where tau is chosen so sum = target_sum
    mu_vals = np.sort(v)[::-1]
    cv = np.cumsum(mu_vals)
    rho = np.arange(1, n+1)
    cond = mu_vals - (cv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_star = np.max(np.where(cond)[0]) + 1
    tau = (cv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(v - tau, 0)
