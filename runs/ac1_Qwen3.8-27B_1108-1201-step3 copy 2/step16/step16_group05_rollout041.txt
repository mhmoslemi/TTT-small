def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cv = np.arange(1, n + 1)
    cond = u * cv > (cssv - 1)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_max(a, top_k=80, beta=0.0):
    """Compute (sub)gradient of smoothed max of autocorrelation.
    Uses exact formula: d(a*a)[k]/d(a[j]) = 2*a[k-j] if 0<=k-j<n else 0.
    """
    n = len(a)
    ac = fast_autocorr(a)
    num_lags = len(ac)
    k_eff = min(top_k, num_lags)
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx].copy()
    max_val = top_vals[0]
    
    if beta > 0 and max_val > 1e-15:
        shifted = (top_vals - max_val) / max_val
        exp_vals = np.exp(-beta * (-shifted))  # = exp(beta*(val - max)/max)
        # Actually: weight_k = exp(beta * (val_k - max_val) / max_val)
        exp_vals = np.exp(beta * (top_vals - max_val) / max_val)
        weights = exp_vals / exp_vals.sum()
    else:
        weights = np.ones(k_eff) / k_eff
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # d/d a[j] (a*a)[k] = 2*a[k-j] if 0 <= k-j < n
        # So grad[j] += 2*w*a[k-j] for valid j
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        # For j in [j_start, j_end): k-j ranges from k-j_start to k-j_end
        # a[k-j] where j goes from j_start to j_end-1
        # k-j goes from k-j_start down to k-(j_end-1) = k-j_end+1
        idx_in_a = np.arange(k_int - j_end + 1, k_int - j_start + 1)
        # We need a[idx_in_a] but in reverse order to match j ordering
        # j = j_start: k-j = k-j_start
        # j = j_end-1: k-j = k-j_end+1
        # So a indices go from k-j_start down to k-j_end+1
        a_slice = a[idx_in_a][::-1]
        grad[j_start:j_end] += 2.0 * w * a_slice
    
    return grad, ac.max()


def nesterov_apg(a0, time_limit, top_k_start=60, top_k_end=200,
                 beta_start=30.0, beta_end=3.0, L_est=5.0,
                 verbose=True, seed=42):
    """Nesterov-accelerated projected gradient with adaptive peak tracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    L = L_est

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 30000.0)
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        cur_beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        g, M_y = compute_gradient_max(y, top_k=cur_top_k, beta=cur_beta)

        # Adaptive L estimate
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15
        L = max(L * 0.999, 2.0 * cur_top_k / n * 2.0)
        eta = 2.0 / (L * (t + 2.0))
        # Cap step size
        eta = min(eta, 1.0 / (t * gn + 1e-10))
        eta = max(eta, 1e-8)

        x_new = project_simplex(y - eta * g)

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            # Restart with perturbation
            sigma = 0.01 + 0.02 * (1.0 - min(1.0, t / 50000.0))
            noise = rng.normal(0, sigma, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-14)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}")

    return best_a, best_m


def polynomial_fit_optimize(a_best, n, time_limit, verbose=True):
    """Fit polynomial shapes and evaluate them for global improvements."""
    x = np.linspace(0.0, 1.0, n)
    best = a_best.copy()
    best_m = fast_autocorr(a_best).max()
    t0 = time.monotonic()

    shapes = []
    # Power laws (increasing)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        shapes.append((x ** p))
    # Power laws (decreasing)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        shapes.append(((1.0 - x) ** p))
    # Beta-like shapes
    for a_p, b_p in [(2, 5), (3, 7), (1, 3), (1, 5), (2, 8), (3, 10),
                     (5, 2), (7, 3), (3, 1), (5, 1)]:
        shapes.append((x ** a_p) * ((1.0 - x) ** b_p))
    # Shifted bumps
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.15, 0.2, 0.25, 0.3]:
            shapes.append(np.exp(-(x - center) ** 2 / (2 * width ** 2)))

    # Also try blends with current best
    best_norm = norm_sum1(a_best)
    for frac in [0.3, 0.5, 0.7]:
        for sh in shapes[:6]:
            shapes.append((1.0 - frac) * best_norm + frac * sh)

    for sh in shapes:
        if time.monotonic() - t0 > time_limit:
            break
        sh = np.maximum(sh, 0.0)
        s = sh.sum()
        if s < 1e-15:
            continue
        sh = sh / s
        m = fast_autocorr(sh).max()
        if m < best_m - 1e-10:
            best_m = m
            best = sh.copy()
            if verbose:
                print(f"      Poly shape: J={2.0*n*best_m:.6f}")

    return best, best_m


def pairwise_coord_descent(a0, time_limit, verbose=True, seed=42):
    """Pairwise coordinate descent: move mass between index pairs to reduce peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:15]

        improved = False
        # Try moves from high-index to low-peak-impact positions
        # Score each index by its contribution to top peaks
        contrib = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start >= j_end:
                continue
            idx_in_a = np.arange(k_int - j_end + 1, k_int - j_start + 1)
            a_slice = a[idx_in_a][::-1]
            contrib[j_start:j_end] += a_slice

        # Source: high contribution. Destination: low contribution.
        src_candidates = np.argsort(contrib)[::-1][:40]
        dst_candidates = np.argsort(contrib)[:40]

        try_count = 0
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                try_count += 1
                if try_count > 500:
                    break
                if dst == src or contrib[dst] > 0.5 * contrib[src]:
                    continue
                # Try moving mass
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if stall > 50:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.003, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()

    # Pre-build Toeplitz matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a  # This will be updated

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Update Toeplitz
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        mc = fast_autocorr(y).max()
        if mc < best_m - 1e-15:
            best_m = mc
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_structured_init(n, seed=0):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    inits = []
    
    # Asymmetric power laws
    for p in [0.5, 1.0, 2.0]:
        inits.append((x ** p))
        inits.append(((1.0 - x) ** p))
    
    # Beta-like
    for ap, bp in [(2, 5), (5, 2), (3, 8), (8, 3), (1, 4), (4, 1)]:
        inits.append((x ** ap) * ((1.0 - x) ** bp) + 1e-6)
    
    # Exponential decay/growth
    for rate in [2, 5, 10]:
        inits.append(np.exp(-rate * x))
        inits.append(np.exp(-rate * (1 - x)))
    
    # Random smooth
    coeff = rng.normal(0, 1, 8)
    inits.append(np.abs(np.sum(coeff[i] * np.cos(np.pi * x * (i+1)) for i in range(8))) + 1e-6)
    
    for sh in inits:
        sh = np.maximum(sh, 0.0)
        s = sh.sum()
        if s > 1e-15:
            inits[inits.index(sh)] = sh / s
        else:
            inits[inits.index(sh)] = np.ones(n) / n
    
    return inits

        print(f"\n=== Phase 2: LP-FW n={len(best_seq)} (budget={t2:.0f}s) ===")
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lpfw_n{len(best_seq)}")

    # Phase 3: APG with different params
    rem = remaining_time()
    t3 = min(rem * 0.15, 240.0)
    if t3 > 30:
        print(f"\n=== Phase 3: APG v2 n={len(best_seq)} (budget={t3:.0f}s) ===")
        a_out, _ = nesterov_apg(best_seq, t3, top_k_start=80, top_k_end=250,
                                beta_start=15.0, beta_end=2.0,
                                L_est=6.0, verbose=True, seed=77)
        try_update(a_out, f"apg2_n{len(best_seq)}")

    # Phase 4: Coordinate descent polish
    rem = remaining_time()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: CD n={len(best_seq)} (budget={t4:.0f}s) ===")
        a_out, _ = pairwise_coord_descent(best_seq, t4, verbose=True, seed=42)
        try_update(a_out, f"cd_n{len(best_seq)}")

    # Phase 5: Polynomial shape exploration + APG
    rem = remaining_time()
    if rem > 200:
        n_test = len(best_seq)
        t5a = min(rem * 0.05, 60.0)
        t5b = min(rem * 0.20, 240.0)
        if t5a > 10 and t5b > 30:
            print(f"\n=== Phase 5: Poly explore n={n_test} ===")
            a_poly, _ = polynomial_fit_optimize(best_seq, n_test, t5a, verbose=True)
            try_update(a_poly, "poly_shape")
            a_out, _ = nesterov_apg(a_poly, t5b, top_k_start=60, top_k_end=200,
                                    beta_start=25.0, beta_end=4.0,
                                    L_est=5.0, verbose=True, seed=123)
            try_update(a_out, f"poly_apg_n{n_test}")

    # Phase 6: Try different lengths
    rem = remaining_time()
    n_best = len(best_seq)
    if rem > 150:
        for delta in [300, -300, 600]:
            n_try = n_best + delta
            if n_try < 500 or n_try > 4000:
                continue
            if rem < 80:
                break
            print(f"\n=== Phase 6: n={n_try} ===")
            a_long = interpolate_sequence(best_seq, n_try)
            t6 = min(rem * 0.15, 120.0)
            if t6 > 20:
                a_out, _ = nesterov_apg(a_long, t6, top_k_start=60, top_k_end=180,
                                        beta_start=25.0, beta_end=5.0,
                                        L_est=5.0, verbose=True, seed=200 + delta)
                try_update(a_out, f"len{n_try}_apg")
            rem = remaining_time()
            if rem > 40:
                t6c = min(rem * 0.1, 60.0)
                if t6c > 15:
                    a_out2, _ = pairwise_coord_descent(a_out, t6c, verbose=False, seed=300)
                    try_update(a_out2, f"len{n_try}_cd")
            rem = remaining_time()

    # Phase 7: Multi-start with structured inits
    rem = remaining_time()
    if rem > 100:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Multi-start n={n_best} ===")
        inits = generate_structured_init(n_best, seed=42)
        # Also blend with current best
        best_norm = norm_sum1(best_seq)
        extra = []
        for sh in inits[:4]:
            extra.append(norm_sum1(0.5 * best_norm + 0.5 * sh))
        all_inits = inits + extra
        
        t_per = min(rem * 0.12, 80.0)
        for idx, init in enumerate(all_inits[:8]):
            rem = remaining_time()
            if rem < 30:
                break
            t7 = min(t_per, rem - 15)
            if t7 > 15:
                a_out, _ = nesterov_apg(init, t7, top_k_start=50, top_k_end=150,
                                        beta_start=30.0, beta_end=5.0,
                                        L_est=5.0, verbose=False, seed=500 + idx)
                try_update(a_out, f"multi{idx}")

    # Phase 8: Final polish
    rem = remaining_time()
    n_best = len(best_seq)
    if rem > 40:
        print(f"\n=== Phase 8: Final polish n={n_best} ===")
        t8 = min(rem * 0.4, 120.0)
        if t8 > 20:
            a_out, _ = nesterov_apg(best_seq, t8, top_k_start=100, top_k_end=300,
                                    beta_start=10.0, beta_end=1.0,
                                    L_est=8.0, verbose=False, seed=999)
            try_update(a_out, "final_apg")
            rem = remaining_time()
        if rem > 20:
            t8c = min(rem * 0.4, 60.0)
            if t8c > 15:
                a_out2, _ = pairwise_coord_descent(best_seq, t8c, verbose=False, seed=777)
                try_update(a_out2, "final_cd")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
