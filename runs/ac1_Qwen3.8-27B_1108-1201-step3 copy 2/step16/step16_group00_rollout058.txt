```python
def build_L_fast(a, top_idx):
    """Build L[top_idx, :] where L[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    K = len(top_idx)
    L_sel = np.zeros((K, n))
    for idx, k in enumerate(top_idx):
        k = int(k)
        # L_sel[idx, j] = a[k - j] if 0 <= k - j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            L_sel[idx, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return L_sel
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated Frank-Wolfe with adaptive constraint selection
2. Sliding step (line search) for better convergence
3. Multi-length exploration with interpolation
4. Targeted coordinate descent polish
5. Periodic restarts to escape plateaus
"""

import numpy as np
import cvxpy as cp
import time

# Global solver
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_L_sel(a, top_idx):
    """Build constraint matrix rows for selected lags. L_sel[i, j] = a[k_i - j] if valid."""
    n = len(a)
    K = len(top_idx)
    L_sel = np.zeros((K, n))
    for idx in range(K):
        k = int(top_idx[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # L_sel[idx, j] = a[k - j] for j in [j_start, j_end]
            # When j = j_start: a[k - j_start]
            # When j = j_end: a[k - j_end]
            L_sel[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return L_sel


def fw_nesterov(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated Frank-Wolfe with adaptive K and sliding step."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    y = a.copy()
    x_prev = a.copy()

    K = 150
    stall_count = 0
    max_stall = 300

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute autocorrelation at y
        ac_y = fast_autocorr(y)
        M_y = float(ac_y.max())

        if M_y < best_m * (1 - 1e-6):
            # y is already better
            best_m = M_y
            best_a = y.copy()

        # Build constraint matrix and solve LP
        top_idx = np.argsort(ac_y)[::-1][:K]
        L_sel = build_L_sel(y, top_idx)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [L_sel @ b_var <= M_y]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            x_prev = x
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            x_prev = x
            continue
        v = b_val / b_sum

        # Sliding step: try multiple alpha values
        alpha_base = 2.0 / (t + 2.0)
        alphas = [alpha_base * 0.5, alpha_base, alpha_base * 1.5, alpha_base * 2.0, min(1.0, alpha_base * 3.0)]
        alphas = [min(1.0, max(0.01, al)) for al in alphas]

        best_alpha = alpha_base
        best_M_step = np.inf
        best_a_step = None
        for al in alphas:
            a_try = (1.0 - al) * x + al * v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            M_try = fast_autocorr(a_try).max()
            if M_try < best_M_step:
                best_M_step = M_try
                best_alpha = al
                best_a_step = a_try

        if best_a_step is None:
            x_prev = x
            continue

        x_prev = x
        x = best_a_step

        # Track best
        if best_M_step < best_m:
            best_m = best_M_step
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive K: increase when stuck
        if stall_count > max_stall and K < 1000:
            K = min(K + 100, 1000)
            stall_count = 0
            # Reset Nesterov momentum
            x_prev = x.copy()
            y = x.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FW t={t} ({elapsed:.0f}s): J={J:.6f}, K-> {K}, reset momentum")

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, stall={stall_count}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted coordinate descent: move mass from peak-contributing positions to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peak lags
        n_top = min(30, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_lags[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Pairs (i, k_i - i) that contribute to this lag
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            jdx = k_int - idx
            contrib = a[idx] * a[jdx]

            if contrib.max() < 1e-20:
                continue

            # Find the pair with maximum contribution
            max_pair = np.argmax(contrib)
            src_i = int(idx[max_pair])
            src_j = int(jdx[max_pair])

            # Candidates for destination: positions with low "peak involvement"
            peak_involved = np.zeros(n, dtype=bool)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                peak_involved[s2:e2 + 1] = True
                peak_involved[k2i - np.arange(s2, e2 + 1)] = True

            candidates = np.where(~peak_involved)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Try moving mass from src_i (and/or src_j) to a candidate
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                for src in set([src_i, src_j]):
                    if a[src] * frac < 1e-15:
                        continue
                    # Pick best destination: minimize increase in max autocorrelation
                    best_dst = None
                    best_M_dst = best_m
                    # Sample up to 20 candidates
                    n_sample = min(20, len(candidates))
                    sample_idx = rng.choice(len(candidates), n_sample, replace=False)
                    for si in sample_idx:
                        dst = int(candidates[si])
                        if dst == src:
                            continue
                        a_try = a.copy()
                        a_try[src] -= a_try[src] * frac
                        a_try[dst] += a_try[src + 0] * frac if False else (a[src] * frac)
                        a_try[src] = a[src] * (1 - frac)
                        a_try[dst] = a[dst] + a[src] * frac
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        M_try = fast_autocorr(a_try).max()
                        if M_try < best_M_dst:
                            best_M_dst = M_try
                            best_dst = (dst, a_try)

                    if best_dst is not None and best_M_dst < best_m:
                        dst_val, a_new = best_dst
                        a = a_new
                        best_m = best_M_dst
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                a[i] = a[i] * (1 - frac)
                a[j] = a[j] + a[i] * frac / (1 - frac)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m
```
