    # flip(a) has length n: flip(a)[m] = a[n-1-m]
    # conv(flip(a), w)[p] = sum_m flip(a)[m] * w[p-m] = sum_m a[n-1-m] * w[p-m]
    # Let j = n-1-m, so m = n-1-j: = sum_j a[j] * w[p-n+1+j]
    # We want: sum_j a[j] * w[j+i] so p-n+1+j = j+i means p = n-1+i
    # So T1[i] = conv(flip(a), w)[n-1+i]
    
    # T2[i] = sum_j a[j] * w[i-j]
    # conv(a, w)[p] = sum_j a[j] * w[p-j]
    # So T2[i] = conv(a, w)[i]
    
    # Great! So:
    # grad[i] = (1/beta) * (conv(flip(a), w)[n-1+i] + conv(a, w)[i])
    
    # Total conv size: n + (2n-1) - 1 = 3n-2
    # Need fft_size >= 3n-2
    
    N_fft = 1 << (3 * n - 2 + 1).bit_length()
    
    a_flip = a[::-1]
    T1 = np.fft.irfft(np.fft.rfft(a_flip, N_fft) * np.fft.rfft(w, N_fft), N_fft)[:3*n-2]
    T2 = np.fft.irfft(np.fft.rfft(a, N_fft) * np.fft.rfft(w, N_fft), N_fft)[:3*n-2]
    
    # T1[i] is at index n-1+i in the convolution
    grad = (T1[n-1:n-1+n] + T2[0:n]) / beta
    return grad

def annealed_logsumexp_descent(a0, time_limit, solver=None, verbose=True, 
                                  beta_start=5.0, beta_end=500.0,
                                  n_anneal_steps=2000, c0=3.0, restart_every=2000):
    """
    Annealed log-sum-exp smoothed optimization with mirror descent on simplex.
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    
    total_iter = max(n_anneal_steps, 50000)
    
    for t in range(1, total_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing: increase beta over time
        progress = min(t / n_anneal_steps, 1.0)
        # Exponential annealing
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(y)
        g = ac_gradient(y, ac, beta=beta)
        
        # Mirror descent step (exponential family / multiplicative weights)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = y
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
        
        # Periodic restart with perturbation
        if t % restart_every == 0:
            pert = rng.normal(0, 0.01, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            M_pert = fast_autocorr(x_new).max()
            if M_pert < best_m:
                best_m = M_pert
                best_a = x_new.copy()
            x = x_new
            x_prev = x_new
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LSE t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m

def lp_fw_with_full_sweep(a0, time_limit, solver, top_k_init=50, top_k_max=300, verbose=True):
    """
    LP Frank-Wolfe with adaptive constraint selection and Nesterov acceleration.
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    stall = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Select active constraints: all lags where ac[k] > 0.95 * M_curr
        # Plus top-k for safety
        threshold = 0.95 * M_curr
        active_mask = ac >= threshold
        active_lags_full = np.where(active_mask)[0]
        top_k_lags = np.argsort(ac)[::-1][:current_top_k]
        # Union
        active_lags = np.unique(np.concatenate([active_lags_full, top_k_lags]))
        
        # Limit number of constraints for LP efficiency
        if len(active_lags) > 400:
            # Keep top 400
            active_lags = np.argsort(ac[active_lags])[::-1][:400]
            active_lags = active_lags[active_lags.argsort()[::-1]]  # reorder
            # Actually just take the top 400 by ac value
            sorted_idx = np.argsort(ac)[::-1][:400]
            active_lags = sorted_idx
        
        # Solve LP: max sum(b) s.t. (y*b)[k] <= M_curr for k in active_lags, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active_lags:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            j_idx = np.arange(j_start, j_end + 1)
            y_coeff = y[k_int - j_idx]
            constraints.append(y_coeff @ b_var[j_idx] <= M_curr)
        
        # Ensure at least some constraints
        if len(constraints) < 5:
            top5 = np.argsort(ac)[::-1][:5]
            for k in top5:
                k_int = int(k)
                j_start = max(0, k_int - n + 1)
                j_end = min(n - 1, k_int)
                if j_start > j_end:
                    continue
                j_idx = np.arange(j_start, j_end + 1)
                y_coeff = y[k_int - j_idx]
                constraints.append(y_coeff @ b_var[j_idx] <= M_curr)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        
        # Update
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        # Adapt top_k
        if t % 100 == 0:
            current_top_k = min(current_top_k + 5, top_k_max)
        
        # Restart if stalled
        if stall > 500:
            stall = 0
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.005, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, k={len(active_lags)}")
    
    return best_a, best_m

def coordinate_polish_v2(a0, time_limit, verbose=True, rng_seed=42):
    """
    Improved coordinate descent: try to reduce the peak by redistributing mass
    from positions that contribute most to the peak to positions that don't.
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    iteration = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = np.argmax(ac)
        
        # Find positions that contribute most to the peak
        k_int = int(peak_k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        idx = np.arange(j_start, j_end + 1)
        contrib = a[idx] * a[k_int - idx]
        total_contrib = contrib.sum()
        
        if total_contrib < 1e-20:
            stall += 10
            continue
        
        # Get top contributing positions
        n_top = min(20, len(idx))
        top_contrib_idx = idx[np.argsort(contrib)[::-1][:n_top]]
        
        # Get top peaks (for evaluating impact)
        n_peaks = min(30, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        
        improved_this_iter = False
        
        for src in top_contrib_idx:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find best destination
            # Score: total contribution to top peaks if we add mass here
            best_dst = -1
            best_score = float('inf')
            
            # Sample candidate destinations
            n_cand = min(200, n)
            candidates = rng.integers(0, n, size=min(500, n))
            
            for dst in candidates:
                if dst == src:
                    continue
                
                # Estimate impact on top peaks
                score = 0.0
                for pk in top_peaks[:10]:
                    pk_int = int(pk)
                    # If we add mass at dst, how much does (a*a)[pk] increase?
                    # (a*a)[pk] += a[dst]*a[pk-dst] + a[pk-dst]*a[dst]... 
                    # Actually: d(a*a)[pk] / a[dst] = a[pk-dst] + a[dst-pk]
                    d1 = pk_int - dst
                    d2 = dst - pk_int
                    sens = 0.0
                    if 0 <= d1 < n:
                        sens += a[d1]
                    if 0 <= d2 < n:
                        sens += a[d2]
                    score += sens * (ac[pk] / M_curr)  # weight by peak height
                
                if score < best_score:
                    best_score = score
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_iter = True
                    stall = 0
                    found = True
                    break
            
            if found:
                break  # Restart with new peak
        
        if not improved_this_iter:
            stall += 1
            # Perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"  CoordV2 ({elapsed:.1f}s): J={J:.6f}, iters={iteration}")
    
    return best_a, best_m

def block_coordinate(a0, time_limit, block_size=64, verbose=True):
    """
    Block coordinate descent: optimize blocks of consecutive coordinates.
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    n_blocks = n // block_size
    if n_blocks < 1:
        return best_a, best_m
    
    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Cycle through blocks
        b = iteration % n_blocks
        start = b * block_size
        end = min(start + block_size, n)
        
        if start >= n:
            continue
        
        # Optimize block [start:end] while keeping rest fixed
        # Use simple gradient descent on the block
        block = a[start:end].copy()
        
        for inner in range(50):
            # Compute gradient for the block
            ac = fast_autocorr(a)
            # Gradient of max(ac) w.r.t. block elements
            top_k = np.argsort(ac)[::-1][:20]
            
            grad_block = np.zeros(end - start)
            for k in top_k:
                k_int = int(k)
                for i in range(start, end):
                    d1 = k_int - i
                    d2 = i - k_int
                    g = 0.0
                    if 0 <= d1 < n:
                        g += a[d1]
                    if 0 <= d2 < n:
                        g += a[d2]
                    grad_block[i - start] += g * (ac[k] / ac.max())
            
            # Gradient step with projection
            lr = 0.01 / (inner + 1)
            block -= lr * grad_block
            block = np.maximum(block, 0.0)
            
            # Renormalize
            a_trial = a.copy()
            a_trial[start:end] = block
            s = a_trial.sum()
            if s < 1e-15:
                break
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                break
            else:
                a = a_trial  # Still update for next iteration
                break
        
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"  BlockCoord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies
    with multi-start and multi-length exploration.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    
    solver = get_cp_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize_a(parent)
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a_arr, label):
        nonlocal best_seq, best_score
        if a_arr is None:
            return
        a_arr = np.maximum(np.asarray(a_arr, dtype=np.float64), 0.0)
        s = a_arr.sum()
        if s < 0.01:
            return
        a_norm = a_arr / s
        sc = eval_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_arr)}, t={elapsed:.0f}s)")
    
    # Generate multiple initializations
    init_candidates = []
    
    if parent is not None:
        init_candidates.append(("parent", normalize_a(parent)))
    
    # Spectral flat
    n_spec = 2400
    init_candidates.append(("spec2400", spectral_flat_init(n_spec, 50)))
    
    # Sine
    init_candidates.append(("sine2400", sine_init(2400)))
    
    # Different lengths
    for n_t in [2000, 2800, 3200]:
        if parent is not None:
            init_candidates.append((f"interp{n_t}", interp_to_length(parent, n_t)))
    
    # Evaluate all inits
    for label, a_init in init_candidates:
        sc = eval_safe(a_init)
        print(f"  Init {label}: score={sc:.6f}")
        try_update(a_init, label)
    
    n0 = len(best_seq)
    print(f"\nStarting with n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Annealed log-sum-exp (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Annealed LSE n={n0}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_in = normalize_a(best_seq)
        a_out, _ = annealed_logsumexp_descent(
            a_in, t1, 
            beta_start=5.0, beta_end=300.0,
            n_anneal_steps=3000, c0=3.0, restart_every=3000,
            verbose=True
        )
        try_update(a_out, f"lse_n{n0}")
    
    # === Phase 2: LP FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW n={n_best}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = normalize_a(best_seq)
        a_out, _ = lp_fw_with_full_sweep(
            a_in, t2, solver, 
            top_k_init=50, top_k_max=300, verbose=True
        )
        try_update(a_out, f"lpfw_n{n_best}")
    
    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_in = normalize_a(best_seq)
        a_out, _ = coordinate_polish_v2(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 4: Second LSE pass with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE v2 n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_in = normalize_a(best_seq)
        a_out, _ = annealed_logsumexp_descent(
            a_in, t4,

            verbose=True
        )
        try_update(a_out, f"lse2_n{n_best}")
    
    # === Phase 5: LP FW again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW v2 n={n_best}, budget={remaining:.0f}s ===")

        )

    # === Phase 6: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)

            sc_alt = eval_safe(a_alt)
            print(f"  Alt init score: {sc_alt:.6f}")
            
            t_alt = min(remaining * 0.25, 200.0)

                )
                try_update(a_out, f"lse_alt_n{n_alt_try}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
