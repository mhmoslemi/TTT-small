For n=2400, the "flat" convolution would have all 4799 lags equal to 1/4799. The ratio of actual max to flat is 1.504 * (2n-1)/(2n) ≈ 1.504. So we're at 1.504x the ideal flatness.

To improve, we need to reduce the "bumpiness" of the convolution.

Let me also consider: what if the optimal solution has a specific structure, like being a "raised cosine" or "Hanning" window? These have well-known spectral properties.

A raised cosine (Hanning) window: a[i] = 0.5 * (1 - cos(2*pi*i/(n-1)))
Its autocorrelation has a specific shape - triangular-ish main lobe with sidelobes.

A Bartlett (triangle) window: a[i] = 1 - |2*i/(n-1) - 1|
Its autocorrelation is more concentrated.

A flat-top window is specifically designed to have flat sidelobes in frequency domain, which might translate to flat convolution.

Let me try different window functions as initializations and see which gives the best starting point.

Actually, I think the most impactful change would be:

1. **Use a larger number of constraints in the LP-FW** - this is the main improvement
2. **Run more LP-FW iterations** - the convergence is O(1/t) so more iterations help
3. **Better Nesterov parameters** - the c0 parameter and temperature schedule

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Frank-Wolfe with full/adaptive constraint set and Nesterov acceleration
2. Smoothed gradient descent (softmax smoothing) with adaptive temperature
3. Multi-scale n exploration
4. Coordinate descent polish
Key: The self-convolution flatness problem. Score = 2n * max_c / (sum_a)^2.
"""

import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(16, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

def build_constraint_matrix(a, n, lags):
    """Build the constraint matrix T where T[k, i] = a[k-i] for valid i."""
    T = np.zeros((len(lags), n))
    for row, k in enumerate(lags):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            cols = np.arange(i_start, i_end)
            T[row, cols] = a[k - cols]
    return T

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iter=10000):
    """
    LP Frank-Wolfe with adaptive constraint selection.
    Uses Nesterov acceleration.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # current iterate
    x = a.copy()  # Nesterov momentum point
    
    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Adaptive constraint selection: start with top lags, expand over time
        # Phase 1: top 100, Phase 2: top 500, Phase 3: all
        if t < 200:
            top_k = min(100, 2 * n - 1)
        elif t < 800:
            top_k = min(500, 2 * n - 1)
        else:
            top_k = 2 * n - 1
        
        # Select top-k lags by value
        if top_k >= (2 * n - 1):
            lags = np.arange(2 * n - 1)
        else:
            lags = np.argsort(ac)[::-1][:top_k]
        
        # Build constraint matrix
        T = build_constraint_matrix(y, n, lags)
        
        # Solve LP: max sum(b) s.t. T@b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Check both y and x
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, top_k={top_k}")
    
    return best_a, best_m

def smooth_gradient_descent(a0, time_limit, verbose=True):
    """
    Smoothed gradient descent using Gaussian smoothing of the max.
    Uses log-sum-exp approximation of the max function.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Parameters
    tau0 = 0.05  # initial smoothing parameter (inverse temperature)
    tau_final = 0.005
    lr0 = 0.5
    momentum = 0.9
    
    v = np.zeros(n)  # momentum
    n_steps_expected = 20000
    
    for t in range(1, n_steps_expected + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Log-sum-exp for stable computation
        # smooth_max ≈ (1/tau) * log(sum(exp(tau * ac)))
        # grad smooth_max ≈ (ac * exp(tau*ac)) / sum(exp(tau*ac))  [weighted average direction]
        # Actually: d/d a [ (1/tau) log(sum_k exp(tau * c_k)) ]
        # = sum_k exp(tau*c_k) / sum_k exp(tau*c_k) * grad c_k
        # = weighted sum of gradients of individual c_k
        
        tau = tau0 * (tau_final / tau0) ** (t / n_steps_expected)
        tau = max(tau, 1e-4)
        
        # Compute weights
        shifted = tau * ac - tau * M  # shift for numerical stability
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros_like(ac)
            weights[np.argmax(ac)] = 1.0
            w_sum = 1.0
        weights /= w_sum
        
        # Compute gradient of smooth max
        # grad c[k] w.r.t. a[i] = a[k-i] (if 0 <= k-i < n) + a[k-i] (if i != k-i)
        # Actually for linear conv: c[k] = sum_j a[j]*a[k-j]
        # d c[k] / d a[i] = a[k-i] (when 0 <= k-i < n) + (a[k-i] if k-2i is valid... )
        # More carefully: c[k] = sum_j a[j] * a[k-j]
        # d c[k] / d a[i] = a[k-i] (term where j=i) + a[k-i] (term where k-j=i, i.e., j=k-i)
        # If i != k-i: d c[k]/d a[i] = a[k-i] + a[k-i] = 2*a[k-i]
        # If i == k-i (i.e., 2i=k): d c[k]/d a[i] = a[k-i] = a[i] (only one term)
        
        grad = np.zeros(n)
        for k in range(len(ac)):
            w = weights[k]
            if w < 1e-12:
                continue
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += w * a[i]  # only one term
                    else:
                        grad[i] += w * a[j]
        
        # Wait, this is O(n^2) per step which is too slow for n=2400
        # Let me use a more efficient computation
        
        # Actually, let me re-derive. c[k] = sum_j a[j]*a[k-j]
        # d c[k] / d a[i] = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j, i)]
        # = a[k-i] + a[k-i]  (if both k-i and k-i are in [0,n-1])
        # = a[k-i] * (1 if k-i in [0,n-1] else 0) + a[k-i] * (1 if k-i in [0,n-1] else 0)
        # Wait, the second term: delta(k-j, i) means j = k-i, so the term is a[j] where j = k-i
        # So d c[k] / d a[i] = a[k-i] (first term) + a[k-i] (second term, which is a[j] evaluated at j=k-i)
        # = 2*a[k-i] if 0 <= k-i < n and k-i != i
        # = a[k-i] = a[i] if 2i = k (i.e., j = i)
        
        # This is still O(n^2) in the worst case. Let me use FFT.
        # Actually, the gradient of c[k] w.r.t. a is just 2 * (reversed and shifted a)
        # Specifically: grad_a c[k] = 2 * (a reversed, shifted by k) with boundary handling
        
        # For efficiency, let's compute the full gradient as a convolution
        # grad[i] = sum_k weights[k] * (a[k-i] + a[k-i]) [with boundary conditions]
        # = 2 * sum_k weights[k] * a[k-i] [with boundary]
        # This is 2 * (weights * a) convolved, evaluated at specific points
        
        # Let me just do it the naive way but with vectorization
        # For each k, the gradient contribution to position i is weights[k] * a[k-i] (for valid k-i)
        # This is like a cross-correlation of weights with a
        
        # grad[i] = sum_k weights[k] * a[k-i] (for 0 <= k-i < n)
        #          + sum_k weights[k] * a[k-i] (for 0 <= k-i < n) [same thing, factor of 2]
        # But when k-i = i (i.e., k = 2i), we only count it once.
        
        # For simplicity, let's compute:
        # tmp[i] = sum_k weights[k] * a[k-i] for valid (k-i) in [0, n-1]
        # This is the cross-correlation of weights and a.
        # = convolve(weights, a_reversed) evaluated at appropriate positions
        
        # Actually, let me just use the direct sum for the top few weights
        top_k = min(200, len(ac))
        top_idx = np.argsort(weights)[::-1][:top_k]
        
        grad = np.zeros(n)
        for k in top_idx:
            w = weights[int(k)]
            if w < 1e-14:
                break
            k_int = int(k)
            # For position i, contribution is a[k_int - i] if 0 <= k_int - i < n
            # i ranges from max(0, k_int-n+1) to min(n-1, k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            vals = a[k_int - idx]
            # Handle the diagonal (where i = k_int - i, i.e., 2*i = k_int)
            diag = (2 * idx == k_int)
            grad[idx] += w * vals * (2 - diag.astype(float))
        
        # Learning rate (decaying)
        lr = lr0 / (1.0 + 0.01 * t)
        
        # Gradient step with momentum (in log-space for non-negativity)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= lr * grad / (a + 1e-30)  # normalized gradient step
        # Actually, let's use mirror descent update:
        # a_new[i] ∝ a[i] * exp(-lr * grad[i])
        log_a = np.log(np.maximum(a, 1e-30)) - lr * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-20)
        a_new = a_new / a_new.sum()
        
        # Momentum
        v = momentum * v + (1 - momentum) * (a_new - a)
        a_proposed = a + v
        a_proposed = np.maximum(a_proposed, 0)
        sp = a_proposed.sum()
        if sp > 1e-15:
            a_proposed = a_proposed / sp
        
        M_new = fast_autocorr(a_proposed).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_proposed.copy()
            a = a_proposed
        else:
            # Try without momentum
            M_new2 = fast_autocorr(a_new).max()
            if M_new2 < best_m:
                best_m = M_new2
                best_a = a_new.copy()
                a = a_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD t={t} ({elapsed:.1f}s): J={J:.6f}, tau={tau:.5f}")
    
    return best_a, best_m

def nesterov_mirror_descent(a0, time_limit, verbose=True, c0=3.0, tau0=0.03, tau1=0.003):
    """Nesterov mirror descent with softmax subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1
    
    n_total = 2 * n - 1
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature (geometric decay)
        frac = min(1.0, t / 10000.0)
        tau = tau0 * (tau1 / tau0) ** frac
        tau = max(tau, 1e-5)
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        
        # Compute softmax-weighted subgradient
        shifted = ac / tau
        shifted -= shifted.max()
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            top_idx = np.argsort(ac)[::-1][:100]
            weights = np.zeros_like(ac)
            weights[top_idx] = 1.0
            w_sum = 100.0
        weights /= w_sum
        
        # Compute gradient efficiently using top weights
        top_k = min(300, n_total)
        top_idx = np.argsort(weights)[::-1][:top_k]
        
        grad = np.zeros(n)
        for k in top_idx:
            w = weights[int(k)]
            if w < 1e-14:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            vals = y[k_int - idx]
            diag = (2 * idx == k_int).astype(float)
            grad[idx] += 2.0 * w * vals * (1.0 - 0.5 * diag)
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_x = np.log(np.maximum(y, 1e-30)) - eta * grad
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-20)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restarting
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(0.5, gamma * 0.7)
        
        # Ergodic average
        running_avg += x
        avg_count += 1
        if avg_count % 1000 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({
