                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective 2*n*max(conv)/sum^2 for a numpy array."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_conv(a)
    return float(2.0 * len(a) * ac.max() / (s * s))


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def logsumexp_smooth_grad(y, ac, beta):
    """
    Compute gradient of the log-sum-exp smoothed objective:
    f_beta(y) = (1/beta) * log(sum_k exp(beta * ac[k]))
    where ac = conv(y, y).
    Gradient: (2/beta) * sum_k w_k * (P_k y) where w_k = exp(beta*ac[k]) / sum exp(beta*ac[j])
    Simplified: 2 * sum_k w_k * P_k y
    """
    n = len(y)
    ac_max = ac.max()
    exp_ac = np.exp(beta * (ac - ac_max))
    w = exp_ac / exp_ac.sum()
    
    # Find significant weights
    threshold = 1e-12
    sig_idx = np.where(w > threshold)[0]
    
    grad = np.zeros(n)
    for k in sig_idx:
        k_int = int(k)
        wi = w[k_int]
        # P_k y: (P_k y)[i] = y[k-i] if valid
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += 2.0 * wi * y[j]
    return grad


def mirror_descent_lse(a0, time_limit, verbose=True, seed=42,
                       beta_start=0.0, beta_end=1.0, beta_schedule='exp',
                       c0=3.0, restart_stall=5000):
    """
    Mirror descent (entropy-based) with log-sum-exp smoothed objective.
    Uses adaptive temperature annealing and Nesterov momentum.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    current_beta = max(beta_start, 1e-6)
    
    # Total iterations to plan annealing over
    # Estimate: ~20000 iterations for the main phase
    total_iter_est = 25000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        frac = min(1.0, t / total_iter_est)
        if beta_schedule == 'exp':
            current_beta = beta_start * math.exp(frac * math.log(max(beta_end / max(beta_start, 1e-6), 1.0)))
            if beta_start < 1e-5:
                current_beta = max(1e-4, beta_end * frac ** 2)
        else:
            current_beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute convolution
        ac = fast_conv(y)
        M_curr = ac.max()

        # Compute smoothed gradient
        g = logsumexp_smooth_grad(y, ac, current_beta)

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart with perturbation
        if stall > restart_stall:
            noise_scale = min(0.02, 0.005 * (stall / restart_stall))
            x_restarted = best_a + rng.normal(0, noise_scale, n) * np.maximum(best_a, 0.01)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0

        # Periodic large jump
        if t % 20000 == 0 and t > 0:
            x_jumped = best_a + rng.normal(0, 0.03, n) * np.maximum(best_a, 0.005)
            x_jumped = np.maximum(x_jumped, 1e-8)
            x_jumped = x_jumped / x_jumped.sum()
            M_jump = fast_conv(x_jumped).max()
            if M_jump < best_m:
                best_m = M_jump
                best_a = x_jumped.copy()
                x = x_jumped
                x_prev = x_jumped

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD-LSE t={t} ({elapsed:.1f}s): J={J:.6f}, beta={current_beta:.3f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=None):
    """LP-based Frank-Wolfe with exact line search and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()
    iter_count = 0
    stall = 0
    max_stall = 50

    if max_iters is None:
        max_iters = 1000000

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix T where (T @ b)[k] = (y * b)[k]
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        direction = b_val / b_sum

        # Exact line search
        alphas = [2.0 / (t + 2.0)] + list(np.linspace(0.05, 0.8, 12))
        best_alpha = alphas[0]
        best_eval = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()

        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov momentum
        if t <= 2:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_mom = x_new
        iter_count += 1

        my = fast_conv(y).max()
        if my < best_m - 1e-15:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        mx = fast_conv(x_mom).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x_mom.copy()
            stall = 0

        if stall > max_stall:
            # Restart from best
            y = best_a.copy()
            x_mom = best_a.copy()
            stall = 0

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def peak_equalization(a0, time_limit, verbose=True, seed=99):
    """
    Targeted peak-shaving: identify top peaks and redistribute mass
    from contributors to the highest peak to positions that don't
    affect other top peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        
        # Find top peaks
        n_peaks = min(8, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        
        improved_this_round = False
        
        for k_idx in range(min(3, n_peaks)):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(top_peaks[k_idx])
            peak_val = ac[k]
            
            if peak_val < M_curr * 0.95 and k_idx > 0:
                continue  # Only work on peaks close to the max
            
            # Find contributing pairs
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k - idx]
            
            if contrib.sum() < 1e-20:
                continue
            
            # Find the most contributing pair
            src_i = idx[np.argmax(contrib)]
            src_j = k - src_i
            
            if src_i == src_j:
                # Diagonal contribution: a[src_i]^2
                # Try to reduce a[src_i]
                src = src_i
                dst_candidates = _find_safe_positions(a, k, top_peaks, n, exclude={src})
            else:
                # Off-diagonal: a[src_i] * a[src_j]
                # Reduce the larger one
                if a[src_i] >= a[src_j]:
                    src = src_i
                else:
                    src = src_j
                dst_candidates = _find_safe_positions(a, k, top_peaks, n, exclude={src_i, src_j})
            
            if len(dst_candidates) == 0:
                continue
            
            # Try reducing src and adding to a safe destination
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                if time.monotonic() - t0 > time_limit:
                    break
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                
                # Pick best destination (lowest impact on all peaks)
                best_dst = None
                best_impact = float('inf')
                n_try = min(len(dst_candidates), 50)
                chosen = dst_candidates[rng.permutation(len(dst_candidates))[:n_try]]
                
                for dst in chosen:
                    # Estimate impact on all top peaks
                    impact = 0
                    for k2 in top_peaks[:4]:
                        k2_int = int(k2)
                        j2 = k2_int - dst
                        if 0 <= j2 < n:
                            impact += a[j2]
                    if impact < best_impact:
                        best_impact = impact
                        best_dst = dst
                
                if best_dst is None:
                    continue
                    
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    stall = 0
                    break
        
        if not improved_this_round:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[j] -= amt
                            a[i] += amt
                            a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakEq ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def _find_safe_positions(a, peak_k, top_peaks, n, exclude):
    """Find positions that don't significantly contribute to top peaks."""
    in_peak = np.zeros(n, dtype=bool)
    for k2 in top_peaks[:5]:
        k2_int = int(k2)
        s2 = max(0, k2_int - n + 1)
        e2 = min(n, k2_int + 1)
        in_peak[s2:e2] = True
    
    for ex in exclude:
        if 0 <= ex < n:
            in_peak[ex] = True
    
    return np.where(~in_peak)[0]


def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with 2-opt mass transfer moves and local search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(12, len(ac))]
        improved = False

        for k in top_peaks[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            
            # Find safe destinations
            safe = _find_safe_positions(a, k_int, top_peaks, n, exclude={int(src)})
            if len(safe) == 0:
                safe = np.arange(n)
            
            n_cand = min(len(safe), 100)
            chosen = safe[rng.permutation(len(safe))[:n_cand]]

            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                # Impact estimate: how much would adding to dst affect top peaks?
                impact = 0.0
                for k2 in top_peaks[:4]:
                    k2_int = int(k2)
                    j2 = k2_int - int(dst)
                    if 0 <= j2 < n:
                        impact += a[j2]
                    j2b = int(dst) - (n - 1 - k2_int)  # other convolution direction
                if impact < best_score:
                    best_score = impact
                    best_dst = int(dst)

            if best_dst < 0:
                continue
            dst = best_dst

            # Try multiple fractions with bisection-like refinement
            found = False
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found and stall > 150:
                # Aggressive random move
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                    else:
                        a[j] -= amt
                        a[i] += amt
                        a = a / a.sum()

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def dual_lp_direction(f, M):
    """
    Solve the LP to get a Frank-Wolfe direction using the dual structure.
    Dual: min M * sum(y) s.t. (f*y)[j] >= 1, y >= 0
    We use cvxpy on the primal for reliability.
    """
    n = len(f)
    # Build T matrix: T[k, j] = f[k-j] if 0 <= k-j < n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = f
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    return b_var, T, prob

            parent = p.copy()

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent with LSE smoothing (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)

                                       beta_start=5e-3, beta_end=4.0, c0=6.0)

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)

    if t3 > 90 and n_curr <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n_curr}")

    # === Phase 4: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 4: Peak Equalization n={n_curr} ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_equalization(a_in, t4, verbose=True, seed=99)
        try_update(a_out, f"peakeq_n{n_curr}")

    # === Phase 5: MD-LSE polish (sharp beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 5: MD-LSE polish n={n_curr} ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_lse(a_in, t5, verbose=True, seed=314,
                                       beta_start=1.0, beta_end=5.0, c0=8.0,
                                       restart_stall=3000)
        try_update(a_out, f"mdlse_polish_n{n_curr}")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 6: Coord n={n_curr} ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_curr}")

    # === Phase 7: LP again (polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 7: LP polish n={n_curr} ===")
    t7 = min(remaining * 0.10, 150.0)
    if t7 > 60 and n_curr <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, max_iters=200)
        try_update(a_out, f"lp_polish_n{n_curr}")

    # === Phase 8: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        for delta in [200, -200, 400]:
            n_try = n_best + delta
            if n_try < 1000 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
            print(f"\n=== Phase 8: Length n={n_try} ===")
            a_long = interpolate_sequence(best_seq, n_try)
            a_in = norm_sum1(a_long)
            t8 = min(remaining * 0.35, 150.0)
            if t8 > 40:
                a_out, _ = mirror_descent_lse(a_in, t8, verbose=False, seed=55 + n_try % 100,
                                              beta_start=1e-3, beta_end=2.0, c0=4.0)
                try_update(a_out, f"mdlse_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    a_out2, _ = coordinate_polish(a_in, min(remaining * 0.4, 60.0), verbose=False, seed=77)
                    try_update(a_out2, f"coord_n{n_try}")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord n={n_best} ===")
    if remaining > 20:
        t9 = min(remaining * 0.7, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=True, seed=271)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 10: Final peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 10: Final peak eq n={n_best} ===")
    if remaining > 15:
        t10 = min(remaining * 0.5, 50.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_equalization(a_in, t10, verbose=False, seed=555)
            try_update(a_out, f"peakeq_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
