def simplex_projection(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            # Perturb from best
            rng = np.random.default_rng(t)
            pert = rng.dirichlet(np.ones(n) * 2.0)
            a = 0.7 * best_a + 0.3 * pert
            a = norm_sum1(a)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, beta={beta:.0f}")

    return best_a, best_m

def proj_subgradient(a0, time_limit, top_k=10, verbose=True):
    """Projected subgradient with Barzilai-Borwein step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    grad_prev = None
    a_prev = a.copy()
    bb_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(a)
        K = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:K]
        vals = ac[idx]
        w = np.exp(20.0 * (vals - vals[0]))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, idx):
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n - 1, k_int)
            if i_lo <= i_hi:
                ii = np.arange(i_lo, i_hi + 1)
                grad[ii] += 2.0 * wi * a[k_int - ii]

        # BB step size
        if grad_prev is not None and t > 1:
            s_vec = a - a_prev
            y_vec = grad - grad_prev
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if sy > 1e-15 and ss > 1e-30:
                eta = ss / sy
                eta = min(eta, 10.0)
                eta = max(eta, 0.01)
                bb_count += 1
            else:
                eta = 1.0 / np.sqrt(t)
        else:
            eta = 1.0 / np.sqrt(t)

        a_prev = a.copy()
        grad_prev = grad.copy()

        a_new = a - eta * grad
        a_new = simplex_projection(a_new, 1.0)

        a = a_new
        M_new = fast_conv(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PSG t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}")

    return best_a, best_m

def build_T_sparse(a, top_m=None):
    """Build sparse constraint matrix for LP oracle. Optionally only top-m constraints."""
    n = len(a)
    num_rows = 2 * n - 1
    if top_m is not None and top_m < num_rows:
        # Only use top-m most binding constraints
        ac = fast_conv(a)
        rows = np.argsort(ac)[::-1][:top_m]
    else:
        rows = np.arange(num_rows)

    # For each row k, the constraint is sum_j a[k-j] * b_j <= M
    # T[k, j] = a[k-j] if 0 <= k-j < n
    row_idx = []
    col_idx = []
    data = []
    for ri, k in enumerate(rows):
        k_int = int(k)
        j_lo = max(0, k_int - n + 1)
        j_hi = min(n - 1, k_int)
        for j in range(j_lo, j_hi + 1):
            row_idx.append(ri)
            col_idx.append(j)
            data.append(a[k_int - j])

    T = sp.csr_matrix((data, (row_idx, col_idx)), shape=(len(rows), n))
    return T, rows

def lp_frank_wolfe(a0, time_limit, top_m=500, verbose=True, max_iters=5000):
    """LP-Frank-Wolfe with active set (top-m constraints) and Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    y_prev = a.copy()

    # Try to get a fast solver
    solver = None
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            xv = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(xv)), [cp.sum(xv) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status == 'optimal':
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraints (top-m most binding)
        T, rows = build_T_sparse(y, top_m=top_m)
        M_vec = np.full(len(rows), M_curr)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_vec]
        objective = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64) if b_var.value is not None else None
        if b_val is None:
            continue
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        for candidate in [y, x]:
            Mc = fast_conv(candidate).max()
            if Mc < best_m:
                best_m = Mc
                best_a = candidate.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, "{_get_context()}")

    return best_a, best_m

def _get_context():
    return "active"

def coord_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Coordinate descent targeting peak convolution lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        peak_idx = int(np.argmax(ac))
        M_curr = ac[peak_idx]

        # Find pairs that contribute most to the peak
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)
        ii = np.arange(i_lo, i_hi + 1)
        contrib = a[ii] * a[peak_idx - ii]
        max_contrib = contrib.max()
        contrib_idx = np.argmax(contrib)
        src = int(ii[contrib_idx])
        dst = int(peak_idx - ii[contrib_idx])

        if src == dst:
            src = int(rng.integers(n))
            dst = int(rng.integers(n))

        # Try moving mass from src to a low-impact position
        best_move = None
        best_trial_m = best_m

        # Compute sensitivity for each position (how much it contributes to top peaks)
        top_peaks = np.argsort(ac)[::-1][:5]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            for pos in range(n):
                d = pk_int - pos
                if 0 <= d < n:
                    sensitivity[pos] += a[d]

        # Find low-sensitivity positions
        low_sens = np.argsort(sensitivity)[:max(20, n // 20)]

        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            for dst_candidate in low_sens[:10]:
                if dst_candidate == src:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst_candidate] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                Mt = fast_conv(a_trial).max()
                if Mt < best_trial_m:
                    best_trial_m = Mt
                    best_move = (src, int(dst_candidate), amt)

        if best_move is not None:
            s, d, amt = best_move
            a[src] -= amt
            a[d] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = best_trial_m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i1] += amt
                        a[i2] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD stall={stall} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def make_u_shape(n, edge_power=0.3):
    """Create a U-shaped initialization (larger at edges)."""
    x = np.linspace(0, 1, n)
    a = x**edge_power + (1 - x)**edge_power + 0.05
    return a / a.sum()

def make_ramp(n, power=1.0):
    """Create a ramp initialization."""
    i = np.arange(1, n + 1)
    a = i**power
    return a / a.sum()

def make_bimodal(n, center_spread=0.1):
    """Create a bimodal (two-peak) initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / center_spread)**2) + np.exp(-((x - 0.85) / center_spread)**2) + 0.01
    return a / a.sum()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-method approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Try to load parent
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_safe(parent)
            best_seq = parent.copy()
            best_score = parent_score
            n_base = len(parent)
            print(f"Parent: n={n_base}, score={parent_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_u_shape(n_base, 0.3)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent (fast, ~25% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 1: Mirror descent n={n0} ===")
        t1 = min(remaining * 0.25, 300.0)
        a_out, _ = mirror_descent(norm_sum1(best_seq), t1, beta_start=30.0,
                                   beta_end=300.0, eta0=3.0, top_k=30, verbose=True)
        try_update(a_out, "md_phase1")

    # === Phase 2: LP-FW (medium, ~25% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW n={n_cur} ===")
        t2 = min(remaining * 0.25, 350.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, top_m=800,
                                   verbose=True, max_iters=3000)
        try_update(a_out, "lp_fw_phase2")

    # === Phase 3: Mirror descent again (refine, ~15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Mirror descent refine n={n_cur} ===")
        t3 = min(remaining * 0.15, 180.0)
        a_out, _ = mirror_descent(norm_sum1(best_seq), t3, beta_start=100.0,
                                   beta_end=1000.0, eta0=1.0, top_k=10, verbose=True)
        try_update(a_out, "md_phase3")

    # === Phase 4: Coordinate descent polish (~15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: Coord descent n={n_cur} ===")
        t4 = min(remaining * 0.15, 150.0)
        a_out, _ = coord_descent(norm_sum1(best_seq), t4, verbose=True)
        try_update(a_out, "cd_phase4")

    # === Phase 5: Projected subgradient (~10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: Proj subgradient n={n_cur} ===")
        t5 = min(remaining * 0.10, 100.0)
        a_out, _ = proj_subgradient(norm_sum1(best_seq), t5, top_k=10, verbose=True)
        try_update(a_out, "psg_phase5")

    # === Phase 6: Final LP-FW (~10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 6: Final LP-FW n={n_cur} ===")
        t6 = min(remaining * 0.10, 120.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t6, top_m=600,
                                   verbose=True, max_iters=2000)
        try_update(a_out, "lp_final")

    # === Phase 7: Try different n values (~5% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Multi-n exploration ===")
        test_ns = [n_best + 200, n_best + 400, n_best - 200]
        for nt in test_ns:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40 or not (500 <= nt <= 4000):
                continue
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_m = min(remaining * 0.3, 80.0)
            a_out, _ = mirror_descent(a_inter, t_m, beta_start=50.0,
                                       beta_end=500.0, eta0=2.0, top_k=20, verbose=False)
            try_update(a_out, f"multin_n{nt}")

    # === Phase 8: Final polish with coord descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final polish n={n_best} ===")
        t8 = min(remaining * 0.5, 60.0)
        a_out, _ = coord_descent(norm_sum1(best_seq), t8, verbose=False, rng_seed=999)
        try_update(a_out, "final_polish")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
