def fast_conv(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s


def get_subgradient(a, conv, active_set):
    """Compute subgradient of max(conv) using active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        k = int(k)
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * a[j]
    return g


def get_active_set(conv, threshold_frac=0.999):
    """Get indices where conv is close to max."""
    M = conv.max()
    if M < 1e-30:
        return [int(conv.argmax())]
    return np.where(conv >= threshold_frac * M)[0]


def logsumexp_subgradient(a, conv, temperature=0.1):
    """Compute smoothed subgradient using log-sum-exp weighting."""
    n = len(a)
    M = conv.max()
    # Soft weights based on exp(conv[k] / temperature)
    shifted = conv / temperature - (conv.max() / temperature)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(conv.argmax())] = 1.0
    else:
        w = w / w_sum
    
    # Weighted subgradient
    g = np.zeros(n)
    # Only use top lags for efficiency
    top_k = min(100, len(conv))
    top_idx = np.argsort(conv)[::-1][:top_k]
    w_top = w[top_idx]
    w_top_sum = w_top.sum()
    if w_top_sum < 1e-30:
        return np.zeros(n)
    w_top = w_top / w_top_sum
    
    for wi, k in zip(w_top, top_idx):
        if wi < 1e-10:
            continue
        k = int(k)
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:

    return g

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    """Solve LP to find Frank-Wolfe direction."""
    n = len(f)
    
    # Build constraint matrix: F_k · b <= M for each k in active_set
    # F_k[j] = f[k-j] if valid
    A_rows = []
    for k in active_set:
        k = int(k)
        row = np.zeros(n)
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row[j] = f[i]
        A_rows.append(row)
    
    A = np.array(A_rows)  # shape: (|active|, n)
    b_vec = np.full(len(active_set), M)

def lp_direction_fast(f, M, active_set):
    """Fast LP direction using dual."""
    n = len(f)
    m = len(active_set)
    
    # Build A: A[k, j] = f[k-j] if valid
    A = np.zeros((m, n))
    for idx, k in enumerate(active_set):
        k = int(k)
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[idx, j] = f[i]
    
    # Dual: minimize M * sum(y) subject to A^T y >= 1, y >= 0
    # In linprog form: minimize [M,...,M] · y subject to -A^T y <= -1, y >= 0
    from scipy.optimize import linprog
    c_dual = np.full(m, M)
    A_dual = -A.T  # shape (n, m)
    b_dual = -np.ones(n)
    
    res = linprog(c_dual, A_ub=A_dual, b_ub=b_dual, bounds=[(0, None)]*m, method='highs')
    if res.status != 0:
        return None
    
    y = res.x
    # Find non-zero positions: where A^T y = 1 (within tolerance)
    At_y = A.T @ y  # shape (n,)
    tol = 1e-6
    nonzero = np.where(np.abs(At_y - 1.0) < tol)[0]
    
    if len(nonzero) == 0:
        # Fallback: uniform
        return np.ones(n) / n
    
    # Solve reduced LP on nonzero positions
    # maximize sum(b[j] for j in nonzero) subject to A[:, nonzero] @ b_nonzero <= M, b_nonzero >= 0
    A_red = A[:, nonzero]
    c_red = -np.ones(len(nonzero))
    res2 = linprog(c_red, A_ub=A_red, b_ub=np.full(m, M), bounds=[(0, None)]*len(nonzero), method='highs')
    if res2.status != 0:
        return None
    
    b_full = np.zeros(n)
    b_full[nonzero] = res2.x
    return b_full

        # For each j, A[idx, j] = f[k-j] if valid
        j_vals = np.arange(n)
        i_vals = k - j_vals
        valid = (i_vals >= 0) & (i_vals < n)
        row[valid] = f[i_vals[valid]]
        A_cols.append(row)
