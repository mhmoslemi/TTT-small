```python
def project_simplex(v, s=1.0):
    """Project v onto {x: sum(x) = s, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
    """Compute smoothed gradient g[m] = 2*sum_j w[m+j]*a[j] for m=0,...,n-1."""
    n = len(a)
    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)
    # g[m] = 2 * conv[m + n - 1]
    return 2.0 * conv[n-1:2*n-1]

def smoothed_gradient_nesterov(a0, time_limit, beta_start=5.0, beta_end=200.0,
                                step_size=0.4, restart_interval=100000,
                                verbose=True, seed=42):
    """Fast smoothed gradient with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Determine FFT size (needs to be >= 3n-2)
    fft_size = 1 << (3*n - 2 - 1).bit_length()
    
    x = a.copy()
    x_prev = a.copy()
    local_best = a.copy()
    local_best_m = best_m
    iterations_since_improve = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive parameters
        frac = min(1.0, t / 500000.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        # Step size: start large, decrease slightly
        step = step_size / (1.0 + 0.001 * frac)
        
        # Nesterov extrapolation
        alpha_t = (t - 1) / (t + 1.5)
        y = x + alpha_t * (x - x_prev)
        y = np.maximum(y, 1e-18)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        
        # Compute softmax weights (the "importance" of each lag)
        beta_ac = beta * ac
        beta_ac -= beta_ac.max()
        w = np.exp(beta_ac)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / w_sum
        
        # Compute smoothed gradient
        g = compute_grad_fft(y, w, fft_size)
        
        # Check gradient norm for early stopping
        g_norm = np.linalg.norm(g)
        
        # Take step and project
        x_new = y - step * g
        x_new = project_simplex(x_new)
        
        x_prev = x
        x = x_new
        
        # Track best
        if t % 10 == 0:  # check every 10 iterations to save time
            m_new = fast_autocorr(x).max()
            if m_new < best_m:
                best_m = m_new
                best_a = x.copy()
                local_best = x.copy()
                local_best_m = m_new
                iterations_since_improve = 0
            else:
                iterations_since_improve += 10
        
        # Periodic restart if stuck
        if iterations_since_improve > restart_interval:
            iterations_since_improve = 0
            # Perturb from local best
            noise_scale = 0.01 * max(1.0, local_best.max())
            pert = rng.normal(0, noise_scale, n)
            x_r = local_best + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            # Also try a "spread" perturbation
            if rng.random() < 0.3:
                # Move mass from peaks to valleys
                peak_idx = np.argsort(x)[::-1][:n//10]
                valley_idx = np.argsort(x)[:n//10]
                move_amt = x[peak_idx].sum() * 0.05
                x[peak_idx] -= move_amt / len(peak_idx)
                x[valley_idx] += move_amt / len(valley_idx)
                x = np.maximum(x, 1e-15)
                x = x / x.sum()
                x_prev = x.copy()
                # Update local best if improved
                m_r = fast_autocorr(x).max()
                if m_r < local_best_m:
                    local_best = x.copy()
                    local_best_m = m_r
        
        if verbose and t % 200000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    SG-Nest t={t} ({elapsed:.1f}s): score={score:.8f}, beta={beta:.1f}, gnorm={g_norm:.4f}")

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=500, verbose=True):
    """LP Frank-Wolfe for polishing."""
    if solver is None:
        return a0, 2.0 * len(a0) * fast_autocorr(a0).max()
    
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select constraints: top max_constraints lags
        if 2*n - 1 <= max_constraints:
            active_lags = np.arange(2*n - 1)
        else:
            # Use all lags above a threshold, or top-k
            threshold = M_curr * 0.95
            active_lags = np.where(ac >= threshold)[0]
            if len(active_lags) > max_constraints:
                active_lags = np.argsort(ac)[::-1][:max_constraints]
        
        # Build the LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        
        # Build constraint matrix
        T = np.zeros((len(active_lags), n))
        rhs = np.zeros(len(active_lags))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
                rhs[row_idx] = ac[k_int]
        
        if len(active_lags) > 0:
            constraints.append(T @ b_var <= rhs)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): score={score:.8f}, n_act={len(active_lags)}")
    
    return best_a, 2.0 * n * best_m

def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Simple coordinate descent polish."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and stall < 20:
        sweep += 1
        improved = False
        n_tries = min(100, n // 2)
        
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.02
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): score={2.0*n*best_m:.8f}")
    
    return best_a, 2.0 * n * best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

```python
def project_simplex(v, s=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - s)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, s / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            y = y / sy

        ac = fast_autocorr(y)

        beta_ac = beta * ac
        beta_ac -= beta_ac.max()
        w = np.exp(beta_ac)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / w_sum

        g = compute_grad_fft(y, w, fft_size)

        x_new = y - step * g
        x_new = project_simplex(x_new)

        x_prev = x
        x = x_new

        if t % 5 == 0:
            m_new = fast_autocorr(x).max()
            if m_new < best_m:
                best_m = m_new
                best_a = x.copy()
                local_best = x.copy()
                local_best_m = m_new
                iterations_since_improve = 0
            else:
                iterations_since_improve += 5

        if iterations_since_improve > restart_interval:
            iterations_since_improve = 0
            noise_scale = 0.01 * max(1.0, local_best.max())
            pert = rng.normal(0, noise_scale, n)
            x_r = local_best + pert
            x_r = np.maximum(x_r, 1e-15)
            sr = x_r.sum()
            if sr > 1e-15:
                x_r = x_r / sr
            x = x_r
            x_prev = x_r.copy()
            m_r = fast_autocorr(x).max()
            if m_r < local_best_m:
                local_best = x.copy()
                local_best_m = m_r

        if verbose and t % 200000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    SG t={t} ({elapsed:.1f}s): score={score:.8f}, beta={beta:.1f}")

    return best_a, 2.0 * n * best_m

def lp_frank_wolfe(a0, time_limit, solver, max_constraints, verbose):
    """LP Frank-Wolfe with constraint selection."""
    if solver is None:
        a = norm_sum1(a0)
        return a, 2.0 * len(a) * fast_autocorr(a).max()

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        total_lags = 2 * n - 1
        if total_lags <= max_constraints:
            active_lags = np.arange(total_lags)
        else:
            threshold = M_curr * 0.97
            active_lags = np.where(ac >= threshold)[0]
            if len(active_lags) > max_constraints:
                active_lags = np.argsort(ac)[::-1][:max_constraints]

        b_var = cp.Variable(n, nonneg=False)
        constraints = [b_var >= 0]
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        if len(active_lags) > 0:
            constraints.append(T @ b_var <= ac[active_lags])

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): score={score:.8f}, n_act={len(active_lags)}")

    return best_a, 2.0 * n * best_m

def coordinate_polish(a0, time_limit, verbose, seed):
    """Coordinate descent / random swap polish."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 30:
        improved = False
        n_tries = min(200, n // 2)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.003, 0.008, 0.02, 0.05]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and stall % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.1f}s): score={2.0*n*best_m:.8f}, stall={stall}")

    return best_a, 2.0 * n * best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """
    Main orchestrator:
    Phase 1 (45%): Fast smoothed gradient + Nesterov (millions of O(n log n) iterations)
    Phase 2 (25%): LP Frank-Wolfe with full constraints
    Phase 3 (15%): Smoothed gradient polish with high beta
    Phase 4 (10%): Alternative length + final polish
    Phase 5 (5%):  Coordinate descent final
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.exp(-3.0 * (x - 0.5) ** 2 / 0.08)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: Smoothed Gradient + Nesterov (45%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 750.0)
    print(f"\n=== Phase 1: SmoothGrad n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_gradient_nesterov(
            a_in, t1, beta_start=5.0, beta_end=300.0,
            step_size=0.4, restart_interval=200000,
            verbose=True, seed=42)
        best_seq, best_score = try_update_seq(a_out, best_seq, best_score, t_global, f"SG1_n{n0}")

    # === Phase 2: LP-FW (25%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP-FW n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30 and solver is not None:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, max_constraints=500, verbose=True)
        best_seq, best_score = try_update_seq(a_out, best_seq, best_score, t_global, f"LP_n{len(best_seq)}")

    # === Phase 3: Smoothed gradient high-beta polish (15%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: SG high-beta n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_gradient_nesterov(
            a_in, t3, beta_start=100.0, beta_end=1000.0,
            step_size=0.15, restart_interval=100000,
            verbose=True, seed=77)
        best_seq, best_score = try_update_seq(a_out, best_seq, best_score, t_global, f"SG3_{len(best_seq)}")

    # === Phase 4: LP-FW final + alt length (10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.06, 120.0)
    print(f"\n=== Phase 4a: LP-FW final n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20 and solver is not None:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, max_constraints=1000, verbose=False)
        best_seq, best_score = try_update_seq(a_out, best_seq, best_score, t_global, f"LPF_{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = min(n0 + 600, 4000)
        t4b = min(remaining * 0.5, 200.0)
        print(f"\n=== Phase 4b: Alt length n={n_alt} (budget={t4b:.0f}s) ===")
        if t4b > 30:
            a_alt = interpolate_sequence(best_seq, n_alt)
            a_out, _ = smoothed_gradient_nesterov(
                a_alt, t4b, beta_start=10.0, beta_end=300.0,
                step_size=0.3, restart_interval=150000,
                verbose=False, seed=99)
            best_seq, best_score = try_update_seq(a_out, best_seq, best_score, t_global, f"SG_alt{n_alt}")

    # === Phase 5: Coordinate polish (5%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.8, 100.0)
    print(f"\n=== Phase 5: Coord n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=False, seed=555)
        best_seq, best_score = try_update_seq(a_out, best_seq, best_score, t_global, f"CD_{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0, 1.0]

    return [float(x) for x in best_seq]
```
