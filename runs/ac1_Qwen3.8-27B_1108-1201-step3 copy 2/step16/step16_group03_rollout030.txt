    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max_grad(y, ac, tau):
    """
    Gradient of smooth max: sm(x) = (1/tau)*log(sum(exp(tau*x)))
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(y)
    max_ac = ac.max()
    exp_vals = np.exp(tau * (ac - max_ac))
    w = exp_vals / exp_vals.sum()
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Y = np.fft.rfft(y, fft_size)
    corr = np.fft.irfft(W * np.conj(Y), fft_size)
    return 2.0 * corr[:n]

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fista_l2(a0, tau_start, tau_end, time_limit, n_anneal, verbose=True, seed=42):
    """FISTA with L2 projection onto simplex, smooth max with tau annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    y_vec = a.copy()
    t_iter = 1.0
    step = 1.0 / (tau_start * 4.0)
    stall = 0
    max_stall = 5000
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, t_iter / n_anneal)
        tau = tau_start + (tau_end - tau_start) * frac
        L = tau * 4.0 + 0.5
        step = 1.0 / L

        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
            beta_mom = (t_iter - 1.0) / t_new
            if t_iter > 1.0:
                y_vec = x + beta_mom * (x - y_vec)
            else:
                y_vec = x.copy()
            t_iter = t_new

            y_vec = np.maximum(y_vec, 0.0)
            sy = y_vec.sum()
            if sy > 1e-15:
                y_vec = y_vec / sy

            ac = fast_conv(y_vec)
            g = smooth_max_grad(y_vec, ac, tau)

            x_new = y_vec - step * g
            x_new = simplex_project(x_new)

            x = x_new
            M_new = fast_conv(x).max()
            if M_new < best_m - 1e-12:
                best_m = M_new
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

            if stall > max_stall:
                perturbation = rng.normal(0, 0.01, n) * best_a
                x = best_a + perturbation
                x = np.maximum(x, 0.0)
                sx = x.sum()
                if sx > 1e-15:
                    x = x / sx
                else:
                    x = best_a.copy()
                y_vec = x.copy()
                t_iter = 1.0
                stall = 0
                restart_count += 1

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA-L2 (elapsed={elapsed:.1f}s): tau={tau:.1f}, J={J:.6f}, restarts={restart_count}")

    return best_a, best_m

def fista_mirror(a0, tau_start, tau_end, time_limit, n_anneal, verbose=True, seed=42):
    """FISTA with mirror (exponential) projection, smooth max with tau annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    y_vec = a.copy()
    t_iter = 1.0
    stall = 0
    max_stall = 5000

    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, t_iter / n_anneal)
        tau = tau_start + (tau_end - tau_start) * frac
        L = tau * 4.0 + 0.5
        step = 1.0 / L

        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
            beta_mom = (t_iter - 1.0) / t_new
            if t_iter > 1.0:
                y_vec = x + beta_mom * (x - y_vec)
            else:
                y_vec = x.copy()
            t_iter = t_new

            y_vec = np.maximum(y_vec, 1e-15)
            sy = y_vec.sum()
            if sy > 1e-15:
                y_vec = y_vec / sy

            ac = fast_conv(y_vec)
            g = smooth_max_grad(y_vec, ac, tau)

            log_x = np.log(y_vec) - step * g
            log_x -= log_x.max()
            x_new = np.exp(log_x)
            s_new = x_new.sum()
            if s_new > 1e-15:
                x_new = x_new / s_new
            else:
                x_new = x.copy()

            x = x_new
            M_new = fast_conv(x).max()
            if M_new < best_m - 1e-12:
                best_m = M_new
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

            if stall > max_stall:
                perturbation = rng.normal(0, 0.01, n) * best_a
                x = best_a + perturbation
                x = np.maximum(x, 1e-10)
                sx = x.sum()
                if sx > 1e-15:
                    x = x / sx
                else:
                    x = best_a.copy()
                y_vec = x.copy()
                t_iter = 1.0
                stall = 0

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA-Mirror (elapsed={elapsed:.1f}s): tau={tau:.1f}, J={J:.6f}")

    return best_a, best_m

def coord_descent(a0, time_limit, verbose=True, seed=42):
    """Peak-aware coordinate descent: move mass from high-contribution to low-contribution indices."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(15, len(ac))]

        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            src = idx[np.argmax(contrib)]

            # Find indices with low contribution to all top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_total = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                total_sens = 0.0
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    j = k2i - dst
                    if 0 <= j < n:
                        total_sens += a[j]
                if total_sens < best_total:
                    best_total = total_sens
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.12]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord (elapsed={elapsed:.1f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done (elapsed={elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def make_init(kind, n):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    xm = (x - (n - 1) / 2.0) / (n / 2.0)
    if kind == 0:
        a = np.ones(n)
    elif kind == 1:
        a = 1.0 - np.abs(xm)
    elif kind == 2:
        a = np.exp(-2.0 * xm * xm)
    elif kind == 3:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
    elif kind == 4:
        a = np.abs(np.sin(np.pi * (x + 0.5) / n)) ** 0.5
    else:
        a = np.ones(n) * (1.0 + 0.1 * np.sin(4.0 * np.pi * x / n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    n0 = len(best_seq) if best_seq is not None else 2400
    if best_seq is None:
        best_seq = make_init(0, n0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    print(f"Initial best: {best_score:.6f}, n={n0}")

    # === Phase 1: FISTA-L2 main run ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA-L2 n={n0} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_l2(a_in, tau_start=5.0, tau_end=80.0,
                            time_limit=t1, n_anneal=30000, verbose=True, seed=42)
        try_update(a_out, "fista_l2_main")

    # === Phase 2: FISTA-Mirror ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: FISTA-Mirror n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_mirror(a_in, tau_start=10.0, tau_end=100.0,
                                time_limit=t2, n_anneal=20000, verbose=True, seed=7)
        try_update(a_out, "fista_mirror")

    # === Phase 3: FISTA-L2 with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: FISTA-L2 variant n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_l2(a_in, tau_start=3.0, tau_end=60.0,
                            time_limit=t3, n_anneal=20000, verbose=True, seed=123)
        try_update(a_out, "fista_l2_v2")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord descent n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent(a_in, time_limit=t4, verbose=True, seed=42)
        try_update(a_out, "coord_1")

    # === Phase 5: FISTA-L2 high tau (fine polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: FISTA-L2 high-tau n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_l2(a_in, tau_start=50.0, tau_end=200.0,
                            time_limit=t5, n_anneal=10000, verbose=False, seed=99)
        try_update(a_out, "fista_high_tau")
        # Quick coord polish after
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:

                                      verbose=False, seed=77)
            try_update(a_out2, "coord_after_high_tau")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length (budget={remaining:.0f}s) ===")
    if remaining > 100:

            n_try = n_best + delta
            if n_try < 800 or n_try > 4000:

            remaining = time_budget - (time.monotonic() - t_global)

            print(f"  Trying n={n_try}...")

    # === Phase 7: Final FISTA + Coord on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 40:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_l2(a_in, tau_start=30.0, tau_end=150.0,
                                time_limit=t7, n_anneal=5000, verbose=False, seed=314)
            try_update(a_out, "fista_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = coord_descent(norm_sum1(best_seq), time_limit=min(remaining * 0.8, 60.0),
                                     verbose=False, seed=555)
            try_update(a_out, "coord_final")

    # === Phase 8: One more FISTA-mirror attempt with high tau ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = fista_mirror(a_in, tau_start=80.0, tau_end=300.0,
                                    time_limit=t8, n_anneal=5000, verbose=False, seed=789)
            try_update(a_out, "fista_mirror_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
