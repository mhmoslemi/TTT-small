            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def subgradient_at_k(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        idx = np.arange(lo, hi)
        grad[lo:hi] += a[k - idx]
    return grad

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    
    if size == 0:
        return None, None
    
    # Build constraint matrix
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def lp_optimize(a0, time_limit, solver, verbose=False, seed=0, relax=1.0):
    """LP best-response with adaptive active set and blending."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    conv_size = 2 * n - 1
    n_iters = 0
    no_improve = 0
    max_no_improve = 80
    
    y = a.copy()
    y_prev = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        
        # Adaptive active set
        progress = min(1.0, t / 500.0)
        # Start with wide active set, narrow over time
        tol = 0.15 * (1.0 - progress) + 0.02 * progress
        base_threshold = M_curr * (1.0 - tol)
        active_set = np.where(conv_y >= base_threshold)[0]
        
        # Always include top peaks
        n_top = max(5, n // 50)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))
        
        # Cap active set size
        max_active = int(2.0 * n)
        if len(active_set) > max_active:
            conv_at_active = conv_y[active_set]
            keep = np.argsort(conv_at_active)[-max_active:]
            active_set = active_set[keep]
        
        # Solve LP with slightly relaxed M for exploration
        M_lp = M_curr * relax
        b, duals = lp_best_response(y, M_lp, active_set, solver)
        
        if b is None:
            # Try with full active set
            active_set = np.argsort(conv_y)[-n_top:]
            b, duals = lp_best_response(y, M_lp, active_set, solver)
            if b is None:
                no_improve += 1
                if no_improve > max_no_improve:
                    # Restart from best
                    y = best_a.copy()
                    y_prev = best_a.copy()
                    no_improve = 0
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm / b_sum
        n_iters += 1
        
        # Blending: adaptive step size
        # Start with larger steps, reduce over time
        alpha = 0.3 * (1.0 - 0.5 * progress) + 0.1 * progress
        # Nesterov momentum
        if t > 1:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
        else:
            momentum = 0.0
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        
        # Perturbation restart
        if no_improve > max_no_improve:
            # Perturb best solution
            noise_scale = 0.05 * rng.random() + 0.01
            perturbed = best_a * (1.0 + noise_scale * rng.standard_normal(n))
            perturbed = np.maximum(perturbed, 0.0)
            s_p = perturbed.sum()
            if s_p > 1e-15:
                y = perturbed * norm / s_p
                y_prev = y.copy()
                a = y.copy()
            no_improve = 0
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def dual_polish(a0, time_limit, solver, verbose=False):
    """Dual-guided subgradient polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 300
    step = 0.3 / n ** 0.25
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Active set for LP
        tol = 0.08
        active_set = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active_set) < 3:
            active_set = np.argsort(conv)[-max(3, n // 100):]
        
        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue
        
        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-5.0 * (M_curr - conv[active_set]) / max(M_curr * 0.02, 1e-10))
            weights = weights / weights.sum()
        
        # Weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            grad += w * subgradient_at_k(a, int(k), n)
        
        # Project gradient to maintain sum constraint
        support = np.where(a > 0.001 * a.max())[0]
        if len(support) < 2:
            break
        
        grad_proj = grad - np.mean(grad[support])
        grad_proj[support] = grad_proj[support] - np.mean(grad_proj[support])
        
        a_trial = a - step * grad_proj
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 1.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualPolish t={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with FFT-based evaluation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 3
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        
        # Focus on coordinates that affect the peak
        peak_k = np.argmax(conv)
        # Coordinates that affect conv[peak_k]: j where peak_k - j is in [0, n-1]
        # i.e., j in [peak_k - n + 1, peak_k] intersected with [0, n-1]
        lo_j = max(0, peak_k - n + 1)
        hi_j = min(n, peak_k + 1)
        important_coords = set(range(lo_j, hi_j))
        
        # Also add some random coordinates for exploration
        n_random = max(5, n // 10)
        random_coords = set(rng.choice(n, n_random, replace=False).tolist())
        important_coords.update(random_coords)
        
        for j in sorted(important_coords):
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            
            # Try a few candidate values
            candidates = [0.0, aj * 0.5, aj * 1.5]
            typical = norm / n
            candidates += [typical, typical * 0.5, typical * 2.0]
            candidates = sorted(set([max(0.0, min(c, norm)) for c in candidates]))
            
            best_x = aj
            best_obj = conv.max()
            
            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                
                # Update convolution
                new_conv = conv.copy()
                delta = x - aj
                lo = max(0, j)
                hi = min(size, j + n)
                new_conv[lo:hi] += 2.0 * delta * a[j:j + (hi - lo)]  # a[lo-j:hi-j] = a[0:hi-lo] when lo=j
                # Wait, this isn't right. Let me fix.
                # (a*a)[k] = sum_{j'} a[j'] * a[k-j']
                # When we change a[j] to x, the change in (a*a)[k] is:
                # (x - aj) * (a[k-j] + a[j-k] - delta*indicator(k-j==j))
                # = (x - aj) * (a[k-j] + a[j-k]) - (x-aj)^2 * indicator(k == 2j)
                # But only when k-j and j-k are valid indices.
                
                # Simpler: just recompute
                old_val = a[j]
                a[j] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = old_val
                    continue
                a_scaled = a * norm / s_new
                new_max = fast_autocorr(a_scaled).max()
                a[j] = old_val
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = true  # Bug: should be True
                    else:
                        a = best_a.copy()
        
        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def multi_subgradient(a0, time_limit, verbose=False, seed=42):
    """Multi-peak subgradient method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.5 / n ** 0.3
    patience = 0
    max_patience = 500
    rng = np.random.default_rng(seed)
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Pick top peaks
        n_peaks = max(3, min(8, n // 100))
        top_indices = np.argsort(conv)[-n_peaks:]
        
        weights = np.exp(-6.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.02, 1e-10))
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k(a, int(k), n)
        
        # Project to sum constraint
        support = np.where(a > 0.001 * a.max())[0]
        if len(support) < 2:
            break
        grad[support] -= np.mean(grad[support])
        
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 2.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-10:
                break
    
    return best_a, best_score

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 3
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        peak_k = np.argmax(conv)

        # Focus on coordinates affecting the peak
        lo_j = max(0, peak_k - n + 1)
        hi_j = min(n, peak_k + 1)
        important_list = list(range(lo_j, hi_j))

        # Add random coordinates
        n_rand = max(10, n // 8)
        rand_coords = rng.choice(n, min(n_rand, n), replace=False).tolist()
        important_list.extend(rand_coords)
        important_list = sorted(set(important_list))

        for j in important_list:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            typical = norm / n
            candidates = sorted(set([
                max(0.0, min(c, norm * 0.5)) for c in
                [0.0, aj * 0.3, aj * 0.6, aj * 1.0, aj * 1.4, aj * 2.0,
                 typical * 0.5, typical, typical * 1.5, typical * 2.5]
            ]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                # Efficient update
                delta = x - aj
                new_conv = conv.copy()
                lo = max(0, j)
                hi = min(size, j + n)
                new_conv[lo:hi] += 2.0 * delta * a[j:j + (hi - lo)]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                # Account for normalization change
                new_sum = a.sum() + delta
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                new_max = (new_conv * scale * scale).max()

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a = best_a.copy()
                conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def multi_subgradient(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.4 / n ** 0.3
    patience = 0
    max_patience = 400
    rng = np.random.default_rng(seed)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        n_peaks = max(3, min(10, n // 80))
        top_indices = np.argsort(conv)[-n_peaks:]

        # Add random near-max for exploration
        if t % 20 == 0:
            threshold = M_curr * (0.85 + 0.1 * rng.random())
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_indices = np.append(top_indices.astype(int), k_rand)

        weights = np.exp(-6.0 * (M_curr - conv[top_indices.astype(int)]) / max(M_curr * 0.02, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k(a, int(k), n)

        support = np.where(a > 0.002 * a.max())[0]
        if len(support) < 2:
            break
        grad[support] -= np.mean(grad[support])

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 2.0)
        else:
            patience += 1
            step *= 0.985
            if step < 1e-10:
                break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def generate_flat_autocorr(n, seed):
    rng = np.random.default_rng(seed)
    phases = rng.uniform(0, 2 * np.pi, n)
    phases[0] = 0.0
    if n % 2 == 0:
        phases[n // 2] = 0.0
    H = np.zeros(n, dtype=complex)
    half = n // 2 + 1
    H[:half] = np.exp(1j * phases[:half])
    for i in range(1, n - n // 2):
        H[i] = np.conj(H[n - i])
    a = np.fft.ifft(H).real
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate():
    """
    Main search: multi-n exploration + LP best-response + dual polish + CD + subgradient.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1760.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Failed to load parent: {e}")

    if best_seq is None:
        n_base = 1600
        best_seq = generate_flat_autocorr(n_base, 42)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting with n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Quick multi-n exploration (80s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_explore = min(80.0, remaining * 0.05)
    if t_explore > 30:
        print(f"\n=== Phase 1: Exploration budget={t_explore:.0f}s ===")
        candidate_ns = []
        for delta in [-400, -200, 200, 400]:
            n_try = n_parent + delta
            if 500 <= n_try <= 4000 and n_try != n_parent:
                candidate_ns.append(n_try)

        t_each = t_explore / max(1, len(candidate_ns))
        for n_try in candidate_ns:
            if time.monotonic() - t_global > time_budget - 100:
                break
            # Resample parent to new length
            x_old = np.linspace(0, 1, n_parent, endpoint=False)
            x_new = np.linspace(0, 1, n_try, endpoint=False)
            a_trial = np.interp(x_new, x_old, best_seq)
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 1e-10:
                a_trial = generate_flat_autocorr(n_try, n_try)

            a_norm = normalize_to(a_trial, n_try)
            a_out, _ = lp_optimize(a_norm, t_each, solver, verbose=False, seed=n_try)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                print(f"  *** Exploration n={n_try}: score={sc:.8f}")
                best_score = sc
                best_seq = a_out.copy()
            else:
                print(f"  Exploration n={n_try}: score={sc:.8f}")

    # === Phase 2: Main LP (largest chunk) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_lp = min(remaining * 0.52, 850.0)
    if t_lp > 60:
        print(f"\n=== Phase 2: Main LP n={n_cur}, budget={t_lp:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = lp_optimize(a_norm, t_lp, solver, verbose=True, seed=0)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            print(f"  *** Phase 2: {sc:.8f}")
            best_score = sc
            best_seq = a_out.copy()
        else:
            print(f"  Phase 2: {sc:.8f}")

    # === Phase 3: Second LP with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_lp2 = min(remaining * 0.18, 280.0)
    if t_lp2 > 30:
        print(f"\n=== Phase 3: LP r2 n={n_cur}, budget={t_lp2:.0f}s ===")
        noise = 0.015 * rng.standard_normal(n_cur)
        perturbed = best_seq * (1.0 + noise)
        perturbed = np.maximum(perturbed, 0.0)
        a_norm = normalize_to(perturbed, n_cur)
        a_out, _ = lp_optimize(a_norm, t_lp2, solver, verbose=True, seed=7777)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            print(f"  *** Phase 3: {sc:.8f}")
            best_score = sc
            best_seq = a_out.copy()

    # === Phase 4: Dual-guided polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_dual = min(remaining * 0.15, 150.0)
    if t_dual > 20:
        print(f"\n=== Phase 4: Dual polish n={n_cur}, budget={t_dual:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = dual_polish(a_norm, t_dual, solver, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            print(f"  *** Phase 4: {sc:.8f}")
            best_score = sc
            best_seq = a_out.copy()

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_cd = min(remaining * 0.15, 130.0)
    if t_cd > 15:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t_cd:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t_cd, verbose=False, seed=55)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            print(f"  *** Phase 5: {sc:.8f}")
            best_score = sc
            best_seq = a_out.copy()

    # === Phase 6: Final LP + subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        t_final_lp = min(remaining * 0.45, 80.0)
        t_sub = remaining - t_final_lp - 5
        if t_final_lp > 15:
            print(f"\n=== Phase 6a: Final LP n={n_cur}, budget={t_final_lp:.0f}s ===")
            a_norm = normalize_to(best_seq, n_cur)
            a_out, _ = lp_optimize(a_norm, t_final_lp, solver, verbose=False, seed=31415)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()

        if t_sub > 10:
            print(f"=== Phase 6b: Subgrad n={n_cur}, budget={t_sub:.0f}s ===")
            a_norm = normalize_to(best_seq, n_cur)
            a_out, _ = multi_subgradient(a_norm, t_sub, verbose=False, seed=271)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
