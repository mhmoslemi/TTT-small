def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    sz = 2 * n - 1
    fs = _next_pow2(sz)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:sz]

def _softmax_obj(x):
    """Top-level objective returning (f, grad) for L-BFGS."""
    n = len(x)
    xs = x - x.max()
    ex = np.exp(xs)
    a = ex / ex.sum()
    
    C = _fast_conv(a)
    bC = _beta_g * C
    bCm = bC.max()
    eb = np.exp(bC - bCm)
    p = eb / eb.sum()
    f_val = (bCm + np.log(eb.sum())) / _beta_g
    
    # Gradient w.r.t. a: g_a[j] = 2 * sum_m p[j+m] * a[m]
    # Computed as 2 * conv(p, a_rev)[j + n - 1]
    a_rev = a[::-1]
    clen = 3 * n - 2
    fs = _next_pow2(clen)
    P = np.fft.rfft(p, fs)
    Ar = np.fft.rfft(a_rev, fs)
    cp = np.fft.irfft(P * Ar, fs)[:clen]
    g_a = 2.0 * cp[n - 1 : 2 * n - 1]
    
    # Gradient w.r.t. x (log-param): g_x = a * (g_a - dot(g_a, a))
    dot = np.dot(g_a, a)
    g_x = a * (g_a - dot)
    
    return f_val, g_x

def _softmax(x):

def _interp_to(a, new_n):

        return a.copy()

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    sz = 2 * n - 1
    fs = _next_pow2(sz)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:sz]

def _softmax_obj(x):
    n = len(x)
    xs = x - x.max()
    ex = np.exp(xs)
    a = ex / ex.sum()
    C = _fast_conv(a)
    bC = _beta_g * C
    bCm = bC.max()
    eb = np.exp(bC - bCm)
    p = eb / eb.sum()
    f_val = (bCm + np.log(eb.sum())) / _beta_g
    a_rev = a[::-1]
    clen = 3 * n - 2
    fs = _next_pow2(clen)
    P = np.fft.rfft(p, fs)
    Ar = np.fft.rfft(a_rev, fs)
    cp = np.fft.irfft(P * Ar, fs)[:clen]
    g_a = 2.0 * cp[n - 1:2 * n - 1]
    dot = np.dot(g_a, a)
    g_x = a * (g_a - dot)
    return f_val, g_x

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-softmax parameterized smooth-max (log-sum-exp) surrogate
2. Beta annealing for progressive sharpening of the max
3. FFT-based O(n log n) gradient via cross-correlation identity
4. Multi-length, multi-initialization exploration for diversity
5. Exponentiated gradient (mirror descent) with Nesterov momentum
6. Coordinate descent with mass-transfer moves
7. Adaptive time management across phases
"""

import numpy as np
from scipy.optimize import minimize
import time

_beta_g = 1.0
_TG = 0.0


def _task_sort_key(item, bn):
    return (0 if item[0] == bn else 1, 0 if item[1] == 'par' else 1, item[0])


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    sz = 2 * n - 1
    fs = _next_pow2(sz)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:sz]


def _softmax_obj(x):
    n = len(x)
    xs = x - x.max()
    ex = np.exp(xs)
    a = ex / ex.sum()
    C = _fast_conv(a)
    bC = _beta_g * C
    bCm = bC.max()
    eb = np.exp(bC - bCm)
    p = eb / eb.sum()
    f_val = (bCm + np.log(eb.sum())) / _beta_g
    a_rev = a[::-1]
    clen = 3 * n - 2
    fs = _next_pow2(clen)
    P = np.fft.rfft(p, fs)
    Ar = np.fft.rfft(a_rev, fs)
    cp = np.fft.irfft(P * Ar, fs)[:clen]
    g_a = 2.0 * cp[n - 1:2 * n - 1]
    dot = np.dot(g_a, a)
    g_x = a * (g_a - dot)
    return f_val, g_x


def _softmax(x):
    xs = x - x.max()
    ex = np.exp(xs)
    return ex / ex.sum()


def _a_to_x(a):
    a = np.maximum(a, 1e-15)
    lg = np.log(a)
    return lg - lg.mean()


def _eval_safe(a):
    try:
        return float(evaluate_sequence([float(v) for v in a]))
    except Exception:
        return float('inf')


def _score_direct(a):
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e18
    C = _fast_conv(a)
    return 2.0 * n * C.max() / (s * s)


def _interp_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    an = np.maximum(a, 0.0)
    s = an.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    an = an / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    r = np.interp(nx, ox, an)
    r = np.maximum(r, 0.0)
    rs = r.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return r / rs


def _init_raised_cos(n):
    j = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def _init_sin2(n):
    j = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (j + 0.5) / n) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def _init_rand_smooth(n, seed, w):
    rng = np.random.default_rng(seed)
    a = rng.random(n).astype(np.float64)
    if w > 1:
        k = np.ones(w) / w
        a = np.convolve(a, k, mode='same')
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def _init_gauss(n, wf):
    j = np.arange(n, dtype=np.float64)
    c = n / 2.0
    w = wf * n
    a = np.exp(-0.5 * ((j - c) / w) ** 2)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def _init_bell(n):
    j = np.arange(n, dtype=np.float64)
    c = n / 2.0
    a = 1.0 - np.abs(j - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def _run_lbfgs(x, betas, maxit, tlim):
    global _beta_g
    t0 = time.monotonic()
    best_x = x.copy()
    best_f = 1e18
    for beta in betas:
        if time.monotonic() - t0 > tlim:
            break
        _beta_g = float(beta)
        x_cur = x - x.mean()
        try:
            res = minimize(_softmax_obj, x_cur, jac=True, method='L-BFGS-B',
                          options={'maxiter': maxit, 'maxfun': maxit * 3,
                                   'ftol': 1e-14, 'gtol': 1e-10})
            x_cur = np.asarray(res.x, dtype=np.float64)
            if not np.isfinite(x_cur).all():
                x_cur = x.copy()
        except Exception as e:
            print(f"      L-BFGS err beta={beta}: {e}")
            break
        _beta_g = 100000.0
        f_true, _ = _softmax_obj(x_cur)
        if f_true < best_f:
            best_f = f_true
            best_x = x_cur.copy()
        x = x_cur
        el = time.monotonic() - t0
        print(f"      beta={beta:>7.1f}  f={f_true:.8f}  ({el:.1f}s  nit={res.nit})")
    return best_x


def _mirror_desc(a_init, tlim, seed, topk_init, c0):
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    xprev = a.copy()
    stall = 0
    max_stall = 3000
    tk = topk_init
    for t in range(1, 100000000):
        if time.monotonic() - t0 > tlim:
            break
        bt = (t - 1.0) / (t + 1.5)
        y = x + bt * (x - xprev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        C = _fast_conv(y)
        km = min(tk, len(C))
        tidx = np.argsort(C)[::-1][:km]
        tv = np.maximum(C[tidx], 1e-30)
        w = tv ** 2.0
        w = w / w.sum()
        g = np.zeros(n)
        for wi, k in zip(w, tidx):
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            for j in range(js, je):
                m = ki - j
                if 0 <= m < n:
                    g[j] += 2.0 * wi * y[m]
        eta = c0 / (np.sqrt(float(t)) + 1.0)
        la = np.log(y) - eta * g
        la -= la.max()
        an = np.exp(la)
        an = np.maximum(an, 1e-30)
        an = an / an.sum()
        xprev = x
        x = an
        Mn = _fast_conv(x).max()
        if Mn < best_m - 1e-15:
            best_m = Mn
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            xr = best_a * (1.0 + rng.normal(0, 0.01, n))
            xr = np.maximum(xr, 1e-10)
            xr = xr / xr.sum()
            x = xr
            xprev = xr
            stall = 0
            tk = min(tk + 1, 10)
        if t % 5000 == 0:
            el = time.monotonic() - t0
            print(f"      MD t={t} ({el:.1f}s) sc={2.0*n*best_m:.6f} tk={tk} stall={stall}")
    return best_a
