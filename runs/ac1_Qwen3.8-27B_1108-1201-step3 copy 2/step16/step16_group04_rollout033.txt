def compute_lse_gradient(a, ac, c):
    """Compute exact gradient of log-sum-exp(c^{-1} * ac) w.r.t. a."""
    n = len(a)
    max_ac = ac.max()
    w = np.exp((ac - max_ac) / c)
    w_sum = w.sum()
    if w_sum < 1e-30:

    else:
        w = w / w_sum

    g = np.empty(n)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_equalize(a, n_iter=20, eps=1e-8):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def solve_lp_direction(y, M, active_lags, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)

    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)

    if not constraints:
        return None

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Adaptive constraint selection: use threshold + top-K
        threshold = M_curr * 0.7
        active_mask = ac >= threshold
        num_active = active_mask.sum()
        # If too many, use top-K
        top_k = max(50, min(200, n // 10))
        if num_active > top_k:
            active_lags = np.argsort(ac)[::-1][:top_k]
        else:
            active_lags = np.where(active_mask)[0]

        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            continue

        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, n_constraints={len(active_lags)}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with adaptive move sizing."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            n_try = min(100, len(candidates))
            chosen = candidates[rng.permutation(len(candidates))[:n_try]]

            best_dst = -1
            best_new_m = best_m

            for dst in chosen:
                if dst == src:
                    continue
                # Try moving a small amount
                for frac in [0.01, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_dst = dst
                        best_frac = frac

            if best_dst >= 0 and best_new_m < best_m - 1e-16:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_new_m
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / max(a.sum(), 1e-15)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / max(a.sum(), 1e-15)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def peak_shaving(a0, time_limit, rng_seed=99):
    """Directly target and reduce the single highest peak."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        peak_lag = int(np.argmax(ac))
        peak_val = ac[peak_lag]

        if peak_val < best_m * (1 - 1e-10):
            best_m = peak_val
            best_a = a.copy()

        # Find the dominant contributors to the peak
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_lag - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            break

        # Top contributors
        top_contrib_idx = idx[np.argsort(contrib)[::-1][:min(10, len(idx))]]

        # Try reducing top contributors and redistributing to low-sensitivity positions
        for src in top_contrib_idx[:5]:
            # Find positions where adding mass has minimal impact on all top peaks
            sensitivity = np.zeros(n)
            top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
            for k2 in top_k:
                k2i = int(k2)
                d = k2i - np.arange(n)
                valid = (d >= 0) & (d < n)
                sensitivity[valid] += a[d[valid]]
                valid2 = (np.arange(n) - k2i) >= 0
                valid2 = (np.arange(n) - k2i) >= 0 & (np.arange(n) - k2i) < n
                sensitivity[valid2] += a[np.arange(n)[valid2] - k2i]

            # Find low-sensitivity positions far from peak
            low_sens = np.where(sensitivity < np.percentile(sensitivity, 20))[0]
            if len(low_sens) == 0:
                low_sens = np.where(sensitivity < np.percentile(sensitivity, 50))[0]

            for dst in low_sens[:20]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        break

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"    PeakShave ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase, multi-length strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    def get_elapsed():
        return time.monotonic() - t_global

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={get_elapsed():.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 0: Spectral pre-conditioning (quick) ===
    print(f"\n=== Phase 0: Spectral Equalization ===")
    a_spec = spectral_equalize(best_seq, n_iter=25)
    try_update(a_spec, "spectral")

    # === Phase 1: Multi-length LSE gradient (40% time) ===
    remaining = time_budget - get_elapsed()
    print(f"\n=== Phase 1: LSE Gradient, budget={remaining:.0f}s ===")
    t1_total = remaining * 0.35

    # Determine which lengths to try
    n_current = len(best_seq)
    lengths_to_try = []
    if n_current == 2400:
        lengths_to_try = [2400, 2800, 3200]
    elif n_current <= 2400:
        lengths_to_try = [n_current, n_current + 400, n_current + 800]
    else:
        lengths_to_try = [n_current, n_current - 400, n_current - 800]
    # Filter to reasonable range
    lengths_to_try = [max(1600, min(l, 4000)) for l in lengths_to_try]
    # Remove duplicates
    seen = set()
    lengths_to_try = [l for l in lengths_to_try if not (l in seen or seen.add(l))]

    per_length_time = t1_total / len(lengths_to_try)

    for n_try in lengths_to_try:
        remaining = time_budget - get_elapsed()
        if remaining < 60:
            break
        t_alloc = min(per_length_time, remaining * 0.4)
        if t_alloc < 30:
            continue

        if n_try == n_current and best_seq is not None:
            a_init = normalize(best_seq)
        elif n_try == n_current:
            a_init = best_seq.copy()
        else:
            a_init = interpolate_sequence(best_seq, n_try)
            a_init = spectral_equalize(a_init, n_iter=10)

        print(f"  LSE-AG n={n_try}, time={t_alloc:.0f}s")
        a_out, m_out = lse_accelerated_gradient(a_init, t_alloc, c_start=3.0, c_end=0.05, verbose=True)
        try_update(a_out, f"lse_n{n_try}")

        # Quick LP polish after LSE
        remaining = time_budget - get_elapsed()
        if solver is not None and remaining > 60:
            t_lp = min(remaining * 0.15, 90.0)
            if t_lp > 20:
                print(f"  LP-FW polish n={n_try}, time={t_lp:.0f}s")
                a_out2, _ = lp_frank_wolfe(a_out, t_lp, solver, verbose=False)
                try_update(a_out2, f"lp_n{n_try}")

    # === Phase 2: Focused LP-FW on best (25% time) ===
    remaining = time_budget - get_elapsed()
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW n={n_best}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.25, 300.0)
    if t2 > 30 and solver is not None:
        a_in = normalize(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 3: LSE gradient again with sharper c (15% time) ===
    remaining = time_budget - get_elapsed()
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE sharp n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.20, 200.0)
    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, _ = lse_accelerated_gradient(a_in, t3, c_start=0.5, c_end=0.02, verbose=True)
        try_update(a_out, f"lse_sharp_n{n_best}")

    # === Phase 4: Peak shaving (10% time) ===
    remaining = time_budget - get_elapsed()
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak Shaving n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.15, 120.0)
    if t4 > 20:
        a_in = normalize(best_seq)
        a_out, _ = peak_shaving(a_in, t4, rng_seed=42)
        try_update(a_out, f"peak_n{n_best}")

    # === Phase 5: Coordinate polish (10% time) ===
    remaining = time_budget - get_elapsed()
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord n={n_best}, budget={remaining:.0f}s ===")
    t5 = min(remaining * 0.15, 120.0)
    if t5 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Final LP (5% time) ===
    remaining = time_budget - get_elapsed()
    n_best = len(best_seq)
    if solver is not None and remaining > 30:
        print(f"\n=== Phase 6: Final LP n={n_best} ===")
        t6 = min(remaining * 0.4, 80.0)
        if t6 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=False)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - get_elapsed()
    if remaining > 20:
        n_best = len(best_seq)
        t7 = min(remaining * 0.8, 60.0)
        if t7 > 10:
            print(f"\n=== Phase 7: Final Coord n={n_best} ===")
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={get_elapsed():.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
