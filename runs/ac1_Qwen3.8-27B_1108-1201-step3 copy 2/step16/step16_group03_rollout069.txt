"""
Minimizes  2*n*max(conv(a,a))/(sum(a))^2  for a non-negative step function a.

Strategy (all pure numpy; cvxpy used only as an optional, time-boxed polish):
  1) Mirror (multiplicative-weights) descent on the simplex with Nesterov
     extrapolation and backtracking on the exact surrogate objective.
  2) Adaptive multi-peak subgradient: peaks are weighted by their excess over a
     slowly-decreasing target level, so the optimizer focuses on the true
     bottleneck peaks instead of the single global max.
  3) Interleaved coordinate/2-opt mass-transfer with a graded-fraction line
     search that fine-tunes individual entries without large surrogate jumps.
  4) Symmetric-projection start and multi-start perturbation to escape basins.
  5) Multi-length interpolation exploration, then a final fine coordinate polish.

The surrogate 2*n*max(fast_conv(a)) over a non-negative, sum-1 vector equals the
value returned by evaluate_sequence, so it is used cheaply for tracking and
acceptance; evaluate_sequence is called once at the end for the reported value.
"""

import numpy as np
import time

# --------------------------- fast primitives ---------------------------

def fast_conv(a):
    """Full linear convolution a*a via rfft (real, squared). Returns length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def surrogate(a):
    """Exact objective for a non-negative, sum-1 vector: 2*n*max(conv(a,a))."""
    n = a.size
    return 2.0 * n * float(fast_conv(a).max())


def evaluate_safe(a):
    try:
        return float(evaluate_sequence([float(x) for x in np.asarray(a).ravel()]))
    except Exception:
        return float("inf")


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size) / a.size
    return a / s


def to_symmetric(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = 0.5 * (a + a[::-1])
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss < 1e-15:
        return np.ones(a.size) / a.size
    return s / ss


def interpolate_sequence(f, new_n):
    old_n = f.size
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


# --------------------------- subgradients ---------------------------

def peak_subgradient(y, k):
    """Subgradient of C[k]=sum_j y[j]y[k-j] w.r.t. y (linear in y)."""
    n = y.size
    k = int(k)
    g = np.zeros(n)
    i0 = max(0, k - n + 1)
    i1 = min(n, k + 1)
    for i in range(i0, i1):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * y[j]
    return g


def weighted_subgradient(y, ac, target, top_k, p):
    """Weighted sum of peak subgradients for peaks above `target` by excess^p."""
    m = ac.size
    thr = top_k if top_k < m else m
    order = np.argsort(ac)[::-1][:thr]
    w = np.zeros(thr)
    idx2 = []
    for ii in range(thr):
        k = order[ii]
        ex = ac[k] - target
        if ex > 1e-12 and ac[k] > 0:
            w[ii] = ex ** p
            idx2.append(k)
    wsum = w.sum()
    if not idx2 or wsum < 1e-30:
        # fall back to global max peak
        return peak_subgradient(y, int(np.argmax(ac))), int(np.argmax(ac))
    w = w / wsum
    g = np.zeros(y.size)
    kk = None
    for wi, k in zip(w, idx2):
        g += wi * peak_subgradient(y, int(k))
        if kk is None or ac[k] > ac[kk]:
            kk = k
    return g, int(kk)


# --------------------------- mirror descent core ---------------------------

def mirror_update(y, g, eta, floor=1e-12):
    """Multiplicative-weights (mirror on simplex) step: a = y*exp(-eta*g), renorm."""
    lg = np.log(np.maximum(y, floor)) - eta * g
    mx = lg.max()
    a = np.exp(lg - mx)
    a = np.maximum(a, floor)
    s = a.sum()
    if s < 1e-15:
        return y.copy()
    return a / s


def amd_run(a0, t_limit, seed=42, use_sym=False, top_k=3, p=1.6,
            c0=1.2, c_min=1e-4, target_decay=0.99985, verbose=False):

    a = norm_sum1(a0)
    if use_sym:
        a = to_symmetric(a)
    best_a = a.copy()
    best_m = surrogate(a)

    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    eta = c0
    target = best_m / (2.0 * n) * 0.9      # level above which peaks are penalized
    target_floor = best_m / (2.0 * n)
    stall = 0
    max_stall = 400
    step = 0

    def elapsed():
        return time.monotonic() - t0

    while elapsed() < t_limit:
        step += 1
        # Nesterov extrapolation
        bt = (step - 1.0) / (step + 1.5)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        g, _ = weighted_subgradient(y, ac, target, top_k, p)
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            break

        # backtracking on exact surrogate
        accepted = False
        for _ in range(12):
            cand = mirror_update(y, g / gn, eta)
            if use_sym and step % 5 == 0:
                cand = to_symmetric(cand)
            sc = surrogate(cand)
            if sc < best_m - 1e-15:
                x = cand
                best_m = sc
                best_a = cand.copy()
                stall = 0
                # grow step a little on success
                eta = min(eta * 1.08, c0)
                accepted = True
                break
            eta *= 0.6
        if not accepted:
            # keep a small exploratory move in the averaged point direction
            x = mirror_update(y, g / gn, max(eta, c_min))
            if use_sym:
                x = to_symmetric(x)
            sc = surrogate(x)
            if sc < best_m - 1e-15:
                best_m = sc
                best_a = x.copy()
                stall = 0
            else:
                x = best_a.copy()
                stall += 1

        # slowly lower the target level to keep focus moving
        target = max(target * target_decay, target_floor)

            x = r

            eta = c0

                  f" eta={eta:.3f} top_k={top_k} p={p:.2f} target={target:.4e}"
                  f" stall={stall}")

    return best_a, best_m


# --------------------------- coordinate / 2-opt polish ---------------------------

def _frac_trial(a, src, dst, amt):
    a2 = a.copy()
    a2[src] -= amt
    a2[dst] += amt
    if a2[src] < -1e-15 or a2[dst] < 0:
        return None
    a2 = np.maximum(a2, 0.0)
    s = a2.sum()
    if s < 1e-15:
        return None
    a2 = a2 / s
    return a2


def coordinate_polish(a0, t_limit, seed=123, verbose=False):
    n = a0.size
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = surrogate(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 250
    fracs = (0.001, 0.003, 0.008, 0.02, 0.05, 0.10)

    def elapsed():
        return time.monotonic() - t0

    while elapsed() < t_limit and stall < max_stall:
        ac = fast_conv(a)
        m = ac.size
        npeaks = min(20, m)
        peaks = np.argsort(ac)[::-1][:npeaks]
        improved = False

        for k in peaks:
            if elapsed() > t_limit:
                break
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            idx = np.arange(i0, i1)
            if idx.size == 0:
                continue
            jidx = k - idx
            valid = (jidx >= 0) & (jidx < n)
            if not valid.any():
                continue
            contrib = np.where(valid, a[idx] * a[idx - (k - idx)], 0.0)
            contrib = np.where(valid, a[idx] * a[k - idx], 0.0)
            if contrib.max() < 1e-20:
                continue
            src = idx[int(np.argmax(contrib))]

            # destinations not lying under any of the top peaks
            hot = np.zeros(n, dtype=bool)
            for k2 in peaks[:10]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                hot[s2:e2] = True
            cold = np.where(~hot)[0]
            pool = cold if cold.size else np.arange(n)
            nsel = min(pool.size, 400)
            chosen = pool[rng.permutation(pool.size)[:nsel]]

            best_dst = -1
            best_sens = np.inf
            for dst in chosen:
                if dst == src:
                    continue
                # sensitivity: how strongly dst couples into the hot peaks
                sens = a[dst]
                for k2 in peaks[:6]:
                    d2 = int(k2) - dst
                    if 0 <= d2 < n:
                        sens += 0.5 * a[d2]
                if sens < best_sens:
                    best_sens = sens
                    best_dst = dst
            if best_dst < 0:
                continue
            dst = best_dst

            found = False
            for fr in fracs:
                amt = a[src] * fr
                if amt < 1e-18:
                    continue
                tr = _frac_trial(a, src, dst, amt)
                if tr is None:
                    continue
                sc = surrogate(tr)
                if sc < best_m - 1e-15:
                    a = tr
                    best_m = sc
                    best_a = tr.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            if not found and stall > 80:
                # escape nudge
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.002, 0.02)
                    tr = _frac_trial(a, i, j, amt)
                    if tr is not None:
                        sc = surrogate(tr)
                        if sc < best_m:
                            a = tr
                            best_m = sc
                            best_a = tr.copy()
                            stall = 0

        if not improved:
            stall += 1

    if verbose:
        print(f"      coord ({time.monotonic() - t0:.1f}s) surr={best_m:.8f}")
    return best_a, best_m

def lp_solver():
    if not _HAS_CP:
        return None
    for s in (cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS):
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ("optimal", "optimal_inaccurate"):
                return s
        except Exception:
            continue
    return None


def lp_polish(a0, t_limit, solver, verbose=False):
    """Frank-Wolfe: LP gives descent direction b with (a*b)_inf<=M, move+renorm."""
    n = a0.size
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = surrogate(a)
    t0 = time.monotonic()

    def elapsed():
        return time.monotonic() - t0

    while elapsed() < t_limit and n <= 6000:
        M = float(fast_conv(a).max())
        if M < 1e-14:
            break
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a
        b = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ("optimal", "optimal_inaccurate"):
            break
        bv = np.maximum(np.asarray(b.value, dtype=np.float64).ravel(), 0.0)
        sb = bv.sum()
        if sb < 1e-15:
            break
        direction = bv / sb
        best_alpha = 0.2
        best_eval = surrogate(norm_sum1(a + 0.2 * (direction - a)))
        for alpha in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5):
            cand = norm_sum1((1 - alpha) * a + alpha * direction)
            sc = surrogate(cand)
            if sc < best_eval:
                best_eval = sc
                best_alpha = alpha
        a_new = norm_sum1((1 - best_alpha) * a + best_alpha * direction)
        sc = surrogate(a_new)
        if sc < best_m - 1e-15:
            best_m = sc
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new
        if verbose and best_m < 1e18:
            print(f"      lp t={elapsed():.1f}s surr={best_m:.8f}")

    return best_a, best_m

def update_best(state, a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return
    a = a / s
    sc = surrogate(a)
    if np.isfinite(sc) and sc < state["score"]:
        state["score"] = sc
        state["seq"] = a.copy()
        print(f"    *** NEW BEST [{label}] surr={sc:.8f} n={a.size} t={state['t0'] and (time.monotonic() - state['t0']):.0f}s")

            tl = min(remaining() * 0.30, 200.0)
            if tl > 40:
                a_out, _ = lp_polish(state["seq"], tl, solv, verbose=True)
                update_best(state, a_out, "lp_p6")

    # ================= Phase 7: final fine coordinate polish =================
    if remaining() > 25:
        print("== Phase 7: final coordinate polish ==")
        tl = min(remaining() * 0.70, 150.0)
        if tl > 10:
            a_out, _ = coordinate_polish(state["seq"], tl, seed=909, verbose=True)
            update_best(state, a_out, "coord_final")

    # ---------------- final validation ----------------
    if state["seq"] is None:
        print("ERROR: no valid sequence; returning trivial.")
        return [1.0]

    final = norm_sum1(state["seq"])
    fs = evaluate_safe(final)
    if not np.isfinite(fs):
        fs = surrogate(final)
    print(f"\nFINAL reported score={fs:.8f}  (surrogate {surrogate(final):.8f})"
          f"  n={final.size}  t={time.monotonic() - t0:.0f}s")
    return [float(v) for v in final]
