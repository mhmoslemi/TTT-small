def fast_autocorr(a):
    """Compute linear autocorrelation (convolution) via FFT."""
    n = len(a)

    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize_v2(a, n_iter=40, eps=1e-10, blend=0.5):
    """Aggressive iterative DFT power flattening with multiple strategies."""
    a = normalize(a).copy()
    n = len(a)
    for it in range(n_iter):
        A = np.fft.fft(a)
        P = (A.real ** 2 + A.imag ** 2) + eps
        # Inverse square root for flattening
        C = 1.0 / np.sqrt(P)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1.0 - blend) * a + blend * a_new
        # Occasionally try a more aggressive blend
        if it % 8 == 7:
            A = np.fft.fft(a)
            P = (A.real ** 2 + A.imag ** 2) + eps
            C = 1.0 / np.power(P, 0.75)
            A_new = A * C
            a_new = np.real(np.fft.ifft(A_new))
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                a = 0.7 * a + 0.3 * a_new
    return a

        # Part 3: diagonal p[2i]*a[i] if valid
        if 2 * i < p_len:
            h[i] += p[2 * i] * a[i]
    return h


def smoothed_obj_grad(z, beta):
    """Compute smoothed LSE objective and gradient for a=softmax(z)."""
    n = len(z)
    z_shift = z - z.max()
    a = np.exp(z_shift)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(n)
    a = a / a_sum

    c = fast_autocorr(a)
    beta_c = beta * c
    bc_max = beta_c.max()
    exp_vals = np.exp(beta_c - bc_max)
    log_sum = bc_max + np.log(exp_vals.sum())
    J = (2.0 * n / beta) * log_sum

    # Softmax weights
    p = exp_vals / exp_vals.sum()

    # Gradient w.r.t. a
    h = compute_grad_a_loop(p, a)
    grad_a = 2.0 * n * h

    # Chain rule through softmax
    a_dot_grad = a @ grad_a
    grad_z = a * (grad_a - a_dot_grad)

    return J, grad_z


def lbgfss_smoothed(a0, betas, time_limit, max_iter_per_beta=200):
    """Run L-BFGS on smoothed objective with increasing beta."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Initialize z from a (log of a)
    z = np.log(np.maximum(a, 1e-30))

    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        print(f"    L-BFGS beta={beta:.1f} (t={time.monotonic()-t0:.1f}s)")

        def fun_g(z_in):
            return smoothed_obj_grad(z_in, beta)

        try:
            res = scipy_minimize(
                fun_g, z, method='L-BFGS-B',
                jac=True, options={'maxiter': max_iter_per_beta,
                                   'gtol': 1e-8, 'ftol': 1e-12}
            )
            if res.success or res.fun < 1e10:
                z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
        except Exception as e:
            print(f"    L-BFGS error at beta={beta}: {e}")
            a_new = a

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        score_new = 2.0 * n * M_new
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"    L-BFGS beta={beta:.1f}: score={score_new:.6f}")

        a = a_new
        z = np.log(np.maximum(a, 1e-30))

    return best_a, best_score


def solve_lp_direction(y, M, active_lags, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    if not constraints:
        return None

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def enhanced_lp_fw(a0, time_limit, solver, verbose=True):
    """Enhanced LP Frank-Wolfe with adaptive constraints and backtracking."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    percentile = 0.75  # Use top 25% of constraints
    min_constraints = 40
    max_constraints = 300

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_M:
            best_M = M_curr
            best_a = y.copy()

        # Adaptive constraint selection: top percentile
        n_active = max(min_constraints, min(max_constraints, int(percentile * len(ac))))
        active_lags = np.argsort(ac)[::-1][:n_active]

        # Increase percentile slowly
        if t % 100 == 0:
            percentile = min(0.95, percentile + 0.005)

        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            continue

        # Backtracking line search
        alpha = 1.0
        J_curr = 2.0 * n * fast_autocorr(y).max()
        found = False
        for _ in range(20):
            x_trial = (1.0 - alpha) * y + alpha * direction
            x_trial = np.maximum(x_trial, 0.0)
            sx = x_trial.sum()
            if sx < 1e-15:
                break
            x_trial = x_trial / sx
            M_trial = fast_autocorr(x_trial).max()
            if M_trial < M_curr - 1e-14 * alpha:
                found = True
                break
            alpha *= 0.5

        if not found:
            alpha = 2.0 / (t + 2.0)
            x_trial = (1.0 - alpha) * y + alpha * direction
            x_trial = np.maximum(x_trial, 0.0)
            sx = x_trial.sum()
            if sx < 1e-15:
                x_trial = y.copy()
            else:
                x_trial = x_trial / sx

        x_prev = y
        x = x_trial

        M_x = fast_autocorr(x).max()
        if M_x < best_M:
            best_M = M_x
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW t={t} ({elapsed:.1f}s): score={2*n*best_M:.6f}, pct={percentile:.2f}")

    return best_a, best_M


def pairwise_search(a0, time_limit, rng_seed=42):
    """Efficient local search via pairwise mass transfers targeting peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Get top peak lags
        n_peaks = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]
        top_vals = ac[top_lags]

        improved = False
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a random peak lag
            k = int(top_lags[int(rng.integers(len(top_lags)))])
            k_int = k

            # Find the pair (i, j) with i+j = k (or |i-j| = k) that contributes most
            # For lag k in convolution: c[k] = sum_j a[j]*a[k-j]
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[k_int - js]
            if contrib.max() < 1e-20:
                continue

            src_j = js[np.argmax(contrib)]
            src_k = k_int - src_j

            # Try moving mass from src_j or src_k to a random location
            src = src_j if a[src_j] >= a[src_k] else src_k
            for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                # Try a few random destinations
                for _ in range(10):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                if a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)

    return best_a, best_M


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def frequency_shaping_step(a, n_steps=5):
    """Apply a few frequency-shaping steps to flatten the spectrum."""
    n = len(a)
    for _ in range(n_steps):
        A = np.fft.fft(a)
        P = (A.real ** 2 + A.imag ** 2) + 1e-12
        C = 1.0 / np.sqrt(P)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.6 * a + 0.4 * a_new
    return a
