def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute objective 2*n*max(autocorr)/(sum)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_linear_ramp(n, power=1.0):
    """Create ramp: a[i] = (i+1)^power."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return a / a.sum()


def make_windowed_ramp(n, window='cosine'):
    """Ramp with a window function to reduce edge effects."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i.copy()
    x = np.linspace(0, 1, n)
    if window == 'cosine':
        w = 0.5 + 0.5 * np.cos(np.pi * x)
    elif window == 'hann':
        w = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif window == 'blackman':
        w = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    else:
        w = np.ones(n)
    a = a * w
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_sine_power(n, power=1.0, shift=0.0):
    """a[i] = sin(pi*(i+1+shift)/(n+1))^power."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 1 + shift) / (n + 1)
    a = np.sin(np.pi * x) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_poly_ramp(n, p1=1.0, p2=1.0):
    """a[i] = (i+1)^p1 * (n-i)^p2 (two-sided power)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = i ** p1 * j ** p2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Coordinate descent that only accepts improvements. Key method for polishing."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_j = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Precompute: for each lag k, the indices (i, k-i) that contribute
    # We'll work with the top lags

    stall = 0
    max_stall = 500
    passes = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))

        # Get top contributing lags
        n_top = min(30, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Indices that contribute to lag k: i where 0<=i<n and 0<=k-i<n
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            partner = k_int - idx
            contrib = a[idx] * a[partner]
            max_contrib_idx = idx[np.argmax(contrib)]
            max_partner = partner[np.argmax(contrib)]

            # Try reducing the largest contributor by moving mass elsewhere
            src = int(max_contrib_idx)
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Compute sensitivity: which positions are "safe" to add mass
            # Positions that don't contribute to top lags are safer
            safe_mask = np.ones(n, dtype=bool)
            for k2 in top_lags[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                safe_mask[s2:e2 + 1] = False
                partner2 = k2i - np.arange(s2, e2 + 1)
                safe_mask[partner2] = False

            safe_positions = np.where(safe_mask)[0]
            if len(safe_positions) == 0:
                safe_positions = np.arange(n)

            # Pick destination: safe position with lowest current value
            # (low value means less impact on autocorrelation)
            scores = a[safe_positions]
            dst = safe_positions[np.argmin(scores)]

            if dst == src:
                dst = safe_positions[np.argmin(scores[safe_positions != src])] if len(safe_positions) > 1 else (src + 1) % n

            # Try moving a small fraction
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-18:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                # Try reducing the partner instead
                src2 = int(max_partner)
                if src2 != src and a[src2] > 1e-14:
                    src2_val = a[src2]
                    dst2 = safe_positions[np.argmin(a[safe_positions])]
                    if dst2 != src2:
                        for frac in [0.001, 0.003, 0.005, 0.01]:
                            amt = src2_val * frac
                            if amt < 1e-15:
                                continue
                            a_trial = a.copy()
                            a_trial[src2] -= amt
                            a_trial[dst2] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            st = a_trial.sum()
                            if st < 1e-15:
                                continue
                            a_trial = a_trial / st
                            M_trial = fast_autocorr(a_trial).max()
                            if M_trial < best_m - 1e-18:
                                a = a_trial
                                best_m = M_trial
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:

                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        passes += 1
        if verbose and passes % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD pass={passes} ({elapsed:.0f}s): J={J:.8f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD final ({elapsed:.0f}s): J={J:.8f}")

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200):
    """LP-based Frank-Wolfe with Nesterov acceleration and top-K constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Build constraint matrix for top-K most binding constraints
        # (f * b)[k] = sum_i b[i] * f[k-i]
        # We want to select K constraints where (y*b)[k] is closest to M_curr
        # For efficiency, compute (y*y)[k] = ac[k] and select top-K
        K_sel = min(top_K, 2 * n - 1)
        # The most binding constraints are where ac[k] is largest
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build T matrix: T[k, i] = y[k-i] if 0 <= k-i < n, else 0
        # For efficiency, only build for selected constraints
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            T_sel[ii, i_start:i_end + 1] = y[k_int - np.arange(i_start, i_end + 1)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel), cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.8f}")

    return best_a, 2.0 * n * best_m


def mirror_descent(a0, time_limit, verbose=True, restart_every=3000, top_k=30, beta_exp=4.0):
    """Mirror descent with soft subgradient, Nesterov acceleration, and restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    c0 = 3.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute soft subgradient
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_exp
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.8f}, stall={stall}")

    return best_a, 2.0 * n * best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        # Quick internal evaluation
        n = len(a_norm)
        ac = fast_autocorr(a_norm)
        J_internal = 2.0 * n * ac.max()
        if np.isfinite(J_internal) and J_internal < best_score:
            # Verify with actual evaluator
            sc = evaluate_safe(a_norm)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_norm.copy()
                elapsed = time.monotonic() - t_global
                print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={n}, t={elapsed:.0f}s)")

    print(f"Total time budget: {time_budget:.0f}s")

    # === Phase 0: Parametric family search ===
    print("\n=== Phase 0: Parametric family search ===")
    n_base = 3200

    # Try multiple lengths
    lengths = [n_base, n_base + 400, n_base + 800, 4000, 4500]

    for n_try in lengths:
        # Linear ramp (key insight: gives J ≈ 4/3)
        a_lin = make_linear_ramp(n_try, power=1.0)
        try_update(a_lin, f"linear_n{n_try}")

        # Power laws
        for p in [0.7, 0.8, 0.9, 1.1, 1.2, 1.5]:
            a_p = make_linear_ramp(n_try, power=p)
            try_update(a_p, f"power{p}_n{n_try}")

        # Two-sided power
        for p1, p2 in [(1.0, 0.5), (1.5, 0.5), (1.0, 0.8), (0.5, 1.0), (2.0, 0.5)]:
            a_poly = make_poly_ramp(n_try, p1=p1, p2=p2)
            try_update(a_poly, f"poly_{p1}_{p2}_n{n_try}")

        # Windowed
        for w in ['cosine', 'hann']:
            a_w = make_windowed_ramp(n_try, window=w)
            try_update(a_w, f"win_{w}_n{n_try}")

        # Sine power
        for sp in [0.5, 1.0, 1.5, 2.0]:
            a_s = make_sine_power(n_try, power=sp)
            try_update(a_s, f"sine{sp}_n{n_try}")
            a_s2 = make_sine_power(n_try, power=sp, shift=0.5)
            try_update(a_s2, f"sine{sp}s0.5_n{n_try}")

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            try_update(p_norm, "parent")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_linear_ramp(n_base)
        best_score = 2.0 * n_base * fast_autocorr(best_seq).max()
        print(f"  Fallback init: J={best_score:.6f}")

    print(f"  Best after parametric search: J={best_score:.8f}, n={len(best_seq)}")

    # === Phase 1: Coordinate descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Coordinate descent (n={len(best_seq)}) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, _ = coordinate_descent(best_seq, t1, verbose=True, rng_seed=42)
        try_update(a_out, f"cd1_n{len(best_seq)}")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30 and n_best <= 4000:
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, verbose=True, top_K=150)
        try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Mirror descent (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = mirror_descent(best_seq, t3, verbose=True, restart_every=5000, top_k=30, beta_exp=4.0)
        try_update(a_out, f"md1_n{len(best_seq)}")

    # === Phase 4: Another coordinate descent round ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord descent round 2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = coordinate_descent(best_seq, t4, verbose=True, rng_seed=999)
        try_update(a_out, f"cd2_n{len(best_seq)}")

    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer = [n_best + 500, n_best + 1000]
    longer = [l for l in longer if l <= 5000]

    for new_n in longer:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"\n=== Phase 5: Scale to n={new_n} ===")

        # Initialize from current best (interpolated)
        a_long_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_long_interp, f"interp_n{new_n}")

        # Also try fresh linear ramp at this length
        a_long_ramp = make_linear_ramp(new_n, power=1.0)
        try_update(a_long_ramp, f"ramp_n{new_n}")

        # Coordinate descent on the better one
        t5 = min(remaining * 0.30, 150.0)
        if t5 > 20:
            a_out, _ = coordinate_descent(best_seq if len(best_seq) == new_n else a_long_ramp,
                                          t5, verbose=False, rng_seed=77)
            try_update(a_out, f"cd_n{new_n}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t6 = min(remaining * 0.30, 200.0)
        if t6 > 20:
            a_out, _ = lp_frank_wolfe(best_seq, t6, solver, verbose=True, top_K=200)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final coord (n={len(best_seq)}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 10:
            a_out, _ = coordinate_descent(best_seq, t7, verbose=False, rng_seed=2024)
            try_update(a_out, f"cd_final_n{len(best_seq)}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = make_linear_ramp(3200)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")

    return [float(x) for x in best_seq]
