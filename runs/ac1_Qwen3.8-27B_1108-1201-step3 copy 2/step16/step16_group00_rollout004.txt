            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_objective(a):
    """Compute 2n * max(conv) / (sum)^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (s * s)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def bottleneck_lags(ac, n_top=10):
    """Return indices of the top n_top largest convolution values."""
    k = min(n_top, len(ac))
    return np.argsort(ac)[::-1][:k]


def get_contributing_pairs(k, n):
    """For lag k in convolution, return (i, j) pairs where i+j=k and 0<=i,j<n."""
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start > i_end:
        return np.array([], dtype=int), np.array([], dtype=int)
    i_idx = np.arange(i_start, i_end + 1)
    j_idx = k - i_idx
    return i_idx, j_idx


def coordinate_descent_bottleneck(a0, time_limit, verbose=True, rng_seed=42, max_stall=500):
    """
    Coordinate descent that specifically targets the bottleneck lag.
    At each step, identify the bottleneck lag and try to reduce it by
    moving mass from contributing positions to non-contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > max_stall:
            break
            
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top bottleneck lags
        n_top = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        # Build "hot" set: positions that contribute significantly to top lags
        hot = np.zeros(n, dtype=float)
        for k in top_lags[:5]:  # Focus on top 5 lags
            k_int = int(k)
            i_idx, j_idx = get_contributing_pairs(k_int, n)
            if len(i_idx) == 0:
                continue
            contrib = a[i_idx] * a[j_idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            # Add weighted contribution to hot score
            hot[i_idx] += contrib / total_contrib
            hot[j_idx] += contrib / total_contrib
        
        # Find cold positions (low hot score)
        cold_threshold = np.percentile(hot, 30)
        cold_positions = np.where(hot <= cold_threshold)[0]
        
        # Try moves from hot to cold
        improved = False
        
        # Sort hot positions by score (most hot first)
        hot_positions = np.argsort(hot)[::-1]
        
        for src in hot_positions[:20]:  # Try top 20 hot positions
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            
            src_val = a[src]
            
            # For each cold destination, try a move
            # Score each destination by how much it would help
            best_dst = -1
            best_frac = 0
            best_new_m = M_curr
            
            for dst in cold_positions[::max(1, len(cold_positions)//50)]:
                if dst == src:
                    continue
                # Try a few fractions
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_dst = dst
                        best_frac = frac
                        break  # Try smaller fractions next
            
            if best_dst >= 0 and best_new_m < M_curr - 1e-15:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
            
            if verbose and stall % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    CD bottleneck: t={elapsed:.0f}s, M={best_m:.8f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CD bottleneck final: t={elapsed:.0f}s, M={best_m:.8f}")
    return best_a, best_m


def subgradient_optimize(a0, time_limit, verbose=True, top_k=30, beta=3.0, c0=3.0, 
                          restart_every=3000, rng_seed=42):
    """
    Subgradient method with Nesterov acceleration.
    Uses a soft subgradient based on top-K convolution peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent step (exponential family / entropic regularization)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            # Diverse restart: different types of perturbations
            pert_type = int(rng.integers(3))
            if pert_type == 0:
                # Gaussian perturbation in linear space
                pert = rng.normal(0, 0.01, n) * best_a
                x_r = best_a + pert
            elif pert_type == 1:
                # Multiplicative perturbation
                pert = rng.normal(0, 0.05, n)
                x_r = best_a * np.exp(pert)
            else:
                # Swap-based perturbation
                x_r = best_a.copy()
                for _ in range(n // 100):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    amt = min(a[i], a[j]) * 0.1
                    x_r[i] -= amt
                    x_r[j] += amt
            
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = min(top_k + 5, 80)
            beta = max(beta - 0.2, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Subgrad restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={top_k}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Subgrad t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200, warm_iters=50):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    At each step, solve an LP to find the best direction, then take a step.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    prev_y = a.copy()
    
    # Precompute T matrix (will be updated each iteration)
    T = np.zeros((2 * n - 1, n))
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update T with current y
        T[:] = 0.0
        for j in range(n):
            T[j:j + n, j] = y
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select most binding constraints
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        
        T_sel = T[binding_idx, :]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        prev_y = y
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def block_redistribute(a0, time_limit, verbose=True, rng_seed=7):
    """
    Block-level mass redistribution.
    Divide the sequence into blocks and try to redistribute mass between blocks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Divide into blocks of size ~n/20
        block_size = max(5, n // 20)
        n_blocks = (n + block_size - 1) // block_size
        
        # Compute "energy" of each block (contribution to top lags)
        top_lags = np.argsort(ac)[::-1][:8]
        block_energy = np.zeros(n_blocks)
        
        for k in top_lags:
            k_int = int(k)
            for b in range(n_blocks):
                start = b * block_size
                end = min(n, (b + 1) * block_size)
                # Compute contribution of this block to lag k
                contrib = 0.0
                for i in range(start, end):
                    j = k_int - i
                    if 0 <= j < n:
                        contrib += a[i] * a[j]
                block_energy[b] += contrib
        
        # Find the most energetic block and the least energetic
        src_block = int(np.argmax(block_energy))
        dst_block = int(np.argmin(block_energy))
        
        if src_block == dst_block:
            stall += 1
            continue
        
        # Try to move mass from src_block to dst_block
        src_start = src_block * block_size
        src_end = min(n, (src_block + 1) * block_size)
        dst_start = dst_block * block_size
        dst_end = min(n, (dst_block + 1) * block_size)
        
        total_src = a[src_start:src_end].sum()
        if total_src < 1e-12:
            stall += 1
            continue
        
        improved = False
        for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = total_src * frac
            if amt < 1e-12:
                continue
            
            a_trial = a.copy()
            # Uniformly reduce src block
            src_count = src_end - src_start
            a_trial[src_start:src_end] -= amt / src_count
            # Uniformly add to dst block
            dst_count = dst_end - dst_start
            a_trial[dst_start:dst_end] += amt / dst_count
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Try random block pair
            b1 = int(rng.integers(n_blocks))
            b2 = int(rng.integers(n_blocks))
            if b1 != b2:
                s1 = b1 * block_size
                e1 = min(n, (b1 + 1) * block_size)
                s2 = b2 * block_size
                e2 = min(n, (b2 + 1) * block_size)
                total1 = a[s1:e1].sum()
                if total1 > 1e-12:
                    frac = rng.uniform(0.002, 0.01)
                    amt = total1 * frac
                    a[s1:e1] -= amt / (e1 - s1)
                    a[s2:e2] += amt / (e2 - s2)
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Block: t={elapsed:.0f}s, M={best_m:.8f}, stall={stall}")
    
    return best_a, best_m
