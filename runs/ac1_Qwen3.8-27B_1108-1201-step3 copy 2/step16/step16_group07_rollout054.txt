"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration and adaptive active-set
2. Exact 1D coordinate descent with analytic critical-point enumeration
3. Peak-flattening via weighted subgradient (KKT-inspired)
4. Multi-scale restarts with structural perturbations
Key innovation: coordinate descent exploits the piecewise-linear structure of
the autocorrelation as a function of a single coordinate to find the exact
1D optimum without ternary search.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    if size == 0:
        return None, None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(a)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            tol = 0.06 * (1.0 - progress) + 0.003 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, n // 60)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > int(2.0 * n):
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-int(2.0 * n):]
                active_set = active_set[keep]

        b, duals = lp_best_response(a, M_curr, active_set, solver)

        if b is None:
            if t % 5 == 0:
                active_set = np.arange(conv_size)
                b, _ = lp_best_response(a, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_a

        score_y = fast_autocorr(a_new).max()
        if score_y < best_score:
            best_score = score_y
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 40:
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_exact(a0, time_limit, verbose=False, seed=0):
    """
    Gauss-Seidel coordinate descent with exact 1D optimization.
    For each coordinate j, the autocorrelation as a function of a[j] is
    piecewise-linear (upper envelope of lines) divided by a quadratic.
    We enumerate critical points (peak minimizers, tie points, boundaries)
    to find the exact 1D optimum without iterative search.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    conv = fast_autocorr(a).copy()
    size = 2 * n - 1
    best_score = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    best_s = s
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 4
    n_passes = 0
    n_evals = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            v = a[j]
            k_lo = j
            k_hi = min(size, j + n)
            seg_len = k_hi - k_lo
            if seg_len == 0:
                continue

            a_seg = a[:seg_len]
            conv_seg = conv[k_lo:k_hi]

            # Outside max
            outside = 0.0
            if k_lo > 0:
                outside = max(outside, conv[:k_lo].max())
            if k_hi < size:
                outside = max(outside, conv[k_hi:].max())

            # Current score
            curr_max = max(conv_seg.max(), outside)
            curr_score = 2.0 * n * curr_max / (s * s)

            # Build candidate deltas
            candidates = [0.0]

            # Top peaks for critical point computation
            n_top = min(12, seg_len)
            top_idx = np.argpartition(conv_seg, -n_top)[-n_top:]

            for idx in top_idx:
                c = conv_seg[idx]
                m = 2.0 * a_seg[idx]
                if m > 1e-15:
                    # Minimize (c + m*d) / (s+d)^2 => d = s - c/m * 1  (derivative zero)
                    # d/dp [(c+mp)/(s+p)^2] = m(s+p)^2 - 2(c+mp)(s+p) = 0
                    # m(s+p) - 2(c+mp) = 0 => ms + mp - 2c - 2mp = 0
                    # ms - 2c - mp = 0 => p = (ms - 2c)/m = s - 2c/m
                    d = s - 2.0 * c / m
                    if d > -v - 1e-12:
                        candidates.append(max(d, -v))

            # Tie points between top peaks
            for i in range(min(6, len(top_idx))):
                for j2 in range(i + 1, min(6, len(top_idx))):
                    ci = conv_seg[top_idx[i]]
                    mi = 2.0 * a_seg[top_idx[i]]
                    cj = conv_seg[top_idx[j2]]
                    mj = 2.0 * a_seg[top_idx[j2]]
                    denom = mj - mi
                    if abs(denom) > 1e-13:
                        d = (cj - ci) / denom
                        if d > -v - 1e-12:
                            candidates.append(max(d, -v))

            # Proportional to current value
            if v > 1e-15:
                for f in [0.3, 0.6, 0.9, -0.3, -0.7, -0.95, 1.5, 2.0]:
                    d = f * v
                    if d > -v - 1e-12:
                        candidates.append(max(d, -v))

            # Evaluate candidates
            best_d = 0.0
            best_local = curr_score
            for d in candidates:
                d = max(d, -v)
                ns = s + d
                if ns < 1e-10:
                    continue
                new_seg = conv_seg + 2.0 * d * a_seg
                new_max = max(new_seg.max(), outside)
                sc = 2.0 * n * new_max / (ns * ns)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_d = d

            if abs(best_d) < 1e-12:
                continue

            # Apply update
            delta = best_d
            a[j] = v + delta
            s += delta
            conv[k_lo:k_hi] += 2.0 * delta * a_seg
            if 2 * j < size:
                conv[2 * j] += delta * delta

            n_evals += 1

            # Verify periodically to prevent drift
            if n_evals % 200 == 0:
                conv = fast_autocorr(a)
                s = a.sum()

            # Check score
            actual_max = conv.max()
            score = 2.0 * n * actual_max / (s * s)

            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                best_s = s
                improved = True
            else:
                # Revert
                a[j] = v
                s -= delta
                conv[k_lo:k_hi] -= 2.0 * delta * a_seg
                if 2 * j < size:
                    conv[2 * j] -= delta * delta

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    # Final verification
    final_conv = fast_autocorr(best_a)
    final_s = best_a.sum()
    final_score = 2.0 * n * final_conv.max() / (final_s * final_s)
    return best_a, final_score


def subgradient_polish(a0, time_limit, verbose=False, seed=42):
    """Weighted subgradient method focusing on the top peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.3 / n ** 0.25
    patience = 0
    max_patience = 300

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        if M_curr < 1e-12:
            break

        # Select top peaks with exponential weighting
        n_peaks = max(5, min(20, n // 50))
        top_indices = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-6.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.015, 1e-10))
        weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            k = int(k)
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            if lo1 < hi1:
                grad[lo1:hi1] += w * a[k - np.arange(lo1, hi1)]
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            if lo2 < hi2:
                grad[lo2:hi2] += w * a[np.arange(lo2, hi2) - k]

        # Project gradient onto the simplex constraint (sum = norm)
        # Direction: reduce where grad is high, increase where low
        support = np.where(a > 0.001 * a.max())[0]
        if len(support) < 2:
            break
        grad_mean = np.mean(grad[support])
        direction = grad - grad_mean
        direction[support] = np.minimum(direction[support], 0.5 * np.abs(grad[support]).max())

        a_trial = a - step * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 2.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-10:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
