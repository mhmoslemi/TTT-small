import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))

def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = norm1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(new_x, old_x, a)
    return norm1(interp)

    a = np.sin(np.pi * x)

    return norm1(a)

    best_m = ac.max()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Lower bound on M (from previous best or from uniform)
    # J >= 2, so M >= 1/n when sum=1
    M_lower = 1.0 / n  # corresponds to J=2
    
    g = 1.0
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            best_j = 2.0 * n * best_m
            stall = 0
            if _GLOBAL_BEST['score'] > best_j:
                _GLOBAL_BEST['score'] = best_j
                _GLOBAL_BEST['seq'] = best_a.copy()
        else:
            stall += 1
        
        # Find top k lag indices
        k_top = min(n_top, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:k_top]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        
        # Soft weights (exponential)
        beta = 3.0
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_top) / k_top
        else:
            w = w / w_sum
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            # grad[j] += a[ki-j] (if valid) + a[j-ki] (if valid)
            for j in range(n):
                v = 0.0
                idx1 = ki - j
                idx2 = j - ki
                if 0 <= idx1 < n:
                    v += a[idx1]
                if 0 <= idx2 < n:
                    v += a[idx2]
                grad[j] = 2.0 * wi * v
        
        # Polyak step size
        step = (M_curr - M_lower) / (np.linalg.norm(grad, 2) ** 2 + 1e-30)
        step = min(step, 1.0)  # Cap the step
        
        a_new = a - g * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = norm1(a_new)
        
        a = a_new
        g /= (1.0 + 0.001)  # Decay step size multiplier
        iter_count += 1
        
        # Restart
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            a = norm1(best_a + pert)
            g = c0
        
        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter={iter_count} ({elapsed:.0f}s): J={best_j:.6f}, M={best_m:.8f}")
    
    return best_a, best_j

Let me optimize the subgradient computation. Instead of the inner loop, I can use array operations:

For a given lag k, the gradient at position j is `a[k-j]` (if 0<=k-j<n) + `a[j-k]` (if 0<=j-k<n).

This is equivalent to: take `a`, reverse it, and shift by k. Plus `a` shifted by k (with wrap-around? No, with zero-padding).

Actually, let me think about this more carefully. For the linear convolution `c_k = sum_i a_i * a_{k-i}` where the sum is over valid i (0<=i<n and 0<=k-i<n):

`dc_k/da_j = a_{k-j} + a_{j-k}` (with the convention that a_i = 0 for i outside [0, n-1])

So for a fixed k, the gradient vector is: `grad_k[j] = a_{k-j} + a_{j-k}` for j=0,...,n-1.

This can be computed as:
- `a_{k-j}` for j=0,...,n-1 is the reverse of `a[0:k+1]` (if k < n) or more generally, it's `a` reversed and shifted.
- `a_{j-k}` for j=0,...,n-1 is `a` shifted right by k (zero-padded).

So `grad_k = reverse_shift(a, k) + shift(a, k)` where:
- `shift(a, k)[j] = a[j-k]` if j>=k else 0 (right shift by k)
- `reverse_shift(a, k)[j] = a[k-j]` if 0<=k-j<n else 0

For the first term: `shift(a, k)` is just `np.roll` with zero-padding, or equivalently `np.concatenate([zeros(k), a[:n-k]])` if k<n, or `zeros(n)` if k>=n.

For the second term: `reverse_shift(a, k)[j] = a[k-j]`. Let i = k-j, so j = k-i. As j goes from 0 to n-1, i goes from k to k-(n-1). We need 0<=i<n, so max(0, k-n+1) <= i <= min(k, n-1).

This is equivalent to: take `a[0:min(k+1,n)]`, reverse it, and place it starting at position max(0, k-n+1)... Actually this is getting confusing. Let me just use a simple vectorized approach:

```python
def compute_gradient_at_lag(a, k):
    n = len(a)
    k = int(k)
    grad = np.zeros(n)
    # a[j-k] for j = k, k+1, ..., n-1 (when k < n)
    if k < n:
        grad[k:] += a[:n-k]
    # a[k-j] for j = 0, ..., min(k, n-1) (when k < n)
    # and for j where 0 <= k-j < n, i.e., k-n+1 <= j <= k
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        j_range = np.arange(j_start, j_end + 1)
        grad[j_range] += a[k - j_range]
    return grad
```

This is O(n) per lag, so O(n * k_top) total. For n=3200 and k_top=20, that's 64000 operations. Should be fast enough with numpy.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:

def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = norm1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(new_x, old_x, a)
    return norm1(interp)

def make_gaussian(n, width=0.3, center=0.5):

def compute_soft_subgradient(a, ac, k_top=20, beta=3.0):
    n = len(a)
    actual_top = min(k_top, 2 * n - 1)
    top_idx = np.argsort(ac)[::-1][:actual_top]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(actual_top) / actual_top
    else:
        w = w / w_sum
    grad = np.zeros(n, dtype=np.float64)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        if ki < n:
            grad[ki:] += wi * a[:n - ki]
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            grad[j_range] += wi * a[ki - j_range]

def pair_swap_optimize(a_init, time_limit, verbose=True, rng_seed=42):
    global _GLOBAL_BEST
    n = len(a_init)
    a = norm1(a_init)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    best_j = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(15, 2 * n - 1)]
        improved = False

        for _ in range(min(200, n)):
            if time.monotonic() - t0 > time_limit:

            src = int(rng.integers(n))
            if a[src] < 1e-8:
                continue

            best_new_m = best_m

            for _t in range(n_tries):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:

                    a_trial /= s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_new_m:

                        best_dst = dst
                        best_frac = frac

            if best_dst >= 0 and best_new_m < best_m - 1e-15:
                a = best_trial
                best_m = best_new_m
                best_j = 2.0 * n * best_m
                best_a = a.copy()
                improved = True
                stall = 0

def coordinate_descent(a_init, time_limit, verbose=True, rng_seed=99):

            else:

                continue
            best_dst = -1

                    if m_trial < best_new_m:
                        best_new_m = m_trial
                        best_dst = dst
                        best_frac = frac
                        best_trial = a_trial.copy()

                a = best_trial
                best_m = best_new_m

def lp_frank_wolfe(a_init, time_limit, verbose=True):
    global _GLOBAL_BEST
    n = len(a_init)
    a = norm1(a_init)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    best_j = 2.0 * n * best_m
    t0 = time.monotonic()

    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS

    y = a.copy()
    x_nest = a.copy()
    prev_y = a.copy()

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y
        ac_y = T_full @ y
        M_curr = ac_y.max()
        if M_curr < 1e-16:
            break
        K_sel = min(100, 2 * n - 1)
        margins = M_curr - ac_y
        binding_idx = np.argsort(margins)[:K_sel]
        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        prev_y = y
        y = y_new
        x_nest = x_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            best_j = 2.0 * n * best_m
            if best_j < _GLOBAL_BEST['score']:
                _GLOBAL_BEST['score'] = best_j
                _GLOBAL_BEST['seq'] = best_a.copy()
        mx = fast_autocorr(x_nest).max()
        if mx < best_m:
            best_m = mx
            best_a = x_nest.copy()
            best_j = 2.0 * n * best_m
            if best_j < _GLOBAL_BEST['score']:
                _GLOBAL_BEST['score'] = best_j
                _GLOBAL_BEST['seq'] = best_a.copy()
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={best_j:.6f}")

    return best_a, best_j

def propose_candidate(budget_s=1800.0):
    global _GLOBAL_BEST
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            _GLOBAL_BEST['score'] = sc
            _GLOBAL_BEST['seq'] = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm1(parent)
            best_score = p_score
            _GLOBAL_BEST['score'] = p_score
            _GLOBAL_BEST['seq'] = best_seq.copy()
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_sine(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Parametric exploration
    print(f"\n=== Phase 1: Parametric exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.10, 150.0)

    if t_phase1 > 30:
        lengths = [n0, max(2000, n0 - 500), min(5000, n0 + 500), min(6000, n0 + 1000)]
        lengths = sorted(set([l for l in lengths if 1000 <= l <= 6000]))
        families = [
            ("gauss_w025", lambda n: make_gaussian(n, width=0.25)),
            ("gauss_w035", lambda n: make_gaussian(n, width=0.35)),
            ("raised_cos", lambda n: make_raised_cosine(n)),
            ("sine", lambda n: make_sine(n)),
            ("u_shape", lambda n: make_u_shape(n, power=2.0)),
            ("u_shape_p3", lambda n: make_u_shape(n, power=3.0)),
            ("bimodal", lambda n: make_bimodal(n, sep=0.25, width=0.12)),
            ("bimodal_wide", lambda n: make_bimodal(n, sep=0.35, width=0.2)),
            ("power1", lambda n: make_power_law(n, p=1.0)),
            ("power2", lambda n: make_power_law(n, p=2.0)),
            ("flat", lambda n: make_flat(n)),
        ]
        per_eval = t_phase1 / (len(lengths) * len(families))
        for ln in lengths:
            for fname, ffunc in families:
                if time.monotonic() - t_global > time_budget - 60:
                    break
                try:
                    a_trial = ffunc(ln)
                    ac = fast_autocorr(a_trial)
                    j_trial = 2.0 * ln * ac.max()
                    if j_trial < best_score:
                        sc = evaluate_safe(a_trial)
                        try_update(a_trial, f"{fname}_n{ln}")
                except Exception:
                    pass

    # Phase 2: Subgradient optimization from best
    print(f"\n=== Phase 2: Subgradient (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        a_in = norm1(best_seq)
        a_out, j_out = subgradient_optimize(a_in, t2, k_top=30, beta=3.0, c0=0.5, restart_every=5000, verbose=True)
        try_update(a_out, f"sgd_n{len(a_out)}")

    # Phase 3: LP Frank-Wolfe
    print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60 and n_best <= 4000:
        a_in = norm1(best_seq)
        a_out, j_out = lp_frank_wolfe(a_in, t3, verbose=True)
        try_update(a_out, f"lp_n{n_best}")

    # Phase 4: Pair swap
    print(f"\n=== Phase 4: Pair swap (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_in = norm1(best_seq)
        a_out, j_out = pair_swap_optimize(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"pswap_n{len(a_out)}")

    # Phase 5: Coordinate descent
    print(f"\n=== Phase 5: Coord descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_in = norm1(best_seq)
        a_out, j_out = coordinate_descent(a_in, t5, verbose=True, rng_seed=123)
        try_update(a_out, f"cd_n{len(a_out)}")

    # Phase 6: Second subgradient pass
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Second SGD (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_in = norm1(best_seq)
        a_out, j_out = subgradient_optimize(a_in, t6, k_top=40, beta=2.5, c0=0.3, restart_every=8000, verbose=True)
        try_update(a_out, f"sgd2_n{len(a_out)}")

    # Phase 7: Length extension
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length extension ===")
    ext_lengths = [n_best + 400, n_best + 800]
    ext_lengths = [l for l in ext_lengths if l <= 5000 and remaining > 100]
    for new_n in ext_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Extending to n={new_n}")
        a_long = interpolate_to_length(best_seq, new_n)
        t7a = min(remaining * 0.35, 150.0)
        if t7a > 20:
            a_out, j_out = subgradient_optimize(a_long, t7a, k_top=30, beta=3.0, c0=0.4, restart_every=4000, verbose=False)
            try_update(a_out, f"sgd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.4, 80.0)
            if t7b > 15:
                a_out, j_out = coordinate_descent(a_long, t7b, verbose=False, rng_seed=555)
                try_update(a_out, f"cd_n{new_n}")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.5, 150.0)
        if t8 > 15:
            a_in = norm1(best_seq)
            a_out, j_out = coordinate_descent(a_in, t8, verbose=True, rng_seed=777)
            try_update(a_out, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_sine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
