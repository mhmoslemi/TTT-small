def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

            if prob.status in ('optimal', 'optimal_inaccurate'):

        direction = lp_direction(a, M, active_lags, solver)
        if direction is None:
            continue
        
        # Line search
        a_new, f_new, alpha = backtracking_line_search(a, direction, f_curr)
        if a_new is None:
            # Try smaller steps
            for alpha in [0.05, 0.03, 0.01]:
                a_trial = (1-alpha)*a + alpha*direction
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                f_trial = eval_obj(a_trial)
                if f_trial < f_curr:
                    a = a_trial
                    break
            else:
                continue
        else:
            a = a_new
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it} ({elapsed:.1f}s): best_f={best_f:.6f}, curr_f={eval_obj(a):.6f}, K={current_top_k}")
    
    return best_a, best_f
