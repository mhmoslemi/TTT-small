def compute_new_conv_max(conv, a, j, x, n):
    """Compute max of new convolution after changing a[j] to x."""
    aj = a[j]
    delta = x - aj
    lo_k = j
    hi_k = min(2*n-2, j + n - 1)
    
    if lo_k > hi_k:
        return conv.max()
    
    # Positions that change
    idx = np.arange(lo_k, hi_k + 1)
    new_vals = conv[lo_k:hi_k+1] + 2.0 * delta * a[idx - j]
    
    # Correction for k=2j
    if 2*j <= hi_k and 2*j >= lo_k:
        # new_conv[2j] should be conv[2j] + x^2 - aj^2, not conv[2j] + 2*delta*aj
        # Current: conv[2j] + 2*delta*aj = conv[2j] + 2*(x-aj)*aj = conv[2j] + 2*x*aj - 2*aj^2
        # Correct: conv[2j] + x^2 - aj^2
        # Difference: x^2 - aj^2 - 2*x*aj + 2*aj^2 = x^2 - 2*x*aj + aj^2 = (x-aj)^2 = delta^2
        new_vals[2*j - lo_k] += delta * delta
    
    # Max of changed positions
    inner_max = new_vals.max()
    
    # Max of unchanged positions
    if lo_k > 0:
        outer_left = conv[:lo_k].max()
    else:
        outer_left = -np.inf
    if hi_k < 2*n-1:
        outer_right = conv[hi_k+1:].max()
    else:
        outer_right = -np.inf
    outer_max = max(outer_left, outer_right)
    
    return max(inner_max, outer_max)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()

        return a, float('inf')
    norm = np.sqrt(2.0 * n)
    a = a * norm / s
    
    best_a = a.copy()

        A = rfft(a, 2*n)  # zero-padded for linear convolution
        magnitude = np.abs(A)

        # Target: |A|^2 = const => |A| = sqrt(const)
        # We want the IDFT of |A|^2 to be flat, which means |A|^2 should be flat
        # But we can't just set |A| to constant because that would give a delta in time domain

        target_mag = np.sqrt(np.sum(magnitude**2) / (n+1))  # for real signal

        s_new = a_new.sum()
        if s_new < 1e-15:

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def golden_section(f, f_args, lo, hi, max_iter=25):
    """Golden section minimization. f takes (x, *f_args)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = f(a_pt, *f_args)
    fb = f(b_pt, *f_args)
    for _ in range(max_iter):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = f(a_pt, *f_args)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = f(b_pt, *f_args)
    return (lo + hi) / 2.0


def exact_coord_descent(a0, time_limit, max_passes=10000, verbose=False, seed=0):
    """Exact coordinate descent with golden section search per coordinate."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve_count = 0
    max_no_improve = 8

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        conv = fast_autocorr(a)
        pass_improved = False
        total_change = 0.0

        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 >= time_limit:
                break

            aj = a[j]
            s_others = norm - aj

            # Compute outer_max (max of conv outside the affected range)
            lo_k = j
            hi_k = min(size - 1, j + n - 1)
            if lo_k == 0 and hi_k == size - 1:
                outer_max = -1e30
            elif lo_k == 0:
                outer_max = conv[hi_k + 1:].max() if hi_k + 1 < size else -1e30
            elif hi_k == size - 1:
                outer_max = conv[:lo_k].max() if lo_k > 0 else -1e30
            else:
                outer_max = max(conv[:lo_k].max(), conv[hi_k + 1:].max())

            # Bounds for x
            x_lo = 0.0
            x_hi = max(2.0 * aj, 4.0 * norm / n) + 0.5 * norm / n

            # Golden section
            f_args = (aj, s_others, norm, conv, a, j, n, outer_max)
            x_opt = golden_section(coord_score_func, f_args, x_lo, x_hi, max_iter=20)

            if abs(x_opt - aj) > 1e-10 * norm:
                old_j = a[j]
                a[j] = x_opt
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    total_change += abs(x_opt - old_j)
                    pass_improved = True
                else:
                    a[j] = old_j

        # Check improvement after pass
        new_score = fast_autocorr(a).max()
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a.copy()
            no_improve_count = 0
        else:
            a = best_a.copy()
            no_improve_count += 1

        if no_improve_count >= max_no_improve:
            break

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, seed=0, verbose=False):
    """Multi-peak subgradient method with adaptive step size."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.8 / np.sqrt(n)
    patience = 0
    max_patience = 300

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit or patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Select peaks
        n_peaks = max(5, min(15, n // 60))
        top_indices = np.argsort(conv)[-n_peaks:]

        # Weighted subgradient
        weights = np.exp(-10.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                idx = np.arange(lo, hi)
                grad[lo:hi] += w * a[k - idx]
                if k >= lo and k < hi:
                    pass  # included above

        # Also add gradient from the constraint sum(a)=norm (project onto simplex)
        grad_proj = grad - np.mean(grad)

        a_trial = a - step * grad_proj
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            patience += 1
            step *= 0.95
            continue
        a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-13:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 3.0 / np.sqrt(n))
        else:
            patience += 1
            step *= 0.98
            if step < 1e-12:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.8f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)

            + 0.05 * np.cos(6 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    print(f"Starting: n={n_base}, score={best_score:.8f}")

    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)

    # === Phase 2: Exact coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Exact CD n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = exact_coord_descent(a_norm, t2, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd1_n{n_cur}", t_global)

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Subgradient n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)

    remaining = time_budget - (time.monotonic() - t_global)

            perturbed = perturbed / perturbed.sum()

            a_out, _ = nesterov_lp_optimize(perturbed, t7, seed=555, verbose=True)

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        t9 = min(remaining * 0.5, 80.0)
        a_norm = normalize_to(best_seq, n_cur)

        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_final_n{n_cur}", t_global)

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):

        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
