- Convex problem structure: min max_k a^T Q_k a s.t. a in simplex
"""

import numpy as np
import time
import math

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    import cvxpy as cp
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.2])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_frank_wolfe(a0, time_limit, solver_name, verbose=True, max_iters=50000):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    Minimizes max(conv(a,a)) over the simplex.
    """
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    y_prev = a.copy()
    iter_count = 0
    lp_time_total = 0.0

    for t in range(1, max_iters +  1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-15:
            break

        # Build the LP: max sum(b) s.t. conv(y, b)_k <= M_curr for all k, b >= 0
        # conv(y,b)[k] = sum_j y[j]*b[k-j] for valid j
        # As a matrix: C[k,j] = y[k-j] if 0<=k-j<n else 0
        # But building full C is expensive. Use the structure.
        # C is (2n-1) x n, C[k,j] = y[k-j] for max(0,k-n+1) <= j <= min(n-1,k)
        
        b_var = cp.Variable(n, nonneg=True)
        
        # Build constraints efficiently using cumulative approach
        # conv(y, b)[k] = sum_{j} y[k-j] * b[j] where 0<=j<n and 0<=k-j<n
        # = sum_{j=max(0,k-n+1)}^{min(n-1,k)} y[k-j] * b[j]
        
        # For the LP, we need all 2n-1 constraints.
        # Build C row by row (sparse)
        import scipy.sparse as sp_holder
        rows_data = []
        cols_data = []
        vals_data = []
        
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            ks = np.full_like(js, k)
            vs = y[js]  # y[k - j] for each j
            # But wait: C[k,j] = y[k-j], so for row k, col j, value is y[k-j]
            # js are the column indices, and y[k-j] for j in js
            vs = y[k - js]
            rows_data.append(ks)
            cols_data.append(js)
            vals_data.append(vs)
        
        if rows_data:
            all_rows = np.concatenate(rows_data)
            all_cols = np.concatenate(cols_data)
            all_vals = np.concatenate(vals_data)
        else:
            all_rows = np.array([], dtype=int)
            all_cols = np.array([], dtype=int)
            all_vals = np.array([])
        
        C = sp_holder.csr_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n))
        
        constraints = [C @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        lp_start = time.monotonic()
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        lp_time_total += time.monotonic() - lp_start
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y_prev = y.copy()
        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if t % 5 == 0:
            mx = fast_autocorr(x).max()
            if mx < best_m:
                best_m = mx
                best_a = x.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            avg_lp = lp_time_total / iter_count if iter_count > 0 else 0
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, lp_avg={avg_lp:.3f}s")

    return best_a, best_m


def soft_subgradient(a0, time_limit, top_k=30, beta_start=3.0, beta_end=20.0, verbose=False, rng_seed=42):
    """
    Nesterov-accelerated mirror descent with soft subgradient on top-k peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0
    stall_count = 0
    max_stall = 2000
    best_beta = beta_start

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        
        # Adaptive beta
        current_beta = beta_start + (beta_end - beta_start) * min(1.0, t / 5000.0)
        w = top_vals ** current_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            noise_scale = 0.05 * (best_m - 1.0 / n) / best_m + 0.01
            perturbation = rng.normal(0, noise_scale, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SoftSGD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """
    Goal-directed coordinate descent: iteratively reduce the current peak
    by moving mass from high-contributing positions to low-impact positions.
    Uses exact line search along each coordinate swap.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 100
    improvement_threshold = 1e-10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = np.argmax(ac)
        
        # Find all positions above 95% of max (they all need to come down)
        threshold = 0.95 * M_curr
        active_peaks = np.where(ac >= threshold)[0]
        
        # For each active peak, compute contribution of each position
        contrib = np.zeros(n)
        for pk in active_peaks[:10]:  # Focus on top peaks
            pk_int = int(pk)
            j_start = max(0, pk_int - n + 1)
            j_end = min(n - 1, pk_int)
            js = np.arange(j_start, j_end + 1)
            contrib[js] += a[pk_int - js]
        
        # Source: positions with highest contribution to peaks and highest value
        source_score = contrib * a
        # Destination: positions with low contribution to peaks
        dest_score = -contrib
        
        # Try moving mass from top sources to best destinations
        n_sources = min(10, n // 10)
        n_dests = min(10, n // 10)
        
        src_candidates = np.argsort(source_score)[::-1][:n_sources]
        dst_candidates = np.argsort(dest_score)[::-1][:n_dests]
        
        improved = False
        
        for src in src_candidates:
            if improved:
                break
            if a[src] < 1e-12:
                continue
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                
                # Try several fractions
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - improvement_threshold:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if not improved and a[src] > 1e-10:
                # Try a random destination
                dst = int(rng.integers(n))
                if dst != src:
                    frac = 0.02
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - improvement_threshold:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0

        if not improved:
            stall += 1
            # Perturb to escape
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a > 1e-15:
                        a = a / s_a
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPolish ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Phase 1: Quick warmup with soft subgradient
    Phase 2: LP-FW main optimization (bulk of time)
    Phase 3: Coordinate polish
    Phase 4: LP-FW second pass
    Phase 5: Multi-n exploration
    Phase 6: Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        best_seq = None

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 1e-10)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Quick soft subgradient warmup (60s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 1: Soft SGD warmup n={n0} ===")
        t1 = min(remaining * 0.04, 60.0)
        a_out, _ = soft_subgradient(norm_sum1(best_seq), t1, top_k=30, beta_start=3.0, beta_end=15.0, verbose=False, rng_seed=42)
        try_update(a_out, "warmup")

    # === Phase 2: LP-FW main (largest chunk, ~45% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW main n={n_cur} ===")
        t2 = min(remaining * 0.42, 700.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, solver, verbose=True)
        try_update(a_out, f"lp_main_n{n_cur}")

    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord polish n={n_cur} ===")
        t3 = min(remaining * 0.08, 100.0)
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 4: LP-FW round 2 (from polished starting point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.25, 400.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t4, solver, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 5: Soft SGD exploration with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: Soft SGD explore n={n_cur} ===")
        t5 = min(remaining * 0.10, 120.0)
        a_out, _ = soft_subgradient(norm_sum1(best_seq), t5, top_k=50, beta_start=5.0, beta_end=30.0, verbose=False, rng_seed=99)
        try_update(a_out, "sgd_explore")

    # === Phase 6: Multi-n exploration ===
    remaining = time_budget - (time.monotonic()
