def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective(a):
    """Compute J = 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def subgradient_at_peak(a, k):
    """Subgradient of c_k = sum_j a_j * a_{k-j} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    batch_size = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()

        # Find top peaks
        n_peaks = min(8, len(ac))
        peak_indices = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity: for each position i, how much does it contribute to the max peak
        sensitivity = np.zeros(n)
        for k in peak_indices[:3]:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                j = ki - i
                if 0 <= j < n:
                    sensitivity[i] += ac[ki] * a[j]

        # Source candidates: high sensitivity
        src_candidates = np.argsort(sensitivity)[::-1][:min(50, n)]

        # Destination candidates: low sensitivity + edge bonus
        dst_score = sensitivity.copy()
        # Edge positions contribute to fewer peaks
        edge = np.zeros(n)
        edge_width = max(1, n // 20)
        edge[:edge_width] = -1.0
        edge[-edge_width:] = -1.0
        dst_score += edge
        dst_candidates = np.argsort(dst_score)[:min(50, n)]

        # Try moves
        improved = False
        moves_made = 0

        for src in src_candidates[:20]:
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates[:20]:
                if dst == src:
                    continue
                # Quick estimate of improvement
                # c_k changes by epsilon * (a_{k-dst} - a_{k-src})
                # We want a_{k-src} > a_{k-dst} for the peak k
                k_peak = int(peak_indices[0])
                a_src_at_peak = a[src] if k_peak - (k_peak - src) < n else 0
                # Simplified: just try the move
                for frac in [0.003, 0.01, 0.03, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a_trial.copy()
                        improved = True
                        moves_made += 1
                        if moves_made >= batch_size:
                            break
                if moves_made >= batch_size:
                    break
            if moves_made >= batch_size:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random exploration
            if stall > 100:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD done ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def spectral_polish(a0, time_limit, n_iter=500, seed=42):
    """
    Iterative time/frequency domain projection:
    - In frequency: try to flatten |A|^2
    - In time: enforce non-negativity
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-20:
            break

        # Soft flattening: compress dynamic range of P
        alpha = 0.3 + 0.4 * min(it / n_iter, 1.0)
        P_flat = P ** (1.0 - alpha)
        P_flat = P_flat / P_flat.max()

        # Reconstruct with flattened magnitude, keep original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_flat) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Enforce non-negativity (soft)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_templates(n, rng, parent=None):
    """Generate diverse structural templates."""
    templates = []
    x = np.linspace(0, 1, n)

    # Uniform
    templates.append(np.ones(n) / n)

    # Sine
    t = np.sin(np.pi * x) ** 2
    templates.append(t / t.sum())

    # Sine with power
    for p in [0.5, 4.0, 8.0]:
        t = np.sin(np.pi * x) ** p
        templates.append(t / t.sum())

    # Bimodal
    t = np.exp(-((x - 0.2) / 0.15) ** 2) + np.exp(-((x - 0.8) / 0.15) ** 2)
    templates.append(t / t.sum())

    # V-shape (edges)
    t = np.abs(2 * x - 1) ** 2 + 0.1
    templates.append(t / t.sum())

    # Inverse V (center)
    t = 1.0 - np.abs(2 * x - 1) ** 2
    t = np.maximum(t, 0.05)
    templates.append(t / t.sum())

    # Logistic
    t = 1.0 / (1.0 + np.exp(-20 * (x - 0.5)))
    templates.append(t / t.sum())

    # Periodic
    for period in [n // 7, n // 11, n // 16]:
        t = 1.0 + 0.4 * np.cos(2 * np.pi * x * n / period)
        t = np.maximum(t, 0.05)
        templates.append(t / t.sum())

    # Random smooth
    for _ in range(3):
        t = rng.random(n)
        ks = max(3, n // 50)
        kernel = np.ones(ks) / ks
        t = np.convolve(t, kernel, mode='same')
        t = np.maximum(t, 0.01)
        templates.append(t / t.sum())

    # Multi-harmonic
    freqs = rng.integers(2, 30, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    phs = rng.uniform(0, 2 * np.pi, size=4)
    t = np.ones(n)
    for f, a_, p in zip(freqs, amps, phs):
        t += a_ * np.cos(2 * np.pi * f * x + p)
    t = np.maximum(t, 0.01)
    templates.append(t / t.sum())

    if parent is not None:
        templates.append(norm_sum1(parent))

    return templates


def propose_candidate(budget_s=1800.0):
    """
    Main orchestrator with adaptive time allocation and multi-phase search.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    rng = np.random.default_rng(42)

    # === Phase 0: Template evaluation (quick) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 0: Templates n={n0}, budget={remaining:.0f}s ===")
    if remaining > 300:
        templates = generate_templates(n0, rng, parent=best_seq)
        best_tmpl = best_seq.copy()
        best_tmpl_obj = objective_val(norm_sum1(best_seq))
        for idx, tmpl in enumerate(templates):
            obj = objective_val(norm_sum1(tmpl))
            if obj < best_tmpl_obj:
                best_tmpl_obj = obj
                best_tmpl = norm_sum1(tmpl)
                print(f"  Better template {idx}: obj={obj:.6f}")
        try_update(best_tmpl, "template")

    # === Phase 1: Main Nesterov MD (workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: NMD main n={n0}, remaining={remaining:.0f}s ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_out, _ = nesterov_md(best_seq, t1, c0=5.0, top_k=3, power=2.5, seed=42, verbose=True)
        try_update(a_out, "nmd_main")

    # === Phase 2: NMD with FW mix ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: NMD+FW n={len(best_seq)}, remaining={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = nesterov_md(best_seq, t2, c0=4.0, top_k=5, power=2.0,
                               seed=123, verbose=False, use_fw=True, fw_every=10)
        try_update(a_out, "nmd_fw")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: CD v2 n={len(best_seq)}, remaining={remaining:.0f}s ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = coordinate_descent_v2(best_seq, t3, seed=42, verbose=True)
        try_update(a_out, "cd_v2")

    # === Phase 4: Spectral polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Spectral n={len(best_seq)}, remaining={remaining:.0f}s ===")
    t4 = min(remaining * 0.10, 100.0)
    if t4 > 20:
        a_out, _ = spectral_polish(best_seq, t4, n_iter=300, seed=99)
        try_update(a_out, "spectral")

    # === Phase 5: NMD restart with different hyperparams ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: NMD restart n={len(best_seq)}, remaining={remaining:.0f}s ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_out, _ = nesterov_md(best_seq, t5, c0=8.0, top_k=1, power=1.5, seed=777, verbose=True)
        try_update(a_out, "nmd_restart")

    # === Phase 6: Another CD pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: CD pass 2 n={len(best_seq)}, remaining={remaining:.0f}s ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 20:
        a_out, _ = coordinate_descent_v2(best_seq, t6, seed=314, verbose=True)
        try_update(a_out, "cd_pass2")

    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length exploration, remaining={remaining:.0f}s ===")
    if remaining > 300:
        for delta in [-200, 200, -400, 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 150:
                break
            n_try = len(best_seq) + delta
            if n_try < 1000 or n_try > 5000:
                continue
            print(f"  Trying n={n_try}")
            a_interp = interpolate_sequence(best_seq, n_try)
            t7 = min(remaining * 0.3, 200.0)
            a_out, _ = nesterov_md(a_interp, t7, c0=5.0, top_k=3, power=2.0,
                                   seed=1000 + n_try, verbose=False)
            try_update(a_out, f"nmd_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 80:
                t7b = min(remaining * 0.15, 80.0)
                a_out2, _ = coordinate_descent_v2(a_out, t7b, seed=2000 + n_try, verbose=False)
                try_update(a_out2, f"cd_n{n_try}")

    # === Phase 8: Final NMD + CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish, remaining={remaining:.0f}s ===")
    if remaining > 60:
        t8a = min(remaining * 0.5, 150.0)
        a_out, _ = nesterov_md(best_seq, t8a, c0=3.0, top_k=2, power=3.0, seed=5555, verbose=False)
        try_update(a_out, "final_nmd")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.6, 80.0)
            a_out2, _ = coordinate_descent_v2(best_seq, t8b, seed=8888, verbose=True)
            try_update(a_out2, "final_cd")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
