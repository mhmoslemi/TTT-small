def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def mirror_descent_topk(a0, time_limit, K=80, beta=4.0, verbose=True):
    """Nesterov-accelerated mirror descent targeting top-K peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = 6.0
    stall_count = 0
    max_stall = 2000

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(K, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * (y[j_idx] + y[idx])

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.02, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coord_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Targeted coordinate descent for fine polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = int(np.argmax(ac))

        # Find contributing pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]

        if len(idx) == 0 or contrib.max() < 1e-20:
            stall += 1
            break

        # Compute sensitivity for all positions (contribution to top peaks)
        top_peaks = np.argsort(ac)[::-1][:15]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            ps = max(0, pk_int - n + 1)
            pe = min(n - 1, pk_int)
            for pos in range(ps, pe + 1):
                d = pk_int - pos
                if 0 <= d < n:
                    sensitivity[pos] += a[d]

        improved = False
        src_ranking = np.argsort(contrib)[::-1]

        for src_rank in src_ranking[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])

            # Find best destination
            candidates = np.argsort(sensitivity)[:min(80, n // 5)]
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                if amt < 1e-18:
                    continue
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPD ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate():
    """Main search orchestrator with active-set LP-FW as primary method."""
    t_global = time.monotonic()
    time_budget = 1740.0  # Leave 60s margin

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: Mirror descent warmup (20s)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 1: MD warmup n={n0} ===")
        t1 = min(remaining * 0.03, 20.0)
        a_out, _ = mirror_descent_topk(norm_sum1(best_seq), t1, K=60, beta=4.0, verbose=False)
        try_update(a_out, "md_warmup")

    # Phase 2: Active-set LP-FW (main, ~45% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: Active-set LP-FW n={n_cur} (main) ===")
        t2 = min(remaining * 0.45, 700.0)
        a_out, _ = active_set_lp_fw(norm_sum1(best_seq), t2, solver, verbose=True,
                                     K_start=15, K_end=80, K_ramp=200)
        try_update(a_out, f"lp_main_n{n_cur}")

    # Phase 3: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord polish n={n_cur} ===")
        t3 = min(remaining * 0.08, 90.0)
        a_out, _ = coord_descent_polish(norm_sum1(best_seq), t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # Phase 4: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.25, 400.0)
        a_out, _ = active_set_lp_fw(norm_sum1(best_seq), t4, solver, verbose=True,
                                     K_start=30, K_end=120, K_ramp=100)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # Phase 5: Try slightly larger n
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_target = n_cur + 200
    if remaining > 100:
        print(f"\n=== Phase 5: Expand to n={n_target} ===")
        a_long = interpolate_sequence(best_seq, n_target)
        t5a = min(remaining * 0.1, 50.0)
        a_out, _ = mirror_descent_topk(norm_sum1(a_long), t5a, K=60, beta=4.0, verbose=False)
        try_update(a_out, f"md_n{n_target}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.35, 300.0)
            a_out, _ = active_set_lp_fw(norm_sum1(norm_sum1(best_seq) if len(best_seq) == n_cur else a_long),
                                         t5b, solver, verbose=True, K_start=20, K_end=80, K_ramp=150)
            try_update(a_out, f"lp_n{n_target}")

    # Phase 6: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 6: Multi-n exploration ===")
        n_best = len(best_seq)
        test_ns = []
        for delta in [100, -100, 200, -200]:
            nt = n_best + delta
            if 800 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.3, 150.0)
            a_out, _ = active_set_lp_fw(norm_sum1(a_inter), t_per, solver, verbose=False,
                                         K_start=20, K_end=60, K_ramp=100)
            try_update(a_out, f"multin_n{nt}")

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final coord polish n={n_best} ===")
        t7 = min(remaining * 0.3, 120.0)
        a_out, _ = coord_descent_polish(norm_sum1(best_seq), t7, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # Phase 8: Final LP if time remaining
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final LP n={n_best} ===")
        t8 = min(remaining * 0.7, 200.0)
        a_out, _ = active_set_lp_fw(norm_sum1(best_seq), t8, solver, verbose=True,
                                     K_start=40, K_end=150, K_ramp=50)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
