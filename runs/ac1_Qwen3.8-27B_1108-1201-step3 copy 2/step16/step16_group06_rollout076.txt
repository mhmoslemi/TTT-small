def cross_corr_w_y(w,y):
    n=len(y)
    wf=np.asarray(w,dtype=np.float64)

    W=np.fft.rfft(wf,fs)
    Y=np.fft.rfft(flipped,fs)
    conv=np.fft.irfft(W*Y, fs)[:size]
    return conv[n-1:n-1+n].copy()

def fast_conv(a):
    a=np.asarray(a, dtype=np.float64)
    n=len(a)
    if n==0: return np.array([0.0], dtype=np.float64)
    size=2*n-1
    fs=1 << (size-1).bit_length()
    A=np.fft.rfft(a, fs)
    C=np.fft.irfft(A*A, fs)
    return C[:size].astype(np.float64, copy=False)

def keep_top(lst, score, a, k):
    if not np.isfinite(score): return lst
    if len(lst) < k:
       lst.append((float(score), np.asarray(a, dtype=np.float64).copy()))
       return lst
    worst=0; worst_s=-1
    for i in range(len(lst)):
       if lst[i][0] > worst_s:
          worst_s=lst[i][0]; worst=i
    if score < worst_s:
       lst[worst]=(float(score), np.asarray(a, dtype=np.float64).copy())
    return lst

"""

Algorithm:
1. Use the provided parent sequence if valid.

4. Optionally explore a nearby length by interpolation and quick polishing.
5. Final safety check and return the best non-negative sequence.

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    C = np.fft.irfft(A * A, fs)
    return np.asarray(C[:size], dtype=np.float64)


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def internal_score(a):

        if not np.isfinite(M):

    except Exception:

def keep_top(lst, score, a, k):
    if not np.isfinite(score):
        return lst
    arr = np.asarray(a, dtype=np.float64).copy()
    if len(lst) < k:

            worst = i

    return lst


def make_beta(n, alpha, beta):
    x = (np.arange(n) + 0.5) / float(n)
    x = np.clip(x, 1e-8, 1.0 - 1e-8)
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    shape = np.where(np.isfinite(shape), shape, 0.0)
    return normalize(shape)


def make_ripple(n, mode, amp, phase, power):
    t = np.arange(n) / float(n)
    base = 1.0 + amp * np.cos(2.0 * np.pi * float(mode) * t + phase)
    base = np.maximum(base, 0.0)
    if power == 1.0:
        f = base
    else:
        f = np.power(base, power)
    f = np.where(np.isfinite(f), f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s < 1e-30:
        return np.ones(n, dtype=np.float64) / float(n)
    return f / s


def make_random_mix(n, rng):
    x = (np.arange(n) + 0.5) / float(n)
    x = np.clip(x, 1e-8, 1.0 - 1e-8)
    alpha = float(rng.uniform(0.7, 3.5))
    beta = float(rng.uniform(0.7, 3.5))
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)

    m = int(rng.integers(1, 4))
    modes = rng.integers(1, 4, size=m)
    amps = rng.uniform(0.3, 1.8, size=m)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=m)
    drift = float(rng.uniform(-1.0, 1.0))

    t = np.arange(n) / float(n)
    z = drift * t
    for i in range(m):
        z += amps[i] * np.cos(2.0 * np.pi * float(modes[i]) * t + phases[i])

    f = shape * np.exp(z)
    f = np.where(np.isfinite(f), f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s < 1e-30:
        return make_beta(n, 1.0, 1.0)
    return f / s


def cross_corr_w_y(w, y):
    n = len(y)
    wf = np.asarray(w, dtype=np.float64)
    yf = np.asarray(y, dtype=np.float64)[::-1]
    size = len(wf) + len(yf) - 1
    fs = 1 << (size - 1).bit_length()
    W = np.fft.rfft(wf, fs)
    Y = np.fft.rfft(yf, fs)
    conv = np.fft.irfft(W * Y, fs)
    conv = conv[:size]
    return np.asarray(conv[n - 1:n - 1 + n], dtype=np.float64)


def gather(arr, idx):

    out = np.zeros(idx.size, dtype=np.float64)

    out[m] = arr[idx[m]]
    return out

def build_peak_matrix(y, peaks):
    n = len(y)
    p = len(peaks)
    A = np.zeros((p, n), dtype=np.float64)
    for r in range(p):
        k = int(peaks[r])
        i0 = k - n + 1
        if i0 < 0:
            i0 = 0
        i1 = k + 1
        if i1 > n:
            i1 = n
        if i0 < i1:
            idx = np.arange(i0, i1)
            A[r, idx] = y[k - idx]
    return A

    if _linprog is not None:
        for relax in (1.0, 1.1, 1.3):
            try:
                b_ub = np.full(p, float(M) * relax, dtype=np.float64)
                bounds = [(0.0, float(B) * relax)] * n
                res = _linprog(
                    c=-np.ones(n, dtype=np.float64),
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs"
                )
                if res is not None and res.success and res.x is not None:
                    val = np.asarray(res.x, dtype=np.float64)
                    val = np.maximum(val, 0.0)
                    s = float(np.sum(val))
                    if np.isfinite(s) and s > 1e-12:
                        return val / s
            except Exception:
                pass

    # CVXPY fallback.
    solver = get_lp_solver()
    if solver is None:
        return None

    for relax in (1.0, 1.1, 1.3):
        try:
            b = cp.Variable(n, nonneg=True)
            constraints = [A @ b <= float(M) * relax, b <= float(B) * relax]
            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate") and b.value is not None:
                val = np.asarray(b.value, dtype=np.float64)
                val = np.maximum(val, 0.0)
                s = float(np.sum(val))
                if np.isfinite(s) and s > 1e-12:
                    return val / s
        except Exception:
            pass

    return None


def lp_frank_wolfe(a0, t_end, verbose=False, P0=None):
    n = len(a0)
    a = normalize(a0)
    best = a.copy()
    bestM = float(np.max(fast_conv(a)))

    if P0 is None:
        P0 = min(400, max(160, int(n * 0.1)))
    P_loc = P0

    y = best.copy()
    x = best.copy()
    it = 0
    mom_it = 1
    stall = 0
    fail_count = 0
    rng = np.random.default_rng(9042)
    t0 = time.monotonic()

    while time.monotonic() < t_end:
        it += 1
        cy = fast_conv(y)
        M = float(np.max(cy))
        if not np.isfinite(M):
            break

        improved = False
        if M < bestM - 1e-15:
            bestM = M
            best = y.copy()
            improved = True
            stall = 0

        P_use = min(P_loc, cy.size)
        if P_use < 1:
            P_use = 1
        peaks = np.argpartition(cy, -P_use)[-P_use:]

        ymax = float(np.max(y))
        if not np.isfinite(ymax) or ymax <= 0.0:
            break
        B = M / (ymax + 1e-15)

        d = solve_lp_direction(y, M, peaks, B)
        if d is None:
            fail_count += 1
            stall += 5
            if fail_count > 80:
                break
            continue
        fail_count = 0

        d = np.asarray(d, dtype=np.float64)
        d = np.maximum(d, 0.0)
        ds = float(np.sum(d))
        if not np.isfinite(ds) or ds < 1e-15:
            d = y.copy()
        else:
            d = d / ds

        alpha = 2.0 / (float(it) + 2.0)
        y1 = normalize((1.0 - alpha) * y + alpha * d)
        M_y1 = float(np.max(fast_conv(y1)))

        if mom_it == 1:
            x1 = y1.copy()
            M_x1 = M_y1
        else:
            beta = (float(mom_it) - 1.0) / (float(mom_it) + 2.0)
            x1 = normalize(y1 + beta * (y1 - y))
            M_x1 = float(np.max(fast_conv(x1)))
            if not np.isfinite(M_x1):
                x1 = y1.copy()
                M_x1 = M_y1

        if np.isfinite(M_y1) and M_y1 < bestM - 1e-15:
            bestM = M_y1
            best = y1.copy()
            improved = True
        if np.isfinite(M_x1) and M_x1 < bestM - 1e-15:
            bestM = M_x1
            best = x1.copy()
            improved = True

        y_new = y1
        x_new = x1
        M_new = M_y1
        new_mom = mom_it + 1

        if np.isfinite(M_x1) and M_x1 <= M_y1:
            y_new = x1
            x_new = x1
            M_new = M_x1
            new_mom = 1

        if 2.0 * alpha < 1.0:
            z = normalize((1.0 - 2.0 * alpha) * y + 2.0 * alpha * d)
            Mz = float(np.max(fast_conv(z)))
            if np.isfinite(Mz):
                if Mz < bestM - 1e-15:
                    bestM = Mz
                    best = z.copy()
                    improved = True
                if Mz < M_new:
                    y_new = z
                    x_new = z
                    M_new = Mz
                    new_mom = 1

        y = y_new
        x = x_new
        mom_it = new_mom

        if improved:
            stall = 0
        else:
            stall += 1

        if stall > 500:
            P_loc = min(700, P_loc + 120)
            stall = 200

        if stall > 2500:
            noise = rng.normal(0.0, 1.0, best.size) * best * 0.02
            y = normalize(best + noise)
            x = y.copy()
            it = 1
            mom_it = 1
            P_loc = P0
            stall = 0
            fail_count = 0

        if verbose and it % 50 == 0:
            print("      lp it=%d t=%.1fs J=%.6f M=%.8f P=%d" % (
                it, time.monotonic() - t0, 2.0 * n * bestM, bestM, P_loc))

    return best, float(bestM)


def coordinate_polish(a0, t_end, verbose=False, seed=123, P=32):
    n = len(a0)
    a = normalize(a0)
    best = a.copy()
    bestM = float(np.max(fast_conv(a)))
    rng = np.random.default_rng(seed)
    frac_list = (0.002, 0.005, 0.01, 0.02, 0.05)
    stall = 0
    rounds = 0
    t0 = time.monotonic()

    while time.monotonic() < t_end:
        rounds += 1
        c = fast_conv(a)
        Mcur = float(np.max(c))
        if not np.isfinite(Mcur):
            break

        if Mcur < bestM - 1e-15:
            bestM = Mcur
            best = a.copy()
            stall = 0

        if P > c.size:
            P_use = c.size
        else:
            P_use = P
        if P_use < 1:
            P_use = 1
        peaks = np.argpartition(c, -P_use)[-P_use:]

        vals = np.maximum(c[peaks], 1e-30)
        w = np.power(vals, 4.0)
        sw = float(np.sum(w))
        if not np.isfinite(sw) or sw < 1e-30:
            w = np.full(peaks.size, 1.0 / float(peaks.size), dtype=np.float64)
        else:
            w = w / sw

        wfull = np.zeros(2 * n - 1, dtype=np.float64)
        wfull[peaks] = w
        G = 2.0 * cross_corr_w_y(wfull, a)
        G = np.where(np.isfinite(G), G, 0.0)

        srcs = np.argsort(G)[-12:]
        dsts = np.argsort(G)[:32]

        improved = False
        for s_idx in srcs:
            if improved:
                break
            src = int(s_idx)
            base = float(a[src])
            if base <= 1e-12:
                continue
            for d_idx in dsts:
                if improved:
                    break
                dst = int(d_idx)
                if dst == src:
                    continue

                lin = gather(a, peaks - dst) - gather(a, peaks - src)
                quad = np.zeros(peaks.size, dtype=np.float64)
                quad = quad + ((2 * dst) == peaks).astype(float)
                quad = quad - 2.0 * ((src + dst) == peaks).astype(float)
                quad = quad + ((2 * src) == peaks).astype(float)

                for frac in frac_list:
                    delta = base * frac
                    if delta <= 0.0:
                        continue
                    est_vals = c[peaks] + 2.0 * delta * lin + (delta * delta) * quad
                    if float(np.max(est_vals)) < bestM - 1e-9:
                        a2 = a.copy()
                        a2[src] -= delta
                        a2[dst] += delta
                        a2 = np.maximum(a2, 0.0)
                        s2 = float(np.sum(a2))
                        if s2 < 1e-15:
                            continue
                        a2 = a2 / s2
                        M2 = float(np.max(fast_conv(a2)))
                        if np.isfinite(M2) and M2 < bestM - 1e-12:
                            a = a2
                            bestM = M2
                            best = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall <= 3000:
                src = int(rng.integers(n))
                dst = int(rng.integers(n))
                frac = float(rng.uniform(0.002, 0.01))
                if src != dst and float(a[src]) > 1e-12:
                    delta = float(a[src]) * frac
                    a2 = a.copy()
                    a2[src] -= delta
                    a2[dst] += delta
                    a2 = np.maximum(a2, 0.0)
                    s2 = float(np.sum(a2))
                    if s2 > 1e-15:
                        a2 = a2 / s2
                        M2 = float(np.max(fast_conv(a2)))
                        if np.isfinite(M2) and M2 < bestM - 1e-12:
                            a = a2
                            bestM = M2
                            best = a.copy()
                            stall = 0

            if stall > 800:
                noise = rng.normal(0.0, 1.0, best.size) * best * 0.02
                a = normalize(best + noise)
                stall = 0

            if stall > 3500:
                break

        if verbose and rounds % 100 == 0:
            print("      coord round=%d t=%.1fs J=%.6f" % (rounds, time.monotonic() - t0, 2.0 * n * bestM))

    return best, float(bestM)


def interpolate_sequence(f, new_n):
    f = normalize(f)
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    if len(f) == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, len(f))
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    return normalize(g)


def propose_candidate():
