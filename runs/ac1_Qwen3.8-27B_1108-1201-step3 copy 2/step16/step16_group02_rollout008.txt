"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov accelerated mirror descent with adaptive subgradient at active lags
2. Backtracking step size with periodic restarts and asymmetric perturbations
3. Softmax-smoothed subgradient for better direction
4. Multi-length exploration and coordinate polish
Key insight: The linear autocorrelation structure means asymmetric sequences
can achieve lower peak autocorrelation than symmetric ones.
"""

import numpy as np
import time

_global_best = None
_global_best_score = float('inf')

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    """Normalize sequence to have sum 1 and non-negative entries."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def subgradient_topk(a, b, top_k=5, weights=None):
    """Compute subgradient at top-k autocorrelation lags."""
    n = len(a)
    idx = np.argsort(b)[::-1][:top_k]
    if weights is None:
        w = np.ones(top_k) / top_k
    else:
        w = weights
    g = np.zeros(n)
    for wk, k in zip(w, idx):
        if wk < 1e-14:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        a_slice = a[k - j_hi : k - j_lo + 1][::-1]
        g[j_lo : j_hi + 1] += 2.0 * wk * a_slice
    return g

def softmax_subgradient(a, b, temperature, top_k=None):
    """Compute smoothed subgradient using softmax over lags."""
    n = len(a)
    b_work = b.copy()
    if top_k is not None:
        idx = np.argsort(b)[::-1][:top_k]
        b_masked = np.zeros_like(b)
        b_masked[idx] = b[idx]
        b_work = b_masked
    
    log_b = np.log(np.maximum(b_work, 1e-30) / temperature)
    mx = log_b.max()
    if np.isinf(mx) or np.isnan(mx):
        idx = np.argsort(b)[::-1][:5]
        w = np.zeros_like(b)
        w[idx] = 1.0 / 5
    else:
        log_b -= mx
        w = np.exp(log_b)
        ws = w.sum()
        if ws < 1e-30:
            idx = np.argsort(b)[::-1][:5]
            w = np.zeros_like(b)
            w[idx] = 1.0 / 5
        else:
            w /= ws
    
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[int(k)]
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        a_slice = a[k - j_hi : k - j_lo + 1][::-1]
        g[j_lo : j_hi + 1] += 2.0 * wk * a_slice
    return g

def mirror_step(a, g, eta):
    """Multiplicative weights (mirror descent) step."""
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def nesterov_mirror_descent(a0, time_budget, eta0=0.03, verbose=True, mode="topk"):
    """Nesterov accelerated mirror descent with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    eta = eta0
    stall = 0
    max_stall = 300
    rng = np.random.default_rng(42)
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        b = fast_autocorr(y)
        M = b.max()
        
        # Compute subgradient
        if mode == "topk":
            top_k = min(5, n)
            # Weight by how close each lag is to max
            top_idx = np.argsort(b)[::-1][:top_k]
            top_vals = b[top_idx]
            w_vals = top_vals / top_vals.sum()
            g = subgradient_topk(y, b, top_k=top_k, weights=w_vals)
        elif mode == "softmax":
            temp = max(0.001, M * 0.05 * (1.0 + 0.5 * np.sin(t * 0.01)))
            g = softmax_subgradient(y, b, temperature=temp, top_k=30)
        else:
            top_k = min(3, n)
            g = subgradient_topk(y, b, top_k=top_k)
        
        # Mirror descent step
        y_new = mirror_step(y, g, eta)
        
        b_new = fast_autocorr(y_new)
        M_new = b_new.max()
        
        if M_new < M * (1.0 - 1e-12):
            x_prev = x
            x = y_new
            stall = 0
            eta = min(eta * 1.005, 0.3)
            if M_new < best_m:
                best_m = M_new
                best_a = x.copy()
                if verbose and (t % 2000 == 0):
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}, mode={mode}")
        else:
            stall += 1
            eta *= 0.8
            if stall > max_stall:
                # Restart with perturbation
                stall = 0
                max_stall = max(100, max_stall // 2)
                eta = eta0 * 0.5
                # Asymmetric perturbation
                pert = rng.normal(0, 0.02, n)
                # Add asymmetry: bias toward one side
                bias = rng.uniform(-0.5, 0.5)
                pert += bias * np.linspace(-1, 1, n) * 0.01
                x_r = best_a * (1.0 + pert)
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                x_prev = x_r
        
        # Periodic check with ergodic average
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            if verbose and t % 10000 == 0:
                print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_m

def coordinate_polish(a0, time_budget, verbose=True, rng_seed=123):
    """Coordinate-level polish: try small transfers between pairs."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 10
    
    while time.monotonic() - t0 < time_budget and stall < max_stall:
        improved = False
        n_pairs = min(100, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_budget:
                    break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if a[i] < 1e-10 and a[j] < 1e-10:
                continue
            
            # Try transferring from i to j
            src, dst = i, j
            if a[src] < 1e-10:
                src, dst = j, i
                if a[src] < 1e-10:
                    continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Try a larger random move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-6:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
