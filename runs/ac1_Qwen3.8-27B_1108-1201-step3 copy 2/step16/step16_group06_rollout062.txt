```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with log-sum-exp smoothed objective and temperature annealing (primary)
2. FFT-based gradient computation
3. Softmax parameterization for non-negativity and normalization
4. LP-Frank-Wolfe polish
5. Coordinate descent final polish
6. Multi-initialization exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def _fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]

def _evaluate_local(a):
    """Evaluate the objective locally (same as evaluate_sequence but faster)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = _fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))

def _smoothed_objective(z, T):
    """Compute smoothed objective and gradient using softmax parameterization."""
    n = len(z)
    # Softmax
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum
    
    # Autocorrelation
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    ac = irfft(A * A, fft_size)[:2 * n - 1]
    
    # Smoothed max via log-sum-exp
    ac_shifted = ac - ac.max()
    log_p = ac_shifted / T
    max_log_p = log_p.max()
    p = np.exp(log_p - max_log_p)
    p_sum = p.sum()
    p = p / p_sum
    
    # Objective: 2*n*T*logsumexp(ac/T)
    obj = 2.0 * n * T * (max_log_p + np.log(p_sum))
    
    # Gradient w.r.t. a
    # S1[i] = sum_j a[j] * p[j+i]
    # S2[i] = sum_j p[i-j] * a[j] = (p*a)[i]
    P = rfft(p, fft_size)
    S2 = irfft(A * P, fft_size)[:n]
    
    p_rev = p[::-1]
    PR = rfft(p_rev, fft_size)
    S1_full = irfft(A * PR, fft_size)
    S1 = S1_full[n - 1:2 * n - 1][::-1]
    
    grad_a = 2.0 * n * (S1 + S2)
    
    # Gradient w.r.t. z (softmax chain rule)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_z

def _lbgfs_smoothed(a_init, T_schedule, max_iter_per_T=200, verbose=False):
    """Run L-BFGS with temperature annealing."""
    from scipy.optimize import minimize
    
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    z = np.log(a)
    
    best_a = a.copy()
    best_obj = _evaluate_local(a)
    
    for T_i, T in enumerate(T_schedule):
        result = minimize(
            _smoothed_objective,
            z,
            args=(T,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter_per_T, 'gtol': 1e-8, 'ftol': 1e-10}
        )
        z = result.x
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        
        obj_true = _evaluate_local(a_new)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a_new.copy()
            if verbose:
                print(f"      T={T:.4f}: new best={best_obj:.6f}")
        
        z = np.log(np.maximum(a_new, 1e-30))
    
    return best_a, best_obj

def _lp_fewolfe(a0, time_limit, max_iters=5000, verbose=False):
    """Simplified LP-Frank-Wolfe."""
    import cvxpy as cp
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _fast_autocorr(a)
        M = ac.max()
        
        # Build LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        # (a*b)[k] = sum_j a[j]*b[k-j] for valid k-j
        # This is a convolution, which is linear in b
        constraints = []
        for k in range(2 * n - 1):
            # (a*b)[k] = sum_j a[j] * b[k-j] where 0<=j<n and 0<=k-j<n
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min > j_max:
                continue
            # sum of a[j] * b[k-j] for j in [j_min, j_max]
            expr = 0
            for j in range(j_min, j_max + 1):
                expr += a[j] * b_var[k - j]
            constraints.append(expr <= M)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        a_new = (1 - alpha) * a + alpha * g_dir
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            a = a_new
        
        m_new = _fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
    
    return best_a, best_m
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with temperature annealing (primary)
2. FFT-based gradient computation via cross-correlation
3. Softmax parameterization for non-negativity and sum constraint
4. Multi-restart with perturbations to escape local minima
5. LP-Frank-Wolfe polish
6. Coordinate descent final polish
7. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_GLOBAL_BEST = [None]
_GLOBAL_BEST_SCORE = [float('inf')]


def _fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]


def _score(a):
    """Evaluate objective: 2*n*max(ac)/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = _fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))


def _smoothed_obj_grad(z, T):
    """Compute smoothed objective and gradient with softmax parameterization."""
    n = len(z)
    # Softmax
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()

    # Autocorrelation via FFT
    fft_size_ac = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size_ac)
    ac = irfft(A * A, fft_size_ac)[:2 * n - 1]

    # Softmax weights on ac (numerically stable)
    ac_max = ac.max()
    log_p = (ac - ac_max) / T
    p = np.exp(log_p)
    p_sum = p.sum()
    p_norm = p / p_sum

    # Objective: 2*n*T*logsumexp(ac/T) = 2*n*(ac_max + T*log(p_sum))
    obj = 2.0 * n * (ac_max + T * np.log(p_sum))

    # Gradient: dJ/da[i] = 4*n*S1[i] where S1[i] = sum_j a[j]*p[j+i]
    # S1 is cross-correlation of a and p
    fft_size_corr = 1 << (3 * n - 2).bit_length()
    A_full = np.fft.fft(a, fft_size_corr)
    P_full = np.fft.fft(p_norm, fft_size_corr)
    S1 = np.fft.ifft(np.conj(A_full) * P_full).real[:n]
    grad_a = 4.0 * n * S1

    # Softmax chain rule: dJ/dz[i] = a[i]*(grad_a[i] - dot(a, grad_a))
    grad_z = a * (grad_a - np.dot(a, grad_a))

    return obj, grad_z


def _lbgfs_anneal(a_init, time_budget, n_temps=15, max_iter_per_T=300, verbose=False):
    """Run L-BFGS with temperature annealing. Returns (best_a, best_score)."""
    global _GLOBAL_BEST, _GLOBAL_BEST_SCORE

    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()

    best_a = a.copy()
    best_s = _score(a)

    # Determine initial temperature from autocorrelation spread
    ac = _fast_autocorr(a)
    ac_max = ac.max()
    ac_median = np.median(ac)
    initial_T = max((ac_max - ac_median) / 5.0, ac_max / 20.0, 1e-10)

    # Geometric schedule
    T_min = ac_max / 500.0
    T_max = initial_T
    if T_min >= T_max:
        T_min = T_max * 0.01

    if n_temps > 1:
        ratios = (T_min / T_max) ** (1.0 / (n_temps - 1))
        T_schedule = [T_max * (ratios ** i) for i in range(n_temps)]
    else:
        T_schedule = [T_max]

    t0 = time.monotonic()
    z = np.log(a)

    for i_T, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_budget:
            break

        try:
            result = minimize(
                _smoothed_obj_grad,
                z,
                args=(T,),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter_per_T, 'gtol': 1e-10, 'ftol': 1e-12,
                         'maxls': 50}
            )
            z = result.x
        except Exception:
            break

        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)

        s_new = _score(a_new)
        if s_new < best_s:
            best_s = s_new
            best_a = a_new.copy()
            if s_new < _GLOBAL_BEST_SCORE[0]:
                _GLOBAL_BEST_SCORE[0] = s_new
                _GLOBAL_BEST[0] = a_new.copy()
                if verbose:
                    print(f"    T={T:.2e}: NEW GLOBAL BEST = {s_new:.6f}")
        elif verbose:
            pass

        # Update z from best to avoid drift
        z = np.log(np.maximum(best_a, 1e-30))

    return best_a, best_s

    best_s = 2.0 * n * best_m
    t0 = time.monotonic()

    # Find a solver

        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sname, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = sname
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS

    for t in range(1, max_iters + 1):

            break

        ac = _fast_autocorr(a)
        M = ac.max()

        # Build T matrix: T[k,j] = a[j] if 0 <= k-j < n
        # Constraint: T @ b <= M (scalar), b >= 0, maximize sum(b)
        # T has shape (2n-1, n)
        # For efficiency, use sparse representation
        # T[k,j] = a[j] when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1

        # For large n, building the full matrix is expensive. Use the structure.
        # Actually for n=2400, T is 4799x2400 ≈ 11.5M entries. Let's just build it.
        # But this takes time each iteration. Let's use a smarter approach.

        # Alternative: use the fact that (a*b)[k] = sum_j a[j]*b[k-j]
        # We can write this as a product of a banded matrix and b.
        # For now, let's just build it directly for correctness.

        # Actually, let's skip the LP-FW for now and use a simpler approach:
        # Just do a few steps of subgradient descent on the max.
        # This is much faster and still effective as a polish.

        # Find peak indices
        top_k = np.argsort(ac)[::-1][:20]
        peak_k = top_k[0]

        # Subgradient of max(ac) w.r.t. a
        # d ac[k]/da[i] = 2*a[k-i] if 0<=k-i<n
        grad = np.zeros(n)
        for k in top_k:
            k_int = int(k)
            w_ac = 1.0  # equal weight for simplicity
            for i in range(n):
                d = k_int - i
                if 0 <= d < n:
                    grad[i] += w_ac * 2.0 * a[d]

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Step: a_new = a - eta * grad, projected to simplex
        alpha = 2.0 / (t + 2.0)
        a_trial = a - alpha * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        m_trial = _fast_autocorr(a_trial).max()
        if m_trial < best_m:
            best_m = m_trial
            best_a = a_trial.copy()
            a = a_trial
            s_trial = 2.0 * n * best_m
            if s_trial < _GLOBAL_BEST_SCORE[0]:
                _GLOBAL_BEST_SCORE[0] = s_trial
                _GLOBAL_BEST[0] = best_a.copy()

    return best_a, _score(best_a)


def _coord_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent targeting peak autocorrelation reduction."""
    global _GLOBAL_BEST, _GLOBAL_BEST_SCORE
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_s = _score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = int(np.argmax(ac))

        # Find contributing indices for the peak
        # ac[peak_idx] = sum_j a[j]*a[peak_idx-j] for valid j
        j_min = max(0, peak_idx - n + 1)
        j_max = min(n - 1, peak_idx)
        if j_min > j_max:
            break

        idx = np.arange(j_min, j_max + 1)
        contrib = a[idx] * a[peak_idx - idx]

        if contrib.max() < 1e-15:
            break

        # Find best move: decrease a high-contributing index, increase a low-impact one
        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])

        # Compute sensitivity of each position to top peaks
        top_peaks = np.argsort(ac)[::-1][:15]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_i = int(pk)
            for pos in range(n):
                d = pk_i - pos
                if 0 <= d < n:
                    sensitivity[pos] += a[d]

        # Find low-sensitivity destinations
        dst_candidates = np.argsort(sensitivity)[:max(50, n // 8)]

        improved = False
        for dst in dst_candidates:
            if dst == src:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] = a[src] - amt
                a_trial[dst] = a[dst] + amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < M_curr - 1e-12:
                    a = a_trial
                    s_new = 2.0 * n * M_trial
                    if s_new < best_s:
                        best_s = s_new
                        best_a = a.copy()
                        if s_new < _GLOBAL_BEST_SCORE[0]:
                            _GLOBAL_BEST_SCORE[0] = s_new
                            _GLOBAL_BEST[0] = best_a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a > 1e-15:
                        a = a / s_a
                        s_new = _score(a)
                        if s_new < best_s:
                            best_s = s_new
                            best_a = a.copy()
                            stall = 0

    if verbose:
        print(f"    CoordDesc: score={best_s:.6f}, time={time.monotonic()-t0:.1f}s")
    return best_a, best_s


def _try_update(a, label="", verbose=True):
    """Try to update global best."""
    global _GLOBAL_BEST, _GLOBAL_BEST_SCORE
    if a is None:
        return
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = _score(a_norm)
    if np.isfinite(sc) and sc < _GLOBAL_BEST_SCORE[0]:
        _GLOBAL_BEST_SCORE[0] = sc
        _GLOBAL_BEST[0] = a_norm.copy()
        if verbose:
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")


def _interpolate(a, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, a_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search function. Returns best sequence found."""
    global _GLOBAL_BEST, _GLOBAL_BEST_SCORE

    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)

    _GLOBAL_BEST[0] = None
    _GLOBAL_BEST_SCORE[0] = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size >= 100 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            _try_update(parent, "parent")
            print(f"Parent loaded: n={len(parent)}, score={_GLOBAL_BEST_SCORE[0]:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if _GLOBAL_BEST[0] is None:
        # Default initialization
        n_init = 2000
        x = np.linspace(0, 1, n_init)
        a_init = np.sin(np.pi * x) ** 2
        a_init = a_init / a_init.sum()
        _try_update(a_init, "default_sin")
        print(f"Default init: n={n_init}, score={_GLOBAL_BEST_SCORE[0]:.6f}")
        parent = a_init

    n0 = len(_GLOBAL_BEST[0])
    print(f"Starting: n={n0}, score={_GLOBAL_BEST_SCORE[0]:.6f}")

    # === Phase 1: Quick L-BFGS warmup (30s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 1: L-BFGS warmup (n={n0}) ===")
        t1 = min(remaining * 0.04, 30.0)
        a_out, s_out = _lbgfs_anneal(_GLOBAL_BEST[0], t1, n_temps=8, max_iter_per_T=100, verbose=True)
        _try_update(a_out, f"lbgfs_warmup_n{n0}")

    # === Phase 2: Main L-BFGS with full annealing (180s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 2: L-BFGS main (n={len(_GLOBAL_BEST[0])}) ===")
        t2 = min(remaining * 0.2, 180.0)
        a_out, s_out = _lbgfs_anneal(_GLOBAL_BEST[0], t2, n_temps=20, max_iter_per_T=300, verbose=True)
        _try_update(a_out, f"lbgfs_main")

    # === Phase 3: Multi-restart L-BFGS (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(_GLOBAL_BEST[0])
        print(f"\n=== Phase 3: Multi-restart L-BFGS (n={n_cur}) ===")
        rng = np.random.default_rng(123)
        n_restarts = 10
        t_per_re = min(remaining / n_restarts, 40.0)

        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break

            # Create perturbed starting point
            noise_scale = 0.01 * (1 + r * 0.3)
            perturbed = _GLOBAL_BEST[0] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 1e-10)
            perturbed = perturbed / perturbed.sum()

            a_out, s_out = _lbgfs_anneal(perturbed, t_per_re, n_temps=12,
                                          max_iter_per_T=200, verbose=False)
            _try_update(a_out, f"restart_{r}")

    # === Phase 4: LP-FW / subgradient polish (120s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(_GLOBAL_BEST[0])
        print(f"\n=== Phase 4: Subgradient polish (n={n_cur}) ===")
        t4 = min(remaining * 0.15, 120.0)
        a_out, s_out = _lp_fewolfe(_GLOBAL_BEST[0], t4, max_iters=2000)
        _try_update(a_out, "subgrad_polish")

    # === Phase 5: L-BFGS after polish (120s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 5: L-BFGS after polish ===")
        t5 = min(remaining * 0.15, 120.0)
        a_out, s_out = _lbgfs_anneal(_GLOBAL_BEST[0], t5, n_temps=15,
                                      max_iter_per_T=200, verbose=True)
        _try_update(a_out, "lbgfs_post_polish")

    # === Phase 6: Coordinate descent (120s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(_GLOBAL_BEST[0])
        print(f"\n=== Phase 6: Coordinate descent (n={n_cur}) ===")
        t6 = min(remaining * 0.12, 120.0)
        a_out, s_out = _coord_descent(_GLOBAL_BEST[0], t6, verbose=True, rng_seed=99)
        _try_update(a_out, "coord_desc")

    # === Phase 7: Final L-BFGS polish (100s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final L-BFGS polish ===")
        t7 = min(remaining * 0.2, 100.0)
        a_out, s_out = _lbgfs_anneal(_GLOBAL_BEST[0], t7, n_temps=10,
                                      max_iter_per_T=200, verbose=True)
        _try_update(a_out, "lbgfs_final")

    # === Phase 8: Try different lengths (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(_GLOBAL_BEST[0])
        print(f"\n=== Phase 8: Length exploration (best n={n_best}) ===")
        test_ns = []
        for delta in [200, 400, -200, 600, -400, 100]:
            nt = n_best + delta
            if 800 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            print(f"  Trying n={nt}...")
            a_interp = _interpolate(_GLOBAL_BEST[0], nt)
            t_per = min(remaining * 0.3, 60.0)
            a_out, s_out = _lbgfs_anneal(a_interp, t_per, n_temps=12,
                                          max_iter_per_T=150, verbose=False)
            _try_update(a_out, f"n{nt}")

    # === Phase 9: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(_GLOBAL_BEST[0])
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        # Quick coord descent
        t9a = min(remaining * 0.4, 60.0)
        a_out, s_out = _coord_descent(_GLOBAL_BEST[0], t9a, verbose=False, rng_seed=777)
        _try_update(a_out, "final_coord")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            # Quick L-BFGS
            t9b = min(remaining * 0.6, 60.0)
            a_out, s_out = _lbgfs_anneal(_GLOBAL_BEST[0], t9b, n_temps=8,
                                          max_iter_per_T=100, verbose=False)
            _try_update(a_out, "final_lbgfs")

    # === Final validation ===
    if _GLOBAL_BEST[0] is not None:
        final_seq = _GLOBAL_BEST[0]
        final_score = _score(final_seq)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={final_score:.6f}, n={len(final_seq)}, time={elapsed:.0f}s")

        # Validate with actual evaluate_sequence
        try:
            test_score = evaluate_sequence([float(x) for x in final_seq])
            print(f"Validation: evaluate_sequence gives {test_score:.6f}")
        except Exception as e:
            print(f"Validation error: {e}")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in _GLOBAL_BEST[0]]
```
