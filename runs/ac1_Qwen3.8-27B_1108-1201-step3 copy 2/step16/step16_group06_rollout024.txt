The "Frank-Wolfe with Nesterov's acceleration" (FW-N) works as:
- Maintain two sequences: $x_t$ (the true iterate) and $y_t$ (the "lookahead" point)
- $y_t = (1 - \beta_t) x_t + \beta_t y_{t-1}$ where $\beta_t = (t-1)/(t+1)$
- Solve the linear subproblem at $y_t$: $v_t = \arg\min_{v \in S} \langle \nabla f(y_t), v \rangle$
- Update: $x_{t+1} = (1 - \alpha_t) x_t + \alpha_t v_t$ where $\alpha_t = 2/(t+2)$

The key difference: the subgradient is evaluated at $y_t$ (the lookahead) rather than at $x_t$ (the true point). This gives a better direction.

For our problem, the "linear subproblem" is: $\min_{v \in \Delta} \langle g(y_t), v \rangle$ where $g(y_t)$ is the subgradient at $y_t$. The solution is $v = e_j$ where $j = \arg\min_i g_i(y_t)$.

This is still O(n * |K*|) per iteration. Very fast.

- Instead of always moving toward a new vertex, sometimes move away from an existing vertex in the convex combination representation.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated Frank-Wolfe with subgradient directions (fast)
2. Periodic LP-based direction finding (slow but high-quality)
3. Multiple diverse initializations with best-of selection
4. Away-step FW for faster convergence
5. Peak-targeted coordinate descent polish
6. Multi-length exploration and interpolation
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).flatten()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def subgradient_at(a, top_k=20):
    """Compute subgradient of max_k (a*a)_k at point a.
    Returns (gradient vector, list of peak indices)."""
    n = len(a)
    ac = fast_autocorr(a)
    # Get top_k peaks
    k_max = min(top_k, len(ac))
    # Use a threshold: peaks within 5% of max
    threshold = ac.max() * 0.95
    peak_mask = ac >= threshold
    peak_indices = np.where(peak_mask)[0]
    if len(peak_indices) == 0:
        peak_indices = np.argmax(ac).reshape(1)
    
    g = np.zeros(n)
    for k in peak_indices:
        k_int = int(k)
        # d(a*a)_k / da_j = 2*a_{k-j} for valid j
        for j in range(max(0, k_int - n + 1), min(n, k_int + 1)):
            g[j] += 2.0 * a[k_int - j]
    
    g /= len(peak_indices)
    return g, peak_indices, ac

def subgradient_fw_nesterov(a0, time_limit, top_k=20, verbose=False, seed=42):
    """Nesterov-accelerated FW with subgradient directions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()  # main iterate
    y = a.copy()  # lookahead
    x_prev = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient at y
        g, peaks, ac = subgradient_at(y, top_k=top_k)
        
        # FW direction: vertex with minimum gradient
        v_idx = np.argmin(g)
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        
        # Update
        x_new = (1.0 - alpha) * x + alpha * np.eye(n)[v_idx]
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        
        x_prev = x
        x = x_new
        
        # Track best
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
        
        my = fast_autocorr(y).max()
        if my < best_m - 1e-15:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      FW t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def lx_direction(y, M):
    """Solve LP: max sum(b) s.t. b>=0, (y*b)_k <= M for all k.
    Returns normalized b or None if infeasible/too slow."""
    import cvxpy as cp
    n = len(y)
    num_rows = 2 * n - 1
    
    # Build constraint matrix sparsely
    # T[k,j] = y[j] if 0 <= k-j < n, else 0
    # T is (2n-1) x n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if y[j] < 1e-15:
            continue
        for k in range(j, j + n):
            if k < num_rows:
                rows.append(k)
                cols.append(j)
                vals.append(y[j])
    
    import scipy.sparse as sp
    T = sp.coo_matrix((vals, (rows, cols)), shape=(num_rows, n)).tocsr()
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    t_start = time.monotonic()
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    
    if time.monotonic() - t_start > 30:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum

def hybrid_fw(a0, time_limit, top_k=20, lp_every=200, verbose=False, seed=42):
    """FW with subgradient directions, occasionally using LP directions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    y = a.copy()
    x_prev = a.copy()

        beta_t = (t - 1.0) / (t + 1.5)

        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Determine direction
        direction = None
        if t % lp_every == 0 and time.monotonic() - t0 < time_limit * 0.9:
            # Try LP direction
            ac_y = fast_autocorr(y)
            M_curr = ac_y.max()
            direction = lx_direction(y, M_curr)
        
        if direction is None:
            # Subgradient direction
            g, peaks, ac = subgradient_at(y, top_k=top_k)
            v_idx = np.argmin(g)
            direction = np.zeros(n)
            direction[v_idx] = 1.0
        
        # Step size (use a slightly adaptive one)
        alpha = 2.0 / (t + 2.0)
        
        # Update
        x_new = (1.0 - alpha) * x + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:

        x = x_new
        
        # Track best
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
        
        # Also check y
        my = fast_autocorr(y).max()
        if my < best_m - 1e-15:
            best_m = my

        if verbose and t % 500 == 0:

def away_step_fw(a0, time_limit, top_k=20, verbose=False, seed=42):
    """FW with away steps for faster convergence."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Store vertices
    vertices = [a.copy()]
    lambdas = np.array([1.0])
    
    x = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at x
        g, peaks, ac = subgradient_at(x, top_k=top_k)
        
        # Standard FW direction
        v_idx = np.argmin(g)
        v = np.zeros(n)
        v[v_idx] = 1.0
        
        # Check for away step
        away_idx = -1
        best_away_gain = 0
        if len(vertices) > 1:
            for i, vi in enumerate(vertices):
                if lambdas[i] < 1e-4:
                    continue
                gain = np.dot(g, vi)
                if gain > best_away_gain:
                    best_away_gain = gain
                    away_idx = i
        
        alpha = 2.0 / (t + 2.0)
        
        if away_idx >= 0 and best_away_gain > 0:
            # Away step: increase lambda_v, decrease lambda_away
            gamma = min(alpha, 0.5 * lambdas[away_idx] / max(best_away_gain, 1e-10))
            lambdas[v_idx if v_idx < len(lambdas) else len(lambdas)-1] += gamma
            if v_idx >= len(lambdas):
                lambdas = np.append(lambdas, gamma)
                vertices.append(v.copy())
            else:
                lambdas[v_idx] += gamma
            lambdas[away_idx] -= gamma
            if lambdas[away_idx] < 0:
                lambdas[away_idx] = 0
            # Compute x from vertices
            x = np.zeros(n)
            for i, vi in enumerate(vertices):
                if lambdas[i] > 0:
                    x += lambdas[i] * vi
            s = x.sum()
            if s > 1e-15:
                x = x / s
            else:
                x = norm_sum1(a0)
            # Clean up small lambdas
            mask = lambdas > 1e-6
            if mask.sum() < len(lambdas):
                lambdas = lambdas[mask]
                vertices = [vertices[i] for i in range(len(vertices)) if mask[i]]
        else:
            # Standard FW update
            if v_idx < n:
                # Check if v is already in vertices
                found = False
                for i, vi in enumerate(vertices):
                    if vi[v_idx] > 0.99:
                        found = True
                        lambdas[i] += alpha
                        break
                if not found:
                    lambdas = np.append(lambdas, alpha)
                    vertices.append(v.copy())
                else:
                    lambdas[np.argmax([vi[v_idx] for vi in vertices])] += alpha
            
            # Normalize lambdas
            s_lam = lambdas.sum()
            if s_lam > 1e-15:
                lambdas = lambdas / s_lam
            
            # Compute x
            x = np.zeros(n)
            for i, vi in enumerate(vertices):
                if lambdas[i] > 1e-10:
                    x += lambdas[i] * vi
            s = x.sum()
            if s > 1e-15:
                x = x / s
        
        # Track best
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      AwayFW t={t} ({elapsed:.1f}s): J={J:.6f}, |V|={len(vertices)}")
            
            # Prune vertices if too many
            if len(vertices) > 200:
                mask = lambdas > 0.005
                if mask.sum() < len(lambdas):
                    lambdas = lambdas[mask]
                    vertices = [vertices[i] for i in range(len(vertices)) if mask[i]]
                    s_lam = lambdas.sum()
                    if s_lam > 0:
                        lambdas = lambdas / s_lam
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, top_k=10, verbose=False, seed=123):
    """Polish using pair-based mass transfer targeting peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        peak_indices = np.argsort(ac)[::-1][:top_k]
        
        # Compute sensitivity: for each position, how much it contributes to peaks
        sensitivity = np.zeros(n)
        for pk in peak_indices:
            pk_int = int(pk)
            for pos in range(n):
                d = pk_int - pos
                if 0 <= d < n:
                    sensitivity[pos] += a[d]
        
        # Try moving mass from high-sensitivity to low-sensitivity positions
        src_candidates = np.argsort(sensitivity)[::-1][:min(50, n // 4)]
        dst_candidates = np.argsort(sensitivity)[:min(50, n // 4)]
        
        improved = False
        for src in src_candidates[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[int(src)] < 1e-10:
                continue
            for dst in dst_candidates[:20]:
                if int(dst) == int(src):
                    continue
                # Try small transfer
                for frac in [0.01, 0.005, 0.002, 0.02]:
                    amt = a[int(src)] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[int(src)] -= amt
                    a_trial[int(dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_try = min(10, n)
            for _ in range(n_try):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDescent ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_initialization(n, kind, seed=0):
    """Generate a diverse initialization."""
    rng = np.random.default_rng(seed)
    if kind == 'uniform':
        a = np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = a / a.sum()
    elif kind == 'gaussian':
        x = np.linspace(0, 1, n)
        sigma = rng.uniform(0.15, 0.35)
        center = rng.uniform(0.3, 0.7)
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        a = a / a.sum()
    elif kind == 'asym_cosine':
        x = np.linspace(0, 1, n)
        shift = rng.uniform(-0.2, 0.2)
        a = np.cos(np.pi * (x - 0.5 + shift)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif kind == 'power_law':
        x = np.linspace(0, 1, n)
        p = rng.uniform(-2, 2)
        a = (x + 0.01) ** p * ((1 - x) + 0.01) ** p
        a = a / a.sum()
    elif kind == 'random_smooth':
        a = rng.uniform(0.5, 1.5, n)
        # Smooth it
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=rng.uniform(5, 20))
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    t_phase1 = remaining * 0.18
    print(f"\n=== Phase 1: Diverse FW runs ({t_phase1:.0f}s) ===")
    
    inits = []
    # Parent
    inits.append(('parent', norm_sum1(best_seq)))
    # Diverse
    diverse_kinds = ['cosine', 'gaussian', 'asym_cosine', 'power_law', 'random_smooth', 'uniform']
    for i, kind in enumerate(diverse_kinds[:4]):
        a_init = generate_initialization(n0, kind, seed=100 + i)
        inits.append((kind, a_init))
    
    per_init_time = t_phase1 / len(inits)
    for label, a_init in inits:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 10:
            break
        t_alloc = min(per_init_time, remaining * 0.9)
        if t_alloc < 5:
            break
        print(f"  Running FW [{label}] for {t_alloc:.0f}s...")
        a_out, _ = subgradient_fw_nesterov(a_init, t_alloc, top_k=30, verbose=False, seed=hash(label) % 10000)
        try_update(a_out, f"phase1_{label}")
    
    # === Phase 2: Hybrid FW with LP on best (35% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = remaining * 0.35
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Hybrid FW+LP n={n_cur} ({t_phase2:.0f}s) ===")
    
    a_out, _ = hybrid_fw(norm_sum1(best_seq), t_phase2, top_k=25, lp_every=100, verbose=True, seed=77)
    try_update(a_out, "phase2_hybrid")
    
    # === Phase 3: Away-step FW on current best (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase3 = remaining * 0.15
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Away-step FW n={n_cur} ({t_phase3:.0f}s) ===")
    
    a_out, _ = away_step_fw(norm_sum1(best_seq), t_phase3, top_k=20, verbose=True, seed=55)
    try_update(a_out, "phase3_away")
    
    # === Phase 4: Coordinate polish (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase4 = remaining * 0.10
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord polish n={n_cur} ({t_phase4:.0f}s) ===")
    
    a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t_phase4, top_k=10, verbose=True, seed=42)
    try_update(a_out, "phase4_coord")
    
    # === Phase 5: Another hybrid FW round (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase5 = remaining * 0.15
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Hybrid FW round 2 n={n_cur} ({t_phase5:.0f}s) ===")
    
    a_out, _ = hybrid_fw(norm_sum1(best_seq), t_phase5, top_k=30, lp_every=50, verbose=True, seed=123)
    try_update(a_out, "phase5_hybrid2")
    
    # === Phase 6: Multi-n exploration (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-n exploration ===")
    
    n_candidates = []
    for delta in [200, 400, -200, -400, 600]:
        nc = n_best + delta
        if 1200 <= nc <= 3600 and nc != n_best:
            n_candidates.append(nc)
    
    for nc in n_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        print(f"  Trying n={nc}...")
        a_inter = interpolate_sequence(best_seq, nc)
        t_nc = min(remaining * 0.3, 120.0)
        a_out, _ = hybrid_fw(a_inter, t_nc, top_k=25, lp_every=200, verbose=False, seed=nc)
        try_update(a_out, f"phase6_n{nc}")
    
    # === Phase 7: Final polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t_phase7 = remaining * 0.5
        print(f"\n=== Phase 7: Final coord polish n={n_best} ({t_phase7:.0f}s) ===")
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t_phase7, top_k=8, verbose=True, seed=999)
        try_update(a_out, "phase7_final_coord")
        
        # If time remains, one more FW pass
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_phase8 = remaining * 0.8
            print(f"\n=== Phase 8: Final FW n={n_best} ({t_phase8:.0f}s) ===")
            a_out, _ = subgradient_fw_nesterov(norm_sum1(best_seq), t_phase8, top_k=30, verbose=False, seed=31415)
            try_update(a_out, "phase8_final_fw")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
