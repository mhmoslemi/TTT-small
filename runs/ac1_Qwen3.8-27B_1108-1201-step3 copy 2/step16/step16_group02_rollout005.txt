def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_soft_subgradient(a, ac, top_L, power):
    """
    Compute soft subgradient: weighted sum of subgradients from top-L lags.
    Subgradient at lag k: g_k[j] = 2*a[k-j] if 0<=k-j<n, else 0.
    Weighted by (ac[k])^power normalized.
    """
    n = len(a)
    num_lags = len(ac)
    L = min(top_L, num_lags)
    
    # Get top L lags by autocorrelation value
    top_idx = np.argsort(ac)[::-1][:L]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    # Power weighting
    w = top_vals ** power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(L) / L
    else:
        w = w / w_sum
    
    # Accumulate subgradient
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # g_k[j] = 2*a[k-j] for valid j (where 0 <= k-j < n)
        # Valid j: max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_vals = np.arange(j_start, j_end + 1)
        a_vals = a[k_int - j_vals]
        grad[j_vals] += 2.0 * wi * a_vals
    
    return grad

def nesterov_projected_subgrad(a0, time_limit, verbose=True, rng_seed=42):
    """
    Nesterov-accelerated projected subgradient method on simplex.
    Objective: 2*n*max_c (with sum a = 1).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0  # step size constant
    stall_count = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        
        # Adaptive: increase L and decrease power over time
        frac = min(1.0, t / 5000.0)
        L = int(20 + 60 * frac)  # 20 to 80
        power = 3.0 + 2.0 * frac  # 3.0 to 5.0
        
        g = compute_soft_subgradient(y, ac, L, power)
        
        # Step size: eta ~ gamma / sqrt(t)
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Projected gradient step
        v = y - eta * g
        # Project onto simplex {a >= 0, sum a = 1}
        v_sorted = np.sort(v)[::-1]
        cssv = np.cumsum(v_sorted)
        rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - 1))[0]
        if len(rho) > 0:
            tau = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
        else:
            tau = 0.0
        a_new = np.maximum(v - tau, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

def make_u_init(n, sharpness=2.0):
    """Create a U-shaped initialization."""
    x = np.linspace(0, 1, n)
    # U-shape: high at edges, low in middle
    a = (x - 0.5) ** 2 + 0.01
    a = a ** sharpness
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_bimodal_init(n, center_width=0.15):
    """Create a bimodal initialization (two peaks)."""
    x = np.linspace(0, 1, n)
    # Two Gaussians at 1/4 and 3/4
    a = np.exp(-((x - 0.25) ** 2) / (2 * center_width ** 2)) + \
        np.exp(-((x - 0.75) ** 2) / (2 * center_width ** 2))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
