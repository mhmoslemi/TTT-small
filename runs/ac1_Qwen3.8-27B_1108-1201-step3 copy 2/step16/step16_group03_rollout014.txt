def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective score for a sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_conv(a)
    return float(2.0 * len(a) * ac.max() / (s * s))


def norm_sum1(a):
    """Normalize so sum = 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_conv_matrix(a):
    """Build sparse matrix T where (T @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start = j
        end = j + n
        rows.extend(range(start, end))
        cols.extend([j] * n)
        vals.extend(a.tolist())
    T = cp.SparseMatrix(np.array(rows), np.array(cols), np.array(vals),
                        shape=(2 * n - 1, n))
    return T


def convolve_with_row(a, k, n):
    """Compute (a*b)[k] coefficient for variable b[j]: returns a[k-j] for valid j."""
    # (a*b)[k] = sum_j a[k-j] * b[j], valid when 0<=k-j<n i.e. k-n+1<=j<=k
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    return j_start, j_end


def multi_peak_subgradient(y, ac, n, top_k=5, threshold_frac=0.8):
    """
    Compute subgradient as weighted average over peaks above threshold.
    Weight by peak value squared for emphasis on largest peaks.
    """
    k_max_val = ac.max()
    if k_max_val < 1e-30:
        return np.zeros(n)
    
    # Find peaks above threshold
    threshold = threshold_frac * k_max_val
    peak_indices = np.where(ac >= threshold)[0]
    
    # Also always include the top-k peaks
    top_idx = np.argsort(ac)[::-1][:top_k]
    all_peaks = np.unique(np.concatenate([peak_indices[:min(len(peak_indices), 20)], top_idx]))
    
    if len(all_peaks) == 0:
        peak_vals = np.array([k_max_val])
        peak_idx = np.array([np.argmax(ac)])
    else:
        peak_vals = ac[all_peaks]
        peak_idx = all_peaks
    
    # Weight by value^2
    w = peak_vals ** 2
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(w)) / len(w)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, peak_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        for j in range(j_start, j_end):
            grad[j] += wi * 2.0 * y[k_int - j]
    
    return grad


def mirror_descent(a0, time_limit, verbose=True, c0=5.0, seed=42, 
                   top_k=3, threshold_frac=0.85, restart_interval=2000):
    """
    Adaptive mirror descent (exponentiated gradient) with multi-peak subgradient.
    Works on the simplex {a >= 0, sum(a)=1}.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    cur_top_k = top_k
    cur_thresh = threshold_frac
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute convolution
        ac = fast_conv(y)
        M_curr = ac.max()
        
        # Compute subgradient
        g = multi_peak_subgradient(y, ac, n, top_k=cur_top_k, threshold_frac=cur_thresh)
        
        # Exponentiated gradient update
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        if stall > max_stall // 2:
            cur_thresh = max(cur_thresh - 0.05, 0.5)
            cur_top_k = min(cur_top_k + 1, 10)
        
        # Restart
        if stall > max_stall:
            # Perturb around best
            noise_scale = max(0.001, 0.01 * stall / max_stall)
            x_restarted = best_a * (1.0 + noise_scale * rng.standard_normal(n))
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            cur_top_k = top_k
            cur_thresh = threshold_frac
            c0 = c0 * 1.2  # Increase step size on restart
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"    MD t={t} ({elapsed:.0f}s): score={score:.6f}, stall={stall}, thresh={cur_thresh:.2f}")
    
    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=500):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    LP: max sum(b) s.t. T@b <= M, b >= 0, where T encodes convolution with a.
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_M = fast_conv(y).max()
    best_score = 2.0 * n * best_M
    t0 = time.monotonic()
    
    x_mom = y.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix
        T = build_conv_matrix(y)
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            t_lp_start = time.monotonic()
            prob.solve(solver=solver, verbose=False)
            t_lp_end = time.monotonic()
        except Exception as e:
            print(f"    LP error at iter {t}: {e}")
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Line search: try multiple step sizes
        alphas = [2.0 / (t + 2.0)]
        # Add some fixed alphas for robustness
        if t < 10:
            alphas += [0.5, 0.3, 0.2]
        elif t < 50:
            alphas += [0.3, 0.2, 0.1]
        else:
            alphas += [0.15, 0.1, 0.05]
        
        best_alpha = alphas[0]
        trial = (1.0 - best_alpha) * y + best_alpha * direction
        trial = np.maximum(trial, 0.0)
        st = trial.sum()
        if st > 1e-15:
            trial = trial / st
        best_eval = fast_conv(trial).max()
        
        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_M = fast_conv(trial).max()
            if trial_M < best_eval:
                best_eval = trial_M
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx
        
        y = y_new
        
        # Track best
        M_y = fast_conv(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            best_score = 2.0 * n * best_M
        
        M_x = fast_conv(x_mom).max()
        if M_x < best_M:
            best_M = M_x
            best_a = x_mom.copy()
            best_score = 2.0 * n * best_M
        
        if verbose and (t <= 5 or t % 10 == 0):
            elapsed = time.monotonic() - t0
            lp_time = t_lp_end - t_lp_start if 't_lp_start' in dir() else 0
            print(f"    LP iter={t} ({elapsed:.0f}s, LP={lp_time:.1f}s): score={best_score:.6f}")
    
    return best_a, best_M

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            print(f"Parent: n={len(parent)}, sum={parent.sum():.4f}")
            best_seq = norm_sum1(parent)
            best_score = eval_score(best_seq)
            print(f"Initial score: {best_score:.6f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = eval_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Fast mirror descent (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror Descent n={n0} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 20:
        a_out, _ = mirror_descent(best_seq, t1, verbose=True, c0=6.0, seed=42,
                                   top_k=5, threshold_frac=0.85)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase 1 result: {best_score:.6f}")
    
    # === Phase 2: LP Frank-Wolfe (main phase, ~700s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Frank-Wolfe n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, verbose=True, max_iters=300)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase 2 result: {best_score:.6f}")
    
    # === Phase 3: Mirror descent refinement after LP (100s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: MD refinement n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 20:
        a_out, _ = mirror_descent(best_seq, t3, verbose=False, c0=8.0, seed=99,
                                   top_k=3, threshold_frac=0.9)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase 3 result: {best_score:.6f}")
    
    # === Phase 4: LP again (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP round 2 n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.2, 250.0)
    if t4 > 60:
        a_out, _ = lp_frank_wolfe(best_seq, t4, solver, verbose=True, max_iters=150)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase 4 result: {best_score:.6f}")
    
    # === Phase 5: Multi-length exploration (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length (remaining={remaining:.0f}s) ===")
    if remaining > 150:
        best_seq, best_score = try_different_lengths(best_seq, best_score, time_budget, t_global)
    
    # === Phase 6: Coordinate polish (100s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coordinate polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.5, 150.0)
    if t6 > 15:
        a_out, _ = coordinate_polish(best_seq, t6, verbose=True, seed=314)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase 6 result: {best_score:.6f}")
    
    # === Phase 7: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_best} (remaining={remaining:.0f}s) ===")
    if n_best <= 3000 and remaining > 80:
        t7 = min(remaining * 0.6, 200.0)
        a_out, _ = lp_frank_wolfe(best_seq, t7, solver, verbose=True, max_iters=100)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase 7 result: {best_score:.6f}")
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final coord n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t8 = min(remaining * 0.8, 100.0)
        a_out, _ = coordinate_polish(best_seq, t8, verbose=True, seed=777, n_sweep=100)
        sc = eval_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Phase 8 result: {best_score:.6f}")
    
    # Final validation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
