import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_weights(ac, mu):
    """Compute softmax weights for the smooth approximation."""
    # Use log-sum-exp trick for numerical stability
    max_ac = ac.max()
    exp_vals = np.exp(mu * (ac - max_ac))
    w = exp_vals / exp_vals.sum()
    return w

def smoothed_gradient(a, ac, mu):
    """Compute gradient of the smoothed objective f_mu(a) = (1/mu)*log(sum exp(mu*(a*a)[k]))."""
    n = len(a)
    w = softmax_weights(ac, mu)
    g = np.zeros(n)
    for k in range(len(ac)):
        if w[k] < 1e-15:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * w[k] * a[j_idx]
    return g

def exact_subgradient_masked(a, ac, top_fraction=0.99):
    """Compute subgradient using lags above a threshold fraction of max."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    threshold = top_fraction * M
    active = ac >= threshold
    # Normalize: subgradient of max at a point where multiple lags are tied
    # is the average of the individual subgradients
    n_active = active.sum()
    if n_active == 0:
        active = ac == M
        n_active = active.sum()
    g = np.zeros(n)
    for k in range(len(ac)):
        if not active[k]:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += (1.0 / n_active) * 2.0 * a[j_idx]
    return g

    x = a.copy()
    z = a.copy()
    t_fista = 1.0

    # L ~ 2 * n * mu (very rough upper bound)
    # Better: use backtracking

    stall = 0
    max_stall = 1000

    L = 10.0 * n * mu_start  # initial Lipschitz estimate

    max_stall = 2000
    it = 0

        it += 1
        frac = min(1.0, it / ramp_iters)
        mu = mu_start + (mu_end - mu_start) * frac

        if it == 1:
            y = z.copy()
        else:
            beta = (t_fista - 1.0) / (t_fista + 2.0)

            if sy < 1e-15:
                y = z.copy()
            else:
                y = y / sy

        ac_y = fast_autocorr(y)

        step = 1.0 / L

        sx = x_new.sum()

            x_new = y.copy()
        else:
            x_new = x_new / sx

        M_new = fast_autocorr(x_new).max()

            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Backtracking on L if we seem to be diverging
        if M_new > best_val * 1.5 and L < 100 * n * mu:
            L *= 2.0
        
        x = x_new
        z = x_new
        t_fista = t_new
        
        # Restart
        if stall > max_stall:
            stall = 0
            scale = 0.005 + 0.02 * min(1.0, stall / max_stall)
            pert = rng.normal(0, scale, n) * best_a
            x = np.maximum(best_a + pert, 1e-10)
            x = x / x.sum()
            z = x.copy()
            t_fista = 1.0

def coordinate_descent_exact(a0, time_limit, verbose=True, rng_seed=123):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    r = rng.integers(0, 6)
    if r == 0:
        # Hann window
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif r == 1:
        # Raised cosine
        f = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif r == 2:
        # Gaussian bump
        center = 0.3 + 0.4 * rng.random()
        width = 0.1 + 0.2 * rng.random()
        f = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif r == 3:
        # Sum of two Gaussians
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        w = 0.05 + 0.15 * rng.random()
        f = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    elif r == 4:
        # Flat with random perturbation
        f = np.ones(n) + rng.normal(0, 0.5, n)
    else:
        # Smooth random
        f = np.ones(n)
        for _ in range(5):
            center = rng.random()
            width = 0.05 + 0.2 * rng.random()
            amp = rng.uniform(-1, 1)
            f += amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f

def mirror_descent_adaptive(a0, time_limit, verbose=True, rng_seed=42):
    """
    Adaptive mirror descent with:
    - Dynamic threshold for active set
    - Backtracking line search
    - Nesterov acceleration
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    threshold = 0.98  # fraction of max that counts as "active"
    c = 3.0
    stall = 0
    max_stall = 3000
    it = 0
    max_iters = 200000
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        
        # Nesterov
        beta = (it - 1.0) / (it + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Active set
        active_thresh = threshold * M
        active_mask = ac >= active_thresh
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask = ac == M
            n_active = 1
        
        # Subgradient
        g = np.zeros(n)
        for k in range(len(ac)):
            if not active_mask[k]:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * a[j_idx] / n_active
        
        # Step size with backtracking
        eta = c / (np.sqrt(it) + 1.0) * np.sqrt(n)
        
        # Mirror update in log-space
        log_a = np.log(np.maximum(y, 1e-15))
        x_new = np.exp(log_a - eta * g / np.linalg.norm(g))
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        M_new = fast_autocorr(x_new).max()
        
        # Backtracking: if M increased too much, reduce step
        while M_new > M * 1.1 and eta > 1e-3:
            eta *= 0.5
            x_new = np.exp(np.log(np.maximum(y, 1e-15)) - eta * g / np.linalg.norm(g))
            x_new = np.maximum(x_new, 1e-30)
            s_new = x_new.sum()
            if s_new > 1e-15:
                x_new = x_new / s_new
            M_new = fast_autocorr(x_new).max()
        
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        
        # Adaptive threshold
        if stall > max_stall // 2 and threshold > 0.95:
            threshold = max(0.95, threshold - 0.01)
        
        # Restart
        if stall > max_stall:
            stall = 0
            noise = rng.normal(0, 0.008, n) * best_a
            x = np.maximum(best_a + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c *= 1.3
            threshold = max(0.95, threshold - 0.02)
            if verbose:
                print(f"    MD restart it={it} ({time.monotonic()-t0:.1f}s): J={2*n*best_m:.6f}, thresh={threshold:.3f}, c={c:.1f}")
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.1f}s): J={2*n*best_m:.6f}, active={n_active}, stall={stall}")
    
    return best_a, best_m

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def softmax_weights(ac, mu):
    max_ac = ac.max()
    exp_vals = np.exp(mu * (ac - max_ac))
    total = exp_vals.sum()
    if total < 1e-30:
        return np.ones(len(ac)) / len(ac)
    return exp_vals / total


def smoothed_gradient(a, ac, mu):
    n = len(a)
    w = softmax_weights(ac, mu)
    g = np.zeros(n)
    for k in range(len(ac)):
        wk = w[k]
        if wk < 1e-14:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wk * a[j_idx]
    return g


def exact_subgradient(a, ac, threshold_frac=0.99):
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    threshold = threshold_frac * M
    active = ac >= threshold
    n_active = active.sum()
    if n_active == 0:
        active = ac == M
        n_active = max(1, active.sum())
    g = np.zeros(n)
    for k in range(len(ac)):
        if not active[k]:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx] / n_active
    return g

def mirror_descent_adaptive(a0, time_limit, c0=3.0, threshold_frac=0.99,
                            verbose=True, rng_seed=42, max_iters=300000):
    """Adaptive mirror descent with dynamic threshold and backtracking in log-space."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    threshold = threshold_frac
    c = c0
    stall = 0
    max_stall = 4000
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        beta = (it - 1.0) / (it + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()

        g = exact_subgradient(y, ac, threshold_frac=threshold)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        eta = c / (np.sqrt(it) + 1.0) * np.sqrt(n) / g_norm

        log_y = np.log(np.maximum(y, 1e-15))
        x_new = np.exp(log_y - eta * g)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        M_new = fast_autocorr(x_new).max()

        backtracks = 0
        while M_new > M * 1.05 and eta > 1e-4 and backtracks < 8:
            eta *= 0.5
            x_new = np.exp(log_y - eta * g)
            x_new = np.maximum(x_new, 1e-30)
            s_new = x_new.sum()
            if s_new > 1e-15:
                x_new = x_new / s_new
            M_new = fast_autocorr(x_new).max()
            backtracks += 1

        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new

        if stall > max_stall // 2 and threshold > 0.95:
            threshold = max(0.95, threshold - 0.005)

        if stall > max_stall:
            stall = 0
            noise = rng.normal(0, 0.005, n) * best_a
            x = np.maximum(best_a + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c *= 1.25
            threshold = max(0.95, threshold - 0.01)
            if verbose:
                print(f"    MD-A restart it={it} ({time.monotonic()-t0:.1f}s): J={2*n*best_m:.6f}, thresh={threshold:.4f}, c={c:.2f}")

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-A it={it} ({elapsed:.1f}s): J={2*n*best_m:.6f}, thresh={threshold:.4f}, stall={stall}")

    return best_a, best_m

        it += 1

            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                hot_positions.add(i)
                j = k_int - i
                if 0 <= j < n:
                    hot_positions.add(j)

                            break

                    break

                    old_val = a_trial[i]

                    delta = new_val - old_val
                    a_trial[i] = new_val
                    # Redistribute to a random cold position
                    if cold_list:
                        dst = int(rng.choice(cold_list))
                        a_trial[dst] -= delta
                    else:
                        a_trial[np.arange(n) != i] -= delta / (n - 1)
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t

                    if M_trial < best_m - 1e-14:

                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        # Strategy 3: Random perturbation to escape

        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif r == 1:
        f = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif r == 2:
        center = 0.3 + 0.4 * rng.random()
        width = 0.1 + 0.2 * rng.random()
        f = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif r == 3:
        c = sorted(rng.uniform(0.15, 0.85, 2))
        w = 0.05 + 0.1 * rng.random()
        f = np.exp(-((x - c[0]) ** 2) / (2 * w ** 2)) + np.exp(-((x - c[1]) ** 2) / (2 * w ** 2))
    elif r == 4:
        f = np.ones(n) + rng.normal(0, 0.3, n)
    elif r == 5:
        f = np.ones(n)
        for _ in range(4):
            center = rng.random()
            width = 0.05 + 0.15 * rng.random()
            amp = rng.uniform(-0.5, 1.0)
            f += amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif r == 6:
        # Power law
        f = (x ** 0.5) * ((1 - x) ** 0.5) + 0.1
    else:
        # Multi-scale
        f = np.ones(n)
        f += 0.3 * np.sin(4 * np.pi * x)
        f += 0.1 * np.sin(10 * np.pi * x)
        f = np.maximum(f, 0.01)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase strategy:
    - Phase 1: Quick MD to establish baseline
    - Phase 2: Smoothed FISTA (primary divergent method)
    - Phase 3: MD with different parameters
    - Phase 4: Coordinate descent polish
    - Phase 5: Multi-length exploration
    - Phase 6: Final combined polish
    """
    global _global_best_score, _global_best_seq
    rng = np.random.default_rng(2025)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            _global_best_score = sc
            _global_best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD n={n0} (budget {remaining:.0f}s) ===")
    t1 = min(remaining * 0.18, 250.0)
    if t1 > 30:
        a_out, _ = mirror_descent_adaptive(norm_sum1(best_seq), t1, c0=3.0, threshold_frac=0.99, verbose=True, rng_seed=1)
        try_update(a_out, f"md_p1_n{n0}")

    # === Phase 2: Smoothed FISTA (main workhorse - different gradient structure) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: FISTA Smoothed n={n_best} (budget {remaining:.0f}s) ===")
    t2 = min(remaining * 0.28, 450.0)
    if t2 > 30:
        a_out, _ = fista_smoothed(norm_sum1(best_seq), t2, mu_start=3.0, mu_end=50.0, ramp_iters=5000, verbose=True, rng_seed=2)
        try_update(a_out, f"fista_p2_n{n_best}")

    # === Phase 3: MD with different hyperparameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MD alt params n={n_best} (budget {remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30:
        a_out, _ = mirror_descent_adaptive(norm_sum1(best_seq), t3, c0=5.0, threshold_frac=0.995, verbose=True, rng_seed=3)
        try_update(a_out, f"md_p3_n{n_best}")

    # === Phase 4: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord Descent n={n_best} (budget {remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = coordinate_descent_smart(norm_sum1(best_seq), t4, verbose=True, rng_seed=4)
        try_update(a_out, f"coord_p4_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length (budget {remaining:.0f}s) ===")
        for delta_n, dl in [(300, 0.20), (600, 0.20), (-300, 0.15)]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            n_try = n_best + delta_n
            if n_try < 1200 or n_try > 3500:
                continue
            print(f"  Trying n={n_try}")
            a_try = interpolate_sequence(best_seq, n_try)
            t5 = min(remaining * dl, 150.0)
            if t5 > 25:
                a_out, _ = mirror_descent_adaptive(norm_sum1(a_try), t5, c0=3.0, threshold_frac=0.99, verbose=False, rng_seed=10 + abs(delta_n) // 100)
                try_update(a_out, f"md_ml_n{n_try}")
                # Quick coord polish on new length
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30 and len(best_seq) == n_try:
                    t5c = min(remaining * 0.15, 80.0)
                    if t5c > 15:
                        a_out2, _ = coordinate_descent_smart(norm_sum1(best_seq), t5c, verbose=False, rng_seed=20 + abs(delta_n) // 100)
                        try_update(a_out2, f"coord_ml_n{n_try}")

    # === Phase 6: Final FISTA polish (high mu) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: FISTA polish n={n_best} (budget {remaining:.0f}s) ===")
    if remaining > 30:
        t6 = min(remaining * 0.25, 150.0)
        if t6 > 20:
            a_out, _ = fista_smoothed(norm_sum1(best_seq), t6, mu_start=15.0, mu_end=80.0, ramp_iters=3000, verbose=True, rng_seed=5)
            try_update(a_out, f"fista_polish_n{n_best}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget {remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.30, 120.0)
        if t7 > 10:
            a_out, _ = coordinate_descent_smart(norm_sum1(best_seq), t7, verbose=True, rng_seed=6)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final MD subpolish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.40, 80.0)
        if t8 > 10:
            a_out, _ = mirror_descent_adaptive(norm_sum1(best_seq), t8, c0=1.5, threshold_frac=0.999, verbose=False, rng_seed=7)
            try_update(a_out, f"md_subpolish_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
