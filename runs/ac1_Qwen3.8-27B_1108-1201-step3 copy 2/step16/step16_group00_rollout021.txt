def lp_fw_step(a, M, K_cons=100, solver=None):
    """One LP-FW step: find direction b maximizing sum(b) s.t. conv(a,b)_k <= M for top-K lags."""
    n = len(a)
    if solver is None:
        return None
    
    ac = fast_autocorr(a)
    # Select the K most "binding" constraints (those closest to M)
    K_sel = min(K_cons, 2*n-1)
    
    # We want conv(a,b)_k <= M for the lags k where ac is large
    # conv(a,b)_k = sum_i a[i]*b[k-i]
    # This is linear in b.
    
    # Build constraint matrix: for each selected lag k, the row is a "shifted" a
    # constraint_k: sum_i a[i] * b[k-i] <= M
    # i.e., sum_j b[j] * a[k-j] <= M (reindexing with j = k-i)
    
    top_k = np.argsort(ac)[::-1][:K_sel]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in top_k:
        k_int = int(k)
        # conv(a,b)[k] = sum_{i} a[i]*b[k-i] = sum_{j=0}^{n-1} b[j]*a[k-j] (where a[k-j]=0 if out of range)
        # So the coefficient of b[j] is a[k-j]
        coeffs = np.zeros(n)
        j_vals = np.arange(n)
        a_idx = k_int - j_vals
        valid = (a_idx >= 0) & (a_idx < n)
        coeffs[valid] = a[a_idx[valid]]
        constraints.append(coeffs @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    bs = b_val.sum()
    if bs < 1e-15:
        return None
    return b_val / bs

def lp_fw_optimize(a0, time_limit, K_cons=100, solver=None, verbose=False):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        b_dir = lp_fw_step(y, M, K_cons=K_cons, solver=solver)
        if b_dir is None:
            break
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        My = fast_autocorr(y).max()
        if My < best_m:
            best_m = My
            best_a = y.copy()
        
        Mx = fast_autocorr(x).max()
        if Mx < best_m:
            best_m = Mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            J = 2.0 * n * best_m
            print(f"  LP-FW t={t}: J={J:.6f}")
    
    return best_a, best_m

def coord_descent(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(15, 2*n-1)
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n-1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je+1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            # Score each position by how much adding mass there would increase peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    scores[d] += add
            
            # Prefer positions with low score
            # Also avoid the source
            scores[src] = float('inf')
            dst = int(np.argmin(scores))
            
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-16:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            if a.sum() > 1e-15:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.02
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except:
            continue
    return cp.ECOS
