"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation L-BFGS with log-sum-exp smoothing and FFT-based gradients
2. LP Frank-Wolfe with adaptive constraint selection
3. Multi-restart refinement
4. Coordinate descent polish
Key innovation: FFT-based gradient computation for the smoothed objective,
enabling fast second-order optimization via L-BFGS.
"""

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_conv(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax(x):
    x = np.asarray(x, dtype=np.float64)
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def logsumexp(x):
    m = x.max()
    return m + np.log(np.sum(np.exp(x - m)))

def smooth_obj_and_grad(x, t):
    """
    Smoothed objective and gradient w.r.t. x (log-parameter).
    a = softmax(x), obj = 2*n*t*log(sum(exp(conv(a,a)/t)))
    """
    n = len(x)
    a = softmax(x)
    
    ac = fast_conv(a)
    ac_scaled = ac / t
    lse = logsumexp(ac_scaled)
    obj = 2.0 * n * t * lse
    
    weights = np.exp(ac_scaled - lse)
    
    # grad_a[i] = 4n * sum_{m=0}^{n-1} weights[i+m] * a[m]
    # = 4n * (weights_rev * a)[2n-2-i]
    b_rev = weights[::-1]
    fft_size = 1 << (3 * n).bit_length()
    B = np.fft.rfft(b_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(B * A, fft_size)
    indices = 2 * n - 2 - np.arange(n)
    grad_a = 4.0 * n * conv_result[indices]
    
    dot_a_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot_a_grad)
    
    return obj, grad_x

def continuation_lbfgs(a0, time_limit, t_start=0.05, t_end=0.001, ratio=0.6,
                       max_iter_per_phase=300, verbose=True):
    """Continuation method: decrease temperature, run L-BFGS at each level."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x = np.log(a0)  # log-parameter
    
    best_a = a0.copy()
    best_obj = fast_conv(a0).max() * 2.0 * n  # true objective (not smoothed)
    t0 = time.monotonic()
    
    t = t_start
    phase = 0
    while t > t_end:
        phase += 1
        phase_start = time.monotonic()
        
        if time.monotonic() - t0 > time_limit:
            break
        
        # Run L-BFGS at this temperature
        try:
            result = minimize(
                smooth_obj_and_grad,
                x,
                args=(t,),
                method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': max_iter_per_phase,
                    'gtol': 1e-8,
                    'maxfun': max_iter_per_phase * 2,
                    'iprint': -1
                }
            )
            x = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at t={t:.5f}: {e}")
            break
        
        a = softmax(x)
        true_obj = fast_conv(a).max() * 2.0 * n
        
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
        
        if verbose:
            phase_time = time.monotonic() - phase_start
            n_iter = result.nit if hasattr(result, 'nit') else 0
            print(f"    Phase {phase}: t={t:.5f}, obj={true_obj:.6f}, best={best_obj:.6f}, "
                  f"iter={n_iter}, time={phase_time:.1f}s")
        
        t *= ratio
    
    return best_a, best_obj / (2.0 * n)  # Return max_conv for consistency

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=400, verbose=True):
    """LP Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    ac = fast_conv(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    x = a.copy()  # Nesterov extrapolate
    y = a.copy()  # Current point
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        # Select constraints: top lags
        n_constraints = min(max_constraints, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:n_constraints]
        
        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((n_constraints, n))
        for row_idx, k in enumerate(top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr * (1 + 1e-10)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            x = y_new + beta_t * (y_new - y)
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx > 1e-15:
                x = x / sx
        
        y = y_new
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Simple coordinate descent: try adjusting pairs of coordinates."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 20:
        improved = False
        n_tries = min(100, n // 3)
        
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            
            for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                a_trial = a_trial / st
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m
