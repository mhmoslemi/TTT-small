def fast_autocorr(a):
    """Compute full autocorrelation (conv with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def fast_conv(a, b):
    """Compute full convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(y):
    """Build convolution matrix T where T @ b = conv(y, b). T is (2n-1) x n."""
    n = len(y)
    size = 2 * n - 1
    k = np.arange(size)[:, None]
    j = np.arange(n)[None, :]
    offset = k - j
    mask = (offset >= 0) & (offset < n)
    T = np.where(mask, y[None, :], 0.0)
    return T


def solve_lp(y, M):
    """Solve: max sum(b) s.t. b >= 0, conv(y,b) <= M element-wise."""
    n = len(y)

    b_ub = np.full(2 * n - 1, M)

                      options={'presolve': True})

            if b.sum() > 1e-15:

    return None


def solve_lp_active(y, M, top_k=200):

            y = x.copy()

        grad, M_curr = subgradient_topk(y, t, top_k=top_k)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_restarted = best_a + pert
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.1f}s): M={best_M:.8f}, J={2*n*best_M:.6f}")

    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, verbose=True, use_active=False, top_k_constraints=500):
    """Nesterov-accelerated LP Frank-Wolfe using scipy HiGHS."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    y_prev = a.copy()
    iter_count = 0
    no_improve = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < 1e-14:
            break

        # Solve LP
        if use_active and n > 500:
            b = solve_lp_active(y, M_curr, top_k=min(top_k_constraints, 2*n-1))
        else:
            b = solve_lp(y, M_curr)

        if b is None or b.sum() < 1e-15:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        g = b / b.sum()
        g = np.maximum(g, 0.0)
        s_g = g.sum()
        if s_g < 1e-15:
            break
        g = g / s_g

        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            x_new = g.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x

        y_prev = y
        y = y_new
        x = x_new
        iter_count += 1

        # Track best
        M_y = fast_autocorr(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        M_x = fast_autocorr(x).max()
        if M_x < best_M:
            best_M = M_x
            best_a = x.copy()
            no_improve = 0

        if no_improve > 200:
            # Perturb to escape
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.005 * best_M, n)
            y_pert = best_a + pert
            y_pert = np.maximum(y_pert, 0.0)
            s_p = y_pert.sum()
            if s_p > 1e-15:
                y_pert = y_pert / s_p
                M_p = fast_autocorr(y_pert).max()
                if M_p < best_M * 1.05:
                    y = y_pert
                    y_prev = y_pert
                    x = y_pert
            no_improve = 0

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.1f}s): M={best_M:.8f}, J={2*n*best_M:.6f}")

    return best_a, best_M


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent polish."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = int(np.argmax(ac))

        # Find contributing pairs to the peak
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        js = np.arange(j_start, j_end + 1)
        contrib = a[js] * a[peak_idx - js]
        max_contrib = contrib.max()
        if max_contrib < 1e-20:
            stall += 1
            break

        # Compute sensitivity: how much each position contributes to top peaks
        top_peaks = np.argsort(ac)[::-1][:8]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk = int(pk)
            ps = max(0, pk - n + 1)
            pe = min(n - 1, pk)
            for pos in range(ps, pe + 1):
                sensitivity[pos] += a[pk - pos]

        # Try moving mass from high-contrib to low-sensitivity positions
        src_ranking = np.argsort(contrib)[::-1]
        improved = False
        for rank in src_ranking[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(js[rank])
            candidates = np.argsort(sensitivity)[:max(20, n // 20)]
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    a_try = np.maximum(a_try, 0.0)
                    s_t = a_try.sum()
                    if s_t < 1e-15:
                        continue
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_M - 1e-14:
                        a = a_try
                        best_M = M_try
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDesc ({elapsed:.1f}s): M={best_M:.8f}, J={2*n*best_M:.6f}")

    return best_a, best_M


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_sequence_wrapper(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_sequence_wrapper(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_sequence_wrapper(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_sequence_wrapper(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Subgradient warmup (10% time, max 60s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 1: Subgrad warmup n={n0} ===")
        t1 = min(remaining * 0.08, 60.0)
        a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t1, top_k=40, beta=4.0, verbose=False)
        try_update(a_out, "subgrad_warmup")

    # === Phase 2: LP-FW main (55% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW n={n_cur} (main) ===")
        t2 = min(remaining * 0.55, 900.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, verbose=True, use_active=True, top_k_constraints=600)
        try_update(a_out, f"lp_main_n{n_cur}")

    # === Phase 3: Coordinate polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord polish n={n_cur} ===")
        t3 = min(remaining * 0.10, 120.0)
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 4: LP-FW round 2 (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.20, 300.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t4, verbose=True, use_active=True, top_k_constraints=400)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 5: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        print(f"\n=== Phase 5: Multi-n exploration ===")
        test_ns = []
        for delta in [300, 600, -300, 900, -600]:
            nt = n_best + delta
            if 800 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Exploring n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            # Quick subgrad
            t5a = min(remaining * 0.06, 30.0)
            a_out, _ = nesterov_subgradient(a_inter, t5a, top_k=30, verbose=False)
            try_update(a_out, f"sg_n{nt}")
            # LP-FW
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 50:
                t5b = min(remaining * 0.15, 150.0)
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5b, verbose=False, use_active=True)
                try_update(a_out, f"lp_n{nt}")

    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 6: Final coord polish n={n_best} ===")
        t6 = min(remaining * 0.25, 100.0)
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t6, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 7: Final LP if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(remaining * 0.6, 200.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7, verbose=True, use_active=True, top_k_constraints=300)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_sequence_wrapper(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]


def norm_sum1(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
