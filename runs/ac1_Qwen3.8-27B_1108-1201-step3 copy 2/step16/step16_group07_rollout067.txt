"""
Minimizes  2*n*max(conv(a,a)) / (sum(a))^2  for a non-negative step function.

Strategy (deliberately different in flavor from the prior program):
  * Primary driver: KL-mirror (exponential-gradient / Hedge) descent with a
    Polyak step size and adaptive active-set "temperature".  This is a
    multiplicative update, so it respects a>=0 without a projection step and
    naturally equalises the autocorrelation peaks through the normalisation.
  * Projected subgradient with a running (damped) ergodic average, also on a
    Polyak step, to polish and stabilise.
  * LP best-response (bilinear saddle / game view) with a progressively
    tightened active set, used as a refinement / escape step.
  * Fast coordinate descent with FFT-based autocorrelation for fine polish.
  * Multi-length exploration by resampling the parent to other n values,

All helpers are top level (no closures / no lambdas).  evaluate_sequence is
assumed to be already imported and is called directly.

def fast_autocorr(a):
    """Full linear autocorrelation a*a via a single FFT (length 2n-1)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

    try:

def normalize_to(a, n):
    """Rescale so that sum(a) = sqrt(2 n)  (score-invariant normalisation)."""
    target = np.sqrt(2.0 * n)
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * target / n
    return a * target / s


def weighted_subgradient(a, conv, n, n_active, temperature):
    """
    Weighted subgradient of  max_k (a*a)[k]  w.r.t. a.

    d(a*a)[k] / da[j] = 2 * a[k-j]   when 0 <= k-j < n, else 0.
    If temperature > 0 use a soft Boltzmann weighting over all lags, otherwise
    use the hard top-n_active peaks weighted by their height.
    """
    M = float(conv.max())
    if M < 1e-15:
        return np.zeros(n)

    if temperature is not None and temperature > 0.0:
        gaps = M - conv
        weights = np.exp(-gaps / temperature)
        wsum = weights.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        weights = weights / wsum
        active = np.where(weights > (weights.max() * 1e-3))[0]
    else:
        if n_active is None:
            n_active = max(5, n // 60)
        active = np.argsort(conv)[-n_active:]
        wv = conv[active]
        wsum = wv.sum()
        if wsum < 1e-30:
            wsum = 1.0
        weights = wv / wsum

    g = np.zeros(n)
    for k, w in zip(active, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * (float(w)) * a[k - np.arange(lo, hi)]
    return g


def mirror_step(a, g, eta, s):
    """KL-mirror (exponential) update followed by sum-normalisation."""
    log_a = np.log(np.maximum(a, 1e-300)) - eta * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 0.0)
    st = a_new.sum()
    if st < 1e-15:
        return a.copy()
    return a_new * (s / st)


def proj_step(a, g, eta, s):
    """Euclidean (projected) subgradient step with sum-normalisation."""
    a_new = np.maximum(a - eta * g, 0.0)
    st = a_new.sum()
    if st < 1e-15:
        return a.copy()
    return a_new * (s / st)


def gap_lower_bound(n):
    """Trivial (average) lower bound on  max_k (a*a)[k]  for the normalised case."""
    return 2.0 * n / (2.0 * n - 1.0)

def mirror_descent_optimize(a0, time_limit, verbose=True, seed=0):
    """Exponential-gradient (Hedge) descent with Polyak step + backtracking."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    s = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * s / a.sum()

    f_lb = gap_lower_bound(n)
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    patience = 0
    max_patience = 300

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M = float(conv.max())
        if M < best_score:
            best_score = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # adaptive active-set: start hard+wide, go soft+narrow
        if t < 120:
            n_act = max(12, n // 25)
            temp = None
        elif t < 400:
            n_act = max(6, n // 70)
            temp = 0.02 * M
        else:
            n_act = None
            temp = 0.012 * M

        g = weighted_subgradient(a, conv, n, n_act, temp)
        g2 = float(np.dot(g, g))
        if g2 < 1e-20:
            break

        eta = (M - f_lb) / (g2 + 1e-10)
        eta = float(np.clip(eta, 1e-5, 5.0))

        a_new = mirror_step(a, g, eta, s)
        new_M = float(fast_autocorr(a_new).max())

        if new_M <= M * (1.0 + 1e-9):
            a = a_new
        else:
            # backtrack once
            a_half = mirror_step(a, g, 0.5 * eta, s)
            hM = float(fast_autocorr(a_half).max())
            if hM < M:
                a = a_half
            else:
                # stuck: inject structured multiplicative noise
                a = a * np.exp(0.06 * rng.standard_normal(n))
                a = np.maximum(a, 1e-15)
                a = a * s / a.sum()

        if verbose and (t % 250 == 0):
            print("    MD t=%d (%.0fs): M=%.10f eta=%.5f best=%.10f"
                  % (t, time.monotonic() - t0, M, eta, best_score))

def lp_best_response(a_fixed, M, active_set, solver):
    """LP: max sum(b)  s.t.  (a_fixed*b)[k] <= M (k in active),  b >= 0."""
    n = len(a_fixed)
    kidx = np.asarray(active_set, dtype=int)
    T = np.zeros((len(kidx), n))
    for r, k in enumerate(kidx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[r, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def coordinate_descent(a0, time_limit, verbose=True, seed=0):
    """Fast coordinate descent with FFT autocorrelation + local peak guidance."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    s = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * s / a.sum()

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    size = 2 * n - 1
    no_improve = 0
    t0 = time.monotonic()
    typ = s / n

    while time.monotonic() - t0 < time_limit:
        if no_improve > 7:
            break

        conv = fast_autocorr(a)
        cur_sum = a.sum()
        improved = False

        # bias the scan order toward coordinates near the tallest peaks
        peak_lags = np.argsort(conv)[-max(8, n // 60):]
        priority = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            priority[lo:hi] += 1.0
        order = rng.permutation(n)
        hot = order[np.argsort(priority[order])[-max(1, n // 2):]]
        seq_scan = np.concatenate([hot, order[order[hot] == 0]][:n])

        for j in seq_scan:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            aj = a[j]
            lo_k = max(0, j)
            hi_k = min(size, j + n)
            base_window = a[lo_k - j: hi_k - j]

            best_obj = float(conv.max())
            best_x = aj
            cands = np.unique(np.clip(
                [0.0, 0.3 * aj, 0.6 * aj, 0.9 * aj, aj, 1.4 * aj, 1.9 * aj,
                 0.5 * typ, typ, 2.0 * typ, 4.0 * typ], 0.0, s))

            for x in cands:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                new_conv[lo_k:hi_k] += 2.0 * delta * base_window
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                new_sum = cur_sum + delta
                if new_sum < 1e-15:
                    continue
                scale = s / new_sum
                nm = float((new_conv * scale * scale).max())
                if nm < best_obj:
                    best_obj = nm
                    best_x = x
                    if 2 * j < size and best_obj < (best_obj - 1e-3):
                        pass

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                a = a * s / a.sum()
                ns = float(fast_autocorr(a).max())
                if ns < best_score:
                    best_score = ns
                    best_a = a.copy()
                    conv = fast_autocorr(a)
                    cur_sum = a.sum()
                    improved = True
                else:
                    a = best_a.copy()
                    conv = fast_autocorr(a)
                    cur_sum = a.sum()

        if not improved:
            no_improve += 1
            a = a * (1.0 + 0.02 * rng.standard_normal(n))
            a = np.maximum(a, 1e-15)
            a = a * s / a.sum()
        else:
            no_improve = 0

        if verbose:
            print("    CD (%.0fs): score=%.10f" % (time.monotonic() - t0, best_score))

    return best_a, best_score

def get_solver():
    for s in (cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS):
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print("  LP solver:", s)
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def resample_to(parent, n):
    """Resample a 1-D pattern to length n by linear interpolation (preserves shape)."""
    m = parent.shape[0]
    if m == 1:
        return np.full(n, float(parent[0]))
    xp = np.linspace(0.0, 1.0, m)
    xp_new = np.linspace(0.0, 1.0, n)
    return np.interp(xp_new, xp, parent)


def attempt_update(best_seq, best_score, seq, label, t0):
    """Validate + evaluate seq; return (best_seq, best_score, improved)."""
    if seq is None:
        return best_seq, best_score, False
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)
    if not np.isfinite(seq).all() or seq.sum() < 0.01:
        return best_seq, best_score, False
    seq = seq / seq.sum()
    sc = evaluate_safe(seq)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = seq.copy()
        print("  *** NEW BEST [%s]: %.8f (n=%d, t=%.0fs)"
              % (label, sc, len(seq), time.monotonic() - t0))
        return best_seq, best_score, True
    return best_seq, best_score, False

    rem = time_budget - (time.monotonic() - t0)
    ncur = len(best_seq)
    t3 = min(0.22 * rem, 360.0)
    if t3 > 40:
        print("\n=== P3 LP n=%d budget=%.0fs ===" % (ncur, t3))
        out, _ = lp_optimize(normalize_to(best_seq, ncur), t3, solver,
                             verbose=True, seed=7)
        best_seq, best_score, _ = attempt_update(best_seq, best_score, out,
                                                 "LP_n%d" % ncur, t0)

    # ---- Phase 4: coordinate descent polish -------------------------------- #
    rem = time_budget - (time.monotonic() - t0)
    ncur = len(best_seq)
    t4 = min(0.12 * rem, 150.0)
    if t4 > 20:
        print("\n=== P4 CD n=%d budget=%.0fs ===" % (ncur, t4))
        out, _ = coordinate_descent(normalize_to(best_seq, ncur), t4,
                                    verbose=False, seed=3)
        best_seq, best_score, _ = attempt_update(best_seq, best_score, out,
                                                 "CD_n%d" % ncur, t0)

    # ---- Phase 5: alternate-length exploration ------------------------------ #
    rem = time_budget - (time.monotonic() - t0)
    alt_lengths = [int(x) for x in [0.5 * n_base, 0.75 * n_base,
                                    1.25 * n_base, 1.5 * n_base]
                   if x not in (n_base,) and x >= 200]
    for al in alt_lengths:
        rem = time_budget - (time.monotonic() - t0)
        if rem < 70:
            break
        tA = min(0.5 * rem, 90.0)
        if tA < 25:
            break
        print("\n=== P5 alt n=%d budget=%.0fs ===" % (al, tA))
        base = resample_to(parent, al)
        out, _ = mirror_descent_optimize(normalize_to(base, al), tA,
                                         verbose=False, seed=11)
        best_seq, best_score, _ = attempt_update(best_seq, best_score, out,
                                                 "altMD_n%d" % al, t0)
        rem = time_budget - (time.monotonic() - t0)
        if rem > 30 and len(best_seq) == al:
            out, _ = coordinate_descent(normalize_to(best_seq, al), min(rem * 0.4, 40.0),
                                        verbose=False, seed=13)
            best_seq, best_score, _ = attempt_update(best_seq, best_score, out,
                                                     "altCD_n%d" % al, t0)

    # ---- Phase 6: structured perturbation restarts on the incumbent -------- #
    rem = time_budget - (time.monotonic() - t0)
    ncur = len(best_seq)
    n_restarts = 2
    for r in range(n_restarts):
        rem = time_budget - (time.monotonic() - t0)
        if rem < 50:
            break
        scale = 0.012 + 0.018 * rng.random()
        noise = 1.0 + scale * rng.standard_normal(ncur)
        noise = np.maximum(noise, 0.25)
        pert = best_seq * noise
        pert = pert / pert.sum()
        tR = min(0.35 * rem, 130.0)
        if tR < 25:
            break
        print("\n=== P6 restart%d n=%d budget=%.0fs ===" % (r, ncur, tR))
        out, _ = mirror_descent_optimize(normalize_to(pert, ncur), tR,
                                         verbose=False, seed=100 + r)
        best_seq, best_score, _ = attempt_update(best_seq, best_score, out,
                                                 "Pert%d_n%d" % (r, ncur), t0)

    # ---- Phase 7: final LP + CD micro-polish ------------------------------- #
    rem = time_budget - (time.monotonic() - t0)
    ncur = len(best_seq)
    if rem > 60:
        tL = min(0.5 * rem, 120.0)
        if tL > 30:
            print("\n=== P7 final LP n=%d ===" % ncur)
            out, _ = lp_optimize(normalize_to(best_seq, ncur), tL, solver,
                                 verbose=False, seed=999)
            best_seq, best_score, _ = attempt_update(best_seq, best_score, out,
                                                     "LPfin_n%d" % ncur)
    rem = time_budget - (time.monotonic() - t0)
    ncur = len(best_seq)
    if rem > 25:
        out, _ = coordinate_descent(normalize_to(best_seq, ncur),
                                    min(rem * 0.7, 60.0), verbose=False, seed=77)
        best_seq, best_score, _ = attempt_update(best_seq, best_score, out,
                                                 "CDfin_n%d" % ncur)

    # ---- final safety ------------------------------------------------------- #
    if best_seq is None or not np.isfinite(best_seq).all() or best_seq.sum() < 0.01:
        best_seq = np.ones(1600)
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("\nFINAL: score=%.8f n=%d time=%.0fs"
          % (best_score, len(best_seq), time.monotonic() - t0))
    return [float(x) for x in best_seq]
