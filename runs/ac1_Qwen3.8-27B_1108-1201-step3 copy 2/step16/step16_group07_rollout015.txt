def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def subgradient_at_k(a, k, n):
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2:hi2] += a[idx - k]
    return grad

def get_active_set(conv, n, progress=1.0):
    """Get active set based on progress (0=loose, 1=tight)."""
    M = conv.max()
    # Progressive tightening
    tol = 0.15 * (1.0 - progress) + 0.01 * progress
    threshold = M * (1.0 - tol)
    active = np.where(conv >= threshold)[0]
    # Always include top peaks
    n_top = max(10, n // 50)
    top_k = np.argsort(conv)[-n_top:]
    active = np.unique(np.concatenate([active, top_k]))
    # Cap size for efficiency
    max_size = int(1.2 * n)
    if len(active) > max_size:
        conv_at_active = conv[active]
        keep = np.argsort(conv_at_active)[-max_size:]
        active = active[keep]
    return active

def lp_solve(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    size = len(active_set)
    if size == 0:
        return None, None
    
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(active_set[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    
    # Try to get duals
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, duals

def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    """LP best-response with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    conv_size = 2 * n - 1
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    n_lp = 0
    n_iters = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        # Update active set periodically
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            active_set = get_active_set(conv_y, n, progress)
        
        b, duals = lp_solve(y, M_curr, active_set, solver)
        if b is None:
            # Fallback: use all constraints
            active_set = np.arange(conv_size)
            b, duals = lp_solve(y, M_curr, active_set, solver)
            if b is None:
                if t % 20 == 0:
                    active_set = None
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_lp += 1
        
        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        n_iters += 1
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, verbose=False, seed=42):
    """Final polish using multi-peak subgradient with adaptive step."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    step = 0.3 / n ** 0.3
    patience = 0
    max_patience = 300
    rng = np.random.default_rng(seed)
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Peak selection with diversity
        n_peaks = max(5, min(15, n // 60))
        # Use top peaks weighted by their excess above threshold
        threshold = M_curr * 0.9
        candidates = np.where(conv >= threshold)[0]
        if len(candidates) < n_peaks:
            candidates = np.argsort(conv)[-n_peaks:]
        else:
            # Sample from candidates
            if len(candidates) > 2 * n_peaks:
                idx = rng.choice(len(candidates), n_peaks * 2, replace=False)
                candidates = candidates[idx]
            n_take = min(n_peaks, len(candidates))
            perm = rng.permutation(len(candidates))[:n_take]
            candidates = candidates[perm]
        
        weights = np.exp(-5.0 * (M_curr - conv[candidates]) / max(M_curr * 0.05, 1e-10))
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, candidates):
            grad += w * subgradient_at_k(a, int(k), n)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        
        # Projected gradient step
        a_trial = a - step * grad * n ** 0.3
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 2.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-12:
                break
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with FFT-based autocorrelation updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 3
    
    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break
        no_improve = 0
        
        conv = fast_autocorr(a)
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            
            # Try a few candidate values
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue
            
            candidates = [0.0, aj * 0.5, aj * 1.5, aj * 2.0, norm / n * 0.5, norm / n * 2.0]
            if aj < norm / n:
                candidates.append(norm / n)
            candidates = sorted(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))
            
            best_x = aj
            best_obj = np.inf
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                
                # Renormalize
                new_sum = a.sum() - aj + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                new_max = np.max(new_conv) * scale * scale
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = new_score * np.ones(size)
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
        
        if not improved:
            no_improve += 1
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def try_update(best_seq, best_score, a, label, t_global):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

    return best_seq, best_score, False

def perturb_seq(seq, scale=0.05, seed=0):
    """Create a perturbed version of the sequence."""
    rng = np.random.default_rng(seed)
    n = len(seq)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    perturbed = seq * noise
    s = perturbed.sum()
    if s < 1e-15:
        return seq.copy()
    return perturbed / s

- y_k >= 0
- sum_k y_k * M = sum(b) (complementary slackness)
- For each j: sum_k y_k * T[k,j] = 1 (if b_j > 0) or >= 1 (if b_j = 0)

where T[k,j] = a[k-j] (the (k,j) entry of the constraint matrix).

This means: sum_k y_k * a[k-j] = 1 for all j where a_j > 0.

In other words, the convolution (y * a)[j] = 1 for all j in the support of a.

This is a strong condition! It means that at the optimal fixed point, the convolution of the "dual weights" y with a is constant (= 1) on the support of a.

This gives us a way to verify (and potentially improve) a candidate solution:
1. Compute the autocorrelation c = a * a
2. Find the active set (peaks)
3. Solve the dual LP to get y
4. Check if (y * a)[j] is approximately constant on the support
5. If not, adjust a to make it more constant

This is essentially the KKT condition, and projecting onto it could give a good polishing step.

Let me implement this "KKT-based polish":

```python
def kkt_polish(a0, time_limit, solver, verbose=False):
    """
    Polish using KKT conditions: at optimum, the dual variables y satisfy
    (y * a)[j] = constant on support of a.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    step = 0.1 / n ** 0.2
    patience = 0
    max_patience = 200
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Get active set
        active_set = get_active_set(conv, n, progress=1.0)
        
        # Solve LP to get duals
        b, duals = lp_solve(a, M_curr, active_set, solver)
        if b is None or duals is None:
            patience += 1
            continue
        
        # Use duals to compute the "KKT residual"
        # At optimum: sum_k y_k * T[k,j] should be constant (=1/something) on support
        # T[k,j] = a[k-j] for valid k-j
        y = np.maximum(duals, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            patience += 1
            continue
        y = y / y_sum  # Normalize
        
        # Compute (y * a) at each position j
        # (y * a)[j] = sum_k y[k_index] * a[k - j] where k_index maps to active_set[k]
        kkt_val = np.zeros(n)
        for idx, k in enumerate(active_set):
            if y[idx] < 1e-15:
                continue
            # T[idx, j] = a[k-j] for valid j
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                kkt_val[lo:hi] += y[idx] * a[k - np.arange(lo, hi)]
        
        # On support, kkt_val should be constant
        support_thresh = 0.01 * a.max()
        support = np.where(a > support_thresh)[0]
        if len(support) < 3:
            break
        
        kkt_on_support = kkt_val[support]
        kkt_mean = np.mean(kkt_on_support)
        kkt_std = np.std(kkt_on_support)
        
        if kkt_std < 1e-10:
            # Already at KKT
            break
        
        # Direction: reduce where kkt_val is high, increase where low
        # (This should reduce the variance of kkt_val on support)
        direction = np.zeros(n)
        direction[support] = kkt_val[support] - kkt_mean
        
        # Apply step
        a_trial = a - step * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience
