def fast_autocorr(a):
    """Compute full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""

def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def subgradient_at_k(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a.
    grad[j] = 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)], else 0."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        grad[lo:hi] = 2.0 * a[k - np.arange(lo, hi)]
    return grad


def lp_best_response_max_sum(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None
    
    # Build constraint matrix (sparse-like, but dense for simplicity with n up to ~2000)
    # T[idx, j] = a_fixed[k - j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_min_max_autocorr(a_fixed, S, active_set, solver):
    """Solve LP: min M s.t. (a_fixed*b)[k] <= M for k in active_set, sum(b)=S, b>=0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None
    
    T = np.zeros((size, n))
    for idx in range(size):

        lo = max(0, k - n + 1)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == S, M_var >= 0]

    return b_val, M_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):

    rng = np.random.default_rng(seed)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 40
    n_iters = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Progressive active set
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            tol = 0.10 * (1.0 - progress) + 0.005 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, n // 60)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > int(1.2 * n):
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-int(1.2 * n):]
                active_set = active_set[keep]

        b = lp_best_response_max_sum(y, M_curr, active_set, solver)

        if b is None:
            if t % 5 == 0:
                active_set = np.arange(conv_size)
                b = lp_best_response_max_sum(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.92)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=False, seed=42):
    """Mirror descent with entropic regularization on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Entropic mirror descent: a_new[j] ∝ a[j] * exp(-η * grad[j])
    # Then normalize to sum = norm
    eta = 0.5 / np.sqrt(n)
    patience = 0
    max_patience = 500

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Pick active peaks
        n_peaks = max(5, min(20, n // 50))
        threshold = M_curr * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(conv)[-n_peaks:]
        
        # Weight by peak value (emphasize the largest)
        weights = np.exp(-10.0 * (M_curr - conv[active.astype(int)]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active.astype(int)):
            grad += w * subgradient_at_k(a, int(k), n)
        
        # Normalize gradient for stable updates
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm * n ** 0.5  # Scale
        
        # Entropic update: a_new ∝ a * exp(-eta * grad)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        # Shift for numerical stability
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-14:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            eta = min(eta * 1.01, 5.0)
        else:
            patience += 1
            eta *= 0.98
            if eta < 1e-10:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent t={t} ({elapsed:.0f}s): best={best_score:.10f}, eta={eta:.6f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=False, seed=1):
    """Weighted subgradient descent with adaptive step."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 800

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        n_peaks = max(3, min(15, n // 60))
        top_indices = np.argsort(conv)[-n_peaks:]

        weights = weights / weights.sum()

            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()

        else:

        if verbose and t % 2000 == 0:

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with efficient updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 5

    n_passes = 0
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes > 200:
            break
        improved = False
        conv = fast_autocorr(a).copy()

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = norm - aj

            lo = j
            hi = min(size, j + n)
            if lo >= hi:
                continue

            # Try a few candidate values
            typical = norm / n
            candidates = [0.0, aj * 0.5, aj * 1.5, typical * 0.3, typical, typical * 2.0, typical * 4.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))

            best_x = aj
            best_obj = best_score  # Use current best as threshold

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration (primary optimizer)
2. Mirror descent (entropic) for fast complementary steps
3. Weighted subgradient descent
4. Coordinate descent polishing
5. Multi-phase time allocation with progressive active-set tightening
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.CBC]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = 'ECOS'
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def subgradient_at_k(a, k, n):
    """Subgradient of (a*a)[k] w.r.t. a. grad[j] = 2*a[k-j] for valid j."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        grad[lo:hi] = 2.0 * a[k - np.arange(lo, hi)]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with progressive active-set tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 35
    n_iters = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 25 == 0:
            progress = min(1.0, t / 600.0)
            tol = 0.10 * (1.0 - progress) + 0.003 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, n // 50)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            max_active = int(1.3 * n)
            if len(active_set) > max_active:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_active:]
                active_set = active_set[keep]

        b = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            if t % 4 == 0:
                active_set = np.arange(conv_size)
                b = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.91)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average check
        if t % 40 == 0:
            a_damped = 0.75 * best_a + 0.25 * a
            a_damped = np.maximum(a_damped, 0.0)
            s_d = a_damped.sum()
            if s_d > 1e-15:
                a_damped = a_damped * norm / s_d
                score_d = fast_autocorr(a_damped).max()
                if score_d < best_score:
                    best_score = score_d
                    best_a = a_damped.copy()
                    stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=False, seed=42):
    """Mirror descent with entropic regularization on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.3 / np.sqrt(n)
    patience = 0
    max_patience = 400

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        n_peaks = max(5, min(25, n // 40))
        threshold = M_curr * 0.94
        active = np.where(conv >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(conv)[-n_peaks:]

        weights = np.exp(-12.0 * (M_curr - conv[active.astype(int)]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, active.astype(int)):
            grad += w * subgradient_at_k(a, int(k), n)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm * np.sqrt(n)

        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()

            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            eta = min(eta * 1.008, 4.0)
        else:
            patience += 1
            eta *= 0.985
            if eta < 1e-12:
                break

        if verbose and t % 400 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror t={t} ({elapsed:.0f}s): best={best_score:.10f}, eta={eta:.5f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=False, seed=1):
    """Weighted subgradient descent with adaptive step and diversity."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.8 / np.sqrt(n)
    patience = 0
    max_patience = 700

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        n_peaks = max(3, min(12, n // 50))
        top_indices = np.argsort(conv)[-n_peaks:]

        if t % 12 == 0:
            thresh = M_curr * (0.88 + 0.08 * rng.random())
            cands = np.where(conv >= thresh)[0]
            if len(cands) > 0:
                k_rand = cands[rng.integers(len(cands))]
                top_indices = np.unique(np.append(top_indices, k_rand))

        weights = np.exp(-8.0 * (M_curr - conv[top_indices.astype(int)]) / max(M_curr * 0.02, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k(a, int(k), n)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn * np.sqrt(n)

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()

            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.004, 2.5)
        else:
            patience += 1
            step *= 0.982
            if step < 1e-12:
                break

        if verbose and t % 1500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.5f}")

    rng = np.random.default_rng(seed)
    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        if n_passes > 300:
            break
        improved = False

        order = rng.permutation(n)

            if time.monotonic() - t0 > time_limit:

                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale

def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence. Returns (best_seq, best_score, improved)."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base, endpoint=False)
        best_seq = (0.5 + 0.3 * np.cos(2 * np.pi * x) +
                    0.15 * np.cos(4 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x))
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.33, 520.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp1_n{n_base}", t_global)

    # === Phase 2: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.13, 180.0)
    if t2 > 25:
        print(f"\n=== Phase 2: Mirror descent n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = mirror_descent_optimize(a_norm, t2, verbose=True, seed=7)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"mirror_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=7777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 4: Subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Subgradient n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_optimize(a_norm, t4, verbose=True, seed=3)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_n{n_cur}", t_global)

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t5, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 6: LP Nesterov round 3 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.14, 180.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP Nesterov r3 n={n_cur}, budget={t6:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False, seed=42424)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp3_n{n_cur}", t_global)

    # === Phase 7: Perturbation restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7_total = min(remaining * 0.15, 150.0)
    if t7_total > 40:
        t7 = t7_total / 2
        if t7 > 20:
            print(f"\n=== Phase 7: Perturbation restart n={n_cur} ===")
            a_out = perturb_and_optimize(best_seq, t7, solver, rng, seed=100)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert_n{n_cur}", t_global)

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 25:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final CD n={n_cur} ===")
            a_norm = normalize_to(best_seq, n_cur)
            a_out, _ = coordinate_descent(a_norm, t8, verbose=False, seed=99)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # Final micro subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.5, 30.0)
        if t9 > 10:
            a_norm = normalize_to(best_seq, n_cur)
            a_out, _ = subgradient_optimize(a_norm, t9, verbose=False, seed=314)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subgFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fallback = 1600
        return [1.0 / n_fallback] * n_fallback

    return [float(x) for x in best_seq]
