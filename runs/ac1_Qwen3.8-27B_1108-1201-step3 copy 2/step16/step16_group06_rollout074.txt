def fast_conv(a, b=None):
    """FFT-based linear convolution."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-14:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        x_prev = x
        iter_count += 1

        # Check improvement
        my = fast_conv(y).max()
        if my < best_M - 1e-14:
            best_M = my
            best_a = y.copy()
            stall_count = 0
        else:
            mx = fast_conv(x).max()
            if mx < best_M - 1e-14:
                best_M = mx
                best_a = x.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Perturbation to escape local minima
        if stall_count > 200:
            stall_count = 0
            pert = rng.normal(0, noise_level, n)
            y_pert = best_a * (1.0 + pert)
            y_pert = np.maximum(y_pert, 0.0)
            sp = y_pert.sum()
            if sp > 1e-14:
                y_pert = y_pert / sp
                Mp = fast_conv(y_pert).max()
                if Mp < best_M:
                    best_M = Mp
                    best_a = y_pert.copy()
                    y = y_pert.copy()
                    x = y_pert.copy()
            noise_level = max(0.001, noise_level * 0.9)

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"    LP-FW t={t} ({elapsed:.1f}s): score={score:.6f}, M={best_M:.8f}, K={K}")

    return best_a, best_M


def subgradient_polish(a_init, time_limit, verbose=True, seed=123):
    """Nesterov-accelerated subgradient descent targeting the peak."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 2.0
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5) if t > 1 else 0.0
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        conv_y = fast_conv(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient of conv(a,a)[k] w.r.t. a
        g = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        jj = np.arange(j_min, j_max + 1)
        g[jj] = y[jj] + y[k_star - jj]

        eta = gamma / (np.sqrt(t) + 1.0)
        a_new = y - eta * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-14:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.02, n)
            x_pert = best_a * (1.0 + pert)
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            x = x_pert.copy()
            x_prev = x_pert.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_M
            print(f"    SGD t={t} ({elapsed:.1f}s): score={score:.6f}")

    return best_a, best_M


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def coordinate_polish(a_init, time_limit, verbose=True, seed=77):
    """Targeted coordinate descent: move mass away from peak-contributing positions."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv_a = fast_conv(a)
        peak_idx = int(np.argmax(conv_a))
        M_curr = conv_a[peak_idx]

        # Compute contribution of each position to the peak
        j_min = max(0, peak_idx - n + 1)
        j_max = min(peak_idx, n - 1)
        jj = np.arange(j_min, j_max + 1)
        contrib = a[jj] * a[peak_idx - jj]

        if contrib.max() < 1e-20:
            stall += 1
            break

        # Sensitivity: how much each position contributes to top peaks
        top_peaks = np.argsort(conv_a)[::-1][:15]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            i_min = max(0, pk_int - n + 1)
            i_max = min(pk_int, n - 1)
            ii = np.arange(i_min, i_max + 1)
            sensitivity[ii] += a[pk_int - ii]

        improved = False
        src_ranking = np.argsort(contrib)[::-1][:5]

        for si in src_ranking:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(jj[si])

            # Find low-sensitivity destinations
            dst_candidates = np.argsort(sensitivity)[:max(30, n // 20)]
            for dst in dst_candidates:
                if dst == src or int(dst) == int(src):
                    continue
                for frac in [0.003, 0.007, 0.015, 0.04]:
                    amt = a[int(src)] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[int(src)] -= amt
                    a_trial[int(dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-14:
                        continue
                    a_trial = a_trial / st
                    Mt = fast_conv(a_trial).max()
                    if Mt < best_M - 1e-15:
                        a = a_trial
                        best_M = Mt
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                amt = a[i] * rng.uniform(0.002, 0.01)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                sa = a.sum()
                if sa > 1e-14:
                    a = a / sa
                    Ma = fast_conv(a).max()
                    if Ma < best_M:
                        best_M = Ma
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        score = 2.0 * n * best_M
        print(f"    CoordPD ({elapsed:.1f}s): score={score:.6f}")

    return best_a, best_M


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    - LP-FW with top-K constraints (primary)
    - Multiple K values and step sizes
    - Subgradient polish
    - Coordinate descent
    - Multi-length exploration
    - Simulated annealing restarts
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        if a is None or len(a) == 0 or len(a) < 10:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    a_start = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 100 and np.isfinite(p).all() and p.sum() > 0.01:
            a_start = normalize(p)
            ps = evaluate_safe(a_start)
            print(f"Parent: n={len(a_start)}, score={ps:.6f}")
            best_seq = a_start.copy()
            best_score = ps
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n_init = 2400
        x = np.linspace(0, 1, n_init)
        a0 = np.sin(np.pi * x) ** 2
        a0 = normalize(a0)
        best_seq = a0.copy()
        best_score = evaluate_safe(a0)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Quick subgradient warmup (30s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 1: SGD warmup n={n0} ===")
        t1 = min(remaining * 0.04, 40.0)
        a_out, _ = subgradient_polish(best_seq, t1, verbose=False, seed=42)
        try_update(a_out, "sgd_warmup")

    # === Phase 2: LP-FW with K=50 (main, ~40% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW K=50 n={n_cur} ===")
        t2 = min(remaining * 0.35, 600.0)
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, K=50, verbose=True, seed=1)
        try_update(a_out, "lp_K50")

    # === Phase 3: LP-FW with K=30 (different constraint set) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LP-FW K=30 n={n_cur} ===")
        t3 = min(remaining * 0.20, 350.0)
        a_out, _ = lp_frank_wolfe(best_seq, t3, solver, K=30, verbose=True, seed=2)
        try_update(a_out, "lp_K30")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: CoordPD n={n_cur} ===")
        t4 = min(remaining * 0.08, 100.0)
        a_out, _ = coordinate_polish(best_seq, t4, verbose=True, seed=55)
        try_update(a_out, "coord")

    # === Phase 5: LP-FW with K=80 ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: LP-FW K=80 n={n_cur} ===")
        t5 = min(remaining * 0.20, 300.0)
        a_out, _ = lp_frank_wolfe(best_seq, t5, solver, K=80, verbose=True, seed=3)
        try_update(a_out, "lp_K80")

    # === Phase 6: SGD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 6: SGD n={n_cur} ===")
        t6 = min(remaining * 0.10, 120.0)
        a_out, _ = subgradient_polish(best_seq, t6, verbose=True, seed=99)
        try_update(a_out, "sgd_polish")

    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_long = n_cur + 400
    if n_long > n_cur and remaining > 120:
        print(f"\n=== Phase 7: Expand to n={n_long} ===")
        a_long = interpolate_seq(best_seq, n_long)
        t7s = min(remaining * 0.06, 50.0)
        a_out, _ = subgradient_polish(a_long, t7s, verbose=False, seed=111)
        try_update(a_out, f"sgd_n{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7l = min(remaining * 0.25, 250.0)
            a_out, _ = lp_frank_wolfe(best_seq, t7l, solver, K=50, verbose=True, seed=4)
            try_update(a_out, f"lp_n{n_long}")

    # === Phase 8: Try shorter sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = n_cur - 200
    if n_short >= 800 and remaining > 80:
        print(f"\n=== Phase 8: Shorten to n={n_short} ===")
        a_short = interpolate_seq(best_seq, n_short)
        t8 = min(remaining * 0.20, 200.0)
        a_out, _ = lp_frank_wolfe(a_short, t8, solver, K=40, verbose=False, seed=5)
        try_update(a_out, f"lp_n{n_short}")

    # === Phase 9: Final LP-FW + coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: Final LP n={n_best} ===")
        t9 = min(remaining * 0.45, 300.0)
        a_out, _ = lp_frank_wolfe(best_seq, t9, solver, K=40, verbose=True, seed=10)
        try_update(a_out, "lp_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 10: Final coord n={n_best} ===")
        t10 = min(remaining * 0.5, 150.0)
        a_out, _ = coordinate_polish(best_seq, t10, verbose=True, seed=200)
        try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
