def fast_conv(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * ac.max() / (s * s)

def fast_conv(a):
    """Compute linear convolution a*a via FFT. O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_from_conv(ac, a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * ac.max() / (s * s)

def get_sparse_structure(n):
    """Precompute sparse structure for convolution matrix A (2n-1 x n)."""
    key = n
    if key in _global_sparse_structure:
        return _global_sparse_structure[key]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    _global_sparse_structure[key] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _global_sparse_structure[key]

def build_conv_matrix(y, n):
    """Build sparse convolution matrix from y."""
    rows, cols = get_sparse_structure(n)
    # A[k, j] = y[j] if 0 <= k-j < n
    # For each (k,j) pair, the value is y[j]
    vals = y[cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_full(y, M, solver):
    """Solve LP: max sum(b) s.t. conv(y,b)[k] <= M for all k, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    A = build_conv_matrix(y, n)
    constraints = [A @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

        coeffs = y[j_start:j_end + 1]

def spectral_equalize(a, n_iter=30):
    """Flatten power spectrum via iterative DFT."""
    a = normalize_nonneg(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + 1e-12)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a

def lp_france_wolfe(a_init, time_limit, solver, top_k_mode='full', top_k=200, verbose=True):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a_init)
    a = normalize_nonneg(a_init)
    best_a = a.copy()
    best_obj = obj_from_conv(fast_conv(a), a)
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac_y = fast_conv(y)
        M_curr = ac_y.max()

        if 2.0 * n * M_curr < best_obj:
            best_obj = 2.0 * n * M_curr
            best_a = y.copy()

        # Solve LP
        if top_k_mode == 'full':
            direction = solve_lp_full(y, M_curr, solver)
        else:
            direction = solve_lp_topk(y, M_curr, solver, top_k)

        if direction is None:
            continue

        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        mx = fast_conv(x).max()
        if 2.0 * n * mx < best_obj:
            best_obj = 2.0 * n * mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFW t={t} ({elapsed:.1f}s): obj={best_obj:.8f}")

    return best_a, best_obj
