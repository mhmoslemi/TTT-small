_csr_col_indices = None  
_csr_row_ptr = None
_structure_n = 0

def _build_structure(n):
    global _csr_data_indices, _csr_col_indices, _csr_row_ptr, _structure_n
    if _structure_n == n:
        return
    _structure_n = n
    data_idx = []
    col_idx = []
    row_ptr = [0]
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            data_idx.append(k-j)
            col_idx.append(j)
        row_ptr.append(len(data_idx))
    _csr_data_indices = np.array(data_idx, dtype=np.int32)
    _csr_col_indices = np.array(col_idx, dtype=np.int32)
    _csr_row_ptr = np.array(row_ptr, dtype=np.int32)

def _build_A(y):

    _build_structure(n)
    data = y[_csr_data_indices]
    return sp.csr_matrix((data, _csr_col_indices, _csr_row_ptr), shape=(2*n-1, n))

def solve_lp_min_peak(y, solver):
    """Solve: min t s.t. (a*y)[k] <= t for all k, sum(a)=1, a>=0."""
    n = len(y)
    A = _build_A(y)
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [
        A @ a_var <= t_var,
        cp.sum(a_var) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None
    a_val = a_val / s
    return a_val, t_var.value

_global_solver = None
_csr_cache = {}

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):

    return a / s

def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_smoothed_subgrad(y, beta):
    """Compute smoothed subgradient of max_k conv(y,y)[k] using log-sum-exp."""
    n = len(y)
    ac = fast_autocorr(y)
    # Softmax weights
    max_ac = ac.max()
    exp_ac = np.exp(beta * (ac - max_ac))
    lam = exp_ac / exp_ac.sum()
    # g[i] = 2 * sum_k lam[k] * y[k-i] = 2 * (lam cross-correlated with y)[i]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    L = np.fft.rfft(lam, fft_size)
    Y = np.fft.rfft(y, fft_size)
    g = 2.0 * np.fft.irfft(L * np.conj(Y), fft_size)[:n]
    return g, ac
