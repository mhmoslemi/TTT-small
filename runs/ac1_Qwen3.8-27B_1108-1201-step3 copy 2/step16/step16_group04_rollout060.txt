5. a_new = IFFT(A_new)[:n], project to non-negative.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed L-BFGS with log-sum-exp and softmax parameterization (FFT-based gradient)
2. Direct projected gradient with adaptive step size
3. Spectral domain shaping iterations
4. Local coordinate descent with pair moves
5. Multi-start with diverse initializations

import numpy as np
from scipy.fft import fft, ifft
import time

def fast_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A, fft_size).real[:size]

def softmax(w):
    """Stable softmax."""
    w = w - w.max()
    e = np.exp(w)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(w) / len(w)
    return e / s

def log_sum_exp(x, beta):
    """Compute beta * LSE(x) = log(sum exp(beta * x)) stably."""
    mx = x.max()
    return beta * (mx + np.log(np.sum(np.exp(beta * (x - mx)))))

def compute_p(c, beta, lse_cache=None):
    """Compute p[k] = exp(beta*c[k]) / sum(exp(beta*c[k]))."""
    if lse_cache is not None:
        max_c, log_sum = lse_cache
    else:
        max_c = c.max()
        log_sum = max_c + np.log(np.sum(np.exp(beta * (c - max_c))))
    return np.exp(beta * c - (log_sum + max_c)), (max_c, log_sum)

def grad_smooth_a(a, beta):
    """Gradient of (2n/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    sum_exp = exp_bc.sum()
    p = exp_bc / sum_exp
    n_conv = 2 * n - 1
    fft_size = 1 << (n_conv - 1).bit_length()
    P = fft(p, fft_size)
    A = fft(a, fft_size)
    cross = ifft(P * np.conj(A), fft_size).real[:n]
    g_a = 4.0 * n * cross
    obj = (2.0 * n / beta) * (max_c + np.log(sum_exp))
    return obj, g_a, c

def obj_val(a):
    """Compute the actual objective 2*n*max(c)/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def obj_and_grad_smooth_w(w, beta):
    """Objective and gradient w.r.t. w for softmax parameterization."""
    a = softmax(w)
    obj, g_a, c = grad_smooth_a(a, beta)
    g_w = a * (g_a - np.dot(a, g_a))
    return obj, g_w

def lbfgs_optimize(a0, beta_init=5000.0, beta_max=80000.0, time_limit=120.0, 
                    max_iter=500, tol=1e-10, verbose=True):
    """
    Optimize using scipy L-BFGS-B on softmax parameterization with adaptive beta.
    """
    from scipy.optimize import minimize as sp_minimize
    
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    w = np.log(a)  # Initial w such that softmax(w) ≈ a
    
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    
    beta = beta_init
    
    def func_grad(w_in):
        obj, grad = obj_and_grad_smooth_w(w_in, beta)
        return obj, grad
    
    current_beta = beta_init
    x = w.copy()
    
    for stage in range(6):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # L-BFGS iteration
        try:
            res = sp_minimize(func_grad, x, method='L-BFGS-B', jac=True,
                             options={'maxiter': max_iter, 'gtol': tol, 
                                     'maxfun': 500, 'iprint': -1,
                                     'ftol': 1e-12, 'maxls': 50})
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error: {e}")
            break
        
        a_new = softmax(res.x)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_true = obj_val(a_new)
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a_new.copy()
            x = res.x.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      L-BFGS stage {stage} ({elapsed:.1f}s): objective={best_obj:.6f}")
        
        # Increase beta
        current_beta = min(current_beta * 2.0, beta_max)
        beta = current_beta
        
        if res.success and res.jac is not None:
            if np.max(np.abs(res.jac)) < 1e-8:
                break
    
    return best_a, best_obj

def spectral_shape(a, n_iter=100, time_limit=30.0, verbose=False):
    """Spectral domain shaping: iterate to flatten the convolution."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:

    a = a / s
    
    n_conv = 2 * n - 1
    fft_size = 1 << (n_conv - 1).bit_length()
    
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = fft(a, fft_size)
        A2 = A * A
        c = ifft(A2, fft_size).real[:n_conv]
        
        # Target: flat c
        c_mean = c.sum() / n_conv  # Should be ~1/s^2 but s=1 so ~1/n_conv
        # Actually c.sum() = (sum(a))^2 = 1
        
        # Compute ideal C
        c_ideal = np.zeros(fft_size)
        c_ideal[:n_conv] = 1.0 / n_conv
        C_ideal = fft(c_ideal)
        
        C = fft(c, fft_size)
        
        # Update: scale A to move C toward C_ideal
        # A_new^2 = A^2 * (C_ideal / C)^(some fraction)
        # In log domain: log|A_new|^2 = log|A|^2 + alpha * (log|C_ideal| - log|C|)
        # Phase: angle(A_new) = angle(A) + alpha * (angle(C_ideal) - angle(C)) / 2
        
        # Simpler: 
        ratio = C_ideal / (C + 1e-30)
        # Clip the magnitude of ratio
        mag_ratio = np.abs(ratio)
        mag_ratio = np.clip(mag_ratio, 0.01, 100.0)
        phase_ratio = np.exp(1j * np.angle(ratio))
        ratio_clipped = mag_ratio * phase_ratio
        
        # A_new = A * sqrt(ratio_clipped)
        sqrt_ratio = np.sqrt(ratio_clipped)
        A_new = A * sqrt_ratio
        
        a_new = ifft(A_new, fft_size).real[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        # Blend for stability
        a = 0.5 * a + 0.5 * a_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        obj_new = obj_val(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
    
    if verbose:
        print(f"    Spectral ({time.monotonic()-t0:.1f}s): objective={best_obj:.6f}")
    return best_a, best_obj

def coordinate_descent(a0, time_limit=60.0, verbose=False, rng_seed=42):

    n = len(a0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:

    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        # Find top few peaks
        top_k = np.argsort(c)[::-1][:min(10, n_conv)]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            
            # Find the pair (j, k-j) with the largest product
            idx = np.arange(j_start, j_end + 1)
            products = a[idx] * a[k_int - idx]
            max_pair_idx = idx[np.argmax(products)]
            
            # Try to reduce this pair by moving mass
            # Option 1: reduce a[max_pair_idx] and add to a random position
            src = int(max_pair_idx)
            if a[src] < 1e-15:
                continue
            
            # Find a good destination
            # We want to add mass where it doesn't increase other peaks much
            # Try several random destinations
            candidates = rng.choice(n, size=min(100, n), replace=False)
            
            best_dst = -1
            best_score = best_obj
            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
