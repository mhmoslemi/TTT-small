Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

3. Multi-length interpolation and exploration
4. Direct evaluation for final verification

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad(a, w):
    """Compute gradient of sum(w * conv(a,a)) w.r.t. a."""
    n = len(a)
    # w has length 2n-1
    term1 = np.correlate(w, a, mode='full')[n-1 : 2*n-1]
    term2 = np.convolve(w, a, mode='full')[0 : n]
    return term1 + term2

def softmax_weights(b, tau):
    """Compute soft-max weights for the autocorrelation."""
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    w = exp_b / exp_b.sum()
    return w

def adam_optimize(a, tau_schedule, n_steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                  eval_fn=None, time_limit=None, t_start=None, verbose=True):
    """Adam optimizer for the soft-max objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for step in range(n_steps):
        if time_limit and time.monotonic() - t_start > time_limit:
            break
        
        t += 1
        tau = tau_schedule(step)
        b = fast_autocorr(a)
        w = softmax_weights(b, tau)
        grad = compute_grad(a, w)
        
        # Normalize gradient (to account for the 1/sum(a)^2 term)
        # The actual objective is 2n * max_b / sum_a^2
        # d(max_b)/da = grad (approximately, with soft-max)
        # d(1/sum_a^2)/da[j] = -2*a[j]/sum_a^3
        # Full gradient: 2n * grad / sum_a^2 - 2n * max_b * 2*a/j / sum_a^3
        # But since we normalize a to sum=1 after each step, we can just use grad
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        b_new = fast_autocorr(a)
        M_new = b_new.max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and step % 1000 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print(f"    Adam step={t}, tau={tau:.4f}, J={J:.6f}, M={best_m:.8f} ({elapsed:.1f}s)")
    
    return best_a, best_m

def peak_shave(a, n_moves=1000, time_limit=None, t_start=None):
    """Directly reduce peaks by moving mass from high-contributing positions."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    rng = np.random.default_rng(42)
    
    for move in range(n_moves):
        if time_limit and time.monotonic() - t_start > time_limit:
            break
        
        b = fast_autocorr(a)
        M = b.max()
        
        # Find top peaks
        top_k = np.argsort(b)[::-1][:5]
        
        # For the top peak, find which positions contribute most
        k_peak = top_k[0]
        js = max(0, int(k_peak) - n + 1)
        je = min(n - 1, int(k_peak))
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[int(k_peak) - idx]
        
        if contrib.max() < 1e-20:
            continue
        
        # Find the position with highest contribution
        src_local = idx[np.argmax(contrib)]
        src = int(src_local)
        
        # Find a good destination: low contribution to all top peaks
        sensitivity = np.zeros(n)
        for pk in top_k:
            pk_int = int(pk)
            i_s = max(0, pk_int - n + 1)
            i_e = min(n, pk_int + 1)
            for pos in range(i_s, i_e):
                sensitivity[pos] += a[pos] * a[abs(pk_int - pos)]
        
        # Find low-sensitivity destination
        sens_sorted = np.argsort(sensitivity)
        
        moved = False
        for dst in sens_sorted[:50]:
            if dst == src:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M - 1e-15:
                    a = a_trial
                    if M_trial < best_m:
                        best_m = M_trial
                        best_a = a.copy()
                    moved = True
                    break
            if moved:
                break
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with adaptive soft-max temperature (primary method)
2. Top-k peak subgradient for sharp local optimization
3. Peak-targeted coordinate moves (mass redistribution)
4. Multi-length exploration with interpolation
5. Symmetrization and diverse restarts
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad(a, w):
    """Compute gradient of sum(w_k * b_k) where b = conv(a,a)."""
    n = len(a)
    term1 = np.correlate(w, a, mode='full')[n - 1: 2 * n - 1]
    term2 = np.convolve(w, a, mode='full')[0: n]
    return term1 + term2

def softmax_weights(b, tau):
    """Compute soft-max weights over autocorrelation values."""
    b_max = b.max()
    shifted = (b - b_max) / tau
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()
    return w

def topk_weights(b, k):
    """Create weights concentrated on top-k peaks."""
    idx = np.argsort(b)[::-1][:k]
    w = np.zeros(len(b))
    vals = b[idx]
    w[idx] = vals
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w[idx] = 1.0 / k
    return w

def score_internal(a):
    """Compute objective without calling evaluate_sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def adam_optimize(a_init, time_limit, t_start, verbose=True, symmetrize=False, topk_mode=False):
    """Adam optimizer with soft-max or top-k subgradient."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, fast_autocorr(a_init).max()
    a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    stall = 0
    best_score = score_internal(best_a)

    tau_start = 0.3
    tau_end = 0.005
    lr_base = 0.05
    decay_steps = max(100, int(time_limit * 30))

    for step in range(20000000):
        if time.monotonic() - t_start > time_limit:
            break
        t += 1

        frac = min(1.0, t / float(decay_steps))
        tau = tau_start * (tau_end / tau_start) ** frac

        b = fast_autocorr(a)
        if topk_mode:
            k = max(5, int(0.01 * len(b)))
            w = topk_weights(b, k)
        else:
            w = softmax_weights(b, tau)
        grad = compute_grad(a, w)

        # Normalize gradient
        gnorm = np.sqrt(np.mean(grad * grad))
        if gnorm > 1e-30:
            grad = grad / gnorm

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        lr_t = lr_base / (1.0 + 0.05 * t / 500.0)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)

        if symmetrize:

        s_new = a_new.sum()
        if s_new < 1e-15:

        else:
            a_new = a_new / s_new

        a = a_new
        b_new = fast_autocorr(a)
        M_new = b_new.max()

            stall = 0
        else:

    return best_a, best_m

def peak_coordinate_descent(a_init, time_limit, t_start, verbose=True):

    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()

    stall = 0

        # Get top peaks

            i_e = min(n, pk_int + 1)

                if 0 <= other < n:
                    sensitivity[pos] += a[pos] * a[other]

        dst_candidates = np.argsort(sensitivity)[:50]

                        a = a_trial

                            best_m = M_trial
                            best_a = a.copy()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
