```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Adam on log-sum-exp smoothed objective with FFT-based gradients
2. Beta annealing for progressive sharpening
3. Multiple diverse initializations (spectral, random, structured)
4. Nesterov-accelerated convergence
5. Final LP refinement and coordinate polish
"""

import numpy as np
import cvxpy as cp
import time

# ============ Core computation ============

def fft_size_for(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def fast_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    N = fft_size_for(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def smooth_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*c)) where c = conv(a,a).
    Returns (objective_value, gradient_array).
    """
    n = len(a)
    c = fast_conv(a)
    bc = beta * c
    max_bc = bc.max()
    exp_bc = np.exp(bc - max_bc)
    s = exp_bc.sum()
    obj = (max_bc + np.log(s)) / beta
    
    # w[k] = exp(beta*c[k]) / sum exp(beta*c[m])
    w = exp_bc / s
    
    # grad[i] = 2 * sum_m w[i+m] * a[m] for m=0..n-1, i=0..n-1
    # = 2 * convolve(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    N2 = fft_size_for((2*n-1) + n - 1)  # = fft_size_for(3n-2)
    W = np.fft.rfft(w, N2)
    AR = np.fft.rfft(a_rev, N2)
    conv = np.fft.irfft(W * AR, N2)
    grad = 2.0 * conv[n-1:2*n-1]
    
    return obj, grad

# ============ Normalization ============

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

# ============ Initialization methods ============

def init_uniform(n):
    return np.ones(n) / n

def init_gaussian(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    sigma = sigma_frac
    a = np.exp(-((x - 0.5)**2) / (2 * sigma**2))
    return a / a.sum()

def init_exponential(n, rate=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-rate * x)
    return a / a.sum()

def init_bilinear(n):
    """Triangular (Bartlett) window."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_spectral_match(n, target_max_iter=50):
    """
    Initialize by matching the spectrum to a target.
    Target: make |FFT(a)|^2 as flat as possible (minimizes peak convolution).
    """
    a = np.ones(n) / n
    N = fft_size_for(2 * n - 1)
    for _ in range(target_max_iter):
        A = np.fft.rfft(a, N)
        C = A * A
        # We want c = ifft(C) to be flat, so we want C to be flat
        # But c must be non-negative (it is, since a >= 0)
        # Alternative: adjust A to make |A|^2 flatter
        P = np.abs(A)**2
        P = np.maximum(P, 1e-30)
        # Target: make P flat (i.e., A has constant magnitude)
        # But we need a = ifft(A) >= 0
        target_mag = np.sqrt(P.sum() / len(P))
        target_phase = np.angle(A)
        A_new = target_mag * np.exp(1j * target_phase)
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.7 * a + 0.3 * a_new
    return a

def init_variadic(n, seed=42):
    """Create several diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    inits.append(init_uniform(n))
    inits.append(init_gaussian(n, 0.25))
    inits.append(init_gaussian(n, 0.4))
    inits.append(init_exponential(n, 1.5))
    inits.append(init_exponential(n, 3.0))
    # Asymmetric
    x = np.linspace(0, 1, n)
    a = np.exp(-3.0 * (x - 0.3)**2)
    inits.append(a / a.sum())
    a = np.exp(-3.0 * (x - 0.7)**2)
    inits.append(a / a.sum())
    # Random
    for trial in range(3):
        a = rng.random(n)
        a = a / a.sum()
        inits.append(a)
    # Spectral match
    inits.append(init_spectral_match(n, 30))
    return inits

# ============ Adam optimizer ============

def adam_optimize(a0, time_limit, beta_start=5.0, beta_end=200.0, 
                  lr_start=0.01, n_betas=10, verbose=True, seed=42):
    """
    Adam optimization with beta annealing in log-space.
    a0: initial point (will be normalized)
    """
    n = len(a0)
    a = normalize(a0)
    # Convert to log-space
    log_a = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_c = fast_conv(a)
    best_max = best_c.max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Beta schedule
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for beta_idx, beta in enumerate(betas):
        phase_t0 = time.monotonic()
        phase_time = time_limit / n_betas
        
        while time.monotonic() - phase_t0 < phase_time:
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            
            # Current a from log-space
            la_max = log_a.max()
            a = np.exp(log_a - la_max)
            a = a / a.sum()
            
            # Compute objective and gradient in a-space
            obj, g_a = smooth_grad(a, beta)
            
            # Check actual objective
            c = fast_conv(a)
            actual_max = c.max()
            if actual_max < best_max:
                best_max = actual_max
                best_a = a.copy()
            
            # Transform gradient to x-space (log-space)
            g_x = a * (g_a - np.dot(g_a, a))
            
            # Adam update
            m = beta1 * m + (1 - beta1) * g_x
            v = beta2 * v + (1 - beta2) * g_x**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            lr = lr_start / np.sqrt(1 + 0.01 * t)
            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic check
            if t % 5000 == 0 and verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max
                print(f"      Adam beta={beta:.1f} t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_max

        phase_t0 = time.monotonic()

        while time.monotonic() - phase_t0 < phase_time:
            if time.monotonic() - t0 > time_limit:

            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

            if t % 10000 == 0 and verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max

# ============ LP-based refinement ============

def lp_refine(a0, time_limit, top_k=200, solver=None):
    """Quick LP-based Frank-Wolfe refinement."""
    if solver is None:
        solver = _get_solver()
    
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        if M < best_m:
            best_m = M
            best_a = a.copy()
        
        # Top-K active constraints
        active = np.argsort(c)[::-1][:top_k]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            j_idx = np.arange(j_start, j_end + 1)
            y_coeff = a[k_int - j_idx]
            constraints.append(y_coeff @ b_var[j_idx] <= M)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        a = np.maximum((1 - alpha) * a + alpha * direction, 0)
        a = a / a.sum()
    
    return best_a, best_m

# ============ Coordinate descent polish ============

def coord_polish(a0, time_limit, rng_seed=42):
    """Coordinate descent polish on the hard objective."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_conv(a)
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:min(30, len(c))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit or improved:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            
            src_pos = np.argmax(contrib)
            src = idx[src_pos]
            
            # Find where to move mass
            for _attempt in range(10):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            
            if not improved:
                no_improve += 1
        
        if not improved:
            no_improve += 1
            # Perturb
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.01
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    no_improve = 0
    
    return best_a, best_m

# ============ Solver selection ============

_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
