def fast_autocorr(a):

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=30, eps=1e-10):
    """Flatten power spectrum via iterative DFT magnitude correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        # Target: flat magnitude -> ifft should be as flat as possible
        target_mag = np.ones(n) / np.sqrt(n)
        # Blend current and target
        new_mag = 0.5 * mag + 0.5 * target_mag
        A_new = phase * new_mag
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.3 * a + 0.7 * a_new
    return normalize(a)


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if n_active == 0:

    constraints = []

        j_end = min(n - 1, k_int)

        y_coeff = y[k_int - j_idx]
        constraints.append(cp.multiply(y_coeff, b_var[j_idx]).sum() <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def fw_line_search(x, direction, n_steps=25):
    """
    Golden section search for optimal step size along FW direction.
    Returns the best alpha in [0,1] and corresponding point.
    """
    # Golden section constants
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
    
    a_lo = 0.0
    a_hi = 1.0
    
    # Initial points
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    
    x_c = np.maximum((1.0 - c) * x + c * direction, 0.0)
    x_c = x_c / max(x_c.sum(), 1e-15)
    f_c = fast_autocorr(x_c).max()
    
    x_d = np.maximum((1.0 - d) * x + d * direction, 0.0)
    x_d = x_d / max(x_d.sum(), 1e-15)
    f_d = fast_autocorr(x_d).max()
    
    for _ in range(n_steps - 2):
        if f_c < f_d:
            a_hi = d
            d = c
            f_d = f_c
            c = a_hi - gr * (a_hi - a_lo)
            x_c = np.maximum((1.0 - c) * x + c * direction, 0.0)
            x_c = x_c / max(x_c.sum(), 1e-15)
            f_c = fast_autocorr(x_c).max()
        else:
            a_lo = c
            c = d
            f_c = f_d
            d = a_lo + gr * (a_hi - a_lo)
            x_d = np.maximum((1.0 - d) * x + d * direction, 0.0)
            x_d = x_d / max(x_d.sum(), 1e-15)
            f_d = fast_autocorr(x_d).max()
    
    alpha = (a_lo + a_hi) / 2.0
    x_best = np.maximum((1.0 - alpha) * x + alpha * direction, 0.0)
    x_best = x_best / max(x_best.sum(), 1e-15)
    return alpha, x_best


def fw_with_linsearch(a0, time_limit, solver, top_k_init=100, top_k_max=300, 
                       ls_steps=20, verbose=True, rng_seed=42):
    """
    Frank-Wolfe with golden-section line search and Nesterov extrapolation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t <= 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 1000:
            stall = 0
            pert = rng.normal(0, 0.01, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
            continue
        
        # Select active constraints
        active_lags = np.argsort(ac)[::-1][:current_top_k]
        
        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            # Fallback: use uniform direction
            direction = np.ones(n) / n
        
        # Line search
        alpha, x_new = fw_line_search(x, direction, n_steps=ls_steps)
        
        x_prev = y
        x = x_new
        
        # Check improvement
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-14:
            best_m = mx
            best_a = x.copy()
            stall = 0
        
        # Adapt top_k
        if t % 100 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 10, top_k_max)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW-LS t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}, alpha={alpha:.4f}")
    
    return best_a, best_m


def coord_polish(a0, time_limit, verbose=True, rng_seed=42):

    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

            if time.monotonic() - t0 > time_limit:
                break

            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            best_reduce = 0.0
            best_dst = -1
            best_frac = 0.0

            candidates = np.where(a < a[src] * 0.5)[0]  # destinations with less mass
            if len(candidates) < 100:
                candidates = np.arange(n)
            rng.shuffle(candidates)
            candidates = candidates[:min(300, len(candidates))]
            
            for dst in candidates:
                if dst == src:
                    continue
                # Compute reduction at lag k
                old_val = a[src] * a[k_int - dst] if 0 <= k_int - dst < n else 0.0
                old_val2 = a[k_int - src] * a[dst] if 0 <= k_int - src < n else 0.0
                # After moving delta from src to dst:
                # term1 = (a[src]-delta) * a[k_int-dst] = old_val - delta*a[k_int-dst]
                # term2 = a[k_int-src] * (a[dst]+delta) = old_val2 + delta*a[k_int-src]
                # Net change = -delta*a[k_int-dst] + delta*a[k_int-src]
                # We want this negative: a[k_int-src] < a[k_int-dst]
                delta_best = a[src] * 0.5  # maximum safe delta

                    delta = a[src] * frac

                        continue

                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue

                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        best_m = M_trial
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        stall = 0
                        delta_best = delta
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with golden-section line search and Nesterov extrapolation
2. Adaptive top-K constraint selection for the LP subproblem
3. Multiplicative weights (mirror descent) as complementary refinement

5. Final peak-targeted coordinate descent polish

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        for x in a_list:
            if not np.isfinite(x):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=30, eps=1e-10):
    """Flatten power spectrum via iterative DFT magnitude correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        target_mag = np.ones(n) / np.sqrt(n)
        new_mag = 0.5 * mag + 0.5 * target_mag
        A_new = phase * new_mag
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.3 * a + 0.7 * a_new
    return normalize(a)


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n_active = len(active_lags)
    if n_active == 0:
        return np.ones(n) / n

    b_var = cp.Variable(n, nonneg=True)
    constraints = []

        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(cp.multiply(y_coeff, b_var[j_idx]).sum() <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def fw_line_search(x, direction, n_ls=18):
    """
    Golden section search for optimal step size along FW direction.
    Returns (best_alpha, best_point).
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    a_lo = 0.0
    a_hi = 1.0

    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)

    x_c = np.maximum((1.0 - c) * x + c * direction, 0.0)
    x_c = x_c / max(x_c.sum(), 1e-15)
    f_c = fast_autocorr(x_c).max()

    x_d = np.maximum((1.0 - d) * x + d * direction, 0.0)
    x_d = x_d / max(x_d.sum(), 1e-15)
    f_d = fast_autocorr(x_d).max()

    for _ in range(max(0, n_ls - 2)):
        if f_c < f_d:
            a_hi = d
            d = c
            f_d = f_c
            c = a_hi - gr * (a_hi - a_lo)
            x_c = np.maximum((1.0 - c) * x + c * direction, 0.0)
            x_c = x_c / max(x_c.sum(), 1e-15)
            f_c = fast_autocorr(x_c).max()
        else:
            a_lo = c
            c = d
            f_c = f_d
            d = a_lo + gr * (a_hi - a_lo)
            x_d = np.maximum((1.0 - d) * x + d * direction, 0.0)
            x_d = x_d / max(x_d.sum(), 1e-15)
            f_d = fast_autocorr(x_d).max()

    alpha = (a_lo + a_hi) / 2.0
    x_best = np.maximum((1.0 - alpha) * x + alpha * direction, 0.0)
    x_best = x_best / max(x_best.sum(), 1e-15)
    return alpha, x_best


def fw_with_linsearch(a0, time_limit, solver, top_k_init=80, top_k_max=300,
                      ls_steps=18, verbose=True, rng_seed=42):
    """
    Frank-Wolfe with golden-section line search and Nesterov extrapolation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    stall = 0

    for t in range(1, 10000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        if t <= 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 800:
            stall = 0
            sigma = 0.005 + 0.01 * (stall // 800)
            pert = rng.normal(0, sigma, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
            continue

        active_lags = np.argsort(ac)[::-1][:current_top_k]

        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            direction = np.ones(n) / n

        alpha, x_new = fw_line_search(x, direction, n_ls=ls_steps)

        x_prev = y
        x = x_new

        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-14:
            best_m = mx
            best_a = x.copy()
            stall = 0

        if t % 80 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 15, top_k_max)

        if verbose and t % 15 == 0:
            J = 2.0 * n * best_m
            print(f"    FW-LS t={t} ({elapsed:.1f}s): J={J:.7f}, top_k={current_top_k}, alpha={alpha:.4f}, M={best_m:.8f}")

    return best_a, best_m


def mirror_descent_refine(a0, time_limit, top_k=30, eta0=2.0, verbose=True, rng_seed=42):
    """
    Multiplicative weights (mirror descent) refinement.
    Uses soft subgradient at top-K peaks.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0

    for t in range(1, 10000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            pert = rng.normal(0, 0.008, n)
            a = best_a * (1.0 + pert)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            continue

        top_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 6.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = eta0 / (np.sqrt(t) + 1.0)
        log_a = np.log(a + 1e-30) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        if verbose and t % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.7f}, M={best_m:.8f}")

    return best_a, best_m


def coord_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with exact evaluation."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(12, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            if a[src] < 1e-15:
                continue

            candidates = np.where(a < a[src] * 0.7)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            perm = rng.permutation(len(candidates))
            candidates = candidates[perm[:min(200, len(candidates))]]

            for dst in candidates:
                if dst == src:
                    continue
                found = False
                for frac in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]:
                    delta = a[src] * frac
                    if delta < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.02, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / max(a.sum(), 1e-15)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.7f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    - Multi-length FW with line search
    - Mirror descent refinement
    - Coordinate polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is not None:
        n_base = len(parent)
        start = normalize(parent)
        try_update(start, "parent")
    else:
        n_base = 2200
        x = np.linspace(0, 1, n_base)
        start = np.sin(np.pi * x) ** 2
        start = start / start.sum()
        try_update(start, "init")

    # Determine candidate lengths
    n_candidates = []
    for dn in [0, -200, -400, 200, 400]:
        nc = n_base + dn
        if 1000 <= nc <= 3500:
            n_candidates.append(nc)
    seen = set()
    unique_ns = []
    for n in n_candidates:
        if n not in seen:
            seen.add(n)
            unique_ns.append(n)
    n_candidates = sorted(unique_ns, key=lambda x: abs(x - n_base))

    print(f"Parent n={n_base}, candidates: {n_candidates}")

    # Phase 1: Spectral equalization on parent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_spec = spectral_equalize(start, n_iter=25)
        try_update(a_spec, f"spec_n{n_base}")

    # Phase 2: FW with line search on primary length
    remaining = time_budget - (time.monotonic() - t_global)
    primary_n = n_base
    print(f"\n=== Phase 2: FW-LS n={primary_n}, budget={min(remaining*0.45, 700):.0f}s ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        a_in = normalize(best_seq) if best_seq is not None and len(best_seq) == primary_n else start
        a_out, _ = fw_with_linsearch(a_in, t2, solver, top_k_init=80, top_k_max=250,
                                     ls_steps=18, verbose=True, rng_seed=42)
        try_update(a_out, f"fwls_n{primary_n}")

    # Phase 3: FW with line search on alternate length
    remaining = time_budget - (time.monotonic() - t_global)
    alt_n = n_candidates[1] if len(n_candidates) > 1 else (n_base + 300)
    if alt_n != primary_n and alt_n != n_candidates[0]:
        pass
    if len(n_candidates) > 1:
        alt_n = n_candidates[1]
    else:
        alt_n = n_base + 300
    if 1000 <= alt_n <= 3500 and alt_n != primary_n and remaining > 100:
        print(f"\n=== Phase 3: FW-LS alt n={alt_n}, budget={min(remaining*0.25, 350):.0f}s ===")
        t3 = min(remaining * 0.25, 350.0)
        if t3 > 50:
            source = best_seq if best_seq is not None else start
            a_interp = interpolate_seq(normalize(source), alt_n)
            a_interp = spectral_equalize(a_interp, n_iter=15)
            try_update(a_interp, f"spec_n{alt_n}")
            a_out, _ = fw_with_linsearch(normalize(a_interp), t3, solver,
                                         top_k_init=80, top_k_max=220,
                                         ls_steps=15, verbose=True, rng_seed=123)
            try_update(a_out, f"fwls_n{alt_n}")

    # Phase 4: Mirror descent on best so far
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: Mirror Descent n={n_best}, budget={min(remaining*0.15, 200):.0f}s ===")
        t4 = min(remaining * 0.15, 200.0)
        if t4 > 30:
            a_out, _ = mirror_descent_refine(normalize(best_seq), t4, top_k=40,
                                             eta0=3.0, verbose=True, rng_seed=77)
            try_update(a_out, f"md_n{n_best}")

    # Phase 5: Final FW refinement
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 5: FW-LS final n={n_best}, budget={min(remaining*0.25, 300):.0f}s ===")
        t5 = min(remaining * 0.25, 300.0)
        if t5 > 50:
            a_out, _ = fw_with_linsearch(normalize(best_seq), t5, solver,
                                         top_k_init=100, top_k_max=300,
                                         ls_steps=20, verbose=True, rng_seed=456)
            try_update(a_out, f"fwls_final_n{n_best}")

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Coord polish n={n_best}, budget={min(remaining*0.3, 150):.0f}s ===")
        t6 = min(remaining * 0.3, 150.0)
        if t6 > 20:
            a_out, _ = coord_polish(normalize(best_seq), t6, verbose=True, rng_seed=42)
            try_update(a_out, f"coord_n{n_best}")

    # Phase 7: Very final coord
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25 and best_seq is not None:
        n_best = len(best_seq)
        t7 = min(remaining * 0.6, 60.0)
        if t7 > 10:
            print(f"\n=== Phase 7: Final coord n={n_best} ===")
            a_out, _ = coord_polish(normalize(best_seq), t7, verbose=False, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
