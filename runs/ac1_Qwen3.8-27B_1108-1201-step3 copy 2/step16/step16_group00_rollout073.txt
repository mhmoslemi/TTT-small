def normalize_positive(a, eps=0.0):
   a = np.asarray(a,float)
   a = np.where(np.isfinite(a), a, 0)
   a = np.maximum(a, 0)
   s = a.sum()
   if s < 1e-12: return np.ones_like(a)/len(a)
   return a/s

def softmax_weights(ac, beta, top_k=0):
   L= len(ac)
   if top_k and top_k < L:
      idx = np.argpartition(ac, -top_k)[-top_k:] if top_k < L else np.arange(L)
      maxv = ac[idx].max()
      w = np.zeros(L)
      vals = np.exp(beta*(ac[idx]-maxv))
      s=vals.sum()
      if s <=0: w[idx] = 1.0/top_k
      else: w[idx] = vals/s
   else:
      maxv = ac.max()
      vals = np.exp(beta*(ac-maxv))
      s=vals.sum()
      if s <=0 or not finite: w = np.ones(L)/L
      else: w = vals/s
   return w

def gradient_from_weights(w, a):
   n=len(a)
   L=2*n-1
   conv_len = L+n-1
   fft_len = 1 << (conv_len -1).bit_length()
   # Use rfft for real
   W = np.fft.rfft(w, fft_len)
   AR = np.fft.rfft(a[::-1], fft_len)
   c = np.fft.irfft(W*AR, fft_len)[:conv_len]
   return 2*c[n-1:2*n-1]

def smooth_gradient(a, beta, top_k=0):

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2*n - 1
    if m <=0: return np.array([0.0])
    fft_len = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A*A, fft_len)[:m]
    return c

def gradient_from_weights_fft(w, a, scale=1.0):
   n=a.size; L=2*n-1
   if w.size != L:
      # pad/crop
      tmp=np.zeros(L); m=min(w.size,L); tmp[:m]=w[:m]; w=tmp
   conv_len = L + n -1
   fft_len = 1 << (conv_len -1).bit_length()
   W=np.fft.rfft(w, fft_len)
   AR=np.fft.rfft(a[::-1], fft_len)
   c=np.fft.irfft(W*AR, fft_len)[:conv_len]

"""
Search for low-peak self-convolution sequences.

This program searches for a non-negative sequence minimizing
    2 * n * max(conv(a, a)) / sum(a)**2
by combining:
  1. Normalized multiplicative-weight mirror descent with FFT/slice
     soft-subgradients of the peak of the full linear convolution.
  2. L-BFGS refinement on a smooth log-sum-exp proxy in a softmax
     parameterization.
  3. Pair-move coordinate polishing.
  4. A short scipy high-s LP Frank-Wolfe-style polish when available.
  5. Length sweeping around the best found length.
"""

import time
import numpy as np

try:
    from scipy.optimize import minimize
    HAVE_MINIMIZE = True
except Exception:
    minimize = None
    HAVE_MINIMIZE = False

try:
    from scipy.optimize import linprog
    HAVE_LINPROG = True
except Exception:
    linprog = None
    HAVE_LINPROG = False

_SMOOTH_BETA = 50.0
_SMOOTH_N = 0
_SMOOTH_DEADLINE = 0.0


def fast_autocorr(a):
    """Full linear convolution a * a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    fft_len = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A * A, fft_len)[:m]
    return c


def normalized_score(a):
    """
    Fast score equivalent to evaluate_sequence for finite nonnegative
    sequences not exceeding the 1000 clip.
    """
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return float("inf")
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        arr = np.minimum(arr, 1000.0)
        s = float(arr.sum())
        if s < 1e-12:
            return float("inf")
        ac = fast_autocorr(arr)
        mb = float(ac.max())
        if mb < 0.0:
            mb = 0.0
        return float(2.0 * arr.size * mb / (s * s))
    except Exception:
        return float("inf")


def evaluate_safe(a):
    """Wrapper for the official scorer with safety."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        arr = np.minimum(arr, 1000.0)
        val = evaluate_sequence([float(x) for x in arr.tolist()])
        return float(val)
    except Exception:
        return float("inf")


def normalize_positive(a):
    """Clip to nonnegative finite values and normalize sum to 1."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([1.0], dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s < 1e-12:
        return np.ones(arr.size, dtype=np.float64) / arr.size
    return arr / s


def interpolate_sequence(f, new_n):
    """Linear interpolation of a normalized sequence to another length."""
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.size
    if old_n == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    if old_n == int(new_n):
        return f.copy()
    s = float(f.sum())
    if s < 1e-12:
        return np.ones(int(new_n), dtype=np.float64) / int(new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    out = np.interp(new_x, old_x, fn)
    return normalize_positive(out)


def make_window(n, kind=0):
    """Construct simple valid seeds."""
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    if kind == 0:
        a = np.sin(np.pi * x) ** 2
    elif kind == 1:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif kind == 2:
        a = np.exp(-6.0 * (x - 0.5) ** 2)
    elif kind == 3:
        a = np.minimum(x, 1.0 - x) + 0.05
    elif kind == 4:
        a = x + 0.1
    elif kind == 5:
        a = (1.0 - x) + 0.1
    else:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 1e-6)
    return normalize_positive(a)


def consider_candidate(a, best_seq, best_score, label=""):
    """
    Top-level candidate updater.
    Returns (best_seq, best_score), replacing only if improved.
    """
    if a is None:
        return best_seq, best_score
    arr = normalize_positive(a)
    if arr.size == 0:
        return best_seq, best_score
    sc = normalized_score(arr)
    if np.isfinite(sc) and sc < best_score - 1e-15:
        if label:
            print(f"    new best [{label}]: {sc:.7f}  n={arr.size}")
        return arr, sc
    return best_seq, best_score


def softmax_weights(ac, beta, top_k=0):
    """
    Convex combination weights over lags.
    If top_k <= 0, use all lags; otherwise use the top-k largest ac values.
    """
    ac = np.asarray(ac, dtype=np.float64)
    L = ac.size
    if L == 0:
        return np.array([], dtype=np.float64)

    use_all = (top_k is None) or (top_k <= 0) or (top_k >= L)

    if use_all:
        vals = ac
        idx = None
    else:
        k = int(top_k)
        if k <= 0:
            return np.ones(L, dtype=np.float64) / L
        idx = np.argpartition(ac, -k)[-k:]
        vals = ac[idx]

    if beta <= 0:
        out = np.zeros(L, dtype=np.float64)
        if use_all:
            out.fill(1.0 / L)
        else:
            out[idx] = 1.0 / len(idx)
        return out

    mx = float(np.max(vals))
    e = np.exp(beta * (vals - mx))
    s = float(e.sum())

    out = np.zeros(L, dtype=np.float64)
    if use_all:
        out = e
    else:
        out[idx] = e

    if s > 0.0 and np.isfinite(s):
        out = out / s
    else:
        out = np.zeros(L, dtype=np.float64)
        if use_all:
            out.fill(1.0 / L)
        else:
            out[idx] = 1.0 / len(idx)
    return out


def gradient_from_weights_fft(w, a, scale=1.0):
    """
    Compute 2 * scale * sum_k w[k] * (T_k a) via FFT.
    This is the gradient direction of sum_k w[k] conv_k.
    """
    w = np.asarray(w, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1

    if w.size != L:
        tmp = np.zeros(L, dtype=np.float64)
        m = min(w.size, L)
        tmp[:m] = w[:m]
        w = tmp

    conv_len = L + n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_len)
    AR = np.fft.rfft(a[::-1], fft_len)
    c = np.fft.irfft(W * AR, fft_len)[:conv_len]
    return (2.0 * scale) * c[n - 1:2 * n - 1]


def gradient_soft(a, ac, beta, top_k=0, scale=1.0):
    """
    Compute soft gradient 2*scale*sum_k w_k T_k a.
    For small top_k, use direct slice accumulation; otherwise FFT.
    """
    a = np.asarray(a, dtype=np.float64)
    ac = np.asarray(ac, dtype=np.float64)
    n = a.size
    L = 2 * n - 1

    if top_k is None or top_k <= 0 or top_k >= L:
        w = softmax_weights(ac, beta, 0)
        return gradient_from_weights_fft(w, a, scale)

    k = int(top_k)
    if k <= 0:
        return np.zeros(n, dtype=np.float64)

    if k <= 80:
        idx = np.argpartition(ac, -k)[-k:]
        vals = ac[idx]
        if beta <= 0:
            weights = np.ones(k, dtype=np.float64) / k
        else:
            mx = float(np.max(vals))
            e = np.exp(beta * (vals - mx))
            s = float(e.sum())
            if s > 0.0 and np.isfinite(s):
                weights = e / s
            else:
                weights = np.ones(k, dtype=np.float64) / k

        ar = a[::-1]
        g = np.zeros(n, dtype=np.float64)

        for wi, kk in zip(weights, idx):
            kk = int(kk)
            st = kk - n + 1
            if st < 0:
                st = 0
            en = kk
            if en >= n:
                en = n - 1
            if st > en:
                continue

            start = n - 1 - kk + st
            stop = n - 1 - kk + en + 1

            if start < 0:
                start = 0
            if stop > n:
                stop = n

            seg = ar[start:stop]
            if seg.size == (en - st + 1):
                g[st:en + 1] += (2.0 * scale * wi) * seg

        return g

    w = softmax_weights(ac, beta, k)
    return gradient_from_weights_fft(w, a, scale)


def _softmax_x_from_z(z, n):
    """Construct softmax([z, 0]) stably."""
    z = np.asarray(z, dtype=np.float64).ravel()
    x = np.zeros(n, dtype=np.float64)
    m = min(z.size, n - 1)
    if m > 0:
        x[:m] = z[:m]
    x[-1] = 0.0
    x -= x.max()
    e = np.exp(x)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return e / s


def smooth_objective(z):
    """Smooth log-sum-exp objective for L-BFGS, using global state."""
    if time.monotonic() > _SMOOTH_DEADLINE:
        return 1e9

    n = _SMOOTH_N
    if n < 2:
        return 0.0

    a = _softmax_x_from_z(z, n)
    ac = fast_autocorr(a)

    beta = float(_SMOOTH_BETA)
    if beta <= 0.0:
        return float(ac.max())

    mx = float(ac.max())
    shifted = np.exp(beta * (ac - mx))
    ss = float(shifted.sum())
    if ss <= 0.0 or not np.isfinite(ss):
        return mx
    return float((beta * mx + np.log(ss)) / beta)


def smooth_gradient(z):
    """Gradient of smooth log-sum-exp objective w.r.t. z = log(a/a_last)."""
    if time.monotonic() > _SMOOTH_DEADLINE:
        return np.zeros(max(1, int(z.size)), dtype=np.float64)

    n = _SMOOTH_N
    if n < 2:
        return np.zeros(max(1, int(z.size)), dtype=np.float64)

    a = _softmax_x_from_z(z, n)
    ac = fast_autocorr(a)

    q = gradient_soft(a, ac, _SMOOTH_BETA, 0, 1.0)
    qdot = float(np.dot(a, q))
    gx = a * (q - qdot)
    return gx[:-1].copy()


def lbfgs_refine(a0, time_limit, beta=50.0, maxiter=150):
    """
    L-BFGS-B refinement of a smooth log-sum-exp proxy.
    Parameterization: a = softmax([z, 0]).
    """
    a = np.asarray(a0, dtype=np.float64).ravel()
    if a.size < 2:
        return normalize_positive(a)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()

    if a[-1] < 1e-10:
        a[-1] = 1e-10
        a /= a.sum()

    z0 = np.log(a[:-1] / a[-1])
    z0 = np.clip(z0, -25.0, 25.0)

    global _SMOOTH_BETA, _SMOOTH_N, _SMOOTH_DEADLINE
    _SMOOTH_BETA = float(beta)
    _SMOOTH_N = a.size
    _SMOOTH_DEADLINE = time.monotonic() + max(1.0, float(time_limit) - 2.0)

    if not HAVE_MINIMIZE:
        return normalize_positive(a)

    try:
        res = minimize(
            smooth_objective,
            z0,
            jac=smooth_gradient,
            method="L-BFGS-B",
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-12,
                "gtol": 1e-5,
                "maxls": 30
            }
        )
        if res is not None and res.x is not None:
            z = np.clip(np.asarray(res.x, dtype=np.float64), -30.0, 30.0)
            aout = _softmax_x_from_z(z, a.size)
            aout = np.maximum(aout, 1e-15)
            return normalize_positive(aout)
    except Exception:
        pass

    return normalize_positive(a)


def mgd_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Mirror-descent (multiplicative weights) optimizer for the exact peak
    objective using soft active-lag subgradients and backtracking.
    """
    a = normalize_positive(a0)
    n = a.size
    if n < 2:
        return a

    a = np.maximum(a, 1e-12)
    a /= a.sum()

    best = a.copy()
    best_F = normalized_score(a)
    t0 = time.monotonic()
    total_deadline = t0 + max(1.0, float(time_limit))

    rng = np.random.default_rng(seed)
    eta_global = 0.5
    best_stall = 0
    curr_stall = 0
    last_print = 0.0

    if time_limit >= 200:
        stages = [
            (2.0, 300, 0.15),
            (5.0, 150, 0.20),
            (10.0, 80, 0.25),
            (20.0, 40, 0.25),
            (50.0, 20, 0.15),
        ]
    elif time_limit >= 60:
        stages = [
            (5.0, 150, 0.25),
            (15.0, 60, 0.45),
            (40.0, 20, 0.30),
        ]
    else:
        stages = [
            (15.0, 60, 0.50),
            (50.0, 20, 0.50),
        ]

    for beta, top_k, frac in stages:
        stage_deadline = t0 + float(frac) * float(time_limit)
        stage_deadline = min(stage_deadline, total_deadline)

        while time.monotonic() < stage_deadline and time.monotonic() < total_deadline:
            sa = float(a.sum())
            ac = fast_autocorr(a)
            F = 2.0 * n * float(ac.max()) / (sa * sa)

            if F < best_F - 1e-15:
                best_F = F
                best = a.copy()
                best_stall = 0
                if verbose and time.monotonic() - last_print > 20.0:
                    last_print = time.monotonic()
                    print(f"      MGD beta={beta:.1f} top={int(top_k)} F={best_F:.7f} t={time.monotonic()-t0:.0f}s")
            else:
                best_stall += 1

            if best_stall > 3000:
                base = best
                if rng.random() < 0.5:
                    a = base + rng.normal(0.0, 0.03, n) * base
                else:
                    seed_ref = make_window(n, int(rng.integers(0, 5)))
                    a = 0.85 * base + 0.15 * seed_ref
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                eta_global = 0.6
                best_stall = 0
                curr_stall = 0

            g = gradient_soft(a, ac, beta, top_k, float(n))
            g = g - np.mean(g)
            gmax = float(np.max(np.abs(g)))

            if gmax < 1e-12:
                break

            eta = min(eta_global, 1.0 / (gmax + 1.0))
            accepted = False

            for _ in range(12):
                logc = np.log(a) - eta * g
                maxlog = logc.max()
                cand = np.exp(logc - maxlog)
                cand = np.maximum(cand, 0.0)
                sc = float(cand.sum())
                if sc <= 1e-300:
                    break
                cand /= sc

                ac2 = fast_autocorr(cand)
                F2 = 2.0 * n * float(ac2.max()) / (sc * sc)

                if np.isfinite(F2) and F2 <= F + 1e-15:
                    if F2 < F or eta < 1e-6 or abs(F2 - F) <= 1e-14:
                        a = cand
                        F = F2
                        accepted = True
                        break

                if eta < 1e-10:
                    break
                eta *= 0.7

            if accepted:
                curr_stall = 0
                eta_global = min(eta_global * 1.15, 1.5)
            else:
                curr_stall += 1
                eta_global *= 0.6
                if curr_stall > 15:
                    base = best if best_stall > 0 else a
                    if rng.random() < 0.3:
                        a = base + 0.02 * rng.normal(0.0, 1.0, n)
                    else:
                        a = a + 0.01 * rng.normal(0.0, 1.0, n)
                    a = np.maximum(a, 1e-12)
                    a /= a.sum()
                    eta_global = 0.4
                    curr_stall = 0

    return best


def coordinate_polish(a0, time_limit, seed=0, verbose=False):
    """
    Pair-move coordinate polish: move small mass from high-gradient
    coordinates to low-gradient coordinates, with exact score checks.
    """
    a = normalize_positive(a0)
    n = a.size
    if n < 2:
        return a

    a = np.maximum(a, 1e-12)
    a /= a.sum()

    best = a.copy()
    best_F = normalized_score(a)

    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(time_limit))
    rng = np.random.default_rng(seed)

    stall = 0
    last_print = 0.0

    stages = [
        (100, 5.0, 0.25),
        (50, 10.0, 0.35),
        (20, 20.0, 0.40),
    ]

    for top_k, beta, frac in stages:
        stage_deadline = min(t0 + float(frac) * float(time_limit), deadline)

        while time.monotonic() < stage_deadline:
            sa = float(a.sum())
            ac = fast_autocorr(a)
            F = 2.0 * n * float(ac.max()) / (sa * sa)

            if F < best_F - 1e-15:
                best_F = F
                best = a.copy()
                stall = 0
                if verbose and time.monotonic() - last_print > 20.0:
                    last_print = time.monotonic()
                    print(f"      Coord top={int(top_k)} F={best_F:.7f} t={time.monotonic()-t0:.0f}s")
            else:
                stall += 1

            if stall > 800:
                a = best + 0.005 * rng.normal(0.0, 1.0, n)
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                stall = 0
                if time.monotonic() > deadline:
                    break

            g = gradient_soft(a, ac, beta, top_k, 1.0)

            src_score = a * np.maximum(g, 0.0)
            src_candidates = np.argsort(src_score)[-4:]
            dst_candidates = np.argsort(g)[:4]

            improved = False

            for ii in src_candidates:
                if a[ii] < 1e-10:
                    continue
                for jj in dst_candidates:
                    if ii == jj:
                        continue
                    for move_frac in (0.05, 0.02, 0.01, 0.005, 0.002):
                        amt = a[ii] * move_frac
                        if amt <= 0.0:
                            continue
                        b = a.copy()
                        b[ii] -= amt
                        b[jj] += amt
                        b = np.maximum(b, 0.0)
                        sb = float(b.sum())
                        if sb <= 1e-12:
                            continue
                        b /= sb
                        Fb = normalized_score(b)
                        if Fb < best_F - 1e-15:
                            a = b
                            best_F = Fb
                            best = b.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

            if not improved:
                for _ in range(10):
                    ii = int(rng.integers(n))
                    jj = int(rng.integers(n))
                    if ii == jj:
                        continue
                    move_frac = float(rng.uniform(0.001, 0.02))
                    amt = a[ii] * move_frac
                    if amt <= 0.0:
                        continue
                    b = a.copy()
                    b[ii] -= amt
                    b[jj] += amt
                    b = np.maximum(b, 0.0)
                    sb = float(b.sum())
                    if sb <= 1e-12:
                        continue
                    b /= sb
                    Fb = normalized_score(b)
                    if Fb < best_F - 1e-15:
                        a = b
                        best_F = Fb
                        best = b.copy()
                        improved = True
                        stall = 0
                        break

            if not improved:
                a = best + 0.002 * rng.normal(0.0, 1.0, n)
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                if stall > 1600:
                    break

    return best


def lp_frank_wolfe(a0, time_limit, top_K=100):
    """
    Short LP-based direction search:
      maximize sum(b) subject to cross-convolution a*b <= peak(a*a), b>=0.
    Then line-search blends toward the normalized b.
    """
    if not HAVE_LINPROG:
        return normalize_positive(a0)

    a = normalize_positive(a0)
    n = a.size
    if n < 8:
        return a

    a = np.maximum(a, 1e-12)
    a /= a.sum()

    best = a.copy()
    best_F = normalized_score(a)

    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(time_limit))

    max_iter = int(min(40, max(5, float(time_limit) // 2)))
    top_K = int(min(top_K, max(10, 2 * n - 2)))

    for _it in range(max_iter):
        if time.monotonic() > deadline:
            break

        sa = float(a.sum())
        ac = fast_autocorr(a)
        M = float(ac.max())
        if M <= 0.0:
            break
        M = M * (1.0 + 1e-9) + 1e-12

        idx = np.argpartition(ac, -top_K)[-top_K:]
        A = np.zeros((top_K, n), dtype=np.float64)

        for r, kk in enumerate(idx):
            kk = int(kk)
            st = max(0, kk - n + 1)
            en = min(n - 1, kk)
            if st <= en:
                A[r, st:en + 1] = a[kk - np.arange(st, en + 1)]

        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(top_K, float(M), dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs"
            )
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            continue

        b = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
        sb = float(b.sum())
        if sb <= 1e-12:
            continue

        g = b / sb

        improved = False
        for alpha in (0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = a + alpha * (g - a)
            cand = np.maximum(cand, 0.0)
            sc = float(cand.sum())
            if sc <= 1e-12:
                continue
            cand /= sc
            Fcand = normalized_score(cand)
            if Fcand < best_F - 1e-15:
                a = cand
                best_F = Fcand
                best = cand.copy()
                improved = True
                break

        if not improved:
            Fg = normalized_score(g)
            if Fg < best_F - 1e-15:
                a = g.copy()
                best_F = Fg
                best = a.copy()

    return best


def propose_candidate(budget_s=1800.0):
    """
    Main search entry point.
    Returns a Python list of nonnegative floats.
    """
    t_global = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    budget = min(max(budget, 120.0), 1800.0)
    # Reserve time for final official validation and return conversion.
    time_budget = budget - 60.0
    if time_budget < 30.0:
        time_budget = 30.0

    best_seq = None
    best_score = float("inf")

    # Load the provided previous construction if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            p = np.minimum(p, 1000.0)
            if p.sum() > 1e-8:
                if p.size > 8000:
                    p = interpolate_sequence(p, 6000)
                best_seq = normalize_positive(p)
                best_score = normalized_score(best_seq)
                print(f"Parent loaded: n={best_seq.size}, score={best_score:.7f}")
    except Exception:
        pass

    if best_seq is None:
        print("No parent found; constructing base seed.")
        for n_try in (2048, 3200, 4096):
            for k in range(6):
                cand = make_window(n_try, k)
                best_seq, best_score = consider_candidate(cand, best_seq, best_score, f"seed n={n_try} k={k}")
        if best_seq is None:
            best_seq = make_window(3200, 0)
            best_score = normalized_score(best_seq)

    print(f"Initial best: {best_score:.7f}, n={best_seq.size}, t={time.monotonic()-t_global:.0f}s")

    # Quick seed scan at current length.
    n_base = int(best_seq.size)
    for k in range(6):
        cand = make_window(n_base, k)
        best_seq, best_score = consider_candidate(cand, best_seq, best_score, f"seed n={n_base} k={k}")

    def remaining():
        return max(0.0, time_budget - (time.monotonic() - t_global))

    # 1) Initial L-BFGS refinements.
    rem = remaining()
    if rem > 90.0 and n_base <= 5000:
        tlim = min(70.0, rem * 0.35)
        if tlim > 20.0:
            a_out = lbfgs_refine(best_seq, tlim, beta=25.0, maxiter=120)
            best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "lbfgs25")

        rem = remaining()
        if rem > 70.0:
            tlim = min(70.0, rem * 0.30)
            if tlim > 15.0:
                a_out = lbfgs_refine(best_seq, tlim, beta=75.0, maxiter=120)
                best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "lbfgs75")

    # 2) Main mirror descent optimization.
    rem = remaining()
    if rem > 120.0:
        tlim = min(rem * 0.45, 700.0)
        a_out = mgd_optimize(best_seq, tlim, verbose=True, seed=12345)
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "main_mgd")

    # 3) High-beta L-BFGS polish.
    rem = remaining()
    if rem > 60.0 and len(best_seq) <= 5000:
        tlim = min(80.0, rem * 0.25)
        if tlim > 20.0:
            a_out = lbfgs_refine(best_seq, tlim, beta=160.0, maxiter=150)
            best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "lbfgs160")

    # 4) Coordinate polish.
    rem = remaining()
    if rem > 60.0:
        tlim = min(120.0, rem * 0.20)
        a_out = coordinate_polish(best_seq, tlim, seed=777, verbose=True)
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "coord1")

    # 5) LP Frank-Wolfe polish (if scipy HiGHS is available).
    rem = remaining()
    if rem > 80.0 and len(best_seq) <= 3500:
        tlim = min(80.0, rem * 0.18)
        a_out = lp_frank_wolfe(best_seq, tlim, top_K=100)
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "lp100")

    # 6) Length sweep.
    base_n = int(best_seq.size)
    candidate_lengths = set()
    for factor in (0.75, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0):
        nn = int(round(base_n * factor))
        if 1024 <= nn <= 8000:
            candidate_lengths.add(nn)
    for nn in (2048, 3072, 4096, 5120, 6144, 8192 if 8192 <= 8000 else 8000):
        if 1024 <= nn <= 8000:
            candidate_lengths.add(nn)

    # Sort by distance from current best length, but try a few larger strengths.
    candidate_lengths = sorted(candidate_lengths, key=lambda x: (abs(x - base_n), x))

    tried = 0
    for new_n in candidate_lengths:
        rem = remaining()
        if rem < 120.0 or tried >= 5:
            break

        a_interp = interpolate_sequence(best_seq, new_n)
        best_seq, best_score = consider_candidate(a_interp, best_seq, best_score, f"interp n={new_n}")

        tlim = min(110.0, rem * 0.35)
        if tlim > 25.0:
            a_out = mgd_optimize(best_seq, tlim, verbose=False, seed=1000 + new_n)
            best_seq, best_score = consider_candidate(a_out, best_seq, best_score, f"mgd n={new_n}")

        rem = remaining()
        if rem > 60.0 and len(best_seq) <= 5000:
            tlim = min(45.0, rem * 0.20)
            if tlim > 15.0:
                a_out = lbfgs_refine(best_seq, tlim, beta=120.0, maxiter=80)
                best_seq, best_score = consider_candidate(a_out, best_seq, best_score, f"lbfgs n={new_n}")

        tried += 1

    # 7) Final mirror descent on the global best.
    rem = remaining()
    if rem > 80.0:
        tlim = min(rem * 0.45, 220.0)
        a_out = mgd_optimize(best_seq, tlim, verbose=True, seed=999983)
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "final_mgd")

    # 8) Final coordinate and one small LP if time remains.
    rem = remaining()
    if rem > 40.0:
        tlim = min(80.0, rem * 0.45)
        a_out = coordinate_polish(best_seq, tlim, seed=424242, verbose=False)
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "final_coord")

    rem = remaining()
    if rem > 60.0 and len(best_seq) <= 3500:
        tlim = min(50.0, rem * 0.45)
        a_out = lp_frank_wolfe(best_seq, tlim, top_K=80)
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "final_lp")

    # Final sanitize.
    best_seq = normalize_positive(best_seq)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s = float(best_seq.sum())
    if s < 1e-8:
        best_seq = make_window(3200, 0)

    official_score = evaluate_safe(best_seq)
    if not np.isfinite(official_score):
        # Should be extremely unlikely; fallback to normalized best.
        best_seq = normalize_positive(best_seq)
        official_score = normalized_score(best_seq)

    print(f"FINAL local estimated best: {best_score:.7f}")
    print(f"FINAL official evaluate_safe: {official_score:.7f}")
    print(f"FINAL n: {best_seq.size}, total time: {time.monotonic()-t_global:.0f}s")

    return [float(x) for x in best_seq.tolist()]
