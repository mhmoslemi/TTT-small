3. Frequency-domain-inspired initializations for diversity

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_gradient(a, top_k=5):
    """
    Compute gradient of max_k conv(a,a)[k] at a.
    Since the problem is non-convex, this is the gradient of the active (peak) constraint(s).
    conv(a,a)[k] = sum_{i+j=k} a[i]*a[j]
    d/d a[m] conv(a,a)[k] = sum_{i+j=k} (delta_{m,i} + delta_{m,j}) * a[i]*a[j] / (a[m]) ... 
    Actually: d/d a[m] sum_{i+j=k} a[i]*a[j] = a[k-m] + a[k-m] (if m != k-m) or 2*a[m] (if m == k-m)
    More carefully: 
    d/d a[m] sum_{i+j=k} a[i]*a[j] = sum_{i+j=k} (delta_{m,i}*a[j] + delta_{m,j}*a[i])
    = a[k-m] (term where i=m, j=k-m) + a[k-m] (term where j=m, i=k-m)
    But if m == k-m (i.e., k = 2m), then it's 2*a[m] (not a[m]+a[m] counted twice).
    Actually: if k = 2m, the term a[m]*a[m] has derivative 2*a[m] w.r.t. a[m].
    If k != 2m, the terms a[m]*a[k-m] and a[k-m]*a[m] both appear (for m != k-m), giving 2*a[k-m].
    Wait, but we need m and k-m to both be in [0, n-1].
    
    So: d/d a[m] conv(a,a)[k] = 
        0 if k-m < 0 or k-m >= n
        2*a[k-m] if m != k-m (i.e., k != 2m)
        2*a[m] if k == 2m
    In both cases (when valid), it's 2*a[k-m] (since if k=2m, k-m=m).
    
    Wait, let me be more careful.
    conv(a,a)[k] = sum_{i=0}^{k} a[i]*a[k-i] (with a[j]=0 for j<0 or j>=n)
    
    d/d a[m] conv(a,a)[k] = derivative of sum_{i} a[i]*a[k-i] w.r.t. a[m]
    = (term where i=m: a[k-m]) + (term where k-i=m, i.e., i=k-m: a[m])
    But if m = k-m (k=2m), these are the same term a[m]^2, whose derivative is 2*a[m].
    If m != k-m, we get a[k-m] + a[k-m] = 2*a[k-m].
    
    So in all cases (when 0 <= k-m < n): d/d a[m] = 2*a[k-m]
    And 0 if k-m < 0 or k-m >= n.
    """
    n = len(a)
    ac = fast_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    
    # Softmax weights for the top peaks
    top_vals = ac[top_idx]
    # Use temperature-based weighting
    temp = max(top_vals[0] * 0.1, 1e-10)
    weights = np.exp((top_vals - top_vals[0]) / temp)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        # d/d a[m] conv[k] = 2*a[k-m] for m in [max(0,k-n+1), min(n-1, k)]
        m_start = max(0, k_int - n + 1)
        m_end = min(n, k_int + 1)
        if m_start >= m_end:
            continue
        m_indices = np.arange(m_start, m_end)
        partner = k_int - m_indices
        # All partners should be in [0, n-1] by construction
        grad[m_start:m_end] += 2.0 * wi * a[partner]
    
    return grad, ac.max()

def peak_direction_search(a0, time_limit, verbose=True, seed=42):
    """
    Direct search using peak gradient with adaptive step size.
    Non-convex, so use backtracking + random restarts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    step_mult = 1.0
    top_k = 5
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, M_curr = peak_gradient(a, top_k=top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Adaptive step size with backtracking
        eta = step_mult / (np.sqrt(1 + t) * gn)
        eta = min(eta, 2.0 / gn)  # Cap
        
        # Try the step
        improved = False
        for _ in range(10):
            a_new = project_simplex(a - eta * grad)
            M_new = fast_autocorr(a_new).max()
            if M_new < M_curr - 1e-12:
                a = a_new
                M_curr = M_new
                improved = True
                break
            eta *= 0.5
        
        if improved:
            if M_curr < best_m - 1e-15:
                best_m = M_curr
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # On stall: try random perturbation
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            # Perturb in a structured way
            noise_scale = 0.05 + 0.1 * (max_stall / 300.0)
            noise = rng.normal(0, noise_scale, n) * a
            a_try = np.maximum(a + noise, 1e-12)
            a_try = a_try / a_try.sum()
            M_try = fast_autocorr(a_try).max()
            if M_try < M_curr:
                a = a_try
                M_curr = M_try
            else:
                # Try perturbations from best
                noise = rng.normal(0, 0.03, n) * best_a
                a_try = np.maximum(best_a + noise, 1e-12)
                a_try = a_try / a_try.sum()
                M_try = fast_autocorr(a_try).max()
                if M_try < M_curr:
                    a = a_try
                    M_curr = M_try
            top_k = min(top_k + 1, 15)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PeakDir t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, n_top_constraints=None, verbose=True):
    """
    LP Frank-Wolfe: at each step, solve LP to find best direction.
    Maximize sum(b) s.t. conv(a,b) <= M, b >= 0
    This is a valid local search direction even for non-convex problems.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if n_top_constraints is None:
        n_top_constraints = min(2 * n - 1, 3 * n)
    
    # Pre-build the toeplitz-like matrix once
    # T[k, j] = a[j] if k-j in [0, n-1], else 0
    # conv(a, b)[k] = sum_j T[k,j] * b[j]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Select top constraints
        if 2 * n - 1 > n_top_constraints:
            top_lags = np.argsort(ac)[::-1][:n_top_constraints]
            T_sel = T[top_lags]
            M_sel = np.full(len(top_lags), M_curr)
        else:
            T_sel = T
            M_sel = np.full(2 * n - 1, M_curr)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=get_solver(), verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        g_dir = b_val / b_sum
        
        # Line search for step size
        best_alpha = 0.0
        best_m_trial = M_curr
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = alpha
        
        if best_alpha > 0:
            a_new = (1 - best_alpha) * a + best_alpha * g_dir
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new
        
        if best_m_trial < best_m - 1e-15:
            best_m = best_m_trial
            best_a = a.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def get_solver():
    # Use a fast solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def freq_domain_init(n, rng):
    """
    Generate a sequence with relatively flat spectrum (to reduce convolution peak).
    Use a random phase in frequency domain, then take inverse FFT and threshold.
    """
    # Random flat-spectrum signal
    # DFT(a) should be roughly flat -> a should be "noise-like"
    # But we need a >= 0, so we take the real part and threshold
    
    # Method 1: Random phases
    fft_size = 1 << (2 * n - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1)
    spectrum[0] = 0  # Remove DC (we'll add it back)
    phases = rng.uniform(0, 2 * np.pi, len(spectrum))
    spectrum = spectrum * np.exp(1j * phases)
    a_complex = np.fft.irfft(spectrum, fft_size)[:n]
    a = np.maximum(a_complex, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def smooth_bump_init(n, rng, sigma=None):
    """Gaussian-like bump with random center and width."""
    if sigma is None:
        sigma = rng.uniform(0.05, 0.4)
    center = rng.uniform(0.2, 0.8)
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # Add some structure
    modulation = 1.0 + rng.uniform(-0.3, 0.3) * np.sin(2 * np.pi * rng.uniform(1, 5) * x)
    a = np.maximum(a * modulation, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent: for each position, find the best new value given others fixed.
    This is exact for the non-convex problem since we're optimizing one variable at a time.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Precompute which lags each position affects
    # Position i affects conv[k] for k in [i, i+n-1] (as a[i]*a[k-i])
    # and k in [0, 2*(n-1)-i] (as a[k-i]*a[i])
    # More precisely, position i appears in conv[k] for k = i + j where j in [0, n-1]
    # i.e., k in [i, i+n-1], with partner a[k-i] = a[j]
    
    stall = 0
    max_stall = 50
    
    # Random order for coordinate descent
    order = rng.permutation(n)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # For position i, find the best value that reduces the peak
            # conv[k] = a[i]*a[k-i] + (other terms not involving a[i])
            # Let base[k] = conv[k] - a[i]*a[k-i] for all k
            # Then conv_new[k] = base[k] + new_val * a[k-i]
            # We want to minimize max_k (base[k] + new_val * a[k-i])
            
            # This is a 1D minimization of max of linear functions -> can be solved exactly
            # but it's expensive. Instead, use a heuristic:
            # Try a few candidate values
            
            current_val = a[i]
            # The "partner" values
            partners = np.zeros(n)
            for k in range(i, min(i + n, 2 * n)):
                j = k - i
                if 0 <= j < n:
                    partners[j] = a[j]
            # partners[j] = a[j] for j in [0, n-1], and conv[i+j] involves a[i]*a[j]
            # Wait, I need to be more careful.
            # conv[k] = sum_{j} a[j] * a[k-j]
            # Terms involving a[i]: a[i]*a[k-i] (when j=i) and a[k-i]*a[i] (when k-j=i, j=k-i)
            # For k-i in [0, n-1]: a[i]*a[k-i] contributes to conv[k]
            # For i in [0, k]: a[k-i]*a[i] contributes to conv[k] (same thing)
            # So a[i] appears in conv[k] with coefficient a[k-i] when 0 <= k-i < n, i.e., i <= k < i+n
            # AND with coefficient a[k-i] when 0 <= k-i < n... wait it's the same condition.
            # Actually: a[i] appears in conv[k] twice if k = 2i (as a[i]*a[i]),
            # and once otherwise (as a[i]*a[k-i] where k-i ≠ i).
            # But since we're computing conv as sum over ALL (j, k-j) pairs, 
            # a[i] appears in conv[k] with multiplicity: 
            #   - once as the "first" factor when j=i: a[i]*a[k-i]
            #   - once as the "second" factor when k-j=i, j=k-i: a[k-i]*a[i]
            # These are the same product, so total contribution is a[i]*a[k-i] (counted once in the sum)
            # Wait no, in the sum sum_j a[j]*a[k-j], if j=i, we get a[i]*a[k-i].
            # If k-j=i, i.e., j=k-i, we get a[k-i]*a[i].
            # If i ≠ k-i (i.e., i ≠ k/2), these are different terms in the sum, so a[i] appears twice.
            # If i = k/2, it's the same term a[i]^2, appearing once.
            
            # For optimization: if we change a[i] to v, then:
            # conv_new[k] = conv[k] + (v - a[i]) * a[k-i] for k in [i, i+n-1] where k-i ≠ i
            # conv_new[2i] = conv[2i] + (v^2 - a[i]^2) if 2i < 2n-1
            
            # This is getting complex. Let's use a simpler heuristic.
            
            # Simple approach: try reducing a[i] and see if the peak decreases
            if current_val < 1e-10:
                continue
            
            # Try a few fractions
            base_conv = fast_autocorr(a)
            current_peak = base_conv.max()
            
            best_frac = 1.0
            best_peak = current_peak
            
            for frac in [0.5, 0.7, 0.85, 0.95, 1.2, 1.5, 2.0]:
                v = current_val * frac
                if v < 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = v
                # Renormalize
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_peak:
                    best_peak = m_trial
                    best_frac = frac
            
            if best_frac != 1.0:
                a[i] = a[i] * best_frac
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                m_new = fast_autocorr(a).max()
                if m_new < best_m - 1e-15:
                    best_m = m_new
                    best_a = a.copy()
                    improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            if a[i] > 1e-8:
                frac = rng.uniform(0.5, 1.5)
                a[i] *= frac
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m
