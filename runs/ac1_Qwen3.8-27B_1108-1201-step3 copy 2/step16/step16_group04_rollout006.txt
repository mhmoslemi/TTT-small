def fast_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    a = normalize(a)
    c = fast_conv(a)
    return 2.0 * len(a) * c.max()


def conv_gradient(c, a):
    """
    Compute gradient of sum_k c[k]^2 w.r.t. a.
    d(c[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n else 0.
    d(sum c^2)/d(a[i]) = sum_k 2*c[k] * 2*a[k-i] = 4 * sum_j a[j]*c[i+j]
    (where j = k-i, so k = i+j)
    """
    n = len(a)
    N = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, N)
    C = np.fft.rfft(c, N)
    # result[i] = sum_j a[j] * c[i+j] = ifft(conj(A) * C)[i]
    result = np.fft.irfft(np.conj(A) * C, N)[:n]
    return 4.0 * result

def spectral_equalize(a0, n_iter=50, verbose=False):
    """
    Iteratively flatten the power spectrum of the zero-padded sequence.
    Targets flat convolution by equalizing |FFT(a_padded)|.
    """
    a = normalize(a0).copy()
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_iter):
        Ap = np.fft.rfft(a, fft_size)
        mag = np.abs(Ap) + 1e-12
        # Scale to flat magnitude
        target_mag = np.ones_like(mag)
        phase = Ap / (mag + 1e-30)
        Ap_new = phase * target_mag
        a_new = np.fft.irfft(Ap_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Blend for stability
        blend = 0.3 + 0.5 * (it / max(n_iter - 1, 1))
        a = (1 - blend) * a + blend * a_new
    
    return a


def solve_lp_direction(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Where (y*b)[k] = sum_j y[j]*b[k-j] (linear in b).
    Returns normalized direction or None.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        # constraint: sum(y_coeff[i] * b[j_idx[i]]) <= M
        # Only add if constraint is actually tight
        if y_coeff.max() > 1e-10:
            constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if not constraints:
        return None
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def line_search_step(y, direction, best_score, n):
    """Find best step size via binary search."""
    best_alpha = 0.0
    best_val = float('inf')
    for alpha in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        x_trial = (1 - alpha) * y + alpha * direction
        x_trial = np.maximum(x_trial, 0.0)
        sx = x_trial.sum()
        if sx < 1e-15:
            continue
        x_trial = x_trial / sx
        c = fast_conv(x_trial)
        val = 2.0 * n * c.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_alpha, best_val


def lp_fw(a0, time_limit, solver, top_k_init=50, top_k_max=200, verbose=True):
    """
    Accelerated LP Frank-Wolfe with adaptive constraint selection and line search.
    """
    n = len(a0)
    a = normalize(a0).astype(np.float64)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    use_ls = True
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-3:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute convolution
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
        
        # Select active constraints
        active_lags = np.argsort(ac)[::-1][:current_top_k]
        
        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            continue
        
        # Step size: use line search for first 200 iters, then diminishing
        if use_ls and t <= 300:
            alpha, val = line_search_step(y, direction, best_m, n)
            if alpha == 0:
                alpha = 2.0 / (t + 2.0)
        else:
            alpha = 2.0 / (t + 2.0)
        
        # Update
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        # Check improvement
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        # Adapt top_k
        if t % 100 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 10, top_k_max)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")
    
    return best_a, 2.0 * n * best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = normalize(a0).astype(np.float64)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:min(30, len(c))]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find low-contribution destination
            low_region = np.where(c < np.median(c))[0]
            # Map conv indices back to a indices
            candidates = []
            for i in range(n):
                # a[i] contributes to c[k] for k in [i, i+n-1]
                # Score: sum of c in the range where a[i] matters
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)
                candidates.append((c[k_start:k_end+1].max(), i))
            
            candidates.sort()
            chosen = [i for _, i in candidates[:50]]
            
            best_dst = -1
            best_trial = best_m
            
            for dst in chosen:
                if dst == src:
                    continue
                found = False
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_trial:
                        best_trial = M_trial
                        best_dst = dst
                        a_best_trial = a_trial.copy()
                        found = True
                        break
            
            if found and best_dst >= 0:
                a = a_best_trial
                best_m = best_trial
                best_a = a.copy()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
            # Random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                a = a / max(s_t, 1e-15)
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i1] += amt
                    a[i2] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / max(a.sum(), 1e-15)
    
    return best_a, 2.0 * n * best_m


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = normalize(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    return normalize(result)

    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

            best_score = p_score

        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global

        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)
