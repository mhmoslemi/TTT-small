    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_optimize(a0, time_limit, mu_init=0.3, mu_decay=0.999, 
                       n_min_iters=100, verbose=True, rng_seed=42):
    """
    Spectral iterative refinement: alternate between frequency-domain
    peak flattening and time-domain non-negativity projection.
    """
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    mu = mu_init
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Frequency domain: compute spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        C_circ = np.fft.irfft(A * A, N)
        
        # Target flat level
        target = 1.0 / (2 * n - 1)
        
        # Modify spectrum to flatten C
        excess = np.maximum(C_circ - target, 0)
        P_new = P * np.exp(-mu * excess)
        P_new = np.maximum(P_new, 1e-30)
        
        # Preserve phase, modify magnitude
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        
        # Time domain: inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project onto {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Fallback: random restart
            a_new = rng.dirichlet(np.ones(n) * 0.5)
        else:
            a_new = a_new / s
        
        # Evaluate
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Decay mu
        mu = max(mu * mu_decay, 0.01)
        
        # Restart if stuck
        if stall > max_stall:
            stall = 0
            # Add noise
            noise = rng.normal(0, 0.01, n) * best_a
            a_restart = np.maximum(best_a + noise, 1e-10)
            a_restart = a_restart / a_restart.sum()
            a = a_restart
            # Reset mu
            mu = mu_init * rng.uniform(0.5, 2.0)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.1f}s): J={J:.6f}, mu={mu:.4f}, stall={stall}")
    
    return best_a, best_m


def exact_subgradient(a, ac, threshold_frac=0.99):
    """Compute exact subgradient from active set."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    return g, len(active)


def mirror_update(a, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def amd_subgradient(a0, time_limit, c0=3.0, threshold_frac=0.99, 
                    verbose=True, rng_seed=42, f_lower=None):
    """
    Accelerated Mirror Descent with exact subgradient and Polyak step.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    best_J = 2.0 * n * best_m  # This is the objective (with sum=1)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    if f_lower is None:
        f_lower = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound ≈ 1
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    current_thresh = threshold_frac
    c_current = c0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        g, n_active = exact_subgradient(y, ac, threshold_frac=current_thresh)
        
        # Polyak step size
        gy = 2.0 * n * ac.max()  # current objective at y
        g_norm_sq = np.dot(g, g)
        if g_norm_sq > 1e-30:
            polyak_eta = (gy - f_lower) / g_norm_sq
            # Blend with standard step
            std_eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)
            eta = min(max(polyak_eta, 0.1 * std_eta), 10.0 * std_eta)
        else:
            eta = std_eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)
        
        # Mirror update
        a_new = mirror_update(y, g, eta)
        
        x_prev = x
        x = a_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        J_new = 2.0 * n * M_new
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive restart
        if stall_count > max_stall:
            stall_count = 0
            noise_scale = 0.003 + 0.01 * (stall_count / max_stall)
            perturbation = rng.normal(0, noise_scale, n) * best_a
            x_restarted = np.maximum(best_a + perturbation, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            c_current *= 1.3
            if current_thresh > 0.98:
                current_thresh = max(0.98, current_thresh - 0.003)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD t={t} ({elapsed:.1f}s): J={best_J:.6f}, eta={eta:.4f}, stall={stall_count}")
    
    return best_a, best_J


def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """
    Improved coordinate descent: for each peak lag, find the optimal
    local redistribution of mass among positions involved in that lag.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = np.argsort(ac)[::-1][:min(10, len(ac))]
        
        improved = False
        
        for k in peak_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            if len(idx) == 0:
                continue
            
            # C[k] = sum_i a[idx_i] * a[j_idx_i]
            # To reduce C[k], reduce products a[idx_i] * a[j_idx_i]
            # Strategy: for each pair, if both are large, move mass from the larger
            # to a position not involved in any peak lag
            
            contrib = a[idx] * a[j_idx]
            
            # Find the largest contributor
            order = np.argsort(contrib)[::-1]
            
            for oi in order[:5]:
                src_i = int(idx[oi])
                partner_j = int(j_idx[oi])
                
                # Determine which to reduce
                if a[src_i] >= a[partner_j]:
                    src = src_i
                else:
                    src = partner_j
                
                if a[src] < 1e-10:
                    continue
                
                # Find target: position with low "peak involvement"
                peak_involve = np.zeros(n)
                for k2 in peak_k[:5]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n, k2i + 1)
                    peak_involve[s2:e2] += 1
                    j2 = k2i - np.arange(s2, e2)
                    peak_involve[j2] += 1
                
                # Find low-involvement positions with small values
                candidates = np.where(peak_involve == 0)[0]
                if len(candidates) == 0:
                    candidates = np.where(peak_involve < 2)[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                
                # Prefer positions with similar value to src
                if len(candidates) > 50:
                    # Sample
                    sampled = rng.choice(candidates, size=50, replace=False)
                    candidates = sampled
                
                for dst in candidates:
                    if dst == src:
                        continue
                    if a[dst] > a[src]:
                        continue
                    
                    # Try transfer
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] = max(a_trial[src] - amt, 0)
                        a_trial[dst] += amt
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Escape mechanism
            n_perturb = max(5, n // 100)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.02, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord v2 ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_init(n, rng):
    """Generate a spectral initialization with random phases."""
    N = 1 << (2 * n - 1).bit_length()
    # Random phase, flat magnitude
    A = np.exp(1j * rng.uniform(0, 2 * np.pi, N // 2 + 1))
    # To ensure real and non-negative, use conjugate symmetry
    # A[0] and A[N//2] must be real
    A[0] = 1.0
    if N % 2 == 0:
        A[N // 2] = 1.0
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
