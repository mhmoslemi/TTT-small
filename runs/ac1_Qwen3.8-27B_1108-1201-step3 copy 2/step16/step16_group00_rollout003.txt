                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")
            else:
                # Light nudge
                nudge = rng.normal(0, 0.01, n) * x
                x = x + nudge
                x = np.maximum(x, 1e-12)
                s = x.sum()
                x = x / s

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200):
    """LP-based Frank-Wolfe with binding constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_y = a.copy()
    iter_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix: T[k, i] = y[k-i] if 0<=k-i<n else 0
        # Only for selected constraints
        K_sel = min(top_K, 2 * n - 1)
        # Most binding constraints: where ac is closest to M_curr
        margins = M_curr - ac
        binding_idx = np.argsort(margins)[:K_sel]

        # Build sparse constraint matrix
        T_sel = np.zeros((K_sel, n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                T_sel[row, i_start:i_end] = y[k_int - np.arange(i_start, i_end)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        if b_var.value is None:
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        prev_y = y
        y = y_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coord_descent_exact(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    ac = fast_autocorr(a)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        improved = False

        # Process coordinates in order of contribution to peaks
        # Find coordinates that most affect the peak autocorrelation
        peak_lags = np.argsort(ac)[::-1][:min(30, len(ac))]

        # Build importance score for each coordinate
        importance = np.zeros(n)
        for k in peak_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                contrib = a[idx] * a[k_int - idx]
                importance[idx] += contrib

        # Sort by importance (descending) - modify most important first
        order = np.argsort(-importance)[:min(200, n)]

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Compute s_k = a[i-k] + a[i+k] for all valid k
            # Effect on R(k): new_R(k) = R(k) + delta * s_k
            s_k = np.zeros(2 * n - 1)
            # a[i-k] contributes to R(k) when 0 <= i-k < n, i.e., i-n+1 <= k <= i
            k_lo1 = max(0, i - n + 1)
            k_hi1 = min(2 * n - 2, i)
            for k in range(k_lo1, k_hi1 + 1):
                j = i - k
                if 0 <= j < n:
                    s_k[k] += a[j]
            # a[i+k] contributes to R(k) when 0 <= i+k < n, i.e., -i <= k <= n-1-i
            k_lo2 = max(0, -i)
            k_hi2 = min(2 * n - 2, n - 1 - i)
            for k in range(k_lo2, k_hi2 + 1):
                j = i + k
                if 0 <= j < n:
                    s_k[k] += a[j]

            # New autocorrelation if a[i] changes by delta:
            # R_new(k) = R(k) + delta * s_k (for k where s_k != 0)
            # We want to minimize max(R_new) subject to a[i] + delta >= 0
            # and maintaining sum = 1 (we'll renormalize after)

            # Current values at affected lags
            affected = np.where(s_k != 0)[0]
            R_aff = ac[affected]
            s_aff = s_k[affected]

            # The objective as function of delta: max_k (R_aff[k] + delta * s_aff[k])
            # This is a convex piecewise-linear function of delta
            # Optimal delta: where the two "highest" lines intersect, or at boundary

            if len(affected) == 0:
                continue

            # Find optimal delta by bisection on the subgradient
            # f(delta) = max_k (R_aff[k] + delta * s_aff[k])
            # f'(delta) = s_aff[k*] where k* = argmax

            lo = -a[i]  # boundary
            hi = a[i] * 0.5  # reasonable upper bound

            # Evaluate at lo and hi
            f_lo = np.max(R_aff + lo * s_aff)
            f_hi = np.max(R_aff + hi * s_aff)
            f_0 = np.max(R_aff)  # delta = 0

            # Find best delta in [lo, 0] and [0, hi]
            best_delta = 0.0
            best_f = f_0

            # Try bisection in [lo, 0]
            if f_lo < f_0 - 1e-15:
                # Optimal is in (lo, something)
                for _ in range(30):
                    mid = (lo + hi) / 2 if mid < 0 else -abs(hi) * 0.1
                    mid = (lo + 0) / 2
                    f_mid = np.max(R_aff + mid * s_aff)
                    # Subgradient direction
                    k_star = np.argmax(R_aff + mid * s_aff)
                    sg = s_aff[k_star]
                    if sg > 0:
                        # f is increasing at mid, go left
                        hi = mid
                    elif sg < 0:
                        # f is decreasing at mid, go right
                        lo = mid
                    else:
                        # At a minimum (flat)
                        break
                    if f_mid < best_f:
                        best_f = f_mid
                        best_delta = mid

            # Try small positive deltas
            for trial in [0.001, 0.01, 0.05, 0.1, 0.2, 0.5]:
                if trial > a[i] * 0.5:
                    break
                f_trial = np.max(R_aff + trial * s_aff)
                if f_trial < best_f - 1e-15:
                    best_f = f_trial
                    best_delta = trial

            if best_delta == 0.0:
                continue

            # Apply the change
            a[i] += best_delta
            a[i] = max(a[i], 0.0)

            # Renormalize (small adjustment)
            s_total = a.sum()
            if s_total < 1e-15:
                break
            a = a / s_total

            # Check improvement
            M_new = fast_autocorr(a).max()
            if M_new < best_m - 1e-14:
                best_m = M_new
                best_a = a.copy()
                improved = True
                ac = fast_autocorr(a)
            else:
                # Revert
                a[i] -= best_delta
                a[i] = max(a[i], 0.0)
                a = a / a.sum()

        if not improved:
            stall += 1
            if stall % 10 == 0:
                # Random perturbation
                n_pert = rng.integers(5, min(50, n // 4))
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] += rng.normal(0, 0.01, n_pert) * a[idx]
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_raised_cosine(n):
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    return norm_sum1(a)


def make_power_law(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_trapezoidal(n, frac=0.3):
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    edge = int(n * frac)
    ramp = np.linspace(0, 1, edge)
    a[:edge] = ramp
    a[-edge:] = ramp[::-1]
    return norm_sum1(a)


def make_bimodal(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-80 * (x - 0.2) ** 2) + np.exp(-80 * (x - 0.8) ** 2) + 0.005
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 6, 8]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def pair_coord_descent(a0, time_limit, verbose=True, rng_seed=456):
    """Pair coordinate descent: simultaneously adjust two coordinates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    ac = fast_autocorr(a)
    M_curr = ac.max()
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lags = np.argsort(ac)[::-1][:20]

        improved = False
        n_trials = min(500, n * 2)

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick source from high-contribution positions
            k = int(peak_lags[rng.integers(len(peak_lags))])
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start >= i_end:
                continue
            src = int(rng.integers(i_start, i_end))
            if a[src] < 1e-10:
                continue

            # Pick destination from low-contribution positions
            dst = int(rng.integers(n))
            if dst == src:
                continue

            # Try multiple transfer amounts
            for frac in [0.005, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-12:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Perturb
            n_pert = int(rng.integers(3, min(20, n // 10)))
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= (1.0 + rng.normal(0, 0.05, n_pert))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairCoord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_raised_cosine(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: AMD on parent length with multiple inits ===
    print(f"\n=== Phase 1: AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 450.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_raised_cosine(n0))
        labels.append("raised_cos")

        inits.append(make_power_law(n0, power=0.5))
        labels.append("pow05")

        inits.append(make_power_law(n0, power=1.0))
        labels.append("pow10")

        inits.append(make_trapezoidal(n0, frac=0.2))
        labels.append("trapez")

        inits.append(make_bimodal(n0))
        labels.append("bimodal")

        inits.append(make_smooth_random(n0, seed=42))
        labels.append("rand")

        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_raised_cosine(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD '{lbl}' (t_avail={t_avail:.0f}s)...", flush=True)
            a_out, _ = amd_optimize(init, t_avail, top_k=50, beta=5.0, c0=4.0,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=250)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_exact(a_in, t3, verbose=True, rng_seed=123)
        try_update(a_out, f"coord_exact_n{n_best}")

    # === Phase 4: Pair coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Pair coord (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_coord_descent(a_in, t4, verbose=True, rng_seed=789)
        try_update(a_out, f"pair_n{n_best}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")

    lengths_to_try = []
    for delta in [-400, -200, 200, 400, 800]:
        ln = n_best + delta
        if 1600 <= ln <= 5000:
            lengths_to_try.append(ln)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        sc_init = evaluate_safe(a_long)
        try_update(a_long, f"interp_n{new_n}")

        t5a = min(remaining * 0.25, 120.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=50, beta=4.5, c0=3.5,
                                     restart_every=4000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 3500:
            t5b = min(remaining * 0.15, 80.0)
            if t5b > 15:
                a_in = norm_sum1(best_seq)
                a_in = interpolate_sequence(a_in, new_n)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=False, top_K=150)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t6 = min(remaining * 0.30, 180.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=350)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord polish (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.40, 150.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_exact(a_in, t7, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: One more AMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.4, 100.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t8, top_k=80, beta=3.0, c0=5.0,
                                     restart_every=3000, verbose=False)
            try_update(a_out, f"amd_highk_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3200
        best_seq = make_raised_cosine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
