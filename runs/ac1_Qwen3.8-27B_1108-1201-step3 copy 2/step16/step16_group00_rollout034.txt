    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_multi_peak_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    n_half = n // 2
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        for i in range(n):
            j1 = k_int - i
            j2 = i - k_int
            contrib = 0.0
            if 0 <= j1 < n:
                contrib += a[j1]
            if 0 <= j2 < n and j2 != j1:
                contrib += a[j2]
            elif 0 <= j2 < n:
                contrib += a[j2] * 0.5
            grad[i] += wi * contrib
    return grad


def compute_fast_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=8000, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_fast_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            strength = min(0.1, 0.02 + 0.01 * (stall_count // restart_every))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 120)
            cur_beta = max(cur_beta - 0.3, 2.0)
            gamma = c0 * (1.0 + 0.1 * min(stall_count // restart_every, 5))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, beta={cur_beta:.1f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def freq_domain_optimize(a0, time_limit, iters=2000, eta0=0.5, verbose=False):
    """Iterative spectrum flattening via Gerchberg-Saxton-like approach."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Soft flattening: compress dynamic range of spectrum
        eta = eta0 * (1.0 + 0.5 * it / max(1, iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        # Update phase, adjust amplitude
        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            a = 0.5 * a + 0.5 * a_new
            a = norm_sum1(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqDom it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=200):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    T_full = np.zeros((2 * n - 1, n))

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full[:] = 0.0
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def mass_transport_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent based on mirror-position mass transport."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute cost at each position for all top peaks
            costs = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                for d in range(n):
                    j1 = k2i - d
                    j2 = d - k2i
                    c = 0.0
                    if 0 <= j1 < n:
                        c += a[j1]
                    if 0 <= j2 < n:
                        c += a[j2]
                    costs[d] += c

            # Find best destination: low cost
            candidates = np.argsort(costs)[:max(1, n // 4)]
            best_dst = candidates[np.argmin(a[candidates])]

            if best_dst == src:
                best_dst = int(rng.integers(n))
                if best_dst == src:
                    continue

            found = False
            for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Transport ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_power_law(n, power=1.0, center=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    if center:
        x = np.abs(x - (n + 1) / 2.0) + 1.0
        a = x ** (-power)
    else:
        a = x ** power
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_sinc_like(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_multihump(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
    a += 0.01
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    rng = np.random.default_rng(42)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-init AMD at base length ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_power_law(n0, power=0.5))
        labels.append("sqrt_ramp")

        inits.append(make_power_law(n0, power=1.0))
        labels.append("linear")

        inits.append(make_power_law(n0, power=2.0))
        labels.append("quadratic")

        inits.append(make_cosine(n0, mode=1))
        labels.append("cos1")

        inits.append(make_cosine(n0, mode=2))
        labels.append("cos2")

        inits.append(make_sinc_like(n0, bandwidth=4.0))
        labels.append("sinc4")

        inits.append(make_sinc_like(n0, bandwidth=8.0))
        labels.append("sinc8")

        inits.append(make_multihump(n0, n_humps=3, spread=0.2))
        labels.append("3hump")

        inits.append(make_multihump(n0, n_humps=5, spread=0.12))
        labels.append("5hump")

        inits.append(make_random_smooth(n0, seed=7, n_freq=6))
        labels.append("smooth1")

        inits.append(make_random_smooth(n0, seed=99, n_freq=10))
        labels.append("smooth2")

        # Centered power laws
        inits.append(make_power_law(n0, power=1.0, center=True))
        labels.append("center_inv")

        inits.append(make_power_law(n0, power=0.5, center=True))
        labels.append("center_inv_sqrt")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 5))

                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits

            if elapsed_so_far > time_budget - 60:

            if t_avail < 8:

        try_update(a_out, f"lp_n{n_best}")

    # === Phase 4: AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AMD polish (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t4, top_k=80, beta=3.5, c0=3.0,
                                 restart_every=8000, verbose=False)
        try_update(a_out, f"amd_polish_n{n_best}")

    # === Phase 5: Mass transport descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Mass transport (n={n_best}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = mass_transport_descent(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"transport_n{n_best}")

    # === Phase 6: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Longer sequences ===")
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 6000]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Scaling to n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t6a = min(remaining * 0.30, 150.0)
        if t6a > 20:
            a_out, _ = amd_optimize(a_long, t6a, top_k=60, beta=4.0, c0=3.5,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.20, 80.0)
            if t6b > 15:
                a_out, _ = freq_domain_optimize(a_long, t6b, iters=1000, eta0=0.5, verbose=False)
                try_update(a_out, f"freqdom_n{new_n}")

    # === Phase 7: Try shorter sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        shorter_lengths = [max(1000, n_best - 1000), max(800, n_best - 2000)]
        for new_n in shorter_lengths[:1]:
            if new_n < 500:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            print(f"  Trying n={new_n}...")
            a_short = interpolate_sequence(best_seq, new_n)
            try_update(a_short, f"interp_short_n{new_n}")
            t7 = min(remaining * 0.4, 100.0)
            if t7 > 20:
                a_out, _ = amd_optimize(a_short, t7, top_k=50, beta=4.0, c0=4.0,
                                         restart_every=3000, verbose=False)
                try_update(a_out, f"amd_short_n{new_n}")

    # === Phase 8: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=350)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final transport (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = mass_transport_descent(a_in, t9, verbose=True, rng_seed=777)
            try_update(a_out, f"transport_final_n{n_best}")

    # === Phase 10: One more AMD if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final AMD (n={n_best}) ===")
        t10 = min(remaining * 0.7, 80.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t10, top_k=100, beta=3.0, c0=2.5,
                                     restart_every=10000, verbose=False)
            try_update(a_out, f"amd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
