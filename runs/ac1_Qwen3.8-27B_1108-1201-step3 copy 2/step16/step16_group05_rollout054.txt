def fast_autocorr(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(a, active_lags):
    """Compute gradient of sum of conv_k(a) for active lags, normalized by count."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        # conv_k(a) = sum_i a[i]*a[i+k] (valid range)
        # d/d a[j] = a[j+k] + a[j-k] (with bounds)
        if k == 0:
            grad += 2.0 * a
        else:
            # a[j+k] for j = 0..n-1-k
            grad[:n-k] += a[k:]
            # a[j-k] for j = k..n-1
            grad[k:] += a[:n-k]
    return grad / len(active_lags)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Iterative peak equalization (water-filling on active lags) - main workhorse
2. Exact max-tracking projected gradient with momentum
3. Multiplicative (exponential) updates for non-negativity preservation

5. Multiple structured initializations (Hann, Blackman-Harris, parent)
6. Two-scale (coarse-to-fine) optimization to escape local minima

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe wrapper around evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project vector onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lag_gradient(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    k = int(k)
    if k == 0:
        grad[:] = 2.0 * a
    elif k < n:
        grad[:n - k] += a[k:]
        grad[k:] += a[:n - k]
    else:
        # By symmetry: conv[k] = conv[2n-2-k]
        k2 = 2 * n - 2 - k
        if k2 == 0:
            grad[:] = 2.0 * a
        elif k2 < n:
            grad[:n - k2] += a[k2:]
            grad[k2:] += a[:n - k2]
    return grad


def compute_pressure(a, active_lags):
    """Compute total pressure from active lags."""
    n = len(a)
    pressure = np.zeros(n)
    for k in active_lags:
        pressure += compute_lag_gradient(a, int(k))
    pressure /= len(active_lags)
    return pressure


def peak_equalization(a0, time_limit=10.0, n_active=20, verbose=False, seed=42):
    """
    Iteratively equalize the top lags using water-filling.
    Push mass from high-pressure to low-pressure positions.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300
    eta_base = 0.005

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_max - 1e-15:
            best_max = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find active lags (near peak)
        top_idx = np.argsort(ac)[::-1][:max(n_active, 10)]
        top_vals = ac[top_idx]
        if top_vals[0] < 1e-15:
            break
        threshold = top_vals[0] * 0.97
        active = top_idx[top_vals >= threshold]
        if len(active) == 0:
            active = top_idx[:3]

        # Compute pressure
        pressure = compute_pressure(a, active)

        # Water-filling direction
        p_mask = pressure > 1e-15
        if p_mask.sum() > 1:
            p_mean = pressure[p_mask].mean()
        else:
            p_mean = pressure.mean()
        direction = -(pressure - p_mean)

        # Adaptive step size with backtracking
        accepted = False
        for attempt in range(6):
            eta = eta_base / (1.0 + 0.3 * attempt)
            a_trial = a + eta * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M - 1e-14 * M:
                a = a_trial
                accepted = True
                break

        if not accepted:
            # Try multiplicative update
            eta = 0.01 / (1.0 + 0.5 * (stall % 10))
            p_shifted = pressure - pressure.min()
            a_trial = a * np.exp(-eta * p_shifted)
            a_trial = np.maximum(a_trial, 1e-18)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M - 1e-15:
                a = a_trial
                accepted = True

        if not accepted:
            if stall > max_stall:
                # Smart restart
                stall = 0
                max_stall = int(max_stall * 1.3)
                # Perturb toward low-pressure regions
                noise = rng.normal(0, 0.002, n)
                a_trial = a + noise * a + 0.005 * direction * a.max()
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                a = a_trial

                # Increase n_active to equalize more lags
                n_active = min(n_active + 5, 60)
            elif stall > 50:
                # Small random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.001, 0.005)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        else:
            # Occasionally increase n_active
            if stall == 0 and n_active < 50:
                n_active += 1

        if verbose and stall % 200 == 0:
            J = 2.0 * n * best_max
            print(f"    PE ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}, n_act={n_active}")

    return best_a, best_max


def exact_gradient_descent(a0, time_limit=60.0, n_active=5, verbose=False, seed=42):
    """
    Projected gradient with exact max tracking and momentum.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    grad_prev = np.zeros(n)
    stall = 0
    max_stall = 1500
    momentum = 0.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_max - 1e-15:
            best_max = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Active lags
        threshold = M * (1.0 - 5e-4)
        active = np.where(ac >= threshold)[0][:n_active]
        if len(active) == 0:
            active = np.array([int(np.argmax(ac))])

        # Gradient of sum of active lags
        grad = np.zeros(n)
        for k in active:
            grad += compute_lag_gradient(a, int(k))
        grad /= len(active)

        # Momentum (heavy ball)
        momentum = 0.9 * momentum + 0.1 * (t / (t + 20.0))
        grad_mom = grad + momentum * grad_prev
        grad_prev = grad.copy()

        # Step size
        eta = 8.0 / (np.sqrt(t) + 2.0)

        # Projected gradient step
        a_new = project_simplex(a - eta * grad_mom)
        a = a_new

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            # Restart with perturbation
            pressure = compute_pressure(a, active)
            p_min = pressure.min()
            direction = -(pressure - p_min)
            direction = direction / (np.abs(direction).max() + 1e-15)
            a = a + 0.005 * direction * a
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            grad_prev = np.zeros(n)
            momentum = 0.0

        if verbose and t % 3000 == 0:
            J = 2.0 * n * best_max
            print(f"    EGD ({time.monotonic()-t0:.0f}s): J={J:.6f}, t={t}")

    return best_a, best_max


def multiplicative_optimize(a0, time_limit=30.0, n_active=10, verbose=False):
    """
    Multiplicative (exponential) update: preserves non-negativity automatically.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()

    for t in range(1, 30000000):

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_max - 1e-15:
            best_max = M
            best_a = a.copy()

        # Active lags
        threshold = M * 0.98
        active = np.where(ac >= threshold)[0][:n_active]
        if len(active) == 0:
            active = np.array([int(np.argmax(ac))])

        # Pressure
        pressure = compute_pressure(a, active)

        # Multiplicative update
        eta = 3.0 / (np.sqrt(t) + 1.0)
        p_centered = pressure - pressure.mean()
        a = a * np.exp(-eta * p_centered)
        a = np.maximum(a, 1e-18)
        a = a / a.sum()

        if verbose and t % 20000 == 0:
            J = 2.0 * n * best_max
            print(f"    MU ({time.monotonic()-t0:.0f}s): J={J:.6f}, t={t}")

    return best_a, best_max


def coarse_fine_optimize(a0, time_limit=120.0, verbose=False):

def structured_initializations(n, parent=None):

    x = np.linspace(0, 1, n)

    bh = np.maximum(bh, 0)

    # Raised cosine with different alpha
    for alpha in [1.5, 2.0, 3.0]:
        rc = np.maximum(np.cos(np.pi * (x - 0.5) / alpha), 0) ** 2
        inits.append(norm_sum1(rc))

    # Gaussian center
    for sigma in [0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(norm_sum1(g))

    # Bimodal
    bim = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
    inits.append(norm_sum1(bim))

    # Parent interpolation
    if parent is not None:
        p = norm_sum1(parent)
        if len(p) != n:
            x_p = np.linspace(0, 1, len(p))
            p_interp = np.interp(x, x_p, p)
        else:
            p_interp = p
        inits.append(norm_sum1(p_interp))

    return inits

def optimize_at_length(a_init, time_limit, verbose=False):
    """Run a full optimization pipeline at a given length."""
    t0 = time.monotonic()
    a = norm_sum1(a_init)
    n = len(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_j = 2.0 * n * best_m

    def update(cand):
        nonlocal best_a, best_m, best_j
        cand = norm_sum1(cand)
        m = fast_autocorr(cand).max()
        j = 2.0 * n * m
        if j < best_j:
            best_a = cand.copy()
            best_m = m
            best_j = j

    # Phase A: Peak equalization (fast, many iterations)
    tA = min(time_limit * 0.35, 60.0)
    a_out, _ = peak_equalization(a, time_limit=tA, n_active=20, verbose=verbose)
    update(a_out)
    a = best_a.copy()

    # Phase B: Exact gradient descent
    rem = time_limit - (time.monotonic() - t0)
    tB = min(rem * 0.3, 60.0)
    if tB > 5:
        a_out, _ = exact_gradient_descent(a, time_limit=tB, n_active=5, verbose=verbose)
        update(a_out)
        a = best_a.copy()

    # Phase C: Multiplicative optimization
    rem = time_limit - (time.monotonic() - t0)
    tC = min(rem * 0.2, 30.0)
    if tC > 5:
        a_out, _ = multiplicative_optimize(a, time_limit=tC, n_active=10, verbose=verbose)
        update(a_out)
        a = best_a.copy()

    # Phase D: Final peak equalization with more active lags
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_out, _ = peak_equalization(a, time_limit=rem, n_active=40, verbose=verbose)
        update(a_out)

    return best_a, best_j
