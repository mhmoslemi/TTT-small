"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Fast Nesterov-accelerated subgradient descent with exact peak subgradients

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1 and clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_T(a):
    """Build convolution constraint matrix T (2n-1 x n)."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j+n, j] = a[j]
    return T

def subgradient_step(a, ac, top_L=20, beta_peak=3.0):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    L = min(top_L, len(ac))
    top_idx = np.argsort(ac)[::-1][:L]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta_peak
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(L) / L
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # d/da[m] (a*a)(k) = 2*a[k-m] if 0 <= k-m < n
        # So for each m in [0,n-1]: contribution is 2*a[k-m] if 0<=k-m<n
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        m_range = np.arange(lo, hi + 1)
        a_vals = a[k_int - m_range]
        grad[m_range] += 2.0 * wi * a_vals
    
    return grad

    best_a = a.copy()

    t0 = time.monotonic()

    gamma = 3.0

        if time.monotonic() - t0 > time_limit:
            break

            y = x.copy()
        else:
            y = y / sy

            ratio = top_vals_all[adaptive_L] / max(M_curr, 1e-30)

        eta = gamma / (np.sqrt(t) + 1.0)
        # Mirror descent update (exponential family)
        log_a = np.log(y + 1e-30) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)

        else:

        x = a_new

        if M_new < best_m - 1e-14:

            x = x_new

            J = 2.0 * n * best_m
            print(f"    NSGD t={t} ({elapsed:.1f}s): J={J:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):

    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def norm_sum1(a):

def compute_subgradient(a, ac, top_L, beta_peak):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    L = min(top_L, len(ac))
    top_idx = np.argsort(ac)[::-1][:L]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta_peak
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(L) / L
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        m_range = np.arange(lo, hi + 1)
        a_vals = a[k_int - m_range]
        grad[m_range] += 2.0 * wi * a_vals
    return grad

def nesterov_subgradient(a0, time_limit, top_L=30, beta_peak=4.0, verbose=False, rng_seed=42):
    """Nesterov-accelerated subgradient descent on the simplex with mirror map."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    gamma = 3.5
    for t in range(1, 10**9):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fast_autocorr(y)
        M_curr = ac.max()
        # Adaptive L
        adaptive_L = min(top_L, len(ac))
        sorted_ac = np.sort(ac)[::-1]
        if len(sorted_ac) > adaptive_L and adaptive_L > 1:
            ratio = sorted_ac[adaptive_L] / max(M_curr, 1e-30)
            if ratio > 0.85:
                adaptive_L = min(int(adaptive_L * 1.5) + 5, len(ac))
        grad = compute_subgradient(y, ac, adaptive_L, beta_peak)
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        eta = gamma / (np.sqrt(t) + 1.0)
        if grad_norm > 1e-15:
            eta = min(eta, 0.5 / grad_norm)
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        x_prev = x
        x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            noise_scale = min(0.05, 0.01 * (stall / max_stall))
            pert = rng.normal(0, noise_scale, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
            stall = 0
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSGD t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Phase 1: Quick NSGD warmup
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 1: NSGD warmup n={n0} ===")
        t1 = min(remaining * 0.05, 40.0)
        a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t1, top_L=40, beta_peak=5.0, verbose=False)
        try_update(a_out, "nsgd_warmup")

    # Phase 2: Spectral shaping (novel)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 2: Spectral shaping n={len(best_seq)} ===")
        t2 = min(remaining * 0.04, 30.0)
        a_out, _ = spectral_shaping(norm_sum1(best_seq), t2, verbose=False)
        try_update(a_out, "spectral1")

    # Phase 3: LP-FW with line search (main)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LP-FW-LS n={n_cur} ===")
        t3 = min(remaining * 0.25, 350.0)
        a_out, _ = lp_frank_wolfe_ls(norm_sum1(best_seq), t3, solver, verbose=True, use_ls=True)
        try_update(a_out, f"lp_ls1_n{n_cur}")

    # Phase 4: NSGD refinement
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: NSGD round 2 n={n_cur} ===")
        t4 = min(remaining * 0.06, 50.0)
        a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t4, top_L=50, beta_peak=6.0, verbose=False)
        try_update(a_out, "nsgd2")

    # Phase 5: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: LP-FW-LS round 2 n={n_cur} ===")
        t5 = min(remaining * 0.25, 300.0)
        a_out, _ = lp_frank_wolfe_ls(norm_sum1(best_seq), t5, solver, verbose=True, use_ls=True)
        try_update(a_out, f"lp_ls2_n{n_cur}")

    # Phase 6: Spectral shaping round 2
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 6: Spectral round 2 n={len(best_seq)} ===")
        t6 = min(remaining * 0.04, 30.0)
        a_out, _ = spectral_shaping(norm_sum1(best_seq), t6, verbose=False)
        try_update(a_out, "spectral2")

    # Phase 7: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 7: Coord descent n={n_cur} ===")
        t7 = min(remaining * 0.08, 60.0)
        a_out, _ = dual_coord_descent(norm_sum1(best_seq), t7, verbose=False)
        try_update(a_out, f"coord1_n{n_cur}")

    # Phase 8: NSGD with different params
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 8: NSGD-alt n={n_cur} ===")
        t8 = min(remaining * 0.07, 50.0)
        a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t8, top_L=20, beta_peak=3.0, verbose=False, rng_seed=123)
        try_update(a_out, "nsgd_alt")

    # Phase 9: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 9: Multi-n exploration ===")
        test_ns = []
        for delta in [200, 400, -200, -400]:
            nt = n_cur + delta
            if 600 <= nt <= 4000 and nt != n_cur:
                test_ns.append(nt)
        for nt in test_ns[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_ns = min(remaining * 0.10, 40.0)
            a_out, _ = nesterov_subgradient(norm_sum1(a_inter), t_ns, top_L=40, beta_peak=5.0, verbose=False)
            try_update(a_out, f"nsgd_n{nt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t_lp = min(remaining * 0.15, 80.0)
                a_out, _ = lp_frank_wolfe_ls(norm_sum1(a_inter), t_lp, solver, verbose=False, use_ls=True)
                try_update(a_out, f"lp_n{nt}")

    # Phase 10: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_final = len(best_seq)
        print(f"\n=== Phase 10: Final LP-FW-LS n={n_final} ===")
        t10 = min(remaining * 0.35, 200.0)
        a_out, _ = lp_frank_wolfe_ls(norm_sum1(best_seq), t10, solver, verbose=True, use_ls=True)
        try_update(a_out, f"lp_final_n{n_final}")

    # Phase 11: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_final = len(best_seq)
        print(f"\n=== Phase 11: Final coord polish n={n_final} ===")
        t11 = min(remaining * 0.4, 80.0)
        a_out, _ = dual_coord_descent(norm_sum1(best_seq), t11, verbose=False, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_final}")

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
