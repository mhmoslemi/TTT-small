```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with smoothed-max (log-sum-exp) objective and FFT-based analytic gradients
2. Softmax parameterization for automatic positivity (a = exp(x)/sum(exp(x)))
3. Multi-start with diverse window-function initializations + parent blending
4. Adaptive temperature schedule (beta increases for sharper max approximation)
5. Multi-length exploration (optimize at smaller n first, then scale up)
6. Peak-aware coordinate descent for final polish
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient_fft(a, w):
    """Compute r[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for j=0..n-1 via FFT.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    N = 1 << (3 * n).bit_length()
    A_fft = np.fft.rfft(a, N)
    w_rev = w[::-1]
    W_fft = np.fft.rfft(w_rev, N)
    conv_result = np.fft.irfft(A_fft * W_fft, N)
    # r[j] = conv_result[2n-2-j], so r = conv_result[n-1:2n-1][::-1]
    return conv_result[n - 1:2 * n - 1][::-1]


def smooth_max_obj_grad(x, beta, n):
    """Objective and gradient for L-BFGS.
    x is the log-parameterization: a = softmax(x).
    Returns (f, df/dx) where f = (1/beta)*log(sum exp(beta*c[k])).
    """
    x_c = np.clip(x, -50.0, 50.0)
    x_shifted = x_c - x_c.max()
    a = np.exp(x_shifted)
    s = a.sum()
    a = a / s

    c = fast_autocorr(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    obj = c_max + log_sum / beta

    w = exp_vals / exp_vals.sum()

    # grad_a[j] = (2/beta) * sum_m a[m] * w[j+m]
    grad_a = (2.0 / beta) * compute_gradient_fft(a, w)

    # Chain rule through softmax: df/dx_j = a_j * (grad_a_j - dot(grad_a, a))
    dot_val = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_val)

    return obj, grad_x


def run_lbfgs(a_init, beta, max_iter=300, verbose=False, label=""):
    """Run L-BFGS optimization starting from a_init (already normalized)."""
    n = len(a_init)
    x0 = np.log(np.maximum(a_init, 1e-30))
    x0 = x0 - x0.mean()  # centering for numerical stability

    t0 = time.monotonic()
    result = minimize(
        smooth_max_obj_grad,
        x0,
        args=(beta, n),
        method='L-BFGS-B',
        jac=True,
        options={
            'maxiter': max_iter,
            'maxfun': max_iter * 5,
            'ftol': 1e-14,
            'gtol': 1e-10,
            'maxls': 30,
            'maxcor': 40,
        }
    )

    elapsed = time.monotonic() - t0
    x_opt = np.clip(result.x, -50.0, 50.0)
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()

    if obj_opt < obj_init:
        return a_opt
    else:
        return a_init


def coordinate_descent(a_init, time_limit, verbose=False, seed=42):

    n = len(a_init)

                if a[src] < 1e-14:
                    continue

                # Score destinations: prefer positions that don't add to peak lags
                n_try = min(n, 300)
                if n > n_try:
                    dsts = rng.choice(n, n_try, replace=False)
                else:
                    dsts = np.arange(n)

                best_dst = -1
                best_m_trial = best_m
                best_amt = 0.0

                for dst in dsts:
                    dst = int(dst)
                    if dst == src:
                        continue
                    # Quick impact score
                    impact = 0.0
                    for k2 in top_lags[:3]:
                        k2_int = int(k2)
                        d = k2_int - dst
                        if 0 <= d < n:
                            impact += a[d]
                        d2 = dst - k2_int
                        if 0 <= d2 < n:
                            impact += a[d2]
                    if impact > a[src] * 2:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < 0 or a_trial[dst] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m_trial:
                            best_m_trial = M_trial
                            best_dst = dst
                            best_amt = amt
                        if M_trial > a[src] * 0.5:
                            break

                if best_dst >= 0 and best_amt > 0 and best_m_trial < best_m - 1e-15:
                    a[src] -= best_amt
                    a[best_dst] += best_amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_m = best_m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Escape: random small perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD ({elapsed:.1f}s): J={J:.6f}")
    return best_a


def make_window(wtype, n):
    """Create window function initializations of length n."""
    x = np.linspace(0.0, 1.0, n)
    if wtype == 'hann':
        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    elif wtype == 'hamming':
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * x)
    elif wtype == 'blackman':
        w = 0.42 - 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x)
    elif wtype == 'quasi_sine':
        w = np.sin(np.pi * x) ** 2
    elif wtype == 'parabolic':
        w = 4.0 * x * (1.0 - x)
    elif wtype == 'gauss':
        w = np.exp(-15.0 * (x - 0.5) ** 2)
    elif wtype == 'quartic':
        w = np.maximum(1.0 - (2.0 * x - 1.0) ** 2, 0.0) ** 2
    elif wtype == 'sech':
        w = 1.0 / np.cosh(4.0 * (x - 0.5))
    elif wtype == 'bump3':
        # Cubic bump: (1-x^2)^3 on [-1,1]
        t = 2.0 * x - 1.0
        w = np.maximum(1.0 - t ** 2, 0.0) ** 3
    elif wtype == 'cos4':
        w = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
        w = w ** 2
    else:
        w = np.ones(n) / n
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def get_parent_sequence():
    """Try to load the parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 1e-15)
    return a_interp / a_interp.sum()


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = 2.0 * len(a_norm) * fast_autocorr(a_norm).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** BEST [{label}]: score={sc:.6f}, n={len(a_norm)}, t={elapsed:.0f}s")

    # ===== Initialization =====
    parent = get_parent_sequence()
    if parent is not None:
        n_base = len(parent)
        p_score = 2.0 * n_base * fast_autocorr(parent).max()
        print(f"Parent loaded: n={n_base}, score={p_score:.6f}")
        try_update(parent, "parent")
    else:
        n_base = 2400
        print(f"No parent, default n={n_base}")
        init_default = make_window('quasi_sine', n_base)
        try_update(init_default, "default_init")

    print(f"\nTime budget: {time_budget:.0f}s")

    # ===== Phase 1: Small-n shape search (fast, finds good shape) =====
    if remaining_time() > 100:
        print(f"\n=== Phase 1: Shape search at n=400 ===")
        n_small = 400
        small_inits = []
        for wt in ['hann', 'blackman', 'parabolic', 'quasi_sine', 'gauss', 'bump3', 'sech', 'quartic']:
            small_inits.append(make_window(wt, n_small))
        if parent is not None:
            small_inits.append(interpolate_to_length(parent, n_small))

        best_small = None
        best_small_m = float('inf')
        for si in small_inits:
            if remaining_time() < 50:
                break
            a_out = si.copy()
            for beta in [5, 15, 40]:
                if remaining_time() < 30:
                    break
                a_out = run_lbfgs(a_out, beta, max_iter=200)
            m = fast_autocorr(a_out).max()
            if m < best_small_m:
                best_small_m = m
                best_small = a_out.copy()

        if best_small is not None:
            print(f"  Small-n best: n={n_small}, max_c={best_small_m:.8f}")
            # Interpolate to target and use as initialization
            a_scaled = interpolate_to_length(best_small, n_base)
            # Quick L-BFGS at target length
            a_refined = a_scaled.copy()
            for beta in [5, 10, 20]:
                if remaining_time() < 30:
                    break
                a_refined = run_lbfgs(a_refined, beta, max_iter=200)
            try_update(a_refined, "shape_interp")

    # ===== Phase 2: Multi-start L-BFGS at target length =====
    print(f"\n=== Phase 2: L-BFGS at n={n_base} ===")
    candidates = []
    if parent is not None:
        candidates.append(('parent', parent.copy()))

    window_types = ['hann', 'blackman', 'parabolic', 'quasi_sine', 'gauss',
                    'bump3', 'sech', 'quartic', 'hamming', 'cos4']
    for wt in window_types:
        candidates.append((wt, make_window(wt, n_base)))

    # Blends with parent
    if parent is not None:
        for wt in ['hann', 'blackman', 'parabolic']:
            blend = 0.5 * parent + 0.5 * make_window(wt, n_base)
            candidates.append((f"blend_{wt}", blend / blend.sum()))

    # If we have a current best, include it
    if best_seq is not None:
        candidates.insert(0, ('current_best', best_seq.copy()))

    print(f"  {len(candidates)} candidates to try")
    for idx, (name, a_init) in enumerate(candidates):
        if remaining_time() < 60:
            break
        t_per = max(20.0, min(remaining_time() / max(1, len(candidates) - idx - 1), 150.0))

        a_cur = a_init.copy()
        for beta in [5, 10, 25]:
            if remaining_time() < 30:
                break
            a_out = run_lbfgs(a_cur, beta, max_iter=250, label=name)
            if fast_autocorr(a_out).max() < fast_autocorr(a_cur).max() - 1e-12:
                a_cur = a_out
        try_update(a_cur, f"lbfgs_{name}")

    # ===== Phase 3: Coordinate descent polish =====
    print(f"\n=== Phase 3: Coordinate descent ===")
    if best_seq is not None and remaining_time() > 80:
        t_cd = min(remaining_time() * 0.25, 250.0)
        a_out = coordinate_descent(best_seq, t_cd, verbose=True, seed=42)
        try_update(a_out, "cd1")

    if best_seq is not None and remaining_time() > 50:
        t_cd = min(remaining_time() * 0.15, 120.0)
        a_out = coordinate_descent(best_seq, t_cd, verbose=False, seed=123)
        try_update(a_out, "cd2")

    # ===== Phase 4: Higher-beta L-BFGS on best =====
    print(f"\n=== Phase 4: High-beta L-BFGS polish ===")
    if best_seq is not None:
        for beta in [50, 100, 200, 500]:
            if remaining_time() < 30:
                break
            a_out = run_lbfgs(best_seq, beta, max_iter=300, verbose=True, label=f"b{beta}")
            try_update(a_out, f"highbeta_{beta}")

    # ===== Phase 5: Try longer lengths =====
    print(f"\n=== Phase 5: Longer lengths ===")
    if best_seq is not None:
        for n_try in [n_base + 400, n_base + 800, n_base + 1600]:
            if remaining_time() < 100:
                break
            if n_try > 5000:
                break
            print(f"  Trying n={n_try}")
            a_long = interpolate_to_length(best_seq, n_try)
            a_cur = a_long.copy()
            for beta in [5, 10, 20]:
                if remaining_time() < 40:
                    break
                a_out = run_lbfgs(a_cur, beta, max_iter=200)
                if fast_autocorr(a_out).max() * n_try < fast_autocorr(a_cur).max() * len(a_cur):
                    a_cur = a_out
            try_update(a_cur, f"n{n_try}")
            if remaining_time() > 50:
                a_cd = coordinate_descent(a_cur, min(60, remaining_time() * 0.3), verbose=False, seed=7)
                try_update(a_cd, f"cd_n{n_try}")

    # ===== Phase 6: Final coordinate descent =====
    print(f"\n=== Phase 6: Final CD ===")
    if best_seq is not None and remaining_time() > 30:
        t_final = min(remaining_time() * 0.5, 200.0)
        a_out = coordinate_descent(best_seq, t_final, verbose=True, seed=314)
        try_update(a_out, "cd_final")

    # ===== Final validation =====
    if best_seq is not None:
        final_score = 2.0 * len(best_seq) * fast_autocorr(best_seq).max()
        print(f"\nFINAL: score={final_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        # Fallback
        x = np.linspace(0, 1, 2400)
        a = np.sin(np.pi * x) ** 2
        return [float(v) for v in a / a.sum()]
```
