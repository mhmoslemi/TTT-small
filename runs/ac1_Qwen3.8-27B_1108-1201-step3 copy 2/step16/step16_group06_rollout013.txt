    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_convolve(a, a)
    return float(2.0 * len(a) * b.max() / (s * s))


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def logsumexp_objective(a, beta):
    ac = fft_autocorr(a)
    scaled = beta * ac
    m = scaled.max()
    if m > 700:
        return (m + np.log(np.exp(scaled - m).sum())) / beta
    return (m + np.log(np.exp(scaled - m).sum())) / beta


def logsumexp_gradient(a, beta):
    ac = fft_autocorr(a)
    scaled = beta * ac
    m = scaled.max()
    exp_vals = np.exp(scaled - m)
    p = exp_vals / exp_vals.sum()
    n = len(a)
    # grad[i] = (2/beta) * sum_k p[k] * a[k-i] where 0<=k-i<n
    # This is cross-correlation of p (length 2n-1) with a (length n)
    # grad[i] = sum_k p[k] * a[k-i] for valid k-i
    # Let's compute via FFT: pad a to length 2n-1, multiply element-wise with p, 
    # but we need sum_k p[k]*a[k-i] which is a correlation
    # corr[i] = sum_k p[k] * a[k-i] = sum_j p[j+i] * a[j]
    # This is the cross-correlation, computeable as:
    # corr = ifft(fft(p) * conj(fft(a_padded)))
    # Or more simply: corr[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # Let's use direct computation via FFT
    # Padded: define q of length 2n-1 where q[k] = p[k]
    # corr[i] = sum_{j=0}^{n-1} q[i+j] * a[j]
    # This is the (i-th) element of the cross-correlation of q and a
    # cross_corr[i] = sum_j q[i+j]*a[j]
    # = conv(q_reversed, a)[i + (2n-2)]  ... let me just use FFT directly
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    Q = np.fft.rfft(q_pad(p, n), fft_size) if False else None
    # Simpler: use the fact that corr[i] = sum_j p[i+j] * a[j]
    # Pad a to length 3n-2: a_ext[j] = a[j] for j<n, 0 otherwise
    # Then corr[i] = sum_j p[i+j]*a[j] is the i-th element of 
    # the "anti-convolution" or we can compute it as:
    # Let r = 3n-2, define a_ext of length r with a_ext[0:n] = a
    # Define p_ext of length r with p_ext[0:2n-1] = p
    # corr[i] = sum_j p_ext[i+j] * a_ext[j] for j=0..n-1
    # This is the cross-correlation, which equals:
    # conv(a_ext, p_ext_reversed)[(2n-2) - i + (2n-3)] ... this is getting complex
    # Let me just do it via FFT properly
    # corr[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # Define b[j] = a[j] for j=0..n-1, 0 otherwise (length 2n-1 is enough)
    # Define c[k] = p[k] for k=0..2n-2, 0 otherwise
    # corr[i] = sum_j c[i+j] * b[j] = (c * b_flipped_conv)[...]
    # Actually: sum_j c[i+j]*b[j] = sum_j c[m]*b[m-i] where m=j+i
    # = conv(c, b)[2*(n-1) - i + ... ]
    # I'll just use numpy correlate or compute via FFT
    # Simplest correct approach:
    # corr = np.correlate(p, a, mode='valid') won't work directly
    # Let me use: corr[i] = np.dot(p[i:i+n], a)
    # For efficiency with FFT:
    # Reverse a: a_r = a[::-1]
    # conv(p, a_r)[k] = sum_j p[j] * a_r[k-j] = sum_j p[j] * a[n-1-(k-j)]
    # = sum_j p[j] * a[n-1-k+j]
    # Let m = n-1-k: = sum_j p[j] * a[m+j]
    # So conv(p, a_r)[n-1-m] = sum_j p[j] * a[m+j] = corr[m]
    # Therefore corr[m] = conv(p, a_r)[n-1-m]
    a_r = a[::-1]
    conv_pa = fft_convolve(a, p)  # length 2n-1
    # corr[m] = conv_pa[n-1-m]
    # So grad[i] = (2/beta) * corr[i] = (2/beta) * conv_pa[n-1-i]
    grad = (2.0 / beta) * conv_pa[n - 1 - np.arange(n)]
    return grad

def logsumexp_nesterov(a_init, time_limit, beta_start=20.0, beta_end=300.0, 
                        verbose=True, seed=42):
    """Nesterov-accelerated mirror descent on log-sum-exp smoothed objective."""
    n = len(a_init)
    a = norm_sum1(a_init.copy())
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 6.0
    max_iters = 500000
    total_iters = max_iters
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        fraction = min(t / (total_iters * 0.8), 1.0)
        beta = beta_start + (beta_end - beta_start) * fraction ** 1.5
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-30)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient of log-sum-exp
        grad = logsumexp_gradient(y, beta)
        
        # Mirror descent step (exponential weights)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Check improvement
        sc = eval_score(x)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = x.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE t={t} ({elapsed:.1f}s): score={best_score:.6f}, beta={beta:.1f}")
    
    return best_a, best_score


def lp_frank_wolfe(a_init, time_limit, solver_name, verbose=True, max_iters=50000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a_init)
    a = norm_sum1(a_init.copy())
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fft_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint: conv(y, b)[k] <= M_curr for all k
        # conv(y,b)[k] = sum_j y[j]*b[k-j] for valid j,k-j
        # So constraint: sum_j y[j]*b[k-j] <= M
        # Let's build the matrix T where T[k,j] = y[j] if 0<=k-j<n
        # T is (2n-1) x n
        # For efficiency, use sparse
        import scipy.sparse as sp
        rows = []
        cols = []
        vals = []
        for j in range(n):
            if y[j] > 1e-15:
                # b[j] appears in conv positions k where 0<=k-j<n, i.e., j<=k<=j+n-1
                k_start = j
                k_end = j + n  # exclusive
                k_end = min(k_end, 2 * n - 1)
                rows.extend(range(k_start, k_end))
                cols.extend([j] * (k_end - k_start))
                vals.extend([y[j]] * (k_end - k_start))
        
        T = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.atleast_1d(b_var.value).flatten(), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        syy = y_new.sum()
        if syy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / syy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fft_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fft_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def aggressive_coord_descent(a_init, time_limit, verbose=True, seed=123):
    """Aggressive coordinate descent targeting the convolution peaks directly."""
    n = len(a_init)
    a = norm_sum1(a_init.copy())
    best_a = a.copy()
    best_m = fft_autocorr(a).max()  # peak of autocorr = peak of conv for symmetric
    # Actually we need peak of conv(a,a), not autocorr
    best_conv_peak = fft_convolve(a, a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fft_convolve(a, a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))
        
        # Find contributing pairs for the peak
        # ac[k] = sum_j a[j] * a[k-j] for valid j
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        partners = k_star - idx
        contrib = a[idx] * a[partners]
        
        if contrib.max() < 1e-20:
            stall += 1
            continue
        
        # Find positions with low "impact" (safe destinations)
        # A position i is "safe" if it doesn't contribute much to any top peak
        top_peaks = np.argsort(ac)[::-1][:min(20, 2 * n - 1)]
        impact = np.zeros(n)
        for pk in top_peaks:
            pk = int(pk)
            for pos in range(min(50, n)):
                d = pk - pos
                if 0 <= d < n:
                    impact[pos] += a[d]
        
        # Also check: moving mass from i to j changes ac by:
        # delta_ac[k] = -a[i]*a[k-i] - a[j]*a[k-j] + delta*(...)
        # Simplification: just try moving from high-contrib to low-impact
        
        improved = False
        src_order = idx[np.argsort(contrib)[::-1]][:10]
        
        for src in src_order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-15:
                continue
            
            # Find best destination
            safe_positions = np.argsort(impact)[:max(30, n // 10)]
            best_frac = None
            best_trial_m = M_curr
            
            for dst in safe_positions:
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0 or a_trial[dst] < 0:
                        continue
                    ct = fft_convolve(a_trial, a_trial)
                    mt = ct.max()
                    if mt < best_trial_m - 1e-14:
                        best_trial_m = mt
                        best_frac = (src, dst, frac)
            
            if best_frac is not None:
                src, dst, frac = best_frac
                amt = a[src] * frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                new_m = fft_convolve(a, a).max()
                if new_m < best_conv_peak - 1e-14:
                    best_conv_peak = new_m
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    new_m = fft_convolve(a, a).max()
                    if new_m < best_conv_peak:
                        best_conv_peak = new_m
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_conv_peak
        print(f"    CoordDescent ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_conv_peak


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_asymmetric_init(n, tilt=0.5, shape='linear'):
    """Create an asymmetric initialization."""
    if shape == 'linear':
        x = np.linspace(0, 1, n)
        a = 1.0 + tilt * (1.0 - 2.0 * x)
    elif shape == 'quadratic':
        x = np.linspace(0, 1, n)
        a = (1.0 + tilt * (1.0 - 2.0 * x)) ** 2
    elif shape == 'exp':
        x = np.linspace(0, 1, n)
        a = np.exp(tilt * (1.0 - 2.0 * x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()
