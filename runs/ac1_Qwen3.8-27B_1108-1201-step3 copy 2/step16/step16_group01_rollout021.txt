def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=None, threshold=0.99):
    """Compute subgradient of max(ac) at point a."""
    n = len(a)
    M = ac.max()
    M_idx = ac.argmax()
    
    if top_k is not None:
        # Use top_k lags
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        # Weight by value (higher lags get more weight)
        w = top_vals / top_vals.sum()
    else:
        # Use all lags above threshold
        active = np.where(ac >= threshold * M)[0]
        if len(active) == 0:
            active = np.array([M_idx])
        w = np.ones(len(active)) / len(active)
        top_idx = active
    
    g = np.zeros(n)
    for k, wi in zip(top_idx, w):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += wi * a[j_idx]
    
    # Project onto tangent space of simplex (remove mean)
    g = g - g.mean()
    return g

def project_simplex(v, target_sum=1.0):
    """Project v onto {x : x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    lam = (cssv[rho[0]] - target_sum) / (rho[0] + 1)
    w = np.maximum(v - lam, 0)
    # Ensure exact sum
    s = w.sum()
    if s > 1e-15:
        w = w * target_sum / s
    else:
        w = np.full(n, target_sum / n)
    return w

def mirror_step(a, g, eta):
    """Mirror descent step in log-space on the simplex."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def backtracking_line_search(a, g, f0, c1=1e-4, rho=0.5, eta_max=10.0):
    """Backtracking line search for mirror descent step."""
    eta = eta_max
    for _ in range(50):
        a_new = mirror_step(a, g, eta)
        f_new = fast_autocorr(a_new).max()
        if f_new <= f0 - c1 * eta * np.sum(g * (a_new - a)):
            return a_new, eta
        eta *= rho
    return a_new, eta

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s


def compute_subgradient_multi(a, ac, top_k=30, power=2.0):
    """Compute weighted subgradient from top-k lags."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    k_use = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_use]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    # Power weighting: peaks get more emphasis
    w = top_vals ** power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_use) / k_use
    else:
        w = w / w_sum
    
    g = np.zeros(n)

        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += wi * a[j_idx]

    g = g - g.mean()
    return g


def mirror_step(a, g, eta):
    """Mirror descent step in log-space on the simplex."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def backtracking_search(a, g, f0, c1=1e-4, rho=0.7, eta_max=20.0, max_iters=40):
    """Backtracking line search for mirror descent."""
    eta = eta_max
    best_a = a.copy()
    best_f = f0
    for _ in range(max_iters):
        a_new = mirror_step(a, g, eta)
        f_new = fast_autocorr(a_new).max()
        if f_new < best_f:
            best_f = f_new
            best_a = a_new
        # Armijo condition (simplified: just require decrease)
        if f_new <= f0 - c1 * eta:
            return a_new, eta
        eta *= rho
        if eta < 1e-8:
            break
    return best_a, max(eta, 1e-8)

        # Adaptive top_k based on spread of peaks
        n_near_peak = int(np.sum(ac_y >= 0.98 * M_y))
        k_eff = max(5, min(60, n_near_peak * 3 + 5))
        
        g = compute_subgradient_multi(y, ac_y, top_k=k_eff, power=power)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Backtracking line search
        a_new, eta_used = backtracking_search(y, g, M_y, eta_max=eta)
        eta = max(eta_used * 1.1, eta * 0.9, 0.01)
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            # Add to solution pool (keep top 5)
            solution_pool.append((M_new, best_a.copy()))
            solution_pool.sort(key=lambda x: x[0])
            solution_pool = solution_pool[:5]
        else:
            stall += 1
        
        # Restart if stuck
        if stall > max_stall:
            stall = 0
            # Pick a random solution from pool
            idx = int(rng.integers(len(solution_pool)))
            base = solution_pool[idx][1].copy()
            noise_scale = 0.003 + 0.005 * np.sqrt(stall / max_stall)
            noise = rng.normal(0, noise_scale, n) * base
            x = np.maximum(base + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            eta = eta_init
            max_stall = int(max_stall * 1.2)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    AMD-BT restart at t={t} ({elapsed:.1f}s)")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD-BT t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.3f}, k_eff={k_eff}")
    
    return best_a, best_m


def peak_flattening(a0, time_limit, verbose=True, rng_seed=55):
    """
    Direct peak flattening: iteratively identify the worst peak and
    move mass to reduce it. Uses a "push-pull" strategy.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find all lags above 99% of max
        threshold = 0.99 * M_curr
        active_lags = np.where(ac >= threshold)[0]
        
        improved = False
        for k in active_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            if len(idx) == 0:
                continue
            
            contrib = a[idx] * a[j_idx]
            if contrib.max() < 1e-20:
                continue
            
            # Find the dominant pair
            max_pos = np.argmax(contrib)
            i_dom = idx[max_pos]
            j_dom = j_idx[max_pos]
            
            # Determine which to reduce (the one that contributes to more peaks)
            peak_count_i = _count_peak_contributions(i_dom, active_lags, n)
            peak_count_j = _count_peak_contributions(j_dom, active_lags, n)
            
            if a[i_dom] * peak_count_i >= a[j_dom] * peak_count_j:
                src = i_dom
            else:
                src = j_dom
            
            if a[src] < 1e-12:
                continue
            
            # Find destination: positions with low peak contribution
            # and low current value
            dest_scores = _position_peak_scores(a, active_lags[:min(10, len(active_lags))], n)
            # Exclude positions already in active lags
            in_active = np.zeros(n, dtype=bool)
            for k2 in active_lags[:min(10, len(active_lags))]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                in_active[s2:e2] = True
                j2 = k2i - np.arange(s2, e2)
                valid = (j2 >= 0) & (j2 < n)
                in_active[j2[valid]] = True
            
            candidates = np.where(~in_active)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Score candidates: low peak score and low value
            cand_scores = dest_scores[candidates] + a[candidates] * 0.1
            sorted_cand = candidates[np.argsort(cand_scores)]
            
            n_try = min(30, len(sorted_cand))
            for dst in sorted_cand[:n_try]:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Escape: random multi-perturbation
            n_pert = max(5, n // 80)
            escaped = False
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.02, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            escaped = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
                if escaped:
                    break
            if not escaped:
                # Give up if we can't escape
                break
        
        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 10 == 0 and int(elapsed) != getattr(peak_flattening, '_last_print', -1):
                peak_flattening._last_print = int(elapsed)
                J = 2.0 * n * best_m
                print(f"    PF ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def _count_peak_contributions(pos, active_lags, n):
    """Count how many active lags involve position pos."""
    count = 0
    for k in active_lags:
        k_int = int(k)
        # pos is involved if k_int - pos is in [0, n-1] or pos - k_int is in [0, n-1]
        if 0 <= k_int - pos < n:
            count += 1
        elif 0 <= pos - k_int < n and k_int != 0:
            count += 1
    return count


def _position_peak_scores(a, active_lags, n):
    """Score each position by its contribution to active peak lags."""
    scores = np.zeros(n)
    for k in active_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        if len(idx) == 0:
            continue
        j_idx = k_int - idx
        scores[idx] += a[j_idx]
    return scores


def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """
    Aggressive coordinate descent focusing on reducing the global peak.
    Uses pairwise mass transfers with line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 80
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iter_count += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lags = np.argsort(ac)[::-1][:min(8, len(ac))]
        
        improved = False
        
        # Focus on the single worst lag
        k_worst = int(peak_lags[0])
        i_start = max(0, k_worst - n + 1)
        i_end = min(n, k_worst + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_worst - idx
        
        if len(idx) > 0:
            contrib = a[idx] * a[j_idx]
            
            # Sort pairs by contribution (highest first)
            pair_order = np.argsort(contrib)[::-1]
            n_pairs_check = min(20, len(pair_order))
            
            for pi in range(n_pairs_check):
                if time.monotonic() - t0 > time_limit:
                    break
                p = pair_order[pi]
                i_p = idx[p]
                j_p = j_idx[p]
                
                # Try reducing the larger of the pair
                if a[i_p] >= a[j_p]:
                    src = i_p
                else:
                    src = j_p
                
                if a[src] < 1e-12:
                    continue
                
                # Try moving to various destinations
                # Prefer destinations far from all peak lags
                dest_candidates = _get_good_destinations(a, peak_lags, n, src, rng, max_n=40)
                
                for dst in dest_candidates:
                    if dst == src:
                        continue
                    # Binary search for optimal transfer amount
                    lo = 0.0
                    hi = min(a[src] * 0.5, a[src])
                    best_frac = None
                    best_mu = best_m
                    
                    for _ in range(6):
                        mid = (lo + hi) / 2.0
                        if mid < 1e-18:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= mid
                        a_trial[dst] += mid
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            break
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_mu:
                            best_mu = M_trial
                            best_frac = mid
                            lo = mid
                        else:
                            hi = mid
                    
                    if best_frac is not None and best_mu < best_m - 1e-14:
                        a[src] -= best_frac
                        a[dst] += best_frac
                        a = np.maximum(a, 0)
                        s_a = a.sum()
                        if s_a > 1e-15:
                            a = a / s_a
                        best_m = best_mu
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random escape
            n_esc = max(3, n // 200)
            for _ in range(n_esc):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.06)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
                if improved:
                    break
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD2 iter={iter_count} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def _get_good_destinations(a, peak_lags, n, exclude_src, rng, max_n=40):
    """Get destination positions that are 'cold' (not in peak lags)."""
    in_peak = np.zeros(n, dtype=bool)
    for k in peak_lags[:min(5, len(peak_lags))]:
        k_int = int(k)
        s = max(0, k_int - n + 1)
        e = min(n, k_int + 1)
        in_peak[s:e] = True
        j = k_int - np.arange(s, e)
        valid = (j >= 0) & (j < n)
        in_peak[j[valid]] = True
    
    in_peak[exclude_src] = True
    cold = np.where(~in_peak)[0]
    
    if len(cold) > max_n:
        # Sample from cold positions, preferring those with low values
        probs = 1.0 / (a[cold] + 0.01)
        probs = probs / probs.sum()
        return rng.choice(cold, size=max_n, replace=False, p=probs)
    return cold


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, style='sine'):
    """Create a structured initialization of length n."""
    x = np.linspace(0, 1, n)
    if style == 'sine':
        a = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2 * np.pi * x) ** 2
    elif style == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 'gaussian':
        a = np.exp(-2.0 * ((x - 0.5) / 0.3) ** 2)
    elif style == 'cosine_pow':
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase optimization with AMD+backtracking, peak flattening,
    coordinate descent, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 2400
        best_seq = make_initialization(n_base, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: AMD with backtracking (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD-BT n={n0} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = amd_backtracking(norm_sum1(best_seq), t1,
                                     rng_seed=1, verbose=True,
                                     top_k=30, power=2.0, eta_init=8.0)
        try_update(a_out, f"amd1_n{n0}")
    
    # === Phase 2: Peak flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak Flattening n={n_best} ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_out, _ = peak_flattening(norm_sum1(best_seq), t2,
                                    verbose=True, rng_seed=55)
        try_update(a_out, f"pf_n{n_best}")
    
    # === Phase 3: AMD with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD-BT (alt params) n={n_best} ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_out, _ = amd_backtracking(norm_sum1(best_seq), t3,
                                     rng_seed=2, verbose=True,
                                     top_k=50, power=3.0, eta_init=5.0)
        try_update(a_out, f"amd2_n{n_best}")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord Descent n={n_best} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, _ = coordinate_descent_v2(norm_sum1(best_seq), t4,
                                          verbose=True, rng_seed=4)
        try_update(a_out, f"cd2_n{n_best}")
    
    # === Phase 5: Peak flattening again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 5: Peak Flattening 2 n={n_best} ===")
        t5 = min(remaining * 0.10, 100.0)
        if t5 > 10:
            a_out, _ = peak_flattening(norm_sum1(best_seq), t5,
                                        verbose=True, rng_seed=66)
            try_update(a_out, f"pf2_n{n_best}")
    
    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 6: Multi-length ===")
        # Try longer
        for delta in [300, 600]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            n_try = n_best + delta
            if n_try > 4000:
                continue
            print(f"  Trying n={n_try}")
            a_long = interpolate_sequence(best_seq, n_try)
            t6 = min(remaining * 0.25, 140.0)
            if t6 > 25:
                a_out, _ = amd_backtracking(norm_sum1(a_long), t6,
                                             rng_seed=10 + delta, verbose=True,
                                             top_k=30, power=2.0, eta_init=6.0)
                try_update(a_out, f"amd_long_n{n_try}")
        
        # Try shorter
        remaining = time_budget - (time.monotonic() - t_global)
        n_best_now = len(best_seq)
        n_short = n_best_now - 300
        if remaining > 50 and n_short > 1200:
            print(f"  Trying n={n_short}")
            a_short = interpolate_sequence(best_seq, n_short)
            t6b = min(remaining * 0.20, 100.0)
            if t6b > 20:
                a_out, _ = amd_backtracking(norm_sum1(a_short), t6b,
                                             rng_seed=99, verbose=True,
                                             top_k=30, power=2.0, eta_init=6.0)
                try_update(a_out, f"amd_short_n{n_short}")
    
    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 7: Final Coord n={n_best} ===")
        t7 = min(remaining * 0.30, 120.0)
        if t7 > 10:
            a_out, _ = coordinate_descent_v2(norm_sum1(best_seq), t7,
                                              verbose=True, rng_seed=200)
            try_update(a_out, f"cd_final_n{n_best}")
    
    # === Phase 8: Final AMD sub-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 10:
            a_out, _ = amd_backtracking(norm_sum1(best_seq), t8,
                                         rng_seed=300, verbose=False,
                                         top_k=50, power=4.0, eta_init=3.0)
            try_update(a_out, f"amd_sub_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
