def smoothed_gradient_fft(a, p):
    """Compute 2*g[i] where g[i] = sum_j a[j]*p[i+j] using FFT."""
    n = len(a)
    p_len = 2 * n - 1
    p_rev = p[::-1].copy()  # p_rev[m] = p[p_len-1-m]
    # conv = a * p_rev (linear convolution, length n + p_len - 1 = 3n-2)
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    PR = np.fft.rfft(p_rev, N)
    conv = np.fft.irfft(A * PR, N)[:3*n-2]
    # g[i] = conv[2n-2-i]
    g = conv[2*n-2- np.arange(n)]  # g[i] = conv[2n-2-i]
    return 2.0 * g

```python
def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if v.sum() <= 1 and (v >= -1e-15).all():
        return np.maximum(v, 0.0) / max(v.sum(), 1e-15)
    
    # Find lambda such that sum(max(v - lambda, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n+1)) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with log-sum-exp smoothing and temperature annealing (FFT-based gradient)
2. Nesterov-accelerated subgradient with adaptive peak weighting
3. Peak-equalization local search with pairwise mass transfers
4. Multi-length exploration and final polish
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = int(rho_candidates[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def smoothed_gradient_fft(a, p):
    """Compute gradient of log-sum-exp smoothed objective using FFT.
    grad[i] = 2 * sum_j a[j] * p[i+j]
    """
    n = len(a)
    p_len = len(p)
    p_rev = p[::-1].copy()
    conv_len = n + p_len - 1
    N = 1 << (conv_len).bit_length()
    A = np.fft.rfft(a, N)
    PR = np.fft.rfft(p_rev, N)
    conv = np.fft.irfft(A * PR, N)[:conv_len]
    # g[i] = conv[p_len - 1 - i]
    g = conv[p_len - 1 - np.arange(n)]
    return 2.0 * g


def fista_smoothed(a0, time_limit, t_start=15.0, t_multiplier=1.4, t_max=800.0,
                   verbose=True, seed=42, max_stall=3000):
    """FISTA with log-sum-exp smoothing and temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    y = a.copy()
    k_fista = 1.0

    current_t = t_start
    anneal_counter = 0
    anneal_interval = 300
    stall = 0
    rng = np.random.default_rng(seed)

    for it in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        anneal_counter += 1
        if anneal_counter >= anneal_interval:
            current_t = min(current_t * t_multiplier, t_max)
            anneal_counter = 0
            anneal_interval = int(anneal_interval * 1.4)

        y_pos = np.maximum(y, 1e-12)
        sy = y_pos.sum()
        y_pos = y_pos / sy

        ac = fast_conv(y_pos)
        ac_max = ac.max()
        ac_shifted = ac - ac_max
        exp_vals = np.exp(current_t * ac_shifted)
        p = exp_vals / exp_vals.sum()

        grad = smoothed_gradient_fft(y_pos, p)

        step = 1.0 / (4.0 * current_t + 8.0)

        z = y_pos - step * grad
        z = project_simplex(z)

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * k_fista * k_fista)) / 2.0
        y_new = z + (k_fista / t_next) * (z - x)
        y_new = project_simplex(y_new)
        x = z
        y = y_new
        k_fista = t_next

        ac_x = fast_conv(x)
        m_x = ac_x.max()
        if m_x < best_m - 1e-16:
            best_m = m_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            pert = rng.normal(0, 0.002 * current_t / 100.0, n) * best_a
            x = best_a + pert
            x = project_simplex(x)
            y = x.copy()
            k_fista = 1.0
            stall = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA it={it} ({elapsed:.1f}s): J={J:.6f}, temp={current_t:.1f}, stall={stall}")

    return best_a, best_m


def nesterov_subgradient(a0, time_limit, top_k=1, beta=2.0, verbose=True, seed=7,
                         c0=3.0, max_stall=5000):
    """Nesterov accelerated subgradient with soft peak weighting on simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(x)
        M_curr = ac.max()

        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * x[j]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(x, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        beta_t = (t - 1.0) / (t + 1.5)
        x_next = a_new + beta_t * (a_new - x_prev)
        x_next = project_simplex(x_next)

        x_prev = x
        x = x_next

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall // 2:
            cur_top_k = min(cur_top_k + 1, 10)
            cur_beta = max(cur_beta * 0.85, 0.3)
        if stall > max_stall:
            x = best_a + rng.normal(0, 0.003, n) * best_a
            x = project_simplex(x)
            x_prev = x.copy()
            stall = 0
            cur_top_k = top_k
            cur_beta = beta

        if t % 5000 == 0:
            x_jump = best_a + rng.normal(0, 0.01, n) * best_a
            x_jump = project_simplex(x_jump)
            M_jump = fast_conv(x_jump).max()
            if M_jump < best_m:
                best_m = M_jump
                best_a = x_jump.copy()
                x = x_jump
                x_prev = x_jump

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSD t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={cur_top_k}, stall={stall}")

    return best_a, best_m


def pairwise_local_search(a0, time_limit, verbose=True, seed=99, max_rounds_no_improve=50):
    """Pairwise mass transfer local search focused on reducing peak convolution values."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_rounds_no_improve:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(8, len(ac))]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-25:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:4]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                j = k_int - dst
                sens = 0.0
                if 0 <= j < n:
                    sens = a[dst] + a[j]
                if sens < best_score:
                    best_score = sens
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = src_val * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-18:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Local search ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
