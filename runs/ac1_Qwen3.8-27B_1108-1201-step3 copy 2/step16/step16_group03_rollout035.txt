    valid_j = k - idx
    mask = (valid_j >= 0) & (valid_j < n)
    grad[idx[mask]] += 2.0 * a[valid_j[mask]]
    return grad

def multi_peak_gradient(a, c, top_k=3, beta=2.0):
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * peak_gradient(a, int(k))
    return grad

def interpolate_sequence_cubic(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    if old_n >= 4:
        from scipy.interpolate import CubicSpline
        cs = CubicSpline(old_x, f_norm, bc_type='natural')
        f_interp = cs(new_x)
    else:
        f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def nesterov_amd(a0, time_limit, verbose=True, top_k_init=1, c0=5.0, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 4000
    current_top_k = top_k_init
    current_beta = 1.0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        M_curr = ac.max()

        if stall > max_stall // 2:
            current_top_k = min(current_top_k + 1, 10)
            current_beta = max(current_beta * 0.85, 0.3)
        elif stall > max_stall:
            current_top_k = top_k_init
            current_beta = 1.0

        g = multi_peak_gradient(y, ac, top_k=current_top_k, beta=current_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            sigma = 0.01 + 0.02 * (stall % 5) * 0.2
            x_restarted = best_a + rng.normal(0, sigma, n) * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            current_top_k = min(current_top_k + 2, 12)
            gamma = max(gamma * 0.8, 1.0)

        if t % 8000 == 0 and t > 0:
            x_jumped = best_a + rng.normal(0, 0.015, n) * best_a
            x_jumped = np.maximum(x_jumped, 1e-8)
            x_jumped = x_jumped / x_jumped.sum()
            M_jump = fast_conv(x_jumped).max()
            if M_jump < best_m:
                best_m = M_jump
                best_a = x_jumped.copy()
                x = x_jumped
                x_prev = x_jumped

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}, stall={stall}, c0={gamma:.2f}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=500):
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    iter_count = 0
    prev_m = best_m

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        alphas = [2.0 / (t + 2.0), 0.4, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]
        best_alpha = alphas[0]
        best_eval = fast_conv(norm_sum1((1 - best_alpha) * y + best_alpha * direction)).max()

        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            trial_m = fast_conv(norm_sum1(trial)).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        y_new = norm_sum1((1.0 - best_alpha) * y + best_alpha * direction)
        y = y_new
        iter_count += 1

        my = fast_conv(y).max()
        if my < best_m - 1e-15:
            best_m = my
            best_a = y.copy()

        if verbose and t % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, alpha={best_alpha:.4f}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    full_passes = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False
        full_passes += 1

        n_candidates = min(300, n)
        candidates = rng.integers(0, n, n_candidates)

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            valid_j = k_int - idx
            mask = (valid_j >= 0) & (valid_j < n)
            valid_idx = idx[mask]
            valid_jv = valid_j[mask]
            contrib = a[valid_idx] * a[valid_jv]
            if contrib.max() < 1e-20:
                continue

            src_pos = valid_idx[np.argmax(contrib)]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:7]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            safe_cand = candidates[~in_peak[candidates]]
            if len(safe_cand) < 20:
                safe_cand = candidates
            sample = safe_cand[rng.permutation(len(safe_cand))[:min(50, len(safe_cand))]]

            best_dst = -1
            best_score = float('inf')
            for dst in sample:
                dst_int = int(dst)
                if dst_int == src_pos:
                    continue
                total_damage = 0.0
                for k2 in top_peaks[:5]:
                    k2_int = int(k2)
                    j2 = k2_int - dst_int
                    if 0 <= j2 < n:
                        total_damage += a[j2]
                if total_damage < best_score:
                    best_score = total_damage
                    best_dst = dst_int

            if best_dst < 0:
                continue
            dst = best_dst

            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src_pos] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst] += amt
                if a_trial[src_pos] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 200 and full_passes % 3 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def simulate_anneal_polish(a0, time_limit, seed=999):
    """Simulated annealing with structured moves for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_start = 0.005
    T_end = 0.0001
    total_iters = 5000

    for t in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = t / total_iters
        T = T_start * (T_end / T_start) ** progress

        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:5]

        move_type = rng.random()
        a_new = a.copy()

        if move_type < 0.4:
            # 2-opt: transfer mass from peak-contributing position to safe position
            k = int(top_peaks[rng.integers(len(top_peaks))])
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            valid_j = k - idx
            mask = (valid_j >= 0) & (valid_j < n)
            valid_idx = idx[mask]
            valid_jv = valid_j[mask]
            contrib = a[valid_idx] * a[valid_jv]
            src = valid_idx[np.argmax(contrib)]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True
            safe = np.where(~in_peak)[0]
            if len(safe) == 0:
                safe = np.arange(n)
            dst = int(safe[rng.integers(len(safe))])
            frac = rng.uniform(0.002, 0.02)
            amt = a[src] * frac
            a_new[src] -= amt
            a_new[dst] += amt
        elif move_type < 0.7:
            # Swap: exchange mass between two positions
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            amt = min(a[i], a[j]) * rng.uniform(0.1, 0.5)
            a_new[i] -= amt
            a_new[j] += amt
        else:
            # Gaussian perturbation
            a_new = a + rng.normal(0, 0.005, n) * a
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        M_new = fast_conv(a_new).max()
        delta = M_new - M_curr

        if delta < 0:
            a = a_new
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
        else:
            if rng.random() < math.exp(-delta / max(T, 1e-8)):
                a = a_new

    J = 2.0 * n * best_m
    print(f"    SA polish: J={J:.6f}")
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Hyperparameter tournament on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD tournament n={n0} ({remaining:.0f}s remaining) ===")
    configs = [
        (5.0, 1, 42),
        (4.0, 2, 7),
        (6.0, 1, 99),
        (3.5, 3, 55),
    ]
    t_per_config = min(remaining * 0.25 / len(configs), 150.0)
    if t_per_config > 20:
        for c0_val, tk_val, seed_val in configs:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_alloc = min(t_per_config, remaining * 0.3)
            if t_alloc < 15:
                break
            print(f"  AMD c0={c0_val}, top_k={tk_val}, seed={seed_val}, t={t_alloc:.0f}s")
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_amd(a_in, t_alloc, verbose=False, top_k_init=tk_val, c0=c0_val, seed=seed_val)
            try_update(a_out, f"amd_c{c0_val}_tk{tk_val}_s{seed_val}_n{n0}")

    # === Phase 2: LP Frank-Wolfe on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP n={len(best_seq)} ({remaining:.0f}s remaining) ===")
    t2 = min(remaining * 0.20, 250.0)

        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True)

    # === Phase 3: AMD polish after LP ===

    print(f"\n=== Phase 3: AMD polish n={len(best_seq)} ===")

        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
