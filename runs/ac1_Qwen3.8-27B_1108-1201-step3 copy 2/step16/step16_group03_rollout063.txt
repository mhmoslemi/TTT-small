def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def sca_qp_optimize(a0, time_limit, solver, verbose=True, num_peaks=8):

        c = fast_conv(a)
        current_peak = c.max()

            best_peak = current_peak
            best_a = a.copy()

        # Identify active peaks
        peak_vals = np.sort(c)[::-1]

        M_target = current_peak * (1.0 - step_fraction)

        active_peaks = [k for k in np.argsort(c)[::-1] if c[k] >= active_threshold][:num_peaks]
        if len(active_peaks) < 2:
            active_peaks = [k for k in np.argsort(c)[::-1]][:num_peaks]

            j_end = min(n, k_int + 1)

                coeffs[j] = a[k_int - j]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var)) <= (M_target + c[k_int]) / 2.0)

            break

        a_new_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s_new = a_new_val.sum()
        if s_new < 1e-15:
            break
        a_new = a_new_val / s_new

        # Check if it actually improved
        peak_new = fast_conv(a_new).max()
        if peak_new < best_peak - 1e-14:
            best_peak = peak_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a convex combination
            for alpha in [0.5, 0.3, 0.7, 0.2]:
                trial = (1.0 - alpha) * a + alpha * a_new
                trial = np.maximum(trial, 0.0)
                st = trial.sum()
                if st < 1e-15:
                    continue
                trial = trial / st
                peak_trial = fast_conv(trial).max()
                if peak_trial < best_peak - 1e-14:
                    best_peak = peak_trial
                    best_a = trial.copy()
                    a = trial
                    break
            else:
                # No improvement, try smaller step
                step_fraction *= 0.5
                if step_fraction < 0.0001:
                    break
        iter_count += 1

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCA iter={iter_count} ({elapsed:.0f}s): obj={2.0*n*best_peak:.6f}")

    return best_a, best_peak


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_peak = fast_conv(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x_mom = a.copy()
    iter_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()

        # Build convolution matrix
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Line search with more opciones
        alphas = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]
        # Add FW step size
        fw_alpha = 2.0 / (t + 2.0)
        alphas = [fw_alpha] + alphas

        best_alpha = alphas[0]
        best_eval = fast_conv(np.maximum((1 - best_alpha) * y + best_alpha * direction, 0)).max()

        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        y = y_new
        x_mom = x_new
        iter_count += 1

        my = fast_conv(y).max()
        if my < best_peak:
            best_peak = my
            best_a = y.copy()
        mx = fast_conv(x_mom).max()
        if mx < best_peak:
            best_peak = mx
            best_a = x_mom.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): obj={2.0*n*best_peak:.6f}")

    return best_a, best_peak


def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with gradient-guided 1D and 2D moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_peak = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        top_peaks = np.argsort(c)[::-1][:min(15, len(c))]
        improved = False

        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]

            # Find best destination using gradient info
            # Gradient at src for peak k: 2*a[k_int-src]
            # We want to move mass to dst where the impact on all top peaks is minimal
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                # Score: total impact on top peaks if we move mass here
                score = 0.0
                for k2 in top_peaks[:4]:
                    k2_int = int(k2)
                    if 0 <= k2_int - dst < n:
                        score += a[k2_int - dst]
                    if 0 <= k2_int - dst < n:
                        score += a[dst]  # approximate
                score = max(score, 1e-15)
                if score < best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_peak - 1e-16:
                    a = a_trial
                    best_peak = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 100:
                # Random perturbation
                for _ in range(50):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_conv(a).max()
                        if M_new < best_peak:
                            best_peak = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_peak
        print(f"    Coord polish ({elapsed:.0f}s): obj={J:.6f}")
    return best_a, best_peak


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n (preserving sum)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
