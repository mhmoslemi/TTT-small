            stall_count += 1

        a = a_new

        # Stall handling: perturbed restart
        if stall_count > 800 and stall_count % 400 == 0:
            pert_scale = min(0.05, 0.01 * (stall_count / 800.0))
            pert = rng.normal(0, pert_scale, n)
            a_try = best_a * (1.0 + pert)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            c_try = fast_conv(a_try)
            if c_try.max() < best_m:
                best_m = c_try.max()
                best_a = a_try.copy()
                a = a_try.copy()
            else:
                a = best_a.copy()
            stall_count = 0

        if verbose and t_iter % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD t={t_iter} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, eta={eta:.6f}")

    return best_a, best_m


def solve_lp_direction(y, M, active_lags, solver, n):
    """
    Solve LP: max sum(b) s.t. (y*b_conv)[k] <= M for k in active_lags, b >= 0.
    Returns normalized direction or None.
    """
    b_var = cp.Variable(n, nonneg=True)

    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)

    if not constraints:
        return None

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def accelerated_lp_fw(a0, time_limit, solver, top_k_init=60, top_k_max=250, verbose=True):
    """
    Accelerated LP Frank-Wolfe with adaptive constraint selection.
    """
    n = len(a0)
    a = normalize(a0)
    c = fast_conv(a)
    best_m = c.max()
    best_a = a.copy()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init

    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        if t > 100000:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute autocorrelation
        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Select active constraints
        active_lags = np.argsort(ac)[::-1][:current_top_k]

        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags, solver, n)
        if direction is None:
            # Try with fewer constraints
            if current_top_k > 20:
                active_lags2 = np.argsort(ac)[::-1][:20]
                direction = solve_lp_direction(y, M_curr, active_lags2, solver, n)
            if direction is None:
                continue

        # Step size (FW rate)
        alpha = 2.0 / (t + 2.0)

        # Update
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        # Check improvement
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Adapt top_k
        if t % 30 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 5, top_k_max)

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """
    Peak-targeted coordinate descent with sensitivity-based destination scoring.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        k_top = min(30, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:k_top]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Score destinations by sensitivity to top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)
            shuffled = candidates[rng.permutation(len(candidates))[:min(300, len(candidates))]]

            best_dst = -1
            best_score = float('inf')

            for dst in shuffled:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    if 0 <= d1 < n:
                        sens += a[d1]
                if sens < best_score:
                    best_score = sens
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in [0.003, 0.008, 0.015, 0.025, 0.04, 0.06, 0.1, 0.15]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if found:
                break  # Restart peak loop with updated a

        if not improved:
            stall += 1
            # Global perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
