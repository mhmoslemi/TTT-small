```python
def log_param_objective(x, beta):
    """
    Objective and gradient for log-parameterized optimization.
    a = softmax(x), i.e., a[j] = exp(x[j]) / sum(exp(x))
    f = (1/beta) * logsumexp(beta * conv(a,a))
    """

    # Softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute f and grad_a
    C = fast_conv(a)
    beta_C = beta * C
    bc_max = beta_C.max()
    exp_bc = np.exp(beta_C - bc_max)
    p = exp_bc / exp_bc.sum()
    f_val = (bc_max + np.log(exp_bc.sum())) / beta
    
    # grad_a[j] = 2 * sum_m p[j+m] * a[m]
    a_rev = a[::-1]
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    conv_pa = np.fft.irfft(P * A, fft_size)[:conv_size]
    grad_a = 2.0 * conv_pa[n-1:2*n-1]
    
    # grad_x[j] = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return f_val, grad_x
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-parameterized smooth-max objective with beta annealing
2. FFT-based fast gradient computation
3. Multi-length, multi-initialization exploration
4. Exponentiated gradient (mirror descent) polish
5. Coordinate descent with local search
6. Adaptive time management
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global for L-BFGS objective
_beta_g = 1.0
_n_g = 0

def _fft_size(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length() if n > 0 else 1


def fast_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fs = _fft_size(size)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def smooth_max_obj_grad(x, beta):
    """
    Log-parameterized smooth max objective and gradient.
    a = softmax(x), f = (1/beta)*logsumexp(beta*conv(a,a))
    Returns (f, grad_x).
    """
    n = len(x)
    # Softmax
    xs = x - x.max()
    ex = np.exp(xs)
    a = ex / ex.sum()
    
    # Convolution
    C = fast_conv(a)
    
    # Softmax over C
    bC = beta * C
    bC_max = bC.max()
    eb = np.exp(bC - bC_max)
    p = eb / eb.sum()
    f_val = (bC_max + np.log(eb.sum())) / beta
    
    # Gradient of f w.r.t. a:
    # g_a[j] = 2 * sum_m p[j+m] * a[m]
    # = 2 * conv(p, a_rev)[j + n - 1]
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    fs = _fft_size(conv_len)
    P = np.fft.rfft(p, fs)
    Ar = np.fft.rfft(a_rev, fs)
    conv_pa = np.fft.irfft(P * Ar, fs)[:conv_len]
    g_a = 2.0 * conv_pa[n - 1 : 2 * n - 1]
    
    # Gradient w.r.t. x: g_x[j] = a[j] * (g_a[j] - dot(g_a, a))
    dot = np.dot(g_a, a)
    g_x = a * (g_a - dot)
    
    return f_val, g_x


def _objective(x):
    """Top-level objective for scipy L-BFGS."""
    return smooth_max_obj_grad(x, _beta_g)


def lbfgs_optimize(x_init, beta_list, maxiter_per_stage=5000, time_limit=60.0):
    """
    Run L-BFGS with beta annealing.
    x_init: initial x (log-scale parameters)
    beta_list: list of beta values (increasing)
    Returns: best x found
    """
    t0 = time.monotonic()
    x = np.asarray(x_init, dtype=np.float64).copy()
    n = len(x)
    best_x = x.copy()
    best_f = np.inf
    
    for beta in beta_list:
        if time.monotonic() - t0 > time_limit:
            break
        _beta_g = beta
        # Subtract mean for stability (softmax is shift-invariant)
        x = x - x.mean()
        try:
            res = minimize(_objective, x, jac=True, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_stage,
                                   'maxfun': maxiter_per_stage * 2,
                                   'ftol': 1e-12, 'gtol': 1e-8})
            x = res.x
        except Exception as e:
            print(f"    L-BFGS error at beta={beta}: {e}")
            break
        
        # Evaluate
        f_curr, _ = smooth_max_obj_grad(x, beta)
        # Also evaluate with high beta for "true" objective
        f_true, _ = smooth_max_obj_grad(x, 10000.0)
        if f_true < best_f:
            best_f = f_true
            best_x = x.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    beta={beta:8.1f}: f_true={f_true:.8f} ({elapsed:.1f}s, nit={res.nit})")
    
    return best_x

def interpolate_to_length(a, new_n):
    """Interpolate sequence a to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = np.maximum(a, 0.0)
    s = a_norm.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a_norm / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return result / rs

def mirror_descent_polish(a_init, time_limit=60.0, seed=42, top_k=3, c0=5.0):
    """
    Exponentiated gradient (mirror descent) with adaptive subgradient.
    a_init: initial sequence (will be normalized)
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    current_top_k = top_k
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Convolution and peaks
        C = fast_conv(y)
        M_curr = C.max()
        
        # Adaptive subgradient
        k_max = min(current_top_k, len(C))
        top_idx = np.argsort(C)[::-1][:k_max]
        top_vals = np.maximum(C[top_idx], 1e-30)
        w = top_vals ** 2.0
        w = w / w.sum()
        
        # Combined gradient
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            # g[j] += 2 * wi * a[k-j] (if valid)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                m = k_int - j
                if 0 <= m < n:
                    g[j] += 2.0 * wi * y[m]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > max_stall:
            x_new = best_a * (1.0 + rng.normal(0, 0.01, n))
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
            stall = 0
            current_top_k = min(current_top_k + 1, 8)
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): score={J:.6f}, top_k={current_top_k}")
    
    return best_a


def coordinate_polish(a_init, time_limit=30.0, seed=123):
    """
    Coordinate descent: try transferring mass between pairs.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_conv(a)
        M_curr = C.max()
        top_peaks = np.argsort(C)[::-1][:min(5, len(C))]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            idx = np.arange(j_start, j_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            
            # Try moving mass from src to random destinations
            n_tries = 50
            for _ in range(n_tries):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if not improved and stall > 50:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
        
        if not improved:
            stall += 1
    
    return best_a
