"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with softmax parametrization
2. Adaptive epsilon (smoothing) schedule
3. Multi-start with diverse initializations  
4. Multi-length exploration
5. Final coordinate descent polishing
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_from_theta(theta):
    theta_max = theta.max()
    a = np.exp(theta - theta_max)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return a / s

def compute_grad_a(a, p, n):
    """Compute grad_a[j] = 2 * sum_k p[k] * a[k-j] * 1[0<=k-j<n]"""
    flip_a = a[::-1]
    # Convolution of flip_a (length n) with p (length 2n-1)
    # Result has length n + 2n - 1 - 1 = 3n - 2
    # We want indices [n-1, n-1+n-1] = [n-1, 2n-2]
    L = 1 << (3 * n - 2 - 1).bit_length()
    F_flip_a = np.fft.rfft(flip_a, L)
    F_p = np.fft.rfft(p, L)
    C = np.fft.irfft(F_flip_a * F_p, L)
    return 2.0 * C[n-1 : 2*n-1]

def obj_and_grad(theta, epsilon):
    a = softmax_from_theta(theta)
    n = len(a)
    ac = fast_conv(a)
    
    ac_max = ac.max()
    shifted = (ac - ac_max) / epsilon
    shifted = np.clip(shifted, -100, 100)
    exp_s = np.exp(shifted)
    exp_s_sum = exp_s.sum()
    f_val = epsilon * (np.log(exp_s_sum) + ac_max / epsilon)
    p = exp_s / exp_s_sum
    
    grad_a = compute_grad_a(a, p, n)
    ga_dot_a = np.dot(grad_a, a)
    grad_theta = a * (grad_a - ga_dot_a)
    
    return f_val, grad_theta

def lbfgs_smoothed(theta0, epsilon, max_iter=2000, tol=1e-10):
    """Run L-BFGS to minimize the smoothed objective."""
    result = minimize(
        fun=lambda t: obj_and_grad(t, epsilon)[0],
        x0=theta0,
        jac=lambda t: obj_and_grad(t, epsilon)[1],
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8, 'maxls': 50}
    )
    return result.x, result.fun, result.nit, result.status

def run_epsilon_schedule(a_init, time_budget, verbose=True):
    """Run the adaptive epsilon schedule starting from a_init."""
    n = len(a_init)
    a = norm_sum1(a_init)
    
    # Initial theta
    theta = np.log(a + 1e-30)
    # Actually, for softmax, we can set theta = log(a) since softmax(log(a)) = a / sum(a) = a (if sum=1)
    
    best_a = a.copy()
    best_score = evaluate_safe(a)
    
    # Initial epsilon
    ac = fast_conv(a)
    top_vals = np.sort(ac)[::-1]
    gap = top_vals[0] - top_vals[min(5, len(top_vals)-1)]
    if gap < 1e-15:
        epsilon = top_vals[0] / 50.0
    else:
        epsilon = max(gap / 5.0, top_vals[0] / 100.0)
    
    t0 = time.monotonic()
    stage = 0
    
    while time.monotonic() - t0 < time_budget * 0.7:
        stage += 1
        # Run L-BFGS
        theta, f_val, nit, status = lbfgs_smoothed(theta, epsilon, max_iter=500, tol=1e-12)
        a_new = softmax_from_theta(theta)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        sc = evaluate_safe(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            theta = np.log(a + 1e-30)
        
        if verbose and stage % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Stage {stage}: eps={epsilon:.2e}, score={best_score:.6f}, iters={nit}, t={elapsed:.0f}s")
        
        # Update epsilon: decrease
        ac_curr = fast_conv(a)
        top_vals = np.sort(ac_curr)[::-1]
        gap = top_vals[0] - top_vals[min(5, len(top_vals)-1)]
        if gap > 1e-15:
            epsilon = max(gap / 5.0, top_vals[0] / 200.0)
        else:
            epsilon = top_vals[0] / 200.0
        
        # Check if we've converged (epsilon too small)
        if epsilon < top_vals[0] * 1e-6:
            break
        epsilon *= 0.7  # Geometric decrease
    
    # Final: use exact max (epsilon = 0 effectively)
    final_a = best_a
    final_score = evaluate_safe(final_a)
    
    return final_a, final_score
