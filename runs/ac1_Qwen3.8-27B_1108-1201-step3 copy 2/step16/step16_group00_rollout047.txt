def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_adaptive_subgradient(a, ac, n_lags_max=200, sharpness=6.0):
    """Compute subgradient using adaptive multi-scale weighting over top lags."""
    n = len(a)
    n_lags = min(n_lags_max, len(ac))
    top_idx = np.argsort(ac)[::-1][:n_lags]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    # Use a two-scale weighting: sharp on the very top, soft on the rest
    w_sharp = top_vals ** (sharpness * 2)
    w_soft = top_vals ** sharpness
    w = 0.3 * w_sharp / w_sharp.sum() + 0.7 * w_soft / w_soft.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def compute_cost_landscape(a, ac, n_lags=200):

    n = len(a)
    n_lags = min(n_lags, len(ac))
    top_idx = np.argsort(ac)[::-1][:n_lags]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals / top_vals.sum()
    cost = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        cost[idx] += wi * (a[j_idx] + a[idx])
    return cost

            for j in range(j_start, j_end):
                T_sel[ii, j] = y[k_int - j]
            M_sel[ii] = M_curr

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new
        solve_iter += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, M={M_curr:.8f}")

    return best_a, best_m


def coordinate_desc_global(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent using global cost landscape for smart mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    n_lags_cost = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        cost = compute_cost_landscape(a, ac, n_lags=n_lags_cost)

        # Find best source-destination pair
        # Source: high cost, high value
        # Dest: low cost, low value (empty space)
        src_score = cost * np.log1p(a * 100)
        dst_score = cost  # want low cost destination

        # Top sources
        n_src = min(50, n)
        top_src = np.argsort(src_score)[::-1][:n_src]
        top_src = top_src[a[top_src] > 1e-10]
        if len(top_src) == 0:
            top_src = np.argsort(a)[::-1][:n_src]

        # Best destinations: lowest cost, excluding sources
        src_set = set(top_src.tolist())
        dst_candidates = np.array([i for i in range(n) if i not in src_set])
        if len(dst_candidates) == 0:
            dst_candidates = np.arange(n)
        n_dst = min(50, len(dst_candidates))
        top_dst = dst_candidates[np.argsort(dst_score[dst_candidates])[:n_dst]]

        improved_this_round = False

        # Try best source-dest pairs
        for src in top_src[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-10:
                continue

            for dst in top_dst[:10]:
                if dst == src:
                    continue
                # Try multiple fractions
                for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved_this_round = True
                        stall = 0
                        break
                if improved_this_round:
                    break

        if not improved_this_round:
            stall += 1
            # Random perturbation to escape
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

            # Larger perturbation if stuck for a long time
            if stall > 100 and stall % 50 == 0:
                pert = rng.normal(0, 0.05, n) * a
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / a_pert.sum()
                M_pert = fast_autocorr(a_pert).max()
                if M_pert < best_m:
                    a = a_pert
                    best_m = M_pert
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def pair_exchange_polish(a0, time_limit, verbose=True, rng_seed=456):
    """Polish using pairwise exchanges with sensitivity-guided search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    ac = fast_autocorr(a)
    # Identify peak lags
    n_peaks = min(100, len(ac))
    peak_lags = np.argsort(ac)[::-1][:n_peaks]

    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lags = np.argsort(ac)[::-1][:n_peaks]

        # For each peak lag, find the pair of positions that most contributes
        # and try to split them
        improved = False
        for k in peak_lags[:30]:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start >= j_end:
                continue
            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            max_j = idx[np.argmax(contrib)]
            max_val = a[max_j]
            if max_val < 1e-10:
                continue

            partner = k_int - max_j
            if partner == max_j:
                # Auto-correlation at same position, need to spread
                # Move mass to random positions
                for dst in rng.choice(n, size=10, replace=False):
                    if dst == max_j:
                        continue
                    for frac in [0.05, 0.1, 0.2]:
                        amt = max_val * frac
                        a_trial = a.copy()
                        a_trial[max_j] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
            elif 0 <= partner < n and a[partner] > 1e-10:
                # Two different positions contribute; try to reduce one or both
                # Move mass from the larger contributor to a low-impact position
                src = max_j if a[max_j] >= a[partner] else partner
                other = partner if src == max_j else max_j
                src_val = a[src]
                # Find a good destination
                dest_scores = np.zeros(n)
                for k2 in peak_lags[:10]:
                    k2i = int(k2)
                    t_start = max(0, k2i - n + 1)
                    t_end = min(n, k2i + 1)
                    for d in range(t_start, t_end):
                        dest_scores[d] += a[k2i - d]
                # Avoid src, other, and high-score positions
                dest_scores[src] = np.inf
                dest_scores[other] = np.inf
                good_dsts = np.argsort(dest_scores)[:20]

                for dst in good_dsts:
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break

            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random pair swap
                i, j = rng.choice(n, size=2, replace=False)
                if a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.02, 0.1)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Pair ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Pair final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with enhanced multi-phase optimization."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: AMD on parent with enhanced parameters ===
    print(f"\n=== Phase 1: Enhanced AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)
    if t_phase1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t_phase1, n_lags_init=100, sharpness_init=8.0,
                                c0=3.0, restart_every=5000, verbose=True)
        try_update(a_out, "amd_parent")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_a0(a_in, t2, solver, verbose=True, top_K=500)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 3: AMD polish after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD polish (n={n_best}) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t3, n_lags_init=200, sharpness_init=5.0,
                                c0=2.5, restart_every=8000, verbose=False)
        try_update(a_out, f"amd_polish_n{n_best}")

    # === Phase 4: Global coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_desc_global(a_in, t4, verbose=True, rng_seed=123)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Pair exchange polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pair polish (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_exchange_polish(a_in, t5, verbose=True, rng_seed=456)
        try_update(a_out, f"pair_n{n_best}")

    # === Phase 6: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    for new_n in [n_best + 400, n_best + 800]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        if new_n > 5000:
            continue
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        sc_long = evaluate_safe(a_long)
        if np.isfinite(sc_long) and sc_long < best_score:
            best_score = sc_long
            best_seq = a_long.copy()
            print(f"    Interpolated n={new_n}: {sc_long:.6f}")
        # Optimize the longer sequence
        t6a = min(remaining * 0.3, 120.0)
        if t6a > 30:
            a_out, _ = amd_optimize(a_long, t6a, n_lags_init=100, sharpness_init=6.0,
                                    c0=3.0, restart_every=5000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60 and new_n <= 3500:
                t6b = min(remaining * 0.2, 80.0)
                if t6b > 20:
                    a_in2 = norm_sum1(best_seq)
                    a_out2, _ = lp_frank_wolfe_a0(a_in2, t6b, solver, verbose=False, top_K=300)
                    try_update(a_out2, f"lp_n{new_n}")

    # === Phase 7: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t7 = min(remaining * 0.3, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_a0(a_in, t7, solver, verbose=True, top_K=600)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.4, 120.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_desc_global(a_in, t8, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 2400
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
