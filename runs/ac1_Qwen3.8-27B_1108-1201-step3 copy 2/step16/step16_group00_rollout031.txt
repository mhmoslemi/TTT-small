"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with adaptive constraint selection and backtracking line search
2. Exact subgradient mirror descent targeting the highest convolution peak
3. Peak-targeted coordinate descent with sensitivity analysis
4. Multi-length exploration with interpolation
Key improvement over prior work: tighter LP constraints using actual peak values,
adaptive step sizes via backtracking, and subgradient method using the true
subdifferential of the max function.

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Compute full linear convolution a*a using FFT. Result has length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def conv_gradient_single(p, k):
    """Gradient of conv(p,p)[k] w.r.t. p. Returns vector of length n."""
    n = len(p)
    g = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    js = np.arange(j_start, j_end)
    idx = k - js
    valid = (idx >= 0) & (idx < n)
    if valid.any():
        g[js[valid]] += 2.0 * p[idx[valid]]
    return g


def build_constraint_matrix(p, k_list):
    """Build K x n matrix T where (T @ b)[r] = conv(p, b)[k_list[r]]."""
    n = len(p)
    K = len(k_list)
    T = np.zeros((K, n))
    for r, k in enumerate(k_list):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        js = np.arange(j_start, j_end)
        idx = k_int - js
        valid = (idx >= 0) & (idx < n)
        if valid.any():
            T[r, js[valid]] = p[idx[valid]]
    return T


def lp_frank_wolfe(a0, time_limit, solver, top_K=300, verbose=True):
    """LP-based Frank-Wolfe with backtracking and adaptive K."""
    n = len(a0)
    p = norm_sum1(a0)
    best_p = p.copy()
    best_obj = fast_conv(p).max()
    t0 = time.monotonic()

    y = p.copy()
    x = p.copy()
    prev_y = p.copy()
    iter_count = 0
    cur_K = top_K
    backtrack_count = 0

    for t in range(1, 200000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv_y = fast_conv(y)
        M_curr = conv_y.max()

        # Select top-K binding constraints
        K_sel = min(cur_K, 2 * n - 1)
        binding_idx = np.argsort(conv_y)[::-1][:K_sel]

        # Build constraint matrix
        T = build_constraint_matrix(y, binding_idx)
        rhs = conv_y[binding_idx]

        # Solve LP: maximize sum(b) s.t. T @ b <= rhs, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: use uniform direction
            b_val = np.ones(n) / n
        else:
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                b_val = np.ones(n) / n
            else:
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

        b_sum = b_val.sum()
        if b_sum < 1e-15:
            b_val = np.ones(n) / n
        else:
            b_val = b_val / b_sum

        # Nesterov-accelerated FW step with backtracking
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_val
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Backtracking: try to ensure objective decreases
        obj_new = fast_conv(y_new).max()
        if obj_new > M_curr * (1 + 1e-10) and t > 10:
            # Backtrack: use smaller step
            for frac in [0.5, 0.25, 0.1, 0.05]:
                y_bt = (1.0 - alpha * frac) * y + alpha * frac * b_val
                y_bt = np.maximum(y_bt, 0.0)
                sy = y_bt.sum()
                if sy < 1e-15:
                    continue
                y_bt = y_bt / sy
                obj_bt = fast_conv(y_bt).max()
                if obj_bt < M_curr:
                    y_new = y_bt
                    break
            else:
                y_new = y.copy()  # No progress this step

        # Update x (Nesterov point)
        if t == 1:
            x = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x = y_new + beta_t * (y_new - y)
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                x = y_new.copy()
            else:
                x = x / sx

        prev_y = y
        y = y_new
        iter_count += 1

        # Track best
        for cand in [y, x]:
            obj_c = fast_conv(cand).max()
            if obj_c < best_obj - 1e-15:
                best_obj = obj_c
                best_p = cand.copy()

        # Adaptive K: increase K when stagnating
        if iter_count % 200 == 0 and best_obj > fast_conv(prev_y).max() - 1e-10:
            cur_K = min(cur_K + 50, 2 * n - 1)

        # Periodic restart to escape local minima
        if iter_count % 500 == 0:
            noise = np.random.default_rng(t).normal(0, 0.05, n) * best_p
            p_restart = best_p + noise
            p_restart = np.maximum(p_restart, 1e-10)
            p_restart = p_restart / p_restart.sum()
            obj_restart = fast_conv(p_restart).max()
            if obj_restart < best_obj:
                best_obj = obj_restart
                best_p = p_restart.copy()
            y = p_restart.copy()
            x = p_restart.copy()

        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={cur_K}")

    return best_p, best_obj


def subgradient_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent on the simplex using exact subgradient at the argmax peak."""
    n = len(a0)
    p = norm_sum1(a0)
    best_p = p.copy()
    best_obj = fast_conv(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    # Initialize log-space variable
    log_p = np.log(np.maximum(p, 1e-30))
    old_log_p = log_p.copy()
    gamma = 5.0
    stall = 0

    for t in range(1, 5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Recover p from log_p
        log_p_shifted = log_p - log_p.max()
        p_curr = np.exp(log_p_shifted)
        p_curr = p_curr / p_curr.sum()

        # Compute convolution and find argmax
        conv_p = fast_conv(p_curr)
        M_curr = conv_p.max()
        k_star = int(np.argmax(conv_p))

        # Compute subgradient: gradient of conv(p,p)[k_star]
        g = conv_gradient_single(p_curr, k_star)

        # Also add small weight to second-highest peak for robustness
        sorted_idx = np.argsort(conv_p)[::-1]
        if len(sorted_idx) > 1:
            k2 = int(sorted_idx[1])
            g2 = conv_gradient_single(p_curr, k2)
            w2 = min(1.0, conv_p[k2] / max(M_curr, 1e-30))
            g = (1.0 - w2) * g + w2 * g2

        # Mirror descent step with Nesterov acceleration
        beta_t = (t - 1.0) / (t + 1.5)
        log_y = log_p + beta_t * (log_p - old_log_p)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_p_new = log_y - eta * g
        log_p_new = log_p_new - log_p_new.max()
        p_new = np.exp(log_p_new)
        p_new = np.maximum(p_new, 1e-30)
        p_new = p_new / p_new.sum()

        old_log_p = log_p
        log_p = log_p_new

        # Evaluate
        M_new = fast_conv(p_new).max()
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_p = p_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stuck
        if stall > 3000:
            stall = 0
            noise = rng.normal(0, 0.03, n) * best_p
            p_r = best_p + noise
            p_r = np.maximum(p_r, 1e-10)
            p_r = p_r / p_r.sum()
            log_p = np.log(np.maximum(p_r, 1e-30))
            old_log_p = log_p.copy()
            gamma = max(gamma * 0.8, 1.0)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"    SGD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_p, best_obj


def coordinate_polish(a0, time_limit, verbose=True, seed=42):
    """Peak-targeted coordinate descent: move mass from hot to cold positions."""
    n = len(a0)
    p = norm_sum1(a0)
    best_p = p.copy()
    best_obj = fast_conv(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv_p = fast_conv(p)
        M_curr = conv_p.max()
        k_star = int(np.argmax(conv_p))

        # Find the "hot" positions for this peak
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        js = np.arange(j_start, j_end + 1)
        pairs = k_star - js
        products = p[js] * p[pairs]

        # Source: position with highest product contribution
        src_local = np.argmax(products)
        src = int(js[src_local])
        src_val = p[src]
        if src_val < 1e-12:
            stall += 1
            continue

        # Compute sensitivity for each destination:
        # How much would adding mass at position d increase the top peaks?
        top_peaks = np.argsort(conv_p)[::-1][:15]
        sens = np.zeros(n)
        for kp in top_peaks:
            kp_int = int(kp)
            for d in range(n):
                d1 = kp_int - d
                if 0 <= d1 < n:
                    sens[d] += p[d1]

        # Find best destination: low sensitivity, not the source
        candidates = np.array([d for d in range(n) if d != src])
        if len(candidates) == 0:
            stall += 1
            continue
        best_dst = candidates[np.argmin(sens[candidates])]

        # Try several move fractions
        improved = False
        for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            amt = src_val * frac
            if amt < 1e-15:
                continue
            p_trial = p.copy()
            p_trial[src] -= amt
            p_trial[best_dst] += amt
            p_trial = np.maximum(p_trial, 0.0)
            st = p_trial.sum()
            if st < 1e-15:
                continue
            p_trial = p_trial / st
            M_trial = fast_conv(p_trial).max()
            if M_trial < best_obj - 1e-15:
                p = p_trial
                best_obj = M_trial
                best_p = p_trial.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Also try moving to a random position
            for trial_src in [src, int(rng.integers(n))]:
                if p[trial_src] < 1e-12:
                    continue
                dst = int(rng.integers(n))
                if dst == trial_src:
                    continue
                frac = rng.uniform(0.002, 0.02)
                amt = p[trial_src] * frac
                p_trial = p.copy()
                p_trial[trial_src] -= amt
                p_trial[dst] += amt
                p_trial = np.maximum(p_trial, 0.0)
                st = p_trial.sum()
                if st < 1e-15:
                    continue
                p_trial = p_trial / st
                M_trial = fast_conv(p_trial).max()
                if M_trial < best_obj - 1e-15:
                    p = p_trial
                    best_obj = M_trial
                    best_p = p_trial.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_obj
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_p, best_obj


def make_analytical_init(n, kind='ramp'):
    """Generate diverse analytical initial sequences."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'linear_ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif kind == 'quadratic_ramp':
        a = np.arange(1, n + 1, dtype=np.float64) ** 2
        return a / a.sum()
    elif kind == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-40 * (x - 0.1) ** 2) + np.exp(-40 * (x - 0.9) ** 2) + 0.05 * np.ones(n)
        return a / a.sum()
    elif kind == 'trapezoid':
        x = np.linspace(0, 1, n)
        a = np.minimum(1.0, np.minimum(x / 0.2, (1 - x) / 0.2))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == 'smooth_random':
        rng = np.random.default_rng(123)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for freq in range(1, 6):
            coeff = rng.normal(0, 0.15)
            phase = rng.uniform(0, 2 * np.pi)
            a += coeff * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.0)
        return a / a.sum()
    else:
        return np.ones(n) / n

            t3b = min(remaining * 0.5, 150.0)
            a_out, _ = coordinate_polish(a_out, t3b, verbose=True, seed=999)
            _try_update(a_out, f"coord2_n{n_best}")

    # === Phase 4: Try different lengths (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-length exploration ===")
    candidate_lengths = [n_best + 500, n_best + 1000, n_best - 400]
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 5500]
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for l in candidate_lengths:
        if l not in seen and l != n_best:
            seen.add(l)
            unique_lengths.append(l)

    for new_n in unique_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        t_lp = min(remaining * 0.4, 180.0)
        if t_lp > 20:
            a_out, _ = lp_frank_wolfe(a_long, t_lp, solver, top_K=200, verbose=False)
            _try_update(a_out, f"lpfw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t_sgd = min(remaining * 0.3, 120.0)
            if t_sgd > 15:
                a_out, _ = subgradient_mirror_descent(a_out, t_sgd, verbose=False)
                _try_update(a_out, f"sgd_n{new_n}")

    # === Phase 5: Final LP on best (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final optimization (n={n_best}, rem={remaining:.0f}s) ===")
    if remaining > 40 and n_best <= 4000:
        t_final = min(remaining * 0.5, 300.0)
        if t_final > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t_final, solver, top_K=400, verbose=True)
            _try_update(a_out, "lp_final")

    # === Final: coordinate polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, remaining * 0.8, verbose=False, seed=2024)
        _try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.05
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
