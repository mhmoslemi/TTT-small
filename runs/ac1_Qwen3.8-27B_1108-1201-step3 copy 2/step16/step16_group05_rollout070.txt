"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative step function a.

Key observations used here:
  * The objective is scale-invariant, so we normalize sum(a)==1 and the score is 2*n*max(conv(a,a)).
  * The problem is NON-CONVEX (max of quadratic forms a^T Q_k a with Q_k indefinite), so convex
    global guarantees do not apply; we combine several non-convex local methods and diverse starts:
      1. peak_direction_search : projected backtracking steps along the (soft) subgradient of the
         active peak lags of the linear convolution, with adaptive restarts.
      2. lp_frank_wolfe        : conditional-gradient / Frank-Wolfe local search using a sampled
         subset of the linear constraints conv(a,b) <= M; gives strong polishing directions.
      3. coordinate_polish     : exact 1-D line search per coordinate against the current autocorr.
      4. freq / bump inits     : flat-spectrum and structured-bump starts to escape the parent basin.
  * All autocorrelations are evaluated via fixed-size FFT for speed.

All helpers are top level (no nested functions / closures / lambdas).

def fast_autocorr(a):
    """Linear convolution a*a via FFT (non-negative real input expected)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    size = 2 * n - 1
    fft_size = 1 << (max(size, 1) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(out, 0.0)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, len(a))
    return a / s


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.shape[0]
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    rho = idx[-1] if idx.size > 0 else 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def peak_gradient(a, top_k=5, temp_frac=0.05):
    """
    Soft-weighted subgradient of max_k conv(a,a)[k].
    d/d a[m] conv(a,a)[k] = 2*a[k-m]  when 0 <= k-m < n, else 0.
    """
    n = a.shape[0]
    ac = fast_autocorr(a)
    order = np.argsort(ac)[::-1]
    top_idx = order[:max(1, top_k)]
    top_vals = ac[top_idx]
    temp = max(top_vals[0] * temp_frac, 1e-12)
    w = np.exp((top_vals - top_vals[0]) / temp)
    w = w / w.sum()

    grad = np.zeros(n)
    for wi, kk in zip(w, top_idx):
        k = int(kk)
        m_start = max(0, k - (n - 1))
        m_end = min(n, k + 1)
        if m_start >= m_end:
            continue
        m_idx = np.arange(m_start, m_end)
        partner = k - m_idx
        grad[m_start:m_end] += 2.0 * wi * a[partner]
    return grad, float(ac.max()), ac


def peak_direction_search(a0, time_limit, verbose=True, seed=42, start_top_k=5):

            break

        if gn < 1e-15:
            break

def lp_frank_wolfe(a0, time_limit, solver, n_top_constraints=None, verbose=True):
    """Maximize sum(b) s.t. conv(a,b) <= M, b >= 0  -> direction; line-search update."""
    n = a0.shape[0]
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()

    if n_top_constraints is None:
        n_top_constraints = min(2 * n - 1, max(300, 2 * n))

    # build Toeplitz-like matrix T[k,j] = a[j] if 0 <= k-j < n else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    b_var = cp.Variable(n, nonneg=True)

    for t in range(1, 40000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        M_curr = float(ac.max())

        if (2 * n - 1) > n_top_constraints:
            top_lags = np.argsort(ac)[::-1][:n_top_constraints]
            T_sel = T[top_lags]
        else:
            T_sel = T
        mrows = T_sel.shape[0]
        M_sel = np.full(mrows, M_curr)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_sel @ b_var <= M_sel])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        if b_var.value is None:
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64).ravel(), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        best_alpha = 0.0
        best_m_trial = M_curr
        a_trial_best = None
        for alpha in (0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
            at = (1.0 - alpha) * a + alpha * g_dir
            at = np.maximum(at, 0.0)
            s = at.sum()
            if s < 1e-15:
                continue
            at /= s
            mt = float(fast_autocorr(at).max())
            if mt < best_m_trial:
                best_m_trial = mt
                best_alpha = alpha
                a_trial_best = at

        if a_trial_best is not None:
            a = a_trial_best
        if best_m_trial < best_m - 1e-15:
            best_m = best_m_trial
            best_a = a.copy()

        if verbose and (t % 4 == 0):
            print("      LP-FW it=%d (%.0fs): J=%.6f" % (t, time.monotonic() - t0, 2.0 * n * best_m))

    return best_a, best_m

def freq_domain_init(n, rng):
    """Flat-spectrum random phase -> threshold to non-negative -> normalize."""
    fft_size = 1 << (2 * n - 1).bit_length()
    spec = np.ones(fft_size // 2 + 1)
    spec[0] = 0.0
    ph = rng.uniform(0.0, 2.0 * np.pi, spec.shape[0])
    spec = spec * np.exp(1j * ph)
    ac = np.fft.irfft(spec, fft_size)[:n]
    a = np.maximum(ac, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def smooth_bump_init(n, rng):
    """Gaussian bump * random sinusoidal modulation -> non-negative -> normalize."""
    sigma = rng.uniform(0.06, 0.40)
    center = rng.uniform(0.20, 0.80)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    mod = 1.0 + rng.uniform(0.1, 0.4) * np.sin(2.0 * np.pi * rng.uniform(1.0, 6.0) * x)
    a = np.maximum(a * mod, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.shape[0]
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    if old_n == 1:
        return np.ones(new_n) / new_n
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        t5 = min(rem * 0.45, 200.0)
        out, _ = peak_direction_search(norm_sum1(al), t5, verbose=True, seed=99, start_top_k=8)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, out
            print("  *** NEW BEST [p5_len%d]: %.6f" % (nt, sc))
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 40 and out.shape[0] <= 3200:
            out2, _ = lp_frank_wolfe(norm_sum1(out), min(rem * 0.5, 90.0), solver, verbose=False)
            sc2 = evaluate_safe(out2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score, best_seq = sc2, out2
                print("  *** NEW BEST [p5_len%d_lpfw]: %.6f" % (nt, sc2))

    # ------------------------------------------------------------------
    # Phase 6: diverse basins (freq + bump) at current best length
    # ------------------------------------------------------------------
    rem = time_budget - (time.monotonic() - t_global)
    nbst = best_seq.shape[0]
    print("\n[Phase6] Diverse inits n=%d rem=%.0fs" % (nbst, rem))
    inits = []
    inits.append((freq_domain_init(nbst, rng), "freq"))
    inits.append((smooth_bump_init(nbst, rng), "bump"))
    bl = 0.55 * norm_sum1(best_seq) + 0.45 * smooth_bump_init(nbst, rng)
    inits.append((norm_sum1(bl), "blend"))
    for label, ini in inits:
        rem = time_budget - (time.monotonic() - t_global)
        if rem < 60:
            break
        t6 = min(rem * 0.4, 110.0)
        out, _ = peak_direction_search(norm_sum1(ini), t6, verbose=False, seed=555, start_top_k=8)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, out
            print("  *** NEW BEST [p6_%s]: %.6f" % (label, sc))

    # ------------------------------------------------------------------
    # Phase 7: final coordinate + LP polish on the global best
    # ------------------------------------------------------------------
    rem = time_budget - (time.monotonic() - t_global)
    nbst = best_seq.shape[0]
    print("\n[Phase7] Final polish n=%d rem=%.0fs" % (nbst, rem))
    if rem > 30:
        out, _ = coordinate_polish(norm_sum1(best_seq), min(rem * 0.55, 120.0),
                                   verbose=True, seed=2718)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, out
            print("  *** NEW BEST [p7_coord]: %.6f" % sc)
    rem = time_budget - (time.monotonic() - t_global)
    nbst = best_seq.shape[0]
    if rem > 50 and nbst <= 3000:
        out, _ = lp_frank_wolfe(norm_sum1(best_seq), min(rem * 0.7, 150.0), solver, verbose=True)
        sc = evaluate_safe(out)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, out
            print("  *** NEW BEST [p7_lpfw]: %.6f" % sc)

    # ------------------------------------------------------------------
    # Final validation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        print("ERROR: no valid sequence, returning single 1.0")
        return [1.0]
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("\nFINAL score=%.6f n=%d t=%.0fs" % (best_score, best_seq.shape[0],
                                               time.monotonic() - t_global))
    return [float(x) for x in best_seq]
