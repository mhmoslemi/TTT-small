    t_val = float(t_var.value)
    
    # Get dual values for the T@c <= t constraints
    dual_w = np.zeros(len(active_lags))
    try:
        for i, con in enumerate(constraints[1:len(active_lags)+1]):
            if con.dual_value is not None:
                dual_w[i] = max(0, float(con.dual_value))
    except Exception:
        pass
    
    return c_val, t_val, dual_w

def minimax_lp_v2(a, M_target, active_lags, solver):
    """
    Alternative: maximize sum(c) - lambda*t s.t. (a*c)(k) <= t, sum(c)=1, c>=0
    This is equivalent to minimax LP but sometimes solves faster.
    Actually let's just do: minimize t with a slightly relaxed constraint.
    """
    n = len(a)
    T = build_T_matrix(a, active_lags)
    
    c = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [cp.sum(c) == 1, T @ c <= t_var]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    c_val = np.maximum(np.asarray(c.value, dtype=np.float64), 0.0)
    c_sum = c_val.sum()
    if c_sum < 1e-15:
        return None, None
    c_val = c_val / c_sum
    t_val = float(t_var.value)
    return c_val, t_val

def lp_fw_minimax(a0, time_limit, solver, verbose=True):
    """
    Frank-Wolfe with minimax LP: at each step, find c that minimizes
    max_k (a*c)(k) over active lags, then take a step towards c.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select active lags: top lags by value
        n_active = min(150, len(ac))
        active_lags = np.argsort(ac)[::-1][:n_active]
        
        # Solve minimax LP
        c_val, t_val = minimax_lp_v2(y, M_curr, active_lags, solver)
        
        if c_val is None:
            # Fallback: simple FW step (maximize sum)
            T = build_T_matrix(y, active_lags)
            c_var = cp.Variable(n, nonneg=True)
            constraints = [T @ c_var <= M_curr, cp.sum(c_var) == 1]
            prob = cp.Problem(cp.Maximize(cp.sum(c_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            c_val = np.maximum(np.asarray(c_var.value, dtype=np.float64), 0.0)
            c_sum = c_val.sum()
            if c_sum < 1e-15:
                continue
            c_val = c_val / c_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * c_val
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-MM iter={t} ({elapsed:.1f}s): J={J:.6f}, t_lp={t_val:.6f}" if t_val else f"    LP-MM iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def peak_repulsion_descent(a0, time_limit, verbose=True):
    """
    Mirror descent using a subgradient that weights by the autocorrelation
    values themselves (power-weighted), pushing mass away from peak-contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Compute power-weighted subgradient
        # Weight each lag by (ac[k]/M_curr)^beta
        beta_power = 8.0  # Higher = more focused on peaks
        weights = (np.maximum(ac, 0) / M_curr) ** beta_power
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones_like(ac) / len(ac)
        else:
            weights = weights / w_sum
        
        # Compute gradient: grad[i] = 2 * sum_k weights[k] * y[k-i] (for valid k-i)
        grad = np.zeros(n)
        # Use only top lags for efficiency
        top_k = min(100, len(ac))
        top_idx = np.argsort(weights)[::-1][:top_k]
        for k in top_idx:
            w = weights[k]
            if w < 1e-10:
                continue
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            grad[idx] += 2.0 * w * y[j_idx]
        
        # Mirror descent step (exponential family)
        eta = 5.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
        
        # Periodic restart
        if t % 5000 == 0:
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PRD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def aggressive_local_search(a0, time_limit, verbose=True, rng_seed=42):
    """
    Aggressive local search with paired moves and adaptive step sizes.
    Tries to reduce the top peaks by moving mass from peak-contributing positions
    to positions that don't contribute to peaks.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find peak lags
        peak_threshold = M_curr * 0.95
        peak_lags = np.where(ac >= peak_threshold)[0]
        
        improved = False
        
        # For each peak lag, find the best "repulsion" move
        for k in peak_lags[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            
            # Positions that contribute to this peak: i where a[k-i] is large
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[k - idx]
            
            # Find positions with high contribution (both a[i] and a[k-i] large)
            # Move mass from i to j where a[k-j] is small
            if len(idx) < 2:
                continue
            
            # Sort by contribution
            contrib_order = np.argsort(contrib)[::-1]
            top_contrib = idx[contrib_order[:min(10, len(idx))]]
            
            # Find good destinations: positions where a[k-dest] is small
            # and the destination isn't a peak contributor
            all_positions = np.arange(n)
            dest_scores = a[all_positions]  # prefer moving to positions with small current value
            # Penalize positions that would create new peaks
            for dest in all_positions:
                kd = int(k)
                if 0 <= kd - int(dest) < n:
                    dest_scores[dest] -= 0.5 * a[kd - int(dest)]
            
            good_dests = np.argsort(dest_scores)[:min(20, n)]
            
            # Try moves
            best_trial_m = M_curr
            best_move = None
            
            for src in top_contrib[:5]:
                if a[src] < 1e-12:
                    continue
                for dst in good_dests[:10]:
                    if dst == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_trial_m - 1e-16:
                            best_trial_m = m_trial
                            best_move = (int(src), int(dst), frac)
            
            if best_move is not None:
                src, dst, frac = best_move
                amt = a[src] * frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Escape: random perturbation
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-8:
                    continue
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if verbose and (stall % 10 == 0 or improved):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ALS ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def nesterov_mirror_descent_improved(a0, time_limit, solver, verbose=True):
    """
    Nesterov Mirror Descent with subgradient from the LP dual.
    At each step, solve the LP and use the dual weights to construct a better subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    x = a.copy()
    x_prev = a.copy()
    lp_count = 0
    lp_interval = 50  # Solve LP every 50 iterations
    
    for t in range(1, 500000):
        if time.monotmonic - t0 > time_limit if False else time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Every lp_interval steps, solve LP for better subgradient
        if t % lp_interval == 1:
            n_active = min(80, len(ac))
            active_lags = np.argsort(ac)[::-1][:n_active]
            c_val, t_val = minimax_lp_v2(y, M_curr, active_lags, solver)
            lp_count += 1
            
            if c_val is not None:
                # Use LP solution to construct subgradient
                # The subgradient at y for the max function is from the active constraints
                # Weight by how "active" each constraint is
                residual = M_curr - ac
                active_weights = np.zeros(len(ac))
                for k in active_lags:
                    k = int(k)
                    active_weights[k] = max(0, 1.0 - ac[k] / M_curr) ** 0.5
                
                # Combine with power weighting
                power_weights = (np.maximum(ac, 0) / M_curr) ** 6.0
                combined = 0.5 * active_weights + 0.5 * power_weights
                w_sum = combined.sum()
                if w_sum > 1e-30:
                    combined = combined / w_sum
                else:
                    combined = power_weights / power_weights.sum()
            else:
                combined = (np.maximum(ac, 0) / M_curr) ** 6.0
                combined = combined / combined.sum()
        else:
            combined = (np.maximum(ac, 0) / M_curr) ** 6.0
            combined = combined / combined.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        top_k = min(80, len(ac))
        top_idx = np.argsort(combined)[::-1][:top_k]
        for k in top_idx:
            w = combined[k]
            if w < 1e-12:
                continue
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            grad[idx] += 2.0 * w * y[j_idx]
        
        # Mirror descent step
        eta = 3.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
        
        # Periodic restart
        if t % 10000 == 0:
            pert = rng.normal(0, 0.002, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD+LP t={t} ({elapsed:.1f}s): J={J:.6f}, lp_solves={lp_count}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Start from parent sequence
    2. LP-FW with minimax LP (primary method, 35% time)
    3. NMD with LP-augmented subgradient (25% time)  
    4. Peak repulsion descent (15% time)
    5. Aggressive local search (10% time)
    6. Multi-length exploration (10% time)
    7. Final polish (5% time)
    """
    rng = np.random.default_rng(2025)
    t_global = time.monotonic()

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: LP-FW with minimax LP (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 1: LP-FW minimax n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_minimax(a_in, t1, solver, verbose=True)
        try_update(a_out, f"LP-MM_n{n0}")
    
    # === Phase 2: NMD with LP-augmented subgradient (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: NMD+LP n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent_improved(a_in, t2, solver, verbose=True)
        try_update(a_out, f"NMDLP_{len(best_seq)}")
    
    # === Phase 3: Peak repulsion descent (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Peak Repulsion n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:

        a_out, _ = peak_repulsion_descent(a_in, t3, verbose=True)
        try_update(a_out, f"PRD_{len(best_seq)}")
    
    # === Phase 4: Aggressive local search (10% time) ===

            try_update(a_out, f"LP_final_{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.6, 60.0)
            if t6b > 10:
                a_out, _ = aggressive_local_search(a_in, t6b, verbose=False, rng_seed=99999)
                try_update(a_out, f"ALS_final_{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
