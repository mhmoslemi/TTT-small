def _fft_convolve(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 assuming sum(a)=1."""
    n = len(a)
    ac = _fft_convolve(a)
    return 2.0 * n * ac.max()


def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _subgradient_topk(a, ac, top_k=5, temperature=100.0):
    """Compute weighted subgradient using top-k peak lags."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Get top-k lags
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    
    # Softmax weights
    vals = ac[top_idx]
    shifted = vals / M - 1.0  # in [-1, 0]
    weights = np.exp(temperature * shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k_max) / k_max
    else:
        weights = weights / wsum
    
    g = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def _mirror_step(a, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(a, 1e-300)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def _run_mirror_descent(a0, time_limit, c0=3.0, top_k=5, temperature=100.0,
                        verbose=True, rng_seed=42, max_iters=10000000):
    """Accelerated mirror descent with Nesterov momentum."""
    n = len(a0)
    a = _normalize(a0)
    ac = _fft_convolve(a)
    best_a = a.copy()
    best_obj = 2.0 * n * ac.max()
    t_start = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    c_cur = c0
    
    for t in range(1, max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-300)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Subgradient
        ac_y = _fft_convolve(y)
        g = _subgradient_topk(y, ac_y, top_k=top_k, temperature=temperature)
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            break
        
        # Step size
        eta = c_cur / (np.sqrt(t) + 1.0)
        a_new = _mirror_step(y, g, eta)
        
        x_prev = x
        x = a_new
        
        # Evaluate
        ac_x = _fft_convolve(x)
        obj_x = 2.0 * n * ac_x.max()
        if obj_x < best_obj - 1e-12:
            best_obj = obj_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            noise = 0.005 + 0.005 * np.sqrt(min(stall, 10))
            x = best_a + rng.normal(0, noise, n) * best_a
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c_cur *= 1.15
            if verbose:
                el = time.monotonic() - t_start
                print(f"      MD restart t={t} ({el:.1f}s): obj={best_obj:.6f}, c={c_cur:.2f}")
        
        if verbose and t % 20000 == 0:
            el = time.monotonic() - t_start
            print(f"      MD t={t} ({el:.1f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj


def _run_coord_descent(a0, time_limit, verbose=True, rng_seed=777, top_peaks=5):
    """Peak-targeted coordinate descent with exact evaluation."""
    n = len(a0)
    a = _normalize(a0)
    ac = _fft_convolve(a)
    best_a = a.copy()
    best_m = ac.max()
    best_obj = 2.0 * n * best_m
    t_start = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    fracs = [0.001, 0.003, 0.007, 0.015, 0.03]
    
    while time.monotonic() - t_start < time_limit and stall < max_stall:
        ac = _fft_convolve(a)
        M = ac.max()
        
        # Get top peak lags
        n_peaks = min(top_peaks, len(ac))
        peak_lags = np.argsort(ac)[::-1][:n_peaks]
        
        # Compute sensitivity for each position (how much it contributes to peaks)
        peak_sens = np.zeros(n)
        for k in peak_lags:
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n, k_i + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_i - idx
            peak_sens[idx] += a[j_idx]
        
        # Find hot positions (high sensitivity + high value)
        hot_score = peak_sens * a
        n_hot = min(30, n)
        hot_pos = np.argsort(hot_score)[::-1][:n_hot]
        
        # Find cold positions (low sensitivity)
        in_peak = np.zeros(n, dtype=bool)
        for k in peak_lags:
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n, k_i + 1)
            in_peak[i_start:i_end] = True
        
        cold_candidates = np.where(~in_peak)[0]
        if len(cold_candidates) > 200:
            cold_candidates = rng.choice(cold_candidates, 200, replace=False)
        
        # Sort cold by sensitivity (prefer lowest)
        if len(cold_candidates) > 0:
            cold_pos = cold_candidates[np.argsort(peak_sens[cold_candidates])[:50]]
        else:
            cold_pos = np.arange(min(50, n))
        
        improved = False
        rng.shuffle(hot_pos)
        
        for src in hot_pos:
            if time.monotonic() - t_start > time_limit:
                break
            if a[src] < 1e-12:
                continue
            
            for dst in cold_pos:
                if dst == src:
                    continue
                if a[dst] > a[src] * 0.5:
                    continue
                
                for frac in fracs:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_t = _fft_convolve(a_trial).max()
                    if M_t < best_m - 1e-14:
                        a = a_trial
                        best_m = M_t
                        best_obj = 2.0 * n * best_m
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                continue  # restart inner loop with new state
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = max(5, n // 100)
            indices = rng.choice(n, n_pert, replace=False)
            for i in indices:
                if a[i] > 1e-10:
                    j = int(rng.integers(n))
                    if j != i:
                        frac = rng.uniform(0.01, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = _fft_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_obj = 2.0 * n * best_m
                                best_a = a.copy()
                                stall = 0
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0.0)
                                if a.sum() > 1e-15:
                                    a = a / a.sum()
        
        if verbose and stall % 50 == 0:
            el = time.monotonic() - t_start
            print(f"      Coord ({el:.1f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj


def _run_lp_frank_wolfe(a0, time_limit, verbose=True, max_iters=100):
    """LP-based Frank-Wolfe direction finding."""
    if not _HAS_CVXPY:
        return _run_mirror_descent(a0, time_limit, c0=3.0, verbose=verbose)
    
    n = len(a0)
    if n > 3000:
        return _run_mirror_descent(a0, time_limit, c0=3.0, verbose=verbose)
    
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fft_convolve(a).max()
    best_obj = 2.0 * n * best_m
    t_start = time.monotonic()
    
    y = a.copy()
    
    # Find solver
    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x_test = cp.Variable(3, nonneg=True)
            prob_test = cp.Problem(cp.Minimize(cp.sum(x_test)), [cp.sum(x_test) == 1])
            prob_test.solve(solver=s)
            if prob_test.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                if verbose:
                    print(f"      LP solver: {s}")
                break
        except Exception:
            continue
    if solver is None:
        return best_a, best_obj
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t_start > time_limit:
            break
        
        ac = _fft_convolve(y)
        M_cur = ac.max()
        if M_cur < 1e-14:
            break
        
        # Build LP: max sum(b) s.t. b >= 0, conv(y,b)[k] <= M_cur for top lags
        b_var = cp.Variable(n, nonneg=True)
        top_lags = np.argsort(ac)[::-1][:min(80, len(ac))]
        
        constraints = [cp.sum(b_var) <= 1.0]
        for k in top_lags:
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n, k_i + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_i - idx
            coeff = np.zeros(n)
            for ii, j in zip(idx, j_idx):
                coeff[j] += y[ii]
            constraints.append(coeff @ b_var <= M_cur)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        
        my = _fft_convolve(y).max()
        if my < best_m:
            best_m = my
            best_obj = 2.0 * n * best_m
            best_a = y.copy()
        
        if verbose and t % 10 == 0:
            el = time.monotonic() - t_start
            print(f"      LP it={t} ({el:.1f}s): obj={best_obj:.6f}")
    
    return best_a, best_obj


def _eval_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _structured_init(n, mode='sin'):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if mode == 'sin':
        a = np.sin(np.pi * x) ** 2 + 0.05 * np.cos(2 * np.pi * x) ** 2
    elif mode == 'bell':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif mode == 'rand':
        a = np.random.default_rng(0).random(n)
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(budget_s=1800.0):
    """
    Main search: minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses mirror descent, coordinate descent, LP Frank-Wolfe,
    and multi-length exploration.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0  # 20s safety margin
    rng = np.random.default_rng(2024)
    
    best_seq = None
    best_score = float('inf')
    
    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _eval_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={el:.0f}s)")
    
    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _eval_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2400
        best_seq = _structured_init(n_base, 'sin')
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Quick mirror descent on parent (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 1: Mirror descent n={n0} (budget={remaining:.0f}s) ===")
        t1 = remaining * 0.25
        a_out, _ = _run_mirror_descent(best_seq, t1, c0=4.0, top_k=3,
                                       temperature=200.0, verbose=True, rng_seed=1)
        _try_update(a_out, f"MD1_n{n0}")
    
    # === Phase 2: Coordinate descent (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 2: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
        t2 = remaining * 0.15
        a_out, _ = _run_coord_descent(best_seq, t2, verbose=True, rng_seed=2, top_peaks=5)
        _try_update(a_out, f"Coord_n{n_best}")
    
    # === Phase 3: Mirror descent again with different params (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: MD2 n={n_best} (budget={remaining:.0f}s) ===")
        t3 = remaining * 0.20
        a_out, _ = _run_mirror_descent(best_seq, t3, c0=2.5, top_k=7,
                                       temperature=50.0, verbose=True, rng_seed=3)
        _try_update(a_out, f"MD2_n{n_best}")
    
    # === Phase 4: LP Frank-Wolfe (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: LP FW n={n_best} (budget={remaining:.0f}s) ===")
        t4 = remaining * 0.15
        a_out, _ = _run_lp_frank_wolfe(best_seq, t4, verbose=True, max_iters=80)
        _try_update(a_out, f"LP_n{n_best}")
    
    # === Phase 5: Multi-length exploration (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 5: Multi-length (budget={remaining:.0f}s) ===")
        n_best = len(best_seq)
        # Try nearby lengths
        lengths_to_try = []
        for dn in [-300, -200, -100, 100, 200, 300, 400, -400, 500]:
            nl = n_best + dn
            if 800 <= nl <= 4000 and nl != n_best:
                lengths_to_try.append(nl)
        
        # Sort by how close to n_best (prefer close)
        lengths_to_try.sort(key=lambda x: abs(x - n_best))
        
        for nl in lengths_to_try[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            print(f"  Trying n={nl}")
            a_interp = _interpolate(best_seq, nl)
            t_sub = min(remaining * 0.25, 120.0)
            if t_sub < 20:
                break
            
            # Quick MD on new length
            a_out, _ = _run_mirror_descent(a_interp, t_sub, c0=3.0, top_k=5,
                                           temperature=100.0, verbose=False,
                                           rng_seed=10 + nl % 100)
            _try_update(a_out, f"ML_n{nl}")
            
            # Quick coord on new length
            remaining = deadline - time.monotonic()
            if remaining > 30:
                t_coord = min(remaining * 0.15, 60.0)
                if t_coord > 15:
                    a_out2, _ = _run_coord_descent(best_seq, t_coord, verbose=False,
                                                   rng_seed=20 + nl % 100, top_peaks=5)
                    _try_update(a_out2, f"MLcoord_n{nl}")
    
    # === Phase 6: Fine coordinate descent (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Fine coord n={n_best} (budget={remaining:.0f}s) ===")
        t6 = remaining * 0.5
        a_out, _ = _run_coord_descent(best_seq, t6, verbose=True, rng_seed=99, top_peaks=8)
        _try_update(a_out, f"FineCoord_n{n_best}")
    
    # === Phase 7: Final mirror descent polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final MD polish n={n_best} ===")
        t7 = remaining * 0.6
        a_out, _ = _run_mirror_descent(best_seq, t7, c0=1.5, top_k=3,
                                       temperature=500.0, verbose=False, rng_seed=111)
        _try_update(a_out, f"FinalMD_n{n_best}")
    
    # === Final validation ===
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
