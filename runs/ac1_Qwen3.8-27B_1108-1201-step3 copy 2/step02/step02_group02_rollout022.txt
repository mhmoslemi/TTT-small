"""
Smoothed gradient descent with Nesterov momentum for minimizing peak
autocorrelation of non-negative step functions. Combines fast FFT-based
gradient computation with periodic LP-based direction finding to escape
local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smoothed_score_and_grad(a, temperature):
    """Compute smoothed score (log-sum-exp) and its gradient."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    
    # Log-sum-exp for numerical stability
    shifted = c / temperature - c_max / temperature
    shifted = np.clip(shifted, -50, 50)
    log_sum_exp = np.log(np.sum(np.exp(shifted))) + c_max / temperature
    
    # Weights for gradient
    weights = np.exp(shifted)
    total_w = np.sum(weights)
    if total_w < 1e-30:
        weights = np.zeros_like(c)
        weights[np.argmax(c)] = 1.0
    else:
        weights /= total_w
    
    # Gradient via cross-correlation
    # grad_j = sum_k weights[k] * a[k-j] for valid k-j
    size = 1
    while size < 2 * n:
        size <<= 1
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    grad = np.fft.irfft(np.conj(A) * W, size)[:n]
    
    smoothed_val = log_sum_exp * temperature
    return smoothed_val, grad


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix A where A[k,i] gives convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve the LP to find the best descent direction."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n


def nesterov_smoothed_descent(a0, time_limit, rng, initial_temp=0.05, min_temp=0.005):

    n = len(a0)
    S = np.sqrt(2.0 * n)
    
    # Normalize

    sum_a = np.sum(a)

        a = np.ones(n) * (S / n)
    else:
        a = a * (S / sum_a)

    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score

    best_a = a.copy()
    best_score = current_score

    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))
    
    while time.monotonic() - t0 < time_limit:

        peak_idx = int(np.argmax(c))
        # Get contributions to the peak
        contribs = np.zeros(n)
        for i in range(min(n, peak_idx + 1)):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_k = max(2, n // 16)
        bot_k = max(2, n // 16)

        bot_idx = np.argsort(contribs)[:bot_k]
        
        # Make a move
        delta = 0.01 * np.mean(a[top_idx])
        a[top_idx] -= delta * 0.5
        a[bot_idx] += delta * 0.5
        a = np.clip(a, 0, None)
        
        new_score = compute_score(a)
        if new_score < current_score:
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            c = fft_convolve(a, a)

            a[top_idx] += delta * 0.5
            a[bot_idx] -= delta * 0.5
            a = np.clip(a, 0, None)
            # Random perturbation

                current_score = new_score
                if current_score < best_score:
                    best_score = current_score

"""
Hybrid optimization for minimizing peak self-convolution of non-negative step functions.
Combines smoothed gradient descent with Nesterov momentum, adaptive temperature scheduling,
structured peak-targeting moves, periodic LP assistance, and coordinate descent polish.
Key insight: the self-convolution peak is not necessarily at the center, so we target
all high-convolution positions simultaneously via log-sum-exp smoothing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smoothed_grad(a, temperature):
    """Compute log-sum-exp smoothed value and gradient of peak convolution."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / temperature
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_vals = np.exp(shifted)
    total_w = np.sum(exp_vals)
    if total_w < 1e-30:
        weights = np.zeros_like(c)
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = exp_vals / total_w
    # Gradient: grad_j = sum_k weights[k] * a[k-j] for valid k-j in [0,n-1]
    # This is cross-correlation of weights (len 2n-1) with a (len n)
    size = 1
    while size < 2 * n:
        size <<= 1
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    grad = np.fft.irfft(np.conj(A) * W, size)[:n]
    # Scale gradient by 1/temperature for the actual gradient of log-sum-exp * T
    smoothed_val = (np.log(total_w) + c_max / temperature) * temperature
    return smoothed_val, grad / temperature


def build_constraint_matrix(f):
    """Build the (2n-1) x n matrix such that A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b>=0, conv(f,b) <= M elementwise."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.ones(n) / n


def nesterov_descent(a0, time_limit, rng, init_temp=0.08, min_temp=0.005,
                     init_lr=0.15, momentum=0.9):
    """Nesterov accelerated smoothed gradient descent with adaptive scheduling."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.clip(a0.astype(np.float64).copy(), 0, None)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) * (S / n)
    else:
        a = a * (S / s)

    v = np.zeros(n)
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    temp = init_temp
    lr = init_lr
    t0 = time.monotonic()
    stall = 0
    n_iter = 0
    check_interval = max(20, n // 10)

    while time.monotonic() - t0 < time_limit:
        _, grad = smoothed_grad(a, temp)
        # Nesterov lookahead
        y = a + momentum * v
        _, y_grad = smoothed_grad(y, temp)
        v = momentum * v - lr * y_grad
        a = np.clip(a + v, 0.0, None)

        n_iter += 1

        if n_iter % check_interval == 0:
            s = np.sum(a)
            if s > 0.01:
                a = a * (S / s)
            current_score = compute_score(a)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                stall = 0
                temp = max(min_temp, temp * 0.97)
            else:
                stall += 1
                if stall > 15:
                    # Reset and perturb
                    a = best_a.copy() * np.exp(rng.normal(0, 0.02, n))
                    a = np.clip(a, 0, None)
                    s = np.sum(a)
                    if s > 0.01:
                        a = a * (S / s)
                    v = np.zeros(n)
                    temp = max(min_temp, temp * 1.5)
                    lr = init_lr * 0.5
                    stall = 0

        if n_iter % 200 == 0:
            lr = max(0.01, lr * 0.97)
            momentum = min(0.98, momentum + 0.001)

    return best_a, best_score


def structured_local_search(a0, time_limit, rng):
    """Targeted local search that directly attacks peak convolution positions."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.clip(a0.astype(np.float64).copy(), 0, None)
    s = np.sum(a)
    if s > 0.01:
        a = a * (S / s)

    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    c = fft_convolve(a, a)
    t0 = time.monotonic()
    temperature = 0.005
    cooling = 0.9998
    move_size = 0.02

    while time.monotonic() - t0 < time_limit:
        r = rng.random()
        candidate = a.copy()

        if r < 0.35:
            # Targeted peak reduction: find top contributors to peak
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            k_start = max(0, peak_idx - n + 1)
            k_end = min(n - 1, peak_idx)
            for j in range(k_start, k_end + 1):
                partner = peak_idx - j
                if 0 <= partner < n:
                    contribs[j] = a[j] * a[partner]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            bot_idx = np.argsort(contribs)[:top_k]
            delta = move_size * np.median(a[top_idx])
            candidate[top_idx] -= delta * 0.4
            candidate[bot_idx] += delta * 0.4 * len(top_idx) / len(bot_idx)
        elif r < 0.55:
            # Multi-peak attack
            c_sorted = np.sort(c)[::-1]
            threshold = c_sorted[min(5, len(c_sorted) - 1)]
            peak_positions = np.where(c >= threshold * 1.001)[0]
            pressure = np.zeros(n)
            for k in peak_positions:
                k_start = max(0, k - n + 1)
                k_end = min(n - 1, k)
                for j in range(k_start, k_end + 1):
                    partner = k - j
                    if 0 <= partner < n:
                        pressure[j] += a[j] * a[partner]
            top_k = max(3, n // 10)
            top_idx = np.argsort(pressure)[-top_k:]
            bot_idx = np.argsort(pressure)[:top_k]
            delta = move_size * 0.5 * np.median(a[top_idx])
            candidate[top_idx] -= delta
            candidate[bot_idx] += delta * len(top_idx) / len(bot_idx)
        elif r < 0.70:
            # Block scale perturbation
            n_blocks = rng.integers(2, 6)
            for _ in range(n_blocks):
                width = max(1, int(rng.integers(1, max(2, n // 8))))
                start = int(rng.integers(0, max(1, n - width)))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, width))
        elif r < 0.85:
            # Pair transfer: move mass from high-contributing to low-contributing
            peak_idx = int(np.argmax(c))
            k_start = max(0, peak_idx - n + 1)
            k_end = min(n - 1, peak_idx)
            contribs = np.zeros(n)
            for j in range(k_start, k_end + 1):
                partner = peak_idx - j
                if 0 <= partner < n:
                    contribs[j] = a[j] * a[partner]
            donor = int(np.argmax(contribs))
            # Find a recipient that doesn't contribute to any peak
            c_threshold = np.max(c) * 0.9
            safe = np.ones(n, dtype=bool)
            for k in np.where(c >= c_threshold)[0]:
                ks = max(0, int(k) - n + 1)
                ke = min(n - 1, int(k))
                for j in range(ks, ke + 1):
                    p = int(k) - j
                    if 0 <= p < n:
                        safe[j] = False
                        safe[p] = False
            if np.any(safe):
                recipients = np.where(safe)[0]
                recipient = int(rng.choice(recipients))
                delta = rng.uniform(0.01, 0.05) * a[donor]
                candidate[donor] -= delta
                candidate[recipient] += delta
            else:
                candidate = a * np.exp(rng.normal(0, 0.01, n))
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0, 0.008, n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = compute_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            c = fft_convolve(a, a)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            move_size = min(0.05, move_size * 1.01)
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-10)):
            a = candidate
            current_score = new_score
            c = fft_convolve(a, a)

        temperature = max(0.0005, temperature * cooling)
        if temperature <= 0.001:
            move_size = max(0.005, move_size * 0.99)

    return best_a, best_score


def lp_refine(a0, time_limit, rng):
    """LP-based direction finding with line search, used periodically."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    f = np.clip(a0.astype(np.float64).copy(), 0, None)
    s = np.sum(f)
    if s > 0.01:
        f = f * (S / s)

    current_score = compute_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.1
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        M = np.max(c)
        b_0 = solve_lp_direction(f, M)
        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = (S / sum_b) * b_0

        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1, 0.05, 0.02, 0.01]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0, None)
            new_score = compute_score(f_new)
            if new_score < current_score:
                f = f_new
                current_score = new_score
                t = min(0.3, trial_t * 1.1)
                improved = True
                stall = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break

        if not improved:
            stall += 1
            t *= 0.7
            if stall > 30:
                f = best_f.copy() * np.exp(rng.normal(0, 0.02, n))
                f = np.clip(f, 0, None)
                s = np.sum(f)
                if s > 0.01:
                    f = f * (S / s)
                current_score = compute_score(f)
                stall = 0
                t = 0.08

    return best_f, best_score


def coordinate_descent_polish(a0, time_limit, rng):
    """Coordinate descent that optimizes pairs of elements jointly."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.clip(a0.astype(np.float64).copy(), 0, None)
    s = np.sum(a)
    if s > 0.01:
        a = a * (S / s)

    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    step_size = 0.01

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        # Find elements contributing to the peak
        k_start = max(0, peak_idx - n + 1)
        k_end = min(n - 1, peak_idx)
        contribs = np.zeros(n)
        for j in range(k_start, k_end + 1):
            partner = peak_idx - j
            if 0 <= partner < n:
                contribs[j] = a[j] * a[partner]

        # Pick top contributor and try to reduce it
        top_idx = int(np.argmax(contribs))
        partner = peak_idx - top_idx
        if not (0 <= partner < n):
            partner = top_idx

        # Try reducing top_idx and increasing a random non-peak element
        threshold = np.max(c) * 0.95
        peak_positions = set(np.where(c >= threshold)[0].tolist())
        safe_indices = [i for i in range(n)
                        if i != top_idx and i != partner
                        and (top_idx + i) not in peak_positions]

        if safe_indices:
            inc_idx = int(rng.choice(safe_indices))
            delta = step_size * a[top_idx] * rng.uniform(0.5, 1.5)
            candidate = a.copy()
            candidate[top_idx] = max(0, candidate[top_idx] - delta)
            candidate[inc_idx] = candidate[inc_idx] + delta
            new_score = compute_score(candidate)
            if new_score < current_score:
                a = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                step_size = min(0.03, step_size * 1.02)
            else:
                # Try the other direction
                candidate = a.copy()
                candidate[top_idx] = candidate[top_idx] + delta * 0.5
                candidate[inc_idx] = max(0, candidate[inc_idx] - delta * 0.5)
                new_score = compute_score(candidate)
                if new_score < current_score:
                    a = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()
                else:
                    step_size = max(0.003, step_size * 0.95)
        else:
            # No safe index, do random pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                candidate = a.copy()
                delta = step_size * abs(a[i] - a[j])
                if a[i] > a[j]:
                    candidate[i] -= delta
                    candidate[j] += delta
                else:
                    candidate[i] += delta
                    candidate[j] -= delta
                candidate = np.clip(candidate, 0, None)
                new_score = compute_score(candidate)
                if new_score < current_score:
                    a = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Generate a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Generate a random initial sequence."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_flat_with_noise(n, rng, noise_level=0.1):
    """Flat sequence with controlled noise."""
    a = np.ones(n) + noise_level * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

            pass

        print(f"Generated starting sequence: n=512, score={s:.6f}")
        best_score = s
        best_seq = start_seq.copy()

    # Phase 1: Nesterov smoothed descent from provided (primary)
