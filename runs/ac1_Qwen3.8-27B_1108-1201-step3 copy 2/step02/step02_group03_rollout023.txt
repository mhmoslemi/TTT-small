So the algorithm is:
1. Fix S (e.g., S = sqrt(2n) or S = 1)
2. Compute c = conv(a, a)
3. Compute w = softmax(tau * c) (normalized)
4. Compute grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for each i (using FFT)
5. Update: a_new = a - lr * grad, project onto {a >= 0, sum(a) = S}
6. Iterate

The projection onto {a >= 0, sum(a) = S} can be done via:
- Clip to non-negative: a = max(a, 0)
- Rescale to achieve sum S: a = a * S / sum(a)

This is not the exact Euclidean projection, but it's a simple and effective heuristic.

For the FFT computation of grad:
grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]

Let's define b[j] = a[j] for j = 0,...,n-1.
Then grad[i] = sum_j w[i+j] * b[j]

This is the cross-correlation of w and b, specifically:
grad[i] = (w ⋆ b)[i] where ⋆ denotes cross-correlation

In terms of convolution: cross-correlation can be computed as:
(w ⋆ b)[i] = conv(w, b_flipped)[i + n - 1]

where b_flipped = b[::-1].

So: grad = conv(w, b[::-1])[n-1 : n-1+n]

This can be done with FFT in O(n log n).

```python
import numpy as np
import time

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def soft_gradient(a, tau):
    """Compute gradient of soft-max objective (with sum fixed)."""
    n = len(a)
    S = a.sum()
    c = np.convolve(a, a)
    
    # Soft-max weights
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is cross-correlation: conv(w, a[::-1])[n-1:n-1+n]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = conv_result[n-1:n-1+n]
    
    # Scale: dF/da[i] = 4n/S² * grad[i] (with S fixed)
    # But we'll absorb the constant into the learning rate
    return grad, c, w

def optimize_sequence(a0, deadline, S_target=None):
    """Optimize a sequence to minimize the objective."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000)
    if a.sum() < 1e-6:
        a = np.ones(n) / np.sqrt(n)
    
    if S_target is None:
        S_target = np.sqrt(2.0 * n)
    
    # Normalize
    a = a * S_target / a.sum()
    best = a.copy()
    best_score = compute_obj(a)
    
    tau = 5.0
    lr = 0.1
    momentum = np.zeros(n)
    momentum_decay = 0.9
    
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Anneal
        tau = min(50.0, 5.0 + 0.01 * it)
        lr = max(0.001, 0.1 * (0.999 ** it))
        
        grad, c, w = soft_gradient(a, tau)
        
        # Project out the mean to maintain sum
        grad -= grad.mean()
        
        # Momentum
        momentum = momentum_decay * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        a_new = a_new * S_target / a_new.sum()
        
        a = a_new
        
        if it % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    # Reset and try different tau
                    momentum[:] = 0
                    no_improve = 0
                    # Try a bigger tau jump
                    tau += 10
        
        if time.monotonic() > deadline - 5:
            break
```

```python
def fft_gradient(a, tau):
    """
    Compute the gradient of the soft-max objective using FFT.
    Returns gradient vector of length n.
    """
    n = len(a)
    S = a.sum()
    
    # Compute convolution
    c = np.convolve(a, a)
    
    # Soft-max weights
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # Using FFT: this is the valid cross-correlation of w (len 2n-1) with a (len n)
    # conv(w, a[::-1])[n-1 : n-1+n]
    
    # Use FFT for speed
    # We need conv(w, a_rev) where w has length 2n-1 and a_rev has length n
    # Result length: 2n-1 + n - 1 = 3n - 2
    # We need indices n-1 to 2n-2 (n values)
    
    size = 2 * (3 * n)  # Next power of 2 for FFT
    size = 1
    while size < 3 * n - 1:
        size *= 2
    
    W = np.fft.rfft(w, size)
    A_rev = np.fft.rfft(a[::-1], size)
    conv_full = np.fft.irfft(W * A_rev, size)[:3*n-2]
    
    grad = conv_full[n-1:n-1+n]
    
    return grad, c, w
```

Uses FFT-based soft-max gradient descent with momentum, adaptive tau annealing,

Key insight: The objective is scale-invariant; we fix sum(a) and minimize peak convolution.
The theoretical lower bound is ~1 (flat convolution); uniform gives 2.

def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def get_grad_fft(a, tau):
    """
    Compute gradient of soft-max objective (with sum fixed) using FFT.
    grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j] where w = softmax(tau * conv(a,a))
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = sum_j w[i+j]*a[j] = conv(w, a_rev)[n-1 : n-1+n]
    a_rev = a[::-1]
    total_len = 3 * n - 2
    fft_size = 1
    while fft_size < total_len:
        fft_size *= 2
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_full = np.fft.irfft(W * AR, fft_size)[:total_len]
    grad = conv_full[n - 1:n - 1 + n]
    return grad, c, w
