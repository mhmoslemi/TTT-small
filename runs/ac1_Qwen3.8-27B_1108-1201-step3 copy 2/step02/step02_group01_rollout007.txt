    x: unconstrained parameter (length n)
    a = softmax(x) (so sum(a) = 1, a > 0)
    Returns: objective value and gradient with respect to x
    """

    a = np.exp(x - x.max())
    a = a / a.sum()

    # Smoothed max
    max_conv = np.max(conv)

    obj = tau * log_sum_exp
    
    # Gradient of smoothed max with respect to a
    softmax_weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n else 0
    # grad_a[i] = sum_k softmax_weights[k] * 2*a[k-i] * [0<=k-i<n]

    grad_a = np.zeros(n)
    for k in range(2*n-1):

            # i ranges from max(0, k-n+1) to min(k, n-1)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                grad_a[i_min:i_max+1] += 2 * softmax_weights[k] * a[k - np.arange(i_min, i_max+1)]
    
    # Chain rule: da/dx for softmax

    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)

def normalize_to_sum(a, S=1.0):
    """Normalize sequence to have a given sum."""
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def objective_and_grad(x, tau):
    """
    Smoothed peak autocorrelation objective using softmax parameterization.
    a = softmax(x), objective = tau * logsumexp(conv(a,a)/tau)
    Returns (obj, grad_x).
    """
    n = len(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()

    max_conv = np.max(conv)
    scaled = conv / tau

    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    lse = np.max(scaled) + np.log(sum_exp)

    history = []

    for step in range(50000):
        if time.time() - t0 > time_limit:
            break

        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]

        # Subgradient of conv(a,a)[peak_idx] wrt a
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = peak_idx - i
            if 0 <= j1 < n:
                val += a[j1]
            j2 = i - peak_idx
            if 0 <= j2 < n:
                val += a[j2]
            grad[i] = 2.0 * val if (0 <= peak_idx - i < n) or (0 <= i - peak_idx < n) else 0.0

        # Actually, gradient of conv(a,a)[k] wrt a[i] = 2*a[k-i] if 0<=k-i<n, else 0
        grad = np.zeros(n)
        i_min = max(0, peak_idx - n + 1)
        i_max = min(peak_idx, n - 1)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] = 2.0 * a[peak_idx - indices]

        # Nesterov momentum
        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum(a_new, S)

        new_peak = np.max(fft_convolve(a_new, a_new))

        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # Still move but don't update best
            stall += 1
            if stall > 100:
                lr *= 0.8
                stall = 0

        # Adaptive step size
        history.append(current_peak)
        if len(history) > 50:
            recent = history[-50:]
            if np.max(recent) - np.min(recent) < 1e-6 * np.max(recent):
                lr *= 0.95

        if verbose and step % 5000 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Subgrad step {step}: score={score:.6f}, lr={lr:.6f}")

    return best_a


def targeted_local_search(a, time_limit=120.0, verbose=False, seed=42):
    """Targeted local search focusing on peak reduction."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)
    t0 = time.time()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(seed)

    for it in range(50000):
        if time.time() - t0 > time_limit:
            break

        conv = fft_convolve(a, a)
        # Find top peaks
        top_k = np.argsort(conv)[-5:]
        current_peak = np.max(conv)

        accepted = False

        # Strategy 1: Reduce top contributors to the main peak
        peak_idx = top_k[-1]
        contribs = np.zeros(n)
        i_min = max(0, peak_idx - n + 1)
        i_max = min(peak_idx, n - 1)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            contribs[indices] = a[indices] * a[peak_idx - indices]

        top_n = max(3, n // 10)
        top_idx = np.argsort(contribs)[-top_n:]

        for _ in range(8):
            candidate = a.copy()
            n_reduce = max(1, top_n // 2)
            reduce_set = rng.choice(top_idx, size=n_reduce, replace=False)
            for idx in reduce_set:
                candidate[idx] *= np.exp(rng.normal(-0.1, 0.05))

            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = max(3, n // 15)
                inc_set = rng.choice(other, size=min(n_inc, len(other)), replace=False)
                candidate[inc_set] *= np.exp(rng.normal(0.15, 0.05, size=len(inc_set)))

            candidate = np.clip(candidate, 0.0, None)
            cs = np.sum(candidate)
            if cs < 0.01:
                continue
            candidate = normalize_to_sum(candidate, S)

            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_peak * 0.999:
                best_peak = new_peak
                best_a = candidate.copy()
                a = candidate
                accepted = True
            elif new_peak < best_peak and rng.random() < 0.5:
                a = candidate
                accepted = True

        # Strategy 2: Block perturbation
        if rng.random() < 0.4:
            width = max(2, n // 20)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_to_sum(candidate, S)
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = candidate.copy()
                a = candidate
                accepted = True

        # Strategy 3: Swap-based moves
        if rng.random() < 0.3:
            candidate = a.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.3, 0.3) * min(candidate[i], candidate[j])
                candidate[i] -= delta
                candidate[j] += delta
                candidate = np.clip(candidate, 0.0, None)
                candidate = normalize_to_sum(candidate, S)
                new_peak = np.max(fft_convolve(candidate, candidate))
                if new_peak < best_peak:
                    best_peak = new_peak
                    best_a = candidate.copy()
                    a = candidate
                    accepted = True

        if verbose and it % 1000 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Local search it {it}: score={score:.6f}")

    return best_a


def make_initialization(n, init_type, rng):
    """Create various initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (0.5 * n))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.2 + 0.4 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 5)))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.4 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.1 + 0.15 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'window':
        a = 0.5 + 0.5 * np.cos(np.pi * t / (n - 1))
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. L-BFGS-B with log-sum-exp smoothing
    2. LP-based iterative refinement
    3. Subgradient method with momentum
    4. Targeted local search
    Explores multiple sequence lengths and initializations.
    """
    t_global = time.time()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)

    # Phase 1: L-BFGS-B optimization on provided sequence
    print("\n=== Phase 1: L-BFGS-B optimization ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(180.0, (deadline - time.time()) * 0.25)
        print(f"  L-BFGS-B on provided (n={n}), limit={t_limit:.0f}s")
        tau_list = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        result = lbfgsb_optimize(provided_seq, tau_list, t_limit, verbose=True)
        if update_best(result):
            print(f"  *** New best after L-BFGS-B: {best_score:.6f} ***")

    # Phase 2: LP refinement on current best
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(150.0, (deadline - time.time()) * 0.2)
        print(f"  LP refine (n={n}), limit={t_limit:.0f}s")
        result = lp_refine(best_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  *** New best after LP: {best_score:.6f} ***")

    # Phase 3: Subgradient method
    print("\n=== Phase 3: Subgradient method ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(120.0, (deadline - time.time()) * 0.15)
        print(f"  Subgradient (n={n}), limit={t_limit:.0f}s")
        result = subgradient_optimize(best_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  *** New best after subgradient: {best_score:.6f} ***")

    # Phase 4: Targeted local search
    print("\n=== Phase 4: Targeted local search ===")
    if best_seq is not None:
        t_limit = min(150.0, (deadline - time.time()) * 0.2)
        print(f"  Local search, limit={t_limit:.0f}s")
        result = targeted_local_search(best_seq, t_limit, verbose=True, seed=seed)
        if update_best(result):
            print(f"  *** New best after local search: {best_score:.6f} ***")

    # Phase 5: Try different sequence lengths
    print("\n=== Phase 5: Other sequence lengths ===")
    if provided_seq is not None:
        n_original = len(provided_seq)
        for n_try in [384, 512, 768, 896]:
            if time.time() > deadline - 180:
                break
            if n_try == n_original:
                continue
            a0 = interpolate_sequence(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            t_sub = min(60.0, (deadline - time.time()) * 0.15)
            print(f"  Trying n={n_try}, limit={t_sub:.0f}s")

            # L-BFGS-B
            tau_list = [0.3, 0.15, 0.07, 0.03]
            result = lbfgsb_optimize(a0, tau_list, t_sub * 0.5, verbose=False)
            if update_best(result):
                print(f"  *** n={n_try}: new best after L-BFGS-B = {best_score:.6f} ***")

            # LP
            if time.time() < deadline - 60:
                result2 = lp_refine(result, t_sub * 0.3, verbose=False)
                if update_best(result2):
                    print(f"  *** n={n_try}: new best after LP = {best_score:.6f} ***")

            # Local search
            if time.time() < deadline - 30:
                result3 = targeted_local_search(result, t_sub * 0.2, verbose=False, seed=seed + n_try)
                if update_best(result3):
                    print(f"  *** n={n_try}: new best after local search = {best_score:.6f} ***")

    # Phase 6: Second round of L-BFGS-B with finer tau on best
    print("\n=== Phase 6: Fine L-BFGS-B polish ===")
    if best_seq is not None:
        remaining = deadline - time.time() - 20
        if remaining > 30:
            t_limit = min(remaining, 120.0)
            n = len(best_seq)
            print(f"  Fine L-BFGS-B (n={n}), limit={t_limit:.0f}s")
            tau_list = [0.05, 0.02, 0.01, 0.005, 0.002]
            result = lbfgsb_optimize(best_seq, tau_list, t_limit, verbose=True)
            if update_best(result):
                print(f"  *** New best after fine L-BFGS-B: {best_score:.6f} ***")

    # Phase 7: Final LP polish
    print("\n=== Phase 7: Final LP polish ===")
    if best_seq is not None:
        remaining = deadline - time.time() - 15
        if remaining > 20:
            t_limit = min(remaining, 90.0)
            n = len(best_seq)
            print(f"  Final LP (n={n}), limit={t_limit:.0f}s")
            result = lp_refine(best_seq, t_limit, verbose=True)
            if update_best(result):
                print(f"  *** New best after final LP: {best_score:.6f} ***")

    # Phase 8: Final local search polish
    print("\n=== Phase 8: Final local search ===")
    if best_seq is not None:
        remaining = deadline - time.time() - 10
        if remaining > 10:
            result = targeted_local_search(best_seq, remaining, verbose=False, seed=seed + 999)
            if update_best(result):
                print(f"  *** New best after final local search: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list

    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
