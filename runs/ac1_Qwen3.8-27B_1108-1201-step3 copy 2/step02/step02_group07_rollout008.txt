Uses scipy HiGHS LP solver, FFT convolution, subgradient polishing, and peak-targeted refinement.

import numpy as np
import time

from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    """Build matrix A where A@b = conv(f,b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0. Returns b or None."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def gradient_polish(f_init, time_limit, verbose=False):
    """Subgradient descent with momentum for polishing."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (float(n) / S)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta_m = 0.85
    lr = 0.005
    lr_decay = 0.9999
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        # Use top K peaks for subgradient
        top_k = min(7, len(c))
        top_idx = np.argsort(c)[-top_k:]

        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(top_k)
        g -= np.mean(g)

        momentum = beta_m * momentum + (1.0 - beta_m) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, None)
        f = f * (float(n) / float(np.sum(f)))
        lr *= lr_decay

        it += 1
        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    [grad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def peak_targeted_search(f_init, time_limit, verbose=False):
    """Target peak reduction: identify peaks and reduce contributing elements."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()

            break

        c = fft_convolve(f, f)
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[-top_k:]

        # For each peak, find top contributing pairs and reduce them
        candidate = f.copy()
        for k in top_idx:
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            contribs = np.zeros(n)
            for j in range(lo, hi + 1):
                contribs[j] = f[j] * f[ki - j]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue
            # Reduce top contributors
            n_reduce = max(3, len(contribs[lo:hi+1]) // 4)
            idx_in_range = np.arange(lo, hi + 1)
            top_contrib_idx = idx_in_range[np.argsort(contribs[lo:hi+1])[-n_reduce:]]
            for idx in top_contrib_idx:
                candidate[idx] *= (1.0 - 0.05 * rng.random())
                # Compensate by increasing a low element
                low_idx = int(rng.integers(0, n))
                candidate[low_idx] *= (1.0 + 0.03 * rng.random())

        candidate = np.clip(candidate, 0.01, 1000.0)
        S_c = float(np.sum(candidate))
        if S_c > 0.01:
            candidate = candidate * (S / S_c)
        new_score = score_fast(candidate)

        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        else:
            # Try a smaller perturbation
            small_cand = f.copy()
            for k in top_idx[:2]:
                ki = int(k)
                lo = max(0, ki - (n - 1))
                hi = min(n - 1, ki)
                for j in range(lo, hi + 1):
                    contrib = f[j] * f[ki - j]
                    if contrib > 0.5 * c[ki]:
                        small_cand[j] *= (1.0 - 0.02)
                        small_cand[ki - j] *= (1.0 - 0.02)
            small_cand = np.clip(small_cand, 0.01, 1000.0)

    if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
        try:
            hs = globals().get('height_sequence_1', None)
            if hs is not None:
                provided = np.clip(np.asarray(hs, dtype=np.float64), 0.01, 1000.0)
                s = score_fast(provided)
                print(f"Provided: n={len(provided)}, score={s:.6f}")
                if s < best_score:
                    best_score = s
                    best_seq = provided.copy()
        except Exception:
            pass

    print("=" * 60)
    print(f"Phase 1: Multi-initialization LP search")
    print("=" * 60)

    # Define tasks: (n, init_func)
    init_funcs = [make_init_hann, make_init_blackman, make_init_bartlett,
                  make_init_chirp, make_init_random, make_init_gaussian_tail,
                  make_init_dual_peak]

    tasks = []
    if provided is not None:
        tasks.append((len(provided), 'provided'))

    # Focus on n=256 (fast LP) with many inits, then larger n
    for n in [256]:
        for idx in range(min(7, len(init_funcs))):
            tasks.append((n, idx))
    for n in [320, 384]:
        for idx in range(3):
            tasks.append((n, idx))
    for n in [512]:
        for idx in range(2):
            tasks.append((n, idx))

    # Time allocation
    results = []
    tasks_done = 0
    for n, init_id in tasks:
        if time.monotonic() > deadline - 200:
            break

        # Generate initialization
        if init_id == 'provided' and provided is not None:
            if len(provided) == n:
                f0 = provided.copy()
            else:
                x_old = np.linspace(0, 1, len(provided))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, provided)
        else:
            f0 = init_funcs[init_id % len(init_funcs)](n, rng)

        # Time allocation based on n
        if n <= 256:
            t_alloc = min(150.0, (deadline - time.monotonic() - 200.0) / max(1, len(tasks) - tasks_done))
            t_alloc = max(t_alloc, 20.0)
        elif n <= 384:
            t_alloc = min(80.0, (deadline - time.monotonic() - 200.0) / max(1, len(tasks) - tasks_done))
            t_alloc = max(t_alloc, 15.0)
        else:
            t_alloc = min(50.0, (deadline - time.monotonic() - 200.0) / max(1, len(tasks) - tasks_done))
            t_alloc = max(t_alloc, 10.0)

        init_name = 'provided' if init_id == 'provided' else f'init{init_id}'
        print(f"  n={n:4d} init={init_name:10s} alloc={t_alloc:.0f}s")
        f_best, s_best = optimize_lp(f0, t_alloc, t_base=0.05, verbose=False)
        print(f"  => score={s_best:.6f} (global best={best_score:.6f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

        results.append((s_best, n, f_best.copy()))
        tasks_done += 1

    # Phase 2: Multi-direction LP on top candidates
    print("\n" + "=" * 60)
    print("Phase 2: Multi-direction LP refinement")
    print("=" * 60)

    results.sort(key=lambda x: x[0])
    top_k = min(3, len(results))
    for i in range(top_k):
        if time.monotonic() > deadline - 150:
            break
        s_i, n_i, f_i = results[i]
        t_alloc = min(100.0, deadline - time.monotonic() - 150.0)
        if t_alloc < 10:
            break
        print(f"  Refining candidate {i} (n={n_i}, score={s_i:.6f}), alloc={t_alloc:.0f}s")
        f_ref, s_ref = multi_lp_directions(f_i, t_alloc, verbose=False)
        print(f"  => score={s_ref:.6f} (global best={best_score:.6f})")
        if s_ref < best_score:
            best_score = s_ref
            best_seq = f_ref.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 3: Gradient polish
    print("\n" + "=" * 60)
    print("Phase 3: Gradient polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 100:
        pt = min(150.0, deadline - time.monotonic() - 100.0)
        if pt > 10:
            print(f"  Polishing (alloc={pt:.0f}s)")
            f_pol, s_pol = gradient_polish(best_seq, pt, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = f_pol.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 4: Peak targeted search
    print("\n" + "=" * 60)
    print("Phase 4: Peak targeted search")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 50:
        pt_s = min(100.0, deadline - time.monotonic() - 50.0)
        if pt_s > 10:
            print(f"  Peak targeting (alloc={pt_s:.0f}s)")
            f_pt, s_pt = peak_targeted_search(best_seq, pt_s, verbose=True)
            if s_pt < best_score:
                best_score = s_pt
                best_seq = f_pt.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 5: Final LP polish
    print("\n" + "=" * 60)
    print("Phase 5: Final LP polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 10:
        fl_t = min(50.0, deadline - time.monotonic() - 5.0)
        if fl_t > 10:
            print(f"  Final LP (alloc={fl_t:.0f}s)")
            f_fin, s_fin = optimize_lp(best_seq, fl_t, t_base=0.02, verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Final validation
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    return [1.0] * 256
