def smooth_objective_grad(f, top_k=50, tau=0.01):
    """Compute gradient of smooth max approximation.
    Uses the top-k peaks with softmax weighting.
    """
    n = len(f)
    c = fft_convolve(f, f)
    
    # Get top k values
    top_idx = np.argsort(c)[-top_k:]
    top_vals = c[top_idx]
    
    # Softmax weights
    max_val = np.max(top_vals)
    weights = np.exp((top_vals - max_val) / tau)
    weights = weights / np.sum(weights)
    
    # Weighted subgradient
    g = np.zeros(n)
    for w, ki in zip(weights, top_idx):
        ki = int(ki)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        g[lo:hi+1] += 2.0 * w * f[ki-hi:ki-lo+1][::-1]
    
    return g, c

import numpy as np
import time
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def normalize_sum1(a):
    S = float(np.sum(a))
    if S < 1e-12:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / S

def subgrad_smooth(f, c, top_k=50, tau=0.005):
    """Smooth subgradient using softmax weighting over top-k peaks."""
    n = len(f)
    top_idx = np.argsort(c)[-top_k:]
    top_vals = c[top_idx].astype(np.float64)
    
    # Numerically stable softmax
    max_v = top_vals.max()
    weights = np.exp((top_vals - max_v) / tau)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(top_k) / top_k
    else:
        weights /= w_sum
    
    g = np.zeros(n)
    for w, ki in zip(weights, top_idx):
        if w < 1e-10:
            continue
        ki = int(ki)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * w * f[ki-hi:ki-lo+1][::-1]
    return g

def optimize_subgrad(f_init, time_limit, top_k_init=10, top_k_max=100, 
                     tau_init=0.002, lr_init=0.01, momentum=0.9, verbose=True):
    """Main optimizer: smooth subgradient descent with momentum and adaptive params."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = normalize_sum1(f)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    vel = np.zeros(n)
    lr = lr_init
    tau = tau_init
    top_k = top_k_init
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break
        
        c = fft_convolve(f, f)
        g = subgrad_smooth(f, c, top_k=top_k, tau=tau)
        
        # Normalize gradient for stable stepping
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Momentum update
        vel = momentum * vel + (1.0 - momentum) * g
        
        # Project gradient to sum-zero subspace (to maintain normalization)
        vel -= np.mean(vel)
        
        # Update
        f = f - lr * vel
        f = np.clip(f, 0.0, None)
        f = normalize_sum1(f)
        
        it += 1
        
        # Adaptive parameters
        if it % 100 == 0:
            # Gradually increase top_k
            top_k = min(top_k_max, top_k_init + it // 200)
            # Gradually decrease tau
            tau = max(0.0005, tau_init * 0.99 ** (it // 100))
        
        if it % 20 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                stall = 0
                # Slightly increase lr on improvement
                lr = min(lr * 1.01, lr_init * 3)
            else:
                stall += 1
                if stall > 50:
                    lr *= 0.9
                    if lr < lr_init * 0.1:
                        # Explore: perturbation
                        rng_p = np.random.default_rng(1000 + it)
                        f = f + 0.005 * rng_p.standard_normal(n)

                        f = normalize_sum1(f)
                        lr = lr_init
                        stall = 0

    if s < best_score:
        best_score = s
        best_f = f.copy()
    
    return best_f, best_score

def optimize_spectral(f_init, time_limit, alpha=0.5, verbose=False):
    """Spectral shaping: flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = normalize_sum1(f)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break
        
        # Compute power spectrum
        F = rfft(f, 2 * n)
        P = np.abs(F) ** 2
        
        # Flatten: P_new = P^alpha (alpha < 1 makes it flatter)
        P_flat = P ** alpha
        # Preserve total power
        if P_flat.sum() > 1e-30:
            P_flat *= P.sum() / P_flat.sum()
        
        # Reconstruct with new magnitude, same phase
        mag = np.sqrt(np.maximum(P_flat, 0))
        phase = np.angle(F)
        F_new = mag * np.exp(1j * phase)
        
        # Inverse transform
        f_new = irfft(F_new, 2 * n)[:n]
        f_new = np.clip(f_new, 0.0, None)
        
        S = float(np.sum(f_new))
        if S < 1e-10:
            break
        f_new = f_new / S
        
        f = f_new
        it += 1
        
        if it % 50 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    [spectral {time.monotonic()-t0:5.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    
    return best_f, best_score

        t = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 + np.cos(2*np.pi*t / max(1, n-1)))
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif method == 'random':
        return np.clip(0.3 + 0.7*rng.random(n), 0.01, None)
    elif method == 'flat':
        return np.ones(n) + 0.05 * rng.standard_normal(n)
    elif method == 'bartlett':
        t = np.arange(n, dtype=np.float64)
        w = 1.0 - np.abs(2.0*t/max(1, n-1) - 1.0)
        return np.clip(0.2 + 0.8*w + 0.03*rng.standard_normal(n), 0.01, None)
    elif method == 'gaussian':
        t = np.arange(n, dtype=np.float64)

        a = np.exp(-0.5 * ((t - center) / (n * 0.25)) ** 2)
        return np.clip(a + 0.02*rng.standard_normal(n), 0.01, None)
    elif method == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.5 * np.cos(2*np.pi*t/n)
        return np.clip(a + 0.03*rng.standard_normal(n), 0.01, None)

        return np.clip(1.0 + 0.1*rng.standard_normal(n), 0.01, None)
