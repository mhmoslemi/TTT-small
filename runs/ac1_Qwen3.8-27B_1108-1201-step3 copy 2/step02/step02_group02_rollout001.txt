def fft_convolve(a):
    """Compute linear autocorrelation using FFT (circulant embedding)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_peaks(a, top_k=10):
    """Compute autocorrelation and return top-k peak positions and values."""
    c = fft_convolve(a)
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    return c, top_idx, c[top_idx]
