import numpy as np

def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2*n-1]

def fast_score(a):
    """Compute evaluation score (scale-invariant version, assumes sum > 0)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def compute_subgradient(a, top_k=5):
    """Compute subgradient of max(conv(a,a)) w.r.t. a.
    Uses the formula: d(conv[k])/d(a[m]) = 2*a[k-m] if 0<=k-m<n else 0.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    
    # Find top-k lags (those closest to max)
    threshold = max_c * 0.999  # lags within 0.1% of max
    peak_lags = np.where(conv >= threshold)[0]

        peak_lags = peak_lags[np.argsort(conv[peak_lags])[-top_k:]]

    for k in peak_lags:
        # grad[m] += 2*a[k-m] for m where 0<=k-m<n

            # k-m goes from k-m_start to k-m_end (decreasing)
            grad[m_start:m_end+1] += 2.0 * a[k - m_start:k - m_end+1][::-1]
    
    grad /= len(peak_lags)
    return grad, max_c, conv

    """Project onto non-negative simplex with given sum."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / s)

    """Optimize using subgradient descent with momentum and adaptive step size."""
    n = len(a_init)
    target_sum = np.sqrt(2.0 * n)

    a = a * (target_sum / np.sum(a))

        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = project(a_new, target_sum)
        
        new_score = fast_score(a_new)

            a = a_new
            current_score = new_score
            stall = 0
            lr = min(lr * 1.1, lr_max)

                    a = a_pert
                    current_score = pert_score
                stall = 0
                lr = 0.01
                momentum = np.zeros(n)

import time

    na = len(a)
    nb = len(b)

    max_c = np.max(conv)
    threshold = max_c * threshold_frac
    peak_lags = np.where(conv >= threshold)[0]
    if len(peak_lags) > top_k:
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[-top_k:]]
    grad = np.zeros(n)
    for k in peak_lags:
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            idx = k - np.arange(m_start, m_end + 1)
            grad[m_start:m_end + 1] += 2.0 * a[idx]
    if len(peak_lags) > 0:
        grad /= len(peak_lags)
    return grad, max_c, conv, peak_lags


def build_constraint_matrix(f):
    """Build constraint matrix for LP: A[k,i] = f[k-i] if valid, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - n + 1)
        hi = min(2 * n - 2, j + n - 1)
        A[j:hi + 1, j] = f[lo:hi - (j - n + 1) + 1] if j >= n - 1 else f[:hi + 1 - j]
        # Simpler: A[k, j] = f[k-j] if 0<=k-j<n
    # Redo properly
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(max(0, j), min(2 * n - 1, j + n)):
            A[k, j] = f[k - j]
    return A


def build_constraint_matrix_fast(f):
    """Build constraint matrix efficiently using structured approach."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2 * n - 2, j + n - 1)
        f_start = 0
        f_end = k_end - j + 1
        A[k_start:k_end + 1, j] = f[f_start:f_end]
    return A


def solve_lp_direction(f, M, max_time=5.0):
    """Solve LP to find best additive direction bounded by M."""
    n = len(f)
    A = build_constraint_matrix_fast(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.ones(n) / n


def subgradient_optimize(a_init, time_limit, rng, verbose=True, use_lp=False, lp_interval=200):
    """Main optimization loop using subgradient descent with momentum."""
    n = len(a_init)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a_init, 0.0).copy()
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) * (target_sum / n)
    else:
        a = a * (target_sum / s)

    best_a = a.copy()
    best_score = fast_score(a)
    current_score = best_score

    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.003
    lr_min = 1e-6
    lr_max = 0.05
    scale = target_sum * target_sum

    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    lp_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        max_c = np.max(conv)
        current_score = 2.0 * n * max_c / scale

        # Subgradient computation
        grad, max_c, conv, peak_lags = compute_subgrad(a, top_k=5, threshold_frac=0.995)

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (target_sum / s_new)

        # Evaluate
        conv_new = fft_convolve(a_new, a_new)
        new_score = 2.0 * n * np.max(conv_new) / scale

        if new_score < current_score:
            a = a_new
            current_score = new_score
            stall = 0
            lr = min(lr * 1.03, lr_max)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                if verbose and (iter_count % 5000 == 0 or best_score < 1.51):
                    print(f"      SG iter={iter_count}, best={best_score:.6f}, lr={lr:.5f}, t={time.monotonic()-t0:.1f}s")
        else:
            stall += 1
            lr = max(lr * 0.92, lr_min)

            if stall > 200:
                # Perturb to escape
                sigma = 0.01 + 0.005 * (stall // 500)
                a = a * np.exp(rng.normal(0, min(sigma, 0.05), n))
                a = np.maximum(a, 0.0)
                a = a * (target_sum / np.sum(a))
                current_score = fast_score(a)
                stall = 0
                lr = 0.003
                momentum = np.zeros(n)

        # Periodic LP direction
        if use_lp and iter_count > 0 and iter_count % lp_interval == 0 and time.monotonic() - t0 < time_limit - 10:
            M = max_c
            b_dir = solve_lp_direction(a, M)
            sum_b = np.sum(b_dir)
            if sum_b > 1e-10:
                g = b_dir * (target_sum / sum_b)
                # Try LP direction with small step
                for t_trial in [0.1, 0.05, 0.02]:
                    a_trial = (1.0 - t_trial) * a + t_trial * g
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial * (target_sum / np.sum(a_trial))
                    trial_score = fast_score(a_trial)
                    if trial_score < current_score:
                        a = a_trial
                        current_score = trial_score
                        stall = 0
                        if current_score < best_score:
                            best_score = current_score
                            best_a = a.copy()
                            if verbose:
                                print(f"      LP improved: {best_score:.6f} (iter={iter_count})")
                        break
            lp_count += 1

        iter_count += 1

    return best_a, best_score


def targeted_local_search(a_init, time_limit, rng, verbose=True):
    """Local search focused on reducing peak contributions."""
    n = len(a_init)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a_init, 0.0).copy()
    a = a * (target_sum / np.sum(a))

    best_a = a.copy()
    best_score = fast_score(a)
    current_score = best_score
    scale = target_sum * target_sum

    t0 = time.monotonic()
    temp = 0.0005
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        max_c = np.max(conv)
        current_score = 2.0 * n * max_c / scale

        candidate = a.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted: reduce elements contributing to top peaks
            peak_lags = np.argsort(conv)[-3:]
            for k in peak_lags:
                m_start = max(0, k - n + 1)
                m_end = min(n - 1, k)
                if m_start > m_end:
                    continue
                # Find top contributing pairs
                idx = k - np.arange(m_start, m_end + 1)
                contribs = a[m_start:m_end + 1] * a[idx]
                top_n = max(3, n // 15)
                local_top = np.argsort(contribs)[-top_n:]
                abs_idx = m_start + local_top
                candidate[abs_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(abs_idx)))
            # Increase some non-peak-contributing elements
            peak_mask = np.zeros(n, dtype=bool)
            for k in peak_lags:
                m_start = max(0, k - n + 1)
                m_end = min(n - 1, k)
                if m_start <= m_end:
                    peak_mask[m_start:m_end + 1] = True
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) > 5:
                inc_idx = rng.choice(non_peak, size=min(10, len(non_peak)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=len(inc_idx)))

        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))

        elif r < 0.60:
            # Pair transfer (preserve sum)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.2, 0.2) * a[i]
                candidate[i] = max(0.0, a[i] - delta)
                candidate[j] = max(0.0, a[j] + delta)

        elif r < 0.75:
            # Multiple small blocks
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.01, size=width))

        elif r < 0.90:
            # Frequency-domain-inspired: modulate with low-frequency pattern
            freq = rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.arange(n)
            mod = 1.0 + 0.02 * np.cos(2 * np.pi * freq * t / n + phase)
            candidate = a * mod
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0.0, 0.005, n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        cs = np.sum(candidate)
        if cs > 0.01:
            candidate = candidate * (target_sum / cs)

        new_score = fast_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                if verbose and (int(time.monotonic() - t0) % 30 < 1):
                    print(f"      LS improved: {best_score:.6f} (t={time.monotonic()-t0:.1f}s)")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            a = candidate
            current_score = new_score

        temp *= cooling

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Create chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_alternating_init(n, ratio, rng):
    """Create alternating high-low initialization (reduces center peak)."""
    t = np.arange(n)
    a = np.where(t % 2 == 0, 1.0 + ratio, 1.0 - 0.5 * ratio)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 5):
        amp = 0.2 / k
        freq = rng.uniform(0.5, 2.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(seq, new_n):
    """Resample sequence to new length via interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimization: subgradient descent + LP + targeted local search.
    Starts from provided sequence if available, explores multiple n values.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = fast_score(start_seq)
            print(f"Starting sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Phase 1: Quick exploration from multiple starts
    print("\n=== Phase 1: Quick exploration ===")
    if time.monotonic() < deadline - 500:
        explore_time = min(150.0, deadline - time.monotonic() - 500)
        n_explores = 6
        t_per = explore_time / n_explores

        configs = [
            (512, 'chirp', 0.6),
            (512, 'alternating', 0.4),
            (512, 'multi_tone', 0),
            (384, 'chirp', 0.5),
            (640, 'chirp', 0.4),
            (512, 'alternating', 0.7),
        ]

        for ci, (n, init_type, alpha) in enumerate(configs):
            if time.monotonic() > deadline - 500:
                break
            # Sometimes resample best
            if best_seq is not None and rng.random() < 0.3:
                a0 = resample_sequence(best_seq, n)
            else:
                if init_type == 'chirp':
                    a0 = make_chirp_init(n, alpha + 0.2 * rng.random(), rng)
                elif init_type == 'alternating':
                    a0 = make_alternating_init(n, alpha + 0.2 * rng.random(), rng)
                else:
                    a0 = make_multi_tone_init(n, rng)

            a_best, s_best = subgradient_optimize(a0, t_per, rng, verbose=False, use_lp=False)
            print(f"  n={n}, {init_type}: score={s_best:.6f}")
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Main optimization on best candidate
    print(f"\n=== Phase 2: Main optimization (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        main_time = min(900.0, deadline - time.monotonic() - 300)
        print(f"  Time budget: {main_time:.0f}s")
        a_best, s_best = subgradient_optimize(best_seq, main_time, rng, verbose=True, use_lp=True, lp_interval=300)
        print(f"  Phase 2 result: {s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Targeted local search
    if best_seq is not None and time.monotonic() < deadline - 100:
        ls_time = min(400.0, max(50.0, deadline - time.monotonic() - 100))
        print(f"\n=== Phase 3: Local search (time={ls_time:.0f}s) ===")
        a_best, s_best = targeted_local_search(best_seq, ls_time, rng, verbose=True)
        print(f"  Phase 3 result: {s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Final subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        polish_time = max(20.0, deadline - time.monotonic() - 30)
        print(f"\n=== Phase 4: Final polish (time={polish_time:.0f}s) ===")
        a_best, s_best = subgradient_optimize(best_seq, polish_time, rng, verbose=False, use_lp=False)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Final short local search
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 15
        if remaining > 30:
            a_best, s_best = targeted_local_search(best_seq, remaining, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
