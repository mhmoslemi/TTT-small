```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent
with logsumexp smoothing, FFT-based gradients, peak-shaving local search,
and multi-length exploration. Starts from parent sequence and diverse inits.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def smooth_obj_and_grad(a, tau):
    """Compute smooth objective and gradient for a on the simplex."""
    n = len(a)
    conv = fast_convolve(a)
    x = conv / tau
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    exp_sum = exp_x.sum()
    probs = exp_x / exp_sum
    obj = 2.0 * n * tau * (x_max + np.log(exp_sum))

    # Gradient: df/da[j] = 4n * sum_{m=0}^{n-1} probs[j+m] * a[m]
    # Compute via FFT cross-correlation
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    fft_size2 = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size2)
    P = np.fft.rfft(probs, fft_size2)
    conv_full = np.fft.irfft(A_rev * P, fft_size2)[:conv_len]
    corr = conv_full[n - 1:2 * n - 1]
    grad = 4.0 * n * corr
    return obj, grad


def obj_exact(a):
    """Exact objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()


def pgd_optimize(a0, time_limit, tau_init=0.05, tau_min=0.001, lr_init=0.01):
    """Projected gradient descent with adaptive temperature and momentum."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = obj_exact(a)
    t0 = time.monotonic()
    n = len(a)

    momentum = np.zeros(n)
    beta = 0.9
    lr = lr_init
    tau = tau_init
    tau_decay = 0.995
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit:
        obj_s, grad = smooth_obj_and_grad(a, tau)

        if not np.isfinite(obj_s):
            break

        # Nesterov momentum
        momentum = beta * momentum + grad
        direction = momentum / (np.linalg.norm(momentum) + 1e-15)

        # Backtracking line search
        a_trial = a - lr * direction
        a_trial = project_simplex(a_trial)
        obj_trial = obj_exact(a_trial)

        if obj_trial < best_obj - 1e-12:
            a = a_trial
            best_obj = obj_trial
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            a_trial2 = a - lr * 0.1 * direction
            a_trial2 = project_simplex(a_trial2)
            obj_trial2 = obj_exact(a_trial2)
            if obj_trial2 < best_obj - 1e-12:
                a = a_trial2
                best_obj = obj_trial2
                best_a = a.copy()
                no_improve = 0
            else:
                lr *= 0.5

        # Decay temperature
        tau = max(tau_min, tau * tau_decay)
        # Restore lr if it got too small
        if lr < lr_init * 0.01:
            lr = lr_init * 0.1

        if no_improve >= max_no_improve:
            break

    return best_a, best_obj


def peak_shave(a, time_limit):
    """Peak shaving: reduce elements contributing most to the peak lag."""
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a)
    else:
        a = a / s
    best_a = a.copy()
    best_obj = obj_exact(a)
    t0 = time.monotonic()
    n = len(a)
    stuck = 0
    max_stuck = 100

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]

        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]

        # Find high and low contributors
        contrib_sorted = np.sort(contrib)
        high_thresh = contrib_sorted[int(0.7 * n)]
        low_thresh = contrib_sorted[int(0.3 * n)]
        high_idx = np.where(contrib > high_thresh)[0]
        low_idx = np.where(contrib < low_thresh)[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            stuck += 1
            continue

        found = False
        # Try moving mass from high to low contributors
        for hi in high_idx[:20]:
            if found:
                break
            for lo in low_idx[:20]:
                if found:
                    break
                if hi == lo:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    amount = a[hi] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[hi] -= amount
                    a_trial[lo] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_exact(a_trial)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        best_a = a.copy()
                        found = True
                        break

        if not found:
            stuck += 1
            # Also try small random perturbations
            for _ in range(10):
                idx = np.random.randint(0, n)
                if a[idx] < 1e-6:
                    continue
                frac = np.random.uniform(0.01, 0.1)
                amount = a[idx] * frac
                target = np.random.randint(0, n)
                if target == idx:
                    continue
                a_trial = a.copy()
                a_trial[idx] -= amount
                a_trial[target] += amount
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                o = obj_exact(a_trial)
                if o < best_obj - 1e-12:
                    best_obj = o
                    a = a_trial.copy()
                    best_a = a.copy()
                    found = True
                    break
        else:
            stuck = 0

    return best_a, best_obj


def get_parent_scaled(n):
    """Get parent sequence scaled/interpolated to length n, normalized."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_inits(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Gaussian
    for sf in [0.2, 0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0.0)
    hn = hann.sum()
    if hn > 0:
        inits['hann'] = hann / hn

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Random Dirichlet
    for alpha in [0.1, 0.3, 0.5, 1.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'dirichlet_{alpha}'] = r

    # Edge-concentrated
    edges = np.zeros(n)
    m = max(1, n // 3)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Cosine ripple
    freqs = [2, 3, 5, 8]
    for f in freqs:
        c = 1.0 + 0.5 * np.cos(2.0 * np.pi * f * x / n)
        c = np.maximum(c, 0.0)
        cs = c.sum()
        if cs > 0:
            inits[f'ripple_{f}'] = c / cs

    # Parent
    parent = get_parent_scaled(n)
    if parent is not None:
        inits['parent'] = parent

    # Perturbed parent
    if parent is not None:
        for noise_level in [0.05, 0.1, 0.2]:
            noise = rng.dirichlet(np.ones(n) * 0.5)
            pp = (1.0 - noise_level) * parent + noise_level * noise
            pp = np.maximum(pp, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits[f'parent_pert_{noise_level}'] = pp / ps

    return inits


def two_phase_optimize(a0, time_limit):
    """Two-phase: smooth PGD then exact peak shaving."""
    t0 = time.monotonic()
    # Phase A: PGD with smoothing (60% of time)
    t_pgd = min(time_limit * 0.6, max(5.0, (time_limit - 10) * 0.6))
    a1, obj1 = pgd_optimize(a0, time_limit=t_pgd, tau_init=0.08, tau_min=0.002, lr_init=0.005)
    # Phase B: Peak shaving (40% of time)
    remaining = time_limit - (time.monotonic() - t0)
    a2, obj2 = peak_shave(a1, time_limit=max(1.0, remaining * 0.7))
    # Phase C: Brief PGD again
    remaining2 = time_limit - (time.monotonic() - t0)
    if remaining2 > 3.0:
        a3, obj3 = pgd_optimize(a2, time_limit=remaining2 * 0.5, tau_init=0.005, tau_min=0.001, lr_init=0.003)
        if obj3 < obj2:
            a2, obj2 = a3, obj3
    return a2, obj2


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-strategy optimization combining projected gradient
    descent with logsumexp smoothing, peak-shaving local search, and multi-length
    exploration to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_time = max(10.0, float(budget_s))
    deadline = time.monotonic() + total_time - 30.0  # 30s safety margin

    best_sequence = None
    best_score = float('inf')

    # Try multiple lengths
    lengths = [256, 384, 512, 640, 768, 1024]

    # Phase 1: Quick scan (35% of budget)
    print("Phase 1: Quick scan over lengths and initializations...")
    phase1_budget = (total_time - 30.0) * 0.35
    phase1_end = time.monotonic() + phase1_budget

    all_configs = []
    for n in lengths:
        inits = make_inits(n, rng)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))

    # Prioritize parent init and medium lengths
    def config_priority(item):
        n, name, _ = item
        if name == 'parent':
            return (0, n)
        if name.startswith('parent_pert'):
            return (1, n)
        if n == 512:
            return (2, n)
        return (3, n)

    all_configs.sort(key=config_priority)

    n_configs = len(all_configs)
    time_per_scan = max(2.0, phase1_budget / max(1, n_configs))

    scan_results = []
    for i, (n, name, a0) in enumerate(all_configs):
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_per_scan, 30.0)
        t_alloc = min(t_alloc, remaining - 20.0)
        if t_alloc < 1.5:
            break

        a_opt, obj_opt = two_phase_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates (45% of budget)
    print("Phase 2: Deep optimization on top candidates...")
    phase2_budget = (total_time - 30.0) * 0.45
    phase2_end = time.monotonic() + phase2_budget

    scan_results.sort(key=lambda x: x[0])
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 12:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    n_deep = len(top_configs)
    time_per_deep = max(10.0, phase2_budget / max(1, n_deep))

    for i, (n, name, a0) in enumerate(top_configs):
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break
        t_alloc = min(time_per_deep, remaining - 15.0)
        if t_alloc < 5.0:
            break

        a_opt, obj_opt = two_phase_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 2 best: {best_score:.6f}")

    # Phase 3: Specialized refinement on best (remaining time)
    print("Phase 3: Intensive refinement...")
    if best_sequence is not None:
        polish_deadline = deadline
        no_global_improve = 0
        max_global_no_improve = 15

        while time.monotonic() < polish_deadline:
            remaining = polish_deadline - time.monotonic()
            if remaining < 10:
                break

            current = best_sequence.copy()
            n = len(current)

            # Strategy 1: PGD with different hyperparams
            t_alloc1 = min(remaining * 0.25, 60.0)
            if t_alloc1 > 3:
                a1, o1 = pgd_optimize(current, time_limit=t_alloc1,
                                      tau_init=0.02, tau_min=0.001, lr_init=0.003)
                s1 = float(evaluate_sequence(a1.tolist()))
                if np.isfinite(s1) and s1 < best_score - 1e-10:
                    best_score = s1
                    best_sequence = a1.copy()
                    no_global_improve = 0
                    print(f"  Polish PGD: score={best_score:.6f}")
                    continue

            # Strategy 2: Peak shaving
            t_alloc2 = min(remaining * 0.25, 60.0)
            if t_alloc2 > 3:
                a2, o2 = peak_shave(current, time_limit=t_alloc2)
                s2 = float(evaluate_sequence(a2.tolist()))
                if np.isfinite(s2) and s2 < best_score - 1e-10:
                    best_score = s2
                    best_sequence = a2.copy()
                    no_global_improve = 0
                    print(f"  Polish Shave: score={best_score:.6f}")
                    continue

            # Strategy 3: Random perturbation + PGD
            t_alloc3 = min(remaining * 0.15, 30.0)
            if t_alloc3 > 2:
                for _ in range(5):
                    if time.monotonic() > polish_deadline - 10:
                        break
                    noise = rng.dirichlet(np.ones(n) * 0.3)
                    for mix in [0.1, 0.2, 0.35]:
                        a_pert = (1.0 - mix) * current + mix * noise
                        a_pert = np.maximum(a_pert, 0.0)
                        a_pert = a_pert / a_pert.sum()
                        a3, o3 = pgd_optimize(a_pert, time_limit=min(t_alloc3 / 5, 5.0),
                                              tau_init=0.01, tau_min=0.001, lr_init=0.005)
                        s3 = float(evaluate_sequence(a3.tolist()))
                        if np.isfinite(s3) and s3 < best_score - 1e-10:
                            best_score = s3
                            best_sequence = a3.copy()
                            no_global_improve = 0
                            print(f"  Polish Perturb: score={best_score:.6f}")

            # Strategy 4: Try a different length by interpolation
            t_alloc4 = min(remaining * 0.15, 30.0)
            if t_alloc4 > 3:
                for new_n in [n // 2, int(n * 1.5)]:
                    if new_n < 64 or new_n > 2048:
                        continue
                    x_old = np.linspace(0, 1, n)
                    x_new = np.linspace(0, 1, new_n)
                    a_interp = np.interp(x_new, x_old, current)
                    a_interp = np.maximum(a_interp, 0.0)
                    s = a_interp.sum()
                    if s < 1e-10:
                        continue
                    a_interp /= s
                    a4, o4 = two_phase_optimize(a_interp, time_limit=min(t_alloc4 / 2, 15.0))
                    s4 = float(evaluate_sequence(a4.tolist()))
                    if np.isfinite(s4) and s4 < best_score - 1e-10:
                        best_score = s4
                        best_sequence = a4.copy()
                        no_global_improve = 0
                        print(f"  Polish ReLength n={new_n}: score={best_score:.6f}")

            no_global_improve += 1
            if no_global_improve >= max_global_no_improve:
                break

    # Final check
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
