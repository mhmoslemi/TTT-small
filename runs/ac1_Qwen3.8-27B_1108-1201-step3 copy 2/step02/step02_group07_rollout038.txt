Accelerated subgradient optimization for minimizing peak auto-convolution

    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    c = fft_convolve(a, a)

30. **Projection onto simplex**:
    The projection of a vector x onto {v : v >= 0, sum(v) = 1} is:
    v_i = max(x_i - τ, 0) where τ is chosen so that sum(v) = 1.
    
    Algorithm (sort-based, O(n log n)):
    1. Sort x in decreasing order: x_{(1)} >= x_{(2)} >= ... >= x_{(n)}.
    2. Find the largest ρ such that sum_{i=1}^{ρ} (x_{(i)} - τ) = 1 where τ = (sum_{i=1}^{ρ} x_{(i)} - 1) / ρ.
    3. Set v_i = max(x_i - τ, 0).

    Simpler O(n) approach using binary search on τ:
    - τ is in range [min(x) - 1, max(x)].
    - f(τ) = sum(max(x_i - τ, 0)) is monotonically decreasing in τ.
    - Binary search for τ such that f(τ) = 1.

    For efficiency, I'll use the sort-based approach since n is at most ~2048.

"""
Mirror descent (multiplicative weights) with top-K weighted subgradients for
minimizing peak auto-convolution of non-negative sequences. Uses FFT-based
convolution, multi-scale search across lengths, and targeted local refinement.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def subgradient_at_lag(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[lo:hi + 1] = 2.0 * a[k - idx]
    return g


def weighted_subgradient(a, c, top_k=3):
    """Compute weighted subgradient using top-K peaks with soft weighting."""
    n = len(a)
    # Use exponential weighting on top peaks
    sorted_idx = np.argsort(c)[::-1][:top_k]
    max_c = c[sorted_idx[0]]
    weights = np.exp((c[sorted_idx] - max_c) * 5.0)
    weights /= weights.sum()
    g = np.zeros(n)
    for w, idx in zip(weights, sorted_idx):
        if w > 1e-8:
            g += w * subgradient_at_lag(a, int(idx))
    return g


def project_simplex(x):
    """Project x onto {v >= 0, sum(v) = 1}."""
    n = len(x)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(n) / n
    x = x / s
    # Ensure exact sum = 1
    x[-1] = 1.0 - x[:-1].sum()
    if x[-1] < 0:
        # Fallback: sort-based projection
        u = np.sort(x)[::-1]
        cssv = np.cumsum(u)
        rho = np.find_last(u * np.arange(1, n + 1) > (cssv - 1.0))
        if rho == -1:
            rho = 0
        tau = (cssv[rho] - 1.0) / (rho + 1)
        x = np.maximum(x - tau, 0.0)
    return x


def mirror_descent(a_init, time_limit, eta=1.0, top_k=3, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a_init)
    x = a_init.astype(np.float64).copy()
    x = np.clip(x, 1e-6, None)
    x = x / x.sum()
    best_x = x.copy()
    best_score = score_fast(x)
    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(x, x)
        g = weighted_subgradient(x, c, top_k)

        # Mirror descent update: x * exp(-eta * g) / Z
        log_x = np.log(np.maximum(x, 1e-30))
        new_log_x = log_x - eta * g
        # Shift for numerical stability
        new_log_x -= new_log_x.max()
        new_x = np.exp(new_log_x)
        new_x = new_x / new_x.sum()

        x = new_x
        it += 1

        if it % 200 == 0:
            s = score_fast(x)
            if s < best_score - 1e-10:
                best_score = s
                best_x = x.copy()
                stale = 0
            else:
                stale += 1

            if verbose and it % 2000 == 0:
                print(f"      MD it={it} cur={s:.6f} best={best_score:.6f} t={time.monotonic()-t0:.1f}s")

            # Adaptive: if stuck, perturb
            if stale >= 20:
                rng_p = np.random.default_rng(it * 31)
                x = x * np.exp(rng_p.normal(0, 0.02, n))
                x = x / x.sum()
                stale = 0

    s = score_fast(x)
    if s < best_score:
        best_score = s
        best_x = x.copy()

    return best_x, best_score


def subgrad_momentum(a_init, time_limit, verbose=False):
    """Subgradient descent with heavy-ball momentum and simplex projection."""
    n = len(a_init)
    x = a_init.astype(np.float64).copy()
    x = np.clip(x, 1e-6, None)
    x = x / x.sum()
    v = np.zeros(n)
    best_x = x.copy()
    best_score = score_fast(x)
    t0 = time.monotonic()
    it = 0
    beta = 0.85
    step0 = 0.02
    decay = 0.001

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(x, x)
        g = weighted_subgradient(x, c, top_k=3)

        eta = step0 / (1.0 + it * decay)
        v = beta * v + (1.0 - beta) * g
        x = x - eta * v
        x = project_simplex(x)
        it += 1

        if it % 300 == 0:
            s = score_fast(x)
            if s < best_score:
                best_score = s
                best_x = x.copy()

            if verbose and it % 3000 == 0:
                print(f"      SG it={it} cur={s:.6f} best={best_score:.6f} t={time.monotonic()-t0:.1f}s")

    s = score_fast(x)
    if s < best_score:
        best_score = s
        best_x = x.copy()

    return best_x, best_score


def local_refine(a_init, time_limit, verbose=False):
    """Targeted local search: reduce peak-contributing variables, boost others."""
    n = len(a_init)
    x = a_init.astype(np.float64).copy()
    best_x = x.copy()
    best_score = score_fast(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    temp = 0.001

    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        candidate = x.copy()
        c = fft_convolve(x, x)
        k_star = int(np.argmax(c))

        # Find top contributors to the peak
        contribs = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            contribs[j] = x[j] * x[k_star - j]

        # Also check neighboring peaks
        sorted_peaks = np.argsort(c)[::-1][:5]
        all_contribs = np.zeros(n)
        for pk in sorted_peaks:
            pk = int(pk)
            lo2 = max(0, pk - (n - 1))
            hi2 = min(n - 1, pk)
            for j in range(lo2, hi2 + 1):
                all_contribs[j] += c[pk] * x[j] * x[pk - j]

        # Strategy: reduce high-contributors, increase low-contributors
        r = rng.random()
        if r < 0.5:
            # Reduce top contributors
            top_n = max(3, n // 16)
            top_idx = np.argsort(contribs)[::-1][:top_n]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.03, top_n))
            # Boost non-contributors
            remaining = np.setdiff1d(np.arange(n), top_idx)
            if len(remaining) > 0:
                boost_n = min(10, len(remaining))
                boost_idx = rng.choice(remaining, boost_n, replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.05, 0.03, boost_n))
        elif r < 0.8:
            # Smooth perturbation targeting all peaks
            mask_high = all_contribs > np.median(all_contribs)
            n_high = mask_high.sum()
            n_low = n - n_high
            if n_high > 0:
                candidate[mask_high] *= np.exp(rng.normal(-0.02, 0.02, n_high))
            if n_low > 0:
                candidate[~mask_high] *= np.exp(rng.normal(0.02 * n_high / max(1, n_low), 0.02, n_low))
        else:
            # Random smooth perturbation
            candidate *= np.exp(rng.normal(0, 0.015, n))

        candidate = np.clip(candidate, 1e-6, None)
        candidate = candidate / candidate.sum()
        new_score = score_fast(candidate)

        if new_score < best_score:
            best_score = new_score
            best_x = candidate.copy()
            x = candidate
            temp = max(0.0005, temp * 1.01)
        elif rng.random() < np.exp(-(new_score - score_fast(x)) / max(temp, 1e-15)):
            x = candidate
        else:
            temp = max(0.0003, temp * 0.999)

        it += 1
        if verbose and it % 100 == 0:
            print(f"      LS it={it} cur={score_fast(x):.6f} best={best_score:.6f} t={time.monotonic()-t0:.1f}s")

    return best_x, best_score


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bartlett_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
    a = 0.2 + 0.8 * w
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng):
    a = np.abs(rng.normal(1.0, 0.4, n)) + 0.05
    return a


def make_raised_cosine_init(n, rng, beta=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 - beta * np.cos(2.0 * np.pi * t / max(1, n - 1))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_to_length(a, new_n):
    """Resample sequence to new length via interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Mirror descent / subgradient on multiple (n, init) combinations.
    Phase 2: Extended optimization on the best found.
    Phase 3: Local refinement.
    Phase 4: Final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            if len(provided) > 0 and np.sum(provided) > 0.01:
                S = np.sum(provided)
                provided_norm = provided / S
                s = score_fast(provided_norm)
                print(f"Provided: n={len(provided)}, score={s:.6f}")
                if s < best_score:
                    best_score = s
                    best_seq = provided_norm.copy()
        except Exception:
            provided = None

    # Phase 1: Multistart
    print("\n=== Phase 1: Multistart optimization ===")

    # Define (n, num_inits) tasks
    n_values = [256, 384, 512, 768, 1024, 1536, 2048]
    tasks = []
    for n in n_values:
        if provided is not None:
            tasks.append((n, 'provided'))
        tasks.append((n, 'hann'))
        tasks.append((n, 'random'))
        if n <= 1024:
            tasks.append((n, 'bartlett'))
            tasks.append((n, 'raised_cos'))

    # Time allocation: faster for smaller n (more iterations possible)
    # But also want to try larger n
    # Give each task a base allocation, adjust by n
    base_time = 45.0
    total_est = 0
    task_times = []
    for n, _ in tasks:
        # Larger n: slightly more time (better potential) but diminishing returns
        t = base_time * (1.0 + 0.1 * np.log2(n / 256.0))
        task_times.append(t)
        total_est += t

    # Scale to fit available time (leave 500s for phases 2-4)
    avail = deadline - t_start - 500.0
    avail = max(100.0, avail)
    scale = avail / total_est
    task_times = [min(t * scale, 120.0) for t in task_times]

    print(f"Total tasks: {len(tasks)}, est total time: {sum(task_times):.0f}s")

    for idx, (n, init_name) in enumerate(tasks):
        if time.monotonic() > deadline - 450:
            print("  [Phase 1 time limit reached]")
            break

        t_alloc = min(task_times[idx], deadline - time.monotonic() - 400)
        if t_alloc < 15:
            continue

        # Generate initial sequence
        if init_name == 'provided' and provided is not None:
            f0 = resample_to_length(provided, n)
        elif init_name == 'hann':
            f0 = make_hann_init(n, rng)
        elif init_name == 'bartlett':
            f0 = make_bartlett_init(n, rng)
        elif init_name == 'raised_cos':
            f0 = make_raised_cosine_init(n, rng, beta=0.3 + 0.2 * rng.random())
        else:
            f0 = make_random_init(n, rng)

        f0 = np.clip(f0, 0.01, None)
        f0 = f0 / f0.sum()

        # Choose method: mirror descent for most, subgrad for some
        use_md = (idx % 2 == 0)
        s_init = score_fast(f0)

        if use_md:
            f_best, s_best = mirror_descent(f0, t_alloc, eta=0.8, top_k=3, verbose=False)
        else:
            f_best, s_best = subgrad_momentum(f0, t_alloc, verbose=False)

        elapsed = time.monotonic() - t_start
        status = ""
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            status = f" *** NEW BEST ***"
        print(f"  [{elapsed:6.1f}s] n={n:4d} init={init_name:10s} init_s={s_init:.4f} "
              f"final={s_best:.6f} best={best_score:.6f}{status}")

    # Phase 2: Extended optimization on best
    print("\n=== Phase 2: Extended optimization on best ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_best = len(best_seq)
        t_ext = min(200.0, deadline - time.monotonic() - 120)
        print(f"  n={n_best}, time={t_ext:.0f}s")

        # Try both methods
        t_half = t_ext / 2
        f_md, s_md = mirror_descent(best_seq, t_half, eta=0.5, top_k=5, verbose=True)
        if s_md < best_score:
            best_score = s_md
            best_seq = f_md
            print(f"  MD improved: {best_score:.6f}")

        if time.monotonic() < deadline - 100:
            t_sg = min(t_half, deadline - time.monotonic() - 100)
            f_sg, s_sg = subgrad_momentum(best_seq, t_sg, verbose=True)
            if s_sg < best_score:
                best_score = s_sg
                best_seq = f_sg
                print(f"  SG improved: {best_score:.6f}")

    # Phase 3: Local refinement
    print("\n=== Phase 3: Local refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        t_ls = min(100.0, deadline - time.monotonic() - 50)
        print(f"  time={t_ls:.0f}s")
        f_ls, s_ls = local_refine(best_seq, t_ls, verbose=True)
        if s_ls < best_score:
            best_score = s_ls
            best_seq = f_ls
            print(f"  LS improved: {best_score:.6f}")

    # Phase 4: Final short MD polish
    print("\n=== Phase 4: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = min(40.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            f_fin, s_fin = mirror_descent(best_seq, t_final, eta=0.3, top_k=5)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin
                print(f"  Final improved: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        # Scale to a reasonable range (the score is scale-invariant, but keep values in [0, 1000])
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with evaluate_sequence
        try:
            final_score = evaluate_sequence(final_list)
        except Exception:
            final_score = best_score
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}, best_internal={best_score:.6f}")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")
        print(f"{'='*60}")
        return final_list

    # Fallback
    print("WARNING: No valid sequence found, returning default.")
    return [1.0] * 256
