"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Key insight: For non-negative sequences, max autocorrelation is always at lag 0 (= sum(a^2)),
so the objective reduces to 2*n*sum(a^2)/(sum(a))^2, minimized by the uniform sequence.
We verify this with gradient-based optimization and return the best sequence found.
"""

import numpy as np
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def gradient_step(a, lr, momentum_buf, momentum_decay):
    """
    Gradient descent step for 2*n*sum(a^2)/S^2 with non-negativity and sum constraint.
    Gradient of sum(a^2)/S^2 w.r.t. a_i = (2*a_i*S - sum(a^2)) / S^2
    We project to maintain sum and non-negativity.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a_curr = normalize(a, S_t)

    S = a_curr.sum()
    sum_sq = np.sum(a_curr ** 2)
    # Gradient of sum(a^2)/S^2: d/da_i = (2*a_i - sum_sq/S) / S
    grad = (2.0 * a_curr - sum_sq / S) / S
    # Project out mean to maintain sum constraint
    grad -= grad.mean()

    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad
    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum_buf


def soft_max_gradient_step(a, lr, momentum_buf, momentum_decay, tau):
    """
    Gradient using soft-max over autocorrelation (in case max is not at lag 0
    due to numerical issues or if my analysis is wrong).
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a_curr = normalize(a, S_t)

    c = np.convolve(a_curr, a_curr)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a_curr[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    grad -= grad.mean()

    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad
    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum_buf


def coordinate_descent_polish(a, deadline, max_rounds=20):
    """Coordinate descent: try small perturbations of each coordinate."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    patience = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:

            orig = a[i]

            S = a.sum()
            target_val = S / n  # uniform value
            candidates = [0.0, orig * 0.5, orig * 1.5, target_val,
                          (orig + target_val) / 2.0]

            best_s = best_score
            for val in candidates:
                a[i] = max(0.0, val)
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_s - 1e-12:

                    best_val = val
                a[i] = orig

            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True

            patience += 1
            if patience >= 3:
                break
        else:
            patience = 0

    return a, best_score
