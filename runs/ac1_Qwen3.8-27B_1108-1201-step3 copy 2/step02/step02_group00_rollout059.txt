"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization with
soft-max smoothing of the max operator, adaptive temperature scheduling,
and multi-start strategy. Uses log-softmax parameterization for the
simplex constraint.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / e.sum()


def log_softmax(x):
    """Numerically stable log-softmax (for initializing b from a)."""
    x_max = np.max(x)
    return x - x_max - np.log(np.sum(np.exp(x - x_max)))


def smooth_objective_and_gradient(b, t, n):
    """
    Compute smoothed objective and gradient.
    a = softmax(b), objective = 2n * softphimax(conv(a,a))
    """
    a = softmax(b)
    R = fast_convolve(a)
    
    max_R = np.max(R)
    scaled = t * (R - max_R)
    w_unnorm = np.exp(scaled)
    w_sum = np.sum(w_unnorm)
    w = w_unnorm / w_sum
    
    # Smoothed max: max_R + log(w_sum)/t
    smooth_max = max_R + np.log(w_sum) / t
    obj = 2.0 * n * smooth_max
    
    # Gradient of R w.r.t. a
    # dR_k/da[j] = 2*a[k-j] if 0 <= k-j <= n-1
    # grad_R[j] = sum_k w[k] * 2*a[k-j] = 2 * (w * a "cross-correlation")[j]
    # Using correlate: g[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    g = np.correlate(w, a, mode='valid')
    grad_a = 2.0 * g
    
    # Full gradient of smoothed obj w.r.t. a (ignoring w dependence for approximation)
    # df/da[j] = 2n * (1/t) * sum_k w[k] * dR_k/da[j] = 2n * (1/t) * 2*g[j] = 4n/t * g[j]
    grad_a_full = (4.0 * n / t) * g
    
    # Convert to b-space: df/db = a * (grad_a - sum(a * grad_a))
    grad_b = a * (grad_a_full - np.dot(a, grad_a_full))
    
    return obj, grad_b

    g = np.correlate(w, a, mode='valid')
    grad_a = (4.0 * n / t) * g

    # Transform to b-space: grad_b = a * (grad_a - sum(a*grad_a))
    grad_b = a * (grad_a - np.dot(a, grad_a))

    return obj, grad_b


def objective_wrapper(b, t, n):
    """Objective only (for scipy)."""
    obj, _ = compute_smooth_obj_grad(b, t, n)
    return obj


def gradient_wrapper(b, t, n):
    """Gradient only (for scipy)."""
    _, g = compute_smooth_obj_grad(b, t, n)
    return g


def lbfgs_refine(a_init, t_values, n, max_iter_per_t=300, gtol=1e-10):
    """
    Run L-BFGS with increasing temperature schedule.
    a_init: initial point on simplex (sum=1, all >= 0)
    t_values: list of temperatures (increasing = sharper peak)
    Returns (best_a, best_hard_obj).
    """
    a = np.maximum(a_init.copy(), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * np.max(fast_convolve(a))

    for t in t_values:
        # Initialize b from current a (b = log(a) works since softmax(log(a)) ~ a for sum(a)=1)
        b = np.log(np.maximum(a, 1e-300))

        try:
            res = minimize(
                objective_wrapper, b, args=(t, n),
                jac=gradient_wrapper, method='L-BFGS-B',
                options={'maxiter': max_iter_per_t, 'gtol': gtol, 'maxls': 50}
            )
            a_new = softmax_fn(res.x)
        except Exception:
            a_new = a.copy()

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        else:
            a_new = np.ones(n) / n
        a = a_new

        # Evaluate with hard max
        obj_new = 2.0 * n * np.max(fast_convolve(a))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

    return best_a, best_obj


def subgradient_fast(a0, time_limit=10.0, top_k=30):
    """Fast subgradient descent on simplex for quick refinement."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    lr = 0.08
    momentum = 0.92
    temp = 0.005
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        max_c = conv.max()
        scaled = temp * (conv - max_c)
        w = np.exp(scaled)
        w_sum = w.sum()
        w_peaks = w[peaks] / w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w_peaks[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w_peaks[i] * 2.0 * a[k - idx]

        v = momentum * v - lr * sg
        a = np.maximum(a + v, 0.0)
        s = a.sum()
        if s > 1e-15:
            a /= s
        else:
            a = np.ones(n) / n

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        lr *= 0.999
        temp *= 1.001

    return best_a, best_obj


def upsample(a, n_new):
    """Upsample a sequence to new length using linear interpolation."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    a = a / s
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0)
    rs = result.sum()
    if rs > 1e-15:
        result /= rs
    else:
        result = np.ones(n_new) / n_new
    return result


def make_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n) / n))

    # Edge-concentrated variants
    for ew in [0.2, 0.35, 0.5]:
        ae = np.ones(n) * 0.5
        ae[0] = ew
        ae[-1] = ew
        ae = np.maximum(ae, 0.01)
        ae /= ae.sum()
        inits.append((f'edge_{ew}', ae))

    # Gaussian centered
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n / 2) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append((f'gauss_{sf}', g))

    # Bimodal (two peaks at 1/4 and 3/4)
    bi = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + \
         np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    bi /= bi.sum()
    inits.append(('bimodal', bi))

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits.append(('ramp', ramp))

    # Random
    inits.append(('random1', rng.dirichlet(np.ones(n) * 0.3)))
    inits.append(('random2', rng.dirichlet(np.ones(n) * 1.0)))

    # From parent
    if parent is not None and len(parent) > 0:
        if n == len(parent):
            inits.append(('parent', parent.copy()))
        else:
            inits.append(('parent_interp', upsample(parent, n)))

    return inits


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        score = float(evaluate_sequence(a_list))
        return score
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization using L-BFGS with soft-max
    smoothing, subgradient refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    # === Phase 1: L-BFGS refinement of parent (high priority) ===
    print("=== Phase 1: L-BFGS on parent ===")
    t1_start = time.monotonic()
    if parent is not None:
        n_p = len(parent)
        # Aggressive temperature schedule
        t_sched = [5, 15, 50, 150, 500, 2000, 8000, 30000, 100000]
        a_opt, obj = lbfgs_refine(parent, t_sched, n_p, max_iter_per_t=400, gtol=1e-12)
        score = evaluate_safe(a_opt)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Parent L-BFGS: score={score:.8f}")
        else:
            print(f"  Parent L-BFGS: score={score:.8f} (no improvement)")

    # Quick subgradient pass too
    if parent is not None:
        t_rem = total_deadline - time.monotonic() - 60
        if t_rem > 5:
            a_sg, _ = subgradient_fast(parent, time_limit=min(15, t_rem * 0.1), top_k=40)
            score = evaluate_safe(a_sg)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_sg.copy()
                print(f"  Parent subgrad: score={score:.8f}")

    # === Phase 2: Multi-length exploration ===
    print("=== Phase 2: Multi-length exploration ===")
    phase2_deadline = total_deadline - 120  # Reserve 120s for phase 3

    lengths = [128, 192, 256, 320, 384, 512]
    # If parent exists, prioritize its length
    if parent is not None and len(parent) in lengths:
        lengths.remove(len(parent))
        lengths.insert(0, len(parent))

    candidates = []
    for n in lengths:
        if time.monotonic() > phase2_deadline - 30:
            break

        inits = make_initializations(n, parent, rng)
        # Limit number of inits to try
        max_inits = 8
        inits = inits[:max_inits]

        time_per_init = max(3.0, (phase2_deadline - time.monotonic() - 20) /
                            (len(lengths) * max_inits))

        for name, a0 in inits:
            if time.monotonic() > phase2_deadline - 20:
                break

            # L-BFGS with moderate schedule
            t_sched = [10, 50, 200, 1000, 5000]
            a_opt, obj = lbfgs_refine(a0, t_sched, n, max_iter_per_t=250, gtol=1e-9)
            score = evaluate_safe(a_opt)
            candidates.append((score, n, name, a_opt.copy()))

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  n={n} {name}: score={score:.8f}")

    # Sort candidates
    candidates.sort(key=lambda x: x[0] if np.isfinite(x[0]) else float('inf'))
    print(f"  Phase 2 best candidates: {[(f'{s:.6f}', n, nm) for s, n, nm, _ in candidates[:5]]}")

    # === Phase 3: Deep refinement of top candidates ===
    print("=== Phase 3: Deep refinement ===")
    top_candidates = candidates[:6]
    # Always include current best
    if best_a is not None:
        top_candidates.append((best_score, len(best_a), 'current_best', best_a.copy()))

    # Remove duplicates (same n and similar score)
    seen = set()
    unique_candidates = []
    for sc, n, nm, a in top_candidates:
        key = n
        if key not in seen:
            unique_candidates.append((sc, n, nm, a))
            seen.add(key)
        if len(unique_candidates) >= 8:
            break

    phase3_deadline = total_deadline - 15
    time_per_deep = max(10.0, (phase3_deadline - time.monotonic()) / max(1, len(unique_candidates)))

    for sc, n, nm, a0 in unique_candidates:
        if time.monotonic() > phase3_deadline - 10:
            break

        # Deep L-BFGS with fine temperature schedule
        t_sched_fine = [50, 200, 800, 3000, 12000, 50000, 200000]
        iters = int(max(200, time_per_deep * 80))
        a_opt, obj = lbfgs_refine(a0, t_sched_fine, n, max_iter_per_t=iters, gtol=1e-12)
        score = evaluate_safe(a_opt)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Deep {nm} n={n}: score={score:.8f}")

        # Subgradient polish
        if time.monotonic() < phase3_deadline - 5:
            a_sg, _ = subgradient_fast(a_opt, time_limit=min(8, phase3_deadline - time.monotonic() - 5), top_k=50)
            score = evaluate_safe(a_sg)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_sg.copy()
                print(f"  SG polish {nm} n={n}: score={score:.8f}")

    # === Phase 4: Final micro-optimization ===
    print("=== Phase 4: Final micro-optimization ===")
    if best_a is not None:
        n_best = len(best_a)
        # Very high temperature L-BFGS (essentially hard max)
        t_very_high = [100000, 500000, 2000000]
        a_opt, obj = lbfgs_refine(best_a, t_very_high, n_best, max_iter_per_t=500, gtol=1e-13)
        score = evaluate_safe(a_opt)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Micro L-BFGS: score={score:.8f}")

        # Small perturbations + re-optimize
        for perturb_scale in [0.0005, 0.001, 0.002]:
            if time.monotonic() > total_deadline - 5:
                break
            noise = rng.standard_normal(n_best) * perturb_scale
            a_pert = np.maximum(best_a + noise, 1e-15)
            a_pert /= a_pert.sum()
            a_opt, obj = lbfgs_refine(a_pert, [5000, 50000, 500000], n_best,
                                       max_iter_per_t=300, gtol=1e-12)
            score = evaluate_safe(a_opt)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  Perturb {perturb_scale}: score={score:.8f}")

        # Coordinate descent style: try zeroing small elements
        if time.monotonic() < total_deadline - 5:
            a_sparse = best_a.copy()
            # Zero out elements below a threshold
            threshold = np.sort(a_sparse)[:max(1, n_best // 10)]
            a_sparse[a_sparse < threshold] = 0
            a_sparse = np.maximum(a_sparse, 0)
            s = a_sparse.sum()
            if s > 0.01:
                a_sparse /= s
                a_opt, obj = lbfgs_refine(a_sparse, [1000, 10000, 100000], n_best,
                                           max_iter_per_t=200, gtol=1e-10)
                score = evaluate_safe(a_opt)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  Sparse init: score={score:.8f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(256) / 256
        best_score = evaluate_safe(best_a)

    # Final verification
    final_score = evaluate_safe(best_a)
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_a)}")
    return [float(x) for x in best_a]
