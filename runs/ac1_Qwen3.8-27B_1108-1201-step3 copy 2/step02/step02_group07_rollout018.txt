import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def smooth_score(a, tau):
    """Smooth approximation using log-sum-exp."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.zeros(n)
    c = fft_convolve(a, a)
    # log-sum-exp for numerical stability
    c_max = np.max(c)
    lse = c_max + np.log(np.sum(np.exp((c - c_max) / tau))) * tau
    score = 2.0 * n * lse / (S * S)
    
    # Gradient
    # d(score)/da_j = (2n/S^2) * (sum_k exp(tau*c_k) * dc_k/da_j / sum_k exp(tau*c_k)) - 2*score/S
    # dc_k/da_j = a[k-j] + a[j-k] (for valid indices)
    weights = np.exp((c - c_max) / tau)
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):
            i1 = k - j
            i2 = j - k
            val = 0.0
            if 0 <= i1 < n:
                val += a[i1]
            if 0 <= i2 < n:
                val += a[i2]
            g[j] += weights[k] * val
    
    g = (2.0 * n / (S * S)) * g - 2.0 * score / S
    
    return score, g

4. Update: a_new[a_j] = a_j * sqrt(c_ideal / c_j) for each j

This is the Euclidean projection onto the simplex!

1. Compute gradient of f at a
2. Take step: a_new = a - lr * grad

"""
Hybrid optimization for minimizing peak aperiodic autocorrelation.
Uses smooth log-sum-exp gradient descent with simplex projection,
combined with LP direction finding and multi-start exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_objective_and_grad(a, tau):
    """
    Compute smooth approximation of score and its gradient.
    Uses log-sum-exp to approximate max.
    Returns (score, gradient) where gradient is w.r.t. a.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(a, a)
    c_max = np.max(c)
    
    # Log-sum-exp
    shifted = (c - c_max) / tau
    log_sum_exp = c_max + tau * np.log(np.sum(np.exp(shifted)))
    
    score = 2.0 * n * log_sum_exp / (S * S)
    
    # Weights for each convolution element
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Gradient of log_sum_exp w.r.t. a_j:
    # d(log_sum_exp)/da_j = sum_k weights[k] * dc_k/da_j
    # dc_k/da_j = a_{k-j} + a_{j-k} (for valid indices)
    
    g = np.zeros(n)
    for j in range(n):
        # Term 1: sum_k w_k * a_{k-j} for valid k-j
        # k-j = m, valid when 0 <= m < n, i.e., j <= k < j+n
        # sum_{m=0}^{n-1} w_{m+j} * a_m
        t1 = 0.0
        for m in range(n):
            idx = m + j
            if idx < 2 * n - 1:
                t1 += weights[idx] * a[m]
        
        # Term 2: sum_k w_k * a_{j-k} for valid j-k
        # j-k = m, valid when 0 <= m < n, i.e., j-n < k <= j
        # sum_{m=0}^{min(j, n-1)} w_{j-m} * a_m
        t2 = 0.0
        m_max = min(j, n - 1)
        for m in range(m_max + 1):
            idx = j - m
            if idx >= 0:
                t2 += weights[idx] * a[m]
        
        g[j] = t1 + t2
    
    # d(score)/da_j = (2n/S^2) * g[j] - 2*score/S
    score_grad = (2.0 * n / (S * S)) * g - 2.0 * score / S
    
    return score, score_grad


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def gradient_descent_smooth(f_init, time_limit, tau_init=1.0, tau_min=0.1, lr_init=0.01, verbose=True):
    """
    Gradient descent on smooth approximation with simplex constraint.
    Uses adaptive learning rate and tau annealing.
    """
    n = len(f_init)
    f = simplex_project(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        s_smooth, g = smooth_objective_and_grad(f, tau)
        
        # Adaptive learning rate: scale by problem size
        f_new = f - lr * g
        f_new = simplex_project(f_new)
        
        s_new = score_fast(f_new)
        
        if s_new < best_score:
            best_score = s_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        it += 1
        
        # Adaptive tau and lr
        if it % 50 == 0:
            tau = max(tau_min, tau * 0.95)
            if no_improve > 30:
                lr *= 0.7
                no_improve = 0
            elif no_improve > 10:
                lr *= 0.9
            else:
                lr = min(lr * 1.1, lr_init * 2)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [grad {elapsed:6.1f}s] it={it} smooth={s_smooth:.6f} true={s_new:.6f} best={best_score:.6f} tau={tau:.3f} lr={lr:.5f}")
    
    return best_f, best_score


def build_A_mat(f):
    """Build matrix A where A@b = conv(f,b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=3.0):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0. Returns b or None."""
    n = len(f)
    A = build_A_mat(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def line_search(f, b, S_f, max_iter=20):
    """
    Find optimal step size t in [0, 1] for f_new = (1-t)*f + t*b.
    Both f and b should be non-negative. We search for t that minimizes score.
    """
    best_t = 0.0
    best_s = score_fast(f)
    
    # Try a grid of step sizes
    ts = np.logspace(-3, 0, 15)  # 0.001 to 1.0
    for t in ts:
        t = min(t, 1.0)
        f_cand = (1.0 - t) * f + t * b
        s_cand = score_fast(f_cand)
        if s_cand < best_s:
            best_s = s_cand
            best_t = t
    
    # Refine around best
    if best_t > 0:
        lo = max(0, best_t * 0.5)
        hi = min(1.0, best_t * 1.5)
        for t in np.linspace(lo, hi, 10):
            f_cand = (1.0 - t) * f + t * b
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t
    
    return best_t, best_s


def hybrid_optimize(f_init, time_limit, verbose=True):
    """
    Combined LP + gradient optimization.
    Alternates between LP direction finding and gradient descent.
    """
    n = len(f_init)
    f = simplex_project(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 5.0:
            break
        
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        
        # Alternate between LP and gradient
        if remaining > 30:
            # LP step
            c = fft_convolve(f, f)
            M = float(np.max(c))
            b = lp_direction(f, M, timeout=min(3.0, remaining * 0.1))
            
            if b is not None and np.sum(b) > 1e-12:
                Sb = float(np.sum(b))
                b_norm = b / Sb  # normalize to sum=1
                t_opt, s_opt = line_search(f, b_norm, float(np.sum(f)))
                
                if s_opt < best_score:
                    best_score = s_opt
                    best_f = ((1 - t_opt) * f + t_opt * b_norm).copy()
                    f = best_f.copy()
                    if verbose:
                        print(f"    [LP {elapsed:6.1f}s] score={s_opt:.6f}")
                else:
                    # LP didn't help, try gradient
                    f, s_g = gradient_descent_smooth(f, min(20.0, remaining * 0.3), 
                                                      tau_init=1.0, tau_min=0.5,
                                                      lr_init=0.005, verbose=False)
                    if s_g < best_score:
                        best_score = s_g
                        best_f = f.copy()
                        if verbose:
                            print(f"    [Grad {elapsed:6.1f}s] score={s_g:.6f}")
            else:
                # LP failed, use gradient
                f, s_g = gradient_descent_smooth(f, min(20.0, remaining * 0.3),
                                                  tau_init=1.0, tau_min=0.5,
                                                  lr_init=0.005, verbose=False)
                if s_g < best_score:
                    best_score = s_g
                    best_f = f.copy()
        else:
            # Final gradient polish
            f, s_g = gradient_descent_smooth(f, remaining - 5.0,
                                              tau_init=0.5, tau_min=0.2,
                                              lr_init=0.003, verbose=False)
            if s_g < best_score:
                best_score = s_g
                best_f = f.copy()
    
    return best_f, best_score


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng):
    a = 0.5 + rng.random(n)
    return a


def make_edge_emphasis_init(n, rng):
    """Sequence with emphasis at edges (inverted window)."""
    t = np.arange(n, dtype=np.float64)
    normalized_t = 2.0 * t / max(1, n - 1) - 1.0  # -1 to 1
    a = 1.0 + 0.5 * normalized_t ** 2
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_cos_emphasis_init(n, rng):
    """Cosine-based edge emphasis."""
    t = np.arange(n, dtype=np.float64)
    normalized_t = 2.0 * t / max(1, n - 1) - 1.0
    a = 1.0 - 0.5 * np.cos(np.pi * normalized_t)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def local_search_polish(f_init, time_limit, verbose=True):
    """
    Local search with targeted perturbations based on convolution peaks.
    """
    n = len(f_init)
    f = simplex_project(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))
        
        # Find elements that contribute most to the peak
        contribs = np.zeros(n)
        for j in range(n):
            i = peak_idx - j
            if 0 <= i < n:
                contribs[j] = f[j] * f[i]
        
        candidate = f.copy()
        strategy = rng.random()
        
        if strategy < 0.4:
            # Reduce top contributors to peak, increase low contributors
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            bot_idx = np.argsort(contribs)[:top_k]
            
            scale_down = np.exp(rng.normal(-0.05, 0.03, size=top_k))
            scale_up = np.exp(rng.normal(0.05, 0.03, size=top_k))
            
            candidate[top_idx] *= scale_down
            candidate[bot_idx] *= scale_up
        elif strategy < 0.7:
            # Smooth perturbation
            candidate *= np.exp(rng.normal(0, 0.02, size=n))
        else:
            # Targeted: reduce the element at peak position most
            if 0 <= peak_idx < n:
                candidate[peak_idx] *= np.exp(rng.normal(-0.08, 0.03))
            elif 0 <= peak_idx - n + 1 < n:
                candidate[peak_idx - n + 1] *= np.exp(rng.normal(-0.08, 0.03))
            # And increase some random other elements
            inc_idx = rng.choice(n, size=5, replace=False)
            candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=5))
        
        candidate = simplex_project(candidate)
        s_new = score_fast(candidate)
        
        if s_new < best_score:
            best_score = s_new
            best_f = candidate.copy()
            f = candidate
            if verbose:
                print(f"    [LS {time.monotonic()-t0:6.1f}s] score={s_new:.6f}")
        else:
            # Accept with some probability (simulated annealing)
            if rng.random() < np.exp(-(s_new - best_score) / 0.01):
                f = candidate
    
    return best_f, best_score

"""
Hybrid LP + smooth gradient optimization for minimizing peak aperiodic autocorrelation
of non-negative sequences. Uses FFT-based convolution, efficient gradient computation
via cross-correlation, simplex projection, and multi-start exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_score_and_grad(a, tau):
    """
    Compute smooth log-sum-exp approximation of the score and its gradient.
    a: sequence (will be clipped to non-negative)
    tau: temperature parameter (smaller = sharper max approximation)
    Returns (smooth_score, gradient_array)
    """
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(a, a)
    c_max = np.max(c)
    
    # Log-sum-exp (numerically stable)
    shifted = (c - c_max) / tau
    log_sum_exp = c_max + tau * np.log(np.sum(np.exp(shifted)))
    
    score = 2.0 * n * log_sum_exp / (S * S)
    
    # Softmax weights
    w = np.exp(shifted)
    w /= np.sum(w)
    
    # Gradient: g[j] = 2 * sum_m w[m+j] * a[m]
    # Compute via FFT cross-correlation
    N = 1
    while N < 2 * (2 * n - 1) - 1:
        N <<= 1
    # Actually we need N >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n - 2
    while N < 3 * n - 1:
        N <<= 1
    
    W = fft(w, N)
    A = fft(a, N)
    # Cross-correlation: ifft(W * conj(A))
    cross = np.real(ifft(W * np.conj(A)))
    
    # g[j] = 2 * cross[n-1+j] for j = 0, ..., n-1
    g = 2.0 * cross[n - 1: 2 * n - 1]
    
    # Full score gradient
    # d(score)/da_j = (2n/S^2) * g[j] - 2*score/S
    score_grad = (2.0 * n / (S * S)) * g - 2.0 * score / S
    
    return score, score_grad


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_descent_smooth(f_init, time_limit, tau_start=1.0, tau_end=0.05, lr=0.005, verbose=False):
    """
    Gradient descent on smooth surrogate with simplex constraint.
    Uses adaptive learning rate and tau annealing.
    """
    n = len(f_init)
    f = simplex_project(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    tau = tau_start
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break
        
        s_smooth, g = smooth_score_and_grad(f, tau)
        
        f_new = f - lr * g
        f_new = simplex_project(f_new)
        
        s_new = score_fast(f_new)
        
        if s_new < best_score:
            best_score = s_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        it += 1
        
        # Anneal tau
        progress = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Adapt lr
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
        elif no_improve > 20:
            lr *= 0.95
    
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      [{elapsed:5.1f}s] it={it:5d} score={s_new:.6f} best={best_score:.6f} tau={tau:.3f} lr={lr:.5f}")
    
    return best_f, best_score


def build_A_mat(f):
    """Build matrix A where A@b = conv(f,b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=3.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)

"""
Hybrid LP + smooth-gradient optimization for minimizing peak aperiodic autocorrelation
of non-negative sequences. Uses FFT-based convolution, efficient gradient via cross-correlation,
simplex projection, adaptive LP+GD alternation, and multi-start exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft as npfft, ifft as npifft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_score_and_grad(a, tau):
    """
    Smooth log-sum-exp approximation of score and its gradient.
    Returns (smooth_score, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, np.zeros(n)

    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    log_sum_exp = c_max + tau * np.log(np.sum(np.exp(shifted)))
    score = 2.0 * n * log_sum_exp / (S * S)

    # Softmax weights over convolution values
    w = np.exp(shifted)
    w /= np.sum(w)

    # Gradient: g[j] = 2 * sum_{m} w[m+j] * a[m]  (cross-correlation)
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    W = npfft(w, N)
    A = npfft(a, N)
    cross = np.real(npifft(W * np.conj(A)))
    g = 2.0 * cross[n - 1: 2 * n - 1]

    score_grad = (2.0 * n / (S * S)) * g - 2.0 * score / S
    return score, score_grad


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_descent_smooth(f_init, time_limit, tau_start=1.0, tau_end=0.05, lr=0.005, verbose=False):
    """Gradient descent on smooth surrogate with simplex constraint."""
    n = len(f_init)
    f = simplex_project(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.3:
            break

        s_smooth, g = smooth_score_and_grad(f, tau_start * (tau_end / tau_start) ** ((time.monotonic() - t0) / max(time_limit, 1.0)))

        f_new = f - lr * g
        f_new = simplex_project(f_new)
        s_new = score_fast(f_new)

        if s_new < best_score:
            best_score = s_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        it += 1

        if no_improve > 80:
            lr *= 0.7
            no_improve = 0
        elif no_improve > 40:
            lr *= 0.9
        elif no_improve == 0:
            lr = min(lr * 1.05, 0.05)

        if verbose and it % 500 == 0:
            print(f"      [{time.monotonic()-t0:5.1f}s] it={it} score={s_new:.6f} best={best_score:.6f} lr={lr:.5f}")

    return best_f, best_score


def build_A_mat(f):
    """Build matrix A where A@b = conv(f,b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=2.5):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M*1, b >= 0."""
    n = len(f)
    A = build_A_mat(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def line_search(f, b, n_points=15):
    """Find optimal step size t for (1-t)*f + t*b."""
    best_t = 0.0
    best_s = score_fast(f)

    for t in np.linspace(0.02, 1.0, n_points):
        f_cand = (1.0 - t) * f + t * b
        s_cand = score_fast(f_cand)
        if s_cand < best_s:
            best_s = s_cand
            best_t = t

    if 0 < best_t < 1:
        lo = max(0.01, best_t * 0.5)
        hi = min(1.0, best_t * 1.5)
        for t in np.linspace(lo, hi, 12):
            f_cand = (1.0 - t) * f + t * b
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t

    return best_t, best_s


def combined_optimize(f_init, time_limit, verbose=True):
    """
    Combined LP + gradient optimization with adaptive chunking.
    """
    n = len(f_init)
    f = simplex_project(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    lp_chunk = 12.0
    grad_chunk = 12.0
    consecutive_no_improve = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            f, s_g = grad_descent_smooth(f, remaining - 2.0, tau_start=0.3, tau_end=0.05, lr=0.003)
            if s_g < best_score:
                best_score = s_g
                best_f = f.copy()
            break

        # --- LP Phase ---
        lp_end = time.monotonic() + min(lp_chunk, remaining * 0.5)
        lp_improved = False

        while time.monotonic() < lp_end:
            c = fft_convolve(f, f)
            M = float(np.max(c))

            if b is None:

            Sb = float(np.sum(b))
            if Sb < 1e-12:
                break
            b = b / Sb

            t_opt, s_opt = line_search(f, b)

                break

        grad_chunk_actual = min(grad_chunk, remaining * 0.5 - 1.0)
        if grad_chunk_actual > 3:
            f, s_g = grad_descent_smooth(f, grad_chunk_actual, tau_start=0.8, tau_end=0.1,
