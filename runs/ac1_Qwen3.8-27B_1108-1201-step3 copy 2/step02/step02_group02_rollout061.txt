def resample_sequence(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq), endpoint=True)
    x_new = np.linspace(0, 1, new_n, endpoint=True)
    return np.interp(x_new, x_old, seq)


def make_random_init(n, rng):
    a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):

    a = 1.0 + 0.5 * np.cos(np.pi * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

"""
Coordinate descent optimization for minimizing peak autocorrelation of
non-negative step functions. Uses efficient 1D sub-optimization at each
coordinate with grid search and refinement, incremental convolution updates,
multiple starting points, different sequence lengths, and perturbation restarts.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def coord_descent(a_init, time_limit, rng, verbose=False):
    """
    Coordinate descent with 1D grid search for each coordinate.
    Maintains convolution incrementally for speed.
    """
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
        S = n
    c = fft_convolve(a, a)
    current_score = 2.0 * n * np.max(c) / (S * S)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    sweep = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            ai = a[i]
            # Affected range in c: [i, i+n-1], always valid since 0<=i<n
            c_seg = c[i:i + n]
            # Outside max
            out_max = -1e18
            if i > 0:
                out_max = max(out_max, np.max(c[:i]))
            if i + n < 2 * n - 1:
                out_max = max(out_max, np.max(c[i + n:]))

            k2i = 2 * i

            # Determine search bounds
            d_lo = -ai
            d_hi = max(1.5 * ai, S / (3.0 * n))
            if ai < 1e-10:
                d_lo = 0.0
                d_hi = max(0.01, S / (4.0 * n))

            if d_hi <= d_lo + 1e-15:
                continue

            best_d = 0.0
            best_f = current_score

            # Coarse grid: 35 points

            for d in np.linspace(d_lo, d_hi, nd_c):
                seg = c_seg + 2.0 * d * a
                pm = np.max(seg)
                qv = c[k2i] + 2.0 * d * ai + d * d
                if qv > pm:

                pk = pm if pm > out_max else out_max
                Sn = S + d
                fv = np.inf if Sn < 0.01 else 2.0 * n * pk / (Sn * Sn)
                if fv < best_f:
                    best_f = fv
                    best_d = d

            # Fine grid: 25 points around best

                f_lo = max(d_lo, best_d - margin)
                f_hi = min(d_hi, best_d + margin)

                    seg = c_seg + 2.0 * d * a
                    pm = np.max(seg)
                    qv = c[k2i] + 2.0 * d * ai + d * d

                    pk = pm if pm > out_max else out_max
                    Sn = S + d
                    fv = np.inf if Sn < 0.01 else 2.0 * n * pk / (Sn * Sn)

                        best_f = fv
                        best_d = d

                seg = c_seg + 2.0 * d * a
                pm = np.max(seg)
                qv = c[k2i] + 2.0 * d * ai + d * d
                if qv > pm:
                    pm = qv
                pk = pm if pm > out_max else out_max
                Sn = S + d

def resample_sequence(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq), endpoint=True)
    x_new = np.linspace(0, 1, new_n, endpoint=True)
    return np.interp(x_new, x_old, seq)


def make_random_init(n, rng):
    """Random initialization with bounded noise."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(np.pi * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_anti_peak_init(n, rng):
    """Initialize with structure that avoids peaks in autocorrelation."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    # Multi-tone with random frequencies
    for k in range(1, 8):
        amp = 0.2 / k
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def targeted_perturbation(a, rng):
    """Perturb by specifically targeting peak-contributing coordinates."""
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))

    # Find coordinates contributing to the peak
    j_start = max(0, peak_idx - n + 1)
    j_end = min(n - 1, peak_idx)
    contrib = np.zeros(n)
    for j in range(j_start, j_end + 1):
        contrib[j] = a[j] * a[peak_idx - j]

    # Reduce top contributors
    top_k = max(3, n // 20)
    top_idx = np.argsort(contrib)[-top_k:]
    a[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))

    # Increase some non-contributing coordinates
    non_top = np.setdiff1d(np.arange(n), top_idx)
    if len(non_top) > 5:
        n_inc = min(10, len(non_top) // 3)
        inc_idx = rng.choice(non_top, size=n_inc, replace=False)
        a[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))

    return np.clip(a, 0.0, None)

            print(f"  *** New best: {best_score:.8f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)}, time={time.monotonic()-t_global:.1f}s ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
