            k = int(k)
            i_vals = np.arange(n)
            j_vals = k - i_vals
            valid = (j_vals >= 0) & (j_vals < n)
            sg[valid] += f[j_vals[valid]]
        sg /= len(top_k)

        # Update with projection
        f_new = project_simplex(f - eta * sg)

        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < obj - 1e-15:
            f = f_new
            eta = min(eta * 1.02, 0.1)
        else:
            # Try smaller step
            for factor in [0.5, 0.1]:
                f_try = project_simplex(f - eta * factor * sg)
                conv_try = fast_convolve(f_try)
                if 2.0 * n * conv_try.max() < obj - 1e-15:
                    f = f_try
                    eta *= factor
                    break
            else:
                eta *= 0.8

    return best_f, best_obj


def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3):
    """Mirror descent polishing on the probability simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f = f / f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Softmax weighting of convolution peaks
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = w / w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # sg[j] += w[k] * f[k-j] for j in [j_start, j_end]
            idx = k - np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] += w[k] * f[idx]

        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f = f / s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.75
                if eta < 0.005:
                    t_param = min(t_param * 1.5, 500.0)
                    eta = 0.2
            if patience > 200:
                break

    return best_f, best_obj


def peak_shave(f0, time_limit, max_moves=50000):
    """Randomized local search that targets the peak of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    moves = 0
    rng = np.random.default_rng(12345)
    epsilon = 1.0 / n  # Amount to transfer

    while time.monotonic() - t0 < time_limit and moves < max_moves:
        moves += 1

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()

        k_star = int(np.argmax(conv))

        # Find the pair with maximum product at the peak
        i_vals = np.arange(n)
        j_vals = k_star - i_vals
        valid = (j_vals >= 0) & (j_vals < n)
        if not np.any(valid):
            continue
        valid_idx = np.where(valid)[0]
        products = f[valid_idx] * f[j_vals[valid_idx]]
        if products.max() < 1e-15:
            continue

        # Pick among top contributors
        top_m = min(10, len(products))
        top_positions = valid_idx[np.argsort(products)[-top_m:]]

        # Try several random moves
        improved_this_iter = False
        for _ in range(5):
            i_src = int(rng.choice(top_positions))
            j_comp = k_star - i_src
            # Source is the larger of f[i_src], f[j_comp]
            if f[i_src] >= f[j_comp]:
                src = i_src
                dst_pool = np.setdiff1d(np.arange(n), [src, j_comp])
            else:
                src = j_comp
                dst_pool = np.setdiff1d(np.arange(n), [i_src, src])
            if len(dst_pool) == 0:
                continue
            dst = int(rng.choice(dst_pool))

            # Transfer mass
            amount = min(epsilon * rng.uniform(0.3, 1.0), f[src] * 0.5)
            if amount < 1e-10:
                continue
            f_trial = f.copy()
            f_trial[src] -= amount
            f_trial[dst] += amount
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-12:
                continue
            f_trial = f_trial / s

            conv_trial = fast_convolve(f_trial)
            obj_trial = 2.0 * n * conv_trial.max()
            if obj_trial < obj - 1e-15:
                f = f_trial
                improved_this_iter = True
                break

        if not improved_this_iter:
            # Try increasing the smaller element at the peak
            # (counterintuitive but sometimes helps by shifting the peak)
            pass

        if moves % 5000 == 0:
            print(f"    PeakShave move {moves}, obj={best_obj:.8f}")

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    # Various smooth shapes
    for sf in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.3 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Reverse ramp
    inits['rramp'] = r[::-1] / r.sum()

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[n//3:n//3 + w] = 1.0
    inits['bi'] = bi / bi.sum()

    # Cosine window (shifted)
    cb = 0.5 * (1 + np.cos(2 * np.pi * (x - n//4) / n))
    cb = np.maximum(cb, 0.01)
    inits['cos'] = cb / cb.sum()

    # Exponential decay
    ed = np.exp(-x / (n * 0.2))
    inits['exp'] = ed / ed.sum()

    # Triangular
    tri = np.maximum(1.0 - 2 * np.abs(x - 0.3 * n) / n, 0.01)
    inits['tri'] = tri / tri.sum()

    # M-sequence inspired (pseudo-random with structure)
    m_seq = np.zeros(n)
    m_seq[rng.choice(n, n//2, replace=False)] = 1.0
    # Smooth it
    kernel = np.ones(5) / 5
    m_seq = np.convolve(m_seq, kernel, mode='same')
    m_seq = np.maximum(m_seq, 0.01)
    inits['mseq'] = m_seq / m_seq.sum()

    # Chirp-like (quadratic phase -> amplitude modulation)
    chirp = np.abs(np.exp(1j * np.pi * (x / n) ** 2)) * np.exp(-x / (n*0.5))
    chirp = np.maximum(chirp, 0.01)
    inits['chirp'] = chirp / chirp.sum()

    # Random Dirichlet
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr

    # Power law
    pl = (x + 1) ** (-0.5)
    inits['pow'] = pl / pl.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: multi-phase optimization with diverse methods."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 15.0

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    def update_best(f, label=""):
        nonlocal best_seq, best_score
        seq = f.tolist()
        sc = float(evaluate_sequence(seq))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()
            print(f"  NEW BEST [{label}]: score={sc:.6f}, n={len(f)}")
        return sc

    # Phase 1: Diverse scan across lengths and initializations
    lengths = [256, 384, 512, 640, 768]
    all_results = []

    # Allocate time for Phase 1
    phase1_time = 400.0
    phase1_deadline = time.monotonic() + phase1_time

    print("=== Phase 1: Diverse LP Scan ===")
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  Length n={n}")
        inits = gen_inits(n, rng)

        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()

        # Prioritize: parent first, then diverse
        ordered_keys = []
        if 'par' in inits:
            ordered_keys.append('par')
        # Sort remaining by expected quality
        for k in inits.keys():
            if k not in ordered_keys:
                ordered_keys.append(k)

        n_configs = len(ordered_keys)
        remaining = phase1_deadline - time.monotonic()
        time_per = max(5.0, remaining / max(1, n_configs * len(lengths)))

        for name in ordered_keys:
            if time.monotonic() > phase1_deadline:
                break
            a0 = inits[name]
            ta = min(time_per, phase1_deadline - time.monotonic())
            if ta < 3:
                break

            f_opt, obj_opt = lp_optimize(a0, time_limit=ta, use_perturbation=False)
            sc = update_best(f_opt, f"LP n={n} {name}")
            if np.isfinite(sc):
                all_results.append((sc, n, name, f_opt))

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep LP optimization on top results
    all_results.sort(key=lambda x: x[0])
    top = all_results[:10]

    phase2_deadline = time.monotonic() + 700.0
    if top and phase2_deadline < deadline:
        # Also add the parent if not already in top
        if parent is not None:
            parent_sc = float('inf')
            for r in top:
                if r[2] == 'par':
                    parent_sc = r[0]
                    break
            if parent_sc == float('inf') and parent.size in [256, 384, 512, 640, 768]:
                pn = parent.size
                pp = np.maximum(parent, 0.0)
                pp = pp / pp.sum()
                top.append((parent_sc, pn, 'par_deep', pp))

        print(f"=== Phase 2: Deep LP ({len(top)} configs) ===")
        n_deep = len(top)
        time_each = max(20.0, min(150.0, (phase2_deadline - time.monotonic()) / n_deep))

        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > phase2_deadline:
                break
            ta = min(time_each, phase2_deadline - time.monotonic())
            if ta < 10:
                break

            print(f"  Deep [{idx}] n={n} {name} (start={sc:.6f})")
            f2, obj2 = lp_optimize(f, time_limit=ta, use_perturbation=True)
            sc2 = update_best(f2, f"DeepLP[{idx}]")

            if sc2 < sc:
                all_results = [r if r[0] != sc or r[2] != name else (sc2, n, name, f2)
                               for r in all_results]

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Multi-method polishing
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"=== Phase 3: Polishing ===")
        f_cur = best_seq.copy()
        n = len(f_cur)
        phase3_deadline = deadline - 30

        # Try multiple polishing methods in sequence
        polish_methods = [
            ("LP_polish", lambda f, t: lp_optimize(f, t, use_perturbation=True)),
            ("Mirror_T50", lambda f, t: mirror_polish(f, t, t_param=50.0, eta0=0.3)),
            ("Mirror_T20", lambda f, t: mirror_polish(f, t, t_param=20.0, eta0=0.5)),
            ("Mirror_T100", lambda f, t: mirror_polish(f, t, t_param=100.0, eta0=0.15)),
            ("SubGrad", lambda f, t: subgradient_polish(f, t, eta0=0.03)),
            ("PeakShave", lambda f, t: peak_shave(f, t)),
            ("Mirror_T300", lambda f, t: mirror_polish(f, t, t_param=300.0, eta0=0.08)),
        ]

        remaining = phase3_deadline - time.monotonic()
        n_methods = len(polish_methods)
        time_per_polish = max(15.0, remaining / n_methods)

        for meth_name, meth_func in polish_methods:
            if time.monotonic() > phase3_deadline:
                break
            ta = min(time_per_polish, phase3_deadline - time.monotonic())
            if ta < 10:
                break

            f_new, obj_new = meth_func(f_cur, ta)
            sc = update_best(f_new, f"Polish_{meth_name}")
            if np.isfinite(sc) and sc <= best_score + 1e-10:
                f_cur = f_new

        # Additional: try peak shave + LP cycle
        if time.monotonic() < phase3_deadline - 20:
            ta = min(60.0, phase3_deadline - time.monotonic() - 20)
            f_ps, _ = peak_shave(f_cur, ta)
            sc = update_best(f_ps, "PS_before_LP")
            if np.isfinite(sc):
                f_cur = f_ps
            if time.monotonic() < phase3_deadline - 10:
                ta2 = min(60.0, phase3_deadline - time.monotonic())
                f_lp, _ = lp_optimize(f_cur, ta2, use_perturbation=True)
                sc = update_best(f_lp, "LP_after_PS")

    # Phase 4: Final intensive optimization
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"=== Phase 4: Final Polish ===")
        f_final = best_seq.copy()
        n = len(f_final)
        remaining = deadline - time.monotonic() - 5

        # Cycle through methods a few times
        for cycle in range(3):
            if remaining < 5:
                break
            ta = max(5.0, remaining / 3.0)
            remaining -= ta

            # Subgradient with small steps
            f_new, _ = subgradient_polish(f_final, min(ta * 0.4, 30.0), eta0=0.01)
            sc = update_best(f_new, f"FinSG_{cycle}")
            if np.isfinite(sc) and sc <= best_score:
                f_final = f_new

            # Mirror with high temperature
            if remaining > 10:
                f_new, _ = mirror_polish(f_final, min(ta * 0.3, 20.0), t_param=150.0, eta0=0.1)
                sc = update_best(f_new, f"FinMir_{cycle}")
                if np.isfinite(sc) and sc <= best_score:
                    f_final = f_new

            # Short LP
            if remaining > 10:
                f_new, _ = lp_optimize(f_final, min(ta * 0.3, 20.0), use_perturbation=True)
                sc = update_best(f_new, f"FinLP_{cycle}")
                if np.isfinite(sc) and sc <= best_score:
                    f_final = f_new

            remaining = deadline - time.monotonic() - 5

    # Final fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
