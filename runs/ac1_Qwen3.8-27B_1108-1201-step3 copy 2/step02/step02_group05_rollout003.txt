import numpy as np
import time
import scipy.optimize

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_value(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def mirror_descent(a, time_limit, lr_init=0.1, momentum=0.9):
    """Mirror descent with multiplicative updates for non-negativity."""
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_obj = obj_value(a)
    v = np.zeros_like(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad[i] = 2*a[k_star - i] for valid i
        # a[j] = 0 for j < 0 or j >= n
        n = len(a)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Actually, let me vectorize this
        # grad[i] = 2*a[k_star - i] if 0 <= k_star-i < n else 0
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] = 2.0 * a[k_star - indices]
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        # Adaptive learning rate
        step = lr / v_norm
        
        # Multiplicative update (mirror descent with entropy)
        a_new = a * np.exp(-step * v)
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Decay learning rate

            lr *= 0.9
            no_improve = 0
        
        if no_improve > 10000:

    return best_a, best_obj

    """Fast subgradient descent with averaged subgradients."""
    a = np.maximum(a.astype(np.float64), 1e-15).copy()

    a /= s

    best_a = a.copy()

    t0 = time.monotonic()

    no_improve = 0

        conv = fast_convolve(a)
        
        if use_averaged and K > 1:
            # Use top-K positions
            top_k = np.argsort(conv)[-K:]
            weights = np.exp(conv[top_k] * 10)  # softmax-like
            weights /= weights.sum()
            grad = np.zeros(n)
            for w, k in zip(weights, top_k):
                lo = max(0, int(k) - n + 1)
                hi = min(n - 1, int(k))
                if lo <= hi:

                    grad[idx] += w * 2.0 * a[int(k) - idx]

        a_new = np.maximum(a - step_size * v, 1e-15)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast subgradient descent
with Nesterov acceleration, averaged subgradients, and hybrid local search.
Explores multiple sequence lengths (512-2048) with diverse initializations.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, k_star):
    """Compute subgradient of (a*a)[k_star] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k_star - idx]
    return grad

def compute_averaged_subgradient(a, conv, K=5, temperature=10.0):
    """Compute averaged subgradient from top-K convolution positions."""
    n = len(a)
    # Softmax weights over top-K
    top_indices = np.argsort(conv)[-K:]
    top_vals = conv[top_indices]
    weights = np.exp(top_vals * temperature)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[int(k) - idx]
    return grad

def subgradient_optimize(a0, time_limit, K=5, temperature=10.0, lr_init=None):
    """Nesterov accelerated subgradient descent with averaged subgradients."""
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    if lr_init is None:
        lr_init = 1.0 / np.sqrt(n)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    momentum = 0.5  # start with lower momentum, increase over time
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        current_obj = conv.max()
        
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient
        if K > 1 and current_obj > 1e-10:
            grad = compute_averaged_subgradient(a, conv, K=K, temperature=temperature)
        else:
            k_star = int(np.argmax(conv))
            grad = compute_subgradient(a, k_star)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        step_size = lr / v_norm
        a_new = a - step_size * v
        a_new = np.maximum(a_new, 1e-15)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            break
        a_new /= a_new_sum
        a = a_new
        iter_count += 1
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
        if no_improve > 5000:
            break
        
        # Slowly increase momentum
        if iter_count % 1000 == 0:
            momentum = min(0.9, momentum + 0.01)
        
        # Occasionally try larger steps
        if iter_count % 5000 == 0 and lr > lr_init * 0.01:
            lr = min(lr_init, lr * 2.0)
    
    return best_a, best_obj * 2 * n  # Convert back to original objective

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a hybrid of fast
subgradient descent, LP-based refinement, and local search.
Explores multiple sequence lengths with diverse initializations.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return out


def obj_from_conv(a, conv):
    """Compute objective from precomputed convolution (assumes sum(a)=1)."""
    n = len(a)
    return 2.0 * n * conv.max()


def subgradient_at(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a. O(n) time."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def averaged_subgradient(a, conv, K, temperature):
    """Compute averaged subgradient from top-K positions with softmax weighting."""
    n = len(a)
    top_idx = np.argsort(conv)[-K:]
    top_vals = conv[top_idx].astype(np.float64)
    # Softmax for weighting
    shifted = top_vals - top_vals.max()
    weights = np.exp(shifted * temperature)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[ki - idx]
    return grad


def subgradient_step(a, conv, lr, K, temperature, momentum_vec, mu):
    """One step of Nesterov-accelerated subgradient descent with multiplicative projection."""
    n = len(a)
    if K > 1:
        grad = averaged_subgradient(a, conv, K, temperature)
    else:
        k_star = int(np.argmax(conv))
        grad = subgradient_at(a, k_star, n)
    
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a, momentum_vec, lr
    
    # Nesterov momentum
    momentum_vec = mu * momentum_vec + (1.0 - mu) * grad
    v_norm = np.linalg.norm(momentum_vec)
    if v_norm < 1e-15:
        return a, momentum_vec, lr
    
    # Step
    a_new = a - (lr / v_norm) * momentum_vec
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new, momentum_vec, lr


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    # Build constraint matrix
    # conv(f, g)[k] = sum_i f[i]*g[k-i] for valid i, k-i
    # For each k (0 to 2n-2), the constraint is:
    #   sum_{i: 0<=i<n, 0<=k-i<n} f[i]*g[k-i] <= M
    # Let j = k-i, so g[j] appears with coefficient f[k-j] = f[i]
    # For a fixed k, g[j] (for j from max(0,k-n+1) to min(n-1,k)) 
    #   has coefficient f[k-j]
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def line_search(f, g, max_iter=30):
    """Golden section line search for (1-t)*f + t*g."""
    def f_of_t(t):
        a = (1.0 - t) * f + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return np.inf
        a /= s
        return fast_convolve(a).max()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)
    t = (lo + hi) / 2.0
    return t, f_of_t(t)


def hybrid_optimize(a0, time_limit, n_sg_per_lp=200, K=5, temperature=5.0):
    """Hybrid: fast subgradient steps + periodic LP refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    mu = 0.5
    momentum_vec = np.zeros(n)
    no_improve = 0
    sg_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # --- Subgradient phase ---
        for _ in range(n_sg_per_lp):
            if time.monotonic() - t0 >= time_limit:
                break
            conv = fast_convolve(a)
            peak = conv.max()
            if peak < best_peak - 1e-14:
                best_peak = peak
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            a, momentum_vec, lr = subgradient_step(a, conv, lr, K, temperature, momentum_vec, mu)
            sg_count += 1
            
            if no_improve > 200:
                lr *= 0.9
                no_improve = 0
            if no_improve > 5000:
                break
        
        # --- LP phase ---
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break
        
        lp_time = min(30.0, remaining * 0.3)
        if lp_time < 2:
            break
        
        # Solve LP (with time limit via signal if needed, but scipy doesn't support that easily)
        g = solve_lp_direction(a, M, n)
        
        if not np.allclose(g, a, atol=1e-6):
            t, obj_t = line_search(a, g, max_iter=20)
            if obj_t < best_peak - 1e-10:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                if ns > 1e-12:
                    a_new /= ns
                    a = a_new
                    no_improve = 0
        
        # Increase momentum over time
        if sg_count % 2000 == 0:
            mu = min(0.85, mu + 0.01)
    
    return best_a, best_peak


def local_search(a, time_limit):
    """Local search: redistribute mass from peak-contributing to safe elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 500:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contributions to peak
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            contrib[idx] = a[idx] * a[k_star - idx]
        
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Find danger and safe sets
        thr = 0.5 * max_c
        danger = np.where(contrib > thr)[0]
        safe = np.where(contrib < 0.3 * max_c)[0]
        
        if len(danger) == 0 or len(safe) == 0:
            # Relax thresholds
            danger = np.argsort(contrib)[-20:]
            safe = np.argsort(contrib)[:20]
        
        found = False
        # Try moving mass from each danger to each safe
        for di in danger[:10]:
            if found:
                break
            for si in safe[:10]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.15, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    trial_peak = fast_convolve(a_trial).max()
                    if trial_peak < best_obj - 1e-12:
                        best_obj = trial_peak
                        a = a_trial.copy()
                        found = True
                        break
        
        if found:
            stuck = 0
        else:
            stuck += 1
    
    return a, best_obj
