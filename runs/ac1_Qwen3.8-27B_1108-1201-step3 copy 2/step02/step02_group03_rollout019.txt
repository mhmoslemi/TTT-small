"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based softmax gradient, Adam optimizer, temperature annealing,
and structural exploration moves (noise injection, pair swaps).
Focuses on n=384 with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def softmax_gradient_full(a, tau):
    """
    Compute the gradient of the objective 2*n*softmax_max_conv/S^2 using FFT.
    Returns (gradient_vector, objective_value).
    """
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    
    # Softmax weights
    c_max = c.max()
    logits = tau * (c - c_max)
    w = np.exp(logits)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    # Softmax objective value (proxy for max)
    F = c_max + np.log(w_sum) / tau  # soft max value
    
    # Gradient of soft-max w.r.t. a:
    # F'_i = 2 * sum_j w[i+j] * a[j]
    # Using FFT: conv(w, a_flipped)[n-1 : n-1+n]
    a_flipped = a[::-1]
    # Need to be careful: conv of length (2n-1) and n gives length 3n-2
    # We want indices n-1 to n-1+n-1
    conv_w_a = np.convolve(w, a_flipped, mode='full')
    F_grad = 2.0 * conv_w_a[n-1:n-1+n]
    
    # S = sum(a)
    S = a.sum()
    
    # Gradient of full objective: 2*n/S^3 * (S * F'_i - 2*F)
    # This is because d/d a_i [F/S^2] = (F'_i * S^2 - F * 2S) / S^4 = (F'_i * S - 2F) / S^3
    grad = (2.0 * n / (S ** 3)) * (S * F_grad - 2.0 * F)
    
    # Objective value
    obj = 2.0 * n * F / (S * S)
    
    return grad, obj, F, S

def softmax_gradient_fft(a, tau):
    """
    FFT-based computation of the softmax gradient.
    """
    n = len(a)
    # Compute autocorrelation using FFT
    size = 2 * n  # power of 2 for efficiency (next pow2 of 2n-1)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    
    A = np.zeros(size, dtype=np.complex128)
    A[:n] = a
    fft_A = np.fft.fft(A)
    c = np.real(np.fft.ifft(fft_A * fft_A))[:2*n-1]
    
    # Softmax weights
    c_max = c.max()
    logits = tau * (c - c_max)
    w = np.exp(logits)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    F = float(c_max + np.log(w_sum) / tau)
    
    # Gradient via FFT cross-correlation
    # F'_i = 2 * sum_j w[i+j] * a[j]
    # This is the cross-correlation of w and a
    # Using FFT: create W (padded) and A (padded), compute W * conj(A) in freq domain
    # Actually, let me use the direct formula:
    # F'_i = 2 * conv(w, a_flipped)[i + n - 1]
    
    # For FFT: pad w to size 2*size, pad a_flipped
    w_pad = np.zeros(2 * size)
    w_pad[:2*n-1] = w
    a_flipped_pad = np.zeros(2 * size)
    a_flipped_pad[:n] = a[::-1]
    
    W = np.fft.fft(w_pad)
    AF = np.fft.fft(a_flipped_pad)
    conv_result = np.real(np.fft.ifft(W * AF))
    
    F_grad = 2.0 * conv_result[n-1:n-1+n]
    
    S = float(a.sum())
    grad = (2.0 * n / (S ** 3)) * (S * F_grad - 2.0 * F)
    obj = 2.0 * n * F / (S * S)
    
    return grad, obj, F, S, c

def adam_step(a, grad, m, v, beta1, beta2, lr, t, eps=1e-8):
    """Adam optimizer step with non-negativity projection."""
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    return a_new

def optimize_inner(a0, deadline, verbose=True):
    """
    Optimize a single starting point using adaptive softmax gradient descent.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    
    # Initialize
    S_t = np.sqrt(2.0 * n)
    a = a * (S_t / a.sum())
    
    best_a = a.copy()
    best_score = compute_obj(best_a)
    
    if verbose:
        print(f"      n={n}, init_score={best_score:.6f}")
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t_adam = 0
    
    # Hyperparameters
    tau = 2.0  # softmax temperature
    lr = 0.1
    tau_start = 1.0
    tau_end = 200.0
    lr_start = 0.1
    lr_end = 0.001
    
    # Schedule: total iterations
    total_iters = 30000
    it = 0
    no_improve = 0
    plateau_limit = 2000
    
    rng = np.random.default_rng(hash(tuple(a[:10])) % 2**32)
    
    while it < total_iters and time.monotonic() < deadline:
        if it % 100 == 0 and time.monotonic() > deadline:
            break
        
        # Anneal parameters
        progress = it / total_iters
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        t_adam += 1
        grad, obj, F, S, c = softmax_gradient_fft(a, tau)
        
        # Check actual objective periodically
        if it % 100 == 0:
            actual_obj = compute_obj(a)
            if actual_obj < best_score - 1e-8:
                best_score = actual_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 100
        
        a = adam_step(a, grad, m, v, beta1, beta2, lr, t_adam)
        
        # Periodic exploration: add noise to escape local minima
        if it % 500 == 400:
            noise_scale = 0.05 * (1 - progress)
            noise = rng.normal(0, noise_scale, n)
            a_explore = np.maximum(a + noise, 0)
            s_explore = compute_obj(a_explore)
            if s_explore < best_score:
                best_score = s_explore
                best_a = a_explore.copy()
                a = a_explore.copy()
                m[:] = 0
                v[:] = 0
                no_improve = 0
        
        # Periodic "jump": replace with perturbed best
        if it % 1000 == 800 and no_improve > 500:
            perturbation = rng.normal(0, 0.1, n)
            a_jump = np.maximum(best_a + perturbation, 0)
            a_jump = a_jump * (S_t / a_jump.sum())
            a = a_jump.copy()
            m[:] = 0
            v[:] = 0
            no_improve = 0
        
        # Early stop if stuck
        if no_improve > plateau_limit:
            # Try a bigger jump
            a = best_a * (1 + rng.normal(0, 0.2, n))
            a = np.maximum(a, 0)
            a = a * (S_t / a.sum())
            m[:] = 0
            v[:] = 0
            no_improve = 0
            plateau_limit = int(plateau_limit * 0.7)
            if plateau_limit < 200:
                break
        
        it += 1
    
    # Final evaluation
    final_score = compute_obj(best_a)
    if verbose:
        print(f"      After {it} iters: score={best_score:.6f}")
    
    return best_a, min(best_score, final_score)

def lp_refine_simple(a, deadline):
    """
    LP-based refinement: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then blend g with a.
    """
    n = len(a)
    if n > 1024:
        return a, compute_obj(a)
    
    S_t = np.sqrt(2.0 * n)
    a = a * (S_t / a.sum())
    c = np.convolve(a, a)
    M = float(c.max())
    best_a = a.copy()
    best_score = compute_obj(best_a)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]
    
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bnds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-6:
                g_norm = g * (S_t / g_sum)
                for blend in [0.3, 0.5, 0.7, 0.9, 1.0]:
                    a_new = (1 - blend) * best_a + blend * g_norm
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new * (S_t / a_new.sum())
                    s_new = compute_obj(a_new)
                    if s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
    except Exception:
        pass
    
    return best_a, best_score
