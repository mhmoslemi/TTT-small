```
def resample_sequence(a,n2):
   n=a.size
   if n2 == n: return a.copy()
   x1=np.linspace(0,1,n)
   x2=np.linspace(0,1,n2)
   b=np.interp(x2,x1,a, left=0.0, right=0.0)
   return np.maximum(b,0)
```

```
    vals = []

        val = float(x); cnt=1

            val = (val*cnt + v*cv)/(cnt+cv)
```

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a)) / sum(a)^2

The program uses:
- diverse structured starts (power laws, smoothed random, Paley difference sets,
  and any provided height_sequence_1),
- exact/sampled line-minimization on the simplex for coordinate and pair moves,
- monotone projection and stochastic re-heating to escape local minima,
- multi-phase time management and final polishing.
"""

import numpy as np
import time


def score_full(a):
    """Full objective, matching evaluate_sequence for valid finite inputs."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * a.size * m / (s * s))


def normalize_p(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        a[:] = 1.0
        s = float(a.size)
    a /= s
    return a


def resample_sequence(a, n2):
    """Linear resample a sequence to length n2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n1 = a.size
    if n1 <= 0:
        return np.zeros(max(1, n2), dtype=np.float64)
    if n2 == n1:
        return a.copy()
    x1 = np.linspace(0.0, 1.0, n1)
    x2 = np.linspace(0.0, 1.0, n2)
    b = np.interp(x2, x1, a, left=0.0, right=0.0)
    return np.maximum(b, 0.0)


def prepare_start(p, max_len=768, min_len=32):
    """Normalize and optionally downsample a raw start candidate."""
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return None
    if p.size > max_len:
        p = resample_sequence(p, max_len)
    if p.size < min_len:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return None
    p /= s
    return p


def eval_quad_points(c, d, e, ts):
    """Evaluate max(c + t*d + t^2*e) for all t in ts."""
    ts = np.asarray(ts, dtype=np.float64).ravel()
    if ts.size == 0:
        return np.empty(0, dtype=np.float64)
    if ts.size == 1:
        t = float(ts[0])
        val = np.max(c + t * d + (t * t) * e)
        if not np.isfinite(val):
            val = 1e100
        return np.array([float(val)], dtype=np.float64)

    ts2 = ts * ts
    q = c[:, None] + d[:, None] * ts[None, :] + e[:, None] * ts2[None, :]
    res = np.max(q, axis=0)
    bad = ~np.isfinite(res)
    if np.any(bad):
        res = np.where(bad, 1e100, res)
    return res


def line_search_quad(c, d, e, lo, hi, M, K=17, refine_K=9):
    """
    Line minimize max(c + t*d + t^2*e) on [lo, hi].
    The point t=0 is assumed feasible and has value M.
    Returns (t_best, f_best).
    """
    if hi < lo:
        lo, hi = hi, lo
    if not np.isfinite(lo) or not np.isfinite(hi):
        return 0.0, M
    if hi - lo < 1e-15:
        return float(lo), M

    K = max(int(K), 3)
    refine_K = max(int(refine_K), 3)

    pts = np.linspace(lo, hi, K, dtype=np.float64)
    if not np.any(np.abs(pts) <= 1e-14):
        pts = np.concatenate((pts, np.array([0.0], dtype=np.float64)))
        pts = np.sort(pts)

    vals = eval_quad_points(c, d, e, pts)
    if vals.size == 0:
        return 0.0, M
    best_idx = int(np.argmin(vals))
    best_t = float(pts[best_idx])
    best_val = float(vals[best_idx])

    if best_val < M - 1e-13:
        if best_idx == 0:
            lo2 = float(pts[0])
            hi2 = float(pts[1]) if pts.size > 1 else lo2
        elif best_idx == pts.size - 1:
            lo2 = float(pts[-2]) if pts.size > 1 else float(pts[-1])
            hi2 = float(pts[-1])
        else:
            lo2 = float(pts[best_idx - 1])
            hi2 = float(pts[best_idx + 1])

        if hi2 > lo2:
            pts2 = np.linspace(lo2, hi2, refine_K, dtype=np.float64)
            vals2 = eval_quad_points(c, d, e, pts2)
            idx2 = int(np.argmin(vals2))
            if vals2[idx2] < best_val:
                best_t = float(pts2[idx2])
                best_val = float(vals2[idx2])

                if refine_K > 5:
                    if idx2 == 0:
                        lo3 = float(pts2[0])
                        hi3 = float(pts2[1])
                    elif idx2 == pts2.size - 1:
                        lo3 = float(pts2[-2]) if pts2.size > 1 else float(pts2[-1])
                        hi3 = float(pts2[-1])
                    else:
                        lo3 = float(pts2[idx2 - 1])
                        hi3 = float(pts2[idx2 + 1])

                    if hi3 > lo3:
                        pts3 = np.linspace(lo3, hi3, max(3, refine_K // 2), dtype=np.float64)
                        vals3 = eval_quad_points(c, d, e, pts3)
                        idx3 = int(np.argmin(vals3))
                        if vals3[idx3] < best_val:
                            best_t = float(pts3[idx3])
                            best_val = float(vals3[idx3])

    if not np.isfinite(best_val):
        return 0.0, M
    return best_t, best_val


def make_order(n, rng, fraction=1.0):
    """Create an order of coordinates, optionally a random subset."""
    if n <= 1:
        return np.arange(n, dtype=np.int64)
    fraction = float(fraction)
    if fraction >= 1.0:
        return rng.permutation(n)
    m = int(max(1, int(n * fraction)))
    m = min(m, n)
    return rng.choice(n, size=m, replace=False)


def coordinate_sweep(p, c, M, order, K, refine_K, deadline):
    """
    Coordinate line moves on the simplex:
      p + t*(e_i - p), t in [0,1]      (toward coordinate i)
      p + t*(e_i - p), t in [t_min,0]  (away from i, reparameterized to u in [0,1])
    """
    n = int(p.size)
    L = 2 * n - 1
    improved = 0
    tol = 1e-12

    for i in order:
        if deadline is not None and time.monotonic() > deadline:
            break

        pi = float(p[i])
        if not np.isfinite(pi) or pi < 0.0:
            pi = 0.0
        if pi > 1.0 - 1e-13:
            continue

        shift = np.zeros(L, dtype=np.float64)
        shift[i:i + n] = p

        d = 2.0 * (shift - c)
        e = c - 2.0 * shift
        idx2 = 2 * i
        if idx2 < L:
            e[idx2] += 1.0

        # Toward basis e_i.
        t_best, val_best = line_search_quad(c, d, e, 0.0, 1.0, M, K, refine_K)

        # Away from i: move along the same line to the point where p_i = 0.
        if 0.0 < pi < 1.0 and pi > 0.05:
            t_min = -pi / (1.0 - pi)
            if t_min < -0.05:
                if t_min < -50.0:
                    t_min = -50.0
                d2 = t_min * d
                e2 = (t_min * t_min) * e
                u, val2 = line_search_quad(c, d2, e2, 0.0, 1.0, M, K, refine_K)
                if val2 < val_best:
                    val_best = val2
                    t_best = t_min * u

        if np.isfinite(val_best) and val_best < M - tol and abs(t_best) > 1e-14:
            scale = 1.0 - t_best
            p *= scale
            p[i] += t_best
            if p[i] < 0.0 and p[i] > -1e-10:
                p[i] = 0.0
            c = c + t_best * d + (t_best * t_best) * e
            M = float(c.max())
            if not np.isfinite(M):
                c = np.convolve(p, p)
                M = float(c.max())
            improved += 1

    return p, c, M, improved


def generate_pair_list(n, rng, count):
    """Generate random and adjacent coordinate pair moves."""
    pairs = []
    if n < 2 or count <= 0:
        return pairs

    for _ in range(int(count)):
        mode = int(rng.integers(3))
        if mode == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        else:
            i = int(rng.integers(n - 1))
            pairs.append((i, i + 1))
    return pairs


def pair_transfer_sweep(p, c, M, pairs, K, deadline):
    """
    Pair line moves: transfer mass from i to j.
    p(s) = p + s*(-e_i + e_j), s in [-p_j, p_i].
    """
    n = int(p.size)
    L = 2 * n - 1
    improved = 0
    tol = 1e-12

    for i, j in pairs:
        if deadline is not None and time.monotonic() > deadline:
            break
        if i == j:
            continue

        pi = float(p[i])
        pj = float(p[j])
        if not np.isfinite(pi) or not np.isfinite(pj):
            continue
        if pi < 0.0:
            pi = 0.0
        if pj < 0.0:
            pj = 0.0
        if pi <= 1e-15 and pj <= 1e-15:
            continue

        shift_i = np.zeros(L, dtype=np.float64)
        shift_i[i:i + n] = p
        shift_j = np.zeros(L, dtype=np.float64)
        shift_j[j:j + n] = p

        d = 2.0 * (shift_j - shift_i)
        e = np.zeros(L, dtype=np.float64)
        e[2 * i] += 1.0
        e[2 * j] += 1.0
        idx_ij = i + j
        if idx_ij < L:
            e[idx_ij] -= 2.0

        lo = -pj
        hi = pi
        if hi - lo < 1e-14:
            continue

        s_best, val_best = line_search_quad(c, d, e, lo, hi, M, K, max(3, K // 2))

        if np.isfinite(val_best) and val_best < M - tol and abs(s_best) > 1e-14:
            p[i] -= s_best
            p[j] += s_best
            if p[i] < 0.0 and p[i] > -1e-10:
                p[i] = 0.0
            if p[j] < 0.0 and p[j] > -1e-10:
                p[j] = 0.0
            c = c + s_best * d + (s_best * s_best) * e
            M = float(c.max())
            if not np.isfinite(M):
                c = np.convolve(p, p)
                M = float(c.max())
            improved += 1

    return p, c, M, improved


def isotonic_decreasing_norm(a):
    """Project onto the non-increasing simplex (PAV + normalization)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)

    vals = []
    counts = []

    for x in a:
        val = float(x)
        if not np.isfinite(val) or val < 0.0:
            val = 0.0
        cnt = 1
        # Non-increasing: if new value is larger than previous block, merge.
        while vals and val > vals[-1] + 1e-15:
            v = vals.pop()
            cv = counts.pop()
            val = (val * cnt + v * cv) / float(cnt + cv)
            cnt += cv
        vals.append(val)
        counts.append(cnt)

    out = np.empty(n, dtype=np.float64)
    pos = 0
    for val, cnt in zip(vals, counts):
        out[pos:pos + cnt] = val
        pos += cnt

    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def is_prime(m):
    """Simple primality test for small moduli."""
    if m < 2:
        return False
    if m % 2 == 0:
        return m == 2
    d = 3
    while d * d <= m:
        if m % d == 0:
            return False
        d += 2
    return True


def generate_starts(rng):
    """Generate diverse normalized starting sequences."""
    raw = []

    # Provided previous state-of-the-art, if present.
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        hs = np.where(np.isfinite(hs), hs, 0.0)
        hs = np.maximum(hs, 0.0)
        if hs.size >= 32:
            if hs.size > 800:
                raw.append(resample_sequence(hs, 768))
            else:
                raw.append(hs.copy())
            raw.append(hs[::-1].copy())
            raw.append(np.sort(hs)[::-1].copy())
            raw.append(np.sort(hs).copy())
    except Exception:
        pass

    lengths = [383, 384, 448, 512, 640, 768]

    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        one = np.ones(n, dtype=np.float64)

        # Abel-type power laws are strong continuous-limit starts.
        raw.append(1.0 / np.sqrt(i + 1.0))
        raw.append(1.0 / np.sqrt(i + 0.5))
        raw.append(1.0 / np.sqrt(i + 1.0 + 1.0))
        raw.append(1.0 / (np.sqrt(i + 1.0) + 1.0))
        raw.append(1.0 / (np.sqrt(i + 1.0) * (1.0 + i / max(1, n - 1))))
        raw.append(1.0 / (np.sqrt(i + 1.0) * (1.0 + 0.5 * i / max(1, n - 1))))

        # Smooth shapes and a few modulations.
        raw.append((one + 0.5 * np.cos(np.pi * i / max(1, n - 1))) / (np.sqrt(i + 1.0) + 0.5))
        raw.append((0.25 + 0.75 * (1.0 - i / max(1, n - 1))) / (np.sqrt(i + 1.0) + 0.25))
        raw.append(1.0 / (np.sqrt(i + 1.0) + np.sqrt(n + 1.0 - i)))

    # Smoothed random starts.
    for n in [384, 512, 640]:
        x = rng.random(n)
        w = max(1, n // 32)
        ker = np.ones(w, dtype=np.float64) / float(w)
        x = np.convolve(x, ker, mode="same")
        x = np.maximum(x, 0.0) + 1e-3
        raw.append(x)

    # Paley cyclic difference sets (P(x)^2 has very regular interior sums).
    for q in [383, 419, 443, 467, 491, 503, 563, 599, 631]:
        if q % 4 != 3 or not is_prime(q):
            continue
        a = np.zeros(q, dtype=np.float64)
        for x in range(1, q):
            a[(x * x) % q] = 1.0
        a += 1e-4
        raw.append(a)

    starts = []
    seen = set()
    for r in raw:
        p = prepare_start(r, max_len=768, min_len=32)
        if p is None:
            continue
        key = (int(p.size), round(float(p[0]), 8), round(float(p[p.size // 2]), 8),
               round(float(p[-1]), 8))
        if key in seen:
            continue
        seen.add(key)
        starts.append(p)

    if len(starts) > 42:
        starts = starts[:42]
    return starts


def first_elem(item):
    return item[0]


def select_candidates(scored, best_p, max_count=5):
    """Choose a few detailed-optimization candidates."""
    cands = []
    length_count = {}

    if best_p is not None:
        best_p = np.asarray(best_p, dtype=np.float64).ravel()
        cands.append(best_p.copy())
        if best_p.size > 0:
            length_count[int(best_p.size)] = 1

    for sc, p in scored:
        if len(cands) >= max_count:
            break
        L = int(np.asarray(p).size)
        if length_count.get(L, 0) >= 2:
            continue
        cands.append(np.asarray(p, dtype=np.float64).copy())
        length_count[L] = length_count.get(L, 0) + 1

    # Top-up with best sorted candidates if needed.
    if len(cands) < max_count:
        for sc, p in scored:
            if len(cands) >= max_count:
                break
            cands.append(np.asarray(p, dtype=np.float64).copy())

    return cands


def optimize_sequence(a, deadline, fast=False, seed_val=0):
    """
    Local optimizer from one initial sequence.
    Uses coordinate line moves, pair transfers, cleaning, monotone projection,
    and stochastic perturbations at stagnation.
    """
    p = normalize_p(a)
    n = int(p.size)
    if deadline is None:
        deadline = time.monotonic() + 30.0
    deadline = float(task_deadline if (task_deadline := deadline) is not None else deadline)

    remaining = deadline - time.monotonic()
    if n <= 1 or remaining < 8.0:
        c = np.convolve(p, p)
        M = float(c.max())
        return p.copy(), float(2.0 * n * M)

    c = np.convolve(p, p)
    M = float(c.max())
    best_p = p.copy()
    best_score = float(2.0 * n * M)

    if seed_val is None:
        seed_arg = 12345
    else:
        seed_arg = int(abs(int(seed_val))) % 2147483646
    rng = np.random.default_rng(seed_arg + 1)

    if fast:
        Kcoord = 9
        refineK = 5
        pairK = 7
        max_sweeps = 14
        coord_fraction = 0.8
        pair_count = 32
        pair_every = 2
        stag_th = 3
    else:
        if n <= 420:
            Kcoord = 33
            refineK = 15
            pairK = 13
        elif n <= 580:
            Kcoord = 27
            refineK = 13
            pairK = 11
        else:
            Kcoord = 21
            refineK = 11
            pairK = 11
        max_sweeps = 800
        coord_fraction = 1.0
        pair_count = 96 if n <= 512 else 64
        pair_every = 1
        stag_th = 6

    sweeps = 0
    stagnant = 0

    while time.monotonic() < deadline - 1.5 and sweeps < max_sweeps:
        order = make_order(n, rng, coord_fraction)
        p, c, M, _ = coordinate_sweep(p, c, M, order, Kcoord, refineK, deadline)

        if (sweeps % pair_every == 0) and time.monotonic() < deadline - 3.0:
            pairs = generate_pair_list(n, rng, pair_count)
            p, c, M, _ = pair_transfer_sweep(p, c, M, pairs, pairK, deadline)

        # Clean, renormalize, and recompute exact convolution.
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s <= 1e-15:
            p[:] = 1.0 / n
        else:
            p /= s
        c = np.convolve(p, p)
        M = float(c.max())
        score = float(2.0 * n * M)
        if not np.isfinite(score) or score >= 1e90:
            score = score_full(p)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            stagnant = 0
        else:
            stagnant += 1

        sweeps += 1

        if stagnant > stag_th:
            # Try monotone projection of the current best.
            p_mono = isotonic_decreasing_norm(best_p)
            c_mono = np.convolve(p_mono, p_mono)
            sc_mono = float(2.0 * n * float(c_mono.max()))
            if np.isfinite(sc_mono) and sc_mono < best_score:
                best_score = sc_mono
                best_p = p_mono.copy()

            # Perturb the best point and accept mild worsening to escape.
            sigma = (0.18 if fast else 0.14) * float(np.exp(-sweeps / (18.0 if fast else 90.0)))
            p_new = best_p * np.exp(rng.normal(0.0, sigma, n))
            p_new = p_new + 1e-6
            p_new = np.maximum(p_new, 0.0)
            sn = float(p_new.sum())
            if sn > 0:
                p_new /= sn
            c_new = np.convolve(p_new, p_new)
            sc_new = float(2.0 * n * float(c_new.max()))

            accept_tol = 0.12 if fast else 0.06
            if np.isfinite(sc_new) and sc_new < best_score * (1.0 + accept_tol):
                p = p_new
                c = c_new
                M = float(c_new.max())
            else:
                p = best_p.copy()
                c = np.convolve(p, p)
                M = float(c.max())

            stagnant = 0

    # Final monotone polish on best.
    p_mono = isotonic_decreasing_norm(best_p)
    c_mono = np.convolve(p_mono, p_mono)
    sc_mono = float(2.0 * n * float(c_mono.max()))
    if np.isfinite(sc_mono) and sc_mono < best_score:
        best_score = sc_mono
        best_p = p_mono.copy()

    return best_p.copy(), best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Returns a list of non-negative floats.
    """
    t0 = time.monotonic()
    budget = float(budget_s) if budget_s is not None else 1800.0
    global_deadline = t0 + max(60.0, budget - 25.0)

    rng = np.random.default_rng(int(abs(int(seed))) % 2147483646 + 7)

    starts = generate_starts(rng)
    scored = []
    for p in starts:
        sc = score_full(p)
        if np.isfinite(sc):
            scored.append((sc, p))

    scored.sort(key=first_elem)

    best_p = None
    best_score = float("inf")
    for sc, p in scored:
        if sc < best_score:
            best_score = sc
            best_p = np.asarray(p, dtype=np.float64).copy()

    if best_p is not None:
        print("initial best: n={:.0f} score={:.8f}".format(best_p.size, best_score))
    else:
        best_p = np.ones(384, dtype=np.float64)
        best_p /= best_p.size
        best_score = score_full(best_p)

    # Phase 1: quick local optimization on the best starting candidates.
    n_fast = min(6, len(scored))
    for idx in range(n_fast):
        rem = global_deadline - time.monotonic()
        if rem < 90.0:
            break
        sc0, p0 = scored[idx]
        t_alloc = min(38.0, max(12.0, rem / max(1, n_fast - idx) * 0.75))
        deadline = min(global_deadline - 30.0, time.monotonic() + t_alloc)
        p_opt, sc_opt = optimize_sequence(p0, deadline, fast=True, seed_val=seed + idx)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = float(sc_opt)
            best_p = np.asarray(p_opt, dtype=np.float64).copy()
            print("fast phase {}: n={:.0f} score={:.8f}".format(idx + 1, best_p.size, best_score))
        else:
            print("fast phase {}: n={:.0f} score={:.8f} (not better)".format(idx + 1, p0.size, sc_opt))

    # Phase 2: detailed optimization on diverse top candidates.
    candidates = select_candidates(scored, best_p, max_count=5)
    n_cand = len(candidates)
    for idx, p0 in enumerate(candidates):
        rem = global_deadline - time.monotonic()
        if rem < 210.0:
            break
        t_alloc = min(420.0, max(30.0, (rem - 90.0) / max(1, n_cand - idx)))
        deadline = min(global_deadline - 20.0, time.monotonic() + t_alloc)
        p_opt, sc_opt = optimize_sequence(p0, deadline, fast=False, seed_val=seed + 100 + idx)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = float(sc_opt)
            best_p = np.asarray(p_opt, dtype=np.float64).copy()
            print("detailed {}: n={:.0f} score={:.8f}".format(idx + 1, best_p.size, best_score))
        else:
            print("detailed {}: n={:.0f} score={:.8f} (not better)".format(idx + 1, p0.size, sc_opt))

    # Phase 3: try nearby/resampled lengths if substantial time remains.
    if best_p is not None:
        rem = global_deadline - time.monotonic()
        if rem > 260.0:
            n_b = int(best_p.size)
            target_lengths = []
            for m in [max(128, n_b - 64), n_b + 1, min(768, n_b + 64), 512, 640, 384]:
                if 96 <= m <= 768 and m != n_b and m not in target_lengths:
                    target_lengths.append(m)

            tried = 0
            for m in target_lengths:
                rem = global_deadline - time.monotonic()
                if rem < 150.0 or tried >= 3:
                    break
                p2 = resample_sequence(best_p, m)
                p2 = normalize_p(p2)
                sc2 = score_full(p2)
                if not np.isfinite(sc2):
                    continue
                t_alloc = min(210.0, max(45.0, rem - 90.0))
                deadline = min(global_deadline - 25.0, time.monotonic() + t_alloc)
                p_opt, sc_opt = optimize_sequence(p2, deadline, fast=False, seed_val=seed + 300 + tried)
                if np.isfinite(sc_opt) and sc_opt < best_score:
                    best_score = float(sc_opt)
                    best_p = np.asarray(p_opt, dtype=np.float64).copy()
                    print("resampled n={:.0f}: score={:.8f}".format(m, best_score))
                else:
                    print("resampled n={:.0f}: score={:.8f} (not better)".format(m, sc_opt))
                tried += 1

        # Final polish on best.
        rem = global_deadline - time.monotonic()
        if rem > 70.0:
            t_alloc = min(420.0, max(20.0, rem - 12.0))
            deadline = time.monotonic() + t_alloc
            p_opt, sc_opt = optimize_sequence(best_p, deadline, fast=False, seed_val=seed + 999)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = float(sc_opt)
                best_p = np.asarray(p_opt, dtype=np.float64).copy()
                print("final polish: n={:.0f} score={:.8f}".format(best_p.size, best_score))
            else:
                print("final polish: n={:.0f} score={:.8f} (not better)".format(best_p.size, sc_opt))

    # Final monotone projection and validation.
    if best_p is None:
        best_p = np.ones(256, dtype=np.float64)
        best_p /= best_p.size

    best_p = np.asarray(best_p, dtype=np.float64).ravel()
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s > 0:
        best_p /= s

    p_mono = isotonic_decreasing_norm(best_p)
    sc_mono = score_full(p_mono)
    sc_best = score_full(best_p)
    if np.isfinite(sc_mono) and (not np.isfinite(sc_best) or sc_mono < sc_best):
        best_p = p_mono
        best_score = float(sc_mono)

    # Cross-check with the official evaluator if available.
    try:
        ev = float(evaluate_sequence([float(x) for x in best_p]))
        if np.isfinite(ev):
            print("official evaluator score: {:.8f}".format(ev))
        else:
            print("official evaluator returned non-finite; using internal score")
    except Exception:
        pass

    final_score = score_full(best_p)
    if not np.isfinite(final_score):
        fallback = np.ones(384, dtype=np.float64)
        fallback /= fallback.size
        return [float(x) for x in fallback]

    print("FINAL: n={:.0f} score={:.8f}".format(best_p.size, final_score))
    return [float(x) for x in best_p]
```
