        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)
        
        # Check improvement
        new_peak = np.max(fft_convolve(a_new, a_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
        
        # Adaptive learning rate
        a = a_new
        
        if step % 1000 == 0 and verbose:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Nesterov step {step}: peak={best_peak:.6f}, score={score:.6f}, lr={lr:.6f}")
    
    return best_a

def block_descent(a, time_limit, verbose=False):
    """
    Block coordinate descent: partition a into blocks and optimize each.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()

    block_sizes = [4, 8, 16, 32, 64]
    
    for bs in block_sizes:
        if time.monotonic() - t0 > time_limit:
            break
        n_blocks = n // bs
        if n_blocks == 0:
            continue
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv_a = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv_a))
            
            # Find which blocks contain indices that contribute to the peak
            # Reduce those blocks

                contrib = 0.0

                    if 0 <= i1 < n:

                            best_a = candidate.copy()

                            break
        
        if verbose:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Block descent (bs={bs}): score={score:.6f}")

        # Blending
        best_t = 0.0
        best_new = np.max(conv_a)
        for t_try in [0.3, 0.2, 0.15, 0.1, 0.05]:
            b = (1 - t_try) * a + t_try * g

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Normalize sequence to target sum."""
    if target_sum is None:

        # Blend with momentum
        if len(direction_history) >= 3:
            # Average recent directions
            g_avg = np.mean(direction_history[-3:], axis=0)
            g_avg = normalize_seq(g_avg, S)
            g_use = 0.5 * g + 0.5 * g_avg
            g_use = normalize_seq(g_use, S)
        else:
            g_use = g
        
        # Line search
        best_t = 0.0
        best_new_peak = current_peak
        for t_try in np.array([0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.03]):
            b = (1 - t_try) * a + t_try * g_use
            b = np.maximum(b, 1e-6)
            b = normalize_seq(b, S)
            new_peak = np.max(fft_convolve(b, b))
            if new_peak < best_new_peak:
                best_new_peak = new_peak
                best_t = t_try
        
        if best_t == 0.0:
            stall += 1
            if stall > 30:
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-6)
                a = normalize_seq(a, S)
                direction_history = []
                stall = 0
            continue
        
        a = (1 - best_t) * a + best_t * g_use
        a = np.maximum(a, 1e-6)
        a = normalize_seq(a, S)
        
        new_peak = np.max(fft_convolve(a, a))
        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and iteration % 100 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    LP iter {iteration}: peak={best_peak:.8f}, score={score:.6f}, stall={stall}")
    
    return best_a

# ============ Spectral Shaping ============

def spectral_shape(a, alpha=0.5, time_limit=30.0, verbose=False):
    """
    Modify the spectrum of a to be flatter (reducing peak autocorrelation).
    
    Uses the fact that circular autocorrelation = ifft(|FFT(a)|^2).
    For linear autocorrelation, this is an approximation but provides
    a good direction for improvement.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    # Pad for FFT
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for iteration in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude
        # Current magnitude spectrum
        mag_target = np.full_like(mag, np.sqrt(np.sum(mag**2) / len(mag)))
        
        # Blend magnitudes
        mag_new = mag ** alpha * mag_target ** (1 - alpha)
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 1e-6)
        a_new = normalize_seq(a_new, S)
        
        new_peak = np.max(fft_convolve(a_new, a_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
        
        # Gradually increase alpha
        alpha = min(0.95, alpha * 1.01)
    
    return best_a

# ============ Local Search ============

def local_search(a, time_limit, verbose=False, rng_seed=0):
    """
    Fine-grained local search with various move types.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    current_peak = best_peak
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        conv_a = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv_a))
        
        # Move type 1: Scale a single element
        if rng.random() < 0.3:
            j = int(rng.integers(n))
            for factor in [0.95, 0.9, 1.05, 1.1, 1.2]:
                candidate = a.copy()
                candidate[j] *= factor
                candidate = np.maximum(candidate, 1e-6)
                candidate = normalize_seq(candidate, S)
                new_peak = np.max(fft_convolve(candidate, candidate))
                if new_peak < current_peak:
                    a = candidate
                    current_peak = new_peak
                    if new_peak < best_peak:
                        best_peak = new_peak
                        best_a = candidate.copy()
                    break
        
        # Move type 2: Scale a pair (one up, one down)
        elif rng.random() < 0.6:
            j1 = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if j1 == j2:
                continue
            for ratio in [0.9, 0.8, 0.7]:
                scale = ratio / (2.0 - ratio)  # Keep sum approximately constant
                candidate = a.copy()
                candidate[j1] *= ratio
                candidate[j2] *= (1 + (1 - ratio) * a[j1] / max(a[j2], 1e-10))
                candidate = np.maximum(candidate, 1e-6)
                candidate = normalize_seq(candidate, S)
                new_peak = np.max(fft_convolve(candidate, candidate))
                if new_peak < current_peak:
                    a = candidate
                    current_peak = new_peak
                    if new_peak < best_peak:
                        best_peak = new_peak
                        best_a = candidate.copy()
                    break
        
        # Move type 3: Block swap (exchange values between two positions)
        elif rng.random() < 0.8:
            j1 = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if j1 == j2:
                continue
            candidate = a.copy()
            # Partial swap
            t = 0.3
            candidate[j1] = (1-t)*a[j1] + t*a[j2]
            candidate[j2] = (1-t)*a[j2] + t*a[j1]
            candidate = normalize_seq(candidate, S)
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < current_peak:
                a = candidate
                current_peak = new_peak
                if new_peak < best_peak:
                    best_peak = new_peak
                    best_a = candidate.copy()
        
        # Move type 4: Reduce peak contributor, increase random other
        else:
            contribs = np.zeros(n)
            for j in range(n):
                i1 = peak_idx - j
                if 0 <= i1 < n:
                    contribs[j] = a[j] * a[i1]
            top_j = int(np.argmax(contribs))
            other_j = int(rng.integers(n))
            if other_j == top_j:
                continue
            candidate = a.copy()
            candidate[top_j] *= 0.9
            candidate[other_j] *= 1.1
            candidate = np.maximum(candidate, 1e-6)
            candidate = normalize_seq(candidate, S)
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < current_peak:
                a = candidate
                current_peak = new_peak
                if new_peak < best_peak:
                    best_peak = new_peak
                    best_a = candidate.copy()
        
        if verbose and step % 1000 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Local search step {step}: score={score:.6f}")
    
    return best_a

# ============ Subgradient Method (Improved) ============

def subgradient_optimize(a, time_limit, verbose=False):
    """
    Subgradient optimization with multiple active constraints.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    lr = 2.0 / n
    K = 5  # Number of active constraints
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        # Get top K lags
        top_k = np.argsort(conv_a)[-K:]
        
        # Compute subgradient as weighted sum of gradients at top K lags
        weights = np.zeros(2 * n - 1)
        peak_vals = conv_a[top_k]
        w_sum = np.sum(peak_vals)
        for i, k_idx in enumerate(top_k):
            weights[k_idx] = peak_vals[i] / w_sum
        
        grad = np.zeros(n)
        for k_idx in top_k:
            w = weights[k_idx]
            for j in range(n):
                i1 = k_idx - j
                if 0 <= i1 < n:
                    grad[j] += w * a[i1]
                i2 = j - k_idx
                if 0 <= i2 < n:
                    grad[j] += w * a[i2]
        
        # Step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-6)
        a_new = normalize_seq(a_new, S)
        
        new_peak = np.max(fft_convolve(a_new, a_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
        
        # Average step (Polyak-style)
        a = 0.9 * a + 0.1 * a_new if new_peak > np.max(conv_a) else a_new
        
        # Adaptive lr
        lr *= 0.9995
        if lr < 1e-6:
            lr = 1e-6
        
        if verbose and step % 5000 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Subgrad step {step}: score={score:.6f}, lr={lr:.6f}")
    
    return best_a

# ============ Initialization ============

def make_init(n, rng, provided=None):
    """Create initial sequence."""
    t = np.arange(n, dtype=np.float64)
    r = rng.random()
    
    if provided is not None and r < 0.4:
        x_old = np.linspace(0, 1, len(provided))
        x_new = np.linspace(0, 1, n)
        base = np.interp(x_new, x_old, provided)
        a = base * np.exp(rng.normal(0, 0.05, n))
    elif r < 0.5:
        alpha = rng.uniform(0.1, 0.4)
        a = 1.0 + alpha * np.cos(2 * np.pi * rng.uniform(0.5, 2.0) * t / n)
    elif r < 0.6:
        a = np.ones(n)
        for _ in range(int(rng.integers(2, 4))):
            f = int(rng.integers(1, n // 5))
            amp = rng.uniform(0.05, 0.15)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif r < 0.7:
        a = 1.0 + 0.2 * rng.standard_normal(n)
        # Smooth
        k = np.ones(5) / 5
        a = np.convolve(a, k, 'same')
    else:
        a = 0.8 + 0.4 * rng.random(n)
    
    return np.maximum(a, 0.01)
