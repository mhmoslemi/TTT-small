```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. L-BFGS-B on log-sum-exp smoothed objective for local optimization
2. LP-based direction finding for escaping local minima
3. Multi-length exploration and simulated annealing

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_obj_grad(a, t):
    """Compute smoothed max of autocorrelation and its gradient.
    a is assumed to have sum=1 and be non-negative.
    Returns (objective_value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Log-sum-exp smoothing
    m = conv.max()
    if m < 1e-30:
        return 0.0, np.zeros(n)
    
    scaled = t * conv
    scaled_max = scaled.max()
    exp_scaled = np.exp(scaled - scaled_max)
    log_sum_exp = (scaled_max + np.log(exp_scaled.sum())) / t
    weights = exp_scaled / exp_scaled.sum()
    
    # Gradient: grad[i] = 2 * sum_k (t * weights[k]) * a[k-i]
    w = t * weights
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-12:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    
    return log_sum_exp, grad

def lbfgs_optimize(a0, time_limit, t_init=10.0, t_max=200.0):
    """Optimize using L-BFGS-B on smoothed objective with increasing t."""
    # Work on the simplex: parameterize as a = exp(x) / sum(exp(x))
    # Or just use bounds [0, inf] and project
    
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    t = t_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Define objective and gradient for L-BFGS
        def obj_func(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10
            x = x / sx
            val, _ = smoothed_obj_grad(x, t)
            return val
        
        def grad_func(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-15:
                return np.ones_like(x)
            x = x / sx
            _, g = smoothed_obj_grad(x, t)
            return g
        
        # Use L-BFGS-B with bounds
        res = minimize(
            obj_func, a.copy(), method='L-BFGS-B',
            jac=grad_func,
            bounds=[(0, 1000)] * n,
            options={'maxiter': 200, 'gtol': 1e-6, 'maxfun': 500}
        )
        
        a_new = np.maximum(res.x, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Increase smoothing parameter
        t = min(t * 1.5, t_max)
        
        if time.monotonic() - t0 > time_limit - 10:
            break
    
    return best_a, best_obj

def lp_direction(f, M):
    """Solve LP to find direction g that reduces cross-correlation with f."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # min -sum(g)
    # s.t. (f*g)(k) <= M for k=0,...,2n-2
    #      g >= 0
    
    # Build constraint matrix
    # (f*g)(k) = sum_{i=0}^{n-1} f[k-i] * g[i] (where f[j]=0 if j out of range)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

def lp_iterate(a, time_limit, max_iters=100, t_start=0.05, t_end=0.01):
    """Iterate using LP-based convex combinations."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        g = lp_direction(a, M)
        if g is None:
            break
        
        # Try convex combination
        # Adaptive step size
        t = t_start * (t_end / t_start) ** (it / max_iters)
        
        improved = False
        for t_try in [t, t*0.5, t*0.1, t*2]:
            h = (1 - t_try) * a + t_try * g
            h = np.maximum(h, 0)
            sh = h.sum()
            if sh < 1e-15:
                continue
            h = h / sh
            conv_h = fast_convolve(h)
            obj_h = conv_h.max()
            if obj_h < best_obj - 1e-10:
                best_obj = obj_h
                best_a = h.copy()
                a = h.copy()
                conv = conv_h
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj
```

```python
def cross_correlate(w, a):
    """Compute h[i] = sum_k w[k] * a[k-i] for i in [0, n-1]."""
    n = len(a)
    len_w = len(w)
    # h[i] = sum_k w[k] * a[k-i]
    # = convolve(w, flip(a))[i + n - 1] ... let me be careful
    # Actually: h[i] = sum_k w[k] * a[k-i]
    # Let b[m] = a[n-1-m] (reversal of a)
    # Then a[k-i] = b[n-1-k+i]
    # h[i] = sum_k w[k] * b[n-1-k+i]
    # Let j = n-1+i, then h[i] = sum_k w[k] * b[j-k] = (w*b)[j] where * is convolution
    # j ranges from n-1 to 2n-2
    # w has length 2n-1, b has length n
    # (w*b) has length 3n-2
    # h[i] = (w*b)[n-1+i] for i in [0, n-1]
    
    fft_size = 1 << (len_w + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    B = np.fft.rfft(b, fft_size)
    conv = np.fft.irfft(W * B, fft_size)
    return conv[n-1 : n-1+n]
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization on log-sum-exp smoothed objective with simplex parameterization
2. LP-based convex combination directions for escaping local minima
3. Multi-length exploration with adaptive time allocation
4. Periodic random restarts and high-resolution polishing
Key improvements over subgradient: quadratic convergence of L-BFGS,
natural simplex constraint via exponential parameterization, and
theoretically-grounded LP directions.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate_fft(w, a):
    """Compute h[i] = sum_k w[k] * a[k-i] for i in [0, n-1] using FFT."""
    n = len(a)
    b = a[::-1]
    len_w = len(w)
    fft_size = 1 << (len_w + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    B = np.fft.rfft(b, fft_size)
    conv = np.fft.irfft(W * B, fft_size)
    return conv[n - 1:n - 1 + n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max_and_grad(a, t):
    """Compute log-sum-exp smoothed max of autocorrelation and gradient w.r.t. a.
    Returns (smoothed_value, h) where h[i] = d(smoothed_max)/d(a_i).
    """
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    if m < 1e-30:
        return 0.0, np.zeros(n)
    scaled = t * conv
    sm = scaled.max()
    e = np.exp(scaled - sm)
    log_sum_exp = (sm + np.log(e.sum())) / t
    w = e / e.sum()
    tw = t * w
    h = 2.0 * cross_correlate_fft(tw, a)
    return log_sum_exp, h


def lbfgs_optimize(a0, time_limit, t_init=50.0, t_max=1000.0, max_inner_iters=150):
    """L-BFGS optimization using exponential parameterization on the simplex.
    Returns (best_a_normalized, best_true_objective).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    x = np.log(a)  # log-parameterization

    t0 = time.monotonic()
    best_val = fast_convolve(a).max()
    best_a = a.copy()
    t = t_init
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        def obj_fn(x_cur):
            log_a = x_cur - x_cur.max()
            a_cur = np.exp(log_a)
            S = a_cur.sum()
            if S < 1e-30:
                return 1e10
            a_cur = a_cur / S
            val, _ = smoothed_max_and_grad(a_cur, t)
            return val

        def grad_fn(x_cur):
            log_a = x_cur - x_cur.max()
            a_cur = np.exp(log_a)
            S = a_cur.sum()
            if S < 1e-30:
                return np.zeros(len(x_cur))
            a_cur = a_cur / S
            _, h = smoothed_max_and_grad(a_cur, t)
            h_a = h @ a_cur
            return a_cur * (h - h_a)

        res = minimize(
            obj_fn, x.copy(), method='L-BFGS-B',
            jac=grad_fn,
            options={'maxiter': max_inner_iters, 'gtol': 1e-10, 'maxfun': 500,
                     'ftol': 1e-12}
        )

        x = res.x
        log_a = x - x.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        conv = fast_convolve(a_new)
        true_obj = conv.max()

        if true_obj < best_val - 1e-12:
            best_val = true_obj
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        a = a_new

        # Increase smoothing parameter
        if stagnation < 3:
            t = min(t * 1.8, t_max)
        else:
            t = min(t * 2.5, t_max)
            stagnation = 0

        # Time check
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 15:
            break

    return best_a, best_val


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)(k)<=M for all k.
    Returns normalized g or None.
    """
    n = len(f)
    # Build constraint matrix: (f*g)(k) = sum_i f[k-i]*g[i]
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        A[k, i_lo:i_hi + 1] = f[k - np.arange(i_lo, i_hi + 1)]

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            g = res.x
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def lp_iterate(a, time_limit, max_iterations=200):
    """Iterate using LP-based convex combinations with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    base_rate = 0.03

    for it in range(max_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        g = lp_direction(a, M)
        if g is None:
            break

        # Adaptive step size: try multiple rates
        rates = [base_rate * (0.5 ** (no_improve // 5)),
                 base_rate * 0.5, base_rate, base_rate * 2.0]

        improved = False
        for rate in rates:
            h = (1 - rate) * a + rate * g
            h = np.maximum(h, 0)
            sh = h.sum()
            if sh < 1e-15:
                continue
            h = h / sh
            conv_h = fast_convolve(h)
            obj_h = conv_h.max()
            if obj_h < best_obj - 1e-11:
                best_obj = obj_h
                best_a = h.copy()
                a = h.copy()
                conv = conv_h
                improved = True
                no_improve = 0
                base_rate = min(0.1, rate * 1.2)
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp_inc'] = ramp / ramp.sum()
    inits['ramp_dec'] = ramp[::-1] / ramp.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x).astype(float) + 0.5
    inits['triangle'] = tri / tri.sum()

    # Gaussians
    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf:.1f}'] = g / g.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Quadratic chirp (non-negative)
    for freq in [1, 2, 3]:
        chirp = 1.0 + 0.8 * np.cos(np.pi * freq * x * (x + 1) / (2 * n))
        chirp = np.maximum(chirp, 0.01)
        inits[f'chirp_{freq}'] = chirp / chirp.sum()

    # Sinusoidal modulation
    for freq in [1, 2, 4]:
        mod = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        mod = np.maximum(mod, 0.01)
        inits[f'sin_{freq}'] = mod / mod.sum()

    # Random
    if rng is not None:
        inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
        inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                inits['parent_ds'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                inits['parent_up'] = pp / pp.sum()

    return inits


def subgradient_polish(a, time_limit, top_k=20, lr=0.02, momentum=0.9):
    """Quick subgradient polish for fine-tuning after L-BFGS."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    temp = max(conv.max() / n, 1e-6)

    while time.monotonic() - t0 < time_limit:
        # Get subgradient at top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            wi = w[i] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)
        conv = fast_convolve(a_new)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # still update for momentum

        temp *= 1.01
        lr *= 0.9995

    return best_a, best_obj

        init_names = list(inits.keys())

        # Prioritize
        priority = ['parent', 'parent_ds', 'parent_up', 'chirp_2', 'gauss_0.2',
                    'random', 'uniform', 'bimodal', 'sin_2', 'ramp_inc']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]

        time_per_init = max(10.0, (phase1_limit - 60) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() - t0 > phase1_limit - 40:
                break
            t_alloc = min(time_per_init, phase1_limit - (time.monotonic() - t0) - 30)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_opt = lbfgs_optimize(a0, time_limit=t_alloc, t_init=30, t_max=100)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # ===== Phase 2: Deep optimization (900s) =====
    print("=== Phase 2: Deep Optimization ===")
    t0 = time.monotonic()

    # Build list of candidates to deeply optimize
    candidates = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = (n, round(a_opt[0], 4), round(a_opt[-1], 4))
        if key not in seen_keys:
            candidates.append((n, name, a_opt))
            seen_keys.add(key)
        if len(candidates) >= 6:
            break

    # Always include current best
    if best_sequence is not None:
        candidates.append((len(best_sequence), 'current_best', best_sequence.copy()))

    # Add some fresh random starts at the most promising length
    if scan_results:
        best_n = scan_results[0][1]
        for r in range(2):
            a_rand = rng.dirichlet(np.ones(best_n) * 0.3)
            candidates.append((best_n, f'fresh_{r}', a_rand))

    phase2_limit = 900
    time_per_candidate = max(20.0, (phase2_limit - 60) / max(1, len(candidates)))

    for i, (n, name, a0) in enumerate(candidates):
        if time.monotonic() - t0 > phase2_limit - 60:
            break
        t_alloc = min(time_per_candidate, phase2_limit - (time.monotonic() - t0) - 50)
        if t_alloc < 15:
            break

        print(f"  Deep[{i}]: n={n}, {name}, t_alloc={t_alloc:.0f}s")

        # Step 1: L-BFGS optimization
        t_lbfgs = t_alloc * 0.5
        a_opt_1, obj_1 = lbfgs_optimize(a0, time_limit=t_lbfgs, t_init=50, t_max=500)

        # Step 2: LP iterations
        t_lp = t_alloc * 0.25
        a_opt_2, obj_2 = lp_iterate(a_opt_1, time_limit=t_lp)

        # Step 3: L-BFGS polish after LP
        t_polish = t_alloc * 0.25
        a_opt_3, obj_3 = lbfgs_optimize(a_opt_2, time_limit=t_polish, t_init=200, t_max=2000)

        # Take the best of the three
        for a_try, obj_try in [(a_opt_1, obj_1), (a_opt_2, obj_2), (a_opt_3, obj_3)]:
            update_best(a_try, f"(deep {name})")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ===== Phase 3: Final polish (500s) =====
    print("=== Phase 3: Final Polish ===")
    t0 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        phase3_limit = 500

        # Multiple L-BFGS passes with different t
        for t_init in [100, 300, 800, 2000]:
            if time.monotonic() - t0 > phase3_limit - 30:
                break
            t_alloc = min(60, phase3_limit - (time.monotonic() - t0) - 25)
            if t_alloc < 10:
                break
            a_opt, obj_opt = lbfgs_optimize(best_sequence, time_limit=t_alloc,
                                            t_init=t_init, t_max=t_init * 3)
            update_best(a_opt, f"(polish t={t_init})")

        # LP iterations on best
        if time.monotonic() - t0 < phase3_limit - 30:
            t_lp = min(80, phase3_limit - (time.monotonic() - t0) - 25)
            if t_lp > 10:
                a_opt, obj_opt = lp_iterate(best_sequence, time_limit=t_lp)
                update_best(a_opt, "(LP polish)")

        # Subgradient fine-tune
        if time.monotonic() - t0 < phase3_limit - 20:
            t_sg = min(40, phase3_limit - (time.monotonic() - t0) - 15)
            if t_sg > 5:
                a_opt, obj_opt = subgradient_polish(best_sequence, time_limit=t_sg,
                                                     top_k=n_best // 10, lr=0.01, momentum=0.92)
                update_best(a_opt, "(subgrad)")

        # Random perturbations + re-optimize
        if time.monotonic() - t0 < phase3_limit - 20:
            for perturb_scale in [0.005, 0.01, 0.02, 0.003]:
                if time.monotonic() - t0 > phase3_limit - 25:
                    break
                noise = rng.standard_normal(n_best) * perturb_scale
                a_pert = project_simplex(best_sequence + noise)
                t_opt = min(30, phase3_limit - (time.monotonic() - t0) - 20)
                if t_opt < 8:
                    break
                a_opt, obj_opt = lbfgs_optimize(a_pert, time_limit=t_opt,
                                                t_init=100, t_max=500)
                update_best(a_opt, f"(perturb {perturb_scale})")

        # Try neighboring lengths via interpolation
        if time.monotonic() - t0 < phase3_limit - 40:
            for delta in [-16, -8, 8, 16]:
                n_try = n_best + delta
                if n_try < 64 or n_try > 1024:
                    continue
                if time.monotonic() - t0 > phase3_limit - 30:
                    break
                a_interp = np.interp(np.linspace(0, 1, n_try),
                                     np.linspace(0, 1, n_best), best_sequence)
                a_interp = np.maximum(a_interp, 0)
                a_interp = a_interp / a_interp.sum()
                t_opt = min(30, phase3_limit - (time.monotonic() - t0) - 20)
                if t_opt < 10:
                    break
                a_opt, obj_opt = lbfgs_optimize(a_interp, time_limit=t_opt,
                                                t_init=50, t_max=500)
                update_best(a_opt, f"(n={n_try})")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
