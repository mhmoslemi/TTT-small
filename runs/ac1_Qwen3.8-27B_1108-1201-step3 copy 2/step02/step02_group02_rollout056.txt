**Idea: Projected gradient with exact projection**

- If the max is achieved at a unique k*, then grad = ∇c_{k*}(a)
- If multiple k's achieve the max, grad = (1/|A|) * sum_{k in A} ∇c_k(a) where A is the active set

This is a weighted average of the gradients, with weights proportional to exp(K*c_k). As K → ∞, this approaches the subgradient of the max.

Minimizes 2*n*max(autocorrelation)/(sum)^2 for non-negative sequences.

        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_grad(a, K):
    n = len(a)
    c = np.convolve(a, a).astype(np.float64)
    max_c = c.max()
    expo = K * (c - max_c)
    # Keep only significant terms
    mask = expo > -30
    if not np.any(mask):
        mask[c.argmax()] = True
    w = np.zeros_like(c)
    w[mask] = np.exp(expo[mask])
    w /= w.sum()
    
    active = np.where(mask)[0]
    grad = np.zeros(n)
    for k in active:
        wk = w[k]
        if wk < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        m_start = k - j_end
        m_end = k - j_start
        grad[j_start:j_end+1] += 2.0 * wk * a[m_start:m_end+1]
    return grad

def optimize_subgrad(a0, time_limit, rng, K_init=30, verbose=False):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best = a.copy()
    best_score = score_fast(a)
    current_score = best_score
    
    lr = 0.05
    K = K_init
    momentum = np.zeros(n)
    beta = 0.5
    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        grad = compute_grad(a, K)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        
        # Try steps with line search
        improved = False
        for trial_lr in [lr, lr*0.7, lr*0.5, lr*0.3, lr*0.1]:
            a_new = a - trial_lr * momentum
            a_new = project_simplex(a_new)
            new_score = score_fast(a_new)
            if new_score < current_score:
                a = a_new
                current_score = new_score
                lr = min(lr * 1.05, 0.3)
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            lr *= 0.8
            K = min(K * 1.05, 500)
            
            if stall > 30:
                # Frank-Wolfe step
                j_min = np.argmin(grad)
                s_fw = np.zeros(n)
                s_fw[j_min] = 1.0
                for t_fw in [0.5, 0.3, 0.2, 0.1]:
                    a_fw = (1 - t_fw) * a + t_fw * s_fw
                    fw_score = score_fast(a_fw)
                    if fw_score < current_score:
                        a = a_fw
                        current_score = fw_score
                        improved = True
                        stall = 0
                        break
                
                if not improved and stall > 60:
                    # Perturbation restart
                    a = a * np.exp(rng.normal(0, 0.1, n))
                    a = project_simplex(a)
                    current_score = score_fast(a)
                    lr = 0.05
                    K = K_init
                    stall = 0
                    momentum = np.zeros(n)
                    beta = 0.5
        
        if current_score < best_score:
            best_score = current_score
            best = a.copy()
        
        iter_count += 1
        if verbose and iter_count % 1000 == 0:
            print(f"    iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, K={K:.1f}, lr={lr:.4f}")
    
    return best, best_score

def polish_scipy(a0, time_limit):
    """Use scipy.optimize.minimize for final polish."""
    n = len(a0)
    a0 = a0.astype(np.float64).copy()
    a0 = np.clip(a0, 0, None)
    a0 = a0 / np.sum(a0)
    
    def objective(x):
        x = np.maximum(x, 0)
        s = np.sum(x)
        if s < 0.01:
            return 1e6
        c = np.convolve(x, x)
        return 2.0 * n * np.max(c) / (s * s)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
                   'jac': lambda x: np.ones(len(x))}]
    bounds = [(0, None)] * n
    
    start_time = time.monotonic()
    result = scipy_minimize(objective, a0, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 100, 'ftol': 1e-10})
    
    if result.success and result.fun < score_fast(a0):
        a_new = np.maximum(result.x, 0)
        a_new = a_new / np.sum(a_new)
        return a_new, result.fun
    
    return a0, score_fast(a0)

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng, spread=0.5):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

def make_dual_peak_init(n, rng):
    """Sequence with two peaks, designed to reduce central autocorrelation."""
    t = np.arange(n)
    a = np.ones(n) * 0.5
    peak1 = int(n * 0.25)
    peak2 = int(n * 0.75)
    width = n // 8
    for shift in range(-width, width):
        for p in [peak1, peak2]:
            idx = p + shift
            if 0 <= idx < n:
                a[idx] += np.exp(-0.5 * (shift/width)**2)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)
