        if i_max >= 0:
            grad[j] = np.dot(a[:i_max+1], w[j:j+i_max+1])
    
    # Center the gradient (for the simplex constraint)
    grad_c = grad - grad.mean()
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad_c)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new

"""
LP-guided optimization with adaptive stepping, gradient refinement, and
simulated-annealing perturbations for minimizing peak auto-convolution.
Implements a hybrid approach: fast LP iterations for global direction,
softmax-gradient steps for local refinement, and annealed perturbations
for escaping local minima.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

# Cache for LP index arrays
_lp_index_cache = {}

def auto_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def get_lp_structure(n):
    """Get or create precomputed LP structure for size n."""
    if n not in _lp_index_cache:
        rows_list = np.repeat(np.arange(n), n) + np.arange(n)  # (i + j) for i in range(n), j in range(n)
        # Actually: for each column i, rows are i, i+1, ..., i+n-1
        cols_list = np.repeat(np.arange(n), n)  # column i repeated n times
        rows_list = np.concatenate([np.arange(i, i+n) for i in range(n)])
        _lp_index_cache[n] = (rows_list.astype(np.int32), cols_list.astype(np.int32), 
                               np.ones(n*n, dtype=np.float64))  # template for values
    return _lp_index_cache[n]

def lp_direction(a, M, n):
    """Solve the LP: max sum(g) s.t. g>=0, (a*g)[k]<=M for all k. Returns g."""
    rows, cols, _ = get_lp_structure(n)
    vals = np.tile(a, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                           method='highs', options={'disp': False, 'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def softmax_grad(a, beta, n):
    """Compute the softmax-weighted gradient of the auto-convolution peak."""
    conv = auto_conv(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        # Fallback: use argmax
        w = np.zeros(2*n-1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # Gradient: grad[j] = sum_i a[i] * w[i+j]
    # Use FFT for efficiency
    # cross-correlation of a and w: corr[j] = sum_i a[i] * w[i+j]
    # This equals convolve(a, w_reversed)[j + n - 1]... 
    # Actually, let's use the direct formula with FFT:
    # corr[j] = sum_{i=0}^{n-1} a[i] * w[i+j]
    # = (a * w_pad)[j + n - 1] where w_pad is w reversed and padded
    # Simpler: use direct computation for n<=1024
    grad = np.zeros(n)
    for j in range(n):
        end = min(n, 2*n-1-j)
        if end > 0:
            grad[j] = np.dot(a[:end], w[j:j+end])
    return grad

def gradient_step(a, beta, eta, n):
    """One step of mirror descent using softmax gradient."""
    grad = softmax_grad(a, beta, n)
    grad_c = grad - grad.mean()
    a_new = a * np.exp(-eta * grad_c)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    return a_new / s

def full_optimize(a0, time_limit, seed=42):
    """Full hybrid optimization: LP + gradient + perturbations."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = auto_conv(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    step = 0.5
    beta = 30.0
    eta = 0.3
    perturb_temp = 0.3  # Annealing temperature for perturbations
    patience = 0
    iter_count = 0
    last_print = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # --- LP Step ---
        g = lp_direction(a, M, n)
        if g is not None and g.sum() > 1e-12:
            g /= g.sum()
            
            if np.max(np.abs(g - a)) > 1e-14:
                # Line search
                improved = False
                for t in [step, min(1.0, step * 1.3), step * 0.6, step * 0.4, 0.05]:
                    t_clipped = min(1.0, max(0.01, t))
                    a_try = (1.0 - t_clipped) * a + t_clipped * g
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st < 1e-12:
                        continue
                    a_try /= st
                    conv_try = auto_conv(a_try)
                    M_try = conv_try.max()
                    obj_try = 2.0 * n * M_try
                    if obj_try < best_obj - 1e-14:
                        a = a_try
                        M = M_try
                        best_obj = obj_try
                        best_a = a.copy()
                        patience = 0
                        step = min(1.0, step * 1.03)
                        improved = True
                        break
                    elif obj_try < 2.0 * n * M - 1e-14:
                        a = a_try
                        M = M_try
                        step = min(1.0, step * 1.01)
                        improved = True
                        break
                
                if not improved:
                    patience += 1
                    step = max(0.02, step * 0.85)
            else:
                patience += 10
        else:
            patience += 5
        
        # --- Gradient Step (every 3 iterations) ---
        if iter_count % 3 == 0:
            a = gradient_step(a, beta, eta, n)
            conv_g = auto_conv(a)
            obj_g = 2.0 * n * conv_g.max()
            if obj_g < best_obj - 1e-14:
                best_obj = obj_g
                best_a = a.copy()
                patience = 0
            M = conv_g.max()
            beta = min(500.0, beta * 1.005)
        
        # --- Perturbation (annealed) ---
        if patience > 30:
            scale = perturb_temp
            a_pert = a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-12:
                a_pert /= sp
                conv_p = auto_conv(a_pert)
                obj_p = 2.0 * n * conv_p.max()
                # Accept if better, or with probability based on "temperature"
                if obj_p < best_obj:
                    a = a_pert
                    M = obj_p / (2.0 * n)
                    best_obj = obj_p
                    best_a = a.copy()
                    patience = 0
                elif patience > 100:
                    # Accept worse solutions to escape (simulated annealing)
                    a = a_pert
                    M = obj_p / (2.0 * n)
                    patience = 0
                perturb_temp = max(0.01, perturb_temp * 0.95)
        
        if patience > 500:
            # Reset perturb temperature and try harder
            perturb_temp = 0.3
            step = 0.5
            beta = 30.0
            patience = 0
            a_pert = a * np.exp(rng.normal(0, 0.2, n))
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-12:
                a_pert /= sp
            a = a_pert
            M = auto_conv(a).max()
        
        if iter_count - last_print >= 500:
            elapsed = time.monotonic() - t0
            print(f"    [{n}d] iter={iter_count}, best={best_obj:.8f}, step={step:.3f}, "
                  f"beta={beta:.1f}, pert={perturb_temp:.3f}, elapsed={elapsed:.1f}s")
            last_print = iter_count
    
    return best_a, best_obj

def gen_initializations(n, seed=42):
    """Generate diverse initializations of length n, each summing to 1."""
    rng = np.random.default_rng(seed)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian with different widths
    for alpha in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - c) / (alpha * n / 2))**2)
        inits[f'gauss_{alpha}'] = g / g.sum()
    
    # Cosine
    cos = 0.5 * (1 + np.cos(2 * np.pi * (x - c) / n))
    inits['cosine'] = cos / cos.sum()
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - c) / (c + 0.5), 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Power-law
    r = np.arange(1, n+1, dtype=np.float64)
    inits['power'] = r**(-0.5) / (r**(-0.5)).sum()
    
    return inits
