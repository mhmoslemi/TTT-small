"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Frequency-domain iterative optimization (Grille-Turin inspired)
2. Time-domain peak shaving with multi-peak awareness
3. LP-based refinement for moderate n
4. Simulated annealing for escaping local minima
5. Multi-start with diverse initializations

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def autocorr_fft(a, n_out=None):
    """Compute linear autocorrelation (convolution) using FFT."""
    n = len(a)
    if n_out is None:
        n_out = next_fast_len(2 * n - 1)
    F = fft(a, n_out)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def freq_domain_iteration(a, n_iter=1000, alpha=0.5, tau=0.01):
    """
    Frequency-domain iterative optimization.
    Uses Grille-Turin style power spectrum shaping.
    """
    n = len(a)
    n_fft = next_fast_len(2 * n - 1)
    S = a.sum()
    
    for it in range(n_iter):
        # Compute current autocorrelation
        c = autocorr_fft(a, n_fft)
        
        # Target: make autocorrelation as flat as possible
        # The average is S^2 / (2n-1)
        avg = S * S / (2 * n - 1)
        
        # Compute power spectrum
        F = fft(a, n_fft)
        P = np.abs(F) ** 2
        
        # GT update: reduce power at frequencies that contribute to peaks
        # For each lag k where c[k] > avg, reduce power proportionally
        # Simple version: scale power by (avg / max(c))^gamma at all frequencies
        # This is too crude. Better: use the "notch" approach
        
        # Actually, the standard GT/POCS approach:
        # 1. Compute desired autocorrelation R_desired (flat)
        # 2. Compute desired power spectrum: P_desired = |FFT(R_desired)|^2
        # 3. Update: |F_new| = |F_old| * (P_desired / P_old)^(1/4) * (something)

            F *= scale
            a_new = np.real(ifft(F))[:n]
        else:
            a_new = a.copy()
        
        # Project to non-negativity
        a_new = np.maximum(a_new, 0)
        
        # Normalize to maintain sum
        S_new = a_new.sum()
        if S_new > 1e-10:

def peak_shaving_time_domain(a, n_iter=500, n_peaks=5, shrink=0.1):
    """
    Time-domain peak shaving: iteratively reduce elements that contribute
    most to the peak autocorrelation values.
    """
    n = len(a)
    S = a.sum()
    
    for it in range(n_iter):
        c = np.convolve(a, a)
        # Find top n_peaks lags
        top_lags = np.argsort(c)[-n_peaks:][::-1]
        
        improved = False
        for k in top_lags:
            # Elements contributing to c[k]: pairs (i, k-i) for valid i
            # c[k] = sum_i a[i]*a[k-i]
            # To reduce c[k], reduce a[i] where a[i]*a[k-i] is large
            contribs = []
            for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
                j = k - i
                if 0 <= j < n:
                    contribs.append((i, j, a[i] * a[j]))
            
            if not contribs:
                continue
            
            # Sort by contribution
            contribs.sort(key=lambda x: -x[2])
            
            # Reduce the top contributors
            for i, j, _ in contribs[:3]:
                # Reduce the larger of a[i] and a[j]
                if a[i] >= a[j]:
                    a[i] *= (1 - shrink)
                else:
                    a[j] *= (1 - shrink)
                improved = True
        
        # Renormalize
        S_new = a.sum()
        if S_new > 1e-10:
            a *= S / S_new
        else:
            break
        
        if not improved:
            break
    
    return a

def lp_refine(f, M, n_vars=None):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n_vars is None:
        n_vars = n
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, min(j + n, 2 * n - 1)):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def multi_peak_gradient(a, tau=None):
    """
    Gradient of soft-max of autocorrelation.
    """
    n = len(a)
    c = np.convolve(a, a)
    
    if tau is None:
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
    else:
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        a_rev = a[::-1]
        conv_result = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad

def gradient_step(a, lr, momentum, decay=0.9, tau=None):
    """Gradient descent with momentum, projected to non-negativity and sum."""
    n = len(a)
    S = a.sum()
    grad = multi_peak_gradient(a, tau)
    grad -= grad.mean()  # Remove mean to preserve sum
    momentum *= decay
    momentum -= lr * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0)
    a_new *= S / max(a_new.sum(), 1e-30)
    return a_new, momentum

def generate_diverse_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Triangular
    center = n / 2
    tri = 1 - np.abs(i - center) / center
    starts.append(tri / tri.sum())
    
    # Gaussian
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((i - center) / (n * sigma_frac)) ** 2)
        starts.append(g / g.sum())
    
    # Cosine modulated
    for freq in [1, 2, 3, 5, 7, 11]:
        c = 0.5 + 0.5 * np.cos(2 * np.pi * i * freq / n)
        starts.append(c / c.sum())
    
    # Random
    for _ in range(5):
        r = rng.random(n) + 0.01
        starts.append(r / r.sum())
    
    # Sparse (geometric-like)
    a = np.zeros(n)
    pos = 0
    while pos < n:
        a[pos] = 1.0
        pos += max(1, int(pos / 2) + 1)
    starts.append(a / a.sum())
    
    # Alternating blocks
    block_size = max(1, n // 16)
    a = np.zeros(n)
    for b in range(0, n, block_size * 2):
        a[b:min(b + block_size, n)] = 1.0
    starts.append(a / a.sum())
    
    return starts

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def autocorr(a):
    return np.convolve(a, a)

def autocorr_fft(a, n_fft=None):
    n = len(a)
    if n_fft is None:
        n_fft = next_fast_len(2 * n - 1)
    F = fft(a, n_fft)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]

def multi_peak_gradient(a, top_k=5, tau=None):
    n = len(a)
    c = autocorr(a)
    
    if tau is None:
        # Top-k peaks with equal weight
        top_idx = np.argsort(c)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad *= 2.0 / top_k
    else:
        # Soft-max
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        a_rev = a[::-1].copy()
        conv_result = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * conv_result[n - 1:n - 1 + n]
    
    return grad

def gradient_step(a, lr, momentum, decay=0.9, top_k=5, tau=None):

    S_new = a_new.sum()

def lp_minmax(f, time_limit=30):
    """
    Solve: min t s.t. conv(f,g)[k] <= t for all k, sum(g)=1, g>=0
    This gives the best g (in min-max sense) for fixed f.
    """
    n = len(f)
    
    # Variables: g[0..n-1], t
    # c = [-1, -1, ..., -1, 0] (minimize t, which is the last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # Minimize t
    
    # Constraints:
    # 1. conv(f,g)[k] <= t for k = 0, ..., 2n-2
    #    conv(f,g)[k] = sum_j f[k-j] * g[j] (where 0<=k-j<n)
    #    i.e., sum_j f[k-j] * g[j] - t <= 0
    #    This is: A1 @ [g; t] <= 0
    A1 = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n-1, k)+1):
            A1[k, j] = f[k-j]
        A1[k, n] = -1.0  # -t
    
    b1 = np.zeros(2*n-1)
    
    # 2. sum(g) = 1: A_eq @ [g; t] = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    # 3. g >= 0: bounds
    bounds = [(0, None)] * n + [(None, None)]  # t is free
    
    try:
        res = linprog(c, A_ub=A1, b_ub=b1, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Primary method: min-max LP iteration (directly optimizes peak autocorrelation).
Secondary: multi-peak gradient descent with adaptive tau.
Tertiary: frequency-domain shaping (Grille-Turin inspired).
Multi-start with diverse structured initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def autocorr(a):
    return np.convolve(a, a)


def top_k_key_sort(x):
    return x[0]


def multi_peak_gradient(a, top_k=5, tau=None):
    n = len(a)
    c = autocorr(a)
    if tau is None:
        top_idx = np.argsort(c)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad *= 2.0 / float(top_k)
    else:
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum
        a_rev = a[::-1].copy()
        conv_result = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def gradient_step(a, lr, momentum, decay=0.9, top_k=5, tau=None):
    n = len(a)
    S = a.sum()
    grad = multi_peak_gradient(a, top_k=top_k, tau=tau)
    grad -= grad.mean()
    momentum = decay * momentum - lr * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new > 1e-30:
        a_new *= S / S_new
    return a_new, momentum


def freq_shaping(a, n_iter=400, exponent=0.15):
    n = len(a)
    n_fft = next_fast_len(2 * n - 1)
    S = a.sum()
    for it in range(n_iter):
        c = autocorr(a)
        avg = (S * S) / (2 * n - 1)
        peak = float(c.max())
        if peak / (avg + 1e-30) < 1.001:
            break
        F = fft(a, n_fft)
        scale = (avg / (peak + 1e-30)) ** exponent
        F = F * scale
        a_new = np.real(ifft(F))[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new > 1e-30:
            a_new *= S / S_new
        else:
            break
        a = a_new
    return a


def lp_minmax(f, time_limit=30):
    n = len(f)
    # Variables: g[0..n-1], t
    # Minimize t
    # Subject to: conv(f,g)[k] <= t for all k; sum(g)=1; g>=0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = f[k - j]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None


def lp_maxsum(f, M, time_limit=30):
    n = len(f)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, min(j + n, 2 * n - 1)):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_minmax_lp(a0, deadline):
    """Optimize using min-max LP iteration as primary engine."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) / n
    S = a.sum()
    a = a / S
    best = a.copy()
    best_score = compute_obj(best)
    t0 = time.monotonic()
    print(f"      LP init: score={best_score:.6f}, n={n}")

    no_improve = 0
    it = 0
    blend_schedule = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03]

    while time.monotonic() < deadline - 10 and it < 200:
        it += 1
        # Adaptive time limit per LP call
        remaining = deadline - time.monotonic() - 10
        t_limit = min(30.0, max(5.0, remaining / max(1, 200 - it)))

        g, t_val = lp_minmax(a, time_limit=t_limit)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g = g / g_sum

        # Determine blend factor
        if len(blend_schedule) > it:
            alpha = blend_schedule[it]
        else:
            alpha = max(0.01, 0.3 * (0.9 ** (it - len(blend_schedule))))

        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new *= S / S_new
        s_new = compute_obj(a_new)

        if s_new < best_score - 1e-12:
            best_score = s_new
            best = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            if it % 10 == 0:
                print(f"      LP it={it}: score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 15:
                break

    return best, best_score


def optimize_gradient(a0, deadline):
    """Optimize using multi-peak gradient descent."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) / n

    a = a / S
    best = a.copy()
    best_score = compute_obj(best)

    momentum = np.zeros(n)

    while time.monotonic() < deadline - 10 and it < max_iters:

def make_starts(rng, n):
    starts = []
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    starts.append(np.ones(n) / n)
    tri = np.maximum(0.0, 1.0 - np.abs(i - center) / center)
    starts.append(tri / max(tri.sum(), 1e-10))
    for sigma_frac in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((i - center) / (n * sigma_frac)) ** 2)
        starts.append(g / max(g.sum(), 1e-10))
    for freq in [1, 2, 3, 5, 7, 11, 17]:
        c = 0.2 + 0.8 * (0.5 + 0.5 * np.cos(2.0 * np.pi * i * freq / n))
        starts.append(c / max(c.sum(), 1e-10))
    a = np.exp(-0.5 * ((i - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((i - n * 0.75) / (n * 0.1)) ** 2)
    starts.append(a / max(a.sum(), 1e-10))
    env = 0.4 + 0.6 * np.cos(np.pi * i / n)
    env = np.maximum(env, 0.0)
    mod = 1.0 + 0.4 * np.cos(2.0 * np.pi * i * 5.0 / n)
    a_mix = env * mod
    starts.append(a_mix / max(a_mix.sum(), 1e-10))
    mod2 = 1.0 + 0.5 * np.cos(2.0 * np.pi * i * 13.0 / n)
    a_mix2 = env * mod2
    starts.append(a_mix2 / max(a_mix2.sum(), 1e-10))
    for _ in range(4):
        r = rng.random(n) + 0.01
        starts.append(r / r.sum())
    # Sparse geometric
    a = np.zeros(n)
    pos = 0
    while pos < n:
        a[pos] = 1.0
        pos += max(1, int(pos * 0.7) + 1)
    starts.append(a / max(a.sum(), 1e-10))
    return starts
