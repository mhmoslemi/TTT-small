import time
from scipy.optimize import minimize
from scipy.signal import correlate

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def logsumexp(x, t):
    """Compute (1/t) * log(sum(exp(t*x))) stably."""
    m = t * x.max()
    return (m + np.log(np.sum(np.exp(t * x - m)))) / t

def smooth_objective(z, t, n):
    """Objective in softmax parameterization."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    return 2.0 * n * logsumexp(conv, t)

def smooth_gradient(z, t, n):

    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()  # softmax
    conv = fast_convolve(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. L-BFGS-B optimization on a log-sum-exp smooth surrogate (softmax parameterization)
2. Multi-scale initialization (small n -> large n)
3. Adaptive temperature annealing for escaping local minima
4. Projected subgradient polishing on the true (non-smooth) objective
5. Multi-restart with diverse initializations and palindromic symmetry option
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def get_score_from_conv(n, conv):
    sm = conv.max()
    return 2.0 * n * sm

def compute_smooth_grad_a(a, conv, t):
    n = len(a)
    tc = t * conv
    tc -= tc.max()
    exp_tc = np.exp(tc)
    p = exp_tc / exp_tc.sum()
    # grad_a[i] = 4n * sum_k p[k] * a[k-i]
    # This is 4n * correlate(p, a, 'full')[:n]
    # Compute via FFT:
    size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)[:n]
    return 4.0 * n * corr

def optimize_lbfgs_smooth(a_init, t_schedule, time_limit, verbose=True):
    n = len(a_init)
    a = np.maximum(a_init, 1e-30)
    a = a / a.sum()
    z = np.log(a) - np.log(a).mean()
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = get_score_from_conv(n, conv)
    
    t0 = time.monotonic()
    
    for t in t_schedule:
        if time.monotonic() - t0 > time_limit - 15:
            break
        
        def fun(zz):
            aa = softmax_fn(zz)
            c = fast_convolve(aa)
            m = t * c.max()
            return 2.0 * n * (m + np.log(np.sum(np.exp(t * c - m)))) / t
        
        def gfun(zz):
            aa = softmax_fn(zz)
            c = fast_convolve(aa)
            ga = compute_smooth_grad_a(aa, c, t) / (2.0 * n)
            # Wait, compute_smooth_grad_a already includes the 4n factor.
            # For the softmax gradient: grad_z_j = a_j * (grad_a_j - <grad_a, a>)
            # where grad_a is the gradient of F_t w.r.t. a.
            # F_t = 2n * LSE(t*conv)/t
            # dF_t/da = 4n * correlate(p, a) (as computed)
            # So grad_a = compute_smooth_grad_a(a, conv, t)
            mean_g = np.dot(aa, ga)
            return aa * (ga - mean_g)
        
        # Time-limited L-BFGS-B
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed - 15
        if remaining < 2:
            break
        
        result = minimize(
            fun, z, jac=gfun, method='L-BFGS-B',
            options={
                'maxiter': min(1000, max(50, int(remaining * 50))),
                'ftol': 1e-15,
                'gtol': 1e-9,
                'maxfun': min(5000, max(200, int(remaining * 200)))
            }
        )
        
        z = result.x.copy()
        a_new = softmax_fn(z)
        conv_new = fast_convolve(a_new)
        obj_new = get_score_from_conv(n, conv_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"    t={t:.0f}: obj={obj_new:.6f} (iters={result.nit})")
        elif verbose:
            print(f"    t={t:.0f}: obj={obj_new:.6f} (no improvement, iters={result.nit})")
    
    return best_a, best_obj

def subgradient_polish(a_init, time_limit, top_k=20, lr_init=0.05, decay=0.9995):
    """Polish using subgradient of the true max objective."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = get_score_from_conv(n, fast_convolve(a))
    
    t0 = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Equal weights on top peaks
        w = np.ones(len(peaks)) / len(peaks)
        
        grad = np.zeros(n)
        for i, k in enumerate(peaks):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += w[i] * 2.0 * a[k - idx]
        
        a = project_simplex(a - lr * grad)
        lr *= decay
        
        conv = fast_convolve(a)
        obj = get_score_from_conv(n, conv)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def generate_inits(n, parent=None, rng=None):
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    # Ramp
    r = np.arange(1, n+1, dtype=float)
    inits['ramp_up'] = r / r.sum()
    inits['ramp_down'] = r[::-1].copy() / r.sum()
    
    # Triangle
    x = np.arange(n, dtype=float)
    tri = np.minimum(x, n-1-x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits['gauss'] = g / g.sum()
    
    # Edge concentration
    edge = np.zeros(n)
    w = max(1, n//5)
    edge[:w] = 1.0
    edge[-w:] = 1.0
    inits['edges'] = edge / edge.sum()
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:

                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    # Random
    if rng is not None:

The gradient w.r.t. a:

= 2n * Σ_k p_k * 2*a[k-i]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp surrogate
optimization via L-BFGS-B on a softmax parameterization, multi-scale
initialization, adaptive temperature annealing, and projected subgradient
polishing on the true non-smooth objective.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def get_obj(a):
    """Compute 2*n*max(conv) for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def compute_grad_a(a, conv, t):
    """
    Compute gradient of F_t(a) = 2n * LSE(t*conv) / t w.r.t. a.
    Returns grad_a of length n.
    """
    n = len(a)
    tc = t * conv
    tc -= tc.max()
    exp_tc = np.exp(tc
    p = exp_tc / exp_tc.sum()
    # grad_a[i] = 4n * sum_k p[k] * a[k-i] = 4n * correlate(p, a)[:n]
    # Via FFT: ifft(FFT(p) * conj(FFT(a)))[:n]
    size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)[:n]
    return 4.0 * n * corr


def lbfgs_one_level(z, t, n, max_iter=500):
    """Single L-BFGS-B optimization at temperature t, starting from z."""

    def fun(zz):
        aa = softmax_fn(zz)
        c = fast_convolve(aa)
        m = t * c.max()
        return 2.0 * n * (m + np.log(np.sum(np.exp(t * c - m)))) / t

    def gfun(zz):
        aa = softmax_fn(zz)
        c = fast_convolve(aa)
        ga = compute_grad_a(aa, c, t)
        mean_g = np.dot(aa, ga)
        return aa * (ga - mean_g)

    result = minimize(fun, z, jac=gfun, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': 1e-15,
                               'gtol': 1e-9, 'maxfun': max_iter * 4})
    return result.x


def optimize_smooth_annealed(a_init, t_schedule, time_limit, verbose=True):
    """
    L-BFGS-B with log-sum-exp surrogate, annealing through t_schedule.
    Each level starts from the previous level's solution.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30)
    a = a / a.sum()
    z = np.log(a) - np.log(a).mean()

    best_a = a.copy()
    best_obj = get_obj(a)[0]

    t0 = time.monotonic()

    for t in t_schedule:
        if time.monotonic() - t0 > time_limit - 10:
            break
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed - 5
        if remaining < 3:
            break
        max_iter = max(30, min(2000, int(remaining * 40)))
        z = lbfgs_one_level(z, t, n, max_iter=max_iter)
        a_new = softmax_fn(z)
        obj_new = get_obj(a_new)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        if verbose:
            print(f"      t={t:.0f}: obj={obj_new:.6f} (best={best_obj:.6f})")

    return best_a, best_obj


def subgradient_polish(a_init, time_limit, top_k=20, lr_init=0.04):
    """Polish using subgradient of the true max-of-conv objective."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = get_obj(a)[0]
    t0 = time.monotonic()
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        w = np.ones(len(peaks)) / len(peaks)
        grad = np.zeros(n)
        for i, k in enumerate(peaks):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += w[i] * 2.0 * a[k - idx]
        a = project_simplex(a - lr * grad)
        lr *= 0.9995
        obj = get_obj(a)[0]
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    r = np.arange(1, n + 1, dtype=float)
    inits['ramp_up'] = r / r.sum()
    inits['ramp_down'] = r[::-1].copy() / r.sum()
    x = np.arange(n, dtype=float)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    g = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
    inits['gauss'] = g / g.sum()
    edge = np.zeros(n)
    w_edge = max(1, n // 5)
    edge[:w_edge] = 1.0
    edge[-w_edge:] = 1.0
    inits['edges'] = edge / edge.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    if parent is not None:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_ds'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_up'] = pp / pp.sum()
    return inits

        t_alloc = max(t_alloc, 20.0)
        a_opt, obj_opt = optimize_smooth_annealed(a0, t_sched, time_limit=t_alloc,
                                                   verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  ** New best from {init_name}: score={best_score:.6f}")

    print(f"  After Phase 2: best_score={best_score:.6f}")

    # Phase 3: High-t refinement and multi-restart
    print(f"=== Phase 3: Refinement (n={n_best}, current={best_score:.6f}) ===")
    phase3_end = min(time.monotonic() + 400.0, deadline - 150.0)

    # High-t annealing from best
    if time.monotonic() < phase3_end - 60:
        remaining = phase3_end - time.monotonic() - 20
        t_sched_hi = [1000, 2000, 5000, 10000]
        a_opt, obj_opt = optimize_smooth_annealed(
            best_a, t_sched_hi, time_limit=min(remaining, 200.0), verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  High-t: score={best_score:.6f}")

    # Random restarts
    if time.monotonic() < phase3_end - 30:
        n_restarts = 5
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break
            for scale in [0.005, 0.01, 0.02]:
                if time.monotonic() > phase3_end - 10:
                    break
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_a + noise)
                a_opt, obj_opt = optimize_smooth_annealed(
                    a_pert, [200, 500, 1500, 4000], time_limit=30, verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  Restart {i} (scale={scale}): score={best_score:.6f}")

    # Phase 4: Subgradient polishing on true objective
    print(f"=== Phase 4: Subgradient Polish (current={best_score:.6f}) ===")
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 15
        a_opt, obj_opt = subgradient_polish(
            best_a, time_limit=remaining,
            top_k=max(10, n_best // 15), lr_init=0.03)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Polish improved: score={best_score:.6f}")

    # Phase 5: Final micro-polish
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        a_opt, obj_opt = optimize_smooth_annealed(
            best_a, [5000, 10000, 20000], time_limit=remaining, verbose=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Final micro: score={best_score:.6f}")

    # Final validation
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}")
    return [float(x) for x in best_a]
