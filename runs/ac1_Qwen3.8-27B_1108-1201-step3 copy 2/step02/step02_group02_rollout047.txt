        return np.ones(n) * np.sqrt(2.0 * n) / n
    target = np.sqrt(2.0 * n)
    return a * (target / s)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ x)[k] = (f*x)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def smooth_obj_and_grad(x, tau, n):
    """
    Smoothed objective using log-sum-exp in softmax parameterization.
    a = softmax(x), so a > 0 and sum(a) = 1 automatically.
    Returns (obj, grad_x).
    """
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)

    c = fft_convolve(a, a)
    c_scaled = c / tau
    c_max = np.max(c_scaled)
    lse = c_max + np.log(np.sum(np.exp(c_scaled - c_max)))
    obj = 2.0 * n * lse

    # Softmax weights
    weights = np.exp(c_scaled - c_max)
    weights /= np.sum(weights)

    # grad_a[j] = (1/tau) * sum_k weights[k] * dc[k]/da[j]
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            idx = k - j
            if idx == j:
                grad_a[j] += 2.0 * a[j] * w
            else:
                grad_a[j] += a[idx] * w
    grad_a /= tau

    # Chain rule for softmax: grad_x = a * (grad_a - a @ grad_a)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def smooth_optimize(a0, time_limit, rng, verbose=False):
    """
    Smooth optimization via log-sum-exp annealing with L-BFGS.
    a0 is the initial sequence (will be converted to softmax params).
    """
    a0 = normalize_to_sqrt2n(a0)
    a0_norm = a0 / np.sum(a0)  # sum = 1 for softmax init
    n = len(a0)
    S = np.sqrt(2.0 * n)

    # Initialize x from a0
    a_clipped = np.clip(a0_norm, 1e-10, None)
    a_clipped = a_clipped / np.sum(a_clipped)
    x = np.log(a_clipped)

    current_score = score(a0)
    best_score = current_score
    best_a = a0.copy()

    t0 = time.monotonic()

    # Annealing schedule
    tau_max = 0.05
    tau_min = 0.002
    n_steps = 12
    tau_values = tau_max * (tau_min / tau_max) ** (np.arange(n_steps) / (n_steps - 1))

    for i, tau in enumerate(tau_values):
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Run L-BFGS at this temperature
        max_iter = max(20, int(80 * (1 + 0.3 * i)))
        res = minimize(
            smooth_obj_and_grad, x, args=(tau, n),
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        x = res.x

        # Convert back to a
        a_new = np.exp(x - np.max(x))
        a_new = a_new / np.sum(a_new)
        a_new = a_new * S  # Restore normalization

        new_score = score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        # Update x for next iteration
        a_clipped = np.clip(a_new / S, 1e-10, None)
        a_clipped = a_clipped / np.sum(a_clipped)
        x = np.log(a_clipped)

        if verbose:
            print(f"    tau={tau:.4f}, score={current_score:.6f}, best={best_score:.6f}")

    return best_a, best_score


def lp_iterate(f0, time_limit, rng, verbose=False):
    """
    LP-based iterative optimization with improved line search.
    """
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = normalize_to_sqrt2n(f0.astype(np.float64).copy())

    current_score = score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.12
    stall_count = 0
    iter_count = 0
    prev_dir = None

    while time.monotonic() - t0 < time_limit:
        # Check time more frequently for large n
        if iter_count % 500 == 0 and time.monotonic() - t0 > time_limit - 2:
            break

        c = fft_convolve(f, f)
        M = np.max(c)
        b_0 = lp_direction(f, M)
        iter_count += 1

        if b_0 is None or np.sum(b_0) < 1e-10:
            # LP failed, try perturbation
            f = f * np.exp(rng.normal(0, 0.05, n))
            f = normalize_to_sqrt2n(f)
            current_score = score(f)
            stall_count += 10
            continue

        g = b_0 * (S / np.sum(b_0))

        # Line search: try multiple step sizes including golden section
        improved = False
        candidates_t = [t, t * 0.7, t * 0.4, t * 0.2, 0.3, 0.5]
        candidates_t = [x for x in candidates_t if 0 < x < 1.0]

        best_t = None
        best_trial_score = current_score
        for trial_t in candidates_t:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0.0, None)
            new_score = score(f_new)
            if new_score < best_trial_score:
                best_trial_score = new_score
                best_t = trial_t

        if best_t is not None:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, None)
            current_score = best_trial_score
            t = min(best_t * 1.1, 0.35)
            stall_count = 0
            prev_dir = g - f
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
        else:
            stall_count += 1
            t *= 0.75

            if stall_count == 30:
                # Conjugate-like perturbation
                if prev_dir is not None:
                    f = f + 0.02 * prev_dir
                    f = np.clip(f, 0.0, None)
                    f = normalize_to_sqrt2n(f)
                else:
                    f = f * np.exp(rng.normal(0, 0.04, n))
                    f = normalize_to_sqrt2n(f)
                current_score = score(f)
                stall_count = 10
                t = 0.08

            elif stall_count >= 60 and stall_count % 30 == 0:
                # Stronger perturbation: target the peak directly
                c = fft_convolve(f, f)
                peak_idx = int(np.argmax(c))
                contribs = np.zeros(n)
                j_min = max(0, peak_idx - n + 1)
                j_max = min(n - 1, peak_idx)
                for j in range(j_min, j_max + 1):
                    contribs[j] = f[j] * f[peak_idx - j]
                # Reduce top contributors, boost others
                top_k = max(5, n // 8)
                top_idx = np.argsort(contribs)[-top_k:]
                f[top_idx] *= np.exp(rng.normal(-0.08, 0.03, size=len(top_idx)))
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 3:
                    boost_idx = rng.choice(other, size=min(15, len(other)), replace=False)
                    f[boost_idx] *= np.exp(rng.normal(0.06, 0.03, size=len(boost_idx)))
                f = normalize_to_sqrt2n(f)
                current_score = score(f)
                stall_count = 0
                t = 0.1

            elif stall_count > 100:
                # Reset to random perturbation of best
                f = best_f * np.exp(rng.normal(0, 0.1, n))
                f = normalize_to_sqrt2n(f)
                current_score = score(f)
                stall_count = 0
                t = 0.15

        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, score={current_score:.6f}, "
                  f"best={best_score:.6f}, t={t:.4f}, stall={stall_count}, "
                  f"elapsed={elapsed:.1f}s")

    return best_f, best_score


def gradient_local_search(a0, time_limit, rng):
    """
    Gradient-informed local search with peak targeting.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sqrt2n(a0.astype(np.float64).copy())
    current_score = score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9995
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        candidate = a.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted peak reduction using gradient
            c = fft_convolve(a, a)
            max_c = np.max(c)
            # Find top peaks
            threshold = max_c * (1.0 - 0.05)
            peak_indices = np.where(c >= threshold)[0]

            # Compute subgradient at peak
            grad = np.zeros(n)
            for k in peak_indices[:10]:  # Limit to top 10 peaks
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                for j in range(j_min, j_max + 1):
                    idx = k - j
                    if idx == j:
                        grad[j] += 2.0 * a[j]
                    else:
                        grad[j] += a[idx]
            grad /= len(peak_indices[:10])

            # Gradient descent step with noise
            lr = 0.01 / (1.0 + iter_count * 0.001)
            candidate -= lr * grad
            candidate += rng.normal(0, 0.005, n) * a

        elif r < 0.55:
            # Block perturbation around peak
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            center = min(n - 1, max(0, (peak_idx) // 2))
            width = max(2, n // 10)
            start = max(0, center - width // 2)
            end = min(n, center + width // 2 + 1)
            candidate[start:end] *= np.exp(rng.normal(-0.02, 0.02, size=end - start))
            # Boost away from peak
            mask = np.ones(n, dtype=bool)
            mask[start:end] = False
            if np.sum(mask) > 5:
                boost_idx = rng.choice(np.where(mask)[0], size=min(20, np.sum(mask)), replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.02, 0.015, size=len(boost_idx)))

        elif r < 0.70:
            # Multi-scale perturbation
            for scale in [n // 4, n // 16, n // 64]:
                width = max(1, scale)
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.015, size=width))

        elif r < 0.85:
            # Pair redistribution (maintain sum)
            for _ in range(3):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    delta = rng.uniform(-0.2, 0.2) * a[min(a[i], a[j])]
                    candidate[i] = max(0, a[i] - delta)
                    candidate[j] = max(0, a[j] + delta)

        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0, 0.008, n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        candidate = normalize_to_sqrt2n(candidate)
        new_score = score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            temp = max(temp * 0.99, 0.0001)
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            a = candidate
            current_score = new_score

        temp *= cooling

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.2 / k
        freq = rng.uniform(0.2, 3.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_envelope_init(n, rng, decay=0.3):
    """Initialize with a decaying envelope from edges."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Inverse of triangular: higher at edges, lower at center
    env = 1.0 + decay * np.abs(t - center) / center
    a = env + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main optimization: combines smooth annealing, LP iteration,
    and gradient local search across multiple starting points and lengths.
    """
    t_global = time.monotonic()
    time_budget = max(60.0, float(budget_s) - 45.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try to use provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if len(start_seq) > 0:
                s = score(start_seq)
                print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
                if s < best_score:
                    best_score = s
                    best_seq = start_seq.copy()
        except Exception:
            start_seq = None

    # Define optimization tasks: (n, init_type, alpha)
    # Prioritize n=512 (from previous best) and nearby sizes
    tasks = []
    if start_seq is not None:
        n0 = len(start_seq)
        tasks.append((n0, 'provided', 0))
        # Resampled versions
        for n_try in [384, 512, 640, 768, 1024]:
            if n_try != n0:
                tasks.append((n_try, 'resample', 0))

    # Additional diverse starts
    for n in [512, 768, 1024]:
        tasks.append((n, 'chirp', 0.5))
        tasks.append((n, 'multi_tone', 0))
        tasks.append((n, 'envelope', 0.4))
        tasks.append((n, 'random', 0))

    n_tasks = len(tasks)
    print(f"Total tasks: {n_tasks}, time budget: {time_budget:.0f}s")

    for ti, (n, init_type, alpha) in enumerate(tasks):
        elapsed = time.monotonic() - t_global
        remaining = deadline - time.monotonic()

        # Stop if not enough time for a meaningful run
        if remaining < 80:
            print(f"Time low ({remaining:.0f}s), stopping task loop")
            break

        # Allocate time: more time for earlier tasks
        remaining_tasks = n_tasks - ti
        # Give primary tasks (provided, n=512) more time
        priority = 1.0
        if init_type == 'provided':
            priority = 2.0
        elif n == 512:
            priority = 1.5
        base_time = max(60.0, (remaining - 50.0) / remaining_tasks) * priority
        t_alloc = min(300.0, base_time)

        # Generate initial sequence
        if init_type == 'provided':
            a0 = start_seq.copy()
        elif init_type == 'resample':
            if best_seq is not None:
                a0 = resample_sequence(best_seq, n)
            else:
                a0 = make_chirp_init(n, 0.5, rng)
        elif init_type == 'chirp':
            a0 = make_chirp_init(n, alpha + 0.3 * rng.random(), rng)
        elif init_type == 'multi_tone':
            a0 = make_multi_tone_init(n, rng)
        elif init_type == 'envelope':
            a0 = make_envelope_init(n, rng, decay=alpha)
        else:
            a0 = 1.0 + 0.4 * rng.standard_normal(n)
            a0 = np.clip(a0, 0.05, None)

        a0 = normalize_to_sqrt2n(a0)
        init_score = score(a0)
        print(f"\n[Task {ti+1}/{n_tasks}] n={n}, init={init_type}, "
              f"init_score={init_score:.6f}, time_alloc={t_alloc:.0f}s")

        # Phase A: Smooth optimization
        t_smooth = t_alloc * 0.25
        if time.monotonic() < deadline - 30:
            a_smooth, s_smooth = smooth_optimize(a0, t_smooth, rng, verbose=False)
            if s_smooth < best_score:
                best_score = s_smooth
                best_seq = a_smooth.copy()
                print(f"  After smooth: {s_smooth:.6f}")
            a_current = a_smooth
        else:
            a_current = a0

        # Phase B: LP iteration
        t_lp = t_alloc * 0.50
        if time.monotonic() < deadline - 20:
            a_lp, s_lp = lp_iterate(a_current, t_lp, rng, verbose=(ti < 3))
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_lp.copy()
                print(f"  After LP: {s_lp:.6f} *** NEW BEST ***")
            a_current = a_lp
        else:
            a_current = a_smooth if s_smooth < init_score else a0

        # Phase C: Gradient local search
        t_local = max(10.0, t_alloc * 0.20)
        if time.monotonic() < deadline - 15:
            a_local, s_local = gradient_local_search(a_current, t_local, rng)
            if s_local < best_score:
                best_score = s_local
                best_seq = a_local.copy()
                print(f"  After local: {s_local:.6f} *** NEW BEST ***")

        # If we found a new best, try a quick LP polish on it
        if time.monotonic() < deadline - 20:
            t_polish = min(60.0, deadline - time.monotonic() - 15)
            if t_polish > 15 and best_seq is not None:
                a_pol, s_pol = lp_iterate(best_seq, t_polish, rng, verbose=False)
                if s_pol < best_score:
                    best_score = s_pol
                    best_seq = a_pol.copy()
                    print(f"  After LP polish: {s_pol:.6f} *** NEW BEST ***")

    # Final: intensive polish on the best sequence
    print(f"\n=== Final polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 60:
            # LP polish
            t_lp_final = remaining * 0.5
            a_fp, s_fp = lp_iterate(best_seq, t_lp_final, rng, verbose=True)
            if s_fp < best_score:
                best_score = s_fp
                best_seq = a_fp.copy()
                print(f"  Final LP: {s_fp:.6f}")

            # Local search
            remaining2 = deadline - time.monotonic() - 5
            if remaining2 > 20:
                a_fs, s_fs = gradient_local_search(best_seq, remaining2, rng)
                if s_fs < best_score:
                    best_score = s_fs
                    best_seq = a_fs.copy()
                    print(f"  Final local: {s_fs:.6f}")

    # Final verification
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0} ===")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with the actual evaluator
        actual_score = evaluate_sequence(final_list)
        print(f"Verified score: {actual_score:.6f}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
