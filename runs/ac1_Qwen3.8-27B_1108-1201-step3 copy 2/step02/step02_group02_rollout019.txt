Now let me also think about the **subgradient approach** as a complement.

The subgradient of max_k (a*a)[k] at point a is:

- The subgradient is (1/|S|) * sum_{k in S} grad_k where grad_k[i] = 2*a_safe(k-i)

We can use NAG (Nesterov Accelerated Gradient) with this subgradient.

The update would be:
1. a_new = a - lr * subgrad
2. a_new = clip(a_new, 0, ∞)
3. Normalize: a_new = a_new / sum(a_new) * S_target

With NAG:
1. z = a + (t-1)/(t+2) * (a - a_old)
2. a_new = z - lr * subgrad
3. a_new = clip(a_new, 0, ∞)
4. Normalize

The learning rate can be set by Polyak: lr = (f(a) - f*) / ||g||^2, but we don't know f*. We can use a heuristic: lr = f(a) / ||g||^2 * alpha for some alpha < 1.

    """Minimize max(base + d*x + e*x^2) / (S_prime + x)^2 over x in [lo, hi]."""
    g = (np.sqrt(5) - 1) / 2
    c = hi - g * (hi - lo)

        s = S_prime + x

            c = dpt
            fc = fd
            dpt = lo + g * (hi - lo)
            fd = F(dpt)

"""
Step function optimization via coordinate descent with golden section search,
Nesterov-accelerated subgradient, and LP-based refinement.
Targets minimization of 2n * max(autocorrelation) / (sum)^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute evaluation score (fast local version)."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_de(a, i):
    """Compute d[k] and e[k] arrays for coordinate i.
    d[k] = coefficient of x in (a*a)[k] when a[i] changes to x (linear term)
    e[k] = coefficient of x^2 (quadratic term, 1 at k=2i)
    """
    n = len(a)
    d = np.zeros(2 * n - 1)
    e = np.zeros(2 * n - 1)
    ki = np.arange(2 * n - 1) - i  # k - i for all k
    valid = (ki >= 0) & (ki < n)
    not_diagonal = (ki != i)  # k - i != i, i.e., k != 2i
    d[valid & not_diagonal] = 2.0 * a[ki[valid & not_diagonal]]
    if 0 <= 2 * i < 2 * n - 1:
        e[2 * i] = 1.0
    return d, e


def golden_section_min(base, d, e, S_prime, lo, hi, n_iter=40):
    """Minimize max(base + d*x + e*x^2) / (S_prime + x)^2 over x in [lo, hi]
    using golden section search."""
    g = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - g * (hi - lo)
    dpt = lo + g * (hi - lo)
    
    def F(x):
        conv = base + d * x + e * x * x
        s = S_prime + x
        if s < 0.01:
            return 1e15
        return np.max(conv) / (s * s)
    
    fc = F(c)
    fd = F(dpt)
    for _ in range(n_iter):
        if fc < fd:
            hi = dpt
            dpt = c
            fd = fc
            c = hi - g * (hi - lo)
            fc = F(c)
        else:
            lo = c
            c = dpt
            fc = fd
            dpt = lo + g * (hi - lo)
            fd = F(dpt)
    return (lo + hi) / 2.0


def coordinate_descent_pass(a, order=None, n_iter=35, warm_start=True):
    """One full pass of coordinate descent with golden section search."""
    n = len(a)
    if order is None:
        order = np.arange(n)
    
    full_conv = fft_convolve(a, a)
    S = np.sum(a)
    max_a = np.max(a)
    
    for i in order:
        x_old = a[i]
        S_prime = S - x_old
        if S_prime < 0.001:
            continue
        
        d, e = compute_de(a, i)
        base = full_conv - d * x_old - e * x_old * x_old
        
        # Determine search range

            lo = max(0.0, x_old - 1.5 * max_a)
            hi = x_old + 2.0 * max_a + 1.0
        else:
            lo = 0.0
            hi = 4.0 * max_a + 2.0

            S += delta
