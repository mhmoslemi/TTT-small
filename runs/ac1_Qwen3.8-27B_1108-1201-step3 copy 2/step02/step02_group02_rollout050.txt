"""
Optimization of non-negative step functions to minimize peak autocorrelation ratio.
Uses log-sum-exp smoothing with L-BFGS, subgradient descent, and coordinate descent.
Key insight: the smoothed objective's gradient is a weighted cross-correlation,
computable in O(n log n) via FFT.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve_full(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute the evaluation score (fast, local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = fft_convolve_full(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def smoothed_obj_and_grad(x, beta):
    """
    Compute smoothed objective and gradient.
    Objective: F = 2n * g / s^2 where g = (1/beta)*log(sum(exp(beta*c)))
    c = full convolution of a with itself, s = sum(a).
    """
    a = np.maximum(x, 0.0)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return 1e10, np.zeros(n)
    
    c = fft_convolve_full(a, a)
    
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    log_sum_exp_val = c_max + np.log(np.sum(np.exp(shifted)))
    g_val = log_sum_exp_val / beta
    
    obj = 2.0 * n * g_val / (s * s)
    
    # Weights for gradient
    w = np.exp(shifted)
    w /= np.sum(w)
    
    # Gradient of g w.r.t. a_k:
    # dg/da_k = 2 * sum_{q=0}^{n-1} w[q+k] * a[q]
    # This is cross-correlation of w (len 2n-1) and a (len n) at lag k.
    # Compute via FFT: (flip(w) * a) at position 2n-2-k
    w_flip = w[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    fw = np.fft.rfft(w_flip, size)
    fa = np.fft.rfft(a, size)
    conv_result = np.fft.irfft(fw * fa, size)
    
    # R[k] = conv_result[2n-2-k] for k=0..n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    indices = 2 * n - 2 - np.arange(n)
    dg = 2.0 * conv_result[indices]
    
    # Full gradient of F = 2n * g / s^2
    # dF/da_k = 2n * (s * dg_k - 2 * a_k * g_val) / s^3
    grad = 2.0 * n * (s * dg - 2.0 * a * g_val) / (s ** 3)
    
    return obj, grad


def subgradient_step(a, lr, peak_tol=1e-6):
    """
    One subgradient descent step on the hard peak objective.
    Returns updated a (projected onto {a>=0} and renormalized).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return a.copy()
    
    c = fft_convolve_full(a, a)
    c_max = np.max(c)
    
    # Find all near-peak positions
    peak_mask = c >= c_max * (1.0 - peak_tol)
    peak_indices = np.where(peak_mask)[0]
    
    # Subgradient: average over peak positions
    # For peak at position p: grad_k = 2 * a[p-k] if 0 <= p-k < n
    grad = np.zeros(n)
    for p in peak_indices:
        lo = max(0, p - n + 1)
        hi = min(n - 1, p)
        if lo <= hi:
            # grad_k = 2 * a[p-k] for k in [lo, hi]
            ks = np.arange(lo, hi + 1)
            grad[ks] += 2.0 * a[p - ks]
    grad /= len(peak_indices)
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = np.sum(a_new)
    if s_new < 0.01:
        return a.copy()
    # Renormalize to maintain same sum
    a_new = a_new * (s / s_new)
    return a_new


def coordinate_descent_polish(a, max_rounds=50, time_limit=None):
    """
    Coordinate descent: try increasing/decreasing each element.
    Only accept changes that improve the score.
    """
    n = len(a)
    current = a.copy()
    current_score = fast_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    step_sizes = [0.5, 0.1, 0.05, 0.01, -0.5, -0.1, -0.05, -0.01]
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        # Randomize the order
        order = np.random.permutation(n)
        
        for k in order:
            for ss in step_sizes:
                candidate = current.copy()
                candidate[k] = max(0.0, current[k] + ss * max(current[k], 0.01))
                new_score = fast_score(candidate)
                if new_score < current_score - 1e-9:
                    current = candidate
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    break  # Move to next coordinate
        
        if not improved:
            # Try smaller steps
            small_steps = [0.005, 0.002, -0.005, -0.002]
            improved2 = False
            for k in order:
                for ss in small_steps:
                    candidate = current.copy()
                    candidate[k] = max(0.0, current[k] + ss * max(current[k], 0.005))
                    new_score = fast_score(candidate)
                    if new_score < current_score - 1e-9:
                        current = candidate
                        current_score = new_score
                        improved2 = True
                        if current_score < best_score:
                            best_score = current_score
                            best = current.copy()
                        break
            if not improved2 and not improved:
                break
    
    return best, best_score


def simulated_annealing_polish(a_init, time_limit, rng):
    """Simulated annealing for local search."""
    current = a_init.astype(np.float64).copy()
    current_score = fast_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted: reduce peak contributors
            conv = fft_convolve_full(current, current)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                contribs[i] = current[i] * current[peak_idx - i]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.015, size=len(top_idx)))
            # Increase some others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.70:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Pair transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.12, 0.12) * current[i]
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = fast_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
            temperature = max(temperature * 0.999, 0.0001)
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
        iters += 1
        
        if iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA iter={iters}, current={current_score:.6f}, best={best_score:.6f}, "
                  f"temp={temperature:.6f}, elapsed={elapsed:.1f}s")
    
    return best, best_score


def lbgfs_smoothed_optimize(a0, time_limit, rng, beta_schedule=None):
    """
    L-BFGS optimization on the smoothed objective with increasing beta.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n)
        s = n
    a = a / s  # Normalize to sum=1
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    if beta_schedule is None:
        beta_schedule = [(5, 0.3), (20, 0.25), (80, 0.25), (300, 0.20)]
    
    total_beta_time = sum(t for _, t in beta_schedule)
    
    for beta, time_frac in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        t_beta = time_limit * time_frac
        
        def objective(x):
            val, _ = smoothed_obj_and_grad(x, beta)
            return val
        
        def gradient(x):
            _, g = smoothed_obj_and_grad(x, beta)
            return g
        
        # Run L-BFGS-B
        result = minimize(
            objective, a, jac=gradient,
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8,
                     'maxfun': 2000}
        )
        
        if result.x is not None:
            a_new = np.maximum(result.x, 0.0)
            s_new = np.sum(a_new)
            if s_new > 0.01:
                a_new = a_new / s_new
                score_new = fast_score(a_new)
                if score_new < best_score:
                    best_score = score_new
                    best_a = a_new.copy()
                a = a_new
        else:
            a = np.maximum(result.get('x', a), 0.0)
        
        elapsed = time.monotonic() - t0
        print(f"      beta={beta}, score_after={best_score:.6f}, elapsed={elapsed:.1f}s")
        
        if time.monotonic() - t0 >= time_limit:
            break
    
    return best_a, best_score


def subgradient_loop(a0, time_limit, rng):
    """Subgradient descent on the hard peak with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n)
        s = n
    a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    current_score = best_score
    t0 = time.monotonic()
    
    lr = 0.05
    stall = 0
    peak_tol = 1e-6
    
    while time.monotonic() - t0 < time_limit:
        # Try a step
        a_new = subgradient_step(a, lr, peak_tol)
        new_score = fast_score(a_new)
        
        if new_score < current_score - 1e-10:
            a = a_new
            current_score = new_score
            lr = min(lr * 1.05, 0.3)
            stall = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            # Try smaller step
            a_try = subgradient_step(a, lr * 0.3, peak_tol)
            try_score = fast_score(a_try)
            if try_score < current_score - 1e-10:
                a = a_try
                current_score = try_score
                lr = lr * 0.3
                stall = 0
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
            else:
                lr *= 0.8
                stall += 1
        
        if stall > 100:
            # Perturb to escape
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 0.01:
                a = a / s
            current_score = fast_score(a)
            stall = 0
            lr = 0.05
            peak_tol = 1e-4  # Widen peak tolerance to get more diverse subgradients
        elif stall > 40:
            peak_tol = max(1e-6, peak_tol * 0.5)
    
    return best_a, best_score


def make_diverse_inits(n, rng):
    """Create diverse initial sequences."""
    inits = []
    
    # Chirp-like
    t = np.arange(n, dtype=np.float64)
    a1 = 1.0 + 0.5 * np.cos(np.pi * t * t / n) + 0.1 * rng.standard_normal(n)
    inits.append(np.clip(a1, 0.01, None))
    
    # Triangle with noise
    a2 = 1.0 - 0.6 * np.abs(2.0 * t / n - 1.0) + 0.15 * rng.standard_normal(n)
    inits.append(np.clip(a2, 0.01, None))
    
    # Two humps
    a3 = 1.0 + 0.4 * np.cos(4.0 * np.pi * t / n) + 0.1 * rng.standard_normal(n)
    inits.append(np.clip(a3, 0.01, None))
    
    # Random
    a4 = 1.0 + 0.4 * rng.standard_normal(n)
    inits.append(np.clip(a4, 0.05, None))
    
    return inits


def resample_sequence(seq, new_n):
    """Resample sequence to new length via interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
