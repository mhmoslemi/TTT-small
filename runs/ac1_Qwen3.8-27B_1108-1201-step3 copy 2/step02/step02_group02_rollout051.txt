def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

def smooth_peak_and_grad(c, T):

    max_tc = np.max(T * c)

    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp  # softmax weights

- The term `j=k-i`: `a[k-i]*a[i] → a[k-i]*(a[i]+δ) = a[k-i]*a[i] + δ*a[k-i]`

"""
Optimized step function search using smoothed gradient descent (log-sum-exp),
L-BFGS-B, coordinate descent, and peak-directed local search.
Improves upon LP-based approach with faster gradient computation via FFT-based
softmax weighting and adaptive temperature scheduling.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def exact_score(a):
    """Compute the exact evaluation score (fast version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def smooth_peak_and_grad(c, T):
    """Return smoothed max of c and softmax weights."""
    max_tc = np.max(T * c)
    exp_shifted = np.exp(T * c - max_tc)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    L = (max_tc + np.log(sum_exp)) / T
    return L, p


def objective_scale_inv(a, T):
    """Scale-invariant smoothed objective: 2n * L / S^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e10
    c = fft_convolve(a, a)
    L, p = smooth_peak_and_grad(c, T)
    return 2.0 * n * L / (S * S), L, p, S


def grad_scale_inv(a, T):
    """Gradient of scale-invariant smoothed objective."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.ones(n)
    c = fft_convolve(a, a)
    L, p = smooth_peak_and_grad(c, T)
    # dL/da_i = (2/T) * sum_m p[m+i] * a[m]
    # Use direct computation (fast enough for n=512)
    corr = np.zeros(n)
    for i in range(n):
        corr[i] = np.dot(p[i:i + n], a)
    dL = (2.0 / T) * corr
    # df/da_i = 2n/S^2 * (dL_i - 2L/S)
    grad = (2.0 * n / (S * S)) * (dL - 2.0 * L / S)
    return grad


def lbgfsb_optimize(a0, T, maxiter=200, time_budget=60.0, verbose=False):
    """Run L-BFGS-B on the scale-invariant smoothed objective."""
    n = len(a0)
    x0 = np.clip(a0, 0.01, 1000.0).astype(np.float64)

    def func(x):
        x = np.maximum(x, 0.01)
        val, _, _, _ = objective_scale_inv(x, T)
        return val

    def grad_func(x):
        x = np.maximum(x, 0.01)
        return grad_scale_inv(x, T)

    t0 = time.monotonic()
    try:
        res = minimize(func, x0, jac=grad_func, method='L-BFGS-B',
                       bounds=[(0.01, 1000.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxfun': 5000, 'iprint': -1})
    except Exception as e:
        if verbose:
            print(f"      L-BFGS error: {e}")
        return a0, exact_score(a0)

    x_best = np.clip(res.x, 0.01, 1000.0)
    score = exact_score(x_best)
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      L-BFGS done: score={score:.6f}, niter={res.nit}, time={elapsed:.1f}s")
    return x_best, score


def gradient_descent_smooth(a0, T, time_budget=30.0, lr_init=0.01, verbose=False):
    """Smoothed projected gradient descent."""
    n = len(a0)
    a = np.clip(a0, 0.01, 1000.0).astype(np.float64).copy()
    a = a * (n / np.sum(a))
    current_score = exact_score(a)
    best_a = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    lr = lr_init
    stuck = 0

    while time.monotonic() - t0 < time_budget:
        grad = grad_scale_inv(a, T)
        # Project gradient to tangent space (sum to zero)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        grad = grad / grad_norm

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.01)
        a_new = a_new * (n / np.sum(a_new))

        new_score = exact_score(a_new)
        if new_score < current_score * (1 - 1e-10):
            a = a_new
            current_score = new_score
            stuck = 0
            lr = min(lr * 1.03, 0.05)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            stuck += 1
            lr *= 0.95
            if stuck > 80:
                break

    return best_a, best_score

    n = len(a0)
    a = np.clip(a0, 0.01, 1000.0).astype(np.float64).copy()
    a = a * (n / np.sum(a))
    current_score = exact_score(a)
    best_a = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    order = list(range(n))
    no_improve_count = 0

    while time.monotonic() - t0 < time_budget:
        rng.shuffle(order)
        pass_improved = False

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break

            best_local = current_score
            best_local_a = a.copy()

            # Try multiplicative adjustments
            for factor in [1 + step_size, 1 - step_size, 1 + step_size * 0.3, 1 - step_size * 0.3]:
                a_trial = a.copy()
                a_trial[i] = max(0.01, a[i] * factor)
                a_trial = a_trial * (n / np.sum(a_trial))
                s = exact_score(a_trial)
                if s < best_local:
                    best_local = s
                    best_local_a = a_trial.copy()

            if best_local < current_score:
                a = best_local_a
                current_score = best_local
                pass_improved = True
                no_improve_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                    if verbose and (int((time.monotonic() - t0) * 10) % 100 == 0):
                        print(f"        CD: score={current_score:.6f}, t={time.monotonic()-t0:.1f}s")
            else:
                no_improve_count += 1

        if not pass_improved:
            step_size *= 0.85
            if step_size < 0.001:
                break
            if no_improve_count > n * 3:
                break

    return best_a, best_score
