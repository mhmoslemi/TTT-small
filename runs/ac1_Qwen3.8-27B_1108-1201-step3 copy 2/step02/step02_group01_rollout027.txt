```python
def fft_convolve(a, b):

    m = len(b)
    size = 1
    while size < n + m - 1:

    fb = np.fft.rfft(b, size)

def subgradient_at_peak(a, peak_idx):
    """Compute the subgradient of conv(a,a)[peak_idx] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        k_minus_j = peak_idx - j
        if 0 <= k_minus_j < n:
            if 2 * j == peak_idx:
                g[j] = 2.0 * a[j]
            else:
                g[j] = 2.0 * a[k_minus_j]
    return g


def project_nonneg_sum(a, S):
    """Project onto {a >= 0, sum(a) = S} using water-filling."""
    n = len(a)
    # We want max(0, a - lambda) to sum to S
    # Sort a in descending order
    sorted_a = np.sort(a)[::-1]
    cumsum = np.cumsum(sorted_a)
    # Find lambda: sum of (a_i - lambda) for a_i > lambda = S
    # n*lambda = sum(a_i > lambda) - S
    # Try each possible "active set" size
    rho = np.arange(1, n + 1)
    # lambda(rho) = (sum of top rho elements - S) / rho
    lambdas = (cumsum[:n] - S) / rho
    # We need the largest rho such that lambda(rho) < sorted_a[rho-1]
    # (0-indexed: lambda(rho) < sorted_a[rho-1] means sorted_a[rho-1] > lambda)
    valid = lambdas < sorted_a
    # Find the correct rho
    # Actually, the standard water-filling algorithm:
    # Start with all elements, remove the smallest one that would go negative
    result = a.copy().astype(float)
    active = np.ones(n, dtype=bool)
    for _ in range(n):
        if not np.any(active):
            break
        remaining = S - np.sum(result[~active])
        n_active = np.sum(active)
        if n_active == 0:
            break
        lambda_val = (np.sum(result[active]) - remaining) / n_active
        result[active] -= lambda_val
        new_inactive = active & (result <= 0)
        if np.sum(new_inactive) == 0:
            break
        result[new_inactive] = 0
        active[new_inactive] = False
    result[~active] = 0
    # Ensure exact sum
    diff = S - np.sum(result)
    if np.sum(active) > 0:
        result[active] += diff / np.sum(active)
        result = np.maximum(result, 0)
    return result


def nesterov_subgradient(a, time_limit, verbose=False):
    """
    Nesterov-accelerated proximal subgradient method.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = S.
    """

    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_mc = max_conv(a)
    
    # Nesterov parameters
    y = a.copy()
    k = 1
    decay = 0.99  # Momentum decay
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(y, y)
        peak_idx = int(np.argmax(conv))
        M = conv[peak_idx]
        
        # Subgradient
        g = subgradient_at_peak(y, peak_idx)
        
        # Step size: alpha_k = 2 / (L * (k + 1)) where L ~ ||g||^2
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        alpha = 2.0 / (g_norm * (k + 1))
        # Cap the step
        alpha = min(alpha, 0.5 / n)
        
        # Update
        y_new = y - alpha * g
        y_new = project_nonneg_sum(y_new, S)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * k * k)) / 2
        beta = (k - 1) / t_new if k > 1 else 0
        a_new = y_new + beta * (y_new - y)
        a_new = project_nonneg_sum(a_new, S)
        
        y = a_new
        k += 1
        
        mc = max_conv(y)
        if mc < best_mc:
            best_mc = mc
            best_a = y.copy()
        
        if verbose and step % 1000 == 0:
            score = 2.0 * n * best_mc / (S * S)
            print(f"    Nesterov step {step}: score={score:.6f}, alpha={alpha:.6f}")
    
    return best_a

        M = np.max(conv_a)

        n_constraints = min(max_constraints, len(conv_a))
        active_idx = np.argsort(conv_a)[-n_constraints:]

        A_sub = A[active_idx, :]

        # (expensive but necessary)

        res = linprog(c, A_ub=A_sub, b_ub=b_sub,
                      bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        c_norm = normalize(g, S)
        
        # Binary search for max feasible t
        lo, hi = 0.0, 1.0
        for _ in range(12):
            mid = (lo + hi) / 2
            b = (1 - mid) * a + mid * c_norm
            b = np.clip(b, 0, None)
            b = normalize(b, S)
            if max_conv(b) <= M + 1e-9:
                lo = mid
            else:
                hi = mid
        t = lo
        
        if t < 0.001:
            break

        a = normalize(a, S)
        
        mc = max_conv(a)
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
```

```python
"""
Multi-method optimization for minimizing peak autocorrelation of non-negative
step functions. Combines: (1) LP-based refinement with active constraint
selection, (2) smooth log-sum-exp gradient descent with Nesterov momentum and
simplex projection, (3) simulated annealing with structured moves informed by
autocorrelation geometry, and (4) multi-length exploration. The log-sum-exp
approach provides a differentiable surrogate for the non-smooth max objective,
enabling efficient gradient-based progress between LP iterations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score: 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def max_conv(a):
    """Compute max of autocorrelation."""
    return float(np.max(fft_convolve(a, a)))


def normalize(a, S):
    """Normalize array to have sum S."""
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(a, S):
    """Project a onto the set {x >= 0, sum(x) = S} via water-filling."""
    n = len(a)
    sorted_a = np.sort(a)[::-1]
    cumsum = np.cumsum(sorted_a)
    rho = np.arange(1, n + 1, dtype=np.float64)
    condition = sorted_a > (cumsum - S) / rho
    valid = np.where(condition)[0]
    if len(valid) == 0:
        return np.full(n, S / n)
    idx = valid[-1]
    lambda_val = (cumsum[idx] - S) / (idx + 1)
    result = np.maximum(a - lambda_val, 0.0)
    diff = S - np.sum(result)
    if abs(diff) > 1e-12:
        mask = result > 0
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.maximum(result, 0.0)
    return result


def build_lp_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - j]
    return A


def lp_solve(a, M, n_constraints=None):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    if n_constraints is not None and n_constraints < len(a) * 2 - 1:
        conv_a = fft_convolve(a, a)
        active_idx = np.argsort(conv_a)[-n_constraints:]
        A_sub = A[active_idx, :]
        b_sub = np.full(len(active_idx), M)
    else:
        A_sub = A
        b_sub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_sub, b_ub=b_sub,
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_refine(a, time_limit, verbose=False):
    """LP-based iterative refinement with adaptive constraint count."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_mc = max_conv(a)
    stagnation = 0

    for iteration in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        # Use all constraints if n is small, otherwise top 300
        nc = n * 2 - 1 if n <= 200 else 300
        g = lp_solve(a, M, n_constraints=nc)
        if g is None:
            # Try full constraints
            g = lp_solve(a, M, n_constraints=None)
            if g is None:
                break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        c = normalize(g, S)

        # Binary search for maximum feasible t
        lo, hi = 0.0, 1.0
        for _ in range(12):
            mid = (lo + hi) / 2.0
            b = (1.0 - mid) * a + mid * c
            b = np.clip(b, 0.0, None)
            b = normalize(b, S)
            if max_conv(b) <= M + 1e-9:
                lo = mid
            else:
                hi = mid
        t = lo

        if t < 0.001:
            stagnation += 1
            if stagnation > 30:
                break
        else:
            stagnation = 0

        a = (1.0 - t) * a + t * c
        a = np.clip(a, 0.0, None)
        a = normalize(a, S)

        mc = max_conv(a)
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()

        if verbose and iteration % 50 == 0:
            score = 2.0 * n * best_mc / (S * S)
            print(f"    LP iter {iteration}: mc={best_mc:.6f}, score={score:.6f}, t={t:.4f}")

    return best_a


def soft_max_value(a, tau):
    """Compute tau * logsumexp(conv(a,a) / tau)."""
    conv = fft_convolve(a, a)
    conv_max = np.max(conv)
    shifted = conv - conv_max
    return tau * (np.log(np.sum(np.exp(shifted / tau))) + conv_max / tau)


def soft_max_gradient(a, tau):
    """Compute gradient of tau*logsumexp(conv(a,a)/tau) w.r.t. a."""
    n = len(a)
    conv = fft_convolve(a, a)
    conv_max = np.max(conv)
    p = np.exp((conv - conv_max) / tau)
    p = p / np.sum(p)

    # grad[j] = (2/tau) * sum_m p[j+m] * a[m] for m=0..n-1
    # = (2/tau) * conv(flip(p), a)[2n-2-j]
    flip_p = p[::-1]
    conv_pa = fft_convolve(flip_p, a)
    # conv_pa has length (2n-1) + n - 1 = 3n-2
    # We need indices 2n-2, 2n-3, ..., n-1
    grad = (2.0 / tau) * conv_pa[n - 1:2 * n - 1][::-1]
    return grad


def gradient_descent_smooth(a, time_limit, verbose=False):
    """
    Projected gradient descent on the log-sum-exp surrogate with decreasing tau
    and Nesterov-like momentum.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_mc = max_conv(a)

    # Initial tau based on the spread of the autocorrelation
    conv = fft_convolve(a, a)
    conv_max = np.max(conv)
    conv_min = np.min(conv)
    tau = max(0.1 * (conv_max - conv_min), 0.01 * conv_max)
    tau_initial = tau
    tau_min = 0.005 * conv_max

    # Step size
    lr = 0.1 / n
    momentum = 0.7
    velocity = np.zeros(n)

    step = 0
    while time.monotonic() - t0 < time_limit:
        if step % 200 == 0:
            # Decrease tau
            tau = max(tau * 0.95, tau_min)
            # Adaptive lr
            mc = max_conv(a)
            lr = max(0.5 / (n * np.sqrt(1 + step * 0.01)), 1e-6 / n)

        grad = soft_max_gradient(a, tau)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Normalize gradient for consistent step size
        grad = grad / (grad_norm + 1e-15)

        # Momentum update
        velocity = momentum * velocity + (1.0 - momentum) * grad
        a_new = a - lr * velocity
        a_new = project_simplex(a_new, S)

        mc_new = max_conv(a_new)
        if mc_new < best_mc:
            best_mc = mc_new
            best_a = a_new.copy()
        a = a_new
        step += 1

        if verbose and step % 2000 == 0:
            score = 2.0 * n * best_mc / (S * S)
            print(f"    GradSmooth step {step}: mc={best_mc:.6f}, score={score:.6f}, tau={tau:.4f}")

    return best_a


def simulated_annealing(a, time_limit, verbose=False):

    step = 0

        T = T_init * (T_final / T_init) ** frac
        step += 1

        # Choose move type

        if r < 0.25:

            candidate = a.copy()
            n_mod = rng.integers(3, max(4, n // 20))
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.08, size=n_mod))
            candidate = np.clip(candidate, 1e-6, None)
            candidate = normalize(candidate, S)

        elif r < 0.45:
            # Block perturbation
            candidate = a.copy()
            width = rng.integers(max(2, n // 60), max(3, n // 10))
            start = rng.integers(0, n - width)
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))
            candidate = np.clip(candidate, 1e-6, None)
            candidate = normalize(candidate, S)

        elif r < 0.65:
            # Peak-targeted move
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            candidate = a.copy()
            # Contributors to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(5, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            # Reduce top contributors slightly
            n_reduce = max(2, top_k // 2)
            red_idx = rng.choice(top_idx, size=n_reduce, replace=False)
            candidate[red_idx] *= np.exp(rng.normal(-0.04, 0.015, size=n_reduce))
            # Increase non-peak elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 2:
                n_inc = max(2, n // 25)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
            candidate = np.clip(candidate, 1e-6, None)
            candidate = normalize(candidate, S)

        elif r < 0.80:
            # Smoothened noise (low-frequency perturbation)
            candidate = a.copy()
            noise = rng.normal(0, 0.06, size=n)
            # Apply a simple low-pass via cumulative sum trick
            kernel_len = max(3, n // 40)
            kernel = np.ones(kernel_len) / kernel_len
            noise = np.convolve(noise, kernel, mode='same')
            candidate = a + noise
            candidate = np.clip(candidate, 1e-6, None)
            candidate = normalize(candidate, S)

        elif r < 0.90:
            # Two-swap move: reduce one group, boost another
            candidate = a.copy()
            n_half = n // 2
            perm = rng.permutation(n)
            red_set = perm[:n_half]
            inc_set = perm[n_half:]
            amt = rng.uniform(0.02, 0.06)
            candidate[red_set] *= (1.0 - amt)
            candidate[inc_set] *= (1.0 + amt)
            candidate = np.clip(candidate, 1e-6, None)
            candidate = normalize(candidate, S)

        else:
            # Mirror perturbation: perturb a and its reverse differently
            candidate = a.copy()
            amount = rng.uniform(0.02, 0.05)
            candidate[:n//2] *= (1.0 + amount * rng.normal(0, 0.5, size=n//2))
            candidate[n//2:] *= (1.0 - amount * rng.normal(0, 0.5, size=n - n//2))
            candidate = np.clip(candidate, 1e-6, None)
            candidate = normalize(candidate, S)

        new_score = 2.0 * n * max_conv(candidate) / (S * S)
        delta = new_score - current_score

        # Metropolis acceptance
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / max(T, 1e-12))

        if accept:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA t={elapsed:.0f}s: best={best_score:.6f}, curr={current_score:.6f}, T={T:.5f}")

    return best_a


def subgradient_refine(a, time_limit, verbose=False):
    """
    Subgradient method on the actual (non-smooth) objective with projection.
    Uses the peak lag subgradient with decreasing step size.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_mc = max_conv(a)

    k = 1
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        M = conv[peak_idx]

        # Subgradient of conv(a,a)[peak_idx] w.r.t. a
        g = np.zeros(n)
        for j in range(n):
            kmj = peak_idx - j
            if 0 <= kmj < n:
                if 2 * j == peak_idx:
                    g[j] = 2.0 * a[j]
                else:
                    g[j] = 2.0 * a[kmj]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Polyak-like step size
        mc_current = M
        alpha = 2.0 * (mc_current - best_mc) / (g_norm * g_norm + 1e-15)
        alpha = min(alpha, 1.0 / n)
        alpha = max(alpha, 1e-8 / n)

        a_new = a - alpha * g
        a_new = project_simplex(a_new, S)

        mc_new = max_conv(a_new)
        if mc_new < best_mc:
            best_mc = mc_new
            best_a = a_new.copy()
        a = a_new
        k += 1

        if verbose and k % 500 == 0:
            score = 2.0 * n * best_mc / (S * S)
            print(f"    Subgrad iter {k}: mc={best_mc:.6f}, score={score:.6f}, alpha={alpha:.6f}")

    return best_a


def interpolate_seq(a, n_new):
    """Linearly interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a)


def make_random_init(n, rng):
    """Create a random non-negative initialization."""
    t = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 5)
    if choice == 0:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(1, 5) * t / n + rng.uniform(0, 2*np.pi))
    elif choice == 1:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif choice == 2:
        center = rng.uniform(0.2, 0.8) * n
        sigma = rng.uniform(0.1, 0.3) * n
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif choice == 3:
        a = 1.0
        for _ in range(3):
            f = rng.integers(1, n // 6)
            a += 0.15 * np.cos(2 * np.pi * f * t / n + rng.uniform(0, 2*np.pi))
    else:
        a = np.ones(n)
        mask = rng.random(n) < 0.3
        a[mask] *= rng.uniform(0.3, 1.5, size=np.sum(mask))
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase, multi-length optimization to minimize the peak autocorrelation
    score of non-negative step functions.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = compute_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False

    # --- Get provided sequence ---
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided)}, score={compute_score(provided):.6f}")
        update_best(provided)

    # --- Define candidate lengths ---
    lengths = [640, 512, 768, 896]
    # Allocate time proportionally, giving more to the provided length
    time_alloc = {}
    remaining_time = deadline - time.monotonic() - 100
    time_alloc[640] = remaining_time * 0.40
    time_alloc[512] = remaining_time * 0.20
    time_alloc[768] = remaining_time * 0.20
    time_alloc[896] = remaining_time * 0.20

    print(f"\nTotal time budget: {remaining_time:.0f}s")
    for n, t in time_alloc.items():
        print(f"  n={n}: {t:.0f}s")

    # --- Main optimization loop over lengths ---
    for n in lengths:
        t_limit_total = time_alloc[n]
        if time.monotonic() > deadline - 60:
            break

        print(f"\n{'='*60}")
        print(f"Optimizing length n={n}, time_budget={t_limit_total:.0f}s")
        print(f"{'='*60}")

        # Initialize
        if provided is not None and n == len(provided):
            a0 = provided.copy()
        elif provided is not None:
            a0 = interpolate_seq(provided, n)
        else:
            a0 = make_random_init(n, rng)
        a0 = np.clip(a0, 0.01, None)

        # Split time among phases
        t_lp1 = t_limit_total * 0.25
        t_grad = t_limit_total * 0.30
        t_sa = t_limit_total * 0.25
        t_lp2 = t_limit_total * 0.10
        t_subgrad = t_limit_total * 0.10

        # Phase 1: LP refinement
        if time.monotonic() < deadline - 30:
            print(f"  Phase 1: LP refinement (limit={t_lp1:.0f}s)")
            a1 = lp_refine(a0, min(t_lp1, deadline - time.monotonic() - 30), verbose=True)
            if update_best(a1):
                print(f"    *** New best: score={best_score:.6f} ***")

        # Phase 2: Smooth gradient descent (log-sum-exp)
        if time.monotonic() < deadline - 30:
            print(f"  Phase 2: Smooth gradient descent (limit={t_grad:.0f}s)")
            a2 = gradient_descent_smooth(a1, min(t_grad, deadline - time.monotonic() - 30), verbose=True)
            if update_best(a2):
                print(f"    *** New best: score={best_score:.6f} ***")

        # Phase 3: Simulated annealing
        if time.monotonic() < deadline - 30:
            print(f"  Phase 3: Simulated annealing (limit={t_sa:.0f}s)")
            a3 = simulated_annealing(a2, min(t_sa, deadline - time.monotonic() - 30), verbose=True)
            if update_best(a3):
                print(f"    *** New best: score={best_score:.6f} ***")

        # Phase 4: Subgradient refinement
        if time.monotonic() < deadline - 20:
            print(f"  Phase 4: Subgradient (limit={t_subgrad:.0f}s)")
            a4 = subgradient_refine(a3, min(t_subgrad, deadline - time.monotonic() - 20), verbose=False)
            if update_best(a4):
                print(f"    *** New best: score={best_score:.6f} ***")

        # Phase 5: Final LP polish
        if time.monotonic() < deadline - 15:
            print(f"  Phase 5: Final LP (limit={t_lp2:.0f}s)")
            a5 = lp_refine(a4, min(t_lp2, deadline - time.monotonic() - 15), verbose=False)
            if update_best(a5):
                print(f"    *** New best: score={best_score:.6f} ***")

        print(f"  Done n={n}, best so far: {best_score:.6f}")

    # --- Final polish on the overall best ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        print(f"\nFinal polish: {remaining:.0f}s remaining")
        n_best = len(best_seq)
        # Try a bit more SA
        t_sa_final = remaining * 0.5
        t_lp_final = remaining * 0.5
        if t_sa_final > 5:
            a_final = simulated_annealing(best_seq, t_sa_final, verbose=False)
            if update_best(a_final):
                print(f"  SA polish improved to {best_score:.6f}")
        if t_lp_final > 5:
            a_final2 = lp_refine(best_seq, t_lp_final, verbose=False)
            if update_best(a_final2):
                print(f"  LP polish improved to {best_score:.6f}")

    # --- Final output ---
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = compute_score(best_seq)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
```
