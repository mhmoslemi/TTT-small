"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp approximation with adaptive temperature
2. Adam optimization in softmax (log-linear) parameterization  
3. Multi-start with diverse initializations
4. Multi-phase search: exploration -> exploitation -> polish
5. Final subgradient polish with true max

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_score_and_grad(z, T):
    """
    Compute smooth score and gradient w.r.t. log-parameters z.
    a = softmax(z), score ≈ 2*n*(1/T)*log(sum(exp(T*conv(a,a))))
    """
    n = len(z)
    
    # a = softmax(z)
    z_s = z - z.max()
    exp_z = np.exp(z_s)
    a = exp_z / exp_z.sum()
    
    # conv = autocorrelation
    conv = fast_convolve(a)
    
    # Softmax weights P on conv
    T_conv = T * conv
    T_max = T_conv.max()
    shifted = T_conv - T_max
    exp_shifted = np.exp(shifted)
    P = exp_shifted / exp_shifted.sum()
    
    # Smooth max (log-sum-exp)
    lse = (T_max + np.log(exp_shifted.sum())) / T
    score = 2.0 * n * lse
    
    # Gradient computation
    # h[m] = 2 * sum_j P[m+j] * a[j] (cross-correlation of P with a)
    N = 1
    while N < 3 * n:
        N <<= 1
    P_fft = np.fft.rfft(P, N)
    a_fft = np.fft.rfft(a, N)
    corr = np.fft.irfft(P_fft * np.conj(a_fft), N)
    h = 2.0 * corr[:n]
    
    # Chain through softmax: grad_z = 2n * (h*a - a*<h,a>)
    ha_dot = np.dot(h, a)
    grad_z = 2.0 * n * (h * a - a * ha_dot)
    
    return score, grad_z

def true_score_from_a(a):
    """Compute the true score from a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def adam_smooth_optimize(z0, time_limit, T_start, T_end, lr=0.05, 
                          beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization with log-sum-exp smooth objective."""
    n = len(z0)
    z = z0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best_score = float('inf')
    best_z = z.copy()
    
    t0 = time.monotonic()
    log_ratio = np.log(T_end / T_start)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * np.exp(progress * log_ratio)
        
        score, grad = smooth_score_and_grad(z, T)
        step += 1
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if score < best_score:
            best_score = score
            best_z = z.copy()
        
        # Periodic check of true score
        if step % 500 == 0:

            a = np.exp(z_s) / np.exp(z_s).sum()
            true_s = true_score_from_a(a)
            if true_s < best_score:
                best_score = true_s
                best_z = z.copy()
    
    return best_z, best_score

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_polish(a0, time_limit, top_k=50, lr_init=0.01, momentum=0.9):
    """Final polish using true subgradient method."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a)) / len(a)
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    lr = lr_init
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        # Get top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += a[k - idx]
        sg *= 2.0
        
        # Update
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= 0.9999
    
    return best_a, 2.0 * n * best_obj

    # Ramp

    # Bimodal

    # Gaussian

    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """
    Compute smooth score (log-sum-exp approx of max) and gradient w.r.t. z.
    a = softmax(z), score = 2*n*(1/T)*log(sum(exp(T*conv(a,a))))
    """
    n = len(z)

    # a = softmax(z)
    z_s = z - z.max()
    exp_z = np.exp(z_s)
    a = exp_z / exp_z.sum()

    # Convolution
    conv = fast_convolve(a)

    # Softmax weights on convolution
    T_conv = T * conv
    T_max = T_conv.max()
    shifted = T_conv
    ex
p_shifted = np.exp(shifted)
    P = exp_shifted / exp_shifted.sum()

    # Log-sum-exp
    lse = (T_max + np.log(exp_shifted.sum())) / T
    score = 2.0 * n * lse

    # Gradient: h[m] = 2 * cross_correlation(P, a)[m]
    N = 1
    while N < 3 * n:
        N <<= 1
    P_fft = np.fft.rfft(P, N)
    a_fft = np.fft.rfft(a, N)
    corr = np.fft.irfft(P_fft * np.conj(a_fft), N)
    h = 2.0 * corr[:n]

    # Chain rule through softmax
    ha_dot = np.dot(h, a)
    grad_z = 2.0 * n * (h * a - a * ha_dot)

    return score, grad_z


def true_score_from_a(a):
    """True objective for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def z_to_a(z):
    """Convert log-parameters to normalized sequence."""
    z_s = z - z.max()
    exp_z = np.exp(z_s)
    a = exp_z / exp_z.sum()
    return a


def a_to_z(a):
    """Convert normalized sequence to log-parameters."""
    return np.log(np.maximum(a, 1e-15)) - np.log(np.maximum(a, 1e-15)).mean()


def adam_optimize(z0, time_limit, T_start, T_end, lr=0.05,
                  beta1=0.9, beta2=0.999, eps=1e-8,
                  restart_every=2000, restart_scale=0.1):
    """
    Adam optimization with log-sum-exp smooth objective.
    Includes periodic random restarts to escape local minima.
    Returns (best_z, best_smooth_score).
    """
    n = len(z0)
    z = z0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best_score = float('inf')
    best_z = z.copy()
    no_improve = 0

    t0 = time.monotonic()
    log_ratio = np.log(max(T_end / T_start, 1.0))

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        # Slightly cyclical T to help escape local minima
        T_base = T_start * np.exp(progress * log_ratio)
        T = T_base

        score, grad = smooth_score_and_grad(z, T)
        step += 1

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if score < best_score - 1e-10:
            best_score = score
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart from best with small perturbation
        if no_improve >= restart_every:
            z = best_z + np.random.default_rng(step).standard_normal(n) * restart_scale / np.sqrt(n)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            step = 0

    return best_z, best_score


def lbfgs_refine(z0, T, maxiter=300, maxfun=5000):
    """L-BFGS refinement with fixed T."""
    def fun_and_grad(z):
        s, g = smooth_score_and_grad(z, T)
        return s, g

    try:
        result = minimize(fun_and_grad, z0, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxfun': maxfun,
                                   'ftol': 1e-14, 'gtol': 1e-8})
        return result.x, result.fun
    except Exception:
        return z0, float('inf')


def subgrad_polish(a0, time_limit, top_k=50, lr_init=0.008, momentum=0.92):
    """
    Subgradient polish using the true max of convolution.
    Works directly on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(len(a)) / len(a)
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    vel = np.zeros(n)
    lr = lr_init

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        # Get top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Multi-peak subgradient
        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += a[k - idx]
        sg *= 2.0

        # Momentum update + projection
        vel = momentum * vel - lr * sg
        a = project_simplex(a + vel)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        lr *= 0.9998

    return best_a, 2.0 * n * best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Multi-start Adam exploration (diverse inits, multiple lengths)
    Phase 2: Deep Adam + L-BFGS refinement on top candidates
    Phase 3: Subgradient polish with true max
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(1.0, total_budget - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Try to get parent sequence
    parent_a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.01, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_a = p / p.sum()
            print(f"Parent available: length={len(p)}")
    except Exception:
        pass

    # ============================
    # Phase 1: Exploration (35% of time)
    # ============================
    print("=== Phase 1: Multi-start Exploration ===")
    phase1_time = total_budget * 0.35
    phase1_deadline = time.monotonic() + phase1_time

    lengths = [128, 192, 256, 320, 384, 512, 768, 1024, 1536]
    all_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 15:
            break

        # Generate diverse initializations in z-space
        z_inits = {}

        # Uniform (z=0)
        z_inits['uniform'] = np.zeros(n)

        # Random
        z_inits['random'] = rng.standard_normal(n) * 0.4

        # From parent (resampled)
        if parent_a is not None:
            if len(parent_a) == n:
                z_inits['parent'] = a_to_z(parent_a)
            else:
                p_r = np.interp(np.linspace(0, 1, n),
                                np.linspace(0, 1, len(parent_a)), parent_a)
                p_r = np.maximum(p_r, 1e-10)
                z_inits['parent'] = a_to_z(p_r)

        # Ramp
        z_inits['ramp'] = np.linspace(-0.5, 0.5, n)

        # Gaussian bump
        x = np.arange(n, dtype=np.float64)
        g = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
        z_inits['gauss'] = np.log(np.maximum(g, 1e-10)) - np.log(np.maximum(g, 1e-10)).mean()

        # Bimodal (edge concentration)
        bi = np.ones(n) * 0.1
        edge = max(2, n // 4)
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        z_inits['bimodal'] = a_to_z(bi / bi.sum())

        # Select subset (prioritize parent)
        priority = ['parent', 'uniform', 'random', 'gauss', 'ramp', 'bimodal']
        init_names = [nm for nm in priority if nm in z_inits]
        max_per_length = 3
        init_names = init_names[:max_per_length]

        time_per = max(8.0, phase1_time / (len(lengths) * max_per_length))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 5:
                break

            t_alloc = min(time_per, phase1_deadline - time.monotonic() - 3)
            if t_alloc < 5:
                break

            z0 = z_inits[name]
            z_opt, _ = adam_optimize(z0, t_alloc, T_start=5.0, T_end=60.0,
                                     lr=0.06, restart_every=3000, restart_scale=0.15)
            a = z_to_a(z_opt)
            score = float(evaluate_sequence(a.tolist()))

            if np.isfinite(score):
                all_results.append((score, n, name, a.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    print(f"Phase 1 complete. Best: {best_score:.6f} (n={len(best_sequence) if best_sequence is not None else '?'})")

    # ============================
    # Phase 2: Deep optimization (50% of time)
    # ============================
    print("=== Phase 2: Deep Optimization ===")
    phase2_deadline = time.monotonic() + total_budget * 0.50

    # Gather top candidates (unique by length)
    all_results.sort(key=lambda x: x[0])
    candidates = []
    seen_lengths = set()
    for score, n, name, a in all_results:
        if n not in seen_lengths and score < best_score + 0.1:
            candidates.append((n, a.copy()))
            seen_lengths.add(n)
        if len(candidates) >= 6:
            break

    # Always include current best
    if best_sequence is not None:
        candidates.append((len(best_sequence), best_sequence.copy()))

    # Deduplicate by length
    final_candidates = []
    seen2 = set()
    for n_c, a_c in candidates:
        if n_c not in seen2:
            final_candidates.append((n_c, a_c))
            seen2.add(n_c)
    final_candidates = final_candidates[:6]

    for n_cand, a_cand in final_candidates:
        if time.monotonic() > phase2_deadline - 60:
            break
        n = n_cand
        z0 = a_to_z(a_cand)

        # Strategy A: Adam with high T (long run)
        if time.monotonic() < phase2_deadline - 40:
            t_alloc = min(120.0, phase2_deadline - time.monotonic() - 30)
            z_opt, _ = adam_optimize(z0, t_alloc, T_start=30.0, T_end=300.0,
                                     lr=0.03, restart_every=5000, restart_scale=0.08)
            a_opt = z_to_a(z_opt)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                z0 = z_opt
                print(f"  Adam(T=30->300) n={n}: score={score:.6f}")

        # Strategy B: L-BFGS with fixed high T
        if time.monotonic() < phase2_deadline - 20:
            maxiter = 200 if n <= 512 else 100
            z_lb, _ = lbfgs_refine(z0, T=150.0, maxiter=maxiter, maxfun=n * 10)
            a_opt = z_to_a(z_lb)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                z0 = z_lb
                print(f"  L-BFGS(T=150) n={n}: score={score:.6f}")

        # Strategy C: Another Adam with different T range
        if time.monotonic() < phase2_deadline - 10:
            t_alloc = min(60.0, phase2_deadline - time.monotonic() - 10)
            if t_alloc > 10:
                z_opt, _ = adam_optimize(z0, t_alloc, T_start=50.0, T_end=500.0,
                                         lr=0.02, restart_every=4000, restart_scale=0.05)
                a_opt = z_to_a(z_opt)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Adam(T=50->500) n={n}: score={score:.6f}")

    print(f"Phase 2 complete. Best: {best_score:.6f}")

    # ============================
    # Phase 3: Subgradient polish (15% of time)
    # ============================
    print("=== Phase 3: Subgradient Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        n_best = len(best_sequence)
        t_per_polish = remaining / 5.0

        polish_configs = [
            {'top_k': max(20, n_best // 6), 'lr_init': 0.006, 'momentum': 0.93},
            {'top_k': max(30, n_best // 4), 'lr_init': 0.004, 'momentum': 0.95},
            {'top_k': max(15, n_best // 8), 'lr_init': 0.008, 'momentum': 0.90},
            {'top_k': max(40, n_best // 3), 'lr_init': 0.003, 'momentum': 0.96},
            {'top_k': n_best, 'lr_init': 0.002, 'momentum': 0.97},
        ]

        for cfg in polish_configs:
            if time.monotonic() > deadline - 5:
                break
            t_alloc = min(t_per_polish, deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a_pol, _ = subgrad_polish(best_sequence, t_alloc, **cfg)
            score = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pol.copy()
                print(f"  Polish k={cfg['top_k']}: score={score:.6f}")

        # Final Adam polish with very high T
        if time.monotonic() < deadline - 10:
            z0 = a_to_z(best_sequence)
            t_alloc = deadline - time.monotonic() - 5
            if t_alloc > 5:
                z_opt, _ = adam_optimize(z0, t_alloc, T_start=200.0, T_end=1000.0,
                                         lr=0.01, restart_every=3000, restart_scale=0.03)
                a_opt = z_to_a(z_opt)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  High-T Adam polish: score={score:.6f}")

    # ============================
    # Final verification
    # ============================
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
