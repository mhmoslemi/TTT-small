    # soft peak value
    soft_peak = c_max + T * np.log(exp_c.sum())
    
    # Gradient: grad[i] = (1/T) * sum_k w[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] if 0 <= k-i < n (and by symmetry a[i] * a[k-i] contributes to both)
    # Actually for c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = a[k-i] + a[i] (if both valid)
    # Wait no: c[k] = sum_j a[j]*a[k-j]. Taking derivative w.r.t. a[i]:
    # dc[k]/da[i] = a[k-i] (from the a[j] term when j=i) + a[i] (from the a[k-j] term when k-j=i, i.e., j=k-i)
    # But if k-i = i (i.e., k=2i), these are the same term and we get 2*a[i]
    # More precisely: dc[k]/da[i] = a[k-i] + a[i] if k-i != i, and 2*a[i] if k-i == i
    # Which is just: a[k-i] + a[i] (counting a[i] once when k-i=i, but since both terms give a[i], it's 2*a[i])
    # Actually: c[k] = sum_j a[j]*a[k-j]. The terms involving a[i] are:
    #   j=i: a[i]*a[k-i]
    #   k-j=i => j=k-i: a[k-i]*a[i] (same term if k-i=i)
    # So dc[k]/da[i] = a[k-i] + a[i] if k-i != i, else 2*a[i]
    # In all cases: dc[k]/da[i] = a[k-i] + a[i] (since when k-i=i, a[k-i]+a[i] = 2*a[i]) ✓
    
    # So grad[i] = (1/T) * sum_k w[k] * (a[k-i] + a[i])
    # = (1/T) * (sum_k w[k]*a[k-i] + a[i]*sum_k w[k])
    # = (1/T) * (sum_k w[k]*a[k-i] + a[i])  [since sum_k w[k] = 1]
    
    # The first term is the cross-correlation of w and a
    # cross[i] = sum_k w[k] * a[k-i] (with a[j]=0 for j outside [0,n-1])
    
    # Build the matrix M where M[i,j] = w[i+j] for i,j in [0,n-1]
    # Then cross[i] = sum_j M[i,j] * a[j] = (M @ a)[i]
    # M is n x n, so this is O(n^2)
    
    idx = np.arange(n)
    M = w[np.arange(n)[:, None] + np.arange(n)[None, :]]
    cross = M @ a
    grad = (1.0 / T) * (cross + a)
    
    return soft_peak, grad


def grad_exact_peak(a):
    """Compute exact subgradient at the peak position."""
    n = len(a)
    c = fft_convolve(a, a)
    peak_val = c.max()
    # Average over all positions that are near the peak
    threshold = peak_val * (1.0 - 1e-6)
    peak_mask = c >= threshold
    peak_indices = np.where(peak_mask)[0]
    
    # Gradient: average of dc[k]/da over peak positions
    # dc[k]/da[i] = a[k-i] + a[i] (as derived above)
    grad = np.zeros(n)
    for k in peak_indices:
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            # For i in [i_min, i_max]: a[k-i] is valid
            # a[k-i] for i in [i_min, i_max] = a[k-i_max : k-i_min+1] reversed
            vals = a[k - i_max : k - i_min + 1][::-1]
            grad[i_min:i_max+1] += vals
        grad += a  # the +a[i] term
    
    grad /= len(peak_indices)
    return peak_val, grad


def flat_conv_init(n, rng):
    """Initialize with approximately flat convolution profile."""
    M = 1.0
    a = np.zeros(n)
    a[0] = np.sqrt(M)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k - i]
        a[k] = (M - s) / (2.0 * a[0])
        if a[k] < 0:
            a[k] = 0.01 * a[0]
    
    # Add small random perturbation for diversity
    a += 0.01 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    S = np.sqrt(2.0 * n)
    a = normalize_to_s(a, S)
    return a


def chirp_init(n, alpha, rng):
    """Initialize with a chirp-like pattern."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * (t * t) / (2.0 * n))
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.05, None)
    S = np.sqrt(2.0 * n)
    return normalize_to_s(a, S)


def multi_tone_init(n, rng, n_tones=5):
    """Initialize with multi-tone pattern."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, n_tones + 1):
        amp = 0.3 / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.05, None)
    S = np.sqrt(2.0 * n)
    return normalize_to_s(a, S)


def u_shape_init(n, alpha, rng):
    """Initialize with U-shape (high at ends, low in middle)."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.05, None)
    S = np.sqrt(2.0 * n)
    return normalize_to_s(a, S)


def peak_targeted_search(a, time_limit, rng):
    """Local search that specifically targets reducing the top peaks."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    current = a.copy()
    current_score = fast_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    step_size = 0.05
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        peak_val = c.max()
        
        # Find top contributing peaks
        top_k = min(10, len(c) // 4)
        peak_indices = np.argsort(c)[-top_k:]
        
        # For each peak, find the elements that contribute most
        all_reduces = []
        all_increases = []
        
        for k in peak_indices:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min > i_max:
                continue
            # Contribution of element i to peak k: a[i] * a[k-i] (times 2 if i != k-i)
            contribs = np.zeros(n)
            for i in range(i_min, i_max + 1):
                j = k - i
                contribs[i] = current[i] * current[j]
                if j != i and j >= 0 and j < n:
                    contribs[j] += current[i] * current[j]
            
            # Top contributors to reduce
            top_contrib = np.argsort(contribs)[-max(3, n // 20):]
            all_reduces.extend(top_contrib.tolist())
        
        # Elements to increase: those with low contribution to any peak
        peak_contrib_total = np.zeros(n)
        for k in peak_indices:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                peak_contrib_total[i] += current[i] * current[j]
        
        low_contrib = np.argsort(peak_contrib_total)[:max(3, n // 20)]
        all_increases.extend(low_contrib.tolist())
        
        # Make a perturbation
        candidate = current.copy()
        
        # Reduce high contributors
        if all_reduces:
            reduce_idx = list(set(all_reduces))[:max(5, n // 10)]
            deltas = step_size * rng.uniform(0.3, 1.0, size=len(reduce_idx))
            for idx, d in zip(reduce_idx, deltas):
                candidate[idx] *= (1.0 - d)
        
        # Increase low contributors
        if all_increases:
            inc_idx = list(set(all_increases))[:max(5, n // 10)]
            # Maintain approximate sum
            total_reduce = np.sum(candidate[i] for i in reduce_idx) - np.sum(current[i] for i in reduce_idx) if reduce_idx else 0
            per_inc = max(0, -total_reduce) / max(1, len(inc_idx))
            for idx in inc_idx:
                candidate[idx] += per_inc + step_size * 0.3 * current[idx] * rng.uniform(-0.5, 0.5)
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        candidate = normalize_to_s(candidate, S)
        
        new_score = fast_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
            step_size = min(0.08, step_size * 1.02)
        else:
            step_size *= 0.95
            if step_size < 0.005:
                # Try a random restart perturbation
                candidate = current * np.exp(rng.normal(0, 0.03, n))
                candidate = np.clip(candidate, 0.0, None)
                candidate = normalize_to_s(candidate, S)
                new_score = fast_score(candidate)
                if new_score < best_score:
                    best_score = new_score
                    best = candidate.copy()
                current = candidate
                current_score = new_score
                step_size = 0.04
    
    return best, best_score


def adam_soft_gradient_descent(a_init, time_limit, rng, n_t_start=2.0, n_t_end=0.3):
    """Adam optimizer with soft-max objective and temperature annealing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_s(a_init.copy(), S)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.02
    t_step = 0
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = fast_score(a)
    last_eval = 0.0
    stall_count = 0
    stall_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal temperature
        T = n_t_start * (n_t_end / n_t_start) ** progress
        T = max(T, 0.1)
        
        # Compute gradient
        soft_peak, grad = grad_soft_max_peak(a, T)
        
        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project: clip and renormalize
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_to_s(a_new, S)
        
        a = a_new
        
        # Periodic exact evaluation
        if time.monotonic() - last_eval > 0.5:
            last_eval = time.monotonic()
            score = fast_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall_count = 0
                stall_best = best_score
            else:
                stall_count += 1
            
            # Adaptive learning rate
            if stall_count > 10:
                lr *= 0.8
                if stall_count > 30:
                    # Perturb to escape
                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = np.clip(a, 0, None)
                    a = normalize_to_s(a, S)
                    lr = 0.01
                    stall_count = 0
        else:
            stall_count += 0.1
        
        # Adjust learning rate based on progress
        if t_step % 500 == 0 and t_step > 0:
            # Try to increase lr if making progress
            if stall_count < 5:
                lr = min(0.05, lr * 1.05)
    
    # Final evaluation
    score = fast_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def fine_tune(a_init, time_limit, rng):
    """Very fine-grained local search for final polish."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    current = a_init.copy()
    current_score = fast_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Pick 2-4 elements to perturb
        n_perturb = rng.integers(2, 5)
        idx = rng.choice(n, size=n_perturb, replace=False)
        
        candidate = current.copy()
        for i in idx:
            factor = np.exp(rng.normal(0, 0.01))
            candidate[i] *= factor
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        candidate = normalize_to_s(candidate, S)
        new_score = fast_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / 0.0005):
            current = candidate
            current_score = new_score
    
    return best, best_score


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining soft-gradient descent, peak-targeted
    local search, and fine-tuning. Uses diverse initializations including
    flat-convolution profiles to minimize peak autocorrelation.
    """
    t_global = time.monotonic()
    time_budget = max(60.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Collect initial candidates
    inits = []
    n_base = 512
    
    # Use provided sequence if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(provided) > 0:
            inits.append(("provided", provided.copy()))
            score_p = fast_score(provided)
            print(f"Provided sequence: n={len(provided)}, score={score_p:.6f}")
            if score_p < best_score:
                best_score = score_p
                best_seq = provided.copy()
    
    # Flat convolution initialization (key novelty)
    n_flat = n_base
    a_flat = flat_conv_init(n_flat, rng)
    score_flat = fast_score(a_flat)
    inits.append(("flat_conv", a_flat))
    print(f"Flat conv init: n={n_flat}, score={score_flat:.6f}")
    if score_flat < best_score:
        best_score = score_flat
        best_seq = a_flat.copy()
    
    # Chirp initialization
    a_chirp = chirp_init(n_base, 0.6, rng)
    score_chirp = fast_score(a_chirp)
    inits.append(("chirp", a_chirp))
    print(f"Chirp init: n={n_base}, score={score_chirp:.6f}")
    if score_chirp < best_score:
        best_score = score_chirp
        best_seq = a_chirp.copy()
    
    # U-shape initialization
    a_u = u_shape_init(n_base, 0.8, rng)
    score_u = fast_score(a_u)
    inits.append(("u_shape", a_u))
    print(f"U-shape init: n={n_base}, score={score_u:.6f}")
    if score_u < best_score:
        best_score = score_u
        best_seq = a_u.copy()
    
    # Multi-tone initialization
    a_mt = multi_tone_init(n_base, rng, n_tones=6)
    score_mt = fast_score(a_mt)
    inits.append(("multi_tone", a_mt))
    print(f"Multi-tone init: n={n_base}, score={score_mt:.6f}")
    if score_mt < best_score:
        best_score = score_mt
        best_seq = a_mt.copy()
    
    # Different length flat conv
    n_flat2 = 640
    a_flat2 = flat_conv_init(n_flat2, rng)
    score_flat2 = fast_score(a_flat2)
    inits.append(("flat_conv_640", a_flat2))
    print(f"Flat conv 640: n={n_flat2}, score={score_flat2:.6f}")
    if score_flat2 < best_score:
        best_score = score_flat2
        best_seq = a_flat2.copy()
    
    # Phase 1: Soft gradient descent on best initializations
    print("\n=== Phase 1: Soft gradient descent ===")
    t_start_p1 = time.monotonic()
    p1_time = min(700.0, deadline - t_start_p1 - 400.0)
    p1_time = max(60.0, p1_time)
    
    # Sort inits by score, take top 3
    inits.sort(key=lambda x: fast_score(x[1]))
    top_inits = inits[:min(3, len(inits))]
    
    n_p1_inits = len(top_inits)
    time_per_init = p1_time / n_p1_inits
    
    for name, a_init in top_inits:
        if time.monotonic() > deadline - 350:
            break
        t_limit = min(time_per_init, deadline - time.monotonic() - 300.0)
        if t_limit < 20:
            break
        
        print(f"  Starting GD on '{name}' (n={len(a_init)}), time={t_limit:.0f}s")
        result_a, result_score = adam_soft_gradient_descent(a_init, t_limit, rng)
        print(f"  '{name}' result: score={result_score:.6f}")
        
        if result_score < best_score:
            best_score = result_score
            best_seq = result_a.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 2: Additional GD runs with perturbed best
    print("\n=== Phase 2: Perturbed GD on best ===")
    t_start_p2 = time.monotonic()
    p2_time = min(300.0, deadline - t_start_p2 - 200.0)
    p2_time = max(30.0, p2_time)
    
    if best_seq is not None and p2_time > 30:
        n_perturb = 3
        time_per_pert = p2_time / n_perturb
        for pi in range(n_perturb):
            if time.monotonic() > deadline - 180:
                break
            t_lim = min(time_per_pert, deadline - time.monotonic() - 150.0)
            if t_lim < 15:
                break
            perturbation_scale = 0.05 * (pi + 1)
            a_pert = best_seq * np.exp(rng.normal(0, perturbation_scale, len(best_seq)))
            a_pert = np.clip(a_pert, 0, None)
            S = np.sqrt(2.0 * len(a_pert))
            a_pert = normalize_to_s(a_pert, S)
            
            result_a, result_score = adam_soft_gradient_descent(a_pert, t_lim, rng)
            print(f"  Perturbed {pi}: score={result_score:.6f}")
            if result_score < best_score:
                best_score = result_score
                best_seq = result_a.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 3: Peak-targeted local search
    print(f"\n=== Phase 3: Peak-targeted search (score={best_score:.6f}) ===")
    if best_seq is not None:
        t_start_p3 = time.monotonic()
        p3_time = min(250.0, deadline - t_start_p3 - 100.0)
        p3_time = max(20.0, p3_time)
        print(f"  Time budget: {p3_time:.0f}s")
        result_a, result_score = peak_targeted_search(best_seq, p3_time, rng)
        print(f"  Peak search result: score={result_score:.6f}")
        if result_score < best_score:
            best_score = result_score
            best_seq = result_a.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 4: Fine-tuning
    print(f"\n=== Phase 4: Fine-tuning (score={best_score:.6f}) ===")
    if best_seq is not None:
        t_start_p4 = time.monotonic()
        p4_time = min(150.0, deadline - t_start_p4 - 30.0)
        p4_time = max(10.0, p4_time)
        print(f"  Time budget: {p4_time:.0f}s")
        result_a, result_score = fine_tune(best_seq, p4_time, rng)
        print(f"  Fine-tune result: score={result_score:.6f}")
        if result_score < best_score:
            best_score = result_score
            best_seq = result_a.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 5: Try different lengths via resampling
    print(f"\n=== Phase 5: Length exploration (score={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for target_n in [384, 640, 768]:
            if time.monotonic() > deadline - 50:
                break
            a_resampled = resample_sequence(best_seq, target_n)
            S = np.sqrt(2.0 * target_n)
            a_resampled = normalize_to_s(a_resampled, S)
            score_r = fast_score(a_resampled)
            print(f"  Resampled to n={target_n}: score={score_r:.6f}")
            
            if score_r < best_score:
                # Optimize at this length
                t_lim = max(10.0, min(60.0, deadline - time.monotonic() - 30.0))
                result_a, result_score = adam_soft_gradient_descent(a_resampled, t_lim, rng)
                print(f"  GD at n={target_n}: score={result_score:.6f}")
                if result_score < best_score:
                    best_score = result_score
                    best_seq = result_a.copy()
                    print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 6: Final fine-tune if time permits
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            print(f"\n=== Phase 6: Final polish ({remaining:.0f}s) ===")
            result_a, result_score = fine_tune(best_seq, remaining, rng)
            if result_score < best_score:
                best_score = result_score
                best_seq = result_a.copy()
            print(f"  Final score: {best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 512
