    n = len(f)
    
    best_f = f.copy()

    patience = 0

    while time.time() - t0 < time_limit:
        b = fast_convolve(f)
        bmax = b.max()
        obj = 2*n*bmax
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                eta *= 0.9
                if patience > 500:
                    break
        
        # Find active set
        threshold = bmax - 1e-6 * bmax
        active = b >= threshold
        
        # Compute subgradient: average gradient over active positions
        sg = np.zeros(n)
        for k in np.where(active)[0]:
            # grad of b_k wrt f: 2*f[k-i] for valid k-i
            # i.e., sg[i] += 2*f[k-i] for i such that 0 <= k-i < n

            i_start = k - j_end
            i_end = k - j_start

        else:
            f_new /= s
        
        f = f_new
    
    return best_f, best_obj

Primal: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0
Dual: min M * sum(y_k) s.t. sum_k y_k * f_{k-i} >= 1 for all i, y >= 0

The dual says: find non-negative weights `y_k` on the convolution positions such that for each coordinate `i`, the total weight "seen" from position `i` is at least 1. The objective is M * sum(y_k), and the optimal primal value equals the optimal dual value.

- The primal player chooses `g` (the next step function shape)
- The dual player chooses `y` (the weights on convolution positions)
- The value of the game is the optimal sum(g)

- Maintain a distribution `p_i` over coordinates (the current `g`)
- The adversary chooses `y_k` proportional to the "violation" of constraint `k`
- Update `p_i` using the losses

Let me also add a key improvement: **use the dual certificate to guide the optimization**.

The dual of the LP "max sum(g) s.t. (f*g)_k <= M" is:
"min M * sum(y) s.t. sum_k y_k * f_{k-i} >= 1 for all i, y >= 0"

At the optimum, `sum(y_k) = 1/sum(g*)` and the dual variables `y_k` tell us which convolution positions are "tight". We can use this information to:
- Focus the mirror descent updates on the tight positions
- Detect when we're near the optimum (dual gap is small)

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive mirror descent with log-sum-exp smoothing, LP-based refinement,
and coordinate descent polishing. Explores multiple sequence lengths and
initializations to avoid local minima.

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_normalized(f):
    """Objective value assuming sum(f)=1."""
    n = len(f)
    b = fast_convolve(f)
    return 2.0 * n * b.max()


def compute_sg(f, w):
    """Compute weighted subgradient of max-convolution.
    f: sequence (length n), w: weights on conv positions (length 2n-1).
    Returns: gradient vector (length n).
    """
    n = len(f)
    sg = np.zeros(n)
    thresh = w.max() * 1e-4
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        i_start = k - j_end
        i_end = k - j_start
        sg[i_start:i_end + 1] += 2.0 * w[k] * f[j_start:j_end + 1][::-1]
    return sg


def mirror_descent(f0, time_limit, t_start=5.0, t_end=300.0, eta0=0.5, seed=42):
    """Adaptive mirror descent on the probability simplex with log-sum-exp smoothing."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(len(f), dtype=np.float64) / len(f)
    else:
        f /= s
    n = len(f)
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = obj_normalized(f)
    
    eta = eta0
    patience = 0
    iter_count = 0
    last_print = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        time_frac = min(1.0, elapsed / time_limit)
        
        # Temperature schedule: start low, increase to focus on peaks
        t_param = t_start * (t_end / t_start) ** (time_frac ** 0.7)
        
        b = fast_convolve(f)
        bmax = b.max()
        if bmax < 1e-15:
            break
        
        obj = 2.0 * n * bmax
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 100:
            eta *= 0.8
            if eta < 0.005:
                # Try a perturbation
                f_pert = f + rng.normal(0, 0.05, n) * f
                f_pert = np.maximum(f_pert, 0)
                sp = f_pert.sum()
                if sp > 1e-12:
                    f_pert /= sp
                    obj_p = obj_normalized(f_pert)
                    if obj_p < best_obj:
                        f = f_pert
                        best_obj = obj_p
                        best_f = f.copy()
                        patience = 0
                        eta = eta0 * 0.5
                    else:
                        break
                else:
                    break
            patience = 0
        
        # Compute log-sum-exp weights
        scaled = t_param * b / bmax
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws > 1e-15:
            w /= ws
        else:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        
        # Subgradient
        sg = compute_sg(f, w)
        sg_c = sg - sg.mean()
        
        # Mirror descent update with momentum-like effect
        # (Use a running average of the subgradient for smoothing)
        lf = -eta * sg_c
        lf -= lf.max()
        f_new = f * np.exp(lf)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            iter_count -= 1
            continue
        f = f_new / s
        
        if iter_count - last_print > 100:
            print(f"    MD iter={iter_count}, obj={best_obj:.6f}, t={t_param:.1f}, eta={eta:.4f}, pat={patience}, time={elapsed:.0f}s")
            last_print = iter_count
    
    return best_f, best_obj


def make_lp_matrix(n):
    """Build sparse row/col structure for cross-convolution constraint."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)_k <= M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, rows=None, cols=None):
    """LP-guided alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    if rows is None:
        rows, cols = make_lp_matrix(n)
    
    t0 = time.monotonic()
    t = 0.5
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        g = solve_lp(f, M, rows, cols, n, timeout=min(8, remaining * 0.5))
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-12:
            break
        g /= gs
        
        if np.max(np.abs(g - f)) < 1e-12:
            break
        
        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t, 0.05*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.2)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.7
            if patience > 30:
                break
    
    if iter_count % 20 == 0 or iter_count < 5:
        print(f"    LP iter={iter_count}, obj={best_obj:.6f}, t={t:.3f}")
    
    return best_f, best_obj


def coord_descent(f0, time_limit, max_sweeps=50):
    """Coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f /= f.sum()
    n = len(f)
    b = fast_convolve(f)
    best_f = f.copy()
    best_obj = 2.0 * n * b.max()
    
    t0 = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            old_val = f[i]
            if old_val < 1e-10:
                continue
            
            # Compute delta_b for changing a_i from old_val to x
            # Affected k: k = i + j for j in [0, n-1], i.e., k in [i, i+n-1]
            k_start = i
            k_end = min(2*n-2, i + n - 1)
            
            if k_start > k_end:
                continue
            
            k_arr = np.arange(k_start, k_end + 1)
            j_arr = k_arr - i  # in [0, n-1]
            b_aff = b[k_start:k_end+1].copy()
            
            # For k = 2i (if in range): quadratic term
            quad_k = 2 * i
            is_quad = (k_arr == quad_k)
            
            # Linear coefficient: 2*f[j] for non-quad, 0 for quad
            lin_coeff = 2.0 * f[j_arr]
            lin_coeff[is_quad] = 0.0
            
            # Constant shift: -2*old_val*f[j] for non-quad, -old_val^2 for quad
            shift = -2.0 * old_val * f[j_arr]
            shift[is_quad] = -old_val ** 2
            
            # b_k(x) = b_aff + shift + x * lin_coeff + (x^2 where quad)
            # Evaluate at candidate x values
            x_candidates = np.array([0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75,
                                      old_val, old_val * 1.25, old_val * 1.5, old_val * 2.0])
            
            best_x = old_val
            best_h = b.max()
            
            for x in x_candidates:
                b_new = b_aff.copy()
                b_new += shift + x * lin_coeff
                if np.any(is_quad):
                    b_new[is_quad] += x ** 2
                h = max(b_new.max(), (b[k_arr != 0]  # unchanged positions
                                      if len(k_arr) < len(b) else np.array([b.max() * 0])).max() if len(k_arr) < len(b) else b_new.max())
                # Simpler: just compute the max over affected positions and compare to max of unaffected
                max_aff = b_new.max()
                if len(k_arr) < len(b):
                    # Max of unaffected positions
                    mask = np.ones(len(b), dtype=bool)
                    mask[k_start:k_end+1] = False
                    max_unaff = b[mask].max() if mask.any() else 0
                    h = max(max_aff, max_unaff)
                else:
                    h = max_aff
                
                if h < best_h:
                    best_h = h
                    best_x = x
            
            if best_x != old_val:
                f[i] = best_x
                improved = True
                # Update b
                # Recompute b (expensive but needed for correctness)
                # Actually, we can update b incrementally:
                delta = best_x - old_val
                for k in range(k_start, k_end + 1):
                    j = k - i
                    if k == 2*i:
                        b[k] += best_x**2 - old_val**2
                    else:
                        b[k] += 2.0 * delta * f[j]
                # But wait, f[j] might have changed if j == i... 
                # Actually no, we only changed f[i], and when j == i, k = 2i which is the quad case.
                # For non-quad cases, f[j] is unchanged (j != i).
                # But we need to be careful: the update formula uses the CURRENT f[j],
                # which is correct since we only changed f[i].
        
        # Check improvement
        if improved:
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-12:
                f = best_f.copy()
            else:
                f /= s
            b = fast_convolve(f)  # Recompute for accuracy
            obj = 2.0 * n * b.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
        
        if not improved:
            break
    
    return best_f, best_obj


def gen_inits(n, rng, parent=None):

        inits[f'g{int(sf*100)}'] = g

    # Cosine

    # Bi-peak

    # Alternating (high-low pattern)

    if parent is not None:
