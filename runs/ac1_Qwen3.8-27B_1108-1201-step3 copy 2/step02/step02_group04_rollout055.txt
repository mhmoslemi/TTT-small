def fast_autoconv(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def subgrad_step(a, k_active, eta):
    """Take a subgradient step to reduce max convolution."""
    n = len(a)
    # Compute subgradient (average over active lags)
    g = np.zeros(n)
    for k in k_active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * a[j]
    g /= len(k_active)
    
    # Projected gradient step
    a_new = a - eta * g
    # Project onto simplex {a >= 0, sum(a) = 1}
    a_new = project_simplex(a_new)
    return a_new

    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    tau = tau_init
    best_a = a.copy()
    best_obj = 2*n*fast_autoconv(a).max()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autoconv(a)
        max_conv = conv.max()
        obj = 2*n*max_conv
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
        
        # Compute soft gradient
        shifted = tau * conv - tau * max_conv  # for numerical stability
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        i_idx = np.arange(n)
        M = w[i_idx[:,None] + i_idx[None,:]]
        grad = 2.0 * (M @ a)
        
        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        a_new = project_simplex(a_new)
        a = a_new
        
        # Anneal temperature
        tau = max(tau_min, tau * tau_decay)
    
    return best_a, best_obj

One more thought: the coordinate descent refiner. Here's a simple but effective approach:

1. Find the peak lag k*
2. Find the index i* that maximizes `a[i*] * a[k*-i*]` (largest contributing pair)
3. Try transferring δ from a[i*] to a random position p
4. If the max conv decreases, accept; otherwise, try another p
5. Repeat

This is O(n) per trial (to check if max conv decreases, assuming we only need to check a few lags).

Actually, checking if max conv decreases after a change requires recomputing the full convolution, which is O(n log n) with FFT. Or we can do a quick check by only looking at the affected lags.

When we change a[i*] -> a[i*] - δ and a[p] -> a[p] + δ:
- conv[k] changes by `δ * (2*a[k-p] - 2*a[k-i*])` (first order, approximately)
- We need this to be negative for k = k* (to reduce the peak)
- And we need it to not be too positive for other k

For k = k*: change = `δ * 2 * (a[k*-p] - a[k*-i*])`. We want this < 0, so `a[k*-p] < a[k*-i*]`.

So we want to find p such that `a[k*-p]` is small (ideally 0). This means p should be far from k* in the sense that `k*-p` is an index where a is small.

If a is mostly concentrated in some region, and k*-i* is in that region, then we want p such that k*-p is outside that region.

This gives a principled way to choose the transfer target!

"""
Minimizes 2*n*max(autoconv(a))/(sum(a))^2 using softmax-annealing optimization
with Adam, simplex projection, coordinate descent refinement, and LP kicks.
Explores multiple initializations and sequence lengths.
"""

import numpy as np
import time
import scipy.optimize as optimize
import scipy.sparse as sparse

def fast_autoconv(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    result /= result.sum()  # ensure exact normalization
    return result

def make_softgrad_matrix(n):
    """Precompute index matrix for soft gradient computation."""
    i_idx = np.arange(n)
    return i_idx[:, None] + i_idx[None, :]

def compute_softgrad(a, w, idx_matrix):
    """Compute gradient of softmax-softmax objective."""
    M = w[idx_matrix]
    return 2.0 * (M @ a)

def softmax_anneal(a0, time_limit, tau_init=0.5, tau_end=0.01, lr=0.1, 
                   beta1=0.9, beta2=0.999, eps=1e-8):
    """Optimize using softmax-annealing with Adam."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    idx_matrix = make_softgrad_matrix(n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_init
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autoconv(a).max()
    
    t0 = time.monotonic()
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autoconv(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if stagnation > 2000:
            # Reset with different tau
            tau = max(tau_init * 0.5, tau_end * 5)
            m *= 0.5
            v *= 0.5
            stagnation = 0
        
        # Softmax weights
        tau_shifted = tau * (conv - max_conv)
        w = np.exp(tau_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        grad = compute_softgrad(a, w, idx_matrix)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        a = project_simplex(a_new)
        
        # Anneal tau
        progress = min(1.0, (time.monotonic() - t0) / max(time_limit, 1))
        tau = tau_init * (tau_end / tau_init) ** progress
        # Or: tau = max(tau_end, tau * 0.999)
    
    return best_a, best_obj

def coord_descent_refine(a0, time_limit):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    a /= s
    n = len(a)
    
    best_a = a.copy()
    conv = fast_autoconv(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autoconv(a)
        max_conv = conv.max()
        k_star = int(np.argmax(conv))
        
        # Find the largest contributing pair for lag k_star
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        valid_i = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        products = a[valid_i] * a[k_star - valid_i]
        i_star = valid_i[np.argmax(products)]
        j_star = k_star - i_star
        
        # Try several transfer moves
        improved = False
        for attempt in range(20):
            # Choose target position: prefer positions where a[k_star - p] is small
            # Random candidate
            p = rng.integers(0, n)
            if p == i_star or p == j_star:
                continue
            
            # Determine transfer amount
            delta = a[i_star] * 0.1  # transfer 10% of a[i_star]

                conv = conv_trial
