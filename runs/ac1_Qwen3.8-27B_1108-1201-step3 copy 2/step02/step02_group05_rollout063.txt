Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed
gradient descent with continuation, FFT-based gradient computation,
and active-set local refinement.

import numpy as np

import time

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_obj_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    F(a) = (1/tau) * log(sum_k exp(tau * (a*a)[k]))
    grad(a) = 2 * cross_correlation(a, w) where w_k = exp(tau*(conv[k]-max)) / sum(...)
    
    Assumes a >= 0. Does NOT assume sum(a) = 1.
    Returns (F, grad).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Stable log-sum-exp
    max_conv = conv.max()
    shifted = tau * (conv - max_conv)
    log_sum = max_conv + np.log(np.sum(np.exp(shifted)))
    F = log_sum / tau
    
    # Weights
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: grad[p] = 2 * sum_{j=0}^{n-1} a[j] * w[p+j]
    # Compute via FFT cross-correlation
    # grad[p] / 2 = sum_j a[j] * w[p+j]
    # This is a cross-correlation. Let's compute it directly for clarity.
    # For n=512, direct computation is O(n^2) = 262144, which is fast enough.
    # But let's use FFT for larger n.
    
    # Method: pad w to length 3n-2, compute convolution with reversed a
    # sum_j a[j] * w[p+j] = (a_rev * w)[p + n - 1] where a_rev[j] = a[n-1-j]
    
    a_rev = a[::-1]
    # Linear convolution of a_rev (length n) and w (length 2n-1)
    # Result has length n + (2n-1) - 1 = 3n-2
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_cross = np.fft.irfft(A_rev * W, fft_size)[:conv_size]
    
    # grad[p] / 2 = conv_cross[p + n - 1] for p = 0, ..., n-1
    grad = 2.0 * conv_cross[n-1 : n-1+n]
    
    return F, grad

    # But since F is scale-invariant (F(a_norm) = F(params)), the gradient w.r.t. 
    # the scale direction should be zero. Let's verify:
    # sum_j a_norm[j] * grad[j] = sum_j a[j]/s * 2*sum_k a[k]*w[j+k]/s ... 
    # This is getting complicated. Let me just use the normalized version directly.
    
    # Simpler: just return F and the gradient with respect to a_norm,
    # and let the optimizer handle the normalization externally.
    # Actually, the cleanest approach: work with a directly (sum=1 enforced by projection).
    
    return F, grad

def optimize_smoothed(a0, tau_schedule, max_iters_per_tau=200):

    (Note: since sum(a)=1, this is already the scale-invariant version
     up to the factor (sum a)^2 = 1.)
    
    Actually, with sum(a) = 1, the objective is 2n * max(conv(a,a)).
    The smoothed version is (1/tau) * log(sum_k exp(tau * (a*a)[k])).
    
    grad_a = 2 * cross_corr(a, w) as before.
    grad_b[i] = (grad_a[i] - sum_j a[j]*grad_a[j]) * a[i]
              = (grad_a[i] - <a, grad_a>) * a[i]
    (This is the softmax backprop: dF/db = a * (grad_a - <a, grad_a>))
    """
    # Compute a from b
    b_shifted = b - b.max()  # for numerical stability
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum()
    
    # Compute smoothed objective and gradient w.r.t. a
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    shifted = tau * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    F = (max_conv + np.log(exp_shifted.sum())) / tau
    
    # grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # (as derived earlier)
    a_rev = a[::-1]
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_cross = np.fft.irfft(A_rev * W, fft_size)[:conv_size]
    grad_a = 2.0 * conv_cross[n-1: n-1+n]
    
    # Backprop through softmax: grad_b = a * (grad_a - dot(a, grad_a))
    a_dot_grad = np.dot(a, grad_a)
    grad_b = a * (grad_a - a_dot_grad)
    
    return F, grad_b

Wait, that's the Hessian of the unconstrained quadratic. The Hessian is a "reversal" matrix scaled by 2.

For the constrained problem (sum a = 1, a >= 0), the KKT conditions give:
H * a - nu * 1 - mu = 0 (for a > 0)
where nu is the Lagrange multiplier for the sum constraint.

So: 2*a[k*-j] - nu = 0 for all j with a[j] > 0.
This gives a[k*-j] = nu/2 = constant for all j in the support.

So at the optimum (ignoring non-negativity), a is constant on the "orbits" of the map j -> k*-j. The orbits are: {j, k*-j} for j != k*/2, and {k*/2} (if k* is even).

This means:
- For each pair {j, k*-j}: a[j] = a[k*-j] = c (some constant).
- For the self (if k* even): a[k*/2] = c (same constant).
- Indices not involved in any pair (i.e., k*-j is out of [0, n-1] for all j in support... wait, that's not right).

Actually, the constraint is: for j with a[j] > 0, we need a[k*-j] = nu/2. This means:
- If j is in the support, then k*-j must also be in the support (unless a[j] = 0).
- And a[j] = a[k*-j] = nu/2.

The "free" indices are those p where k*-p is not in [0, n-1]. For these, the derivative d(a*a)[k*]/d(a[p]) = 0, so they don't appear in the KKT condition. We can put arbitrary mass on them.

So the optimal solution for a single active lag k* is:
- Put mass c on each "pair" {j, k*-j} where both j and k*-j are in [0, n-1] and j != k*-j.
- Put mass c on the "self" k*/2 (if valid).
- Put any remaining mass on "free" indices (where k*-p is out of range).

The total mass on the paired/self indices is: c * (number of paired indices + number of self indices).
The remaining mass (1 - c * n_pairs) goes to free indices.

But we also need to minimize the total, which means we want to MINIMIZE the mass on the paired indices (since each paired index contributes to (a*a)[k*]). We want c as small as possible, i.e., we want to put as much mass as possible on free indices.

If there are F free indices and P paired/self indices (total), then:
- Minimum mass on paired: 0 (if F > 0, put all mass on free).
- Then (a*a)[k*] = 0.

But this is only optimal for the single-lag problem. For the full problem (all lags), we need to balance.

    """
    Compute smoothed log-sum-exp objective and gradient w.r.t. b (softmax params).
    a = softmax(b), then F = (1/tau)*log(sum exp(tau*(a*a)[k])).
    """
    n = len(b)
    # Softmax
    b_max = b.max()
    exp_b = np.exp(b - b_max)
    a = exp_b / exp_b.sum()
    
    # Convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp with stability

    shifted = tau * (conv - max_c)

    # Cross-correlation via FFT

def coordinate_descent(a, time_limit=60.0):
    """Fast coordinate descent: try small perturbations at random indices."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a /= s
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        for frac in [0.02, 0.05, 0.1, 0.2, -0.02, -0.05, -0.1, -0.2]:
            amount = a[i] * frac
            if amount < -a[i] * 0.95:
                amount = -a[i] * 0.95
            if amount > a[i]:
                amount = a[i]
            a_t = a.copy()
            a_t[i] -= amount
            a_t[j] += amount
            a_t = np.maximum(a_t, 0.0)
            ts = a_t.sum()
            if ts < 1e-12:
                continue
            a_t /= ts
            o = float(evaluate_sequence(a_t.tolist()))
            if np.isfinite(o) and o < best_score - 1e-10:
                best_score = o
                best_a = a_t.copy()
                a = a_t.copy()
                break
    return best_a, best_score
