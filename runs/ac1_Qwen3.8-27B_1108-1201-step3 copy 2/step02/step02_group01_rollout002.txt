"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Combines LP-based blending, smooth power-sum optimization (L-BFGS-B),
and stochastic local search with simulated annealing.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def autocorr(a):
    """Compute linear autocorrelation."""
    return fft_convolve(a, a)

def compute_score(a):
    """Compute the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = autocorr(a)
    return float(2.0 * n * np.max(conv) / (s * s))

def normalize_sum(a, target_sum):
    """Normalize sequence to have target sum."""
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = (a * g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_blend_step(a, M, verbose=False):
    """One LP blending step. Returns new a and the blend parameter used."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_sum(a, target_sum)
    
    g = solve_lp(a, M)
    if g is None:
        # Fallback: use a itself
        return a, 0.0
    
    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return a, 0.0
    
    g = normalize_sum(g, target_sum)
    
    # Find best blend parameter
    best_t = 0.0
    best_conv = M + 1e-10
    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]:
        blend = (1 - t) * a + t * g
        blend = np.clip(blend, 0.0, None)
        blend = normalize_sum(blend, target_sum)
        c_max = np.max(autocorr(blend))
        if c_max <= M + 1e-10:
            best_t = t
            best_conv = c_max
            break
    
    if best_t == 0.0:

            blend = np.clip(blend, 0.0, None)

            if c_max <= M + 1e-10:

def power_sum_objective(x, n, p, target_sum):
    """Smooth objective using power sum of autocorrelation values."""
    a = np.clip(x, 0.0, None)
    a = normalize_sum(a, target_sum)
    conv = autocorr(a)
    # Power sum as smooth max
    ps = np.sum(conv ** p) ** (1.0 / p)
    return 2.0 * n * ps / (target_sum * target_sum)

def power_sum_gradient(x, n, p, target_sum):
    """Numerical gradient of the power sum objective."""
    eps = 1e-6
    f0 = power_sum_objective(x, n, p, target_sum)
    grad = np.zeros(len(x))
    for i in range(len(x)):
        x_pert = x.copy()
        x_pert[i] += eps
        grad[i] = (power_sum_objective(x_pert, n, p, target_sum) - f0) / eps
    return grad

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines LP-based blending with bisection,
smooth power-sum L-BFGS-B optimization, multi-peak subgradient descent,

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def autocorr(a):
    """Compute linear autocorrelation."""
    return fft_convolve(a, a)


def compute_score(a):
    """Compute the score 2*n*max(conv)/sum^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = autocorr(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize_sum(a, target_sum=None):
    """Normalize sequence to have target sum (default sqrt(2n))."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = (a * g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M, min_g_bound=0.0):
    """Solve LP: max sum(g) + min_g_bound * t s.t. conv(a,g) <= M, g >= t, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    if min_g_bound > 0:
        # Add variable t with g[i] >= t
        # max sum(g) + min_g_bound * t
        # s.t. A@z <= M (where z = g)
        #      g[i] - t >= 0  =>  -g[i] + t <= 0
        #      t >= 0, g >= 0
        c = np.concatenate([-np.ones(n), [-min_g_bound]])
        A_ub = A
        b_ub = np.full(2 * n - 1, M)
        # Add constraints -g[i] + t <= 0
        A_extra = np.zeros((n, n + 1))
        for i in range(n):
            A_extra[i, i] = -1.0
            A_extra[i, n] = 1.0
        A_ub = np.vstack([A_ub, A_extra])
        b_ub = np.concatenate([b_ub, np.zeros(n)])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
        return None
    else:
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            return res.x
        return None

def power_sum_obj(x, n, p, target_sum):
    """Power-sum smooth approximation of the objective."""
    a = np.clip(x, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-10:

    a = a * (target_sum / s)
    conv = autocorr(a)
    # Use power sum as smooth max
    p_conv = np.sum(conv ** p) ** (1.0 / p)
    return 2.0 * n * p_conv / (target_sum * target_sum)

        t_start = time.monotonic()
        
        def obj(x):
            return power_sum_obj(x, n, p, target_sum)
        
        try:
            result = minimize(obj, best_a, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'maxfun': 50000,
                                       'ftol': 1e-14, 'gtol': 1e-10})
            new_a = np.clip(result.x, 0.0, None)
            new_a = normalize_sum(new_a, target_sum)
            new_score = compute_score(new_a)
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                if verbose:
                    print(f"    Smooth p={p}: score={best_score:.6f} ({time.monotonic()-t_start:.1f}s)")
        except Exception:
            continue
    
    return best_a


def multi_peak_subgradient_optimize(a, time_limit, verbose=False, epsilon=0.02):
    """Subgradient descent using multiple peaks."""
    t0 = time.monotonic()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_sum(a, target_sum)
    best_a = a.copy()
    best_score = float(np.max(autocorr(a)))
    
    lr = 1.0 / n
    step = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = autocorr(a)
        peak_val = float(np.max(conv))
        
        # Find all peaks within epsilon of max
        threshold = peak_val * (1.0 - epsilon)
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        
        # Compute subgradient (average over active peaks)
        grad = np.zeros(n)
        for k in peak_indices:
            for j in range(n):
                i1 = k - j
                if 0 <= i1 < n:
                    grad[j] += a[i1]
                i2 = j - k
                if 0 <= i2 < n:
                    grad[j] += a[i2]
        grad /= len(peak_indices)
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_sum(a_new, target_sum)
        
        new_peak = float(np.max(autocorr(a_new)))
        if new_peak < best_score - 1e-12:
            best_score = new_peak
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_peak < peak_val:
            a = a_new
            stall += 1
        else:
            stall += 1
            lr *= 0.95
        
        if stall > 100:
            lr *= 0.5
            stall = 0
            # Re-center
            a = best_a + 0.01 * np.random.default_rng(step).standard_normal(n)
            a = np.clip(a, 0.0, None)
            a = normalize_sum(a, target_sum)
        
        step += 1
        if verbose and step % 200 == 0:
            sc = 2.0 * n * best_score / (target_sum * target_sum)
            print(f"    Subgrad step={step}: score={sc:.6f}, lr={lr:.4f}, peaks={len(peak_indices)}")
    
    return best_a


def stochastic_search(a, time_limit, verbose=False):
    """Stochastic local search with simulated annealing and informed moves."""
    t0 = time.monotonic()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_sum(a, target_sum)
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(777)
    
    T = 0.005
    T_min = 1e-9
    cooling = 0.9995
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        current = best_a
        current_score = best_score
        
        strategy = rng.random()
        
        if strategy < 0.25:
            # Reduce top peak contributors
            conv = autocorr(current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            for idx in top_idx:
                current[idx] *= np.exp(rng.normal(-0.03, 0.01))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = max(3, n // 15)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                current[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))
        elif strategy < 0.45:
            # Block perturbation
            width = int(rng.integers(max(2, n // 30), max(3, n // 8)))
            start = int(rng.integers(0, max(1, n - width)))
            current[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif strategy < 0.65:
            # Gaussian multiplicative noise
            current *= np.exp(rng.normal(0.0, 0.015, size=n))
        elif strategy < 0.8:
            # Swap segments
            w1 = int(rng.integers(max(2, n // 20), max(3, n // 6)))
            w2 = int(rng.integers(max(2, n // 20), max(3, n // 6)))
            s1 = int(rng.integers(0, max(1, n - w1)))
            s2 = int(rng.integers(0, max(1, n - w2)))
            if abs(s1 - s2) > max(w1, w2):
                current[s1:s1 + w1] = current[s1:s1 + w1] * 0.5 + current[s2:s2 + w2] * 0.5
                current[s2:s2 + w2] = current[s1:s1 + w1] * 0.5 + current[s2:s2 + w2] * 0.5
        else:
            # Mirror/symmetry perturbation
            center = n // 2
            current[:center] = 0.7 * current[:center] + 0.3 * current[center - 1: -1: -1] if center > 1 else current[:center]
        
        current = np.clip(current, 0.0, None)
        current = normalize_sum(current, target_sum)
        
        new_score = compute_score(current)
        delta = new_score - current_score
        
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            best_a = current.copy()
            best_score = new_score
        
        T *= cooling
        step += 1
        
        if verbose and step % 10000 == 0:
            print(f"    Stochastic step={step}: score={best_score:.6f}, T={T:.2e}")
    
    return best_a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def make_structured_init(n, seed):
    """Create a structured initialization with good autocorrelation properties."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    
    # Start with a low-frequency modulation
    a = np.ones(n)
    n_harmonics = int(rng.integers(2, 5))
    for i in range(n_harmonics):
        freq = rng.uniform(0.3, 3.0) * (i + 1)
        amp = rng.uniform(0.1, 0.3) / (i + 1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    
    # Add a chirp component
    a += 0.1 * rng.random() * np.cos(np.pi * t * t / n)
    
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP blending with bisection,
    smooth power-sum L-BFGS-B, multi-peak subgradient, stochastic search,
    and length exploration.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 40.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        a_clip = np.maximum(a_clip, 1e-10)
        s = compute_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.asarray(height_sequence_1, dtype=np.float64)
        provided_seq = np.clip(provided_seq, 0.0, 1000.0)
        provided_seq = np.maximum(provided_seq, 1e-6)
        print(f"Provided: n={len(provided_seq)}, score={compute_score(provided_seq):.6f}")
        update_best(provided_seq)
    
    # === Phase 1: LP refinement on provided sequence ===
    print("\n=== Phase 1: LP refinement ===")
    if provided_seq is not None:
        n = len(provided_seq)
        remaining = deadline - time.monotonic()
        t_limit = min(150.0, remaining * 0.2)
        print(f"  LP refine n={n}, limit={t_limit:.0f}s")
        result = lp_refine(provided_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  After LP: score={best_score:.6f}")
    
    # === Phase 2: Smooth optimization ===
    print("\n=== Phase 2: Smooth L-BFGS-B ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(120.0, remaining * 0.15)
        print(f"  Smooth optimize, limit={t_limit:.0f}s")
        result = smooth_optimize(best_seq, t_limit, verbose=True,
                                 p_values=(5.0, 10.0, 20.0, 50.0))
        if update_best(result):
            print(f"  After smooth: score={best_score:.6f}")
    
    # === Phase 3: Multi-peak subgradient ===
    print("\n=== Phase 3: Multi-peak subgradient ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(100.0, remaining * 0.12)
        print(f"  Subgradient, limit={t_limit:.0f}s")
        result = multi_peak_subgradient_optimize(best_seq, t_limit, verbose=True, epsilon=0.02)
        if update_best(result):
            print(f"  After subgrad: score={best_score:.6f}")
        # Try with different epsilon
        if remaining > 60:
            result = multi_peak_subgradient_optimize(best_seq, min(60.0, remaining * 0.1),
                                                     verbose=False, epsilon=0.05)
            if update_best(result):
                print(f"  After subgrad (eps=0.05): score={best_score:.6f}")
    
    # === Phase 4: More LP refinement ===
    print("\n=== Phase 4: LP refinement round 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(120.0, remaining * 0.15)
        print(f"  LP refine round 2, limit={t_limit:.0f}s")
        # Try with min_g constraint
        result = lp_refine(best_seq, t_limit, verbose=True, use_min_g=0.1)
        if update_best(result):
            print(f"  After LP2 (min_g=0.1): score={best_score:.6f}")
        # Without min_g
        remaining = deadline - time.monotonic()
        if remaining > 30:
            result = lp_refine(best_seq, min(60.0, remaining * 0.3), verbose=True)
            if update_best(result):
                print(f"  After LP2 (no min_g): score={best_score:.6f}")
    
    # === Phase 5: Stochastic search ===
    print("\n=== Phase 5: Stochastic search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()

    # === Phase 6: Length exploration ===

                if update_best(result):
                    print(f"  After final LP: score={best_score:.6f}")
            
            remaining = deadline - time.monotonic() - 10
            if remaining > 10:
                result = stochastic_search(best_seq, remaining, verbose=False)
                if update_best(result):
                    print(f"  After final stochastic: score={best_score:.6f}")
    
    # === Phase 8: Try structured initializations at best length ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 60:
            n_best = len(best_seq)
            for trial in range(3):
                if time.monotonic() > deadline - 15:

                a_init = make_structured_init(n_best, seed + 1000 + trial)

        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list
    
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
