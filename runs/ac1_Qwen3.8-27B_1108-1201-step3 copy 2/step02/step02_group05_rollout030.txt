        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        progress = min(1.0, it / 2000.0)
        tau_frac = tau_start + (tau_end - tau_start) * progress
        tau = max(C_max * tau_frac, 1e-12)

        w = softmax_stable(C, tau)
        grad = compute_smooth_grad(a, w)

        grad_range = grad.max() - grad.min()
        if grad_range < 1e-14:
            break
        eta = step_size / grad_range
        exponent = -eta * (grad - grad.mean())
        exponent = np.clip(exponent, -40.0, 40.0)
        a = a * np.exp(exponent)
        a = normalize_a(a)

    return best_a, best_obj


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    n_constr = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(n_constr, j + n)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(n_constr, n))

    c = -np.ones(n)
    b = np.full(n_constr, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_refine(f0, time_limit):
    """LP-based iterative refinement with line search."""
    f = normalize_a(f0.astype(np.float64).copy())
    n = len(f)
    best_f = f.copy()
    best_obj = obj_from_conv(fast_convolve(f), n)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 40:
        C = fast_convolve(f)
        M = C.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(f, M)

        if np.allclose(g, f, atol=1e-10):
            no_improve += 5
            continue

        best_t = 0.0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]:
            a_t = normalize_a((1.0 - t) * f + t * g)
            obj_t = obj_from_conv(fast_convolve(a_t), n)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t

        if best_t > 0.0 and best_obj_t < best_obj - 1e-10:
            f = normalize_a((1.0 - best_t) * f + best_t * g)
            best_obj = best_obj_t
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def peak_reduction(a, time_limit):
    """Targeted peak reduction: move mass from peak-contributing to safe positions."""
    a = normalize_a(a.astype(np.float64).copy())
    n = len(a)
    t0 = time.monotonic()
    best_obj = obj_from_conv(fast_convolve(a), n)
    best_a = a.copy()
    stuck = 0

    while time.monotonic() - t0 < time_limit and stuck < 300:
        C = fast_convolve(a)
        k_star = int(np.argmax(C))

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        danger = np.where(contrib > 0.4 * max_c)[0]
        safe = np.where(contrib < 0.05 * max_c)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        for di in danger[:25]:
            if found:
                break
            for si in safe[:25]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = normalize_a(a_trial)
                    o = obj_from_conv(fast_convolve(a_trial), n)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        best_a = a.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return best_a, best_obj


def get_parent_clipped(n):
    """Get parent sequence clipped/resized to length n, normalized."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_initializations(n, rng, parent):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    x = np.arange(n, dtype=np.float64)
    for sf in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits['gauss_' + str(sf)] = g / g.sum()

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    hann = np.maximum(hann, 0.0)
    inits['hann'] = hann / hann.sum()

    if parent is not None:
        inits['parent'] = parent.copy()
        for noise_level in [0.05, 0.1, 0.2]:
            p = parent * (1.0 + noise_level * rng.standard_t(df=4, size=n))
            inits['parent_pert_' + str(noise_level)] = normalize_a(p)

    # Edge-concentrated
    edges = np.zeros(n)
    m = max(1, n // 5)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    return inits


def sort_key_phase1(item):
    """Sort key: parent inits first, then by length."""
    n, name, _ = item
    if 'parent' in name:
        return (0, n)
    return (1, n)


def sort_by_obj(item):
    """Sort by objective value (first element)."""
    return item[0]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses mirror descent with smooth max as the primary
    optimizer, combined with sparse LP refinement and targeted peak reduction.
    Multi-phase strategy: explore, refine, polish.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 25.0

    best_sequence = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  NEW BEST: " + label + " n=" + str(len(a)) + " score=" + format(score, ".6f"))

    # Phase 1: Mirror descent exploration (40% budget)
    print("Phase 1: Mirror descent exploration...")
    phase1_time = total_budget * 0.40
    phase1_end = time.monotonic() + phase1_time

    lengths = [256, 512, 1024]
    all_configs = []
    for n in lengths:
        parent = get_parent_clipped(n)
        inits = make_initializations(n, rng, parent)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))

    all_configs.sort(key=sort_key_phase1)
    n_configs = max(1, len(all_configs))
    phase1_remaining = max(5.0, phase1_end - time.monotonic())
    time_per_config = max(1.5, phase1_remaining / n_configs)

    md_results = []
    for i in range(len(all_configs)):
        n, name, a0 = all_configs[i]
        remaining_to_dead = deadline - time.monotonic()
        if remaining_to_dead < 45.0:
            break
        t_alloc = min(time_per_config, remaining_to_dead - 40.0)
        if t_alloc < 1.0:
            break

        a_opt, obj_opt = mirror_descent_optimize(a0, time_limit=t_alloc,
                                                   tau_start=0.1, tau_end=0.005,
                                                   step_size=0.3)
        update_best(a_opt, "MD/" + str(n) + "/" + name)
        md_results.append((obj_opt, n, name, a_opt))

    print("  Phase 1 done. Best score: " + format(best_score, ".6f"))

    # Phase 2: Deep optimization (35% budget)
    print("Phase 2: Deep optimization on top candidates...")
    phase2_time = total_budget * 0.35
    phase2_end = time.monotonic() + phase2_time

    md_results.sort(key=sort_by_obj)
    top_count = min(10, len(md_results))

    remaining_p2 = max(5.0, phase2_end - time.monotonic())
    time_per_deep = max(3.0, remaining_p2 / max(1, top_count))

    for i in range(top_count):
        remaining_to_dead = deadline - time.monotonic()
        if remaining_to_dead < 45.0:
            break
        obj_opt, n, name, a0 = md_results[i]
        t_alloc = min(time_per_deep, remaining_to_dead - 40.0)
        if t_alloc < 2.0:
            break

        # Deeper MD with lower tau
        t_md = t_alloc * 0.55
        a_md, _ = mirror_descent_optimize(a0, time_limit=t_md,
                                           tau_start=0.05, tau_end=0.002,
                                           step_size=0.2)
        update_best(a_md, "DeepMD/" + str(n) + "/" + name)

        # LP refinement
        remaining_to_dead = deadline - time.monotonic()
        t_lp = min(t_alloc * 0.45, remaining_to_dead - 35.0)
        if t_lp > 2.0:
            a_lp, _ = lp_refine(a_md, time_limit=t_lp)
            update_best(a_lp, "LP/" + str(n) + "/" + name)

    print("  Phase 2 done. Best score: " + format(best_score, ".6f"))

    # Phase 3: Intensive polish on global best (25% budget)
    print("Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_deadline = deadline - 10.0
        polish_iter = 0
        while time.monotonic() < polish_deadline and polish_iter < 50:
            remaining = polish_deadline - time.monotonic()
            if remaining < 5.0:
                break
            polish_iter += 1

            # Alternate: MD, LP, peak reduction
            phase = polish_iter % 3

            if phase == 0:
                t_alloc = min(remaining * 0.35, 60.0)
                if t_alloc > 2.0:
                    a_md, _ = mirror_descent_optimize(best_sequence, time_limit=t_alloc,
                                                       tau_start=0.02, tau_end=0.001,
                                                       step_size=0.15)
                    update_best(a_md, "PolishMD-" + str(polish_iter))

            elif phase == 1:
                remaining = polish_deadline - time.monotonic()
                t_alloc = min(remaining * 0.35, 90.0)
                if t_alloc > 2.0:
                    a_lp, _ = lp_refine(best_sequence, time_limit=t_alloc)
                    update_best(a_lp, "PolishLP-" + str(polish_iter))

            else:
                remaining = polish_deadline - time.monotonic()
                t_alloc = min(remaining * 0.3, 30.0)
                if t_alloc > 2.0:
                    a_peak, _ = peak_reduction(best_sequence, time_limit=t_alloc)
                    update_best(a_peak, "PolishPeak-" + str(polish_iter))

    # Final safety check
    if best_sequence is None:
        best_sequence = np.ones(128)
        try:
            best_score = float(evaluate_sequence(best_sequence.tolist()))
        except Exception:
            best_score = float('inf')

    try:
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
    except Exception:
        pass

    print("FINAL SCORE: " + format(best_score, ".6f"))
    print("Sequence length: " + str(len(best_sequence)))
    return [float(x) for x in best_sequence]
