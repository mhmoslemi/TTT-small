def fft_convolve(a, b):

    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

        A[j:j + n, j] = f
    return A

"""

peak aperiodic autocorrelation of non-negative sequences.

log-sum-exp gradient polishing, multi-length exploration, and peak-aware local search.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A_rows(f, active_lags):
    """Build only the rows of A matrix for active lags. A[k, j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    for k in active_lags:
        ki = int(k)
        row = np.zeros(n)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            row[j] = f[ki - j]
        rows.append(row)
    return np.array(rows)


def solve_min_t_lp(f, active_lags, timeout=5.0):
    """
    Solve: min t s.t. conv(f, b)_k <= t for k in active_lags, b >= 0, sum(b) = 1.
    Returns (b, t) or (None, None).
    """
    n = len(f)
    A_act = build_A_rows(f, active_lags)
    m = len(active_lags)

    # Variables: x = [b_0, ..., b_{n-1}, t]
    # Objective: minimize t => c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Constraints:
    # A_act @ b - t <= 0  (m constraints)
    # sum(b) <= 1         (1 constraint)
    # -sum(b) <= -1       (1 constraint)
    A_ub = np.zeros((m + 2, n + 1))
    b_ub = np.zeros(m + 2)

    A_ub[:m, :n] = A_act
    A_ub[:m, n] = -1.0
    b_ub[:m] = 0.0

    A_ub[m, :n] = 1.0
    b_ub[m] = 1.0
    A_ub[m + 1, :n] = -1.0
    b_ub[m + 1] = -1.0

    bounds = [(0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9})
        if res.success:
            b = res.x[:n]
            t_val = res.x[n]
            return b, t_val
    except Exception:
        pass
    return None, None


def solve_max_sum_lp(f, M, timeout=5.0):
    """Fallback: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    A_full = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_full[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(a_init, time_limit, verbose=True):
    """LP-based iteration with direct min-t formulation and adaptive active set."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    eps_lo = 0.15  # start relaxed
    eps_hi = 0.01  # end tight
    min_active = 5
    max_active = 30

    it = 0
    stall = 0
    patience = 100

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(a, a)
        M = float(np.max(c))

        # Adaptive active set
        progress = min(1.0, it / patience)
        eps = eps_lo - (eps_lo - eps_hi) * progress
        threshold = M * (1.0 - eps)
        active = np.where(c >= threshold)[0]
        if len(active) < min_active:
            active = np.argsort(c)[-min_active:]
        if len(active) > max_active:
            active = np.argsort(c)[-max_active:]

        # Solve LP
        b, t_val = solve_min_t_lp(a, active, timeout=4.0)
        if b is None:
            # Fallback
            M_curr = float(np.max(fft_convolve(a, a))) * 1.0
            b = solve_max_sum_lp(a, M_curr, timeout=4.0)
            if b is None:
                break
            Sb = float(np.sum(b))
            if Sb < 1e-12:
                break
            b = b / Sb
        else:
            Sb = float(np.sum(b))
            if Sb < 1e-12:
                break
            b = b / Sb

        # Line search
        best_t = 0.1
        best_s = np.inf
        for t_try in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_cand = (1.0 - t_try) * a + t_try * b
            a_cand = np.clip(a_cand, 0.0, None)
            S_cand = float(np.sum(a_cand))
            if S_cand < 0.01:
                continue
            a_cand = a_cand / S_cand
            s_cand = score_fast(a_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        a = (1.0 - best_t) * a + best_t * b
        a = np.clip(a, 0.0, None)
        S = float(np.sum(a))
        if S < 0.01:
            break
        a = a / S
        it += 1

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{elapsed:6.1f}s] it={it:4d} M={M:.10f} score={score_fast(a):.6f} "
                  f"best={best_score:.6f} active={len(active)} stall={stall}")

        if stall >= patience:
            # Escape: add small random perturbation
            rng_esc = np.random.default_rng(42 + it * 7919)
            a += 0.05 * rng_esc.standard_normal(n)
            a = np.clip(a, 0.01, None)
            a = a / float(np.sum(a))
            stall = 0

    return best_a, best_score


def logsumexp_gradient_polish(a_init, time_limit, verbose=True):
    """Polish using log-sum-exp approximation of max, with projected gradient descent."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    T = 50.0  # temperature for log-sum-exp (higher = sharper)
    lr = 0.005
    beta = 0.9  # momentum
    momentum = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(a, a)
        # Compute log-sum-exp gradient
        c_max = np.max(c)
        exp_c = np.exp(T * (c - c_max))
        p = exp_c / float(np.sum(exp_c))  # softmax weights

        # Gradient of LSE: sum_k p_k * 2 * a[k-i] (valid indices)
        grad = np.zeros(n)
        for k in range(len(c)):
            if p[k] < 1e-12:
                continue
            ki = k
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * p[k] * a[ki - j]

        # Project: remove mean (to maintain sum constraint direction)
        grad -= np.mean(grad)
        momentum = beta * momentum + (1.0 - beta) * grad
        a = a - lr * momentum
        a = np.clip(a, 0.001, None)
        a = a / float(np.sum(a))

        it += 1
        if it % 200 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose:
                print(f"    [lse {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")

    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def targeted_local_search(a_init, time_limit, verbose=True):
    """Local search that targets the peak-contributing elements."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    temp = 0.001
    cooling = 0.9997
    step_size = 0.02
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(a, a)
        peak_lag = int(np.argmax(c))

        # Identify contributors to the peak
        contribs = np.zeros(n)
        lo = max(0, peak_lag - (n - 1))
        hi = min(n - 1, peak_lag)
        for j in range(lo, hi + 1):
            contribs[j] = a[j] * a[peak_lag - j]

        total_contrib = float(np.sum(contribs))
        if total_contrib < 1e-15:
            contribs = np.ones(n) / n
        else:
            contribs = contribs / total_contrib

        candidate = a.copy()
        r = rng.random()

        if r < 0.5:
            # Decrease top contributors to peak
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            dec_amt = step_size * rng.uniform(0.5, 1.5)
            candidate[top_idx] *= np.exp(-dec_amt * rng.uniform(0.5, 1.0, size=top_k))
            # Increase non-contributors
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = max(3, len(other_idx) // 10)
                inc_idx = rng.choice(other_idx, size=min(n_inc, len(other_idx)), replace=False)
                candidate[inc_idx] *= np.exp(step_size * rng.uniform(0.3, 1.0, size=len(inc_idx)))
        elif r < 0.8:
            # Smooth perturbation weighted by contribution
            noise = rng.normal(0, step_size, size=n)
            candidate *= np.exp(0.5 * noise)
        else:
            # Swap-like operation: decrease high-contrib, increase low-contrib
            sorted_idx = np.argsort(contribs)
            n_swap = max(2, n // 20)
            dec_idx = sorted_idx[-n_swap:]
            inc_idx = sorted_idx[:n_swap]
            amt = step_size * rng.uniform(0.5, 1.5)
            candidate[dec_idx] *= np.exp(-amt)
            candidate[inc_idx] *= np.exp(amt)

        candidate = np.clip(candidate, 0.001, None)
        S_cand = float(np.sum(candidate))
        if S_cand < 0.01:
            continue
        candidate = candidate / S_cand
        new_score = score_fast(candidate)

        cur_score = score_fast(a)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
        elif rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-15)):
            a = candidate

        temp *= cooling
        step_size *= 0.9999
        it += 1

        if verbose and it % 500 == 0:
            print(f"    [ls {time.monotonic()-t0:6.1f}s] it={it} score={score_fast(a):.6f} best={best_score:.6f}")

    return best_a, best_score


def make_init(n, rng, method='hann'):
    """Generate initial sequence."""
    t = np.arange(n, dtype=np.float64)
    if method == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.03 * rng.standard_normal(n)
    elif method == 'asymmetric':
        # Try asymmetric: high on one side
        a = np.exp(-3.0 * t / n) + 0.1
        a += 0.03 * rng.standard_normal(n)
    elif method == 'bimodal':
        # Two peaks
        center1 = n * 0.25
        center2 = n * 0.75
        a = np.exp(-((t - center1) ** 2) / (2 * (n * 0.1) ** 2)) + \
            np.exp(-((t - center2) ** 2) / (2 * (n * 0.1) ** 2))
        a += 0.02 * rng.standard_normal(n)
    elif method == 'random':
        a = 0.5 + rng.random(n)
    else:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    return np.clip(a, 0.01, None)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.01, None)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP iteration + LSE gradient + targeted local search.
    Explores multiple sequence lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = score_fast(provided)
            print(f"Provided: n={len(provided)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = provided.copy()
        except Exception:
            pass

    # Define search tasks: (n, init_method)
    tasks = []
    if provided is not None:
        n_prov = len(provided)
        tasks.append((n_prov, 'provided'))

    # Explore various lengths
    for n in [128, 192, 256, 320, 384, 512, 768, 1024]:
        if n == (n_prov if provided is not None else -1):
            continue
        tasks.append((n, 'asymmetric'))
        tasks.append((n, 'hann'))

    # Sort by expected value (smaller n first for quick wins)
    tasks.sort(key=lambda x: x[0])

    print(f"\n=== Starting optimization with {len(tasks)} tasks ===")

    # Phase 1: LP iteration for each task
    for idx, (n, init_name) in enumerate(tasks):
        if time.monotonic() > deadline - 120:
            break

        remaining = deadline - time.monotonic() - 120
        if remaining < 15:
            break

        # Time allocation: give more time to promising tasks
        if init_name == 'provided' and provided is not None:
            t_alloc = min(remaining * 0.25, 300.0)
        elif n <= 256:
            t_alloc = min(remaining * 0.12, 200.0)
        elif n <= 512:
            t_alloc = min(remaining * 0.08, 150.0)
        else:
            t_alloc = min(remaining * 0.05, 100.0)
        t_alloc = max(10.0, t_alloc)

        # Generate initial sequence
        if init_name == 'provided' and provided is not None:
            f0 = resample_sequence(provided, n)
        else:
            f0 = make_init(n, rng, method=init_name)

        print(f"\n  [{idx+1}/{len(tasks)}] n={n:4d} init={init_name:10s} alloc={t_alloc:.0f}s")
        f_best, s_best = lp_iteration(f0, t_alloc, verbose=False)

        if verbose_check(s_best, best_score):
            print(f"  => score={s_best:.6f} (global best={best_score:.6f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}) ***")

    # Phase 2: Gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        pt = min(120.0, deadline - time.monotonic() - 50.0)
        if pt > 20:
            print(f"\n=== Phase 2: LSE gradient polish ({pt:.0f}s, n={len(best_seq)}) ===")
            f_pol, s_pol = logsumexp_gradient_polish(best_seq, pt, verbose=False)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = f_pol.copy()
                print(f"  Polished: {best_score:.6f}")

    # Phase 3: Targeted local search
    if best_seq is not None and time.monotonic() < deadline - 30:
        ls_t = min(90.0, deadline - time.monotonic() - 20.0)
        if ls_t > 15:
            print(f"\n=== Phase 3: Local search ({ls_t:.0f}s, n={len(best_seq)}) ===")
            f_ls, s_ls = targeted_local_search(best_seq, ls_t, verbose=False)
            if s_ls < best_score:
                best_score = s_ls
                best_seq = f_ls.copy()
                print(f"  After local search: {best_score:.6f}")

    # Phase 4: Final LP refinement on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        fl_t = min(60.0, deadline - time.monotonic() - 5.0)
        if fl_t > 10:
            print(f"\n=== Phase 4: Final LP ({fl_t:.0f}s, n={len(best_seq)}) ===")
            f_fin, s_fin = lp_iteration(best_seq, fl_t, verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  Final LP: {best_score:.6f}")

    # Phase 5: One more local search pass
    if best_seq is not None and time.monotonic() < deadline - 5:
        ls2_t = min(30.0, deadline - time.monotonic() - 3.0)
        if ls2_t > 5:
            f_ls2, s_ls2 = targeted_local_search(best_seq, ls2_t, verbose=False)
            if s_ls2 < best_score:
                best_score = s_ls2
                best_seq = f_ls2.copy()
                print(f"  Final LS: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 50}")
        return final_list

    # Fallback
    return [1.0] * 256
