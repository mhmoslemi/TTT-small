subgradient descent, coordinate descent, and perturbation-based local search.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def compute_subgradient(a, top_k=5):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    # Find top positions

        idx_a = k - np.arange(lo, hi + 1)
        g[lo:hi+1] += 2.0 * a[idx_a]
    g /= len(top_idx)
    return g

"""
Step-function optimization minimizing peak autocorrelation.
Uses coordinate descent with 1D sub-optimization, subgradient polish,
and multi-start from parametric initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def make_ramp(n, p=1.0):
    """Power-law ramp: a[j] = ((j+1)/n)^p"""
    j = np.arange(n, dtype=np.float64)
    return ((j + 1) / n) ** p

            #   a[i]^2 if k = 2i
            #   0 otherwise

                k_vals = j_k[valid]
                a_ks = a[k_vals - i]  # a[k-i] for valid k
                # Check which are k = 2i (quadratic case)
                quad_mask = (k_vals == 2 * i)
                lin_mask = ~quad_mask

                    R[k_vals[lin_mask]] -= 2.0 * a_i_old * a_ks[lin_mask]

                    R[k_vals[quad_mask]] -= a_i_old * a_i_old

                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            x_lo = max(0, best_x - x_max / num_points)
            x_hi = best_x + x_max / num_points

            break
    
    return best_a, best_score

"""
Step-function optimization minimizing peak autocorrelation of non-negative sequences.
Uses parametric initializations (power ramps, exponentials), cyclic coordinate descent
with 1D sub-optimization, subgradient polish, and perturbation-based local search.
Key insight: score is scale-invariant, so we fix sum(a)=1 and minimize 2n*max(conv(a,a)).
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def score_normalized(a):
    """Score assuming sum(a)=1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c))


def normalize_sum(a, target=1.0):
    """Normalize a to have sum = target."""
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a * (target / s)


def make_ramp(n, p=1.0):
    """Power-law ramp: a[j] = ((j+1)/n)^p, normalized to sum=1."""
    j = np.arange(n, dtype=np.float64)
    a = ((j + 1.0) / n) ** p
    return normalize_sum(a, 1.0)


def make_exp_ramp(n, alpha=2.0):
    """Exponential ramp."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * j / max(1, n - 1))
    return normalize_sum(a, 1.0)


def make_sigmoid(n, beta=5.0):
    """Sigmoid shape."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / max(1, n - 1) - 1.0
    a = 1.0 / (1.0 + np.exp(-beta * x))
    a = a - a[0] + 0.1
    a = np.clip(a, 0.01, None)
    return normalize_sum(a, 1.0)


def make_chebyshev(n, m=3):
    """Chebyshev polynomial (shifted) as profile."""
    from numpy.polynomial.chebyshev import chebval
    x = np.linspace(-1, 1, n)
    a = np.abs(chebval(x, np.zeros(m+1)))

    j = np.arange(n, dtype=np.float64)
    a = (j / max(1, n-1)) ** 2 * (1 + 0.5 * np.cos(np.pi * j / n))
    a = np.clip(a, 0.01, None)
    return normalize_sum(a, 1.0)

                vals = alpha_affected * x + beta_affected * x * x + R_affected
                peak = max(max_unchanged, float(np.max(vals)))
                S_new = S_other + x
                if S_new < 1e-10:
                    return 1e10
                return 2.0 * n * peak / (S_new * S_new)
            
            # Bounded golden section
            g_lo = 0.0
            g_hi = x_max
            
            # First evaluate at initial point and boundaries
            f_lo = obj_1d(g_lo)
            f_hi = obj_1d(g_hi)
            f_init = obj_1d(a_i)
            
            # Expand if initial point is better than both ends
            if f_init < f_lo and f_init < f_hi:
                # Try expanding
                if a_i > 0:
                    g_lo = a_i - x_max
                    g_lo = max(0.0, g_lo)
                g_hi = a_i + x_max
                f_lo = obj_1d(g_lo)
                f_hi = obj_1d(g_hi)
                if f_init < f_lo:
                    g_lo = g_lo * 0.5
                    f_lo = obj_1d(g_lo)
                    if f_init < f_lo:
                        best_x = a_i
                        best_f = f_init
                        # Just use initial
                    else:
                        best_x = a_i  # will be updated in golden section
                        best_f = f_init
                elif f_init < f_hi:
                    g_hi = a_i + (g_hi - a_i) * 0.5
                    f_hi = obj_1d(g_hi)
                else:
                    best_x = a_i
                    best_f = f_init
            
            if not hasattr(coordinate_descent, '_skip_golden') or not getattr(coordinate_descent, '_skip_golden'):
                # Golden section iterations
                c1 = g_hi - gr * (g_hi - g_lo)
                c2 = g_lo + gr * (g_hi - g_lo)
                f_c1 = obj_1d(c1)
                f_c2 = obj_1d(c2)
                
                for _ in range(40):
                    if f_c1 < f_c2:
                        g_hi = c2
                        c2 = c1
                        f_c2 = f_c1
                        c1 = g_hi - gr * (g_hi - g_lo)
                        f_c1 = obj_1d(c1)
                    else:
                        g_lo = c1
                        c1 = c2
                        f_c1 = f_c2
                        c2 = g_lo + gr * (g_hi - g_lo)
                        f_c2 = obj_1d(c2)
                
                best_x = (g_lo + g_hi) / 2.0
                best_f = obj_1d(best_x)
            
            # Check if improvement
            if best_f < f_init - 1e-12:
                a[i] = best_x
        
        num_passes += 1
        s = score_fast(a)
        if s < best_score - 1e-10:
            best_score = s
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and num_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [cd {elapsed:.1f}s] pass={num_passes} score={s:.6f} best={best_score:.6f} stall={stall}")
        
        if stall >= stall_limit:
            break
    
    return best_a, best_score


def subgradient_polish(a_init, time_limit, verbose=False):
    """Subgradient descent with momentum on the score function."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    a = normalize_sum(a, 1.0)
    
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    base_lr = 0.005
    it = 0
    stall = 0
    stall_limit = 200
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(a, a)
        S = float(np.sum(a))
        max_c = float(np.max(c))
        
        # Subgradient of max_c with respect to a
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(c)[-top_k:]
        g_max = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if hi >= lo:
                idx_a = k - np.arange(lo, hi + 1)
                g_max[lo:hi+1] += 2.0 * a[idx_a]
        g_max /= top_k
        
        # Gradient of score = 2n*max_c/S^2
        g_score = (2.0 * n / (S * S)) * (g_max - max_c / S)
        
        lr = base_lr / (1.0 + 0.05 * it)
        momentum = beta * momentum + (1.0 - beta) * g_score
        a = a - lr * momentum
        a = np.clip(a, 0, None)
        a = normalize_sum(a, 1.0)
        
        it += 1
        if it % 200 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if verbose and it % 1000 == 0:
                print(f"    [sgd {time.monotonic()-t0:.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
            if stall >= stall_limit:
                rng = np.random.default_rng(it * 13 + 7)
                a = best_a * (1.0 + 0.03 * rng.standard_normal(n))
                a = np.clip(a, 1e-6, None)
                a = normalize_sum(a, 1.0)
                stall = 0
                momentum = np.zeros(n)
    
    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def perturbation_search(a_init, time_limit, verbose=False):
    """Perturbation-based local search with simulated annealing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = normalize_sum(a, 1.0)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    temp = 0.001
    cooling = 0.99995
    it = 0
    stall = 0
    stall_limit = 500
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.3:
            # Modify a few random elements
            num_mod = rng.integers(2, max(3, n // 20))
            idxs = rng.choice(n, size=num_mod, replace=False)
            for idx in idxs:
                factor = np.exp(rng.normal(0, 0.1))
                candidate[idx] *= factor
        elif r < 0.5:
            # Scale a contiguous region
            width = rng.integers(3, max(4, n // 10))
            start = rng.integers(0, max(1, n - width))
            factor = np.exp(rng.normal(0, 0.08))
            candidate[start:start+width] *= factor
        elif r < 0.7:
            # Ridge modification: reduce peak-contributing elements
            c = fft_convolve(a, a)
            peak_k = int(np.argmax(c))
            # Find which elements contribute most to c[peak_k]
            j_lo = max(0, peak_k - (n-1))
            j_hi = min(n-1, peak_k)
            contribs = np.zeros(n)
            contribs[j_lo:j_hi+1] = a[j_lo:j_hi+1] * a[peak_k - j_lo:peak_k - j_hi+1]
            top_contrib = np.argsort(contribs)[-max(3, n//15):]
            candidate[top_contrib] *= np.exp(rng.normal(-0.05, 0.03, size=len(top_contrib)))
        elif r < 0.85:
            # Shape modification: multiply by a smooth factor
            t = np.arange(n, dtype=np.float64) / max(1, n-1)
            factor = 1.0 + rng.normal(0, 0.05) * np.sin(np.pi * t * rng.integers(1, 4))
            candidate *= np.clip(factor, 0.5, 2.0)
        else:
            # Global scale fluctuation
            candidate *= np.exp(rng.normal(0, 0.02))
        
        candidate = np.clip(candidate, 1e-6, None)
        candidate = normalize_sum(candidate, 1.0)
        
        new_score = score_fast(candidate)
        current_score = score_fast(a)
        
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
            stall = 0
        elif new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            a = candidate
        
        temp *= cooling
        it += 1
        
        if it % 500 == 0:
            stall_check_score = score_fast(a)
            if stall_check_score < best_score:
                best_score = stall_check_score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if verbose and it % 5000 == 0:
                print(f"    [perturb {time.monotonic()-t0:.1f}s] it={it} score={stall_check_score:.6f} best={best_score:.6f}")
            if stall >= stall_limit:
                # Restart
                rng2 = np.random.default_rng(it * 7 + 42)
                a = best_a * (1.0 + 0.05 * rng2.standard_normal(n))
                a = np.clip(a, 1e-6, None)
                a = normalize_sum(a, 1.0)
                stall = 0
                temp = 0.001
    
    return best_a, best_score


def parametric_optimization(n, p_init, time_limit, verbose=False):
    """Optimize a power ramp a[j] = ((j+1)/n)^p by adjusting p and then doing CD."""
    # First optimize p via grid search
    best_p = p_init
    best_score_p = np.inf
    
    for p in np.arange(0.2, 4.0, 0.1):
        a = make_ramp(n, p)
        s = score_fast(a)
        if s < best_score_p:
            best_score_p = s
            best_p = p
    
    # Fine-tune p
    for p in np.arange(best_p - 0.1, best_p + 0.1, 0.01):
        if p <= 0:
            continue
        a = make_ramp(n, p)
        s = score_fast(a)
        if s < best_score_p:
            best_score_p = s
            best_p = p
    
    a_best = make_ramp(n, best_p)
    if verbose:
        print(f"      Best p={best_p:.3f}, score={best_score_p:.6f}")
    
    # Now do coordinate descent from this starting point
    a_opt, s_opt = coordinate_descent(a_best, time_limit, verbose=verbose)
    return a_opt, s_opt


def propose_candidate(seed=42, budget_s=1800):
    """Main entry point: multi-strategy search for optimal non-negative sequence."""
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    print(f"Starting optimization with budget {budget_s}s")
    
    # Phase 0: Evaluate provided sequence and quick parametric scan
    print("\n=== Phase 0: Initial evaluation ===")
    
    candidates = []
    
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        s = score_fast(provided)
        print(f"  Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()
        candidates.append((provided, s))
    
    # Quick parametric scan
    for n in [64, 128, 256, 384, 512]:
        for p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]:
            a = make_ramp(n, p)
            s = score_fast(a)
            candidates.append((a, s))
            if s < best_score:
                best_score = s
                best_seq = a.copy()
        
        for alpha in [1.0, 2.0, 3.0, 5.0]:
            a = make_exp_ramp(n, alpha)
            s = score_fast(a)
            candidates.append((a, s))
            if s < best_score:
                best_score = s
                best_seq = a.copy()
    
    # Sort candidates by score
    candidates.sort(key=lambda x: x[1])
    print(f"  After parametric scan: best={best_score:.6f}")
    
    # Phase 1: Coordinate descent on top candidates
    print("\n=== Phase 1: Coordinate descent ===")
    
    top_n = 8
    top_candidates = [c[0] for c in candidates[:top_n]]
    
    # Deduplicate by (n, score) to avoid redundant work
    seen = set()
    unique_candidates = []
    for cand in top_candidates:
        n = len(cand)
        s = round(score_fast(cand), 4)
        key = (n, s)
        if key not in seen:
            seen.add(key)
            unique_candidates.append(cand)
    
    for idx, cand in enumerate(unique_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        n = len(cand)
        # Time allocation: more time for smaller/faster sequences
        t_alloc = min(remaining - 20, max(20, 120.0 * 256.0 / n))
        t_alloc = min(t_alloc, remaining - 20)
        
        print(f"  [{time.monotonic()-t_start:.0f}s] CD on n={n}, init_score={score_fast(cand):.6f}, time={t_alloc:.0f}s")
        
        a_opt, s_opt = coordinate_descent(cand, t_alloc, verbose=True)
        
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={len(a_opt)}) ***")
        
        # Also try subgradient polish on the CD result
        if s_opt < best_score + 0.01 and time.monotonic() < deadline - 60:
            t_sg = min(60.0, deadline - time.monotonic() - 15)
            if t_sg > 10:
                a_sg, s_sg = subgradient_polish(a_opt, t_sg, verbose=False)
                if s_sg < best_score:
                    best_score = s_sg
                    best_seq = a_sg.copy()
                    print(f"  *** SG POLISH NEW BEST: {best_score:.6f} ***")
    
    # Phase 2: Try different lengths with optimized shapes
    print("\n=== Phase 2: Multi-length refinement ===")
    
    if best_seq is not None:
        base_n = len(best_seq)
        
        # Try interpolating to different lengths
        for target_n in [base_n // 2, base_n, base_n * 2, base_n * 3]:
            if target_n < 32 or target_n > 4096:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            
            # Interpolate
            x_old = np.linspace(0, 1, base_n)
            x_new = np.linspace(0, 1, target_n)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.clip(a_interp, 0, None)
            a_interp = normalize_sum(a_interp, 1.0)
            
            s_interp = score_fast(a_interp)
            t_alloc = min(remaining - 20, 100.0 * 256.0 / target_n)
            t_alloc = min(t_alloc, remaining - 20)
            
            print(f"  [{time.monotonic()-t_start:.0f}s] CD on interp n={target_n}, score={s_interp:.6f}, time={t_alloc:.0f}s")
            a_opt, s_opt = coordinate_descent(a_interp, t_alloc, verbose=False)
            
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  *** NEW BEST: {best_score:.6f} (n={target_n}) ***")
    
    # Phase 3: Subgradient polish on the best
    print("\n=== Phase 3: Subgradient polish ===")
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        t_pg = min(150.0, deadline - time.monotonic() - 20)
        if t_pg > 20:
            print(f"  Polish with {t_pg:.0f}s")
            a_pol, s_pol = subgradient_polish(best_seq, t_pg, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = a_pol.copy()
                print(f"  *** POLISH NEW BEST: {best_score:.6f} ***")
    
    # Phase 4: Perturbation search
    print("\n=== Phase 4: Perturbation search ===")
    
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_pert = min(200.0, deadline - time.monotonic() - 10)
        if t_pert > 20:
            print(f"  Perturbation with {t_pert:.0f}s")
            a_pert, s_pert = perturbation_search(best_seq, t_pert, verbose=True)
            if s_pert < best_score:
                best_score = s_pert
                best_seq = a_pert.copy()
                print(f"  *** PERTURB NEW BEST: {best_score:.6f} ***")
    
    # Phase 5: Final coordinate descent
    print("\n=== Phase 5: Final CD ===")
    
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_final = max(10.0, deadline - time.monotonic() - 5)
        print(f"  Final CD with {t_final:.0f}s")
        a_fin, s_fin = coordinate_descent(best_seq, t_final, verbose=True)
        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin.copy()
            print(f"  *** FINAL NEW BEST: {best_score:.6f} ***")
    
    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list
    
    # Fallback
    fallback = make_ramp(256, 1.5)
    final_list = [float(max(0.0, min(1000.0, x))) for x in fallback]
    print(f"FALLBACK: n={len(final_list)}, score={evaluate_sequence(final_list):.6f}")
    return final_list
