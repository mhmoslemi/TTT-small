```python
"""
Hybrid optimization combining Fienup IO (phase retrieval) algorithm for flat-spectrum
initialization with smooth Adam gradient descent on the simplex.
The IO algorithm directly targets flat power spectrum (=> flat auto-convolution),
then Adam with adaptive temperature polishes the result.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smooth(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*conv(a,a)[k])) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    tconv = tau * conv
    tconv -= tconv.max()
    w = np.exp(tconv)
    w /= w.sum()
    # grad[l] = 2 * sum_{j=0}^{n-1} w[l+j] * a[j] for l=0..n-1
    # Use FFT-based correlation
    b = a[::-1]
    L = 3 * n - 2
    fft_size = 1 << (L - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    B = np.fft.rfft(b, fft_size)
    corr = np.fft.irfft(W * B, fft_size)[:L]
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, conv, w


def io_flat_spectrum(a0, n_iters, relaxation=0.8, verbose=False):
    """
    Fienup Input-Output algorithm to find non-negative a with flat auto-convolution.
    Alternates between:
      - Fourier domain: enforce |DFT(a_padded)| = target (flat)
      - Real domain: enforce a >= 0, sum(a) = 1 (simplex)
    """
    n = len(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    # Target magnitude: for flat auto-convolution with sum=1,
    # each (a*a)[k] should be ~ 1/(2n-1), so |A[k]|^2 ~ 1/(2n-1)
    target_flat = 1.0 / np.sqrt(2 * n - 1)

    for it in range(n_iters):
        # Forward: zero-pad and FFT
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)

        # Output step: flatten magnitude
        mag = np.abs(A)
        target_mag = np.full_like(mag, target_flat)
        target_mag[0] = mag[0]  # Keep DC (= sum(a) = 1)

        phase = A / (mag + 1e-30)
        A_target = target_mag * phase

        # Relaxed update
        A_new = (1.0 - relaxation) * A + relaxation * A_target

        # Inverse
        a_padded_new = np.fft.irfft(A_new, fft_size)
        a_new = a_padded_new[:n]

        # Input step: project onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()  # fallback
        else:
            a = a_new / s

        # Track best
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if verbose and it % 200 == 0:
            print(f"    IO iter {it}: obj={obj:.6f}")

    return best_a, best_obj


def adam_smooth_descent(a0, time_limit, lr=0.05, tau_init=5.0, tau_rate=0.999,
                        beta1=0.9, beta2=0.999, eps=1e-8, perturb_every=2000,
                        perturb_scale=0.003):
    """
    Adam optimizer on the simplex with smooth log-sum-exp approximation to max.
    Temperature tau decreases over time to sharpen the max approximation.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_init

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient
        grad, conv, w = compute_gradient_smooth(a, tau)

        # Project gradient onto simplex tangent space
        grad_proj = grad - np.mean(grad)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_proj
        v = beta2 * v + (1.0 - beta2) * grad_proj ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

        # Update tau (decrease to sharpen)
        tau *= tau_rate

        # Track best using true (non-smooth) objective
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation to escape local minima
        if no_improve >= perturb_every:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            m *= 0.5
            v *= 0.5
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            tau = max(tau, 2.0 * conv.max())
            lr = max(0.01, lr * 0.7)

    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations of length n on the simplex."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    # Triangle
    tri = np.minimum(x.astype(np.float64), n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Gaussians
    for sf in [0.2, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random sparse
    n_nonzero = max(2, n // 4)
    idx = rng.choice(n, size=n_nonzero, replace=False)
    sparse = np.zeros(n)
    sparse[idx] = rng.dirichlet(np.ones(n_nonzero) * 0.3)
    inits['sparse'] = sparse

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) >= n:
                pp = p[:n]
                pp /= pp.sum()
                inits['parent_crop'] = pp
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
            if len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Fienup IO algorithm for fast flat-spectrum initialization
    Phase 2: Adam smooth gradient descent for fine-tuning
    Phase 3: Final polish with low temperature
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    results = []  # (score, n, a_array)

    # === Phase 1: IO Algorithm Exploration ===
    print("=== Phase 1: Fienup IO Exploration ===")
    t0 = time.monotonic()
    n_values = [128, 256, 512, 1024, 2048]

    phase1_budget = 400.0
    time_per_n = phase1_budget / len(n_values)

    for n in n_values:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        # Limit inits per n
        max_inits = 4
        priority = ['parent_crop', 'parent_up', 'parent_ds', 'random', 'gauss_0.4',
                    'uniform', 'ramp_inc', 'bimodal', 'triangle', 'sparse']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in inits if nm not in priority]
        init_names = init_names[:max_inits]

        time_per_init = time_per_n / len(init_names)

        for name in init_names:
            if time.monotonic() > deadline - 580:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 570)
            if t_alloc < 2:
                break

            a0 = inits[name]
            n_iters = min(500, max(50, int(t_alloc * 50)))

            # Run IO with different relaxation parameters
            for relax in [0.8, 0.5]:
                if time.monotonic() > deadline - 570:
                    break
                a_opt, obj_norm = io_flat_spectrum(a0, n_iters=n_iters // 2,
                                                   relaxation=relax)
                # Evaluate with actual objective (sum=1, so score = 2n*max_conv)
                conv = fast_convolve(a_opt)
                score = 2.0 * n * conv.max()
                results.append((score, n, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  IO: n={n}, {name}, relax={relax}, score={score:.6f}")

    print(f"Phase 1 done. Best so far: {best_score:.6f} (time: {time.monotonic()-t0:.1f}s)")

    # Sort results and keep top ones
    results.sort(key=lambda x: x[0])
    top_results = results[:10]

    # === Phase 2: Adam Smooth Descent ===
    print("=== Phase 2: Adam Smooth Descent ===")
    t1 = time.monotonic()
    phase2_budget = min(1000.0, deadline - time.monotonic() - 200.0)

    # Deduplicate: keep best per n
    best_per_n = {}
    for score, n, a_arr in top_results:
        if n not in best_per_n or score < best_per_n[n][0]:
            best_per_n[n] = (score, a_arr)

    candidates = []
    for n, (score, a_arr) in best_per_n.items():
        candidates.append((n, a_arr))
        # Also add a "un-sparse" version
        unsp = a_arr * 0.7 + 0.3 / n
        unsp /= unsp.sum()
        candidates.append((n, unsp))

    # Add fresh random candidates at promising n values
    for n in [256, 512]:
        if n not in best_per_n:
            a_rand = rng.dirichlet(np.ones(n) * 0.3)
            candidates.append((n, a_rand))

    time_per_cand = phase2_budget / max(1, len(candidates))

    for i, (n, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(time_per_cand, deadline - time.monotonic() - 170)
        if t_alloc < 5:
            break

        # Run Adam with different hyperparameters
        configs = [
            {'lr': 0.05, 'tau_init': 5.0, 'tau_rate': 0.999},
            {'lr': 0.02, 'tau_init': 10.0, 'tau_rate': 0.9995},
            {'lr': 0.1, 'tau_init': 3.0, 'tau_rate': 0.998},
        ]

        for cfg in configs[:2]:  # Limit to 2 configs per candidate
            if time.monotonic() > deadline - 160:
                break
            t_single = t_alloc / 2.0
            if t_single < 3:
                break

            a_opt, obj_norm = adam_smooth_descent(a0, time_limit=t_single, **cfg)
            conv = fast_convolve(a_opt)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Adam: n={n}, cand {i}, lr={cfg['lr']}, score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f} (time: {time.monotonic()-t1:.1f}s)")

    # === Phase 3: Final Polish ===
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        t2 = time.monotonic()
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)

        # Multiple polish attempts
        polish_configs = [
            {'lr': 0.01, 'tau_init': 2.0, 'tau_rate': 0.9999, 'perturb_every': 3000},
            {'lr': 0.005, 'tau_init': 1.0, 'tau_rate': 0.99995, 'perturb_every': 5000},
            {'lr': 0.03, 'tau_init': 5.0, 'tau_rate': 0.999, 'perturb_scale': 0.005},
        ]

        t_per_polish = remaining / (len(polish_configs) + 2)

        for cfg in polish_configs:
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(t_per_polish, deadline - time.monotonic() - 25)
            if t_alloc < 3:
                break

            # IO + Adam combo
            a_io, _ = io_flat_spectrum(best_sequence, n_iters=200, relaxation=0.6)
            a_opt, _ = adam_smooth_descent(a_io, time_limit=t_alloc * 0.6, **{k: v for k, v in cfg.items() if k != 'perturb_every'})
            # Also try Adam directly from best
            a_opt2, _ = adam_smooth_descent(best_sequence, time_limit=t_alloc * 0.4, **{k: v for k, v in cfg.items() if k != 'perturb_every'})

            for a_cand in [a_opt, a_opt2]:
                conv = fast_convolve(a_cand)
                score = 2.0 * n_best * conv.max()
                if score < best_score:
                    best_score = score
                    best_sequence = a_cand.copy()
                    print(f"  Polish: score={score:.6f}")

        # Random perturbation + re-optimize
        if time.monotonic() < deadline - 30:
            for _ in range(5):
                if time.monotonic() > deadline - 30:
                    break
                noise = rng.standard_normal(n_best) * 0.005
                a_pert = project_simplex(best_sequence + noise)
                t_rem = deadline - time.monotonic() - 25
                if t_rem < 3:
                    break
                a_opt, _ = adam_smooth_descent(a_pert, time_limit=min(t_rem / 3, 20),
                                               lr=0.02, tau_init=3.0, tau_rate=0.999)
                conv = fast_convolve(a_opt)
                score = 2.0 * n_best * conv.max()
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb+opt: score={score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = 2.0

    # Verify with evaluate_sequence
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
```
