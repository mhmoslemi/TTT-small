        # Gradient of max at argmax: 2*a_{k*-j}
        k = idx
        grad = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx_arr = np.arange(j_start, j_end + 1)
        grad[idx_arr] = 2.0 * a[k - idx_arr]
        smooth_obj = 2.0 * n * b.max()
        return smooth_obj, grad

    w /= w_sum
    smooth_obj = 2.0 * n * (Tb.max() + np.log(w_sum)) / T  # = 2*n*logsumexp(T*b)/T
    
    # Gradient: 4*n * c where c[j] = sum_k w[k] * a[k-j] (if 0 <= k-j < n)
    # c[j] = (w * a_rev)[n-1+j]
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(W * AR, fft_size)
    c = conv_result[n - 1:2 * n - 1]
    grad = 4.0 * n * c
    return smooth_obj, grad, b


def mirror_descent(a0, time_limit, T_init=20.0, T_max=2000.0, eta_init=1.0,
                   eta_decay=0.99995, batch=20, restart_freq=200):
    """
    Mirror descent (multiplicative weights) with smooth max.
    Works in the simplex {a >= 0, sum(a) = 1}.
    Returns (best_a, best_exact_obj) where best_exact_obj = 2*n*max(conv).
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_exact = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_init
    eta = eta_init
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            smooth_obj, grad, b = smooth_objective_and_grad(a, T)
            
            # Multiplicative weights update
            # a <- a * exp(-eta * grad) / sum(a * exp(-eta * grad))
            # For numerical stability, work in log-space
            log_update = -eta * grad
            log_update -= log_update.max()
            a_new = a * np.exp(log_update)
            s = a_new.sum()
            if s > 1e-300:
                a_new /= s
            else:
                a_new = a.copy()
            
            # Ensure non-negativity (should be automatic from multiplicative update)
            a = np.maximum(a_new, 0.0)
            s = a.sum()
            if s < 1e-300:
                a = np.ones(n) / n
            else:
                a /= s
            
            # Check exact objective
            conv_new = fast_convolve(a)
            cur_exact = 2.0 * n * conv_new.max()
            
            if cur_exact < best_exact - 1e-12:
                best_exact = cur_exact
                best_a = a.copy()
            
            iter_count += 1
        
        # Adaptive schedule
        eta *= eta_decay
        T = min(T_max, T * 1.005)
        
        # Periodic restart
        if iter_count % restart_freq == 0:
            # Check if we've plateaued
            # Perturb if needed
            pass
    
    return best_a, best_exact


def coordinate_descent_polish(a0, time_limit, max_iters=100000):
    """
    Coordinate descent: optimize one or two coordinates at a time.
    Minimizes the exact max of conv(a,a) subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick a random coordinate to adjust
        j = rng.integers(0, n)
        
        # We want to adjust a[j] while keeping sum(a) = 1
        # Move mass from a[j] to another coordinate k, or vice versa
        k = rng.integers(0, n)
        if j == k:
            continue
        
        # Try moving small amounts of mass from j to k (and k to j)
        # Parameterize: a'[j] = a[j] - delta, a'[k] = a[k] + delta
        # Need a[j] - delta >= 0 and a[k] + delta >= 0
        # So delta in [-a[k], a[j]]
        
        # The objective as a function of delta is:
        # 2*n*max_l (a'*a')_l where (a'*a')_l only changes for l that involve j or k
        # Specifically, (a'*a')_l = (a*a)_l + delta*(a_{l-j} + a_{l-k}) + delta^2*(delta_{l,2j} or cross terms)
        # This is a quadratic in delta for each l.
        # The max of quadratics is convex, so we can ternary search.
        
        # Affected l values: those where l-j in [0,n-1] or l-k in [0,n-1]
        # l-j in [0,n-1] => l in [j, j+n-1]
        # l-k in [0,n-1] => l in [k, k+n-1]
        l_min = max(0, min(j, k))
        l_max = min(2*n-2, max(j+n-1, k+n-1))
        
        # Get current values at affected l
        affected_l = np.arange(l_min, l_max + 1)
        
        # (a*a)_l = sum_i a[i]*a[l-i]
        # d/delta (a'*a')_l = sum_i [d a'[i]/d delta * a'_{l-i} + a'_i * d a'_{l-i}/d delta]
        # = (a'_{l-j} + a'_{l-k}) + delta * (cross terms from delta^2)
        # Actually it's simpler: (a'*a')_l as a function of delta is quadratic.
        # Let's just evaluate at a few points and find the minimum.
        
        # Current conv at affected positions
        # We can compute the change efficiently
        # (a'*a')_l - (a*a)_l = delta*(a_{l-j} + a_{l-k}) + delta^2 * I(l = 2j or l = 2k or j+k = l... )
        # Wait, more carefully:
        # a'[i] = a[i] - delta*I(i=j) + delta*I(i=k)
        # (a'*a')_l = (a*a)_l + delta*(a_{l-j}*I(0<=l-j<n) + a_{l-k}*I(0<=l-k<n))
        #              + delta*(a_{l-j}*I(0<=l-j<n) + a_{l-k}*I(0<=l-k<n))  [from the other side]
        #              - 2*delta^2*I(l=j+k, 0<=j<n, 0<=k<n)  [cross term]
        #              + delta^2*I(l=2j, 0<=j<n) + delta^2*I(l=2k, 0<=k<n)
        # 
        # Hmm, this is getting complicated. Let me just do a small grid search over delta.
        
        # Try a few delta values
        max_delta = min(a[j], 0.5 / n)  # don't move too much
        deltas = np.linspace(-max_delta, max_delta, 11)
        
        best_delta = 0
        best_val = best_obj
        
        current_conv = conv.copy()
        for d in deltas:
            if a[j] - d < 0 or a[k] + d < 0:
                continue
            # Compute change in conv
            # For each affected l:
            # change_l = d*(a[l-j] if 0<=l-j<n else 0 + a[l-k] if 0<=l-k<n else 0)
            #           + d*(a[l-j] if 0<=l-j<n else 0 + a[l-k] if 0<=l-k<n else 0)
            #           - 2*d^2*(1 if l==j+k and 0<=j<n and 0<=k<n else 0)  [wait, this isn't right]
            # Let me just recompute the affected conv values directly
            
            # (a'*a')_l for affected l
            # = current_conv[l] + d * 2 * (a[l-j] if 0<=l-j<n else 0 + a[l-k] if 0<=l-k<n else 0)
            #   - d^2 * 2 * (1 if j+k==l)  [because a'_j * a'_k = (a_j-d)(a_k+d) = a_j*a_k + d(a_j-a_k) - d^2]
            #   + d^2 * (1 if l==2j or l==2k)
            #
            # This is getting messy. Let me just recompute the full conv for candidate deltas
            # but only for the affected positions.
            
            pass
        
        # Actually, for n=256, recomputing the full conv is fast (~10μs).
        # Let me just try a few deltas and recompute.
        for d in deltas:
            if a[j] - d < -1e-15 or a[k] + d < -1e-15:
                continue
            a_test = a.copy()
            a_test[j] = max(0, a[j] - d)
            a_test[k] = max(0, a[k] + d)
            s_test = a_test.sum()
            if s_test < 1e-15:
                continue
            a_test /= s_test  # renormalize (should be ~1 already)
            conv_test = fast_convolve(a_test)
            val = 2.0 * n * conv_test.max()
            if val < best_val - 1e-12:
                best_val = val
                best_delta = d
        
        if best_delta != 0:
            a[j] = max(0, a[j] - best_delta)
            a[k] = max(0, a[k] + best_delta)
            s = a.sum()
            if s > 1e-15:
                a /= s
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
    
    return best_a, best_obj


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp_inc'] = ramp / ramp.sum()
    inits['ramp_dec'] = ramp[::-1] / ramp.sum()
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Gaussian (center)
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits['gauss_center'] = g / g.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Random sparse
    sparse = np.zeros(n)
    n_nonzero = max(2, n // 4)
    idx = rng.choice(n, size=n_nonzero, replace=False)
    sparse[idx] = 1.0
    inits['sparse'] = sparse / sparse.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

    Main entry point. Multi-phase search using smooth gradient mirror descent,
    coordinate descent polish, and multi-length exploration.
    """

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    def update_best(a):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            return True
        return False
    
    # Phase 1: Quick exploration across lengths and inits
    print("=== Phase 1: Quick Exploration ===")
    lengths = [128, 256, 512, 1024]
    phase1_deadline = time.monotonic() + 400.0
    
    results = []
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        
        # Prioritize
        priority = ['parent', 'parent_ds', 'parent_up', 'cosine', 'uniform', 'ramp_inc', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        
        t_per = max(5.0, 300.0 / (len(lengths) * 4))

            if time.monotonic() > phase1_deadline - 10:
                break

            if t_alloc < 3:

            a_opt, _ = mirror_descent(
                a0, time_limit=t_alloc,
                T_init=20.0, T_max=500.0, eta_init=0.5,
                eta_decay=0.9999, batch=10)

                print(f"  Phase1: n={n}, {name}, score={best_score:.6f}")

    # Phase 2: Focused optimization
    print(f"=== Phase 2: Focused Optimization (best so far: {best_score:.6f}) ===")
    
    # Use the best from phase 1 and parent

    if best_sequence is not None:
        candidates.append(best_sequence.copy())
    if parent is not None:

            pp = np.maximum(pp, 0)

            if ps > 1e-10:

    n_best = len(best_sequence) if best_sequence is not None else 256

    for name in ['cosine', 'random', 'ramp_dec']:
