"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Combines softmax gradient descent, LP-based peak reduction, and coordinate descent.
Explores multiple sequence lengths with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def compute_soft_gradient(a, tau):
    """Compute gradient of soft-max autocorrelation w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        end = min(n, len(w) - i)
        if end > 0:
            grad[i] = 2.0 * np.dot(w[i:i+end], a[:end])
    return grad, c


def lp_solve_peak(f, M, n_max=1024):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if n > n_max:
        return None
    # Build constraint matrix A (2n-1) x n
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        m_start = j
        m_end = min(j + n, 2*n-1)
        A[m_start:m_end, j] = f[:m_end-m_start]
    
    c_obj = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def gradient_descent_loop(a, deadline, tau_start=5.0, tau_end=50.0, 
                          lr_start=0.3, momentum=0.9):
    """Run softmax gradient descent with adaptive parameters."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    mo_buf = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / 10000.0)
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_start * (0.9998 ** it)
        lr = max(lr, 0.01)
        
        grad, c = compute_soft_gradient(a, tau)
        # Project out mean to maintain sum constraint
        grad -= grad.mean()
        mo_buf = momentum * mo_buf - lr * grad
        a_new = a + mo_buf
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        
        if it % 500 == 0:
            s = compute_obj(a_new)
            if s < best_score - 1e-9:
                best_score = s
                best = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    break
            a = a_new
        else:
            a = a_new
    
    return best, best_score


def lp_refinement(a, deadline, n_max=768):
    """Iterative LP-based refinement."""
    n = len(a)
    if n > n_max:
        return a, compute_obj(a)
    
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 10:
        c = np.convolve(a, a)
        M = float(c.max())
        
        # Solve LP with slightly reduced M
        M_reduced = M * 0.95
        g = lp_solve_peak(a, M_reduced, n_max)
        
        if g is None or g.sum() < 1e-6:
            # Try with original M
            g = lp_solve_peak(a, M, n_max)
            if g is None or g.sum() < 1e-6:
                break
        
        g_sum = g.sum()
        g_norm = g * (S_t / g_sum)
        
        improved = False
        for t_blend in [0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best, best_score


def coordinate_polish(a, deadline, max_rounds=5):
    """Coordinate descent polishing."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            # Try a few candidate values
            candidates = [0.0, orig * 0.5, orig * 1.5, orig * 0.8, orig * 1.2]
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a_n = normalize(a, S_t)
                s = compute_obj(a_n)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True
            else:
                a[i] = orig
        
        if not improved:
            break
    
    return a, best_score

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses a combination of: (1) top-M peak averaging gradient descent with adaptive
step size, (2) periodic LP-based direction finding, (3) coordinate descent

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def top_m_gradient(a, c, m=5):
    """
    Compute gradient that averages over the top m peaks of the autocorrelation.
    Returns gradient vector and the top-m weights.
    """
    n = len(a)
    # Find top m lags
    c_sorted_idx = np.argsort(c)[::-1][:m]
    c_top = c[c_sorted_idx]
    c_max = c_top[0]
    # Soft weighting among top m
    w = np.exp(10.0 * (c_top - c_max))
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(c_sorted_idx):
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += w[idx] * 2.0 * a[j]
    return grad


def softmax_gradient(a, c, tau=10.0):
    """Compute gradient of soft-max of autocorrelation."""
    n = len(a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))

    return grad


def lp_direction(f, M, n_max=800):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns optimal g or None.
    """

    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0

    for j in range(n):
        m_start = j
        m_end = min(j + n, 2 * n - 1)
        length = m_end - m_start
        A[m_start:m_end, j] = f[:length]

    b = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n

                      options={'presolve': True, 'time_limit': 30})

        pass
    return None

            orig = a[i]
            c = np.convolve(a, a)
            k_star = int(np.argmax(c))
            j = k_star - i
            candidates = [0.0, orig * 0.5, orig * 1.5]
            if 0 <= j < n and j != i:
                candidates.append(orig * 0.7)
                candidates.append(orig * 1.2)
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a_n = normalize(a, S_t)
                s = compute_obj(a_n)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True
            else:
                a[i] = orig
        if not improved:
            break
    return a, best_score


def generate_inits(rng, n):
    """Generate diverse initial sequences."""
    inits = []
    i = np.arange(n, dtype=np.float64)
    
    # From height_sequence_1
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= n and np.all(np.isfinite(gs)):
            inits.append(gs[:n].copy())
    except (NameError, Exception):
        pass
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Random
    inits.append(rng.random(n) * 0.5 + 0.25)
    
    # Cosine modulations
    for freq in [1, 2, 3, 5, 7]:
        inits.append(0.5 + 0.3 * np.cos(2 * np.pi * i * freq / n))
    
    # Edge-weighted
    inits.append(0.3 + 0.4 * np.abs(2.0 * i / n - 1.0))
    
    # Gaussian
    inits.append(np.exp(-0.5 * ((i - n / 2) / (n * 0.25)) ** 2))
    
    # Sparse
    a_s = np.zeros(n)
    mask = rng.random(n) < 0.4
    a_s[mask] = rng.random(mask.sum()) * 0.5 + 0.2
    inits.append(a_s)
    
    # Sine
    inits.append(np.abs(np.sin(2 * np.pi * i * 3.0 / n)) + 0.1)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-start optimization combining gradient descent
    (top-M peak + softmax), LP-based refinement, and coordinate descent.
    Returns the best non-negative sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    print(f"=== Starting optimization, budget={budget_s}s ===")
    
    # Phase 0: Optimize the existing solution first
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            n_g = len(gs)
            s_g = compute_obj(gs)
            print(f"  height_sequence_1: n={n_g}, score={s_g:.6f}")
            t_alloc = min(120.0, (deadline - t_start) * 0.25)
            seq, score = main_optimize(gs, t_start + t_alloc, use_lp=(n_g <= 768))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Optimized existing: {best_score:.6f}")
    except (NameError, Exception) as e:
        print(f"  No height_sequence_1: {e}")
    
    # Phase 1: Try different lengths with diverse starts
    lengths = [512, 384, 640]
    
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        inits = generate_inits(rng, n)
        n_inits = len(inits)
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 40:
                break
            
            remaining = deadline - time.monotonic() - 30
            # Allocate time based on remaining budget and how many inits left
            inits_left = n_inits - idx
            lengths_left = max(1, len(lengths) - lengths.index(n) - (1 if idx == n_inits - 1 else 0))
            total_left = inits_left + lengths_left
            t_alloc = min(120.0, remaining / max(1, total_left))
            t_alloc = max(25.0, t_alloc)
            
            if time.monotonic() + t_alloc > deadline - 25:
                t_alloc = deadline - time.monotonic() - 25
                if t_alloc < 15:
                    break
            
            print(f"  n={n}, init {idx+1}/{n_inits}, alloc={t_alloc:.0f}s")
            use_lp = (n <= 700)
            seq, score = main_optimize(a0, time.monotonic() + t_alloc, use_lp=use_lp)
            
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  *** New best: {best_score:.6f}, n={n} ***")
            else:
                print(f"  score={score:.6f} (not better)")
    
    # Phase 2: Polish the best solution
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"\n=== Final polish: n={n_best}, score={best_score:.6f} ===")
        
        # Coordinate descent polish
        if n_best <= 512:
            a_cd, s_cd = coordinate_polish(best_seq, deadline - 15, max_rounds=3)
            if s_cd < best_score:
                best_score = s_cd
                best_seq = a_cd
                print(f"  After coordinate polish: {best_score:.6f}")
        
        # A few perturbed restarts
        for trial in range(3):
            if time.monotonic() > deadline - 15:
                break
            pert = rng.normal(0, 0.03, n_best)
            a0 = np.maximum(best_seq + pert, 0)
            t_alloc = min(60.0, deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            seq, score = main_optimize(a0, time.monotonic() + t_alloc,
                                       use_lp=(n_best <= 700))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Perturb {trial}: {best_score:.6f}")
        
        # Final LP pass
        if n_best <= 640 and time.monotonic() < deadline - 10:
            c = np.convolve(best_seq, best_seq)
            M = float(c.max())
            g = lp_direction(best_seq, M * 0.88, n_max=640)
            if g is not None and g.sum() > 1e-6:
                S_t = np.sqrt(2.0 * n_best)
                g_norm = g * (S_t / g.sum())
                for t_b in [0.4, 0.3, 0.2, 0.1, 0.05]:
                    a_new = (1.0 - t_b) * best_seq + t_b * g_norm
                    a_new = np.maximum(a_new, 0.0)
                    a_new = normalize(a_new, S_t)
                    s_new = compute_obj(a_new)
                    if s_new < best_score:
                        best_score = s_new
                        best_seq = a_new
                        print(f"  Final LP blend (t={t_b}): {best_score:.6f}")
                        break
    
    # Validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"eval error: {e}")
        # Ensure all values are valid
        result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
        return result
    
    # Fallback
    print("Falling back to uniform sequence of length 384.")
    return [0.5] * 384
