"""
Multi-strategy optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines improved LP iterations with adaptive
peak tightening, softmax-gradient descent, and focused local search.
Uses structured initializations (step patterns, chirps) and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_A_matrix(f):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<len(f)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_tight(f, M, delta=0.0, time_limit=60):
    """
    Solve LP: max sum(g) s.t. conv(f,g)_k <= M*(1-delta) for all k, g >= 0.
    Returns (g, feasibility) where feasibility is True if solution found.
    """
    n = len(f)
    A = build_A_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * (1.0 - delta))
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            return result.x, True
    except Exception:
        pass
    return None, False


def lp_optimize_improved(f_init, time_limit, verbose=False, t_init=0.25):
    """
    Improved LP-based iterative optimization with adaptive tightness and line search.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    delta = 0.0
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break

        conv = fft_convolve(f, f)
        M = np.max(conv)

        # Try tightening: start with small delta, increase if infeasible
        g = None
        for d_try in [delta, 0.0, 0.005, 0.01, 0.02, 0.05]:
            if d_try < 0:

            g, ok = solve_lp_tight(f, M, delta=d_try, time_limit=20)
            if ok and g is not None and np.sum(g) > 1e-10:

                break

        if g is None or np.sum(g) < 1e-10:

        # = a_{k-m} (if k-m == m, i.e., k = 2m)
        # Hmm, this is getting complicated. Let me just use:
        # d(conv(a,a)_k)/da_m = a[k-m] (if 0<=k-m<n) + a[k-m] (if 0<=k-m<n and k-m != m)
        # No that's wrong too. Let me be very careful.
        #
        # conv(a,a)_k = sum_{j=0}^{n-1} a[j] * a[k-j] (with a[x]=0 if x<0 or x>=n)
        # d/d(a[m]) conv(a,a)_k = a[k-m] + a[k-m] if k-m != m
        #                         = a[k-m] if k-m == m (i.e., k = 2m)
        # Wait no. Let's expand:
        # conv(a,a)_k = ... + a[m]*a[k-m] + ... + a[k-m]*a[m] + ...
        # If k-m != m, both terms a[m]*a[k-m] and a[k-m]*a[m] are present and distinct,
        # so derivative = a[k-m] + a[m]... no, derivative of a[j]*a[k-j] w.r.t. a[m]:
        # If j=m: derivative = a[k-j] = a[k-m]
        # If k-j=m i.e. j=k-m: derivative = a[j] = a[k-m]
        # So if m != k-m (i.e., k != 2m): d/d(a[m]) = a[k-m] + a[k-m] = 2*a[k-m]
        # If m == k-m (i.e., k = 2m): only one term, d/d(a[m]) = a[m] (the term is a[m]*a[m])
        #
        # So: dC_soft/da_m = sum_k w_k * [2*a[k-m] if k!=2m; a[m] if k==2m]
        #                    (where a[x]=0 if x not in [0,n))
        #
        # This is approximately 2 * (w * a)[m] in the circular sense, with a correction at k=2m.
        # For large n, the correction is negligible. Let's use the approximation:
        # dC_soft/da_m ≈ 2 * sum_k w_k * a[(k-m) mod n]  ... but we need linear conv.
        #
        # Actually for linear convolution: dC/da_m = sum_k w_k * a[k-m] (if 0<=k-m<n)
        # plus sum_k w_k * a[k-m] (same term again) = 2 * sum_k w_k * a[k-m] (if 0<=k-m<n)
        # EXCEPT when k=2m where we only get a[m] once.
        #
        # For simplicity, let's use: grad[m] = 2 * sum_{k} w_k * a[k-m] * I(0<=k-m<n)
        # This slightly overestimates the gradient at the diagonal but is a good approximation.

        # Compute grad efficiently: grad[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
        # This is essentially the convolution of w and a, evaluated at specific positions
        # Let's compute it via FFT for efficiency
        wa = fft_convolve(w, a)  # wa[k] = sum_j w[j] * a[k-j]
        # grad[m] = 2 * sum_k w[k] * a[k-m] ... hmm this is not quite wa
        # wa[k] = sum_j w[j]*a[k-j], so wa[k+m] = sum_j w[j]*a[k+m-j]
        # What we want: sum_k w[k]*a[k-m] = sum_k w[k]*a[k-m]
        # Let l = k-m: = sum_l w[l+m]*a[l] = cross-correlation
        # This is wa reversed... let me just compute it directly for n up to ~2000
        # For larger n, use FFT-based approach

        if n <= 2000:
            grad = np.zeros(n)
            for m in range(n):
                # sum_k w[k]*a[k-m] for 0<=k-m<n i.e. m<=k<n+m
                lo = m
                hi = min(2 * n - 1, n + m - 1)
                if lo < hi:
                    grad[m] = 2.0 * np.dot(w[lo:hi], a[hi - m - 1:lo - 1 - 1:-1] if False else a[(hi - 1 - m):(lo - 1 - m):-1][::-1])
            # That indexing is getting messy. Let me use a simpler approach.
            # grad[m] = 2 * sum_{k=m}^{n+m-1} w[k] * a[k-m] (where w has length 2n-1)
            #          = 2 * dot(w[m:n+m], a[0:n]) reversed...
            # Actually: sum_{k=m}^{m+n-1} w[k] * a[k-m] = dot(w[m:m+n], a)
            # But w has length 2n-1, so w[m:m+n] is valid for m <= n-1
            # And a[k-m] for k in [m, m+n-1] gives a[0] to a[n-1]
            grad = np.zeros(n)
            for m in range(n):
                seg = w[m:m + n]
                if len(seg) == n:
                    grad[m] = 2.0 * np.dot(seg, a)
                else:
                    # Pad with zeros
                    pad = np.zeros(n - len(seg))
                    grad[m] = 2.0 * np.dot(np.concatenate([pad, seg]), a)
        else:
            # FFT-based: grad[m] = 2 * (wپadded * a)[m] ... approximate
            w_pad = np.zeros(2 * n - 1)
            w_pad[:len(w)] = w
            # grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j] = 2 * (w * reverse(a))[m + n - 1] ...
            # Let's just use the direct formula with FFT
            size = 1
            while size < 2 * n:
                size <<= 1
            W = np.fft.rfft(w, size)
            A = np.fft.rfft(a, size)
            # grad[m] = 2 * sum_j w[m+j]*a[j] = 2 * circular correlation
            # = 2 * ifft(W * conj(A))[m] ... approximately
            corr = np.fft.irfft(W * np.conj(np.fft.rfft(a, size)), size)
            grad = 2.0 * corr[:n]

        # Full gradient of F (with sum(a)=1 constraint, F = 2n * C)
        # We're optimizing on the simplex, so we project the gradient
        g_full = 2.0 * n * grad  # gradient of 2n*C w.r.t. a

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * g_full
        v = beta2 * v + (1 - beta2) * (g_full ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        lr = base_lr / np.sqrt(1.0 + 0.05 * step)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project onto simplex {a >= 0, sum(a) = 1}
        f_new = np.clip(f_new, 0.0, None)
        s = np.sum(f_new)
        if s < 1e-10:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s

        f = f_new
        score = local_score(f * np.sqrt(2 * n))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f.copy()

        if verbose and step % 100 == 0:
            print(f"      Grad step={step}, score={score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.5f}, lr={lr:.6f}")

    return best_f * np.sqrt(2 * n), best_score


def local_search(f_init, time_limit, verbose=False):
    """
    Focused local search: perturb elements that contribute most to the peak.
    Uses coordinate-style updates with geometric sampling.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    current_score = best_score
    temperature = 0.002
    cooling = 0.9999
    skipped = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break

        candidate = f.copy()
        r = rng.random()

        if r < 0.4:
            # Peak-focused move: reduce top contributors to peak, increase others
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            total_c = np.sum(contribs) + 1e-10
            # Top contributors
            k_top = max(5, n // 8)
            top_idx = np.argsort(contribs)[-k_top:]
            other_idx = np.setdiff1d(np.arange(n), top_idx)

            # Reduce top contributors
            frac_reduce = rng.uniform(0.02, 0.08)
            candidate[top_idx] *= (1.0 - frac_reduce * rng.random(k_top))
            # Increase others (proportional to their "safety" - distance from peak)
            if len(other_idx) > 0:
                n_inc = min(max(3, n // 15), len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= (1.0 + frac_reduce * 1.5 * rng.random(n_inc))
        elif r < 0.65:
            # Block perturbation
            n_blocks = rng.integers(2, 6)
            block_size = max(1, n // (n_blocks * 2))
            for _ in range(n_blocks):
                start = int(rng.integers(0, max(1, n - block_size)))
                length = int(rng.integers(1, block_size + 1))
                end = min(start + length, n)
                candidate[start:end] *= np.exp(rng.normal(0.0, 0.03, size=end - start))
        elif r < 0.8:
            # Single element or pair update
            n_mod = rng.integers(1, 4)
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.05, size=n_mod))
        elif r < 0.9:
            # Smooth oscillation
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.arange(n)
            candidate *= np.exp(0.02 * np.sin(2 * np.pi * freq * t / n + phase))
        else:
            # Asymmetric shift (breaks symmetry)
            shift = int(rng.integers(1, max(2, n // 16)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))

        candidate = np.clip(candidate, 1e-10, 1000.0)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        skipped += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
                temperature = min(temperature * 1.2, 0.005)
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score

        temperature *= cooling
        if temperature < 1e-6:
            temperature = 1e-6

        if verbose and skipped % 500 == 0:
            print(f"      LS iter={skipped}, temp={temperature:.6f}, "
                  f"score={current_score:.6f}, best={best_score:.6f}")

    return best_f, best_score


def make_step_init(n, rng, n_high=None, ratio=0.4):
    """Step-like initialization: high on left, low on right."""
    if n_high is None:
        n_high = n // 2
    a = np.zeros(n)
    a[:n_high] = 1.0
    a[n_high:] = ratio
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):
    """Quadratic chirp initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.5 * rng.random()
    a = 1.0 + alpha * np.cos(2 * np.pi * t * t / (2 * n) + rng.uniform(0, 2 * np.pi))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng, n_tones=8):
    """Sum of random sinusoids."""
    a = np.ones(n)
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_ramp_init(n, rng, power=1.5):
    """Ramp-like initialization."""
    t = np.linspace(0, 1, n)
    a = t ** power
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_antisymmetric_init(n, rng):
    """Antisymmetric-ish: high where mirror is low."""
    a = np.ones(n)
    t = np.arange(n, dtype=np.float64)
    # Make left side high, right side low (and vice versa for odd positions)
    a[t < n // 2] = 1.5
    a[t >= n // 2] = 0.5
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-based optimization from multiple initializations
    Phase 2: Gradient-based polishing
    Phase 3: Local search refinement
    Phase 4: Final LP polish
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    def update_best(f, score, label=""):
        nonlocal best_score, best_seq
        if score < best_score - 1e-12:
            best_score = score
            best_seq = f.copy()
            print(f"  *** New best: {best_score:.6f} ({label}) ***")

    # === Phase 1: LP-based optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: LP-based iterative optimization")
    print("=" * 60)

    n_prov = len(provided_seq) if provided_seq is not None else 256

    # Define search strategies: (n, init_func, time_alloc_fraction)
    # We'll allocate time proportionally
    strategies = []

    if provided_seq is not None:
        strategies.append((n_prov, 'provided', 0.30))
    strategies.append((n_prov, 'step', 0.15))
    strategies.append((n_prov, 'antisym', 0.10))
    strategies.append((n_prov, 'chirp', 0.12))
    strategies.append((n_prov, 'multitone', 0.10))
    strategies.append((512, 'step', 0.10))
    strategies.append((512, 'chirp', 0.08))
    strategies.append((n_prov, 'ramp', 0.05))

    total_frac = sum(s[2] for s in strategies)
    available_time = deadline - time.monotonic()

    for n, init_name, frac in strategies:
        if time.monotonic() > deadline - 60:
            break

        time_alloc = frac / total_frac * available_time * 0.80
        time_alloc = max(20.0, time_alloc)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 10)
        if time_alloc < 15:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = make_antisymmetric_init(n, rng) if n != n_prov else provided_seq.copy()
            if n == n_prov:
                a0 = provided_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(provided_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, provided_seq)
        elif init_name == 'step':
            ratio = 0.3 + 0.4 * rng.random()
            n_high = int(n * (0.3 + 0.4 * rng.random()))
            a0 = make_step_init(n, rng, n_high=n_high, ratio=ratio)
        elif init_name == 'chirp':
            a0 = make_chirp_init(n, rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng, n_tones=6 + int(3 * rng.random()))
        elif init_name == 'ramp':
            a0 = make_ramp_init(n, rng, power=0.5 + 2.0 * rng.random())
        elif init_name == 'antisym':
            a0 = make_antisymmetric_init(n, rng)
        else:
            a0 = 1.0 + 0.3 * rng.standard_normal(n)
            a0 = np.clip(a0, 0.01, None)

        print(f"\n  Strategy: n={n}, init='{init_name}', time={time_alloc:.0f}s")
        t_start = time.monotonic()

        f_best, score = lp_optimize_improved(a0, time_alloc, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  LP done: score={score:.6f}, time={elapsed:.1f}s")

        update_best(f_best, score, f"n={n},init={init_name}")

    # === Phase 2: Gradient-based polishing ===
    print("\n" + "=" * 60)
    print("Phase 2: Gradient-based polishing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(400.0, remaining * 0.40)
        if grad_time > 30:
            n_best = len(best_seq)
            print(f"  Polishing n={n_best}, time={grad_time:.0f}s")
            t_start = time.monotonic()
            a_grad, score_grad = gradient_optimize(best_seq, grad_time, verbose=True)
            elapsed = time.monotonic() - t_start
            print(f"  Gradient polish: score={score_grad:.6f}, time={elapsed:.1f}s")
            update_best(a_grad, score_grad, "gradient")

    # === Phase 3: Local search ===
    print("\n" + "=" * 60)
    print("Phase 3: Local search refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        ls_time = min(350.0, remaining * 0.45)
        if ls_time > 20:
            n_best = len(best_seq)
            print(f"  Local search n={n_best}, time={ls_time:.0f}s")
            t_start = time.monotonic()
            a_ls, score_ls = local_search(best_seq, ls_time, verbose=True)
            elapsed = time.monotonic() - t_start
            print(f"  Local search: score={score_ls:.6f}, time={elapsed:.1f}s")
            update_best(a_ls, score_ls, "local_search")

    # === Phase 4: Final LP polish ===
    print("\n" + "=" * 60)
    print("Phase 4: Final LP refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic()
        final_lp_time = min(200.0, remaining * 0.80)
        if final_lp_time > 15:
            n_best = len(best_seq)
            print(f"  Final LP n={n_best}, time={final_lp_time:.0f}s")
            t_start = time.monotonic()
            a_final, score_final = lp_optimize_improved(best_seq, final_lp_time, verbose=False)
            elapsed = time.monotonic() - t_start
            print(f"  Final LP: score={score_final:.6f}, time={elapsed:.1f}s")
            update_best(a_final, score_final, "final_lp")

    # === Phase 5: Quick gradient + LS on best ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            half = remaining * 0.5
            print(f"\n  Quick gradient+LS: {remaining:.0f}s remaining")
            a_qg, s_qg = gradient_optimize(best_seq, half * 0.6, verbose=False)
            update_best(a_qg, s_qg, "quick_grad")

            if time.monotonic() < deadline - 10:
                remaining2 = deadline - time.monotonic() - 5
                if remaining2 > 10:
                    a_ql, s_ql = local_search(best_seq, remaining2, verbose=False)
                    update_best(a_ql, s_ql, "quick_ls")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
