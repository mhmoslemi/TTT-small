def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_and_grad(conv, T):
    """Compute smooth max and its gradient (as weights)."""
    scaled = conv / T
    scaled_max = scaled.max()
    exp_scaled = np.exp(scaled - scaled_max)
    w = exp_scaled / exp_scaled.sum()
    smooth_val = T * (np.log(exp_scaled.sum()) + scaled_max)
    return smooth_val, w
