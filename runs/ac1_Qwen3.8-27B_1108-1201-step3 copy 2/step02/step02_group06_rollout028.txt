            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.65:
            # Local window perturbation
            width = max(2, int(rng.integers(2, max(3, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.8:
            # Two-window perturbation
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.9:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Frequency-domain perturbation
            fa = np.fft.rfft(f)
            fa *= np.exp(rng.normal(0.0, 0.01, size=len(fa)) + 1j * rng.normal(0.0, 0.005, size=len(fa)))
            candidate = np.fft.irfft(fa, n)
            candidate = np.clip(candidate, 1e-10, None)

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (S / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            f = candidate
            current_score = new_score

        temperature *= cooling

    return best_f, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 1.5))
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = int(rng.integers(4, 10))
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_tapered_init(n, rng, beta=1.0):
    """Raised-cosine taper with random modulation."""
    t = np.arange(n, dtype=np.float64)
    taper = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
    a = 0.5 + 0.5 * taper ** beta
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n * rng.uniform(0.15, 0.4)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a += 0.05
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Quick GD from diverse initializations (multiple n values)
    Phase 2: Extended GD with annealing on best candidates
    Phase 3: SA refinement
    Phase 4: Final GD polish at very low temperature
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s_prov = local_score(provided_seq)
            print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
            if s_prov < best_score:
                best_score = s_prov
                best_seq = provided_seq.copy()
        except Exception:
            pass

    n_base = len(provided_seq) if provided_seq is not None else 256

    # === Phase 1: Quick GD from diverse initializations ===
    print("\n" + "=" * 60)
    print("Phase 1: Quick GD from diverse initializations")
    print("=" * 60)

    initializations = []

    # From provided
    if provided_seq is not None:
        initializations.append(('provided', provided_seq.copy()))

    # Chirp
    for alpha in [0.3, 0.5, 0.7]:
        initializations.append((f'chirp_{alpha}', make_chirp_init(n_base, alpha, rng)))

    # Multi-tone
    for _ in range(2):
        initializations.append(('multitone', make_multitone_init(n_base, rng)))

    # Tapered
    for beta in [0.5, 1.0, 2.0]:
        initializations.append((f'taper_{beta}', make_tapered_init(n_base, rng, beta)))

    # Gaussian
    initializations.append(('gauss', make_gaussian_init(n_base, rng)))

    # Random
    a_rand = 1.0 + 0.3 * rng.standard_normal(n_base)
    initializations.append(('random', np.clip(a_rand, 0.01, None)))

    # Try different lengths
    for n_alt in [384, 512]:
        if n_alt != n_base:
            initializations.append((f'chirp_{n_alt}', make_chirp_init(n_alt, 0.4, rng)))
            initializations.append((f'taper_{n_alt}', make_tapered_init(n_alt, rng, 1.0)))

    # Quick GD: allocate ~40s per initialization, cap at available time
    phase1_time_per = 35.0
    max_phase1_evals = len(initializations)

    for idx, (name, a0) in enumerate(initializations):
        if time.monotonic() > deadline - 300:
            break
        n = len(a0)
        print(f"  [{idx+1}/{len(initializations)}] n={n}, init='{name}', score_init={local_score(a0):.6f}")
        t_start = time.monotonic()
        a_new, score = soft_peak_gd(a0, phase1_time_per, seed=seed + idx, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"    -> score={score:.6f} ({elapsed:.1f}s)")
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}, init={name}) ***")

    # === Phase 2: Extended GD on best ===
    print("\n" + "=" * 60)
    print("Phase 2: Extended GD with annealing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        gd_time = min(400.0, remaining * 0.4)
        if gd_time > 30:
            print(f"  n={len(best_seq)}, time={gd_time:.0f}s, current best={best_score:.6f}")
            a_new, score = soft_peak_gd(best_seq, gd_time, seed=seed + 1000, verbose=True)
            print(f"  GD2 result: {score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Multi-start from best with perturbations ===
    print("\n" + "=" * 60)
    print("Phase 3: Multi-start refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 150:
        n = len(best_seq)
        n_starts = 5
        time_per_start = min(50.0, (deadline - time.monotonic() - 100) / n_starts)
        if time_per_start > 10:
            for s in range(n_starts):
                if time.monotonic() > deadline - 100:
                    break
                # Create perturbed version
                perturbation_scale = 0.1 * (s + 1) / n_starts
                a_pert = best_seq * np.exp(rng.normal(0, perturbation_scale, n))
                a_pert = np.clip(a_pert, 0.01, None)
                a_pert = a_pert * (np.sum(best_seq) / np.sum(a_pert))
                print(f"  Start {s+1}/{n_starts}, perturb={perturbation_scale:.3f}")
                a_new, score = soft_peak_gd(a_pert, time_per_start, seed=seed + 2000 + s)
                if score < best_score:
                    best_score = score
                    best_seq = a_new.copy()
                    print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: SA refinement ===
    print("\n" + "=" * 60)
    print("Phase 4: Simulated annealing refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        sa_time = min(200.0, remaining * 0.6)
        if sa_time > 20:
            print(f"  SA, n={len(best_seq)}, time={sa_time:.0f}s")
            a_new, score = sa_refine(best_seq, sa_time, seed=seed + 5000)
            print(f"  SA result: {score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Final GD polish ===
    print("\n" + "=" * 60)
    print("Phase 5: Final GD polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        final_time = min(100.0, remaining * 0.8)
        if final_time > 10:
            print(f"  Final polish, n={len(best_seq)}, time={final_time:.0f}s")
            a_new, score = soft_peak_gd(best_seq, final_time, seed=seed + 9999)
            print(f"  Final GD result: {score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
