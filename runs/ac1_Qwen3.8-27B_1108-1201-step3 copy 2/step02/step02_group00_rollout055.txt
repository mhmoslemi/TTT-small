1. Successive Linear Programming (SLP) - solve LP relaxation at each step
2. Projected subgradient descent with Nesterov acceleration
3. Coordinate descent polish
4. Multi-start exploration
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_at_peaks(a, conv, peaks, weights=None):
    """Compute weighted subgradient at given peaks."""
    n = len(a)
    if weights is None:
        weights = np.ones(len(peaks)) / len(peaks)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        wi = 2.0 * weights[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg

def slp_step(a, conv, active_peaks, tol=1e-10):
    """
    Successive Linear Programming step.
    Solves: min t s.t. b_k + 2*(S_k a)^T (a_new - a) <= t for active k
            a_new >= 0, sum(a_new) = 1
    """
    n = len(a)
    m = len(active_peaks)
    
    # Build constraint matrix
    # For each active peak k: b_k + 2*(S_k a)^T a_new - 2*(S_k a)^T a <= t
    # => -2*(S_k a)^T a_new + t >= b_k - 2*(S_k a)^T a
    # => 2*(S_k a)^T a_new - t <= 2*(S_k a)^T a - b_k
    # In standard form: A @ [a_new; t] <= b
    
    # A is m x (n+1), b is m
    A = np.zeros((m, n + 1))
    b_rhs = np.zeros(m)
    
    for i, k in enumerate(active_peaks):
        k = int(k)
        # S_k a: (S_k a)_j = a_{k-j} for valid j
        # 2*(S_k a)^T a_new = 2 * sum_j a_{k-j} * a_new_j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        A[i, idx] = 2.0 * a[k - idx]
        A[i, n] = -1.0
        b_rhs[i] = 2.0 * np.dot(a[k - idx], a[k - idx]) - conv[k]
    
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: a_new >= 0 (implicitly), t unbounded below (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    # Equality: sum(a_new) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Solve LP
    try:
        result = linprog(c, A_ub=A, b_ub=b_rhs, A_eq=A_eq, b_eq=b_eq, 
                        bounds=bounds, method='highs')
        if result.success:
            return result.x[:n]
    except Exception:
        pass
    
    return None

def sobeves_step(a, conv, peak_threshold_frac=0.95):
    """
    Smoothed subgradient step with adaptive peak selection.
    Uses softmax weighting over peaks above a threshold.
    """
    n = len(a)
    max_b = conv.max()
    
    # Select peaks above threshold
    threshold = max_b * peak_threshold_frac
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Temperature-based weighting
    temp = max_b * 0.01
    scaled = conv[active] / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = compute_subgradient_at_peaks(a, conv, active, w)
    return sg

def optimize_nesterov(a0, time_limit, lr_init=0.05, momentum_init=0.85, 
                       peak_frac=0.95, backtracking=True):
    """
    Nesterov-accelerated projected subgradient descent with backtracking line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead point
    conv = fast_convolve(y)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    momentum = momentum_init
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov step: y = (1+m)*a - m*prev_a... 
        # Actually, standard Nesterov: v_new = m * v - lr * grad(y)
        # a_new = y - v_new... 
        # Let me use the standard fast gradient method:
        # y_k = a_k + (t_{k-1}-1)/t_k * (a_k - a_{k-1})
        # g_k = grad(y_k)
        # a_{k+1} = project(y_k - g_k/t_k)
        
        # Simpler: heavy-ball with lookahead
        conv_y = fast_convolve(y)
        sg = sobeves_step(y, conv_y, peak_frac)
        
        v = momentum * v - lr * sg
        a_new = project_simplex(y + v)
        
        # Check if we improved
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = a_new
        a = a_new
        conv = conv_new
        
        # Adapt parameters
        if no_improve >= max_no_improve:
            # Restart with perturbation
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            y = a.copy()
            v = np.zeros(n)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)
            lr = max(lr_init * 0.3, lr * 0.5)
        
        # Schedule
        lr *= 0.9999
        momentum = min(0.98, momentum + 0.0002)
    
    return best_a, best_obj

def optimize_slp(a0, time_limit, peak_frac=0.9, line_search=True):
    """
    Successive Linear Programming optimizer.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() - t0 < time_limit:
        max_b = conv.max()
        threshold = max_b * peak_frac
        active = np.where(conv >= threshold)[0]
        
        if len(active) < 1:
            active = np.array([np.argmax(conv)])
        
        # Solve LP
        a_lp = slp_step(a, conv, active)
        
        if a_lp is not None:
            a_lp = np.maximum(a_lp, 0)
            a_lp_sum = a_lp.sum()
            if a_lp_sum > 1e-10:
                a_lp = a_lp / a_lp_sum
            
            # Line search
            if line_search:
                # Try several step sizes
                best_alpha = 0
                best_f = 2.0 * n * fast_convolve(a).max()
                for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
                    a_trial = project_simplex((1 - alpha) * a + alpha * a_lp)
                    f_trial = 2.0 * n * fast_convolve(a_trial).max()
                    if f_trial < best_f - 1e-12:
                        best_f = f_trial
                        best_alpha = alpha
                if best_alpha > 0:
                    a = project_simplex((1 - best_alpha) * a + best_alpha * a_lp)
            else:
                a = a_lp
            
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve >= max_no_improve:
            # Perturb and restart
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            no_improve = 0
            peak_frac = max(0.8, peak_frac - 0.02)
    
    return best_a, best_obj

def coordinate_descent_polish(a0, time_limit, max_sweeps=100):
    """
    Cyclic coordinate descent: for each pair (i,j), optimize the 1D problem
    of moving mass between coordinates i and j.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            # For coordinate i, the best update considering the simplex constraint:
            # We want to find the optimal value of a_i, with the constraint that
            # sum(a) = 1 and a >= 0.
            # 
            # As a_i changes, all other a_j (j≠i) must adjust to maintain sum=1.
            # The simplest approach: fix a_i = t, and scale the rest proportionally.
            # a_j' = a_j * (1-t) / (1-a_i) for j ≠ i
            #
            # This is a 1D optimization over t ∈ [0, 1].
            
            if a[i] >= 1.0 - 1e-10:
                continue
                
            rest_sum = 1.0 - a[i]
            if rest_sum < 1e-10:
                continue
                
            # Define objective as function of t = a_i'
            def obj_1d(t):
                if t < 0 or t > 0.9999:
                    return 1e10
                a_trial = a.copy()
                a_trial[i] = t
                scale = (1.0 - t) / rest_sum
                a_trial[np.arange(n) != i] *= scale
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum() if a_trial.sum() > 1e-15 else 1
                c = fast_convolve(a_trial)
                return c.max()
            
            # Use golden section or scipy minimize_scalar
            try:
                result = minimize_scalar(obj_1d, bounds=(0, min(0.99, 1 - 1e-6)), 
                                        method='bounded', options={'xatol': 1e-8})
                if result.fun < conv.max() - 1e-10:
                    t_opt = result.x
                    a[i] = t_opt
                    scale = (1.0 - t_opt) / rest_sum
                    mask = np.arange(n) != i
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj - 1e-10:
                        best_obj = cur_obj
                        best_a = a.copy()
                    improved = True
            except Exception:
                pass
        
        sweep += 1
        if not improved:
            break
    
    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc
    inits['ramp_dec'] = ramp_inc[::-1].copy()
    
    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    # Gaussians
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    # Bimodal (edges)
    for frac in [0.2, 0.3, 0.4]:
        bi = np.zeros(n)
        edge = max(1, int(n * frac))
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        bi /= bi.sum()
        inits[f'bimodal_{frac:.1f}'] = bi
    
    # Exponential decay
    bump = np.exp(-x / (0.2 * n))
    bump /= bump.sum()
    inits['bump_left'] = bump
    inits['bump_right'] = bump[::-1].copy()
    
    # Cosine
    cos_pick = np.cos(np.pi * x / (n - 1)) + 1.0
    cos_pick /= cos_pick.sum()
    inits['cosine'] = cos_pick
    
    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        # Interpolation
        pp = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent_interp'] = pp / ps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Successive Linear Programming (SLP) with line search for direction finding
2. Nesterov-accelerated subgradient descent with backtracking
3. Cyclic coordinate descent for local polish

Key improvements over prior: LP-based direction (exploits all active constraints),

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_subgradient(a, conv, peaks, weights):
    """Compute weighted subgradient at given peaks with given weights."""
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        wi = 2.0 * weights[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def softmax_weighted_sg(a, conv, peak_frac=0.95, temp_frac=0.01):
    """Compute softmax-weighted subgradient over active peaks."""
    n = len(a)
    max_b = conv.max()
    if max_b < 1e-15:
        return np.zeros(n)
    threshold = max_b * peak_frac
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    temp = max_b * temp_frac
    scaled = conv[active] / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(active))
        w[np.argmax(conv[active])] = 1.0
    else:
        w /= w_sum
    return compute_weighted_subgradient(a, conv, active, w)


def slp_direction(a, conv, peak_frac=0.90):
    """
    Solve the SLP subproblem: find a_new on simplex that minimizes
    max over active peaks of the linearized b_k at a.
    Returns the direction (a_new - a) or None.
    """
    n = len(a)
    max_b = conv.max()
    threshold = max_b * peak_frac
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    m = len(active)
    
    # Build LP: min t s.t. 2*(S_k a_new)^T a_new... 
    # Actually the linearization is: b_k(a) + 2*(S_k a)^T (x - a) <= t
    # => 2*(S_k a)^T x - t <= 2*(S_k a)^T a - b_k(a)
    # Note: 2*(S_k a)^T a = 2 * sum_j a_{k-j} * a_j = 2 * b_k(a)
    # So RHS = 2*b_k(a) - b_k(a) = b_k(a)
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for i, k in enumerate(active):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        A_ub[i, idx] = 2.0 * a[k - idx]
        A_ub[i, n] = -1.0
        b_ub[i] = 2.0 * np.dot(a[k - idx], a[k - idx]) - conv[k]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            return result.x[:n]
    except Exception:
        pass
    return None

def _cd_clean(a0, time_limit, max_sweeps=80):

    a = np.maximum(a0.astype(np.float64), 0.0)

            if rest_sum < 1e-12:

                                      method='bounded', options={'xatol': 1e-8})

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    for frac in [0.2, 0.3, 0.4]:
        bi = np.zeros(n)
        edge = max(1, int(n * frac))
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        bi /= bi.sum()
        inits[f'bimodal_{frac:.1f}'] = bi
    
    bump = np.exp(-x / (0.2 * n))
    bump /= bump.sum()
    inits['bump_left'] = bump
    inits['bump_right'] = bump[::-1].copy()
    
    cosv = np.cos(np.pi * x / max(1, n - 1)) + 1.01
    cosv /= cosv.sum()
    inits['cosine'] = cosv
    
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random3'] = rng.dirichlet(np.ones(n) * 2.0)
    
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent_interp'] = pp / ps
        if len(parent) >= n:
            pp2 = np.maximum(parent[:n], 0)
            ps2 = pp2.sum()
            if ps2 > 1e-10:
                inits['parent_clip'] = pp2 / ps2
    
    return inits

            try_update(a_cd, "P3 CD best")
        
        # Polish from perturbed best
        for pscale in [0.003, 0.006, 0.012]:
            if time.monotonic() > deadline - 30:
                break
            rem = deadline - time.monotonic() - 20
            t_pert = max(10.0, rem / 3.0)
            noise = rng.standard_normal(n_b) * pscale
            a_p = project_simplex(best_sequence + noise)
            # Quick Nesterov then CD
            t_n = t_pert * 0.5
            a_cn, _ = optimize_nesterov(a_p, time_limit=t_n,
                                        lr_init=0.01, momentum_init=0.9,
                                        peak_frac=0.93, temp_frac=0.002)
            sc = try_update(a_cn, f"P3 pert{pscale} Neg")
            t_c2 = min(rem * 0.3, 120.0)
            if t_c2 > 5 and a_cn is not None:
                a_cd2, _ = _cd_clean(a_cn, time_limit=t_c2, max_sweeps=40)
                try_update(a_cd2, f"P3 pert{pscale} CD")
            
            # Also SLP from perturbed
            rem = deadline - time.monotonic() - 20
            if rem > 15:
                a_ss, _ = optimize_slp(a_p, time_limit=min(rem * 0.4, 100), peak_frac=0.82)
                try_update(a_ss, f"P3 pert{pscale} SLP")
        
        # Final long CD on absolute best
        rem = deadline - time.monotonic() - 15
        if rem > 20 and best_sequence is not None:
            a_final_cd, _ = _cd_clean(best_sequence, time_limit=rem, max_sweeps=50)
            try_update(a_final_cd, "P3 Final CD")
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.8f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
