"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral initialization (Gerchberg-Saxton), smooth-to-sharp gradient descent
with adaptive tau, LP best-response polishing, and stochastic refinement.
Multi-start with diverse initializations at multiple lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize_to(a, target):
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a) * target / len(a)
    return a * (target / S)

def softmax_gradient(a, tau):
    """
    Compute gradient of soft-max approximation to max(conv(a,a)).
    soft_max = (1/tau) * log(sum(exp(tau * c_k)))
    grad_i = 2 * sum_k w_k * a[k-i] (for valid k-i)
    where w_k = exp(tau * c_k) / sum_j exp(tau * c_j)
    """
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the "valid" cross-correlation of w (length 2n-1) with a (length n)
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad, w

def adam_step(a, grad, m, v, t, lr, beta1, beta2, eps, target_sum):
    """Adam optimizer step with projection."""
    n = len(a)
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_to(a_new, target_sum)
    return a_new, m, v

def spectral_init(n, rng, target_sum=None):
    """
    Initialize using Gerchberg-Saxton / error reduction for flat spectrum.
    """
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    a = np.maximum(rng.random(n), 0.01)
    a = normalize_to(a, target_sum)
    
    for _ in range(50):
        # Forward: compute FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        # Target: uniform magnitude
        # sum(|A|^2) = N * sum(a^2) — we want this to be consistent with flat conv
        # For flat conv: each conv value = 2n/(2n-1), so sum(conv^2) = (2n-1)*(2n/(2n-1))^2 = 4n^2/(2n-1)
        # sum(|A|^4) = N * sum(conv^2) (by Parseval on the convolution)
        # This constrains the L4 norm but not L2.
        # For simplicity, just make |A| uniform:
        # sum(|A|^2) = N * sum(a_pad^2) = N * sum(a^2)
        # If |A_k| = C for all k, then N*C^2 = N*sum(a^2), so C = sqrt(sum(a^2))
        
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude: make it flat
        C = np.sqrt(np.sum(mag**2) / N)
        A_new = C * np.exp(1j * phase)
        
        # Enforce conjugate symmetry (for real a)
        for k in range(1, N // 2):
            A_new[N-k] = np.conj(A_new[k])
        A_new[N//2] = np.conj(A_new[N//2])  # Nyquist
        
        # Inverse FFT
        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, target_sum)
        
        if np.allclose(a_new, a, atol=1e-8):
            break
        a = a_new
    
    return a

def gradient_descent_smooth(a, deadline, target_sum):
    """
    Gradient descent with soft-max objective, increasing tau.
    Uses Adam optimizer.
    """
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    tau = 1.0
    lr = 0.5
    
    last_improve = time.monotonic()
    
    while time.monotonic() < deadline:
        t += 1
        # Increase tau gradually
        elapsed = time.monotonic() - (deadline - (deadline - last_improve))
        # Simpler: increase tau with iterations
        tau = min(50.0, 1.0 + 0.1 * (t // 100))
        lr = 0.5 * (0.9999 ** t)
        lr = max(lr, 0.01)
        
        grad, w = softmax_gradient(a, tau)
        # Project out mean component (to maintain sum constraint approximately)
        grad -= grad.mean()
        
        a, m, v = adam_step(a, grad, m, v, t, lr, 0.9, 0.999, 1e-8, target_sum)
        
        if t % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                best = a.copy()
                last_improve = time.monotonic()
            elif time.monotonic() - last_improve > 30:
                # Try to escape: reset momentum, restart tau
                m[:] = 0
                v[:] = 0
                tau = 1.0
                lr = 0.5
                last_improve = time.monotonic()
    
    return best, best_score

def lp_best_response(f, target_sum):
    """
    Given fixed f, find g >= 0, sum(g) = target_sum, minimizing max(conv(f, g)).
    This is an LP.
    """
    n = len(f)
    if n > 1024:
        return None
    
    # Variables: g[0], ..., g[n-1], M
    # Minimize M
    # s.t. sum_j f[j] * g[k-j] <= M for k = 0, ..., 2n-2
    #      sum(g) = target_sum
    #      g >= 0
    
    # Objective: [0, ..., 0, 1] (last variable is M)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_j f[k-j] * g[j] - M <= 0
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0; A[k, n] = -1
    A = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
        A[k, n] = -1.0
    
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds: g >= 0, M is free
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def lp_polish(a, deadline, target_sum):
    """
    Alternate LP best-response: fix a, find g; fix g, find a; repeat.
    """
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    
    current = a.copy()
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 10:
        # Solve LP for g given current a
        g, M = lp_best_response(current, target_sum)
        if g is None:
            break
        
        # Update: blend current and g
        improved = False
        for t_blend in [1.0, 0.7, 0.5, 0.3]:
            a_new = (1 - t_blend) * current + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, target_sum)
            s = compute_obj(a_new)
            if s < best_score - 1e-10:
                current = a_new
                best = a_new.copy()
                best_score = s
                improved = True
                break
            elif s < best_score:
                # Slight improvement
                current = a_new
                if s < best_score - 1e-12:
                    best = a_new.copy()
                    best_score = s
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            # Small perturbation to escape
            rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            pert = rng_local.normal(0, 0.01, n)
            current = np.maximum(current + pert, 0)
            current = normalize_to(current, target_sum)
        else:
            no_improve = 0
    
    return best, best_score

def stochastic_polish(a, deadline, target_sum, rng):
    """
    Stochastic coordinate descent: randomly perturb groups of coordinates.
    """
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    current = a.copy()
    
    group_sizes = [1, 2, 3, 5, 8]
    
    while time.monotonic() < deadline:
        # Choose random group
        gs = group_sizes[rng.integers(len(group_sizes))]
        indices = rng.integers(0, n, size=gs)
        
        orig_vals = current[indices].copy()
        
        # Try several perturbations
        best_local = best_score
        best_perturb = None
        
        for trial in range(5):
            perturbation = (rng.random(gs) - 0.5) * 0.2 * current[indices]
            new_vals = np.maximum(orig_vals + perturbation, 0)
            current[indices] = new_vals
            current = normalize_to(current, target_sum)
            s = compute_obj(current)
            if s < best_local:
                best_local = s
                best_perturb = new_vals.copy()
        
        if best_perturb is not None:
            current[indices] = best_perturb
            current = normalize_to(current, target_sum)
            if best_local < best_score:
                best_score = best_local
                best = current.copy()
        else:
            # Restore
            current[indices] = orig_vals
            current = normalize_to(current, target_sum)
    
    return best, best_score

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) spectral initialization via error reduction, (2) L-BFGS-B with
increasingly sharp soft-max objective, (3) LP best-response polishing,
(4) stochastic coordinate refinement. Multi-start at multiple lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_to(a, target):
    """Normalize sequence to have the given sum."""
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a) * target / len(a)
    return a * (target / S)


def soft_max_obj_and_grad(a, tau):
    """
    Compute soft-max objective and its gradient.
    Objective: (1/tau)*log(sum(exp(tau*c_k))) where c = conv(a,a)
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = float(c.max())
    c_shifted = c - c_max
    w = np.exp(tau * c_shifted)
    w_sum = float(w.sum())

    if w_sum < 1e-300:
        k_star = int(np.argmax(c))
        obj = c_max
        w = np.zeros_like(c)
        w[k_star] = 1.0
    else:
        obj = c_max + np.log(w_sum) / tau
        w = w / w_sum

    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_w = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_w[n - 1:n - 1 + n]

    return obj, grad


def lbfgs_optimize(a0, tau, maxiter=200, tol=1e-8):
    """Run L-BFGS-B on the soft-max objective with given tau."""
    n = len(a0)
    bounds = [(0.0, None)] * n

    def fun(x):
        val, _ = soft_max_obj_and_grad(x, tau)
        return val

    def jac(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g

    res = minimize(fun, a0, method='L-BFGS-B', jac=jac, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-7,
                            'maxls': 50})
    return np.maximum(res.x, 0.0)


def spectral_init(n, rng, target_sum=None):
    """
    Initialize sequence using Gerchberg-Saxton for flat spectrum.
    """
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)

    N = 1
    while N < 2 * n - 1:
        N *= 2

    a = np.maximum(rng.random(n), 0.05)
    a = normalize_to(a, target_sum)

    for _ in range(30):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        mag = np.abs(A)
        phase = np.angle(A)

        # Target: uniform magnitude
        C = np.sqrt(np.sum(mag ** 2) / N)
        A_new = C * np.exp(1j * phase)

        # Enforce conjugate symmetry
        for k in range(1, N // 2 + 1):
            A_new[N - k] = np.conj(A_new[k])
        if N % 2 == 0:
            A_new[N // 2] = float(A_new[N // 2].real)

        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, target_sum)

        if np.max(np.abs(a_new - a)) < 1e-10:
            break
        a = a_new

    return a


def lp_best_response(f, target_sum):
    """
    Given fixed f, solve LP: min M s.t. conv(f,g)[k]<=M, sum(g)=target, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(f)
    if n > 1024:
        return None, None

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Build constraint matrix
    A = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
        A[k, n] = -1.0

    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None


def lp_polish(a, deadline, target_sum):
    """Alternate LP best-response steps."""
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    current = a.copy()
    no_improve = 0

    while time.monotonic() < deadline and no_improve < 15:
        g, M = lp_best_response(current, target_sum)
        if g is None:
            break

        improved = False
        for t_blend in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_new = (1.0 - t_blend) * current + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, target_sum)
            s = compute_obj(a_new)
            if s < best_score - 1e-10:
                current = a_new
                best = a_new.copy()
                best_score = s
                improved = True
                break

        if not improved:
            no_improve += 1
            # Perturb to escape
            rng_local = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            pert = rng_local.normal(0, 0.05, n)
            current = np.maximum(current + pert, 0)
            current = normalize_to(current, target_sum)
        else:
            no_improve = 0

    return best, best_score


def stochastic_polish(a, deadline, target_sum, rng, max_steps=5000):
    """Random group perturbations with greedy acceptance."""
    n = len(a)
    best = a.copy()
    best_score = compute_obj(a)
    current = a.copy()
    steps = 0

    while time.monotonic() < deadline and steps < max_steps:
        steps += 1
        gs = int(rng.integers(1, 9))  # group size 1-8
        indices = rng.integers(0, n, size=gs)
        orig_vals = current[indices].copy()

        best_local = best_score
        best_pert = None

        for _ in range(4):
            scale = 0.1 + 0.3 * rng.random()
            pert = (rng.random(gs) - 0.5) * 2.0 * scale * np.maximum(orig_vals, 0.01)
            new_vals = np.maximum(orig_vals + pert, 0.0)
            current[indices] = new_vals
            current = normalize_to(current, target_sum)
            s = compute_obj(current)
            if s < best_local:
                best_local = s
                best_pert = new_vals.copy()

        if best_pert is not None:
            current[indices] = best_pert
            current = normalize_to(current, target_sum)
            if best_local < best_score - 1e-12:
                best_score = best_local
                best = current.copy()
        else:
            current[indices] = orig_vals
            current = normalize_to(current, target_sum)

    return best, best_score

    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    lengths = [256, 320, 384, 448, 512]

    # Build list of starts
    all_starts = []

    # From height_sequence_1
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size >= 16 and np.all(np.isfinite(hs)):
            for L in lengths:
                if L <= len(hs):
                    all_starts.append(hs[:L].copy())
    except NameError:
        pass

    # Spectral inits
    for n in lengths:
        for trial in range(2):
            all_starts.append(spectral_init(n, rng))

    # Random
    for n in lengths:
        all_starts.append(rng.random(n) * 0.5 + 0.25)

    # Structured
    for n in [256, 384, 512]:
        i = np.arange(n, dtype=np.float64)
        all_starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 7.0 / n))
        all_starts.append(np.abs(np.sin(2 * np.pi * i * 3.0 / n)) + 0.05)
        c = n / 2.0
        all_starts.append(np.exp(-0.5 * ((i - c) / (n * 0.2)) ** 2) + 0.05)

    # Sort by length (shorter first for quick feedback)
    all_starts.sort(key=lambda x: len(x))

    # Deduplicate
    seen = set()
    unique_starts = []
    for s in all_starts:
        key = (len(s), round(float(s[:8].sum() * 100), 0))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    all_starts = unique_starts

    print(f"Total starts: {len(all_starts)}, budget={budget_s:.0f}s")

    # First pass
    for idx, a0 in enumerate(all_starts):
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Time allocation: shorter sequences faster, give reasonable time
        n = len(a0)
        t_alloc = min(180.0, remaining * 0.2)
        if n <= 256:
            t_alloc = max(20.0, min(60.0, t_alloc))
        elif n <= 384:
            t_alloc = max(30.0, min(120.0, t_alloc))
        else:
            t_alloc = max(30.0, min(150.0, t_alloc))

        if time.monotonic() + t_alloc > deadline - 60:
            t_alloc = deadline - time.monotonic() - 60
            if t_alloc < 20:
                break

        seq, score = optimize_single(a0, time.monotonic() + t_alloc, verbose=True)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [start {idx}, n={len(seq)}] NEW BEST: {best_score:.6f}")

        # Check global deadline
        if time.monotonic() > deadline - 120:
            break

    # Second pass: focus on best length
    if best_seq is not None and time.monotonic() < deadline - 60:
        best_n = len(best_seq)
        print(f"\n=== Second pass: n={best_n}, best={best_score:.6f} ===")

        for trial in range(8):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic() - 30
            if remaining < 20:
                break

            if trial % 3 == 0:
                a0 = np.maximum(best_seq + rng.normal(0, 0.15, best_n), 0)
            elif trial % 3 == 1:
                a0 = np.maximum(best_seq * (0.5 + 0.5 * rng.random(best_n)), 0)
            else:
                # Reorder subsets
                a0 = best_seq.copy()
                k = best_n // 4
                perm = rng.permutation(best_n)
                a0 = a0[perm]

            a0 = normalize_to(a0, np.sqrt(2.0 * best_n))
            t_alloc = min(150.0, remaining * 0.4)
            if t_alloc < 15:
                break

            seq, score = optimize_single(a0, time.monotonic() + t_alloc, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [pass2-{trial}] NEW BEST: {best_score:.6f}")

    # Try larger n if time remains
    if time.monotonic() < deadline - 120:
        for n_try in [640, 768]:
            if time.monotonic() > deadline - 60:
                break
            rng2 = np.random.default_rng(seed + n_try)
            a_init = spectral_init(n_try, rng2)
            seq, score = optimize_single(a_init, deadline - 30, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [n={n_try}] NEW BEST: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, internal_score={final_score:.6f}")
        try:
            ev = evaluate_sequence([float(x) for x in best_seq])
            print(f"evaluate_sequence: {ev:.6f}")
        except Exception as e:
            print(f"eval error: {e}")
        return [float(x) for x in best_seq]

    print("Falling back to uniform n=384")
    return [1.0] * 384
