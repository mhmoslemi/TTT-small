import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def normalize_seq(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a * S / s

def freq_domain_iterate(a, num_iters=100, flatten_power=0.3, verbose=False):
    """
    Iterative method to flatten the power spectrum.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    
    for it in range(num_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: interpolate between current mag and uniform mag
        uniform_mag = np.full_like(mag, S / np.sqrt(n))
        # Use a blending: new_mag = mag^(1-alpha) * uniform_mag^alpha
        # This is equivalent to: new_mag = mag * (uniform_mag/mag)^alpha
        # Or: log(new_mag) = (1-alpha)*log(mag) + alpha*log(uniform_mag)
        # Simpler: new_mag = mag * (uniform_mag / (mag + eps))^flatten_power
        eps = 1e-10
        ratio = uniform_mag / (mag + eps)
        new_mag = mag * ratio ** flatten_power
        # Clip to avoid extreme values
        new_mag = np.clip(new_mag, 0, 10 * uniform_mag[0])
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, n)
        a_new = np.real(a_new)
        a_new = np.maximum(a_new, 0.0)
        
        if np.sum(a_new) < 1e-10:

        a = a_new
        
        if verbose and it % 20 == 0:

            print(f"      Freq iter {it}: score={score:.6f}")
    
    return a

def subgradient_step(a, lr):
    """One subgradient step to reduce the peak autocorrelation."""
    n = len(a)
    conv = fft_convolve(a, a)
    peak_idx = int(np.argmax(conv))
    
    # Subgradient of conv(a,a)[peak_idx] w.r.t. a[j]
    # conv(a,a)[k] = sum_i a[i]*a[k-i] (for valid i, k-i)
    # d/d a[j] of conv(a,a)[k] = a[k-j] + a[j-k] (if valid)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = peak_idx - j
        if 0 <= i1 < n:
            val += a[i1]
        i2 = j - peak_idx
        if 0 <= i2 < n:
            val += a[i2]
        grad[j] = val
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    S = np.sum(a)
    a_new = a_new * S / (np.sum(a_new) + 1e-10)
    
    return a_new, peak_idx

def subgradient_optimize(a, time_limit, verbose=False):

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)

        return np.inf

def normalize(a, target_sum=None):
    """Normalize to target sum (default: sqrt(2n))."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))

    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
