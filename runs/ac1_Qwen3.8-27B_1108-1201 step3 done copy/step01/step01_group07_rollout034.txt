def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax parameterization
with L-BFGS optimization and temperature annealing. Combines gradient-based
optimization in log-space with multiplicative update polishing.
Tries multiple initializations and sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def smooth_objective_and_grad(v, t):
    """
    Compute the smoothed objective F(v) = (2n/t)*log(sum exp(t*c_k)) and
    its gradient with respect to v, where a = softmax(v), c = conv(a,a).
    """
    n = len(v)

    # Softmax parameterization
    v_shifted = v - v.max()
    exp_v = np.exp(v_shifted)
    a = exp_v / exp_v.sum()

    # Compute convolution c = conv(a, a), length 2n-1
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:conv_size]

    # Softmax weights over c
    tc = t * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    w = exp_tc / exp_tc.sum()

    # Smoothed objective
    log_sum_exp = tc_max + np.log(exp_tc.sum())
    F = (2.0 * n / t) * log_sum_exp

    # Gradient computation
    # g[i] = 4n * sum_k w[k] * a[k-i]
    # Using FFT: g[i] = 4n * conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    grad_fft_size = 1 << (conv_size + n - 2).bit_length()
    W = np.fft.rfft(w, grad_fft_size)
    A_rev = np.fft.rfft(a_rev, grad_fft_size)
    conv_w_arev = np.fft.irfft(W * A_rev, grad_fft_size)
    g = 4.0 * n * conv_w_arev[n - 1:n - 1 + n]

    # grad_v = a * (g - dot(g, a))
    dot_ga = np.dot(g, a)
    grad_v = a * (g - dot_ga)

    return F, grad_v


def obj_func(v, t):
    """Wrapper for scipy minimize - objective only."""
    F, _ = smooth_objective_and_grad(v, t)
    return F


def grad_func(v, t):
    """Wrapper for scipy minimize - gradient only."""
    _, grad = smooth_objective_and_grad(v, t)
    return grad


def run_lbfgs(v0, n, temps, max_iter_per_temp, time_limit):
    """
    Run temperature annealing with L-BFGS.
    Returns (best_a, best_score).
    """
    v = v0.copy()
    best_score = np.inf
    best_a = None

    t_start = time.monotonic()

    for t in temps:
        if (time.monotonic() - t_start) > time_limit:
            break

        try:
            result = minimize(
                obj_func, v,
                jac=grad_func,
                args=(t,),
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_temp, 'gtol': 1e-6, 'ftol': 1e-10}
            )
            v = result.x
        except Exception:
            pass

        # Evaluate actual (unsmoothed) objective
        v_shifted = v - v.max()
        a = np.exp(v_shifted)
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum
        c = fast_convolve(a)
        score = 2.0 * n * c.max()

        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

    temps_full = [200, 100, 60, 40, 30, 20, 15, 12, 10, 8, 6, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0, 0.8, 0.6]
    temps_short = [100, 60, 40, 30, 20, 15, 10, 8, 6, 5, 4, 3, 2, 1.5, 1.0, 0.7]

    # Build list of (n, init_name, v_init) configurations
    configs = []

    lengths = [128, 256]
    if (deadline - time.monotonic()) > 900:
        lengths.append(512)

    for n in lengths:
        x = np.arange(n, dtype=np.float64)

        # Uniform
        configs.append((n, 'uniform', np.zeros(n)))

        # Random smooth
        v_rand = np.cumsum(rng.normal(0, 0.5, n))
        v_rand -= v_rand.mean()
        configs.append((n, 'rand_smooth', v_rand))

        # Ramp
        configs.append((n, 'ramp', np.linspace(-3, 3, n)))

        # Reverse ramp
        configs.append((n, 'ramp_rev', np.linspace(3, -3, n)))

        # Bimodal (edges high)
        bimodal = 4.0 * ((x / n - 0.5) ** 2) * 8
        configs.append((n, 'bimodal', bimodal))

        # Gaussian centered
        configs.append((n, 'gauss_c', -((x - n / 2) / (0.15 * n)) ** 2 * 2))

        # Gaussian off-center
        configs.append((n, 'gauss_off', -((x - 0.3 * n) / (0.12 * n)) ** 2 * 3))

        # Two bumps
        two_bump = -((x - 0.2 * n) / (0.08 * n)) ** 2 * 3 - ((x - 0.8 * n) / (0.08 * n)) ** 2 * 3
        configs.append((n, 'two_bump', two_bump))

        # Cosine window
        cos_win = -0.5 * (1 + np.cos(2 * np.pi * x / n)) * 4
        configs.append((n, 'cosine', cos_win))

        # From parent
        if parent is not None:
            if parent.size == n:
                v_p = np.log(parent + 1e-10)
                configs.append((n, 'parent', v_p))
            elif parent.size > n:
                v_p = np.log(parent[:n] + 1e-10)
                configs.append((n, 'parent_trunc', v_p))
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, parent.size), parent)
                v_p = np.log(p_interp + 1e-10)
                configs.append((n, 'parent_interp', v_p))

    # Sort: parent configs first
    configs.sort(key=lambda x: (0 if 'parent' in x[1] else 1, x[0]))

    print(f"Total configs: {len(configs)}")

    # Phase 1: Scan all configs
    phase1_deadline = deadline - 120.0
    time_per_config = max(15.0, (phase1_deadline - time.monotonic()) / max(1, len(configs)))

    scan_results = []
    for i, (n, name, v0) in enumerate(configs):
        if time.monotonic() > phase1_deadline - 30:
            break

        t_alloc = min(time_per_config, phase1_deadline - time.monotonic() - 10)
        if t_alloc < 8:
            break

        if t_alloc < 40:
            temps_use = temps_short[:8]
            max_iter = 50
        elif t_alloc < 80:
            temps_use = temps_short
            max_iter = 100
        else:
            temps_use = temps_full
            max_iter = 150

        best_a, score = run_lbfgs(v0, n, temps_use, max_iter, t_alloc)

        if best_a is not None and np.isfinite(score):
            real_score = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(real_score):
                scan_results.append((real_score, n, name, best_a))
                if real_score < best_score:
                    best_score = real_score
                    best_sequence = best_a.copy()
                    print(f"  Scan {i}: n={n}, {name}, score={real_score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}. Top 5: {[f'{s:.5f}' for s, *_ in scan_results[:5]]}")

    # Phase 2: Deep optimization on top candidates
    if time.monotonic() < deadline - 60:
        top_candidates = []
        seen = set()
        for score, n, name, a in scan_results:
            key = (n, name)
            if key not in seen:
                top_candidates.append((n, name, a))
                seen.add(key)
            if len(top_candidates) >= 8:
                break

        # Always include current best
        if best_sequence is not None:
            top_candidates.append((len(best_sequence), 'best', best_sequence.copy()))

        remaining = deadline - time.monotonic() - 40
        time_per_deep = max(20.0, remaining / max(1, len(top_candidates)))
        deep_temps = [50, 30, 20, 15, 12, 10, 8, 6, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5, 0.4]

        for i, (n, name, a0) in enumerate(top_candidates):
            if time.monotonic() > deadline - 40:
                break

            t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
            if t_alloc < 15:
                break

            v0 = np.log(np.maximum(a0, 1e-15))

            # Run L-BFGS with deep temps
            best_a, score = run_lbfgs(v0, n, deep_temps, 200, t_alloc * 0.6)

            if best_a is not None and np.isfinite(score):
                real_score = float(evaluate_sequence(best_a.tolist()))
                if np.isfinite(real_score) and real_score < best_score:
                    best_score = real_score
                    best_sequence = best_a.copy()
                    print(f"  Deep {i}: n={n}, {name}, score={real_score:.6f}")

                # Polish with multiplicative updates
                remaining_polish = deadline - time.monotonic() - 30
                if remaining_polish > 10:
                    polished = refine_multiplicative(best_a, remaining_polish * 0.5,
                                                     t_start=10.0, eta_start=0.3)
                    pol_score = float(evaluate_sequence(polished.tolist()))
                    if np.isfinite(pol_score) and pol_score < best_score:
                        best_score = pol_score
                        best_sequence = polished.copy()
                        print(f"  Deep {i} polished: score={pol_score:.6f}")

    # Phase 3: Final refinement on the best solution
    if best_sequence is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        n = len(best_sequence)

        # Multiplicative polish
        t_p1 = remaining * 0.4
        polished = refine_multiplicative(best_sequence, t_p1, t_start=8.0, eta_start=0.25)
        pol_score = float(evaluate_sequence(polished.tolist()))
        if np.isfinite(pol_score) and pol_score < best_score:
            best_score = pol_score
            best_sequence = polished.copy()
            print(f"  Final multiplicative: score={pol_score:.6f}")

        # Coordinate descent polish
        if time.monotonic() < deadline - 10:
            t_p2 = min(60.0, (deadline - time.monotonic() - 10) * 0.3)
            coord_result = refine_coordinate(best_sequence, t_p2)
            coord_score = float(evaluate_sequence(coord_result.tolist()))
            if np.isfinite(coord_score) and coord_score < best_score:
                best_score = coord_score
                best_sequence = coord_result.copy()
                print(f"  Final coordinate: score={coord_score:.6f}")

        # One more L-BFGS polish with very low temps
        if time.monotonic() < deadline - 10:
            t_p3 = min(60.0, (deadline - time.monotonic() - 10) * 0.5)
            v0 = np.log(np.maximum(best_sequence, 1e-15))
            polish_temps = [10, 6, 4, 3, 2, 1.5, 1, 0.8, 0.6, 0.5, 0.4, 0.3]
            best_a, score = run_lbfgs(v0, n, polish_temps, 200, t_p3)
            if best_a is not None and np.isfinite(score):
                real_score = float(evaluate_sequence(best_a.tolist()))
                if np.isfinite(real_score) and real_score < best_score:
                    best_score = real_score
                    best_sequence = best_a.copy()
                    print(f"  Final L-BFGS: score={real_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses softmax parameterization with L-BFGS and
    temperature annealing to minimize 2n*max(conv(a,a))/(sum(a))^2.
    Multiple phases: scan, deep optimize, polish.
    """
    result = main_optimize(seed=seed, budget_s=budget_s)
    return [float(max(0.0, x)) for x in result]
