def fast_convolve(f):
    """Fast convolution using FFT."""
    n = len(f)

    # Pad to next power of 2 for efficiency
    fft_size = 1 << (size - 1).bit_length()

def gradient_soft_max(f, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(f*f))) w.r.t. f."""
    n = len(f)
    conv_size = 2 * n - 1
    fft_size_conv = 1 << (conv_size - 1).bit_length()
    
    # Step 1: Compute convolution f*f
    F = fft(f, fft_size_conv)
    conv = np.real(ifft(F * F, fft_size_conv))[:conv_size]
    
    # Step 2: Compute softmax
    v = softmax(conv, tau)
    
    # Step 3: Compute cross-correlation f ⊛ v
    fft_size_cc = 1 << (3 * n - 2 - 1).bit_length()
    F2 = fft(f, fft_size_cc)
    V2 = fft(v, fft_size_cc)
    grad = np.real(ifft(F2 * np.conj(V2), fft_size_cc))[:n]

Now for the simplex projection. The problem is: given z, find the closest point to z on the simplex {x : x >= 0, sum(x) = 1}.

The standard algorithm:
1. Sort z in decreasing order: z_(1) >= z_(2) >= ... >= z_(n)
2. Find the largest rho such that z_(rho) - (sum_{i=1}^{rho} z_(i)) / rho > 0
3. theta = (sum_{i=1}^{rho} z_(i) - 1) / rho
4. x_i = max(z_i - theta, 0)

Let me code this:

```python
def project_simplex(z):
    """Project z onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(z)
    if z.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(z)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(z - theta, 0)
```

Multi-strategy optimization for minimizing the autocorrelation peak of a
non-negative step function. Score = 2*n*max(conv(f,f))/sum(f)^2.

Main approach: Mirror descent (multiplicative weights) with soft-max
approximation of the max, using FFT for fast convolution.
Combined with LP-based polishing and random restarts.

import numpy as np

from scipy.optimize import linprog

def fast_convolve(f):
    """Fast linear convolution f*f using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = fft(f, fft_size)
    return np.real(ifft(F * F, fft_size))[:size]

def compute_gradient(f, tau):
    """Compute gradient of soft-max objective and the convolution."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Soft-max: v_k = exp(tau*conv_k) / sum(exp(tau*conv_k))
    tau_conv = tau * conv
    tau_conv -= tau_conv.max()  # for numerical stability
    exp_t = np.exp(tau_conv)
    v = exp_t / exp_t.sum()
    
    # Gradient: (A^T v)_j = sum_k f[k-j] * v_k = cross-correlation
    # We need ifft(fft(f) * conj(fft(v)))
    cc_size = 2 * n + (2 * n - 1) - 1  # = 4n - 2
    fft_size_cc = 1 << (cc_size - 1).bit_length()
    F = fft(f, fft_size_cc)
    V = fft(v, fft_size_cc)
    grad = np.real(ifft(F * np.conj(V), fft_size_cc))[:n]
    
    return grad, conv

def project_simplex(z):
    """Project z onto the probability simplex."""
    n = len(z)
    if z.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(z)[::-1]
    cv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(z - theta, 0)

def mirror_step(f, grad, eta):
    """Multiplicative weights update: f_j <- f_j * exp(-eta * grad_j), renormalized."""
    log_f = np.log(np.maximum(f, 1e-300))
    new_log = log_f - eta * grad
    new_log -= new_log.max()
    f_new = np.exp(new_log)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

def score_fast(f):
    """Fast computation of the score."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(f)
    return float(2.0 * n * conv.max() / (s * s))

def make_initial(n, rng, method='uniform'):
    """Create initial sequence."""
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1 + np.cos(x))
    elif method == 'sin_sq':
        x = np.linspace(0, np.pi, n)
        f = np.sin(x) ** 2
    elif method == 'random':
        f = rng.random(n) + 0.01
    elif method == 'two_peaks':
        x = np.arange(n, dtype=float)
        s = max(1, n // 8)
        f = np.exp(-0.5 * ((x - n//3) / s)**2) + np.exp(-0.5 * ((x - 2*n//3) / s)**2)
    elif method == 'noise_cosine':

    elif method == 'linear':

    else:

def run_mirror_descent(f_init, tau_start=1.0, tau_end=50.0, n_iter=5000, eta_scale=0.5, deadline=None):
    """Run mirror descent with annealed temperature."""
    f = f_init.copy()
    best_f = f.copy()
    best_score = score_fast(f)
    n = len(f)
    
    for i in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        progress = i / max(1, n_iter - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        eta = eta_scale / tau
        
        grad, conv = compute_gradient(f, tau)
        f_new = mirror_step(f, grad, eta)
        
        sc = score_fast(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score

def run_lp_polish(f, budget_s=30.0):
    """Polish using LP-based approach (similar to original but with early stopping)."""
    n = len(f)
    f = f / f.sum()
    best = f.copy()
    best_score = score_fast(f)
    
    if n > 4096:
        return best, best_score  # LP too slow for large n
    
    t = 0.1
    deadline = time.monotonic() + budget_s
    
    while time.monotonic() < deadline:
        conv = fast_convolve(f)
        M = float(conv.max())
        
        # Build constraint matrix (sparse)
        i_idx = np.tile(np.arange(n), n)
        j_idx = np.repeat(np.arange(n), n)
        row_idx = j_idx + i_idx
        data = f[i_idx]
        A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
        
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n

                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    f_new = (1 - t) * f + t * g
                    f_new = np.maximum(f_new, 0)
                    f_new = f_new / f_new.sum()
                    sc = score_fast(f_new)

                        best_score = sc
                        best = f_new.copy()
                    f = f_new

            break

            break

- Update: p_k ∝ p_k * exp(-eta * (Af)_k)

"""
Multi-strategy optimization for minimizing the autocorrelation peak of a
non-negative step function. Score = 2*n*max(conv(f,f))/sum(f)^2.

Uses FFT-based soft-max mirror descent (multiplicative weights) with annealed
temperature, combined with LP polishing and random perturbation.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import scipy.sparse as sp
import time


def fast_convolve(f):
    """Fast linear convolution f*f using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = fft(f, fft_size)
    return np.real(ifft(F * F, fft_size))[:size]


def cross_corr_fft(a, b, out_len):
    """Cross-correlation (a circ b)[j] = sum_i a[i]*b[i+j] for j=0..out_len-1."""
    n_a = len(a)
    n_b = len(b)
    L = n_a + n_b - 1
    fft_size = 1 << (L - 1).bit_length()
    A = fft(a, fft_size)
    B = fft(b, fft_size)
    result = np.real(ifft(A * np.conj(B), fft_size))
    return result[:out_len]


def compute_soft_gradient(f, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(f*f))) and the convolution."""
    n = len(f)
    conv = fast_convolve(f)
    conv_size = 2 * n - 1
    # Stable softmax
    tconv = tau * conv
    tconv -= tconv.max()
    exp_t = np.exp(tconv)
    v = exp_t / exp_t.sum()
    # Gradient = A^T @ v = cross-correlation of f and v
    grad = cross_corr_fft(f, v, n)
    return grad, conv, v


def project_simplex(z):
    """Project z onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(z)
    if z.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(z)[::-1]
    cv = np.cumsum(u)
    rho_mask = u * np.arange(1, n + 1) > (cv - 1)
    if not rho_mask.any():
        return np.ones(n) / n
    rho = np.max(np.where(rho_mask)[0])
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(z - theta, 0)


def score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(f)
    return float(2.0 * n * conv.max() / (s * s))


def make_initial(n, rng, method='uniform'):
    """Create initial non-negative sum-normalized sequence."""
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1.0 + np.cos(x))
    elif method == 'sin_sq':
        x = np.linspace(0, np.pi, n)
        f = np.sin(x) ** 2
    elif method == 'random':
        f = rng.random(n) + 0.01
    elif method == 'two_peaks':
        x = np.arange(n, dtype=float)
        s = max(1, n // 8)
        f = np.exp(-0.5 * ((x - n // 3) / s) ** 2) + np.exp(-0.5 * ((x - 2 * n // 3) / s) ** 2)
    elif method == 'noise_cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1.0 + np.cos(x)) + 0.3 * rng.random(n)
    elif method == 'linear':
        f = np.linspace(0.1, 1.0, n)
    elif method == 'bimodal_gauss':
        x = np.arange(n, dtype=float)
        f = np.exp(-((x - n * 0.3) / (n * 0.1)) ** 2) + np.exp(-((x - n * 0.7) / (n * 0.1)) ** 2)
    elif method == 'ramp':
        f = np.abs(np.linspace(-1, 1, n)) + 0.1
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.01)
    return f / f.sum()


def run_mirror_descent(f_init, tau_start, tau_end, n_iter, eta_scale, deadline):
    """Mirror descent with annealed soft-max temperature."""
    f = f_init.copy()
    n = len(f)
    best_f = f.copy()
    best_score = score_fast(f)
    stall = 0

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        progress = i / max(1, n_iter - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        eta = eta_scale / tau

        grad, conv, v = compute_soft_gradient(f, tau)
        # Mirror descent step (multiplicative weights)
        log_f = np.log(np.maximum(f, 1e-300))
        new_log = log_f - eta * grad
        new_log -= new_log.max()
        f_new = np.exp(new_log)
        s = f_new.sum()
        if s < 1e-300:
            f_new = f.copy()
        else:
            f_new = f_new / s

        sc = score_fast(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        f = f_new

        # Recovery: if too concentrated, blend with uniform
        if stall > 200:
            f = 0.5 * f + 0.5 * (1.0 / n)
            stall = 0

    return best_f, best_score


def run_lp_polish(f, budget_s):
    """LP-based polishing step."""
    n = len(f)
    f = f / f.sum()
    best = f.copy()
    best_score = score_fast(f)
    deadline = time.monotonic() + budget_s

    if n > 4096:
        return best, best_score

    # Build sparse convolution matrix once
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx

    t = 0.15
    iteration = 0
    while time.monotonic() < deadline:
        conv = fast_convolve(f)
        M = float(conv.max())
        if M < 1e-15:
            break

        data = f[i_idx]
        A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'time_limit': 5.0, 'presolve': True})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs < 1e-10:
                break
            g = g / gs
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-10:
                break
            f_new = f_new / fs
            sc = score_fast(f_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = f_new.copy()
            f = f_new
        except Exception:
            break
        t *= 0.97
        if t < 0.005:
            break
        iteration += 1
        if iteration % 20 == 0:
            print("    LP iter %d, score=%.6f" % (iteration, best_score), flush=True)

    return best, best_score


def random_polish(f, budget_s, seed):
    """Polish with adaptive random block perturbations."""
    rng = np.random.default_rng(seed)
    n = len(f)
    best = f.copy()
    best_score = score_fast(f)
    deadline = time.monotonic() + budget_s
    temp = 0.4

    while time.monotonic() < deadline:
        w = int(rng.integers(1, max(2, min(n // 5, 200))))
        start = int(rng.integers(0, max(1, n - w + 1)))
        stop = min(n, start + w)
        f_new = best.copy()
        factor = float(np.exp(rng.normal(0.0, temp)))
        f_new[start:stop] *= factor
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = score_fast(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = f_new.copy()
        else:
            temp *= 0.999
            if temp < 0.05:
                temp = 0.3  # restart temperature

    return best, best_score


def run_native_grad_descent(f_init, n_iter, deadline):
    """Subgradient method using the active max indices directly."""
    f = f_init.copy()
    n = len(f)
    best_f = f.copy()
    best_score = score_fast(f)
    eta = 0.01
    patience = 0

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        conv = fast_convolve(f)
        max_val = conv.max()
        # Find all indices achieving (near) the max
        threshold = max_val - 1e-10 * max_val
        active = np.where(conv >= threshold)[0]
        # Subgradient: average of A^T e_k for active k
        # (A^T e_k)_j = f[k-j] for 0<=k-j<n
        # Average over active set
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                m = k - j
                if 0 <= m < n:
                    grad[j] += f[m]
        grad /= max(1, len(active))

        # Projected subgradient step
        f_new = project_simplex(f - eta * grad)
        sc = score_fast(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            patience = 0
        else:
            patience += 1
        f = f_new
        eta *= 0.999
        if patience > 500:
            eta = 0.01
            patience = 0
            f = 0.7 * f + 0.3 / n

    return best_f, best_score


def to_list(arr):
    """Convert numpy array to list of floats."""
    return [float(x) for x in arr]
