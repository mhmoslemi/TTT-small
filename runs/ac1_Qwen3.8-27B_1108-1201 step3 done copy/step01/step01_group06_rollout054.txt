from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def soft_max_gradient(a, T):
    n = len(a)
    c = fft_convolve_self(a)
    c_shifted = c - c.max()
    weights = np.exp(c_shifted / T)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros_like(c)
        weights[np.argmax(c)] = 1.0
    else:
        weights /= wsum
    
    L = 1 << (3 * n - 1).bit_length()
    A = fft(a, L)
    W = fft(weights, L)
    corr = np.real(ifft(np.conj(A) * W))
    grad = 2.0 * corr[:n]
    return grad, c.max()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    total_time = max_time
    
    while time.monotonic() - start < max_time:
        t += 1
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - start) / total_time)
        T = T_init * (T_min / T_init) ** progress
        
        grad, cur_max = soft_max_gradient(a, T)
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr = lr_init * (0.5 ** (float(t) / 2000.0))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Periodic perturbation
        if t % 5000 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.005 * S / n, n), S)
    
    return best_local, best_score

                best_local = a.copy()
                break
        a[idx] = old_val  # restore (we'll use best_local at the end)

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]

def soft_max_gradient_and_val(a, T):
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    c_shifted = c - c_max

"""
Minimize 2*n*max_self_conv / sum^2 over non-negative sequences.

Strategy:
  * Normalize to sum = sqrt(2n) so objective = max of self-convolution.
  * Use soft-max (log-sum-exp) approximation for smooth gradients.
  * Adam optimizer with simplex projection and adaptive temperature annealing.
  * Start from the provided best-known sequence.
  * Periodic perturbations to escape local minima.
  * L-BFGS-B polish in reduced space for the final phase.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def soft_max_grad(a, T):
    """Compute gradient of T*logsumexp(c/T) where c = a*a (self-conv).
    Returns (gradient, soft_max_value, true_max)."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    c_shifted = c - c_max
    exp_c = np.exp(c_shifted / T)
    wsum = exp_c.sum()
    weights = exp_c / wsum
    
    # Gradient: dF/da[m] = 2 * sum_k weights[k] * a[k-m]
    # This is 2 * cross_correlation(a, weights)[m]
    L = 1 << (3 * n - 1).bit_length()
    A = fft(a, L)
    W = fft(weights, L)
    corr = np.real(ifft(np.conj(A) * W))
    grad = 2.0 * corr[:n]
    soft_max_val = c_max + T * np.log(wsum)
    return grad, soft_max_val, c_max


def run_adam_softmax(a0, n, S, max_seconds, rng, T_start, T_end, lr_init):
    """Adam with soft-max gradient and simplex projection."""
    a = a0.copy()
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_seconds:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_seconds)
        # Linear interpolation in log space for T
        log_T = np.log(T_start) + progress * (np.log(T_end) - np.log(T_start))
        T = np.exp(log_T)
        
        grad, soft_val, true_max = soft_max_grad(a, T)
        if true_max < best_score:
            best_score = true_max
            best_local = a.copy()
            if t % 5000 == 0:
                print("  t=%d T=%.4f maxc=%.6f" % (t, T, best_score))
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Learning rate with decay
        lr = lr_init * (0.7 ** (float(t) / 3000.0))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Periodic perturbation to escape local minima
        if t % 10000 == 0 and rng.random() < 0.5:
            noise_scale = 0.01 * S / n
            a = project_simplex(a + rng.normal(0.0, noise_scale, n), S)
    
    return best_local, best_score


def run_lbfgs_reduced(a0, n, S, max_seconds, T):
    """L-BFGS-B optimization in reduced space (last variable eliminated)."""
    
    def obj_and_grad(x):
        a_last = S - x.sum()
        if a_last < -1e-10:
            return 1e10, np.zeros_like(x)
        a = np.empty(n, dtype=np.float64)
        a[:n-1] = x
        a[n-1] = max(0.0, a_last)
        
        grad_full, soft_val, _ = soft_max_grad(a, T)
        if a_last < 0:
            # Infeasible, push back
            return 1e10 + a_last**2, grad_full[:n-1]
        
        # dF/dx[i] = dF/da[i] - dF/da[n-1] for i = 0..n-2
        # (because a[n-1] = S - sum(x), so da[n-1]/dx[i] = -1)
        gx = grad_full[:n-1] - grad_full[n-1]
        return soft_val, gx
    
    x0 = a0[:n-1].copy()
    
    result = minimize(
        obj_and_grad, x0,
        jac=True,  # function returns (f, g)
        method='L-BFGS-B',
        bounds=[(0.0, 1000.0)] * (n - 1),
        options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-10, 'gtol': 1e-8}
    )
    
    x_opt = result.x
    a_opt = np.empty(n, dtype=np.float64)
    a_opt[:n-1] = x_opt
    a_opt[n-1] = max(0.0, S - x_opt.sum())
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s > 0:
        a_opt = a_opt * (S / s)
    
    c = fft_convolve_self(a_opt)
    return a_opt, c.max()
