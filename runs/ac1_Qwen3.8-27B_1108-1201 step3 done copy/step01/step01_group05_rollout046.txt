def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def conv_full(p):
    """Full self-convolution via FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    """Score matching evaluate_sequence for non-negative p."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_full(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax):
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def logsumexp_surrogate(c, T, cmax):
    """Smoothed max via log-sum-exp."""
    if c.size == 0:
        return 0.0
    if cmax <= 0 or not np.isfinite(cmax):
        return 0.0
    T = max(T, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return float(cmax)
    return float(cmax + T * np.log(s))


def gradient_smoothed_max(p, T_factor):

    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)

        return np.empty(0), np.empty(0), 1.0, 0.0

    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    if cmax <= 0 or not np.isfinite(cmax):
        return np.zeros(n), c.astype(np.float64), 1.0, 0.0

    diff = (c - cmax) / T

    # This is (w * p) at appropriate offset, which is a cross-correlation.

    # Actually simpler: grad_j = sum_k w_k * p_{(k-j) mod L} ... no.
    
    # Let's think: c_k = sum_i p_i p_{k-i} (where indices are valid: 0<=i<n, 0<=k-i<n)
    # dc_k/dp_j: the terms where i=j gives p_{k-j} (if valid), and where k-i=j gives p_{i} = p_j * p_{k-j}... 
    # Actually: c_k = sum_i p_i * p_{k-i}. 
    # d c_k / d p_j = p_{k-j} (from the term i=j) + p_j * 0 ... 
    # No: d/dp_j [sum_i p_i * p_{k-i}] = p_{k-j} (if 0<=k-j<n, from i=j term) + sum over i where k-i=j, i.e., i=k-j: p_{k-j} (if 0<=k-j<n)
    # Wait, that's the same term counted twice if p_j appears in both positions.
    # c_k = sum_i p_i * p_{k-i}. The j-th factor: if j is the "first" index (i=j), contributes p_{k-j}. If j is the "second" index (k-i=j, so i=k-j), contributes p_{k-j}.
    # So dc_k/dp_j = p_{k-j} (if 0<=k-j<n) + p_{k-j} (if 0<=k-j<n)? No, that's double counting.
    # Actually: c_k = sum_i p_i * p_{k-i}. When we differentiate w.r.t. p_j:
    # - Term where i=j: gives p_{k-j} (if 0 <= k-j < n)
    # - Term where k-i=j, i.e., i=k-j: gives p_{k-j} (if 0 <= k-j < n)
    # But these are the SAME term when i=j AND k-i=j, i.e., k=2j. In that case we get 2*p_j.
    # Otherwise they're different terms and we get p_{k-j} + p_{k-j} = 2*p_{k-j}.
    # Wait no. Let me be more careful.
    # c_k = sum_{i: 0<=i<n, 0<=k-i<n} p_i * p_{k-i}
    # d c_k / d p_j = [k-j in range(n)] * p_{k-j} + [j in range(n) and k-j in range(n)] * p_{k-j}
    # Since j is always in [0,n-1], the second condition is just k-j in [0,n-1].
    # So d c_k / d p_j = 2 * p_{k-j} if 0 <= k-j < n, else 0.
    # (The factor of 2 comes from p_j appearing in both positions when k is fixed and we differentiate.)
    # Actually wait, that's not right either. Let me use a concrete example.
    # p = [a, b, c], c_1 = a*b + b*a = 2ab (for k=1)
    # dc_1/da = b, dc_1/db = a + a = 2a, dc_1/dc = 0
    # Hmm, so it's not simply 2*p_{k-j}.
    # c_k = sum_i p_i * p_{k-i}. For k=1, n=3: i can be 0 (gives p_0*p_1) or i=1 (gives p_1*p_0).

    # dc_1/dp_0 = 2*p_1. dc_1/dp_1 = 2*p_0. dc_1/dp_2 = 0.

    # For k=2: c_2 = p_0*p_2 + p_1*p_1 + p_2*p_0 = 2*p_0*p_2 + p_1^2.
    # dc_2/dp_0 = 2*p_2. dc_2/dp_1 = 2*p_1. dc_2/dp_2 = 2*p_0.

    # Actually for k=2j: the term i=j gives p_j*p_j = p_j^2, and differentiating gives 2*p_j.
    # For k != 2j but 0<=k-j<n: term i=j gives p_j*p_{k-j}, derivative w.r.t. p_j is p_{k-j}.
    #   Also term i=k-j gives p_{k-j}*p_j, derivative w.r.t. p_j is p_{k-j}.
    #   Total: 2*p_{k-j}.
    # For k=2j: only the term i=j contributes (since i=k-j=j is the same). p_j*p_j = p_j^2, derivative = 2*p_j.
    # So actually in ALL cases where 0<=k-j<n: dc_k/dp_j = 2*p_{k-j}.
    # (When k=2j, p_{k-j} = p_j, so 2*p_j, which matches.)
    
    # Therefore: grad_j = 2 * sum_{k: 0<=k-j<n} w_k * p_{k-j}
    # = 2 * sum_{m: 0<=m<n} w_{m+j} * p_m (substituting m = k-j)
    # = 2 * (w circulant applied to p) at position j

    # Alternative: grad = 2 * ifft(fft(p, L) * fft(w_reversed, L)) ...

    # Let me just use: grad_j = 2 * sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w_k * p_{k-j}

    # Then cross_corr = ifft(fft(w_pad, L) * conj(fft(p_pad, L)))
    # and grad = 2 * cross_corr[n-1 : 2n-2] ...

    # grad_j = 2 * sum_{m=0}^{n-1} w_{j+m} * p_m (where w_{j+m} is 0 if j+m >= 2n-1)

    # Using FFT: pad w to length L, pad p to length L.
    # cross_corr_j = sum_m p_m * w_{j+m}  (circular)
    # = ifft(conj(fft(p, L)) * fft(w, L))[j]

    # cross[j] = sum_m p_m * w_{j+m} (circular convolution equivalent)
    # We want grad_j = 2 * cross[j] for j = 0,...,n-1
    # But only for valid indices: j+m < 2n-1 for all m, which is true for j < n since m < n, so j+m < 2n.

    G = 2.0 * cross[:n]
    G = np.maximum(G, 0.0)  # gradient should be non-negative for non-negative p
    
    surr = logsumexp_surrogate(c, T, cmax)
    return G, c.astype(np.float64, copy=False), T, float(surr)


def make_initials(n, rng, parent=None):
    """Generate diverse initial sequences."""
    out = []
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        if parent.size == n and parent.sum() > 0:
            out.append(safe_normalize(parent))
    
    out.append(np.full(n, 1.0/n, dtype=np.float64))
    
    # Power laws
    x = np.linspace(0, 1, n)
    for a in (0.5, 1.0, 2.0, 4.0, 8.0):
        out.append(safe_normalize(x**a + 1e-10))
        out.append(safe_normalize((1-x)**a + 1e-10))
    
    # Cosine and sine shapes
    for f in (0.5, 1, 2, 3, 5, 8):
        for ph in (0, np.pi/2, np.pi):
            p = 0.5 + 0.4 * np.cos(2 * np.pi * f * x + ph)
            out.append(safe_normalize(np.maximum(p, 0.01)))
    
    # Gaussian bumps
    for mu in (0.2, 0.5, 0.8):
        for sig in (0.05, 0.1, 0.2, 0.4):
            p = np.exp(-((x - mu)**2) / (2 * sig**2))
            out.append(safe_normalize(p + 1e-10))
    
    # Two bumps
    out.append(safe_normalize(np.exp(-((x-0.2)**2)/0.01) + np.exp(-((x-0.8)**2)/0.01) + 1e-10))
    out.append(safe_normalize(np.exp(-((x-0.3)**2)/0.04) + np.exp(-((x-0.7)**2)/0.04) + 1e-10))
    
    # Random smooth
    for _ in range(8):
        f1 = int(rng.integers(1, max(2, n//50)))
        f2 = int(rng.integers(1, max(2, n//20)))
        p = 1.0 + 0.3*np.sin(2*np.pi*f1*x + rng.uniform(0, 2*np.pi)) + 0.2*np.sin(2*np.pi*f2*x + rng.uniform(0, 2*np.pi))
        out.append(safe_normalize(np.maximum(p, 0.01)))
    
    # Random perturbations of uniform
    for _ in range(4):
        noise = rng.normal(0, 0.3, n)
        # Smooth the noise
        kernel = np.ones(5)/5
        noise_smooth = np.convolve(noise, kernel, mode='same')
        out.append(safe_normalize(1.0 + 0.5 * noise_smooth))
    
    return out


def optimize_pgd(p0, rng, time_limit, label=""):
    """Projected gradient descent with adaptive step size."""
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(best_p)
    cur_score = best_score
    prev_p = p.copy()
    
    end = time.monotonic() + max(0.0, float(time_limit))
    T_factor = 0.5
    iter_no = 0
    stall = 0
    
    while time.monotonic() < end:
        iter_no += 1
        
        g, c, T, surr = gradient_smoothed_max(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        gmax = float(np.max(g)) if g.size else 0.0
        if gmax < 1e-30:
            break
        
        # Adaptive step sizes
        scale = 1.0 / (n * gmax + 1e-30)
        lrs = (0.5*scale, 0.2*scale, 0.08*scale, 0.03*scale, 0.01*scale)
        
        accepted = False
        for lr in lrs:
            v = p - lr * g
            pt = project_simplex(v)
            sc = internal_score(pt)
            
            if np.isfinite(sc) and sc < cur_score - 1e-15:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    stall = 0
                else:
                    stall += 1
                cur_score = sc
                prev_p = p
                p = pt
                accepted = True
                break
        
        if not accepted:
            stall += 1
            T_factor = max(0.05, T_factor * 0.9)
            
            # Random kick when stalled
            if stall > 100:
                sigma = 0.1 / n
                p = project_simplex(0.7*best_p + 0.3*p + rng.normal(0, sigma, n))
                cur_score = internal_score(p)
                prev_p = p
                stall = 0
                T_factor = 0.3
            elif stall > 50:
                p = project_simplex(0.9*best_p + 0.1*p + rng.normal(0, 0.02/n, n))
                cur_score = internal_score(p)
                prev_p = p
        
        if iter_no % 500 == 0:
            print(f"{label} iter={iter_no} best={best_score:.10f} cur={cur_score:.10f}", flush=True)
        
        if best_score <= 2.0 + 1e-12 and stall > 200:
            break
    
    return best_p, best_score


def lp_best_response(p, budget_s):
    """LP-based best response for short sequences."""
    if not _HAVE_LP:
        return None, np.inf
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > 500:
        return None, np.inf
    if budget_s < 2:
        return None, np.inf
    
    c = conv_full(p)
    M = float(np.max(c))
    if M <= 0 or not np.isfinite(M):
        return None, np.inf
    
    f = p / p.sum()
    
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel()
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n)
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=(0, None), method='highs', options={'disp': False, 'time_limit': max(1, budget_s-1)})
    except Exception:
        return None, np.inf
    
    if res is None or res.x is None:
        return None, np.inf
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    gs = float(g.sum())
    if gs <= 1e-12:
        return None, np.inf
    g = g / gs
    
    for t in (1.0, 0.7, 0.5, 0.3, 0.1):
        q = project_simplex((1-t)*p + t*g)
        sc = internal_score(q)
        if np.isfinite(sc) and sc < 2.0:
            return q, sc
    
    return None, np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the best non-negative sequence minimizing evaluate_sequence."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(60, float(budget_s) - 15)
    
    # Get parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 10 and parent_arr.sum() > 0:
            parent = parent_arr
    except Exception:
        parent = None
    
    # Determine sequence length
    if parent is not None and 16 <= parent.size <= 5000:
        n = int(parent.size)
    else:
        n = 512
    
    # Track best
    best_p = None
    best_score = np.inf
    best_eval = np.inf
    
    def try_candidate(p):
        nonlocal best_p, best_score, best_eval
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0:
            return
        p = safe_normalize(p)
        sc = internal_score(p)
        # Verify with actual evaluator
        try:
            ev = evaluate_sequence([float(x) for x in p])
            if np.isfinite(ev):
                sc = min(sc, ev)
        except Exception:
            pass
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            best_eval = float(sc)
            print(f"  NEW BEST: score={sc:.10f} n={p.size}", flush=True)
    
    print(f"Starting search: n={n}, budget={budget_s}s", flush=True)
    
    # Generate and evaluate initial candidates
    inits = make_initials(n, rng, parent)
    scored = []
    for p in inits:
        sc = internal_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    
    for sc, p in scored[:4]:
        try_candidate(p)
    
    print(f"Best initial: {best_score:.10f}", flush=True)
    
    # If we already hit 2.0 (theoretical min), we can stop early
    if best_score <= 2.0 + 1e-10:
        print("Achieved theoretical minimum of 2.0!", flush=True)
        return [float(x) for x in best_p]
    
    # Local optimization on top candidates
    n_opt = min(4, len(scored))
    remaining = deadline - time.monotonic()
    if remaining > 30:
        time_per = max(10, remaining / (n_opt + 1))
        for i in range(n_opt):
            if time.monotonic() > deadline - 30:
                break
            sc, p = scored[i]
            limit = min(time_per, deadline - time.monotonic() - 15)
            if limit < 5:
                break
            p_best, sc_best = optimize_pgd(p, rng, limit, label=f"opt{i}")
            try_candidate(p_best)
    
    # Try LP on short sequences
    if time.monotonic() < deadline - 30 and n <= 500:
        lp_budget = min(60, deadline - time.monotonic() - 20)
        p_lp, sc_lp = lp_best_response(best_p, lp_budget)
        if p_lp is not None:
            try_candidate(p_lp)
            if sc_lp < best_score:
                # Continue optimizing from LP result
                if time.monotonic() < deadline - 30:
                    limit = min(120, deadline - time.monotonic() - 10)
                    p_best, sc_best = optimize_pgd(p_lp, rng, limit, label="LP-refine")
                    try_candidate(p_best)
    
    # Final polish
    if best_p is not None and time.monotonic() < deadline - 10:
        limit = max(5, deadline - time.monotonic() - 5)
        p_best, sc_best = optimize_pgd(best_p, rng, limit, label="final")
        try_candidate(p_best)
    
    # Fallback
    if best_p is None:
        best_p = np.full(n, 1.0/n, dtype=np.float64)
        best_score = internal_score(best_p)
        print(f"Using uniform fallback: score={best_score:.10f}", flush=True)
    
    # Final verification
    result = [float(x) for x in best_p]
    try:
        final_score = evaluate_sequence(result)
        print(f"Final evaluation: {final_score:.10f} (internal: {best_score:.10f})", flush=True)
    except Exception as e:
        print(f"Final eval error: {e}", flush=True)
    
    print(f"Total time: {time.monotonic() - start_time:.1f}s", flush=True)
    return result
