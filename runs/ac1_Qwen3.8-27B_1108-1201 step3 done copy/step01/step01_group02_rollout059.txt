def lp_direction(a, M, timeout=30.0):
    """Find g >= 0 maximizing sum(g) subject to conv(a,g) <= M."""
    n = len(a)
    n_rows = 2 * n - 1
    # Build the convolution matrix A where conv(a, g)[k] = sum_j A[k,j] * g[j]
    # A[k,j] = a[j+k-(n-1)] if 0 <= j+k-(n-1) < n, else 0
    # Actually, using numpy's convention:
    # conv(a,g)[k] = sum_i a[i] * g[k-i] for valid i
    # = sum_j a[k-j] * g[j] for valid k-j
    # So A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # A is a banded Toeplitz matrix
    
    # For efficiency, use the banded structure
    # A has bandwidth n (each row has at most n nonzeros)
    
    # Build as CSR
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        # Nonzero columns j where 0 <= k-j <= n-1
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[k - j])
    
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
    
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

    if s < 0.01:

    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:

    return a / s

    best_score = score_fast(a, conv, s)

        for k in range(nr):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if js <= je:
                idx = np.arange(js, je + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx)
                data.extend(a[k - idx])
        return csr_matrix((data, (rows, cols)), shape=(nr, n))


def lp_step(a, M, timeout=20.0):
    """Solve LP to find best direction g."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x)
            return g
    except Exception:
        pass
    return None


def lp_refine(a, time_limit=300.0, max_iters=200, verbose=False):
    """LP-based iterative refinement."""
    t0 = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_score = score_fast(a)
    best_a = a.copy()
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        s = a.sum()
        cur_score = 2.0 * n * M / (s * s)
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.8f}")
            break
        
        if verbose and (it % 10 == 0 or it < 5):
            print(f"    LP iter {it}: score={cur_score:.8f}, M={M:.6f}")
        
        timeout = max(3.0, min(30.0, remaining * 0.2))
        g = lp_step(a, M, timeout=timeout)
        if g is None:
            break
        
        sg = g.sum()
        if sg <= 1.0 + 1e-8:
            if verbose:
                print(f"    LP no improvement (sum_g={sg:.6f})")
            break
        
        g = g / sg
        alpha = min(0.5, max(0.03, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        
        new_score = score_fast(a_new)
        if new_score <= cur_score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a = (1.0 - alpha2) * a + alpha2 * g
            a = normalize_nonneg(a)
    
    final_score = score_fast(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.8f}, time={time.monotonic()-t0:.1f}s")
    return best_a, best_score
