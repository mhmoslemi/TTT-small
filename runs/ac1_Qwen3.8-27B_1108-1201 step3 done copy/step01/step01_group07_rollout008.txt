def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def cd_one_step(a, j, conv):
    """
    Find optimal a[j] to minimize 2n*max(conv)/(sum)^2 given other coords fixed.
    Solves 1D subproblem: min_t max_k(B_k + alpha_k*t + beta_k*t^2)/(S+t)^2
    via golden section search.

    n = len(a)
    old = a[j]
    S = a.sum() - old

    if 0 <= k2j < 2 * n - 1:

        den = S + t

        return num / (den * den)

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    order = np.arange(n)

    no_improve = 0
    max_no_improve = 150
    perturbation_count = 0
    sweep_count = 0

    while time.monotonic() - start < time_limit:
        if sweep_count % 5 == 0:
            order = rng.permutation(n)

        for j in order:
            j = int(j)
            new_val = cd_one_step(a, j, conv)
            old_val = a[j]

            if abs(new_val - old_val) > 1e-15:
                delta = new_val - old_val
                idx = np.arange(2 * n - 1) - j
                valid = (idx >= 0) & (idx < n)
                conv[valid] += 2.0 * delta * a[idx[valid]]
                k2j = 2 * j
                if k2j < 2 * n - 1:
                    conv[k2j] += delta * delta
                a[j] = new_val

        # Recompute objective (use stored conv)
        s = a.sum()
        if s < 1e-15:
            break
        cur_obj = 2.0 * n * conv.max() / (s * s)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        sweep_count += 1

        if no_improve >= max_no_improve:
            if perturbation_count < 30:
                # Perturb: randomly zero out some coordinates and redistribute
                strength = 0.1 + 0.2 * perturbation_count / 30.0
                if perturbation_count % 3 == 0:
                    # Gaussian perturbation
                    a_p = a + rng.normal(0, strength * a.mean(), n)
                elif perturbation_count % 3 == 1:
                    # Randomly zero some entries
                    mask = rng.random(n) > 0.3
                    a_p = a * mask
                    a_p += rng.uniform(0, a.mean() * 0.5, n)
                else:
                    # Swap-based perturbation
                    a_p = a.copy()
                    for _ in range(n // 4):
                        i, jj = rng.integers(0, n, 2)
                        swap = a_p[i]
                        a_p[i] = a_p[jj]
                        a_p[jj] = swap
                a_p = np.maximum(a_p, 0)
                s_p = a_p.sum()
                if s_p > 1e-10:
                    a_p /= s_p
                    conv_new = fast_convolve(a_p)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj * 1.15:
                        a = a_p
                        conv = conv_new
                        cur_obj = obj_new
                        perturbation_count += 1
                        no_improve = 0
                        max_no_improve = max(50, max_no_improve - 20)
            else:
                break

    # Normalize the result
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a, best_obj


def make_initialization(n, name, rng, parent=None):
    """Create a diverse initialization."""
    x = np.arange(n, dtype=np.float64)
    if name == 'uniform':
        return np.ones(n) / n
    elif name.startswith('u_shape'):
        alpha = float(name.split('_')[1])
        centered = 2.0 * x / max(1, n - 1) - 1.0
        a = 1.0 + alpha * centered * centered
        a /= a.sum()
        return a
    elif name.startswith('u_shape_strong'):
        alpha = float(name.split('_')[1])
        centered = 2.0 * x / max(1, n - 1) - 1.0
        a = np.exp(alpha * centered * centered)
        a /= a.sum()
        return a
    elif name == 'bimodal':
        a = np.zeros(n)
        a[:max(1, n // 6)] = 1.0
        a[-max(1, n // 6):] = 1.0
        a += 0.1
        a /= a.sum()
        return a
    elif name == 'gauss':
        g = np.exp(-0.5 * ((x - n * 0.5) / (0.15 * n)) ** 2)
        g /= g.sum()
        return g
    elif name == 'random':
        a = rng.dirichlet(np.ones(n) * 0.3)
        return a
    elif name == 'parent':
        if parent is not None and len(parent) > 0:
            if len(parent) >= n:
                p = parent[:n].copy()
            else:
                p = np.interp(np.linspace(0, 1, n),
                              np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            p = np.maximum(p, 1e-10)
            p /= p.sum()
            return p
        return np.ones(n) / n
    elif name == 'random_sparse':
        # Sparse random: only ~30% of positions nonzero
        mask = rng.random(n) < 0.3
        a = np.zeros(n)
        a[mask] = rng.uniform(0.5, 1.5, mask.sum())
        a += 1e-6
        a /= a.sum()
        return a
    else:
        return np.ones(n) / n


def run_optimization(seed=42, budget_s=1800):
    """Main optimization: scan lengths/inits, then deep optimize best."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    # Phase 1: Quick scan
    print("Phase 1: Quick scan...")
    t_start = time.monotonic()

    lengths = [128, 256]
    if budget_s > 1000:
        lengths.append(512)

    init_names = ['u_shape_2', 'u_shape_5', 'u_shape_strong_2', 'u_shape_strong_4',
                  'bimodal', 'uniform', 'random', 'parent', 'random_sparse',
                  'u_shape_1', 'u_shape_strong_1']

    scan_configs = []
    for n in lengths:
        for name in init_names:
            scan_configs.append((n, name))

    # Time allocation for scan
    phase1_time = min(600.0, (deadline - time.monotonic()) * 0.4)
    time_per_scan = max(5.0, phase1_time / len(scan_configs))

    scan_results = []
    for n, name in scan_configs:
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 120)
        if t_alloc < 3:
            break

        a0 = make_initialization(n, name, rng, parent)
        a_opt, obj_opt = coordinate_descent(a0, time_limit=t_alloc,
                                            seed=seed + hash((n, name)) % 10000)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 done. Best: {best_score:.6f}. Time: {time.monotonic()-t_start:.1f}s")

    # Phase 2: Deep optimization on best configs
    print("Phase 2: Deep optimization...")
    t_start2 = time.monotonic()
    scan_results.sort(key=lambda x: x[0])

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt.copy()))
            seen.add(key)
        if len(top_configs) >= 8:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 60
    n_configs = len(top_configs)
    time_per_deep = max(15.0, remaining / n_configs)

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 40:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 40)
        if t_alloc < 5:
            break

        a_opt, obj_opt = coordinate_descent(a0, time_limit=t_alloc,
                                            seed=seed + 100 + i)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}. Time: {time.monotonic()-t_start2:.1f}s")

    # Phase 3: Final polish with longer runs on the best
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("Phase 3: Final polish...")
        remaining = deadline - time.monotonic() - 15
        n_polish = min(4, max(1, int(remaining // 30)))
        time_per_polish = (deadline - time.monotonic() - 15) / n_polish

        for i in range(n_polish):
            if time.monotonic() > deadline - 15:
                break
            t_alloc = min(time_per_polish, deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break

            # Slightly perturb the best for diversity
            if i == 0:
                a0 = best_sequence.copy()
            else:
                a0 = best_sequence + rng.normal(0, 0.01, len(best_sequence))
                a0 = np.maximum(a0, 0)
                a0 /= a0.sum()

            a_opt, obj_opt = coordinate_descent(a0, time_limit=t_alloc,
                                                seed=seed + 500 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {i}: n={len(best_sequence)}, score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}, length={len(best_sequence)}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses coordinate descent with exact 1D subproblem solving
    to minimize 2n*max(conv(a,a))/(sum(a))^2. Multiple restarts, perturbations,
    and diverse initializations are used to avoid local minima.
    """
    result = run_optimization(seed=seed, budget_s=budget_s)
    return [float(max(0.0, min(1000.0, x))) for x in result]
