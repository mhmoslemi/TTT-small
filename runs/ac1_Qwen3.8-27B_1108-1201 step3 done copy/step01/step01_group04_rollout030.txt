"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: FFT-accelerated Adam subgradient descent with top-k softmax smoothing,
combined with periodic LP-based active-set refinement. Explores multiple
sequence lengths and diverse starting points within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    fa = rfft(a, fft_size)
    fb = rfft(b, fft_size)
    return irfft(fa * fb, fft_size)[:n]


def _compute_conv(a):
    """Compute autocorrelation (linear convolution) of a with itself."""
    return _fft_convolve(a, a)


def _compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _compute_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _compute_grad(a, w):
    """Compute gradient given weights w on autocorrelation values.
    grad[k] = 2 * sum_i a[i] * w[i+k]
    Computed as: 2 * conv(w, a_rev)[n-1 : 2n-1] where a_rev = a[::-1]
    """
    n = len(a)
    a_rev = a[::-1]
    conv_w_arev = _fft_convolve(w, a_rev)
    return 2.0 * conv_w_arev[n - 1: 2 * n - 1]


def _topk_weights(c, k, beta):
    """Compute top-k softmax weights on the autocorrelation values."""
    m = len(c)
    k = min(k, m)
    # Get indices of top-k values
    if k == m:
        idx = np.arange(m)
    else:
        # Use partition for efficiency
        part = np.argpartition(c, -k)[-k:]
        idx = part
    vals = c[idx]
    if beta < 1e-6:
        w = np.zeros(m)
        w[idx] = 1.0 / k
    else:
        exp_vals = np.exp(beta * (vals - vals.max()))
        w = np.zeros(m)
        w[idx] = exp_vals / exp_vals.sum()
    return w


def _normalize(a, target):
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)


def _lp_solve(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    # A[m, j] = f[m-j] if 0 <= m-j < n
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        A[m, lo:hi+1] = f[hi:lo-1:-1] if hi >= lo else 0
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _adam_step(a, grad, m_v, v_v, t, lr, b1, b2, eps, target_sum):
    """One Adam step with projection onto non-negative, fixed-sum simplex."""
    m_v = b1 * m_v + (1 - b1) * grad
    v_v = b2 * v_v + (1 - b2) * grad ** 2
    m_hat = m_v / (1 - b1 ** t)
    v_hat = v_v / (1 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, target_sum)
    return a_new, m_v, v_v


def _optimize(a0, deadline, max_iters=5000, topk=16, beta_init=1.0, beta_end=50.0,

    if a.sum() < 1e-6:

    target_sum = np.sqrt(2.0 * n)
    a = _normalize(a, target_sum)
    
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    print(f"    Start: n={n}, score={score:.6f}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        t += 1
        progress = it / max_iters
        
        # Anneal hyperparameters
        beta = beta_init + (beta_end - beta_init) * progress
        lr = lr_init + (lr_end - lr_init) * progress
        
        # Compute autocorrelation
        c = _compute_conv(a)
        max_c = float(c.max())

        k_eff = max(1, min(k_eff, 2 * n - 1))

def _lp_refine(a, deadline, max_iters=50):
    """Pure LP-based refinement. Iterate: solve LP, blend, repeat."""
    n = len(a)
    if n > 1024:
        return a, _compute_obj(a)
    
    target_sum = np.sqrt(2.0 * n)
    a = _normalize(a.copy(), target_sum)
    best = a.copy()
    best_score = _compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = _compute_conv(a)
        M = float(c.max())
        
        g = _lp_solve(a, M, n)
        if g is None:
            break
        
        improved = False
        for alpha in [0.7, 0.5, 0.3, 0.2, 0.1]:

            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_score


def _make_start(n, rng, style='random'):
    """Create a starting sequence of length n."""
    if style == 'random':
        return rng.random(n) * 0.5 + 0.25
    elif style == 'uniform':
        return np.ones(n) * 0.5
    elif style == 'chirp':
        i = np.arange(n, dtype=np.float64)
        return 0.5 + 0.4 * np.cos(2 * np.pi * i * 3.0 / n)
    elif style == 'raised_cos':
        i = np.arange(n, dtype=np.float64)
        return 0.3 + 0.7 * np.cos(np.pi * i / n) ** 2
    elif style == 'gaussian':
        i = np.arange(n, dtype=np.float64)
        center = n / 2.0
        return np.exp(-0.5 * ((i - center) / (n * 0.3)) ** 2)
    elif style == 'bimodal':
        i = np.arange(n, dtype=np.float64)
        c1, c2 = n * 0.3, n * 0.7
        return np.exp(-0.5 * ((i - c1) / (n * 0.15)) ** 2) + np.exp(-0.5 * ((i - c2) / (n * 0.15)) ** 2)
    elif style == 'm_seq':
        # Generate an m-sequence inspired pattern
        # Use a simple LFSR pattern, then smooth
        x = np.zeros(n)
        state = rng.integers(1, 255)
        for i in range(n):
            x[i] = state & 1
            new_bit = ((state >> 3) ^ ((state >> 1) & state) ^ (state >> 4) ^ (state & 1)) 
            state = ((state >> 1) | (new_bit << 7)) & 0xFF
        # Convert to smooth non-negative
        x = (x + 1) / 2  # [0, 1]
        # Smooth with a small window
        kernel = np.ones(3) / 3
        return np.convolve(x, kernel, 'same')
    else:
        return np.ones(n) * 0.5

    Returns optimal y (weights on autocorrelation values).

    c_obj = np.ones(2 * n - 1)

            if 0 <= k < n:

            return res.x
    except Exception:
        pass
    return None

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-dual-guided subgradient descent with Adam optimizer and FFT acceleration.
The LP dual provides optimal subgradient weights for the min-max objective.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    fa = rfft(a, fft_size)
    fb = rfft(b, fft_size)
    return irfft(fa * fb, fft_size)[:n]


def _autocorr(a):
    return _fft_convolve(a, a)


def _obj(a):
    n = len(a)
    c = _autocorr(a)
    mc = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return 2.0 * n * mc / (S * S)


def _grad_weights(w, a):
    """Compute gradient given subgradient weights w (on autocorrelation lags).
    grad[k] = 2 * sum_i a[i] * w[i+k]
    """
    n = len(a)
    # Use FFT: grad = 2 * ifft(FFT(a) * conj(FFT(w_padded)))... 
    # Actually: sum_i a[i] * w[i+k] is cross-correlation.
    # cross_corr[k] = sum_i a[i] * w[i+k] = conv(a, w_flipped)[k + len(w) - 1]
    w_flipped = w[::-1]
    conv_result = _fft_convolve(a, w_flipped)
    offset = len(w) - 1
    return 2.0 * conv_result[offset:offset + n]

    m = 2 * n - 1
    c_obj = np.ones(m)
    
    # Build B: B[j, k] = f[k-j] if 0 <= k-j < n
    B = np.zeros((n, m))
    for j in range(n):
        lo = j
        hi = min(m - 1, j + n - 1)
        B[j, lo:hi+1] = f[hi - j + 1 : 0 : -1]  # f[j], f[j-1], ..., f[lo-j]
        # Wait, let me redo this.
        # B[j, k] = f[k-j] for k from j to j+n-1 (clamped to m-1)
        # So for k = j: f[0], k = j+1: f[1], ..., k = j+n-1: f[n-1]
    
    # Let me rebuild more carefully
    B = np.zeros((n, m))
    for j in range(n):
        for k in range(j, min(m, j + n)):
            B[j, k] = f[k - j]
    
    # This is O(n^2) which is fine for n up to ~2000
    
    # linprog: min c^T y s.t. -B y <= -1, y >= 0
    A_ub = -B
    b_ub = -np.ones(n)
    bounds = [(0, None)] * m
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.fun < 1e10:
            return res.x
    except Exception:
        pass
    return None


def _adam_update(a, grad, state, lr, target_sum):
    """Adam update with non-negativity and sum constraint."""
    t = state['t'] + 1
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_v = state['m']
    v_v = state['v']
    
    m_v = b1 * m_v + (1 - b1) * grad
    v_v = b2 * v_v + (1 - b2) * grad * grad
    m_hat = m_v / (1 - b1 ** t)
    v_hat = v_v / (1 - b2 ** t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-15:
        a_new = np.ones_like(a)
    else:
        a_new = a_new * (target_sum / S)
    
    state['t'] = t
    state['m'] = m_v
    state['v'] = v_v
    return a_new


def _optimize_lp(a0, deadline, max_iters=3000, lr=0.1):
    """Optimize using LP-dual subgradient with Adam."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000.0).copy()
    if a.sum() < 1e-10:
        a = np.ones(n)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    
    best = a.copy()
    best_score = _obj(a)
    
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    
    print(f"      LP-opt start: n={n}, score={best_score:.6f}")
    
    # Solve initial LP to get weights
    c = _autocorr(a)
    M = float(c.max())
    y = _solve_dual_lp(a / a.sum(), n)  # Normalize f for numerical stability
    if y is None:
        y = np.zeros(2 * n - 1)
        y[c.argmax()] = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        # Compute gradient using current weights
        grad = _grad_weights(y, a)
        # Project out mean for sum constraint
        grad = grad - grad.mean()
        
        # Adam step with annealed lr
        progress = it / max_iters
        lr_cur = lr * (1.0 + np.cos(np.pi * progress)) * 0.5
        lr_cur = max(lr_cur, 0.005)
        
        a = _adam_update(a, grad, state, lr_cur, target_sum)
        
        # Periodic LP re-solve
        if (it + 1) % 500 == 0:
            if time.monotonic() > deadline - 30.0:
                break
            c = _autocorr(a)
            M = float(c.max())
            y_new = _solve_dual_lp(a / a.sum(), n)
            if y_new is not None:
                y = y_new
            # Also try direct LP primal for blending
            # (skip for now, the dual gradient should be sufficient)
        
        # Periodic evaluation
        if (it + 1) % 300 == 0:
            s = _obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                if it % 1500 == 0:
                    print(f"        it={it}, score={best_score:.6f}")
    
    s = _obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score
