```python
                delta = new - old
                # Update c
                k_min = i
                k_max = min(i + n - 1, 2*n - 2)
                # Vectorized update for affected lags
                partners = f[k_min - i : k_max - i + 1]  # f[0], f[1], ..., f[k_max-i]
                # For each k in [k_min, k_max]: c[k] += 2*delta*f[k-i]
                # But if k-i == i (i.e., k == 2i): c[k] += 2*delta*old + delta^2 instead of 2*delta*f[i]
                # Since f[i] is being changed, we need to be careful
                # Actually, the "partner" for lag k is f[k-i], and we're changing f[i].
                # The contribution of f[i] to c[k] is:
                #   If j=i: f[i]*f[k-i] -> changes by delta*f[k-i]
                #   If j=k-i: f[k-i]*f[i] -> changes by f[k-i]*delta (if k-i != i)

                # So total change for k != 2i: 2*delta*f[k-i]

                c_new = c.copy()  # O(n)

                if k_min <= 2*i <= k_max:

                    # So actually c_new[2*i] = c[2*i] + 2*delta*old (from the vectorized update)
                    # But the true change is 2*delta*old + delta^2.
                    # So we need to add delta^2 extra.
                    c_new[2*i] += delta**2

                if M_new < best_M:
                    best_M = M_new
                    best_new = new
                    best_c = c_new.copy()

                f[i] = best_new
                c = best_c
                M = best_M
```

```python
import numpy as np
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def normalize(f):
    s = f.sum()
    if s < 1e-10:
        return f
    return f / s

def cd_sweep(f, c, M, frac_candidates, n_inner=1):
    """One sweep of coordinate descent. Returns (f, c, M, improved)."""
    n = len(f)
    improved = False
    for i in range(n):
        if f[i] < 1e-12:
            f[i] = 1e-12
            improved = True
            continue
        best_M = M
        best_f = f.copy()
        best_c = c.copy()
        for frac in frac_candidates:
            f_trial = f.copy()
            f_trial[i] *= frac
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 0.01:
                continue
            # Don't renormalize (scale-invariant)
            c_trial = np.convolve(f_trial, f_trial)
            M_trial = c_trial.max()
            if M_trial < best_M - 1e-15:
                best_M = M_trial
                best_f = f_trial
                best_c = c_trial
        if best_M < M - 1e-15:
            f = best_f
            c = best_c
            M = best_M
            improved = True
    return f, c, M, improved

def run_cd(f, deadline, base_fracs=None):
    """Run cyclic coordinate descent until deadline."""
    if base_fracs is None:
        base_fracs = [0.5, 0.7, 0.85, 0.95, 1.05, 1.2, 1.5]
    f = np.maximum(f, 0.0).astype(np.float64)
    f = normalize(f)
    n = len(f)
    c = np.convolve(f, f)
    M = c.max()
    
    # Adaptive fracs: start wide, narrow down
    all_fracs = [

        [0.8, 0.9, 0.95, 0.98, 1.0, 1.02, 1.05, 1.1],  # fine
    ]
    phase = 0
    stall = 0
    while time.monotonic() < deadline:
        fracs = all_fracs[min(phase, len(all_fracs)-1)]
        f, c, M, improved = cd_sweep(f, c, M, fracs)

                if s > 0.01:

                    c = np.convolve(f, f)
                    M = c.max()
                stall = 0
```

```python
def fft_cd(f, deadline):
    n = len(f)
    f = np.maximum(f, 0.0).astype(np.float64)
    f = f / f.sum()
    N = 2 * n - 1
    ff = np.fft.rfft(f, N)
    # Precompute the exponential factors
    freqs = np.arange(len(ff))  # 0, 1, ..., n
    exp_factors = np.exp(-2j * np.pi * np.outer(np.arange(n), freqs) / N)
    # exp_factors[i, k] = exp(-2*pi*i*k/N)
    
    conv_fft = ff * ff
    c = np.fft.irfft(conv_fft, N)[:2*n-1]
    M = float(c.max())
    
    fracs = [0.5, 0.75, 0.9, 1.1, 1.25, 1.5]
    
    while time.monotonic() < deadline:
        improved = False
        for i in range(n):
            if f[i] < 1e-12:
                continue
            best_M = M
            best_ff = ff
            best_c = c
            for frac in fracs:
                delta = f[i] * (frac - 1.0)
                if abs(delta) < 1e-15:
                    continue
                # Check non-negativity
                if f[i] + delta < 0:
                    continue
                # Update FFT
                ff_trial = ff + delta * exp_factors[i]
                conv_trial = ff_trial * ff_trial
                c_trial = np.fft.irfft(conv_trial, N)[:2*n-1]
                M_trial = float(c_trial.max())
                if M_trial < best_M - 1e-12:
                    best_M = M_trial
                    best_ff = ff_trial
                    best_c = c_trial
                    best_delta = delta
            if best_M < M - 1e-12:
                f[i] += best_delta
                ff = best_ff
                c = best_c
                M = best_M
                improved = True
        if not improved:
            break
    return f
```

"""
Multi-phase optimization for step function sequences minimizing
score = 2*n*max(convolve(f,f)) / (sum(f))^2.

Combines FFT-accelerated coordinate descent, LP-based refinement,
and population-based search across multiple starting points and lengths.

import numpy as np

from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute score faithfully matching evaluate_sequence."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def normalize_f(f):
    """Normalize to sum=1, ensure non-negative."""
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        return f
    return f / s


def fft_cd_sweep(f, ff, c, M, fracs, exp_factors, N):
    """One sweep of FFT-based coordinate descent. Returns updated state."""
    n = len(f)
    improved = False
    for i in range(n):
        if f[i] < 1e-12:
            continue
        best_M = M
        best_ff = None
        best_c = None
        best_delta = 0.0
        for frac in fracs:
            delta = f[i] * (frac - 1.0)
            if abs(delta) < 1e-16:
                continue
            if f[i] + delta < 0.0:
                continue
            ff_trial = ff + delta * exp_factors[i]
            conv_trial = ff_trial * ff_trial
            c_trial = np.fft.irfft(conv_trial, N)[:2 * n - 1]
            M_trial = float(c_trial.max())
            if M_trial < best_M - 1e-13:
                best_M = M_trial
                best_ff = ff_trial
                best_c = c_trial
                best_delta = delta
        if best_M < M - 1e-13:
            f[i] += best_delta
            ff = best_ff
            c = best_c
            M = best_M
            improved = True
    return f, ff, c, M, improved


def run_fft_cd(f, deadline, verbose=True):
    """Run FFT-based coordinate descent with adaptive step sizes."""
    n = len(f)
    f = normalize_f(f)
    N = 2 * n - 1
    ff = np.fft.rfft(f, N)
    freqs = np.arange(len(ff))
    exp_factors = np.exp(-2j * np.pi * np.outer(np.arange(n), freqs) / N)
    conv_fft = ff * ff
    c = np.fft.irfft(conv_fft, N)[:2 * n - 1]
    M = float(c.max())
    s = f.sum()
    best_score = 2.0 * n * M / (s * s)
    best_f = f.copy()

    # Adaptive step size phases
    phase_steps = [
        ([0.2, 0.4, 0.6, 0.8, 1.2, 1.6, 2.5], 4),
        ([0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7], 6),
        ([0.5, 0.7, 0.85, 0.93, 1.0, 1.07, 1.15, 1.5], 8),
        ([0.7, 0.85, 0.93, 0.97, 1.0, 1.03, 1.07, 1.15], 10),
        ([0.85, 0.93, 0.97, 0.99, 1.0, 1.01, 1.03, 1.07], 12),
        ([0.95, 0.98, 0.99, 1.0, 1.005, 1.02, 1.05], 15),
    ]

    stall = 0
    phase = 0
    total_sweeps = 0

    while time.monotonic() < deadline:
        if phase >= len(phase_steps):
            phase = len(phase_steps) - 1
        fracs, max_no_improve = phase_steps[phase]

        f, ff, c, M, improved = fft_cd_sweep(f, ff, c, M, fracs, exp_factors, N)
        total_sweeps += 1

        if improved:
            score = 2.0 * n * M / (f.sum() ** 2)
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall >= max_no_improve:
                stall = 0
                if phase < len(phase_steps) - 1:
                    phase += 1
                    if verbose:
                        print("    CD phase ->", phase, "sweeps:", total_sweeps,
                              "score: %.6f" % best_score, flush=True)
                else:
                    # Perturb to escape local minimum
                    rng = np.random.default_rng(total_sweeps)
                    n_perturb = max(1, n // 50)
                    for _ in range(n_perturb):
                        idx = rng.integers(0, n)
                        f[idx] *= rng.uniform(0.6, 1.4)
                    f = normalize_f(f)
                    ff = np.fft.rfft(f, N)
                    conv_fft = ff * ff
                    c = np.fft.irfft(conv_fft, N)[:2 * n - 1]
                    M = float(c.max())

        if total_sweeps % 100 == 0 and verbose:
            score = 2.0 * n * M / (f.sum() ** 2)
            print("    CD sweep %d: current=%.6f best=%.6f M=%.6e" %
                  (total_sweeps, score, best_score, M), flush=True)

    return best_f, best_score


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) for convolution constraint."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A
