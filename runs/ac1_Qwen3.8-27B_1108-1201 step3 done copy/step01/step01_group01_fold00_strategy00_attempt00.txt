                    if ob < best_objective:
                        best_objective = ob
                        best_candidate = ter
                        print(f"Found new best at objective {best_objective:.6f}")
                
                starter_idx += 1
        
        # Escape loop in case time_remaining becomes negative
        time_remaining = max(time_remaining, 0.0)
        
        #If we've lost track, return current best
        if best_candidate is None:
            if constraints and base_length > 0:
                best_candidate = starter_parent
            else:
                # Combine all zero and then the starting points (must be non-empty)
                try:
                    best_candidate = np.concatenate([starter_val for _, starter_val in starters_per_length.items()])
                except Exception:
                    best_candidate = np.ones(base_length) if base_length > 0 else np.array([])
        
        time_remaining = total_time - (time.monotonic() - start_times)
    
    # Stage 7: Force a last optimization if time allows
    if best_objective == float('inf'):
        # Suggest the parent sequence if only that exists
        if exists:
            best_candidate = starter_parent
            best_objective = 2 * base_length * np.max(np.convolve(best_candidate, best_candidate)) / (np.sum(best_candidate) ** 2)
        else:
            # Use the union of all instances to form a single sequence (edge case handling)
            best_candidate = np.ones(6121)
            best_objective = 2 * 6121 * np.max(np.convolve(best_candidate, best_candidate)) / (np.sum(best_candidate) ** 2)
    
    # Stage 8: Final acceptance and potential fallback
    # We must provide a valid sequence if nothing else has been found
    # Let the parent be our fallback if necessary
    if best_objective == float('inf'):
        # Use the parent sequence as fallback (if available)
        if exists:
            best_candidate = starter_parent
            # Force a simplex projection in case negative numbers crept in
            best_candidate = project_simplex(best_candidate, 1.0)
            best_objective = 2 * base_length * np.max(np.convolve(best_candidate, best_candidate)) / (np.sum(best_candidate) ** 2)
        else:
            # Initialize with a uniform sequence for performance
            best_candidate = np.ones(6121) / 1000.0
            best_objective = optimize_start_point(best_candidate, 10.0)[1]
    
    # Stage 9: Print the final candidate and return it
    print("Optimization complete with best candidate")
    return best_candidate if best_objective < float('inf') else (best_candidate, best_objective)[0]
