"""
Minimize 2*n*max(self_convolution)/sum^2 over non-negative sequences.

Algorithm:
- Continuation method with softmax-smoothed max objective
- FFT-based gradient computation via cross-correlation
- Adam optimizer with simplex projection and Nesterov momentum
- Multiple diverse initializations
- Final polishing with exact subgradient

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_max_grad(a, beta):
    """Softmax-smoothed max of self-convolution and its gradient."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    weights = exp_bc / exp_bc.sum()
    val = (c_max + np.log(exp_bc.sum())) / beta
    # Gradient: grad[m] = 2 * sum_j a[j]*weights[m+j]
    # = 2 * (a_rev * weights)[n-1+m]
    a_rev = a[::-1]
    conv_len = n + size - 1
    fsize2 = 1 << (conv_len - 1).bit_length()
    AR = fft(a_rev, fsize2)
    W = fft(weights, fsize2)
    corr = np.real(ifft(AR * W))
    grad = 2.0 * corr[n-1:n-1+n]
    return val, grad


def exact_max_subgrad(a, top_k=16):
    """Exact max and subgradient from top-k active lags."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    max_val = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total <= 1e-30:
        return max_val, np.zeros(n)
    weights = top_vals / total
    grad = np.zeros(n)
    for w_val, k in zip(weights, top_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * w_val * a[k-hi:k-lo+1][::-1]
    return max_val, grad


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def adam_continuation(a_init, n, S, max_time, rng, beta_schedule, lr_init):
    """Adam with simplex projection and continuation on beta."""
    a = project_simplex(np.clip(a_init, 0, 1000.0), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - start >= max_time:
            break
        # Reset momentum when changing beta significantly
        beta_start = time.monotonic()
        max_iters_for_beta = int(50000 * (1.0 / max(1, len(beta_schedule))))
        
        while time.monotonic() - start < max_time and t < 500000:
            t += 1
            val, grad = smooth_max_grad(a, beta)
            # Clip gradient for numerical stability
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e6:
                grad = grad / gnorm * 1e6
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            lr = lr_init / (1.0 + 0.001 * t)
            a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
            
            # Check improvement using exact objective
            if t % 500 == 0:
                exact_max, _ = exact_max_subgrad(a, top_k=1)
                if exact_max < best_score:
                    best_score = exact_max
                    best_local = a.copy()
    
    # Final exact check
    exact_max, _ = exact_max_subgrad(a, top_k=1)
    if exact_max < best_score:
        best_score = exact_max
        best_local = a.copy()
    
    return best_local, best_score


def adam_exact_subgrad(a_init, n, S, max_time, rng, lr_init, top_k=8):
    """Adam with exact subgradient (no smoothing)."""
    a = project_simplex(np.clip(a_init, 0, 1000.0), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        t += 1
        max_val, grad = exact_max_subgrad(a, top_k)
        if max_val < best_score:
            best_score = max_val
            best_local = a.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr = lr_init * (0.5 ** (t / 2000.0))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Random restart within the search
        if t % 5000 == 0 and rng.random() < 0.05:
            noise = rng.normal(0, 0.005 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score
