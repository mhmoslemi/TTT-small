import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_losm_exp_gradient(a, conv, T):
    """Compute gradient of (1/T) * log(sum(exp(T * conv))) w.r.t. a."""
    n = len(a)
    # Softmax weights
    max_c = conv.max()
    exp_T = np.exp(T * (conv - max_c))
    p = exp_T / exp_T.sum()
    
    # grad[m] = 2 * sum_k p[k] * a[k-m] for 0 <= k-m < n
    # This is a cross-correlation: (p ⊛ a)[m] = sum_j p[m+j] * a[j]
    # Use FFT
    size = len(conv)  # 2n-1
    N = 1 << (size + n - 1 - 1).bit_length()  # pad to next power of 2 for linear cross-corr
    P = fft(p, N)
    A = fft(a, N)
    # Cross-correlation: IFFT(P * conj(A))
    cross = np.real(ifft(P * np.conj(A)))[:n]
    return 2.0 * cross

def fista_losm(a_init, n, S, max_time, T_schedule, lr, rng):
    """FISTA with log-sum-exp smoothing and adaptive temperature."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    a = a * (S / max(a.sum(), 1e-15))
    
    y = a.copy()
    t_momentum = 1.0
    
    best_conv_max = np.inf

import numpy as np

from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    rho_candidates = np.where(w > (cum - S) / np.arange(1, n + 1))[0]
    if len(rho_candidates) == 0:
        theta = (cum[-1] - S) / n
    else:
        rho = rho_candidates[-1] + 1
        theta = (cum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def losm_grad(a, conv, T):
    """Gradient of (1/T)*log(sum(exp(T*conv))) w.r.t. a, via FFT cross-correlation."""
    n = len(a)
    size = len(conv)  # 2n-1
    # Softmax
    shifted = T * (conv - conv.max())
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    # Cross-correlation: (p ⊛ a)[m] = sum_{j=0}^{n-1} p[m+j]*a[j]
    # FFT size >= size + n - 1 = 3n-2
    N = 1 << (size + n - 2).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    cross = np.real(ifft(P * np.conj(A)))[:n]
    return 2.0 * cross

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: FISTA with log-sum-exp smoothing on the max-convolution objective,
normalized so sum(a) = sqrt(2n) (making the objective equal to max(conv)).
Multiple initializations with varying asymmetry, adaptive temperature schedule,
FFT-based gradient computation, and spectral flattening for escape from local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cum - S) / idx
    valid = np.where(w >= theta_cand)[0]
    if len(valid) == 0:
        theta = (cum[-1] - S) / n
    else:
        rho = valid[-1] + 1
        theta = (cum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def losm_grad(a, conv, T):
    """
    Gradient of (1/T)*log(sum_k exp(T*conv[k])) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(a)
    size = len(conv)
    # Softmax weights (numerically stable)
    shifted = T * (conv - conv.max())
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()

    # Cross-correlation: (p ⊛ a)[m] = sum_{j=0}^{n-1} p[m+j]*a[j]
    N = 1 << (size + n - 2).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    cross = np.real(ifft(P * np.conj(A)))[:n]
    return 2.0 * cross


def spectral_step(a, S, alpha=0.4):
    """One step of spectral flattening: blend |DFT| towards geometric mean."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    # Geometric mean of non-DC, non-zero magnitudes
    mask = np.ones(N, dtype=bool)
    mask[0] = False
    valid = mask & (mag > 1e-12)
    if valid.sum() > 10:
        gm = np.exp(np.mean(np.log(mag[valid])))
        target = np.full(N, gm, dtype=np.float64)
        target[0] = mag[0]
        new_mag = (1.0 - alpha) * mag + alpha * target
    else:
        new_mag = mag.copy()

    new_A = new_mag * np.exp(1j * phase)
    # Enforce Hermitian symmetry for real output
    new_A[0] = np.real(new_A[0])
    if N % 2 == 0:
        new_A[N // 2] = np.real(new_A[N // 2])
        idx = np.arange(1, N // 2)
        new_A[N - idx] = np.conj(new_A[idx])

    a_new = np.real(ifft(new_A))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    else:
        a_new = np.full(n, S / n)
    return a_new


def fista_run(a_init, S, time_budget, rng, T_schedule=None):
    """
    FISTA optimization with log-sum-exp smoothing and temperature schedule.
    Returns (best_sequence, best_max_conv).
    """
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * (S / s)

    if T_schedule is None:
        T_schedule = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0]
    frac = [0.06, 0.10, 0.16, 0.20, 0.26, 0.22]
    while len(frac) < len(T_schedule):
        frac.append(0.15)

    start = time.monotonic()
    best_a = a.copy()
    best_score = fft_conv_self(a).max()

    for pi, T in enumerate(T_schedule):
        phase_budget = time_budget * frac[pi]
        phase_start = time.monotonic()

        y = a.copy()
        t_mom = 1.0
        max_a = max(np.max(a), S / n)
        lr = 4.0 / (T * max_a + 1e-12)
        iter_count = 0
        no_improve = 0
        local_best = best_score
        use_spectral = (pi >= 3)

        while time.monotonic() - phase_start < phase_budget:
            conv = fft_conv_self(y)
            grad = losm_grad(y, conv, T)

            a_new = proj_simplex(y - lr * grad, S)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
            a = a_new
            t_mom = t_new
            iter_count += 1

            if iter_count % 150 == 0:
                cur = fft_conv_self(a).max()
                if cur < local_best - 1e-8:
                    local_best = cur
                    no_improve = 0
                    if cur < best_score:
                        best_score = cur
                        best_a = a.copy()
                else:
                    no_improve += 1

                if no_improve >= 8:
                    # Escape strategy
                    r = rng.random()
                    if use_spectral and r < 0.5:
                        a = spectral_step(a, S, alpha=0.3 + 0.3 * r)
                    else:
                        a = proj_simplex(a + rng.normal(0.0, 0.005 * S / n, n), S)
                    y = a.copy()
                    t_mom = 1.0
                    no_improve = 0
                    max_a = max(np.max(a), S / n)
                    lr = 4.0 / (T * max_a + 1e-12)

            # Slow lr decay
            if iter_count % 500 == 0:
                lr *= 0.995

        # End of phase
        cur = fft_conv_self(a).max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

    return best_a, best_score


def make_init_ramp(n, slope, rng):
    """Raised cosine with linear tilt."""
    i = np.arange(n, dtype=np.float64)
    base = 0.5 * (1.0 + np.cos(2.0 * np.pi * (i + 0.5) / n))
    tilt = 1.0 + slope * (2.0 * (i + 0.5) / n - 1.0)
    a = base * np.maximum(tilt, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a


def make_init_gauss(n, center_frac, width, rng):
    """Gaussian with offset center."""
    i = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    a = np.exp(-((i - center) / width) ** 2)
    return a


def make_init_linear(n, power, rng):
    """Power-law ramp: a[i] = ((i+1)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1.0) / n) ** power
    return a
