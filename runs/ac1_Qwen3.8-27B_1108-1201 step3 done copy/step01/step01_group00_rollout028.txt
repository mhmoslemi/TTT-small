For a single non-negative sequence of length n, the minimum possible peak of the linear autocorrelation (given fixed sum) is related to the "ideal" autocorrelation. 

Actually, let me think about this problem from the perspective of the power spectrum. The DFT of the sequence `a` (padded to length 2n-1) has magnitude squared `|A[k]|^2`. The circular autocorrelation is `IFFT(|A[k]|^2)`. The linear autocorrelation is a "wrapped" version of the circular one.

If we could make `|A[k]|^2` perfectly flat, the circular autocorrelation would be a delta at 0, and the linear autocorrelation peak would be minimized.

The "flattest" power spectrum achievable with a non-negative sequence of fixed sum is related to the "minimum distance" problem in signal processing.

If we parameterize `a` through its DFT coefficients, we can try to minimize the variance of `|A[k]|^2`. This is equivalent to minimizing the peak of the autocorrelation (by Parseval's theorem, the L-infinity norm of the autocorrelation is related to the L2 norm of the power spectrum deviations from flat).

"""
Optimized search for minimizing the normalized peak of linear autocorrelation
of non-negative step functions. Uses hybrid gradient (soft-max + exact subgradient),
backtracking line search, peak-directed coordinate descent polish, and multi-start
with diverse initializations including spectral-flatness-based starts.
"""

import numpy as np
import time
import math


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the objective function value."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf, np.inf, -1
    c = fft_convolve(a, a)
    max_c = np.max(c)
    peak_idx = int(np.argmax(c))
    score = 2.0 * n * max_c / (s * s)
    return score, max_c, peak_idx


def softmax_normalized(theta):
    """Stable softmax normalization."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def grad_softmax_conv(a, tau):
    """
    Compute soft-max value and gradient w.r.t. a.
    a: the sequence (normalized, non-negative)
    tau: temperature
    Returns: value, gradient (w.r.t. a), convolution
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    
    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * cross_correlation(w, a)[m]
    # Compute via: convolve(w, flip(a)) and extract
    flip_a = a[::-1]
    conv = fft_convolve(w, flip_a)
    # conv has length (2n-1) + n - 1 = 3n-2
    # We want R[m] = sum_k w[k] * a[k-m] = conv[m + n - 1]
    R = conv[n-1:2*n-1]
    grad = 2.0 * R
    return value, grad, c


def grad_exact_peak(a, peak_idx):
    """
    Exact subgradient at the peak index.
    grad[m] = 2 * a[peak_idx - m] if 0 <= peak_idx - m < n, else 0
    """
    n = len(a)
    grad = np.zeros(n)
    # peak_idx ranges from 0 to 2n-2
    # For m in [0, n-1]: j = peak_idx - m must be in [0, n-1]
    # So m must be in [peak_idx - (n-1), peak_idx] ∩ [0, n-1]
    m_min = max(0, peak_idx - (n - 1))
    m_max = min(n - 1, peak_idx)
    if m_min <= m_max:
        ms = np.arange(m_min, m_max + 1)
        js = peak_idx - ms
        grad[ms] = 2.0 * a[js]
    return grad


def optimize_hybrid(n, theta_init, time_limit, rng, S=None):
    """
    Hybrid gradient optimization combining soft-max and exact peak gradients.
    Uses backtracking line search for step size.
    """
    if S is None:
        S = np.sqrt(2.0 * n)
    
    theta = theta_init.astype(np.float64).copy()
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    
    tau = 50.0 / n  # Start with relatively high temperature
    lr = 1.0
    min_lr = 0.01
    max_lr = 10.0
    
    step = 0
    stall_count = 0
    stall_best = best_score
    
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)
        
        # Adaptive temperature: start high, decrease over time
        tau = (50.0 / n) * (0.01) ** (elapsed_frac ** 0.7)
        
        a = softmax_normalized(theta) * S
        score, max_c, peak_idx = compute_score(a)
        
        if score < best_score:
            best_score = score
            best_theta = theta.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # If stalled for too long, perturb
        if stall_count > 500:
            theta += rng.normal(0, 0.1, n)
            stall_count = 0
            tau = max(tau, 20.0 / n)  # Reheat
            continue
        
        # Compute hybrid gradient
        # Soft-max gradient
        _, g_smooth, _ = grad_softmax_conv(a, tau)
        
        # Exact peak subgradient
        g_peak = grad_exact_peak(a, peak_idx)
        
        # Normalize both gradients to have similar scale
        g_smooth_norm = np.linalg.norm(g_smooth)
        g_peak_norm = np.linalg.norm(g_peak)
        
        # Blend: weight the exact gradient more as we cool
        blend = min(1.0, elapsed_frac * 2.0)  # 0 at start, 1 at end
        if g_smooth_norm > 1e-10 and g_peak_norm > 1e-10:
            g = (1 - blend) * (g_smooth / g_smooth_norm) + blend * (g_peak / g_peak_norm)
        else:
            g = g_smooth if g_smooth_norm > 1e-10 else g_peak
        
        # Project gradient to the tangent space of the simplex constraint
        # If a = softmax(theta) * S, then da/dtheta = a * (I - a/S_outer)
        # The constraint is sum(a) = S, so we need grad_theta to satisfy
        # the constraint. With softmax parameterization, this is automatic.
        # grad_theta = a * (g - a · g / S)
        inner = np.dot(a, g) / S
        g_theta = a * (g - inner)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * g_theta
        
        # Backtracking line search
        alpha = lr
        found = False
        for _ in range(10):
            theta_candidate = theta - alpha * momentum
            a_candidate = softmax_normalized(theta_candidate) * S
            score_candidate, _, _ = compute_score(a_candidate)
            
            if np.isfinite(score_candidate) and score_candidate < score:
                theta = theta_candidate
                found = True
                break
            alpha *= 0.5
        
        if not found:
            # If no improvement found, reduce learning rate and try small step
            alpha = lr * 0.1
            theta = theta - alpha * momentum
            lr = max(min_lr, lr * 0.7)
        else:
            lr = min(max_lr, lr * 1.05)
        
        step += 1
        if step % 500 == 0:
            a_check = softmax_normalized(theta) * S
            score_check, _, _ = compute_score(a_check)
            print(f"    step={step}, score={score_check:.6f}, best={best_score:.6f}, tau={tau:.2e}, lr={lr:.3f}, t={time.monotonic()-t0:.1f}s")
    
    return best_theta, best_score


def coordinate_descent_polish(a_init, time_limit, rng):
    """
    Coordinate descent: try adjusting individual elements or small groups.
    Focus on reducing the peak.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    score, max_c, peak_idx = compute_score(current)
    best = current.copy()
    best_score = score
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution and identify peak contributors
        c = fft_convolve(current, current)
        pk = int(np.argmax(c))
        
        # Find top contributors to the peak
        contribs = np.zeros(n)
        for i in range(min(n, pk + 1)):
            j = pk - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]
        
        # Strategy 1: Reduce top contributors
        r = rng.random()
        
        if r < 0.4:
            # Reduce top contributor
            top_k = max(1, n // 32)
            top_idx = np.argsort(contribs)[-top_k:]
            idx = top_idx[rng.integers(len(top_idx))]
            factor = rng.uniform(0.7, 0.95)
            candidate = current.copy()
            candidate[idx] *= factor
        elif r < 0.7:
            # Increase a non-contributing element
            sorted_idx = np.argsort(contribs)
            bottom_half = sorted_idx[:n//2]
            idx = bottom_half[rng.integers(len(bottom_half))]
            factor = rng.uniform(1.05, 1.3)
            candidate = current.copy()
            candidate[idx] *= factor
        elif r < 0.85:
            # Pair swap: reduce one, increase another (keep sum roughly same)
            top_idx = np.argsort(contribs)[-n//16:]
            bot_idx = np.argsort(contribs)[:n//16]
            i1 = top_idx[rng.integers(len(top_idx))]
            i2 = bot_idx[rng.integers(len(bot_idx))]
            transfer = current[i1] * rng.uniform(0.1, 0.3)
            candidate = current.copy()
            candidate[i1] -= transfer
            candidate[i2] += transfer
            candidate = np.clip(candidate, 0, None)
        else:
            # Block operation
            width = rng.integers(1, max(2, n // 16))
            start = rng.integers(0, n - width)
            candidate = current.copy()
            candidate[start:start+width] *= rng.uniform(0.8, 1.2)
        
        new_score, _, new_pk = compute_score(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best = candidate.copy()
            current = candidate.copy()
            score = new_score
        elif new_score < score or rng.random() < 0.05:
            current = candidate.copy()
            score = new_score
    
    return best, best_score


def make_zadoff_chu_init(n, rng):
    """Create a Zadoff-Chu-like initialization."""
    # Zadoff-Chu sequences have ideal autocorrelation properties
    # For a sequence of length N (odd), zc[k] = exp(-j*pi*k*(k+1)/N)
    # We use the real part (or magnitude-based) as initialization
    N = n if n % 2 == 1 else n + 1
    k = np.arange(N, dtype=np.float64)
    phase = -np.pi * k * (k + 1) / N
    zc = np.exp(1j * phase)
    # Use magnitude + small perturbation as non-negative sequence
    a = 1.0 + 0.3 * np.abs(zc[:n]) + 0.1 * np.cos(2 * np.pi * rng.integers(1, 5) * np.arange(n) / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_chirp_init(n, rng):
    """Create a chirp (FM sweep) initialization."""
    t = np.arange(n, dtype=np.float64)
    # Frequency sweep
    f0 = rng.uniform(2, 8)
    f1 = rng.uniform(f0 + 5, f0 + 20)
    phase = 2 * np.pi * (f0 * t / n + (f1 - f0) * t**2 / (2 * n**2))
    a = 1.0 + 0.3 * np.cos(phase) + 0.1 * np.cos(2 * np.pi * rng.uniform(3, 10) * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_flat_spectrum_init(n, rng):
    """
    Create an initialization that aims for flat power spectrum.
    Start with random phases, take inverse FFT, take absolute value.
    """
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    # Random phases with flat magnitude
    n_freq = size // 2 + 1
    magnitudes = np.ones(n_freq)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    # Make it real and even-symmetric
    F = np.zeros(size, dtype=np.complex128)
    F[:n_freq] = magnitudes * np.exp(1j * phases)
    # Ensure symmetry for real output
    for k in range(1, size // 2 + 1):
        F[size - k] = np.conj(F[k])
    a_complex = np.fft.ifft(F)
    a = np.abs(a_complex[:n])
    a = a / np.mean(a)  # Normalize to mean 1
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_golay_like_init(n, rng):
    """
    Create a Golay-like initialization with low autocorrelation.
    Uses the Golay complementary pair construction idea.
    """
    # Start with a binary Golay-like sequence and smooth it
    # Golay pair of length 4: [1,1,1,-1] and [1,1,-1,0]
    # For our purposes, use a random ±1 sequence and smooth
    binary = rng.choice([0.3, 1.0], size=n)  # Non-negative "binary-like"
    # Smooth with a small kernel
    kernel = np.ones(3) / 3
    a = np.convolve(binary, kernel, mode='same')
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random_init(n, rng):
    """Random initialization."""
    a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

"""
Hybrid optimization for minimizing the normalized peak of linear autocorrelation
of non-negative step functions. Uses: (1) diverse spectral-based initializations,
(2) gradient descent with adaptive blend of soft-max and exact peak subgradient,
(3) backtracking line search, (4) peak-directed coordinate descent polish,
(5) multi-scale length exploration with time-aware budgeting.
"""

import numpy as np
import time
import math


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute (score, max_conv, peak_idx) for a non-negative sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf, np.inf, -1
    c = fft_convolve(a, a)
    max_c = np.max(c)
    peak_idx = int(np.argmax(c))
    score = 2.0 * n * max_c / (s * s)
    return score, max_c, peak_idx


def softmax_normalized(theta):
    """Stable softmax returning values summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def grad_softmax_conv(a, tau):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns (value, gradient_wrt_a, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * math.log(sum_exp)
    # grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * conv(w, a_flip)[i + n - 1]
    a_flip = a[::-1]
    conv_wa = fft_convolve(w, a_flip)
    R = conv_wa[n - 1: 2 * n - 1]
    grad = 2.0 * R
    return value, grad, c


def grad_exact_peak(a, peak_idx):
    """Exact subgradient at the peak: grad[m] = 2*a[peak_idx - m] if valid."""
    n = len(a)
    grad = np.zeros(n)
    m_min = max(0, peak_idx - (n - 1))
    m_max = min(n - 1, peak_idx)
    if m_min <= m_max:
        ms = np.arange(m_min, m_max + 1)
        js = peak_idx - ms
        grad[ms] = 2.0 * a[js]
    return grad


def optimize_gradient_hybrid(n, theta_init, time_limit, rng):
    """
    Gradient descent with adaptive soft-max/exact gradient blend and
    backtracking line search. Returns (best_theta, best_score).
    """
    S = math.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf

    momentum = np.zeros(n)
    beta = 0.85
    lr_base = 3.0
    min_lr = 0.05
    max_lr = 20.0
    lr = lr_base

    step = 0
    stall = 0
    reheat_cooldown = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        # Temperature schedule: exponential decay with floor
        tau = (40.0 / n) * (0.02) ** (frac ** 0.6)
        tau = max(tau, 0.1 / n)

        a = softmax_normalized(theta) * S
        score, max_c, peak_idx = compute_score(a)

        if score < best_score:
            best_score = score
            best_theta = theta.copy()
            stall = 0
        else:
            stall += 1

        # Reheat if stalled
        if stall > 300 and reheat_cooldown <= 0:
            theta += rng.normal(0, 0.15, n)
            stall = 0
            reheat_cooldown = 100
            lr = lr_base
        if reheat_cooldown > 0:
            reheat_cooldown -= 1

        # Compute gradient
        _, g_smooth, _ = grad_softmax_conv(a, tau)
        g_peak = grad_exact_peak(a, peak_idx)

        # Adaptive blend: more weight on exact peak as we cool
        blend = min(1.0, frac * 1.5 + 0.2)
        g_s_norm = np.linalg.norm(g_smooth)
        g_p_norm = np.linalg.norm(g_peak)

        if g_s_norm > 1e-12 and g_p_norm > 1e-12:
            g = (1.0 - blend) * (g_smooth / g_s_norm) + blend * (g_peak / g_p_norm)
        elif g_s_norm > 1e-12:
            g = g_smooth / g_s_norm
        else:
            g = g_peak

        # Convert to theta-space gradient
        inner = np.dot(a, g) / S
        g_theta = a * (g - inner)

        # Momentum
        momentum = beta * momentum + (1.0 - beta) * g_theta

        # Backtracking line search
        alpha = lr
        best_candidate = None
        best_candidate_score = score
        for _ in range(12):
            theta_cand = theta - alpha * momentum
            a_cand = softmax_normalized(theta_cand) * S
            score_cand, _, _ = compute_score(a_cand)
            if np.isfinite(score_cand) and score_cand < best_candidate_score:
                best_candidate_score = score_cand
                best_candidate = theta_cand
            alpha *= 0.5

        if best_candidate is not None:
            theta = best_candidate
            lr = min(max_lr, lr * 1.08)
        else:
            # No improvement found; take a tiny step anyway and reduce lr
            theta = theta - 0.01 * lr * momentum
            lr = max(min_lr, lr * 0.6)

        step += 1
        if step % 300 == 0:
            a_chk = softmax_normalized(theta) * S
            s_chk, _, _ = compute_score(a_chk)
            print(f"    step={step}, cur={s_chk:.6f}, best={best_score:.6f}, tau={tau:.2e}, lr={lr:.2f}, t={elapsed:.1f}s")

    return best_theta, best_score


def peak_directed_cd(a_init, time_limit, rng):
    """
    Coordinate descent with peak-directed mutations.
    Aggressively reduces elements contributing to the current peak.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    score, max_c, peak_idx = compute_score(current)
    best = current.copy()
    best_score = score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(current, current)
        pk = int(np.argmax(c))

        # Compute per-element contribution to the peak
        contribs = np.zeros(n)
        m_min = max(0, pk - (n - 1))
        m_max = min(n - 1, pk)
        for i in range(m_min, m_max + 1):
            j = pk - i
            if 0 <= j < n:
                contribs[i] = current[i] * current[j]

        r = rng.random()
        candidate = current.copy()

        if r < 0.30:
            # Reduce top contributors to peak
            top_k = max(2, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            n_reduce = max(1, top_k // 3)
            chosen = rng.choice(top_idx, size=min(n_reduce, len(top_idx)), replace=False)
            candidate[chosen] *= np.exp(rng.normal(-0.06, 0.02, size=len(chosen)))
            # Increase non-contributors to maintain sum
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = max(1, min(6, len(other)))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.02, size=n_inc))
        elif r < 0.50:
            # Targeted reduction of single worst element
            worst = int(np.argmax(contribs))
            candidate[worst] *= rng.uniform(0.6, 0.9)
            # Compensate elsewhere
            others = np.setdiff1d(np.arange(n), [worst])
            n_comp = min(4, len(others))
            comp_idx = rng.choice(others, size=n_comp, replace=False)
            candidate[comp_idx] *= rng.uniform(1.05, 1.2, size=n_comp)
        elif r < 0.65:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.80:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.90:
            # Partial symmetrization
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.015, size=n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.012, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score, _, _ = compute_score(candidate)

        if new_score < best_score:
            best_score = new_score
            best = candidate.copy()
            current = candidate.copy()
            score = new_score
        elif new_score < score:
            current = candidate.copy()
            score = new_score
        elif rng.random() < math.exp(-(new_score - score) / max(temperature, 1e-10)):
            current = candidate.copy()
            score = new_score

        temperature *= cooling

    return best, best_score


def make_chirp_init(n, rng):
    """Chirp (FM sweep) initialization."""
    t = np.arange(n, dtype=np.float64)
    f0 = rng.uniform(3, 10)
    f1 = f0 + rng.uniform(5, 20)
    phase = 2 * np.pi * (f0 * t / n + (f1 - f0) * t**2 / (2.0 * n * n))
    a = 1.0 + 0.35 * np.cos(phase) + 0.15 * np.cos(
        2 * np.pi * rng.uniform(2, 8) * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_flat_spectrum_init(n, rng):
    """Initialize with flat power spectrum (zero-phase inverse FFT of flat magnitude)."""
    size = 1
    while size < 2 * n:
        size <<= 1
    n_freq = size // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    F = np.zeros(size, dtype=np.complex128)
    F[:n_freq] = np.exp(1j * phases)
    for k in range(1, size // 2 + 1):
        F[size - k] = np.conj(F[k])
    a_complex = np.fft.ifft(F)
    a = np.abs(a_complex[:n])
    a = a / (np.mean(a) + 1e-10)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_zadoff_chu_init(n, rng):
    """Zadoff-Chu-inspired initialization."""
    N = n if n % 2 == 1 else n + 1
    k = np.arange(N, dtype=np.float64)
    root = rng.integers(1, N // 2)
    phase = -np.pi * root * k * (k + 1) / N
    zc = np.exp(1j * phase)
    a = 1.0 + 0.3 * np.cos(np.angle(zc[:n])) * np.abs(zc[:n])
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_binary_smooth_init(n, rng):
    """Smooth binary-like initialization (Golay-inspired)."""
    binary = rng.choice([0.2, 1.0], size=n)
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(binary, kernel, mode='same')
    a += 0.08 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random_init(n, rng):
    """Random initialization around 1."""
    a = 1.0 + 0.25 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

        polished, polish_score = peak_directed_cd(best_seq, cd_time, rng)

            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After CD: {best_score:.6f}")
        else:
            print(f"  CD did not improve (best still {best_score:.6f})")

    # Phase 3: Final gradient refinement on the best
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Phase 3: Final gradient refinement ===")
        n = len(best_seq)
        S = math.sqrt(2.0 * n)
        theta_init = np.log(best_seq / np.sum(best_seq) + 1e-10)
        time_left = max(30.0, (deadline - time.monotonic()) * 0.40)
        theta_best, grad_score = optimize_gradient_hybrid(n, theta_init, time_left, rng)
        a_best = softmax_normalized(theta_best) * S
        actual_score, _, _ = compute_score(a_best)
        print(f"  After final gradient: {actual_score:.6f}")
        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Final CD polish
    if time.monotonic() < deadline - 20 and best_seq is not None:
        final_cd_time = max(10.0, (deadline - time.monotonic()) * 0.5)
        print(f"\n=== Phase 4: Final CD polish (time={final_cd_time:.0f}s) ===")
        polished, polish_score = peak_directed_cd(best_seq, final_cd_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  Final CD score: {best_score:.6f}")

    # Final verification with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
