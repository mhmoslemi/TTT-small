**Approach: Projected Gradient Descent on the "peak reduction" problem**

At any point, we know which convolution entries are at the maximum (or near-maximum). We want to reduce these. The key insight is:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max (softmax) approximation
with L-BFGS optimization in log-softmax parameterization. Uses temperature annealing
(starting smooth, becoming sharp) combined with diverse initializations and
multiple sequence lengths. Includes coordinate-wise polish as final step.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    result = np.fft.irfft(A * A, size)
    return result[:2 * n - 1]


def softmax(x):
    """Numerically stable softmax."""
    xm = x - x.max()
    e = np.exp(xm)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def smooth_obj_grad(x, t):
    """Compute smooth objective and gradient in log-softmax parameterization.
    Objective: 2*n*smooth_max(conv(softmax(x), softmax(x)))
    """
    n = len(x)
    a = softmax(x)
    conv = fast_convolve(a)

    # Smooth max
    scaled = t * conv
    m = scaled.max()
    exp_s = np.exp(scaled - m)
    Z = exp_s.sum()
    w = exp_s / Z
    S = (np.log(Z) + m) / t

    # Gradient: dS/dx_j = sum_k w_k * d(conv_k)/dx_j
    # d(conv_k)/dx_j = 2*a_j*(a_{k-j} - conv_k)
    # So dS/dx_j = 2*a_j * (s_j - E_w[conv])
    # where s_j = sum_k w_k * a_{k-j}
    E_conv = np.dot(w, conv)

    # Compute s_j = sum_k w[k] * a[k-j] using FFT
    # s_j = (w * a_flip)[n-1+j] where a_flip = a[::-1]
    a_flip = a[::-1]
    conv_size = 1
    while conv_size < (2 * n - 1) + n - 1:
        conv_size <<= 1
    W = np.fft.rfft(w, conv_size)
    AF = np.fft.rfft(a_flip, conv_size)
    conv_w_af = np.fft.irfft(W * AF, conv_size)
    s = conv_w_af[n - 1:2 * n - 1]

    grad = 2.0 * a * (s - E_conv)
    obj = 2.0 * n * S
    return obj, grad


def true_obj(a):
    """True (non-smooth) objective: 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def optimize_anneal(a0, t_schedule, max_iter_per_stage, time_limit, seed=0):
    """Run L-BFGS with temperature annealing. Returns best sequence found."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    a = a / a.sum()
    x = np.log(a)

    best_a = a.copy()
    best_true = true_obj(a)
    t0 = time.monotonic()

    for t in t_schedule:
        if time.monotonic() - t0 > time_limit - 2:
            break

        def fun(xv):
            o, _ = smooth_obj_grad(xv, t)
            return o

        def gfun(xv):
            _, g = smooth_obj_grad(xv, t)
            return g

        x_current = np.log(np.maximum(softmax(x), 1e-30))

        try:
            res = minimize(fun, x_current, jac=gfun, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_stage, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxls': 50})
        except Exception:
            break

        if res.fun < fun(x_current):
            x = res.x.copy()

        a_new = softmax(x)
        a_new = a_new / a_new.sum()
        t_obj = true_obj(a_new)
        if t_obj < best_true - 1e-10:
            best_true = t_obj
            best_a = a_new.copy()

        # Also check the smooth objective's implied max
        conv = fast_convolve(a_new)
        scaled = t * conv
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()
        S = (np.log(exp_s.sum()) + m) / t
        smooth_obj = 2.0 * n * S

    return best_a, best_true


def coordinate_polish(a, time_limit, seed=0):
    """Try small coordinate perturbations to reduce the max of convolution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.copy()
    best_a = a.copy()
    best_obj = true_obj(a)
    t0 = time.monotonic()
    step_size = 0.05

    while time.monotonic() - t0 < time_limit:
        # Find peak position
        conv = fast_convolve(a)
        peak_val = conv.max()

        # Try random coordinate pairs
        for _ in range(200):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try moving mass from i to j or vice versa
            delta = step_size * a[i] * rng.uniform(0.1, 1.0)
            a_trial = a.copy()
            a_trial[i] -= delta
            a_trial[j] += delta
            if a_trial[i] < 0:
                a_trial[i] = 0
                a_trial[j] += a[i] * 0.1  # just nudge

            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s

            obj = true_obj(a_trial)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial.copy()
                step_size = max(0.01, step_size * 0.95)
            else:
                step_size = max(0.01, step_size * 0.99)

    return best_a, best_obj


def make_initializations(n, parent, rng):
    """Create diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # U-shaped (bimodal at edges)
    u = np.ones(n)
    edge = max(1, n // 4)
    u[:edge] = 2.0
    u[-edge:] = 2.0
    u /= u.sum()
    inits['u_shape'] = u

    # Strong U-shape
    su = np.exp(-0.5 * ((x - n * 0.5) / (0.3 * n)) ** 2)
    su = 1.0 / (su + 0.1)
    su /= su.sum()
    inits['strong_u'] = su

    # Cosine
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    c /= c.sum()
    inits['cosine'] = c

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r

    # Random Dirichlet
    for alpha in [0.3, 1.0, 3.0]:
        inits[f'dirichlet_{alpha}'] = rng.dirichlet(np.full(n, alpha))

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            xx = np.linspace(0, 1, n)
            pp = np.interp(xx, np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    # "Anti-peak" - try to make convolution flat by design
    # Use a periodic-like pattern
    period = max(2, n // 8)
    anti = np.ones(n)
    anti[::period] = 0.3
    anti /= anti.sum()
    inits['periodic'] = anti

    return inits

        score = float(evaluate_sequence(seq.tolist() if isinstance(seq, np.ndarray) else seq))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy() if isinstance(seq, np.ndarray) else np.array(seq)
            print(f"  NEW BEST: score={score:.8f}, n={len(best_seq)}")
        return score

    # Phase 1: Quick scan over n values and initializations
    print("=== Phase 1: Quick Scan ===")
    n_values = [128, 192, 256, 384, 512, 768, 1024]
    if parent is not None:
        pn = len(parent)
        if pn not in n_values:
            n_values.insert(0, pn)

    scan_time_total = min(400.0, (deadline - time.monotonic()) * 0.3)
    configs = []

    for n in n_values:
        inits = make_initializations(n, parent, rng)
        for name, a0 in inits.items():
            configs.append((n, name, a0))

    n_configs = len(configs)
    time_per_scan = max(2.0, scan_time_total / max(1, n_configs))

    scan_results = []
    for i, (n, name, a0) in enumerate(configs):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 60)
        if t_alloc < 1.5:
            break

        # Quick optimization: t=15, 50 iters
        a_opt, obj_opt = optimize_anneal(a0, t_schedule=[15.0],
                                         max_iter_per_stage=60,
                                         time_limit=max(0.5, t_alloc - 0.2),
                                         seed=seed + i)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()

    print(f"Phase 1 done. Best: {best_score:.8f}")

    # Phase 2: Deep optimization on top configs
    print("=== Phase 2: Deep Optimization ===")
    scan_results.sort(key=lambda x: x[0])

    # Select top configs ensuring diversity in n
    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen_n.add(n)
        elif n not in seen_n and len(top_configs) < 14:
            top_configs.append((n, name, a_opt))
            seen_n.add(n)

    # Always include current best
    if best_seq is not None:
        top_configs.append((len(best_seq), 'best', best_seq.copy()))

    remaining = deadline - time.monotonic() - 60
    time_per_deep = max(10.0, remaining / max(1, len(top_configs)))

    t_schedules = [
        [10.0, 25.0, 50.0, 100.0, 200.0],
        [20.0, 50.0, 100.0, 200.0, 400.0],
        [30.0, 80.0, 150.0, 300.0, 600.0],
    ]

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 40:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 40)
        if t_alloc < 5:
            break

        a_current = a0.copy()
        for t_sched in t_schedules:
            if time.monotonic() > deadline - 40:
                break
            t_single = t_alloc / len(t_schedules)
            a_opt, obj_opt = optimize_anneal(a_current, t_schedule=t_sched,
                                             max_iter_per_stage=150,
                                             time_limit=max(1.0, t_single - 0.2),
                                             seed=seed + 1000 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Deep: n={n}, {name}, sched={t_sched[0]}+, score={score:.8f}")
            a_current = a_opt

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        n_best = len(best_seq)

        # Multiple polish attempts with different t
        polish_schedules = [
            [50.0, 100.0, 200.0, 500.0, 1000.0],
            [100.0, 200.0, 500.0, 1000.0, 2000.0],
            [200.0, 500.0, 1000.0, 2000.0, 5000.0],
        ]

        a_polish = best_seq.copy()
        for sched in polish_schedules:
            if time.monotonic() > deadline - 20:
                break
            t_single = remaining / (len(polish_schedules) + 1)
            a_opt, obj_opt = optimize_anneal(a_polish, t_schedule=sched,
                                             max_iter_per_stage=300,
                                             time_limit=max(1.0, t_single - 0.2),
                                             seed=seed + 5000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                a_polish = a_opt.copy()
                print(f"  Polish: {sched[0]}+, score={score:.8f}")

        # Coordinate polish
        if time.monotonic() < deadline - 10:
            t_coord = min(30.0, deadline - 10 - time.monotonic())
            if t_coord > 2:
                a_coord, obj_coord = coordinate_polish(best_seq, time_limit=t_coord,
                                                       seed=seed + 9999)
                score = float(evaluate_sequence(a_coord.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_coord.copy()
                    print(f"  Coord polish: score={score:.8f}")

        # Random restart around best
        if time.monotonic() < deadline - 15:
            n_r = len(best_seq)
            for attempt in range(3):
                if time.monotonic() > deadline - 15:
                    break
                noise_level = [0.01, 0.005, 0.002][attempt]
                a_noisy = best_seq + rng.normal(0, noise_level, n_r)
                a_noisy = np.maximum(a_noisy, 0)
                a_noisy /= a_noisy.sum()
                t_single = (deadline - 15 - time.monotonic()) / (3 - attempt)
                a_opt, _ = optimize_anneal(a_noisy, t_schedule=[100.0, 300.0, 1000.0],
                                           max_iter_per_stage=200,
                                           time_limit=max(1.0, t_single - 0.2),
                                           seed=seed + 7777 + attempt)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  Restart {attempt}: score={score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"=== FINAL SCORE: {best_score:.8f}, n={len(best_seq)} ===")

    # Ensure all values are valid (non-negative, finite, not too large)
    best_seq = np.clip(best_seq, 0, 1000.0)
    return [float(x) for x in best_seq]
