```python
def lp_step(f, M):
    """Find g >= 0 maximizing sum(g) subject to (f*g)[k] <= M for all k."""
    n = len(f)
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # This is a linear function of g for fixed f.
    # We want to maximize sum(g) subject to these constraints and g >= 0.
    
    # Using scipy.optimize.linprog
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k-j < n:
                A_ub[k, j] = f[j]
    b_ub = np.full(2*n-1, M)
    
    # g >= 0: bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        return g
    return None
```

```python
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    c = -np.ones(n)

    A_ub = np.zeros((2 * n - 1, n))
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with FISTA refinement. The LP (from the state-of-the-art paper) finds the
optimal feasible direction at each step, enabling monotone decrease of the
max auto-convolution. FISTA is used for fine-tuning between LP steps.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(n):
    """Build the convolution constraint matrix (2n-1 x n) such that
    A @ g[k] = (f*g)[k] where f is provided separately.
    Actually, the constraint is (f*g)[k] = sum_j f[j]*g[k-j],
    which is linear in g with coefficients f[j].
    So we need A[k,j] = f[j] if 0 <= k-j < n, else 0.
    This depends on f, so we build it each time.
    """
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    mask = (k - j >= 0) & (k - j < n)
    return mask  # Boolean mask, multiply by f later


def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build constraint matrix: A_ub[k, j] = f[j] if 0 <= k-j < n
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    mask = (k_idx - j_idx >= 0) & (k_idx - j_idx < n)
    A_ub = mask * f[None, :]
    b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and result.x is not None and result.x.max() > 1e-12:
        g = np.maximum(result.x, 0)
        g_sum = g.sum()
        if g_sum > 1e-12:
            return g / g_sum
    return None


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_step(a, conv, M, step_size):
    """
    One FISTA-like step: compute subgradient, take step, project.
    """
    n = len(a)
    # Active set: indices where conv is close to max
    tol = 1e-6
    active = np.where(conv >= M - tol * M)[0]
    
    # Subgradient: average of gradients at active peaks
    sg = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # grad_k[i] = 2*a[k-i] for valid i
        # i ranges from j_start to j_end, k-i ranges from k-j_start to k-j_end
        sg[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
    sg /= len(active)
    
    # Take step and project
    v = a - step_size * sg
    a_new = project_simplex(v)
    return a_new


def optimize_lp_fista(a0, time_limit, max_lp_calls=5000):
    """
    Combined LP + FISTA optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / a_sum
    
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M
    
    t0 = time.monotonic()
    lp_count = 0
    fista_eta = 0.01
    lp_stagnation = 0
    fista_stagnation = 0
    t_step = 0.5
    mode = 'lp'
    
    while time.monotonic() - t0 < time_limit:
        if mode == 'lp' and lp_count < max_lp_calls:
            g = solve_lp_direction(a, M)
            lp_count += 1
            if g is not None:
                found = False
                for t in [t_step, 0.4*t_step, 0.2*t_step, 0.1, 0.05, 0.01]:
                    h = (1-t)*a + t*g
                    h_conv = fast_convolve(h)
                    h_max = h_conv.max()
                    if h_max < M - 1e-14:
                        a = h
                        conv = h_conv
                        M = h_max
                        best_a = a.copy()
                        best_M = M
                        lp_stagnation = 0
                        found = True
                        break
                if not found:
                    lp_stagnation += 1
                    t_step = max(0.01, t_step * 0.7)
                    if lp_stagnation > 15:
                        mode = 'fista'
                        fista_stagnation = 0
                        fista_eta = 0.05
            else:
                mode = 'fista'
        else:
            # FISTA step (multiple small steps between LP calls)
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                a_new = fista_step(a, conv, M, fista_eta)
                new_conv = fast_convolve(a_new)
                new_M = new_conv.max()
                if new_M < M - 1e-14:
                    a = a_new
                    conv = new_conv
                    M = new_M
                    best_a = a.copy()
                    best_M = M
                    fista_stagnation = 0
                else:
                    fista_stagnation += 1
                    fista_eta *= 0.9

                lp_stagnation = 0
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
(the state-of-the-art active-set LP method) combined with projected subgradient
descent for fine-tuning and escape from stagnation. Uses multi-start with
diverse initializations and adaptive sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1)
    if not valid.any():
        return np.ones(n) / n
    rho = np.nonzero(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k=0..2n-2.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    c = -np.ones(n)

    # Build constraint matrix: A_ub[k, j] = f[j] if 0 <= k-j < n
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    mask = (k_idx - j_idx >= 0) & (k_idx - j_idx < n)
    A_ub = mask.astype(np.float64) * f[None, :]
    b_ub = np.full(2 * n - 1, M)

    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={"presolve": True})
    except Exception:
        return None

    if result.success and result.x is not None and np.isfinite(result.x).all():
        g = np.maximum(result.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-12:
            return g / g_sum
    return None


def compute_subgradient(a, conv, M, tol_frac=0.001):
    """Compute the (approximate) subgradient of max(conv) at a."""
    n = len(a)
    thresh = M - tol_frac * M
    active = np.where(conv >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    sg = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
    sg /= len(active)
    return sg


def optimize_lp_fista(a0, time_limit, max_lp_calls=20000, verbose=False):
    """
    Combined LP + subgradient optimization to minimize max auto-convolution.
    Returns (best_a_normalized, best_objective_2nM).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / a_sum

    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    lp_count = 0
    t_step = 0.5
    lp_stagnation = 0
    sg_eta = 0.05
    sg_stagnation = 0
    mode = 'lp'
    relax = 1.0  # relaxation factor for LP constraint

    while time.monotonic() - t0 < time_limit:
        if mode == 'lp' and lp_count < max_lp_calls:
            # Solve LP with current bound
            g = solve_lp_direction(a, M * relax)
            lp_count += 1

            if g is not None:
                # Check if g is meaningfully different from a
                g_diff = np.linalg.norm(g - a)
                if g_diff < 1e-8:
                    # At a fixed point, try relaxing
                    relax = min(relax * 1.02, 1.5)
                    if relax > 1.3:
                        mode = 'sg'
                        sg_stagnation = 0
                        sg_eta = 0.1
                    continue

                # Try step sizes
                found = False
                for t in [t_step, 0.5 * t_step, 0.25 * t_step, 0.1, 0.05, 0.01]:
                    h = (1 - t) * a + t * g
                    h_conv = fast_convolve(h)
                    h_max = h_conv.max()
                    if h_max < M - 1e-14:
                        a = h
                        conv = h_conv
                        M = h_max
                        best_a = a.copy()
                        best_M = M
                        lp_stagnation = 0
                        relax = 1.0
                        found = True
                        break

                if not found:
                    lp_stagnation += 1
                    t_step = max(0.01, t_step * 0.7)
                    relax = min(relax + 0.01, 1.5)
                    if lp_stagnation > 20:
                        mode = 'sg'
                        sg_stagnation = 0
                        sg_eta = 0.05
            else:
                mode = 'sg'
        else:
            # Subgradient phase: multiple small steps
            steps_done = 0
            while steps_done < 100 and time.monotonic() - t0 < time_limit:
                sg = compute_subgradient(a, conv, M, tol_frac=0.001)
                v = a - sg_eta * sg
                a_new = project_simplex(v)
                new_conv = fast_convolve(a_new)
                new_M = new_conv.max()

                if new_M < M - 1e-14:
                    a = a_new
                    conv = new_conv
                    M = new_M
                    best_a = a.copy()
                    best_M = M
                    sg_stagnation = 0
                    sg_eta = min(sg_eta * 1.02, 0.5)
                else:
                    sg_eta *= 0.85
                    sg_stagnation += 1

                steps_done += 1

            if sg_stagnation > 10:
                mode = 'lp'
                lp_stagnation = 0
                t_step = 0.3
                sg_stagnation = 0
                relax = 1.0

        if verbose and lp_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP+FISTA: obj={2*n*best_M:.8f}, M={M:.10f}, "
                  f"lp_calls={lp_count}, t={elapsed:.1f}s, mode={mode}")

    return best_a, 2.0 * n * best_M


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence length n."""
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussians with various widths
    for sf in [0.1, 0.15, 0.2, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Raised cosine (single bump)
    if n > 2:
        rc = 0.5 * (1 + np.cos(np.pi * (x - mid) / max(1, mid)))
        rc = np.maximum(rc, 1e-10)
        rc /= rc.sum()
        inits['cosine'] = rc

    # Bimodal (edges high)
    w = max(1, n // 4)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Asymmetric Gaussian (offset from center)
    for offset in [0.2, 0.3, 0.4]:
        ag = np.exp(-0.5 * ((x - n * offset) / (0.2 * n)) ** 2)
        ag /= ag.sum()
        inits[f'asym_{offset}'] = ag

    # Tukey-like (flat middle, tapered edges)
    if n > 4:
        r = n // 8
        tu = np.ones(n)
        for i in range(min(r, n // 2)):
            tu[i] = 0.5 * (1 + np.cos(np.pi * (i + 0.5) / r))
            tu[n - 1 - i] = tu[i]
        tu = np.maximum(tu, 1e-10)
        tu /= tu.sum()
        inits['tukey'] = tu

    # Random Dirichlet
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits
```
