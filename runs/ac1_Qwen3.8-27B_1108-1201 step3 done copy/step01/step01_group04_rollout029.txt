"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses DFT-accelerated gradient descent with softmax approximation,
adaptive parameters, multiple starting points of various lengths,
and local search for final polishing.
"""

import numpy as np
import time

def _next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n]

def _compute_conv(a):
    """Compute a * a (autocorrelation) using FFT."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)
    return conv[:2*n-1]

def _compute_obj(a):
    """Compute 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    conv = _compute_conv(a)
    return float(2.0 * n * conv.max() / (S * S))

def _normalize(a, target=1.0):
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def _grad_dft(a, beta):
    """
    Compute gradient of smooth max(conv) approximation using DFT.
    Returns (gradient, conv).
    """
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[:2*n-1]
    
    # Softmax weights over the convolution
    conv_max = conv.max()
    # Clamp to avoid overflow
    beta_safe = min(beta, 500.0)
    w = np.exp(beta_safe * (conv - conv_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2*n-1)
        w[conv.argmax()] = 1.0
    else:
        w = w / w_sum
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * cross_correlation(w, a)[j]
    # Cross-correlation via FFT: corr(w,a)[j] = ifft(conj(FFT(w)) * FFT(a))[j]
    L2 = _next_pow2(2 * n - 1 + n - 1)  # = next_pow2(3n-2)
    W = np.fft.rfft(w, L2)
    A2 = np.fft.rfft(a, L2)
    corr = np.fft.irfft(np.conj(W) * A2, L2)
    grad = 2.0 * corr[:n]
    
    return grad, conv

def _grad_step(a, lr, beta):
    """One projected gradient step."""
    grad, conv = _grad_dft(a, beta)
    # Project out mean for sum constraint
    grad = grad - grad.mean()
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, 1.0)
    return a_new

Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses DFT-accelerated gradient descent with softmax approximation, spectrum flattening,
adaptive parameters, momentum, multiple starting points of various lengths,
and local search for final polishing.

import numpy as np
import time


def _next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _compute_conv(a):
    """Compute a * a (linear autocorrelation) using FFT."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)
    return conv[:2 * n - 1]


def _compute_obj(a):
    """Compute 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return float('inf')
    conv = _compute_conv(a)
    return float(2.0 * n * float(conv.max()) / (S * S))

    """Normalize to have a specific sum."""
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[:2 * n - 1]

    # Softmax weights
    conv_max = float(conv.max())
    beta_safe = min(beta, 1000.0)
    w = np.exp(beta_safe * (conv - conv_max))
    w_sum = float(w.sum())
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(conv.argmax())] = 1.0
    else:
        w = w / w_sum

    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] for valid indices
    # = 2 * (convolve(w_rev, a))[2n-2-j]
    L2 = _next_pow2(3 * n - 2)
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, L2)
    A2 = np.fft.rfft(a, L2)
    c = np.fft.irfft(W_rev * A2, L2)
    grad = 2.0 * c[2 * n - 2 : n - 2 : -1]

    return grad, conv


def _grad_step(a, lr, beta, momentum=0.0, velocity=None):
    """One projected gradient step with optional momentum."""
    grad, conv = _grad_dft(a, beta)
    # Project out mean for sum constraint
    grad = grad - np.mean(grad)

    if momentum > 0.0:
        if velocity is None:
            velocity = np.zeros_like(a)
        velocity = momentum * velocity + (1.0 - momentum) * grad
        eff_grad = velocity
    else:
        eff_grad = grad
        velocity = None

    a_new = a - lr * eff_grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, 1.0)
    return a_new, velocity


def _flatten_spectrum(a, alpha=0.7, num_iters=200):
    """Flatten power spectrum to reduce peak autocorrelation (Gilbert-like)."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    a = a.copy()

    for _ in range(num_iters):
        A = np.fft.rfft(a, L)
        P = np.abs(A) ** 2
        P_max = float(P.max())
        if P_max < 1e-30:
            break
        # Flatten: reduce dynamic range
        P_new = P_max * np.power(P / P_max, alpha)
        mag_new = np.sqrt(P_new)
        # Maintain phase
        phase_A = A / (np.abs(A) + 1e-30)
        A_new = mag_new * phase_A
        a = np.fft.irfft(A_new, L)[:n]
        # Project to non-negative
        a = np.maximum(a, 0.0)
        a = _normalize(a, 1.0)

    return a


def _optimize(a0, deadline, max_iters=100000, beta_start=3.0, beta_end=100.0,
              lr_start=0.08, lr_min=0.001, momentum=0.5, check_interval=100):
    """
    Main optimization loop using DFT-based gradient descent with adaptive parameters.
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None)
    a = _normalize(a, 1.0)

    best = a.copy()
    best_score = _compute_obj(a)

    beta = beta_start
    velocity = None
    patience = 0
    patience_limit = max(2000, check_interval * 20)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal beta (increase to focus on peak)
        progress = min(1.0, float(it) / float(max(1, int(max_iters * 0.8))))
        beta = beta_start + (beta_end - beta_start) * progress

        # Learning rate: cosine schedule
        frac = min(1.0, float(it) / float(max_iters))
        lr = lr_start * 0.5 * (1.0 + np.cos(np.pi * min(frac * 1.2, 1.0)))
        lr = max(lr, lr_min)

        a, velocity = _grad_step(a, lr, beta, momentum=momentum, velocity=velocity)

        # Periodic evaluation
        if (it + 1) % check_interval == 0:
            score = _compute_obj(a)
            if score < best_score:
                best_score = score
                best = a.copy()
                patience = 0
            else:
                patience += 1

            if patience >= patience_limit:
                # Perturbation to escape local minimum
                a = best + 0.02 * np.random.randn(n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, 1.0)
                velocity = None
                patience = 0

    # Final check
    score = _compute_obj(a)
    if score < best_score:
        best_score = score
        best = a.copy()

    return best, best_score


def _optimize_two_phase(a0, deadline, max_phase1=50000, max_phase2=500000):
    """Two-phase optimization: broad exploration then fine-tuning."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None)
    a = _normalize(a, 1.0)

    # Phase 1: Broad exploration (lower beta, higher lr)
    t_mid = time.monotonic() + 0.3 * (deadline - time.monotonic())
    a, score1 = _optimize(a, t_mid, max_iters=max_phase1,
                          beta_start=2.0, beta_end=30.0,
                          lr_start=0.15, lr_min=0.01, momentum=0.3,
                          check_interval=50)

    # Phase 2: Fine-tuning (higher beta, lower lr)
    a, score2 = _optimize(a, deadline, max_iters=max_phase2,
                          beta_start=30.0, beta_end=300.0,
                          lr_start=0.03, lr_min=0.0005, momentum=0.7,
                          check_interval=200)

    return a, score2


def _local_search(a, deadline, num_trials=20000, seed=42):
    """Local search with pairwise transfers and small perturbations."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a = _normalize(a.copy(), 1.0)
    best = a.copy()
    best_score = _compute_obj(a)

    step_size = 0.03
    no_improve = 0
    max_no_improve = 2000

    for it in range(num_trials):
        if time.monotonic() > deadline:
            break

        # Random pair of indices for transfer
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        delta = step_size * rng.random() * (1.0 if rng.random() > 0.4 else -1.0)

        a_new = a.copy()
        a_new[i] = max(0.0, a_new[i] + delta)
        a_new[j] = max(0.0, a_new[j] - delta)
        a_new = _normalize(a_new, 1.0)

        score = _compute_obj(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay step size and check for stagnation
        step_size *= 0.9995
        if no_improve > max_no_improve:
            break

    return best, best_score


def _spectral_refine(a, deadline, alpha=0.5, max_iters=500):
    """Spectrum flattening as a refinement step."""
    n = len(a)
    best = a.copy()
    best_score = _compute_obj(a)
    L = _next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, L)
        P = np.abs(A) ** 2
        P_max = float(P.max())
        if P_max < 1e-30:
            break

        # Adaptive alpha: reduce dynamic range
        P_new = P_max * np.power(P / P_max, alpha)
        mag_new = np.sqrt(P_new)
        phase_A = A / (np.abs(A) + 1e-30)
        A_new = mag_new * phase_A
        a_candidate = np.fft.irfft(A_new, L)[:n]
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _normalize(a_candidate, 1.0)

        score = _compute_obj(a_candidate)
        if score < best_score:
            best_score = score
            best = a_candidate.copy()
            a = a_candidate.copy()
        else:
            # Blend
            blend = 0.5 * a + 0.5 * a_candidate
            blend = np.maximum(blend, 0.0)
            blend = _normalize(blend, 1.0)
            score_blend = _compute_obj(blend)
            if score_blend < best_score:
                best_score = score_blend
                best = blend.copy()
                a = blend.copy()
            else:
                break

    return best, best_score


def _generate_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []

    # Try to use provided global
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            starts.append(('warm_full', gs.copy(), len(gs)))
            for L in [64, 128, 256, 512]:
                if L < len(gs):
                    starts.append(('warm_trunc', gs[:L].copy(), L))
            # Interpolated versions
            for L in [1024, 2048]:
                idx = np.arange(L, dtype=np.float64)
                gi = np.arange(len(gs), dtype=np.float64)
                interp = np.interp(idx, gi, gs)
                starts.append(('warm_interp', interp, L))
    except (NameError, Exception):
        pass

    for n in lengths:
        i = np.arange(n, dtype=np.float64)

        # Cosine window
        starts.append(('cos', 0.5 + 0.5 * np.cos(2.0 * np.pi * i / n), n))

        # Gaussian (varying widths)
        center = (n - 1) / 2.0
        for sf in [0.15, 0.25, 0.4]:
            g = np.exp(-0.5 * ((i - center) / (sf * n)) ** 2)
            starts.append(('gauss', g, n))

        # Chirp
        for f in [2, 3, 5]:
            c = 0.5 + 0.5 * np.cos(2.0 * np.pi * f * i * i / (2.0 * n))
            starts.append(('chirp', c, n))

        # Random
        starts.append(('rand', rng.random(n) + 0.1, n))

        # Sinc-like (via windowed rectangular)
        t = i / (n - 1) * 2.0 - 1.0
        s = np.abs(np.sinc(t)) + 0.01
        starts.append(('sinc', s, n))

        # Power law
        p = np.power(i + 1.0, -0.3)
        starts.append(('power', p, n))

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple strategies and starting points
    to find the best sequence minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(1.0, float(budget_s))
    deadline = t_start + total_budget - 15.0  # Reserve 15s for final eval

    best_seq = None
    best_score = float('inf')

    def _try_update(seq, score, label=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            elapsed = time.monotonic() - t_start
            print(f"  {label} NEW BEST: {best_score:.6f} (n={len(seq)}, t={elapsed:.0f}s)")
        return best_seq, best_score

    print(f"=== Starting optimization. Budget: {budget_s}s ===")
    print(f"Deadline: {deadline - t_start:.0f}s from start")

    # ========== PHASE 1: Quick screening of shorter sequences ==========
    print("\n--- Phase 1: Quick screening ---")
    phase1_end = t_start + min(250.0, total_budget * 0.15)

    # Quick optimize the warm start if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8:
            t0 = time.monotonic()
            seq, score = _optimize(gs, phase1_end, max_iters=30000,
                                   beta_start=3.0, beta_end=50.0,
                                   lr_start=0.1, momentum=0.5)
            _try_update(seq, score, "Phase1-warm")
            print(f"  Warm start (n={len(gs)}): {score:.6f} [{time.monotonic()-t0:.1f}s]")
    except (NameError, Exception):
        pass

    # Try a few shorter structured sequences
    for n in [128, 256, 512]:
        if time.monotonic() > phase1_end:
            break
        i = np.arange(n, dtype=np.float64)
        candidates = [
            0.5 + 0.5 * np.cos(2.0 * np.pi * i / n),
            np.exp(-0.5 * ((i - n / 2) / (0.3 * n)) ** 2),
            rng.random(n) + 0.1,
        ]
        for ci, a0 in enumerate(candidates):
            if time.monotonic() > phase1_end:
                break
            seq, score = _optimize(a0, phase1_end, max_iters=5000,
                                   beta_start=3.0, beta_end=40.0,
                                   lr_start=0.12, momentum=0.4)
            _try_update(seq, score, f"P1 n={n} c{ci}")

    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # ========== PHASE 2: Thorough optimization ==========
    print("\n--- Phase 2: Thorough optimization ---")
    phase2_end = t_start + total_budget * 0.50

    if best_seq is not None:
        # Continue optimizing the best found
        seq, score = _optimize_two_phase(best_seq, phase2_end,
                                         max_phase1=20000, max_phase2=100000)
        _try_update(seq, score, "Phase2-best")

        # Spectrum flattening
        if time.monotonic() < phase2_end:
            seq, score = _spectral_refine(best_seq, phase2_end, alpha=0.6, max_iters=200)
            _try_update(seq, score, "Phase2-spectral")

            # Gradient descent after spectral
            if time.monotonic() < phase2_end:
                seq, score = _optimize(best_seq, phase2_end, max_iters=50000,
                                       beta_start=20.0, beta_end=200.0,
                                       lr_start=0.05, lr_min=0.001, momentum=0.7)
                _try_update(seq, score, "Phase2-grad2")
    else:
        # No good start yet, try longer
        n = 512
        i = np.arange(n, dtype=np.float64)
        a0 = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / n)
        seq, score = _optimize_two_phase(a0, phase2_end,
                                         max_phase1=30000, max_phase2=50000)
        _try_update(seq, score, "Phase2-cos512")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # ========== PHASE 3: Longer sequences ==========
    print("\n--- Phase 3: Longer sequences ---")
    phase3_end = t_start + total_budget * 0.78

    for n in [1024, 2048, 4096]:
        if time.monotonic() > phase3_end:
            break

        i = np.arange(n, dtype=np.float64)
        candidates = []

        # Cosine
        candidates.append(('cos', 0.5 + 0.5 * np.cos(2.0 * np.pi * i / n)))

        # Gaussian
        center = (n - 1) / 2.0
        candidates.append(('gauss', np.exp(-0.5 * ((i - center) / (0.25 * n)) ** 2)))

        # Chirp
        candidates.append(('chirp', 0.5 + 0.5 * np.cos(2.0 * np.pi * 3.0 * i * i / (2.0 * n))))

        # Interpolated best
        if best_seq is not None:
            gi = np.linspace(0, n - 1, len(best_seq))
            interp = np.interp(i, gi, best_seq)
            candidates.append(('interp_best', interp))

        # Random
        candidates.append(('rand', rng.random(n) + 0.05))

        for cname, a0 in candidates:
            if time.monotonic() > phase3_end:
                break
            remaining = phase3_end - time.monotonic()
            if remaining < 10:
                break
            max_it = max(2000, int(remaining * 30))
            seq, score = _optimize(a0, phase3_end, max_iters=max_it,
                                   beta_start=3.0, beta_end=60.0,
                                   lr_start=0.08, lr_min=0.002, momentum=0.5)
            _try_update(seq, score, f"P3 n={n} {cname}")

    print(f"  Phase 3 done. Best: {best_score:.6f}")

    # ========== PHASE 4: Final polishing ==========
    print("\n--- Phase 4: Final polishing ---")
    if best_seq is not None:
        polish_deadline = deadline - 5.0

        # Fine gradient with high beta
        if time.monotonic() < polish_deadline - 30:
            seq, score = _optimize(best_seq, polish_deadline - 30,
                                   max_iters=300000,
                                   beta_start=100.0, beta_end=500.0,
                                   lr_start=0.02, lr_min=0.0003,
                                   momentum=0.8, check_interval=500)
            _try_update(seq, score, "P4-fine")

        # Spectrum refinement
        if time.monotonic() < polish_deadline - 20:
            seq, score = _spectral_refine(best_seq, polish_deadline - 20,
                                         alpha=0.8, max_iters=100)
            _try_update(seq, score, "P4-spec")

        # Another gradient pass
        if time.monotonic() < polish_deadline - 10:
            seq, score = _optimize(best_seq, polish_deadline - 10,
                                   max_iters=100000,
                                   beta_start=200.0, beta_end=800.0,
                                   lr_start=0.01, lr_min=0.0001,
                                   momentum=0.9, check_interval=500)
            _try_update(seq, score, "P4-grad3")

        # Local search
        if time.monotonic() < polish_deadline - 5:
            seq, score = _local_search(best_seq, polish_deadline - 5,
                                       num_trials=5000)
            _try_update(seq, score, "P4-local")

    # ========== Final validation ==========
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

        # Safety: ensure all values are valid
        result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
        if sum(result) < 0.01:
            result = [1.0] * 256
        return result
    else:
        print("No valid sequence found, using fallback.")
        return [1.0] * 256
