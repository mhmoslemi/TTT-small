```python
"""
Optimized search for non-negative step functions minimizing the normalized
peak of linear self-convolution. Uses gradient-based optimization with
subgradient/soft-max on softmax-parameterized sequences, multiple lengths,
structured initializations, and local polish via targeted peak reduction.
Key insight: score = max(conv(a,a)) when a is normalized to sum=sqrt(2n).
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    s = np.sum(e)
    return e / s


def score_and_subgrad(a):
    """
    Compute score and hard subgradient for a (assumed sum = sqrt(2n)).
    score = max(conv(a,a)) [since 2n*max_c/S^2 = max_c when S^2=2n]
    subgrad[j] = a[j] * (a[p-j] - max_c) where p=argmax(conv)
    """
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    p = int(np.argmax(c))
    S = np.sum(a)
    score = 2.0 * n * max_c / (S * S)
    # Subgradient of max_c w.r.t. a:
    g_a = np.zeros(n)
    # d(c[p])/da[i] = a[p-i] if 0 <= p-i < n
    lo = max(0, p - n + 1)
    hi = min(n, p + 1)
    for i in range(lo, hi):
        g_a[i] = a[p - i]
    # d(score)/da[i] = 2n/S^2 * g_a[i] - 4n*max_c/S^3 * 1
    d_score_da = 2.0 * n / (S * S) * g_a - 4.0 * n * max_c / (S ** 3)
    # d(score)/d(theta[j]) = sum_i d_score_da[i] * d(a[i])/d(theta[j])
    # d(a[i])/d(theta[j]) = S*a[i]*(delta(i,j) - a[i])  [a = softmax(theta)*S]
    # = S * (d_score_da[j]*a[j] - a[j]*sum_i d_score_da[i]*a[i])
    inner = np.dot(d_score_da, a)
    d_score_dtheta = S * a * (d_score_da - inner / S)
    # Simplify: S * a * (d_score_da - inner/S) = S*a*d_score_da - a*inner
    # = S*a*(2n/S^2 * g_a - 4n*max_c/S^3) - a*inner
    # The constant part (-4n*max_c/S^3) sums to 0 when dotted with a (since sum a = S)
    # So d_score_dtheta = S*a*(2n/S^2 * g_a) - a*inner
    # Let me just use the formula directly
    d_score_dtheta = S * a * (d_score_da - inner / S)
    return score, d_score_dtheta, max_c, p, c


def score_and_softgrad(a, tau):
    """
    Compute score and smooth gradient using soft-max for the max function.
    """
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    soft_max_c = c_max + tau * np.log(sum_exp)
    score = 2.0 * n * soft_max_c / (S * S)

    # d(soft_max_c)/da[i] = sum_k w[k] * a[k-i] (valid k-i)
    # = cross-correlation of w and a at lag i
    # Computed via: np.correlate(w, a, 'full')[n-1 : 2n-1]
    # But for efficiency with FFT:
    # Use: g_a[i] = sum_m a[m]*w[i+m]
    # This is: ifft(fft(a_pad) * conj(fft(w_pad))) with proper indexing
    # Simpler: use the convolution trick
    # g_a[i] = sum_m a[m]*w[i+m] = (a * w_flipped)[(2n-2)-i] where w_flipped = w[::-1]
    total_len = n + len(c) - 1  # = 3n-2
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w[::-1], size)
    conv_awf = np.fft.irfft(fa * fw, size)
    # g_a[i] = conv_awf[2n-2-i]
    g_a = conv_awf[2 * n - 2: 2 * n - 2 + n][::-1]

    d_score_da = 2.0 * n / (S * S) * g_a - 4.0 * n * soft_max_c / (S ** 3)
    inner = np.dot(d_score_da, a)
    d_score_dtheta = S * a * (d_score_da - inner / S)
    return score, d_score_dtheta, soft_max_c


def optimize_sequence(n, a0, time_limit, rng, use_soft_start=True):
    """
    Optimize a single sequence using Adam on theta with annealed temperature.
    """
    S = np.sqrt(2.0 * n)
    a0 = np.clip(a0, 1e-6, None)
    a0_norm = a0 / np.sum(a0)
    theta = np.log(a0_norm + 1e-10)
    theta_best = theta.copy()

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_score = np.inf
    t0 = time.monotonic()

    # Temperature schedule
    tau_start = 5.0 / n
    tau_end = 0.0  # 0 means switch to hard subgradient
    tau_switch_frac = 0.3  # switch to hard at 30% of time

    # LR schedule
    lr_init = 1.0
    lr_decay = 0.9997

    a = softmax_normalize(theta) * S
    current_score = local_score(a)
    best_score = current_score
    best_theta = theta.copy()
    lr = lr_init
    patience = 0
    patience_limit = 3000

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit

        a = softmax_normalize(theta) * S

        # Compute gradient
        if use_soft_start and elapsed_frac < tau_switch_frac:
            tau = tau_start * (1.0 - elapsed_frac / tau_switch_frac)
            if tau < 0.01 / n:
                tau = 0.01 / n
            score, grad, _ = score_and_softgrad(a, tau)
        else:
            score, grad, max_c, p, c = score_and_subgrad(a)

        # Adam update
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * grad
        momentum2 = beta2 * momentum2 + (1 - beta2) * (grad * grad)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        lr *= lr_decay

        # Check score
        a_check = softmax_normalize(theta) * S
        check_score = local_score(a_check)

        if check_score < best_score:
            best_score = check_score
            best_theta = theta.copy()
            patience = 0
        else:
            patience += 1

        # Restart if stuck
        if patience > patience_limit:
            # Perturb theta
            theta_best_perturbed = best_theta + rng.normal(0, 0.1, size=n)
            a_pert = softmax_normalize(theta_best_perturbed) * S
            pert_score = local_score(a_pert)
            if pert_score < best_score + 0.01:
                theta = theta_best_perturbed
            else:
                theta = best_theta + rng.normal(0, 0.2, size=n)
            momentum1 = np.zeros(n)
            momentum2 = np.zeros(n)
            lr = lr_init * 0.5
            patience = 0

    a_best = softmax_normalize(best_theta) * S
    return a_best, local_score(a_best)


def local_score(a):
    """Fast local score computation (same formula as evaluate_sequence)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def targeted_polish(a_init, time_limit, rng):
    """
    Local polish: identify peak contributors and make targeted adjustments.
    Uses a combination of peak reduction and redistribution.
    """
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    S = np.sqrt(2.0 * n)

    temperature = 0.002
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.4:
            # Targeted: reduce peak contributors, boost others
            c = fft_convolve(current, current)
            max_c = np.max(c)
            p = int(np.argmax(c))
            # Find contributors to the peak
            contribs = np.zeros(n)
            lo = max(0, p - n + 1)
            hi = min(n, p + 1)
            for i in range(lo, hi):
                contribs[i] = current[i] * current[p - i]
            # Reduce top contributors
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            # Boost bottom contributors
            bottom_idx = np.argsort(contribs)[:top_k]
            candidate[bottom_idx] *= np.exp(rng.normal(0.03, 0.01, size=len(bottom_idx)))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.75:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.85:
            # Second peak targeting
            c = fft_convolve(current, current)
            # Find top 3 peaks
            top3 = np.argsort(c)[-3:]
            for p in top3:
                lo = max(0, p - n + 1)
                hi = min(n, p + 1)
                idx = rng.choice(range(lo, hi), size=min(3, hi - lo), replace=False)
                candidate[idx] *= np.exp(rng.normal(-0.02, 0.008, size=len(idx)))
            # Boost random other positions
            others = np.setdiff1d(np.arange(n), top3)
            if len(others) > 0:
                inc_idx = rng.choice(others, size=min(5, len(others)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=len(inc_idx)))
        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 1e-6, 1000.0)
        # Renormalize to keep score comparison fair
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_anti_sym_init(n, rng):
    """Create an anti-symmetric-like initial sequence (high at edges, low center)."""
    t = np.arange(n, dtype=np.float64) / n
    # V-shape: high at edges, low at center
    a = 1.0 + 0.3 * (2.0 * t - 1.0) ** 2
    a += 0.05 * np.sin(4.0 * np.pi * t)  # add some ripple
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k in range(1, 8):
        amp = 0.15 / k
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_pseudorandom_init(n, rng):
    """Create a sequence based on a pseudo-random binary pattern (smoothed)."""
    # Use a simple LFSR-like pattern
    bits = (rng.integers(0, 2, size=n) * 2 - 1).astype(np.float64)
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(bits, kernel, mode='same')
    a = 1.0 + 0.3 * a
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: (1) gradient-based optimization over multiple
    initializations and sequence lengths, (2) targeted local polish.
    Returns best sequence found.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 20.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        if len(start_seq) > 0:
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Phase 1: Gradient optimization on multiple (length, init) combinations
    print("\n=== Phase 1: Gradient optimization ===")

    # Define search configurations
    configs = []
    if start_seq is not None:
        configs.append((len(start_seq), 'provided', 60))

    # Various lengths and initializations
    lengths_inits = [
        (256, 'chirp'), (256, 'anti_sym'), (256, 'multi_tone'),
        (384, 'chirp'), (384, 'anti_sym'),
        (512, 'chirp'), (512, 'anti_sym'), (512, 'multi_tone'),
        (768, 'chirp'), (768, 'anti_sym'),
        (1024, 'chirp'), (1024, 'anti_sym'),
        (2048, 'chirp'), (2048, 'anti_sym'),
    ]

    # Allocate time proportionally - give more time to medium lengths
    for n, init_type in lengths_inits:
        if time.monotonic() > deadline - 30:
            break
        time_left = deadline - time.monotonic()
        # Allocate time based on remaining configs
        remaining_configs = [c for c in lengths_inits if True]  # simplified
        time_per = min(45.0, time_left / 8.0)

        # Generate initial sequence
        if init_type == 'provided' and start_seq is not None:
            a0 = start_seq.copy()
        elif init_type == 'chirp':
            alpha = 0.3 + 0.3 * rng.random()
            a0 = make_chirp_init(n, alpha, rng)
        elif init_type == 'anti_sym':
            a0 = make_anti_sym_init(n, rng)
        elif init_type == 'pseudorandom':
            a0 = make_pseudorandom_init(n, rng)
        else:
            a0 = make_multi_tone_init(n, rng)

        a0 = np.clip(a0, 0.01, 1000.0)

        t0 = time.monotonic()
        a_best, grad_score = optimize_sequence(n, a0, time_per, rng)
        elapsed = time.monotonic() - t0
        actual_score = local_score(a_best)
        print(f"  n={n}, init='{init_type}', score={actual_score:.6f}, time={elapsed:.1f}s")

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Deeper optimization on the best candidate
    print(f"\n=== Phase 2: Deeper optimization ===")
    if best_seq is not None:
        n_best = len(best_seq)
        polish_time = min(400.0, max(30.0, deadline - time.monotonic() - 100.0))
        print(f"  Deep optimizing n={n_best}, score={best_score:.6f}, time={polish_time:.0f}s")
        a_deep, deep_score = optimize_sequence(n_best, best_seq, polish_time, rng)
        if deep_score < best_score:
            best_score = deep_score
            best_seq = a_deep.copy()
            print(f"    *** New best after deep: {best_score:.6f} ***")

    # Phase 3: Targeted polish
    print(f"\n=== Phase 3: Targeted polish ===")
    if best_seq is not None:
        polish_time = min(200.0, max(20.0, deadline - time.monotonic() - 10.0))
        print(f"  Polishing, score={best_score:.6f}, time={polish_time:.0f}s")
        polished, polish_score = targeted_polish(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"    *** New best after polish: {best_score:.6f} ***")

    # Phase 4: Try a couple more creative initializations if time permits
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Extra exploration ===")
        remaining = deadline - time.monotonic()
        extra_attempts = [
            (512, 'pseudorandom'),
            (1024, 'pseudorandom'),
            (768, 'multi_tone'),
        ]
        for n, init_type in extra_attempts:
            if time.monotonic() > deadline - 30:
                break
            if init_type == 'pseudorandom':
                a0 = make_pseudorandom_init(n, rng)
            else:
                a0 = make_multi_tone_init(n, rng)
            t_per = min(50.0, remaining / 3.0)
            a_best, s = optimize_sequence(n, a0, t_per, rng)
            print(f"  extra n={n}, init='{init_type}', score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Final polish on the very best
    if best_seq is not None and time.monotonic() < deadline - 15:
        final_polish_time = max(10.0, deadline - time.monotonic() - 10.0)
        polished, polish_score = targeted_polish(best_seq, final_polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"    *** Final polish improved to: {best_score:.6f} ***")

    # Final: verify and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
