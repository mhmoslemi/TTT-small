    best_candidate = None

    # For the given search space (even at larger n), we can considerlazy evaluation vs explicit.
    for meth_in in initializations:

            # Use a dense array of initial size

                best_score = score_sub
                best_candidate = f.copy().tolist()
        except:
            continue
    
    if best_candidate is not None:
        # Final fallback if not found but candidate available
        final_candidate = np.at least_reshape(best_candidate or fallback) if best_candidate else fallback
    else:
        # No optimization done - the fallback is provided
        final_candidate = fallback

        s_total = np.sum(final_candidate)

    except:
        pass
    
    print("Proposed score=%.6f(n=%d)" % (evaluate_sequence(final_candidate), len(final_candidate)))
    return final_candidate

def normalize_sequence(seq):
    s = sum(seq)
    scale = 1000 / s if s >= 0.01 else s  # This is just a thought, in reality the clipping is done by the evaluator and this function's purpose is to explain the constraints.
    return [x * scale for x in seq]

def evaluate_with_bounds(seq):
    """Evaluate with bounds handling - return inf if not valid."""
    try:
        # Ensure it's a list of numbers
        seq = [x for x in seq if isinstance(x, (int, float)) or np.isfinite(float(x))]
        s = sum(seq)
        if s < 0.01:
            return np.inf
        conv = np.convolve(seq, seq, mode='full')
        return float(2 * len(seq) * np.max(conv) / (s**2))
    except:
        return np.inf
