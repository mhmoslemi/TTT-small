```python
"""
Multi-phase optimization for low max-convolution step functions.
Phase 1: LP best-response iteration at moderate n to discover optimal shape.
Phase 2: FFT-based projected gradient descent with smoothed-max surrogate at target n.
Phase 3: Coordinate/pair polish with occasional LP re-solve.
Key insight: non-symmetric shapes allow max(conv) < S^2/n, breaking the symmetric lower bound.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_full(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def score_of(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s <= 1e-12:
        return np.inf
    c = conv_full(p)
    cm = float(c.max()) if c.size else 0.0
    if cm < 0:
        cm = 0.0
    return float(2.0 * n * cm / (s * s))


def softmax_weights(c, T):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c, 0.0
    cm = float(c.max())
    if cm <= 0 or not np.isfinite(cm):
        w = np.full(c.size, 1.0 / c.size)
        return w, cm
    T = max(T, 1e-300)
    d = (c - cm) / T
    e = np.exp(d)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
        wsum = 1.0
    else:
        w = e / s
        wsum = s
    return w, cm


def grad_smoothed_max(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), 0.0
    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(c.max()) if c.size else 0.0
    if cmax <= 0 or not np.isfinite(cmax):
        return np.zeros(n), 0.0
    T = max(T_factor * cmax, 1e-300)
    w, _ = softmax_weights(c, T)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    g = 2.0 * G
    surr = cmax + T * np.log(max(float(w.sum()), 1e-300))
    return g, surr


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return out / s


def normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(p.size, 1.0 / max(1, p.size))
    return p / s


def lp_solve(p, M, timeout=30.0):
    """Solve: max sum(g) s.t. p*g <= M, g >= 0. Returns normalized g or None."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n < 2 or n > 3000:
        return None
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(p, n).astype(np.float64)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method='highs',
                      options={'disp': False, 'time_limit': timeout})
        if res is None or not res.success:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s <= 1e-12:
            return None
        return g / s
    except Exception:
        return None

    starts.append(normalize(1.0 + 1.5 * x))
    starts.append(normalize(1.0 - 0.5 * x))
    starts.append(normalize(np.exp(-4.0 * (x - 0.5) ** 2)))
    for i in range(n_starts - len(starts)):
        starts.append(normalize(np.maximum(np.exp(-rng.uniform(1, 5) * (x - rng.uniform(0.2, 0.8)) ** 2 / 0.03), 0.01)))

    for p0 in starts:

def upsample(p_small, n_large):
    """Interpolate a shape from small n to large n using cubic-like interpolation."""
    p_small = np.asarray(p_small, dtype=np.float64)
    n_s = int(p_small.size)
    n_l = int(n_large)
    x_s = np.linspace(0, 1, n_s)
    x_l = np.linspace(0, 1, n_l)
    p_l = np.interp(x_l, x_s, p_small)
    p_l = np.maximum(p_l, 0.0)
    return normalize(p_l)

    lp_budget = min(350.0, max(60.0, (deadline - time.monotonic()) * 0.25))
    print("Phase 1: LP shape search n=%d budget=%.0fs" % (n_lp, lp_budget), flush=True)
    shape_p, shape_sc = lp_iterate(n_lp, rng, lp_budget, n_starts=5)
    record(upsample(shape_p, n_lp) if parent is not None and parent.size != n_lp else shape_p, shape_sc)

    # Phase 2: Test multiple target lengths
    # Determine target n values
    if parent is not None and parent.size >= 100 and parent.size <= 5000:
        target_ns = [parent.size]
    else:
        target_ns = [1000, 2000]

    # Add the LP n itself as a candidate
    if n_lp not in target_ns:
        target_ns.insert(0, n_lp)

    # Sort by expected difficulty (shorter first for quick wins)
    target_ns = sorted(set(target_ns))

    # Allocate time
    remaining = deadline - time.monotonic()
    n_targets = len(target_ns)
    # Give more time to longer sequences (they have more potential)
    weights = [1.0 + 0.3 * i for i in range(n_targets)]
    total_w = sum(weights)
    budgets = [max(60.0, remaining * w / total_w - 20.0) for w in weights]

    for ni, n in enumerate(target_ns):
        if time.monotonic() > deadline - 80:
            break
        tb = min(budgets[ni], deadline - time.monotonic() - 60.0)
        if tb < 30:
            break
        print("Phase 2: n=%d budget=%.0fs" % (n, tb), flush=True)

        # Initialize from shape
        if parent is not None and parent.size == n:
            p_init = normalize(parent.copy())
        else:
            p_init = upsample(shape_p, n)

        sc_init = score_of(p_init)
        record(p_init, sc_init)

        # Split time: 70% gradient, 30% polish
        t_grad = tb * 0.65
        t_pol = tb * 0.35

        # Gradient phase
        p_out, sc_out = grad_descent(p_init, n, rng, t_grad)
        record(p_out, sc_out)

        # Polish phase
        if time.monotonic() < deadline - 20:
            t_pol_actual = min(t_pol, deadline - time.monotonic() - 15.0)
            if t_pol_actual > 10:
                p_pol, sc_pol = polish(p_out, n, rng, t_pol_actual)
                record(p_pol, sc_pol)

    # Final: if we still have time, do one more LP+gradient cycle at best n
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_final = int(best_seq.size)
        t_final = deadline - time.monotonic() - 15
        print("Final polish: n=%d budget=%.0fs" % (n_final, t_final), flush=True)
        # Quick LP
        if n_final <= 1200 and t_final > 30:
            c = conv_full(best_seq)
            M = float(c.max())
            g_lp = lp_solve(best_seq, M, timeout=min(20.0, t_final * 0.3))
            if g_lp is not None:
                for t in (0.5, 0.3, 0.15, 0.08):
                    q = project_simplex((1 - t) * best_seq + t * g_lp)
                    sc_q = score_of(q)
                    if np.isfinite(sc_q) and sc_q < best_score:
                        best_score = sc_q
                        best_seq = q.copy()
                        print("  LP blend t=%.2f -> %.10f" % (t, best_score), flush=True)
        # Gradient
        if time.monotonic() < deadline - 10:
            t_g = deadline - time.monotonic() - 5
            p_g, sc_g = grad_descent(best_seq, n_final, rng, t_g)
            record(p_g, sc_g)

    # Safety fallback
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)
        best_score = score_of(best_seq)

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = best_seq * (0.01 / max(s, 1e-300))
    best_seq = np.minimum(best_seq, 1000.0)

    elapsed = time.monotonic() - t_start
    print("DONE: score=%.10f n=%d elapsed=%.1fs" % (best_score, best_seq.size, elapsed), flush=True)
    return [float(x) for x in best_seq]
```
