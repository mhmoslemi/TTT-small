from scipy.optimize import linprog

def solve_lp_for_b(f, M, n):
    """
    Solve: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k.
    """
    # Objective: maximize sum(b) = minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_i f[i] * b[k-i] (valid indices)
    # This is a linear function of b
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, j] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Non-negativity: b >= 0 (handled by bounds)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x

def solve_lp_active_set(a, S, n, active_lags=None, max_iter=100):
    """
    Solve: min t s.t. (a*b)[k] <= t for active lags k, sum(b) = S, b >= 0.
    Uses cutting-plane method to add violated constraints.
    """
    if active_lags is None:
        active_lags = list(range(n))  # Start with first n lags
    
    for iteration in range(max_iter):
        # Build constraint matrix for active lags
        m = len(active_lags)
        A_ub = np.zeros((m, n+1))
        for idx, k in enumerate(active_lags):
            j_min = max(0, k - (n-1))
            j_max = min(k, n-1)
            for j in range(j_min, j_max+1):
                A_ub[idx, j] = a[k-j]
            A_ub[idx, -1] = -1.0
        
        b_ub = np.zeros(m)
        
        # Solve
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        bounds = [(0, None)] * (n+1)
        
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if not result.success:
            return None
        
        b_sol = result.x[:n]
        t_sol = result.x[-1]
        
        # Check all constraints
        conv = fft_convolve(a, b_sol)
        violated = np.where(conv > t_sol + 1e-10)[0]
        
        if len(violated) == 0:
            return b_sol, t_sol
        
        # Add violated constraints
        active_lags = list(set(active_lags) | set(violated.tolist()))
        
    return b_sol, t_sol
