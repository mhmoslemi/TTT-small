def compute_grad_sofmax(a, tau):

    Compute gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Returns (value, gradient, convolution).

    exp_c = np.exp((c - c_max) / tau)
    w = exp_c / np.sum(exp_c)
    value = c_max + tau * np.log(np.sum(exp_c))

    # This is a cross-correlation
    # sum_k a[k] * w[j+k] for j=0..n-1
    # = for each j, sum over k of a[k] * w[j+k]
    # This is (a * w)[j] in the "cross-correlation" sense
    # = sum_k a[k] * w[j+k]
    # Let's compute this using the fact that it's a convolution-like operation
    
    # w[j+k] for k=0..min(n-1, len(w)-1-j)
    # We need sum_k a[k] * w[j+k] for each j
    # This is like a cross-correlation of a and w
    # = (a correlated with w) evaluated at various points
    
    # Using FFT: 
    # sum_k a[k] * w[j+k] = sum_k a[k] * w_rev[w_rev_len-1-(j+k)]  ... this is getting complicated

    # sum_k a[k] * w[j+k] = (flip(a) * w)[j + n - 1] where * is linear convolution
    flip_a = a[::-1]
    conv = fft_convolve(flip_a, w)
    grad = 2.0 * conv[n-1:2*n-1]
    
    return value, grad, c

            comp_idx = rng.integers(n)
            candidate[comp_idx] += freed
        
        candidate = np.maximum(candidate, 1e-10)
