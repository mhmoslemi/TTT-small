def subgradient_step(a, lr, n_active=5):
    """One subgradient step on the peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    # Find top n_active peaks
    top_indices = np.argsort(c)[-n_active:]
    grad = np.zeros(n)
    for m in top_indices:
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            if m - j == j:  # m = 2j
                grad[j] += 2.0 * a[j]
            else:
                grad[j] += a[m - j]
    grad /= n_active  # average over active constraints
    # Project out sum constraint
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses iterative LP-based refinement with ternary search for optimal blending,
adaptive constraint levels, and multiple optimization strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def compute_peak(a):
    c = np.convolve(a, a)
    return float(c.max())

def build_A_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A

def solve_lp(f, M):
    n = len(f)
    A = build_A_matrix(f)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def ternary_search_blend(f, g, max_iter=30):
    """Find t in [0,1] that minimizes the objective of (1-t)f + tg (normalized)."""
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        
        a1 = np.maximum((1 - m1) * f + m1 * g, 0.0)
        s1 = a1.sum()
        if s1 < 1e-15:
            score1 = float('inf')
        else:
            a1 = a1 / s1
            score1 = compute_peak(a1)  # sum is 1, so obj = 2n * peak
        
        a2 = np.maximum((1 - m2) * f + m2 * g, 0.0)
        s2 = a2.sum()
        if s2 < 1e-15:
            score2 = float('inf')
        else:
            a2 = a2 / s2
            score2 = compute_peak(a2)
        
        if score1 < score2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    a = np.maximum((1 - t) * f + t * g, 0.0)
    s = a.sum()
    if s < 1e-15:
        return f, 0.0
    a = a / s
    return a, t
