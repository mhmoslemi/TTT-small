```python
    # Minimize t
    # Subject to:
    #   2*conv(a,d)[k] - t <= -conv_a[k] for k=0..2n-2
    #   sum(d) = 0
    #   d[i] >= -a[i]

    c[n] = 1.0  # minimize t
    
    # Convolution constraints
    # For each k, the coefficient of d[i] is 2*a[k-i] if 0 <= k-i < n
```

```python
"""
LP-based iterative optimization to minimize 2n*max(conv(a,a))/sum(a)^2
for non-negative sequences. Uses linearized subgradient method with
LP at each step to find the best descent direction.
"""

import numpy as np
from scipy.sparse import csr_matrix, hstack
from scipy.optimize import linprog
import time


def score_seq(a):
    """Compute the score for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(a, n):
    """Build sparse matrix C where C[k,i] = a[k-i] if 0<=k-i<n."""
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])
    data = np.tile(a.astype(np.float64), n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def solve_direction(a, conv_a, n, timeout=30.0):
    """Solve the LP to find the best first-order descent direction."""
    C = build_conv_matrix(a, n)
    
    # A_ub = [2C | -1], shape (2n-1, n+1)
    neg_one_col = csr_matrix(-np.ones((2 * n - 1, 1)))
    A_ub = hstack([2.0 * C, neg_one_col], format='csr')
    b_ub = -conv_a
    
    # A_eq: sum(d) = 0
    A_eq = csr_matrix(np.ones((1, n)), shape=(1, n + 1))
    b_eq = np.array([0.0])
    
    # Bounds: d[i] >= -a[i], t >= 0
    bounds = [(-float(a[i]), None) for i in range(n)]
    bounds.append((0.0, None))
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return res.x[:n], res.x[n]
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None, None


def lp_optimize(a, time_limit_s, verbose=True):
    """Iteratively optimize using LP-based linearized subgradient."""
    n = len(a)
    a = np.maximum(0.0, a.astype(np.float64).copy())
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # normalize to sum = 1
    
    start_time = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    stagnant = 0
    it = 0
    
    if verbose:
        print(f"  LP start: score={best_score:.6f}, n={n}")
    
    while time.monotonic() - start_time < time_limit_s:
        conv_a = np.convolve(a, a)
        current_peak = conv_a.max()
        current_score = 2.0 * n * current_peak  # sum = 1
        
        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            break
        
        # Check if we're at a local optimum (peak at center with uniform-like distribution)
        # Solve LP for direction
        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 5.0:
            break
        timeout = max(2.0, min(60.0, remaining * 0.15))
        
        d, t_val = solve_direction(a, conv_a, n, timeout=timeout)
        
        if d is None:
            break
        
        if t_val >= current_peak - 1e-8:
            # Can't improve
            if verbose:
                print(f"  LP: no improvement at iter {it}, t={t_val:.6f} >= peak={current_peak:.6f}")
            break
        
        # Line search
        best_alpha = 0.0
        best_local = current_score
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            a_candidate = a + alpha * d
            if (a_candidate < -1e-12).any():
                continue
            a_candidate = np.maximum(0.0, a_candidate)
            a_candidate = a_candidate / a_candidate.sum()
            s_local = 2.0 * n * np.convolve(a_candidate, a_candidate).max()
            if s_local < best_local:
                best_local = s_local
                best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        a = a + best_alpha * d
        a = np.maximum(0.0, a)
        a = a / a.sum()
        it += 1
        
        if verbose and it % 10 == 0:
            print(f"  LP iter {it:4d}: score={current_score:.6f}, alpha={best_alpha:.3f}, t_val={t_val:.6f}")
    
    final_score = 2.0 * n * np.convolve(a, a).max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"  LP done: best={best_score:.6f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    
    return best_a, best_score
```

```python
"""
LP-based iterative optimization to minimize 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses a linearized subgradient method where at each step an LP is solved to find the best
first-order descent direction for reducing the convolution peak. Combined with local search
polishing and multi-start from diverse initializations.
"""

import numpy as np
from scipy.sparse import csr_matrix, hstack
from scipy.optimize import linprog
import time


def score_seq(a):
    """Compute the score 2n*max(conv(a,a))/sum(a)^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(a, n):
    """Build sparse matrix C where C[k,i] = a[k-i] if 0<=k-i<n, else 0. Shape (2n-1, n)."""
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])
    data = np.tile(a.astype(np.float64), n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def lp_step(a, conv_a, n, timeout=30.0):
    """Solve LP to find best first-order direction d to reduce convolution peak.
    
    Minimize t
    Subject to: 2*conv(a,d)[k] - t <= -conv_a[k] for all k
                sum(d) = 0
                d[i] >= -a[i]
                
    Returns (d, t_value) or (None, None) on failure.
    """
    C = build_conv_matrix(a, n)
    neg_one_col = csr_matrix(-np.ones((2 * n - 1, 1), dtype=np.float64))
    A_ub = hstack([2.0 * C, neg_one_col], format='csr')
    b_ub = -conv_a.astype(np.float64)

    A_eq = csr_matrix(np.ones((1, n), dtype=np.float64), shape=(1, n + 1))
    b_eq = np.array([0.0])

    bounds = [(-float(a[i]), None) for i in range(n)]
    bounds.append((0.0, None))

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return res.x[:n], float(res.x[n])
    except Exception as e:
        print(f"    LP error: {e}")

    return None, None


def lp_optimize(a, time_limit_s, verbose=True):
    """Iteratively optimize using LP-based linearized subgradient method."""
    n = len(a)
    a = np.maximum(0.0, a.astype(np.float64).copy())
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    start_time = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    stagnant = 0
    it = 0

    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    max_iters = 2000
    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        conv_a = np.convolve(a, a)
        current_peak = conv_a.max()
        current_score = 2.0 * n * current_peak

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 40:
            break

        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 5.0:
            break
        timeout = max(1.0, min(45.0, remaining * 0.12))

        d, t_val = lp_step(a, conv_a, n, timeout=timeout)

        if d is None:
            break

        if t_val >= current_peak - 1e-9:
            break

        # Line search
        best_alpha = 0.0
        best_local = current_score
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            a_cand = a + alpha * d
            if (a_cand < -1e-12).any():
                a_cand = np.maximum(0.0, a_cand)
            s_cand = a_cand.sum()
            if s_cand < 1e-12:
                continue
            a_cand = a_cand / s_cand
            s_local = 2.0 * n * np.convolve(a_cand, a_cand).max()
            if s_local < best_local:
                best_local = s_local
                best_alpha = alpha

        if best_alpha == 0.0:
            # Try smaller steps
            for alpha in [0.01, 0.05]:
                a_cand = a + alpha * d
                a_cand = np.maximum(0.0, a_cand)
                s_cand = a_cand.sum()
                if s_cand < 1e-12:
                    continue
                a_cand = a_cand / s_cand
                s_local = 2.0 * n * np.convolve(a_cand, a_cand).max()
                if s_local < best_local:
                    best_local = s_local
                    best_alpha = alpha

        if best_alpha == 0.0:
            break

        a = a + best_alpha * d
        a = np.maximum(0.0, a)
        a = a / a.sum()
        it += 1

        if verbose and it % 20 == 0:
            print(f"    LP iter {it:4d}: score={current_score:.6f}, alpha={best_alpha:.3f}, t={t_val:.6f}")

    final_score = 2.0 * n * np.convolve(a, a).max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f}, {it} iters, {time.monotonic()-start_time:.1f}s")

    return best_a, best_score


def local_search(a, time_limit_s, rng, verbose=True):
    """Local search with random multiplicative perturbations."""
    n = len(a)
    a = np.maximum(0.0, a.astype(np.float64).copy())
    a = a / a.sum()

    start_time = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    accept_count = 0

    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(n * (0.01 + 0.2 * rng.random())))
        start = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        noise = np.exp(rng.normal(0, 0.08, size=width))
        candidate[start:start + width] *= noise
        candidate = np.maximum(0, candidate)
        s = candidate.sum()
        if s < 1e-12:
            continue
        candidate = candidate / s

        sc = 2.0 * n * np.convolve(candidate, candidate).max()
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 200 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")

    return best_a, best_score


def init_power(n, p=1.0):
    """Monotonically increasing power-law initialization."""
    t = np.arange(1, n + 1, dtype=np.float64) / n
    return t ** p


def init_geometric(n, r=1.1):
    """Geometric initialization."""
    return r ** np.arange(n, dtype=np.float64)


def init_mixed(n, rng, p1=1.0, p2=2.0):
    """Mixed initialization combining power laws with random perturbation."""
    a = init_power(n, p1) + rng.random() * 0.1 * init_power(n, p2)
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses LP-based iterative refinement with multiple initializations and lengths,
    followed by local search and final LP polishing.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 20.0)

    best_score = float('inf')
    best_seq = None

    # Try parent sequence if available
    try:
        parent = np.array([float(x) for x in height_sequence_1], dtype=np.float64)
        parent = np.maximum(0, parent)
        if parent.sum() > 0.01 and len(parent) > 0:
            parent_score = score_seq(parent)
            print(f"Parent sequence: score={parent_score:.6f}, n={len(parent)}")
            if parent_score < best_score:
                best_score = parent_score
                best_seq = parent.copy()
            # Refine parent with LP
            t_limit = min(180.0, budget_s * 0.15)
            remaining = deadline - time.monotonic()
            t_limit = min(t_limit, remaining * 0.4)
            if t_limit > 20:
                print(f"\n=== LP refine parent (n={len(parent)}, t={t_limit:.0f}s) ===")
                refined, s = lp_optimize(parent, time_limit_s=t_limit, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = refined.copy()
                    print(f"  *** New best from parent: {best_score:.6f}")
    except (NameError, TypeError):
        pass

    # Define configurations: (n, init_type, param, time_fraction)
    configs = [
        (500, 'power', 1.0, 0.22),
        (500, 'power', 1.5, 0.14),
        (500, 'power', 2.0, 0.12),
        (1000, 'power', 1.0, 0.18),
        (1000, 'power', 1.5, 0.12),
        (500, 'geo', 1.05, 0.10),
        (1000, 'geo', 1.03, 0.10),
        (300, 'power', 1.0, 0.06),
        (2000, 'power', 1.0, 0.12),
        (1500, 'power', 1.5, 0.10),
    ]

    for n, init_type, param, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        t_limit = min(time_frac * budget_s, remaining * 0.85)
        if t_limit < 15.0:
            continue

        print(f"\n=== n={n}, init={init_type}({param}), t={t_limit:.0f}s ===")

        if init_type == 'power':
            a = init_power(n, p=param)
        elif init_type == 'geo':
            a = init_geometric(n, r=param)
        else:
            a = init_power(n)

        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        init_score = 2.0 * n * np.convolve(a, a).max()
        print(f"  Init score: {init_score:.6f}")

        # LP optimization
        a_refined, score = lp_optimize(a, time_limit_s=t_limit, verbose=True)

        if score < best_score:
            best_score = score
            best_seq = a_refined.copy()
            print(f"  *** NEW BEST: score={score:.6f}, n={n}")

    # Final polishing phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print(f"\n=== Final polish (remaining={remaining:.0f}s) ===")

            # LP polish
            t_lp = remaining * 0.45
            polished, s = lp_optimize(best_seq, time_limit_s=t_lp, verbose=True)
            if s < best_score:
                best_score = s
                best_seq = polished.copy()
                print(f"  *** After LP polish: {best_score:.6f}")

            # Local search
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20.0:
                t_ls = remaining2 * 0.4
                polished2, s2 = local_search(best_seq, time_limit_s=t_ls, rng=rng, verbose=True)
                if s2 < best_score:
                    best_score = s2
                    best_seq = polished2.copy()
                    print(f"  *** After LS: {best_score:.6f}")

            # Final LP after LS
            remaining3 = deadline - time.monotonic()
            if remaining3 > 15.0:
                t_final = remaining3 * 0.85
                polished3, s3 = lp_optimize(best_seq, time_limit_s=t_final, verbose=True)
                if s3 < best_score:
                    best_score = s3
                    best_seq = polished3.copy()
                    print(f"  *** Final LP: {best_score:.6f}")

    # If no good result, try one more time with a fresh start
    if best_score > 1.8:
        print("\n=== Retry with different init ===")
        remaining = deadline - time.monotonic()
        if remaining > 60:
            for p_try in [0.5, 0.7, 1.3, 1.8]:
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                n_try = 500
                a = init_power(n_try, p=p_try)
                a = a / a.sum()
                t_left = min(remaining * 0.4, 120.0)
                if t_left < 15:
                    break
                print(f"  Retry: n={n_try}, p={p_try}, t={t_left:.0f}s")
                refined, s = lp_optimize(a, time_limit_s=t_left, verbose=False)
                if s < best_score:
                    best_score = s
                    best_seq = refined.copy()
                    print(f"  *** Retry best: {best_score:.6f}")

    if best_seq is None:
        a = init_power(500, 1.0)
        a = a / a.sum()
        best_seq = a
        best_score = score_seq(a)
        print(f"  Fallback: score={best_score:.6f}")

    result = [float(x) for x in best_seq]
    verify_score = evaluate_sequence(result)
    print(f"\nFinal result: score={best_score:.6f}, verified={verify_score:.6f}, n={len(result)}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return result
```
