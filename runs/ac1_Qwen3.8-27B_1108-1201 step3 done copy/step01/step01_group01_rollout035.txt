                perturb = rng.normal(0, 0.01 * np.mean(best), n)

Adaptive soft-max subgradient optimization with FFT-accelerated gradients,

import numpy as np

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_b(a):
    """Full autocorrelation b = convolve(a, a), length 2n-1."""
    return np.convolve(a, a)


def _score(a):
    """Compute 2*n*max(b) assuming sum(a)=1."""
    n = len(a)
    b = _compute_b(a)

def _fft_grad(a, w, L):
    """
    Compute grad[j] = 2 * sum_m w[j+m] * a[m] using FFT.
    w has length 2n-1, a has length n, L is FFT size >= 2n-1.
    Returns grad of length n.
    """
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    c = np.fft.irfft(A * np.conj(W), L)
    return 2.0 * c[:len(a)]
