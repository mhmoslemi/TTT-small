def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def lp_direction(a, M):
    """
    Find the best direction b that maximizes sum(b) subject to:
    - b >= 0
    - conv(a, b) <= M (element-wise)
    
    This is an LP with n variables and (2n-1) constraints.
    """
    n = len(a)
    # conv(a, b)[k] = sum_i a[i] * b[k-i] for valid i
    # This is a convolution, so the constraint matrix is a Toeplitz-like matrix
    
    # Build constraint matrix A such that A @ b <= M_vec
    # conv(a, b)[k] for k = 0, ..., 2n-2
    # conv(a, b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]
    
    # For efficiency, build the matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                A[k, j] = a[i]
    
    b_vec = np.ones(n)  # maximize sum(b) = 1^T b
    # linprog minimizes, so we minimize -1^T b
    res = linprog(-b_vec, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

def lp_optimize(a_init, time_limit, rng):

                best_theta = theta.copy()
            print(f"    grad step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.0f}s")
    
    a_best = np.exp(best_theta - np.max(best_theta))
    final_score = local_score(a_best)
    if final_score < best_score:
        best_score = final_score
    
    return a_best, best_score


def build_lp_matrix(a):
    """Build the constraint matrix for the LP. A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_optimize(a_init, time_limit, rng):
    """
    LP-based optimization: iteratively find optimal direction and take a step.
    The LP maximizes sum(b) subject to b >= 0 and conv(a, b) <= M.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init.astype(np.float64).copy()
    sum_a = np.sum(a)
    a = a * S / sum_a  # normalize
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    step_size = 0.4
    no_improve_count = 0
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fft_convolve(a, a)
        M = np.max(c)
        
        # Build LP matrix
        A = build_lp_matrix(a)
        
        # Solve LP: maximize sum(b) s.t. A@b <= M, b >= 0
        c_obj = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except:
            break
        
        if not res.success:
            break
        
        b_dir = res.x
        sum_b = np.sum(b_dir)
        
        if sum_b < 1e-10:
            break
        
        # Check if b_dir is essentially the same as a (no improvement direction)
        b_norm = b_dir * S / sum_b
        if np.allclose(b_norm, a, atol=1e-4):
            no_improve_count += 1
            if no_improve_count > 20:
                # Try reducing M to force a different direction
                M_reduced = M * 0.99
                b_ub2 = np.full(2 * n - 1, M_reduced)
                try:
                    res2 = linprog(c_obj, A_ub=A, b_ub=b_ub2, bounds=bounds, method='highs')
                    if res2.success:
                        b_dir = res2.x
                        sum_b = np.sum(b_dir)
                        if sum_b > 1e-10:
                            b_norm = b_dir * S / sum_b
                            no_improve_count = 0
                        else:
                            break
                    else:
                        break
                except:
                    break
            continue
        
        no_improve_count = 0
        
        # Take a step
        a_new = (1.0 - step_size) * a + step_size * b_norm
        a_new = np.clip(a_new, 0.0, None)
        sum_new = np.sum(a_new)
        if sum_new < 1e-10:
            step_size *= 0.5
            continue
        a_new = a_new * S / sum_new
        
        score_new = local_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            # Slightly increase step size on success
            step_size = min(0.6, step_size * 1.02)
        else:
            # Reduce step size
            step_size = max(0.05, step_size * 0.85)
            # Try a smaller step
            a_new2 = (1.0 - step_size) * a + step_size * b_norm
            a_new2 = np.clip(a_new2, 0.0, None)
            sum_new2 = np.sum(a_new2)
            if sum_new2 > 1e-10:
                a_new2 = a_new2 * S / sum_new2
                score_new2 = local_score(a_new2)
                if score_new2 < best_score:
                    best_score = score_new2
                    best_a = a_new2.copy()
                    a = a_new2
        
        if it % 100 == 0:
            print(f"    LP it={it}, score={score_new:.6f}, best={best_score:.6f}, step={step_size:.3f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def targeted_local_search(a_init, time_limit, rng):
    """
    Aggressive local search with simulated annealing, focused on peak contributors.
    Uses paired moves (reduce peak contributor + increase non-peak contributor).
    """
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    temp = 0.002
    cooling = 0.99995
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution and find peak
        conv = fft_convolve(current, current)
        peak_val = np.max(conv)
        peak_idx = int(np.argmax(conv))
        
        # Find contributors to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                if i == j:
                    contribs[i] = current[i] * current[i]
                else:
                    contribs[i] = current[i] * current[j]
        
        # Also find positions that contribute to SECOND peak
        peak2_val = conv.copy()
        peak2_val[peak_idx] = -1
        peak2_idx = int(np.argmax(peak2_val))
        contribs2 = np.zeros(n)
        for i in range(n):
            j = peak2_idx - i
            if 0 <= j < n:
                if i == j:
                    contribs2[i] = current[i] * current[i]
                else:
                    contribs2[i] = current[i] * current[j]
        
        # Combined "importance" for being a peak contributor
        importance = 0.7 * contribs + 0.3 * contribs2
        
        r = rng.random()
        
        if r < 0.25:
            # Paired move: reduce top peak contributor, increase bottom contributor
            n_top = min(16, n)
            top_idx = np.argsort(contribs)[-n_top:]
            n_bot = min(16, n)
            bot_idx = np.argsort(contribs)[:n_bot]
            
            j = top_idx[rng.integers(n_top)]
            l = bot_idx[rng.integers(n_bot)]
            if j == l:
                l = bot_idx[(rng.integers(n_top) + 1) % n_bot]
            
            candidate = current.copy()
            frac = rng.uniform(0.02, 0.1)
            delta = current[j] * frac
            candidate[j] = max(0.0, current[j] - delta)
            candidate[l] = current[l] + delta * rng.uniform(0.5, 1.5)
            
        elif r < 0.45:
            # Reduce multiple top contributors
            n_top = min(8, n)
            top_idx = np.argsort(contribs)[-n_top:]
            n_reduce = rng.integers(2, min(5, n_top) + 1)
            reduce_idx = rng.choice(top_idx, size=n_reduce, replace=False)
            candidate = current.copy()
            total_reduced = 0.0
            for j in reduce_idx:
                frac = rng.uniform(0.03, 0.12)
                delta = current[j] * frac
                candidate[j] = max(0.0, current[j] - delta)
                total_reduced += delta
            # Distribute to random low-importance positions
            n_inc = min(4, n)
            bot_idx = np.argsort(importance)[:n_inc]
            inc_idx = rng.choice(bot_idx, size=min(3, len(bot_idx)), replace=False)
            per_inc = total_reduced / len(inc_idx) * rng.uniform(0.8, 1.2)
            for l in inc_idx:
                candidate[l] += per_inc
                
        elif r < 0.6:
            # Small random perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))
            
        elif r < 0.8:
            # Block move on peak contributors
            n_top = min(32, n)
            top_idx = np.argsort(contribs)[-n_top:]
            # Pick a contiguous block of top contributors
            block_size = rng.integers(2, min(6, n_top))
            start_pos = rng.integers(0, n_top - block_size + 1)
            block = top_idx[start_pos:start_pos + block_size]
            candidate = current.copy()
            candidate[block] *= np.exp(rng.normal(-0.03, 0.02, size=len(block)))
            # Compensate with low-importance positions
            n_bot = min(32, n)
            bot_idx = np.argsort(importance)[:n_bot]
            comp_idx = rng.choice(bot_idx, size=min(len(block), n_bot), replace=False)
            candidate[comp_idx] *= np.exp(rng.normal(0.03, 0.02, size=len(comp_idx)))
            
        else:
            # Symmetrization attempt
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.005, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)
        iter_count += 1
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if iter_count % 2000 == 0:
                    print(f"    SA it={iter_count}, best={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            current = candidate
            current_score = new_score
        
        temp *= cooling
        if temp < 1e-8:
            temp = 1e-8
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.05, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_ramp_init(n, rng):
    """Create a ramp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    # Try different ramp shapes
    shape = rng.random()
    if shape < 0.3:
        a = 1.0 + 0.5 * t / n
    elif shape < 0.6:
        a = 1.5 - 0.5 * t / n
    else:
        a = 1.0 + 0.3 * np.sin(np.pi * t / n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for _ in range(rng.integers(3, 7)):
        k = rng.integers(1, 15)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: gradient optimization, LP-based refinement,
    and aggressive targeted local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Try to use provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
        except:
            start_seq = None
    
    n_main = 256
    if start_seq is not None:
        n_main = len(start_seq)
    
    # Phase 1: Gradient-based optimization
    print("\n=== Phase 1: Gradient optimization ===")
    grad_time = min(400.0, time_budget * 0.25)
    
    initializations = []
    if start_seq is not None and start_seq.shape[0] == n_main:
        initializations.append(('provided', start_seq.copy()))
    initializations.append(('chirp', make_chirp_init(n_main, 0.5 + 0.3 * rng.random(), rng)))
    initializations.append(('ramp', make_ramp_init(n_main, rng)))
    initializations.append(('multi_tone', make_multi_tone_init(n_main, rng)))
    
    time_per_init = grad_time / len(initializations)
    
    for name, a0 in initializations:
        if time.monotonic() > deadline - 900:
            break
        a0 = np.clip(a0, 0.01, 1000.0)
        if np.sum(a0) < 0.01:
            continue
        
        t0 = time.monotonic()
        a_best, score = optimize_gradient_exp(n_main, a0, min(time_per_init, deadline - time.monotonic() - 900), lr_base=0.3)
        elapsed = time.monotonic() - t0
        print(f"  Grad ({name}): score={score:.6f}, time={elapsed:.0f}s")
        
        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 2: LP-based optimization
    print(f"\n=== Phase 2: LP optimization (best = {best_score:.6f}) ===")
    if best_seq is not None:
        lp_time = min(700.0, max(200.0, deadline - time.monotonic() - 500.0))
        if lp_time > 60:
            print(f"  LP time budget: {lp_time:.0f}s")
            a_lp, lp_score = lp_optimize(best_seq, lp_time, rng)
            print(f"  LP result: score={lp_score:.6f}")
            if lp_score < best_score:
                best_score = lp_score
                best_seq = a_lp.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 3: Targeted local search
    print(f"\n=== Phase 3: Local search polish (best = {best_score:.6f}) ===")
    if best_seq is not None:
        sa_time = max(60.0, min(500.0, deadline - time.monotonic() - 20.0))
        print(f"  SA time budget: {sa_time:.0f}s")
        a_sa, sa_score = targeted_local_search(best_seq, sa_time, rng)
        print(f"  SA result: score={sa_score:.6f}")
        if sa_score < best_score:
            best_score = sa_score
            best_seq = a_sa.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 4: Try a second round of local search if time permits
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Second local search round ===")
        sa2_time = max(30.0, min(300.0, deadline - time.monotonic() - 20.0))
        # Start from a slightly perturbed version
        a_perturbed = best_seq * np.exp(rng.normal(0.0, 0.01, size=len(best_seq)))
        a_sa2, sa2_score = targeted_local_search(a_perturbed, sa2_time, rng)
        print(f"  SA2 result: score={sa2_score:.6f}")
        if sa2_score < best_score:
            best_score = sa2_score
            best_seq = a_sa2.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Also try a different length briefly if time permits
    if best_seq is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Phase 5: Try different length ===")
        for n_try in [384, 512]:
            if time.monotonic() > deadline - 40:
                break
            # Interpolate or create new initialization
            if start_seq is not None:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n_try)
                a0 = np.interp(x_new, x_old, start_seq)
            else:
                a0 = make_chirp_init(n_try, 0.6, rng)
            
            a0 = np.clip(a0, 0.01, 1000.0)
            # Quick gradient optimization
            t_remaining = min(80.0, deadline - time.monotonic() - 30)
            if t_remaining > 20:
                a_try, try_score = optimize_gradient_exp(n_try, a0, t_remaining, lr_base=0.3)
                print(f"  n={n_try}: score={try_score:.6f}")
                if try_score < best_score:
                    best_score = try_score
                    best_seq = a_try.copy()
                    print(f"    *** New best: {best_score:.6f} ***")
                
                # Quick local search
                if time.monotonic() < deadline - 30:
                    sa_t = min(60.0, deadline - time.monotonic() - 20)
                    a_ps, ps_score = targeted_local_search(a_try, sa_t, rng)
                    if ps_score < best_score:
                        best_score = ps_score
                        best_seq = a_ps.copy()
                        print(f"    *** New best: {best_score:.6f} ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
