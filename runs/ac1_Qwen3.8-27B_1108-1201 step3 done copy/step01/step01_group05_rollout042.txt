"""
Step-function search for low max autocorrelation using LP-based best response
iteration with diverse initializations, coarse-to-fine scaling, adaptive
blending, and simulated annealing to escape local minima. Key improvements:
- Min-t LP formulation for tighter best-response
- Coarse-to-fine: optimize at small n, interpolate up
- Multi-length exploration
- Adaptive temperature for annealing
- Diverse shape initializations including power-law tapers

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv_sq(p):
    """Full self-convolution via FFT."""
    n = len(p)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(max(1, 2 * n - 1))
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def _score(p):
    """Score = 2n * max(conv) / (sum)^2, scale-invariant."""
    n = len(p)
    s = float(np.sum(p))
    if s < 1e-12:
        return np.inf
    c = _conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def _norm(p):
    """Normalize to sum 1, non-negative."""
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s < 1e-12:
        return np.full(len(p), 1.0 / len(p))
    return p / s


def _project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(np.sum(out))
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _build_conv_sparse(f, n):
    """Build sparse convolution matrix: (A @ g)[k] = sum_j f[j]*g[k-j]."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        cnt = j_hi - j_lo + 1
        if cnt > 0:
            rows.extend([k] * cnt)
            cols.extend(range(j_lo, j_hi + 1))
            data.extend(f[j_lo:j_hi + 1].tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_best_response(f, n, time_limit=30.0):
    """
    Solve: min t
           s.t. (f*g)[k] <= t, k=0..2n-2
                sum(g) = 1
                g >= 0
    Returns (g_normalized, t_star) or (None, None).
    """
    t_start = time.monotonic()
    if time.monotonic() - t_start > time_limit:
        return None, None

    A_conv = _build_conv_sparse(f, n)
    # Dense for HiGHS efficiency at these sizes
    A_dense = A_conv.toarray()

    # Variables: [g_0, ..., g_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequalities: (f*g)[k] - t <= 0
    A_ub = np.column_stack([A_dense, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(-1e9, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'disp': False, 'time_limit': max(1.0, time_limit - (time.monotonic() - t_start))})
    except Exception:
        return None, None

    if res is None or res.status != 0:
        return None, None

    g = np.asarray(res.x[:n], dtype=np.float64)
    t_star = float(res.x[-1])
    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if s < 1e-12:
        return None, None
    g = g / s
    return g, t_star


def _make_geometric(n, rate, reverse=False):
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-rate * x)
    if reverse:
        p = p[::-1]
    return _norm(p)


def _make_power_taper(n, alpha, beta=0.0):
    """p[j] proportional to (j+1)^alpha * (n-j)^beta, normalized."""
    j = np.arange(1, n + 1, dtype=np.float64)
    p = np.power(j, alpha) * np.power(n - j + 1, beta)
    p = np.maximum(p, 0.0)
    return _norm(p)


def _make_quadratic_taper(n, a, b):
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + a * x + b * x * x
    return _norm(np.maximum(p, 0.0))


def _make_cubic_taper(n, a, b, c):
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + a * x + b * x * x + c * x * x * x
    return _norm(np.maximum(p, 0.0))


def _make_exp_taper(n, a, b):
    """p[j] = exp(a*x + b*x^2), x in [0,1]."""
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(a * x + b * x * x)
    return _norm(p)


def _make_diadic(n, rng, smooth=2.0):
    """Random smooth function via sum of sines."""
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n)
    for _ in range(3):
        f = float(rng.integers(1, max(2, n // 20)))
        ph = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.05, 0.5)) / smooth
        p += amp * np.sin(2 * np.pi * f * x + ph)
    return _norm(p)


def _interp_extend(p, new_n):
    """Interpolate a sequence to a new length."""
    n = len(p)
    if n == new_n:
        return p.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return _norm(p_new)

def _gradient_optimize(f, n, rng, time_limit, label=""):
    """
    Gradient-based optimization using smoothed max surrogate.
    For larger sequences where LP is too slow.
    """
    f = _norm(f.astype(np.float64).copy())
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    cur_score = best_score
    end = time.monotonic() + time_limit
    it = 0
    T_factor = 0.1
    momentum_vec = np.zeros(n)
    momentum = 0.3

    L = next_fast_len(max(1, 3 * n - 2))

    while time.monotonic() < end:
        it += 1
        c = _conv_sq(f)
        cmax = float(np.max(c)) if c.size else 0.0
        if cmax <= 0:
            break

        T = max(T_factor * cmax, 1e-30)
        diff = (c - cmax) / T
        e = np.exp(diff)
        s = float(np.sum(e))
        if s < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / s

        # Gradient: grad_j = 2 * sum_k w_k * f[k-j] (with bounds)
        # Use FFT: grad = 2 * ifft(fft(f, L) * fft(w[::-1], L)) sliced appropriately
        Ff = fft(f, L)
        Fw = fft(w[::-1], L)
        h = np.real(ifft(Ff * Fw))

        # The gradient of sum_k w_k * c[k] w.r.t. f[j]
        # c[k] = sum_i f[i]*f[k-i], dc[k]/df[j] = f[k-j] + f[j-k] (if in range)
        # grad_j = sum_k w_k * (f[k-j] + f[j-k]) for valid k
        # This equals 2 * (w * f)[j] in the circular sense, but we need linear.
        # h[j] = sum_k Ff_freq * Fw_freq... actually let me use the direct formula.
        # Full circular convolution: (w circ f)[j] = sum_k w[k] * f[(j-k) mod L]
        # We want the linear part.
        # Actually, the simplest: grad = 2 * ifft(fft(f, L) * fft(w, L)) gives circular
        # For linear, we need to be careful. Let me just compute it directly for correctness.
        # Actually the formula from the previous code works:
        # G = h[2n-2 : n-2 : -1] gives the gradient
        if n == 1:
            G = np.array([h[0]])
        else:
            G = h[2 * n - 2:n - 2:-1]
        G = np.asarray(G, dtype=np.float64)
        grad = 2.0 * G

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-300:
            break

        # Line search
        base_lr = 1.0 / (gnorm * np.sqrt(n))
        accepted = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult
            v = -lr * grad + momentum * momentum_vec
            f_new = _project_simplex(f + v)
            sc = _score(f_new)
            if sc < cur_score or (sc < best_score):
                momentum_vec = 0.9 * momentum_vec - lr * grad
                f = f_new
                cur_score = sc
                accepted = True
                break
            else:
                momentum_vec = 0.5 * momentum_vec

        if not accepted:
            # Random perturbation
            for sig in [1.0/n, 0.3/n, 0.1/n]:
                f_p = _project_simplex(f + rng.normal(0, sig, n))
                sc_p = _score(f_p)
                if sc_p < cur_score:
                    f = f_p
                    cur_score = sc_p
                    momentum_vec = np.zeros(n)
                    break
            else:
                T_factor = max(0.01, T_factor * 0.7)
                if T_factor < 0.005:
                    # Large restart
                    f = _project_simplex(0.7 * best_f + 0.3 * f + rng.normal(0, 0.5/n, n))
                    cur_score = _score(f)
                    T_factor = 0.05
                    momentum_vec = np.zeros(n)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        if it % 200 == 0:
            print(f"  {label}: it={it} cur={cur_score:.8f} best={best_score:.8f}", flush=True)

        if best_score < 1.5020:
            break

    return best_f, best_score


def _generate_initials(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    names = []

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if len(p) == n and np.sum(p) > 0:
            inits.append(_norm(p))
            names.append("parent")
        elif len(p) > 0 and np.sum(p) > 0:
            inits.append(_interp_extend(p, n))
            names.append("parent_interp")

    # Uniform
    inits.append(np.full(n, 1.0 / n))
    names.append("uniform")

    # Geometric tapers
    for rate in [0.5, 1.0, 2.0, 4.0, 6.0, 10.0]:
        inits.append(_make_geometric(n, rate, False))
        names.append(f"geo_{rate}")
        if rate <= 4.0:
            inits.append(_make_geometric(n, rate, True))
            names.append(f"geo_rev_{rate}")

    # Power tapers
    for alpha in [0.3, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(_make_power_taper(n, alpha))
        names.append(f"pow_{alpha}")
    for alpha, beta in [(1.0, 0.5), (2.0, 1.0), (0.5, 1.5), (1.5, 2.0)]:
        inits.append(_make_power_taper(n, alpha, beta))
        names.append(f"pow_{alpha}_{beta}")

    # Quadratic tapers
    for a, b in [(-1, 0), (-2, 0), (-3, 0), (-1, 1), (-2, 1), (-3, 2),
                 (-4, 3), (-1, -0.5), (-2, 1.5), (1, -2), (2, -3)]:
        inits.append(_make_quadratic_taper(n, a, b))
        names.append(f"quad_{a}_{b}")

    # Cubic tapers
    for a, b, c in [(-2, 1, 0), (-3, 2, -1), (-4, 3, -1), (-2, -1, 2),
                    (-3, 0, 1), (-5, 4, -1)]:
        inits.append(_make_cubic_taper(n, a, b, c))
        names.append(f"cub_{a}_{b}_{c}")

    # Exponential tapers
    for a, b in [(-1, 0), (-2, 1), (-3, 2), (-1, -0.5), (-4, 3),
                 (-2, -1), (1, -3), (2, -4)]:
        inits.append(_make_exp_taper(n, a, b))
        names.append(f"exp_{a}_{b}")

    # Random smooth
    for _ in range(8):
        inits.append(_make_diadic(n, rng, smooth=1.0))
        names.append(f"rand_{_}")
    for _ in range(4):
        inits.append(_make_diadic(n, rng, smooth=3.0))
        names.append(f"rand_smooth_{_}")

    return inits, names


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a nonnegative step function minimizing the evaluation score.
    Uses LP-based best response iteration with diverse initializations,
    coarse-to-fine scaling, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0  # Reserve 15s for finalization

    # Get parent if available
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        p_arr = np.where(np.isfinite(p_arr), p_arr, 0.0)
        p_arr = np.maximum(p_arr, 0.0)
        if p_arr.ndim == 1 and p_arr.size >= 16 and float(p_arr.sum()) > 0.0:
            parent = p_arr
            print(f"Parent: len={p_arr.size}, score={_score(_norm(p_arr)):.8f}", flush=True)
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_update(p, sc, label=""):
        nonlocal best_p, best_score
        if p is None or len(p) == 0:
            return
        p = _norm(p)
        if not np.isfinite(sc):
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"*** NEW BEST: score={best_score:.8f} n={len(p)} {label}", flush=True)

    # Phase 1: Coarse-to-fine shape discovery at small n
    print("=" * 60, flush=True)
    print("Phase 1: Coarse shape search", flush=True)
    print("=" * 60, flush=True)

    coarse_n = 200
    coarse_inits, _ = _generate_initials(coarse_n, rng, parent)
    coarse_scored = []
    for p in coarse_inits:
        sc = _score(p)
        if np.isfinite(sc):
            coarse_scored.append((sc, p))
    coarse_scored.sort(key=lambda x: x[0])

    # Run quick LP iterations on top coarse inits
    coarse_time = min(180.0, (deadline - time.monotonic()) * 0.15)
    for sc, p in coarse_scored[:5]:
        if time.monotonic() > deadline - 600:
            break
        remaining = min(40.0, coarse_time - (time.monotonic() - start_time) + 60.0)
        if remaining < 10:
            break
        f_best, sc_best = _lp_iteration(p, coarse_n, rng, remaining, f"coarse_{sc:.4f}")
        try_update(f_best, sc_best, f"coarse n={coarse_n}")

    # Get best coarse shape for interpolation
    coarse_best_p = best_p.copy() if best_p is not None else coarse_scored[0][1] if coarse_scored else np.full(coarse_n, 1.0/coarse_n)

    # Phase 2: Multi-length LP optimization
    print("=" * 60, flush=True)
    print("Phase 2: Multi-length LP optimization", flush=True)
    print("=" * 60, flush=True)

    # Try several lengths
    target_lengths = [500, 700, 1000, 1400]

    # Parent length is important
    if parent is not None:
        plen = len(parent)
        if plen not in target_lengths and 300 <= plen <= 2000:
            target_lengths.append(plen)
            target_lengths.sort()

    # Allocate time to each length
    n_lengths = len(target_lengths)
    time_per_length = min(350.0, max(80.0, (deadline - time.monotonic()) * 0.55 / n_lengths))

    for n in target_lengths:
        if time.monotonic() > deadline - 200:
            break

        # Generate initializations for this length
        # Interpolate from coarse best if available
        if len(coarse_best_p) != n:
            interp_p = _interp_extend(coarse_best_p, n)
        else:
            interp_p = coarse_best_p.copy()

        inits, names = _generate_initials(n, rng, parent)
        # Add interpolated coarse best
        inits.append(interp_p)
        names.append("coarse_interp")

        # Score and sort
        scored = []
        for i, p in enumerate(inits):
            sc = _score(p)
            if np.isfinite(sc):
                scored.append((sc, p, names[i]))
        scored.sort(key=lambda x: x[0])

        print(f"\nLength n={n}: top initializations:", flush=True)
        for sc, p, nm in scored[:8]:
            print(f"  {nm}: {sc:.8f}", flush=True)

        # Try LP on top initializations
        n_tries = min(6, len(scored))
        time_budget_here = min(time_per_length, deadline - time.monotonic() - 150)
        time_per_try = max(15.0, time_budget_here / max(1, n_tries))

        for i in range(n_tries):
            if time.monotonic() > deadline - 150:
                break
            sc_init, p_init, nm = scored[i]
            remaining = min(time_per_try, deadline - time.monotonic() - 150)
            if remaining < 10:
                break
            print(f"  Trying LP on {nm} (init score={sc_init:.6f}, budget={remaining:.0f}s)", flush=True)
            f_best, sc_best = _lp_iteration(p_init, n, rng, remaining, f"n{n}_{nm}")
            try_update(f_best, sc_best, f"n={n} init={nm}")

    # Phase 3: Gradient-based optimization on best solutions
    print("=" * 60, flush=True)
    print("Phase 3: Gradient polish", flush=True)
    print("=" * 60, flush=True)

    if best_p is not None and time.monotonic() < deadline - 60:
        n_best = len(best_p)
        grad_time = min(200.0, deadline - time.monotonic() - 30)
        print(f"  Polishing best (n={n_best}, score={best_score:.8f}, budget={grad_time:.0f}s)", flush=True)
        f_best, sc_best = _gradient_optimize(best_p, n_best, rng, grad_time, "grad_polish")
        try_update(f_best, sc_best, "gradient_polish")

    # Phase 4: Final LP refinement on the best
    print("=" * 60, flush=True)
    print("Phase 4: Final LP refinement", flush=True)
    print("=" * 60, flush=True)

    if best_p is not None and len(best_p) <= 1500 and time.monotonic() < deadline - 30:
        final_time = min(120.0, deadline - time.monotonic() - 10)
        if final_time > 20:
            n_best = len(best_p)
            print(f"  Final LP (n={n_best}, budget={final_time:.0f}s)", flush=True)
            f_best, sc_best = _lp_iteration(best_p, n_best, rng, final_time, "final_lp")
            try_update(f_best, sc_best, "final_lp")

    # Fallback
    if best_p is None:
        best_p = np.full(700, 1.0 / 700)
        best_score = _score(best_p)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(np.sum(best_p))
    if s < 0.01:
        best_p = np.full(700, 1.0 / 700)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: n={len(best_p)}, internal_score={_score(best_p):.10f}", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_p]
