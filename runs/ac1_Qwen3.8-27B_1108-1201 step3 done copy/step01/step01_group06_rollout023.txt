"""
Minimize 2n*max(self-conv)/sum^2 using accelerated soft-subgradient with
Nesterov momentum and temperature annealing. FFT-based cross-correlation
for O(n log n) subgradient computation per step.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):

"""
Minimize 2*n*max(self-convolution)/sum(a)^2 using accelerated soft-subgradient
with Nesterov momentum, temperature annealing, and FFT-based cross-correlation
for O(n log n) subgradient computation. Includes LP-based polishing via cvxpy
for the best found solution, and periodic authoritative verification.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import scipy.sparse as sp
import cvxpy as cp


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size].copy()


def compute_soft_grad(a, c, T, fsize):
    """Compute soft-subgradient via softmax + FFT cross-correlation."""
    n = len(a)
    c_shift = c - c.max()
    # Clamp to avoid overflow
    c_shift = np.clip(c_shift / T, -500, 500)
    w = np.exp(c_shift)
    w_sum = w.sum()
    if w_sum <= 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    # Cross-correlation: grad[j] = 2 * sum_k w[k] * a[k-j]
    flip_a = a[::-1]
    W = fft(w, fsize)
    FA = fft(flip_a, fsize)
    cc = np.real(ifft(W * FA))
    grad = 2.0 * cc[n - 1: 2 * n - 1]
    return grad


def run_soft_nesterov(a_init, n, S, max_time, rng, verbose=True):
    """Nesterov-accelerated soft-subgradient with temperature annealing."""
    a = safe_normalize(a_init, n)
    fsize = 4 * n
    y = a.copy()  # Nesterov extrapolate
    t_mom = 1.0  # momentum parameter
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    step = 0

    # Temperature schedule
    T_max = 8.0
    T_min = 0.05
    # Base learning rate
    base_lr = S / n * 2.0

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Annealing: exponential
        T = T_min + (T_max - T_min) * np.exp(-5.0 * progress)

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        y_new = y + beta * (y - a)
        a = y_new
        t_mom = t_new

        # Project y into feasible set (keeps Nesterov point valid)
        y = project_simplex(y, S)

        # Compute objective and subgradient
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = compute_soft_grad(a, c, T, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Learning rate: proportional to T (smoothness scales as 1/T)
        lr = base_lr * T / (gnorm + 1e-10) * 0.5
        lr = min(lr, S / (2.0 * np.sqrt(step + 1)))

        a = project_simplex(a - lr * grad, S)
        step += 1

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - start
            print("  soft_nest step=%d maxc=%.8f T=%.4f time=%.1f" %
                  (step, best_score, T, elapsed))

    return best_local, best_score


def run_topk_subgrad(a_init, n, S, max_time, rng, top_k=20):
    """Top-k hard subgradient with momentum (faster, less smooth)."""
    a = safe_normalize(a_init, n)

    y = a.copy()

    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()

    base_lr = S / n * 1.5

        k_use = max(3, int(top_k * np.exp(-3.0 * progress)))

        beta = (t_mom - 1.0) / t_new
        a = y + beta * (y - a)
        t_mom = t_new
        y = project_simplex(y, S)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:

            best_local = a.copy()

        step += 1

        if step % 10000 == 0:

def compute_soft_from_weights(a, w, fsize):
    """Compute subgradient from a pre-computed weight vector w (length 2n-1)."""
    n = len(a)
    flip_a = a[::-1]
    W = fft(w, fsize)
    FA = fft(flip_a, fsize)
    cc = np.real(ifft(W * FA))
    return 2.0 * cc[n - 1: 2 * n - 1]


def solve_lp_polish(f, max_iter=5, verbose=True):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M*1, g >= 0.
    Returns the optimal g (normalized)."""
    n = len(f)
    if n > 4096:
        # Too large for explicit LP, skip
        return None
    M = fft_convolve_self(f).max()
    if M <= 0:
        return None

    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    # Use CSC format for efficiency
    data = np.zeros(n * n)  # over-allocate
    row_idx = np.zeros(n * n, dtype=np.int32)
    col_idx = np.zeros(n * n, dtype=np.int32)
    ptr = 0
    for j in range(n):
        start_row = j
        end_row = min(j + n, 2 * n - 1)
        count = end_row - start_row
        data[ptr:ptr + count] = f[j]
        row_idx[ptr:ptr + count] = np.arange(start_row, end_row)
        col_idx[ptr:ptr + count] = j
        ptr += count

    A = sp.csc_matrix((data[:ptr], (row_idx[:ptr], col_idx[:ptr])),
                      shape=(2 * n - 1, n))

    # Set up LP
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M, g >= 0]

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
            except Exception:
                return None

    if g.value is None:
        return None

    g_val = np.asarray(g.value, dtype=np.float64)
    g_val = np.maximum(g_val, 0.0)
    if g_val.sum() < 1e-10:
        return None
    return g_val


def run_lp_iteration(f, n, S, max_time, max_lp_calls=20, verbose=True):
    """Iterate the LP-based update: f = (1-t)*f + t*g_normalized."""
    a = safe_normalize(f, n)
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    lp_count = 0

    while time.monotonic() - start < max_time and lp_count < max_lp_calls:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        g = solve_lp_polish(a, max_iter=1)
        if g is None:
            break

        # Normalize g to sum = S
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g * (S / g_sum)

        # Update with decreasing step
        t = 0.5 / (1.0 + 0.3 * lp_count)
        a = (1.0 - t) * a + t * g
        a = project_simplex(a, S)
        lp_count += 1

        if verbose:
            new_c = fft_convolve_self(a)
            print("  LP iter=%d maxc=%.8f step_t=%.4f time=%.1f" %
                  (lp_count, new_c.max(), t, time.monotonic() - start))

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng):
    """Mirror descent (multiplicative weights) as a diverse alternative."""
    a = safe_normalize(a_init, n)

    final_score = evaluate_sequence(final_a.tolist())
    print("\n=== FINAL: score=%.6f, n=%d, sum=%.4f ===" %
          (final_score, len(final_a), final_a.sum()))

    return [float(x) for x in final_a]
