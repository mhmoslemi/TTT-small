"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set gradient descent with backtracking line search,
edge-aware initialization, spectral perturbation for local minimum escape,
and LP-based refinement. Combines multiple strategies within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _grad_active(a, top_k=None):
    """Compute gradient using only top-k active lags. Returns (gradient, c)."""
    n = len(a)
    c = np.convolve(a, a)
    if top_k is None:
        top_k = min(50, 2 * n - 1)
    M = c.max()
    if M < 1e-30:
        return np.zeros(n), c
    # Get top-k indices
    k_indices = np.argpartition(c, -top_k)[-top_k:]
    # Weight by activity level
    weights = np.zeros(2 * n - 1)
    for k in k_indices:
        weights[k] = (c[k] / M) ** 2
    # Compute gradient: grad[j] = sum_k weights[k] * 2 * a[|k-j|] for valid |k-j|
    grad = np.zeros(n)
    for k in k_indices:
        wk = weights[k]
        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += wk * 2.0 * a[abs(k - j)]
    grad -= grad.mean()
    return grad, c


def _grad_softmax(a, tau):
    """Compute gradient using soft-max weighting over all lags."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        k_lo = 0
        k_hi = min(n, 2 * n - 1 - j)
        if k_hi > k_lo:
            grad[j] = 2.0 * np.dot(a[k_lo:k_hi], w[j + k_lo:j + k_hi])
    grad -= grad.mean()
    return grad, c


def _step_with_backtrack(a, grad, max_lr=0.5, min_lr=1e-6, max_backtracks=15):
    """Take a gradient step with backtracking line search. Returns new a."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    f0 = _compute_obj(a)
    lr = max_lr
    for _ in range(max_backtracks):
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S_t)
        f1 = _compute_obj(a_new)
        if f1 < f0:
            return a_new
        lr *= 0.5
        if lr < min_lr:
            break
    # If no improvement found, take a tiny step
    a_new = a - min_lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, S_t)
    return a_new


def _optimize_gradient(a0, deadline, max_iters=5000, method='active'):
    """Optimize using gradient descent with active-set or softmax gradient."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = _normalize(a, S_t)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score

    # Annealing schedule
    lr_init = 0.3
    top_k_init = 20
    top_k_final = 200

    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        # Anneal learning rate and top_k
        frac = it / max_iters
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, 0.005)
        top_k = int(top_k_init + (top_k_final - top_k_init) * frac)

        if method == 'active':
            grad, c = _grad_active(a, top_k=top_k)
        else:
            tau = 10.0 + 40.0 * frac
            grad, c = _grad_softmax(a, tau)

        a = _step_with_backtrack(a, grad, max_lr=lr)

        if (it + 1) % 50 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    # Final check
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def _spectral_perturb(a, rng, strength=0.3):
    """Apply a spectral perturbation to escape local minima."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    L = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(L)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    # Add random phase perturbation
    phase_perturb = rng.standard_normal(len(A)) * strength * 0.1
    A_perturbed = A * np.exp(1j * phase_perturb)
    # Also slightly modify magnitude
    mag_mod = 1.0 + rng.standard_normal(len(A)) * strength * 0.05
    mag_mod = np.maximum(mag_mod, 0.1)
    A_perturbed = A_perturbed * mag_mod / np.abs(A_perturbed + 1e-30) * np.abs(A)
    # Reconstruct
    a_new = np.fft.irfft(A_perturbed, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-12:
        a_new = _normalize(a_new, S_t)
    else:
        a_new = a.copy()
    return a_new


def _peak_reduction_move(a, deadline):
    """Directly reduce the a[j] values that contribute most to the peak."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t).copy()
    c = np.convolve(a, a)
    k_star = int(c.argmax())
    # Find which j contribute most to c[k_star]
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    contributions = np.zeros(n)
    for j in range(lo, hi + 1):
        contributions[j] = a[abs(k_star - j)]
    # Reduce high-contribution a[j], increase low-contribution ones
    total_contrib = contributions.sum()
    if total_contrib < 1e-12:
        return a
    # Proportional reduction
    reduce_frac = 0.1
    a[lo:hi+1] -= reduce_frac * contributions[lo:hi+1]
    a = np.maximum(a, 0.0)
    a = _normalize(a, S_t)
    return a


def _lp_refine(f, M, n_vars=None):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    if n_vars is not None:
        n = n_vars
    if n > 2048:
        return None
    c_obj = -np.ones(n)
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for m in range(n_constraints):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    b = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_based_optimize(a0, deadline, max_iters=100):
    """LP-based refinement using the state-of-the-art approach."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0), S_t)
    best = a.copy()
    best_score = _compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        c = np.convolve(a, a)
        M = float(c.max())
        g = _lp_refine(a, M)
        if g is None:
            break
        improved = False
        for t in [0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S_t)
            s_new = _compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        if not improved:
            # Try a small gradient step
            grad, _ = _grad_active(a, top_k=30)
            a = _step_with_backtrack(a, grad, max_lr=0.05)
            s = _compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
    return best, best_score


def _edge_heavy_profile(n, alpha=1.0):
    """Create an edge-heavy profile: large at edges, small in center."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Distance from edge (0 at edges, max at center)
    dist = np.minimum(i, n - 1 - i) / max(n - 1, 1)
    # Profile: 1 at edges, decays toward center
    profile = (1.0 - dist) ** alpha
    # Ensure non-negative and not all zero
    if profile.sum() < 1e-10:
        profile = np.ones(n)
    return profile

    # Use global if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            for L in [256, 384, 512, min(768, len(gs)), min(1024, len(gs))]:
                if L <= len(gs):
                    starts.append((gs[:L].copy(), f"global_n{L}"))
            # Also try the full length
            if len(gs) not in [256, 384, 512, 768, 1024]:
                starts.append((gs.copy(), f"global_full_n{len(gs)}"))
    except (NameError, TypeError):
        pass

    # Random starts
    for n in [256, 384, 512]:
        starts.append((rng.random(n) * 0.5 + 0.25, f"random_n{n}"))

    # Edge-heavy profiles (motivated by theory: edges should be ~sqrt(M))
    for n in [256, 512, 768]:
        for alpha in [0.5, 1.0, 1.5, 2.0]:
            starts.append((_edge_heavy_profile(n, alpha), f"edge_a{alpha}_n{n}"))

    # Cosine / structured profiles
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        # Raised cosine
        starts.append((0.5 + 0.5 * np.cos(2 * np.pi * i / n), f"raisedcos_n{n}"))
        # Double edge
        edge = np.minimum(i, n - 1 - i).astype(float)
        starts.append((np.exp(-edge / (n * 0.15)), f"doubleedge_n{n}"))
        # Linear ramp (edge-heavy)
        starts.append((np.minimum(i.astype(float), (n-1-i).astype(float)) / (n * 0.5) + 0.1,
                       f"ramp_n{n}"))
        # Alternating (to break symmetry)
        alt = np.ones(n)
        alt[::2] = 0.3
        starts.append((alt * np.exp(-np.minimum(i, n-1-i).astype(float) / (n*0.2)),
                       f"alt_n{n}"))

    # Sort by length (shorter first for quick wins)
    starts.sort(key=lambda x: len(x[0]))

    # --- Run optimization on each start ---
    n_starts = len(starts)
    time_per_start = (deadline - t_start - 30.0) / max(n_starts, 1)
    print(f"  {n_starts} starting sequences, ~{time_per_start:.0f}s each")

    for idx, (a0, label) in enumerate(starts):
        if time.monotonic() > deadline - 30.0:
            break
        # Time-limited: don't spend too long on any single start
        remaining = deadline - time.monotonic()
        alloc = min(time_per_start * 1.5, remaining - 30.0)
        if alloc < 15.0:
            break
        print(f"\n[{idx+1}/{n_starts}] Starting: {label}")
        _try_and_update(a0, label)

    # --- Final refinement of the best sequence ---
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        print(f"\n=== Final refinement (current best: {best_score:.6f}) ===")
        for pass_i in range(10):
            if time.monotonic() > deadline - 10.0:
                break
            # Alternate between gradient and LP
            if pass_i % 2 == 0:
                seq, score = _optimize_gradient(best_seq, deadline - 10.0,
                                                max_iters=1500, method='active')
            else:
                if len(best_seq) <= 1024:
                    seq, score = _lp_based_optimize(best_seq, deadline - 10.0, max_iters=30)
                else:
                    seq, score = _optimize_gradient(best_seq, deadline - 10.0,
                                                    max_iters=1500, method='softmax')
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Refine pass {pass_i}: score={best_score:.6f}")
            else:
                # Try a perturbation to escape
                if time.monotonic() < deadline - 20.0:
                    perturbed = _spectral_perturb(best_seq, rng, strength=0.3)
                    seq2, score2 = _optimize_gradient(perturbed, deadline - 10.0,
                                                      max_iters=1000, method='active')
                    if score2 < best_score:
                        best_score = score2
                        best_seq = seq2.copy()
                        print(f"  Perturb+optimize pass {pass_i}: score={best_score:.6f}")

    # --- Final evaluation ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function that optimizes a non-negative sequence to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    Uses active-set gradient descent, spectral perturbation, and LP refinement.
    """
    return _propose_candidate_impl(seed=seed, budget_s=budget_s)
