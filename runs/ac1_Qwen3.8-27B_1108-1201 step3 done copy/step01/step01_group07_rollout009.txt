import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_subgradient(a, conv, t=50.0):
    """Compute smoothed subgradient of max(conv) w.r.t. a."""
    n = len(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n)
    
    # Softmax weights
    scaled = t * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Subgradient
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg[j_start:j_end+1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    
    return sg

def adam_optimize(a0, time_limit, t_param=50.0, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the probability simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_step = 0
    t0 = time.monotonic()
    
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t0 < time_limit:
        t_step += 1
        conv = fast_convolve(a)
        max_conv = conv.max()
        
        sg = compute_subgradient(a, conv, t=t_param)
        
        # Center the subgradient (we're on the simplex, so the direction
        # component along (1,1,...,1) doesn't matter)
        sg_centered = sg - sg.mean()
        
        m = beta1 * m + (1 - beta1) * sg_centered
        v = beta2 * v + (1 - beta2) * sg_centered**2
        m_hat = m / (1 - beta1**t_step)
        v_hat = v / (1 - beta2**t_step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a = a_new / a_sum
        
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            # Reduce learning rate and change temperature
            lr *= 0.5
            t_param *= 1.5
            patience = 0
            if lr < 0.0001:
                break
    
    return best_a, best_obj

                pert = rng.dirichlet(np.ones(n) * 0.1)
                a = alpha * pert + (1 - alpha) * best_a
                a = normalize_nonneg(a)
                conv = fast_convolve(a)
                # Reset params
                current_eta = max(0.3, eta_init * 0.7)
                current_t = t_init * (0.5 + rng.random())
                current_k = max(2, top_k_init + int(rng.integers(0, 4)))
            else:
                current_eta *= 0.6
                current_t *= 1.5
                current_k = min(current_k + 2, 20)
            patience = 0

            if current_eta < 0.02:
                current_eta = 0.4
                current_t = t_init * 0.5
                current_k = 2
                # Reset to best
                a = best_a.copy()

    return best_a, best_obj


def optimize_adam(a0, time_limit, t_init=60.0, lr_init=0.1, top_k_init=5, seed=123):
    """Adam optimizer on the simplex with peak-focused subgradient."""
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)

    conv = fast_convolve(a)
    best_obj = obj_from_conv(a, conv)
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    current_t = t_init
    current_lr = lr_init
    current_k = top_k_init
    patience = 0
    max_patience = 250
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        sg = compute_subgradient(a, conv, current_k, current_t)
        sg_c = sg - sg.mean()

        m = 0.9 * m + 0.1 * sg_c
        v = 0.999 * v + 0.001 * sg_c ** 2
        m_hat = m / (1.0 - 0.9 ** step)
        v_hat = v / (1.0 - 0.999 ** step)

        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = normalize_nonneg(a_new)

        conv = fast_convolve(a)
        obj = obj_from_conv(a, conv)

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            current_lr *= 0.5
            current_t *= 1.4
            current_k = min(current_k + 3, 25)
            patience = 0

            # Occasional kick
            if rng.random() < 0.3:
                alpha = 0.2 + 0.3 * rng.random()
                pert = rng.dirichlet(np.ones(n) * 0.2)
                a = alpha * pert + (1 - alpha) * best_a
                a = normalize_nonneg(a)
                m = np.zeros(n)
                v = np.zeros(n)
                step = 0
                current_lr = lr_init * 0.5

            if current_lr < 0.002:
                a = best_a.copy()
                current_lr = 0.05
                current_t = t_init * 0.5
                current_k = 3
                m = np.zeros(n)
                v = np.zeros(n)
                step = 0
                patience = 0

    return best_a, best_obj


def optimize_projected(a0, time_limit, lr_init=0.05, t_init=50.0, top_k_init=5, seed=77):
    """Projected subgradient with adaptive step size."""
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)

    conv = fast_convolve(a)
    best_obj = obj_from_conv(a, conv)
    best_a = a.copy()

    current_lr = lr_init
    current_t = t_init
    current_k = top_k_init
    patience = 0
    max_patience = 200
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    # Polyak step size tracking
    best_obj_tracking = best_obj

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        sg = compute_subgradient(a, conv, current_k, current_t)

        a_new = a - current_lr * sg
        a = project_simplex(a_new)

        conv = fast_convolve(a)
        obj = obj_from_conv(a, conv)

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            current_lr *= 0.7
            current_t *= 1.3
            current_k = min(current_k + 2, 20)
            patience = 0

            if current_lr < 0.0005:
                alpha = 0.3 + 0.3 * rng.random()
                pert = rng.dirichlet(np.ones(n) * 0.2)
                a = alpha * pert + (1 - alpha) * best_a
                a = normalize_nonneg(a)
                current_lr = lr_init * 0.5
                current_t = t_init * 0.7

    return best_a, best_obj


def local_search(a0, time_limit, seed=99):
    """Local search: try small perturbations and accept improvements."""
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = obj_from_conv(a, conv)
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit:
        step += 1
        # Random perturbation: pick 2-4 indices and redistribute
        n_pert = int(rng.integers(2, min(5, n + 1)))
        idx = rng.choice(n, size=n_pert, replace=False)

        # Get current values at these indices
        old_vals = a[idx].copy()
        # New random distribution (Dirichlet over these indices)
        new_vals = rng.dirichlet(np.ones(n_pert) * 0.5)
        total_old = old_vals.sum()
        new_vals *= total_old

        a_new = a.copy()
        a_new[idx] = new_vals
        a_new = normalize_nonneg(a_new)

        conv_new = fast_convolve(a_new)
        obj_new = obj_from_conv(a_new, conv_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Occasionally accept worse solutions (simulated annealing style)
            if no_improve > 50 and rng.random() < 0.05:
                a = a_new
                conv = conv_new

        if no_improve >= max_no_improve:
            # Reset to best and try different perturbation size
            a = best_a.copy()
            no_improve = 0
            max_no_improve = int(max_no_improve * 0.7) + 20
            if max_no_improve < 50:
                break

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for a given length."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = x - n * 0.5

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Gaussian (narrow)
    g = np.exp(-0.5 * (xm / (0.15 * n)) ** 2)
    inits.append(('gauss_narrow', g / g.sum()))

    # Gaussian (medium)
    g = np.exp(-0.5 * (xm / (0.3 * n)) ** 2)
    inits.append(('gauss_med', g / g.sum()))

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 0)
    if rc.sum() > 0:
        inits.append(('raised_cos', rc / rc.sum()))

    # Bimodal (edges)
    bi = np.zeros(n)
    w = max(3, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    # Smooth the edges
    for i in range(min(w, 5)):
        bi[i] *= (i + 1) / w
        bi[-1 - i] *= (i + 1) / w
    bi = np.maximum(bi, 0)
    if bi.sum() > 0:
        inits.append(('bimodal', bi / bi.sum()))

    # Quadratic bump
    qb = 1.0 - (xm / (0.5 * n)) ** 2
    qb = np.maximum(qb, 0)
    if qb.sum() > 0:
        inits.append(('quadratic', qb / qb.sum()))

    # Random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(('random', r))

    # W-shape (two peaks)
    if n >= 8:
        wshape = np.zeros(n)
        wshape[n // 4] = 1.0
        wshape[3 * n // 4] = 1.0
        # Smooth
        kernel = np.ones(3) / 3
        wshape = np.convolve(wshape, kernel, mode='same')
        wshape = np.maximum(wshape, 0)
        if wshape.sum() > 0:
            inits.append(('wshape', wshape / wshape.sum()))

    # Parent-based
    if parent is not None:
        if parent.size == n:
            pp = np.maximum(parent, 0).astype(np.float64)
            if pp.sum() > 1e-10:
                inits.append(('parent', pp / pp.sum()))
        elif parent.size > n:
            pp = parent[:n].astype(np.float64)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits.append(('parent_trunc', pp / pp.sum()))
        else:
            # Interpolate
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits.append(('parent_interp', pp / pp.sum()))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid multi-phase optimization:
    Phase 1: Quick scan over lengths and initializations
    Phase 2: Deep optimization on top candidates using multiple optimizers
    Phase 3: Local search polish on the best candidate
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Loaded parent sequence: n={p.size}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Quick scan
    print("=== Phase 1: Quick scan ===")
    lengths = [64, 128, 256, 512]
    if budget_s > 1200:
        lengths.append(1024)

    scan_configs = []
    for n in lengths:
        inits = generate_initializations(n, rng, parent)
        for name, a0 in inits:
            scan_configs.append((n, name, a0))

    n_configs = len(scan_configs)
    phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.3)
    time_per_scan = max(4.0, phase1_budget / max(1, n_configs))

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_configs):
        remaining = deadline - time.monotonic() - 60
        if remaining < 10:
            break
        t_alloc = min(time_per_scan, remaining * 0.15)
        if t_alloc < 3:
            t_alloc = 3

        a_opt, obj_opt = optimize_mirror(a0, time_limit=t_alloc, seed=i + 1)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt, obj_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep optimization ===")
    # Select top configs with length diversity
    top_configs = []
    seen = set()
    for score, n, name, a_opt, obj_opt in scan_results:
        if n not in seen or len(top_configs) < 4:
            top_configs.append((n, name, a_opt.copy()))
            seen.add(n)
        if len(top_configs) >= 10:
            break

    # Always include the current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    # Add parent at its original length if not already included
    if parent is not None and parent.size not in [c[0] for c in top_configs]:
        top_configs.append((parent.size, 'parent_full', parent.copy()))

    remaining = deadline - time.monotonic() - 120
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)) / 3.0)

    for i, (n, name, a0) in enumerate(top_configs):
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break

        t_single = min(time_per_deep, remaining / 3.0)
        if t_single < 5:
            t_single = 5

        for opt_idx, (opt_name, opt_func, t_p, lr_p, k_p) in enumerate([
            ('mirror', optimize_mirror, 50.0, 0.6, 3),
            ('adam', optimize_adam, 80.0, 0.08, 5),
            ('proj', optimize_projected, 60.0, 0.04, 5),
        ]):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(t_single, deadline - time.monotonic() - 80)
            if t_alloc < 3:
                break

            if opt_name == 'mirror':
                a_opt, obj_opt = opt_func(a0, time_limit=t_alloc, t_init=t_p,
                                          eta_init=lr_p, top_k_init=k_p,
                                          seed=seed + i * 100 + opt_idx + 1)
            elif opt_name == 'adam':
                a_opt, obj_opt = opt_func(a0, time_limit=t_alloc, t_init=t_p,
                                          lr_init=lr_p, top_k_init=k_p,
                                          seed=seed + i * 100 + opt_idx + 1)
            else:
                a_opt, obj_opt = opt_func(a0, time_limit=t_alloc, t_init=t_p,
                                          lr_init=lr_p, top_k_init=k_p,
                                          seed=seed + i * 100 + opt_idx + 1)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep: n={n}, {name}, {opt_name}, score={score:.6f}")

    # Phase 3: Local search polish
    print("=== Phase 3: Local search polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 15
        n_best = len(best_sequence)

        # Run local search for a portion of remaining time
        ls_time = min(remaining * 0.5, 200.0)
        a_opt, obj_opt = local_search(best_sequence, time_limit=ls_time, seed=seed + 999)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Local search: score={score:.6f}")

        # Final mirror descent polish with very low temperature
        if time.monotonic() < deadline - 20:
            remaining2 = deadline - time.monotonic() - 10
            t_single = min(remaining2 * 0.6, 150.0)
            a_opt, obj_opt = optimize_mirror(best_sequence, time_limit=t_single,
                                             t_init=200.0, eta_init=0.3,
                                             top_k_init=2, seed=seed + 777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final polish: score={score:.6f}")

        # One more local search round
        if time.monotonic() < deadline - 10:
            remaining3 = deadline - time.monotonic() - 5
            if remaining3 > 10:
                a_opt, obj_opt = local_search(best_sequence, time_limit=remaining3,
                                              seed=seed + 555)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final local: score={score:.6f}")

    if best_sequence is None:
        # Fallback
        n_fb = 128
        best_sequence = np.ones(n_fb) / n_fb
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"  Fallback: score={best_score:.6f}")

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
