"""
Improved subgradient optimization with FFT-accelerated gradient computation,
Nesterov momentum, adaptive step size, and local refinement for minimizing
the normalized autocorrelation peak of a non-negative sequence.
"""

import numpy as np
import time

def _compute_score(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)

def _fft_convolve(a, b, N=None):
    """Linear convolution via FFT."""
    if N is None:
        N = len(a) + len(b) - 1
    return np.real(np.fft.ifft(np.fft.fft(a, N) * np.fft.fft(b, N)))

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _make_anti_symmetric(n, eps, pattern='linear', rng=None):
    """Create anti-symmetric starting points."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0  # in [-1, 1]
    if pattern == 'linear':
        a = 1.0 + eps * t
    elif pattern == 'cosine':
        a = 1.0 + eps * np.cos(np.pi * t)
    elif pattern == 'quadratic':
        a = 1.0 + eps * t * np.abs(t)
    elif pattern == 'random':
        if rng is None:
            rng = np.random.default_rng(0)
        a = 1.0 + eps * (2.0 * rng.random(n) - 1.0)
    else:
        a = 1.0 + eps * t
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def _optimize_nesterov(a, deadline, temp_init=1.0, temp_end=100.0,
                        eta_init=None, beta_momentum=0.9,
                        eta_decay=0.9999, min_eta=1e-12,
                        check_interval=50):
    """
    Nesterov-accelerated subgradient descent with FFT-accelerated gradients.
    Minimizes max_k b_k subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    N = 2 * n - 1
    
    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if eta_init is None:
        eta_init = 0.5 / np.sqrt(n)
    
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    momentum = np.zeros(n)
    eta = eta_init
    it = 0
    no_improve = 0
    temp = temp_init
    
    while time.monotonic() < deadline:
        # Compute b = convolve(a, a) via FFT
        fa = np.fft.fft(a, N)
        b = np.real(np.fft.ifft(fa * fa))
        
        # Compute current score
        cur_score = 2.0 * n * np.max(b)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 10000:
            break
        
        # Anneal temperature
        progress = min(no_improve / 5000.0, 1.0)  # or use iteration count
        temp = temp_init + (temp_end - temp_init) * min(it / 10000.0, 1.0)
        
        # Compute softmax weights over peaks
        b_shifted = b - np.max(b)
        w = np.exp(b_shifted / temp)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(N) / N
        else:
            w = w / w_sum
        
        # Compute subgradient: 2 * convolve(w, a)[n-1 : 2*n-1]
        fw = np.fft.fft(w, N)
        wa = np.real(np.fft.ifft(fw * fa))
        subgrad = 2.0 * wa[n-1:2*n-1]
        
        # Center for sum constraint
        subgrad -= np.mean(subgrad)
        
        # Nesterov momentum update
        momentum = beta_momentum * momentum + (1 - beta_momentum) * subgrad
        a = a - eta * momentum
        
        # Project onto simplex
        a = _project_simplex(a, 1.0)
        
        eta *= eta_decay
        it += 1
        
        if it % (check_interval * 100) == 0:
            print(f"  Iter {it}: score={best_score:.8f}, eta={eta:.2e}, temp={temp:.2f}")
    
    return best, best_score

    fa = np.fft.fft(a, N)
    b = np.real(np.fft.ifft(fa * fa))
    
    # Try perturbations
    rng = np.random.default_rng(123)
    delta = 0.01 / n  # small perturbation
    
    while time.monotonic() < deadline:

        i_inc = rng.integers(0, n)
        i_dec = rng.integers(0, n)
        if i_inc == i_dec:
            continue
        
        # Try moving mass from i_dec to i_inc
        for sign in [1, -1]:
            a_new = a.copy()
            a_new[i_inc] += sign * delta
            a_new[i_dec] -= sign * delta
            if a_new[i_inc] < 0 or a_new[i_dec] < 0:
                continue

            # Only need to check if max increases
            # b changes at specific lags
            # For efficiency, just recompute full b via FFT

            b_new = np.real(np.fft.ifft(fa_new * fa_new))

                best = a_new.copy()
                best_score = score_new
                a = a_new
                fa = fa_new
                b = b_new
    
    return best, best_score

"""
FFT-accelerated Nesterov subgradient optimization with adaptive temperature annealing,
diverse anti-symmetric starting points, and local coordinate refinement for minimizing
the normalized autocorrelation peak of a non-negative sequence.
Key innovation: the subgradient of max_k b_k is computed as 2*convolve(weights, a),
enabling O(n log n) per iteration via FFT.

import numpy as np
import time


def _compute_fft_convolve(a, N):
    """Compute linear convolution a*a via FFT, returning array of length N."""
    fa = np.fft.fft(a, N)
    return np.real(np.fft.ifft(fa * fa))


def _compute_score_fast(a, N):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = _compute_fft_convolve(a, N)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_anti_symmetric(n, eps, pattern='linear', rng=None):
    """Create anti-symmetric starting point with specified pattern."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    if pattern == 'linear':
        a = 1.0 + eps * t
    elif pattern == 'cosine':
        a = 1.0 + eps * np.cos(np.pi * t)
    elif pattern == 'sine':
        a = 1.0 + eps * np.sin(np.pi * t)
    elif pattern == 'quadratic':
        a = 1.0 + eps * t * np.abs(t)
    elif pattern == 'cubic':
        a = 1.0 + eps * t * t * t
    elif pattern == 'random_smooth':
        if rng is None:
            rng = np.random.default_rng(0)
        # Generate smooth random profile
        freqs = rng.integers(1, max(2, n // 10), size=3)
        a = np.ones(n)
        for f in freqs:
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(-1, 1)
            a += amp * eps * np.cos(2 * np.pi * f * i / n + phase)
    elif pattern == 'random':
        if rng is None:
            rng = np.random.default_rng(0)
        a = 1.0 + eps * (2.0 * rng.random(n) - 1.0)
    else:
        a = 1.0 + eps * t
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _optimize_nesterov(a_init, deadline, temp_init=0.5, temp_end=50.0,
                       eta_init=None, beta_mom=0.9, eta_decay=0.99995,
                       min_eta=1e-14, verbose=False, restart_every=5000):
    """
    Nesterov-accelerated subgradient method with FFT-accelerated gradient.
    Minimizes max_k b_k subject to a >= 0, sum(a) = 1.
    The subgradient of max_k b_k is approximated via softmax-weighted peaks:
      subgrad = 2 * convolve(softmax(b/tau), a) [restricted to first n elements]
    This is computed in O(n log n) via FFT.
    """
    n = len(a_init)
    N = 2 * n - 1

    a = np.asarray(a_init, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    if eta_init is None:
        eta_init = 0.3 / np.sqrt(n)

    best = a.copy()
    best_score = _compute_score_fast(a, N)

    momentum = np.zeros(n)
    prev_momentum = np.zeros(n)
    eta = eta_init
    it = 0
    no_improve = 0
    temp = temp_init

    t0 = time.monotonic()

    while time.monotonic() < deadline:
        # Compute b = convolve(a, a) via FFT
        fa = np.fft.fft(a, N)
        b = np.real(np.fft.ifft(fa * fa))
        b_max_val = np.max(b)
        cur_score = 2.0 * n * b_max_val

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: increase over time to explore more peaks
        frac = min(float(it) / float(max(restart_every, 1)), 1.0)
        temp = temp_init + (temp_end - temp_init) * frac

        # Softmax weights over all peaks
        b_shifted = b - b_max_val
        w = np.exp(b_shifted / temp)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(N) / N
        else:
            w /= w_sum

        # Subgradient: 2 * convolve(w, a)[n-1 : 2*n-1]
        fw = np.fft.fft(w, N)
        wa = np.real(np.fft.ifft(fw * fa))
        subgrad = 2.0 * wa[n - 1:2 * n - 1]

        # Center for sum constraint
        subgrad -= np.mean(subgrad)

        # Nesterov momentum
        y = a + beta_mom * (a - prev_momentum)  # Nesterov lookahead
        # Actually standard FISTA: y = x_k + (t_{k-1}-1)/t_k * (x_k - x_{k-1})
        # Simplified: use heavy ball momentum
        momentum = beta_mom * momentum + subgrad
        a_new = a - eta * momentum

        # Project onto simplex
        a_new = _project_simplex(a_new, 1.0)

        # Update
        prev_momentum = a
        a = a_new
        eta *= eta_decay
        it += 1

        # Periodic restart of momentum if stuck
        if no_improve >= restart_every:
            momentum *= 0.1
            no_improve = 0
            eta = max(eta * 2.0, min_eta)

        if eta < min_eta:
            break

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Iter {it}, t={elapsed:.1f}s, score={best_score:.8f}, "
                  f"eta={eta:.2e}, temp={temp:.2f}, no_improve={no_improve}")

    return best, best_score

def _local_refine(a_init, deadline, rng=None):

    Local refinement via coordinate pair swaps with FFT-based score evaluation.
    Tries to move small amounts of mass between coordinates to reduce the peak.
    """

    delta = base_delta
    n_trials = 0
    n_accepts = 0

    while time.monotonic() < deadline:

        # The peak at location k* is sensitive to a[i] via a[k*-i]
        # So coordinates i where a[k*-i] is large are most impactful
        k_star = peak_loc
        if 0 <= k_star < N:
            # Sensitivity: |a[k_star - i]| for valid i
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)

                sens = np.abs(a[k_star - np.arange(lo, hi + 1)])
                # Pick from sensitive coordinates

                    i_inc = lo + idx
                else:
                    i_inc = rng.integers(0, n)
            else:
                i_inc = rng.integers(0, n)
        else:
            i_inc = rng.integers(0, n)

        i_dec = rng.integers(0, n)
        if i_inc == i_dec:
            continue

        # Try moving mass
        for sign in [1, -1]:
            move = sign * delta
            if a[i_inc] + move < 0 or a[i_dec] - move < 0:
                continue

            a[i_inc] += move
            a[i_dec] -= move

            # Quick score check via FFT
            fa = np.fft.fft(a, N)
            b_new = np.real(np.fft.ifft(fa * fa))
            score_new = 2.0 * n * np.max(b_new)

            if score_new < best_score - 1e-13:
                best = a.copy()
                best_score = score_new
                b = b_new
                peak_loc = np.argmax(b)
                n_accepts += 1
                # Slightly increase delta on success
                delta = min(delta * 1.1, base_delta * 20)
            else:
                # Revert
                a[i_inc] -= move
                a[i_dec] += move
                # Decrease delta on failure
                delta = max(delta * 0.95, base_delta * 0.1)

        # Periodically update peak location
        if n_trials % 1000 == 0:
            peak_loc = np.argmax(_compute_fft_convolve(a, N))

    if n_trials > 0:
        print(f"    Local refine: {n_trials} trials, {n_accepts} accepts, "
              f"final score={best_score:.8f}")

    return best, best_score

            print(f"  Start {i+1}/{n_starts}: NEW BEST {global_best_score:.8f}")
        else:
            print(f"  Start {i+1}/{n_starts}: {best_s:.8f} (best={global_best_score:.8f})")

    # Phase 2: Adaptive multi-round refinement (25% of time)
    phase2_deadline = time.monotonic() + 0.25 * (total_deadline - time.monotonic())
    print(f"Phase 2: Adaptive refinement (deadline in {phase2_deadline - time.monotonic():.1f}s)")
    best_a, best_s = _optimize_adaptive(global_best, phase2_deadline)
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
        print(f"  Phase 2: NEW BEST {global_best_score:.8f}")
    else:
        print(f"  Phase 2: {best_s:.8f} (best={global_best_score:.8f})")

    # Phase 3: Local refinement (10% of time)
    phase3_deadline = time.monotonic() + 0.10 * (total_deadline - time.monotonic())
    print(f"Phase 3: Local refinement (deadline in {phase3_deadline - time.monotonic():.1f}s)")
    best_a, best_s = _local_refine(global_best, phase3_deadline, rng)
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
        print(f"  Phase 3: NEW BEST {global_best_score:.8f}")
    else:
        print(f"  Phase 3: {best_s:.8f} (best={global_best_score:.8f})")

    # Phase 4: Perturbation + re-optimize (10% of time)
    remaining = total_deadline - time.monotonic()
    if remaining > 10:
        print(f"Phase 4: Perturbation escape ({remaining:.1f}s remaining)")
        n_escapes = max(1, int(remaining / 40))
        time_per_escape = remaining / n_escapes
        for e in range(n_escapes):
            if time.monotonic() >= total_deadline:
                break
            esc_dl = min(time.monotonic() + time_per_escape, total_deadline)
            scale = 0.2 + 0.1 * e  # increasing perturbation
            a_pert = _perturb_and_optimize(global_best, esc_dl, rng, scale=scale)
            best_a, best_s = _optimize_nesterov(
                a_pert, esc_dl,
                temp_init=1.0, temp_end=50.0,
                eta_init=0.2 / np.sqrt(n),
                beta_mom=0.9,
                eta_decay=0.99995
            )
            if best_s < global_best_score:
                global_best = best_a.copy()
                global_best_score = best_s
                print(f"  Escape {e+1}: NEW BEST {global_best_score:.8f}")

    # Final: verify with actual evaluator
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.8f}")
    return result
