def spectral_error_reduction(a, alpha=0.1):
    """Blend the convolution toward flat, then recover a."""
    n = len(a)
    S = a.sum()
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    S = a.sum()  # Now S = S_t
    
    c = np.convolve(a, a)  # length 2n-1
    target = S * S / (2 * n - 1)  # flat target
    
    # Blend
    c_new = (1 - alpha) * c + alpha * target
    
    # Ensure non-negativity of c_new
    c_new = np.maximum(c_new, 0)
    
    # Recover a: FFT(a)^2 = FFT(c_new) (with zero-padding)
    N = 2 * n - 1  # or next power of 2
    C = np.fft.fft(c_new, N)
    
    # A = sqrt(C), taking principal branch
    A = np.sqrt(C)  # complex square root
    
    # Ensure Hermitian symmetry (for real a)
    # A[k] should = conj(A[N-k])
    # Due to numerical errors, enforce this

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    if target is None:

            if s_spec < s_curr:
                a = a_spec

        # Occasionally try a "kick" with different tau
        if (it + 1) % 800 == 0:
            a_kick = grad_step(a, lr * 3.0, tau=3.0)
            s_kick = compute_obj(a_kick)
            if s_kick < compute_obj(a):
                a = a_kick

        # Check improvement
        if (it + 1) % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-8:
                best_score = s
                best = a.copy()
                improve_count = 0
            else:
                improve_count += 1

        # Early stop if no improvement for a long time and tau is high
        if improve_count > 30 and tau > 50.0:
            # Try a reset with spectral shaping
            if use_spectral:
                a_reset = spectral_shaping(best, alpha=0.4)
                a_reset = grad_step(a_reset, 0.05, tau=10.0)
                s_reset = compute_obj(a_reset)
                if s_reset < best_score:
                    a = a_reset
                    improve_count = 0
                else:
                    a = best.copy()
                    improve_count = 0
            else:
                break

    # Final evaluation
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    # LP refinement
    if n <= 1024 and time.monotonic() < deadline - 20.0:
        c = np.convolve(best, best)
        M = float(c.max())

        for lp_it in range(200):
            if time.monotonic() > deadline - 10.0:
                break

            g = lp_solve(best, M)
            if g is None:
                break

            improved = False
            for t in [0.8, 0.6, 0.4, 0.2, 0.1]:
                a_new = (1.0 - t) * best + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_seq(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    best = a_new
                    best_score = s_new
                    improved = True
                    break

            if not improved:
                # Nudge with gradient
                best = grad_step(best, 0.015, tau=40.0)
                s = compute_obj(best)
                if s < best_score - 1e-10:
                    best_score = s
                    improved = True
                else:
                    # Try spectral
                    best_spec = spectral_shaping(best, alpha=0.2)
                    s_spec = compute_obj(best_spec)
                    if s_spec < best_score - 1e-10:
                        best = best_spec
                        best_score = s_spec
                        improved = True

            if not improved and lp_it > 5:
                break

            c = np.convolve(best, best)
            M = float(c.max())

    return best, best_score


def generate_starts(rng, lengths):
    """Generate diverse starting sequences for given lengths."""
    starts = []

    # Try to use previous best
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            for L in lengths:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
                elif L > len(gs):
                    # Pad by repeating
                    reps = int(np.ceil(L / len(gs)))
                    padded = np.tile(gs, reps)[:L]
                    starts.append(padded)
    except NameError:
        pass

    for n in lengths:
        i = np.arange(n, dtype=np.float64)

        # Random
        starts.append(rng.random(n) * 0.7 + 0.15)

        # Chirp
        starts.append(0.3 + 0.7 * np.abs(np.cos(2.0 * np.pi * i * 2.0 / n)))

        # Raised cosine
        starts.append(0.3 + 0.7 * np.cos(np.pi * i / n) ** 2)

        # Lopsided exponential
        starts.append(0.2 + 0.8 * np.exp(-i / (n * 0.4)))

        # Inverse lopsided
        starts.append(0.2 + 0.8 * np.exp(-(n - 1 - i) / (n * 0.4)))

        # U-shaped (high at edges)
        starts.append(0.2 + 0.8 * np.cos(np.pi * i / n) ** 2)

        # Anti-U (high in middle)
        starts.append(0.2 + 0.8 * np.exp(-2.0 * ((i - n / 2.0) / (n * 0.3)) ** 2))

        # Sine-based
        starts.append(0.2 + 0.8 * np.abs(np.sin(np.pi * i / n)))

        # Periodic with noise
        base = 0.5 + 0.3 * np.cos(2.0 * np.pi * i * 5.0 / n)
        starts.append(np.maximum(base + rng.random(n) * 0.1, 0.05))

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Quick screening of diverse starts (short optimization each)
    Phase 2: Focused optimization on top candidates
    Phase 3: LP refinement and final polishing
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Phase 1: Quick screening
    # Try multiple lengths with short optimization budgets
    print("=== Phase 1: Quick Screening ===")
    lengths_screen = [512, 1024, 2048]
    starts = generate_starts(rng, lengths_screen)
    starts.sort(key=len)

    # Limit screening to fit in 40% of budget
    screen_budget = (float(budget_s) - 15.0) * 0.35
    screen_deadline = t_start + screen_budget

    candidates = []  # (score, sequence)

    for idx, a0 in enumerate(starts):
        if time.monotonic() > screen_deadline - 5.0:
            break
        n = len(a0)
        a0 = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
        if a0.sum() < 1e-6:
            continue

        # Quick optimization: fewer iterations
        quick_deadline = min(time.monotonic() + min(30.0, (screen_deadline - time.monotonic()) / max(1, len(starts) - idx)), screen_deadline - 5.0)
        seq, score = optimize_single(a0, quick_deadline, max_grad_iters=800, use_spectral=True)
        candidates.append((score, seq))
        print(f"  [{idx+1}/{len(starts)}] n={n}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidate: {candidates[0][0]:.6f}, n={len(candidates[0][1])}")

    # Phase 2: Focused optimization on top candidates
    print("\n=== Phase 2: Focused Optimization ===")
    phase2_deadline = t_start + (float(budget_s) - 15.0) * 0.75

    # Take top 3 unique candidates (different lengths if possible)
    top_candidates = []
    seen_lengths = set()
    for score, seq in candidates:
        L = len(seq)
        if L not in seen_lengths:
            top_candidates.append((score, seq))
            seen_lengths.add(L)
        if len(top_candidates) >= 3:
            break

    # If we don't have enough, add next best
    if len(top_candidates) < 3:
        for score, seq in candidates:
            if seq not in [tc[1] for tc in top_candidates]:
                top_candidates.append((score, seq))
            if len(top_candidates) >= 3:
                break

    # Also add the previous best if available and good
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            gs_test = normalize_seq(gs.copy())
            gs_score = compute_obj(gs_test)
            if gs_score < best_score * 1.05:
                top_candidates.append((gs_score, gs_test))
    except NameError:
        pass

    for idx, (init_score, a0) in enumerate(top_candidates):
        if time.monotonic() > phase2_deadline - 20.0:
            break
        print(f"  Optimizing candidate {idx+1}: n={len(a0)}, init_score={init_score:.6f}")

        seq, score = optimize_single(a0, phase2_deadline, max_grad_iters=8000,
                                     use_spectral=True, tau_start=5.0, tau_end=100.0)
        print(f"    => score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    *** New global best: {best_score:.6f} ***")

    # Phase 3: Final refinement
    print(f"\n=== Phase 3: Final Refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        n_best = len(best_seq)
        print(f"  Refining best sequence (n={n_best}, score={best_score:.6f})")

        # Multiple refinement passes with different strategies
        for pass_i in range(5):
            if time.monotonic() > deadline - 5.0:
                break

            # Alternate between gradient and spectral
            if pass_i % 2 == 0:
                seq, score = optimize_single(best_seq, deadline - 5.0,
                                             max_grad_iters=3000, use_spectral=True,
                                             tau_start=20.0, tau_end=120.0,
                                             lr_start=0.05, lr_end=0.001)
            else:
                # Spectral-heavy refinement
                a = best_seq.copy()
                S_t = np.sqrt(2.0 * n_best)
                for it in range(500):
                    if time.monotonic() > deadline - 5.0:
                        break
                    a = spectral_shaping(a, alpha=0.1)
                    a = grad_step(a, 0.02, tau=30.0)
                    s = compute_obj(a)
                    if s < best_score:
                        best_score = s
                        best_seq = a.copy()

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Pass {pass_i}: {best_score:.6f}")
            else:
                # Small nudge attempt
                a_nudge = grad_step(best_seq, 0.01, tau=20.0)
                s_nudge = compute_obj(a_nudge)
                if s_nudge < best_score:
                    best_score = s_nudge
                    best_seq = a_nudge.copy()
                    print(f"  Nudge: {best_score:.6f}")
                else:
                    break

    # Return best
    if best_seq is not None:
        final = [float(x) for x in best_seq]
        print(f"\nFinal result: n={len(final)}, score={best_score:.6f}")
        try:
            eval_s = float(evaluate_sequence(final))
            print(f"evaluate_sequence verification: {eval_s:.6f}")
        except Exception as e:
            print(f"Verification error: {e}")
        return final

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
