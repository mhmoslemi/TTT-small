def project_simplex(v):

    k = int(np.count_nonzero(cond) - 1)

    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(max(1, p.size), 1.0 / max(1, p.size), dtype=np.float64)
    return p / s


def lp_best_response(p, budget_s=30.0):
    """
    LP best-response: max sum(g) s.t. conv(p,g) <= M, g >= 0.
    Returns (g_normalized, M) or (None, None) on failure.
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > 1000:
        return None, None
    c = conv_full(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0:
        return None, None
    
    f = p + 1e-15
    fs = float(f.sum())
    f = f / fs
    
    try:
        # Build constraint matrix: A[i, j] = f[j] for i,j where i = j + k for k in 0..n-1
        # conv(f, g)[i] = sum_j f[j] * g[i-j] for valid j
        # We need: for each i in 0..2n-2, sum_j f[j] * g[i-j] <= M
        # Row i: columns j where 0 <= i-j < n, i.e., j in [max(0,i-n+1), min(i, n-1)]
        rows = []
        cols = []
        data = []
        for i in range(2 * n - 1):
            j_lo = max(0, i - n + 1)
            j_hi = min(i, n - 1)
            for j in range(j_lo, j_hi + 1):
                rows.append(i)
                cols.append(j)
                data.append(f[j])
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'disp': False})
    except Exception:
        return None, None
    
    if res is None or res.status != 0:
        return None, None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None, None
    g = g / s
    return g, M


def make_beta_shape(n, alpha, beta):
    """Beta-like shape: p_j ∝ (j+0.5)^alpha * (n-1-j+0.5)^beta."""
    x = np.arange(n, dtype=np.float64) + 0.5
    p = np.power(x, alpha) * np.power(n - x, beta)
    return safe_normalize(p)


def make_u_shape(n, power):
    """U-shaped: p_j ∝ (j+1)^(-power) * (n-j)^(-power)."""
    x = np.arange(1, n + 1, dtype=np.float64)
    p = np.power(x, -power) * np.power(n + 1 - x, -power)
    return safe_normalize(p)


def make_cos_taper(n, exponent):
    """Cosine taper: p_j ∝ cos^exponent(pi * j / (n-1))."""
    x = np.linspace(0.0, np.pi, n)
    p = np.power(np.abs(np.cos(x)), exponent)
    p = np.maximum(p, 1e-15)
    return safe_normalize(p)


def make_power_taper(n, alpha):
    """Power taper: p_j ∝ (j+1)^(-alpha) for one-sided, or symmetric."""
    x = np.arange(1, n + 1, dtype=np.float64)
    p = np.power(x, -alpha) * np.power(n + 1 - x, -alpha)
    return safe_normalize(p)

    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def score_exact(p):
    """Compute the exact score 2n*max(conv(p,p))/sum(p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = conv_full(p)
    cmax = float(np.max(c)) if c.size else 0.0
    score = 2.0 * n * cmax / (s * s)
    return score


def smooth_val(c, T, cmax_hint):
    """Stable log-sum-exp smoothed max value."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, None
    cmax_hint = float(cmax_hint)
    if not np.isfinite(cmax_hint) or cmax_hint <= 0:
        cmax_hint = float(np.max(c))
    T = max(float(T), 1e-300)
    diff = (c - cmax_hint) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return cmax_hint, None
    val = cmax_hint + T * np.log(s)
    w = e / s
    return float(val), w


def gradient_smoothed(p, T):
    """
    Gradient of smoothed max of conv(p,p).
    Returns (grad, smoothed_value, weights).
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), 0.0, None
    c = conv_full(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax <= 0:
        return np.zeros(n), 0.0, None
    T = max(float(T), 1e-300)
    surr_val, w = smooth_val(c, T, cmax)
    if w is None:
        return np.zeros(n), cmax, None
    
    # grad_j = 2 * sum_{k: 0<=k-j<n} w[k] * p[k-j]
    grad = np.zeros(n, dtype=np.float64)
    for j in range(n):
        lo = max(j, 0)
        hi = min(j + n, 2 * n - 1)
        if hi > lo:
            grad[j] = 2.0 * float(np.dot(w[lo:hi], p[:hi - lo]))
    
    return grad, surr_val, w


def gradient_exact(p, c_idx):
    """Exact subgradient of max(conv(p,p)) at the given argmax index."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    grad = np.zeros(n, dtype=np.float64)
    for j in range(n):
        k_minus_j = c_idx - j
        if 0 <= k_minus_j < n:
            grad[j] = 2.0 * p[k_minus_j]
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
