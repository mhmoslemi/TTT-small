```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fft_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    n_fft = 1
    while n_fft < 2*n - 1:
        n_fft *= 2
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)
    return conv[:2*n-1]

def score_sequence(a):
    n = len(a)
    conv = fft_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)
```

```python
"""
Smoothed gradient descent with log-sum-exp for minimizing peak autocorrelation
of non-negative sequences. Combines adaptive smoothing, Adam optimizer,
multi-length search, and LP-based polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def next_pow2(x):
    p = 1
    while p < x:
        p *= 2
    return p


def fft_convolve_full(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    n_fft = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)
    return conv[:2 * n - 1]


def score_sequence(a):
    """Compute the evaluation score."""
    n = len(a)
    conv = fft_convolve_full(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def initial_parent():
    """Get the parent sequence if available."""
    try:
        seq = list(height_sequence_1)
        a = np.array([float(x) for x in seq])
        a = np.maximum(0, a)
        if a.sum() > 0.01:

    except Exception:

    return None


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_rectangular(n):
    return np.ones(n, dtype=np.float64)


def random_bumps(n, rng, n_bumps=3):
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def smoothed_gradient_descent(a, time_limit, rng, M_start=50.0, M_end=50000.0,
                               lr_base=0.5, verbose=False, anneal_iters=None):
    """
    Smoothed gradient descent using log-sum-exp to approximate max(conv).
    Uses Adam optimizer with non-negativity projection and sum normalization.
    """
    n = len(a)
    a = np.maximum(0.0, a.astype(np.float64)).copy()
    a = normalize_nonneg(a)

    best_score = score_sequence(a)
    best_a = a.copy()

    n_fft = next_pow2(2 * n - 1)
    n_fft_cross = next_pow2(3 * n - 2)

    # Adam state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_count = 0

    if anneal_iters is None:
        anneal_iters = max(100, int(time_limit * 200))

    start_time = time.monotonic()
    last_print = start_time
    patience = 0
    patience_limit = max(500, int(anneal_iters * 0.3))

    while time.monotonic() - start_time < time_limit:
        t_count += 1
        # Anneal M (smoothing parameter)
        frac = min(1.0, t_count / anneal_iters)
        # Use exponential annealing
        M = M_start * (M_end / M_start) ** frac

        # Compute convolution via FFT
        fa = np.fft.rfft(a, n=n_fft)
        conv = np.fft.irfft(fa * fa, n=n_fft)[:2 * n - 1]

        # Compute log-sum-exp smoothed objective value and softmax weights
        conv_max = conv.max()
        if conv_max < 1e-15:
            break
        scaled_M = M / conv_max  # This gives us weights proportional to exp(M*conv/conv_max)
        # Actually we want weights proportional to exp(M * conv)
        # Use log-sum-exp trick for numerical stability
        log_w = M * conv
        log_w_max = log_w.max()
        w = np.exp(log_w - log_w_max)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[conv.argmax()] = 1.0
        else:
            w /= w_sum

        # Compute gradient: grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
        # This is cross-correlation of w and a
        a_rev = a[::-1]
        # Use FFT for the cross-correlation
        fw = np.fft.rfft(w, n=n_fft_cross)
        fa_rev = np.fft.rfft(a_rev, n=n_fft_cross)
        cross = np.fft.irfft(fw * fa_rev, n=n_fft_cross)
        grad = 2.0 * cross[n - 1:2 * n - 1]

        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** t_count)
        v_hat = v_buf / (1.0 - beta2 ** t_count)

        # Adaptive learning rate
        lr = lr_base / (1.0 + 0.1 * t_count / 1000.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to non-negativity
        a_new = np.maximum(0.0, a_new)

        # Renormalize to sum = 1
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a_new / s

        # Evaluate and track best
        if t_count % 10 == 0 or t_count < 10:
            score = score_sequence(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                patience = 0
            else:
                patience += 10

            if verbose and time.monotonic() - last_print > 10.0:
                print(f"      SGD iter {t_count}: score={score:.6f}, best={best_score:.6f}, M={M:.1f}")
                last_print = time.monotonic()

            if patience > patience_limit and t_count > anneal_iters // 2:
                # Restart with perturbation
                if verbose:
                    print(f"      SGD restart at iter {t_count}, best={best_score:.6f}")
                a = best_a.copy()
                noise = rng.normal(0, 0.05, size=n)
                a = a + noise
                a = normalize_nonneg(a)
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                t_count = 0
                patience = 0

    # Final evaluation
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def lp_refine(a, time_limit=120.0):
    """LP-based refinement: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)

    start_time = time.monotonic()
    stagnant = 0

    while time.monotonic() - start_time < time_limit:
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 10.0:
            break

        conv = fft_convolve_full(a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                break

        # Build constraint matrix: conv(a, g)[k] = sum_j a[j] * g[j+k-n+1]
        # For each row k (0..2n-2): sum_j A[k,j] * g[j] <= M
        # A[k,j] = a[j] if 0 <= k+j-n+1 < n... 
        # Actually conv(a,g)[m] = sum_j a[j]*g[m-j+n-1]... 
        # Let's use: conv(a,g) = np.convolve(a, g), so conv[m] = sum_j a[j]*g[m-j]
        # For 0 <= m-j < n, i.e., j in [max(0,m-n+1), min(n-1,m)]
        # A[m, j] = a[j] if max(0, m-n+1) <= j <= min(n-1, m), else 0
        # Wait: conv(a,g)[m] = sum_{j} a[j] * g[m-j] where 0<=j<n and 0<=m-j<n
        # = sum_{j=max(0,m-n+1)}^{min(n-1,m)} a[j] * g[m-j]
        # Let i = m-j, so j = m-i, i ranges from max(0,m-n+1) to min(n-1,m)
        # = sum_{i} a[m-i] * g[i] where i in [max(0,m-n+1), min(n-1,m)]
        # So A[m, i] = a[m-i] if max(0,m-n+1) <= i <= min(n-1,m)

        n_rows = 2 * n - 1
        # Build sparse matrix
        row_indices = []
        col_indices = []
        data = []
        for m in range(n_rows):
            i_start = max(0, m - n + 1)
            i_end = min(n - 1, m)
            for i in range(i_start, i_end + 1):
                val = a[m - i]
                if val > 0:
                    row_indices.append(m)
                    col_indices.append(i)
                    data.append(val)

        A_ub = csr_matrix((data, (row_indices, col_indices)), shape=(n_rows, n))
        c = -np.ones(n)
        b_ub = np.full(n_rows, M)
        bounds = [(0.0, None)] * n

        timeout_lp = max(5.0, min(30.0, remaining * 0.4))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': timeout_lp})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g
                    alpha = min(0.5, max(0.05, 2.0 / (1.0 + stagnant)))
                    a_new = (1.0 - alpha) * a + alpha * g
                    a_new = normalize_nonneg(a_new)
                    new_score = score_sequence(a_new)
                    if new_score <= score + 1e-12:
                        a = a_new
                    else:
                        alpha2 = alpha * 0.3
                        a_new = (1.0 - alpha2) * a + alpha2 * g
                        a = normalize_nonneg(a_new)
                else:
                    break
            else:
                break
        except Exception:
            break

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def local_search_refine(a, time_limit=60.0, rng=None):
    """Perturbation-based local search."""
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()

    start_time = time.monotonic()
    accept_count = 0

    while time.monotonic() - start_time < time_limit:

        width = int(rng.integers(max(2, n // 200), max(3, n // 5)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()
        perturbation = np.exp(rng.normal(0.0, 0.15, size=stop - start))
        candidate[start:stop] *= perturbation

            mask = rng.random(n) < 0.5
            candidate[mask] *= rng.uniform(0.5, 2.0)

        score = score_sequence(candidate)

        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1

    return best_a, best_score

        if time_limit < 15.0:
            continue

        print(f"\n=== n={n}, init={init_type}, time_limit={time_limit:.0f}s, remaining={remaining:.0f}s ===")

        # Generate initial sequence
        if init_type == 'parent':
            a = parent.copy()
        elif init_type == 'parent_resampled':
            if parent is not None:
                # Resample parent to new length
                old_t = np.linspace(0, 1, len(parent))
                new_t = np.linspace(0, 1, n)
                a = np.interp(new_t, old_t, parent)
            else:
                a = initial_raised_cosine(n)
        elif init_type == 'parent_extended':
            if parent is not None:
                # Pad parent with zeros and add a smooth tail
                a = np.zeros(n)
                a[:len(parent)] = parent
                # Add smooth continuation
                tail_start = len(parent)
                tail_len = n - tail_start
                if tail_len > 0:
                    decay = np.exp(-np.arange(tail_len) / max(1, tail_len / 3))
                    a[tail_start:] = parent[-1] * decay
            else:
                a = initial_raised_cosine(n)
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
            # Add small random perturbation
            a += rng.normal(0, 0.01, size=n)
            a = np.maximum(0, a)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=rng.uniform(0.15, 0.4))
        elif init_type == 'blackman':
            a = initial_blackman(n)
            a += rng.normal(0, 0.005, size=n)
            a = np.maximum(0, a)
        elif init_type == 'random':
            a = random_bumps(n, rng, n_bumps=int(rng.integers(2, 7)))
        elif init_type == 'rectangular':
            a = initial_rectangular(n)
            # Taper the edges
            taper = n // 10
            if taper > 2:
                a[:taper] *= np.linspace(0.3, 1.0, taper)
                a[-taper:] *= np.linspace(1.0, 0.3, taper)
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a)
        init_score = score_sequence(a)
        print(f"    Init score: {init_score:.6f}")

        if init_score < best_score:
            best_score = init_score
            best_seq = a.copy()
            print(f"    *** New best from init: {best_score:.6f}")

        # Run smoothed gradient descent
        if time.monotonic() < deadline - 20:
            t_sgd = time_limit * 0.7
            a_sgd, score_sgd = smoothed_gradient_descent(
                a, time_limit=t_sgd, rng=rng,
                M_start=30.0, M_end=100000.0,
                lr_base=0.8, verbose=True
            )
            if score_sgd < best_score:
                best_score = score_sgd
                best_seq = a_sgd.copy()
                print(f"    *** New best from SGD: {best_score:.6f}")

        # LP refinement if still time
        remaining = deadline - time.monotonic()
        if remaining > 30.0 and best_seq is not None:
            t_lp = min(time_limit * 0.2, remaining * 0.5)
            if len(best_seq) == n:
                a_lp, score_lp = lp_refine(best_seq, time_limit=t_lp)
                if score_lp < best_score:
                    best_score = score_lp
                    best_seq = a_lp.copy()
                    print(f"    *** New best from LP: {best_score:.6f}")

    # Final polishing phase on the best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")

        if remaining > 60.0:
            # More aggressive SGD
            n_best = len(best_seq)
            t_sgd2 = remaining * 0.4
            a_polish, score_polish = smoothed_gradient_descent(
                best_seq, time_limit=t_sgd2, rng=rng,
                M_start=100.0, M_end=200000.0,
                lr_base=0.3, verbose=True
            )
            if score_polish < best_score:
                best_score = score_polish
                best_seq = a_polish.copy()
                print(f"    After SGD polish: {best_score:.6f}")

            # LP refinement
            remaining = deadline - time.monotonic()
            if remaining > 20.0:
                t_lp = remaining * 0.4
                a_lp, score_lp = lp_refine(best_seq, time_limit=t_lp)
                if score_lp < best_score:
                    best_score = score_lp
                    best_seq = a_lp.copy()
                    print(f"    After LP polish: {best_score:.6f}")

            # Local search
            remaining = deadline - time.monotonic()
            if remaining > 15.0:
                t_ls = remaining * 0.6
                a_ls, score_ls = local_search_refine(best_seq, time_limit=t_ls, rng=rng)
                if score_ls < best_score:
                    best_score = score_ls
                    best_seq = a_ls.copy()
                    print(f"    After LS polish: {best_score:.6f}")

            # One more SGD round
            remaining = deadline - time.monotonic()
            if remaining > 10.0:
                a_final, score_final = smoothed_gradient_descent(
                    best_seq, time_limit=remaining * 0.8, rng=rng,
                    M_start=500.0, M_end=500000.0,
                    lr_base=0.1, verbose=False
                )
                if score_final < best_score:
                    best_score = score_final
                    best_seq = a_final.copy()
                    print(f"    After final SGD: {best_score:.6f}")

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Convert to list and verify
    best_seq = np.maximum(0.0, best_seq)
    result = [float(x) for x in best_seq]
    
    print(f"\nFinal result: score={best_score:.6f}, n={len(result)}")
    verify_score = evaluate_sequence(result)
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return result
```
