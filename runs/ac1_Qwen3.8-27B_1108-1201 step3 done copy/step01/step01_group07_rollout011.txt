mirror descent with adaptive softmax peak weighting, FFT-accelerated
subgradients, and a multi-phase search strategy.
"""

import numpy as np
import time


def next_pow2(m):
    return 1 << (m - 1).bit_length() if m > 1 else 1


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    S = next_pow2(size)
    A = np.fft.rfft(a, S)
    return np.fft.irfft(A * A, S)[:size]


def compute_subgradient_fft(a, w, fft_size, A_cached=None):
    """
    sg[i] = sum_k w[k] * a[k-i] (valid indices only)
    = conv(a, w_flipped)[size-1-i]
    """
    n = len(a)
    size = 2 * n - 1
    w_flipped = w[::-1]
    
    if A_cached is not None and A_cached.size == fft_size // 2 + 1:
        A = A_cached
    else:
        A = np.fft.rfft(a, fft_size)
    
    W = np.fft.rfft(w_flipped, fft_size)
    conv = np.fft.irfft(A * W, fft_size)

    idx = size - 1 - np.arange(n)
    return conv[idx]


def proj_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _bessel_i0(x):
    """I0 Bessel function (vectorized, approximation)."""
    # Use the series for |x| < 8, asymptotic for |x| >= 8
    x = np.abs(x)
    out = np.empty_like(x)
    small = x < 8
    if small.any():
        xs = x[small]
        out[small] = 1 + xs**2/4 + xs**4/64 + xs**6/2304 + xs**8/147456
    if (~small).any():
        xl = x[~small]
        out[~small] = np.exp(xl) / np.sqrt(2 * np.pi * xl) * (1 + 1/(8*xl) + 9/(128*xl**2))
    return out

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    S = _next_pow2(size)
    A = np.fft.rfft(a, S)

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
