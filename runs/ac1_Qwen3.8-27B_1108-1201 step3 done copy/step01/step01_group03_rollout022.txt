    """
    For conv[k_max] = sum_i f[i]*f[k_max-i],

def subgradient_step(f, step_size, normalize=True):
    """
    Take a step to reduce the max convolution value.
    f is normalized to sum=1.
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = conv.max()
    # Find all positions at (or near) the max
    threshold = max_val - 1e-10
    active = np.where(conv >= threshold)[0]
    
    # Sum of gradients at active positions
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = int(k) - i
            if 0 <= j < n:
                grad[i] += f[j]
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return f
    
    # Take step: f_new = f - step_size * grad
    f_new = f - step_size * grad / grad_norm
    
    # Project: make non-negative and renormalize
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    f_new = f_new / s
    return f_new

"""
Hybrid smoothed-max gradient + LP + random perturbation optimizer
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np

import time

def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2, or inf if invalid."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def compute_grad_smoothed_max(f, beta):
    """
    Compute gradient of log-sum-exp(beta * conv(f,f)) w.r.t. f.
    Returns the gradient as a vector of length n.
    """
    n = len(f)
    conv = np.convolve(f, f)
    # Softmax for numerical stability
    conv_shifted = conv - conv.max()
    exp_conv = np.exp(beta * conv_shifted)
    weights = exp_conv / exp_conv.sum()
    # grad[i] = sum_j f[j] * weights[i+j]
    # = np.convolve(f[::-1], weights)[n-1 : 2*n-1]
    grad = np.convolve(f[::-1], weights)[n-1:2*n-1]
    return grad, conv.max()

def smoothed_max_step(f, beta, eta):
    """One step of gradient descent on smoothed max, with projection."""
    grad, max_val = compute_grad_smoothed_max(f, beta)
    # Normalize gradient to unit norm for consistent step size
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return f
    f_new = f - eta * grad / gnorm
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    f_new = f_new / s
    return f_new

def build_conv_sparse(n, f):
    """Build sparse matrix A with A[k,j] = f[k-j] for valid indices."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    # Only include non-zero entries
    mask = data > 1e-15
    A = sp.coo_matrix((data[mask], (row_idx[mask], j_idx[mask])),
                      shape=(2*n-1, n)).tocsr()
    return A

def solve_lp(f, M, time_limit_s=5.0):
    """Solve max sum(g) s.t. A(g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_sparse(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lp_step(f, t, time_limit_s=5.0):
    """One LP-based update step."""
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f, False
    g = solve_lp(f, M, time_limit_s)
    if g is None:
        return f, False
    gs = g.sum()
    if gs < 1e-10:
        return f, False
    g = g / gs

    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f, False
    return f_new / s, True
