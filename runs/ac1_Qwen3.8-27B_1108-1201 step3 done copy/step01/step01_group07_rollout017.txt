"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS in log-space with
soft-max smoothed convolution peaks and FFT-based exact gradients.
Tries multiple sequence lengths (128-4096) with diverse initializations,
adaptive temperature schedule, and final subgradient polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def smooth_obj_and_grad_x(x, t, n):
    """
    Compute the soft-max smoothed objective and gradient in log-parameter space.
    x: log-parameters (a = softmax(x))
    t: temperature (higher = sharper focus on peaks)
    n: sequence length
    Returns (objective_value, gradient_wrt_x).
    """
    # Compute a = softmax(x)
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    Z = exp_x.sum()
    a = exp_x / Z

    # Compute convolution
    conv = fast_convolve(a)

    # Soft-max weights over convolution positions
    tc = t * conv
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    p_sum = exp_tc.sum()
    p = exp_tc / p_sum

    # Smoothed objective: 2n * (1/t) * log(sum_k exp(t*conv_k))
    # = 2n * (tc_max/t + log(p_sum)/t)
    # = 2n * (conv_max + log(p_sum)/t)
    f = 2.0 * n * (tc_max / t + np.log(p_sum) / t)

    # Gradient of F = 2n * f_t with respect to a:
    # grad_a[i] = 4n * sum_k p[k] * a[k-i] (a[j]=0 for j outside [0,n-1])
    # Computed as: conv_full = linear_convolve(a, flip(p)), then grad_a[i] = 4n * conv_full[2n-2-i]

    r = p[::-1]  # flip p, length 2n-1
    fft_size2 = 1 << (3 * n - 1).bit_length()
    A2 = np.fft.rfft(a, fft_size2)
    R2 = np.fft.rfft(r, fft_size2)
    conv_full = np.fft.irfft(A2 * R2, fft_size2)

    # grad_a[i] = 4n * conv_full[2n-2-i] for i=0,...,n-1
    # conv_full has length 3n-2, indices 0 to 3n-3
    # We need indices 2n-2, 2n-3, ..., n-1
    grad_a = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]

    # Convert to x-gradient: grad_x[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    dot_ga_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga_a)

    return f, grad_x


def optimize_one(a0, time_limit, t_values=None):
    """
    Optimize a single starting point using L-BFGS in log-space with temperature schedule.
    a0: initial non-negative sequence
    time_limit: seconds allotted
    t_values: list of temperatures to try in order
    Returns: (best_a_normalized, best_objective_value)
    """
    if t_values is None:
        t_values = [50.0, 100.0, 200.0, 400.0, 800.0]

    a = np.maximum(a0, 1e-300).astype(np.float64)
    total = a.sum()
    if total < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / total

    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t_start = time.monotonic()

    for t in t_values:
        if time.monotonic() - t_start > time_limit * 0.95:
            break

        x0 = np.log(a + 1e-300)
        elapsed = time.monotonic() - t_start
        remaining = time_limit * 0.95 - elapsed
        if remaining < 1.0:
            break

        # Estimate max iterations based on remaining time
        # Each L-BFGS iteration costs ~3 FFTs of size ~2n, roughly O(n log n)
        # For n=512, one iteration is ~1ms, so 1000 iters = 1s
        # Be conservative: allow at most remaining/0.005 iterations
        max_iter = max(50, min(2000, int(remaining / max(0.002, n / 200000.0))))

        time_per_phase = remaining / len(t_values)
        phase_iter = max(50, min(max_iter, int(time_per_phase / max(0.003, n / 100000.0))))

        try:
            result = minimize(
                smooth_obj_and_grad_x,
                x0,
                method='L-BFGS-B',
                args=(t, n),
                jac=True,
                options={'maxiter': phase_iter, 'gtol': 1e-10,
                         'maxfun': phase_iter * 3, 'ftol': 1e-12}
            )
        except Exception as e:
            print(f"    L-BFGS error at t={t}: {e}")
            continue

        x_opt = result.x
        a_new = np.exp(x_opt - x_opt.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def subgradient_polish(a, time_limit, steps=None):
    """
    Polish using exact subgradient at the argmax of convolution.
    Uses multiplicative updates with adaptive step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t_start = time.monotonic()
    eta = 0.3
    patience = 0
    max_patience = 30

    for step in range(steps if steps else 5000):
        if time.monotonic() - t_start > time_limit:
            break

        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Find all positions where conv is within 1% of max (handle near-ties)
        thresh = max_conv * 0.99
        peak_indices = np.where(conv >= thresh)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Compute subgradient: grad_a[i] = sum_{k in peaks} a[k-i] (averaged)
        grad_a = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad_a[i] += a[j]
        grad_a /= len(peak_indices)

        # Multiplicative update: a[i] *= exp(-eta * grad_a[i])
        # Center the gradient to preserve sum approximately
        ga_centered = grad_a - grad_a.mean()
        log_update = -eta * ga_centered
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            eta *= 0.5
            patience = 0
            if eta < 0.003:
                break

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for a given length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian center
    for sf in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n / 2.0) / (sf * n)) ** 2)
        g = np.maximum(g, 0)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Cosine (peaked center, zero edges)
    cos_seq = np.cos(np.pi * x / (n - 1 + 1e-10))
    cos_seq = np.maximum(cos_seq, 0)
    if cos_seq.sum() > 1e-10:
        cos_seq /= cos_seq.sum()
        inits['cosine'] = cos_seq

    # Edge-concentrated (high at edges)
    edge = np.zeros(n)
        w = max(1, n // 8)
        edge[:w] = 1.0
        edge[-w:] = 1.0
        edge /= edge.sum()
        inits['edge'] = edge

    # Alternating
    alt = np.ones(n)
    alt[1::2] = 0.3
    alt /= alt.sum()
    inits['alternating'] = alt

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'rand_{alpha}'] = r

    # Bump at 1/4 and 3/4
    bump = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + \
           np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    bump /= bump.sum()
    inits['bimodal'] = bump

    # From parent
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.size >= n:
            pp = p[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_trunc'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, p.size), p)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    return inits
