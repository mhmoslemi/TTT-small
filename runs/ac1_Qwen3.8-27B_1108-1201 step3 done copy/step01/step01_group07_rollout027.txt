            max_stall = max(100, max_stall // 2)

def simulated_annealing(a0, time_limit, seed=0):
    """Simulated annealing on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / a_sum
    
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    current_obj = best_obj
    
    T_init = 0.01
    T_final = 1e-5
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** progress
        
        # Propose perturbation
        n_perturb = max(1, n // 10)
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_new = a.copy()
        a_new[idx] += rng.normal(0, 0.05, n_perturb)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a
        else:
            a_new = a_new / s
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        delta = obj_new - current_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            conv = conv_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
    
    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for a given length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal_edge'] = bi
    
    # Center-biased (opposite of bimodal)
    cb = np.zeros(n)
    mid = max(1, n // 4)
    cb[n//2 - mid : n//2 + mid] = 1.0
    cb /= cb.sum()
    inits['center'] = cb
    
    # Cosine bump
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits['cosine'] = cos_b
    
    # Random Dirichlet
    inits['random_dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_dirichlet2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Random binary-like
    binary = (rng.random(n) < 0.5).astype(float)
    if binary.sum() > 0:
        inits['binary'] = binary / binary.sum()
    
    # Modulated uniform
    for freq in [1, 2, 3]:
        mod = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n)
        mod = np.maximum(mod, 0.01)
        inits[f'mod_{freq}'] = mod / mod.sum()
    
    # Parent-based
    if parent is not None:
        if parent.size >= n:
            p = np.maximum(parent[:n], 0.0).copy()
            if p.sum() > 1e-10:
                inits['parent'] = p / p.sum()
        else:
            p = np.interp(np.linspace(0, 1, n), 
                         np.linspace(0, 1, parent.size), parent)
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-10:
                inits['parent_interp'] = p / p.sum()
        # Perturbed parent
        if 'parent' in inits or 'parent_interp' in inits:
            p = inits.get('parent', inits.get('parent_interp'))
            perturbed = p + rng.normal(0, 0.05, n) * p
            perturbed = np.maximum(perturbed, 0.0)
            if perturbed.sum() > 1e-10:
                inits['parent_perturbed'] = perturbed / perturbed.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization with
FFT-based subgradients, temperature annealing, simulated annealing,
and diverse multi-scale search over sequence lengths 64-2048.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def smooth_max_weights(conv, t):
    """Return softmax(t*conv) weights (numerically stable)."""
    m = conv.max()
    if m < 1e-15:
        return np.zeros_like(conv)
    shifted = t * (conv - m)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_shifted / total


def compute_subgradient(weights, a):
    """Compute 2*(weights cross-correlate a) via FFT. Length n result."""
    n = len(a)
    a_rev = a[::-1]
    total_len = len(weights) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A, fft_size)[:total_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def normalize_simplex(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

        lr = lr_init / np.sqrt(1.0 + 0.01 * step)

        sg = sg - sg.mean()

        scale = 0.02 + 0.05 * (1.0 - progress)
        a_new = a.copy()
        a_new[idx] *= (1.0 + rng.normal(0, scale, n_perturb))
        a_new = normalize_simplex(a_new)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        delta = obj_new - current_obj

        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            a = a_new
            conv = conv_new
            current_obj = obj_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    half = n / 2.0

    inits['uniform'] = np.ones(n) / n

    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - half) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Edge-concentrated (inverse Gaussian)
    inv_g = 1.0 / (1.0 + 0.5 * ((x - half) / (n * 0.3)) ** 2)
    inv_g /= inv_g.sum()
    inits['inv_gauss'] = inv_g

    # Bimodal edges
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine ring (high at edges, low center)
    ring = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    ring = np.maximum(ring, 0.01)
    ring /= ring.sum()
    inits['cosine_ring'] = ring

    # Power-law decay from center
    pl = 1.0 / (1.0 + 0.5 * np.abs(x - half) / (n * 0.2))
    pl /= pl.sum()
    inits['powerlaw'] = pl

    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 3.0)

    # Binary-like
    for p in [0.3, 0.5, 0.7]:
        binary = (rng.random(n) < p).astype(float)
        if binary.sum() > 0:
            inits[f'binary_{p}'] = binary / binary.sum()

    # Modulated uniform
    for freq in [1, 2, 5]:
        mod = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        mod = np.maximum(mod, 0.01)
        inits[f'mod_{freq}'] = mod / mod.sum()

    # Parent-based
    if parent is not None:
        if parent.size >= n:
            p = np.maximum(parent[:n], 0.0).copy()
        else:
            p = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, parent.size), parent)
            p = np.maximum(p, 0.0)
        if p.sum() > 1e-10:
            p_norm = p / p.sum()
            inits['parent'] = p_norm
            # Perturbed parent
            pert = p_norm * rng.lognormal(0, 0.2, n)
            pert = normalize_simplex(pert)
            inits['parent_pert'] = pert

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: hybrid Adam + SA optimizer with FFT subgradients,
    temperature annealing, multi-length search, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(0.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 25.0

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
            print(f"Parent loaded: len={p.size}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Quick scan
    print("=== Phase 1: Quick Scan ===")
    lengths = [128, 256, 512, 1024, 2048]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.append(pn)
            lengths.sort()
        # Always test the parent directly
        p_norm = parent / parent.sum()
        score_p = float(evaluate_sequence(p_norm.tolist()))
        if np.isfinite(score_p):
            best_score = score_p
            best_sequence = p_norm.copy()
            print(f"  Parent score: {score_p:.6f}")

    phase1_budget = min(450.0, (deadline - time.monotonic()) * 0.25)
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 80:
            break
        inits = generate_initializations(n, rng, parent)
        n_configs = len(inits)
        time_per = max(2.0, phase1_budget / (n_configs * max(1, len(lengths))))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 80)
            if t_alloc < 1.5:
                break

            a_opt, obj_opt = adam_optimize(a0, time_limit=t_alloc,
                                           t_start=8.0, t_end=150.0,
                                           lr_init=0.04, seed=seed + n + len(name))
            seq = a_opt.tolist()
            score = float(evaluate_sequence(seq))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    # Phase 2: Deep optimization
    print(f"=== Phase 2: Deep Opt (best={best_score:.6f}) ===")
    scan_results.sort(key=lambda x: x[0])

    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_n and len(top_configs) < 10:
            top_configs.append((n, name, a_opt))
            seen_n.add(n)

    if best_sequence is not None:
        n_b = len(best_sequence)
        a_b = best_sequence / best_sequence.sum()
        top_configs.append((n_b, 'best', a_b))
        seen_n.add(n_b)

    phase2_budget = deadline - time.monotonic() - 60
    phase2_budget = max(100.0, phase2_budget * 0.55)
    time_per_deep = max(15.0, phase2_budget / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 50:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 50)
        if t_alloc < 5:
            break

        param_sets = [
            (10, 300, 0.03, 0.85),
            (5, 100, 0.06, 0.9),
            (30, 600, 0.02, 0.8),
            (15, 200, 0.04, 0.85),
        ]
        t_each = t_alloc * 0.7 / len(param_sets)

        for j, (ts, te, lr, b1) in enumerate(param_sets):
            if time.monotonic() > deadline - 50:
                break
            if t_each < 3:
                break
            a_opt, _ = adam_optimize(a0, time_limit=t_each,
                                     t_start=ts, t_end=te,
                                     lr_init=lr, beta1=b1,
                                     seed=seed + i * 100 + j)
            seq = a_opt.tolist()
            score = float(evaluate_sequence(seq))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n} {name} p{j}: {score:.6f}")
            # Chain: use result as next start
            a0 = a_opt

        # Simulated annealing on best from Adam chain
        t_sa = max(5.0, t_alloc * 0.3)
        if time.monotonic() < deadline - 55:
            a_sa, _ = simulated_annealing_optimize(a0, time_limit=t_sa,
                                                   seed=seed + 5000 + i)
            score = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_sa.copy()
                print(f"  SA n={n} {name}: {score:.6f}")

    # Phase 3: Final polish
    print(f"=== Phase 3: Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        a_best = best_sequence / best_sequence.sum()
        n_best = len(a_best)

        polish_settings = [
            (50, 800, 0.008, 0.9),
            (100, 1500, 0.005, 0.92),
            (200, 3000, 0.003, 0.95),
            (30, 500, 0.006, 0.88),
        ]
        t_each = remaining / len(polish_settings)

        for j, (ts, te, lr, b1) in enumerate(polish_settings):
            if remaining < 5:
                break
            a_opt, _ = adam_optimize(a_best, time_limit=t_each,
                                     t_start=ts, t_end=te,
                                     lr_init=lr, beta1=b1,
                                     seed=seed + 9000 + j)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                a_best = a_opt / a_opt.sum()
                n_best = len(a_opt)
                print(f"  Polish p{j}: {score:.6f}")
            remaining = deadline - 20 - time.monotonic()

        # Final SA polish
        if remaining > 10:
            a_sa, _ = simulated_annealing_optimize(a_best, time_limit=remaining,
                                                   seed=seed + 7777)
            score = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_sa.copy()
                print(f"  Final SA: {score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128) / 128.0
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
