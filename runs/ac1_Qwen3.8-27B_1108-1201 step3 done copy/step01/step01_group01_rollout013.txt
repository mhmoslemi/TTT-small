```python
Multi-phase optimization to minimize the normalized autocorrelation peak.
Uses Nesterov-accelerated subgradient descent, coordinate descent polishing,
and simulated annealing for escaping local minima.

import numpy as np
import time
from scipy import sparse

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c)

def _nesterov_subgradient(a, deadline, rng, M_peaks=50, eta_init=None, 
                           cooling=0.999, min_eta=1e-10, momentum_init=0.9,
                           reheat_every=500):

    """

    if eta_init is None:
        eta_init = 0.5 / np.sqrt(n)

    S = np.sum(a)
    if S > 1e-10:
        a /= S
    y = a.copy()  # Nesterov momentum point
    momentum = momentum_init
    
    best = a.copy()
    best_score = _compute_score(best)
    eta = eta_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        
        # Compute autocorrelation at momentum point
        c_y = np.convolve(y, y)
        c_max = np.max(c_y)
        
        # Get top peaks
        if M_peaks >= len(c_y):
            top_idx = np.arange(len(c_y))
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            top_idx = np.argpartition(c_y, -M_peaks)[-M_peaks:]
            # Softmax weights
            c_sel = c_y[top_idx]
            c_range = c_max - c_y[top_idx]
            weights = np.exp(-10.0 * c_range / max(c_max, 1e-15))
            if np.sum(weights) < 1e-15:
                weights = np.ones(len(top_idx)) / len(top_idx)
            else:
                weights /= np.sum(weights)
        
        # Compute gradient at y
        grad = np.zeros(n)
        for ki, k in enumerate(top_idx):
            w = weights[ki]
            if w < 1e-8:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * y[k - idx]
        
        # Center gradient
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Nesterov update
        a_new = y - eta * grad
        a_new = _project_simplex(a_new, 1.0)
        
        # Update momentum point
        t = (1.0 + np.sqrt(1.0 + 4.0 * momentum**2)) / 2.0
        y = a_new + ((momentum - 1.0) / t) * (a_new - a)
        y = _project_simplex(y, 1.0)
        
        a = a_new
        
        # Evaluate
        cur_score = _compute_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Cooling

           _eta = eta * cooling
            if _eta < min_eta:

                a = best + 0.05 * (rng.random(n) - 0.5)
                a = _project_simplex(a, 1.0)
                y = a.copy()
            eta = eta
            no_improve = 0

            pert = 0.02 * (rng.random(n) - 0.5)
            a_warm = best + pert
            a_warm = _project_simplex(a_warm, 1.0)
            y = a_warm.copy()
    
    return best, best_score
```

```python
return float(2 * n * max_b / (sum_a**2))
```

So it clips to [0, 1000] and checks if sum >= 0.01. The score is `2*n*max_b/sum_a^2`. This is scale-invariant (if we multiply all elements by c, max_b scales by c^2 and sum_a^2 scales by c^2, so the ratio is unchanged).

So I can work with `sum(a) = 1` and the final sequence can be any scaling (as long as elements are in [0, 1000]).

Let me now write the complete program.

Actually, one more consideration: the `np.convolve` function. For n=6121, it produces an array of length 2n-1 = 12241. This is a one-time cost per evaluation. In the inner loop, I want to avoid recomputing the full convolution if possible.

For the subgradient method, I need the gradient, which requires knowing which lags are at the peak. I compute the full convolution, find the peaks, and compute the gradient. This is O(n^2) per iteration (for the convolution) + O(n * M_peaks) for the gradient (for M_peaks peaks, each contributing to O(n) coordinates).

Wait, the gradient computation: for each peak lag k, the gradient is `2 * a[k-i]` for valid i. This is O(n) per peak. With M_peaks peaks, it's O(n * M_peaks). If M_peaks = 50, that's O(50 * 6121) ≈ 300,000 operations, which is negligible compared to the O(n^2) = 37 million for the convolution.

So the bottleneck is the convolution. Each iteration takes about 10-20 ms. In 600 seconds, I can do about 30,000-60,000 iterations. That should be enough for good convergence with Nesterov acceleration.

Multi-phase optimizer for minimizing the normalized autocorrelation peak.
Phase 1: Nesterov-accelerated subgradient descent (multi-start)
Phase 2: Coordinate descent with incremental updates
Phase 3: Simulated annealing
Phase 4: Final polish
"""

import numpy as np
import time

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c)

def _nesterov_optimize(a, deadline, rng, n_peaks=100):
    """Nesterov-accelerated subgradient descent."""

    best = a.copy()

    eta = 0.3 / np.sqrt(n)
    mu = 0.9  # momentum
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(y, y)
        c_max = np.max(c)

            wts = np.ones(len(top_idx)) / len(top_idx)

            c_top = c[top_idx]

            else:

            w = wts[k_i]

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * y[k - idx]

        if gnorm < 1e-15:

        a, y = a_new, y_new
