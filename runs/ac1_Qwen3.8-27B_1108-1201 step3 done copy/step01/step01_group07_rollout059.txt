        k = int(k_int)
        wk = w[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            idx1 = k - js  # k-j values
            idx2 = js - k  # j-k values
            contrib = np.zeros(j_end - j_start + 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            contrib[v1] += a[idx1[v1]]
            contrib[v2] += a[idx2[v2]]
            g[j_start:j_end + 1] += wk * contrib
    return g


def do_optimize(a0, time_limit, temp0=50.0, eta0=0.4, momentum_beta=0.7):
    """
    Adaptive mirror descent with backtracking, momentum, and periodic FW steps.
    a0: initial non-negative vector (will be normalized).
    Returns (best_a, best_obj).
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fft_convolve(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()

    eta = eta0
    temp = temp0
    momentum = np.zeros(n)
    iter_count = 0
    patience = 0
    max_patience = 100
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Soft-max weights
        scaled = temp * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        g = compute_subgrad(a, w)
        # Center (constant doesn't matter on simplex)
        g_centered = g - g.mean()

        # Momentum
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * g_centered

        # Backtracking line search
        found = False
        for _ in range(20):
            log_factor = -eta * momentum
            log_factor -= log_factor.max()
            a_new = a * np.exp(log_factor)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-12:
                break
            a_new = a_new / s
            new_conv = fft_convolve(a_new)
            new_obj = 2.0 * n * new_conv.max()
            if new_obj < best_obj - 1e-10:
                a = a_new
                best_obj = new_obj
                best_a = a_new.copy()
                conv = new_conv
                patience = 0
                found = True
                break
            elif new_obj < compute_obj_normalized(a) * 1.001:
                # Acceptable but not improving best; still take the step
                a = a_new
                conv = new_conv
                found = True
                break
            eta *= 0.5
            if eta < 0.001:
                break

        if not found:
            # Reset step size and try a FW step for diversification
            eta = eta0 * 0.5
            # FW step: move toward coordinate with lowest subgradient
            j = int(np.argmin(g_centered))
            t_fw = 0.1
            a = (1.0 - t_fw) * a + t_fw * np.zeros(n)
            a[j] = t_fw
            a = normalize(a)
            conv = fft_convolve(a)
            # Reset momentum
            momentum *= 0.5

        iter_count += 1

        # Adaptive temperature: gradually sharpen
        if iter_count % 50 == 0:
            temp *= 1.15
            if temp > 500.0:
                temp = 500.0

        # Periodic perturbation to escape local minima
        if iter_count % 200 == 0:
            rng_local = np.random.default_rng(iter_count)
            noise = rng_local.normal(0, 0.05, n)
            a = a + noise * a
            a = normalize(a)
            momentum *= 0.3
            conv = fft_convolve(a)

        # Patience-based restart of parameters
        if patience > max_patience:
            patience = 0
            eta = min(eta0, eta * 1.5)
            temp = max(20.0, temp * 0.7)
            momentum *= 0.3
            max_patience = max(50, max_patience - 5)

        # Check progress
        if patience == 0:
            max_patience = 100

    return best_a, best_obj


def do_optimize_v2(a0, time_limit):
    """
    Alternative optimizer: combines mirror descent with periodic
    exact coordinate refinement and harder active set focus.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = 0.3
    temp = 30.0
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        phase = (iter_count % 300)

        if phase < 200:
            # Mirror descent phase
            conv = fft_convolve(a)
            max_conv = conv.max()
            if max_conv < 1e-15:
                break

            # Focus on top peaks (harder active set)
            top_k = max(5, int(0.05 * len(conv)))
            top_indices = np.argsort(conv)[-top_k:]
            top_vals = conv[top_indices]
            # Soft-max only over top peaks
            scaled = temp * top_vals / top_vals.max()
            scaled -= scaled.max()
            w_top = np.exp(scaled)
            w_top /= w_top.sum()

            # Build full weight vector
            w = np.zeros(len(conv))
            w[top_indices] = w_top

            g = compute_subgrad(a, w)
            g_centered = g - g.mean()

            # Multiplicative update
            log_factor = -eta * g_centered
            log_factor -= log_factor.max()
            a_new = a * np.exp(log_factor)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-12:
                a_new = a_new / s
                new_conv = fft_convolve(a_new)
                new_obj = 2.0 * n * new_conv.max()
                if new_obj <= best_obj * 1.01:
                    a = a_new
                    conv = new_conv
                    if new_obj < best_obj - 1e-10:
                        best_obj = new_obj
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
                    eta *= 0.7
            else:
                no_improve += 1
                eta = 0.3

            # Adapt temperature
            if no_improve > 50:
                temp = min(temp * 1.3, 300.0)
                no_improve = 0

        elif phase < 250:
            # Coordinate refinement phase
            # For each coordinate, try small adjustments
            rng_local = np.random.default_rng(iter_count)
            for _ in range(5):
                j = rng_local.integers(0, n)
                # Try increasing a[j] and proportionally decreasing others
                delta = rng_local.uniform(-0.1, 0.1) * a[j]
                a_try = a.copy()
                a_try[j] = max(0, a_try[j] + delta)
                a_try = normalize(a_try)
                try_conv = fft_convolve(a_try)
                try_obj = 2.0 * n * try_conv.max()
                if try_obj < best_obj - 1e-12:
                    a = a_try
                    best_obj = try_obj
                    best_a = a.copy()

        else:
            # Diversification phase: random perturbation
            rng_local = np.random.default_rng(iter_count + 777)
            strength = rng_local.uniform(0.05, 0.2)
            a_perturbed = a * (1.0 + strength * rng_local.normal(0, 1, n))
            a = normalize(a_perturbed)
            eta = 0.3
            temp = 30.0

        iter_count += 1
        if time.monotonic() - t0 > time_limit - 5:
            break

    return best_a, best_obj


def run_search(seed, budget_s):
    """Main search combining multiple strategies."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Try to access parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    def update_best(seq, score):
        nonlocal best_sequence, best_score
        if score < best_score:
            best_score = score
            best_sequence = seq.copy()
            if best_score < 1.51:
                print(f"  NEW BEST: {score:.6f} (n={len(seq)})")

    # Phase 1: Scan different n and initializations
    print("Phase 1: Scanning...")
    lengths = [128, 256, 512, 1024]

    scan_configs = []
    for n in lengths:
        x = np.arange(n, dtype=np.float64)
        # Uniform
        scan_configs.append((n, 'uniform', np.ones(n) / n))
        # Raised cosine
        rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        scan_configs.append((n, 'raised_cos', rc / rc.sum()))
        # Exponential (left-heavy)
        exp_l = np.exp(-x / (n * 0.3))
        scan_configs.append((n, 'exp_left', exp_l / exp_l.sum()))
        # Exponential (right-heavy)
        exp_r = np.exp(-(n - 1 - x) / (n * 0.3))
        scan_configs.append((n, 'exp_right', exp_r / exp_r.sum()))
        # Bimodal (edges heavy)
        bi = np.zeros(n)
        edge = max(1, n // 6)
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        # Add some middle
        bi[n//4:3*n//4] = 0.3
        scan_configs.append((n, 'bimodal', bi / bi.sum()))
        # Random Dirichlet
        r = rng.dirichlet(np.ones(n) * 1.0)
        scan_configs.append((n, 'random', r))
        # Linear ramp
        ramp = np.arange(1, n + 1, dtype=np.float64)
        scan_configs.append((n, 'ramp', ramp / ramp.sum()))
        # Parent (if available)
        if parent is not None:
            if parent.size >= n:
                pp = parent[:n].copy()
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    scan_configs.append((n, 'parent', pp / pp.sum()))
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    scan_configs.append((n, 'parent_interp', pp / pp.sum()))

    # Quick scan: limited time per config
    n_total = len(scan_configs)
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.3)
    time_per_scan = max(5.0, phase1_budget / max(1, n_total))

    scan_results = []
    for i, (n, name, a0) in enumerate(scan_configs):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(time_per_scan, deadline - time.monotonic() - 120)
        if t_alloc < 3:
            break

        a_opt, obj_opt = do_optimize(a0, time_limit=t_alloc, temp0=40.0, eta0=0.4)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            update_best(a_opt, score)
            print(f"  Scan n={n:4d} {name:15s} score={score:.6f}")

    # Phase 2: Deep optimization on top results
    print(f"Phase 2: Deep optimization. Best: {best_score:.6f}")
    scan_results.sort(key=lambda x: x[0])

    # Select top configs with diversity in n
    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        top_configs.append((n, name, a_opt))
        seen_n.add(n)
        if len(top_configs) >= 8:
            break
    # Add some diversity
    for score, n, name, a_opt in scan_results:
        if n not in {c[0] for c in top_configs} and len(top_configs) < 12:
            top_configs.append((n, name, a_opt))

    remaining = deadline - time.monotonic() - 60
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break

        # Try optimizer v1
        t_half = t_alloc * 0.5
        a_opt1, _ = do_optimize(a0, time_limit=t_half, temp0=60.0, eta0=0.35)
        score1 = float(evaluate_sequence(a_opt1.tolist()))
        if np.isfinite(score1):
            update_best(a_opt1, score1)

        # Try optimizer v2 with the result
        a_opt2, _ = do_optimize_v2(a_opt1, time_limit=t_alloc - t_half)
        score2 = float(evaluate_sequence(a_opt2.tolist()))
        if np.isfinite(score2):
            update_best(a_opt2, score2)
            print(f"  Deep n={n:4d} {name:15s} score={min(score1, score2):.6f}")

    # Phase 3: Final polish on the best
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print(f"Phase 3: Final polish. Current best: {best_score:.6f}")
        remaining = deadline - 15 - time.monotonic()
        n_best = len(best_sequence)

        # Multiple polish attempts with different parameters
        attempts = [
            (60.0, 0.3, 500.0),
            (100.0, 0.2, 300.0),
            (200.0, 0.15, 200.0),
            (50.0, 0.25, 400.0),
        ]
        time_per_attempt = max(5.0, remaining / len(attempts))

        for temp_p, eta_p, _ in attempts:
            if time.monotonic() > deadline - 10:
                break
            t_single = min(time_per_attempt, deadline - time.monotonic() - 10)
            if t_single < 3:
                break
            a_opt, _ = do_optimize(best_sequence, time_limit=t_single,
                                   temp0=temp_p, eta0=eta_p)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                update_best(a_opt, score)

        # One more round with v2
        if time.monotonic() < deadline - 10:
            t_single = deadline - 10 - time.monotonic()
            a_opt, _ = do_optimize_v2(best_sequence, time_limit=t_single)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                update_best(a_opt, score)

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Searches for the best sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2 using adaptive subgradient methods.
    """
    result = run_search(seed=seed, budget_s=budget_s)
    # Ensure validity
    result = np.clip(result, 0.0, 1000.0)
    return [float(x) for x in result]
