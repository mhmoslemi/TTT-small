The "iterative spectral inversion" method is interesting: 
1. Start with p
2. Compute its autocorrelation R = p*p
3. Modify R to have lower peak (e.g., R_new = min(R, threshold) at the peak, keep rest)
4. Invert: find p_new such that p_new*p_new ≈ R_new (this is spectral factorization)
5. Iterate

But step 4 is non-trivial for linear convolution. For circular convolution, it's easy (take the square root of the DFT). For linear, it's harder.

Actually, we can embed the linear convolution into a circular convolution by zero-padding. 1. Zero-pad p to length L >= 2n-1 (use next power of 2)
2. Compute circular autocorrelation: R_circ = IFFT(|FFT(p_padded)|²)
3. The first 2n-1 values of R_circ equal the linear autocorrelation
4. Modify R_circ to reduce the peak
5. Spectral factorization: p_new_padded = IFFT(sqrt(FFT(R_new)))  (taking the principal square root)
6. Truncate to first n values, renormalize

This is a form of the " direct Fourier method" or "Spectral Inversion Method" (SIM) used in signal processing.

"""
Smooth gradient-based optimization to minimize the autocorrelation peak
of a non-negative step function, using log-sum-exp smoothing and
multiplicative updates. Combines with structured exploration to escape
local minima.
"""
import numpy as np
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def smooth_objective(p, beta):
    """Compute smooth approximation of max(p*p) using log-sum-exp."""
    conv = np.convolve(p, p)
    # Log-sum-exp for numerical stability
    max_c = conv.max()
    log_sum_exp = max_c + np.log(np.sum(np.exp(beta * (conv - max_c))))
    return log_sum_exp / beta

def compute_gradient(p, beta):
    """Compute gradient of the smooth objective w.r.t. p."""
    n = len(p)
    conv = np.convolve(p, p)
    # Softmax weights
    max_c = conv.max()
    exp_bc = np.exp(beta * (conv - max_c))
    weights = exp_bc / exp_bc.sum()
    # Gradient: 2 * (weights * flip(p)) - 2/sum(p)
    # (weights * flip(p))[j] = sum_k weights[k] * p[k-j] for valid k-j
    # This is the convolution of weights and flip(p), truncated to length n
    flipped_p = p[::-1].copy()
    # We need grad[j] = 2 * sum_k weights[k] * p[k-j] for 0<=k-j<n
    # = 2 * (weights * flip(p))[j] where * is full convolution, take first n elements
    # Actually: sum_k weights[k]*p[k-j] where 0<=k-j<n means k >= j and k <= j+n-1
    # This is the cross-correlation of weights and p, or convolution of weights and flip(p)
    full_conv = np.convolve(weights, flipped_p, mode='full')
    # full_conv has length len(weights) + n - 1 = (2n-1) + n - 1 = 3n-2
    # We want indices j = 0,...,n-1 of sum_k weights[k]*p[k-j]
    # weights has indices 0..2n-2, p has indices 0..n-1
    # sum_k weights[k]*p[k-j] for k=0..2n-2: p[k-j] is valid when 0<=k-j<=n-1
    # = convolution of weights with p (not flip), but with a sign flip in the index
    # Actually, let me think again.
    # grad[j] = 2 * sum_k weights[k] * p[k-j] * 1[0<=k-j<n]
    # Let m = k-j, so k = j+m, m = 0..n-1
    # = 2 * sum_{m=0}^{n-1} weights[j+m] * p[m] * 1[j+m <= 2n-2]
    # This is the cross-correlation: (weights ⋆ p)[j] = sum_m weights[j+m]*p[m]
    # = np.correlate(weights, p, mode='valid')? No...
    # Actually it's: for each j, dot(weights[j:j+n], p) (with truncation)
    # This is the "sliding dot product" = cross-correlation
    # In numpy: np.correlate(weights, p, mode='full')[n-1:n-1+n] 
    # Wait, np.correlate(a, v, mode='full')[k] = sum_n a[n+k]*conj(v[n]) (for suitable range)
    # Hmm, let me just use convolution.
    # sum_m weights[j+m]*p[m] = (weights * flip(p))[j + n-1] ... no
    # Let's use: convolve(weights, p[::-1], 'full')[n-1 + j] for j=0..n-1?
    # convolve(a, b, 'full')[k] = sum_i a[i]*b[k-i]
    # If b = p[::-1], then b[k-i] = p[n-1-(k-i)] = p[n-1-k+i]
    # convolve(weights, p[::-1], 'full')[k] = sum_i weights[i]*p[n-1-k+i]
    # Let j = k - (n-1), so k = j + n - 1
    # = sum_i weights[i]*p[n-1-(j+n-1)+i] = sum_i weights[i]*p[i-j]
    # We want sum_m weights[j+m]*p[m]. Let i = j+m, m = i-j:
    # = sum_i weights[i]*p[i-j] for valid i, j
    # This matches! So grad[j]/2 = convolve(weights, p[::-1], 'full')[j + n - 1]
    
    grad_conv = np.convolve(weights, flipped_p, mode='full')
    # indices n-1 to n-1+n-1 = 2n-2
    grad_part = grad_conv[n-1:2*n-1]  # length n
    grad = 2.0 * grad_part - 2.0 / p.sum()
    return grad

def multiplicative_update(p, grad, eta):
    """Multiplicative weight update on the simplex."""
    p_new = p * np.exp(-eta * grad)
    p_new = np.maximum(p_new, 1e-30)  # avoid exact zeros
    p_new = p_new / p_new.sum()
    return p_new

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    # Simple approach: clip and normalize
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

def smooth_optimize(p_init, n_steps=5000, beta_start=5.0, beta_end=200.0,
                    eta_start=1.0, eta_end=0.01, deadline=None):
    """Run smooth gradient optimization."""
    p = p_init.copy()
    p = np.maximum(p, 1e-10)
    p = p / p.sum()
    best = p.copy()
    best_sc = fast_score(p)
    
    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Anneal beta and eta
        frac = it / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac
        
        # Compute gradient
        grad = compute_gradient(p, beta)
        
        # Multiplicative update
        p_new = multiplicative_update(p, grad, eta)
        
        # Evaluate
        sc = fast_score(p_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = p_new.copy()
        
        p = p_new
        
        if it % 500 == 0:
            print("  smooth_opt it=%d beta=%.1f eta=%.4f score=%.6f" % (it, beta, eta, best_sc), flush=True)
    
    return best, best_sc

def block_perturb(f, rng, width_range=(2, 50)):
    """Multiplicative random block perturbation."""
    n = len(f)
    cand = f.copy()
    w = int(rng.integers(width_range[0], min(width_range[1], n) + 1))
    if w >= n:
        w = max(2, n // 10)
    start = int(rng.integers(0, n - w + 1))
    stop = min(n, start + w)
    # Random multiplicative factor
    factor = float(np.exp(rng.normal(0.0, 0.3)))
    cand[start:stop] *= factor
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

def local_search(f_init, deadline, rng):
    """Block perturbation local search."""
    p = f_init.copy()
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return f_init, fast_score(f_init)
    p = p / s
    best = p.copy()
    best_sc = fast_score(p)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Choose perturbation width based on progress
        n = len(p)
        if it % 100 == 0:
            w_high = max(5, n // 20)
        else:
            w_high = max(3, n // 100)
        cand = block_perturb(p, rng, width_range=(1, w_high))
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
        # Accept with some probability (simulated annealing)
        # For simplicity, always accept if better, sometimes accept if worse (early on)
        if sc < best_sc + 0.01:  # accept small worsening
            p = cand
    return best, best_sc

def make_initial(n, rng, method='uniform'):
    """Generate initial sequence."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'perturbed_uniform':
        a = np.ones(n) + rng.normal(0.0, 0.3, n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sine_sq':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - n/2) / (n/4)) ** 2)
    elif method == 'two_bumps':
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - n*0.25) / (n*0.1)) ** 2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.1)) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

"""
Smooth gradient-based optimization for minimizing the autocorrelation peak
of a non-negative step function. Uses log-sum-exp smoothing, multiplicative
updates, spectral initialization, and multi-scale local search.
"""
import numpy as np
import time

def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def compute_conv(p):
    """Compute linear convolution p*p."""
    return np.convolve(p, p)

def smooth_grad(p, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*(p*p))) - 2*log(sum p).
    Returns gradient vector of same shape as p.
    """
    n = len(p)
    conv = compute_conv(p)
    # Log-sum-exp weights
    max_c = conv.max()
    exp_bc = np.exp(beta * (conv - max_c))
    sum_exp = exp_bc.sum()
    if sum_exp < 1e-300:
        weights = np.ones(len(conv)) / len
        weights = np.ones(len(conv)) / len(conv)
    else:
        weights = exp_bc / sum_exp
    
    # Gradient: 2 * (weights * flip(p)) - 2/sum(p)
    # (weights * flip(p))[j] = sum_k weights[k] * p[k-j] for valid indices
    # = convolve(weights, flip(p), 'full')[j + n - 1]
    flipped = p[::-1]
    full_conv = np.convolve(weights, flipped, mode='full')
    grad_part = full_conv[n-1:2*n-1]  # length n
    grad = 2.0 * grad_part - 2.0 / p.sum()
    return grad

def multiplicative_step(p, grad, eta):
    """Apply multiplicative weight update."""
    log_p = np.log(np.maximum(p, 1e-30))
    log_p_new = log_p - eta * grad
    # Clip for numerical stability
    log_p_new = np.clip(log_p_new, -30, 30)
    p_new = np.exp(log_p_new)
    p_new = np.maximum(p_new, 1e-30)
    p_new = p_new / p_new.sum()
    return p_new

def smooth_optimize(p_init, deadline, beta_start=5.0, beta_end=500.0,
                    eta_start=0.5, eta_end=0.001, max_iters=50000):
    """Run smooth gradient optimization with annealing."""
    p = p_init.copy()
    p = np.maximum(p, 1e-10)
    p = p / p.sum()
    best = p.copy()
    best_sc = fast_score(p)
    
    n = len(p)
    # Precompute convolution length
    conv_len = 2 * n - 1
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        # Anneal
        beta = beta_start * (beta_end / beta_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac
        
        # Compute gradient
        grad = smooth_grad(p, beta)
        
        # Step
        p_new = multiplicative_step(p, grad, eta)
        
        # Evaluate (every few steps for speed)
        if it % 10 == 0:
            sc = fast_score(p_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = p_new.copy()
        
        p = p_new
        it += 1
    
    # Final evaluation
    sc = fast_score(p)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best = p.copy()
    
    return best, best_sc, it

def spectral_init(n, sidelobe_ratio=0.3, rng=None):
    """Initialize using spectral factorization for a target flat-ish autocorrelation."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    L = 1
    while L < 2 * n:
        L *= 2
    
    # Target autocorrelation
    R = np.zeros(L)
    # Try different sidelobe ratios
    for trial in range(5):
        ratio = sidelobe_ratio * (1 + 0.2 * rng.standard_normal())
        ratio = max(0.05, min(0.9, ratio))
        R[0] = 1.0  # Main lobe
        # Sidelobes with some structure
        for k in range(1, min(2*n-1, L)):
            # Taper the sidelobes to make the spectrum valid
            taper = 1.0 - 0.5 * k / max(1, 2*n-2)
            R[k] = ratio * taper
    
    # Compute spectrum
