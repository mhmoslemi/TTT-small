from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, n, top_indices, c, temperature=1.0):
    """Soft-weighted subgradient using exponential weighting over top lags."""
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    # Use exponential weighting based on how close each lag is to the max
    max_val = top_vals[0]
    if max_val <= 1e-15:
        return grad
    # Weight by exp(-temperature * (max_val - val))
    weights = np.exp(-temperature * (max_val - top_vals))
    weights /= weights.sum()
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad
