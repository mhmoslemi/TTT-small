```python
def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def soft_max_gradient(c, tau):
    """Compute weights for soft-max."""
    shifted = c - c.max()
    w = np.exp(tau * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w

def compute_grad(a, w):
    """Compute gradient of soft_max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    # w has length 2n-1
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k (0 <= k < 2n-1, 0 <= k-j < n)

        if k_lo <= k_hi:
```

```python
def optimize_soft_max(a0, deadline, tau_start=5.0, tau_end=200.0, 
                      max_iters=50000, adam_lr=0.01, 
                      beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a)
    S_t = np.sqrt(2.0 * n)
    
    best = a.copy()
    best_score = compute_obj(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    # FFT size
    L = 1
    while L < 3 * n:
        L *= 2
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Anneal tau exponentially
        progress = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute convolution and soft-max weights
        c = np.convolve(a, a)
        w = compute_soft_weights(c, tau)
        
        # Compute gradient via FFT
        W = np.fft.rfft(w, L)
        A = np.fft.rfft(a, L)
        corr = np.fft.irfft(W * np.conj(A), L)
        grad = 2.0 * corr[:n]
        
        # Center gradient (preserve sum)
        grad -= grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a = a - adam_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project: non-negative and normalize
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)
        
        # Check improvement periodically
        if (it + 1) % 500 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Final check
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score
```

However, the LP matrix is sparse (each row has at most n non-zeros, but on average about n/2). Actually, each row `k` has non-zeros at positions `j` where `0 <= k-j < n`, i.e., `max(0, k-n+1) <= j <= min(n-1, k)`. The number of non-zeros in row `k` is `min(k+1, n, 2n-1-k)`.

For the LP, I'll use scipy's `linprog` with the HiGHS method.

Actually, for the LP to be useful, I need ALL constraints to be enforced (otherwise the solution might violate some constraint and the blend won't actually reduce the peak). So I need the full LP.

For n=512, the LP has 512 variables and 1023 inequality constraints. The constraint matrix is 1023 x 512 with about 512*512/2 ≈ 131K non-zeros (triangular structure). This should be solvable in < 1 second with HiGHS.

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Strategy: Soft-max gradient descent with Adam optimizer and FFT-accelerated
gradients, followed by LP-based refinement and top-k polishing.
Multiple starting points of various lengths are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = np.sum(a)
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def soft_weights(c, tau):
    """Compute soft-max weights for convolution values."""
    shifted = c - np.max(c)
    w = np.exp(tau * shifted)
    s = np.sum(w)
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def topk_weights(c, k):
    """Compute uniform weights on the top-k largest convolution values."""
    w = np.zeros_like(c)
    idx = np.argsort(c)[-k:]
    w[idx] = 1.0 / k
    return w


def gradient_fft(a, w, fft_size):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a, using FFT cross-correlation."""
    n = len(a)
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    return 2.0 * corr[:n]


def gradient_direct(a, w):
    """Compute gradient directly (for small n)."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            grad[j] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[:k_hi - k_lo + 1])
    return grad


def optimize_adam(a0, deadline, tau_start=3.0, tau_end=300.0,
                  max_iters=100000, lr=0.05, beta1=0.9, beta2=0.999,
                  eps=1e-8, eval_every=200):
    """Soft-max gradient descent with Adam optimizer."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = normalize(a)
    S_t = np.sqrt(2.0 * n)

    best = a.copy()
    best_score = compute_obj(a)

    # FFT size
    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)

    t0 = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Exponential tau annealing
        progress = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute gradient
        c = np.convolve(a, a)
        w = soft_weights(c, tau)
        grad = gradient_fft(a, w, fft_size)

        # Center gradient to preserve sum
        grad -= np.mean(grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        # Evaluate
        if (it + 1) % eval_every == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    # Final evaluation
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def optimize_topk(a0, deadline, k_start=100, k_end=1,
                  max_iters=50000, lr=0.02, beta1=0.9, beta2=0.999,
                  eps=1e-8, eval_every=100):
    """Top-k gradient descent with Adam."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)

    best = a.copy()
    best_score = compute_obj(a)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2

    m = np.zeros(n)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal k from k_start to k_end
        progress = min(1.0, it / max_iters)
        k = max(1, int(k_start * (k_end / k_start) ** progress))

        c = np.convolve(a, a)
        w = topk_weights(c, k)
        grad = gradient_fft(a, w, fft_size)
        grad -= np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        if (it + 1) % eval_every == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score


def lp_refine(a0, deadline, max_iters=200):
    """LP-based refinement: alternate LP solve and gradient step."""
    n = len(a0)
    if n > 1024:
        return a0, compute_obj(a0)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        M = float(np.max(c))

        # Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, j] = a[k - j]
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        if g is None or np.sum(g) < 1e-10:
            break

        # Try blending
        improved = False
        for t in [0.7, 0.5, 0.3, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Small gradient step to escape
            c = np.convolve(a, a)
            w = soft_weights(c, 50.0)
            grad = gradient_fft(a, w, fft_size)
            grad -= np.mean(grad)

            adam_t += 1
            m_adam = 0.9 * m_adam + 0.1 * grad
            v_adam = 0.999 * v_adam + 0.001 * grad * grad
            m_hat = m_adam / (1 - 0.9 ** adam_t)
            v_hat = v_adam / (1 - 0.999 ** adam_t)
            a = a - 0.01 * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = np.maximum(a, 0.0)
            a = normalize(a, S_t)

            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s

    return best, best_score


def perturb_refine(a0, deadline, n_perturbs=500, scale=0.1):
    """Try random perturbations and keep improvements (hill climbing)."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)

    rng = np.random.default_rng(123)

    for _ in range(n_perturbs):
        if time.monotonic() > deadline:
            break

        # Random perturbation
        delta = rng.normal(0, scale, size=n)
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        s_new = compute_obj(a_new)

        if s_new < best_score:
            a = a_new
            best = a.copy()
            best_score = s_new

    return best, best_score


def coordinate_descent(a0, deadline, n_sweeps=10):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    S = np.sum(a)
    best = a.copy()
    best_score = compute_obj(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for j in range(n):
            # Try increasing and decreasing a[j]
            for delta_frac in [0.1, 0.5, 1.0, -0.1, -0.5]:
                a_trial = a.copy()
                a_trial[j] = a[j] * (1.0 + delta_frac)
                if a_trial[j] < 0:
                    continue
                s = compute_obj(a_trial)
                if s < best_score:
                    a = a_trial
                    best = a.copy()
                    best_score = s

    return best, best_score
```

```python
"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-accelerated soft-max gradient descent with Adam optimizer, top-k gradient
polishing, LP-based refinement, and perturbation-based local search.
Explores multiple starting points from height_sequence_1 and structured patterns.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def convolve_fft(a):
    """Compute convolve(a, a) using FFT for speed."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2 * n - 1]


def normalize(a, target=None):
    """Normalize sequence to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = np.sum(a)
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def soft_weights(c, tau):
    """Compute soft-max weights for convolution values."""
    shifted = c - np.max(c)
    w = np.exp(tau * shifted)
    s = np.sum(w)
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def topk_weights(c, k):
    """Uniform weights on top-k largest convolution values."""
    w = np.zeros_like(c)
    idx = np.argsort(c)[-k:]
    w[idx] = 1.0 / k
    return w


def gradient_fft(a, w, fft_size):
    """Gradient of sum_k w[k]*c[k] w.r.t. a via FFT cross-correlation."""
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    n = len(a)
    return 2.0 * corr[:n]


def optimize_adam(a0, deadline, tau_start=3.0, tau_end=300.0,
                  max_iters=200000, lr=0.05, beta1=0.9, beta2=0.999,
                  eps=1e-8, eval_every=200, restart_patience=20000):
    """Soft-max gradient descent with Adam. Includes restart on stagnation."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)

    best = a.copy()
    best_score = compute_obj(a)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2

    m = np.zeros(n)
    v = np.zeros(n)
    last_improve = 0
    rng = np.random.default_rng(7)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress

        c = convolve_fft(a)
        w = soft_weights(c, tau)
        grad = gradient_fft(a, w, fft_size)
        grad -= np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        if (it + 1) % eval_every == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
                last_improve = it
            elif it - last_improve > restart_patience:
                # Restart with perturbation from best
                a = best + rng.normal(0, 0.05 * best.mean(), n)
                a = np.maximum(a, 0.0)
                a = normalize(a, S_t)
                m = np.zeros(n)
                v = np.zeros(n)
                last_improve = it

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    return best, best_score


def optimize_topk(a0, deadline, k_start=200, k_end=1,
                  max_iters=100000, lr=0.015, beta1=0.9, beta2=0.999,
                  eps=1e-8, eval_every=200):
    """Top-k gradient descent with Adam. Anneals k from large to 1."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)

    best = a.copy()
    best_score = compute_obj(a)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2

    m = np.zeros(n)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        k = max(1, int(k_start * (k_end / k_start) ** progress))

        c = convolve_fft(a)
        w = topk_weights(c, k)
        grad = gradient_fft(a, w, fft_size)
        grad -= np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        if (it + 1) % eval_every == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    return best, best_score


def lp_refine(a0, deadline, max_iters=200):
    """LP-based refinement: alternate LP solve and blend."""
    n = len(a0)
    if n > 1024:
        return a0.copy(), compute_obj(a0)

    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = convolve_fft(a)
        M = float(np.max(c))

        # Build LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        if g is None or np.sum(g) < 1e-10:
            break

        improved = False
        for t in [0.6, 0.4, 0.25, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                stall_count = 0
                break

        if not improved:
            stall_count += 1
            if stall_count > 10:
                break
            # Gradient escape
            c = convolve_fft(a)
            w = soft_weights(c, 80.0)
            grad = gradient_fft(a, w, fft_size)
            grad -= np.mean(grad)
            adam_t += 1
            m_adam = 0.9 * m_adam + 0.1 * grad
            v_adam = 0.999 * v_adam + 0.001 * grad * grad
            m_hat = m_adam / (1 - 0.9 ** adam_t)
            v_hat = v_adam / (1 - 0.999 ** adam_t)
            a = a - 0.015 * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = np.maximum(a, 0.0)
            a = normalize(a, S_t)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
                stall_count = 0

    return best, best_score


def perturb_search(a0, deadline, n_trials=5000, scales=None):
    """Hill-climbing with random perturbations of varying scale."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    best = a.copy()
    best_score = compute_obj(a)
    rng = np.random.default_rng(42)

    if scales is None:
        scales = [0.02, 0.05, 0.1, 0.2, 0.01, 0.005]

    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        scale = scales[rng.integers(len(scales))]
        # Sparse perturbation: only change a few coordinates
        n_change = rng.integers(1, max(2, n // 10) + 1)
        idx = rng.choice(n, size=n_change, replace=False)
        delta = rng.normal(0, scale * best.mean(), size=n_change)
        a_trial = a.copy()
        a_trial[idx] = np.maximum(a_trial[idx] + delta, 0.0)
        s = compute_obj(a_trial)
        if s < best_score:
            a = a_trial
            best = a.copy()
            best_score = s

    return best, best_score


def high_tau_polish(a0, deadline, tau=500.0, max_iters=100000, lr=0.005):
    """Final polish with very high tau (near-hard-max gradient)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """

    total_budget = float(budget_s)
    deadline = t_start + total_budget - 8.0

    best_seq = None
    best_score = float('inf')

    def try_update(seq, score, label=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [{label}] New best: {score:.6f}, n={len(seq)}")

    # --- Collect starting points ---
    starts = []
    start_labels = []

    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 64 and np.all(np.isfinite(gs)):
            for L in [512, min(768, len(gs)), min(384, len(gs))]:
                if L <= len(gs) and L >= 64:
                    starts.append(gs[:L].copy())
                    start_labels.append(f"hs1[{L}]")
    except (NameError, Exception):
        pass

    # Structured starting points
    for n in [384, 512, 640]:
        i = np.arange(n, dtype=np.float64)
        starts.append(0.2 + 0.8 * np.abs(np.sin(2 * np.pi * i * 7.0 / n)))
        start_labels.append(f"sin7[{n}]")
        starts.append(np.exp(-0.5 * ((i - n * 0.3) / (n * 0.25)) ** 2) + 0.03)
        start_labels.append(f"gauss[{n}]")
        starts.append(0.3 + 0.7 * np.abs(np.sin(np.pi * i * i / (n * 2.0))))
        start_labels.append(f"chirp[{n}]")
        starts.append(rng.random(n) * 0.7 + 0.3)
        start_labels.append(f"rand[{n}]")

    # Prioritize: height_sequence_1 starts first, then by length
    starts_with_priority = []
    for idx, (s, lbl) in enumerate(zip(starts, start_labels)):
        if lbl.startswith("hs1"):
            priority = 0
        elif len(s) == 512:
            priority = 1
        else:
            priority = 2
        starts_with_priority.append((priority, len(s), idx))
    starts_with_priority.sort()
    ordered_indices = [x[2] for x in starts_with_priority]

    print(f"Starting optimization: {len(starts)} starts, budget={total_budget:.0f}s")

    # --- Phase 1: Adam soft-max on multiple starts ---
    phase1_end = t_start + 0.50 * (deadline - t_start)
    print(f"\n=== Phase 1: Soft-max Adam ({phase1_end - t_start:.0f}s budget) ===")

    for idx in ordered_indices:
        if time.monotonic() > phase1_end - 15:

        a0 = starts[idx]
        lbl = start_labels[idx]

        elapsed = time.monotonic() - t0
```
