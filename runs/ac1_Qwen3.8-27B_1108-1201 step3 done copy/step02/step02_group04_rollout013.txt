"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothed gradient descent
with Adam optimizer, periodic LP corrections, and diversity-preserving restarts.
Key innovation: Uses FFT-based gradient computation for the soft-max approximation
of the max-convolution objective, enabling fast per-iteration updates.
"""

import numpy as np
import time
import scipy.optimize as optimize
import scipy.sparse as sparse


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_softgrad(a, weights, n, fft_size):
    """Compute gradient of soft-max objective using FFT cross-correlation.
    grad_i = sum_k w_k * a_{k-i} for valid indices.
    """
    # Cross-correlation: (a * w_rev)[2n-2-i] gives grad_i
    w_rev = weights[::-1]
    # Need FFT size >= n + (2n-1) - 1 = 3n-2
    cg_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, cg_size)
    W = np.fft.rfft(w_rev, cg_size)
    conv_aw = np.fft.irfft(A * W, cg_size)[:3 * n - 2]
    grad = conv_aw[n - 1:2 * n - 1][::-1]
    return grad


def softmax_smoothgrad(f, tau):
    """Compute soft-max smoothed objective and gradient for normalized f.
    f should be non-negative with sum 1.
    Returns (obj_value, gradient_wrt_f).
    The gradient is for the unconstrained problem; we project afterwards.
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = tau * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    
    # Objective: (1/tau) * log(sum(exp(tau * conv)))
    # = (1/tau) * (tau * conv.max() + log(sum(exp(tau*(conv - conv.max())))))
    # = conv.max() + (1/tau) * log(sum(exp_vals))
    obj = conv.max() + np.log(exp_vals.sum()) / tau
    
    # Gradient
    grad = compute_softgrad(f, w, n, None)
    return obj, grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def make_lp_structure(n):
    """Precompute sparse matrix for LP cross-convolution constraints."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def adam_smooth_optimize(f0, time_limit, tau_init=50.0, lr_init=0.01, 
                         tau_decay=0.999, tau_min=500.0, lp_every=50):
    """Optimize using Adam on the soft-max smoothed objective with periodic LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_structure(n)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr = lr_init
    tau = tau_init
    
    t0 = time.monotonic()
    patience = 0
    last_lp_time = t0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute soft-max gradient
        obj_sm, grad = softmax_smoothgrad(f, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex
        f_new = project_simplex(f_new)
        
        # Evaluate exact objective
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            M = M_new
            patience = 0
        else:
            patience += 1
        
        f = f_new
        
        # Adaptive tau: increase tau over time (more exact max)
        if t % 10 == 0:
            tau = min(tau * 1.05, tau_min)
            lr *= 0.995
        
        # Periodic LP correction
        if t % lp_every == 0 and time.monotonic() - last_lp_time > 3:
            last_lp_time = time.monotonic()
            g = solve_lp(f, M, rows, cols, n, timeout=8)
            if g is not None:
                gs = g.sum()
                if gs > 1e-12:
                    g /= gs
                    # Blend

                        f_blend = (1 - alpha) * f + alpha * g
                        f_blend = project_simplex(f_blend)
                        cb = fast_convolve(f_blend)
                        ob = 2.0 * n * cb.max()
                        if ob < best_obj - 1e-13:
                            f = f_blend
                            best_obj = ob
                            best_f = f.copy()
                            M = cb.max()
                            break

            # Try a random perturbation
            noise = np.random.exponential(0.01, n)
            f = (1 + noise) * f
            f = project_simplex(f)
            patience = 0
            lr = lr_init * 0.5
            tau = max(tau_init, tau * 0.5)
            m = np.zeros(n)
            v = np.zeros(n)
            
            if patience > 500:
                break

def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3):
    """Polish with multiplicative mirror descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Gradient: cross-correlation of w with f
        grad = compute_softgrad(f, w, n, None)
        
        sg_c = grad - grad.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s
        
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 25 == 0:
                eta *= 0.7
                if eta < 0.008:
                    t_param = min(t_param * 1.5, 500.0)
                    eta = 0.2
            if patience > 300:
                break
    
    return best_f, best_obj


def lp_optimize(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_structure(n)
    
    t0 = time.monotonic()
    step = 1.0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        g = solve_lp(f, M, rows, cols, n, timeout=10)
        if g is None:
            break

        if gs < 1e-12:
            break
        g /= gs
        
        if np.max(np.abs(g - f)) < 1e-14:
            break
        
        improved = False
        for t_try in [step, 0.7*step, 0.5*step, 0.3*step, 0.1*step]:
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue

            obj_new = 2.0 * n * conv_new.max()

                M = conv_new.max()

                step = min(1.0, step * 1.1)
                improved = True
                break

def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr
    
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Adam optimizer with soft-max smoothed gradient (FFT-accelerated) as primary method
2. Periodic LP corrections for global direction guidance
3. Nesterov momentum mirror descent for polishing
4. Diverse initializations and multiple sequence lengths

Key innovation over previous approach: Uses true gradient of the objective (via
soft-max smoothing) rather than relying solely on the LP proxy direction,
allowing escape from LP-induced local optima.
"""

import numpy as np
import time
import scipy.optimize as optimize
import scipy.sparse as sparse


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_grad_fft(a, weights, n):
    """Compute gradient of soft-max objective via FFT cross-correlation.
    grad_i = sum_k w_k * a_{k-i} for valid indices.
    Uses (a * w_rev)[2n-2-i] trick.
    """
    cg_size = 1 << (3 * n - 2 - 1).bit_length()
    w_rev = weights[::-1]
    A = np.fft.rfft(a, cg_size)
    W = np.fft.rfft(w_rev, cg_size)
    conv_aw = np.fft.irfft(A * W, cg_size)[:3 * n - 2]
    grad = conv_aw[n - 1:2 * n - 1][::-1]
    return grad


def softmax_grad(f, tau):
    """Compute soft-max smoothed objective value and gradient for f on simplex."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = tau * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    obj = conv.max() + np.log(exp_vals.sum()) / tau
    grad = compute_grad_fft(f, w, n)
    return obj, grad, w


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def make_lp_sparse(n):
    """Build sparse matrix for cross-conv LP: A[k,i] = f[k-i] if valid."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_quick(f, M, rows, cols, n, timeout=8):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lp_alternate_optimize(f0, time_limit, lp_timeout=10):
    """LP alternating optimization with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f = f / f.sum()
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    rows, cols = make_lp_sparse(n)
    step = 1.0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        g = solve_lp_quick(f, M, rows, cols, n, timeout=lp_timeout)
        if g is None:

            break
        g = g / gs
        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [step, 0.7 * step, 0.5 * step, 0.3 * step, 0.1 * step]:
            f_new = (1 - t_try) * f + t_try * g

                f = f_new
                M = cn.max()
                best_obj = on
                best_f = f.copy()
                patience = 0
                step = min(1.0, step * 1.08)
                improved = True
                break

            patience += 1
            step *= 0.85
            if patience > 100:

def gen_inits(n, rng):
    """Generate diverse initial probability vectors of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uni'] = np.ones(n) / n
    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = g / g.sum()
        inits[f'g{int(sf * 100)}'] = g
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    cb = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()
    # Synthesized wave
    sw = 0.5 + 0.3 * np.cos(4.0 * np.pi * x / n) + 0.2 * np.cos(8.0 * np.pi * x / n)
    sw = np.maximum(sw, 0.05)
    inits['wave'] = sw / sw.sum()
    return inits
