import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def coordinate_ascent_sweep(a, peak_count=10, n_sweeps=1):
    """
    Perform coordinate ascent: for each coordinate, find the value that
    minimizes the score, given all other coordinates fixed.
    """
    n = len(a)
    S = np.sum(a)
    a = a.copy()
    best_score = local_score(a)
    best_a = a.copy()
    
    for sweep in range(n_sweeps):
        # Compute convolution once
        c = fft_convolve(a, a)
        S = np.sum(a)
        
        # Process coordinates in random order
        order = np.random.permutation(n)
        
        for j in order:
            # d_k = a[k-j] for valid k
            # k ranges from j to j+n-1 (where k-j is in [0, n-1])
            # c'[k] = c[k] + (x - a[j]) * a[k-j]
            
            # Get the interaction vector
            # For k in [0, 2n-2], d_k = a[k-j] if 0 <= k-j < n, else 0
            d = np.zeros(2*n-1)
            d[j:j+n] = a[:n]  # d[k] = a[k-j] for k in [j, j+n-1]
            
            B = S - a[j]  # Minimum possible new sum (when x=0)
            
            # Find top peaks
            peak_indices = np.argpartition(c, -peak_count)[-peak_count:]
            
            # Generate candidates
            candidates = set([0.0, a[j]])
            for k in peak_indices:
                dk = d[k]
                if dk > 1e-12:
                    xk = B - 2.0 * c[k] / dk
                    if xk >= 0:
                        candidates.add(xk)
                    # Also try x that makes c'[k] = 0 (if possible)
                    xk2 = a[j] - c[k] / dk
                    if xk2 >= 0:
                        candidates.add(xk2)
            
            # Evaluate each candidate
            best_x = a[j]
            best_local_score = 2.0 * n * np.max(c) / (S ** 2)  # current score
            
            for x in candidates:
                # New convolution
                c_new = c + (x - a[j]) * d
                S_new = B + x
                if S_new < 0.01:
                    continue
                max_c_new = np.max(c_new)
                # Ensure non-negative
                if np.min(c_new) < -1e-10:
                    # This shouldn't happen for non-negative a, but just in case
                    continue
                score = 2.0 * n * max_c_new / (S_new ** 2)
                if score < best_local_score:
                    best_local_score = score
                    best_x = x
            
            # Update
            if best_x != a[j]:
                c = c + (best_x - a[j]) * d
                S = B + best_x
                a[j] = best_x
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
    
    return best_a, best_score

def make_init(n, rng, style=0):
    t = np.arange(n, dtype=np.float64)
    if style == 0:
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / n * (1 + rng.random()))
    elif style == 1:
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * rng.uniform(0.5, 2.0) * t / n)
    elif style == 2:
        a = 1.0 + 0.2 * t / n
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kw):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)
    best_sc = np.inf
    best_seq = None

    # Initialize from provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        s0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        sc0 = score_of(s0)
        print(f"Provided: n={len(s0)}, score={sc0:.6f}")
        best_sc = sc0
        best_seq = s0.copy()

    if best_seq is None:
        best_seq = np.ones(512)
        best_sc = score_of(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_sc:.6f}")

    # Phase 1: Coordinate descent on the provided/best sequence
    print("\n=== Phase 1: Coord descent on initial ===")
    tl = min(300.0, deadline - time.monotonic() - 400.0)
    if tl > 30:
        seq, sc = coord_descent(best_seq, rng, peak=18, max_sweeps=500, tlimit=tl)
        if sc < best_sc:
            best_sc, best_seq = sc, seq
            print(f"  Improved: {best_sc:.6f}")
        print(f"  After Phase 1: {best_sc:.6f}")

    # Phase 2: Try different lengths by resampling
    print("\n=== Phase 2: Multi-length exploration ===")
    target_ns = [384, 512, 640, 768, 1024, 1280, 1536]
    for tn in target_ns:
        if time.monotonic() > deadline - 250.0:
            break
        a0 = resample(best_seq, tn)
        tl = min(100.0, max(30.0, (deadline - 250.0 - time.monotonic()) / max(1, len(target_ns))))
        seq, sc = coord_descent(a0, rng, peak=15, max_sweeps=400, tlimit=tl)
        if sc < best_sc:
            best_sc, best_seq = sc, seq
            print(f"  n={tn}: {best_sc:.6f} ***")
        else:
            print(f"  n={tn}: {sc:.6f}")

    # Phase 3: Fresh starts with different initializations
    print("\n=== Phase 3: Fresh starts ===")
    fresh_ns = [512, 768, 1024]
    for ci, (tn, style) in enumerate([(512, 0), (512, 1), (768, 0), (768, 2), (1024, 1), (512, 3), (768, 1), (1024, 0)]):
        if time.monotonic() > deadline - 150.0:
            break
        a0 = make_init(tn, rng, style)
        # Sometimes resample best
        if rng.random() < 0.3 and best_seq is not None:
            a0 = resample(best_seq, tn) * np.exp(rng.normal(0, 0.03, tn))
            a0 = np.clip(a0, 0.01, None)
        tl = min(80.0, max(20.0, (deadline - 150.0 - time.monotonic()) / max(1, len(fresh_ns))))
        seq, sc = coord_descent(a0, rng, peak=15, max_sweeps=300, tlimit=tl)
        if sc < best_sc:
            best_sc, best_seq = sc, seq
            print(f"  Fresh n={tn} s={style}: {best_sc:.6f} ***")
        else:
            print(f"  Fresh n={tn} s={style}: {sc:.6f}")

    # Phase 4: Perturbation restarts on best
    print("\n=== Phase 4: Perturbation restarts ===")
    for r in range(30):
        if time.monotonic() > deadline - 80.0:
            break
        noise_scale = 0.03 + 0.02 * r
        a0 = best_seq * np.exp(rng.normal(0, noise_scale, len(best_seq)))
        a0 = np.clip(a0, 0.0, None)
        tl = min(50.0, max(15.0, (deadline - 80.0 - time.monotonic()) / max(1, 30 - r)))
        if tl < 10:
            break
        seq, sc = coord_descent(a0, rng, peak=18, max_sweeps=300, tlimit=tl)
        if sc < best_sc:
            best_sc, best_seq = sc, seq
            print(f"  Restart {r}: {best_sc:.6f} ***")

    # Phase 5: Final polish with high peak count
    print(f"\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 20.0:
        tl = deadline - time.monotonic() - 15.0
        seq, sc = coord_descent(best_seq, rng, peak=25, max_sweeps=1000, tlimit=tl)
        if sc < best_sc:
            best_sc, best_seq = sc, seq
        print(f"  After final polish: {best_sc:.6f}")

    # Final verification
    result = [float(max(0.0, min(1000.0, v))) for v in best_seq]
    fs = evaluate_sequence(result)
    print(f"\n=== FINAL: score={fs:.6f}, n={len(result)} ===")
    return result
