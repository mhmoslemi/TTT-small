```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def normalize(a, target_sum=None):
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / s)

def lp_solve(a, M, max_iter=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix efficiently
    # conv(a,g)[k] = sum_i a[i] * g[k-i] for valid k-i
    # This is a Toeplitz-like matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k+1, n)):
            A[k, i] = a[k-i]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max_iter or 30})
    if res.status == 0:
        return res.x
    return np.zeros(n)
```

"""
Multi-phase optimization for minimizing the normalized peak convolution
of non-negative step functions. Combines LP-based iterative refinement with
proper line search, simulated annealing for global exploration, and targeted
local search. Exploits FFT-based convolution for speed and tries multiple
sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def conv_max(a):
    """Compute max of conv(a,a)."""
    return float(np.max(fft_convolve(a, a)))


def normalize(a, target_sum=None):
    """Normalize sequence to have a given sum."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / s)


def build_lp_matrix(a):
    """Build convolution constraint matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k + 1, n)
        for i in range(i_start, i_end):
            A[k, i] = a[k - i]
    return A


def lp_solve(a, M, time_limit=15.0):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M, g >= 0.
    Returns optimal g or None if infeasible/timeout.
    """
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status == 0:
        return res.x
    return None


def lp_blend_line_search(a, g, M, sum_target):
    """
    Find the best blend parameter t such that the blended sequence
    (1-t)*a + t*g (normalized) has conv_max <= M.
    Returns the best blended sequence and the peak value.
    """
    n = len(a)
    g_norm = normalize(g, sum_target)
    a_norm = normalize(a, sum_target)

    # Binary search for largest t where peak <= M
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_seq = a_norm.copy()
    best_peak = np.max(fft_convolve(a_norm, a_norm))

    for _ in range(30):
        mid = (lo + hi) / 2.0
        candidate = (1.0 - mid) * a_norm + mid * g_norm
        candidate = np.clip(candidate, 0.0, None)
        cnorm = normalize(candidate, sum_target)
        peak = np.max(fft_convolve(cnorm, cnorm))
        if peak <= M + 1e-10:
            best_t = mid
            best_seq = cnorm
            best_peak = peak
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-6:
            break

    # Also try: even if peak slightly exceeds M, if the score improves
    # (because the peak distribution is more even), accept it
    # Try t = best_t * 1.5 for a more aggressive step
    for t_try in [best_t * 1.5, best_t * 2.0, 0.5, 0.3, 0.2]:
        if t_try <= 0 or t_try > 1:
            continue
        candidate = (1.0 - t_try) * a_norm + t_try * g_norm
        candidate = np.clip(candidate, 0.0, None)
        cnorm = normalize(candidate, sum_target)
        peak = np.max(fft_convolve(cnorm, cnorm))
        # Accept if the score (which is peak since sum is fixed) is better
        if peak < best_peak:
            best_peak = peak
            best_seq = cnorm
            best_t = t_try

    return best_seq, best_peak, best_t


def lp_refine(a, time_limit, verbose=False):
    """LP-based iterative refinement with proper line search."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = conv_max(a)
    stall_count = 0

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        g = lp_solve(a, M, time_limit=min(10.0, time_limit - (time.monotonic() - t0)))
        if g is None:
            break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        new_a, new_peak, t_used = lp_blend_line_search(a, g, M, S)

        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_a = new_a.copy()
            a = new_a
            stall_count = 0
        elif new_peak < conv_max(a):
            a = new_a
            stall_count += 1
        else:
            stall_count += 1
            if stall_count > 30:
                break

        if verbose and iteration % 20 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    LP iter {iteration}: peak={best_peak:.8f}, score={score:.8f}, t={t_used:.4f}")

        iteration += 1

    return best_a


def simulated_annealing(a, time_limit, seed=42, verbose=False):
    """
    Simulated annealing with structured moves for global exploration.
    Moves are designed to reduce convolution peaks.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = conv_max(a)
    rng = np.random.default_rng(seed)

    T_init = (best_peak - 0.5 * best_peak)  # Initial temperature
    T_final = T_init * 1e-6
    cooling_rate = (T_final / T_init) ** (1.0 / max(1, int(2.0 / 0.01)))

    T = T_init
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T_init * (T_final / T_init) ** elapsed_frac

        conv = fft_convolve(a, a)
        current_peak = np.max(conv)

        # Generate a move
        move_type = rng.random()

        if move_type < 0.3:
            # LP-guided move
            g = lp_solve(a, current_peak, time_limit=5.0)
            if g is not None and np.sum(g) > 1e-10:
                g_norm = normalize(g, S)
                t_blend = rng.uniform(0.1, 0.7)
                candidate = (1.0 - t_blend) * a + t_blend * g_norm
                candidate = np.clip(candidate, 0.0, None)
                candidate = normalize(candidate, S)
            else:
                continue
        elif move_type < 0.5:
            # Smooth perturbation (add cosine/bump)
            candidate = a.copy()
            center = rng.uniform(0.1, 0.9) * n
            width = rng.uniform(0.05, 0.3) * n
            amp = rng.uniform(0.02, 0.15) * np.mean(a)
            i = np.arange(n)
            candidate += amp * np.exp(-0.5 * ((i - center) / width) ** 2) * rng.choice([1, -1])
            candidate = np.clip(candidate, 0.01, None)
            candidate = normalize(candidate, S)
        elif move_type < 0.65:
            # Block perturbation
            candidate = a.copy()
            width = int(rng.integers(max(2, n // 20), max(3, n // 4)))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.1))
            candidate[start:start + width] *= factor
            candidate = np.clip(candidate, 0.01, None)
            candidate = normalize(candidate, S)
        elif move_type < 0.75:
            # Peak-targeted reduction
            peak_idx = int(np.argmax(conv))
            candidate = a.copy()
            # Reduce elements contributing to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(2, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.03, size=top_k))
            # Increase some other elements
            other = np.setdiff1d(np.arange(n), top_idx)
            n_inc = max(2, n // 20)
            inc_idx = rng.choice(other, size=n_inc, replace=False)
            candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.03, size=n_inc))
            candidate = np.clip(candidate, 0.01, None)
            candidate = normalize(candidate, S)
        elif move_type < 0.85:
            # Sinusoidal perturbation
            candidate = a.copy()
            freq = rng.uniform(1, max(2, n // 10))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.1) * np.mean(a)
            i = np.arange(n)
            candidate += amp * np.cos(2 * np.pi * freq * i / n + phase)
            candidate = np.clip(candidate, 0.01, None)
            candidate = normalize(candidate, S)
        else:
            # Multi-peak targeted move
            top_peaks = np.argsort(conv)[-3:]
            candidate = a.copy()
            for pk in top_peaks:
                contribs = np.zeros(n)
                for i in range(n):
                    j = pk - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                top_k = max(1, n // 25)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.02, size=top_k))
            candidate = np.clip(candidate, 0.01, None)
            candidate = normalize(candidate, S)

        new_peak = conv_max(candidate)
        delta = new_peak - current_peak

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = candidate
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = candidate.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        step += 1

        if verbose and step % 100 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    SA step {step}: best_score={score:.6f}, T={T:.4f}, elapsed={time.monotonic()-t0:.0f}s")

        # Restart if stuck too long
        if no_improve > 500:
            # Add noise to escape
            a = a + rng.normal(0, 0.01 * np.mean(a), n)
            a = np.clip(a, 0.01, None)
            a = normalize(a, S)
            no_improve = 0
            T = min(T * 2, T_init * 0.1)

    return best_a


def targeted_local_search(a, time_limit, verbose=False):

def make_structured_init(n, rng):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    init_type = rng.integers(0, 6)

    if init_type == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(np.pi * t / (n - 1)))
    elif init_type == 1:
        # Gaussian centered
        center = n // 2
        sigma = n * (0.2 + 0.2 * rng.random())
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 2:
        # Multi-frequency cosine (all positive)
        a = np.ones(n)
        for _ in range(3):
            f = rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.3 * rng.random() * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 3:
        # Hann window
        a = 0.5 * (1 - np.cos(2 * np.pi * t / (n - 1)))
    elif init_type == 4:
        # Blackman window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t / (n - 1)) + 0.08 * np.cos(4 * np.pi * t / (n - 1))
    else:
        # Constant with small noise
        a = 1.0 + 0.1 * rng.standard_normal(n)

    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

        if time.monotonic() > deadline - 500:
            break
        for init_seed in range(3):
            if time.monotonic() > deadline - 480:
                break
            a0 = make_structured_init(n_try, np.random.default_rng(seed + init_seed * 100))
            t_per = min(30.0, max(10.0, (deadline - time.monotonic()) * 0.05))
            result = lp_refine(a0, t_per, verbose=False)
            update_best(result)

    print(f"  After phases 1-2: best_score={best_score:.8f}")

    # === Phase 3: Simulated annealing on best ===
    print("\n=== Phase 3: Simulated Annealing ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(400.0, max(100.0, (deadline - time.monotonic()) * 0.35))
        print(f"  SA on n={n}, budget={t_limit:.0f}s")
        result = simulated_annealing(best_seq, t_limit, seed=seed + 7, verbose=True)
        update_best(result)

    # === Phase 4: LP refinement again on best ===
    print("\n=== Phase 4: LP refinement (after SA) ===")
    if best_seq is not None:
        n = len(best_seq)
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_limit = min(remaining * 0.5, 200.0)
            print(f"  LP refine best (n={n}), budget={t_limit:.0f}s")
            result = lp_refine(best_seq, t_limit, verbose=True)
            update_best(result)

    # === Phase 5: Targeted local search ===
    print("\n=== Phase 5: Targeted local search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            t_limit = remaining
            print(f"  TLS budget={t_limit:.0f}s")
            result = targeted_local_search(best_seq, t_limit, verbose=True)
            update_best(result)

    # === Phase 6: Final LP polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            result = lp_refine(best_seq, remaining, verbose=False)
            update_best(result)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.8f}, local_score={best_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 640
