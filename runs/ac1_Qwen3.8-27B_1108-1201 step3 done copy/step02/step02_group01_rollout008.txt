```python
import numpy as np
import time

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)

def smoothed_max_gradient(a, tau):
    """Compute gradient of tau * logsumexp(conv(a,a) / tau) w.r.t. a."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(weights, size)
    
    # First term: corr[j] = sum_n a[n] * weights[n+j] for j in [0, n-1]
    corr = np.fft.irfft(np.conj(fw) * fa, size)[:n]
    
    # Second term: conv_fw[j] = sum_k a[j-k] * weights[k] for j in [0, n-1]
    conv_fw = np.fft.irfft(fa * fw, size)[:n]
    
    grad = corr + conv_fw
    return grad

def optimize_smoothed(a, tau_start, tau_end, n_steps, lr, momentum=0.9):
    """Optimize using smoothed max with annealing."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    v = np.zeros(n)  # Momentum
    
    for step in range(n_steps):
        # Anneal tau
        progress = step / max(1, n_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        grad = smoothed_max_gradient(a, tau)
        
        # Momentum update
        v = momentum * v + lr * grad
        a_new = a - v
        
        # Project: non-negative and sum to 1
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # Check score
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

Let me also think about **coordinate descent**: instead of updating all elements at once, update one element (or a small group) at a time. For each coordinate a[j], we can compute the optimal value analytically (by minimizing the smoothed objective over a single coordinate).

For a single coordinate update: if we change a[j] to a[j] + delta, the change in conv is:
conv_new[k] = conv[k] + delta * (a[k-j] + a[j-k]) (for valid indices)

The new max is max_k (conv[k] + delta * g_k) where g_k = a[k-j] + a[j-k].

We want to choose delta to minimize max_k (conv[k] + delta * g_k), subject to a[j] + delta >= 0.

This is a 1D optimization problem! The function max_k (conv[k] + delta * g_k) is a piecewise-linear function of delta (the max of linear functions). We can find its minimum by checking the breakpoints.

This is essentially a subgradient method in 1D, which is very efficient.

Let me code this up:

```python
def coordinate_update(a, j, conv):
    """Find optimal value of a[j] given fixed other coordinates."""
    n = len(a)
    # g_k = d conv[k] / d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    g = np.zeros(2*n-1)
    for k in range(2*n-1):
        val = 0.0
        if 0 <= k-j < n:
            val += a[k-j]
        if 0 <= j-k < n:
            val += a[j-k]
        g[k] = val
    
    # We want to minimize max_k (conv[k] + delta * g[k]) subject to a[j] + delta >= 0
    # i.e., delta >= -a[j]
    # The function f(delta) = max_k (conv[k] + delta * g[k]) is convex in delta
    # (it's the max of linear functions)
    # Its minimum is where the subgradient contains 0
    # Subgradient at delta: {sum of g[k] for k in argmax(conv[k] + delta*g[k])}
    # Minimum is where 0 is in the subgradient, i.e., there exist k1, k2 with g[k1] < 0 < g[k2]
    # and conv[k1] + delta*g[k1] = conv[k2] + delta*g[k2] = max
    
    # Simple approach: start with delta = 0 and do bisection on the subgradient
    # Or: since it's 1D convex, use ternary search or golden section
    
    # For speed, let's use the fact that the function is piecewise linear
    # and the minimum occurs at a breakpoint or where the subgradient contains 0
    
    # Practical approach: evaluate at a few candidate points
    # The minimum of max_k (c_k + delta * g_k) over delta >= -a[j]
    
    # If all g[k] >= 0, the minimum is at delta = -a[j] (pushing a[j] to 0)
    # If all g[k] <= 0, the minimum is at delta -> +inf (but bounded by normalization)
    # Otherwise, the minimum is where the two "active" constraints cross
    
    # For efficiency, let's just do a small grid search + refinement
    # Better: use the fact that the optimum is where the two largest "effective" values are equal
    
    # Let's use a simple approach: ternary search on [delta_min, delta_max]
    # delta_min = -a[j]
    # delta_max = a[j] * 2 (reasonable upper bound)
    
    delta_min = -a[j]
    delta_max = a[j] * 2 + 1e-6
    
    # Evaluate at a few points to narrow down
    for _ in range(50):
        m1 = delta_min + (delta_max - delta_min) / 3
        m2 = delta_max - (delta_max - delta_min) / 3
        f1 = np.max(conv + m1 * g)
        f2 = np.max(conv + m2 * g)
        if f1 < f2:
            delta_max = m2
        else:
            delta_min = m1
    
    delta_opt = (delta_min + delta_max) / 2
    new_val = max(0, a[j] + delta_opt)
    return new_val
```

The function f(delta) = max_k (conv[k] + delta * g[k]) is convex and piecewise linear. Its minimum is at a point where the subgradient contains 0. The subgradient at delta is:
∂f(∂elta) = co{g[k] : k in argmax(conv[k] + delta * g[k])}

The minimum is where min(∂f(delta)) <= 0 <= max(∂f(delta)).

We can find this by:
1. For each k, the "breakpoint" where conv[k] + delta * g[k] becomes the max is at the intersection with another line.
2. The minimum is at the leftmost breakpoint where the subgradient crosses 0.

- If at delta_min, all active constraints have d_k >= 0, the minimum is at delta_min
- Otherwise, the minimum is at the delta where the two "extreme" active constraints are equal

```python
def optimal_delta(conv, g, delta_min):
    # Find delta >= delta_min minimizing max_k (conv[k] + delta * g[k])
    # f(delta) is convex, piecewise linear
    
    # Check at delta_min
    vals_min = conv + delta_min * g
    max_val_min = np.max(vals_min)
    active_at_min = np.where(vals_min == max_val_min)[0]
    min_g_at_min = np.min(g[active_at_min])
    
    if min_g_at_min >= 0:
        # All active constraints have non-negative slope, so
        # the function is non-decreasing at delta_min
        return delta_min
    
    # Otherwise, the minimum is at an interior point
    # Binary search for where the subgradient crosses 0
    # As delta -> +inf, the active constraint is the one with max g[k]
    # As delta -> -inf, the active constraint is the one with min g[k]
    
    delta_lo = delta_min
    delta_hi = delta_min + 10 * np.max(np.abs(conv)) / max(1e-10, np.max(np.abs(g)) - np.min(np.abs(g)))
    
    for _ in range(100):
        delta_mid = (delta_lo + delta_hi) / 2
        vals = conv + delta_mid * g
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-10)[0]
        subgrad_min = np.min(g[active])
        subgrad_max = np.max(g[active])
        
        if subgrad_min <= 0 <= subgrad_max:
            return delta_mid
        elif subgrad_min > 0:
            # Function is increasing, move left
            delta_hi = delta_mid
        else:
            # Function is decreasing, move right
            delta_lo = delta_mid
    
    return (delta_lo + delta_hi) / 2
```
