```python
def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-30:
        return np.ones(len(z)) / len(z)
    return exp_z / s

        conv = fast_convolve(a)
```

2. Compute `conv = a * a`

4. Compute `grad_a = 2 * (weighted convolution of w and a)`

7. `z = z - lr * m_hat / (sqrt(v_hat) + eps)`
8. Anneal temperature: `T *= decay` or `T = T0 * (decay^t)`

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with log-sum-exp
smoothing, softmax parameterization, multi-resolution search, and coordinate
refinement. Key differences from prior approach: Adam's per-coordinate adaptive
learning rates, fully differentiable LSE objective enabling stable small steps,
softmax reparameterization avoiding projection issues, and a final coordinate-wise
refinement pass.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    """Convert unconstrained z to probability simplex via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-30:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def softmax_grad(grad_a, a):
    """Gradient through softmax: grad_z = a * (grad_a - <a, grad_a>)/(sum a)."""
    # a is already normalized (sum=1)
    return a * (grad_a - np.dot(a, grad_a))


def compute_lse_grad_and_val(a, temperature):
    """
    Compute gradient and value of T*log(sum(exp(conv/T))) w.r.t. a.
    Returns (grad_a, lse_value, conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-30:
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] = 2.0 * a[k_max - idx]
        lse_val = conv.max()
        return grad, lse_val, conv
    weights = exp_scaled / total
    lse_val = conv.max() + temperature * np.log(total)

    # Compute grad via FFT: grad[j] = 2 * sum_m a[m] * w[j+m]
    # = 2 * (a * w_rev)[2n-2-j] where w_rev = w[::-1]
    w_rev = weights[::-1]
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    Wr = np.fft.rfft(w_rev, N)
    conv_aw = np.fft.irfft(A * Wr, N)
    grad = 2.0 * conv_aw[2 * (n - 1) - np.arange(n)]

    return grad, lse_val, conv


def adam_optimize(a0, time_limit, lr=0.01, b1=0.9, b2=0.999, eps=1e-8,
                  temp_init=5.0, temp_decay=0.9995, temp_min=0.01,
                  perturb_every=500, perturb_scale=0.05, seed=0):
    """
    Adam optimizer with softmax parameterization and LSE smoothing.
    Operates on a0 normalized to sum=1.
    Returns (best_a, best_obj) where best_obj = 2*n*max(conv).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    # Initialize z so that softmax(z) = a
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()  # remove mean for numerical stability

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    temp = temp_init
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Compute LSE gradient
        grad_a, lse_val, conv = compute_lse_grad_and_val(a, temp)
        grad_z = softmax_grad(grad_a, a)

        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * grad_z * grad_z
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Get new a
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal temperature
        temp = max(temp_min, temp * temp_decay)

        # Perturb if stuck
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            z = z + noise
            m *= 0.5
            v *= 0.5
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_param=50.0, eta_init=0.3, seed=0):
    """
    Multiplicative updates (mirror descent) for refinement.
    Returns (best_a, best_obj).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    t_cur = t_param
    no_improve = 0
    max_no_improve = 80

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_cur * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        # Compute gradient
        grad = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for k_int in active:
            k = int(k_int)
            wi = w[k_int] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += wi * a[k - idx]

        # Centered gradient
        grad_centered = grad - np.dot(a, grad)

        # Multiplicative update
        log_update = -eta * grad_centered
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            no_improve = 0
            eta *= 0.7
            t_cur *= 1.4
            if eta < 0.05:
                # Restart with perturbation
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(best_a + noise)
                conv = fast_convolve(a)
                eta = eta_init * 0.5
                t_cur = t_param
                max_no_improve = 60

    return best_a, best_obj


def coordinate_refine(a, time_limit, step=0.005, seed=0):
    """
    Coordinate-wise local search. Try increasing/decreasing each coordinate
    and keep improvements.
    Returns (best_a, best_obj).
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Use random coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            original = a[i]
            # Try increasing
            a[i] = min(original + step, 1.0)
            s = a.sum()
            a = a / s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()
                improved = True
            else:
                a[i] = original
                s = a.sum()
                a = a / s

            # Try decreasing
            a[i] = max(original - step, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = original
                    s = a.sum()
                    a = a / s
            else:
                a[i] = original
                s = a.sum()
                a = a / s

        sweep += 1
        if not improved:
            break
        # Shrink step
        step *= 0.85

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n, normalized to sum=1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated
    edge_w = np.exp(-2.0 * np.minimum(x, n - 1 - x) / (0.2 * n))
    inits['edge'] = edge_w / edge_w.sum()

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Sine
    sinit = np.abs(np.sin(np.pi * x / n)) + 0.1
    inits['sine'] = sinit / sinit.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        # Resample
        pp = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent_res'] = pp / ps

    return inits
```
