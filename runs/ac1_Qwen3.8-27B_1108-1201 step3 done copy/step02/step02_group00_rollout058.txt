```python
def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Project gradient onto tangent space of simplex at point v."""
    # Tangent space: {d : sum(d) = 0}
    # Projection: d - (mean(d)) * 1
    return grad - np.mean(grad)
```

The cross-correlation theorem states:

then `FFT(R_{xy}) = FFT(x) * conj(FFT(y))`

So `R_{xy} = IFFT(FFT(x) * conj(FFT(y)))`

In our case: x = w, y = a, and we want R_{wa}[j] = sum_k w[k] * a[k-j].

So: `sg[j] = 2 * IFFT(FFT(w) * conj(FFT(a)))[j]`

With zero-padding to size N ≥ (2n-1) + n - 1 = 3n - 2.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Riemannian gradient descent
with FFT-based smooth max gradient, adaptive annealing, and multi-phase search.
Key innovations: O(n log n) gradient via FFT cross-correlation, anti-symmetric
initializations to reduce center-peak, and aggressive temperature annealing.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_grad_fft(a, conv, T):
    """
    Compute gradient of smooth max of conv using FFT cross-correlation.
    Gradient of m_T(conv) = (1/T) log(sum exp(T*c_k)) w.r.t. a is:
      sum_k softmax_k(T) * (2 * a[k-j]) for each j
    = 2 * cross_correlate(softmax_weights, a)
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Softmax weights (shifted for numerical stability)
    max_c = conv.max()
    scaled = T * (conv - max_c)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(conv_len)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Cross-correlation via FFT: sg[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * IFFT(FFT(w) * conj(FFT(a)))
    fft_size = 1 << (conv_len + n - 1 + 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    sg = 2.0 * np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    return sg


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute objective 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def optimize_riemannian(a0, time_limit, T_init=50.0, T_final=5000.0,
                        lr_init=0.05, momentum_init=0.7, momentum_final=0.9,
                        batch=30, perturb_interval=300, perturb_scale=0.01):
    """
    Riemannian gradient descent with smooth max, momentum, and adaptive schedule.
    Works on the simplex {a >= 0, sum(a) = 1}.
    Returns (best_a, best_obj_value).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    obj_val, conv = compute_obj(a)
    best_a = a.copy()
    best_obj = obj_val
    
    t0 = time.monotonic()
    v_mom = np.zeros(n)
    T = T_init
    lr = lr_init
    momentum = momentum_init
    total_iters = 0
    no_improve = 0
    perturb_count = 0
    
    # Estimate how many iterations we can do
    # Each iter: ~3 FFTs of size O(n) → ~O(n log n) ops
    # For n=256: ~20k ops per iter → ~50k iters in 1800s (conservative)
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute smooth gradient
            sg = smooth_max_grad_fft(a, conv, T)
            
            # Riemannian gradient (project to tangent space: sum(d) = 0)
            rg = sg - np.mean(sg)
            
            # Momentum update
            v_mom = momentum * v_mom - lr * rg
            
            # Update and project to simplex
            a = project_simplex(a + v_mom)
            
            # Recompute
            obj_val, conv = compute_obj(a)
            total_iters += 1
            
            if obj_val < best_obj - 1e-14:
                best_obj = obj_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Progress schedule based on fraction of time elapsed
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** frac
        momentum = momentum_init + (momentum_final - momentum_init) * frac
        
        # Perturb if stuck
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Riemannian gradient descent
with FFT-based smooth max gradient (O(n log n) per iteration), adaptive
temperature annealing, momentum scheduling, and multi-phase search strategy.
Key innovations: FFT cross-correlation for gradient, anti-symmetric
initializations, and aggressive exploration-exploitation balance.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_grad_fft(a, conv, T):
    """
    Compute gradient of smooth max m_T(conv) = (1/T)log(sum exp(T*c_k)).
    Uses FFT cross-correlation for O(n log n) complexity.
    sg[j] = 2 * sum_k softmax_k(T) * a[k-j]
    """
    n = len(a)
    conv_len = 2 * n - 1

    # Softmax weights (shifted for numerical stability)
    max_c = conv.max()
    scaled = T * (conv - max_c)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(conv_len)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    # Cross-correlation via FFT: sg[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * IDFT(FFT(w) * conj(FFT(a)))
    fft_size = 1 << (conv_len + n - 1 + 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    sg = 2.0 * np.fft.irfft(W * np.conj(A), fft_size)[:n]

    return sg


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute objective 2*n*max(conv(a,a)) assuming sum(a)=1. Returns (obj, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def optimize_riemannian(a0, time_limit, T_init=30.0, T_final=5000.0,
                        lr_init=0.04, momentum_init=0.65, momentum_final=0.9,
                        batch=25, perturb_interval=300, perturb_scale=0.008,
                        lr_decay=0.999, seed=0):
    """
    Riemannian gradient descent with smooth max, momentum, and adaptive schedule.
    Works on the simplex {a >= 0, sum(a) = 1}.
    Returns (best_a, best_obj_value).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    obj_val, conv = compute_obj(a)
    best_a = a.copy()
    best_obj = obj_val

    t0 = time.monotonic()
    v_mom = np.zeros(n)
    T = T_init
    lr = lr_init
    momentum = momentum_init
    no_improve = 0
    perturb_count = 0
    current_interval = perturb_interval

    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute smooth gradient via FFT
            sg = smooth_max_grad_fft(a, conv, T)

            # Riemannian gradient (tangent space: sum(d) = 0)
            rg = sg - np.mean(sg)

            # Momentum update
            v_mom = momentum * v_mom - lr * rg

            # Update and project to simplex
            a = project_simplex(a + v_mom)

            # Recompute objective
            obj_val, conv = compute_obj(a)

            if obj_val < best_obj - 1e-14:
                best_obj = obj_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        frac = min(1.0, (time.monotonic() - t0) / time_limit)

        momentum = momentum_init + (momentum_final - momentum_init) * frac

            a = project_simplex(a + noise)

            no_improve = 0
            perturb_count += 1
            current_interval = int(current_interval * 1.4)

        # Learning rate decay
        lr *= lr_decay

    return best_a, best_obj

    phase1_budget = total_budget * 0.35
    phase1_deadline = t1_start + phase1_budget

    lengths = [256, 512, 1024]
    init_names = ['parent_interp', 'anti_symmetric', 'gauss_left', 'random',
                  'bimodal_edge', 'cosine', 'uniform', 'two_bumps', 'sawtooth']

    results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 25:
            break

        # Time budget per (length, init) pair
        remaining_inits = len(init_names)
        t_remaining = phase1_deadline - time.monotonic() - 10
        t_per = max(12.0, t_remaining / (remaining_inits + 0.5))

        for init_name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break

            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 8)
            if t_alloc < 8:
                continue

            a0 = make_initialization(n, init_name, rng, parent)
            a_opt, obj_opt = optimize_riemannian(
                a0, time_limit=t_alloc,
                T_init=30.0, T_final=3000.0,
                lr_init=0.04, momentum_init=0.6, momentum_final=0.85,
                batch=25, perturb_interval=200, perturb_scale=0.008,
                seed=seed + n + hash(init_name) % 10000
            )

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, init_name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {init_name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # === Phase 2: Deep optimization (45% of time) ===
    print("=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()
    phase2_budget = (deadline - t2_start) * 0.48
    phase2_deadline = t2_start + phase2_budget

    # Build list of configs to optimize deeply
    top_configs = []

    # Top results from phase 1
    for score, n, name, a_opt in results[:5]:
        top_configs.append((n, name, a_opt))

    # Current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    # Fresh starts at best length with different seeds
    if best_sequence is not None:
        best_n = len(best_sequence)
        for extra_name in ['random', 'anti_symmetric_2', 'gauss_right']:
            top_configs.append((best_n, f'fresh_{extra_name}',
                               make_initialization(best_n, extra_name, rng, parent)))

    # Deduplicate
    seen = set()
    unique_configs = []
    for n, name, a in top_configs:
        key = (n, name)
        if key not in seen:
            unique_configs.append((n, name, a))
            seen.add(key)
    top_configs = unique_configs[:7]

    if top_configs:
        t_per_deep = max(20.0, phase2_budget / len(top_configs))
    else:
        t_per_deep = phase2_budget

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_deadline - 15:
            break

        t_alloc = min(t_per_deep, phase2_deadline - time.monotonic() - 10)
        if t_alloc < 10:
            break

        # Vary parameters slightly for diversity
        lr_var = 0.03 + 0.01 * (i % 3)
        mom_var = 0.7 + 0.05 * (i % 2)

        a_opt, obj_opt = optimize_riemannian(
            a0, time_limit=t_alloc,
            T_init=20.0, T_final=8000.0,
            lr_init=lr_var, momentum_init=mom_var, momentum_final=0.92,
            batch=30, perturb_interval=400, perturb_scale=0.005,
            lr_decay=0.9992,
            seed=seed + 100 + i
        )

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}, {name}: {score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # === Phase 3: Final polish (remaining time) ===
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        phase3_deadline = deadline - 10
        n_best = len(best_sequence)

        polish_configs = [
            {'T_init': 10.0, 'T_final': 15000.0, 'lr_init': 0.01,
             'momentum_init': 0.8, 'momentum_final': 0.96,
             'batch': 50, 'perturb_interval': 500, 'perturb_scale': 0.002,
             'lr_decay': 0.9995},
            {'T_init': 5.0, 'T_final': 30000.0, 'lr_init': 0.005,
             'momentum_init': 0.85, 'momentum_final': 0.98,
             'batch': 80, 'perturb_interval': 1000, 'perturb_scale': 0.001,
             'lr_decay': 0.9998},
            {'T_init': 50.0, 'T_final': 5000.0, 'lr_init': 0.015,
             'momentum_init': 0.7, 'momentum_final': 0.9,
             'batch': 40, 'perturb_interval': 300, 'perturb_scale': 0.005,
             'lr_decay': 0.999},
        ]

        t_per_polish = (phase3_deadline - time.monotonic()) / (len(polish_configs) + 2)

        for pi, cfg in enumerate(polish_configs):
            if time.monotonic() > phase3_deadline - 10:
                break
            t_alloc = min(t_per_polish, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break

            a_opt, obj_opt = optimize_riemannian(
                best_sequence, time_limit=t_alloc, seed=seed + 200 + pi, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {pi}: {score:.6f}")

        # Random perturb + re-optimize
        if time.monotonic() < phase3_deadline - 15:
            remaining = phase3_deadline - time.monotonic() - 10
            for _ in range(3):
                if time.monotonic() > phase3_deadline - 10:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_sequence + noise)
                t_alloc = remaining / 3
                if t_alloc < 5:
                    break
                a_opt, obj_opt = optimize_riemannian(
                    a_pert, time_limit=t_alloc,
                    T_init=10.0, T_final=15000.0,
                    lr_init=0.01, momentum_init=0.8, momentum_final=0.96,
                    batch=50, perturb_interval=500, perturb_scale=0.002,
                    lr_decay=0.9995, seed=seed + 300)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb+opt: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
