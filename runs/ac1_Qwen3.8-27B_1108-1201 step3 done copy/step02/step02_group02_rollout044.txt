import time

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Fast score computation."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

import numpy as np

def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a non-negative sequence (fast)."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

    Compute the (smoothed) subgradient of max_k conv_k(p).

    p: array of length n (probability vector)
    w: array of length 2n-1 (weights on constraints)
    Returns: array of length n

    p_rev = p[::-1]
    cw = fft_convolve(p_rev, w)
    return 2.0 * cw[n - 1:n - 1 + n]

            if len(other) > 10:
                n_inc = min(15, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.70:
            # Multi-block
            for _ in range(6):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.2, 0.2) * max(current[i], current[j]) * 0.1
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.005, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif np.isfinite(new_score) and rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling
        iteration += 1

        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    polish iter={iteration}, score={current_score:.6f}, "
                  f"best={best_score:.6f}, T={temperature:.6f}, elapsed={elapsed:.1f}s")

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.05, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_multi_tone_init(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.2 / k
        freq = rng.uniform(0.3, 2.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length via interpolation."""
    seq = np.asarray(seq, dtype=np.float64)
    x_old = np.linspace(0, 1, len(seq), endpoint=True)
    x_new = np.linspace(0, 1, new_n, endpoint=True)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses mirror descent (multiplicative weight) on the
    probability simplex to minimize peak autocorrelation, combined with local
    search polish. Tries multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
        try:
            hs1 = globals().get('height_sequence_1', None)
            if hs1 is not None:
                start_seq = np.clip(np.asarray(hs1, dtype=np.float64), 0.0, 1000.0)
                if len(start_seq) > 0:
                    s = local_score(start_seq)
                    print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
                    if s < best_score:
                        best_score = s
                        best_seq = start_seq.copy()
        except Exception:
            pass

    # Phase 1: Mirror descent on the provided sequence (if available)
    print("\n=== Phase 1: Mirror descent (primary, n=512) ===")
    if start_seq is not None and time.monotonic() < deadline - 600:
        # Normalize to probability vector
        p0 = start_seq.copy()
        p0 = np.clip(p0, 1e-10, None)
        p0 = p0 / p0.sum()
        t1 = min(400.0, deadline - time.monotonic() - 500)
        p_best, s_best = mirror_descent_optimize(p0, t1, rng, verbose=True)
        a_best = np.sqrt(2.0 * len(p_best)) * p_best
        a_score = local_score(a_best)
        print(f"  Phase 1 result: score={a_score:.6f}")
        if a_score < best_score:
            best_score = a_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Mirror descent on other sequence lengths and initializations
    print(f"\n=== Phase 2: Mirror descent (exploration) ===")
    configs = [
        (512, 'chirp', 0.5),
        (512, 'multi_tone', 0),
        (512, 'random', 0.4),
        (768, 'chirp', 0.4),
        (768, 'random', 0.3),
        (1024, 'chirp', 0.3),
        (1024, 'random', 0.3),
        (768, 'multi_tone', 0),
    ]

    n_configs = len(configs)
    for ci, (n, init_type, alpha) in enumerate(configs):
        if time.monotonic() > deadline - 200:
            break
        time_left = deadline - time.monotonic() - 150
        remaining = n_configs - ci
        t_per = max(20.0, min(80.0, time_left / max(1, remaining)))

        if init_type == 'chirp':
            a0 = make_chirp_init(n, alpha + 0.2 * rng.random(), rng)
        elif init_type == 'multi_tone':
            a0 = make_multi_tone_init(n, rng)
        else:
            a0 = make_random_init(n, rng)

        # Occasionally resample best to this length
        if best_seq is not None and rng.random() < 0.3:
            a0 = resample_sequence(best_seq, n)
            print(f"  [resampled best to n={n}]")

        p0 = a0 / a0.sum()
        p_best, s_best = mirror_descent_optimize(p0, t_per, rng, verbose=False)
        a_best = np.sqrt(2.0 * n) * p_best
        a_score = local_score(a_best)
        print(f"  n={n}, init={init_type}: score={a_score:.6f}")
        if a_score < best_score:
            best_score = a_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Local search polish
    print(f"\n=== Phase 3: Local search polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 80))
        print(f"  Starting polish, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, polish_score = local_search_polish(best_seq, polish_time, rng, verbose=True)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  Polish didn't improve (best={best_score:.6f})")

    # Phase 4: Final mirror descent on best sequence
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 4: Final mirror descent polish ===")
        p0 = best_seq.copy()
        p0 = np.clip(p0, 1e-10, None)
        p0 = p0 / p0.sum()
        final_md_time = min(200.0, max(15.0, deadline - time.monotonic() - 50))
        p_best, s_best = mirror_descent_optimize(p0, final_md_time, rng, verbose=False)
        a_best = np.sqrt(2.0 * len(p_best)) * p_best
        a_score = local_score(a_best)
        print(f"  After final MD: {a_score:.6f}")
        if a_score < best_score:
            best_score = a_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Final local search if time permits
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            print(f"\n=== Phase 5: Final local search ({remaining:.0f}s) ===")
            polished, polish_score = local_search_polish(best_seq, remaining, rng, verbose=False)
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polished.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
