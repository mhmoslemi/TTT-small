```python
def soft_subgrad_optimize(a_init, time_limit, rng):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    tau = 5.0 / n  # initial temperature for soft-max
    tau_min = 0.1 / n
    eta = 0.1  # step size
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        conv_max = np.max(conv)
        
        # Soft-max weights
        shifted = (conv - conv_max) / tau
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient: g[j] = 2 * sum_m a[m] * w[j+m]
        g = np.zeros(n)
        for j in range(n):
            g[j] = 2.0 * np.dot(a, w[j:j+n])
        
        # Center to preserve sum
        g -= np.mean(g)
        
        # Adaptive step size
        g_scale = np.max(np.abs(g))
        if g_scale < 1e-10:
            break
        step = eta * g / g_scale
        
        # Multiplicative update
        a = a * np.exp(-step)
        a = np.clip(a, 1e-10, None)
        a = a * (S / np.sum(a))
        
        score = local_score(a)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 100:
                # Perturb
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.clip(a, 1e-10, None)
                a = a * (S / np.sum(a))
                stagnant = 0
        
        # Anneal
        tau *= 0.9995
        if tau < tau_min:
            tau = tau_min
        
        # Occasionally increase step size if improving
        if stagnant == 0 and eta < 0.5:
            eta *= 1.01
        elif stagnant > 50:
            eta *= 0.9
```

```python
"""
Soft-subgradient optimization with multiplicative updates, peak-aware raking,
and simulated annealing for minimizing normalized peak autocorrelation of
non-negative step functions. Uses annealed soft-max gradients computed via
Hankel matrix structure, with adaptive step sizes and multi-length exploration.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def compute_gradient_hankel(a, w):
    """Compute g[j] = 2 * sum_m a[m] * w[j+m] using Hankel matvec."""
    n = len(a)
    # w has length 2n-1; we need w[j:j+n] for j=0..n-1
    # This forms a Hankel matrix H where H[j,m] = w[j+m]
    # Use stride tricks to avoid copying
    H = np.lib.stride_tricks.as_strided(
        w, shape=(n, n), strides=(w.strides[0], w.strides[0])
    )
    return 2.0 * (H @ a)


def soft_subgrad_optimize(a_init, time_limit, rng, verbose=False):
    """Main optimization: soft-subgradient with multiplicative updates."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)

    t0 = time.monotonic()
    tau = 8.0 / n
    tau_min = 0.05 / n
    eta = 0.15
    stagnant = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        conv_max = np.max(conv)

        # Soft-max weights (with numerical stability)
        shifted = (conv - conv_max) / tau
        # Clip to prevent underflow issues
        shifted = np.clip(shifted, -50, 0)
        e = np.exp(shifted)
        e_sum = np.sum(e)
        if e_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = e / e_sum

        # Gradient via Hankel matvec
        g = compute_gradient_hankel(a, w)

        # Center to preserve sum
        g -= np.mean(g)

        # Adaptive step size
        g_scale = np.max(np.abs(g))
        if g_scale < 1e-12:
            break

        step = eta * g / g_scale
        # Clamp step to prevent extreme updates
        step = np.clip(step, -1.0, 1.0)

        # Multiplicative update
        a = a * np.exp(-step)
        a = np.clip(a, 1e-15, None)
        a = a * (S / np.sum(a))

        iteration += 1
        score = local_score(a)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if eta < 0.6:
                eta = min(0.6, eta * 1.02)
        else:
            stagnant += 1
            if stagnant > 80:
                # Add perturbation to escape local minimum
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.clip(a, 1e-15, None)
                a = a * (S / np.sum(a))
                stagnant = 0
                eta = max(0.05, eta * 0.7)
            elif stagnant > 30:
                eta = max(0.03, eta * 0.95)

        # Anneal tau
        tau = max(tau_min, tau * 0.9992)

        if verbose and iteration % 200 == 0:
            print(f"    iter={iteration}, tau={tau:.5f}, eta={eta:.3f}, "
                  f"score={score:.6f}, best={best_score:.6f}, "
                  f"elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def raking_optimize(a_init, time_limit, rng, verbose=False):
    """Peak-aware raking: transfer mass from peak-contributing to non-contributing elements."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)

    t0 = time.monotonic()
    alpha = 0.02  # transfer fraction
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))

        # Compute contributions to the peak: c[i] = a[i] * a[peak_idx - i]
        c = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                c[i] = a[i] * a[j]

        # Also compute total contribution to ALL conv values (to avoid creating new peaks)
        # Approximate: use the soft-max weighted gradient
        conv_max = np.max(conv)
        tau_r = 3.0 / n
        shifted = (conv - conv_max) / tau_r
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum > 1e-300:
            w /= w_sum
            g_total = compute_gradient_hankel(a, w)
        else:
            g_total = c.copy()

        # Top donors (high peak contribution) and bottom receivers (low total contribution)
        top_k = max(2, n // 16)
        donor_idx = np.argsort(c)[-top_k:]
        # Exclude donors from receiver pool
        not_donor = np.setdiff1d(np.arange(n), donor_idx)
        if len(not_donor) < top_k:
            continue
        receiver_idx = not_donor[np.argsort(g_total[not_donor])[:top_k]]

        # Transfer mass
        donor_amount = np.mean(a[donor_idx]) * alpha
        for i in donor_idx:
            a[i] = max(1e-10, a[i] - donor_amount)
        for i in receiver_idx:
            a[i] = a[i] + donor_amount * len(donor_idx) / len(receiver_idx)

        a = np.clip(a, 1e-10, None)
        a = a * (S / np.sum(a))
        iterations += 1

        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Occasionally perturb
        if iterations % 50 == 0:
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.clip(a, 1e-10, None)
            a = a * (S / np.sum(a))

        if verbose and iterations % 100 == 0:
            print(f"    rake iter={iterations}, score={score:.6f}, "
                  f"best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def sa_polish(a_init, time_limit, rng, verbose=False):
    """Simulated annealing with peak-aware mutations."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score

    t0 = time.monotonic()
    temperature = 0.004
    cooling = 0.99985
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))

        if r < 0.30:
            # Peak-targeted: reduce top contributors to peak, boost others
            c = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    c[i] = a[i] * a[j]
            top_k = max(2, n // 20)
            top_idx = np.argsort(c)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            n_inc = min(4, len(other))
            inc_idx = rng.choice(other, size=n_inc, replace=False)
            candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.50:
            # Local block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.70:
            # Dual-block perturbation
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.025, size=width))
        elif r < 0.85:
            # Global small noise
            candidate *= np.exp(rng.normal(0, 0.008, n))
        else:
            # Mirror/flip perturbation on a segment
            width = max(2, n // 8)
            start = int(rng.integers(0, n - width))
            seg = candidate[start:start + width].copy()
            candidate[start:start + width] = seg[::-1] * np.exp(rng.normal(0, 0.01, width))

        candidate = np.clip(candidate, 1e-10, None)
        candidate = candidate * (S / np.sum(candidate))

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature *= cooling

        if verbose and iter_count % 500 == 0:
            print(f"    SA iter={iter_count}, T={temperature:.6f}, "
                  f"score={current_score:.6f}, best={best_score:.6f}, "
                  f"elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_score

        a = a * np.exp(-step)
        a = np.clip(a, 1e-15, None)
        a = a * (S / np.sum(a))

        score = local_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                break  # We've converged enough

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Soft-subgradient optimization with multiplicative updates (main engine)
    2. Peak-aware raking
    3. Focused gradient polish
    4. Simulated annealing for final refinement
    Explores multiple lengths and initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # Define search configurations: (n, init_type, time_fraction)
    n_prov = len(provided_seq) if provided_seq is not None else 256

    search_configs = [
        (n_prov, 'provided', 0.22),
        (n_prov, 'chirp', 0.12),
        (n_prov, 'multitone', 0.08),
        (512, 'chirp', 0.12),
        (512, 'raised_cosine', 0.08),
        (1024, 'chirp', 0.12),
        (1024, 'raised_cosine', 0.08),
        (2048, 'chirp', 0.10),
        (2048, 'raised_cosine', 0.08),
    ]

    total_frac = sum(c[2] for c in search_configs)
    print(f"\nTotal time budget: {budget_s - 30:.0f}s, {len(search_configs)} configs")

    # Phase 1: Soft-subgradient optimization on multiple configs
    print("\n" + "=" * 60)
    print("Phase 1: Soft-subgradient optimization")
    print("=" * 60)

    for n, init_name, frac in search_configs:
        if time.monotonic() > deadline - 60:
            break

        time_alloc = max(20.0, frac / total_frac * (budget_s - 60) * 0.75)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 30)
        if time_alloc < 15:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            if n == len(provided_seq):
                a0 = provided_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(provided_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, provided_seq)
        elif init_name == 'chirp':
            a0 = make_chirp_init(n, rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng)
        elif init_name == 'raised_cosine':
            a0 = make_window_init(n, rng, 'raised_cosine')
        elif init_name == 'hann':
            a0 = make_window_init(n, rng, 'hann')
        else:
            a0 = make_random_init(n, rng)

        init_score = local_score(a0)
        print(f"\n  n={n}, init='{init_name}', init_score={init_score:.6f}, alloc={time_alloc:.0f}s")
        t_start = time.monotonic()

        f_best, score = soft_subgrad_optimize(a0, time_alloc, rng, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  Subgrad done: score={score:.6f}, time={elapsed:.1f}s, iters~{elapsed*200:.0f}")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")

    # Phase 2: Raking on the best sequence
    print("\n" + "=" * 60)
    print("Phase 2: Peak-aware raking")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        rake_time = min(150.0, remaining * 0.25)
        if rake_time > 20:
            print(f"  Raking n={len(best_seq)}, time={rake_time:.0f}s")
            a_rake, score_rake = raking_optimize(best_seq, rake_time, rng, verbose=False)
            print(f"  Raking result: {score_rake:.6f}")
            if score_rake < best_score:
                best_score = score_rake
                best_seq = a_rake.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 3: Focused gradient polish (very small tau)
    print("\n" + "=" * 60)
    print("Phase 3: Focused gradient polish")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(200.0, remaining * 0.30)
        if grad_time > 20:
            print(f"  Gradient polish n={len(best_seq)}, time={grad_time:.0f}s")
            a_grad, score_grad = gradient_polish_focused(best_seq, grad_time, rng)
            print(f"  Gradient polish result: {score_grad:.6f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: One more round of subgradient on best
    print("\n" + "=" * 60)
    print("Phase 4: Final subgradient pass")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic()
        final_time = min(150.0, remaining * 0.35)
        if final_time > 20:
            print(f"  Final subgrad n={len(best_seq)}, time={final_time:.0f}s")
            # Slightly perturb to avoid pure exploitation
            a_pert = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a_final, score_final = soft_subgrad_optimize(a_pert, final_time, rng)
            print(f"  Final subgrad result: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 5: SA polish
    print("\n" + "=" * 60)
    print("Phase 5: Simulated annealing polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        sa_time = min(120.0, remaining * 0.8)
        if sa_time > 15:
            print(f"  SA polish n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng, verbose=False)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final focused gradient on the very best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        polish_time = min(60.0, remaining * 0.7)
        if polish_time > 10:
            a_poly, score_poly = gradient_polish_focused(best_seq, polish_time, rng)
            if score_poly < best_score:
                best_score = score_poly
                best_seq = a_poly.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")

        # Update best if evaluation differs (shouldn't, but be safe)
        if final_score < best_score or best_seq is None:
            best_score = final_score
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
```
