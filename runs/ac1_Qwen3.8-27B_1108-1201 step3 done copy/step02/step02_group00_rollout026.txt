Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth L-BFGS-B optimization
with log-sum-exp approximation of max, multi-start over different lengths,

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    sm = x_max + np.log(exp_shifted.sum()) / beta
    return sm, w


def grad_smooth_max_conv(a, beta):
    """Gradient of smooth_max(conv(a,a), beta) with respect to a.
    Returns gradient of the same shape as a."""
    n = len(a)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    
    # grad = 2 * ifft(conj(fft(w, N)) * fft(a, N))[:n]
    N = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n]
    grad *= 2.0
    return grad, sm


def objective_and_grad(x, n, beta):
    """Objective and gradient in log-space (softmax parametrization).
    x is the log-space variable, a = softmax(x)."""
    # Compute a from x
    a = x - x.max()
    a = np.exp(a)
    a = a / a.sum()

    grad_a, sm = grad_smooth_max_conv(a, beta)

def optimize_lbfgsb(a0, beta, maxiter=500, max_time=None):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)

    a0 = a0 / a0.sum()
    x0 = np.log(a0)
    
    best_obj = float('inf')

        obj, _ = objective_and_grad(x, n, beta)
        return obj
    
    def jac(x):
        _, g = objective_and_grad(x, n, beta)
        return g
    
    result = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxfun': maxiter * 3})
    
    if result.success or result.fun < best_obj:
        best_x = result.x
        best_obj = result.fun
    
    a = np.exp(best_x - best_x.max())
    a = a / a.sum()
    return a, best_obj


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def projected_gradient_smooth(a0, beta, lr=0.01, n_iter=1000):
    """Projected gradient descent on simplex with smooth max objective."""
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a)
    
    best_obj = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):

        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

            best_obj = obj
            best_a = a.copy()

        N = 1 << (len(w) + n - 1 - 1).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
        
        # Gradient step + projection
        a = project_simplex(a - lr * grad)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth L-BFGS-B optimization
with log-sum-exp approximation of max, frequency-domain inspired initializations,
multi-start over different lengths, and adaptive temperature scheduling.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a using FFT cross-correlation."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj(x, n, beta):
    """Top-level objective for scipy minimize (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    """Top-level Jacobian for scipy minimize (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=500):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)

    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})

    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def optimize_pg_smooth(a0, beta, lr=0.05, n_iter=500):
    """Projected gradient descent on simplex with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if obj < 1e-10:
            break

        grad_a = compute_grad_smooth_conv(a, w)

        # Adaptive learning rate
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-15:
            break

        # Step with backtracking
        t = lr
        for _ in range(10):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-10:
                break
            t *= 0.5

        a = a_new
        # Decay learning rate
        lr *= 0.9995

    return best_a, best_obj

def peak_reduction_step(a, n_steps=10, reduction=0.97):
    """Iteratively reduce the peak of the convolution."""
    a = a.copy()
    n = len(a)
    for _ in range(n_steps):
        conv = fast_convolve(a)
        max_k = np.argmax(conv)
        i_start = max(0, max_k - n + 1)
        i_end = min(n - 1, max_k)
        # Reduce contributing elements proportionally
        contrib = np.abs(a[i_start:i_end + 1])
        total = contrib.sum()
        if total > 1e-15:
            reduction_factor = reduction * (contrib / total)
            a[i_start:i_end + 1] -= reduction_factor * (conv.max() - conv.mean()) / max(total, 1e-15)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            a = np.ones(n) / n
    return a


def coordinate_refine(a, n_iters=200):
    """Coordinate descent to reduce max convolution."""
    n = len(a)
    a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iters):
        improved = False
        for i in range(n):
            conv = fast_convolve(a)
            max_k = int(np.argmax(conv))
            # Try adjusting a[i]
            old_val = a[i]
            # Find how much a[i] contributes to the peak
            delta = a[max_k - i] if 0 <= max_k - i < n else 0.0
            if delta < 1e-15:
                delta = a[i - max_k] if 0 <= i - max_k < n else 0.0

            # Try reducing
            for scale in [0.9, 0.95, 0.8]:
                a[i] = old_val * scale
                conv_new = fast_convolve(a)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    improved = True
                    break
                # Try increasing (if it helps flatten other peaks)
            if not improved:
                a[i] = old_val
                # Try increasing slightly
                for scale in [1.1, 1.05]:
                    a[i] = old_val * scale
                    conv_new = fast_convolve(a)

                    if obj_new < best_obj:

                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        improved = True

                    a[i] = old_val

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s

    return a

                    break

                maxiter = int(min(400, max(50, t_single * 30)))
                a_opt, _ = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
                score = score_and_update(a_opt)
                a_cur = a_opt

    print("Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Fine-tuning
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20.0

        # L-BFGS-B with very low beta for precise optimization
        for beta in [3.0, 2.0, 1.5]:
            if remaining < 10:
                break
            t_single = remaining / 4
            if t_single < 5:
                break
            maxiter = int(min(500, max(100, t_single * 40)))
            a_opt, _ = optimize_lbfgsb(best_sequence / best_sequence.sum(), beta=beta, maxiter=maxiter)
            score = score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 20.0

        # Peak reduction + re-optimize
        if remaining > 20:
            a_reduced = peak_reduction_step(best_sequence / best_sequence.sum(), n_steps=50, reduction=0.98)
            a_opt, _ = optimize_lbfgsb(a_reduced, beta=10.0, maxiter=200)
            score = score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 20.0

        # Random perturbations + re-optimize
        if remaining > 30:
            for _ in range(5):
                if remaining < 10:
                    break
                scale = 0.003 + 0.002 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
                t_single = min(60.0, remaining / 3)
                if t_single < 5:
                    break
                maxiter = int(min(300, max(50, t_single * 20)))
                a_opt, _ = optimize_lbfgsb(a_pert, beta=15.0, maxiter=maxiter)
                score = score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 20.0

        # Coordinate refinement on the best
        if remaining > 15:
            a_coord = coordinate_refine(best_sequence / best_sequence.sum(), n_iters=50)
            score = score_and_update(a_coord)
            if score < best_score:
                # Re-optimize from here
                if remaining > 10:
                    a_opt, _ = optimize_lbfgsb(a_coord, beta=10.0, maxiter=200)
                    score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 20.0

        # Final projected gradient polish with low beta
        if remaining > 10:
            n_iters = int(min(500, remaining * 20))
            a_opt, _ = optimize_pg_smooth(best_sequence / best_sequence.sum(), beta=8.0, lr=0.02, n_iter=n_iters)
            score = score_and_update(a_opt)

    # Try a different length one more time if we have time
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 60:
            # Try n = 384 or 320 (nearby powers)
            alt_n = 384 if len(best_sequence) == 256 else 256
            if time.monotonic() + 50 < deadline:
                inits_alt = generate_initializations(alt_n, parent, rng)
                if best_sequence is not None:
                    pp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, len(best_sequence)), best_sequence)
                    pp = np.maximum(pp, 0)
                    ps = pp.sum()
                    if ps > 1e-10:
                        inits_alt['from_best'] = pp / ps
                a0_alt = inits_alt.get('from_best', inits_alt.get('flat_spec', inits_alt['uniform']))
                maxiter = int(min(400, remaining * 3))
                a_opt, _ = optimize_lbfgsb(a0_alt, beta=20.0, maxiter=maxiter)
                score_and_update(a_opt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
