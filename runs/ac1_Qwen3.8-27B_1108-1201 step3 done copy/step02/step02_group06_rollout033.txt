    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_conv_and_grad(a, T):
    """Compute convolution, soft-max weights, and gradient wrt a."""
    n = len(a)
    conv = fft_convolve(a, a)
    conv_shifted = (conv - np.max(conv)) / T
    exp_conv = np.exp(conv_shifted)
    weights = exp_conv / np.sum(exp_conv)
    
    # g_a[i] = 2 * sum_j a[j] * weights[i+j] for i=0..n-1
    g_a = np.zeros(n)
    for i in range(n):
        g_a[i] = 2.0 * np.dot(a, weights[i:i + n])
    return conv, weights, g_a


def soft_grad_optimize(a_init, time_limit, seed=42, lr_scale=1.0):
    """
    Adam gradient descent on softmax-parameterized sequence.
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    a_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(a_safe) - np.log(np.sum(a_safe))
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    
    T = 8.0 / n
    T_min = 0.15 / n
    T_decay = 0.9992
    
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    t0 = time.monotonic()
    stagnant = 0
    stagnant_threshold = 500
    
    while time.monotonic() - t0 < time_limit:
        theta_shifted = theta - np.max(theta)
        exp_theta = np.exp(theta_shifted)
        a = exp_theta / np.sum(exp_theta) * S
        
        conv, weights, g_a = compute_conv_and_grad(a, T)
        
        # Project gradient to maintain sum constraint
        G = np.sum(g_a)
        grad_theta = a * g_a - (a ** 2 / S) * G
        
        step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        lr = 4.0 * lr_scale / np.sqrt(1.0 + 0.01 * step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        T *= T_decay
        T = max(T, T_min)
        
        if step % 500 == 0:
            theta = theta - np.mean(theta)
        
        if step % 5 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
            
            if stagnant > stagnant_threshold:
                # Diversification: random perturbation
                theta += 0.1 * rng.standard_normal(n)
                theta = theta - np.mean(theta)
                T = max(T, 3.0 / n)
                stagnant = 0
                stagnant_threshold = int(stagnant_threshold * 1.3)
    
    return best_a, best_score


def build_sparse_A(f):
    """Build constraint matrix A for LP (sparse). A[k,j] = f[j] if 0<=k-j<n."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_step(f, M, time_limit_lp=5.0):
    """Single LP step: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_sparse_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': time_limit_lp})
    if result.success:
        return result.x
    return None


def lp_optimize(f_init, time_limit, t_init=0.3, verbose=False):
    """LP-based iterative optimization."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        lp_time = min(10.0, max(1.0, (time_limit - (time.monotonic() - t0)) * 0.3))
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = lp_step(f, M, time_limit_lp=lp_time)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)
        
        score_new = compute_score(f_new)
        iteration += 1
        
        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            t = min(0.8, t * 1.03)
        else:
            f = f_new
            stagnant += 1
            if stagnant > 15:
                t *= 0.8
                stagnant = 0
            if stagnant > 50:
                f = f + 0.03 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.2
                stagnant = 0
        
        if verbose and iteration % 20 == 0:
            print(f"    LP iter={iteration}, score={score_new:.6f}, best={best_score:.6f}, "
                  f"elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def sa_polish(f_init, time_limit, rng, n_alt=None):
    """Simulated annealing with peak-targeting mutations."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = compute_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.005
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.30:
            # Peak-targeting: reduce contributors to peak, boost others
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * (f[j] if j >= 0 else 0)
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.55:
            # Gaussian perturbation
            candidate *= np.exp(rng.normal(0.0, 0.012, size=n))
        elif r < 0.70:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.85:
            # Asymmetric shift
            shift = int(rng.integers(2, max(3, n // 10)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            # Local swap
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                delta = f[i1] * 0.1
                candidate[i1] -= delta
                candidate[i2] += delta

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = compute_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score

        temperature *= cooling

    return best_f, best_score


def water_filling_step(a, n_steps=10, alpha=0.1):
    """Water-filling: reduce peak contributors, boost low contributors."""
    a = a.copy()
    n = len(a)
    S = np.sum(a)
    
    for _ in range(n_steps):
        conv = fft_convolve(a, a)
        peak_val = np.max(conv)
        peak_idx = int(np.argmax(conv))
        
        # Find top contributors to peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Reduce top contributors
        top_k = max(3, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]
        reduction = alpha * a[top_idx] * 0.1
        a[top_idx] -= reduction
        
        # Boost bottom contributors (spread mass)
        bottom_idx = np.argsort(contribs)[:top_k]
        a[bottom_idx] += reduction.sum() / len(bottom_idx)
        
        a = np.clip(a, 1e-10, None)
        a = a * (S / np.sum(a))
    
    return a


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(8):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def interpolate_sequence(seq, n):
    """Interpolate a sequence to length n."""
    if len(seq) == n:
        return seq.copy()
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: gradient descent, LP refinement, water-filling, SA.
    Explores multiple initializations and sequence lengths.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = compute_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    n_prov = len(provided_seq) if provided_seq is not None else 256

    # Define search configurations: (n, init_function, time_fraction)
    search_configs = [
        (n_prov, 'provided', 0.25),
        (n_prov, 'chirp', 0.12),
        (n_prov, 'multitone', 0.10),
        (n_prov, 'random', 0.10),
        (512, 'chirp', 0.10),
        (512, 'random', 0.08),
        (384, 'chirp', 0.08),
        (768, 'random', 0.07),
        (n_prov, 'chirp2', 0.05),
        (n_prov, 'multitone2', 0.05),
    ]

    total_frac = sum(c[2] for c in search_configs)
    available_time = deadline - time.monotonic()

    # Phase 1: Gradient-based optimization on multiple configs
    print("\n" + "=" * 60)
    print("Phase 1: Soft-gradient optimization (multiple configs)")
    print("=" * 60)

    for n, init_name, frac in search_configs:
        if time.monotonic() > deadline - 60:
            break

        time_alloc = max(10.0, frac / total_frac * available_time * 0.75)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 20)
        if time_alloc < 8:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = interpolate_sequence(provided_seq, n)
        elif init_name == 'chirp':
            alpha = 0.3 + 0.4 * rng.random()
            a0 = make_chirp_init(n, alpha, rng)
        elif init_name == 'chirp2':
            alpha = 0.5 + 0.3 * rng.random()
            a0 = make_chirp_init(n, alpha, rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng)
        elif init_name == 'multitone2':
            a0 = make_multitone_init(n, rng)
        else:
            a0 = make_random_init(n, 0.2 + 0.3 * rng.random(), rng) if False else make_random_init(n, 0.2 + 0.3 * rng.random())
        
        # Fix random init to use rng
        if init_name not in ('provided',) and provided_seq is None and init_name == 'random':
            a0 = make_random_init(n, rng, 0.2 + 0.3 * rng.random())

        print(f"  n={n}, init='{init_name}', alloc={time_alloc:.0f}s")
        t_start = time.monotonic()

        f_best, score = soft_grad_optimize(a0, time_alloc, seed=int(rng.integers(0, 1000000)))
        elapsed = time.monotonic() - t_start
        print(f"  Grad done: score={score:.6f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")

    # Phase 2: LP refinement on best sequence
    print("\n" + "=" * 60)
    print("Phase 2: LP-based refinement")
    print("=" * 60)

    if best_seq is not None:

                print(f"  *** New best: {best_score:.6f} ***")
