    if S > 0:

    return a_new

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Combines LP-based direction finding, softmax subgradient descent, spectral methods,
and coordinate descent to find sequences with low peak-to-average autocorrelation ratio.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def conv_matrix(f):
    """Build matrix A where (A @ b)[k] = conv(f, b)[k]."""
    n = len(f)
    out_len = 2 * n - 1
    A = np.zeros((out_len, n), dtype=np.float64)
    for j in range(n):
        lo = max(0, j - (n - 1))  # not needed, k-j >= 0 means k >= j
        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
        # So for column j, A[j+j', j] = f[j'] for j' = 0, ..., n-1
        # i.e., A[j:j+n, j] = f
        A[j:j+n, j] = f
    return A


def solve_direct_lp(f, S, timeout=5.0):
    """Solve: min t s.t. conv(f,b)_k <= t, sum(b)=S, b>=0.
    Returns (b, t) or (None, None).
    """
    n = len(f)
    A = conv_matrix(f)
    # Variables: b_0,...,b_{n-1}, t
    # Objective: minimize t -> c = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A @ b - t <= 0 for all k
    # [A | -1] @ [b; t] <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = S
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            b = res.x[:n]
            t = res.x[n]
            return b, t
    except Exception:
        pass
    return None, None

def softmax_subgradient_fast(f, beta=10.0):
    """Faster softmax subgradient using convolution structure."""
    n = len(f)
    c = fft_convolve(f, f)
    c_max = np.max(c)
    c_shifted = c - c_max
    weights = np.exp(beta * c_shifted)
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        # Contribution to g[m] from coefficient k:
        # g[m] += w * (f[k-m] if 0<=k-m<n else 0 + f[m-k] if 0<=m-k<n else 0)
        # This is: g += w * (f reversed and shifted + f shifted)
        # Specifically: for m in [0, n-1]:
        #   f[k-m] is non-zero when m in [max(0,k-n+1), min(n-1,k)]
        #   f[m-k] is non-zero when m in [max(0,k), min(n-1,k+n-1)]
        
        # First part: m in [max(0, k-n+1), min(n-1, k)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            # g[lo1:hi1+1] += w * f[k-hi1 : k-lo1+1] (reversed slice of f)
            # f[k-m] for m = lo1,...,hi1 gives f[k-lo1], f[k-lo1-1], ..., f[k-hi1]
            # That's f[k-hi1 : k-lo1+1] reversed
            seg = f[k - hi1 : k - lo1 + 1]
            g[lo1:hi1+1] += w * seg[::-1]
        
        # Second part: m in [max(0, k), min(n-1, k+n-1)]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            # f[m-k] for m = lo2,...,hi2 gives f[lo2-k], f[lo2-k+1], ..., f[hi2-k]
            seg = f[lo2 - k : hi2 - k + 1]
            g[lo2:hi2+1] += w * seg
    
    return g, c_max, c


def spectral_init(n, rng, iterations=50):
    """Initialize using iterative spectral flattening (Gerchberg-Saxton style)."""
    # Start with a random non-negative sequence
    a = rng.random(n) + 0.1
    
    for it in range(iterations):
        # Compute FFT (zero-padded)
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: make |A|^2 as flat as possible
        # Compress dynamic range of |A|
        target_mag = np.sqrt(np.mean(mag**2))
        # Blend: new_mag = target_mag * (mag / target_mag)^alpha
        alpha = 0.5  # compression factor
        new_mag = target_mag * (mag / target_mag) ** alpha
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, size)[:n]
        
        # Project to non-negative
        a = np.clip(a_new, 0.01, None)
    
    return a

def softmax_descent(f_init, time_limit, beta_init=5.0, verbose=True):
    """Softmax subgradient descent with Nesterov momentum and adaptive step size."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()

    beta_m = 0.9
    lr = 0.005
    beta = beta_init
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        g, c_max, c = softmax_subgradient_fast(f, beta)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g /= g_norm
        
        # Center gradient (to maintain sum constraint)
        g -= np.mean(g)
        
        momentum = beta_m * momentum + g
        f = f - lr * momentum
        f = np.clip(f, 0.01, 1000.0)
        f = f * (S / float(np.sum(f)))
        
        it += 1
        if it % 100 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                stall = 0
                # Slowly increase beta (sharpen the softmax)
                beta = min(beta * 1.1, 100.0)
                # Decrease learning rate
                lr *= 0.98
            else:
                stall += 1
                if stall > 50:
                    # Restore best and try different beta
                    f = best_f.copy()
                    beta = max(beta * 0.5, 1.0)
                    lr = 0.005
                    momentum = np.zeros(n)
                    stall = 0
            
            if verbose:
                print(f"    [softmax {time.monotonic()-t0:6.1f}s] it={it} beta={beta:.1f} score={s:.6f} best={best_score:.6f}")
        
        # Periodically try to restart from best
        if it % 500 == 0 and stall > 20:
            f = best_f.copy()
            momentum = np.zeros(n)
            beta = beta_init
            lr = 0.005
            stall = 0
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=True):
    """Coordinate descent: try small perturbations of individual coordinates."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    S = float(np.sum(f))
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    step_size = 0.05
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        # Pick a random coordinate (or the one with highest impact)
        c = fft_convolve(f, f)
        peak_k = int(np.argmax(c))
        
        # Find coordinates that contribute most to the peak
        contribs = np.zeros(n)
        for m in range(n):
            if 0 <= peak_k - m < n:
                contribs[m] = f[peak_k - m]
            if 0 <= m - peak_k < n and m - peak_k != peak_k - m:
                contribs[m] += f[m - peak_k]
        
        # Try reducing a high-contribution coordinate
        top_contrib = np.argsort(contribs)[-max(3, n//20):]
        m = int(rng.choice(top_contrib))
        
        # Also try increasing a low-contribution coordinate
        low_contrib = np.argsort(contribs)[:max(3, n//20)]
        m2 = int(rng.choice(low_contrib))
        
        # Random perturbation
        f[m] *= np.exp(rng.normal(-step_size, step_size/2))
        f[m2] *= np.exp(rng.normal(step_size, step_size/2))
        f = np.clip(f, 0.01, 1000.0)
        f = f * (S / float(np.sum(f)))
        
        s = score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
            it += 1
        else:
            # Revert
            f = best_f.copy()
        
        if it % 50 == 0 and it > 0 and verbose:
            # Slowly decrease step size
            step_size *= 0.99
    
    return best_f, best_score

"""
Multi-phase optimizer for minimizing peak-to-sum autocorrelation ratio.
Uses LP-based direction finding, softmax subgradient, spectral init, and coordinate descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


# === Core computation functions ===

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build matrix A where (A @ b)[k] = sum_j f[k-j]*b[j] (convolution)."""
    n = len(f)
    out_len = 2 * n - 1
    A = np.zeros((out_len, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A


# === LP solver ===

def solve_direct_lp(f, S, timeout=5.0):
    """min t s.t. conv(f,b)_k <= t, sum(b)=S, b>=0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0, None)] * n + [(0, None)]  # t >= 0 (since all values are non-negative)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def spectral_init(n, rng, iterations=40):
    """Gerchberg-Saxton style spectral flattening initialization."""
    a = rng.random(n) + 0.1
    for _ in range(iterations):
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        target = np.sqrt(np.mean(mag**2))
        alpha = 0.5
        new_mag = target * np.power(mag / target, alpha)
        A_new = new_mag * np.exp(1j * phase)
        a = np.clip(irfft(A_new, size)[:n], 0.01, None)
    return a

    """Hann window with noise."""

    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    a = 0.3 + 0.7 * window + 0.2 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def softmax_descent(f_init, time_limit, verbose=True):
    """Softmax subgradient descent with momentum."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    
    momentum = np.zeros(n)
    beta_m = 0.90
    lr = 0.003
    beta = 5.0
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f, f)
        c_max = np.max(c)
        weights = np.exp(beta * (c - c_max))
        weights /= np.sum(weights)
        
        # Compute gradient
        g = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-10:
                continue
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                g[lo1:hi1+1] += w * f[k-hi1:k-lo1+1][::-1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                g[lo2:hi2+1] += w * f[lo2-k:hi2-k+1]
        
        g -= np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g /= g_norm
        
        momentum = beta_m * momentum + g
        f = f - lr * momentum
        f = np.clip(f, 0.01, 1000.0)
        f *= S / float(np.sum(f))
        
        it += 1
        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score - 1e-8:
                best_score = s
                best_f = f.copy()
                stall = 0
                beta = min(beta * 1.05, 50.0)
                lr *= 0.99
            else:
                stall += 1
                if stall > 30:
                    f = best_f.copy()
                    momentum = np.zeros(n)
                    beta = 3.0
                    lr = 0.003
                    stall = 0
            if verbose:
                print(f"    [SGD {time.monotonic()-t0:6.1f}s] it={it} beta={beta:.1f} score={s:.6f} best={best_score:.6f}")
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score
