Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Fast subgradient descent for coarse optimization

import numpy as np

from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def exact_obj(a):
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def smooth_obj(a, beta):
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    # log-sum-exp for numerical stability
    c = beta * conv
    c_max = c.max()
    log_sum = c_max + np.log(np.sum(np.exp(c - c_max)))
    M = log_sum / beta
    return 2.0 * n * M / (s * s)

def smooth_grad(a, beta):
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    
    # Softmax weights
    c = beta * conv
    c_max = c.max()
    exp_c = np.exp(c - c_max)
    w = exp_c / exp_c.sum()
    
    # Smooth max value
    log_sum = c_max + np.log(exp_c.sum())
    M = log_sum / beta
    
    # dM/da[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Using convolution: h = convolve(w, a[::-1], 'full')[n-1:2n-1]
    W = w  # length 2n-1
    a_rev = a[::-1]
    # Use FFT for efficiency
    conv_len = (2*n-1) + n - 1  # = 3n-2
    L = 1 << (conv_len - 1).bit_length()
    h = np.fft.irfft(np.fft.rfft(W, L) * np.fft.rfft(a_rev, L), L)[:2*n-1]
    h = h[n-1:2*n-1]  # Extract the correct slice

    grad = 2.0 * n * (dM_da * s - 2.0 * M) / (s**3)
    
    obj = 2.0 * n * M / (s * s)
    return grad, obj

    conv = fast_convolve(a)
    best_obj = exact_obj(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:

        scaled -= scaled.max()
        w = np.exp(scaled)

def generate_inits(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['ramp'] = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] /= inits['ramp'].sum()
    inits['ramp_dec'] = inits['ramp'][::-1].copy()
    
    tri = np.minimum(x, n-1-x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    for sf in [0.2, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf*n))**2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    bi = np.zeros(n)
    edge = max(1, n//4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    if parent is not None:
        if len(parent) >= n:
            p = np.maximum(parent[:n], 0)
            if p.sum() > 1e-10:
                inits['parent'] = p / p.sum()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            if p.sum() > 1e-10:

        if len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            if p.sum() > 1e-10:
                inits['parent_ds'] = p / p.sum()
    
    return inits
