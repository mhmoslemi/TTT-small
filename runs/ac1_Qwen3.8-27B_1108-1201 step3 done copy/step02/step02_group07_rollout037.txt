```python
"""
Smoothed-max subgradient descent with LP refinement for minimizing
peak autocorrelation of non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def compute_grad(a, p):
    """
    Compute gradient of smoothed max objective.
    p: softmax weights over conv lags (length 2n-1)
    Returns grad (length n) where grad[j] = 2 * sum_m p[j+m] * a[m]
    """
    n = len(a)
    p_rev = p[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    Fa = rfft(a, size)
    Fp = rfft(p_rev, size)
    conv_val = irfft(Fa * Fp, size)
    return 2.0 * conv_val[n-1:2*n-1][::-1]


def subgradient_optimize(a_init, time_budget, T_init=5.0, T_final=50.0,
                         lr_init=0.02, beta=0.9, verbose=True):
    """
    Smoothed-max subgradient descent with momentum and annealing T.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = a.sum()
    if S > 0:
        a = a / S
    else:
        a = np.ones(len(a)) / len(a)
    
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    momentum = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_budget:
        if time.monotonic() - t0 >= time_budget - 1.0:
            break
        
        # Anneal T
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        T = T_init + (T_final - T_init) * progress
        lr = lr_init / (1.0 + 0.5 * it * 0.01)
        
        # Compute convolution
        c = fft_convolve(a, a)
        
        # Compute softmax
        Tc = T * c
        Tc -= Tc.max()
        p = np.exp(Tc)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p /= p_sum
        
        # Compute gradient
        grad = compute_grad(a, p)
        
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        a = a - lr * momentum
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        
        it += 1
        
        # Check score every 50 iterations
        if it % 50 == 0 or time.monotonic() - t0 > time_budget - 2.0:
            s_val = score(a)
            if s_val < best_score:
                best_score = s_val
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            if verbose and it % 500 == 0:
                print(f"      [subgrad {time.monotonic()-t0:5.1f}s] "
                      f"it={it:5d} T={T:6.1f} score={s_val:.6f} best={best_score:.6f}")
        
        # Restart if stalled
        if stall_count >= 20:
            a += 0.02 * rng.standard_normal(n)
            a = np.clip(a, 0.01, None)
            a = a / a.sum()
            momentum = np.zeros(n)
            stall_count = 0
    
    # Final check
    s_val = score(a)
    if s_val < best_score:
        best_score = s_val
        best_a = a.copy()
    
    return best_a, best_score


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b) (length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def lp_refine(f, time_budget, verbose=True):
    """LP-based refinement: find best b s.t. conv(f,b) <= M, b >= 0."""
    f = np.asarray(f, dtype=np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_score = score(f)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        if time.monotonic() - t0 >= time_budget - 5.0:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        # Solve LP: max sum(b) s.t. A@b <= M, b >= 0
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 5.0, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break
        
        Sb = b.sum()
        if Sb < 1e-12:
            break
        b = b * (f.sum() / Sb)
        
        # Line search
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.1, 0.05, 0.02, 0.01]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1
        
        if best_s < best_score:
            best_score = best_s
            best_f = f.copy()
            if verbose:
                print(f"      [LP {time.monotonic()-t0:5.1f}s] it={it:3d} score={best_s:.6f}")
    
    return best_f, best_score


def local_search(a_init, time_budget, verbose=True):
    """Perturbation-based local search."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    temp = 0.001
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_budget:
        if time.monotonic() - t0 >= time_budget - 1.0:
            break
        
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.3:
            # Peak-directed perturbation
            c = fft_convolve(a, a)
            peak = c.max()
            # Reduce contributions to peak
            p = c / (c.max() + 1e-10)
            grad = compute_grad(a, p / p.sum())
            candidate -= 0.01 * grad
        elif r < 0.6:
            # Random multiplicative perturbation
            candidate *= np.exp(rng.normal(0, 0.02, n))
        elif r < 0.8:
            # Coordinate perturbation
            idx = rng.integers(0, n)
            width = max(1, n // 32)
            start = max(0, idx - width // 2)
            end = min(n, start + width)
            candidate[start:end] *= np.exp(rng.normal(0, 0.05, end - start))
        else:
            # Swap-like: reduce high, increase low
            c = fft_convolve(a, a)
            high_mask = a > np.median(a)
            candidate[high_mask] *= 0.98
            candidate[~high_mask] *= 1.02
        
        candidate = np.clip(candidate, 1e-6, None)
        candidate = candidate / candidate.sum()
        new_score = score(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
        elif rng.random() < np.exp(-(new_score - score(a)) / max(temp, 1e-15)):
            a = candidate
        
        temp *= cooling
    
    return best_a, best_score


def make_initialization(n, seed, method='hann'):
    """Create initial sequence."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    
    if method == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.05 * rng.standard_normal(n)
    elif method == 'bartlett':
        w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
        a = 0.3 + 0.7 * w
        a += 0.05 * rng.standard_normal(n)
    elif method == 'chirp':
        window = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
        a = 0.5 + 0.5 * window + 0.15 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
        a += 0.03 * rng.standard_normal(n)
    elif method == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-(t - center)**2 / (2 * sigma**2))
        a += 0.03 * rng.standard_normal(n)
    elif method == 'asymmetric':
        # Skewed toward one side
        a = np.exp(-t / (n / 3.0))
        a += 0.05 * rng.standard_normal(n)
    elif method == 'two_bump':
        # Two bumps
        c1, c2 = n * 0.3, n * 0.7
        sigma = n / 8.0
        a = np.exp(-(t - c1)**2 / (2*sigma**2)) + np.exp(-(t - c2)**2 / (2*sigma**2))
        a += 0.03 * rng.standard_normal(n)
    elif method == 'random':
        a = 0.5 + rng.random(n)
    else:
        a = np.ones(n)
    
    return np.clip(a, 0.01, None)
```

```python
"""
Smoothed-max subgradient descent with momentum, LP refinement, and local search
for minimizing peak autocorrelation of non-negative sequences.
Uses FFT-based gradient computation for O(n log n) per iteration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def compute_grad(a, p):
    """
    Compute gradient of smoothed max objective.
    p: softmax weights over conv lags (length 2n-1)
    Returns grad (length n): grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    """
    n = len(a)
    p_rev = p[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    Fa = rfft(a, size)
    Fp = rfft(p_rev, size)
    conv_val = irfft(Fa * Fp, size)
    return 2.0 * conv_val[n - 1:2 * n - 1][::-1]


def subgradient_optimize(a_init, time_budget, T_init=5.0, T_final=200.0,
                         lr_init=50.0, beta=0.92, verbose=True, tag=""):
    """
    Smoothed-max subgradient descent with momentum and annealing temperature.
    lr is in "units of gradient scale" - we scale gradient to O(1) then use lr.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = a.sum()
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(len(a)) / len(a)

    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    momentum = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    rng = np.random.default_rng(777 + n)
    last_improve = 0

    while time.monotonic() - t0 < time_budget:
        if time.monotonic() - t0 >= time_budget - 1.0:
            break

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_budget, 1.0))
        # Anneal T: start smooth, end sharp
        T = T_init + (T_final - T_init) * (progress ** 0.7)
        # Decay learning rate
        lr = lr_init / (1.0 + 0.3 * progress * 10.0)

        # Compute convolution
        c = fft_convolve(a, a)

        # Compute softmax weights
        Tc = T * c
        Tc -= Tc.max()
        p = np.exp(Tc)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p /= p_sum

        # Compute gradient
        grad = compute_grad(a, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Normalize gradient for stable step sizing
        grad_normalized = grad / grad_norm

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad_normalized
        a = a - lr * momentum / n
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n

        it += 1

        # Check score periodically
        check_interval = max(10, it // 50)
        if it % check_interval == 0 or elapsed > time_budget - 2.0:
            s_val = score(a)
            if s_val < best_score:
                best_score = s_val
                best_a = a.copy()
                stall_count = 0
                last_improve = it
            else:
                stall_count += 1

            if verbose and it % (check_interval * 10) == 0:
                print(f"      [subgrad{tag} {elapsed:5.1f}s] "
                      f"it={it:6d} T={T:7.1f} score={s_val:.6f} best={best_score:.6f}")

        # Restart if stalled for too long
        if stall_count >= 30:
            perturbation = 0.03 * (1.0 + progress)
            a += perturbation * rng.standard_normal(n)
            a = np.clip(a, 1e-4, None)
            a = a / a.sum()
            momentum = np.zeros(n)
            stall_count = 0

    # Final check
    s_val = score(a)
    if s_val < best_score:
        best_score = s_val
        best_a = a.copy()

    return best_a, best_score


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b) (length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(f, time_budget, verbose=True, tag=""):
    """LP-based refinement: find best b s.t. conv(f,b) <= M, b >= 0."""
    f = np.asarray(f, dtype=np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_score = score(f)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_budget:
        if time.monotonic() - t0 >= time_budget - 8.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        # Solve LP
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 8.0, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break

        Sb = float(b.sum())
        if Sb < 1e-12:
            break
        b = b * (f.sum() / Sb)

        # Line search
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.15, 0.1, 0.05, 0.03, 0.01]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_f = f.copy()
            stall = 0
            if verbose:
                print(f"      [LP{tag} {time.monotonic()-t0:5.1f}s] "
                      f"it={it:3d} score={best_s:.6f}")
        else:
            stall += 1

        if stall >= 15:
            break

    return best_f, best_score


def local_search(a_init, time_budget, verbose=True, tag=""):
    """Perturbation-based local search with diverse moves."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    temp = 0.0005
    cooling = 0.99995

    while time.monotonic() - t0 < time_budget:
        if time.monotonic() - t0 >= time_budget - 1.0:
            break

        candidate = a.copy()
        r = rng.random()

        if r < 0.25:
            # Gradient-directed: reduce peak contributions
            c = fft_convolve(a, a)
            Tc = 30.0 * c
            Tc -= Tc.max()
            p = np.exp(Tc)
            p /= p.sum()
            grad = compute_grad(a, p)
            candidate -= 0.005 * grad
        elif r < 0.45:
            # Multiplicative perturbation (global)
            candidate *= np.exp(rng.normal(0, 0.015, n))
        elif r < 0.65:
            # Local window perturbation
            width = max(2, n // 16)
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, width))
        elif r < 0.8:
            # Anti-peak: reduce where a[j]*a[n-1-j] is large
            products = a * a[::-1]
            top_k = max(3, n // 10)
            top_idx = np.argsort(products)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, top_k))
            bottom_idx = np.argsort(products)[:-top_k]
            num_inc = min(5, len(bottom_idx))
            if num_inc > 0:
                chosen = rng.choice(bottom_idx, num_inc, replace=False)
                candidate[chosen] *= np.exp(rng.normal(0.02, 0.01, num_inc))
        else:
            # Random coordinate
            idx = int(rng.integers(0, n))
            candidate[idx] *= np.exp(rng.normal(0, 0.1))

        candidate = np.clip(candidate, 1e-6, None)
        s = candidate.sum()
        if s > 1e-10:
            candidate = candidate / s
        else:
            continue

        new_score = score(candidate)
        curr_score = score(a)

        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
        elif rng.random() < np.exp(-(new_score - curr_score) / max(temp, 1e-15)):
            a = candidate

        temp *= cooling

    return best_a, best_score


def make_initialization(n, seed, method='hann'):
    """Create initial sequence with given shape."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)

    if method == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.05 * rng.standard_normal(n)
    elif method == 'bartlett':
        w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
        a = 0.3 + 0.7 * w
        a += 0.05 * rng.standard_normal(n)
    elif method == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-(t - center) ** 2 / (2 * sigma ** 2))
        a += 0.03 * rng.standard_normal(n)
    elif method == 'asymmetric':
        a = np.exp(-t / (n / 3.0)) + 0.1
        a += 0.05 * rng.standard_normal(n)
    elif method == 'two_bump':
        c1, c2 = n * 0.3, n * 0.7
        sigma = n / 8.0
        a = np.exp(-(t - c1) ** 2 / (2 * sigma ** 2)) + np.exp(-(t - c2) ** 2 / (2 * sigma ** 2))
        a += 0.03 * rng.standard_normal(n)
    elif method == 'chirp':
        window = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
        a = 0.5 + 0.5 * window + 0.2 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
        a += 0.03 * rng.standard_normal(n)
    elif method == 'random':
        a = 0.3 + 0.7 * rng.random(n)
    elif method == 'concave':
        # Concave shape: high in middle, low at edges (opposite of convex)
        a = 1.0 - (2.0 * t / max(1, n - 1) - 1.0) ** 2
        a += 0.03 * rng.standard_normal(n)
    elif method == 'sawtooth':
        period = max(4, n // 8)
        frac = t % period
        a = frac / period
        a += 0.05 * rng.standard_normal(n)
    else:
        a = np.ones(n) + 0.05 * rng.standard_normal(n)

    a = np.clip(a, 0.01, None)
    return a / a.sum()

        progress_in_phase = rank / max(1, len(ordered_indices))
        if progress_in_phase < 0.3:
            T_init, T_final = 3.0, 50.0
            lr_init = 30.0
        elif progress_in_phase < 0.7:
            T_init, T_final = 10.0, 200.0
            lr_init = 50.0
        else:
            T_init, T_final = 5.0, 100.0
            lr_init = 20.0

        verbose = (rank < 5) or (n <= 256)
        f_best, s_best = subgradient_optimize(f0, t_alloc,
                                              T_init=T_init, T_final=T_final,
                                              lr_init=lr_init, verbose=verbose,
                                              tag=f"({n},{init_name[:4]})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}, init={init_name}) ***")
        elif verbose:
            print(f"  n={n:4d} init={init_name:12s} score={s_best:.6f}")

    print(f"\nPhase 1 done. Best so far: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    # Phase 2: LP refinement (only for manageable sizes)
    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30 and n_best <= 512:
            lp_time = min(250.0, remaining)
            print(f"\n=== Phase 2: LP refinement (n={n_best}, t={lp_time:.0f}s) ===")
            f_lp, s_lp = lp_refine(best_seq, lp_time, verbose=True)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = f_lp.copy()
                print(f"  LP improved to: {best_score:.6f}")
            else:
                print(f"  LP no improvement: {s_lp:.6f}")
        else:
            print(f"\n=== Phase 2: Skipped LP (n={n_best}, remaining={max(0,remaining):.0f}s) ===")

    # Phase 3: Local search
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 50.0
        if remaining > 15:
            ls_time = min(120.0, remaining)
            print(f"\n=== Phase 3: Local search (t={ls_time:.0f}s) ===")
            f_ls, s_ls = local_search(best_seq, ls_time, verbose=True)
            if s_ls < best_score:
                best_score = s_ls
                best_seq = f_ls.copy()
                print(f"  Local search: {best_score:.6f}")

    # Phase 4: Final subgradient polish with high T
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 15:
            print(f"\n=== Phase 4: Final polish (t={remaining:.0f}s) ===")
            f_fin, s_fin = subgradient_optimize(best_seq, remaining,
                                                 T_init=100.0, T_final=300.0,
                                                 lr_init=15.0, beta=0.95,
                                                 verbose=False, tag="(final)")
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 50}")
        return final_list

    return [1.0] * 256
```
