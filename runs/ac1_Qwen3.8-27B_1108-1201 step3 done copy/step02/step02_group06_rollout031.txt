    size = 1

        size <<= 1

    n = len(a)
    if n == 0:
        return np.inf

"""
Subgradient + Informed SA optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses FFT-based soft subgradient computation,
annealed temperature schedules, informed SA moves guided by autocorrelation structure,
population-based diversity, and coordinate descent for final polish.

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_soft_subgradient(a, tau):
    """Compute soft subgradient of max(conv(a,a)) using softmax weights."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    w = np.exp((c - c_max) / tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    Ar = np.fft.rfft(a_rev, size)
    Wr = np.fft.rfft(w, size)
    conv_result = np.fft.irfft(Ar * Wr, size)
    g = 2.0 * conv_result[n - 1:2 * n - 1]
    return g, c, w


def subgradient_phase(a_init, time_limit, seed=42):
    """Phase 1: Subgradient descent with annealed temperature."""
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    sum_target = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    iter_count = 0
    stagnant = 0

    tau_start = 0.3
    tau_end = 0.02
    lr_start = 0.15
    lr_end = 0.005

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** elapsed_frac

        g, c, w = compute_soft_subgradient(a, tau)

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = np.sum(a_new)
        if a_new_sum < 1e-10:
            stagnant += 1
            continue
        a_new = a_new * (sum_target / a_new_sum)

        score = local_score(a_new)
        iter_count += 1

        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best_a = a.copy()
            stagnant = 0
        elif score < best_score + 0.005:
            if rng.random() < 0.4:
                a = a_new
            stagnant += 1
        else:
            stagnant += 1

        if stagnant > 200:
            a = a + 0.08 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a * (sum_target / np.sum(a))
            stagnant = 0

    return best_a, best_score


def sa_phase(a_init, time_limit, seed=123):
    """Phase 2: Informed simulated annealing with adaptive moves."""
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    sum_a = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score
    t0 = time.monotonic()

    temperature = 0.004
    min_temp = 0.00005
    iter_count = 0
    no_improve = 0
    best_iter_gap = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.25:
            # Peak-directed move
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            n_inc = min(6, len(other_idx))
            inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
            candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.45:
            # Block modification
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.6:
            # Global multiplicative noise
            candidate *= np.exp(rng.normal(0.0, 0.007, size=n))
        elif r < 0.75:
            # Asymmetric redistribution
            shift = rng.uniform(0.02, 0.06) * rng.choice([1, -1])
            mid = n // 2
            candidate[:mid] *= (1.0 - shift)
            candidate[mid:] *= (1.0 + shift)
        elif r < 0.9:
            # Two-block exchange
            bs = max(2, int(rng.integers(2, n // 8)))
            s1 = int(rng.integers(0, n - bs))
            s2 = int(rng.integers(0, n - bs))
            if s1 != s2:
                swap = rng.uniform(0.03, 0.08)
                candidate[s1:s1 + bs] *= (1.0 - swap)
                candidate[s2:s2 + bs] *= (1.0 + swap)
        else:
            # Random restart with small perturbation
            candidate = a * np.exp(rng.normal(0.0, 0.02, size=n))

        candidate = np.maximum(candidate, 0.0)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_a / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                best_iter_gap = iter_count
        else:
            no_improve += 1
            if rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
                a = candidate
                current_score = new_score

        # Adaptive temperature
        if no_improve > 1000:
            temperature = max(min_temp, temperature * 0.85)
            no_improve = 0
        elif no_improve < 200:
            temperature = min(0.006, temperature * 1.005)

        temperature = max(min_temp, temperature * 0.99995)

    return best_a, best_score, iter_count


def coordinate_polish(a_init, time_limit, seed=77):
    """Phase 3: Coordinate descent with multiplicative updates."""
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    sum_target = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            orig_val = a[i]
            base_sum = sum_target - orig_val
            best_local = best_score
            best_factor = 1.0
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]:
                new_val = max(0.0, orig_val * factor)
                new_sum = base_sum + new_val
                if new_sum < 0.01:
                    continue
                a[i] = new_val
                cand = a * (sum_target / new_sum)
                s = local_score(cand)
                if s < best_local:
                    best_local = s
                    best_factor = factor
            a[i] = max(0.0, orig_val * best_factor)

        a = np.maximum(a, 0.0)
        a = a * (sum_target / np.sum(a))
        passes += 1

        current_score = local_score(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    return best_a, best_score


def make_diverse_inits(n, rng, provided_seq):
    """Generate diverse initializations."""
    inits = []
    if provided_seq is not None:
        inits.append(provided_seq.copy())
    # Random
    inits.append(np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, None))
    # Chirp
    t = np.arange(n, dtype=np.float64)
    inits.append(np.clip(1.0 + 0.3 * np.cos(np.pi * t * t / n) + 0.05 * rng.standard_normal(n), 0.01, None))
    # Decreasing envelope
    env = np.linspace(1.5, 0.5, n)
    inits.append(np.clip(env + 0.1 * rng.standard_normal(n), 0.01, None))
    # Bimodal
    centers = n // 4, 3 * n // 4
    a = np.ones(n)
    for c in centers:
        a += 0.5 * np.exp(-((t - c) ** 2) / (2 * (n // 8) ** 2))
    inits.append(np.clip(a + 0.05 * rng.standard_normal(n), 0.01, None))
    # High-contrast
    a = np.where(rng.random(n) < 0.3, 0.1, 2.0)
    inits.append(np.clip(a + 0.05 * rng.standard_normal(n), 0.01, None))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: subgradient descent -> informed SA -> coordinate polish.
    Uses FFT-based subgradients, autocorrelation-guided SA moves, and adaptive
    temperature management to minimize the normalized peak autocorrelation.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 35.0
    rng = np.random.default_rng(seed)

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(provided_seq) == 0:
            provided_seq = None

    n = 256
    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    if provided_seq is not None:
        if len(provided_seq) != n:
            x_old = np.linspace(0, 1, len(provided_seq))
            x_new = np.linspace(0, 1, n)
            provided_seq = np.interp(x_new, x_old, provided_seq)
        provided_seq = np.clip(provided_seq, 0.0, 1000.0)
        s = local_score(provided_seq)
        print(f"Provided: n={n}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided_seq.copy()

    if best_seq is None:
        best_seq = np.ones(n)
        best_score = local_score(best_seq)
        print(f"Default init: score={best_score:.6f}")

    # Generate diverse inits
    inits = make_diverse_inits(n, rng, provided_seq)
    init_scores = []
    for idx, init in enumerate(inits):
        init = np.clip(init, 0.0, None)
        s_init = local_score(init)
        init_scores.append(s_init)
        print(f"  Init {idx}: score={s_init:.6f}")
        if s_init < best_score:
            best_score = s_init
            best_seq = init.copy()

    # === Phase 1: Subgradient descent on top initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Subgradient descent")
    print(f"{'='*60}")

    # Sort inits by score, take top 3
    top_init_indices = np.argsort(init_scores)[:3]
    top_init_indices = list(top_init_indices)
    if provided_seq is not None and 0 not in top_init_indices:
        top_init_indices[0] = 0

    remaining = deadline - time.monotonic()
    sg_time_each = min(80.0, remaining * 0.15 / len(top_init_indices))

    for idx in top_init_indices:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(sg_time_each, deadline - time.monotonic() - 30)
        if t_alloc < 10:
            break
        print(f"  SubGrad on init {idx} (score={init_scores[idx]:.6f}), time={t_alloc:.0f}s")
        a_sg, score_sg = subgradient_phase(inits[idx], t_alloc, seed=seed + idx)
        print(f"  SubGrad result: {score_sg:.6f}")
        if score_sg < best_score:
            best_score = score_sg
            best_seq = a_sg.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 2: Informed SA (main phase) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Informed SA")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    sa_time = min(remaining * 0.65, 1100.0)
    sa_time = max(sa_time, 100.0)
    print(f"  SA time budget: {sa_time:.0f}s")

    a_sa, score_sa, n_iters = sa_phase(best_seq, sa_time, seed=seed + 100)
    print(f"  SA result: {score_sa:.6f} ({n_iters} iterations)")
    if score_sa < best_score:
        best_score = score_sa
        best_seq = a_sa.copy()
        print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Coordinate descent polish")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    cd_time = min(remaining * 0.3, 250.0)
    cd_time = max(cd_time, 30.0)
    if cd_time > 20:
        print(f"  CD time budget: {cd_time:.0f}s")
        a_cd, score_cd = coordinate_polish(best_seq, cd_time, seed=seed + 200)
        print(f"  CD result: {score_cd:.6f}")
        if score_cd < best_score:
            best_score = score_cd
            best_seq = a_cd.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final SA on best ===
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*60}")
        print(f"Phase 4: Final SA polish")
        print(f"{'='*60}")
        remaining = deadline - time.monotonic() - 20
        sa2_time = max(remaining, 30.0)
        a_final, score_final, n_iters2 = sa_phase(best_seq, sa2_time, seed=seed + 300)
        print(f"  Final SA result: {score_final:.6f} ({n_iters2} iterations)")
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = local_score(np.array(final_list))
    print(f"\n{'='*60}")
    print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
    print(f"{'='*60}")

    return final_list
