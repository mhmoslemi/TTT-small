"""
Improved optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Uses enhanced LP-blend with dual-guided shaking,
FISTA on smoothed objective, multiplicative updates, and multi-restart
strategy. Exploits the structure that the peak is typically at the center lag
for non-negative sequences.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize to target sum S."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def build_sparse_lp_matrix(a):
    """Build sparse convolution constraint matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, duals)."""
    n = len(a)
    A = build_sparse_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        duals = -res.uneqlin if res.uneqlin is not None else np.zeros(2 * n - 1)
        return res.x, duals
    return np.zeros(n), np.zeros(2 * n - 1)

def _apply_shake(a, conv_a, peak_idx, rng, S, strength=0.5):
    """Perturb a using subgradient information at peak and nearby lags."""
    n = len(a)
    # Compute weighted subgradient at top few lags
    top_lags = np.argsort(conv_a)[-min(5, len(conv_a)):]
    weights = conv_a[top_lags] / (np.sum(conv_a[top_lags]) + 1e-10)

    grad = np.zeros(n)
    for lag in top_lags:
        w = 0
        for lag_i in top_lags:
            if lag_i == lag:
                w = weights[list(top_lags).index(lag)]
                break
        for j in range(n):
            m = lag - j
            if 0 <= m < n:
                grad[j] += w * 2.0 * a[m]

    # Apply multiplicative perturbation in negative gradient direction
    eta = strength / (np.max(grad) + 1e-10)
    a_new = a * np.exp(-eta * grad)

    # Add small random noise
    a_new *= np.exp(rng.normal(0, 0.02, n))
    a_new = np.maximum(a_new, 1e-6)
    a_new = normalize_seq(a_new, S)
    return a_new


def fista_smooth(a, time_limit, verbose=True):
    """
    FISTA on log-sum-exp smoothed objective.
    Minimizes (1/tau)*log(sum_k exp(tau * conv_k(a)))
    subject to sum(a) = S, a >= 0.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_val = _smoothed_value(a, 5.0)

    # Start with moderate temperature, decrease over time
    tau = 5.0
    L = _estimate_L(a, tau)  # Lipschitz constant estimate
    step_prev = 1.0
    y = a.copy()
    k = 1

    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at y
        conv_y = fft_convolve(y, y)
        shifted = conv_y - np.max(conv_y)  # for numerical stability
        p = np.exp(tau * shifted)
        p /= np.sum(p)

        # Gradient: (2/tau) * cross_corr(p, y)
        # grad[j] = (2/tau) * sum_m p[j+m] * y[m]
        grad = np.zeros(n)
        for j in range(n):
            # sum over m=0..n-1 of p[j+m] * y[m], where p is zero outside [0,2n-2]
            m_max = min(n - 1, 2 * n - 2 - j)
            if m_max >= 0:
                grad[j] = np.dot(p[j:j + m_max + 1], y[:m_max + 1])
        grad *= 2.0 / tau

        # FISTA step
        step = 1.0 / L
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0)
        # Project to sum constraint
        s = np.sum(x_new)
        if s < 1e-10:
            x_new = np.full(n, S / n)
        else:
            x_new = x_new * (S / s)

        # Check if we improved
        val = _smoothed_value(x_new, tau)
        if val < best_val:
            best_val = val
            best_a = x_new.copy()

        # Update momentum
        k_new = (-1 + np.sqrt(1 + 4 * k * k)) / 2
        beta = (k_new - 1) / (k_new + 2)
        y = x_new + beta * (x_new - y)
        y = np.maximum(y, 0)
        s_y = np.sum(y)
        if s_y < 1e-10:
            y = x_new.copy()
        else:
            y = y * (S / s_y)

        k = k_new + 1

        # Decrease temperature slowly
        if iteration % 500 == 0 and tau > 1.0:
            tau *= 0.95
            L = _estimate_L(a, tau)

        if verbose and iteration % 2000 == 0:
            score = 2.0 * n * np.max(fft_convolve(best_a, best_a)) / (S * S)
            print(f"    FISTA iter {iteration}: score={score:.6f}, tau={tau:.2f}")

    return best_a


def _smoothed_value(a, tau):
    """Compute smoothed objective value."""
    conv = fft_convolve(a, a)
    shifted = conv - np.max(conv)
    return np.max(conv) + np.log(np.sum(np.exp(tau * shifted))) / tau


def _estimate_L(a, tau):
    """Estimate Lipschitz constant of the smoothed gradient."""
    n = len(a)
    # The Hessian of (1/tau)*log(sum exp(tau*conv_k)) involves second derivatives of conv
    # A crude bound: L <= (4/tau) * max_k |Q_k|_op where Q_k is the matrix for conv_k
    # |Q_k|_op <= 2 * max_i a[i] (rough bound)
    return (4.0 / tau) * (2.0 * np.max(a) + 1.0) * n * 0.1


def multiplicative_polish(a, time_limit, verbose=True):
    """
    Multiplicative update polish. Iteratively reduces elements contributing
    most to the peak lag.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(42)
    eta = 0.1

    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_convolve(a, a)
        peak = np.max(conv)
        peak_idx = int(np.argmax(conv))

        # Subgradient at peak
        grad = np.zeros(n)
        for j in range(n):
            m = peak_idx - j
            if 0 <= m < n:
                grad[j] = 2.0 * a[m]

        if np.max(grad) < 1e-10:
            break

        # Multiplicative update
        scale = eta / np.max(grad)
        a = a * np.exp(-scale * grad)
        a = np.maximum(a, 1e-8)
        a = normalize_seq(a, S)

        new_conv = np.max(fft_convolve(a, a))
        if new_conv < best_conv - 1e-12:
            best_conv = new_conv
            best_a = a.copy()
        else:
            # Revert and try smaller step
            a = best_a.copy()
            eta *= 0.9
            if eta < 0.001:
                # Add random perturbation
                a *= np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-8)
                a = normalize_seq(a, S)
                eta = 0.05

        if verbose and iteration % 5000 == 0:
            score = 2.0 * n * best_conv / (S * S)
            print(f"    Multi iter {iteration}: score={score:.6f}, eta={eta:.4f}")

    return best_a


def interpolate_sequence(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def multi_restart(lp_result, time_limit, verbose=True):
    """
    Multi-restart: perturb the best solution and re-optimize.
    """
    n = len(lp_result)
    S = np.sqrt(2.0 * n)
    t0 = time.monotonic()
    best_a = lp_result.copy()
    best_score = local_score(lp_result)
    rng = np.random.default_rng(999)

    n_restarts = 8
    time_per_restart = (time_limit - 10) / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 10:
            break
        t_rem = time.PerfCounter()
        time_left_for_this = min(time_per_restart, time_limit - (time.monotonic() - t0) - 5)
        if time_left_for_this < 10:
            break

        # Different perturbation strategies
        a_pert = best_a.copy()
        strategy = r % 4

        if strategy == 0:
            # Gaussian noise
            a_pert *= np.exp(rng.normal(0, 0.1, n))
        elif strategy == 1:
            # Block flip
            blocks = n // 4
            for b in range(blocks):
                a_pert[b*4:(b+1)*4] *= rng.uniform(0.5, 1.5)
        elif strategy == 2:
            # Reversed + noise
            a_pert = a_pert[::-1] * np.exp(rng.normal(0, 0.05, n))
        else:
            # Sparse perturbation
            indices = rng.choice(n, size=n//10, replace=False)
            a_pert[indices] *= rng.uniform(0.3, 2.0, size=len(indices))

        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = normalize_seq(a_pert, S)

        # Short LP refinement
        result = lp_refine_v2(a_pert, time_left_for_this, verbose=False, shake_interval=20)
        sc = local_score(result)
        if sc < best_score:
            best_score = sc
            best_a = result
            if verbose:
                print(f"    Restart {r}: new best score={best_score:.6f}")

    return best_a


def make_structured_init(n, rng):
    """Create diverse structured initializations."""
    t = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 6)

    if choice == 0:
        # Chirp-like
        alpha = rng.uniform(0.2, 0.5)
        a = 1.0 + alpha * np.cos(2 * np.pi * alpha * t * t / n)
    elif choice == 1:
        # Bell curve (concentrated in middle)
        center = n / 2
        sigma = n * rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2) + 0.1
    elif choice == 2:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        sigma = n * 0.08
        a = np.exp(-0.5 * ((t - c1) / sigma) ** 2) + np.exp(-0.5 * ((t - c2) / sigma) ** 2) + 0.1
    elif choice == 3:
        # Sine
        f = rng.integers(1, 5)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * f * t / n + rng.uniform(0, 2 * np.pi)) + 0.3
    elif choice == 4:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            f = rng.integers(1, 10)
            a += 0.2 * np.cos(2 * np.pi * f * t / n + rng.uniform(0, 2 * np.pi))
    else:
        # Edge-concentrated
        decay = rng.uniform(0.05, 0.15)
        a = np.exp(-decay * np.minimum(t, n - 1 - t)) + 0.1

    return np.maximum(a, 0.01)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-blend, FISTA, multiplicative updates,
    and multi-restart to minimize 2n*max_conv/sum^2.
    """
    t_global = time.monotime() if hasattr(time, 'monotime') else time.monotonic()
    deadline = t_global + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        n_prov = len(provided_seq)
        print(f"Provided sequence: n={n_prov}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)

    # === Phase 1: LP refinement from provided sequence ===
    print("\n=== Phase 1: LP refinement (provided) ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(300.0, (deadline - time.monotonic()) * 0.4)
        print(f"  LP v2 refine (n={n}), limit={t_limit:.0f}s")
        result = lp_refine_v2(provided_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  After LP: score={best_score:.6f}")
        else:
            print(f"  LP did not improve: score={best_score:.6f}")

    # === Phase 2: FISTA smoothing ===
    print("\n=== Phase 2: FISTA smoothing ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.15)
        if t_limit > 20:
            print(f"  FISTA (n={n}), limit={t_limit:.0f}s")
            result = fista_smooth(best_seq, t_limit, verbose=True)
            if update_best(result):
                print(f"  After FISTA: score={best_score:.6f}")

    # === Phase 3: Multiplicative polish ===
    print("\n=== Phase 3: Multiplicative polish ===")
    if best_seq is not None:
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.15)
        if t_limit > 20:
            print(f"  Multi (n={len(best_seq)}), limit={t_limit:.0f}s")
            result = multiplicative_polish(best_seq, t_limit, verbose=True)
            if update_best(result):
                print(f"  After multi: score={best_score:.6f}")

    # === Phase 4: Multi-restart ===
    print("\n=== Phase 4: Multi-restart ===")
    if best_seq is not None:
        t_limit = min(300.0, (deadline - time.monotonic()) * 0.35)
        if t_limit > 60:
            print(f"  Multi-restart, limit={t_limit:.0f}s")
            result = multi_restart(best_seq, t_limit, verbose=True)
            if update_best(result):
                print(f"  After restart: score={best_score:.6f}")

    # === Phase 5: Final LP on best ===
    print("\n=== Phase 5: Final LP refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            t_limit = min(remaining, 150.0)
            print(f"  Final LP, limit={t_limit:.0f}s")
            result = lp_refine_v2(best_seq, t_limit, verbose=True, shake_interval=15)
            if update_best(result):
                print(f"  After final LP: score={best_score:.6f}")

    # === Phase 6: Try different lengths ===
    print("\n=== Phase 6: Different lengths ===")
    if provided_seq is not None:
        for n_try in [512, 768, 896]:
            if time.monotonic() > deadline - 30:
                break
            t_per = min(45.0, (deadline - time.monotonic()) * 0.15)
            if t_per < 15:
                break
            a0 = interpolate_sequence(provided_seq, n_try)
            a0 = np.maximum(a0, 0.01)
            print(f"  Trying n={n_try}, limit={t_per:.0f}s")
            result = lp_refine_v2(a0, t_per, verbose=False)
            if update_best(result):
                print(f"  *** n={n_try}: new best score={best_score:.6f} ***")

    # === Phase 7: Structured initializations ===
    print("\n=== Phase 7: Structured inits ===")
    for n in [512, 640, 768]:
        if time.monotonic() > deadline - 20:
            break
        for init_i in range(3):
            if time.monotonic() > deadline - 10:
                break
            t_per = min(25.0, (deadline - time.monotonic()) * 0.1)
            if t_per < 10:
                continue
            a0 = make_structured_init(n, rng)
            result = lp_refine_v2(a0, t_per, verbose=False, shake_interval=15)
            if update_best(result):
                print(f"  *** n={n}, init{init_i}: new best score={best_score:.6f} ***")

    # Final
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
