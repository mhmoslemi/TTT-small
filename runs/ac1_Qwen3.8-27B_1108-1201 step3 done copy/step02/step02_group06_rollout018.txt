   - Find the peak location k* of conv(a,a)
   - The peak is c[k*] = sum_j a[j]*a[k*-j]
   - To reduce c[k*], we need to reduce the products a[j]*a[k*-j]
   - The most effective way is to reduce the large elements that contribute to the peak
   - But reducing a[j] also affects other convolution entries
   - A good heuristic: reduce a[j] proportionally to its contribution to the peak, and distribute the "saved" mass to elements that contribute least to the peak

"""
Multi-strategy optimization for minimizing normalized peak auto-convolution
of non-negative step functions. Uses: (1) analytical gradient descent in 
softmax parameterization with adaptive annealing, (2) peak-shaving coordinate
descent, (3) LP-based direction finding, and (4) cosine-modulated 
initializations. Leverages the key insight that d(conv(a,a)[m])/d(a[j]) = 2*a[m-j]
for valid indices, enabling efficient gradient computation via cross-correlation.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_gradient(a, tau):
    """Compute gradient of soft-max(conv(a,a)) w.r.t. a using analytical formula.
    
    Uses the identity: d(conv(a,a)[m])/d(a[j]) = 2*a[m-j] if 0<=m-j<n.
    Gradient = 2 * cross_correlation(W, a, 'valid') where W is softmax weights.
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    W = exp_c / sum_exp
    soft_max_val = c_max + tau * np.log(sum_exp)
    # Gradient via cross-correlation
    grad = 2.0 * np.correlate(W, a, mode='valid')
    return grad, soft_max_val

    # Initialize theta from a_init
    a_safe = np.clip(a_init, 1e-6, None)
    theta = np.log(a_safe / np.sum(a_safe)) * S  # so that softmax(theta)*S ≈ a_safe
    
    # Actually, let's use: a[j] = S * exp(theta[j]) / sum(exp(theta))
    # Initialize: theta[j] = log(a_safe[j]) (the S factor is implicit)
    theta = np.log(a_safe)
    theta -= np.max(theta)  # for numerical stability
    
    def theta_to_a(th):
        th_shifted = th - np.max(th)
        e = np.exp(th_shifted)
        return e / np.sum(e) * S
    
    a = theta_to_a(theta)
    best_a = a.copy()
    best_score = local_score(a)
    
    # Adam optimizer
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    base_lr = 1.0
    
    t0 = time.monotonic()
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Cosine annealing for temperature
        tau = tau_end + 0.5 * (tau_start - tau_end) * (1 + np.cos(np.pi * elapsed_frac))
        tau = max(tau, tau_end)
        
        # Compute gradient of soft-max w.r.t. a
        grad_a, _ = compute_gradient(a, tau)
        
        # Reparameterize to theta: grad_theta[k] = a[k] * (grad_a[k] - dot(grad_a,a)/S)
        inner = np.dot(grad_a, a) / S
        grad_theta = a * (grad_a - inner)
        
        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        # Learning rate with decay
        lr = base_lr / np.sqrt(1 + 0.05 * t_step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Update a
        a = theta_to_a(theta)
        
        # Compute score
        score = local_score(a)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # If stuck, perturb and restart with new temperature
        if stagnation > 200:
            theta += 0.1 * rng.standard_normal(n)
            theta -= np.max(theta)
            a = theta_to_a(theta)
            stagnation = 0
    
    return best_a, best_score


def peak_shave(a_init, time_limit, seed=42):
    """Iterative peak shaving: identify peak, reduce contributing elements,
    redistribute mass to low-contribution elements."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S = np.sum(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    lr = 0.05
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        
        # Compute contribution of each element to the peak
        # c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contribs = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contribs[j] = a[j] * a[k]
        
        # Also check the symmetric peak (since conv is symmetric)
        sym_idx = (2 * n - 2) - peak_idx
        if sym_idx != peak_idx:
            for j in range(n):
                k = sym_idx - j
                if 0 <= k < n:
                    contribs[j] += a[j] * a[k]
        
        # Identify top contributors to reduce
        total_contrib = np.sum(contribs)
        if total_contrib < 1e-10:
            break
        
        # Rank by contribution
        sorted_idx = np.argsort(contribs)[::-1]
        
        # Reduce top contributors, increase bottom contributors
        n_reduce = max(3, n // 8)
        n_incre = max(3, n // 8)
        
        reduce_idx = sorted_idx[:n_reduce]
        incre_idx = sorted_idx[-n_incre:]
        
        # Avoid modifying the same indices
        overlap = set(reduce_idx) & set(incre_idx)
        if overlap:
            incre_idx = [i for i in incre_idx if i not in set(reduce_idx)]
        
        if len(incre_idx) == 0:
            break
        
        # Scale factors
        red_scale = 1.0 - lr * (1.0 + rng.random())
        inc_scale = 1.0 + lr * (1.0 + rng.random())
        
        a_new = a.copy()
        a_new[reduce_idx] *= red_scale
        a_new[incre_idx] *= inc_scale
        a_new = np.clip(a_new, 0.0, None)
        
        # Renormalize to maintain sum
        new_sum = np.sum(a_new)
        if new_sum < 1e-10:
            break
        a_new *= S / new_sum
        
        score_new = local_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            # Gradually reduce learning rate
            lr *= 0.999
        else:
            # Try with smaller step
            lr *= 0.9
            if lr < 0.001:
                break
    
    return best_a, best_score


def cosine_init(n, k, alpha, phase, rng):
    """Generate a cosine-modulated initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def multitone_init(n, rng, n_tones=5):
    """Generate a multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def build_constraint_matrix(f, n):
    """Build constraint matrix A for LP. A[m, j] = f[m-j] if 0<=m-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M, time_limit_lp=5.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns g or None."""
    n = len(f)
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None
    
    A = build_constraint_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit_lp})
    if result.success:
        return result.x
    return None


def lp_optimize(f_init, time_limit, t_init=0.3, seed=42):
    """LP-based iterative optimization."""
    from scipy.optimize import linprog
    rng = np.random.default_rng(seed)
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = lp_step(f, M, time_limit_lp=min(remaining * 0.3, 10.0))
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new *= sum_f / sum_new
        
        score_new = local_score(f_new)
        
        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            t = min(0.8, t * 1.02)
        else:
            stagnant += 1
            if stagnant > 30:
                t *= 0.7
                stagnant = 0
            if stagnant > 100:
                f = f + 0.01 * rng.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f *= sum_f / np.sum(f)
                t = 0.2
                stagnant = 0
    
    return best_f, best_score


def coordinate_descent(a_init, time_limit, seed=42):
    """Coordinate descent: optimize one element at a time."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S = np.sum(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            # Try different values for a[j]
            current_val = a[j]
            if current_val < 1e-6:
                continue
            
            # Optimal value: we want to minimize max(conv) while keeping sum fixed
            # Try a grid of multipliers
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a_test = a.copy()
                a_test[j] = current_val * mult
                if a_test[j] < 0:
                    continue
                # Renormalize
                test_sum = np.sum(a_test)
                if test_sum < 1e-10:
                    continue
                a_test *= S / test_sum
                score_test = local_score(a_test)
                
                if score_test < best_score:
                    best_score = score_test
                    best_a = a_test.copy()
                    a = a_test
                    improved = True
                    break
        
        if not improved:
            # Try adding mass to zero elements
            zero_idx = np.where(a < 1e-6)[0]
            if len(zero_idx) > 0:
                j = int(rng.choice(zero_idx))
                for mult in [0.1, 0.3, 0.5, 1.0, 2.0]:
                    a_test = a.copy()
                    a_test[j] = S / n * mult
                    test_sum = np.sum(a_test)
                    a_test *= S / test_sum
                    score_test = local_score(a_test)
                    if score_test < best_score:
                        best_score = score_test
                        best_a = a_test.copy()
                        a = a_test
                        improved = True
                        break
            if not improved:
                break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining gradient descent, peak shaving,
    LP refinement, and coordinate descent with multiple initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Get provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided)
        print(f"Provided: n={len(provided)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided.copy()
    
    n_base = len(provided) if provided is not None else 256
    
    # Phase 1: Gradient descent from multiple initializations
    print("\n=== Phase 1: Gradient descent (multi-init) ===")
    phase1_time = min(600.0, (deadline - time.monotonic()) * 0.4)
    
    inits = []
    if provided is not None and n_base == 256:
        inits.append(('provided', provided.copy()))
    
    # Cosine modulations with various frequencies
    for k in [1, 2, 3, 5, 8]:
        alpha = 0.15 + 0.1 * (k % 3)
        phase = rng.uniform(0, 2 * np.pi)
        inits.append((f'cos_k{k}', cosine_init(n_base, k, alpha, phase, rng)))
    
    # Multi-tone
    inits.append(('multitone1', multitone_init(n_base, rng, 5)))
    inits.append(('multitone2', multitone_init(n_base, rng, 8)))
    
    # Uniform (baseline = 2.0, useful as reference)
    inits.append(('uniform', np.ones(n_base)))
    
    # Chirp-like
    t = np.arange(n_base, dtype=np.float64)
    chirp = 1.0 + 0.3 * np.cos(np.pi * t * t / n_base)
    inits.append(('chirp', np.clip(chirp, 0.01, None)))
    
    n_inits = len(inits)
    time_per_init = phase1_time / n_inits
    
    for name, a0 in inits:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(time_per_init, remaining - 50)
        if alloc < 30:
            break
        
        print(f"  Grad descent: {name}, alloc={alloc:.0f}s")
        a_opt, score_opt = softmax_descent(a0, alloc, tau_start=0.3, tau_end=0.005, seed=seed)
        elapsed = time.monotonic() - (deadline - remaining)
        print(f"    -> score={score_opt:.6f}")
        
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} ({name}) ***")
    
    # Phase 2: LP-based refinement on best sequence
    print(f"\n=== Phase 2: LP refinement (score={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    phase2_time = min(300.0, remaining * 0.3)
    
    if phase2_time > 30 and best_seq is not None:
        print(f"  LP optimize, n={len(best_seq)}, alloc={phase2_time:.0f}s")
        a_lp, score_lp = lp_optimize(best_seq, phase2_time, t_init=0.25, seed=seed + 1)
        print(f"  LP result: {score_lp:.6f}")
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"  *** New best: {best_score:.6f} (LP) ***")
    
    # Phase 3: Peak shaving
    print(f"\n=== Phase 3: Peak shaving (score={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    phase3_time = min(200.0, remaining * 0.25)
    
    if phase3_time > 20 and best_seq is not None:
        print(f"  Peak shave, alloc={phase3_time:.0f}s")
        a_ps, score_ps = peak_shave(best_seq, phase3_time, seed=seed + 2)
        print(f"  Peak shave result: {score_ps:.6f}")
        if score_ps < best_score:
            best_score = score_ps
            best_seq = a_ps.copy()
            print(f"  *** New best: {best_score:.6f} (peak shave) ***")
    
    # Phase 4: Gradient polish on best
    print(f"\n=== Phase 4: Fine gradient polish (score={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    phase4_time = min(200.0, remaining * 0.3)
    
    if phase4_time > 20 and best_seq is not None:
        # Use lower temperature for fine polishing
        print(f"  Fine grad, alloc={phase4_time:.0f}s")
        a_fine, score_fine = softmax_descent(best_seq, phase4_time, 
                                              tau_start=0.1, tau_end=0.002, seed=seed + 3)
        print(f"  Fine grad result: {score_fine:.6f}")
        if score_fine < best_score:
            best_score = score_fine
            best_seq = a_fine.copy()
            print(f"  *** New best: {best_score:.6f} (fine grad) ***")
    
    # Phase 5: Coordinate descent
    print(f"\n=== Phase 5: Coordinate descent (score={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    phase5_time = min(150.0, remaining * 0.5)
    
    if phase5_time > 15 and best_seq is not None:
        print(f"  Coord descent, alloc={phase5_time:.0f}s")
        a_cd, score_cd = coordinate_descent(best_seq, phase5_time, seed=seed + 4)
        print(f"  Coord descent result: {score_cd:.6f}")
        if score_cd < best_score:
            best_score = score_cd
            best_seq = a_cd.copy()
            print(f"  *** New best: {best_score:.6f} (coord desc) ***")
    
    # Phase 6: Final peak shave + gradient combo
    print(f"\n=== Phase 6: Final refinement (score={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 30 and best_seq is not None:
        # Quick peak shave
        t_ps = min(60.0, remaining * 0.4)
        if t_ps > 10:
            a_ps2, score_ps2 = peak_shave(best_seq, t_ps, seed=seed + 5)
            if score_ps2 < best_score:
                best_score = score_ps2
                best_seq = a_ps2.copy()
                print(f"  Peak shave 2: {best_score:.6f}")
        
        # Quick gradient
        remaining2 = deadline - time.monotonic()
        t_g = min(60.0, remaining2 * 0.6)
        if t_g > 10:
            a_g2, score_g2 = softmax_descent(best_seq, t_g,
                                             tau_start=0.05, tau_end=0.001, seed=seed + 6)
            if score_g2 < best_score:
                best_score = score_g2
                best_seq = a_g2.copy()
                print(f"  Final grad: {best_score:.6f}")
    
    # Final check
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list
    
    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
