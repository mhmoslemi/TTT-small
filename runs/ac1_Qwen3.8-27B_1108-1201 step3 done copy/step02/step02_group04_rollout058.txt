```python
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def subgradient_optimize(a0, time_limit, eta0=0.1, beta=0.9, patience_limit=2000):
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        g = np.zeros(n)
        g[j_start:j_end+1] = 2.0 * a[k_star - j_end : k_star - j_start + 1][::-1]
        
        # Center gradient (stay on simplex)
        g = g - g.mean()
        
        # Momentum update
        v = beta * v + (1.0 - beta) * g
        a_new = a - eta * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
            eta = min(eta * 1.01, 0.5)
        else:
            patience += 1
            eta *= 0.95
            if patience > patience_limit:
                break
    
    return best_a, best_obj
```

Actually, here's another idea: **water-filling on the convolution**

The convolution values c[k] = (a*a)[k] for k=0,...,2n-2. We want to minimize max c[k]. The "water-filling" idea:

- Compute c = a*a
- The position with the largest c is the "bottleneck"
- Adjust a to reduce c at that position
- The adjustment should be "cheap" (not increase other c values too much)

This is essentially what the LP does, but we can do it more cheaply with a simple gradient step.

Let me also consider: **what if we work in the log-domain?**

Let a_i = exp(b_i) / sum(exp(b_j)). Then the problem becomes minimizing max_k sum_i exp(b_i + b_{k-i}) - log(sum exp(b)). This is a smooth function of b (the max can be smoothed with log-sum-exp).

The log-sum-exp approximation: max_k c[k] ≈ (1/T) * log(sum_k exp(T*c[k])). As T increases, this approaches the true max.

This gives us a smooth optimization problem in b! We can use gradient-based methods (Adam, LBFGS) on b.

This might be more effective than subgradient methods on a directly.

```python
def logspace_optimize(a0, time_limit, T=10.0):
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a /= a.sum()
    n = len(a)
    
    b = np.log(a)  # log space
    
    # Adam optimizer
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.1
    
    best_obj = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a_temp = np.exp(b - b.max())
        a_temp /= a_temp.sum()
        
        conv = fast_convolve(a_temp)
        # Smooth max
        T_eff = min(T * t / 100.0, 500.0)  # anneal T
        smooth_max = np.logsumexp(T_eff * conv) / T_eff
        obj = 2.0 * n * smooth_max
        
        # Gradient w.r.t. b
        # dSmooth_max/db_i = (1/T) * sum_k T * exp(T*c[k]) * dc[k]/db_i / sum_k exp(T*c[k])
        # = sum_k softmax_k * dc[k]/db_i where softmax_k = exp(T*c[k])/sum exp(T*c[k])
        
        w = np.exp(T_eff * conv - T_eff * conv.max())
        w /= w.sum()
        
        # dc[k]/db_i = dc[k]/da_i * da_i/db_i = (2*a[k-i]) * a_i (for valid k-i)
        # This is expensive to compute for all k, but we can use the weighted sum:
        # grad_b[i] = a_i * sum_k w[k] * 2*a[k-i] (for valid k-i)
        # = 2 * a_i * (w*a)[i]  ... wait, that's the cross-convolution of w and a!
        
        # Actually: sum_k w[k] * 2*a[k-i] = 2 * (w * a)[i] where * is convolution
        # But w has length 2n-1 and a has length n, so (w*a) has length 3n-2
        # We only need indices 0 to n-1 of this convolution
        
        # Let me compute (w * a)[i] for i = 0, ..., n-1
        # (w * a)[i] = sum_k w[k] * a[k-i] (where a is zero-padded)
        # = sum_{k=i}^{i+n-1} w[k] * a[k-i]  (since a[j] = 0 for j < 0 or j >= n)
        
        w_ext = np.zeros(3*n-2)  # zero-extend w
        # Actually, let me just use FFT
        fft_size = 1 << (3*n-3).bit_length()
        W = np.fft.rfft(w, fft_size)
        A_ext = np.fft.rfft(np.concatenate([a, np.zeros(n-1)]), fft_size)
        # (w * a_ext)[i] for i = 0,...,n-1
        cross = np.fft.irfft(W * A_ext, fft_size)[:n]
        
        grad_b = 2.0 * a_temp * cross
        # Subtract mean to maintain the normalization (since b is defined up to a constant)
        grad_b -= grad_b.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * grad_b ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        b -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Evaluate actual objective
        a_check = np.exp(b - b.max())
        a_check /= a_check.sum()
        actual_obj = 2.0 * n * fast_convolve(a_check).max()
        
        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a_check.copy()
    
    return best_a, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-space Adam optimization
with log-sum-exp smoothing and annealing. The problem is reformulated in
log-space (a_i = softmax(b_i)) where the objective becomes smooth, enabling
efficient gradient-based optimization. Log-sum-exp annealing gradually
sharpenes the smoothed max towards the true max.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def cross_conv_feat(w, a):
    """Compute (w * a)[0:n] where * is convolution, w has length 2n-1, a has length n."""
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(W * A, fft_size)[:n]


def compute_grad(a, T):
    """Compute gradient of 2n*logsumexp(T*conv(a,a))/T w.r.t. b=log(a)."""
    n = len(a)
    conv = fast_convolve(a)
    Tc = T * conv
    Tc -= Tc.max()
    w = np.exp(Tc)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: use argmax
        k = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[k] = 1.0
    else:
        w /= w_sum
    
    cross = cross_conv_feat(w, a)
    wc = np.dot(w, conv)
    grad = 4.0 * n * a * (cross - wc)
    return grad


def logspace_adam(a0, time_limit, T_init=5.0, T_max=300.0, lr=0.1, beta1=0.9, beta2=0.999):
    """Optimize in log-space using Adam with log-sum-exp annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    n = len(a)
    b = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    
    conv0 = fast_convolve(a)
    best_obj = 2.0 * n * conv0.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        T = min(T_init + (T_max - T_init) * t / max(t, 1), T_max)
        # Actually, let's use exponential annealing
        T = T_init * (1.001) ** t
        T = min(T, T_max)
        
        grad = compute_grad(a, T)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a = np.exp(b - b.max())
        a /= a.sum()
        
        obj = 2.0 * n * fast_convolve(a).max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 200:
                lr *= 0.5
                no_improve = 0
            elif no_improve == 400:
                lr *= 0.5
                no_improve = 0
            elif no_improve > 1000:
                break
    
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, eta0=0.05, beta=0.9):

def make_lp_structure(n):
    """Precompute row/col indices for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=20):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_step(f, time_limit, max_iters=50):

```python
    return 4.0 * n * a * (cross - wc)


def logspace_adam(a0, time_limit, T_init=5.0, T_max=500.0, lr=0.1):
    """Adam optimizer in log-space with log-sum-exp annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    n = len(a)
    b = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    conv0 = fast_convolve(a)
    best_obj = 2.0 * n * conv0.max()
    best_a = a.copy()
    t0 = time.monotonic()
    t = 0
    no_improve = 0
    cur_lr = lr
    while time.monotonic() - t0 < time_limit:
        t += 1
        T = min(T_init * (1.002 ** t), T_max)
        grad = compute_grad_logspace(a, T)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        b = b - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(b - b.max())
        a /= a.sum()
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 300:
                cur_lr *= 0.5
                no_improve = 0
            elif no_improve > 2000:
                break
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, eta0=0.05, beta=0.9):
    """Subgradient descent with momentum on the probability simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    momentum = np.zeros(n)
    eta = eta0
    no_improve = 0
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        k_star = int(np.argmax(conv))
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        g = np.zeros(n)
        g[j_start:j_end + 1] = 2.0 * a[k_star - j_end:k_star - j_start + 1][::-1]
        g -= g.mean()
        momentum = beta * momentum + (1.0 - beta) * g
        a_new = a - eta * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj - 1e-14:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.005, 0.3)
        else:
            no_improve += 1
            eta *= 0.98
            if no_improve > 5000:
                break
    return best_a, best_obj


def make_lp_structure(n):
    """Precompute row/col indices for cross-convolution LP matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def lp_step(f, time_limit, max_iters=30):
    """LP-guided alternating optimization for a few steps."""
    f = np.maximum(f.astype(np.float64), 0.0)
    n = len(f)
    fs = f.sum()
    if fs < 1e-12:
        f = np.ones(n) / n
    else:
        f /= fs
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    rows, cols = make_lp_structure(n)
    t0 = time.monotonic()
    t_blend = 1.0
    for _ in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        vals = np.tile(f, n)
        A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = optimize.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                               method='highs', options={'disp': False, 'time_limit': 15})
        if not res.success:
            break
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-12:
            break
        g /= gs
        if np.max(np.abs(g - f)) < 1e-12:
            break
        improved = False
        for t_try in [t_blend, 0.5 * t_blend, 0.1 * t_blend]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s2 = f_new.sum()
            if s2 < 1e-12:
                continue
            f_new /= s2
            cn = fast_convolve(f_new)
            on = 2.0 * n * cn.max()
            if on < best_obj - 1e-12:
                f = f_new
                M = cn.max()
                best_obj = on
                best_f = f.copy()
                t_blend = min(1.0, t_blend * 1.05)
                improved = True
                break
        if not improved:
            t_blend *= 0.7
            if t_blend < 0.01:
                break
    return best_f, best_obj


def cross_conv_perturb(a, rng):
    """Perturb a by pushing down cross-convolution peaks with other sequences."""
    n = len(a)
    b = np.maximum(a.copy(), 1e-30)
    b /= b.sum()
    for _ in range(50):
        idx = int(rng.integers(n))
        perturb = np.zeros(n)
        sig = 0.05 + 0.1 * rng.random()
        center = rng.random() * n
        perturb = np.exp(-0.5 * ((np.arange(n) - center) / (sig * n)) ** 2)
        if rng.random() > 0.5:
            perturb = perturb[::-1]
        perturb /= perturb.sum()
        cross = np.convolve(a, perturb)[:2 * n - 1]
        peak_pos = int(np.argmax(cross))
        w = peak_pos / max(n - 1, 1)
        a_new = (1.0 - w) * a + w * perturb
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        b = a_new
    return b


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uni'] = np.ones(n) / n
    for sf in [0.1, 0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb[np.minimum(cb, 1e-10)] = 1e-10
    inits['cos'] = cb / cb.sum()
    t1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    t2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['twohump'] = (t1 + t2) / (t1.sum() + t2.sum())
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search: log-space Adam + subgradient + LP + cross-conv perturbation."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    def try_update(a_opt, label):
        nonlocal best_seq, best_score
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"    *** NEW BEST [{label}]: score={score:.6f}")
        else:
            print(f"    [{label}]: score={score:.6f}")
        return score

    # Phase 1: Log-space Adam on diverse configs
    print("Phase 1: Log-space Adam optimization")
    lengths = [512, 1024]
    total_configs = 0
    for n in lengths:
        inits = gen_inits(n, rng)
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 1e-30)
                pp /= pp.sum()
                inits['parent'] = pp
            else:
                xq = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                xq = np.maximum(xq, 1e-30)
                xq /= xq.sum()
                inits['parent'] = xq
            if n == 512 and parent.size >= 512:
                ps = np.maximum(parent[:512], 1e-30)
                ps = (ps + ps[::-1]) / 2.0
                ps /= ps.sum()
                inits['parent_sym'] = ps
        total_configs += len(inits)

    p1_budget = (deadline - time.monotonic()) * 0.55
    t_each = max(8.0, p1_budget / total_configs)

    for n in lengths:
        inits = gen_inits(n, rng)
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 1e-30)
                pp /= pp.sum()
                inits['parent'] = pp
            else:
                xq = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                xq = np.maximum(xq, 1e-30)
                xq /= xq.sum()
                inits['parent'] = xq
            if n == 512 and parent.size >= 512:
                ps = np.maximum(parent[:512], 1e-30)
                ps = (ps + ps[::-1]) / 2.0
                ps /= ps.sum()
                inits['parent_sym'] = ps
        for name, a0 in inits.items():
            if time.monotonic() > deadline - 90:
                break
            ta = min(t_each, deadline - time.monotonic() - 90)
            if ta < 5:
                break
            print(f"  Adam n={n} {name} ({ta:.0f}s)")
            a_opt, _ = logspace_adam(a0, time_limit=ta)
            try_update(a_opt, f"adam n={n} {name}")

    print(f"\nPhase 1 done. Best: {best_score:.6f}")

    # Phase 2: LP refinement on best
    remaining = deadline - time.monotonic() - 30
    if remaining > 30 and best_seq is not None:
        print(f"\nPhase 2: LP refinement ({remaining:.0f}s)")
        n_best = len(best_seq)
        f2, _ = lp_step(best_seq, time_limit=min(remaining * 0.6, 300))
        try_update(f2, "LP phase2")

    # Phase 3: Cross-conv perturbation + re-optimize
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("\nPhase 3: Cross-conv perturbation")
        n_pert = len(best_seq)
        for attempt in range(5):
            if time.monotonic() > deadline - 20:
                break
            pert = cross_conv_perturb(best_seq, rng)
            a_opt, _ = logspace_adam(pert, time_limit=30, T_init=20.0, T_max=500.0, lr=0.08)
            try_update(a_opt, f"perturb {attempt}")

    # Phase 4: Final deep polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print("\nPhase 4: Final polish")
        tr = deadline - time.monotonic() - 10
        for T_i, T_m in [(30.0, 800.0), (50.0, 1000.0), (100.0, 500.0)]:
            if tr < 10:
                break
            a_opt, _ = logspace_adam(best_seq, time_limit=tr / 3.0,
                                     T_init=T_i, T_max=T_m, lr=0.05)
            try_update(a_opt, f"polish T={T_i:.0f}")
            tr = deadline - time.monotonic() - 5

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
```
