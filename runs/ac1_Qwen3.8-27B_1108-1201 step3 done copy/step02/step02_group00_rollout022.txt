```python
def objective_and_grad(x, temperature):
    """
    x: unconstrained parameters, a = softmax(x)
    Returns (objective, gradient) of 2n * smooth_max(conv(a,a), T)
    """
    a = softmax(x)  # a_i = exp(x_i) / sum(exp(x_j))
    n = len(a)
    conv = fast_convolve(a)
    
    # Smooth max
    t = conv / temperature
    t_max = t.max()
    w = np.exp(t - t_max)
    w /= w.sum()
    smooth_max = temperature * (t_max + np.log(w.sum()))
    
    obj = 2.0 * n * smooth_max
    
    # Gradient: d(obj)/dx = d(obj)/da * da/dx
    # d(obj)/da[j] = 2n * (1/T) * sum_k w[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = a[k-j] + (a[k-j] if k==2j) ... 
    # Simplified: = a[k-j] for k != 2j, 2*a[k-j] for k=2j
    # Main term: (1/T) * sum_k w[k]*a[k-j] = (1/T) * crosscorr(w,a)[j]
    
    # Compute crosscorr as described above
    a_rev = a[::-1]
    L = 1 << (3*n - 2).bit_length()
    cross_full = np.fft.irfft(np.fft.rfft(w, L) * np.fft.rfft(a_rev, L), L)
    crosscorr = cross_full[n-1:2*n-1]
    
    # Diagonal correction
    diag_idx = 2*np.arange(n)
    valid = diag_idx < 2*n-1
    grad_a = (2.0*n/temperature) * crosscorr
    grad_a[valid] += (2.0*n/temperature) * w[diag_idx[valid]] * a[valid]
    
    # da/dx: a_i * (delta_ij - a_j) for softmax
    # grad_x[j] = sum_i grad_a[i] * a[i]*(delta_ij - a[j])
    # = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_obj_grad(x, T, n):
    """
    x: (n,) unconstrained parameters
    T: temperature for softmax
    n: sequence length
    Returns (objective, gradient)
    """
    # Softmax parameterization
    xm = x - x.max()
    ex = np.exp(xm)
    a = ex / ex.sum()
    
    # Auto-convolution
    conv = fast_convolve(a)
    
    # Smooth max
    tc = conv / T
    tc_max = tc.max()
    ec = np.exp(tc - tc_max)
    ec_sum = ec.sum()
    w = ec / ec_sum
    smax = T * (tc_max + np.log(ec_sum))
    
    obj = 2.0 * n * smax
    
    # Gradient
    a_rev = a[::-1]
    L = 1 << (3*n - 2).bit_length()
    cross_full = np.fft.irfft(np.fft.rfft(w, L) * np.fft.rfft(a_rev, L), L)
    crosscorr = cross_full[n-1:2*n-1]
    
    grad_a = crosscorr / T
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < 2*n - 1
    grad_a[valid] += w[diag_idx[valid]] * a[valid] / T
    
    grad_a *= 2.0 * n
    
    # Jacobian of softmax
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

def anneal_lBFGS(x0, n, T_start, T_end, time_limit, verbose=False):
    """
    Temperature-annealed L-BFGS-B optimization.
    x0: initial unconstrained parameters
    n: sequence length
    T_start, T_end: temperature range
    """
    T = T_start
    x = x0.copy()
    best_obj = float('inf')
    best_x = x0.copy()
    
    # Number of temperature levels
    n_levels = 15
    T_ratios = np.geomspace(T_start, T_end, n_levels)
    
    t0 = time.monotonic()
    
    for i, T in enumerate(T_ratios):
        if time.monotonic() - t0 > time_limit:

            args=(T, n),

        a = np.exp(x - x.max())

        conv = fast_convolve(a)
        true_obj = 2.0 * n * conv.max()

            best_x = x.copy()
        
        if verbose:
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using temperature-annealed L-BFGS-B
on a smooth softmax approximation of the max, with log-sum-exp parameterization.
Key innovations: (1) smooth objective via softmax of convolution peaks, (2) 
constrained-to-unconstrained via exp parameterization, (3) multi-length search,
(4) final polish with projected subgradient on the non-smooth objective.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(x, T, n):
    """
    Smooth approximation of 2n*max(conv(a,a)) using softmax parameterization.
    x: unconstrained parameters (length n)
    T: softmax temperature
    Returns (objective, gradient).
    """
    # Softmax: a_i = exp(x_i) / sum(exp(x_j))
    xm = x - x.max()
    ex = np.exp(xm)
    esum = ex.sum()
    a = ex / esum

    # Auto-convolution
    conv = fast_convolve(a)

    # Softmax of conv for smooth max
    tc = conv / T
    tc_max = tc.max()
    ec = np.exp(tc - tc_max)
    ec_sum = ec.sum()
    w = ec / ec_sum
    smax = T * (tc_max + np.log(ec_sum))

    obj = 2.0 * n * smax

    # Gradient w.r.t. a: grad_a[j] = (2n/T) * sum_k w[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = a[k-j] + a[j]*[k==2j]
    # Main term is cross-correlation of w and a
    a_rev = a[::-1]
    L = 1 << (3 * n - 2).bit_length()
    cross_full = np.fft.irfft(np.fft.rfft(w, L) * np.fft.rfft(a_rev, L), L)
    crosscorr = cross_full[n - 1:2 * n - 1]

    grad_a = (2.0 * n / T) * crosscorr
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < 2 * n - 1
    grad_a[valid] += (2.0 * n / T) * w[diag_idx[valid]] * a[valid]

    # Chain rule through softmax: grad_x[j] = a[j]*(grad_a[j] - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def true_objective(a):
    """Compute the true (non-smooth) objective."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def anneal_lbfgsb(x0, n, T_start, T_end, time_limit, maxiter_per_level=300):
    """
    Temperature-annealed L-BFGS-B optimization.
    """
    n_levels = 20
    T_values = np.geomspace(T_start, T_end, n_levels)
    x = x0.copy()
    best_x = x0.copy()
    best_true_obj = true_objective(x0_n = _x_to_a(x0))

    t0 = time.monotonic()

    for i, T in enumerate(T_values):
        if time.monotonic() - t0 > time_limit:
            break

        try:
            res = minimize(
                smooth_obj_grad, x,
                args=(T, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_level, 'gtol': 1e-10,
                         'maxfun': maxiter_per_level * 2, 'ftol': 1e-12}
            )
            x_new = res.x
        except Exception:
            x_new = x

        a_new = _x_to_a(x_new)
        true_obj = true_objective(a_new)

        if np.isfinite(true_obj) and true_obj < best_true_obj:
            best_true_obj = true_obj
            best_x = x_new.copy()
        else:
            x = x  # keep old if new is worse

        # Always move to new x for next temperature level (even if slightly worse in true obj)
        x = x_new

    return best_x, best_true_obj


def _x_to_a(x):
    """Convert unconstrained parameters to probability vector."""
    xm = x - x.max()
    ex = np.exp(xm)
    return ex / ex.sum()


def subgradient_polish(a0, time_limit, top_k=50, lr_init=0.02, momentum=0.9):
    """
    Polished subgradient descent on the simplex for the non-smooth objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = true_objective(a)
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        # Softmax weights over peaks
        scaled = peak_vals / (peak_vals.max() * 0.01 + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_arr = np.arange(j_start, j_end + 1)
            sg[idx_arr] += wi * a[k - idx_arr]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = true_objective(a)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            # Perturbation restart
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.002, lr * 0.5)

        lr *= 0.9998

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_param=80.0, eta_init=0.3):
    """
    Multiplicative weights / mirror descent for the non-smooth objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = true_objective(a)
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    t_cur = t_param
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Softmax weights over convolution
        scaled = t_cur * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Subgradient
        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_arr = np.arange(j_start, j_end + 1)
            sg[idx_arr] += w[k_int] * 2.0 * a[k - idx_arr]

        sg_centered = sg - sg.mean()
        log_update = -eta * sg_centered
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

        conv = fast_convolve(a)
        cur_obj = true_objective(a)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 150:
            no_improve = 0
            eta *= 0.7
            t_cur *= 1.5
            if eta < 0.01:
                eta = 0.25
                t_cur = t_param

    return best_a, best_obj

            if remaining < 5:
                break
            noise = rng.standard_normal(n_best) * 0.001
            a_pert = project_simplex(best_sequence + noise)
            x0 = a_to_x(a_pert)
            conv_c = fast_convolve(a_pert)
            mc = conv_c.max()
            t_attempt = min(remaining / 2, 30.0)
            x_best, obj_best = anneal_lbfgsb(x0, n_best, mc * 0.1, mc * 0.005, t_attempt, maxiter_per_level=100)
            a_best = _x_to_a(x_best)
            score = float(evaluate_sequence(a_best.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_best.copy()
                print(f"  Perturb+L-BFGS: score={score:.6f}")
            remaining = deadline - time.monotonic() - 10.0

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
