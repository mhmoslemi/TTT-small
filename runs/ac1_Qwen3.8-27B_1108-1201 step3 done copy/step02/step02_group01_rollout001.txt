from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def normalize_seq(a, target_sum=1.0):
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s

def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def softmax_gradient(a, beta, lr, sum_target=1.0):
    """
    One projected gradient step minimizing log-sum-exp(beta * conv(a,a)).
    """
    n = len(a)
    conv = fft_convolve(a, a)
    
    # Soft max weights (numerically stable)
    conv_shifted = conv - np.max(conv)
    weights = np.exp(beta * conv_shifted)
    weights /= np.sum(weights)
    
    # Gradient: grad[j] = sum_k weights[k] * (a[k-j] + a[j-k])
    # = cross_correlation(weights, a)[j] + convolution(weights, a)[j]

"""
Multi-phase optimization for minimizing the normalized peak convolution
of non-negative step functions. Uses soft-max gradient descent with FFT-based
gradient computation, LP-based refinement, and adaptive length search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def normalize_seq(a, target_sum=1.0):
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s

def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))

def softmax_gradient_step(a, beta, lr, target_sum=1.0):
    n = len(a)
    conv = fft_convolve(a, a)
    
    # Soft max weights (numerically stable)
    conv_shifted = conv - np.max(conv)
    weights = np.exp(beta * conv_shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum
    
    # Compute gradient using FFT
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
    # This is cross_correlation(w, a) + convolution(w, a)
    
    size = 1
    while size < max(2 * len(weights), 2 * n) + 1:
        size <<= 1
    
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    
    # Cross-correlation w ⊛ a: (w⊛a)[j] = sum_k w[k] * a[k-j]
    # = ifft(conj(FFT(w)) * FFT(a))
    corr = np.fft.irfft(np.conj(W) * A, size)[:n]
    
    # Convolution w * a: (w*a)[j] = sum_k w[k] * a[j-k]
    # = ifft(FFT(w) * FFT(a))
    conv_wa = np.fft.irfft(W * A, size)[:n]
    
    grad = corr + conv_wa
    
    # Gradient step with projection
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, target_sum)
    
    return a_new, np.max(conv)

def build_sparse_lp_matrix(a):
    """Build sparse LP constraint matrix where row k has a[k-j] at column j."""
    n = len(a)

    cols = []
    data = []

            cols.append(j)
            data.append(a[k - j])
