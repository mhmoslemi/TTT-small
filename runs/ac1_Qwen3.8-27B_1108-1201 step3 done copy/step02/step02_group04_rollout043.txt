    # Make symmetric

    a = np.maximum(a, 0)
    if a.sum() > 0:

    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

4. Multi-length exploration (256-1024)

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def flat_conv_init(n, symmetric=True):
    """Initialize via flat-convolution recurrence."""
    c = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(c)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k - i]
        a[k] = (c - s) / (2.0 * a[0])
        a[k] = max(a[k], 0.0)
    if symmetric:
        a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def make_lp_structure(n):
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    # Wait, I need the cross-convolution matrix.
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid k, i
    # For fixed k: sum over i where 0<=i<n and 0<=k-i<n
    # A[k, i] = f[i] if 0 <= k-i < n, else 0
    # Column i: rows where 0 <= k-i < n, i.e., i <= k < i+n, i.e., k in [i, i+n-1]
    # So column i has non-zeros at rows i, i+1, ..., i+n-1
    # The value at row k, column i is f[i]
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, use_symmetry=False, adaptive_M=False):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    if use_symmetry:
        f = 0.5 * (f + f[::-1])
        f = np.maximum(f, 0)
        f /= f.sum()

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Adaptive M: try slightly tighter bound
        M_try = M
        if adaptive_M and iter_count > 10:
            M_try = M * 0.995  # 0.5% tighter
        
        g = solve_lp(f, M_try, rows, cols, n)
        if g is None and adaptive_M:
            g = solve_lp(f, M, rows, cols, n)
            M_try = M
        
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if use_symmetry:
            g = 0.5 * (g + g[::-1])
            g = np.maximum(g, 0)
            g /= g.sum()

        if np.max(np.abs(g - f)) < 1e-14:
            break

        # Backtracking line search
        improved = False
        for t_try in [t, 0.8*t, 0.6*t, 0.4*t, 0.2*t, 0.1*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            if use_symmetry:
                f_new = 0.5 * (f_new + f_new[::-1])
                f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            obj_new = 2.0 * n * M_new
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = M_new
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.8
            if patience > 100:
                break

    return best_f, best_obj


def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3, use_symmetry=False):
    """Mirror descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    if use_symmetry:
        f = 0.5 * (f + f[::-1])
        f = np.maximum(f, 0)
        f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Softmax weights
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            break
        w /= w_sum

        # Compute gradient (cross-correlation of w with f)
        sg = np.zeros(n)
        # Only process significant weights for speed
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        # Center the gradient
        sg_c = sg - sg.mean()
        
        # Multiplicative update
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        if use_symmetry:
            f = 0.5 * (f + f[::-1])
            f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.75
                if eta < 0.01:
                    t_param = min(t_param * 1.3, 300.0)
                    eta = 0.25
            if patience > 300:
                break

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Flat-convolution recursive initialization (key new idea - solves for the
   sequence whose auto-convolution is closest to flat via a recurrence relation)
2. LP-guided alternating optimization with adaptive M (tighter constraints
   force more aggressive exploration)
3. Mirror descent polishing with adaptive temperature
4. Multi-length exploration (256-1024)
5. Symmetry exploitation (palindrome constraint)
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def flat_conv_init(n, symmetric=True):
    """Initialize via the flat-convolution recurrence.
    Solves conv[k] = c for all k, giving a[k] = (c - sum_{i=1}^{k-1} a[i]*a[k-i]) / (2*a[0]).
    This produces a sequence whose auto-convolution is as flat as possible,
    starting large at edges and decaying toward the middle."""
    c = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(c)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k - i]
        a[k] = (c - s) / (2.0 * a[0])
        if a[k] < 0:
            a[k] = 0.0
    if symmetric:
        a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def make_lp_structure(n):
    """Precompute sparse structure for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, use_symmetry=False, adaptive_M=False, tight_factor=0.995):
    """LP-based alternating optimization with optional adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    if use_symmetry:
        f = 0.5 * (f + f[::-1])
        f = np.maximum(f, 0)
        f /= f.sum()

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        M_try = M
        if adaptive_M and iter_count > 5:
            M_try = M * tight_factor

        g = solve_lp(f, M_try, rows, cols, n)
        if g is None and adaptive_M:
            g = solve_lp(f, M, rows, cols, n)

        if g is None:
            patience += 5
            if patience > 200:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if use_symmetry:
            g = 0.5 * (g + g[::-1])
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs > 0:
                g /= gs

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.8 * t, 0.6 * t, 0.4 * t, 0.2 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            if use_symmetry:
                f_new = 0.5 * (f_new + f_new[::-1])
                f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            obj_new = 2.0 * n * M_new
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = M_new
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.85
            if patience > 150:
                break

    return best_f, best_obj


def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3, use_symmetry=False):
    """Mirror descent (multiplicative weights) polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    if use_symmetry:
        f = 0.5 * (f + f[::-1])
        f = np.maximum(f, 0)
        f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            break
        w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        if use_symmetry:
            f = 0.5 * (f + f[::-1])
            f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.75
                if eta < 0.01:
                    t_param = min(t_param * 1.4, 400.0)
                    eta = 0.25
            if patience > 300:
                break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations prioritizing flat-convolution structure."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Key: flat convolution recurrence (symmetric and asymmetric)
    inits['flat_sym'] = flat_conv_init(n, symmetric=True)
    inits['flat_asym'] = flat_conv_init(n, symmetric=False)

    # Blended flat + uniform
    flat = inits['flat_sym']
    uni = np.ones(n) / n
    for alpha in [0.7, 0.85]:
        blk = alpha * flat + (1 - alpha) * uni
        blk = np.maximum(blk, 0)
        blk /= blk.sum()
        inits[f'flat_blended_{int(alpha*100)}'] = blk

    # U-shape / bathtub
    u = 1.0 / (0.05 + 4.0 * (x / n - 0.5) ** 2)
    u /= u.sum()
    inits['bathtub'] = u

    # Edge-concentrated
    edge = np.zeros(n)
    w = max(1, n // 6)
    edge[:w] = 1.0
    edge[-w:] = 1.0
    edge /= edge.sum()
    inits['edge'] = edge

    # Gaussian (various widths)
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g

    # Cosine bump
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb = np.maximum(cb, 0.01)
    cb /= cb.sum()
    inits['cos'] = cb

    # Random
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5)

    return inits

        t_each = remaining / len(top)
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 30:
                break
            ta = min(t_each * 1.2, deadline - time.monotonic() - 30)
            if ta < 8:
                break

            sym = name.endswith('_sym')

            # Longer LP with slightly different tight factor
            f2, _ = lp_optimize(f, time_limit=ta * 0.5, use_symmetry=sym,
                                adaptive_M=True, tight_factor=0.998)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                print(f"  Deep LP [{idx}] n={n} {name} score={sc2:.6f}")

            # Mirror polish with multiple temperatures
            if time.monotonic() < deadline - 15:
                tr = min(ta * 0.35, deadline - time.monotonic() - 15)
                for tp in [40.0, 100.0]:
                    if tr < 3:
                        break
                    f3, _ = mirror_polish(f2, time_limit=tr, t_param=tp, use_symmetry=sym)
                    sc3 = float(evaluate_sequence(f3.tolist()))
                    if np.isfinite(sc3) and sc3 < best_score:
                        best_score = sc3
                        best_seq = f3.copy()
                        print(f"  Polish [{idx}] tp={tp:.0f} score={sc3:.6f}")
                    tr = deadline - time.monotonic() - 15

    # Phase 3: Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 5:
        print("Phase 3: Final polish")
        tr_total = deadline - time.monotonic() - 5
        configs = [(60.0, 0.2), (120.0, 0.12), (200.0, 0.06), (300.0, 0.03)]
        n_cfgs = len(configs)
        tr_each = tr_total / n_cfgs
        for tp, eta in configs:
            if tr_each < 2:
                break
            f5, _ = mirror_polish(best_seq, time_limit=tr_each, t_param=tp, eta0=eta)
            sc5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_seq = f5.copy()
                print(f"  Final tp={tp:.0f} eta={eta} score={sc5:.6f}")
            tr_each = (deadline - time.monotonic() - 5) / max(1, n_cfgs - 1) if n_cfgs > 1 else 0

    # Phase 4: Quick restart with perturbation
    if best_seq is not None and time.monotonic() < deadline - 3:
        tr = deadline - time.monotonic() - 3
        if tr > 10:
            n = len(best_seq)
            # Try a perturbed version
            noise = rng.normal(0, 0.01, n)
            perturbed = np.maximum(best_seq + noise * best_seq.mean(), 0)
            perturbed /= perturbed.sum()
            f6, _ = lp_optimize(perturbed, time_limit=min(tr, 30), adaptive_M=True)
            sc6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(sc6) and sc6 < best_score:
                best_score = sc6
                best_seq = f6.copy()
                print(f"  Perturb score={sc6:.6f}")

    if best_seq is None:
        best_seq = flat_conv_init(512, symmetric=True)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
