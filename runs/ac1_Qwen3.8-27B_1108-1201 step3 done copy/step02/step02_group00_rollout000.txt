        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(f, n):
    """Build matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0.
    Then A @ g gives conv(f, g)."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[k, lo:hi] = f[k - (lo:hi)]
    return A


def lp_direction(f, n, reg=1e-7, solver_pref=None):
    """Solve LP: min t + reg*||g||^2 s.t. conv(f,g) <= t, sum(g)=1, g>=0."""
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()

    A = build_conv_matrix(f, n)
    constraints = [cp.sum(g) == 1, A @ g <= t]
    prob = cp.Problem(cp.Minimize(t + reg * cp.sum_squares(g)), constraints)

    if solver_pref is not None:
        try:
            prob.solve(solver=solver_pref)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            pass

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue

    return f.copy()


def golden_search(f, g, n, max_iter=35):
    """Golden section search for alpha in [0,1] minimizing max(conv(h,h))."""
    cache = {}

    def f_obj(alpha):
        key = round(alpha, 8)
        if key in cache:
            return cache[key]
        h = (1 - alpha) * f + alpha * g
        conv = fast_convolve(h)
        val = conv.max()
        cache[key] = val
        return val

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_obj(c)
    fd = f_obj(d)

    for _ in range(max_iter):
        if b - a < 1e-7:
            break
        if fc < fd:
            b, d = d, c
            fd = fc
            c = b - gr * (b - a)
            fc = f_obj(c)
        else:
            a, c = c, d
            fc = fd
            d = a + gr * (b - a)
            fd = f_obj(d)

    alpha = (a + b) / 2.0
    # Check endpoints
    f0 = f_obj(0.0)
    f1 = f_obj(1.0)
    fm = f_obj(alpha)
    if f0 <= fm and f0 <= f1:
        alpha = 0.0
    elif f1 <= fm and f1 <= f0:
        alpha = 1.0
    return alpha


def lp_optimize(f0, time_limit, reg=1e-7, restart_every=80, noise_scale=0.005,
                solver_pref=None, max_ls=30):
    """Run LP-based optimization loop. Returns (best_f, best_obj, iters)."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_obj = objective_value(f)

    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = lp_direction(f, n, reg=reg, solver_pref=solver_pref)
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            g = f.copy()
        else:
            g = g / gs

        alpha = golden_search(f, g, n, max_ls)

        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        obj = objective_value(f_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        iters += 1

        if no_improve >= restart_every:
            # Perturb to escape local minimum
            scale = noise_scale * (1.0 + no_improve / restart_every)
            noise = rng.standard_normal(n) * scale
            f = np.maximum(f + noise, 0.0)
            s = f.sum()
            f = f / s if s > 1e-15 else np.ones(n) / n
            no_improve = 0
            reg = max(1e-9, reg * 0.5)

    return best_f, best_obj, iters


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_polish(f0, time_limit, lr=0.01, momentum=0.9, top_k=20, temp=0.0002):
    """Polish with projected subgradient on the simplex."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_f = f.copy()
    best_obj = objective_value(f)

    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * f[k - idx]

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        f = f_new
        conv = fast_convolve(f)
        obj = objective_value(f)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 2000:
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(f + noise)
            v *= 0.3
            no_improve = 0
            lr = max(0.002, lr * 0.7)

        lr *= 0.9999
        temp *= 1.003

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Gaussian
    for sf in [0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random (Dirichlet with low concentration -> sparse-ish)
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Sinusoidal
    for freq in [2, 4, 8]:
        sin = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        sin = np.maximum(sin, 0.1)
        sin /= sin.sum()
        inits[f'sin_{freq}'] = sin

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase LP-based search with subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    results = []

    # Phase 1: LP optimization from diverse starts at multiple lengths
    print("=== Phase 1: Multi-start LP Optimization ===")
    lengths_to_try = [128, 192, 256, 320, 384, 512]

    # Determine time per (length, init) pair
    n_pairs = sum(len(generate_inits(n, parent, rng)) for n in lengths_to_try)
    phase1_budget = min(600.0, (deadline - time.monotonic()) * 0.35)
    time_per_pair = max(3.0, phase1_budget / max(1, n_pairs))
    print(f"Phase 1 budget: {phase1_budget:.0f}s, ~{time_per_pair:.1f}s per pair, {n_pairs} pairs")

    phase1_t0 = time.monotonic()
    for n in lengths_to_try:
        if time.monotonic() >= deadline - 120:
            break
        inits = generate_inits(n, parent, rng)
        # Prioritize parent, then uniform, then random
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'random1',
                    'bimodal', 'triangle', 'gauss_0.2', 'sin_4', 'random2']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]

        for name in init_names:
            if time.monotonic() >= deadline - 100:
                break
            t_alloc = min(time_per_pair, deadline - time.monotonic() - 80)
            if t_alloc < 2:
                break

            f0 = inits[name]
            f_opt, obj_opt, iters = lp_optimize(
                f0, time_limit=t_alloc, reg=1e-7, restart_every=60,
                noise_scale=0.005, max_ls=25)

            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, f_opt.copy(), obj_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = f_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f} (iters={iters})")

    print(f"Phase 1 done. Best so far: {best_score:.6f}")

    # Phase 2: Focus on best configurations with longer LP runs
    print("=== Phase 2: Deep LP Optimization ===")
    results.sort(key=lambda x: x[0])
    top = results[:6] if len(results) >= 6 else results

    # Add best sequence explicitly
    if best_sequence is not None:
        bn = len(best_sequence)
        top.append((best_score, bn, 'best', best_sequence.copy(), best_score))

    # Remove duplicates (same length, similar sequence)
    seen_lengths = {}
    top_filtered = []
    for score, n, name, f, obj in top:
        if n not in seen_lengths or score < seen_lengths[n][0]:
            seen_lengths[n] = (score, name, f)
            top_filtered.append((score, n, name, f, obj))
        if len(top_filtered) >= 8:
            break

    remaining = deadline - time.monotonic() - 60.0
    time_per_deep = max(20.0, remaining / max(1, len(top_filtered) * 2))

    for score, n, name, f0, obj in top_filtered:
        if time.monotonic() > deadline - 50:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 40)
        if t_alloc < 10:
            break

        # Run LP with different regularization
        for reg in [1e-8, 1e-6]:
            if time.monotonic() > deadline - 35:
                break
            t_single = t_alloc / 2.0
            if t_single < 5:
                break
            f_opt, obj_opt, iters = lp_optimize(
                f0, time_limit=t_single, reg=reg, restart_every=100,
                noise_scale=0.003, max_ls=35)

            f_opt2, obj_opt2 = subgradient_polish(
                f_opt, time_limit=min(5.0, t_single * 0.3),
                lr=0.005, momentum=0.92, top_k=n // 15, temp=0.0001)

            for cand in [f_opt, f_opt2]:
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_sequence = cand.copy()
                    print(f"  Deep n={n}, {name}, reg={reg}: score={sc:.6f}")

    print(f"Phase 2 done. Best so far: {best_score:.6f}")

    # Phase 3: Intensive polishing of the best solution
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        n_best = len(best_sequence)

        # Multiple subgradient passes with different params
        polish_configs = [
            {'lr': 0.005, 'momentum': 0.93, 'top_k': max(15, n_best // 10), 'temp': 0.0001},
            {'lr': 0.003, 'momentum': 0.95, 'top_k': max(30, n_best // 5), 'temp': 0.00005},
            {'lr': 0.01, 'momentum': 0.88, 'top_k': max(20, n_best // 8), 'temp': 0.0003},
        ]

        t_per = remaining / (len(polish_configs) + 3)

        for cfg in polish_configs:
            if remaining < 5:
                break
            f_opt, obj_opt = subgradient_polish(
                best_sequence, time_limit=min(t_per, 60.0), **cfg)
            sc = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = f_opt.copy()
                print(f"  SG polish: score={sc:.6f}")
            remaining = deadline - time.monotonic() - 10.0

        # Short LP polish
        if remaining > 10:
            f_opt, obj_opt, _ = lp_optimize(
                best_sequence, time_limit=min(remaining / 3, 120.0),
                reg=1e-8, restart_every=150, noise_scale=0.002, max_ls=35)
            sc = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = f_opt.copy()
                print(f"  LP polish: score={sc:.6f}")

        # Perturbation + re-optimize
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 15:
            for trial in range(5):
                if remaining < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                f_pert = np.maximum(best_sequence + noise, 0.0)
                f_pert = f_pert / f_pert.sum()
                f_opt, obj_opt, _ = lp_optimize(
                    f_pert, time_limit=min(remaining / (5 - trial), 30.0),
                    reg=1e-7, restart_every=50, noise_scale=0.004, max_ls=25)
                sc = float(evaluate_sequence(f_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_sequence = f_opt.copy()
                    print(f"  Perturb+LP: score={sc:.6f}")
                remaining = deadline - time.monotonic() - 10.0

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
