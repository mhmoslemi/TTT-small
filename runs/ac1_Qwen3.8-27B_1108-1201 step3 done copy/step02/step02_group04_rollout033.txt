def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def fast_cross_convolve(f, g):
    """Cross-convolution of f and g using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    result = np.fft.irfft(F * G, fft_size)
    return result[:size]

def make_lp_structure(n):

            cols_list.append(i)

def solve_lp(f, M, rows, cols, n, timeout=10):

    b = np.full(2 * n - 1, M)

                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def golden_section_step(f, d, conv_f, conv_fd, conv_d, eps_max, max_iter=80):
    """Find optimal step size using golden section search."""
    if eps_max < 1e-12:
        return 0.0

        h = conv_f + 2.0 * eps * conv_fd + eps * eps * conv_d
        return h.max()

    a, b = 0.0, eps_max
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = max_at(c_pt)
    fd = max_at(d_pt)

    for _ in range(max_iter):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = max_at(c_pt)

            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = max_at(d_pt)
        if abs(b - a) < 1e-12 * max(1.0, eps_max):

            pp = np.maximum(pp, 0.0)
        if pp.sum() > 1e-10:
            inits['par'] = pp / pp.sum()

    return inits

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: hybrid LP peak descent + accelerated gradient + mirror descent."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    all_results = []
    lengths = [256, 512]

    # Phase 1: LP peak descent (main optimization)
    print("Phase 1: LP peak descent")
    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        inits = gen_inits(n, rng, parent)

        remaining = deadline - time.monotonic() - 180
        time_per = max(15.0, remaining / max(1, len(inits) * len(lengths)))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 150:
                break
            ta = min(time_per, deadline - time.monotonic() - 150)
            if ta < 10:
                break

            t_start = time.monotonic()
            f_opt, obj_opt = lp_peak_descent(a0, time_limit=ta)
            score = float(evaluate_sequence(f_opt.tolist()))

            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    elapsed = time.monotonic() - t_start
                    print(f"  n={n:4d} {name:5s} score={score:.6f} ({elapsed:.0f}s)")

    print(f"Phase 1 done. Best score: {best_score:.6f}")

    # Phase 2: Polish top results with accelerated gradient + mirror descent
    all_results.sort(key=lambda x: x[0])
    top = all_results[:8]

    remaining = deadline - time.monotonic() - 60
    if remaining > 30 and top:
        print(f"Phase 2: Polish top {len(top)}, {remaining:.0f}s remaining")
        t_each = remaining / len(top)
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 30:
                break
            ta = min(t_each, deadline - time.monotonic() - 30)
            if ta < 5:
                break

            # Accelerated gradient
            f1, _ = accelerated_gradient_polish(f, time_limit=ta * 0.35)
            sc1 = float(evaluate_sequence(f1.tolist()))
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_seq = f1.copy()
                print(f"  AG [{idx}] n={n} {name} score={sc1:.6f}")

            # Mirror descent with different params
            if time.monotonic() < deadline - 15:
                for tp, eta in [(80.0, 0.2), (200.0, 0.1), (500.0, 0.05)]:
                    if time.monotonic() > deadline - 10:
                        break
                    tr = min(ta * 0.2, deadline - time.monotonic() - 10)
                    if tr < 2:
                        break
                    f2, _ = mirror_descent_polish(f1, time_limit=tr, t_param=tp, eta0=eta)
                    sc2 = float(evaluate_sequence(f2.tolist()))
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = f2.copy()
                        f1 = f2
                        print(f"  MD [{idx}] n={n} tp={tp:.0f} score={sc2:.6f}")

    # Phase 3: Final LP on the best + one more polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("Phase 3: Final optimization")
        tr = deadline - time.monotonic() - 20
        f3, _ = lp_peak_descent(best_seq, time_limit=tr * 0.4, delta=0.08)
        sc3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_seq = f3.copy()
            print(f"  Final LP score={sc3:.6f}")

        if time.monotonic() < deadline - 10:
            f4, _ = mirror_descent_polish(best_seq, time_limit=min(20, deadline - time.monotonic() - 10),
                                          t_param=300.0, eta0=0.08)
            sc4 = float(evaluate_sequence(f4.tolist()))
            if np.isfinite(sc4) and sc4 < best_score:
                best_score = sc4
                best_seq = f4.copy()
                print(f"  Final MD score={sc4:.6f}")

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
