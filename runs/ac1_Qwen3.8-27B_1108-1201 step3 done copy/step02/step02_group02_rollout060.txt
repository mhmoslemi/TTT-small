"""
Smooth gradient-based optimization for minimizing peak autocorrelation
of non-negative step functions. Uses log-sum-exp smoothing with
progressively increasing sharpness, L-BFGS in log-space, and
coordinate descent refinement on the true objective.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local evaluation of the objective."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smooth_obj_and_grad(z, beta, n):
    """
    Compute the smooth objective and its gradient w.r.t. z (log-space).
    Objective: (2n/beta) * logsumexp(beta * c) / (sum a)^2
    where a = exp(z), c = convolve(a, a)
    """
    a = np.exp(z)
    c = fft_convolve(a, a)
    
    # Numerically stable logsumexp
    mc = np.max(c)
    exp_vals = np.exp(beta * (c - mc))
    lse = mc + np.log(np.sum(exp_vals))
    
    # Softmax probabilities
    p = exp_vals / np.sum(exp_vals)
    
    # Correlation G[i] = sum_m a[m] * p[m+i]
    G = np.correlate(p, a, mode='valid')
    assert len(G) == n, f"G length {len(G)} != n={n}"
    
    s = np.sum(a)
    
    # Objective value
    obj = (2.0 * n / beta) * lse / (s ** 2)
    
    # Gradient: d(obj)/d(a[i]) = 4n*G[i]/s^2 - 4n*lse/(beta*s^3)
    d_obj_da = (4.0 * n * G) / (s ** 2) - (4.0 * n * lse) / (beta * s ** 3)
    
    # Chain rule: d(obj)/d(z[i]) = d_obj_da[i] * a[i]
    grad = d_obj_da * a
    
    return obj, grad


def run_lbfgs(z0, beta, n, max_iter=500, time_limit=60.0):
    """Run L-BFGS on the smooth objective."""
    t0 = time.monotonic()
    
    # For scipy, we need separate functions
    # But we can't use closures per the rules, so we use a class or pass args
    
    best_z = z0.copy()
    best_val = smooth_obj_and_grad(z0, beta, n)[0]
    
    result = minimize(
        lambda x: smooth_obj_and_grad(x, beta, n)[0],
        z0,
        jac=lambda x: smooth_obj_and_grad(x, beta, n)[1],
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    if result.fun < best_val:
        best_val = result.fun
        best_z = result.x.copy()
    
    return best_z, best_val

"""
Smooth gradient-based optimization for minimizing peak autocorrelation
of non-negative step functions. Uses log-sum-exp smoothing with
progressively increasing sharpness, L-BFGS in log-space, fast coordinate
descent on the true objective, and simulated annealing polish.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global params for scipy minimize (to avoid lambdas/closures)
_gp = {}


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local evaluation of the objective (no validation)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smooth_obj_and_grad(z, beta, n):
    """
    Compute the smooth objective and its gradient w.r.t. z (log-space).
    Objective: (2n/beta) * logsumexp(beta * c) / (sum a)^2
    where a = exp(z), c = convolve(a, a)
    """
    a = np.exp(z)
    c = fft_convolve(a, a)

    # Numerically stable logsumexp
    mc = np.max(c)
    exp_vals = np.exp(beta * (c - mc))
    se = np.sum(exp_vals)
    lse = mc + np.log(se)

    # Softmax probabilities
    p = exp_vals / se

    # Correlation G[i] = sum_m a[m] * p[m+i]
    G = np.correlate(p, a, mode='valid')

    s = np.sum(a)

    # Objective value
    obj = (2.0 * n / beta) * lse / (s ** 2)

    # Gradient: d(obj)/d(a[i]) = 4n*G[i]/s^2 - 4n*lse/(beta*s^3)
    d_obj_da = (4.0 * n * G) / (s ** 2) - (4.0 * n * lse) / (beta * s ** 3)

    # Chain rule: d(obj)/d(z[i]) = d_obj_da[i] * a[i]
    grad = d_obj_da * a

    return obj, grad


def _smooth_fun(x):
    return smooth_obj_and_grad(x, _gp['beta'], _gp['n'])[0]


def _smooth_jac(x):
    return smooth_obj_and_grad(x, _gp['beta'], _gp['n'])[1]


def run_lbfgs(z0, beta, n, maxiter=300):
    """Run L-BFGS on the smooth objective in log-space."""
    _gp['beta'] = beta
    _gp['n'] = n

    result = minimize(
        _smooth_fun,
        z0,
        jac=_smooth_jac,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
    )

    return result.x, result.fun


def coordinate_descent(a, n_passes=20):
    """
    Fast coordinate descent on the true (non-smooth) objective.
    For each coordinate, uses golden section search with O(n) evaluations.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    current_score = local_score(a)
    c_full = fft_convolve(a, a)

    for _pass in range(n_passes):
        improved = False
        for i in range(n):
            x_orig = a[i]
            S = np.sum(a) - x_orig

            # Remove a[i]'s contribution from convolution
            # c[k] without a[i] = c[k] - contribution
            # Contribution to c[k]: 2*a[i]*a[k-i] for k in [i,i+n-1], k!=2i
            #                       a[i]^2 for k=2i
            c_base = c_full.copy()
            for k in range(i, min(i + n, 2 * n - 1)):
                if k == 2 * i:
                    c_base[k] -= x_orig * x_orig
                else:
                    c_base[k] -= 2.0 * x_orig * a[k - i]

            # Find max of unaffected lags
            # Affected lags are [i, i+n-1]
            lo_k = i
            hi_k = min(i + n, 2 * n - 1)
            if lo_k > 0 or hi_k < 2 * n - 1:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[lo_k:hi_k] = False
                max_unaffected = np.max(c_base[mask])
            else:
                max_unaffected = 0.0

            # Golden section search for optimal x
            lo = 0.0
            hi = max(3.0 * x_orig, 0.5)

            def _eval_x(x):
                """Evaluate objective at coordinate value x."""
                s = S + x
                if s < 0.01:
                    return 1e10
                max_c = max_unaffected
                for k in range(lo_k, hi_k):
                    if k == 2 * i:
                        gk = c_base[k] + x * x
                    else:
                        gk = c_base[k] + 2.0 * x * a[k - i]
                    if gk > max_c:
                        max_c = gk
                return 2.0 * n * max_c / (s * s)

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _eval_x(c1)
            f2 = _eval_x(c2)

            for _it in range(35):
                if hi - lo < 1e-10 * max(1.0, hi):
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _eval_x(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _eval_x(c2)

            best_x = (lo + hi) / 2.0
            best_f = _eval_x(best_x)

            if best_f < current_score - 1e-12:
                a[i] = best_x
                current_score = best_f
                improved = True
                # Update c_full for affected lags
                for k in range(lo_k, hi_k):
                    if k == 2 * i:
                        c_full[k] = c_base[k] + best_x * best_x
                    else:
                        c_full[k] = c_base[k] + 2.0 * best_x * a[k - i]
            else:
                a[i] = x_orig

        if not improved:
            break

    return a, current_score


def polish_annealing(a_init, time_limit, rng):
    """Simulated annealing local search for final polish."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.99985

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(4, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 2)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.60:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.75:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * current[i]
                candidate[i] = max(0.0, current[i] - delta)
                candidate[j] = max(0.0, current[j] + delta)
        elif r < 0.85:
            # Triple perturbation
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) == 3:
                delta = rng.uniform(-0.08, 0.08) * current[i]
                candidate[i] = max(0.0, current[i] - 2 * delta)
                candidate[j] = max(0.0, candidate[j] + delta)
                candidate[k] = max(0.0, candidate[k] + delta)
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.005, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def make_init(n, init_type, alpha, rng):
    """Create various initialization sequences."""
    if init_type == 'uniform':
        return np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n)) + 0.5
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
        a = np.clip(a, 0.05, None)
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.05, None)
    elif init_type == 'multi_tone':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in range(1, 8):
            amp = 0.3 / k
            freq = rng.uniform(0.5, 3.0) * k
            phase = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        a += 0.04 * rng.standard_normal(n)
        return np.clip(a, 0.05, None)
    elif init_type == 'symmetric':
        half = n // 2
        a = 1.0 + 0.3 * rng.standard_normal(half)
        a = np.clip(a, 0.1, None)
        if n % 2 == 0:
            full = np.concatenate([a, a[::-1]])
        else:
            full = np.concatenate([a, 1.0, a[::-1]])
        return full[:n]
    else:
        return np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.05, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length via interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def smooth_optimize(z0, n, beta_schedule, time_per_level, rng):
    """
    Run smooth gradient optimization with increasing beta.
    Returns best z found.
    """
    current_z = z0.copy()
    best_a = np.exp(current_z).copy()
    best_score = local_score(best_a)

    for level, beta in enumerate(beta_schedule):
        elapsed_level = 0.0
        while elapsed_level < time_per_level:
            remaining = time_per_level - elapsed_level
            maxiter = max(20, int(200 * remaining / time_per_level))

            t_start = time.monotonic()
            result_z, result_smooth = run_lbfgs(current_z, beta, n, maxiter=maxiter)
            elapsed_level = time.monotonic() - t_start

            a_new = np.exp(result_z)
            a_new = np.clip(a_new, 0.0, 1000.0)
            true_score = local_score(a_new)

            if true_score < best_score:
                best_score = true_score
                best_a = a_new.copy()
                current_z = result_z.copy()

            # Check if converged (smooth value not improving)
            if result_smooth < 0:  # safety
                break
            # If true score hasn't improved much, try next level
            if elapsed_level > time_per_level * 0.5:
                break

            # If we still have time, do another L-BFGS call at same beta
            if elapsed_level < time_per_level * 0.7:
                continue
            else:
                break

        print(f"    beta={beta:.0f}: true_score={best_score:.6f}")

    return best_a, best_score, current_z


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Use provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = local_score(start_seq)
            print(f"Provided: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Try multiple n values and initializations
    n_candidates = [512, 384, 256, 768, 1024]
    if start_seq is not None:
        n_candidates = [len(start_seq)] + [x for x in n_candidates if x != len(start_seq)]

    init_types = ['chirp', 'multi_tone', 'uniform', 'symmetric', 'random']

    # Phase 1: Quick screening with low beta
    print("\n=== Phase 1: Quick screening ===")
    screen_configs = []
    for n in n_candidates[:4]:
        for it in init_types[:3]:
            screen_configs.append((n, it))
    if start_seq is not None:
        screen_configs.insert(0, (len(start_seq), 'provided'))

    time_per_screen = max(10.0, min(40.0, (deadline - time.monotonic() - 1200) / len(screen_configs)))

    for ci, (n, it) in enumerate(screen_configs):
        if time.monotonic() > deadline - 1200:
            break

        if it == 'provided' and start_seq is not None:
            a0 = resample_sequence(start_seq, n)
        elif it == 'random':
            a0 = make_init(n, 'uniform', 0, rng)
        else:
            a0 = make_init(n, it, 0.5, rng)

        a0 = np.clip(a0, 0.01, None)
        z0 = np.log(a0)

        # Quick L-BFGS with low beta
        _gp['beta'] = 10.0
        _gp['n'] = n
        result = minimize(
            _smooth_fun, z0, jac=_smooth_jac,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-6, 'iprint': -1}
        )
        a_test = np.clip(np.exp(result.x), 0.0, 1000.0)
        score = local_score(a_test)

        print(f"  n={n}, init={it}: score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a_test.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Full smooth optimization on best candidates
    print(f"\n=== Phase 2: Full smooth optimization ===")
    print(f"  Starting score: {best_score:.6f}")

    # Get the best from screening and the provided sequence
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq.copy())
    if start_seq is not None:
        candidates.append(start_seq.copy())

    # Add a couple more diverse candidates
    for n in [512, 384]:
        a0 = make_init(n, 'multi_tone', 0.4, rng)
        candidates.append(np.clip(a0, 0.01, None))

    beta_schedules = [
        [10, 20, 50, 100, 200, 400],
        [5, 15, 40, 100, 250],
    ]

    time_per_candidate = max(60.0, min(300.0, (deadline - time.monotonic() - 600) / len(candidates)))

    for ci, a_cand in enumerate(candidates):
        if time.monotonic() > deadline - 600:
            break

        n = len(a_cand)
        a_cand = np.clip(a_cand, 0.01, None)
        z0 = np.log(a_cand)
        # Normalize z for numerical stability
        z0 -= np.mean(z0)

        schedule = beta_schedules[ci % len(beta_schedules)]
        t_avail = min(time_per_candidate, deadline - time.monotonic() - 600)
        t_per_level = max(15.0, t_avail / len(schedule))

        print(f"  Candidate {ci+1}/{len(candidates)}: n={n}, time={t_avail:.0f}s")
        a_best, s_best, z_best = smooth_optimize(z0, n, schedule, t_per_level, rng)
        print(f"  Candidate {ci+1} result: score={s_best:.6f}")

        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Coordinate descent refinement
    print(f"\n=== Phase 3: Coordinate descent ===")
    print(f"  Before: {best_score:.6f}")
    if time.monotonic() < deadline - 200:
        cd_time = min(400.0, deadline - time.monotonic() - 200)
        n_passes = max(5, int(cd_time / 3))  # roughly 3s per pass
        a_cd, s_cd = coordinate_descent(best_seq, n_passes=n_passes)
        print(f"  After CD: {s_cd:.6f}")
        if s_cd < best_score:
            best_score = s_cd
            best_seq = a_cd.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Annealing polish
    print(f"\n=== Phase 4: Annealing polish ===")
    if time.monotonic() < deadline - 80:
        anneal_time = min(300.0, deadline - time.monotonic() - 80)
        print(f"  Before: {best_score:.6f}, time={anneal_time:.0f}s")
        a_pol, s_pol = polish_annealing(best_seq, anneal_time, rng)
        print(f"  After anneal: {s_pol:.6f}")
        if s_pol < best_score:
            best_score = s_pol
            best_seq = a_pol.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 5: Final coordinate descent
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Final CD ===")
        cd_time2 = deadline - time.monotonic() - 30
        n_passes2 = max(3, int(cd_time2 / 3))
        a_cd2, s_cd2 = coordinate_descent(best_seq, n_passes=n_passes2)
        print(f"  Final CD: {s_cd2:.6f}")
        if s_cd2 < best_score:
            best_score = s_cd2
            best_seq = a_cd2.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
