"""
Step function optimization using soft-max smoothing + subgradient methods.
Minimizes 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses FFT-based gradient computation for O(n log n) per iteration.

import numpy as np
import time

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the objective value."""
    a = np.clip(a, 0, None)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    n = len(a)
    b = fft_convolve(a)
    M = np.max(b)
    return 2.0 * n * M / (S * S)

def softmax_weights(b, sigma):
    """Compute softmax weights for the soft-max approximation."""
    b_max = np.max(b)
    shifted = (b - b_max) / sigma
    exp_shifted = np.exp(shifted)
    return exp_shifted / np.sum(exp_shifted), b_max

def compute_gradient_smoothed(a, sigma):
    """Compute gradient of soft-max of conv(a,a) w.r.t. a.
    dM/da = 2 * convolve(w, a)[:n] where w = softmax(b/sigma)
    """
    b = fft_convolve(a)
    w, b_max = softmax_weights(b, sigma)
    # dM/da_i = 2 * sum_j w[i+j] * a[j] (for valid indices)
    # This is 2 * (w * a) convolved, truncated to n
    grad_full = fft_convolve(w, a)
    grad = 2.0 * grad_full[:len(a)]
    return grad, b_max, w

def project_nonneg_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S} using bisection."""
    # Find t such that sum(max(a - t, 0)) = S
    if np.sum(np.maximum(a, 0)) <= S:
        return np.maximum(a, 0) * (S / max(np.sum(np.maximum(a, 0)), 1e-15))
    
    lo = np.min(a) - 1
    hi = np.max(a) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(a - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - (lo + hi) / 2, 0)

def smooth_descent(a0, time_limit, rng, sigma_init=0.1):

        grad_proj = grad - np.mean(grad)

    return best_a, best_score

def subgradient_descent(a0, time_limit, rng):
    """Pure subgradient descent (no smoothing)."""
    n = len(a0)
    a = a0.astype(np.float64).copy()

    v = np.zeros(n)
    beta = 0.9

        b = fft_convolve(a)
        M = np.max(b)
        
        # Find peaks (within 1% of max for robustness)
        threshold = M * 0.99
        peaks = np.where(b >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peaks:
            # grad[i] += 2*a[k-i] if 0 <= k-i < n
            valid_i = np.arange(max(0, k - n + 1), min(k + 1, n))
            valid_j = k - valid_i
            grad[valid_i] += 2.0 * a[valid_j]

        grad_proj = grad - np.mean(grad)
        
        # Momentum
        v = beta * v + grad_proj
        a_new = a - step * v
        a_new = project_nonneg_sum(a_new, S)

def make_antisyms_init(n, eps, rng):
    """Anti-symmetric perturbation about center."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + eps * np.sin(np.pi * (t - center) / (n / 2.0))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng, spread=0.2):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

def make_bump_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + alpha * np.exp(-((t - center) ** 2) / (2 * (n / 4) ** 2))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

The key insight is that I can compute the gradient of a soft-max approximation of the peak efficiently using FFT (via `grad = 2 * convolve(softmax_weights, a)`), enabling fast smooth gradient descent. ```python
"""
Step function optimization using soft-max smoothed gradient descent with FFT-based
gradient computation, subgradient kicks, block coordinate refinement, and multi-scale
search. Minimizes 2n*max(conv(a,a))/sum(a)^2 for non-negative step sequences.
"""

import numpy as np
import time


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, None)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    n = len(a)
    b = fft_convolve(a)
    M = np.max(b)
    return 2.0 * n * M / (S * S)


def softmax_weights(b, sigma):
    """Compute softmax weights and max for soft-max approximation."""
    b_max = np.max(b)
    shifted = (b - b_max) / sigma
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = exp_shifted / total
    return w, b_max


def compute_grad_smoothed(a, sigma):
    """Compute gradient of soft-max(conv(a,a)) w.r.t. a using FFT.
    dM/da_i = 2 * sum_j w[i+j] * a[j] = 2 * (w * a)_i (truncated).
    """
    b = fft_convolve(a)
    w, b_max = softmax_weights(b, sigma)
    grad_full = fft_convolve(w, a)
    grad = 2.0 * grad_full[:len(a)]
    return grad, b_max


def project_nonneg_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a_pos = np.maximum(a, 0)
    s = np.sum(a_pos)
    if s <= S + 1e-15:
        if s < 1e-15:
            return np.full(len(a), S / len(a))
        return a_pos * (S / s)
    # Need to find t such that sum(max(a - t, 0)) = S
    lo = np.min(a)
    hi = np.max(a)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(a - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - (lo + hi) / 2.0, 0)


def smooth_descent(a0, time_limit, rng, sigma_init=0.5, verbose=True):
    """Smoothed gradient descent with momentum and adaptive sigma/step."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n) * S
        S = n
    a = a * S / np.sum(a)

    sigma = sigma_init
    step = 0.05
    v = np.zeros(n)
    beta = 0.8

    best_a = a.copy()
    best_score = compute_score(a)
    stale = 0
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        grad, b_max = compute_grad_smoothed(a, sigma)
        grad_proj = grad - np.mean(grad)

        v = beta * v + grad_proj
        v_norm = np.linalg.norm(v)
        if v_norm > 1e10:
            v = v * (1e10 / v_norm)

        a_new = a - step * v
        a_new = project_nonneg_sum(a_new, S)
        new_score = compute_score(a_new)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            stale = 0
            step = min(0.3, step * 1.04)
            # Gradually decrease sigma
            sigma = max(0.01, sigma * 0.998)
        else:
            stale += 1
            if stale % 10 == 0:
                step *= 0.85
            if stale > 40:
                # Sharpen sigma
                sigma = max(0.005, sigma * 0.6)
                step *= 0.5
                v = np.zeros(n)
                stale = 10
            if stale > 200:
                # Reset to best and try fresh
                a = best_a.copy()
                v = np.zeros(n)
                step = 0.05
                sigma = max(0.1, sigma_init * 0.3)
                stale = 0
                # Add small perturbation
                a = a * (1.0 + 0.02 * rng.standard_normal(n))
                a = project_nonneg_sum(a, S)

        a = a_new
        it += 1

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={it}, score={new_score:.7f}, best={best_score:.7f}, "
                  f"sigma={sigma:.5f}, step={step:.5f}, el={elapsed:.0f}s")

    return best_a, best_score


def subgradient_descent(a0, time_limit, rng):
    """Pure subgradient descent with momentum."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
        S = n
    a = a * S / np.sum(a)

    step = 0.003
    v = np.zeros(n)
    beta = 0.85

    best_a = a.copy()
    best_score = compute_score(a)
    stale = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = fft_convolve(a)
        M = np.max(b)
        threshold = M * 0.995
        peaks = np.where(b >= threshold)[0]

        grad = np.zeros(n)
        for k in peaks:
            lo = max(0, k - n + 1)
            hi = min(k + 1, n)
            valid_i = np.arange(lo, hi)
            valid_j = k - valid_i
            grad[valid_i] += 2.0 * a[valid_j]
        grad /= len(peaks)

        grad_proj = grad - np.mean(grad)
        v = beta * v + grad_proj

        a_new = a - step * v
        a_new = project_nonneg_sum(a_new, S)
        new_score = compute_score(a_new)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            stale = 0
            step = min(0.01, step * 1.03)
        else:
            stale += 1
            if stale % 15 == 0:
                step *= 0.9
            if stale > 100:
                v = np.zeros(n)
                step = 0.003
                stale = 0

        a = a_new

    return best_a, best_score


def block_refine(a0, time_limit, rng, block_size=8):
    """Block coordinate descent: optimize blocks of a at a time."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    n_blocks = (n + block_size - 1) // block_size
    order = np.arange(n_blocks)

    while time.monotonic() - t0 < time_limit:
        rng.shuffle(order)
        any_improved = False
        for bi in order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            # Optimize this block via gradient steps
            block = a[start:end].copy()
            for _ in range(20):
                b = fft_convolve(a)
                M = np.max(b)
                w, _ = softmax_weights(b, 0.1)
                grad_full = fft_convolve(w, a)
                grad_block = 2.0 * grad_full[start:end]
                # Project within block
                grad_block = grad_block - np.mean(grad_block)
                lr = 0.1
                block = block - lr * grad_block
                block = np.maximum(block, 0.0)
                if np.sum(block) < 1e-10:
                    block = np.ones(end - start) * 0.1
            a[start:end] = block
            # Re-normalize
            a = project_nonneg_sum(a, S)
            sc = compute_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                any_improved = True
        if not any_improved:
            break

    return best_a, best_score


def coordinate_refine(a0, time_limit, rng):
    """Fine coordinate-wise optimization."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        rng.shuffle(order)
        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for frac in [0.85, 0.92, 1.08, 1.15]:
                a_new = a.copy()
                a_new[i] = a[i] * frac
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s < 0.01:
                    continue
                a_new = a_new * S / s
                sc = compute_score(a_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    a = a_new
                    best_a = a_new.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_a, best_score


def make_antisyms_init(n, eps, rng):
    """Anti-symmetric perturbation about center."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + eps * np.sin(np.pi * (t - center) / (n / 2.0))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.25):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_bump_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + alpha * np.exp(-((t - center) ** 2) / (2.0 * (n / 4.0) ** 2))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 6):
        amp = 0.2 / k
        freq = rng.uniform(0.5, 2.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return np.clip(new_seq, 0.01, None)
