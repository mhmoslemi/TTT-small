import numpy as np

import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fast_score(a):
    """Compute 2n * max(conv(a,a)) / (sum a)^2"""
    n = len(a)
    if n == 0 or np.sum(a) < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (np.sum(a) ** 2)

def subgradient_step(a, lr, top_k=1):
    """One subgradient step on the peak autocorrelation (normalized to sum=1)."""
    n = len(a)
    c = fft_convolve(a, a)
    # Get top-k peaks
    if top_k >= 2*n-1:
        peak_vals = c
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        peak_indices = np.argsort(c)[-top_k:]
        peak_vals = c[peak_indices]
        # Weight by how close to max
        max_val = peak_vals[0]
        weights = np.exp(-(peak_vals - min(peak_vals)) * 10)  # soft weighting
        weights /= np.sum(weights)
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = 0.0
        if top_k >= 2*n-1:
            w = weights[k]
        else:
            if k in set(peak_indices.tolist()):
                w = weights[peak_indices.tolist().index(k)]
        if w == 0:
            continue
        # grad[i] += 2 * w * a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * w * a[k - i]
    
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    s = np.sum(a_new)
    if s < 0.01:
        return a.copy()
    return a_new / s  # renormalize to sum=1

"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines: spectral equalization, LP-based direction finding, subgradient with Adam,
coordinate descent, and simulated annealing polish. Multiple n values and restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute 2n * max(conv(a,a)) / (sum a)^2 efficiently."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)


def normalize_sum1(a):
    """Normalize array to sum = 1, staying non-negative."""
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix where row k gives coefficients of conv[k] as linear function of changes."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n


def spectral_equalization(a, n_iter=200, rng=None):
    """Flatten the power spectrum via Gerchberg-Saxton-like iteration."""
    n = len(a)
    a_work = a.astype(np.float64).copy()
    a_work = normalize_sum1(a_work)
    
    # Target: flat magnitude spectrum
    for it in range(n_iter):
        # Forward FFT
        F = np.fft.rfft(a_work, 2 * n)
        # Target magnitude: flat (approximately 1/sqrt(n) for non-DC, 1 for DC)
        target_mag = np.ones_like(np.abs(F))
        # Keep DC component large (it's the sum)
        # Actually for normalized seq, DC = 1, others should be small
        # We want |F[k]| to be roughly constant for k>=1
        mags = np.abs(F)
        mags[0] = 1.0  # fix DC
        if len(mags) > 1:
            target_level = np.mean(mags[1:]) * 0.5  # push toward flat
        else:
            target_level = 0.1
        new_mags = np.clip(mags, target_level, None)
        new_mags[0] = 1.0
        # New spectrum with same phases
        phases = np.angle(F)
        F_new = new_mags * np.exp(1j * phases)
        # Inverse FFT
        a_new = np.fft.irfft(F_new, 2 * n)[:n]
        a_new = np.clip(a_new, 0, None)
        a_new = normalize_sum1(a_new)
        a_work = a_new
    
    return a_work

def coordinate_descent(a0, time_limit, rng, verbose=False):
    """Coordinate descent: for each element, try multiplier factors and pick best."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    a = normalize_sum1(a)
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    factors = [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.3, 1.5, 2.0, 3.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for fac in factors:
                trial = a.copy()
                trial[i] = a[i] * fac
                trial = np.clip(trial, 0, None)
                trial = normalize_sum1(trial)
                trial_score = fast_score(trial)
                if trial_score < current_score:
                    a = trial
                    current_score = trial_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()
                    break
        
        if not improved:
            # Try with fewer, smaller factors
            small_factors = [0.9, 0.95, 1.0/0.95, 1.0/0.9]
            for i in order[:n//2]:
                if time.monotonic() - t0 > time_limit:
                    break
                for fac in small_factors:
                    trial = a.copy()
                    trial[i] = a[i] * fac
                    trial = np.clip(trial, 0, None)
                    trial = normalize_sum1(trial)
                    trial_score = fast_score(trial)
                    if trial_score < current_score * 0.9999:
                        a = trial
                        current_score = trial_score
                        if current_score < best_score:
                            best_score = current_score
                            best_a = a.copy()
                        break
    
    if verbose:
        print(f"    CoordDesc: final score={best_score:.6f}")
    return best_a, best_score

    best_score = current_score
    
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.25:
            # Peak-targeted reduction
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                contribs[i] = a[i] * a[peak_idx - i]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 3:
                n_inc = min(8, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.40:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.55:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.2, 0.2) * max(a[i], a[j])
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        elif r < 0.70:
            # Multi-pair
            for _ in range(3):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    delta = rng.uniform(-0.1, 0.1) * max(a[i], a[j])
                    candidate[i] = max(0, a[i] - delta)
                    candidate[j] = max(0, a[j] + delta)
        elif r < 0.85:
            # Small global
            candidate = a * np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Concentration: boost some, reduce others
            mask = rng.random(n) < 0.3
            candidate[mask] *= 0.8
            candidate[~mask] *= 1.05
        
        candidate = np.clip(candidate, 0, None)
        candidate = normalize_sum1(candidate)
        new_score = fast_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
        if temperature < 1e-9:
            temperature = 1e-9
    
    if verbose:
        print(f"    SA: final score={best_score:.6f}")
    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.4):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_spectral_flat_init(n, rng):
    """Initialize by trying to get flat spectrum."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.1, None)
    return spectral_equalization(a, n_iter=50, rng=rng)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
