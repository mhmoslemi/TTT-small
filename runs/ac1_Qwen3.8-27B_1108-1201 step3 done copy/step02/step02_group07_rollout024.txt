import numpy as np
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def compute_softmax_grad(a, tau):
    """Compute softmax-weighted gradient of max(c)."""
    n = len(a)
    c = fft_convolve(a, a)
    # Softmax weights
    c_max = np.max(c)
    weights = np.exp((c - c_max) / tau)
    weights /= np.sum(weights)
    # Gradient: g[j] = sum_k weights[k] * a[k-j] (for valid k-j)
    g = np.zeros(n)
    for j in range(n):
        # g[j] = sum_k weights[k] * a[k-j] where 0 <= k-j < n
        # i.e., k ranges from j to j+n-1
        lo = j
        hi = min(2*n-1, j+n)  # k goes from j to min(j+n-1, 2n-2)
        if lo < 2*n-1:
            g[j] = np.dot(weights[lo:hi], a[:hi-lo])
    return g, c, weights

                            lr_init=0.05, momentum=0.9, cd_every=50, verbose=True):

    n = len(f0)
    a = f0.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a * (S / S)  # already normalized
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    it = 0
    g_mom = np.zeros(n)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        # Anneal temperature
        frac = min(1.0, it / max(1, int(time_limit / 0.5)))
        tau = tau_init * (tau_end / tau_init) ** frac
        # Actually, let's use a schedule based on iteration count
        # Total expected iterations ~ time_limit / 0.01 = 100000 * (1/0.5) ... this is hard to estimate
        # Let's just use a fixed schedule: tau decreases linearly over the time limit
        elapsed = time.monotonic() - t0
        tau = tau_init + (tau_end - tau_init) * min(1.0, elapsed / time_limit)
        
        # Compute gradient
        g, c, w = compute_softmax_grad(a, tau)
        
        # Normalize gradient to have zero mean (preserves sum approximately)
        g_mean = np.mean(g)
        g = g - g_mean
        
        # Momentum
        g_mom = momentum * g_mom + (1 - momentum) * g
        
        # Line search
        best_lr = lr_init
        best_s = np.inf
        for lr in [lr_init, 2*lr_init, 0.5*lr_init, 0.25*lr_init]:
            a_cand = a * np.exp(-lr * g_mom)
            a_cand = np.clip(a_cand, 1e-10, 1000.0)
            a_cand = a_cand * (S / float(np.sum(a_cand)))  # renormalize
            s_cand = score_fast(a_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_lr = lr
        
        a = a * np.exp(-best_lr * g_mom)
        a = np.clip(a, 1e-10, 1000.0)
        a = a * (S / float(np.sum(a)))
        it += 1
        
        # Check improvement
        if best_s < best_score - 1e-10:
            best_score = best_s
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Coordinate descent every cd_every iterations
        if cd_every > 0 and it % cd_every == 0 and time.monotonic() - t0 < time_limit - 5:
            a, s_cd = coord_pass(a)
            if s_cd < best_score:
                best_score = s_cd
                best_a = a.copy()
                stall = 0
        
        # Restart if stuck too long
        if stall > 200:
            rng = np.random.default_rng(it)
            a = a * np.exp(rng.normal(0, 0.1, n))
            a = np.clip(a, 1e-10, 1000.0)
            a = a * (S / float(np.sum(a)))
            stall = 0
            g_mom = np.zeros(n)
        
        if verbose and it % 100 == 0:
            print(f"    [{time.monotonic()-t0:6.1f}s] it={it:5d} tau={tau:.4f} score={best_s:.6f} best={best_score:.6f}")
    
    return best_a, best_score

def coord_pass(a, max_rounds=3):
    """One or more passes of coordinate descent."""
    n = len(a)
    S = float(np.sum(a))
    best_score = score_fast(a)
    
    for round in range(max_rounds):
        for j in range(n):
            old_val = a[j]
            # Try several candidate values
            best_v = old_val
            best_s_new = best_score  # will be recomputed
            
            # The effect of changing a[j] on c:
            # c[k] changes by a_new[j]*a[k-j] - old_val*a[k-j] for k where k-j is valid
            # and by (a_new[j]^2 - old_val^2) for k = 2j
            
            # Quick evaluation: try a few values
            candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val, old_val * 1.3, old_val * 1.5]
            for v in candidates:
                a[j] = v
                s_new = score_fast(a)
                if s_new < best_s_new:
                    best_s_new = s_new
                    best_v = v
            a[j] = best_v
            
        # Renormalize
        a = a * (S / float(np.sum(a)))
    
    return a, score_fast(a)

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def compute_softmax_grad(a, tau):
    """Compute softmax-weighted subgradient of max(c). Returns (grad, c, peak_max)."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    # Softmax weights (numerically stable)
    weights = np.exp((c - c_max) / tau)
    weights /= np.sum(weights)
    # Gradient: g[j] = 2 * sum_k weights[k] * a[k-j] for valid k-j in [0, n-1]
    g = np.zeros(n)
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        if lo < hi:
            g[j] = 2.0 * np.dot(weights[lo:hi], a[:hi - lo])
    return g, c, c_max

def make_asymmetric_init(n, rng):
    """Create an asymmetric smooth initialization."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    # Skewed shape: exponential decay from left, with a bump
    alpha = rng.uniform(1.0, 3.0)
    a = np.exp(-alpha * t) * (1.0 + 0.5 * np.sin(np.pi * t * rng.uniform(1, 3)))
    a = np.clip(a, 0.01, None)
    # Add slight noise
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a


def make_cosine_asym(n, rng):
    """Asymmetric cosine-based initialization."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)

    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - phase))
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a


def make_waterfall(n, rng):
    """Waterfall/ramp with smoothing."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    direction = rng.choice([1, -1])
    if direction == 1:
        a = 1.0 - t
    else:
        a = t
    # Add a bump
    bump_center = rng.uniform(0.2, 0.8)
    bump_width = rng.uniform(0.1, 0.3)
    a += 0.3 * np.exp(-((t - bump_center) ** 2) / (2 * bump_width ** 2))
    a = np.clip(a, 0.01, None)
    return a
