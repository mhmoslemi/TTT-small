    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_to_sum1(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a)
    return a / s


def build_conv_matrix_rows(a, p):
    """Compute gradient of <p, conv(a,a)> w.r.t. a using FFT."""
    n = len(a)
    p_flipped = p[::-1]
    full_conv = fft_convolve(a, p_flipped)
    indices = (2 * n - 2) - np.arange(n)
    C_arr = full_conv[indices]
    return 2.0 * C_arr


def smooth_gradient_descent(a_init, n_steps, tau_start, tau_end, lr_start, lr_end, 
                            beta1=0.9, beta2=0.999, eps=1e-8, time_limit=None,
                            t_start=None, verbose=False):
    """
    Adam-based smooth gradient descent on the log-sum-exp approximation.
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = normalize_to_sum1(a)
    n = len(a)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    step_count = 0
    
    if t_start is None:
        t_start = time.monotonic()
    
    best_score = local_score(a)
    best_a = a.copy()
    
    for step in range(n_steps):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        
        # Anneal tau
        progress = min(1.0, step / max(1, n_steps * 0.8))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Anneal learning rate
        lr = lr_start * (lr_end / lr_start) ** (step / max(1, n_steps))
        
        # Compute convolution
        conv = fft_convolve(a, a)
        
        # Compute softmax weights
        conv_clipped = np.clip(tau * conv, -500, 500)
        conv_shifted = conv_clipped - np.max(conv_clipped)
        p = np.exp(conv_shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.ones_like(conv) / len(conv)
        else:
            p = p / p_sum
        
        # Compute gradient
        grad = build_conv_matrix_rows(a, p)
        
        # Adam update
        step_count += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_hat = m_hat / (1 - beta1 ** step_count)
        v_hat_hat = v_hat / (1 - beta2 ** step_count)
        
        a_new = a - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum1(a_new)
        
        # Check improvement
        score = local_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      SGD step {step}: score={best_score:.6f}, tau={tau:.2f}, lr={lr:.6f}, time={elapsed:.1f}s")
    
    return best_a, best_score


def lp_refine_one_step(a, M=None):
    """
    One LP refinement step: find g that maximizes sum(g) subject to conv(a,g) <= M, g >= 0.
    Returns (g, M) or (None, M) if LP fails.
    """
    n = len(a)
    if M is None:
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
    
    # Build constraint matrix: A[k, j] = a[k-j] if 0<=k-j<n else 0
    # conv(a, g)[k] = sum_j A[k,j] * g[j] <= M
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max+1] = a[k - j_max : k - j_min + 1]  # a[k-j] for j from j_min to j_max
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            return res.x, M
    except Exception:
        pass
    return None, M


def lp_refine(a, time_limit, max_iters=100):
    """
    Iterative LP refinement with adaptive blending.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = normalize_to_sum1(a)
    
    t0 = time.monotonic()
    best_score = local_score(a)
    best_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        g, M = lp_refine_one_step(a, M)
        if g is None or np.sum(g) < 1e-10:
            break
        
        # Normalize g
        g = g / np.sum(g)
        
        # Find best blend via bisection
        best_t = 0.0
        best_conv_max = M
        
        for t_try in [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]:
            a_blend = (1 - t_try) * a + t_try * g
            a_blend = np.maximum(a_blend, 0.0)
            a_blend = normalize_to_sum1(a_blend)
            conv_blend = fft_convolve(a_blend, a_blend)
            conv_max = np.max(conv_blend)
            if conv_max < best_conv_max - 1e-12:
                best_conv_max = conv_max
                best_t = t_try
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = normalize_to_sum1(a)
        
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def targeted_local_search(a, time_limit, rng, verbose=False):
    """
    Targeted local search focusing on reducing the peak lag.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = normalize_to_sum1(a)
    
    t0 = time.monotonic()
    best_score = local_score(a)
    best_a = a.copy()
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Compute contributions to the peak
        # conv[peak_idx] = sum_j a[j]*a[peak_idx-j] (valid j)
        contribs = np.zeros(n)
        j_min = max(0, peak_idx - n + 1)
        j_max = min(peak_idx, n - 1)
        for j in range(j_min, j_max + 1):
            contribs[j] = a[j] * a[peak_idx - j]
        
        # Strategy: reduce top contributors, increase low contributors
        n_reduce = max(2, n // 15)
        n_increase = max(2, n // 10)
        
        top_idx = np.argsort(contribs)[-n_reduce:]
        bottom_idx = np.argsort(contribs)[:n_increase]
        
        candidate = a.copy()
        
        # Reduce top contributors
        for idx in top_idx:
            factor = np.exp(rng.normal(-0.03, 0.01))
            candidate[idx] *= factor
        
        # Increase bottom contributors
        for idx in bottom_idx:
            factor = np.exp(rng.normal(0.03, 0.01))
            candidate[idx] *= factor
        
        candidate = np.maximum(candidate, 0.0)
        candidate = normalize_to_sum1(candidate)
        
        new_score = local_score(candidate)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Try a more aggressive perturbation
                candidate = a.copy()
                n_perturb = max(5, n // 8)
                perturb_idx = rng.choice(n, size=n_perturb, replace=False)
                candidate[perturb_idx] *= np.exp(rng.normal(0, 0.05, size=n_perturb))
                candidate = np.maximum(candidate, 0.0)
                candidate = normalize_to_sum1(candidate)
                new_score = local_score(candidate)
                if new_score < best_score:
                    best_score = new_score
                    best_a = candidate.copy()
                    a = candidate
                    stall = 0
                else:
                    stall = 100  # Reset to try again
    
    return best_a, best_score


def make_random_init(n, rng, init_type='random'):
    """Create a random initialization."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'random_positive':
        a = 0.5 + 0.5 * rng.random(n)
    elif init_type == 'chirp':
        alpha = 0.2 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.3))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.1 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_sine':
        a = np.ones(n)
        for _ in range(2 + int(rng.integers(0, 3))):
            f = int(rng.integers(1, n // 5))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'linear_ramp':
        slope = rng.uniform(-0.3, 0.3)
        a = 1.0 + slope * t / n
    elif init_type == 'arch':
        a = 1.0 + 0.3 * rng.random() * np.sin(np.pi * t / n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    return normalize_to_sum1(a)


def interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

"""
Minimizes the normalized peak autocorrelation score 2*n*max(conv(a,a))/(sum(a))^2
for non-negative step functions. Uses smooth gradient descent (log-sum-exp of max)
with Adam optimizer, combined with LP-based directional refinement and targeted
local search across multiple sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_sum1(a):
    """Normalize to unit sum."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a)
    return a / s


def compute_grad(a, p):
    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    Uses the identity: grad[m] = 2 * sum_k p[k]*a[k-m] (valid range)
    = 2 * cross_correlation(p, a)[m], computed via FFT.
    """
    n = len(a)
    p_flipped = p[::-1]
    full_conv = fft_convolve(a, p_flipped)
    indices = (2 * n - 2) - np.arange(n)
    return 2.0 * full_conv[indices]


def smooth_gradient_descent(a_init, time_limit, t_start, tau_scale=1.0,
                            lr_base=0.005, verbose=False):
    """
    Adam-based smooth gradient descent on log-sum-exp(max(conv(a,a))).
    Anneals tau from smooth to sharp.
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = normalize_sum1(a)
    n = len(a)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    step_count = 0

    best_score = local_score(a)
    best_a = a.copy()

    # Annealing schedule
    tau_start = 10.0 * tau_scale
    tau_end = 200.0 * tau_scale
    total_steps_estimated = int(time_limit * 3000)  # rough estimate
    anneal_frac = 0.7

    stall_count = 0
    max_stall = 500

    while time.monotonic() - t_start < time_limit:
        step_count += 1
        progress = min(1.0, step_count / max(1, total_steps_estimated * anneal_frac))
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_base * (1.0 + step_count * 1e-5) ** (-0.5)

        # Compute convolution
        conv = fft_convolve(a, a)

        # Compute softmax weights
        tc = tau * conv
        tc_max = np.max(tc)
        tc_shifted = tc - tc_max
        tc_shifted = np.clip(tc_shifted, -50, 50)
        exp_tc = np.exp(tc_shifted)
        p_sum = np.sum(exp_tc)
        if p_sum < 1e-300:
            p = np.ones_like(conv) / len(conv)
        else:
            p = exp_tc / p_sum

        # Compute gradient
        grad = compute_grad(a, p)

        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad * grad
        m_hat_c = m_hat / (1 - beta1 ** step_count)
        v_hat_c = v_hat / (1 - beta2 ** step_count)

        a_new = a - lr * m_hat_c / (np.sqrt(v_hat_c) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum1(a_new)

        score = local_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # If stalled, perturb
        if stall_count > max_stall:
            a = best_a.copy()
            noise = np.random.default_rng(step_count).normal(0, 0.01, n)
            a = a * np.exp(noise)
            a = np.maximum(a, 0.0)
            a = normalize_sum1(a)
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            stall_count = 0

        if verbose and step_count % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"        SGD step {step_count}: best={best_score:.6f}, "
                  f"tau={tau:.1f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_score


def one_lp_step(a, M=None):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns g or None.
    """
    n = len(a)
    if M is None:
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

    # Build constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1]

    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 60})
        if res.status == 0:
            return res.x, M
    except Exception:
        pass
    return None, M


def lp_refine(a, time_limit, max_iters=80):
    """
    Iterative LP refinement with bisection on blend parameter.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = normalize_sum1(a)

    t0 = time.monotonic()
    best_score = local_score(a)
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        g, M = one_lp_step(a, M)
        if g is None or np.sum(g) < 1e-10:
            break

        g = normalize_sum1(g)

        # Bisection on blend parameter t
        best_t = 0.0
        best_conv_max = M + 1e-10

        lo, hi = 0.0, 1.0
        for _ in range(12):
            mid = (lo + hi) / 2.0
            a_blend = (1 - mid) * a + mid * g
            a_blend = np.maximum(a_blend, 0.0)
            a_blend = normalize_sum1(a_blend)
            cmax = np.max(fft_convolve(a_blend, a_blend))
            if cmax < best_conv_max:
                best_conv_max = cmax
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t < 0.005:
            break

        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = normalize_sum1(a)

        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def targeted_local_search(a, time_limit, rng):
    """
    Targeted local search: reduce peak-contributing elements,
    increase low-contributing elements.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = normalize_sum1(a)

    t0 = time.monotonic()
    best_score = local_score(a)
    best_a = a.copy()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))

        # Compute per-element contribution to the peak
        contribs = np.zeros(n)
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(peak_idx, n - 1)
        for j in range(j_lo, j_hi + 1):
            contribs[j] = a[j] * a[peak_idx - j]

        # Also compute "total" contribution (sum over all lags)
        # This tells us which elements are "dangerous" overall
        total_contrib = np.zeros(n)
        for j in range(n):
            # Element a[j] contributes a[j]*a[k-j] to conv[k] for each k
            # Total contribution = a[j] * sum_k a[k-j] = a[j] * sum(a) = a[j]
            total_contrib[j] = a[j]

        # Strategy: reduce top peak-contributors, increase bottom ones
        n_reduce = max(3, n // 12)
        n_increase = max(3, n // 8)

        top_peak = np.argsort(contribs)[-n_reduce:]
        bottom_peak = np.argsort(contribs)[:n_increase]

        candidate = a.copy()
        for idx in top_peak:
            candidate[idx] *= np.exp(rng.normal(-0.04, 0.015))
        for idx in bottom_peak:
            candidate[idx] *= np.exp(rng.normal(0.04, 0.015))

        candidate = np.maximum(candidate, 0.0)
        candidate = normalize_sum1(candidate)

        new_score = local_score(candidate)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # More aggressive: random block perturbation
                width = max(2, n // 20)
                start = int(rng.integers(0, n - width))
                candidate = a.copy()
                candidate[start:start + width] *= np.exp(
                    rng.normal(0, 0.06, size=width))
                candidate = np.maximum(candidate, 0.0)
                candidate = normalize_sum1(candidate)
                new_score = local_score(candidate)
                if new_score < best_score:
                    best_score = new_score
                    best_a = candidate.copy()
                    a = candidate
                stall = 0

    return best_a, best_score


def make_init(n, rng, vtype):
    """Create various initializations."""
    t = np.arange(n, dtype=np.float64)
    if vtype == 'constant':
        a = np.ones(n)
    elif vtype == 'random':
        a = 0.3 + 0.7 * rng.random(n)
    elif vtype == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.4))
    elif vtype == 'sine':
        f = 1 + int(rng.integers(0, 6))
        amp = 0.15 + 0.25 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif vtype == 'multi':
        a = np.ones(n)
        for _ in range(2 + int(rng.integers(0, 3))):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.15)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif vtype == 'ramp':
        a = 0.5 + 0.5 * t / max(1, n - 1)
    elif vtype == 'arch':
        a = 0.5 + 0.5 * np.sin(np.pi * t / max(1, n - 1))
    else:
        a = np.ones(n)
    return normalize_sum1(np.maximum(a, 0.01))


def interp_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Smooth gradient descent from diverse initializations and lengths
    2. LP-based directional refinement
    3. Further gradient polish
    4. Targeted local search
    Returns the best sequence found.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def try_update(a):
        nonlocal best_score, best_seq
        a = np.clip(a, 0.0, 1000.0)
        s = local_score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            return True
        return False

    # Load provided sequence if available
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.asarray(height_sequence_1, dtype=np.float64)
        provided = np.maximum(provided, 0.0)
        if np.sum(provided) > 0.01:
            provided = normalize_sum1(provided)
            try_update(provided)
            print(f"Provided: n={len(provided)}, score={best_score:.6f}")

    # Determine lengths to try
    if provided is not None:
        n_prov = len(provided)
        lengths = [n_prov]
        for dn in [-128, -64, -32, 32, 64, 128, 256]:
            nl = n_prov + dn
            if 128 <= nl <= 4096:
                lengths.append(nl)
        for nl in [256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]:
            if 128 <= nl <= 4096 and nl not in lengths:
                lengths.append(nl)
    else:
        lengths = [256, 384, 512, 640, 768, 1024, 1536]

    lengths = sorted(set(lengths))

    # Phase 1: Gradient descent from multiple starts
    print("\n=== Phase 1: Smooth gradient descent ===")
    n_lengths = len(lengths)
    phase1_time = (deadline - time.monotonic()) * 0.55
    time_per_len = max(20.0, phase1_time / n_lengths)

    init_types = ['random', 'chirp', 'sine', 'multi', 'ramp', 'arch']

    for n in lengths:
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic() - 200
        if remaining < 30:
            break

        t_len_start = time.monotonic()
        t_len_budget = min(time_per_len, remaining * 0.4)
        if t_len_budget < 15:
            continue

        print(f"  n={n} (budget={t_len_budget:.0f}s):")

        # Try provided sequence interpolated
        if provided is not None:
            a_init = interp_seq(provided, n)
            a_init = np.maximum(a_init, 0.01)
            a_init = normalize_sum1(a_init)
            t_sub = min(40.0, t_len_budget * 0.4)
            if t_sub > 10:
                ts = time.monotonic()
                res, sc = smooth_gradient_descent(a_init, t_sub, ts,
                                                   tau_scale=n / 512.0,
                                                   lr_base=0.004)
                if try_update(res):
                    print(f"    interp: score={best_score:.6f}")
                    t_len_budget -= (time.monotonic() - t_len_start)
                    t_len_start = time.monotonic()

        # Try random initializations
        n_inits = max(1, int(t_len_budget / 25.0))
        for ii in range(n_inits):
            if time.monotonic() > deadline - 200:
                break
            rem = t_len_budget - (time.monotonic() - t_len_start)
            if rem < 10:
                break

            vtype = init_types[ii % len(init_types)]
            a_init = make_init(n, rng, vtype)
            t_sub = min(40.0, rem * 0.6)
            if t_sub < 8:
                break

            ts = time.monotonic()
            res, sc = smooth_gradient_descent(a_init, t_sub, ts,
                                               tau_scale=n / 512.0,

            if try_update(res):
                print(f"    {vtype}: score={best_score:.6f}")
            t_len_start = time.monotonic()

    print(f"\n  Phase 1 best: {best_score:.6f}")

    # Phase 2: LP refinement

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100

            t_lp = min(120.0, remaining * 0.4)

            res, sc = lp_refine(best_seq, t_lp, max_iters=60)

    # Phase 3: More gradient from refined result

            res, sc = smooth_gradient_descent(best_seq, t_grad, ts,

            print(f"  After grad: {best_score:.6f}")

    # Phase 4: LP again

        remaining = deadline - time.monotonic() - 60

            t_lp2 = min(90.0, remaining * 0.4)

    # Phase 5: Targeted local search

    if best_seq is not None:

    # Final

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        verified = evaluate_sequence(final_list)
        print(f"Verified: {verified:.6f}, n={len(final_list)}")
        return final_list

    return [1.0] * 512
