import numpy as np

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def normalize_sum1(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def exp_grad_step(a, lr, tau=None, top_k=5):

    Exponentiated gradient step for minimizing max_k (a*a)[k]
    with sum(a) = 1, a >= 0.

    """

    conv = fft_convolve(a, a)
    
    if tau is not None and tau > 0:
        # Soft weights using softmax
        shifted = conv - np.max(conv)
        weights = np.exp(shifted / tau)
        weights /= np.sum(weights)
    else:

        if top_k is not None and top_k < len(conv):
            top_indices = np.argsort(conv)[-top_k:]

            weights[top_indices] = 1.0 / top_k
        else:

            pass

    # For linear: grad[i] = 2 * sum_k weights[k] * a[k-i] where 0 <= k-i < n

def exp_grad_optimize(a, time_limit, verbose=False):
    """Exponentiated gradient optimization with adaptive step size."""
    n = len(a)
    a = normalize_sum1(a.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_max_conv = np.max(fft_convolve(a, a))
    
    lr = 0.5
    stall = 0
    tau = best_max_conv * 0.1  # Softness parameter
    
    while time.monotonic() - t0 < time_limit:
        a_new, current_max = exp_grad_step(a, lr, tau=tau)
        
        # Check if we made progress
        new_max = np.max(fft_convolve(a_new, a_new))
        if new_max < best_max_conv - 1e-12:
            best_max_conv = new_max
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_max < current_max:
            a = a_new
            stall += 1
        else:
            a = a_new  # Still move, but reduce lr
            stall += 1
            lr *= 0.9
        
        if stall > 50:
            lr *= 0.5
            tau = best_max_conv * 0.05
            stall = 0
        
        # Adaptive tau
        if stall == 0:
            tau = max(best_max_conv * 0.05, tau * 1.1)
        
        if verbose and (int(time.monotonic() - t0) % 5 == 0):
            score = 2.0 * n * best_max_conv
            print(f"    ExpGrad t={time.monotonic()-t0:.0f}s: score={score:.6f}, lr={lr:.4f}, tau={tau:.4f}")
    
    return best_a

def simulated_annealing(a, time_limit, verbose=False):
    """Simulated annealing on the sequence."""
    n = len(a)
    a = normalize_sum1(a.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_max_conv = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(12345)
    
    T = best_max_conv * 0.5  # Initial temperature
    T_min = best_max_conv * 0.001
    cooling = 0.999
    
    while time.monotonic() - t0 < time_limit:
        # Propose perturbation
        candidate = a.copy()
        # Random subset of elements
        n_perturb = max(3, n // 10)
        idx = rng.choice(n, size=n_perturb, replace=False)
        eps = rng.normal(0, 0.1, size=n_perturb)
        candidate[idx] *= np.exp(eps)
        candidate = np.maximum(candidate, 1e-10)
        candidate = normalize_sum1(candidate)
        
        # Evaluate
        new_max = np.max(fft_convolve(candidate, candidate))
        delta = new_max - best_max_conv
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = candidate
            if new_max < best_max_conv:
                best_max_conv = new_max
                best_a = candidate.copy()
        
        T *= cooling
        if T < T_min:
            T = T_min
    
    return best_a

def make_initialization(n, init_type, rng):
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.2 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.15 + 0.1 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'smooth_random':
        # Generate smooth random via low-pass filtering
        a = 1.0 + 0.3 * rng.standard_normal(n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n*0.05)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    return a

def interpolate_sequence(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

"""
Optimization of non-negative step functions to minimize normalized peak
autocorrelation using exponentiated gradient with soft subgradient,
simulated annealing, peak-shaving heuristics, and multi-length search.
Key innovation: uses multiplicative weight updates (exponentiated gradient)
which are faster than LP and naturally handle non-negativity constraints,
combined with soft subgradients for smoother optimization landscape.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_sum1(a):
    """Normalize sequence to have sum = 1."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, conv, tau=None, top_k=None):
    """
    Compute (soft) subgradient of max_k conv[k] w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] where 0 <= k-i < n.
    Uses softmax weights over lags for smoothness.
    """
    n = len(a)
    conv_len = len(conv)
    
    if tau is not None and tau > 1e-15:
        shifted = conv - np.max(conv)
        weights = np.exp(shifted / tau)
        weights /= np.sum(weights)
        # Zero out negligible weights for efficiency
        weights[weights < 1e-10] = 0.0
    elif top_k is not None and top_k < conv_len:
        top_indices = np.argsort(conv)[-top_k:]
        weights = np.zeros(conv_len)
        weights[top_indices] = 1.0 / top_k
    else:
        peak_idx = int(np.argmax(conv))
        weights = np.zeros(conv_len)
        weights[peak_idx] = 1.0
    
    # Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] * 1(0<=k-i<n)
    grad = np.zeros(n)
    nonzero_k = np.where(weights > 0)[0]
    for k in nonzero_k:
        wk = weights[k]
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            grad[idx] += 2.0 * wk * a[k - idx]
    
    return grad

    lr = lr_init

    velocity = np.zeros(n)
    tau = best_max_conv * 0.15
    stall = 0
    step_count = 0

        current_max = float(np.max(conv))

        grad = compute_subgradient(a, conv, tau=tau)

            beta = max(0.3, beta * 0.95)

def peak_shave(a, time_limit, verbose=False):
    """
    Targeted peak shaving: identify top peaks in convolution and
    specifically modify elements that contribute to them.
    """
    n = len(a)
    a = normalize_sum1(a.copy()).astype(np.float64)
    t0 = time.monotonic()
    best_a = a.copy()
    best_max_conv = float(np.max(fft_convolve(a, a)))
    rng = np.random.default_rng(54321)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        current_max = float(np.max(conv))
        
        # Find top 3 peaks
        top_peaks = np.argsort(conv)[-3:]
        
        for peak_k in top_peaks:
            # Elements contributing to this peak: a[i] where a[peak_k - i] > 0
            i_min = max(0, peak_k - n + 1)
            i_max = min(n - 1, peak_k)
            contrib_range = np.arange(i_min, i_max + 1)
            if len(contrib_range) == 0:
                continue
            
            contrib = a[contrib_range] * a[peak_k - contrib_range]
            
            # Try reducing top contributors and increasing non-contributors
            n_reduce = max(2, len(contrib_range) // 5)
            top_contrib_idx = contrib_range[np.argsort(contrib)][-n_reduce:]
            
            # Non-contributor indices
            non_contrib = np.setdiff1d(np.arange(n), contrib_range)
            
            for _ in range(7):
                candidate = a.copy()
                # Reduce top contributors
                n_red = max(1, n_reduce // 2)
                red_idx = rng.choice(top_contrib_idx, size=min(n_red, len(top_contrib_idx)), replace=False)
                candidate[red_idx] *= np.exp(rng.uniform(-0.3, -0.05, size=len(red_idx)))
                
                # Increase non-contributors
                if len(non_contrib) > 0:
                    n_inc = max(2, n // 25)
                    inc_idx = rng.choice(non_contrib, size=min(n_inc, len(non_contrib)), replace=False)
                    candidate[inc_idx] *= np.exp(rng.uniform(0.05, 0.3, size=len(inc_idx)))
                
                candidate = np.maximum(candidate, 1e-10)
                candidate = normalize_sum1(candidate)
                new_max = float(np.max(fft_convolve(candidate, candidate)))
                
                if new_max < best_max_conv:
                    best_max_conv = new_max
                    best_a = candidate.copy()
                    a = candidate
                elif new_max < current_max * 1.02:
                    # Accept small regressions for exploration
                    if rng.random() < 0.3:
                        a = candidate
    
    return best_a


def simulated_annealing(a, time_limit, verbose=False):
    """Simulated annealing with adaptive perturbation size."""
    n = len(a)
    a = normalize_sum1(a.copy()).astype(np.float64)
    t0 = time.monotonic()
    best_a = a.copy()
    best_max_conv = float(np.max(fft_convolve(a, a)))
    rng = np.random.default_rng(13579)
    
    T = best_max_conv * 0.3
    T_min = best_max_conv * 0.005
    cooling = 0.9995
    perturb_scale = 0.15
    step = 0
    accept_count = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Adaptive perturbation size
        n_perturb = max(3, int(n * 0.05 * (1.0 + 0.5 * np.sin(step * 0.01))))
        idx = rng.choice(n, size=n_perturb, replace=False)
        eps = rng.normal(0, perturb_scale, size=n_perturb)
        
        candidate = a.copy()
        candidate[idx] *= np.exp(eps)
        candidate = np.maximum(candidate, 1e-10)
        candidate = normalize_sum1(candidate)
        
        new_max = float(np.max(fft_convolve(candidate, candidate)))
        delta = new_max - best_max_conv
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = candidate
            accept_count += 1
            if new_max < best_max_conv:
                best_max_conv = new_max
                best_a = candidate.copy()
                # Shrink perturbation on improvement
                perturb_scale = max(0.02, perturb_scale * 0.95)
        
        T *= cooling
        if T < T_min:
            # Restart with higher temperature occasionally
            if step % 500 == 0:
                T = best_max_conv * 0.1
                perturb_scale = 0.15
        
        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_max_conv
            print(f"    SA step {step} t={elapsed:.0f}s: score={score:.6f}, T={T:.4f}")
    
    return best_a

            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def refine_from_sequence(a, remaining_time):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a)
    a = normalize_sum1(a.copy()).astype(np.float64)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = float(local_score(a))
    
    def _update(candidate):
        nonlocal best_a, best_score
        s = local_score(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            return True
        return False
    
    # Phase 1: Exponentiated gradient
    t1 = min(40.0, (time.monotonic() - t0) + remaining_time * 0.3)
    t1_limit = 40.0
    result = exp_grad_optimize(a, t1_limit, lr_init=0.3)
    _update(result)
    
    # Phase 2: Peak shaving
    remaining = remaining_time - (time.monotonic() - t0)
    if remaining > 15:
        t2_limit = min(25.0, remaining * 0.3)
        result = peak_shave(best_a, t2_limit)
        _update(result)
    
    # Phase 3: Simulated annealing
    remaining = remaining_time - (time.monotonic() - t0)
    if remaining > 15:
        t3_limit = min(25.0, remaining * 0.3)
        result = simulated_annealing(best_a, t3_limit)
        _update(result)
    
    # Phase 4: Second exponentiated gradient (polish)
    remaining = remaining_time - (time.monotonic() - t0)
    if remaining > 10:
        t4_limit = min(20.0, remaining * 0.25)
        result = exp_grad_optimize(best_a, t4_limit, lr_init=0.1)
        _update(result)
    
    # Phase 5: Block optimization
    remaining = remaining_time - (time.monotonic() - t0)
    if remaining > 10:
        t5_limit = min(15.0, remaining * 0.2)
        result = block_optimize(best_a, t5_limit, block_size=max(4, n // 20))
        _update(result)
    
    # Phase 6: Final peak shaving
    remaining = remaining_time - (time.monotonic() - t0)
    if remaining > 5:
        result = peak_shave(best_a, min(remaining * 0.5, 10.0))
        _update(result)
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining exponentiated gradient,
    simulated annealing, peak shaving, and block optimization.
    Tries multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"    *** NEW BEST: score={s:.6f}, n={len(a_clip)} ***")
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)
    
    n_prov = len(provided_seq) if provided_seq is not None else 512
    
    # Define lengths to try, prioritizing the provided length and nearby
    lengths = [n_prov]
    for delta in [32, 64, 96, 128, 192, 256]:
        for sign in [-1, 1]:
            n_try = n_prov + sign * delta
            if 64 <= n_try <= 4096 and n_try not in lengths:
                lengths.append(n_try)
    # Add some structured lengths
    for n_try in [256, 320, 384, 448, 512, 640, 768, 1024, 1280]:
        if 64 <= n_try <= 4096 and n_try not in lengths:
            lengths.append(n_try)
    
    # Sort: provided length first, then by proximity
    lengths = sorted(set(lengths), key=lambda x: (x != n_prov, abs(x - n_prov)))
    
    print(f"Trying lengths: {lengths[:8]}")
    
    # Allocate time: more time for the provided length, less for others
    # First sequence gets ~40% of time, others get decreasing amounts
    total_time = budget_s - 60
    n_lengths = min(len(lengths), 8)  # Limit to 8 lengths
    
    time_alloc = []
    remaining_time = total_time
    for i in range(n_lengths):
        if i == 0:
            alloc = min(remaining_time * 0.35, 400.0)
        elif i == 1:
            alloc = min(remaining_time * 0.2, 250.0)
        elif i == 2:
            alloc = min(remaining_time * 0.15, 180.0)
        else:
            alloc = min(remaining_time * 0.1, 120.0)
        alloc = max(alloc, 30.0)
        time_alloc.append(alloc)
        remaining_time -= alloc
    
    for idx, n in enumerate(lengths[:n_lengths]):
        if time.monotonic() > deadline - 30:
            break
        
        t_limit = time_alloc[idx]
        actual_limit = min(t_limit, deadline - time.monotonic() - 20)
        if actual_limit < 20:
            break
        
        print(f"\n--- Length n={n}, time_budget={actual_limit:.0f}s ---")
        
        # Get initial sequence
        if provided_seq is not None and np.isclose(n, n_prov):
            a0 = provided_seq.copy()
        elif provided_seq is not None:
            a0 = interpolate_sequence(provided_seq, n)
        else:
            a0 = make_initialization(n, 'chirp', rng)
        a0 = np.clip(a0, 0.01, None)
        
        # Run optimization pipeline
        result, score = refine_from_sequence(a0, actual_limit)
        update_best(result)
    
    # Final round: intensive optimization on the best sequence found
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            print(f"\n--- Final intensive optimization on best (n={len(best_seq)}), {remaining:.0f}s ---")
            # Run multiple short rounds with different random seeds
            for round_i in range(3):
                if time.monotonic() > deadline - 10:
                    break
                t_round = min(60.0, (deadline - time.monotonic() - 10) / (3 - round_i))
                if t_round < 10:
                    break
                
                # Slightly perturb the best sequence for diversity
                n = len(best_seq)
                a_perturbed = best_seq * np.exp(rng.normal(0, 0.03, size=n))
                a_perturbed = np.maximum(a_perturbed, 1e-10)
                
                result = exp_grad_optimize(a_perturbed, t_round * 0.5, lr_init=0.2)
                update_best(result)
                
                result = peak_shave(result, t_round * 0.3)
                update_best(result)
                
                result = simulated_annealing(result, t_round * 0.2)
                update_best(result)
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    return [1.0] * 256
