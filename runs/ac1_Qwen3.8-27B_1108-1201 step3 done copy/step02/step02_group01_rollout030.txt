"""
Optimized step function search using subgradient descent with active-set
methods, LP-based refinement, and coordinate descent. Exploits the insight
that monotone (ramp-like) initializations provide much better starting
points than random or oscillatory ones for minimizing normalized peak
autocorrelation of non-negative sequences.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / max(1, len(a))
    return a * S / s


def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 800:
        return None  # Too slow for large n
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None


def compute_subgradient(a, active_lags):
    """Compute the subgradient of max_k conv(a,a)[k] at given active lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        for i in range(n):
            j1 = k - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k
            if 0 <= j2 < n:
                grad[i] += a[j2]
    return grad


def compute_subgradient_fast(a, active_lags):
    """Fast subgradient using FFT-based approach for each lag."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        # conv(a,a)[k] = sum_i a[i]*a[k-i]
        # d/da[i] = a[k-i] + a[i-k] (with bounds)
        j1_start = max(0, k - (n - 1))
        j1_end = min(n - 1, k)
        # a[k-i] for i in [j1_start, j1_end] gives values a[k-j1_end]..a[k-j1_start]
        if j1_end >= j1_start:
            idx = np.arange(j1_start, j1_end + 1)
            grad[idx] += a[k - idx]
        j2_start = max(0, k - (n - 1))
        j2_end = min(n - 1, k)
        if k > 0:
            idx2 = np.arange(0, n)
            valid = (idx2 - k) >= 0 & (idx2 - k) < n
            if np.any(valid):
                grad[idx2[valid]] += a[idx2[valid] - k]
    return grad


def subgradient_descent(a, time_limit, verbose=True):
    """Subgradient descent with adaptive step size and active set tracking."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    lr = 1.0 / n
    patience = 0
    max_patience = 500
    history = []
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak = np.max(conv)
        peak_idx = int(np.argmax(conv))
        
        if peak < best_peak - 1e-10:
            best_peak = peak
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Find active lags (within 2% of peak)
        threshold = peak * 0.98
        active = np.where(conv >= threshold)[0]
        # Limit to at most 20 active lags for efficiency
        if len(active) > 20:
            active = active[np.argsort(conv[active])[-20:]]
        
        # Compute subgradient
        n = len(a)
        grad = np.zeros(n)
        for k in active:
            j1_start = max(0, k - (n - 1))
            j1_end = min(n - 1, k)
            if j1_end >= j1_start:
                idx = np.arange(j1_start, j1_end + 1)
                grad[idx] += a[k - idx]
            if k > 0:
                idx2 = np.arange(k, k + n)
                valid = idx2 < n
                if np.any(valid):
                    grad[idx2[valid]] += a[idx2[valid] - k]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
        
        # Adaptive learning rate
        if patience > max_patience // 2:
            lr *= 0.95
        elif patience < max_patience // 4:
            lr = min(lr * 1.01, 3.0 / n)
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        
        # Check if improvement
        new_conv_peak = np.max(fft_convolve(a_new, a_new))
        if new_conv_peak < peak + 0.01 * peak:
            a = a_new
        else:
            # Try smaller step
            a_new = a - lr * 0.1 * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, S)
            new_conv_peak = np.max(fft_convolve(a_new, a_new))
            if new_conv_peak < peak:
                a = a_new
            else:
                lr *= 0.9
                patience += 50
        
        history.append(best_peak)
        
        if verbose and len(history) % 200 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    SubGrad: best_peak={best_peak:.6f}, score={score:.6f}, "
                  f"lr={lr:.4f}, patience={patience}")
        
        if patience > max_patience * 3:
            break
    
    return best_a


def lp_blend_step(a, M_override=None):
    """One LP blending step. Returns improved a or None if no improvement."""
    n = len(a)
    if n > 800:
        return None
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    conv = fft_convolve(a, a)
    M = np.max(conv) if M_override is None else M_override
    
    g = lp_solve(a, M)
    if g is None:
        return None
    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return None
    
    c = normalize_seq(g, S)
    
    # Try several blend parameters
    best_t = 0.0
    best_conv_max = M
    for t_try in [0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
        b = (1 - t_try) * a + t_try * c
        b = np.maximum(b, 0.0)
        b = normalize_seq(b, S)
        b_conv_max = np.max(fft_convolve(b, b))
        if b_conv_max < best_conv_max:
            best_conv_max = b_conv_max
            best_t = t_try
    
    if best_t == 0.0:
        return None
    
    result = (1 - best_t) * a + best_t * c
    result = np.maximum(result, 0.0)
    return normalize_seq(result, S)


def lp_refine_loop(a, time_limit, verbose=True):
    """Repeated LP blending."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        result = lp_blend_step(a)
        if result is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        new_peak = np.max(fft_convolve(result, result))
        if new_peak < best_peak - 1e-10:
            best_peak = new_peak
            best_a = result.copy()
            a = result
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                break
        
        if verbose and no_improve % 50 == 0 and no_improve > 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    LP: peak={best_peak:.6f}, score={score:.6f}, stall={no_improve}")
    
    return best_a


def coordinate_descent(a, time_limit, verbose=True):
    """Coordinate descent: optimize each element one at a time."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(99)
    
    # Order: process coordinates that contribute most to the peak first
    conv = fft_convolve(a, a)
    peak_idx = int(np.argmax(conv))
    
    while time.monotonic() - t0 < time_limit:
        # Process coordinates in random order (to avoid cyclic patterns)
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try several multiplicative factors
            best_val = a[i]
            best_local_peak = np.max(fft_convolve(a, a))
            
            # Current contribution to peak
            for factor in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = normalize_seq(a_trial, S)
                trial_peak = np.max(fft_convolve(a_trial, a_trial))
                if trial_peak < best_local_peak - 1e-10:
                    best_local_peak = trial_peak
                    best_val = a_trial[i]
            
            if best_val != a[i]:
                a[i] = best_val
                a = normalize_seq(a, S)
                improved = True
                if np.max(fft_convolve(a, a)) < best_peak - 1e-10:
                    best_peak = np.max(fft_convolve(a, a))
                    best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_peak / (S * S)
            if int(elapsed) % 10 == 0:
                print(f"    CoordDesc t={elapsed:.0f}s: peak={best_peak:.6f}, score={score:.6f}")
        
        if not improved:
            # Try random perturbations when stuck
            idx = rng.choice(n, size=max(3, n // 10), replace=False)
            for j in idx:
                a[j] *= np.exp(rng.normal(0, 0.3))
            a = np.maximum(a, 0.0)
            a = normalize_seq(a, S)
            if np.max(fft_convolve(a, a)) < best_peak:
                best_peak = np.max(fft_convolve(a, a))
                best_a = a.copy()
    
    return best_a


def make_ramp(n, curvature=1.0, rng=None):
    """Create a ramp-like initial sequence with optional curvature."""
    t = np.linspace(0, 1, n)
    if curvature == 1.0:
        a = t + 0.1
    elif curvature == 0.5:
        a = np.sqrt(t) + 0.1
    elif curvature == 2.0:
        a = t**2 + 0.1
    elif curvature == 1.5:
        a = t**1.5 + 0.1
    else:
        a = t**curvature + 0.1
    
    if rng is not None and rng.random() < 0.5:
        a = a[::-1]  # Sometimes reverse
    
    a = np.maximum(a, 0.01)
    return a


def make_sigmoid_ramp(n, steepness, shift, rng=None):
    """Sigmoid-shaped ramp."""
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-steepness * (t - shift)))
    a = a / a.max() + 0.05
    return a


def mixed_optimize(a, time_limit, verbose=True):
    """
    Mixed optimization: alternates between subgradient, LP, and coordinate descent.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    # Phase A: Subgradient (40% of time)
    t_sub = min(time_limit * 0.35, (time_limit * 0.8 - 30))
    a_sub = subgradient_descent(a, t_sub, verbose=verbose)
    peak_sub = np.max(fft_convolve(a_sub, a_sub))
    if peak_sub < best_peak:
        best_peak = peak_sub
        best_a = a_sub.copy()
        a = a_sub
    if verbose:
        score = 2.0 * n * best_peak / (S * S)
        print(f"    After subgrad: peak={best_peak:.6f}, score={score:.6f}")
    
    remaining = (time_limit - 30) - (time.monotonic() - t0)
    if remaining < 10:
        return best_a
    
    # Phase B: LP refinement (30% of remaining)
    t_lp = min(remaining * 0.4, 120)
    a_lp = lp_refine_loop(a, t_lp, verbose=verbose)
    peak_lp = np.max(fft_convolve(a_lp, a_lp))
    if peak_lp < best_peak:
        best_peak = peak_lp
        best_a = a_lp.copy()
        a = a_lp
    if verbose:
        score = 2.0 * n * best_peak / (S * S)
        print(f"    After LP: peak={best_peak:.6f}, score={score:.6f}")
    
    remaining = (time_limit - 30) - (time.monotonic() - t0)
    if remaining < 10:
        return best_a
    
    # Phase C: Coordinate descent (25% of remaining)
    t_cd = min(remaining * 0.4, 120)
    a_cd = coordinate_descent(a, t_cd, verbose=verbose)
    peak_cd = np.max(fft_convolve(a_cd, a_cd))
    if peak_cd < best_peak:
        best_peak = peak_cd
        best_a = a_cd.copy()
        a = a_cd
    if verbose:
        score = 2.0 * n * best_peak / (S * S)
        print(f"    After coord desc: peak={best_peak:.6f}, score={score:.6f}")
    
    # Phase D: Final subgradient polish
    remaining = (time_limit - 20) - (time.monotonic() - t0)
    if remaining > 15:
        a_final = subgradient_descent(a, remaining, verbose=False)
        peak_final = np.max(fft_convolve(a_final, a_final))
        if peak_final < best_peak:
            best_peak = peak_final
            best_a = a_final.copy()
    
    return best_a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization starting from ramp initializations and the
    provided sequence. Uses subgradient descent, LP blending, and
    coordinate descent in an alternating fashion.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)
    
    # Generate initial candidates
    candidates = []
    
    # Ramp-based initializations at various lengths and curvatures
    ramp_configs = [
        (256, 1.0), (256, 1.5), (256, 0.7),
        (384, 1.0), (384, 1.5), (384, 0.7),
        (512, 1.0), (512, 1.5), (512, 0.7), (512, 2.0),
        (640, 1.0), (640, 1.5), (640, 0.7),
        (768, 1.0), (768, 1.5),
    ]
    
    for n, curv in ramp_configs:
        a = make_ramp(n, curv, rng)
        candidates.append((a, f"ramp_n{n}_c{curv}"))
    
    # Sigmoid ramps
    for n in [256, 384, 512, 640]:
        for steep in [3.0, 5.0, 8.0]:
            for shift in [0.3, 0.5, 0.7]:
                a = make_sigmoid_ramp(n, steep, shift, rng)
                candidates.append((a, f"sigmoid_n{n}_s{steep}_sh{shift}"))
    
    # Provided sequence at its native length and interpolated
    if provided_seq is not None:
        candidates.append((provided_seq.copy(), "provided_native"))
        for n_interp in [256, 384, 512, 640, 768, 1024]:
            if n_interp != len(provided_seq):
                a_interp = interpolate_sequence(provided_seq, n_interp)
                candidates.append((a_interp, f"provided_interp_n{n_interp}"))
    
    # Evaluate all candidates and sort by score
    print(f"\nGenerated {len(candidates)} initial candidates")
    scored_candidates = []
    for a, name in candidates:
        s = local_score(a)
        scored_candidates.append((s, a, name))
    scored_candidates.sort(key=lambda x: x[0])
    
    print("Top 10 candidates:")
    for s, a, name in scored_candidates[:10]:
        print(f"  {name}: score={s:.6f}, n={len(a)}")
    
    # Update best from candidates
    for s, a, name in scored_candidates[:5]:
        update_best(a)
    
    print(f"\nInitial best: {best_score:.6f}")
    
    # Allocate time to optimize top candidates
    remaining = deadline - time.monotonic()
    
    # Strategy: spend time on the top 4-6 candidates
    top_k = min(6, len(scored_candidates))
    # But don't spend too much time if we already beat the target
    if best_score < 1.5027:
        top_k = min(3, top_k)
    
    time_per_candidate = (remaining - 60) / top_k
    time_per_candidate = max(30, min(time_per_candidate, 300))
    
    print(f"\nWill optimize top {top_k} candidates, ~{time_per_candidate:.0f}s each")
    
    for rank in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        
        s_init, a_init, name = scored_candidates[rank]
        n = len(a_init)
        print(f"\n--- Optimizing candidate {rank+1}/{top_k}: {name} (n={n}, init_score={s_init:.6f}) ---")
        
        t_limit = min(time_per_candidate, deadline - time.monotonic() - 60)
        if t_limit < 20:
            break
        
        # Run mixed optimization
        result = mixed_optimize(a_init, t_limit, verbose=True)
        if update_best(result):
            print(f"  *** NEW BEST: score={best_score:.6f} (from {name}) ***")
    
    # Final polish on the best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            print(f"\n=== Final polish (time={remaining:.0f}s) ===")
            
            # Try coordinate descent on best
            n = len(best_seq)
            t_cd = min(remaining * 0.4, 60)
            result_cd = coordinate_descent(best_seq, t_cd, verbose=True)
            update_best(result_cd)
            
            remaining = deadline - time.monotonic() - 10
            if remaining > 15:
                t_sub = min(remaining * 0.5, 60)
                result_sub = subgradient_descent(best_seq, t_sub, verbose=True)
                update_best(result_sub)
            
            # Try LP if n is small enough
            remaining = deadline - time.monotonic() - 10
            if remaining > 20 and len(best_seq) <= 700:
                t_lp = min(remaining * 0.6, 60)
                result_lp = lp_refine_loop(best_seq, t_lp, verbose=True)
                update_best(result_lp)
            
            # One more targeted attempt: use the best as seed for a fresh subgradient run
            remaining = deadline - time.monotonic() - 5
            if remaining > 10:
                result = subgradient_descent(best_seq, remaining, verbose=False)
                update_best(result)
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, "
              f"local_score={best_score:.6f} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    # Ultimate fallback: simple ramp
    n = 512
    t = np.linspace(0.1, 1.0, n)
    return [float(x) for x in t]
