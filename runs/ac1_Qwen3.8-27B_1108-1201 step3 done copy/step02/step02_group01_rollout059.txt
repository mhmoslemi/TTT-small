Fast iterative optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses soft-max gradient descent with FFT-based
convolution, momentum, and adaptive step sizes, combined with targeted
peak-reduction strategies and structured perturbations.

import numpy as np

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a * S / s

def soft_max_grad(conv, tau):
    """Compute the gradient weights using soft-max."""
    # Shift to avoid overflow
    shifted = conv - np.max(conv)
    weights = np.exp(shifted / tau) / np.sum(np.exp(shifted / tau))
    return weights

def peak_targeted_descent(a, time_limit, verbose=True):
    """
    Target the specific peak lag and reduce contributions to it.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    lr = 0.02
    stall = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        for j in range(max(0, peak_idx - n + 1), min(peak_idx, n - 1) + 1):
            contrib[j] = a[peak_idx - j]
            if peak_idx - j != j:
                contrib[peak_idx - j] += a[j]
        
        # Direction: reduce elements contributing to peak, boost others
        # Weight by contribution
        total_contrib = np.sum(contrib)
        if total_contrib < 1e-12:
            break
        
        # Scale factor: we want to reduce the peak while maintaining sum
        # Reduce proportional to contribution, increase others proportionally to (max - contrib)
        max_contrib = np.max(contrib)

        # Add small positive for non-contributors to maintain diversity
        non_contrib_mask = contrib < 0.1 * max_contrib
        direction[non_contrib_mask] += 0.5
        
        # Normalize direction to have zero sum (to preserve sum)
        direction -= np.mean(direction)
        
        a_new = a + lr * direction * S
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        
        new_score = compute_score(a_new)

            best_score = new_score

            a = a_new
            stall = 0

def multiplicative_update(a, time_limit, verbose=True):

    t0 = time.monotonic()

    stall = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))
        
        # Compute "pressure" on each element from the peak
        pressure = np.zeros(n)
        for j in range(max(0, peak_idx - n + 1), min(peak_idx, n - 1) + 1):
            pressure[j] += a[peak_idx - j]
            if peak_idx - j != j:
                pressure[peak_idx - j] += a[j]

def structured_perturbation(a, rng):
    """Apply a structured perturbation to explore different regions."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    t = np.arange(n, dtype=np.float64)
    mod_type = rng.integers(0, 5)
    
    if mod_type == 0:
        # Add a small chirp
        freq = rng.uniform(0.1, 2.0)
        a_new = a * (1.0 + 0.05 * np.cos(2 * np.pi * freq * t / n))
    elif mod_type == 1:
        # Add a small sine
        freq = rng.integers(1, n // 4)
        amp = rng.uniform(0.01, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a_new = a + amp * a.max() * np.cos(2 * np.pi * freq * t / n + phase)
    elif mod_type == 2:
        # Window shape modulation
        center = n / 2
        sigma = n * rng.uniform(0.2, 0.8)
        a_new = a * (1.0 + 0.1 * np.exp(-0.5 * ((t - center) / sigma) ** 2))
    elif mod_type == 3:
        # Random block scaling
        num_blocks = rng.integers(3, 10)
        a_new = a.copy()
        for _ in range(num_blocks):
            start = int(rng.integers(0, n))
            length = int(rng.integers(1, n // num_blocks))
            scale = np.exp(rng.normal(0, 0.05))
            a_new[start:start + length] *= scale
    else:
        # Random smooth perturbation
        noise = rng.normal(0, 0.02, n)
        # Smooth it
        kernel = np.ones(5) / 5
        noise = np.convolve(noise, kernel, mode='same')
        a_new = a * np.exp(noise)
    
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, S)
    return a_new


def lp_lite(a, M, n_constraints=50):
    """
    A lightweight LP-like approach: only use the top constraint violations.
    Instead of solving a full LP, do a few rounds of projected gradient
    on the active constraints.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    
    conv = fft_convolve(a)
    # Get top k lags with highest convolution values
    top_lags = np.argsort(conv)[-n_constraints:]
    
    # For each top lag, compute the gradient
    total_grad = np.zeros(n)
    for k in top_lags:
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            total_grad[j] += a[k - j]
            if k - j != j:
                total_grad[k - j] += a[j]
    
    # Normalize and apply
    total_grad /= n_constraints
    direction = -total_grad
    direction -= np.mean(direction)
    
    a_new = a + 0.01 * direction * S

"""
FFT-based iterative optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses soft-max gradient descent with momentum,
peak-targeted coordinate descent, multiplicative updates, and structured
perturbations. All gradient computations use FFT for O(n log n) efficiency.
"""

import numpy as np
import time
import math


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def compute_soft_gradient(a, tau):
    """
    Compute the gradient of the soft-max of conv(a,a) using FFT.
    Returns grad where grad[i] = 2 * sum_j w[i+j] * a[j]
    with w = softmax(conv(a,a) / tau).
    """
    n = len(a)
    conv = fft_convolve(a)
    # Softmax weights (shifted for numerical stability)
    shifted = conv - np.max(conv)
    exp_vals = np.exp(shifted / tau)
    w = exp_vals / np.sum(exp_vals)
    
    # Compute cross-correlation (w ⋆ a) using FFT
    # (w ⋆ a)[i] = sum_j w[i+j] * a[j]
    # DFT: ifft(fft(w_padded) * conj(fft(a_padded)))
    N = 1
    while N < n + (2 * n - 1):
        N <<= 1
    
    w_padded = np.zeros(N)
    w_padded[:2 * n - 1] = w
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    fw = np.fft.rfft(w_padded)
    fa = np.fft.rfft(a_padded)
    
    cross_corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
    grad = 2.0 * cross_corr
    return grad, conv, w


def compute_peak_gradient(a):
    """
    Compute the gradient at the single peak location of conv(a,a).
    grad[i] = 2 * a[k*-i] if 0 <= k*-i < n, else 0, where k* = argmax.
    """
    n = len(a)
    conv = fft_convolve(a)
    peak_idx = int(np.argmax(conv))
    
    grad = np.zeros(n)
    # grad[i] = 2 * a[peak_idx - i] if 0 <= peak_idx - i < n
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, conv, peak_idx


def soft_gradient_descent(a, time_limit, tau_init=0.05, tau_min=0.002, lr_init=0.05, verbose=True):
    """
    Gradient descent with soft-max smoothing and momentum.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    tau = tau_init
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Adaptively decrease temperature
        if step > 0 and step % 200 == 0:
            tau = max(tau_min, tau * 0.9)
        
        grad, conv, w = compute_soft_gradient(a, tau)
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        a_new = a + v * (S / n)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        
        new_score = compute_score(a_new)
        if new_score < best_score - 1e-8:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase learning rate on success
            lr = min(lr * 1.02, 0.5)
        elif new_score < compute_score(a):
            a = a_new
            stall += 1
            lr *= 0.98
        else:
            stall += 1
            lr *= 0.95
            v *= 0.5
            if stall > 100:
                lr = max(lr * 0.5, 0.001)
                v = np.zeros(n)
                stall = 0
        
        step += 1
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        SoftGrad step={step}, t={elapsed:.1f}s, score={best_score:.6f}, tau={tau:.4f}, lr={lr:.4f}")
    
    return best_a, best_score

def multiplicative_update_descent(a, time_limit, verbose=True):
    """
    Multiplicative update: scale elements inversely to their contribution
    to the peak convolution value.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    rate = 0.03
    stall = 0
    step = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))
        
        # Pressure on each element from the peak
        pressure = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                pressure[i] = 2.0 * a[j]
        
        # Multiplicative update
        max_p = np.max(pressure)
        if max_p < 1e-12:
            break
        update = np.exp(-rate * pressure / max_p)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-10)
        a_new = normalize_seq(a_new, S)
        
        new_score = compute_score(a_new)
        if new_score < best_score - 1e-8:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try half rate
            update_half = np.exp(-rate * 0.5 * pressure / max_p)
            a_half = a * update_half
            a_half = np.maximum(a_half, 1e-10)
            a_half = normalize_seq(a_half, S)
            if compute_score(a_half) < best_score - 1e-8:
                best_score = compute_score(a_half)
                best_a = a_half.copy()
                a = a_half
                rate *= 0.5
                stall = 0
            else:
                rate *= 0.5
                stall += 1
                if stall > 80:
                    a = a * np.exp(rng.normal(0, 0.03, n))
                    a = np.maximum(a, 1e-10)
                    a = normalize_seq(a, S)
                    rate = 0.03
                    stall = 0
        
        step += 1
        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        MultUpd step={step}, t={elapsed:.1f}s, score={best_score:.6f}, rate={rate:.4f}")
    
    return best_a, best_score

def structured_perturbation(a, rng):
    """Apply a structured perturbation to explore different regions."""
    n = len(a)
    S = np.sqrt(2.0 * n)

    if mod_type == 0:
        freq = rng.uniform(0.1, 3.0)
        a_new = a * (1.0 + 0.04 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 6.28)))
    elif mod_type == 1:
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.01, 0.04)
        a_new = a + amp * np.max(a) * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 6.28))
    elif mod_type == 2:
        center = n * rng.uniform(0.2, 0.8)
        sigma = n * rng.uniform(0.1, 0.5)
        a_new = a * (1.0 + 0.08 * np.exp(-0.5 * ((t - center) / sigma) ** 2))
    elif mod_type == 3:
        num_blocks = int(rng.integers(3, 12))
        a_new = a.copy()
        for _ in range(num_blocks):
            start = int(rng.integers(0, n))
            length = int(rng.integers(1, max(2, n // num_blocks)))
            scale = np.exp(rng.normal(0, 0.04))
            a_new[start:min(n, start + length)] *= scale
    elif mod_type == 4:
        noise = rng.normal(0, 0.015, n)
        kernel_size = max(1, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        a_new = a * np.exp(noise)
    else:
        # Antisymmetric-ish perturbation (small)
        amp = rng.uniform(0.005, 0.02)
        a_new = a * (1.0 + amp * np.sin(2 * np.pi * t / n * int(rng.integers(1, 10))))
    
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, S)
    return a_new


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining FFT-based gradient methods:
    1. Soft-max gradient descent (smooth, fast)
    2. Peak-targeted descent (focused on the maximum)
    3. Multi-peak gradient descent (addresses top-k peaks)
    4. Multiplicative updates (robust for non-negative constraints)
    5. Structured perturbation exploration
    6. Final fine-tuning
    
    All methods use FFT for O(n log n) gradient computation.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 40.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        provided_score = compute_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={provided_score:.6f}")
        best_score = provided_score
        best_seq = provided_seq.copy()
    
    # Prepare candidate list
    candidates = []
    if provided_seq is not None:
        candidates.append(('native', provided_seq.copy()))
        for n_try in [512, 768, 896, 1024]:
            a_interp = interpolate_sequence(provided_seq, n_try)
            candidates.append((f'interp_{n_try}', np.maximum(a_interp, 0.001)))
    
    # Fresh structured initializations
    for n_fresh in [640, 768, 1024]:
        t = np.arange(n_fresh, dtype=np.float64)
        # Chirp
        a_chirp = np.maximum(1.0 + 0.3 * np.cos(np.pi * t * t / (n_fresh * 0.25)), 0.01)
        candidates.append((f'chirp_{n_fresh}', a_chirp))
        # Multi-tone
        a_mt = np.ones(n_fresh)
        for f in [2, 5, 11, 23]:
            a_mt += 0.08 * np.cos(2 * np.pi * f * t / n_fresh + rng.uniform(0, 6.28))
        candidates.append((f'multitone_{n_fresh}', np.maximum(a_mt, 0.01)))
        # Random smooth
        noise = rng.normal(0, 0.1, n_fresh)
        kernel = np.ones(max(1, n_fresh // 50)) / (n_fresh // 50)
        noise = np.convolve(noise, kernel, mode='same')
        a_rand = np.maximum(1.0 + noise, 0.01)
        candidates.append((f'rand_{n_fresh}', a_rand))
    
    # Sort candidates by initial score
    candidates_with_scores = []
    for name, a in candidates:
        s = compute_score(a)
        candidates_with_scores.append((s, name, a))
    candidates_with_scores.sort()
    
    print(f"\nTop candidates:")
    for s, name, a in candidates_with_scores[:5]:
        print(f"  {name}: n={len(a)}, score={s:.6f}")
    
    # Phase 1: Soft-gradient descent on top candidates
    print(f"\n=== Phase 1: Soft-gradient descent ===")
    time_remaining = deadline - time.monotonic()
    n_top = min(3, len(candidates_with_scores))
    time_per = min(250.0, time_remaining / max(1, n_top) - 10)
    
    for i in range(n_top):
        if time.monotonic() > deadline - 200:
            break
        s, name, a = candidates_with_scores[i]
        t_limit = min(time_per, deadline - time.monotonic() - 150)
        if t_limit < 10:
            break
        print(f"\n  Soft-gradient on {name} (n={len(a)}), limit={t_limit:.0f}s")
        result, res_score = soft_gradient_descent(a, t_limit, tau_init=0.05, tau_min=0.003,
                                                   lr_init=0.05, verbose=True)
        if res_score < best_score:
            best_score = res_score
            best_seq = result.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 2: Peak-targeted descent
    print(f"\n=== Phase 2: Peak-targeted descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_limit = min(200.0, remaining * 0.5)
            print(f"  Peak-targeted (n={len(best_seq)}), limit={t_limit:.0f}s")
            result, res_score = peak_targeted_descent(best_seq, t_limit, verbose=True)
            if res_score < best_score:
                best_score = res_score
                best_seq = result.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 3: Multi-peak gradient descent
    print(f"\n=== Phase 3: Multi-peak gradient descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_limit = min(180.0, remaining * 0.5)
            print(f"  Multi-peak (n={len(best_seq)}), limit={t_limit:.0f}s")
            result, res_score = multi_peak_gradient_descent(best_seq, t_limit, top_k=8, verbose=True)
            if res_score < best_score:
                best_score = res_score
                best_seq = result.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 4: Multiplicative updates
    print(f"\n=== Phase 4: Multiplicative updates ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_limit = min(150.0, remaining * 0.5)
            print(f"  Mult-update (n={len(best_seq)}), limit={t_limit:.0f}s")
            result, res_score = multiplicative_update_descent(best_seq, t_limit, verbose=True)
            if res_score < best_score:
                best_score = res_score
                best_seq = result.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # Phase 5: Exploration with perturbation + quick gradient
    print(f"\n=== Phase 5: Exploration ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 15:
            t_limit = remaining * 0.7
            print(f"  Exploration, limit={t_limit:.0f}s")
            t_exp = time.monotonic()
            n_exp = len(best_seq)
            S_exp = np.sqrt(2.0 * n_exp)
            exp_stall = 0
            
            while time.monotonic() - t_exp < t_limit:
                a_pert = structured_perturbation(best_seq, rng)
                s_pert = compute_score(a_pert)
                
                if s_pert < best_score:
                    best_score = s_pert
                    best_seq = a_pert.copy()
                    exp_stall = 0
                else:
                    # If close, try quick gradient fix
                    if s_pert < best_score * 1.01:
                        a_fix, s_fix = soft_gradient_descent(a_pert, 3.0,
                                                              tau_init=0.02, tau_min=0.005,
                                                              lr_init=0.03, verbose=False)
                        if s_fix < best_score:
                            best_score = s_fix
                            best_seq = a_fix.copy()
                            exp_stall = 0
                        else:
                            exp_stall += 1
                    else:
                        exp_stall += 1
                
                if exp_stall > 40:
                    # Bigger perturbation from best
                    best_seq = best_seq * np.exp(rng.normal(0, 0.04, n_exp))
                    best_seq = np.maximum(best_seq, 0.0)
                    best_seq = normalize_seq(best_seq, S_exp)
                    # Quick gradient clean-up
                    best_seq, best_score = soft_gradient_descent(best_seq, 2.0,
                                                                  tau_init=0.03, tau_min=0.01,
                                                                  lr_init=0.05, verbose=False)
                    exp_stall = 0
                
                if int((time.monotonic() - t_exp)) % 30 == 0:
                    print(f"    Explo t={time.monotonic()-t_exp:.0f}s, best={best_score:.6f}")
    
    # Phase 6: Final alternation of methods
    print(f"\n=== Phase 6: Final refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            # Alternate between peak-targeted and multi-peak
            methods = [peak_targeted_descent, multi_peak_gradient_descent, soft_gradient_descent]
            for method in methods:
                if time.monotonic() > deadline - 15:
                    break
                t_limit = min(30.0, deadline - time.monotonic() - 15)
                if t_limit < 5:
                    break
                print(f"  {method.__name__}, limit={t_limit:.0f}s")
                if method.__name__ == 'multi_peak_gradient_descent':
                    result, res_score = method(best_seq, t_limit, top_k=6, verbose=False)
                elif method.__name__ == 'soft_gradient_descent':
                    result, res_score = method(best_seq, t_limit, tau_init=0.01, tau_min=0.002,
                                               lr_init=0.02, verbose=False)
                else:
                    result, res_score = method(best_seq, t_limit, verbose=False)
                if res_score < best_score:
                    best_score = res_score
                    best_seq = result.copy()
                    print(f"    *** New best: {best_score:.6f} ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    return [1.0] * 256
