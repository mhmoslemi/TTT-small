Uses analytical subgradient with Adam, peak-shaving local search,

    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    """Adam gradient descent on softmax parameterization."""
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    
    # Initialize theta from a_init
    a_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(a_safe / np.sum(a_safe))
    
    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.2 / n
    
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    step = 0
    best_a = a_init.copy()
    best_score = local_score(a_init)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        # Compute a from theta
        t_shift = theta - np.max(theta)
        e = np.exp(t_shift)
        a = e / np.sum(e) * S_target
        
        # Compute gradient w.r.t. theta
        dF_da = gradient_step(a, tau=tau, top_k=top_k)
        G_sum = np.dot(dF_da, a) / S_target
        dF_dtheta = a * (dF_da - G_sum)
        
        # Adam update
        step += 1
        m1 = b1 * m1 + (1 - b1) * dF_dtheta
        m2 = b2 * m2 + (1 - b2) * dF_dtheta ** 2
        m1_hat = m1 / (1 - b1**step)
        m2_hat = m2 / (1 - b2**step)
        
        lr_t = lr / np.sqrt(1 + 0.01 * step)
        theta = theta - lr_t * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Check score
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
