"""
Optimized step function search using asymmetric (geometric/power-law) initialization,
LP-based refinement, subgradient descent, and coordinate descent.
Key insight: asymmetric sequences (tilted toward one end) achieve lower peak autocorrelation
than symmetric windows, as they better equalize convolution values across lags.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build matrix A where A@b = conv(f,b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_cross(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_peakeq(f, M, timeout=5.0):
    """
    Alternative LP: min M' s.t. conv(f,b) <= M'*1, b >= 0, sum(b) = sum(f).
    Returns (b, M') or (None, None).
    """
    n = len(f)
    S = float(np.sum(f))
    A = build_conv_matrix(f)
    # Variables: (b_0, ..., b_{n-1}, t) where t is the peak bound
    # Minimize: t
    # Subject to: A@b <= t*1, sum(b) = S, b >= 0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    # A@b - t*1 <= 0  =>  A - [1,1,...,1]^T as columns
    A_ub = np.column_stack([A, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(b) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None


def make_geometric(n, r):
    """Geometric sequence a[i] = r^i."""
    return np.array([r ** i for i in range(n)], dtype=np.float64)


def make_power_law(n, p):
    """Power law: a[i] = ((n-1-i)/(n-1))^p, decreasing."""
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.array([1.0])
    return ((n - 1 - t) / (n - 1)) ** p


def make_asymmetric_window(n, t_param, p=1.0):
    """
    Asymmetric window: a[i] = (1 + t*(1 - 2i/(n-1)))^p * w_sym[i]
    where w_sym is a cosine window.
    """
    if n == 1:
        return np.array([1.0])
    t = np.arange(n, dtype=np.float64)
    asym = 1.0 + t_param * (1.0 - 2.0 * t / (n - 1))
    asym = np.clip(asym, 0.01, None)
    w = 0.5 * (1.0 + np.cos(np.pi * t / (n - 1)))
    a = (asym ** p) * (0.5 + 0.5 * w)
    return np.clip(a, 0.01, None)


def make_mixed(n, alpha, beta):
    """Mix of geometric and uniform: a[i] = alpha * r^i + beta, r = 1 - 2/n."""
    r = 1.0 - 2.0 / n
    t = np.arange(n, dtype=np.float64)
    a = alpha * (r ** t) + beta
    return np.clip(a, 0.01, None)


def parameter_sweep(n, time_limit):
    """Sweep parameters of parametric families to find best initial candidate."""
    t0 = time.monotonic()
    best_score = np.inf
    best_seq = None

    # Geometric sweep
    for r in np.arange(0.3, 0.995, 0.01):
        if time.monotonic() - t0 > time_limit:
            break
        a = make_geometric(n, r)
        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()

    # Power law sweep
    for p in np.arange(0.1, 5.0, 0.1):
        if time.monotonic() - t0 > time_limit:
            break
        a = make_power_law(n, p)
        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()

    # Asymmetric window sweep
    for t_param in np.arange(-1.5, 1.5, 0.1):
        if time.monotonic() - t0 > time_limit:
            break
        for p in [0.5, 1.0, 1.5, 2.0]:
            a = make_asymmetric_window(n, t_param, p)
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_seq = a.copy()

    # Mixed sweep
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5]:
        if time.monotonic() - t0 > time_limit:
            break
        for beta in [0.1, 0.3, 0.5, 1.0]:
            a = make_mixed(n, alpha, beta)
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_seq = a.copy()

    print(f"    Parameter sweep n={n}: best_score={best_score:.6f} (time={time.monotonic()-t0:.1f}s)")
    return best_seq, best_score


def optimize_lp_steps(f_init, time_limit, t_step=0.05, verbose=False):
    """LP-iteration optimizer with cross-correlation constraint."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        b = solve_lp_cross(f, M, timeout=4.0)
        if b is None:
            break

        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        # Line search
        best_t = t_step
        best_s = np.inf
        for t_try in [t_step, 0.3 * t_step, 0.7 * t_step, 1.3 * t_step]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      [{elapsed:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f}")

        if stall >= 60:
            rng_esc = np.random.default_rng(42 + it * 7919)
            f += 0.02 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0

    return best_f, best_score


def subgradient_descent(f_init, time_limit, verbose=False):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (np.sqrt(2.0 * n) / S)  # normalize
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta_m = 0.85
    lr = 0.005
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        S = float(np.sum(f))
        M = float(np.max(c))

        # Top-k peaks for smoother subgradient
        top_k = min(10, len(c))
        top_idx = np.argsort(c)[-top_k:]
        top_mask = np.zeros(len(c), dtype=bool)
        top_mask[top_idx] = True
        # Weight by how close to max
        weights = np.zeros(len(c))
        for k in top_idx:
            weights[k] = c[k] / M  # normalized weight

        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            w = weights[ki]
            for j in range(lo, hi + 1):
                g[j] += w * f[ki - j]

        g /= max(1.0, np.sum(weights))
        # Full gradient of score = 2n*M/S^2
        # d(score)/da[j] = 2n*g[j]/S^2 - 4n*M*a[j]/S^3
        grad = 2.0 * n * g / (S * S) - 4.0 * n * M * f / (S * S * S)

        momentum = beta_m * momentum + (1.0 - beta_m) * grad
        f = f - lr * momentum
        f = np.clip(f, 0.001, 1000.0)
        f = f * (np.sqrt(2.0 * n) / float(np.sum(f)))

        it += 1
        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"      [subgrad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
            # Adaptive learning rate
            if s > best_score * 1.05:
                lr *= 0.7
            elif s < best_score * 0.99:
                lr *= 1.1

        if it % 5000 == 0 and lr < 0.0001:
            lr = 0.001

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        # Pick a random coordinate or the one with largest gradient contribution
        if rng.random() < 0.3:
            c = fft_convolve(f, f)
            k_star = int(np.argmax(c))
            # The coordinates that contribute most to c[k_star]
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            candidates = list(range(lo, hi + 1))
            j = int(rng.choice(candidates))
        else:
            j = int(rng.integers(0, n))

        # Try to optimize f[j] by trying nearby values
        best_val = f[j]
        best_s = score_fast(f)
        for scale in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.3]:
            f[j] = f[j] * scale if f[j] > 0 else 0.01
            s = score_fast(f)
            if s < best_s:
                best_s = s
                best_val = f[j]
        f[j] = best_val

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_f = f.copy()

        it += 1
        if verbose and it % 1000 == 0:
            print(f"      [coord {time.monotonic()-t0:6.1f}s] it={it} best={best_score:.6f}")

    return best_f, best_score


def lp_peakeq_iteration(f_init, time_limit, verbose=False):
    """LP iteration using peak-equality formulation."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))
        S = float(np.sum(f))

        b, M_new = solve_lp_peakeq(f, M, timeout=4.0)
        if b is None:
            break

        # Blend towards b
        for t_try in [0.1, 0.3, 0.5, 0.7]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_score:
                f = f_cand.copy()
                best_score = s_cand
                best_f = f.copy()
                break
        else:
            # Accept the smallest step
            f_cand = (1.0 - 0.1) * f + 0.1 * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand <= best_score:
                f = f_cand.copy()
                if s_cand < best_score:
                    best_score = s_cand
                    best_f = f.copy()
            else:
                break

        it += 1
        if verbose and it % 10 == 0:
            print(f"      [lp_peakeq {time.monotonic()-t0:6.1f}s] it={it} best={best_score:.6f}")

    return best_f, best_score

        print(f"{'='*60}")
        return final_list

    return [1.0] * 256
