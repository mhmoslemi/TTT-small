    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

The problem is: minimize max_k c_k subject to sum(a) = 1, a >= 0.

This is a convex optimization problem (minimize a convex function over a convex set). The optimal solution satisfies KKT conditions.

The Lagrangian is: L = max_k c_k - λ(sum(a) - 1) - μ^T a

At the optimum, the subgradient of max_k c_k must be in the span of {1, active constraint indicators}.

"""
Smoothed subgradient descent with Nesterov acceleration for minimizing peak
autocorrelation of non-negative step functions. Uses softmax-smoothed gradient
of the convolution peak, adaptive step sizes, multiple starting points, and
coordinate-wise local search polish.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (fast version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s ** 2)


def softmax_gradient(c, a, T):
    """Compute the smoothed gradient of the peak using softmax weights."""
    Tc = T * c
    Tc -= np.max(Tc)
    w = np.exp(Tc)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    n = len(a)
    grad = np.correlate(w, a, mode='valid')
    return grad


def hard_subgradient(c, a):
    """Compute the hard subgradient at all peak positions."""
    n = len(a)
    max_c = np.max(c)
    peak_mask = c >= max_c * (1.0 - 1e-6)
    peak_indices = np.where(peak_mask)[0]
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += a[idx]
    grad /= len(peak_indices)
    return grad


def nesterov_smoothed_descent(a_init, time_limit, rng, T_init=30.0, T_max=300.0,
                              momentum_init=0.80, verbose=False):
    """Smoothed subgradient descent with Nesterov acceleration."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)

    T = T_init
    momentum = momentum_init
    v = np.zeros(n)
    step_size = 0.04
    stall_count = 0
    no_improve = 0

    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_val = np.max(c)

        # Compute smoothed gradient
        grad = softmax_gradient(c, a, T)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Backtracking line search (limited)
        alpha = step_size
        improved = False
        for _ in range(8):
            a_trial = a - alpha * v
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = np.sum(a_trial)
            if s_trial < 0.01:
                alpha *= 0.5
                continue
            a_trial = a_trial / s_trial
            c_trial = fft_convolve(a_trial, a_trial)
            if np.max(c_trial) < peak_val:
                a = a_trial
                peak_val = np.max(c_trial)
                improved = True
                break
            alpha *= 0.5

        if improved:
            step_size = min(step_size * 1.01, 0.15)
            stall_count = 0
            no_improve += 1
        else:
            stall_count += 1
            step_size *= 0.85
            if step_size < 1e-6:
                step_size = 0.02
                v = np.zeros(n)
                stall_count = 0
            if stall_count > 150:
                # Perturb to escape
                scale = rng.uniform(0.02, 0.06)
                a = a * np.exp(rng.normal(0.0, scale, n))
                a = np.maximum(a, 0.0)
                a = a / np.sum(a)
                v = np.zeros(n)
                step_size = 0.03
                T = max(20.0, T * 0.7)
                momentum = max(0.7, momentum - 0.05)
                stall_count = 0
                no_improve = 0

        # Gradually increase T and momentum
        T = min(T * 1.003, T_max)
        momentum = min(momentum * 1.0005, 0.92)

        # Update best
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0

        iteration += 1
        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iteration}, score={best_score:.6f}, T={T:.1f}, "
                  f"alpha={step_size:.4f}, mom={momentum:.3f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def coordinate_polish(a_init, time_limit, rng, verbose=False):
    """Coordinate-wise local search: try adjusting individual elements."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        # Random coordinate selection (focus on high-contribution elements)
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        # Find which coordinates contribute most to the peak
        contribs = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            idx = peak_idx - j
            if 0 <= idx < n:
                contribs[j] = a[j] * a[idx]

        # Sample coordinates to try: mix of high-contribution and random
        n_tries = min(n, 64)
        high_contrib_idx = np.argsort(contribs)[-n_tries // 2:]
        random_idx = rng.choice(n, size=n_tries // 2, replace=False)
        try_idx = np.concatenate([high_contrib_idx, random_idx])
        np.random.shuffle(try_idx) if hasattr(np, 'random') else rng.shuffle(try_idx)

        improved_this_sweep = False
        for j in try_idx:
            if time.monotonic() - t0 > time_limit:
                break
            # Try several multiplicative adjustments
            best_local = a[j]
            best_local_score = best_score
            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_trial = a.copy()
                a_trial[j] = a[j] * factor
                s_t = np.sum(a_trial)
                if s_t < 0.01:
                    continue
                a_trial = a_trial / s_t
                sc = compute_score(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = a_trial

            if best_local_score < best_score:
                a = best_local.copy() if isinstance(best_local, np.ndarray) else a
                a[j] = best_local
                s = np.sum(a)
                a = a / s
                best_score = best_local_score
                best_a = a.copy()
                improved_this_sweep = True

        sweeps += 1
        if not improved_this_sweep and sweeps % 50 == 0:
            # Add small noise to escape
            a = a * np.exp(rng.normal(0.0, 0.01, n))
            a = np.maximum(a, 0.0)
            a = a / np.sum(a)

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    coord sweep={sweeps}, score={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def simulated_annealing_polish(a_init, time_limit, rng, verbose=False):
    """Simulated annealing for final polish."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / np.sum(a)

    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score

    temperature = 0.002
    cooling = 0.9998
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.25:
            # Peak-targeted reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            contribs = np.zeros(n)
            for j in range(lo, hi + 1):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = a[j] * a[idx]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 3:
                n_inc = min(8, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width + 1))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.60:
            # Multi-small-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width + 1))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.01, size=width))
        elif r < 0.75:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * a[i]
                candidate[i] = max(0.0, a[i] - delta)
                candidate[j] = max(0.0, a[j] + delta)
        elif r < 0.88:
            # Triple redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) == 3:
                delta = rng.uniform(-0.08, 0.08) * a[i]
                candidate[i] = max(0.0, a[i] - 2 * delta)
                candidate[j] = max(0.0, a[j] + delta)
                candidate[k] = max(0.0, a[k] + delta)
        else:
            # Small global noise
            candidate = a * np.exp(rng.normal(0.0, 0.005, n))

        candidate = np.maximum(candidate, 0.0)
        s_c = np.sum(candidate)
        if s_c < 0.01:
            continue
        candidate = candidate / s_c
        new_score = compute_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            accept_prob = np.exp(-(new_score - current_score) / max(temperature, 1e-12))
            if rng.random() < accept_prob:
                a = candidate
                current_score = new_score

        temperature = max(temperature * cooling, 1e-6)
        step += 1

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA step={step}, score={best_score:.6f}, T={temperature:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng, n_tones=5):
    """Create a multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, n_tones + 1):
        amp = 0.2 / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Create a random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq), endpoint=True)
    x_new = np.linspace(0, 1, new_n, endpoint=True)
    return np.interp(x_new, x_old, seq)


def lp_direction_step(f, M, time_limit, rng):
    """One LP-based direction step (for final polish)."""
    n = len(f)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f

    from scipy.optimize import linprog
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    t0 = time.monotonic()
    remaining = time_limit - (time.monotonic() - t0)
    if remaining < 30:
        return f, compute_score(f)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        b_0 = np.maximum(res.x, 0.0)
        sum_b = np.sum(b_0)
        if sum_b > 1e-10:
            S = np.sqrt(2.0 * n)
            g = (S / sum_b) * b_0
            # Line search
            S_f = np.sum(f)
            f_norm = f * (S / S_f)
            best_f = f_norm.copy()
            best_s = compute_score(f_norm)
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                f_new = (1 - t) * f_norm + t * g
                f_new = np.maximum(f_new, 0)
                if np.sum(f_new) < 0.01:
                    continue
                sc = compute_score(f_new)
                if sc < best_s:
                    best_s = sc
                    best_f = f_new
            return best_f, best_s
    return f, compute_score(f)


def propose_candidate(seed=42, budget_s=1800, **kwargs):

                                                    T_init=20.0, T_max=200.0,

        print(f"  n={n}, init={init_type}: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    if best_seq is not None and time.monotonic() < deadline - 30:
        sa_time = min(150.0, max(15.0, deadline - time.monotonic() - 20))
        print(f"  SA time={sa_time:.0f}s, score={best_score:.6f}")
        polished, polish_score = simulated_annealing_polish(best_seq, sa_time, rng, verbose=False)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After SA: {best_score:.6f}")

    # Phase 6: Final Nesterov run with high T
    print(f"\n=== Phase 6: Final Nesterov (high T) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        final_time = min(100.0, max(10.0, deadline - time.monotonic() - 10))
        f_best, s_best = nesterov_smoothed_descent(best_seq, final_time, rng,
                                                    T_init=100.0, T_max=500.0,
                                                    momentum_init=0.85, verbose=False)
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  After final Nesterov: {best_score:.6f}")

    # Phase 7: Final short SA if time permits
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            print(f"\n=== Phase 7: Final SA ({remaining:.0f}s) ===")
            polished, polish_score = simulated_annealing_polish(best_seq, remaining, rng,
                                                                 verbose=False)
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polished.copy()
                print(f"  Final: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
