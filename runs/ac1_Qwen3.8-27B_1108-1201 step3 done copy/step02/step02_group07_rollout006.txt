Multi-phase optimization for minimizing peak convolution of non-negative sequences.
Uses LP-based direction finding with geometric mean updates, subgradient polish,
and peaking-based local search across multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=3.0):
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def geometric_update(f, g, t, floor=1e-6):
    """Geometric mean update: f_new = f * (g/f)^t"""
    f_safe = np.maximum(f, floor)
    g_safe = np.maximum(g, floor)
    f_new = f_safe * np.power(g_safe / f_safe, t)
    return np.clip(f_new, 0.0, 1000.0)


def additive_update(f, g, t):
    """Additive update: f_new = (1-t)*f + t*g"""
    f_new = (1.0 - t) * f + t * g
    return np.clip(f_new, 0.0, 1000.0)


def line_search(f, g, f_score, max_t=1.0, min_t=1e-4, n_trials=8, use_geometric=True):
    """Find best step size t in [min_t, max_t]"""

    best_s = f_score

    for t in np.geomspace(min_t, max_t, n_trials):
        if use_geometric:
            f_new = geometric_update(f, g, t)
        else:
            f_new = additive_update(f, g, t)
        s = score_fast(f_new)
        if s < best_s:
            best_s = s
            best_t = t
    return best_t, best_s, (best_t if not use_geometric else None)

"""
Multi-phase optimization for minimizing peak convolution score of non-negative sequences.
Combines LP-based direction finding with geometric mean updates, adaptive line search,
subgradient polishing, and peaking-targeted local search across multiple lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=2.5):
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def geometric_update(f, g, t, floor=1e-8):
    f_safe = np.maximum(f, floor)
    g_safe = np.maximum(g, floor)
    f_new = f_safe * np.power(g_safe / f_safe, t)
    return np.clip(f_new, 0.0, 1000.0)

        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        k = min(5, len(c))
        top_idx = np.argsort(c)[-k:]

        g = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += f[ki - j]

        g -= np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-12:
            g = g / g_norm

        momentum = beta * momentum + (1.0 - beta) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, None)
        f = f * (float(n) / float(np.sum(f)))

        it += 1
        if it % 500 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                if verbose:
                    print(f"    [subgrad {time.monotonic()-t0:6.1f}s] it={it} new_best={s:.6f}")
        lr *= lr_decay

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def peaking_search(f_init, time_limit, verbose=False):
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)
    it = 0
    temp = 0.003

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))

        contribs = np.zeros(n)
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            contribs[j] = f[j] * f[peak_idx - j]

        candidate = f.copy()
        r = rng.random()

        if r < 0.5:
            # Reduce top contributors to the peak
            n_red = max(3, n // 12)
            top_idx = np.argsort(contribs)[-n_red:]
            candidate[top_idx] *= np.exp(rng.normal(-0.015, 0.01, size=n_red))
            # Increase some bottom contributors
            n_inc = max(3, n // 18)
            bot_idx = np.argsort(contribs)[:n_inc]
            bot_idx = bot_idx[~np.isin(bot_idx, top_idx)]
            if len(bot_idx) > 0:
                candidate[bot_idx] *= np.exp(rng.normal(0.015, 0.01, size=len(bot_idx)))
        elif r < 0.8:
            # Local window perturbation
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0, 0.01, n))

        candidate = np.clip(candidate, 0.001, 1000.0)
        new_score = score_fast(candidate)
        cur_score = score_fast(f)

        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-15)):
            f = candidate

        temp *= 0.99995
        it += 1

        if verbose and it % 100 == 0:
            print(f"    [peaking {time.monotonic()-t0:6.1f}s] it={it} cur={cur_score:.6f} best={best_score:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Main search: multi-start LP optimization + subgradient + peaking polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Evaluate provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    # Phase 1: LP optimization from multiple starts and lengths
    print("\n=== Phase 1: LP optimization ===")

    tasks = []
    if best_seq is not None:
        tasks.append((len(best_seq), 'provided'))

    # Key lengths to try
    for n in [128, 192, 256, 320, 384, 512, 640, 768]:
        tasks.append((n, 'hann'))
    for n in [256, 384, 512]:
        tasks.append((n, 'flat'))
        tasks.append((n, 'triangular'))
    for n in [384, 512, 768]:
        tasks.append((n, 'kaiser'))

    # Time budget for phase 1: 55% of total
    phase1_budget = 0.55 * (budget_s - 12.0)
    time_per_task = phase1_budget / len(tasks)
    time_per_task = max(8.0, min(time_per_task, 100.0))

    for idx, (n, init_type) in enumerate(tasks):
        remaining = deadline - time.monotonic() - 0.25 * (budget_s - 12.0)
        if remaining < 10:
            break

        # Generate initialization
        if init_type == 'provided' and best_seq is not None and len(best_seq) == n:
            f0 = best_seq.copy()
        elif init_type == 'hann':
            f0 = make_hann(n, rng, noise=0.04)
        elif init_type == 'flat':
            f0 = make_flat(n, rng, noise=0.08)
        elif init_type == 'triangular':
            f0 = make_triangular(n, rng, noise=0.04)
        elif init_type == 'kaiser':
            f0 = make_kaiser_like(n, rng, noise=0.03)
        else:
            f0 = make_hann(n, rng, noise=0.05)

        t_alloc = min(time_per_task, remaining)
        if t_alloc < 5:
            continue

        use_geo = (n <= 512)  # geometric update works well for moderate n
        print(f"  [{idx+1}/{len(tasks)}] n={n:4d} init={init_type:10s} t={t_alloc:.0f}s")
        f_best, s_best = lp_optimize(f0, t_alloc, use_geometric=use_geo, verbose=(n <= 256 and idx < 3))

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}, {init_type}) ***")
        else:
            print(f"  score={s_best:.6f} (best={best_score:.6f})")

    # Phase 2: Subgradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        pt = min(90.0, deadline - time.monotonic() - 50)
        if pt > 10:
            print(f"\n=== Phase 2: Subgradient polish ({pt:.0f}s) ===")
            f_pol, s_pol = subgradient_polish(best_seq, pt, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = f_pol.copy()
                print(f"  Polished: {best_score:.6f}")
            else:
                print(f"  No improvement: {s_pol:.6f} vs {best_score:.6f}")

    # Phase 3: Peaking search
    if best_seq is not None and time.monotonic() < deadline - 35:
        ps_t = min(70.0, deadline - time.monotonic() - 25)
        if ps_t > 10:
            print(f"\n=== Phase 3: Peaking search ({ps_t:.0f}s) ===")
            f_ps, s_ps = peaking_search(best_seq, ps_t, verbose=True)
            if s_ps < best_score:
                best_score = s_ps
                best_seq = f_ps.copy()
                print(f"  After peaking: {best_score:.6f}")
            else:
                print(f"  No improvement: {s_ps:.6f} vs {best_score:.6f}")

    # Phase 4: Final LP refinement
    if best_seq is not None and time.monotonic() < deadline - 8:
        fl_t = min(50.0, deadline - time.monotonic() - 5)
        if fl_t > 8:
            print(f"\n=== Phase 4: Final LP refinement ({fl_t:.0f}s) ===")
            f_fin, s_fin = lp_optimize(best_seq, fl_t, use_geometric=True, verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  Final: {best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    return [1.0] * 256
