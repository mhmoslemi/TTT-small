"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization with
log-sum-exp smoothing of the max function, softmax parameterization for
non-negativity and normalization, multi-start with beta annealing,
and multi-length exploration.

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a, fft_size=None):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_grad_lse(x, beta, n, fft_size, corr_fft_size):
    """
    Objective and gradient in x-space where a = softmax(x).
    f(x) = 2n * m_beta(conv(softmax(x), softmax(x)))
    where m_beta(v) = (1/beta) * log(sum(exp(beta*v)))
    """
    # Compute a = softmax(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute convolution
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:2*n-1]
    
    # Smooth max
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    exp_bc = np.exp(beta_conv - max_bc)
    log_sum_exp = np.log(exp_bc.sum())
    m_beta = (max_bc + log_sum_exp) / beta
    f = 2.0 * n * m_beta
    
    # Softmax probabilities
    p = exp_bc / exp_bc.sum()
    
    # Cross-correlation r[i] = sum_k p[k] * a[k-i] for i=0..n-1
    # = sum_{l=0}^{n-1} p[l+i] * a[l]
    # Using FFT: IFFT(FFT(p, N) * conj(FFT(a, N)))[:n]
    # where N >= (2n-1) + n - 1 = 3n-2
    P_full = np.fft.fft(p, corr_fft_size)
    A_full = np.fft.fft(a, corr_fft_size)
    r = np.fft.ifft(P_full * np.conj(A_full)).real[:n]
    
    # Gradient of f w.r.t. a:
    # df/da_i = 2n * (2/beta) * r_i = (4n/beta) * r_i
    grad_a = (4.0 * n / beta) * r
    
    # Gradient w.r.t. x (softmax chain rule):
    # dx_f_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    a_dot_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - a_dot_grad)
    
    return f, grad_x

def optimize_lse(a0, time_limit, beta_schedule=None, max_iter=5000):
    """
    Optimize using L-BFGS with log-sum-exp smoothing.
    a0: initial probability vector
    beta_schedule: list of beta values to anneal through
    """
    n = len(a0)
    fft_size = 1 << (2*n - 2).bit_length()
    corr_fft_size = 1 << (3*n - 3).bit_length()
    
    # Initialize x from a0
    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    x = np.log(a)
    
    best_a = a.copy()
    best_score = float('inf')
    
    if beta_schedule is None:
        beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000]
    
    t0 = time.monotonic()
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Evaluate current
        f, _ = obj_grad_lse(x, beta, n, fft_size, corr_fft_size)
        
        result = minimize(
            obj_grad_lse,
            x,
            args=(beta, n, fft_size, corr_fft_size),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': max_iter * 10}
        )
        
        x = result.x
        f, _ = obj_grad_lse(x, beta, n, fft_size, corr_fft_size)
        
        # Convert back to a
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = exp_x / exp_x.sum()
        
        # Evaluate actual objective
        conv = fast_convolve(a, fft_size)
        score = 2.0 * n * conv.max()  # since sum(a) = 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def optimize_subgrad_simple(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=50):
    """Simple projected subgradient as backup."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        # Subgradient at top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        
        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * a[k - idx]
        
        sg /= len(peaks)
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= 0.9999
    
    return best_a, best_obj
