"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses adaptive soft-max gradient descent, multi-peak LP refinement,
block coordinate descent, and simulated annealing local search.
Focuses on n=384 with exploration of nearby lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import correlate


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_sum1(a):
    """Normalize so sum(a) = 1."""
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) / len(a)
    return a / S


def softmax_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta * c[k])) where c = conv(a,a).
    Returns gradient vector of length n.
    """
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max() * beta
    exp_vals = np.exp(beta * c_shifted)
    w = exp_vals / exp_vals.sum()
    # grad[i] = sum_j w[i+j] * a[j] for j=0..n-1
    # Build index matrix
    idx = np.arange(n)
    w_sub = w[idx[:, None] + idx[None, :]]
    grad = w_sub @ a
    return grad, w, c


def hardmax_grad(a):
    """Gradient of max(conv(a,a)) using hard max."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    return grad, k_star


def project_sum1_nonneg(a):
    """Project onto {a >= 0, sum(a) = 1} (simple: clip and renormalize)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) / len(a)
    return a / S


def gradient_step(a, lr, momentum, decay, beta=None):
    """One step of gradient descent with momentum, projected."""
    if beta is not None:
        grad, w, c = softmax_grad(a, beta)
    else:
        grad, k_star = hardmax_grad(a)
    # Project out mean component (to maintain sum constraint approximately)
    grad_centered = grad - grad.mean()
    momentum[:] = decay * momentum - lr * grad_centered
    a_new = a + momentum
    a_new = project_sum1_nonneg(a_new)
    return a_new, momentum


def lp_mulipeak(f, M_thresholds, n_vars=None):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M_thresholds[k] for all k, g >= 0.
    M_thresholds is an array of length 2n-1.
    Returns optimal g or None.
    """
    n = len(f)
    if n_vars is None:
        n_vars = n
    if n > 1024:
        return None

    c_obj = -np.ones(n)
    # Build constraint matrix: A[k,j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = f[:end - start]

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=M_thresholds, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_single_peak(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M (uniform), g >= 0."""
    n = len(f)
    if n > 1024:
        return None
    M_thresh = np.full(2 * n - 1, M)
    return lp_mulipeak(f, M_thresh)


def coordinate_descent(a, deadline, max_rounds=5, block_size=8):
    """
    Block coordinate descent with line search.
    """
    n = len(a)
    a = normalize_sum1(a.copy())
    best_score = compute_obj(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random permutation of block start positions
        n_blocks = n // block_size
        starts = rng.permutation(n_blocks) * block_size

        for s in starts:
            if time.monotonic() > deadline:
                break
            block = a[s:s+block_size].copy()
            orig = block.copy()

            # Try several perturbation strategies for this block
            candidates = []
            # Scale variations
            for scale in [0.5, 0.7, 1.3, 1.5]:
                candidates.append(orig * scale)
            # Flip-like: make large values smaller
            mx = block.max()
            if mx > 1e-10:
                candidates.append(1.0 - block / mx)
            # Random
            candidates.append(rng.random(block_size) * mx * 0.5)

            best_val = orig.copy()
            best_s = compute_obj(a)

            for cand in candidates:
                cand = np.maximum(cand, 0.0)
                a[s:s+block_size] = cand
                a = normalize_sum1(a)
                s_val = compute_obj(a)
                if s_val < best_s - 1e-12:
                    best_s = s_val
                    best_val = cand.copy()
                a[s:s+block_size] = orig
            a = normalize_sum1(a)

            if not np.array_equal(best_val, orig):
                a[s:s+block_size] = best_val
                a = normalize_sum1(a)
                if best_s < best_score - 1e-12:
                    best_score = best_s
                    improved = True

        if not improved:
            break

    return a, best_score


def local_search_anneal(a, deadline, n_iter=20000, T_start=0.001, T_end=0.0001):
    """
    Simulated annealing local search: perturb random pairs of coordinates.
    """
    n = len(a)
    a = normalize_sum1(a.copy())
    best = a.copy()
    best_score = compute_obj(a)
    current_score = best_score
    rng = np.random.default_rng(777)
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iter))

        # Pick two random indices to swap/adjust
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        orig_i, orig_j = a[i], a[j]

        # Try: swap, or mix
        for mode in range(3):
            if mode == 0:
                # Swap
                a[i], a[j] = orig_j, orig_i
            elif mode == 1:
                # Mix
                mix = 0.3 + 0.4 * rng.random()
                a[i] = orig_i * (1 - mix) + orig_j * mix
                a[j] = orig_j * (1 - mix) + orig_i * mix
            else:
                # Perturb
                perturb = rng.normal(0, 0.05 * max(orig_i, orig_j, 0.01))
                a[i] = max(0, orig_i + perturb)

            a[time.monotonic() > deadline and 0 or i] = a[i]  # no-op, keep i
            a = normalize_sum1(a)
            s_new = compute_obj(a)

            if s_new < current_score or rng.random() < np.exp(-(s_new - current_score) / max(T, 1e-10)):
                current_score = s_new
                if s_new < best_score:
                    best_score = s_new
                    best = a.copy()
                break
            else:
                a[i] = orig_i
                a[j] = orig_j
                a = normalize_sum1(a)

        if it % 5000 == 0 and it > 0:
            print(f"    Anneal iter {it}: current={current_score:.6f}, best={best_score:.6f}")

    return best, best_score

                    starts.append(padded)
            # Shuffled versions
            L_use = min(len(gs), 384)
            for _ in range(3):
                idx = rng.permutation(L_use)
                starts.append(gs[idx].copy())
    except NameError:
        pass

    # Random starts
    for n in lengths:
        starts.append(rng.random(n) * 0.6 + 0.4)
        # Biased random
        i_arr = np.arange(n, dtype=np.float64)
        starts.append(rng.random(n) * (1.0 + 0.5 * np.sin(2 * np.pi * i_arr / n)))
        starts[-1] = np.maximum(starts[-1], 0.1)

    # Structured starts
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        # Chirp
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 5.0 / n + 0.5))
        starts.append(0.5 + 0.3 * np.cos(2 * np.pi * i * 7.0 / n) + 0.2 * np.cos(2 * np.pi * i * 13.0 / n))
        # Gaussian
        starts.append(np.exp(-2.0 * ((i - n / 2.0) / (n * 0.3)) ** 2) + 0.05)
        # Two Gaussians
        starts.append(np.exp(-4.0 * ((i - n * 0.3) / (n * 0.15)) ** 2) + np.exp(-4.0 * ((i - n * 0.7) / (n * 0.15)) ** 2) + 0.02)
        # Linear ramp + noise
        starts.append(0.3 + 0.7 * i / n + rng.normal(0, 0.02, n))
        starts[-1] = np.maximum(starts[-1], 0.01)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization with gradient descent, LP refinement,
    coordinate descent, and local search. Returns best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    print(f"Starting optimization, budget={budget_s}s")

    # Primary lengths to try
    primary_lengths = [384, 320, 448, 352, 416]

    # Generate starts
    starts = generate_starts(rng, primary_lengths)

    # Sort: prioritize n=384 starts first (known good length), then others
    def length_priority(s):
        L = len(s)
        if L == 384:
            return 0
        if L == 320:
            return 1
        if L == 448:
            return 2
        return 3
    starts.sort(key=length_priority)

    # Deduplicate
    unique_starts = []
    seen = set()
    for s in starts:
        key = (len(s), round(float(s.sum()), 2))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    starts = unique_starts

    print(f"Total starts: {len(starts)}")

    # Allocate time across starts
    n_starts = len(starts)
    # First start gets 40% of budget, second gets 25%, rest split
    allocations = []
    remaining_budget = deadline - t_start - 60  # keep 60s for final polish
    if n_starts > 0:
        alloc1 = max(30, remaining_budget * 0.35)
        allocations.append(alloc1)
        remaining_budget -= alloc1
    if n_starts > 1:
        alloc2 = max(20, remaining_budget * 0.25)
        allocations.append(alloc2)
        remaining_budget -= alloc2
    # Split rest among remaining starts
    if n_starts > 2:
        per_start = max(15, remaining_budget / (n_starts - 2))
        for _ in range(n_starts - 2):
            allocations.append(min(per_start, 60.0))

    # Run optimization on each start
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 60:
            break
        if idx >= len(allocations):
            break
        t_alloc = min(allocations[idx], deadline - time.monotonic() - 50)
        if t_alloc < 10:
            break

        print(f"\n[Start {idx}] n={len(a0)}, allocated={t_alloc:.0f}s")
        try:
            seq, score = optimize_single(a0, t_alloc, verbose=True)
        except Exception as e:
            print(f"    Error: {e}")
            continue

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    ** NEW BEST: score={best_score:.6f}, n={len(seq)}")
        else:
            print(f"    score={score:.6f} (not better than {best_score:.6f})")

    # Second pass: refine the best with more time
    if best_seq is not None and time.monotonic() < deadline - 80:
        print(f"\n=== Second pass: refining best (score={best_score:.6f}, n={len(best_seq)}) ===")
        n_best = len(best_seq)
        t_alloc2 = min(300.0, deadline - time.monotonic() - 40)

        # Try perturbed versions
        for trial in range(3):
            if time.monotonic() > deadline - 40:
                break
            pert = rng.normal(0, 0.03 / n_best, n_best)
            a0 = np.maximum(best_seq + pert, 0)
            t_alloc = min(120.0, (deadline - time.monotonic() - 30) / max(1, 3 - trial))
            try:
                seq, score = optimize_single(a0, t_alloc, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"    ** NEW BEST: score={best_score:.6f}")
            except Exception as e:
                print(f"    Error: {e}")

    # Third pass: local search / annealing on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Local search on best (score={best_score:.6f}) ===")
        n_best = len(best_seq)
        t_alloc3 = min(120.0, deadline - time.monotonic() - 20)
        if t_alloc3 > 20:
            try:
                seq, score = local_search_anneal(best_seq, time.monotonic() + t_alloc3, *
                                                 [20000, 0.0005, 0.00005])
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"    ** NEW BEST after anneal: score={best_score:.6f}")
            except Exception as e:
                print(f"    Anneal error: {e}")

    # Final: one more gradient pass on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_final = min(60.0, deadline - time.monotonic() - 10)
        if t_final > 10:
            try:
                seq, score = optimize_single(best_seq, t_final, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"    ** FINAL NEW BEST: score={best_score:.6f}")
            except Exception as e:
                print(f"    Final error: {e}")

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        # Ensure all values are in valid range
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return [float(x) for x in best_seq]

    # Fallback
    print("FALLBACK: returning uniform sequence n=256")
    return [1.0] * 256
