def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def build_lp_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

"""
Multi-phase optimization combining improved LP iteration, multi-peak gradient
descent, structured simulated annealing, and adaptive restarts for minimizing
the normalized peak autocorrelation of non-negative step functions.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S):
    """Normalize sequence to sum S."""
    s = np.sum(a)
    if s < 1e-12:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def build_lp_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix for conv(a, g) constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix_sparse(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=False):
    """LP-based iterative refinement with adaptive M."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a, 0.01, None), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    iteration = 0

    while time.monotonic() - t0 < time_limit and iteration < 20000:
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        # Try slightly tighter M for more aggressive reduction
        M_tight = M * 0.995
        g = lp_solve(a, M_tight)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            g = lp_solve(a, M)
            g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        c_norm = normalize_seq(g, S)

        # Find best blend
        best_t = 0.0
        best_cv = M
        for t_try in [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]:
            b = (1.0 - t_try) * a + t_try * c_norm
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            cv = np.max(fft_convolve(b, b))
            if cv < best_cv:
                best_cv = cv
                best_t = t_try
            if cv <= M * 1.001:
                break

        if best_t == 0.0:
            break

        a = (1.0 - best_t) * a + best_t * c_norm
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)

        new_cv = np.max(fft_convolve(a, a))
        if new_cv < best_conv_max - 1e-13:
            best_conv_max = new_cv
            best_a = a.copy()
        elif new_cv > best_conv_max * 1.005:
            a = best_a.copy()

        iteration += 1
        if verbose and iteration % 50 == 0:
            sc = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: score={sc:.6f}, M={best_conv_max:.6f}")

        if best_t < 0.005:
            break

    return best_a


def compute_multi_peak_gradient(a, k_peaks=3):
    """Compute gradient penalizing top-k peaks of conv(a,a)."""
    n = len(a)
    conv = fft_convolve(a, a)
    # Get top-k peak positions
    top_idx = np.argsort(conv)[-k_peaks:]
    top_idx = sorted(set(int(i) for i in top_idx))

    grad = np.zeros(n)
    weights = np.zeros(len(top_idx))
    max_val = conv[top_idx[0]] if len(top_idx) > 0 else 1.0
    for i, lag in enumerate(top_idx):
        weights[i] = exp(-0.5 * (conv[lag] - max_val) / (max_val * 0.1)) if max_val > 0 else 1.0

    for i, lag in enumerate(top_idx):
        w = weights[i] / sum(weights)
        for j in range(n):
            i1 = lag - j
            if 0 <= i1 < n:
                grad[j] += w * a[i1]
            i2 = j - lag
            if 0 <= i2 < n and j != i1:
                grad[j] += w * a[i2]

    return grad, conv[top_idx[0]] if len(top_idx) > 0 else 0


def gradient_optimize(a, time_limit, verbose=False):
    """Multi-peak gradient descent with adaptive learning rate."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a, 0.01, None), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = np.max(fft_convolve(a, a))

    lr = 2.0 / n
    step = 0
    stall = 0
    k_peaks = 3

    while time.monotonic() - t0 < time_limit:
        grad, peak_val = compute_multi_peak_gradient(a, k_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        # Adaptive step: try to reduce peak by ~10% per step initially
        target_reduction = 0.05 * peak_val
        lr_effective = lr * target_reduction / max(grad_norm * peak_val, 1e-12)
        lr_effective = min(lr_effective, 3.0 / n)
        lr_effective = max(lr_effective, 0.01 / n)

        a_new = a - lr_effective * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        new_score = np.max(fft_convolve(a_new, a_new))
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 5.0 / n)
        else:
            stall += 1
            if stall > 100:
                lr *= 0.9
                k_peaks = min(k_peaks + 1, 8)
                if stall > 500:
                    k_peaks = 3
            if stall > 2000:
                break

        step += 1
        if verbose and step % 500 == 0:
            sc = 2.0 * n * best_score / (S * S)
            print(f"    Grad {step}: score={sc:.6f}, lr={lr:.4f}, k={k_peaks}")

    return best_a


def exp(x):
    return np.exp(x)


def simulated_annealing(a, time_limit, verbose=False):
    """SA with structured perturbations: multiplicative noise + chirp phase."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a, 0.01, None), S)
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score

    T = 0.005
    T_min = 0.00001
    cooling = 0.9997
    rng = np.random.default_rng(777)
    step = 0
    t_array = np.arange(n, dtype=np.float64)

    while time.monotonic() - t0 < time_limit and T > T_min:
        candidate = a.copy()

        # Strategy: mix of different perturbation types
        mode = rng.random()
        if mode < 0.4:
            # Block multiplicative perturbation
            n_blocks = int(rng.integers(1, 4))
            for _ in range(n_blocks):
                width = int(rng.integers(max(1, n // 50), max(2, n // 10)))
                start = int(rng.integers(0, max(1, n - width)))
                scale = np.exp(rng.normal(0, 0.08))
                candidate[start:start + width] *= scale
        elif mode < 0.7:
            # Chirp-like phase perturbation
            f = rng.uniform(0.01, 0.1)
            phi = rng.uniform(0, 2 * np.pi)
            amp = 0.05 + 0.05 * rng.random()
            candidate *= (1.0 + amp * np.cos(2 * np.pi * f * t_array / n + phi))
        elif mode < 0.9:
            # Sparse element perturbation
            n_els = int(rng.integers(2, max(3, n // 20)))
            idx = rng.choice(n, size=n_els, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.15, size=n_els))
        else:
            # Gaussian noise
            candidate += rng.normal(0, 0.02, size=n)

        candidate = np.clip(candidate, 0.0, None)
        candidate = normalize_seq(candidate, S)

        new_score = local_score(candidate)
        delta = new_score - current_score

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()

        T *= cooling
        step += 1

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            sc = 2.0 * n * best_score / (S * S)
            print(f"    SA {step}: score={sc:.6f}, T={T:.6f}, t={elapsed:.0f}s")

    return best_a


def make_initialization(n, init_type, rng):
    """Create various structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.2 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.1 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'tapered':
        a = 0.3 + 0.7 * np.cos(np.pi * t / (2.0 * n)) ** 2
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.1 + 0.15 * rng.random())
        a = 0.5 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'random_smooth':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)


def interpolate_sequence(a, n_new):
    """Linear interpolation to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)
