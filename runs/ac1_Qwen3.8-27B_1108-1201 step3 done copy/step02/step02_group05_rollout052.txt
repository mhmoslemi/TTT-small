import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_convolve(a / s).max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if np.any(cond):
        rho = np.max(np.where(cond)[0])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected subgradient with Nesterov momentum, LP-based direction finding,

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    return 2.0 * n * fast_convolve(a).max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if np.any(cond):
        rho = np.max(np.where(cond)[0])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = (cssv[n-1] - 1.0) / n
    w = np.maximum(v - theta, 0.0)
    return w


def compute_subgradient(a, conv=None):
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    k_star = int(np.argmax(conv))
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[mask] = a[idx[mask]]
    return grad


def subgradient_optimize(a0, time_limit=60.0):

def line_search(f, g, obj_f):
    """Simple line search: try several step sizes and pick the best."""
    best_t = 0.0
    best_obj = obj_f
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:

        a = np.maximum(a, 0.0)

            continue
        a /= s
        obj = obj_norm(a)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj

def local_search(a, time_limit=60.0):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 500
    
    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each pair to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Find indices to move mass FROM (high contribution) and TO (low contribution)
        danger = np.where(contrib > 0.5 * max_c)[0]
        safe = np.where(contrib < 0.1 * max_c)[0]
        
        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue
        
        found = False
        # Try moving mass from top danger to top safe
        for di in danger[:20]:
            if found:
                break
            for si in safe[:20]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break
        
        if not found:
            stuck += 1
        else:
            stuck = 0
    
    return a, best_obj


def get_parent(n):

def make_initializations(n, rng):
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Geometric progression (heavy on one side)
    for ratio in [1.05, 1.1, 1.15]:
        g = ratio ** x
        inits[f'geo_{ratio}'] = g / g.sum()
    for ratio in [1.05, 1.1, 1.15]:
        g = ratio ** (n - 1 - x)
        inits[f'geo_r_{ratio}'] = g / g.sum()
    
    # Two bumps
    for pos in [0.25, 0.4, 0.6]:
        bump = np.exp(-0.5 * ((x - pos * n) / (0.1 * n)) ** 2)
        inits[f'bump_{pos}'] = bump / bump.sum()
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
        inits['parent_mirror'] = parent[::-1] / parent[::-1].sum()
        # Perturbed parent
        perturbed = parent * (1.0 + 0.1 * rng.standard_normal(n))
        perturbed = np.maximum(perturbed, 0)
        if perturbed.sum() > 1e-10:
            inits['parent_pert'] = perturbed / perturbed.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    return inits

Wait, that's not quite right. The log-sum-exp of `conv/T` is `log(sum(exp(conv_i/T)))`. To approximate `max(conv)`, I use `(1/T) * log(sum(exp(T * conv_i)))` which approaches `max(conv_i)` as `T -> inf`.

Let me use: `smooth_max = (1/T) * log(sum(exp(T * conv_i)))` for `T > 0` large.

Then the gradient of `smooth_max` with respect to `a[j]` is:
`sum_k softmax(T * conv)[k] * T * d(conv[k])/da[j] / T = sum_k softmax(T*conv)[k] * a[k-j]`

where `softmax(T*conv)[k] = exp(T*conv[k]) / sum_i exp(T*conv[i])`.

This is a weighted average of the subgradients, weighted by how close each `conv[k]` is to the max. This is much smoother than the hard max and allows for better gradient-based methods.

Let me implement this:

```python
def smooth_obj_and_grad(a, T=50.0):
    """Smooth objective and gradient."""
    conv = fast_convolve(a)
    n = len(a)
    
    # Log-sum-exp for numerical stability
    max_conv = conv.max()
    shifted = T * (conv - max_conv)
    log_sum = max_conv * T + np.log(np.sum(np.exp(shifted)))
    smooth_max = log_sum / T  # This is the smoothed max
    
    # Weights
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient: sum_k weights[k] * a[k-j]
    grad = np.zeros(n)
    for k in range(len(conv)):
        w = weights[k]
        if w < 1e-10:
            continue
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] += w * a[idx[mask]]
    
    obj = 2.0 * n * smooth_max
    grad_obj = 2.0 * n * grad
    return obj, grad_obj
```

This is O(n^2) per iteration due to the loop over k, but for n=512, that's about 262k operations which is fast.
