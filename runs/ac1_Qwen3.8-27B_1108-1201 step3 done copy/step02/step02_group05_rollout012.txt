import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_xy(x, y):
    """Linear convolution of x and y using FFT."""
    n = len(x) + len(y) - 1
    fft_size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:n]


def objective_smoothing(a, t):
    """Smoothed objective: 2n * logsumexp(t*conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf, None
    conv = fast_convolve(a)
    # logsumexp for numerical stability
    max_c = conv.max()
    if max_c < 0:
        max_c = 0
    lse = max_c + np.log(np.sum(np.exp(t * (conv - max_c)))) / t
    val = 2.0 * n * lse / (s * s)
    return val, conv


def grad_smoothed(a, t):
    """Gradient of smoothed objective (with sum(a) fixed to 1)."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    weights = np.exp(t * (conv - max_c))
    weights /= weights.sum()
    # grad of logsumexp(t*conv) = t * weights @ dconv/da
    # d(conv[k])/da[m] = 2*a[k-m] if 0 <= k-m < n
    # So grad[m] = sum_k weights[k] * 2 * a[k-m] (if 0 <= k-m < n)
    # This is 2 * convolve(weights, a)[m] (reversed)
    # Actually: grad[m] = 2 * sum_k weights[k] * a[k-m] for 0<=k-m<n
    # = 2 * (convolve(weights_reversed, a))[m] ... let me think
    
    # sum_k weights[k] * a[k-m] where 0 <= k-m < n means k-m = j, 0<=j<n
    # = sum_j weights[j+m] * a[j] for j+m in [0, 2n-2]
    # This is the convolution of weights and a at position m... 
    # Actually it's (a * weights)[m] = sum_j a[j] * weights[m-j]
    # No wait: sum_j weights[j+m] * a[j] = sum_j a[j] * weights[j+m]
    # This is cross-correlation of a and weights at lag m.
    # = convolve(flip(a), weights)[m + n - 1] ... this is getting complicated.
    
    # Let me just compute it directly for clarity (O(n^2) but n <= 2048 so it's fine)
    grad = np.zeros(n)
    for m in range(n):
        # d(conv[k])/da[m] = 2*a[k-m] if 0 <= k-m < n
        # grad[m] = sum_k t * weights[k] * 2 * a[k-m] (if 0 <= k-m < n)
        # Let j = k-m, so k = j+m, j in [0, n-1], k in [m, m+n-1]
        # grad[m] = 2*t * sum_{j=0}^{n-1} weights[j+m] * a[j]
        # (where weights[k] = 0 if k >= 2n-1)
        k_start = m
        k_end = min(m + n, 2 * n - 1)
        grad[m] = 2.0 * t * np.dot(weights[k_start:k_end], a[:k_end - k_start])
    
    return grad


def objective_and_grad(a, t):
    """Compute objective and gradient for L-BFGS-B."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf, np.zeros(n)
    # Work with normalized a
    a_norm = a / s
    conv = fast_convolve(a_norm)
    max_c = conv.max()
    if max_c <= 0:
        max_c = 0
    exp_t = np.exp(t * (conv - max_c))
    sum_exp = exp_t.sum()
    lse = max_c + np.log(sum_exp) / t
    val = 2.0 * n * lse  # (sum a_norm = 1, so denominator is 1)
    
    weights = exp_t / sum_exp
    grad = np.zeros(n)
    for m in range(n):
        k_start = m
        k_end = min(m + n, 2 * n - 1)
        grad[m] = 2.0 * t * np.dot(weights[k_start:k_end], a_norm[:k_end - k_start])
    
    return val, grad


def gradient_descent_smoothed(a0, t=100.0, lr=0.01, time_limit=60.0, max_iter=1000):
    """Gradient descent with smoothed max objective."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a /= s
    
    best_a = a.copy()
    best_val = 2.0 * len(a) * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        val, grad = objective_and_grad(a, t)
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-15:
            break
        a_new /= ns
        
        # Check actual (unsmoothed) objective
        actual_val = 2.0 * len(a_new) * fast_convolve(a_new).max()
        if actual_val < best_val:
            best_val = actual_val
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate
        if it % 50 == 0 and it > 0:
            lr *= 0.9
    
    return best_a, best_val


def multiplicative_updates(a0, time_limit=60.0, max_iter=500):
    """Mirror descent with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a) / len(a)
    a /= a.sum()
    
    best_a = a.copy()
    best_val = 2.0 * len(a) * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.1
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        k_star = np.argmax(conv)
        
        # Subgradient of max at k_star
        # d(conv[k_star])/da[m] = 2*a[k_star - m] if 0 <= k_star - m < n
        n = len(a)
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-15:
            break
        a_new /= ns
        
        val = 2.0 * n * fast_convolve(a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        elif it % 100 == 0:
            eta *= 0.8
        
        a = a_new
    
    return best_a, best_val


def subgradient_momentum(a0, time_limit=60.0, max_iter=1000):
    """Subgradient method with Nesterov momentum, targeting multiple peak bins."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a /= a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_val = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    v = np.zeros(n)
    mu = 0.9  # momentum
    lr = 0.05
    top_k = 5  # number of peak bins to target
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top-k peak bins
        top_indices = np.argsort(conv)[-top_k:][::-1]
        # Weight them by how close they are to the max
        weights_k = np.exp((conv[top_indices] - M) * 10.0)
        weights_k /= weights_k.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for wk, k in zip(weights_k, top_indices):
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += wk * 2.0 * a[j]
        
        # Nesterov momentum
        v = mu * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-15:
            break
        a_new /= ns
        
        val = 2.0 * n * fast_convolve(a_new).max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.95
        
        a = a_new
    
    return best_a, best_val


def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:

    return f.copy()


def lp_step(f, time_limit=30.0):
    """One LP-based optimization step."""
    conv = fast_convolve(f)
    M = conv.max()
    g = solve_lp(f, M)
    if np.allclose(g, f, atol=1e-10):
        return f, 2.0 * len(f) * M
    
    # Line search
    best_t = 0.0
    best_val = 2.0 * len(f) * M
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_new = (1 - t) * f + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new /= s
        val = 2.0 * len(a_new) * fast_convolve(a_new).max()
        if val < best_val:
            best_val = val
            best_t = t
    
    a_new = (1 - best_t) * f + best_t * g
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new, best_val


def lp_optimize(f0, time_limit):
    """Iterative LP-based optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    f /= f.sum()
    best_f = f.copy()
    best_val = 2.0 * len(f) * fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        f_new, val = lp_step(f, time_limit=5.0)
        if val < best_val - 1e-10:
            best_val = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return best_f, best_val


def perturb_sequence(a, rng, strength=0.1):
    """Add multiplicative perturbation and renormalize."""
    noise = rng.lognormal(mean=0, sigma=strength, size=len(a))
    a_new = a * noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def extend_parent(parent, target_n):
    """Extend parent sequence to target length via interpolation + noise."""
    p = np.asarray(parent, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    p = p / s
    
    if target_n <= len(p):
        # Downsample
        indices = np.linspace(0, len(p) - 1, target_n).astype(int)
        return p[indices] / p[indices].sum()
    else:
        # Upsample
        x_orig = np.linspace(0, 1, len(p))
        x_new = np.linspace(0, 1, target_n)
        extended = np.interp(x_new, x_orig, p)
        extended = np.maximum(extended, 0.0)
        s2 = extended.sum()
        if s2 < 1e-10:
            return np.ones(target_n) / target_n
        return extended / s2


def local_search_multi(a, time_limit=30.0):
    """Local search targeting multiple peak bins simultaneously."""
    n = len(a)

    t0 = time.monotonic()

        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top 3 peak bins

        safe = np.zeros(n)

        safe_indices = np.where(safe_mask)[0]
        danger_indices = np.where(danger > 0.5 * danger.max())[0]

            stuck += 1
            continue
        
        found = False

            if found:

def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

    if np.allclose(g, f, atol=1e-10):
        return f, 2.0 * n * M

    best_t = 0.0
    best_val = 2.0 * n * M
    # Try several step sizes
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_new = (1.0 - t) * f + t * g
        a_new = np.maximum(a_new, 0.0)

        a_new /= s

    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new, best_val

            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break

    return best_f, best_val


def perturb_sequence(a, rng, strength=0.1):
    """Add multiplicative lognormal perturbation and renormalize."""
    noise = rng.lognormal(mean=0.0, sigma=strength, size=len(a))
    a_new = a * noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def extend_parent(parent, target_n):
    """Extend or downsample parent sequence to target length via interpolation."""
    p = np.asarray(parent, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    p = p / s

    if target_n <= len(p):
        indices = np.linspace(0, len(p) - 1, target_n).astype(int)
        result = p[indices]
    else:
        x_orig = np.linspace(0, 1, len(p))
        x_new = np.linspace(0, 1, target_n)
        result = np.interp(x_new, x_orig, p)
        result = np.maximum(result, 0.0)

    s2 = result.sum()
    if s2 < 1e-10:
        return np.ones(target_n) / target_n
    return result / s2


def local_search_multi(a, time_limit=30.0):
    """Local search targeting multiple peak convolution bins."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a /= a.sum()
    best_val = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit and stuck < 300:
        conv = fast_convolve(a)
        M = conv.max()

        # Find top peak bins
        top_indices = np.argsort(conv)[-3:][::-1]

        # Compute danger indices (high contribution to peaks)
        danger = np.zeros(n)
        for k in top_indices:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib = a[i] * a[j]
                    danger[i] = max(danger[i], contrib)
                    danger[j] = max(danger[j], contrib)

        if danger.max() < 1e-15:
            break

        danger_thresh = 0.5 * danger.max()
        safe_thresh = 0.1 * danger.max()
        danger_idx = np.where(danger > danger_thresh)[0]
        safe_idx = np.where(danger < safe_thresh)[0]

        if len(danger_idx) == 0 or len(safe_idx) == 0:
            stuck += 1
            continue

        found = False
        for di in danger_idx[:12]:
            if found:
                break
            for si in safe_idx[:12]:
                if found:
                    break
                for frac in [0.05, 0.1, 0.2, 0.35]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[di] -= amount
                    a_try[si] += amount
                    a_try = np.maximum(a_try, 0.0)
                    ts = a_try.sum()
                    if ts < 1e-15:
                        continue
                    a_try /= ts
                    val = 2.0 * n * fast_convolve(a_try).max()
                    if val < best_val - 1e-10:
                        best_val = val
                        a = a_try.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_val


def make_chirp(n, alpha, beta):
    """Create a chirp-like non-negative sequence."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + beta * np.cos(2.0 * np.pi * alpha * x / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-strategy optimization:
    Phase 1: Quick LP scan over diverse initializations and lengths.
    Phase 2: Deep optimization with LP, gradient descent, and subgradient.
    Phase 3: Intensive polish with alternating methods and perturbations.
    """
    rng = np.random.default_rng(seed)
    effective_budget = max(10.0, float(budget_s) - 40.0)
    deadline = time.monotonic() + effective_budget

    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        parent = np.maximum(parent, 0.0)
        parent = parent / parent.sum()
        parent_len = len(parent)
        parent_score = float(evaluate_sequence(parent.tolist()))
    except Exception:
        parent = np.ones(512) / 512
        parent_len = 512
        parent_score = float(evaluate_sequence(parent.tolist()))

    print(f"Parent: len={parent_len}, score={parent_score:.6f}")

    # Generate initial candidates
    candidates = {}
    candidates['parent'] = parent.copy()

    # Extended lengths
    target_lengths = [640, 768, 1024, 1280, 1536, 2048]
    for tn in target_lengths:
        if tn > parent_len:
            candidates[f'ext_{tn}'] = extend_parent(parent, tn)
            # Perturbed extension
            p_pert = perturb_sequence(parent, rng, 0.08)
            candidates[f'ext_p_{tn}'] = extend_parent(p_pert, tn)
        elif tn < parent_len:
            candidates[f'down_{tn}'] = extend_parent(parent, tn)

    # Diverse initializations
    for n in [512, 1024, 2048]:
        for alpha, beta in [(0.31, 0.4), (0.73, 0.3), (1.37, 0.5), (2.61, 0.35)]:
            candidates[f'chirp_{n}_{alpha}'] = make_chirp(n, alpha, beta)

        # Edge-concentrated
        a_edge = np.zeros(n)
        m = max(1, n // 4)
        a_edge[:m] = 1.0
        a_edge[-m:] = 1.0
        a_edge /= a_edge.sum()
        candidates[f'edge_{n}'] = a_edge

        # Smooth bump (Gaussian)
        x = np.arange(n, dtype=np.float64)
        for sf in [0.2, 0.4, 0.6]:
            g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
            g /= g.sum()
            candidates[f'gauss_{n}_{sf}'] = g

    print(f"Total candidates: {len(candidates)}")

    # ===== Phase 1: Quick scan (25% of budget) =====
    phase1_time = effective_budget * 0.25
    phase1_deadline = time.monotonic() + phase1_time
    print(f"\nPhase 1: Quick scan ({phase1_time:.0f}s budget)")

    # Prioritize: parent, extensions of parent, then shorter others
    def priority_key(k):
        if k == 'parent':
            return (0, 0)
        if k.startswith('ext_') and not k.startswith('ext_p_'):
            parts = k.split('_')
            try:
                return (1, int(parts[1]))
            except (ValueError, IndexError):
                return (1, 9999)
        if k.startswith('ext_p_'):
            return (2, 512)
        # Extract length
        parts = k.rsplit('_', 1)
        try:
            return (3, int(parts[1]))
        except (ValueError, IndexError):
            return (3, 9999)

    ordered = sorted(candidates.keys(), key=priority_key)
    n_scan = len(ordered)
    time_per_scan = max(2.0, phase1_time / max(1, n_scan))

    scan_results = []
    for key in ordered:
        if time.monotonic() >= phase1_deadline:
            break
        remaining = deadline - time.monotonic() - 60.0
        if remaining < 10:
            break
        t_alloc = min(time_per_scan, 20.0, remaining)
        if t_alloc < 2.0:
            break

        a = candidates[key]
        a_opt, val = lp_optimize(a, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, key, a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  {key} (n={len(a_opt)}): score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # ===== Phase 2: Deep optimization (50% of budget) =====
    phase2_time = effective_budget * 0.50
    phase2_deadline = time.monotonic() + phase2_time
    print(f"\nPhase 2: Deep optimization ({phase2_time:.0f}s budget)")

    top_n = min(5, len(scan_results))
    top_candidates = [(sr[1], sr[2]) for sr in scan_results[:top_n]]
    if best_sequence is not None:
        top_candidates.append(('current_best', best_sequence.copy()))
    # Also add the parent if not already included
    if not any(name == 'parent' for name, _ in top_candidates):
        top_candidates.append(('parent', parent.copy()))

    time_per_deep = max(15.0, phase2_time / max(1, len(top_candidates)))

    for name, a0 in top_candidates:
        if time.monotonic() >= phase2_deadline:
            break
        remaining = deadline - time.monotonic() - 30.0
        if remaining < 15:
            break
        t_alloc = min(time_per_deep, remaining - 10.0)
        if t_alloc < 5.0:
            break

        a_work = a0.copy()

        # Strategy A: Extended LP
        t_lp = t_alloc * 0.35
        a_lp, _ = lp_optimize(a_work, time_limit=t_lp)
        score = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_lp.copy()
            a_work = a_lp.copy()
            print(f"  [LP] {name} (n={len(a_lp)}): {score:.6f}")

        # Strategy B: Subgradient momentum
        remaining = deadline - time.monotonic() - 30.0
        t_sg = min(t_alloc * 0.25, remaining * 0.25)
        if t_sg > 5:
            a_sg, _ = subgrad_momentum(a_work, time_limit=t_sg)
            score = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_sg.copy()
                a_work = a_sg.copy()
                print(f"  [SG] {name} (n={len(a_sg)}): {score:.6f}")

        # Strategy C: Gradient descent with smoothing
        remaining = deadline - time.monotonic() - 30.0
        t_gd = min(t_alloc * 0.25, remaining * 0.25)
        if t_gd > 5:
            for t_smooth in [50.0, 150.0, 300.0]:
                remaining2 = deadline - time.monotonic() - 30.0
                if remaining2 < 5:
                    break
                t_gd_sub = min(t_gd / 3.0, remaining2 * 0.4)
                a_gd, _ = gd_smoothed(a_work, t=t_smooth, lr=0.01, time_limit=t_gd_sub)
                score = float(evaluate_sequence(a_gd.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_gd.copy()
                    a_work = a_gd.copy()
                    print(f"  [GD t={t_smooth}] {name} (n={len(a_gd)}): {score:.6f}")

        # Strategy D: Local search
        remaining = deadline - time.monotonic() - 30.0
        t_ls = min(t_alloc * 0.15, remaining * 0.15)
        if t_ls > 3:
            a_ls, _ = local_search_multi(a_work, time_limit=t_ls)
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ls.copy()
                print(f"  [LS] {name} (n={len(a_ls)}): {score:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # ===== Phase 3: Intensive polish (25% of budget) =====
    phase3_time = effective_budget * 0.25
    phase3_deadline = time.monotonic() + phase3_time
    print(f"\nPhase 3: Intensive polish ({phase3_time:.0f}s budget)")

    if best_sequence is not None:
        polish_count = 0
        while time.monotonic() < phase3_deadline:
            remaining = phase3_deadline - time.monotonic()
            if remaining < 10:
                break

            a_cur = best_sequence.copy()
            improved = False

            # LP step
            t_lp = min(remaining * 0.25, 45.0)
            if t_lp > 3:
                a_try, _ = lp_optimize(a_cur, time_limit=t_lp)
                score = float(evaluate_sequence(a_try.tolist()))
                if np.isfinite(score) and score < best_score - 1e-9:
                    best_score = score
                    best_sequence = a_try.copy()
                    a_cur = a_try.copy()
                    improved = True
                    print(f"  Polish LP: {score:.6f}")

            # Subgradient
            remaining = phase3_deadline - time.monotonic()
            t_sg = min(remaining * 0.2, 25.0)
            if t_sg > 3:
                a_try, _ = subgrad_momentum(a_cur, time_limit=t_sg)
                score = float(evaluate_sequence(a_try.tolist()))
                if np.isfinite(score) and score < best_score - 1e-9:
                    best_score = score
                    best_sequence = a_try.copy()
                    a_cur = a_try.copy()
                    improved = True
                    print(f"  Polish SG: {score:.6f}")

            # Gradient descent
            remaining = phase3_deadline - time.monotonic()
            t_gd = min(remaining * 0.15, 15.0)
            if t_gd > 3:
                a_try, _ = gd_smoothed(a_cur, t=200.0, lr=0.005, time_limit=t_gd)
                score = float(evaluate_sequence(a_try.tolist()))
                if np.isfinite(score) and score < best_score - 1e-9:
                    best_score = score
                    best_sequence = a_try.copy()
                    a_cur = a_try.copy()
                    improved = True
                    print(f"  Polish GD: {score:.6f}")

            # Local search
            remaining = phase3_deadline - time.monotonic()
            t_ls = min(remaining * 0.15, 15.0)
            if t_ls > 2:
                a_try, _ = local_search_multi(a_cur, time_limit=t_ls)
                score = float(evaluate_sequence(a_try.tolist()))
                if np.isfinite(score) and score < best_score - 1e-9:
                    best_score = score
                    best_sequence = a_try.copy()
                    a_cur = a_try.copy()
                    improved = True
                    print(f"  Polish LS: {score:.6f}")

            # Perturbation + restart if stuck
            if not improved:
                polish_count += 1
                if polish_count >= 3:
                    # Try perturbation
                    a_pert = perturb_sequence(a_cur, rng, strength=0.03)
                    t_pert = min(remaining * 0.2, 20.0)
                    a_try, _ = lp_optimize(a_pert, time_limit=t_pert)
                    score = float(evaluate_sequence(a_try.tolist()))
                    if np.isfinite(score) and score < best_score - 1e-9:
                        best_score = score
                        best_sequence = a_try.copy()
                        print(f"  Perturb+LP: {score:.6f}")
                        polish_count = 0
                    elif polish_count >= 6:
                        break
            else:
                polish_count = 0

    # Final: try a few more perturbations on the best if time remains
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_sequence is not None:
        print(f"\nFinal perturbations ({remaining:.0f}s remaining)")
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            a_pert = perturb_sequence(best_sequence, rng, strength=0.02 + 0.02 * trial)
            t_try = min(remaining * 0.15, 30.0)
            a_try, _ = lp_optimize(a_pert, time_limit=t_try)
            score = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_try.copy()
                print(f"  Perturb trial {trial}: {score:.6f}")

    if best_sequence is None:
        best_sequence = parent.copy()

    # Final evaluation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
