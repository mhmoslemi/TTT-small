def solve_qp(C, n):
    """Solve minimize a^T C a subject to sum(a)=1, a>=0.
    C is an n x n symmetric PSD matrix.
    Uses active set method.
    """
    # Start with all variables active
    active = set(range(n))
    for iteration in range(100):
        # Solve the equality-constrained problem on the active set
        # minimize a_S^T C_SS a_S subject to sum(a_S) = 1
        # KKT: 2 * C_SS * a_S = lambda * 1
        C_SS = C[np.ix_(list(active), list(active))]
        # Solve C_SS * a_S = (lambda/2) * 1
        # lambda = 2 / (1^T C_SS^{-1} 1)
        # a_S = C_SS^{-1} 1 / (1^T C_SS^{-1} 1)
        ones = np.ones(len(active))
        try:
            inv_C_ones = np.linalg.solve(C_SS, ones)
            lam = 2.0 / (ones @ inv_C_ones)
            a_S = inv_C_ones / (ones @ inv_C_ones)
        except np.linalg.LinAlgError:
            # C_SS is singular, use pseudo-inverse
            ...
        
        # Check feasibility
        a = np.zeros(n)
        for i, j in enumerate(active):
            a[j] = a_S[i]
        
        # Check KKT conditions for inactive variables
        grad = 2 * C @ a
        violations = [i for i in range(n) if i not in active and grad[i] < lam - 1e-10]
        
        if not violations:
            # Check no active variable is negative
            if a_S.min() >= -1e-12:
                return np.maximum(a, 0)
            else:
                # Remove the most violated variable from active set
                worst = min(active, key=lambda i: a[i])
                active.remove(worst)
        else:
            # Add the most violated variable to active set
            worst = min(violations, key=lambda i: grad[i])
            active.add(worst)
    
    return np.maximum(a, 0)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_box_qp(C, n, max_iter=200):
    """Solve minimize a^T C a subject to sum(a)=1, a>=0 using active set method."""
    # Active set: variables with a[i] > 0
    active = set(range(n))
    
    for it in range(max_iter):
        if len(active) == 0:
            return np.ones(n) / n
        
        # Solve equality-constrained problem on active set
        # minimize a_S^T C_SS a_S subject to 1^T a_S = 1
        # KKT: 2 * C_SS * a_S = lambda * 1
        # => a_S = C_SS^{-1} 1 / (1^T C_SS^{-1} 1)
        S = sorted(active)
        C_SS = C[np.ix_(S, S)]
        ones = np.ones(len(S))
        
        try:
            # Use Cholesky for symmetric PSD
            try:
                L = np.linalg.cholesky(C_SS + 1e-12 * np.eye(len(S)))
                Cinv_1 = np.linalg.solve(L.T, np.linalg.solve(L, ones))
            except np.linalg.LinAlgError:
                Cinv_1 = np.linalg.solve(C_SS + 1e-12 * np.eye(len(S)), ones)
            
            denom = ones @ Cinv_1
            if denom < 1e-15:
                a_S = np.ones(len(S)) / len(S)
            else:
                a_S = Cinv_1 / denom
            
        except np.linalg.LinAlgError:
            a_S = np.ones(len(S)) / len(S)
        
        # Build full solution
        a = np.zeros(n)
        for idx, i in enumerate(S):
            a[i] = a_S[idx]
        
        # Compute gradient
        grad = 2 * C @ a
        
        # Check KKT for inactive variables
        lam = grad[S[len(S)//2]] if len(S) > 0 else 0  # approximate lambda
        # Actually, lambda = 2 / denom (from KKT)
        if denom > 1e-15:
            lam = 2.0 / denom
        
        # Find violated constraints
        inactive = [i for i in range(n) if i not in active]
        min_grad_inactive = min((grad[i] for i in inactive), default=0)
        
        if min_grad_inactive < lam - 1e-10 and inactive:
            # Add the most violated variable
            worst = min(inactive, key=lambda i: grad[i])
            active.add(worst)
        elif a_S.min() < -1e-10:
            # Remove the most negative variable
            worst_idx = np.argmin(a_S)
            worst = S[worst_idx]
            active.remove(worst)
        else:
            # KKT satisfied
            a = np.maximum(a, 0)
            a = a / a.sum()
            return a
    
    # Fallback
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n

The key is to use MULTIPLE active lags simultaneously. The more lags we include, the more "constrained" the solution is, and the better it balances all lags.

**Refined algorithm**:
1. Start with `a` (e.g., uniform or from parent).
2. Compute `conv = convolve(a, a)`.
3. Select the top `m` lags (where `m` is a parameter, e.g., 10-50).
4. Assign equal weights to these `m` lags (or use softmax weights).
5. Form `C_weighted` (Hankel matrix with `C[i,j] = w[i+j]`).
6. Solve the QP: `minimize a^T C_weighted a` subject to `sum(a) = 1, a >= 0`.
7. Update `a`.
8. Go to 2.

The number of active lags `m` is a key hyperparameter. More lags → better balance but slower QP solve.
