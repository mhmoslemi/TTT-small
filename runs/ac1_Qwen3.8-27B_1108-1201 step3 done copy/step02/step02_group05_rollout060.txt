import scipy.signal

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(a, tau):

def softmax_grad(b, tau):

    a = a / a.sum()
    n = len(a)

    # Actually, let me compute d(f_tau)/d(b) = d(f_tau)/d(a) * da/db
    # da/db[i] = a[i] * (delta_{ij} - a[j]) for the softmax parameterization
    # So d(f_tau)/db[j] = sum_i (df/da[i]) * a[i] * (delta_{ij} - a[j])
    # = a[j] * (df/da[j] - sum_i (df/da[i]) * a[i])

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized gradient
descent with continuation on the soft-max parameter tau, followed by L-BFGS-B
polishing and local search. Explores multiple sequence lengths starting from
the parent sequence.

import numpy as np
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_grad_and_obj(b, tau):
    """Compute f_tau(b) and its gradient, where a = softmax(b)."""
    a = np.exp(b - b.max())
    a = a / a.sum()
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()
    sm = c_max + (1.0 / tau) * np.log(exp_c.sum())
    obj = 2.0 * n * sm
    
    # Gradient wrt a: 4n * (p * flip(a))[n-1 : n-1+n]
    flip_a = a[::-1]
    fft_size = 1 << (len(p) + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    FA = np.fft.rfft(flip_a, fft_size)
    conv_result = np.fft.irfft(P * FA, fft_size)[:len(p) + n - 1]
    df_da = 4.0 * n * conv_result[n-1:n-1+n]
    
    # Gradient wrt b
    df_da_dot_a = np.dot(df_da, a)
    grad_b = a * (df_da - df_da_dot_a)
    
    return obj, grad_b

def continuation_optimize(a0, time_limit, tau_values=None):
    """Run continuation method with increasing tau."""
    if tau_values is None:
        tau_values = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    
    # Initial b from a (log-softmax parameterization)
    b = np.log(a + 1e-30)
    # Center b (addition of constant doesn't change softmax)
    b = b - b.mean()
    
    best_a = a.copy()
    best_true_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for tau in tau_values:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Use L-BFGS-B for this tau
        def obj_func(bx):
            o, _ = softmax_grad_and_obj(bx, tau)
            return o
        
        def grad_func(bx):
            _, g = softmax_grad_and_obj(bx, tau)
            return g
        
        try:
            res = minimize(
                obj_func, b, method='L-BFGS-B',
                jac=grad_func,
                options={'maxiter': 500, 'maxfun': 5000, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            if res.success or res.fun < obj_func(b):

                a_new = np.exp(b - b.max())

                a = a_new

        except Exception as e:
            print(f"  L-BFGS-B error at tau={tau}: {e}")
            break
    
    return best_a, best_true_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized continuation
gradient descent (L-BFGS-B) with annealing, multi-length exploration, coordinate
descent refinement, and targeted local search. Starts from parent sequence and
diverse initializations, using a continuation method on the soft-max parameter
tau to navigate the non-smooth landscape of the max-convolution objective.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def true_obj(a):
    """True objective with sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def softmax_obj_and_grad(b, tau):
    """
    Compute f_tau(b) and gradient, where a = softmax(b).
    f_tau(b) = 2n * [c_max + (1/tau)*log(sum(exp(tau*(c-c_max))))]
    This is a smooth upper bound on 2n*max(c), tight as tau->inf.
    """
    a = np.exp(b - b.max())
    a = a / a.sum()
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()
    sm = c_max + (1.0 / tau) * np.log(exp_c.sum())
    obj = 2.0 * n * sm

    # Gradient wrt a: df/da[i] = 2n * sum_k p[k] * 2*a[k-i] (if 0<=k-i<n)
    # = 4n * (p * flip(a))[n-1 : n-1+n]
    flip_a = a[::-1]
    fft_size = 1 << (len(p) + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    FA = np.fft.rfft(flip_a, fft_size)
    conv_result = np.fft.irfft(P * FA, fft_size)[:len(p) + n - 1]
    df_da = 4.0 * n * conv_result[n - 1:n - 1 + n]

    # Chain rule: d/db[j] = a[j] * (df/da[j] - sum_i df/da[i]*a[i])
    df_da_dot_a = np.dot(df_da, a)
    grad_b = a * (df_da - df_da_dot_a)

    return obj, grad_b


def continuation_lbfgs(a0, time_limit, tau_schedule=None):
    """
    Continuation method: minimize f_tau using L-BFGS-B, gradually increasing tau.
    """
    if tau_schedule is None:
        tau_schedule = [1.0, 2.0, 5.0, 10.0, 25.0, 50.0, 100.0, 250.0, 500.0, 1000.0, 2500.0, 5000.0]

    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    b = np.log(a + 1e-300)
    b = b - b.mean()

    best_a = a.copy()
    best_true = true_obj(a)
    t0 = time.monotonic()

    for tau in tau_schedule:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 2.0:
            break

        remaining = time_limit - elapsed

        def obj_func(bx):
            o, _ = softmax_obj_and_grad(bx, tau)
            return o

        def grad_func(bx):
            _, g = softmax_obj_and_grad(bx, tau)
            return g

        try:
            maxiter = max(50, int(remaining * 200))
            maxfun = max(100, int(remaining * 1000))
            res = minimize(
                obj_func, b, method='L-BFGS-B',
                jac=grad_func,
                options={'maxiter': maxiter, 'maxfun': maxfun,
                         'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
            )
            if res.x is not None:
                b_new = res.x - res.x.mean()
                a_new = np.exp(b_new - b_new.max())
                a_new = a_new / a_new.sum()
                t = true_obj(a_new)
                if t < best_true:
                    best_true = t
                    best_a = a_new.copy()
                a = a_new
                b = b_new
        except Exception:
            break

    return best_a, best_true


def coordinate_descent(a0, time_limit, n_coords_per_iter=50):
    """
    Coordinate descent: for each coordinate i, find optimal a[i] given rest fixed.
    Uses the softmax parameterization implicitly by just adjusting one element
    and renormalizing.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = true_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    max_no_improve = 200

    order = np.arange(n)

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_convolve(a)
        improved = False

        # Pick coordinates in order of "importance" (gradient magnitude)
        flip_a = a[::-1]
        fft_size = 1 << (2 * n - 1 + n - 1).bit_length()
        # Just use random order for simplicity
        idx = rng.choice(n, size=min(n_coords_per_iter, n), replace=False)

        for i in idx:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            ai = a[i]
            # Try different values of a[i]
            # Objective as function of x (new value for a[i]):
            # Set a[i]=x, scale rest by (1-x)/(1-ai)
            for frac in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 0.3, 2.0, 0.1]:
                x = ai * frac
                if x < 0 or x >= 1.0:
                    continue
                a_trial = a.copy()
                scale = (1.0 - x) / (1.0 - ai) if ai < 1.0 else 1.0
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] = a[mask] * scale
                a_trial[i] = x
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                o = true_obj(a_trial)
                if o < best_obj - 1e-13:
                    best_obj = o
                    a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return a, best_obj


def local_search(a0, time_limit):
    """
    Targeted local search: move mass from peak-contributing to safe positions.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = true_obj(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 300
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        c = fast_convolve(a)
        # Consider top 5 peaks
        top_peaks = np.argsort(c)[-5:]

        contrib = np.zeros(n)
        for k in top_peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib[i] += 2.0 * a[i] * a[j]

        thr = np.median(contrib[contrib > 0]) if np.any(contrib > 0) else 0
        danger = np.where(contrib > thr * 1.5)[0]
        safe = np.where(contrib < thr * 0.5)[0]

        found = False

        if len(danger) > 0 and len(safe) > 0:
            for di in danger[:20]:
                if found:
                    break
                for si in safe[:20]:
                    if found:
                        break
                    if di == si:
                        continue
                    for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                        amount = a[di] * frac
                        if amount < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[di] -= amount
                        a_trial[si] += amount
                        a_trial = np.maximum(a_trial, 0.0)
                        ts = a_trial.sum()
                        if ts < 1e-12:
                            continue
                        a_trial /= ts
                        o = true_obj(a_trial)
                        if o < best_obj - 1e-13:
                            best_obj = o
                            a = a_trial.copy()
                            found = True
                            break

        if not found:
            # Random perturbation attempt
            for _ in range(30):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                frac = rng.choice([0.02, 0.05, 0.1, 0.2, 0.3])
                direction = rng.choice([-1, 1])
                amount = a[i] * frac * direction
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                o = true_obj(a_trial)
                if o < best_obj - 1e-13:
                    best_obj = o
                    a = a_trial.copy()
                    found = True
                    break

        if not found:

        else:
            stuck = 0

def make_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.15, 0.25, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Two-sided Gaussian
    for sf in [0.1, 0.2, 0.3]:
        g1 = np.exp(-0.5 * ((x - 0.2 * n) / (sf * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.8 * n) / (sf * n)) ** 2)
        g = (g1 + g2)
        g /= g.sum()
        inits[f'double_gauss_{sf}'] = g

    # Hamming window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0.0)
        if hann.sum() > 0:
            inits['hann'] = hann / hann.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Random (Dirichlet)
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # Edge concentration
    edges = np.zeros(n)
    m = max(1, n // 4)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Middle concentration
    mid = np.zeros(n)
    m = max(1, n // 3)
    mid[n//2 - m:n//2 + m] = 1.0
    inits['middle'] = mid / mid.sum()

    # From parent
    if parent is not None and parent.size > 0:
        if parent.size >= n:
            p = parent[:n].copy()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 0.0)
        if p.sum() > 1e-10:
            inits['parent'] = p / p.sum()

    return inits

    all_configs = []
    for n in lengths:
        inits = make_initializations(n, rng, parent)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))

    # Prioritize: parent first, then n=512, then others
    def config_priority(item):
        n, name, _ = item
        score = 0
        if name == 'parent':
            score += 0
        elif name == 'uniform':
            score += 2
        else:
            score += 3
        score += abs(n - 512) / 512.0
        return score

    all_configs.sort(key=config_priority)

    n_configs = len(all_configs)
    results = []

    for i, (n, name, a0) in enumerate(all_configs):
        remaining = phase1_end - time.monotonic()
        if remaining < 5.0:
            break

        # Allocate time based on priority
        if name == 'parent':
            t_alloc = min(remaining * 0.3, 180.0)
        elif n == 512:
            t_alloc = min(remaining * 0.2, 120.0)
        else:
            t_alloc = min(remaining * 0.15, 60.0)

        t_alloc = max(3.0, t_alloc)
        t_alloc = min(t_alloc, remaining - 2.0)
        if t_alloc < 2.0:
            break

        try:
            a_opt, obj_opt = continuation_lbfgs(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")
        except Exception as e:
            print(f"  Error n={n}, {name}: {e}")

    print(f"  Phase 1 done. Best so far: {best_score:.6f}")

    # Sort results and keep top ones
    results.sort(key=lambda x: x[0])

    # ============================================================
    # Phase 2: Deep optimization on top candidates
    # ============================================================
    print("Phase 2: Deep optimization on top candidates...")
    phase2_end = time.monotonic() + (total_budget - 30.0) * 0.30

    top_results = results[:6]
    # Also add the current best
    if best_sequence is not None:
        top_results.append((best_score, len(best_sequence), 'best', best_sequence.copy()))

    # Remove duplicates (same n and similar starting point)
    seen = set()
    unique_top = []
    for score, n, name, a_opt in top_results:
        key = (n, round(score, 4))
        if key not in seen:
            unique_top.append((n, name, a_opt))
            seen.add(key)

    for i, (n, name, a0) in enumerate(unique_top):
        remaining = phase2_end - time.monotonic()
        if remaining < 10.0:
            break

        t_alloc = min(remaining * 0.7, 120.0)
        t_alloc = max(5.0, t_alloc)
        t_alloc = min(t_alloc, remaining - 5.0)
        if t_alloc < 3.0:
            break

        # Deep continuation with more aggressive tau schedule
        tau_deep = [10.0, 50.0, 100.0, 500.0, 1000.0, 5000.0, 10000.0, 50000.0]
        try:
            a_opt, obj_opt = continuation_lbfgs(a0, time_limit=t_alloc, tau_schedule=tau_deep)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n}, {name}: score={score:.6f}")
        except Exception:
            pass

        # Also try coordinate descent on the result
        remaining = phase2_end - time.monotonic()
        if remaining > 10.0 and best_sequence is not None:
            cd_time = min(remaining * 0.4, 60.0)
            a_cd, obj_cd = coordinate_descent(best_sequence, time_limit=cd_time)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD n={n}: score={score:.6f}")

    # ============================================================
    # Phase 3: Intensive polish on the best candidate
    # ============================================================
    print(f"Phase 3: Intensive polish (best={best_score:.6f})...")
    if best_sequence is not None:
        polish_deadline = deadline - 10.0

        # Alternate: continuation with high tau + local search + coordinate descent
        iterations = 0
        while time.monotonic() < polish_deadline:
            iterations += 1
            remaining = polish_deadline - time.monotonic()
            if remaining < 10.0:
                break

            n = len(best_sequence)

            # Step A: High-tau L-BFGS-B
            if remaining > 20:
                tau_high = [100.0, 1000.0, 5000.0, 50000.0]
                t_alloc = min(remaining * 0.3, 60.0)
                try:
                    a_opt, obj_opt = continuation_lbfgs(best_sequence, time_limit=t_alloc, tau_schedule=tau_high)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Polish iter {iterations} (L-BFGS): score={score:.6f}")
                except Exception:
                    pass

            # Step B: Local search
            remaining = polish_deadline - time.monotonic()
            if remaining > 15:
                ls_time = min(remaining * 0.25, 60.0)
                a_ls, obj_ls = local_search(best_sequence, time_limit=ls_time)
                score_ls = float(evaluate_sequence(a_ls.tolist()))
                if np.isfinite(score_ls) and score_ls < best_score:
                    best_score = score_ls
                    best_sequence = a_ls.copy()
                    print(f"  Polish iter {iterations} (LS): score={score_ls:.6f}")

            # Step C: Coordinate descent
            remaining = polish_deadline - time.monotonic()
            if remaining > 15:
                cd_time = min(remaining * 0.25, 60.0)
                a_cd, obj_cd = coordinate_descent(best_sequence, time_limit=cd_time)
                score_cd = float(evaluate_sequence(a_cd.tolist()))
                if np.isfinite(score_cd) and score_cd < best_score:
                    best_score = score_cd
                    best_sequence = a_cd.copy()
                    print(f"  Polish iter {iterations} (CD): score={score_cd:.6f}")

            # Step D: Random perturbation + re-optimize (escape local min)
            remaining = polish_deadline - time.monotonic()
            if remaining > 30:
                a_pert = best_sequence.copy()
                noise = rng.normal(0, 0.01, size=n)
                a_pert = a_pert + noise * a_pert
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 1e-10:
                    a_pert /= s_pert
                    t_pert = min(remaining * 0.2, 30.0)
                    try:
                        a_reopt, _ = continuation_lbfgs(a_pert, time_limit=t_pert,
                                                        tau_schedule=[50.0, 500.0, 5000.0])
                        score_pert = float(evaluate_sequence(a_reopt.tolist()))
                        if np.isfinite(score_pert) and score_pert < best_score:
                            best_score = score_pert
                            best_sequence = a_reopt.copy()
                            print(f"  Polish iter {iterations} (perturb): score={score_pert:.6f}")
                    except Exception:
                        pass

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
