```python
import scipy.optimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_lse(a, tau):
    """Log-sum-exp smoothed objective."""
    n = len(a)
    conv = fast_convolve(a)
    # Subtract max for numerical stability
    cmax = conv.max()
    lse = cmax + np.log(np.sum(np.exp(tau * (conv - cmax)))) / tau
    return 2.0 * n * lse

def grad_lse(a, tau):

    n = len(a)
    conv = fast_convolve(a)
    cmax = conv.max()
    # Softmax weights
    weights = np.exp(tau * (conv - cmax))
    weights /= weights.sum()

        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += weights[k] * a[j]
    return 2.0 * n * grad
```

```python
import numpy as np
import scipy.optimize
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def lse_obj_and_grad_x(x, tau):
    """Log-sum-exp smoothed objective and its gradient in log-parameterization."""
    n = len(x)
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()
    
    conv = fast_convolve(a)
    cmax = conv.max()
    # LSE
    lse = cmax + np.log(np.sum(np.exp(tau * (conv - cmax)))) / tau
    obj = 2.0 * n * lse
    
    # Gradient wrt a
    weights = np.exp(tau * (conv - cmax))
    weights /= weights.sum()
    g = np.zeros(n)
    for i in range(n):
        end = min(n, 2 * n - 1 - i)
        g[i] = np.dot(weights[i:i+end], a[:end])
    g *= 2.0 * n
    
    # Project to tangent space of simplex
    g_tan = g - np.dot(g, a)
    # Gradient wrt x = a * g_tan
    grad_x = a * g_tan
    
    return obj, grad_x

def lse_optimize(a0, tau_schedule, max_iter=200, time_limit=30):
    """Optimize using L-BFGS on LSE-smoothed objective with tau schedule."""
    a = np.maximum(a0, 1e-10).copy()
    a = a / a.sum()
    x = np.log(a)
    n = len(x)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def func_and_grad(x):
            return lse_obj_and_grad_x(x, tau)
        
        try:
            result = scipy.optimize.minimize(
                func_and_grad, x, jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-6}
            )
            x = result.x
            x_shift = x - x.max()
            a = np.exp(x_shift)
            a = a / a.sum()
            
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_obj

def multiplicative_weights(a0, n_iter=1000, eta=0.1, time_limit=10):
    """Multiplicative weight updates to reduce peak."""
    a = np.maximum(a0, 1e-10).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: sg[i] = a[k_star - i] for valid k_star - i
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = a[j]
        
        # Multiplicative update
        a = a * np.exp(-eta * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step size
        eta = max(0.01, eta * 0.999)
    
    return best_a, best_obj

def peak_reduction_search(a0, time_limit=10):
    """Targeted peak reduction by redistributing mass."""
    a = np.maximum(a0, 1e-10).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 100:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contributions to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Find elements that contribute most to the peak
        # and elements that contribute little
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        
        # Threshold for "dangerous" and "safe" elements
```

```python
def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()


def lse_func_and_grad(x, tau):
    """LSE-smoothed objective and gradient in log-parameterization."""
    n = len(x)
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()
    
    conv = fast_convolve(a)
    cmax = conv.max()
    lse = cmax + np.log(np.sum(np.exp(tau * (conv - cmax)))) / tau
    obj = 2.0 * n * lse
    
    weights = np.exp(tau * (conv - cmax))
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.full(len(conv), 1.0 / len(conv))
    else:
        weights /= wsum
    
    g = np.zeros(n)
    for i in range(n):
        end = min(n, 2 * n - 1 - i)
        g[i] = np.dot(weights[i:i+end], a[:end])
    g *= 2.0 * n
    
    g_tan = g - np.dot(g, a)
    grad_x = a * g_tan
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. L-BFGS on log-sum-exp smoothed objective with tau annealing in log-parameterization
2. Multiplicative weight (mirror descent) updates guided by convolution peak subgradient
3. Peak reduction local search for final polishing
Explores multiple sequence lengths and diverse initializations including the parent sequence.
"""
import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()


def lse_func_and_grad(x, tau):
    """LSE-smoothed objective and gradient in log-parameterization. Returns (obj, grad)."""
    n = len(x)
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()

    conv = fast_convolve(a)
    cmax = conv.max()
    lse = cmax + np.log(np.sum(np.exp(tau * (conv - cmax)))) / tau
    obj = 2.0 * n * lse

    weights = np.exp(tau * (conv - cmax))
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.full(len(conv), 1.0 / len(conv))
    else:
        weights = weights / wsum

    g = np.zeros(n)
    for i in range(n):
        end = min(n, 2 * n - 1 - i)
        g[i] = np.dot(weights[i:i + end], a[:end])
    g *= 2.0 * n

    g_dot_a = np.dot(g, a)
    g_tan = g - g_dot_a
    grad_x = a * g_tan

    return obj, grad_x


def lse_optimize(a0, tau_list, time_limit):
    """Run L-BFGS on LSE-smoothed objective with tau schedule."""
    a = np.maximum(a0, 1e-10).copy()
    a = a / a.sum()
    x = np.log(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            result = scipy.optimize.minimize(
                lse_func_and_grad, x, args=(tau,), jac=True,
                method='L-BFGS-B',
                options={'maxiter': 400, 'gtol': 1e-7}
            )
            x = result.x
            x_shift = x - x.max()
            a_new = np.exp(x_shift)
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_obj


def multiplicative_update(a0, n_iter, eta_init, time_limit):
    """Mirror descent with multiplicative weights to reduce convolution peak."""
    a = np.maximum(a0, 1e-10).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = eta_init
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Use top few positions for a "soft" subgradient
        top_k = min(5, len(conv))
        top_indices = np.argsort(conv)[-top_k:]
        weights = np.zeros(len(conv))
        for idx in top_indices:
            weights[idx] = conv[idx] / cmax if cmax > 0 else 1.0
        weights = weights ** 4  # sharpen
        wsum = weights.sum()
        if wsum > 0:
            weights = weights / wsum

        sg = np.zeros(n)
        for i in range(n):
            end = min(n, 2 * n - 1 - i)
            sg[i] = np.dot(weights[i:i + end], a[:end])

        a = a * np.exp(-eta * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        eta = max(0.005, eta * 0.997)

    return best_a, best_obj


def peak_reduction(a0, time_limit):
    """Local search: move mass from peak-contributing to safe positions."""
    a = np.maximum(a0, 1e-10).copy()

                contrib[i] = a[i] * a[j]

        thr_s = 0.1 * cmax
        danger = np.where(contrib > thr_d)[0]

                if found:

                        a = a_t.copy()
                        best_a = a.copy()
                        found = True
                        break

        else:
            stuck = 0

def lp_refine(f, M, n):
    """Solve LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A_ub[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                                     method='highs', options={'time_limit': 10})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize_step(f, time_limit):
    """Single LP-based descent step with line search."""
    f = np.maximum(f, 0.0).copy()
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    if M < 1e-15:
        return f, compute_obj(f)

    g = lp_refine(f, M, n)
    if np.allclose(g, f, atol=1e-10):
        return f, compute_obj(f)

    # Line search
    best_t = 0.0
    best_obj = compute_obj(f)
    for t in np.linspace(0.05, 1.0, 20):
        a_trial = (1.0 - t) * f + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-12:
            continue
        a_trial = a_trial / s
        obj = compute_obj(a_trial)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    a_new = (1.0 - best_t) * f + best_t * g
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    return a_new, best_obj

def make_inits(n, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits['gauss_' + str(sf)] = g / g.sum()

    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(1)
    hann = np.maximum(hann, 1e-10)
    inits['hann'] = hann / hann.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    parent = get_parent(n)

        inits['parent'] = parent
        for nl in [0.03, 0.08, 0.15]:
            perturbed = parent * (1.0 + nl * rng.normal(0, 1, n))
            perturbed = np.maximum(perturbed, 1e-10)
            inits['parent_p' + str(nl)] = perturbed / perturbed.sum()
