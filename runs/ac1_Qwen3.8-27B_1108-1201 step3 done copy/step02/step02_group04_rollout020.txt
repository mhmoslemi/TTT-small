Wait, I need to double-check the gradient. Let me redo it.

`F(a) = (2n/beta) * log(sum_k exp(beta * c[k]))` where `c = a * a` (full convolution)

`dF/dc[k] = (2n/beta) * beta * exp(beta*c[k]) / sum_l exp(beta*c[l]) = 2n * p[k]`

`dc[k]/da[j] = 2*a[k-j]` (as derived, where a[m]=0 for invalid m)

`dF/da[j] = sum_k dF/dc[k] * dc[k]/da[j] = sum_k 2n*p[k] * 2*a[k-j] = 4n * sum_k p[k]*a[k-j] = 4n * g[j]`

where `g[j] = sum_k p[k]*a[k-j]` (sum over k where 0<=k-j<n).

Now, `a = softmax(x)`, so `da[i]/dx[j] = a[i]*(delta_{ij} - a[j])`

`dF/dx[j] = sum_i dF/da[i] * da[i]/dx[j] = sum_i 4n*g[i] * a[i]*(delta_{ij} - a[j])`
`= 4n * a[j] * (g[j] - a[j] * sum_i g[i]*a[i])`

Wait: `sum_i 4n*g[i]*a[i]*(delta_{ij} - a[j]) = 4n*a[j]*g[j] - 4n*a[j]*sum_i g[i]*a[i] = 4n*a[j]*(g[j] - sum_i g[i]*a[i])`

Let `s = sum_i g[i]*a[i]`. Then `grad_x[j] = 4n*a[j]*(g[j] - s)`.

And `s = sum_i a[i]*g[i] = sum_i a[i]*sum_k p[k]*a[k-i] = sum_k p[k]*sum_i a[i]*a[k-i] = sum_k p[k]*c[k]`

So `s = p @ c` (dot product of p and the convolution).

OK so the gradient computation is:
1. Compute c = conv(a, a)
2. Compute p = softmax(beta * c)
3. Compute g[j] = sum_k p[k]*a[k-j] (cross-correlation, extract relevant part)
4. Compute s = p @ c
5. grad_x = 4*n*a*(g - s)

This is O(n log n) per evaluation, which is fast.

Hybrid optimizer combining LP-guided alternating minimization with 
L-BFGS polishing on a smoothed objective. Uses multiple initializations,
reduced LP for speed, and progressive beta annealing for the polish phase.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_lp_indices(n):
    rows = []
    cols = []
    for i in range(n):
        for j in range(n):
            rows.append(i + j)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(f, M, n, rows, cols, vals, active_mask=None, timeout=5.0):
    if active_mask is not None:
        mask = active_mask[rows]
        data = vals[mask]
        r = rows[mask]
        c = cols[mask]
    else:
        data = vals
        r = rows
        c = cols
    b = np.full(len(r), M)
    A = sparse.csc_matrix((data, (r, c)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                               method='highs', options={'disp': False, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    n = len(f)
    f_sum = f.sum()

    else:
        f = f / f_sum

    M = conv.max()

    best_f = f.copy()

            active_mask = np.zeros(2*n-1, dtype=bool)

            active_mask = None

def smooth_objective_and_grad(x, n, beta):
    """Compute smoothed objective and gradient for log-parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    a = a / a_sum
    
    conv = fast_convolve(a)
    
    bc = beta * conv
    m = bc.max()
    exp_bc = np.exp(bc - m)
    sum_exp = exp_bc.sum()
    p = exp_bc / sum_exp
    
    obj = 2.0 * n * (m + np.log(sum_exp)) / beta
    
    # g[j] = sum_k p[k] * a[k-j]
    # Use cross-correlation
    corr = np.correlate(p, a, 'full')
    g = corr[n-1:2*n-1]
    
    # s = p @ conv
    s = np.dot(p, conv)
    
    # grad_x = 4*n*a*(g - s)
    grad = 4.0 * n * a * (g - s)
    
    return obj, grad

    n = len(f)
    f = f / f.sum()
    
    # Initial log-parametrization
    x = np.log(f)

        n_betas = [10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0, 10000.0]
    
    t0 = time.monotonic()
    remaining = time_limit - (time.monotonic() - t0)
    time_per_beta = remaining / len(n_betas)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    for beta in n_betas:
        if time.monotonic() - t0 > time_limit:
            break

            obj, _ = smooth_objective_and_grad(x, n, beta)
            return obj
        
        def func_grad(x):
            _, grad = smooth_objective_and_grad(x, n, beta)
            return grad
        
        time_budget = min(time_per_beta, time_limit - (time.monotonic() - t0))
        if time_budget < 2.0:
            break
        
        try:
            res = optimize.minimize(func, x, jac=func_grad, method='L-BFGS-B',
                                    options={'maxiter': 200, 'time_limit': max(1.0, time_budget*0.8),
                                             'gtol': 1e-10})
        except Exception:
            break
        
        if res.success or res.fun < best_obj:
            x = res.x
            x_shifted = x - x.max()
            f_new = np.exp(x_shifted)
            f_new = f_new / f_new.sum()
            f_new = np.maximum(f_new, 0.0)
            
            conv = fast_convolve(f)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_new.copy()
        
        x = np.log(np.maximum(f_new, 1e-30))
    
    return best_f, best_obj

def multiplicative_weights(f0, time_limit):
    """Multiply update: a[j] *= exp(-eta * subgrad[j]) where subgrad comes from the max convolution."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    n = len(f)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    eta = 0.1
    t_param = 50.0
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        
        # Soft-max weights on the convolution
        bc = t_param * conv / M
        bc -= bc.max()
        w = np.exp(bc)
        w = w / w.sum()
        
        # Subgradient: g[j] = sum_k w[k] * a[k-j]
        corr = np.correlate(w, f, 'full')
        g = corr[n-1:2*n-1]
        
        # Update: f[j] *= exp(-eta * g[j])
        # Center the gradient for numerical stability
        g_centered = g - np.dot(f, g)
        f = f * np.exp(-eta * g_centered)
        f = np.maximum(f, 1e-30)
        f = f / f.sum()

            best_obj = obj
            best_f = f.copy()
            patience = 0

            patience += 1

                eta *= 0.8
                t_param *= 1.5
                if eta < 0.01:
                    eta = 0.05
                    t_param = 50.0
            if patience > 500:
                break
    
    return best_f, best_obj

"""

Strategy: (1) diverse structured initializations, (2) LP-guided alternating

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_lp_indices(n):

def solve_lp(f, M, n, rows, cols, vals, active_mask=None, timeout=4.0):
    if active_mask is not None:
        m = active_mask[rows]
        data = vals[m]
        r = rows[m]
        c = cols[m]
    else:
        data = vals
        r = rows
        c = cols
    b = np.full(len(r), M)
    A = sparse.csc_matrix((data, (r, c)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                               method='highs',
                               options={'disp': False, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit):
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    rows, cols = make_lp_indices(n)
    vals = np.tile(f, n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)

def smooth_obj_grad(x, n, beta):
    xs = x - x.max()
    a = np.exp(xs)
    a = a / a.sum()
    conv = fast_convolve(a)
    bc = beta * conv
    m = bc.max()
    e = np.exp(bc - m)
    se = e.sum()
    p = e / se
    obj = 2.0 * n * (m + np.log(se)) / beta
    corr = np.correlate(p, a, 'full')
    g = corr[n - 1:2 * n - 1]
    s = np.dot(p, conv)
    grad = 4.0 * n * a * (g - s)
    return obj, grad


def lbfgs_polish(f0, time_limit):
    f = np.maximum(f0.astype(np.float64), 1e-30)
    n = len(f)
    f = f / f.sum()
    x = np.log(f)
    betas = [5.0, 15.0, 50.0, 150.0, 500.0, 1500.0, 5000.0]
    t0 = time.monotonic()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        per = remaining / (len(betas) - bi)
        max_it = max(20, int(per / 0.03))
        max_it = min(max_it, 800)
        t_up = [False]

        def callback(xk):
            if time.monotonic() - t0 > time_limit - 3:
                t_up[0] = True
                raise RuntimeError("timeout")

        def func(xv):
            return smooth_obj_grad(xv, n, beta)[0]

        def funcg(xv):
            return smooth_obj_grad(xv, n, beta)[1]

        try:
            res = optimize.minimize(func, x, jac=funcg, method='L-BFGS-B',
                                    callback=callback,
                                    options={'maxiter': max_it, 'gtol': 1e-9})
            x = res.x
        except Exception:
            pass

        xs = x - x.max()
        fn = np.exp(xs)
        fn = fn / fn.sum()
        fn = np.maximum(fn, 0.0)
        conv = fast_convolve(fn)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_f = fn.copy()
        x = np.log(np.maximum(fn, 1e-30))
        print(f"    LBF beta={beta:.0f} obj={obj:.8f}")

    return best_f, best_obj


def mirror_refine(f0, time_limit, t0_param=30.0, eta0=0.2):
    f = np.maximum(f0.astype(np.float64), 1e-15)
    n = len(f)

    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        bc = t_param * conv / M
        bc -= bc.max()
        w = np.exp(bc)
        w = w / w.sum()
        corr = np.correlate(w, f, 'full')
        g = corr[n - 1:2 * n - 1]
        gc = g - np.dot(f, g)

def gen_init(name, n, rng):
    x = np.arange(n, dtype=np.float64)
    if name == 'uni':
        return np.ones(n) / n
    elif name == 'gauss10':
        g = np.exp(-0.5 * ((x - 0.5 * n) / (0.10 * n)) ** 2)
        return g / g.sum()
    elif name == 'gauss20':
        g = np.exp(-0.5 * ((x - 0.5 * n) / (0.20 * n)) ** 2)
        return g / g.sum()
    elif name == 'gauss35':
        g = np.exp(-0.5 * ((x - 0.5 * n) / (0.35 * n)) ** 2)
        return g / g.sum()
    elif name == 'cos':
        c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        return c / c.sum()
    elif name == 'ramp':
        r = np.arange(1, n + 1, dtype=np.float64)
        return r / r.sum()
    elif name == 'bi':
        w = max(1, n // 3)
        b = np.zeros(n)
        b[:w] = 1.0
        b[-w:] = 1.0
        return b / b.sum()
    elif name == 'tri':
        t = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
        return t / t.sum()
    elif name == 'rnd':
        r = rng.dirichlet(np.ones(n) * 0.5)
        return r
    elif name == 'sin2':
        s = 0.5 + 0.4 * np.sin(4 * np.pi * x / n)
        return s / s.sum()
    elif name == 'parent':
        return None
    else:
        return np.ones(n) / n
