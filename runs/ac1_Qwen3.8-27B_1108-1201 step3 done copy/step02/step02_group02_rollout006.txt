import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def soft_max_grad(a, T=100):
    """Compute gradient of the soft-max of the convolution."""
    n = len(a)
    conv = fft_convolve(a, a)
    m = np.max(conv)
    # Soft max weights
    shifted = T * (conv - m)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: grad[i] = sum_k weights[k] * 2 * a[k-i] (if valid)
    grad = np.zeros(n)
    # Only consider peaks with significant weight
    significant = weights > 1e-10
    for k in np.where(significant)[0]:
        w = weights[k] * 2.0
        # a[k-i] for i in range(n) is valid when 0 <= k-i < n
        # i.e., k-n+1 <= i <= k
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            # a[k-i] for i in [i_min, i_max]
            # k-i ranges from k-i_min to k-i_max
            idx_start = k - i_max
            idx_end = k - i_min
            grad[i_min:i_max+1] += w * a[idx_start:idx_end+1]
    
    return grad, conv

    best_a = a.copy()

    lr = 0.01
    stall = 0

        if grad_norm < 1e-15:

            new_score = score(a_new)
            if new_score < current_score:
                a = a_new
                current_score = new_score
                lr = trial_lr * 1.1
                improved = True
                stall = 0
                if current_score < best_score:
                    best_score = current_score

                break

"""
Peak autocorrelation minimization for non-negative step functions using corrected
subgradient/soft-max gradient descent, LP-based direction finding, and simulated
annealing. Key improvement over previous: correct gradient computation (the prior
code used f[k-j]+f[j] instead of 2*f[k-j] for the subgradient).

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Fast local score computation."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


def soft_max_gradient(a, T):
    """Compute soft-max of convolution and its gradient w.r.t. a.
    
    Returns (soft_max_value, gradient, full_convolution)
    """
    n = len(a)
    conv = fft_convolve(a, a)
    m = np.max(conv)
    # Compute weights p_k = exp(T*(conv_k - m)) / sum
    shifted = T * (conv - m)
    max_shifted = np.max(shifted)  # = 0
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        return m, np.zeros(n), conv
    p = exp_shifted / total  # probability weights, sum to 1
    
    soft_max_val = m + np.log(total) / T
    
    # Gradient: grad[i] = sum_k p_k * 2*a[k-i] (if 0 <= k-i < n)
    grad = np.zeros(n)
    # Only process peaks with meaningful weight
    threshold = 1e-12
    active = np.where(p > threshold)[0]
    for k in active:
        w = 2.0 * p[k]
        # a[k-i] valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            # For i in [i_min, i_max], k-i ranges from k-i_max to k-i_min
            idx_lo = k - i_max
            idx_hi = k - i_min
            grad[i_min:i_max + 1] += w * a[idx_lo:idx_hi + 1]
    
    return soft_max_val, grad, conv

def lp_solve(f, M, n):
    """Solve LP: max sum(b) s.t. b>=0, conv(f,b)_k <= M for all k."""
    # Build constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n
    # For efficiency, use the fact that each column is a shifted copy of f
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        A[start:end, j] = f
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        b_0 = lp_solve(f, M, n)
        iter_count += 1
        if b_0 is None or np.sum(b_0) < 1e-10:
            # LP failed, try gradient
            smax, grad, _ = soft_max_gradient(f, 50.0)
            b_0 = f - 0.1 * grad
            b_0 = np.clip(b_0, 0, None)
            if np.sum(b_0) < 1e-10:
                break
        
        sum_b = np.sum(b_0)
        g = (S / sum_b) * b_0
        
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1, 0.02]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0, None)
            new_score = compute_score(f_new)
            if new_score < current_score:
                f = f_new
                current_score = new_score
                t = min(trial_t * 1.1, 0.3)
                improved = True
                stall = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break
        
        if not improved:
            stall += 1
            t *= 0.7
            if stall > 60:
                # Try gradient-based escape
                smax, grad, _ = soft_max_gradient(f, 20.0)
                f = f - 0.3 * grad
                f = np.clip(f, 0, None)
                sf = np.sum(f)
                if sf > 0.01:
                    f = f * (S / sf)
                    current_score = compute_score(f)
                stall = 0
                t = 0.03
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_f, best_score


def local_polish(a_init, time_limit, seed=0, verbose=False):
    """Local search with targeted peak reduction."""
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted peak reduction
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            i_min = max(0, peak_idx - n + 1)
            i_max = min(n - 1, peak_idx)
            for i in range(i_min, i_max + 1):
                contribs[i] = a[i] * a[peak_idx - i]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.70:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.2, 0.2) * a[i]
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        elif r < 0.85:
            # Small global
            candidate = a * np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_a, best_score


def resample_sequence(seq, new_n):
    """Linearly resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def make_init(n, init_type, rng):
    """Create initial sequence of given length and type."""
    if init_type == 'uniform':
        a = np.ones(n)
        a += 0.05 * rng.standard_normal(n)
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
        a += 0.04 * rng.standard_normal(n)
    elif init_type == 'multi_tone':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in range(1, 5):
            amp = 0.2 / k
            freq = rng.uniform(0.5, 2.0) * k
            phase = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        t = np.arange(n, dtype=np.float64)
        center = n // 2
        a = 1.0 + 0.5 * np.exp(-((t - center) ** 2) / (2 * (n // 6) ** 2))
        a += 0.04 * rng.standard_normal(n)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function combining corrected gradient SA, LP iteration,
    and local polish to minimize peak autocorrelation of non-negative sequences.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 20.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # ============ Phase 1: Corrected Gradient SA from provided sequence ============
    print("\n=== Phase 1: Gradient SA (primary) ===")
    if start_seq is not None and time.monotonic() < deadline - 600:
        t1 = min(450.0, deadline - time.monotonic() - 600)
        a_best, s_best = gradient_sa(start_seq, t1, T_start=5.0, T_end=300.0,
                                     lr_start=0.5, seed=seed, verbose=True)
        print(f"  Phase 1 result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # ============ Phase 2: LP iteration ============
    print(f"\n=== Phase 2: LP iteration ===")
    if best_seq is not None and time.monotonic() < deadline - 350:
        t2 = min(300.0, deadline - time.monotonic() - 350)
        a_best, s_best = lp_iterate(best_seq, t2, seed=seed + 1, verbose=True)
        print(f"  Phase 2 result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # ============ Phase 3: Explore different lengths and init types ============
    print(f"\n=== Phase 3: Exploration ===")
    n_options = [256, 384, 512, 768, 1024]
    init_types = ['chirp', 'multi_tone', 'bimodal', 'random']
    
    # Try different lengths for the best sequence
    if best_seq is not None and time.monotonic() < deadline - 200:
        for n_target in n_options:
            if time.monotonic() > deadline - 150:
                break
            if n_target == len(best_seq):
                continue
            resampled = resample_sequence(best_seq, n_target)
            s_res = compute_score(resampled)
            print(f"  Resampled to n={n_target}: score={s_res:.6f}")
            if s_res < best_score:
                best_score = s_res
                best_seq = resampled.copy()
                print(f"    *** New best from resampling: {best_score:.6f} ***")
            # Quick gradient SA on resampled
            t_quick = min(60.0, deadline - time.monotonic() - 100)
            if t_quick > 10:
                a_b, s_b = gradient_sa(resampled, t_quick, T_start=10.0, T_end=200.0,
                                       lr_start=0.3, seed=seed + n_target, verbose=False)
                if s_b < best_score:
                    best_score = s_b
                    best_seq = a_b.copy()
                    print(f"    *** New best after SA n={n_target}: {best_score:.6f} ***")

    # Try novel initializations
    if time.monotonic() < deadline - 150:
        for n_try in [256, 512, 1024]:
            for it in init_types:
                if time.monotonic() > deadline - 100:
                    break
                a0 = make_init(n_try, it, rng)
                t_q = min(50.0, deadline - time.monotonic() - 80)
                if t_q < 10:
                    break
                a_b, s_b = gradient_sa(a0, t_q, T_start=5.0, T_end=150.0,
                                       lr_start=0.4, seed=seed + 1000 + n_try, verbose=False)
                if s_b < best_score:
                    best_score = s_b
                    best_seq = a_b.copy()
                    print(f"  *** New best from {it} n={n_try}: {best_score:.6f} ***")

    # ============ Phase 4: Local polish ============
    print(f"\n=== Phase 4: Local polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 50))
        print(f"  Polish time: {polish_time:.0f}s, current best: {best_score:.6f}")
        a_p, s_p = local_polish(best_seq, polish_time, seed=seed + 7777, verbose=False)
        if s_p < best_score:
            best_score = s_p
            best_seq = a_p.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  Polish: no improvement (best={best_score:.6f})")

    # ============ Phase 5: Final gradient SA ============
    if time.monotonic() < deadline - 20:
        final_time = max(10.0, deadline - time.monotonic() - 10)
        print(f"\n=== Phase 5: Final gradient SA ({final_time:.0f}s) ===")
        a_f, s_f = gradient_sa(best_seq, final_time, T_start=50.0, T_end=500.0,
                               lr_start=0.3, seed=seed + 99999, verbose=True)
        if s_f < best_score:
            best_score = s_f
            best_seq = a_f.copy()
            print(f"  Final: {best_score:.6f}")

    # ============ Final verification ============
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
