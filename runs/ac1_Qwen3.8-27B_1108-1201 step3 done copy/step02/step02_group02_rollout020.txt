def compute_subgradient_fast(a, T):
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    weights = np.exp((c - c_max) / T)
    weights /= weights.sum()
    
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    # For each i, this is a dot product of weights[i:i+n] with a
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(weights[i:i+n], a)
    return grad

    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a * (np.sqrt(2.0 * n) / S)  # normalize
    S_target = np.sum(a)
    
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score
    
    lr = 0.1
    T = T_init
    t0 = time.monotonic()
    stall = 0

        # Gradient descent step

        a_new = np.maximum(a_new, 0)  # non-negativity constraint
        
        # Renormalize to maintain sum
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (S_target / S_new)
        
        new_score = compute_score(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            stall = 0
            lr = min(lr * 1.02, 0.3)
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            stall += 1
            lr *= 0.8
            if stall > 50:
                # Perturb
                a = a * np.exp(np.random.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a * (S_target / np.sum(a))
                current_score = compute_score(a)
                stall = 0
                lr = 0.1
                T = min(T * 2, 1.0)  # increase temperature to explore
        
        # Gradually decrease T
        T = max(T * 0.999, T_min)
    
    return best, best_score

    c = fft_convolve(a, a)
    c_max = np.max(c)

    grad = np.zeros(n)

def coordinate_descent_step(a, i):

def annealing_polish(a_init, time_limit, seed=123):

    rng = np.random.default_rng(seed)
    n = len(a_init)
    S_target = np.sum(a_init)

    current_score = compute_score(a)

    cooling = 0.9998
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            # Find contributing pairs
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):

                if 0 <= j < n:

            top_k = max(8, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.015, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(20, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.70:
            # Multi-block
            for _ in range(8):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.85:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.2, 0.2) * a[i]
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        else:
            # Small global
            candidate = a * np.exp(rng.normal(0.0, 0.008, size=n))

        S_cand = np.sum(candidate)
        if S_cand > 1e-10:

                best = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

    return best, best_score

def resample_sequence(seq, new_n):

    x_old = np.linspace(0, 1, len(seq))

    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score (fast local version)."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_subgradient(a, T):
    """Compute subgradient of max(c) using soft-max with temperature T."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    weights = np.exp((c - c_max) / T)
    wsum = weights.sum()
    if wsum < 1e-300:
        # Fallback: use argmax
        peak = np.argmax(c)
        grad = np.zeros(n)
        for j in range(n):
            k = peak - j
            if 0 <= k < n:
                grad[j] = 2.0 * a[k]
        return grad
    weights /= wsum

    # Build n x n matrix W[i,j] = weights[i+j], then grad = 2 * W @ a
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    W = weights[idx]
    grad = 2.0 * (W @ a)
    return grad


def subgradient_optimize(a_init, time_limit, T_init=0.5, T_min=0.005, seed=42, momentum=0.5):
    """
    Subgradient descent with soft-max smoothing, adaptive learning rate,
    momentum, and perturbation-based escape from local minima.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)

    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) * (S_target / n)
    else:
        a = a * (S_target / S)

    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score

    lr = 0.08
    T = T_init
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    stall = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if time_limit < 5.0 and iter_count > 200:
            break

        grad = compute_subgradient(a, T)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new * (S_target / S_new)

        new_score = compute_score(a_new)
        iter_count += 1

        if new_score < current_score:
            a = a_new
            current_score = new_score
            stall = 0
            lr = min(lr * 1.015, 0.25)
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        else:
            stall += 1
            lr *= 0.82
            v *= 0.5  # dampen momentum on failure

            if stall > 40:
                # Perturb to escape local minimum
                sigma = 0.02 + 0.02 * min(stall / 200.0, 1.0)
                a = a * np.exp(rng.normal(0, sigma, n))
                a = np.maximum(a, 0)
                a = a * (S_target / np.sum(a))
                current_score = compute_score(a)
                v = np.zeros(n)
                stall = 0
                lr = 0.06
                T = min(T * 2.5, 3.0)

        # Adaptively adjust T
        if stall < 5:
            T = max(T * 0.999, T_min)
        else:
            T = max(T * 0.998, T_min)

    return best, best_score


def coordinate_refine(a_init, time_limit, seed=77):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the peak autocorrelation.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S_target = np.sum(a_init)
    a = a_init.astype(np.float64).copy()
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    # Precompute the "rest" contributions
    def eval_coord(a_full, i, v):
        """Evaluate max of convolution with a[i] = v."""
        a_trial = a_full.copy()
        a_trial[i] = v
        c = fft_convolve(a_trial, a_trial)
        return np.max(c)

    # Multiple passes through coordinates
    max_passes = 50
    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        # Randomize coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                return best, best_score

            current_val = a[i]
            if current_val < 1e-8:
                continue

            # Define 1D objective
            def obj_1d(v, i=i):
                return eval_coord(a, i, v)

            # Bounded search
            v_max = max(current_val * 3.0, S_target / n * 3.0)
            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, v_max),
                                      method='bounded', options={'xatol': 1e-7})
                if res.success and res.fun < obj_1d(current_val):
                    a[i] = res.x
            except Exception:
                pass

        # Renormalize
        S_now = np.sum(a)
        if S_now > 1e-10:
            a = a * (S_target / S_now)

        new_score = compute_score(a)
        if new_score < best_score:
            best_score = new_score
            best = a.copy()

    return best, best_score


def annealing_polish(a_init, time_limit, seed=123):
    """Simulated annealing for final polish with diverse moves."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S_target = np.sum(a_init)

    a = a_init.astype(np.float64).copy()
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score

    temperature = 0.0012
    cooling = 0.99985
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.30:
            # Targeted peak reduction via pair manipulation
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(8, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(15, len(other) // 4)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.012, size=n_inc))
        elif r < 0.50:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.018, size=width))
        elif r < 0.65:
            # Multi-block small changes
            for _ in range(6):
                width = max(1, int(rng.integers(1, max(2, n // 25))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.012, size=width))
        elif r < 0.80:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.18, 0.18) * max(a[i], a[j]) * 0.5
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        elif r < 0.92:
            # Triple redistribution
            idx3 = rng.choice(n, size=3, replace=False)
            deltas = rng.normal(0, 0.08, 3) * max(a[idx3])
            for k in range(3):
                candidate[idx3[k]] = max(0, a[idx3[k]] + deltas[k])
        else:
            # Small global scaling
            candidate = a * np.exp(rng.normal(0.0, 0.005, size=n))

        candidate = np.maximum(candidate, 0)
        S_cand = np.sum(candidate)
        if S_cand > 1e-10:
            candidate = candidate * (S_target / S_cand)

        new_score = compute_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = candidate.copy()
        else:
            acceptance = np.exp(-(new_score - current_score) / max(temperature, 1e-12))
            if rng.random() < acceptance:
                a = candidate
                current_score = new_score

        temperature = max(temperature * cooling, 1e-7)

    return best, best_score


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length using cubic interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def make_initializations(n, rng):
    """Create diverse initializations for a sequence of length n."""
    inits = {}
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)

    # Uniform
    inits['uniform'] = np.ones(n)

    # Bimodal / edge-heavy
    inits['edge_heavy'] = 0.4 + 0.6 * (1 - 4 * t * (1 - t)) ** 2
    inits['edge_heavy'] += 0.03 * rng.standard_normal(n)
    inits['edge_heavy'] = np.maximum(inits['edge_heavy'], 0.01)

    # Centre-heavy (cosine)
    inits['centre'] = 0.4 + 0.6 * np.cos(np.pi * (t - 0.5)) ** 2
    inits['centre'] += 0.03 * rng.standard_normal(n)
    inits['centre'] = np.maximum(inits['centre'], 0.01)

    # Chirp
    inits['chirp'] = 1.0 + 0.4 * np.cos(2 * np.pi * 2.0 * t**2)
    inits['chirp'] += 0.04 * rng.standard_normal(n)
    inits['chirp'] = np.maximum(inits['chirp'], 0.01)

    # Multi-tone
    a = np.ones(n)
    for k in range(1, 5):
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.2 / k * np.cos(2 * np.pi * freq * t + phase)
    a += 0.03 * rng.standard_normal(n)
    inits['multi_tone'] = np.maximum(a, 0.01)

    # Random
    inits['random'] = np.maximum(1.0 + 0.35 * rng.standard_normal(n), 0.05)

    # Smooth random (low-pass filtered)
    raw = 1.0 + 0.5 * rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(raw, kernel, mode='same')
    inits['smooth_random'] = np.maximum(smoothed, 0.05)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: subgradient descent with soft-max, coordinate
    descent refinement, and simulated annealing. Uses multiple starting points,
    sequence lengths, and initialization strategies.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 90.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # === Phase 1: Quick subgradient from provided/best sequence ===
    if best_seq is not None:
        n1 = len(best_seq)
        t1 = min(80.0, max(30.0, (deadline - time.monotonic()) * 0.05))
        print(f"\n=== Phase 1: Quick subgradient (n={n1}, t={t1:.0f}s) ===")
        f_b, s_b = subgradient_optimize(best_seq, t1, T_init=0.3, T_min=0.01, seed=1)
        print(f"  Score: {s_b:.6f}")
        if s_b < best_score:
            best_score = s_b
            best_seq = f_b
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 2: Extended subgradient from best with exploration ===
    if best_seq is not None:
        n2 = len(best_seq)
        t2 = min(500.0, max(100.0, (deadline - time.monotonic()) * 0.30))
        print(f"\n=== Phase 2: Extended subgradient (n={n2}, t={t2:.0f}s) ===")

        # Run multiple subgradient passes with different T schedules
        a = best_seq.copy()
        for pass_num in range(3):
            t_remaining = min(t2, deadline - time.monotonic() - 500.0)
            if t_remaining < 20:
                break
            T_start = 0.4 - pass_num * 0.1
            T_end = 0.005 - pass_num * 0.001
            f_b, s_b = subgradient_optimize(a, t_remaining, T_init=max(T_start, 0.1),
                                            T_min=max(T_end, 0.002), seed=10 + pass_num)
            print(f"  Pass {pass_num+1}: score={s_b:.6f}")
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b
                print(f"  *** New best: {best_score:.6f} ***")
            a = f_b

    # === Phase 3: Exploratory runs with different n and initializations ===
    print(f"\n=== Phase 3: Exploration ===")
    n_values = [256, 384, 512, 640, 768, 1024]
    init_names = ['edge_heavy', 'chirp', 'smooth_random', 'multi_tone', 'random', 'centre']

    configs = []
    for n in n_values:
        for init_name in init_names:
            configs.append((n, init_name))
    if best_seq is not None:
        for n in n_values:
            if n != len(best_seq):
                configs.append((n, 'resampled'))

    rng.shuffle(configs)
    n_configs = len(configs)
    explore_deadline = deadline - 350.0  # reserve time for phases 4-5
    per_config = max(10.0, (explore_deadline - time.monotonic()) / max(1, n_configs))

    explored = 0
    for ci, (n, init_name) in enumerate(configs):
        if time.monotonic() > explore_deadline:
            break
        time_left = explore_deadline - time.monotonic()
        if time_left < 15:
            break
        remaining = n_configs - ci
        t_per = min(per_config * 1.5, max(12.0, time_left / max(1, remaining)))
        if t_per < 8:
            break

        if init_name == 'resampled' and best_seq is not None:
            a0 = resample_sequence(best_seq, n)
        else:
            inits = make_initializations(n, rng)
            a0 = inits.get(init_name, inits['random'])

        f_b, s_b = subgradient_optimize(a0, t_per, T_init=0.5, T_min=0.01,
                                        seed=100 + ci, momentum=0.4)
        explored += 1
        if s_b < best_score:
            best_score = s_b
            best_seq = f_b
            print(f"  [{explored}] n={n}, {init_name}: {s_b:.6f} *** NEW BEST ***")
        else:
            if explored <= 10 or s_b < best_score + 0.05:
                print(f"  [{explored}] n={n}, {init_name}: {s_b:.6f}")

    # === Phase 4: Coordinate descent refinement ===
    if best_seq is not None:
        t4 = min(250.0, max(30.0, deadline - time.monotonic() - 200.0))
        if t4 > 20:
            print(f"\n=== Phase 4: Coordinate descent (t={t4:.0f}s) ===")
            f_b, s_b = coordinate_refine(best_seq, t4, seed=200)
            print(f"  Score: {s_b:.6f}")
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Annealing polish ===
    if best_seq is not None:
        t5 = min(250.0, max(30.0, deadline - time.monotonic() - 80.0))
        if t5 > 20:
            print(f"\n=== Phase 5: Annealing polish (t={t5:.0f}s) ===")
            f_b, s_b = annealing_polish(best_seq, t5, seed=300)
            print(f"  Score: {s_b:.6f}")
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 6: Final subgradient refinement ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        t6 = max(10.0, deadline - time.monotonic() - 15)
        if t6 > 15:
            print(f"\n=== Phase 6: Final subgradient (t={t6:.0f}s) ===")
            f_b, s_b = subgradient_optimize(best_seq, t6, T_init=0.08, T_min=0.003,
                                            seed=400, momentum=0.6)
            print(f"  Score: {s_b:.6f}")
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b
                print(f"  *** New best: {best_score:.6f} ***")

    # Final short annealing if time permits
    if best_seq is not None and time.monotonic() < deadline - 10:
        t7 = max(5.0, deadline - time.monotonic() - 8)
        if t7 > 10:
            f_b, s_b = annealing_polish(best_seq, t7, seed=500)
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 512
