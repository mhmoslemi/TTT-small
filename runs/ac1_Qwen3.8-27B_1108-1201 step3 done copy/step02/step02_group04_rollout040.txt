def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def subgradient_step(a, conv, M, step_size, alpha=0.1):
    """One subgradient step to reduce the peak of auto-convolution.
    a: probability vector (sum=1, non-negative)
    conv: auto-convolution of a
    M: current max of conv
    step_size: learning rate
    alpha: weight for near-peak positions
    """
    n = len(a)
    # Find positions near the peak
    threshold = M * (1 - alpha)
    peak_mask = conv >= threshold
    
    # Compute subgradient: for each i, sg[i] = sum_{k in peaks} a[k-i] (where valid)
    sg = np.zeros(n)
    for k in np.where(peak_mask)[0]:
        k = int(k)

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            sg[j] += a[k - j]
    
    # Update: a_new = a - step_size * sg, then project
    a_new = a - step_size * sg
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:

        if w < 1e-10:

        j_start = max(0, k - n + 1)

    sg_centered = sg - sg.mean()
    log_update = -eta * sg_centered
    log_update -= log_update.max()  # numerical stability
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:

def lp_step(a, M, timeout=5):
    """One LP step: find g that maximizes sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix A where A[k,i] = a[k-i] for valid k-i
    # (a*g)(k) = sum_i a[max(0,k-n+1):min(n-1,k)] restricted appropriately
    # Actually (a*g)(k) = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * g[k-i]
    
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):  # k = i + j for j in 0..n-1
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])  # a[j] where j = k - i
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',

    if res.success:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-15:
            g /= s
            return g
    return None

def compute_subgrad(a, weights):
    """Compute sg[j] = sum_k weights[k] * a[k-j] for valid indices.
    a: length n
    weights: length 2n-1
    Returns: sg of length n
    """
    n = len(a)
    L = 2*n - 1
    # sg[j] = sum_{m=0}^{n-1} a[m] * weights[j+m] (where j+m < L)
    # This is the cross-correlation we can compute via FFT
    
    # Pad to appropriate size
    N = 1 << (n + L - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses ramp-like initializations (which give ~1.336 for n=512) refined
with FFT-based mirror descent and LP steps.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_subgrad_fft(a, weights):
    """Compute sg[j] = sum_k weights[k] * a[k-j] via FFT cross-correlation."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (n + L - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    corr = np.fft.irfft(A * np.conj(W), N)
    return corr[:n]

def mirror_descent(a, conv, eta, temperature):
    """One mirror descent step."""
    n = len(a)
    M = conv.max()
    if M < 1e-15:
        return a
    # Weights for each convolution position
    # High weight for positions near the peak
    shifted = conv / M
    weights = np.exp(temperature * (shifted - 1.0))
    # This gives weight ~1 at peak, ~exp(-temperature*(1-conv/M)) elsewhere
    
    # Compute subgradient
    sg = compute_subgrad_fft(a, weights)
    
    # Mirror descent update
    sg_centered = sg - sg.mean()
    log_update = -eta * sg_centered
    log_update -= log_update.max()
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s
    return a_new

def lp_step(a, M, timeout=5):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0. Return normalized g or None."""
    n = len(a)
