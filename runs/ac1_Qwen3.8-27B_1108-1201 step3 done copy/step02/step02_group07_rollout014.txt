"""
Multi-phase optimization minimizing peak self-convolution of non-negative sequences.
Combines: (1) LP-based direction finding with adaptive M-tightening, (2) Nesterov
subgradient descent with simplex projection, (3) coordinate descent polish,
(4) targeted mass redistribution. Multi-start across lengths and initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build matrix A (2n-1 x n) such that A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, timeout=5.0):
    """Solve LP: max 1'b s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_max_conv(a):
    """Compute subgradient of max_k conv(a,a)[k] with respect to a.
    Uses top-k convolution entries for a smoother gradient estimate."""
    n = len(a)
    c = fft_convolve(a)
    top_k = min(10, len(c))
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        # d c[ki] / da[i] = 2*a[ki-i] if 0 <= ki-i < n
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for i in range(lo, hi + 1):
            g[i] += 2.0 * a[ki - i]
    g /= float(top_k)
    return g


def make_hann(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian(n, rng):
    t = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-t ** 2 / (2 * sigma ** 2))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bartlett(n, rng):
    t = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
    a = 0.2 + 0.8 * w
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_windowed(n, rng):
    t = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a = (0.5 + rng.random(n)) * (0.3 + 0.7 * w)
    return np.clip(a, 0.01, None)


def make_flattened(seq, alpha=0.3):
    """Blend sequence toward its mean to flatten."""
    a = np.asarray(seq, dtype=np.float64)
    mean_a = np.mean(a)
    return np.clip((1 - alpha) * a + alpha * mean_a, 0.01, None)

            # Escape: try different strategy
            rng_esc = np.random.default_rng(42 + it * 7919)
            if stall % 2 == 0:
                f += 0.05 * rng_esc.standard_normal(n)
            else:
                # Targeted perturbation on high-contributing elements
                c = fft_convolve(f)
                peak_k = int(np.argmax(c))
                contribs = np.zeros(n)
                for j in range(n):
                    idx = peak_k - j
                    if 0 <= idx < n:
                        contribs[j] = f[j] * f[idx]
                top_idx = np.argsort(contribs)[-max(5, n // 10):]
                f[top_idx] *= np.exp(rng_esc.normal(-0.1, 0.05, size=len(top_idx)))
                bottom_idx = np.argsort(contribs)[:max(5, n // 10)]
                f[bottom_idx] *= np.exp(rng_esc.normal(0.1, 0.05, size=len(bottom_idx)))
            f = np.clip(f, 0.01, 1000.0)
            M_factor = min(1.0, M_factor + 0.1)
            stall = 0

    return best_f, best_score


def nesterov_polish(f_init, time_limit, verbose=True):
    """Nesterov-accelerated subgradient descent with simplex projection."""
    n = len(f_init)
    S = float(np.sum(f_init))
    f = f_init.astype(np.float64).copy()
    f = f / S  # normalize to sum=1
    x = f.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0

    # Adaptive learning rate
    lr = 0.01
    lr_decay = 0.9995
    momentum = 0.75

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        # Nesterov lookahead
        x_eval = x + momentum * (x - f)
        x_eval = np.clip(x_eval, 0.0, None)
        Sx = np.sum(x_eval)
        if Sx < 0.01:
            x_eval = np.clip(f, 0.01, None)
            Sx = np.sum(x_eval)
        x_eval = x_eval / Sx

        g = subgradient_max_conv(x_eval)
        # Remove mean to maintain sum constraint approximately
        g = g - np.mean(g)

        # Update
        v = momentum * v + lr * g
        x = x - v
        x = simplex_project(x, 1.0)

        f = x
        it += 1

        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"      [nest {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
            lr *= lr_decay

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def coordinate_descent_polish(f_init, time_limit, verbose=True):
    """Coordinate descent with multiplicative updates."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    best_f = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    factors = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        # Each sweep: update each coordinate
        for i in range(n):
            if time.monotonic() - t0 >= time_limit - 2.0:
                break
            current = a[i]
            if current < 1e-6:
                # Try to increase from near-zero
                a[i] = 0.01
                s = score_fast(a)
                if s > best_score * 1.1:
                    a[i] = current
                continue

            best_local = current
            best_local_score = score_fast(a)

            for fac in factors:
                a[i] = current * fac
                s = score_fast(a)
                if s < best_local_score:
                    best_local_score = s
                    best_local = fac

            a[i] = current * best_local

        sweep += 1
        s = score_fast(a)
        if s < best_score:
            if verbose and sweep % 2 == 0:
                print(f"      [coord {time.monotonic()-t0:6.1f}s] sweep={sweep} score={s:.6f} best={best_score:.6f}")
            best_score = s
            best_f = a.copy()
        elif sweep > 3:
            # Randomize order for diversity
            pass

    return best_f, best_score


def mass_redistribute(f, time_limit, verbose=True):
    """Targeted mass redistribution: reduce peak contributors, increase troughs."""
    n = len(f)
    a = f.astype(np.float64).copy()
    best_f = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(a)
        peak_k = int(np.argmax(c))
        S = float(np.sum(a))

        # Compute contribution of each element to the peak
        contribs = np.zeros(n)
        for j in range(n):
            idx = peak_k - j
            if 0 <= idx < n:
                contribs[j] = a[j] * a[idx]

        # Also compute contribution to the second-highest
        peak_k2 = int(np.argsort(c)[-2]) if len(c) > 1 else 0
        contribs2 = np.zeros(n)
        for j in range(n):
            idx = peak_k2 - j
            if 0 <= idx < n:
                contribs2[j] = a[j] * a[idx]

        total_contrib = contribs + 0.5 * contribs2

        # Reduce top contributors, increase bottom contributors
        n_top = max(3, n // 15)
        n_bot = max(3, n // 15)
        top_idx = np.argsort(total_contrib)[-n_top:]
        bot_idx = np.argsort(total_contrib)[:n_bot]

        reduction = 0.05
        mass_removed = 0.0
        for i in top_idx:
            mass = a[i] * reduction
            a[i] -= mass
            mass_removed += mass
        a = np.clip(a, 0.0, None)

        # Distribute mass to bottom contributors
        if mass_removed > 0 and len(bot_idx) > 0:
            per = mass_removed / len(bot_idx)
            for i in bot_idx:
                a[i] += per
            a = np.clip(a, 0.0, 1000.0)

        s = score_fast(a)
        it += 1
        if s < best_score:
            best_score = s
            best_f = a.copy()
        else:
            a = best_f.copy()
            # Try more aggressive reduction
            reduction = 0.1
            for i in top_idx:
                mass = a[i] * reduction
                a[i] -= mass
                mass_removed += mass
            a = np.clip(a, 0.0, None)
            if mass_removed > 0 and len(bot_idx) > 0:
                per = mass_removed / len(bot_idx)
                for i in bot_idx:
                    a[i] += per
            a = np.clip(a, 0.0, 1000.0)
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_f = a.copy()

        if verbose and it % 50 == 0:
            print(f"      [redist {time.monotonic()-t0:6.1f}s] it={it} best={best_score:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search: LP + Nesterov + coordinate descent + redistribution."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = score_fast(provided)
            print(f"Provided: n={len(provided)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = provided.copy()
        except Exception:
            provided = None

    # Define tasks: (n, init_func_name)
    # Focus on n=256 primarily (where previous best was found), also try 320, 384
    tasks = []
    if provided is not None:
        pn = len(provided)
        tasks.append((pn, 'provided'))
        tasks.append((pn, 'flattened'))
    for n in [256, 320, 384]:
        tasks.append((n, 'hann'))
        tasks.append((n, 'gaussian'))
        tasks.append((n, 'random'))
    if provided is not None and len(provided) not in [256, 320, 384]:
        tasks.append((512, 'hann'))

    # Phase 1: LP optimization with multi-start
    print("\n=== Phase 1: LP-based optimization (multi-start) ===")
    time_per_task_base = (deadline - t_start - 300.0) / len(tasks)  # Reserve 300s for later phases
    time_per_task_base = max(30.0, time_per_task_base)

    for idx, (n, init_name) in enumerate(tasks):
        if time.monotonic() > deadline - 400.0:
            break

        # Generate initialization
        if init_name == 'provided' and provided is not None:
            if len(provided) == n:
                f0 = provided.copy()
            else:
                x_old = np.linspace(0, 1, len(provided))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, provided)
        elif init_name == 'flattened' and provided is not None:
            if len(provided) == n:
                f0 = make_flattened(provided, alpha=0.3)
            else:
                x_old = np.linspace(0, 1, len(provided))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, provided)
                f0 = make_flattened(f0, alpha=0.3)
        elif init_name == 'hann':
            f0 = make_hann(n, rng)
        elif init_name == 'gaussian':
            f0 = make_gaussian(n, rng)
        elif init_name == 'random':
            f0 = make_random_windowed(n, rng)
        elif init_name == 'bartlett':
            f0 = make_bartlett(n, rng)
        else:
            f0 = make_hann(n, rng)

        # Allocate time: give more to provided/better initializations
        t_alloc = time_per_task_base
        if init_name == 'provided':
            t_alloc = time_per_task_base * 1.5
        t_alloc = min(t_alloc, 300.0)
        t_alloc = max(20.0, min(t_alloc, deadline - time.monotonic() - 350.0))

        print(f"\n  Task {idx+1}/{len(tasks)}: n={n}, init={init_name}, time={t_alloc:.0f}s")
        f_best, s_best = lp_optimize(f0, t_alloc, verbose=True)
        print(f"  => score={s_best:.6f} (global best={best_score:.6f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}) ***")

    # Phase 2: Nesterov polish on best
    if best_seq is not None and time.monotonic() < deadline - 120:
        pt = min(120.0, deadline - time.monotonic() - 100.0)
        if pt > 20:
            print(f"\n=== Phase 2: Nesterov polish ({pt:.0f}s) ===")
            f_pol, s_pol = nesterov_polish(best_seq, pt, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_seq = f_pol.copy()
                print(f"  After Nesterov: {best_score:.6f}")

    # Phase 3: Coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 80:
        cd_t = min(100.0, deadline - time.monotonic() - 60.0)
        if cd_t > 15:
            print(f"\n=== Phase 3: Coordinate descent ({cd_t:.0f}s) ===")
            f_cd, s_cd = coordinate_descent_polish(best_seq, cd_t, verbose=True)
            if s_cd < best_score:
                best_score = s_cd
                best_seq = f_cd.copy()
                print(f"  After coord descent: {best_score:.6f}")

    # Phase 4: Mass redistribution
    if best_seq is not None and time.monotonic() < deadline - 60:
        mr_t = min(60.0, deadline - time.monotonic() - 40.0)
        if mr_t > 10:
            print(f"\n=== Phase 4: Mass redistribution ({mr_t:.0f}s) ===")
            f_mr, s_mr = mass_redistribute(best_seq, mr_t, verbose=True)
            if s_mr < best_score:
                best_score = s_mr
                best_seq = f_mr.copy()
                print(f"  After redistribution: {best_score:.6f}")

    # Phase 5: Final LP refinement
    if best_seq is not None and time.monotonic() < deadline - 20:
        fl_t = min(60.0, deadline - time.monotonic() - 10.0)
        if fl_t > 10:
            print(f"\n=== Phase 5: Final LP refinement ({fl_t:.0f}s) ===")
            f_fin, s_fin = lp_optimize(best_seq, fl_t, verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  Final: {best_score:.6f}")

    # Phase 6: One more coordinate descent pass if time allows
    if best_seq is not None and time.monotonic() < deadline - 5:
        cd_t2 = min(30.0, deadline - time.monotonic() - 3.0)
        if cd_t2 > 5:
            f_cd2, s_cd2 = coordinate_descent_polish(best_seq, cd_t2, verbose=False)
            if s_cd2 < best_score:
                best_score = s_cd2
                best_seq = f_cd2.copy()

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    fallback = [1.0] * 256
    return fallback
