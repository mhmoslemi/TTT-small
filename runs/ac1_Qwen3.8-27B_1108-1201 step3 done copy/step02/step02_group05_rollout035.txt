    if s < 1e-12:
        return 1e10
    a_norm = a / s
    c = fast_convolve(a_norm)
    c = np.maximum(c, 0)
    return (c**p).sum() ** (1.0/p)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            y = project_simplex(a + momentum * (a - y))  # Nesterov extrapolation
        else:

            y = a.copy()

            break
    
    # Final score
    score = 2.0 * n * fast_convolve(best_a).max()

import numpy as np
import scipy.optimize
import time

# --- Helper functions ---

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s*s)

def pnorm_obj(a, p):
    """Scale-invariant p-norm objective."""
    s = a.sum()
    if s < 1e-12:
        return 1e10
    a_n = a / s
    c = fast_convolve(a_n)
    return (c**p).sum() ** (1.0/p)

def pnorm_grad(a, p):
    """Gradient of p-norm objective."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.zeros(n), 1e10
    a_n = a / s
    c = fast_convolve(a_n)
    c = np.maximum(c, 1e-30)
    F = (c**p).sum() ** (1.0/p)
    w = c ** (p-1)
    # cross-correlation: g[j] = sum_m a_n[m] * w[m+j]
    N = 1 << (3*n - 2 - 1).bit_length()
    A = np.fft.rfft(a_n, N)
    W = np.fft.rfft(w, N)
    g = np.fft.irfft(np.conj(A) * W, N)[:n]
    grad = 2.0 * F**(1-p) * g / s
    return grad, F

    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

# --- Optimization routines ---

def pgd_nesterov(a0, time_limit, p_start=10, p_max=200):
    """Nesterov-accelerated projected gradient on the simplex."""
    n = len(a0)
    a = project_simplex(a0.copy())
    y = a.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = 1.0 / (n * 0.1)  # initial learning rate
    best_a = a.copy()
    best_score = compute_obj(a)
    p = p_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Every 200 iters, increase p
        if it % 200 == 0:
            p = min(int(p * 1.3), p_max)
            momentum = min(0.9 + it * 0.001, 0.98)
        
        grad, F = pnorm_grad(y, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Backtracking
        step = lr
        found = False
        for _ in range(20):
            y_trial = project_simplex(y - step * grad)
            F_trial = pnorm_obj(y_trial, p)
            if F_trial <= F * (1 - 1e-8 * step):
                found = True
                break
            step *= 0.5
            if step < 1e-15:
                break
        
        if found:
            a_new = y_trial
            F_new = F_trial
        else:
            a_new = a
            F_new = pnorm_obj(a, p)
            lr *= 0.5
        
        # Nesterov update
        y = project_simplex(a_new + momentum * (a_new - a))
        a = a_new
        
        score = compute_obj(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_score

def lp_direction(f, M):
    """LP: find g >= 0, sum(g)=1, maximizing sum(g) s.t. conv(f,g) <= M."""
    n = len(f)
    # conv(f, g)[k] = sum_i f[i]*g[k-i] <= M for all k=0,...,2n-2
    # This is linear in g!
    # A_ub: (2n-1) x n matrix where A_ub[k,j] = f[k-j] if 0<=k-j<n else 0
    # b_ub: M * ones(2n-1)
    # c: -ones(n) (maximize sum(g))
    # bounds: g >= 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[k, j] = f[k-j]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                                       method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

def lp_refine(a0, time_limit):
    """Iterative LP refinement (like parent)."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = compute_obj(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        c = fast_convolve(a)
        M = c.max()
        g = lp_direction(a, M)
        
        if np.allclose(g, a, atol=1e-10):
            no_improve += 5
        else:
            # Line search for best t
            best_t, best_F = 0, pnorm_obj(a, 20)
            for t in np.linspace(0.05, 1.0, 10):
                a_t = project_simplex((1-t)*a + t*g)
                F_t = pnorm_obj(a_t, 20)
                if F_t < best_F:
                    best_F = F_t
                    best_t = t
            
            if best_t > 0:
                a_new = project_simplex((1-best_t)*a + best_t*g)
                score = compute_obj(a_new)
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
    
    return best_a, best_score

def lbfgs_refine(a0, time_limit, p=30):
    """L-BFGS-B refinement using log-parametrization."""
    n = len(a0)
    a_init = np.maximum(a0, 1e-10)
    a_init = a_init / a_init.sum()
    b_init = np.log(a_init)
    
    def obj_b(b):
        b_clip = np.clip(b, -50, 50)
        a = np.exp(b_clip - b_clip.max())
        a = a / a.sum()
        s = a.sum()
        c = fast_convolve(a)
        return (c**p).sum() ** (1.0/p)
    
    def grad_b(b):
        b_clip = np.clip(b, -50, 50)
        a = np.exp(b_clip - b_clip.max())
        a = a / a.sum()
        c = fast_convolve(a)
        c = np.maximum(c, 1e-30)
        F = (c**p).sum() ** (1.0/p)
        w = c ** (p-1)
        N = 1 << (3*n - 2 - 1).bit_length()
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        g = np.fft.irfft(np.conj(A) * W, N)[:n]
        grad_a = 2.0 * F**(1-p) * g
        # db -> da: a = softmax(b), da/db = a * (I - a^T a) (Jacobian of softmax)
        # grad_b = (grad_a / a) - a * (grad_a . a) / a (approximately)
        # More precisely: grad_b[j] = a[j] * (grad_a[j] - sum_k a[k]*grad_a[k])
        dot = np.dot(a, grad_a)
        grad_b = a * (grad_a - dot)
        return grad_b, obj_b(b)
    
    try:
        res = scipy.optimize.minimize(
            obj_b, b_init, method='L-BFGS-B', 
            jac=lambda b: grad_b(b)[0],
            options={'maxiter': 200, 'maxfun': 500, 'factr': 1e7}
        )
        if res.success or res.fun < obj_b(b_init):
            b_opt = np.clip(res.x, -50, 50)
            a_opt = np.exp(b_opt - b_opt.max())
            a_opt = a_opt / a_opt.sum()
            score = compute_obj(a_opt)
            if np.isfinite(score):
                return a_opt, score
    except Exception:
        pass
    
    score = compute_obj(a0)
    return a0, score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated projected

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)

    except Exception:
        pass
    
    def update_best(a, score):
        nonlocal best_sequence, best_score

    candidates = []
    
    for n in lengths:
        # Parent-based
        if parent is not None:
            if parent.size >= n:
                a = parent[:n].copy()
            else:
                a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            a = np.maximum(a, 0)
            a = project_simplex(a)
            candidates.append(('parent', a))
        
        # Uniform
        candidates.append(('uniform', np.ones(n) / n))
        
        # Raised cosine
        x = np.arange(n) / (n - 1) if n > 1 else np.array([0.5])
        rc = 0.5 * (1 + np.cos(2 * np.pi * x))
