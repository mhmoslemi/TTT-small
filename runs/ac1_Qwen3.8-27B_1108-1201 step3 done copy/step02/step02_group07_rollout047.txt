Hybrid optimization for minimizing peak autocorrelation of non-negative sequences.
Uses projected gradient descent with adaptive steps, LP-based escape, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

# Helper functions (top-level)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def peak_grad(a, top_k=10):
    """Compute subgradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    c = fft_convolve(a, a)
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        for j in range(n):
            idx = ki - j
            if 0 <= idx < n:
                g[j] += a[idx]
    g /= float(top_k)
    return g, c

def projected_gradient_descent(a_init, time_limit, lr=0.01, top_k=10, verbose=True):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S * np.sqrt(n)  # normalize
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    it = 0
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2:
            break
        
        g, c = peak_grad(a, top_k)
        g -= np.mean(g)  # remove mean to preserve sum (approximately)
        momentum = beta * momentum + (1 - beta) * g
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 1e-6, None)
        # Renormalize

        if stall > 200:
            break
    
    return best_a, best_score

def lp_step(a, M, timeout=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Build constraint matrix implicitly using the structure
    # conv(a,b)[k] = sum_i a[i]*b[k-i] <= M
    # This is a system of (2n-1) constraints on n variables
    
    # For efficiency, use the Toeplitz-like structure
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except:
        pass
    return None

        i = rng.integers(0, n)
        
        # Try different values
        base = a[i]
        best_val = base
        best_s = best_score
        
        for scale in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:

            if a[i] < 0:

                best_val = a[i]
        
        a[i] = best_val
        
        if best_s < best_score:

            best_a = a.copy()
    
    return best_a, best_score

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def peak_subgradient(a, top_k=8):
    """Subgradient of max(conv(a,a)) w.r.t. a. Returns (grad, conv)."""
    n = len(a)
    c = fft_convolve(a, a)
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        # d/d a[j] conv(a,a)[ki] = 2*a[ki-j] if 0 <= ki-j < n
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        g[lo:hi + 1] += 2.0 * a[ki - np.arange(lo, hi + 1)]
    g /= float(top_k)
    return g, c


def build_A_matrix(f):
    """Build matrix A where A@b = conv(f,b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. conv(f,b) <= M*1, b >= 0."""
    n = len(f)
    A = build_A_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            best_val = base
            for scale in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a[i] = base * scale
                s = score_fast(a)
                if s < best_score:
                    best_score = s
                    best_val = a[i]
                    best_a = a.copy()
            a[i] = best_val

        it += 1
        if verbose and it % 50 == 0:
            print(f"    [CD {time.monotonic()-t0:5.1f}s] it={it:4d} best={best_score:.6f}")

    return best_a, best_score


def simple_coord_descent(a_init, time_limit, verbose=False):
    """Simple random coordinate descent."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        i = rng.integers(0, n)
        base = a[i]
        if base < 1e-10:
            continue
        for scale in [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]:
            a[i] = base * scale
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                break
        # Restore if no improvement
        if a[i] != base and score_fast(a) >= best_score:
            pass  # keep the best

        it += 1
        if verbose and it % 200 == 0:
            print(f"    [SCD {time.monotonic()-t0:5.1f}s] it={it:5d} best={best_score:.6f}")

    return best_a, best_score


def make_init(n, kind, seed=0):
    """Generate initial sequence of given length and type."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)

    if kind == 'flat':
        a = np.ones(n)
    elif kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    elif kind == 'raised_cosine':
        roll = max(1, n // 8)
        a = np.ones(n)
        for i in range(roll):
            val = 0.5 * (1.0 + np.cos(np.pi * i / (2.0 * roll)))
            a[i] = val
            a[n - 1 - i] = val
    elif kind == 'hann_sq':
        base = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a = base ** 2 + 0.1
    elif kind == 'peaked':
        center = n / 2.0
        a = np.exp(-0.3 * ((t - center) / (n / 4.0)) ** 2)
    elif kind == 'u_shape':
        a = 0.4 + 0.6 * (1.0 - 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1))))
    elif kind == 'w_shape':
        a = 0.5 + 0.3 * np.cos(4.0 * np.pi * t / max(1, n - 1))
    elif kind == 'smooth_random':
        # Random with smoothing
        raw = 0.5 + rng.random(n)
        # Simple moving average smoothing
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(raw, kernel, mode='same')
    else:
        a = np.ones(n)

    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Explore different lengths and initializations (250s)
    Phase 2: Deep hybrid optimization on best (700s)
    Phase 3: PGD + multiplicative fine-tuning (400s)
    Phase 4: Peak-targeted coordinate descent (300s)
    Phase 5: Final LP refinement (100s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 40.0)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    print(f"\n{'='*60}")
    print(f"Starting optimization. Budget: {budget_s}s, deadline in {deadline-t_start:.0f}s")
    print(f"Initial best: {best_score:.6f}")
    print(f"{'='*60}")

    # === Phase 1: Initial exploration ===
    print(f"\n=== Phase 1: Exploration (until t={250}s) ===")
    phase1_end = t_start + 250.0

    tasks = []
    if provided is not None:
        pn = len(provided)
        tasks.append((pn, 'provided', pn * 100 + 1))
        # Try different lengths from provided
        for factor in [2, 3]:
            new_n = pn * factor
            if new_n <= 4096:
                tasks.append((new_n, 'from_provided', new_n * 100 + 2))

    for n in [512, 1024, 2048, 4096]:
        tasks.append((n, 'hann', n))
        tasks.append((n, 'raised_cosine', n + 10))
        tasks.append((n, 'hann_sq', n + 20))
        tasks.append((n, 'smooth_random', n + 30))

    # Time allocation: give more time to larger n
    for idx, (n, kind, kseed) in enumerate(tasks):
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 15.0:
            break

        if kind == 'provided' and provided is not None:
            a0 = provided.copy()
        elif kind == 'from_provided' and provided is not None:
            a0 = interpolate_sequence(provided, n)
        else:
            a0 = make_init(n, kind, seed + kseed)

        t_alloc = min(50.0, remaining * 0.35)
        a_best, s_best = hybrid_optimize(a0, t_alloc, verbose=False, tag=f"P1-n{n}")

        print(f"  n={n:4d} init={kind:15s} score={s_best:.6f} (global best={best_score:.6f})")

        if s_best < best_score - 1e-12:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}, {kind}) ***")

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep optimization (best={best_score:.6f}) ===")
    phase2_end = t_start + 950.0
    if best_seq is not None and time.monotonic() < phase2_end:
        t_alloc = min(phase2_end - time.monotonic() - 10.0, 700.0)
        print(f"  Allocating {t_alloc:.0f}s for deep optimization")
        a_best, s_best = hybrid_optimize(best_seq, t_alloc, verbose=True, tag="P2")
        if s_best < best_score - 1e-12:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  Phase 2 result: {best_score:.6f}")
        else:
            print(f"  Phase 2: no improvement (stayed at {best_score:.6f})")

    # === Phase 3: PGD with different parameters ===
    print(f"\n=== Phase 3: PGD/MUL fine-tuning (best={best_score:.6f}) ===")
    phase3_end = t_start + 1350.0
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        # Try PGD with small lr and large top_k
        t1 = remaining * 0.4
        a1, s1 = pgd_optimize(best_seq, t1, lr=0.003, top_k=20,
                              momentum_beta=0.95, verbose=True, tag="P3a-PGD")
        if s1 < best_score - 1e-12:
            best_score = s1
            best_seq = a1.copy()
            print(f"  Phase 3a result: {best_score:.6f}")

        remaining = phase3_end - time.monotonic()
        # Try multiplicative
        if remaining > 30:
            t2 = remaining * 0.5
            a2, s2 = multiplicative_optimize(best_seq, t2, eta=0.002, top_k=15,
                                             verbose=True, tag="P3b-MUL")
            if s2 < best_score - 1e-12:
                best_score = s2
                best_seq = a2.copy()
                print(f"  Phase 3b result: {best_score:.6f}")

    # === Phase 4: Peak-targeted coordinate descent ===
    print(f"\n=== Phase 4: Peak-targeted CD (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 50.0:
        t_alloc = min(deadline - time.monotonic() - 50.0, 300.0)
        if t_alloc > 20.0:
            a_best, s_best = peak_targeted_coord_descent(best_seq, t_alloc, verbose=True)
            if s_best < best_score - 1e-12:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"  Phase 4 result: {best_score:.6f}")

    # === Phase 5: Final LP refinement ===
    print(f"\n=== Phase 5: Final LP (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        t_alloc = min(deadline - time.monotonic() - 20.0, 100.0)
        if t_alloc > 15.0:
            c = fft_convolve(best_seq, best_seq)
            M = float(np.max(c))
            b = solve_lp(best_seq, M, timeout=t_alloc * 0.5)
            if b is not None and np.sum(b) > 1e-10:
                b = b * (np.sum(best_seq) / np.sum(b))
                for t in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    a_try = (1.0 - t) * best_seq + t * b
                    a_try = np.clip(a_try, 1e-8, None)
                    s_try = score_fast(a_try)
                    if s_try < best_score - 1e-12:
                        best_score = s_try
                        best_seq = a_try.copy()
                        print(f"  Final LP (t={t}): {best_score:.6f}")

    # === Final verification ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"Elapsed: {time.monotonic()-t_start:.1f}s")
        print(f"{'='*60}")
        return final_list

    # Fallback
    print("WARNING: No valid sequence found, returning fallback.")
    return [1.0] * 256
