```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines soft-max smoothed gradient descent
with Adam optimizer, LP-based direction finding, and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_grad_smoothed(a, tau):
    """Compute smoothed objective and gradient using soft-max."""
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    
    # Soft-max
    c_shifted = (c - np.max(c)) / tau
    exp_c = np.exp(c_shifted)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    F = tau * (np.log(sum_exp) + np.max(c) / tau)
    
    # Gradient of F: cross-correlation of w and a
    N = 1
    while N < len(c) + n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    grad_F = np.fft.irfft(W * np.conj(A), N)[:n]
    
    obj = 2.0 * n * F / (S ** 2)
    grad = (2.0 * n / (S ** 3)) * (S * grad_F - 2.0 * F)
    
    return obj, grad

def adam_smoothed_optimize(a_init, time_limit, tau_start=0.05, tau_end=0.005, 
                           lr=0.05, verbose=False):
    """Adam optimization with annealed soft-max temperature."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a * (S / np.sum(a))  # normalize
    
    best_a = a.copy()
    best_score = local_score(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Anneal temperature: geometric from tau_start to tau_end
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        obj, grad = compute_grad_smoothed(a, tau)
        t += 1
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate schedule: decay over time
        lr_t = lr / (1.0 + 0.5 * elapsed_frac)
        
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        
        S_new = np.sum(a)
        if S_new < 1e-10:
            a = a_init.copy()
            continue
        a = a * (S / S_new)
        
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and t % 100 == 0:
            print(f"      step={t}, tau={tau:.5f}, obj={obj:.6f}, score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def build_constraint_A(f):
    """Build constraint matrix: (f*g)[k] for k=0,...,2n-2."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp(f, M, active_only=True):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    conv = fft_convolve(f, f)
    
    if active_only:
        # Only use constraints where conv is within 5% of max
        threshold = 0.95 * M
        active = np.where(conv >= threshold)[0]
        A = build_constraint_A(f)[active]
        b_ub = conv[active]  # Use actual values for tighter constraints
    else:
        A = build_constraint_A(f)
        b_ub = np.full(2 * n - 1, M)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 20})
    if result.success:
        return result.x
    return None

def lp_optimize(f_init, time_limit, t_init=0.3, verbose=False):
    """LP-based iterative optimization."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    rng_local = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = solve_lp(f, M)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (S / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        S_new = np.sum(f_new)
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (S / S_new)
        
        score_new = local_score(f_new)
        
        if score_new < best_score:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant < 10:
                f = f_new  # Accept anyway
            else:
                f = f + 0.01 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (S / np.sum(f))
                t = max(0.1, t * 0.9)
                stagnant = 0
    
    return best_f, best_score

def sa_polish(f_init, time_limit, rng, verbose=False):
    """Simulated annealing polish."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-focused mutation
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.7:
            # Local perturb
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Symmetric perturb
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            scale = np.exp(rng.normal(0.0, 0.02))
            candidate[i] *= scale
            candidate[j] *= scale
        else:
            # Global gentle
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        S_cand = np.sum(candidate)
        if S_cand < 1e-10:
            continue
        candidate = candidate * (S / S_cand)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif temperature > 1e-8 and rng.random() < np.exp(-(new_score - current_score) / temperature):
            f = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score

def make_u_shape(n, rng):
    """U-shaped initialization: high at edges, low in middle."""
    t = np.linspace(0, 1, n)
    a = 0.3 + 0.7 * (2 * t - 1) ** 2
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_cosine_init(n, rng):
    """Cosine-based initialization."""
    t = np.arange(n)
    a = 0.5 + 0.5 * np.cos(np.pi * t / (n - 1 + 1e-10))
    # Make it U-shaped (low in middle)
    a = 1.0 - a
    a = np.clip(a, 0.05, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_bimodal_init(n, rng):
    """Bimodal initialization: two bumps."""
    t = np.arange(n, dtype=np.float64)

    a += 0.02 * rng.standard_normal(n)
```

```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines soft-max smoothed gradient descent
(Adam), LP-based direction finding, and simulated annealing with peak-focused
mutations. Uses U-shaped and bimodal initializations that directly target
reduction of the central autocorrelation peak.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def cross_corr_w_a(w, a):
    """Compute h[j] = sum_k w[k]*a[k-j] for j=0..n-1 via FFT."""
    n = len(a)
    N = 1
    while N < len(w) + n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(W * np.conj(A), N)[:n]


def compute_grad_smoothed(a, tau):
    """Compute gradient of 2n*soft_max(c,tau) for fixed-scale optimization."""
    n = len(a)
    c = fft_convolve(a, a)
    c_shifted = (c - np.max(c)) / tau
    exp_c = np.exp(c_shifted)
    w = exp_c / np.sum(exp_c)
    grad_FC = cross_corr_w_a(w, a)
    grad = 2.0 * n * grad_FC
    return grad


def adam_smoothed_optimize(a_init, time_limit, tau_start=0.1, tau_end=0.005,
                           lr=0.1, beta1=0.8, beta2=0.999, verbose=False):
    """Adam optimization with annealed soft-max temperature on fixed-scale problem."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a * (S / np.sum(a))

    best_a = a.copy()
    best_score = local_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    eps = 1e-8
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        grad = compute_grad_smoothed(a, tau)
        step += 1

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        lr_t = lr / (1.0 + 2.0 * elapsed_frac)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0.0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (S / S_new)

        a = a_new
        score = local_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and step % 200 == 0:
            print(f"      step={step}, tau={tau:.5f}, score={score:.6f}, best={best_score:.6f}")

    return best_a, best_score


def build_constraint_A(f):
    """Build constraint matrix A: (f*g)[k] = sum_j f[k-j]*g[j]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, frac_active=0.9):
    """Solve LP: max sum(g) s.t. f*g <= M (active constraints only), g >= 0."""
    n = len(f)
    conv = fft_convolve(f, f)
    threshold = frac_active * M
    active = np.where(conv >= threshold)[0]

    A_full = build_constraint_A(f)
    A = A_full[active]
    b_ub = M * np.ones(len(active))

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 15})
    if result.success:
        return result.x
    return None


def lp_optimize(f_init, time_limit, t_init=0.3, verbose=False):
    """LP-based iterative optimization with adaptive step size."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    rng_local = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)

        g = solve_lp(f, M)
        if g is None:
            stagnant += 5
            if stagnant > 30:
                break
            f = f + 0.005 * rng_local.standard_normal(n)
            f = np.clip(f, 0.0, None)
            f = f * (S / np.sum(f))
            continue

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        g_rescaled = g * (S / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        S_new = np.sum(f_new)
        if S_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (S / S_new)

        score_new = local_score(f_new)

        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant <= 15:
                f = f_new
            else:
                f = f * (1.0 - 0.02) + 0.02 * rng_local.dirichlet(np.ones(n))
                t = max(0.05, t * 0.8)
                stagnant = 0

    return best_f, best_score


def sa_polish(f_init, time_limit, rng, verbose=False):
    """Simulated annealing with peak-focused and symmetric mutations."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.99985
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-focused: reduce top contributors, increase others
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.01, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(10, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.01, size=n_inc))
        elif r < 0.55:
            # Symmetric pair mutation
            for _ in range(3):
                i = int(rng.integers(0, n // 2))
                j = n - 1 - i
                scale = np.exp(rng.normal(0.0, 0.02))
                candidate[i] *= scale
                candidate[j] *= scale
        elif r < 0.75:
            # Local window perturbation
            width = max(2, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.85:
            # Asymmetric shift
            shift = int(rng.integers(1, max(2, n // 30)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))
        else:
            # Global gentle noise
            candidate *= np.exp(rng.normal(0.0, 0.004, size=n))

        candidate = np.clip(candidate, 0.0, None)
        S_cand = np.sum(candidate)
        if S_cand < 1e-10:
            continue
        candidate = candidate * (S / S_cand)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif temperature > 1e-10:
            delta = new_score - current_score
            if rng.random() < np.exp(-delta / temperature):
                f = candidate
                current_score = new_score

        temperature *= cooling

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA iter={iter_count}, temp={temperature:.6f}, "
                  f"score={current_score:.6f}, best={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s")

    return best_f, best_score


def make_u_shape(n, rng):
    """U-shaped: high at edges, low in middle - directly reduces central peak."""
    t = np.linspace(0, 1, n)
    a = 0.2 + 0.8 * (2.0 * t - 1.0) ** 2
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_deep_u(n, rng):
    """Deeper U-shape with more aggressive center suppression."""
    t = np.linspace(0, 1, n)
    a = 0.05 + 0.95 * (2.0 * t - 1.0) ** 3
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bimodal(n, rng):
    """Two Gaussian bumps at 25% and 75% positions."""
    t = np.arange(n, dtype=np.float64)
    c1, c2 = n * 0.25, n * 0.75
    w = n * 0.12
    a = np.exp(-((t - c1) ** 2) / (2 * w ** 2)) + np.exp(-((t - c2) ** 2) / (2 * w ** 2))
    a += 0.4
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_trimodal(n, rng):
    """Three Gaussian bumps."""
    t = np.arange(n, dtype=np.float64)
    centers = [n * 0.15, n * 0.5, n * 0.85]
    w = n * 0.1
    a = np.ones(n) * 0.2
    for c in centers:
        a += np.exp(-((t - c) ** 2) / (2 * w ** 2))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chebyshev(n, rng):
    """Based on Chebyshev polynomial (alternating high/low)."""
    t = np.linspace(0, np.pi, n)
    a = 0.5 + 0.5 * np.abs(np.cos(n * t / 2))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: gradient descent (soft-max + Adam),
    LP-based refinement, and simulated annealing.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided: n={len(provided_seq)}, score={s_prov:.6f}")
        best_score = s_prov
        best_seq = provided_seq.copy()

    n_base = len(provided_seq) if provided_seq is not None else 256

    # Build list of (name, array) initializations
    inits = []
    if provided_seq is not None:
        inits.append(('provided', provided_seq.copy()))
    inits.append(('u_shape', make_u_shape(n_base, rng)))
    inits.append(('deep_u', make_deep_u(n_base, rng)))
    inits.append(('bimodal', make_bimodal(n_base, rng)))
    inits.append.trimodal if False else None
    inits.append(('trimodal', make_trimodal(n_base, rng)))
    inits.append(('chebyshev', make_chebyshev(n_base, rng)))

    # Try a larger length
    n_large = 384 if n_base <= 256 else 512
    inits.append((f'u_{n_large}', make_u_shape(n_large, rng)))
    inits.append((f'bimo_{n_large}', make_bimodal(n_large, rng)))

    # === Phase 1: Gradient optimization on multiple inits ===
    print("\n" + "=" * 60)
    print("Phase 1: Soft-max gradient optimization")
    print("=" * 60)

    phase1_budget = (deadline - time.monotonic()) * 0.50
    per_init = phase1_budget / len(inits)

    for name, a0 in inits:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(per_init, deadline - time.monotonic() - 30)
        if t_alloc < 8:
            break

        s0 = local_score(a0)
        print(f"  [{name}] n={len(a0)}, init={s0:.6f}, alloc={t_alloc:.0f}s")

        a_grad, sc_grad = adam_smoothed_optimize(a0, t_alloc, tau_start=0.1,
                                                  tau_end=0.003, lr=0.1)
        print(f"  [{name}] grad result: {sc_grad:.6f}")

        if sc_grad < best_score:
            best_score = sc_grad
            best_seq = a_grad.copy()
            print(f"  *** New best: {best_score:.6f} ({name}) ***")

    # === Phase 2: LP refinement ===
    print("\n" + "=" * 60)
    print("Phase 2: LP-based refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        lp_time = min(remaining * 0.25, 250.0)
        if lp_time > 15:
            print(f"  n={len(best_seq)}, alloc={lp_time:.0f}s")
            a_lp, sc_lp = lp_optimize(best_seq, lp_time, t_init=0.25)
            print(f"  LP result: {sc_lp:.6f}")
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Second gradient pass (lower tau, lower lr) ===
    print("\n" + "=" * 60)
    print("Phase 3: Fine gradient refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        g2_time = min(remaining * 0.25, 200.0)
        if g2_time > 10:
            print(f"  n={len(best_seq)}, alloc={g2_time:.0f}s")
            a_g2, sc_g2 = adam_smoothed_optimize(best_seq, g2_time, tau_start=0.03,
                                                  tau_end=0.001, lr=0.04)
            print(f"  Fine grad result: {sc_g2:.6f}")
            if sc_g2 < best_score:
                best_score = sc_g2
                best_seq = a_g2.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: SA polish ===
    print("\n" + "=" * 60)
    print("Phase 4: Simulated annealing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(remaining * 0.5, 350.0)
        if sa_time > 10:
            print(f"  n={len(best_seq)}, alloc={sa_time:.0f}s")
            a_sa, sc_sa = sa_polish(best_seq, sa_time, rng)
            print(f"  SA result: {sc_sa:.6f}")
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Final micro-optimization ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Quick LP
            a_f, sc_f = lp_optimize(best_seq, min(remaining * 0.5, 60), t_init=0.1)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()

            remaining2 = deadline - time.monotonic() - 5
            if remaining2 > 10:
                # Quick SA
                a_s, sc_s = sa_polish(best_seq, remaining2, rng)
                if sc_s < best_score:
                    best_score = sc_s
                    best_seq = a_s.copy()

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
```
