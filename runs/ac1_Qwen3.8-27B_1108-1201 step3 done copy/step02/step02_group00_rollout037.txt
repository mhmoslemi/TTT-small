Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothing with FISTA,
continuation method, and multi-start strategy.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(a, w):
    """Compute g[i] = 2 * sum_j a[j] * w[j+i] using FFT cross-correlation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(np.conj(A) * W, fft_size)
    return 2.0 * corr[:n]

def softmax_weights(conv, temp):
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w

def fista_step(a, temp, time_per_temp, n):
    """Run FISTA for a given temperature level."""
    z = a.copy()
    t_fista = 1.0
    t0 = time.monotonic()
    
    # Estimate step size
    conv_z = fast_convolve(z)
    g = compute_gradient_fft(z, softmax_weights(conv_z, temp))
    L = 4.0 / temp  # rough estimate
    alpha = 1.0 / L
    # Cap step size
    alpha = min(alpha, 0.5)
    
    best_a = a.copy()
    conv_a = fast_convolve(a)
    best_local = conv_a.max()
    
    while time.monotonic() - t0 < time_per_temp:
        conv_z = fast_convolve(z)
        w = softmax_weights(conv_z, temp)
        g = compute_gradient_fft(z, w)
        
        # Gradient step
        a_new = project_simplex(z - alpha * g)
        
        # Backtracking line search (simplified: just check if objective decreased)
        conv_new = fast_convolve(a_new)
        f_new = temp * np.log(np.sum(np.exp((conv_new - conv_new.max()) / temp))) + conv_new.max()
        # Actually, let's just check the max value
        if conv_new.max() > conv_z.max() * 1.5:
            alpha *= 0.5
            a_new = project_simplex(z - alpha * g)
            conv_new = fast_convolve(a_new)

        z_new = a_new + ((t_fista - 1) / t_new) * (a_new - a)
        z = z_new
        a = a_new
        t_fista = t_new

            best_a = a.copy()

def optimize_continuation(a0, time_limit, max_temp=0.1, min_temp=0.001, n_temps=10):
    """Continuation method: optimize from high to low temperature."""
    n = len(a0)
    a = a0.copy()
    
    # Geometric temperature schedule
    temps = np.geomspace(max_temp, min_temp, n_temps)
    time_per_temp = time_limit / n_temps
    
    t_start = time.monotonic()
    
    for i, temp in enumerate(temps):
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 1.0:
            break
        t_alloc = min(time_per_temp * 2, remaining - 0.5)  # allow some extra time
        if t_alloc < 0.5:

        obj = 2.0 * n * conv.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Soft-max smoothing with FISTA (accelerated projected gradient) and continuation
2. Active-set exact subgradient polishing
3. Multi-start across different lengths and initializations
Exploits convexity of the problem for global convergence guarantees.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, temp):
    """Compute softmax weights over convolution values."""
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def gradient_soft_max(a, w):
    """Compute gradient of soft-max objective using FFT cross-correlation.
    g[i] = 2 * sum_k w[k] * a[k-i]
    """
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    R = np.fft.irfft(np.conj(A) * W, fft_size)
    return 2.0 * R[:n]


def gradient_active_set(a, conv, epsilon_frac=0.01):
    """Compute exact subgradient using the active set (peaks near max)."""
    n = len(a)
    max_conv = conv.max()
    threshold = max_conv * (1.0 - epsilon_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    # Uniform weights over active set
    w = np.zeros(len(conv))
    w[active] = 1.0 / len(active)
    return gradient_soft_max(a, w)


def fista_optimize(a0, time_limit, temp, L_init=None):
    """FISTA (accelerated projected gradient) for a fixed temperature.
    Returns (best_a, best_max_conv).
    """
    n = len(a0)
    z = a0.copy()
    t_fista = 1.0
    t0 = time.monotonic()

    conv_z = fast_convolve(z)
    w = softmax_weights(conv_z, temp)
    g = gradient_soft_max(z, w)

    if L_init is not None:
        L = L_init
    else:
        L = 6.0 / temp
    alpha = 1.0 / L
    alpha = min(alpha, 1.0)

    best_a = a0.copy()
    conv_a = fast_convolve(a0)
    best_local = conv_a.max()
    prev_val = best_local
    stagnant = 0

        conv_z = fast_convolve(z)
        w = softmax_weights(conv_z, temp)
        g = gradient_soft_max(z, w)

        # Gradient step with backtracking
        a_new = project_simplex(z - alpha * g)
        conv_new = fast_convolve(a_new)

        # Simple backtracking: if max increased significantly, halve step
        if conv_new.max() > conv_z.max() * 1.1 + 1e-10:

    return best_a, best_local


def optimize_continuation(a0, time_limit, max_temp=0.05, min_temp=0.001, n_temps=8, verbose=False):
    """Continuation method: optimize from high to low temperature."""
    n = len(a0)
    a = a0.copy()

    temps = np.geomspace(max_temp, min_temp, n_temps)
    t_start = time.monotonic()
    L_prev = None

    for i, temp in enumerate(temps):
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 1.0:
            break
        t_alloc = min(remaining * 1.5 / (n_temps - i), remaining - 0.5)
        if t_alloc < 1.0:
            break

        a, cur_max = fista_optimize(a, t_alloc, temp, L_init=L_prev)

        # Update L estimate for next iteration
        L_prev = 6.0 / temp

        if verbose and (i % 2 == 0 or i == n_temps - 1):
            elapsed = time.monotonic() - t_start
            obj = 2.0 * n * cur_max
            print(f"    temp={temp:.5f}, obj={obj:.6f}, time={elapsed:.1f}s/{time_limit:.0f}s")

    return a


def gradient_active_polish(a, time_limit, epsilon_frac=0.01):
    """Polish using exact active-set subgradient with adaptive step size."""
    n = len(a)
    a_cur = a.copy()
    t0 = time.monotonic()

    conv = fast_convolve(a_cur)
    best_max = conv.max()
    best_a = a_cur.copy()

    # Polyak step size using lower bound
    lower_bound = 1.0 / (2 * n - 1)  # trivial bound on max(conv)
    f_current = best_max
    f_lower = lower_bound
    alpha = 0.1

    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a_cur)
        cur_max = conv.max()
        f_current = cur_max

        # Compute active set subgradient
        g = gradient_active_set(a_cur, conv, epsilon_frac)

        # Polyak step size
        if f_current - f_lower > 1e-10:
            g_norm_sq = np.dot(g, g)
            if g_norm_sq > 1e-15:
                alpha_polyak = (f_current - f_lower) / g_norm_sq
                alpha = min(0.5, max(0.01, alpha_polyak * 0.5))

        # Gradient step
        a_new = project_simplex(a_cur - alpha * g)
        conv_new = fast_convolve(a_new)

        # Backtracking
        steps = 0
        while conv_new.max() > cur_max and steps < 10:
            alpha *= 0.5
            a_new = project_simplex(a_cur - alpha * g)
            conv_new = fast_convolve(a_new)
            steps += 1

        a_cur = a_new

        if conv_new.max() < best_max - 1e-12:
            best_max = conv_new.max()
            best_a = a_cur.copy()
            stagnant = 0
            # Increase epsilon to include more peaks
            epsilon_frac = min(0.1, epsilon_frac * 1.2)
        else:
            stagnant += 1
            if stagnant % 100 == 0:
                # Perturb
                rng_local = np.random.default_rng()
                noise = rng_local.standard_normal(n) * 0.002
                a_cur = project_simplex(a_cur + noise)

        if stagnant > 500:
            break

    return best_a, best_max


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    inits['ramp_rev'] = inits['ramp'][::-1].copy()

    # Asymmetric bump (key for beating symmetric bound)
    bump = np.exp(-x / (0.2 * n))
    inits['bump_left'] = bump / bump.sum()
    bump_r = np.exp(-(n - 1 - x) / (0.2 * n))
    inits['bump_right'] = bump_r / bump_r.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.4 * n) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal edges
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['edges'] = bi / bi.sum()

    # Asymmetric bimodal
    abi = np.zeros(n)
    abi[:n // 3] = 1.0
    abi[-n // 6:] = 0.5
    inits['asym_edges'] = abi / abi.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # Sine-based
    sine = np.abs(np.sin(2 * np.pi * x / (2 * n))) + 0.05
    inits['sine'] = sine / sine.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        p_sum = p.sum()
        if p_sum > 1e-10:
            p_norm = p / p_sum
            if len(p_norm) == n:
                inits['parent'] = p_norm.copy()
            elif len(p_norm) > n:
                idx = np.linspace(0, len(p_norm) - 1, n)
                sampled = np.interp(idx, np.arange(len(p_norm)), p_norm)
                sampled = np.maximum(sampled, 0)
                s = sampled.sum()
                if s > 1e-10:
                    inits['parent_ds'] = sampled / s
            else:
                idx = np.linspace(0, len(p_norm) - 1, n)
                sampled = np.interp(idx, np.arange(len(p_norm)), p_norm)
                sampled = np.maximum(sampled, 0)
                s = sampled.sum()
                if s > 1e-10:
                    inits['parent_up'] = sampled / s

    return inits


def propose_candidate():
    """
    Main search function. Uses multi-start FISTA with soft-max continuation,
    followed by active-set polishing.
    """
    seed = 123
    budget_s = 1800.0
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: len={len(p)}, score={parent_score:.6f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Multi-start exploration (450s)
    print("=== Phase 1: Multi-Start Exploration ===")
    lengths_to_try = [128, 256, 384, 512]

    if parent is not None:
        p_len = len(parent)
        if p_len in lengths_to_try:
            lengths_to_try.remove(p_len)
        lengths_to_try.insert(0, p_len)
        # Add nearby lengths
        for alt in [p_len - 64, p_len + 64]:
            if 64 <= alt <= 1024 and alt not in lengths_to_try:
                lengths_to_try.append(alt)
        lengths_to_try = lengths_to_try[:5]

    phase1_time = 450.0
    time_per_n = phase1_time / len(lengths_to_try)

    scan_results = []

    for n in lengths_to_try:
        if time.monotonic() > start_time + phase1_time + 60:
            break
        t_alloc = min(time_per_n, deadline - time.monotonic() - (budget_s - 30 - phase1_time - 60))
        if t_alloc < 20:
            continue
        t_alloc = min(t_alloc, phase1_time)

        inits = generate_initializations(n, rng, parent)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_ds', 'parent_up', 'bump_left', 'ramp', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        time_per_init = t_alloc / len(init_names)

        for name in init_names:
            if time.monotonic() > start_time + phase1_time + 60:
                break
            t_init = min(time_per_init, start_time + phase1_time + 60 - time.monotonic())
            if t_init < 10:
                break

            a0 = inits[name]
            a_opt = optimize_continuation(a0, t_init, max_temp=0.04, min_temp=0.002, n_temps=5, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization (900s)
    print("=== Phase 2: Deep Optimization ===")
    scan_results.sort(key=lambda x: x[0])

    top_configs = []
    seen_lengths = set()
    for s, n, name, a in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, name, a))
            seen_lengths.add(n)
        if len(top_configs) >= 4:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_time = 900.0
    time_per_deep = phase2_time / max(1, len(top_configs))

    for n, name, a0 in top_configs:
        if time.monotonic() > start_time + phase1_time + 60 + phase2_time:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 300)
        if t_alloc < 30:
            break

        # Deep continuation with more temperature levels
        a_opt = optimize_continuation(a0, t_alloc, max_temp=0.02, min_temp=0.0005, n_temps=10, verbose=True)

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}, {name}: score={score:.6f}")

    # Phase 3: Active-set polishing (300s)
    print(f"=== Phase 3: Active-Set Polish (best={best_score:.6f}) ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        phase3_time = 300.0
        t_start3 = time.monotonic()

        # Multiple polish rounds with different epsilon
        for epsilon in [0.005, 0.01, 0.02, 0.003]:
            elapsed3 = time.monotonic() - t_start3
            if elapsed3 > phase3_time or time.monotonic() > deadline - 30:
                break
            t_polish = min(40.0, phase3_time - elapsed3, deadline - time.monotonic() - 30)
            if t_polish < 10:
                break

            a_opt, obj = gradient_active_polish(best_sequence, t_polish, epsilon_frac=epsilon)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish eps={epsilon}: score={score:.6f}")

        # Final FISTA polish at very low temperature
        elapsed3 = time.monotonic() - t_start3
        remaining3 = phase3_time - elapsed3
        if remaining3 > 30 and time.monotonic() < deadline - 30:
            a_opt = optimize_continuation(best_sequence, min(remaining3, 120),
                                          max_temp=0.005, min_temp=0.0002, n_temps=8, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final FISTA: score={score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
