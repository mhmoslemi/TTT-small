```python
def solve_lp(f, M):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    # (f*g)[k] = sum_j f[k-j]*g[j] for valid j
    # This is a linear function of g with coefficients f[k-j]
    
    # Build the constraint matrix A where A[k,j] = f[k-j] if valid, 0 otherwise
    # A has shape (2n-1, n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (vector of M's), g >= 0
    b = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x
    else:
        return None
```
