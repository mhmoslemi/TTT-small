import numpy as np

import time


def compute_score(a):
    """Compute the evaluation score for sequence a."""
    n = len(a)
    if np.sum(a) < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = np.max(c)
    return float(2 * n * max_c / (np.sum(a))**2)


def build_conv_matrix(n, a):
    """Build sparse convolution matrix for LP constraints."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_direction(a, M):

    n = len(a)
    A = build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub,

    if result.success and result.x is not None:

    return None

Uses multiple initializations (ramp, quadratic, sine, random) and scipy HiGHS
LP solver for direction finding. Includes adaptive step sizes and a fallback 
local search for robustness.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_score(a):
    """Compute the evaluation score for a non-negative sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')

            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None

"""
LP-based Frank-Wolfe optimization minimizing 2n*max(conv(a,a))/(sum(a))^2
for non-negative sequences. Uses scipy HiGHS LP for direction finding with
multiple analytical initializations and adaptive step sizes.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def build_conv_matrix(n, a):
    """Sparse matrix A: (A @ g)[k] = sum_j a[k-j]*g[j] = (a*g)[k]."""
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_direction(a, M, top_k=None):
    """LP: max sum(g) s.t. (a*g)[k]<=M (optionally restricted), g>=0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    
    if top_k is not None and top_k < A.shape[0]:
        # Select most binding constraints
        binding = np.array(A.sum(axis=1)).flatten()
        idx = np.argsort(binding)[::-1][:top_k]
        A_use = A[idx, :]
        b_use = np.full(len(idx), M)
    else:
        A_use = A
        b_use = np.full(A.shape[0], M)
    
    c_obj = -np.ones(n)
    bounds = [(0, 100.0)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_use, b_ub=b_use,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None


def lp_optimize(a0, max_iters=25, deadline=None, verbose=True, top_k=None):
    """Frank-Wolfe iterations with LP direction finding."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, None)
    if np.sum(a) < 1e-10:
        return a0.copy(), float('inf')

    best_a = a.copy()
    best_score = compute_score(a)
    t_list = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.05, 0.15, 0.4]
    
    if verbose:
        print(f"  Start: score={best_score:.6f}")
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline - 30:
            break
        
        c = np.convolve(a, a)
        M = float(np.max(c))
        
        g = solve_direction(a, M, top_k=top_k)
        if g is None or np.sum(g) < 1e-10:
            break
        
        g = g / np.sum(g)
        
        best_local_score = best_score
        best_local_a = best_a.copy()
        improved = False
        
        for t in t_list:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            s = np.sum(cand)
            if s < 1e-10:

            cand = cand / s
