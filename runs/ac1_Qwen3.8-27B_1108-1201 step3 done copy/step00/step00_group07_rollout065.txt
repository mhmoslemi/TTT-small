import numpy as np
import time


# ----------------------------------------------------------------------------
# Minimize  2 * n * max(conv(a,a)) / sum(a)^2  over non-negative sequences a.
# The objective is scale invariant, so we optimize with sum(a)=1 and minimize
# max(c) where c = full linear self-convolution of a.  A flat convolution has
# max(c) ~ 1/(2n) giving objective ~1; uniform gives max(c)=1/n -> objective 2.
# We push peaks below 1/n with a projected sub-gradient (momentum + adaptive
# Polyak step) that targets the top-K convolution peaks, breaking the symmetry
# that keeps symmetric sequences stuck at objective 2.
# ----------------------------------------------------------------------------


def _fft_conv(a):
    """Full linear self-convolution via FFT (returns length 2n-1)."""
    n = a.size
    N = 1 << int(np.ceil(np.log2(2 * n - 1)))
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)
    return c[:2 * n - 1]


def _subgrad_topk(a, c, K):
    """Gradient of the sum of the top-K entries of the self-convolution."""
    n = a.size
    if c.size < K:
        idx_all = np.argsort(c)[::-1]
    else:
        idx_all = np.argpartition(c, -K)[-K:]
    G = np.zeros(n, dtype=np.float64)
    for k in idx_all:
        L = k - n + 1
        if L < 0:
            L = 0
        U = k
        if U > n - 1:
            U = n - 1
        if U < L:
            continue
        idx = np.arange(L, U + 1)
        # a[k - idx] == a[idx[::-1]] because R_k is symmetric about k/2
        G[idx] += 2.0 * a[idx][::-1]
    return G


def _nn_normalize(a):
    a = np.where(a < 0.0, 0.0, a)
    s = a.sum()
    if s <= 0.0:
        return np.full_like(a, 1.0 / a.size)
    return a / s


def _optimize_one(a0, deadline, tag=""):
    a = _nn_normalize(np.asarray(a0, dtype=np.float64).copy())
    n = a.size
    v = np.zeros_like(a)
    mu = 0.7
    target = 1.04 / (2 * n - 1)      # aim a touch above the flat average
    best_a = a.copy()
    best_c = None
    best_v = float("inf")
    K = 1
    lr_cap = 0.5
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        cur = float(c.max())
        if cur < best_v - 1e-12:
            best_v = cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # adaptive Polyak step for the (sub-)gradient of the top-K max surrogate
        G = _subgrad_topk(a, c, K)
        g2 = float(G @ G)
        gap = max(cur - target, 0.0)
        if g2 > 1e-30:
            lr = min(lr_cap, gap / (g2 + 1e-15))
            lr = max(lr, 1e-6)
        else:
            lr = 1e-4
        v = mu * v - lr * G
        a = a + v
        a = np.where(a < 0.0, 0.0, a)
        a = _nn_normalize(a)

        # anneal momentum and occasionally widen the active set to escape ridges
        it += 1
        if it % 2000 == 0:

            a = _nn_normalize(a)
            v = np.zeros_like(a)
            mu = 0.7
            K = 1
            stall = 0

    return best_a, best_v

            except Exception:
                pass

        per_start = max(20, (deadline - time.monotonic()) / max(1, len(starts)))
        for si, s in enumerate(starts):
            tl = min(deadline - 15.0, time.monotonic() + per_start)
            if time.monotonic() >= tl:
                break
            a, vmax = _optimize_one(s, tl, tag=f"n{n}-s{si}")
            if a is None:
                continue
            # evaluate the true objective (sum not necessarily 1 after norm, use formula)
            obj = 2.0 * n * vmax / (a.sum() ** 2)
            print(f"[{n}/{si}] iters done  local_max_c={vmax:.3e}  obj={obj:.5f}  global={best_obj:.5f}",
                  flush=True)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    if best_a is None:
        # safe valid fallback: uniform (objective 2.0)
        a = np.ones(2048, dtype=np.float64)
        return [float(x) for x in a]

    # final confirmation with the authoritative evaluator
    fl = [float(x) for x in best_a]
    try:
        ev = float(evaluate_sequence(fl))
        print("final evaluate_sequence =", ev, " est =", best_obj, flush=True)
        if np.isfinite(ev):
            best_obj = ev
    except Exception as e:
        print("eval error", e, flush=True)

    return fl
