def _fft_convolve(a, L):

    A = np.fft.rfft(a, L)

def _gs_flat(a_init, n_iter, L):
    """Gerchberg-Saxton iteration: alternate between flat-spectrum and non-negativity."""
    n = len(a_init)
    a_pad = np.zeros(L)
    a_pad[:n] = a_init.copy()
    for _ in range(n_iter):
        A = np.fft.rfft(a_pad)
        mag = np.abs(A)
        target_mag = np.full(mag.shape, mag[1:].mean())
        target_mag[0] = mag[0]
        A_new = target_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, L)
        a_pad = np.clip(a_new[:n], 0.0, None)
    return a_pad

    """Compute softmax-weighted subgradient from top-K convolution lags."""
    n = len(a)
    idx = np.argpartition(conv, -K)[-K:]
    vals = conv[idx]

    weights = np.exp((vals - max_val) / T)
