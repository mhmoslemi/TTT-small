"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative step sequences.

Method: LP-based conditional gradient (Frank-Wolfe style). At each step we
solve  max 1^T g  s.t.  conv(a, g) <= M,  g >= 0  where M = max conv(a,a).
The cross-convolution bound guarantees the peak cannot grow to first order in
the blend, so a small step a <- (1-t)a + t*g (renormalized) reduces the peak.
We use the active-set LP (only near-binding lags) for speed, with a full LP
fallback, and randomly-restarted random/bump perturbations to escape local
minima. The best sequence ever found is always returned before the deadline.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False


def _normalize(a):
    """Clip to non-negative and renormalize to unit sum; returns (a, ok)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-8 or not np.isfinite(s):
        return a, False
    return a / s, True


def _build_conv_matrix(a):
    """Sparse matrix T (shape (2n-1, n)) with (T @ g)[k] = sum_i a[i]*g[k-i]."""
    n = a.size
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    mat = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return mat.tocsr()


def _solve_lp_full(a, M, timeout=20.0):
    """Solve the full LP; return g >= 0 or None."""
    if not _HAVE_SCIPY:
        return None
    n = a.size
    try:
        T = _build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"time_limit": timeout, "output_flag": 0})
        if res is not None and res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def _solve_lp_active(a, M, frac=0.85, timeout=15.0):
    """Solve LP on near-active lags only; verify against full convolution."""

        return None

    cols = []
    data = []

        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            rows.extend([int(k)] * js.size)
            cols.extend([int(j) for j in js])
            vals = a[k - hi:k - lo + 1][::-1]
            data.extend([float(v) for v in vals])
    if not rows:
        return None
    try:
        T = coo_matrix((data, (rows, cols)), shape=(active.size, n)).tocsr()
        c = -np.ones(n)
        b_ub = np.full(active.size, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"time_limit": timeout, "output_flag": 0})

            g = np.maximum(res.x, 0.0)

                check = np.convolve(a, g)
                mx = check.max()
                if mx > M * (1.0 + 1e-6):
                    g = g * (M / mx)
                return g
    except Exception:
        pass
    return None
