import numpy as np

def compute_convolution(a, N):
    """Compute linear convolution a*a via FFT."""
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)
    return b[:2*len(a)-1]

def compute_objective_and_grad(a, T, N):
    """Compute logsumexp(b/T) and its gradient."""
    n = len(a)
    b = compute_convolution(a, N)
    # Log-sum-exp for numerical stability
    b_max = np.max(b)
    log_sum_exp = b_max/T + np.log(np.sum(np.exp((b - b_max)/T)))
    # Softmax weights
    w = np.exp((b - b_max)/T)
    w /= np.sum(w)
    # Gradient
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(W * np.conj(A), N)
    grad = (2.0/T) * c[:n]
    return log_sum_exp, grad

"""
Minimizes 2n*max(a*a)/(sum(a))^2 over non-negative sequences using
log-sum-exp smoothed gradient optimization in log-softmax parameterization.
Multi-phase T-annealing with L-BFGS-B, multiple lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize

_DEADLINE = float('inf')

def _fft_convolve(a, N):
    """Linear convolution a*a via real FFT, returning first 2n-1 elements."""
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * len(a) - 1]

def _obj_grad(z, T, N, n):
    """
    Compute logsumexp(b/T) and its gradient w.r.t. z where a = softmax(z).
    Returns (objective, gradient).
    """
    zs = z - np.max(z)
    exp_z = np.exp(zs)
    a = exp_z / np.sum(exp_z)

    b = _fft_convolve(a, N)
    bmax = np.max(b)
    L = bmax / T + np.log(np.sum(np.exp((b - bmax) / T)))

    w = np.exp((b - bmax) / T)
    w /= np.sum(w)

    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(W * np.conj(A), N)
    g = (2.0 / T) * c[:n]

    ag = np.dot(a, g)
    grad_z = a * (g - ag)

    return L, grad_z

def _next_pow2(m):
    p = 1
    while p < m:
        p *= 2
    return p

def _opt_from_z(z0, n, T_schedule, max_iter, seed):

    for T in T_schedule:

            break

        def f(x):
            return _obj_grad(x, T, N, n)[0]

        def g(x):
            return _obj_grad(x, T, N, n)[1]

        try:
            res = minimize(f, z, jac=g, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-12})
            z = res.x
        except Exception:
            pass

        a = np.exp(z - np.max(z))
        a = a / np.sum(a)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_z = z.copy()
            print(f"    T={T:.2e} score={score:.6f}")

    return best_z, best_score
