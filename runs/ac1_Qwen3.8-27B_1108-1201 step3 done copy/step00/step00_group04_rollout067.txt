    inits.append(s_asym / s_asym.sum())

    # 8. Random
    rand = 0.1 + rng.random(n)
    inits.append(rand / rand.sum())

    # 9. Quadratic ramp
    quad = 0.1 + 0.9 * (x / n) ** 2
    inits.append(quad / quad.sum())

    # 10. Bimodal asymmetric
    bm = np.exp(-8.0 * ((x - 0.2 * n) / n) ** 2) + 0.4 * np.exp(-8.0 * ((x - 0.7 * n) / n) ** 2)
    inits.append(bm / bm.sum())

    # 11. Anti-symmetric emphasis: high at start, very low near mirror positions
    anti = np.ones(n)
    anti[:n // 6] = 4.0
    anti[n // 6:n // 3] = 2.0
    anti[n // 3:2 * n // 3] = 0.8
    anti[2 * n // 3:5 * n // 6] = 0.3
    anti[5 * n // 6:] = 0.05
    inits.append(anti / anti.sum())

    # 12. Frequency-structured: alternating high/low with decay
    freq = np.ones(n)
    freq[::2] = 2.0
    freq[1::2] = 0.3
    freq *= np.exp(-2.0 * x / n)
    inits.append(freq / freq.sum())

    # 13. Another chirp variant
    chirp2 = 1.0 + 0.8 * np.cos(4.0 * np.pi * x * x / (n * 1.0) + 1.0)
    chirp2 = np.clip(chirp2, 0.0, None)
    if chirp2.sum() > 1e-10:
        inits.append(chirp2 / chirp2.sum())

    num_inits = len(inits)
    best = None
    best_score = float('inf')

    # Phase 1: Gradient-based optimization (75% of time)
    time_for_opt = max(30.0, (float(budget_s) - 20.0) * 0.75)
    time_per_init = time_for_opt / num_inits
    opt_start = time.monotonic()

    for idx in range(num_inits):
        if time.monotonic() > opt_start + time_for_opt - 2.0:
            break

        init_deadline = min(time.monotonic() + time_per_init, opt_start + time_for_opt - 1.0)

        a_opt, local_score = _optimize_from_init(inits[idx], N, n, rng, init_deadline)

        # Evaluate with exact method
        if local_score < best_score * 1.01:
            exact = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best = a_opt.copy()
                print(f"Init {idx}: new best exact score = {best_score:.6f}")

        if (idx + 1) % 4 == 0 or idx == num_inits - 1:
            elapsed = time.monotonic() - opt_start
            print(f"  Phase 1 progress: {idx+1}/{num_inits} inits, elapsed={elapsed:.1f}s, best={best_score:.6f}")

    # Phase 2: Coordinate descent polish (20% of time)
    if best is not None:
        polish_deadline = time.monotonic() + max(5.0, (float(budget_s) - 20.0) * 0.2)
        a_polish, ps = _polish_coordinate(best, N, n, rng, polish_deadline)
        exact = float(evaluate_sequence(a_polish.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best_score = exact
            best = a_polish.copy()
            print(f"After polish: score = {best_score:.6f}")

    # Phase 3: More gradient optimization from the best found (remaining time)
    if best is not None and time.monotonic() < deadline - 10.0:
        rng2 = np.random.default_rng(seed + 12345)
        a_restart = best + rng2.normal(0.0, 0.01 * best.max(), n)
        a_restart = np.clip(a_restart, 0.0, None)
        a_restart = a_restart / a_restart.sum()
        a_opt2, _ = _optimize_from_init(a_restart, N, n, rng2, deadline - 3.0)
        exact2 = float(evaluate_sequence(a_opt2.tolist()))
        if np.isfinite(exact2) and exact2 < best_score:
            best_score = exact2
            best = a_opt2.copy()
            print(f"Phase 3: new best score = {best_score:.6f}")

    # Final
    if best is None:
        best = np.full(n, 1.0 / n)

    final_score = float(evaluate_sequence(best.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    return [float(v) for v in best]
