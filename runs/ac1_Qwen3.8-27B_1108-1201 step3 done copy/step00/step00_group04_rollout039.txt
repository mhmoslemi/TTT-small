```python
def subgradient_step(a, lr, n):
    c = np.convolve(a, a)
    k_star = np.argmax(c)
    # Subgradient of max(c) w.r.t. a[j] is 2*a[k_star - j] (with a[x]=0 for x out of [0,n-1])
    # Build the subgradient vector
    sg = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            sg[j] = 2 * a[idx]
    # Update
    a_new = a - lr * sg
    # Project onto {a >= 0, sum(a) = sqrt(2n)}
    S = np.sqrt(2 * n)
    # Water-filling: find lambda such that sum(max(0, a_new - lambda)) = S
    # Binary search for lambda
    lo, hi = a_new.min() - 1, a_new.max() + 1
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.maximum(0, a_new - mid)
        if proj.sum() > S:
            lo = mid
        else:
            hi = mid
    lambda_star = (lo + hi) / 2
    a_new = np.maximum(0, a_new - lambda_star)
    # Normalize to exact sum (numerical error)
    if a_new.sum() > 0:
        a_new *= S / a_new.sum()
    return a_new
```

```python
subgradient polishing, and multi-length random restarts.
Key insight: the objective is scale-invariant; we normalize sum(a)=sqrt(2n)
so the objective equals max(convolve(a,a)). Asymmetric perturbations of
the uniform sequence reduce central convolution peaks.
"""
import numpy as np

def _build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_iteration(a, t=0.3):
    """One LP-based iteration: find best g with (a*g)[k] <= max(a*a)."""
    n = len(a)
    c = np.convolve(a, a)
    M = np.max(c)
    S = np.sum(a)
    if M <= 0 or S < 1e-12:
        return a, M
    A = _build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return a, M
    g = res.x
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return a, M
    g = g * (S / sum_g)
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 1e-12:
        a_new *= (S / s)
    return a_new, M


def _subgradient_step(a, lr):
    """One subgradient step on min max(convolve(a,a)) s.t. sum(a)=const, a>=0."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    # Subgradient of max(c) w.r.t. a[j] = 2*a[k_star - j] (if in range)
    sg = np.zeros(n)
    for j in range(n):

        if proj.sum() > S:
            lo = mid
        else:

    a_new = np.maximum(0.0, a_new - lam)
    s = np.sum(a_new)
    if s > 1e-12:
        a_new *= (S / s)
    return a_new


def _run_lp_opt(a, max_iters=500, t=0.25, time_limit=60.0):
    """Run LP iterations with adaptive step size."""
    n = len(a)
    S = np.sum(a)
    c = np.convolve(a, a)
    best_M = np.max(c)
    best_a = a.copy()
    t = 0.25

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:

        a, M = _lp_iteration(a, t=t)
        c = np.convolve(a, a)
        cur_M = np.max(c)

            best_a = a.copy()
            t = min(0.6, t * 1.05)
        else:

    return best_a, best_M


def _run_subgradient_opt(a, max_iters=2000, time_limit=60.0):
    """Subgradient method with Polyak-like step size."""
    n = len(a)
    S = np.sum(a)
    best_M = np.max(np.convolve(a, a))
    best_a = a.copy()
    lr = 1.0 / np.sqrt(max(n, 1))

            best_M = cur_M
            best_a = a.copy()
        # Decay step size
        lr *= 0.9995
        if lr < 1e-6:
            break
    return best_a, best_M
```
