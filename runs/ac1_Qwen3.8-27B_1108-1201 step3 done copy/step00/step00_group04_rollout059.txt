import time

def compute_convolution_fft(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    return C[:2*n-1]

def compute_gradient_fft(a, active_mask):
    """Compute the subgradient of the max convolution value."""
    n = len(a)
    N = 1
    while N < 3*n - 2:
        N *= 2
    A = np.fft.rfft(a, N)
    M = np.fft.rfft(active_mask, N)

    # But with rfft, the conjugate is straightforward
    C = np.fft.irfft(np.conj(A) * M, N)
    # C[i] for i=0,...,n-1 gives the gradient
    return C[:n]

def project_simplex(v, S=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = S}."""
    n = len(v)
    # Binary search for the threshold
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0) * (S / np.maximum(v - theta, 0).sum() if np.maximum(v - theta, 0).sum() > 0 else 1.0)
