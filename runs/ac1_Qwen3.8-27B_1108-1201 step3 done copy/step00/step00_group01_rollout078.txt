```python
"""
Search for a nonnegative coefficient vector minimizing
    2*n*max_k((a*a)_k) / (sum(a))^2.

The program combines:
  * a strong explicit initialization from the central-binomial shape;
  * smooth log-sum-exp gradient descent with FFT-based convolutions;
  * linear-program directions (full LP for small n, active-set top-constraint
    LP for larger n);
  * a short multiplicative local search.
"""

import numpy as np
import time

try:
    from scipy.sparse import csc_matrix
    from scipy.optimize import linprog
except Exception:
    csc_matrix = None
    linprog = None


def _fft_len(m):
    """Smallest power of two at least m."""
    L = 1
    while L < int(m):
        L <<= 1
    return L


def _normalize(a):
    """Clip to valid nonnegative range and normalize sum to one."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        a.fill(1.0)
        a /= float(a.size)
    else:
        a /= s
    return a


def _self_conv(a):
    """Full linear self-convolution using FFT, clipped at zero."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _fft_len(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _quick_obj(a):
    """Fast objective using FFT; matches evaluate_sequence up to roundoff."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return float("inf")
    c = _self_conv(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(c.max()) / (s * s))


def _central_initial(n):
    """
    First n coefficients of sqrt(1/(1-z)), i.e. binomial(2k,k)/4^k.
    This already has all early convolution coefficients equal and gives
    asymptotic objective pi/2.
    """
    n = int(max(1, n))
    x = np.empty(n, dtype=np.float64)
    x[0] = 1.0
    for i in range(1, n):
        x[i] = x[i - 1] * float((2 * i - 1)) / float((2 * i))
    return _normalize(x)


def _grad_from_p(a, p):
    """
    Gradient of log-sum-exp of a*a:
        g_i = 2 * sum_j p_{i+j} a_j,
    computed by FFT.
    """
    n = a.size
    L = _fft_len(3 * n - 2)
    Af = np.fft.rfft(a, L)
    pf = np.fft.rfft(p[::-1], L)
    h = np.fft.irfft(Af * pf, L)
    return 2.0 * h[n - 1:2 * n - 1][::-1]


def _gradient_descent(a, deadline, name="grad"):
    """Smooth log-sum-exp descent under sum(a)=1 and a>=0."""
    a = _normalize(a)
    n = a.size
    best = a.copy()
    best_obj = _quick_obj(a)

    rng = np.random.default_rng(12345)
    start = time.monotonic()
    total_plan = max(1.0, deadline - start)
    norm_a = float(np.linalg.norm(a))
    eta = 0.20
    no_accept = 0
    it = 0
    last_print = start

    while time.monotonic() < deadline and it < 1000000:
        it += 1

        c = _self_conv(a)
        cmax = float(c.max())
        s = float(a.sum())
        obj = float(2.0 * n * cmax / (s * s)) if s > 1e-12 else float("inf")

        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best = a.copy()
            no_accept = 0

        elapsed = max(1e-9, time.monotonic() - start)
        frac = min(1.0, elapsed / total_plan)
        T_factor = 0.25 * (0.035 + 0.965 * (1.0 - frac))
        T = max(cmax * T_factor, 1e-15)

        shift = (c - cmax) / T
        shift = np.clip(shift, -50.0, 0.0)
        ex = np.exp(shift)
        sumex = float(ex.sum())
        if not np.isfinite(sumex) or sumex <= 0.0:
            # fallback: uniform over current maxima
            ex = (c >= cmax - 1e-14).astype(np.float64)
            sumex = float(ex.sum())
            if sumex <= 0:
                ex.fill(1.0)
                sumex = float(ex.size)
        LSE = cmax + T * float(np.log(sumex))
        p = ex / sumex

        g = _grad_from_p(a, p)
        g = np.asarray(g, dtype=np.float64)
        g -= g.mean()
        ng = float(np.linalg.norm(g))

        accepted = False
        if np.isfinite(ng) and ng > 1e-14:
            d = g / ng  # gradient direction; descent is -d
            for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
                if time.monotonic() > deadline:
                    break
                step = mult * eta * norm_a
                cand = a - step * d
                cand = np.clip(cand, 0.0, 1000.0)
                cs = float(cand.sum())
                if not np.isfinite(cs) or cs < 1e-8:
                    continue
                cand /= cs

                cn = _self_conv(cand)
                cnmax = float(cn.max())
                obj_new = float(2.0 * n * cnmax)  # cand normalized
                cshift = (cn - cnmax) / T
                cshift = np.clip(cshift, -50.0, 0.0)
                s2 = float(np.exp(cshift).sum())
                if not np.isfinite(s2) or s2 <= 0:
                    continue
                LSE_new = cnmax + T * float(np.log(s2))

                allow = 1e-9 + 0.001 * max(0.0, obj - 1.0)
                if (np.isfinite(LSE_new) and LSE_new < LSE - 1e-13 and obj_new < obj + allow) or \
                   (np.isfinite(obj_new) and obj_new < obj - 1e-12):
                    a = cand
                    norm_a = float(np.linalg.norm(a))
                    accepted = True
                    if np.isfinite(obj_new) and obj_new < best_obj:
                        best_obj = obj_new
                        best = a.copy()
                    break

        if not accepted:
            no_accept += 1
            eta = max(0.003, eta * 0.70)
            if no_accept > 35:
                # multiplicative random restart of the working point
                noise = rng.normal(0.0, 0.12, n)
                cand = a * np.exp(noise)
                cand = np.clip(cand, 0.0, 1000.0)
                cs = float(cand.sum())
                if np.isfinite(cs) and cs > 1e-8:
                    a = cand / cs
                    norm_a = float(np.linalg.norm(a))
                no_accept = 0
                eta = max(0.02, eta)
            if no_accept > 250:
                break
        else:
            eta = min(0.45, eta * 1.02)

        now = time.monotonic()
        if now - last_print > 30.0 or it % 1000 == 0:
            print(f"{name}: iter={it} best={best_obj:.8f} cur_obj={obj:.8f} eta={eta:.4f}")
            last_print = now

    return best, best_obj


def _select_lp_rows(c, K, rng):
    """Select active constraints: high-convolution positions."""
    m = c.size
    K = int(max(1, min(K, m)))
    cmax = float(c.max())
    cmin = float(c.min())
    spread = cmax - cmin
    if spread > 0.0:
        thresh = cmax - 0.05 * spread
    else:
        thresh = cmax

    idx = np.nonzero(c >= thresh)[0]
    if idx.size > K:
        return idx[rng.choice(idx.size, K, replace=False)]

    if K < m:
        top_all = np.argpartition(c, -K)[-K:]
    else:
        top_all = np.arange(m)

    combined = np.unique(np.concatenate((idx, top_all)))
    if combined.size > K:
        combined = combined[rng.choice(combined.size, K, replace=False)]
    return combined[:K]


def _solve_top_lp(a, selected):
    """
    For the selected rows, solve:
        maximize sum(g)
        subject to (a*g)[k] <= 1, g >= 0.
    The returned g is only a direction and is normalized later.
    """
    if linprog is None:
        return None
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    K = len(selected)
    if K <= 0 or n <= 0:
        return None

    a_lp = np.maximum(a, 1e-12)
    try:
        A = np.zeros((K, n), dtype=np.float64)
    except Exception:
        return None

    for i in range(K):
        k = int(selected[i])
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            # Columns js..je need values a[k-js], ..., a[k-je] reversed.
            A[i, js:je + 1] = a_lp[k - je:k - js + 1][::-1]

    b = np.ones(K, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)

    res = None
    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            bounds=(0.0, 100.0),
            method="highs",
            options={"presolve": True, "time_limit": 35.0}
        )
    except Exception:
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b,
                bounds=(0.0, 100.0),
                method="highs",
                options={"presolve": True}
            )
        except Exception:
            return None

    if res is not None and getattr(res, "success", False) and res.x is not None:
        x = np.asarray(res.x, dtype=np.float64)
        x[~np.isfinite(x)] = 0.0
        return np.maximum(x, 0.0)
    return None


def _lp_heuristic(a, deadline, name="lp"):
    """Top-constraint LP方向的 Frank-Wolfe-like refinement."""
    a = _normalize(a)
    n = a.size
    if n <= 0 or linprog is None:
        return a, _quick_obj(a)

    best = a.copy()
    best_obj = _quick_obj(a)

    K = int(max(256, min(768, n // 16)))
    rng = np.random.default_rng(7777)
    alpha = 0.18
    no_accept = 0
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline and it < 100000:
        it += 1
        c = _self_conv(a)
        s = float(a.sum())
        obj = float(2.0 * n * float(c.max()) / (s * s)) if s > 1e-12 else float("inf")
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best = a.copy()

        selected = _select_lp_rows(c, K, rng)
        g = _solve_top_lp(a, selected)

        if g is None or g.size != n:
            no_accept += 1
            if no_accept >= 5:
                K = max(128, K // 2)
                no_accept = 0
            continue

        gs = float(g.sum())
        if not np.isfinite(gs) or gs < 1e-10:
            continue
        g /= gs

        accepted = False
        for al in (0.30, 0.20, 0.10, 0.05, 0.02, 0.01):
            cand = (1.0 - al) * a + al * g
            cand = np.clip(cand, 0.0, 1000.0)
            cs = float(cand.sum())
            if not np.isfinite(cs) or cs < 1e-8:
                continue
            cand /= cs
            obj_new = _quick_obj(cand)
            if np.isfinite(obj_new) and obj_new < obj - 1e-12:
                a = cand
                accepted = True
                if obj_new < best_obj:
                    best_obj = obj_new
                    best = a.copy()
                break

        if accepted:
            no_accept = 0
            alpha = min(0.40, alpha * 1.10)
        else:
            no_accept += 1
            alpha *= 0.7
            if no_accept >= 6:
                K = max(128, K // 2)
                no_accept = 0

        now = time.monotonic()
        if now - last_print > 20.0 or it % 10 == 0:
            print(f"{name}: iter={it} best={best_obj:.8f} cur={obj:.8f} K={K} al={alpha:.3f}")
            last_print = now

        if K < 128 and no_accept > 12:
            break

    return best, best_obj


def _exact_lp_refinement(a, deadline, name="exactLP"):
    """
    Full-constraint LP iteration for small n.
    Uses the exact sparse convolution matrix A[k,j] = a[k-j].
    """
    a = _normalize(a)
    n = a.size
    if n <= 0 or linprog is None or csc_matrix is None:
        return a, _quick_obj(a)
    if n > 2300:
        return _lp_heuristic(a, deadline, name)

    best = a.copy()
    best_obj = _quick_obj(a)

    rng = np.random.default_rng(991)
    alpha = 0.22
    no_accept = 0
    it = 0
    last_print = time.monotonic()

    try:
        base = np.arange(n, dtype=np.int32)
        cols = np.tile(base, n)
        rows = cols.copy()
        rows += np.tile(base, n)
        b = np.ones(2 * n - 1, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
    except Exception:
        return best, best_obj

    while time.monotonic() < deadline and it < 100000:
        it += 1
        c = _self_conv(a)
        s = float(a.sum())
        obj = float(2.0 * n * float(c.max()) / (s * s)) if s > 1e-12 else float("inf")
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best = a.copy()

        a_lp = np.maximum(a, 1e-12)
        vals = np.tile(a_lp, n)
        g = None

        for attempt in range(2):
            try:
                A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
                if attempt == 0:
                    res = linprog(
                        c_obj,
                        A_ub=A,
                        b_ub=b,
                        bounds=(0.0, 100.0),
                        method="highs",
                        options={"presolve": True, "time_limit": 60.0}
                    )
                else:
                    res = linprog(
                        c_obj,
                        A_ub=A,
                        b_ub=b,
                        bounds=(0.0, 100.0),
                        method="highs",
                        options={"presolve": True}
                    )
                break
            except Exception:
                res = None

        if res is not None and getattr(res, "success", False) and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g[~np.isfinite(g)] = 0.0
            g = np.maximum(g, 0.0)

        if g is None:
            no_accept += 1
        else:
            gs = float(g.sum())
            if not np.isfinite(gs) or gs < 1e-10:
                no_accept += 1
            else:
                g /= gs
                accepted = False
                for al in (0.30, 0.20, 0.12, 0.06, 0.03, 0.015):
                    cand = (1.0 - al) * a + al * g
                    cand = np.clip(cand, 0.0, 1000.0)
                    cs = float(cand.sum())
                    if not np.isfinite(cs) or cs < 1e-8:
                        continue
                    cand /= cs
                    obj_new = _quick_obj(cand)
                    if np.isfinite(obj_new) and obj_new < obj - 1e-12:
                        a = cand
                        accepted = True
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best = a.copy()
                        break

                if accepted:
                    no_accept = 0
                    alpha = min(0.40, alpha * 1.10)
                else:
                    no_accept += 1
                    alpha *= 0.7

        if no_accept > 10:
            # Small perturbation to avoid stagnation.
            noise = rng.normal(0.0, 0.05, n)
            cand = a * np.exp(noise)
            cand = np.clip(cand, 0.0, 1000.0)
            cs = float(cand.sum())
            if np.isfinite(cs) and cs > 1e-8:
                a = cand / cs
            no_accept = 0
            if it > 40:
                break

        now = time.monotonic()
        if now - last_print > 20.0 or it % 5 == 0:
            print(f"{name}: iter={it} best={best_obj:.8f} cur={obj:.8f} al={alpha:.3f}")
            last_print = now

    return best, best_obj


def _local_search(a, deadline, name="local"):
    """Multiplicative block perturbations. Used as final polish."""
    a = _normalize(a)
    n = a.size
    best = a.copy()
    best_obj = _quick_obj(a)
    cur = a.copy()
    cur_obj = best_obj

    rng = np.random.default_rng(314159)
    sigma = 0.35
    no_improve = 0
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline and it < 1000000:
        it += 1
        cand = cur.copy()
        n_blocks = int(rng.integers(1, 5))
        wlo = max(1, n // 3000)
        whi = max(wlo + 1, n // 25)
        for _ in range(n_blocks):
            w = int(rng.integers(wlo, whi + 1))
            st = int(rng.integers(0, max(1, n - w)))
            mult = float(np.exp(rng.normal(0.0, sigma)))
            cand[st:st + w] *= mult

        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        cs = float(cand.sum())
        if not np.isfinite(cs) or cs < 1e-8:
            continue
        cand /= cs
        obj = _quick_obj(cand)

        if np.isfinite(obj) and obj < best_obj:
            best = cand.copy()
            best_obj = obj
            cur = cand
            cur_obj = obj
            no_improve = 0
            sigma = min(0.60, sigma * 1.15)
        else:
            no_improve += 1
            if np.isfinite(obj):
                # mild simulated-annealing acceptance
                if rng.random() < float(np.exp(-max(0.0, obj - cur_obj) * 20000.0)):
                    cur = cand
                    cur_obj = obj
            if no_improve > 120:
                sigma *= 0.65
                no_improve = 0
                if sigma < 0.01:
                    sigma = 0.12
                    cur = best.copy()
                    cur_obj = best_obj

        now = time.monotonic()
        if now - last_print > 20.0 or it % 500 == 0:
            print(f"{name}: iter={it} best={best_obj:.8f} sigma={sigma:.4f}")
            last_print = now

    return best, best_obj


def _interpolate_sequence(a, n_target):
    """Stretch a shorter valid sequence to a target length."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    n_target = int(max(1, n_target))
    if m == 0:
        return _central_initial(n_target)
    if m == 1:
        out = np.full(n_target, float(a[0]), dtype=np.float64)
    elif m == n_target:
        out = a.copy()
    else:
        x_old = np.linspace(0.0, 1.0, m, dtype=np.float64)
        x_new = np.linspace(0.0, 1.0, n_target, dtype=np.float64)
        out = np.interp(x_new, x_old, a).astype(np.float64)
    return _normalize(out)


def _final_select(candidates, limit):
    """Score top FFT candidates with the official evaluator when available."""
    scored = []
    for a in candidates:
        if a is None:
            continue
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            continue
        arr = _normalize(arr)
        try:
            q = _quick_obj(arr)
        except Exception:
            q = float("inf")
        if np.isfinite(q):
            scored.append((float(q), arr))

    if not scored:
        return _central_initial(64)

    scored.sort()
    exact = []
    seen = []
    checked = 0
    eval_fn = globals().get("evaluate_sequence", None)

    for q, arr in scored:
        if checked >= 12:
            break
        if time.monotonic() > limit:
            break

        dup = False
        for _, old in seen:
            if old.size == arr.size:
                head_ok = np.allclose(arr[:8], old[:8], rtol=1e-8, atol=1e-10)
                tail_ok = np.allclose(arr[-8:], old[-8:], rtol=1e-8, atol=1e-10)
                sum_ok = abs(float(arr.sum()) - float(old.sum())) < 1e-6
                if head_ok and tail_ok and sum_ok:
                    dup = True
                    break
        if dup:
            continue

        seq = [float(x) for x in arr]
        val = q
        if eval_fn is not None:
            try:
                val = float(eval_fn(seq))
                checked += 1
            except Exception:
                val = q

        if np.isfinite(val):
            exact.append((float(val), arr))
            seen.append((q, arr))

    if exact:
        exact.sort()
        if np.isfinite(exact[0][0]):
            return exact[0][1]

    return scored[0][1]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point.
    Returns the best valid sequence found within budget_s seconds.
    """
    t0 = time.monotonic()
    total = float(budget_s)
    if not np.isfinite(total) or total <= 0.0:
        total = 1800.0

    # Reserve time for official scoring and list conversion.
    final_limit = t0 + max(60.0, total - 30.0)

    candidates = []

    # Main length: 8192 is a good balance of FFT cost and search surface.
    n_main = 8192
    main_work = _central_initial(n_main)
    main_score = _quick_obj(main_work)
    best_arr = main_work.copy()
    best_score = main_score

    candidates.append(best_arr.copy())

    # Optional fallback / reference start.
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            h_arr = _normalize(h)
            if h_arr.size > 0:
                h_score = _quick_obj(h_arr)
                if np.isfinite(h_score):
                    candidates.append(h_arr.copy())
                    if h_score < best_score:
                        best_score = h_score
                        best_arr = h_arr.copy()
        except Exception:
            pass

    # ---------------- Small full-LP phase ----------------
    # This phase often gives a strong shape that can be stretched to main length.
    now = time.monotonic()
    remaining = final_limit - now
    if remaining > 900.0:
        small_n = 2048
        small_init = _central_initial(small_n)
        candidates.append(small_init.copy())

        small_total = min(0.18 * remaining, 300.0)
        small_grad_time = min(80.0, 0.30 * small_total)
        small_grad_deadline = time.monotonic() + small_grad_time

        small_a, small_obj = _gradient_descent(small_init, small_grad_deadline, name="small-grad")
        if small_a is not None:
            candidates.append(small_a.copy())
            if np.isfinite(small_obj) and small_obj < best_score:
                best_score = small_obj
                best_arr = small_a.copy()

        now = time.monotonic()
        remaining_small = final_limit - now - small_grad_deadline
        remaining_small = max(0.0, final_limit - now)
        small_lp_time = min(0.16 * remaining, 220.0)
        if small_lp_time > 20.0:
            small_a2, small_obj2 = _exact_lp_refinement(small_a, time.monotonic() + small_lp_time, name="small-LP")
            candidates.append(small_a2.copy())
            if np.isfinite(small_obj2) and small_obj2 < best_score:
                best_score = small_obj2
                best_arr = small_a2.copy()
            if np.isfinite(small_obj2) and small_obj2 < small_obj:
                small_a = small_a2
                small_obj = small_obj2

    # If the small phase produced a better shape, stretch it to the main length.
    if best_arr.size != n_main:
        try:
            stretched = _interpolate_sequence(best_arr, n_main)
            stretched_score = _quick_obj(stretched)
            if np.isfinite(stretched_score) and stretched_score < main_score:
                main_work = stretched
                main_score = stretched_score
                candidates.append(main_work.copy())
                if stretched_score < best_score:
                    best_score = stretched_score
                    best_arr = main_work.copy()
        except Exception:
            pass

    # ---------------- Main gradient phase ----------------
    now = time.monotonic()
    remaining = final_limit - now
    if remaining > 60.0:
        grad_time = 0.55 * remaining
        grad_deadline = time.monotonic() + grad_time
        m_a, m_obj = _gradient_descent(main_work, grad_deadline, name="main-grad")
        if m_a is not None:
            candidates.append(m_a.copy())
            if np.isfinite(m_obj) and m_obj < main_score:
                main_work = m_a
                main_score = m_obj
            if np.isfinite(m_obj) and m_obj < best_score:
                best_score = m_obj
                best_arr = m_a.copy()

    # ---------------- Main top-constraint LP phase ----------------
    now = time.monotonic()
    remaining = final_limit - now
    if remaining > 40.0:
        lp_time = 0.30 * remaining
        lp_deadline = time.monotonic() + lp_time
        m_a2, m_obj2 = _lp_heuristic(main_work, lp_deadline, name="main-LP")
        if m_a2 is not None:
            candidates.append(m_a2.copy())
            if np.isfinite(m_obj2) and m_obj2 < main_score:
                main_work = m_a2
                main_score = m_obj2
            if np.isfinite(m_obj2) and m_obj2 < best_score:
                best_score = m_obj2
                best_arr = m_a2.copy()

    # ---------------- Final local polish ----------------
    now = time.monotonic()
    remaining = final_limit - now
    if remaining > 25.0:
        local_time = max(10.0, remaining * 0.85)
        m_a3, m_obj3 = _local_search(best_arr, time.monotonic() + local_time, name="local")
        if m_a3 is not None:
            candidates.append(m_a3.copy())
            if np.isfinite(m_obj3) and m_obj3 < best_score:
                best_score = m_obj3
                best_arr = m_a3.copy()

    # Ensure the currently best working sequence is explicitly represented.
    if best_arr is not None:
        candidates.append(_normalize(best_arr))

    # Final official verification and selection.
    result = _final_select(candidates, time.monotonic() if global_eval_deadline_missing else final_limit)
    result = _normalize(result)

    # Last guard: if somehow invalid, return a simple valid central sequence.
    if result.size == 0 or not np.isfinite(result).all():
        result = _central_initial(256)

    return [float(x) for x in result]
```
