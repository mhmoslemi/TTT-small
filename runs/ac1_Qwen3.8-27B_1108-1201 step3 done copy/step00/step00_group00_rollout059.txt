    a = a.copy()
    for it in range(n_iters):
        c = np.convolve(a, a)
        k_star = np.argmax(c)
        # Find indices contributing to c[k_star]
        i_start = max(0, k_star - (len(a)-1))
        i_end = min(len(a)-1, k_star)
        for i in range(i_start, i_end+1):
            j = k_star - i
            # Reduce a[i] and a[j]
            # Multiplicative update
            a[i] *= 0.999
            a[j] *= 0.999
    return a
