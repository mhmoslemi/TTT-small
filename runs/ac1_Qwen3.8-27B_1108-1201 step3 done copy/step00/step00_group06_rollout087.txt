def fft_convolve_self(a):
    """Compute a*a (linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

    """Compute np.correlate(p, a, 'valid') using FFT.
    p has length M >= len(a) = N.
    Returns array of length M - N + 1.
    """
