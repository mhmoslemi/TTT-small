"""
Minimizes 2*n*max(f*f)/(sum(f))^2 using:
1. Volterra sequence initialization (flat ramp-up autocorrelation by construction)
2. Gradient descent with soft-max smoothing of the max operator
3. Multiplicative (exponential) updates guaranteeing non-negativity
4. Momentum for accelerated convergence
5. Adaptive temperature annealing
6. Random perturbations and restarts to escape local minima

def generate_volterra(n, C=1.0):
    """Generate sequence with first n autocorrelation values equal to C."""
    f = np.zeros(n)
    f[0] = np.sqrt(C)
    for k in range(1, n):
        if k > 1:
            s = np.dot(f[1:k], f[1:k][::-1])
        else:
            s = 0.0
        f[k] = (C - s) / (2.0 * f[0])
    return f


def soft_max_gradient(f, T):
    """Compute gradient of soft-max approximation to max(full_convolve(f,f))."""
    n = len(f)
    b = np.convolve(f, f, mode='full')
    b_max = b.max()
    exp_b = np.exp(T * (b - b_max))
    weights = exp_b / exp_b.sum()
    # grad[i] = sum_k weights[k] * f[k-i] for valid k,i
    grad = np.correlate(weights, f, mode='full')[n - 1:2 * n - 1]
    return grad, b


def fast_score(f):
    """Fast score computation (no list conversion, no clamping)."""
    n = len(f)
    b = np.convolve(f, f, mode='full')
    max_b = b.max()
    sum_a = f.sum()
    if sum_a < 0.01:

    return float(2.0 * n * max_b / (sum_a ** 2))
