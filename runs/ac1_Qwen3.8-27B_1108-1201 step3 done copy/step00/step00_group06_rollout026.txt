Let me also think about a complementary approach: **coordinate descent**.

At each step, pick a coordinate $j$ and try to optimize $a[j]$ while keeping others fixed. The score as a function of $a[j]$ (with $a[j] = t$) involves:
- $C[k]$ changes by $t \cdot a[k-j] + a[j]_{old} \cdot 0$... actually it's more complex because $C[k] = \sum_i a[i] a[k-i]$ and changing $a[j]$ affects all $C[k]$ where $j$ or $k-j$ is in $[0, n-1]$.

import numpy as np
import time

def _fft_size(m):
    """Next power of 2 >= m."""
    size = 1
    while size < m:
        size <<= 1
    return size

def _compute_C(a, fft_size):
    """Compute full convolution a*a using pre-allocated FFT size."""
    n = len(a)
    fa = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(fa * fa, fft_size)[:2*n-1]
    return C

def _soft_max_objective(C, beta):
    """Compute soft-max of C and the softmax weights."""
    C_max = np.max(C)
    exp_C = np.exp(beta * (C - C_max))
    return exp_C / np.sum(exp_C)

def _compute_grad(a, C, p, grad_size):
    """Compute gradient: g[j] = 2 * sum_k p[k] * a[k-j]."""
    n = len(a)
    a_rev = a[::-1]
    P = np.fft.rfft(p, grad_size)
    Ar = np.fft.rfft(a_rev, grad_size)
    conv = np.fft.irfft(P * Ar, grad_size)
    R = conv[n-1:2*n-1]
