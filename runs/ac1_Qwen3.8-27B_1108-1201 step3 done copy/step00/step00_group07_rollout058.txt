Let me code up the subgradient approach:

```python
import numpy as np
import time

def compute_conv(a):
    return np.convolve(a, a)

def subgradient_step(a, lr):
    c = np.convolve(a, a)
    k_star = np.argmax(c)
    S = np.sum(a)
    M = c[k_star]
    
    # Gradient of c[k_star] w.r.t. a
    # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
    n = len(a)
    grad = np.zeros(n)
    # a[k_star - i] for i = 0, ..., n-1
    # k_star - i ranges from k_star down to k_star - n + 1
    valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
    indices = k_star - np.arange(n)
    grad[valid] = 2 * a[indices[valid]]
```

```python
def fast_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, k_near_peaks):
    """Compute aggregated subgradient for given peak positions."""

        grad[valid] += 2 * a[indices[valid]]
    return grad
```

Actually, I realize I should think more carefully about the step. Let me use a proper line search or at least a reasonable step size.

The key issue with subgradient methods is the step size. A common choice is:
lr_t = c / sqrt(t) for some constant c.

But this requires knowing the Lipschitz constant of the gradient, which is hard to estimate.

An alternative is to use a "normalized" step:
a_new = a - lr * direction / (||direction|| * ||a||)

This makes the step scale-invariant.

Or I can use the "projection" approach: after taking a step, project onto the constraint set (a >= 0, sum(a) = 1).

Let me think about this differently. Since the problem is scale-invariant, I can work on the "simplex" {a >= 0, sum(a) = 1}. The projection onto this simplex is well-known and can be computed in O(n log n).

Algorithm:
1. Start with a on the simplex.
2. Compute the subgradient g (of the "active" component max_k c[k]).

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using projected subgradient
descent with momentum, simplex projection, and multiple restarts.
"""
import numpy as np
import time

def _fft_convolve(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[:size]

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgradient_direction(a, c, n_peaks=5):
    """Compute descent direction using aggregated subgradient of top peaks."""
    n = len(a)
    S = np.sum(a)
    
    # Get top peaks
    if 2 * n - 1 <= n_peaks:
        peak_indices = np.arange(2 * n - 1)
    else:
        peak_indices = np.argpartition(c, -n_peaks)[-n_peaks:]
    
    # Weight peaks by their values (higher peak = more weight)
    weights = c[peak_indices]
    if weights.sum() > 0:
        weights = weights / weights.sum()
    else:
        weights = np.ones(len(peak_indices)) / len(peak_indices)
    
    # Compute aggregated gradient
    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        k = int(k)
        indices = k - np.arange(n)
        valid = (indices >= 0) & (indices < n)
        grad[valid] += w * 2.0 * a[indices[valid]]
    
    # The subgradient of max_k c_k is this weighted combination
    # The subgradient of F = 2n * max_c / S^2 w.r.t. a[i] is:
    # 2n * (grad[i] * S - 2*max_c) / S^3 (using just the top peak)
    # For the weighted version, replace max_c with weighted sum:
    max_c = c[peak_indices].max()
    direction = grad * S - 2.0 * max_c
    # We want to DESCEND, so the update is a - lr * direction_normalized
    # But direction here is the gradient (we subtract it)
    return direction

    m = np.zeros(n)
    beta = 0.9
    
    # Learning rate schedule
    lr = 0.5
    lr_decay = 0.999
    
    eval_interval = 50
    no_improve = 0
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < time_limit:

        direction = _subgradient_direction(a, c)
        
        # Momentum update
        m = beta * m + (1 - beta) * direction
        
        # Step

        # Project onto simplex
        a = _project_simplex(a)
        
        it += 1
