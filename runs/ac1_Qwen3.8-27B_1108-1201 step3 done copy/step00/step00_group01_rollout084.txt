```
        return np.convolve(a, a)
    m = 2*n -1

    if out.min() < 0:
       out = np.where(out < 0, 0, out) # small negatives
    return out
```

```python
"""
Search for a low-peak self-convolution step function.

The evaluator minimizes:
    2 * n * max(a * a) / (sum(a))^2
where a * a is numpy's full linear convolution.  The objective is scale
invariant, so the code normalizes sequences to a safe scale and runs a
mirror-descent / active-peak shaping loop.  Multiple Fourier-parameter and
random starts are used, then a round-robin full-vector optimizer polishes the
best sequences.

Algorithm outline:
1. Build a diverse set of initial sequences:
   - previous global sequence if present
   - uniform sequence
   - simple sinusoidal asymmetric sequences
   - random smooth sequences
   - low-dimensional Fourier searches using Nelder-Mead when SciPy is present
2. Run a full-vector optimizer on each candidate in round-robin slices.
   The optimizer tracks the top convolution lags, constructs a centered
   gradient for the weighted sum of those lags, and updates log-weights
   multiplicatively.  This moves mass away from coordinates that contribute
   most to the active peaks while preserving the sum approximately.
3. Keep the exact-evaluated best sequence and use the remaining time to
   polish that best sequence.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None

_BASIS_CACHE = {}


def _safe_normalize(a, n):
    """Return a clean length-n nonnegative sequence with sum at most O(n)."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size != int(n):
        if a.size < int(n):
            a = np.concatenate((a, np.zeros(int(n) - a.size, dtype=np.float64)))
        else:
            a = a[:int(n)]
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(int(n), dtype=np.float64)

    amax = float(np.max(a))
    if not np.isfinite(amax) or amax <= 0.0:
        return np.ones(int(n), dtype=np.float64)

    target = float(n)
    max_allowed_sum = 1000.0 * s / amax
    if max_allowed_sum < target:
        target = max_allowed_sum
    if target < 0.01:
        target = s

    if target > 0.0:
        a *= target / s
    a = np.clip(a, 0.0, 1000.0)

    s2 = float(np.sum(a))
    if not np.isfinite(s2) or s2 <= 0.01:
        return np.ones(int(n), dtype=np.float64)

    amax2 = float(np.max(a))
    if amax2 > 1000.0:
        a *= 1000.0 / amax2
        a = np.clip(a, 0.0, 1000.0)
    return a


def _fft_conv_self(a):
    """Full linear convolution a*a using FFT for moderate/large n."""
    n = int(a.size)
    m = 2 * n - 1
    if n <= 512:
        out = np.convolve(a, a)
        if out.min() < 0.0:
            out = np.maximum(out, 0.0)
        return out

    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    out = out[:m]
    if out.min() < 0.0:
        out = np.where(out < 0.0, 0.0, out)
    return out


def _approx_score(a):
    """Fast approximation of evaluate_sequence using FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    s = float(np.sum(a))
    if n == 0 or not np.isfinite(s) or s < 0.01:
        return np.inf
    b = _fft_conv_self(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * n * mb / (s * s))


def _fourier_basis(n, K):
    """Cache sine/cosine Fourier bases for frequencies 1..K."""
    n = int(n)
    K = int(K)
    key = (n, K)
    if key in _BASIS_CACHE:
        return _BASIS_CACHE[key]
    x = np.arange(n, dtype=np.float64)
    basis = []
    for m in range(1, K + 1):
        th = 2.0 * np.pi * m * x / float(n)
        basis.append(np.sin(th))
        basis.append(np.cos(th))
    _BASIS_CACHE[key] = basis
    return basis


def _fourier_seq(n, coeffs, K, scale=0.8):
    """Build a nonnegative sequence 1 + scale * linear Fourier sum."""
    n = int(n)
    K = int(K)
    coeffs = np.asarray(coeffs, dtype=np.float64).reshape(-1)
    if coeffs.size != 2 * K:
        coeffs = np.zeros(2 * K, dtype=np.float64)
    basis = _fourier_basis(n, K)
    g = np.zeros(n, dtype=np.float64)
    for c, b in zip(coeffs, basis):
        if c != 0.0:
            g += c * b
    a = 1.0 + float(scale) * g
    return _safe_normalize(a, n)


def _fourier_objective(coeffs, n, K):
    """Approximate score plus mild coefficient-size penalty."""
    coeffs = np.asarray(coeffs, dtype=np.float64).reshape(-1)
    n = int(n)
    K = int(K)
    if coeffs.size != 2 * K:
        return 1.0e9
    if not np.all(np.isfinite(coeffs)):
        return 1.0e9
    sc = _approx_score(_fourier_seq(n, coeffs, K, scale=0.8))
    if not np.isfinite(sc):
        return 1.0e9
    cpen = float(np.dot(coeffs, coeffs))
    if cpen > 9.0:
        sc += 0.05 * (float(np.sqrt(cpen)) - 3.0)
    return float(sc)


def _add_result(results, score, seq, limit=3):
    """Keep up to `limit` lowest-score candidates without using lambda."""
    if not np.isfinite(score):
        return results
    results.append([float(score), np.asarray(seq, dtype=np.float64).copy()])
    if len(results) > limit:
        worst = 0
        for i in range(1, len(results)):
            if results[i][0] > results[worst][0]:
                worst = i
        del results[worst]
    return results


def _fourier_search(n, end, rng, K=4):
    """Small-dimensional Fourier search to obtain strong diversifying starts."""
    n = int(n)
    K = int(K)
    dim = 2 * K
    results = []

    starts = []

    x0 = np.zeros(dim, dtype=np.float64)
    if dim > 0:
        x0[0] = 0.8
    starts.append(x0)

    x0 = np.zeros(dim, dtype=np.float64)
    if dim > 0:
        x0[0] = 0.6
    if dim > 2:
        x0[2] = 0.22
    if dim > 4:
        x0[4] = 0.12
    starts.append(x0)

    for _ in range(2):
        starts.append(rng.normal(0.0, 0.5, dim))

    maxiter = 90
    local_seconds = 10.0

    for x0 in starts:
        if time.monotonic() >= end:
            break
        xbest = np.asarray(x0, dtype=np.float64).copy()
        fbest = _fourier_objective(xbest, n, K)
        if not np.isfinite(fbest):
            xbest = np.zeros(dim, dtype=np.float64)
            if dim > 0:
                xbest[0] = 0.7
            fbest = _fourier_objective(xbest, n, K)

        if _scipy_minimize is not None:
            try:
                res = _scipy_minimize(
                    _fourier_objective,
                    xbest,
                    args=(n, K),
                    method="Nelder-Mead",
                    options={
                        "maxiter": maxiter,
                        "xatol": 1.0e-4,
                        "fatol": 1.0e-5,
                        "disp": False,
                    },
                )
                if np.isfinite(res.fun) and res.fun < fbest:
                    xbest = np.asarray(res.x, dtype=np.float64).reshape(-1).copy()
                    fbest = float(res.fun)
            except Exception:
                pass

        local_end = min(end, time.monotonic() + local_seconds)
        while time.monotonic() < local_end:
            jump = 0.25 if rng.random() < 0.1 else 0.05
            xtry = xbest + rng.normal(0.0, jump, dim)
            ftry = _fourier_objective(xtry, n, K)
            if np.isfinite(ftry) and ftry < fbest:
                xbest = np.asarray(xtry, dtype=np.float64).reshape(-1).copy()
                fbest = float(ftry)

        seq_best = _fourier_seq(n, xbest, K, scale=0.8)
        _add_result(results, fbest, seq_best, limit=3)
        seq_start = _fourier_seq(n, xbest, K, scale=0.8)
        _add_result(results, _approx_score(seq_start), seq_start, limit=3)

    out = []
    for item in results:
        out.append(item[1])
    return out


def _perturb_sequence(a, rng, amp=0.1):
    """Random smooth/block perturbations to escape flat regions."""
    n = int(a.size)
    a = np.asarray(a, dtype=np.float64).copy()
    r = float(rng.random())
    if r < 0.40:
        a *= np.exp(rng.normal(0.0, amp, n))
    elif r < 0.70:
        lo_w = max(1, n // 200)
        hi_w = max(2, n // 8)
        if hi_w <= lo_w:
            hi_w = lo_w + 1
        width = int(rng.integers(lo_w, hi_w))
        if width > n:
            width = n
        start_hi = max(1, n - width + 1)
        start = int(rng.integers(0, start_hi))
        a[start:start + width] *= np.exp(rng.normal(0.0, amp * 1.5, width))
    else:
        x = np.arange(n, dtype=np.float64)
        m = int(rng.integers(1, 6))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a *= np.exp(amp * np.sin(2.0 * np.pi * m * x / float(n) + ph))

    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return _safe_normalize(np.ones(n, dtype=np.float64), n)
    return _safe_normalize(a, n)


def _optimize(a0, end, rng, exact_every=80, label="opt"):
    """
    Full-vector active-peak mirror descent.

    Returns (best_sequence, best_exact_score, current_sequence).
    """
    n = int(np.asarray(a0).size)
    if n <= 0:
        n = 1
    a = _safe_normalize(a0, n)

    best_approx = _approx_score(a)
    best = a.copy()
    best_exact = float(evaluate_sequence(best.tolist()))

    eta = 0.5
    it = 0
    stagnation = 0
    last_print = 0.0

    while time.monotonic() < end:
        b = _fft_conv_self(a)
        maxb = float(np.max(b))
        s = float(np.sum(a))
        if s < 0.01 or not np.isfinite(maxb):
            a = best.copy()
            best_approx = _approx_score(a)
            stagnation += 1
            continue

        score = float(2.0 * n * maxb / (s * s))
        if np.isfinite(score):
            if score < best_approx:
                best_approx = score
                best = a.copy()
                stagnation = 0
            else:
                stagnation += 1

            if score > best_approx * 3.0:
                a = best.copy()
                continue
        else:
            a = best.copy()
            stagnation += 1
            continue

        it += 1

        ml = int(b.size)
        top = max(8, 8 + it // 200)
        if top > 96:
            top = 96
        if top > ml:
            top = ml

        if top >= ml:
            top_idx = np.arange(ml, dtype=np.int64)
        else:
            top_idx = np.argpartition(b, -top)[-top:].astype(np.int64)

        bt = b[top_idx]
        delta = float(bt.max() - bt.min())
        tau = max(1.0e-6, 0.25 * delta + 0.03 * maxb)
        wt = np.exp((bt - bt.max()) / tau)
        wsum = float(wt.sum())
        if not np.isfinite(wsum) or wsum <= 0.0:
            wt = np.ones(top, dtype=np.float64) / float(max(1, top))
        else:
            wt /= wsum

        cost = np.zeros(n, dtype=np.float64)
        for j in range(top):
            k = int(top_idx[j])
            w = float(wt[j])
            if w <= 0.0:
                continue

            lo = k - n + 1
            if lo < 0:
                lo = 0
            hi = k + 1
            if hi > n:
                hi = n
            if hi <= lo:
                continue

            s_start = k - lo
            s_end = k - hi + 1

            if s_end == 0:
                cost[lo:hi] += w * a[s_start::-1]
            else:
                cost[lo:hi] += w * a[s_start:s_end - 1::-1]

        mean_cost = float(np.dot(wt, bt)) / s
        d = mean_cost - cost
        dmax = float(np.max(np.abs(d)))
        if not np.isfinite(dmax) or dmax < 1.0e-12:
            a = _perturb_sequence(a, rng, 0.10)
            continue

        d /= dmax
        np.clip(d, -1.5, 1.5, out=d)

        a = a * np.exp(eta * d)
        a = _safe_normalize(a, n)

        if stagnation > 400:
            a = _perturb_sequence(best.copy(), rng, 0.25)
            eta = 0.60
            stagnation = 0
        elif stagnation > 150:
            a = _perturb_sequence(a, rng, 0.15)
            eta = max(0.10, eta * 0.80)
        elif it % 200 == 0:
            eta = min(1.00, eta * 1.03)

        if it % exact_every == 0:
            ex = float(evaluate_sequence(a.tolist()))
            if np.isfinite(ex) and ex < best_exact:
                best_exact = ex
                best = a.copy()
            if time.monotonic() - last_print > 30.0:
                print(label, "iter", it, "approx", round(score, 7),
                      "best_exact", round(best_exact, 7),
                      "eta", round(eta, 4), "top", top, flush=True)
                last_print = time.monotonic()

    ex = float(evaluate_sequence(a.tolist()))
    if np.isfinite(ex) and ex < best_exact:
        best_exact = ex
        best = a.copy()

    exb = float(evaluate_sequence(best.tolist()))
    if np.isfinite(exb) and exb < best_exact:
        best_exact = exb

    scor = _approx_score(a)
    best_scor = _approx_score(best)
    if (not np.isfinite(scor)) or (scor > best_scor * 3.0):
        a = best.copy()

    return best, float(best_exact), a


def propose_candidate(seed=42, budget_s=1800):
    """Main entry point: return a list of nonnegative heights."""
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    master_deadline = t0 + budget - 20.0
    rng = np.random.default_rng(int(seed))

    n = 6121
    base_arr = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if arr.size > 0 and np.all(np.isfinite(arr)):
            base_arr = arr.copy()
            n = int(arr.size)
    except Exception:
        pass

    if n < 1:
        n = 6121

    selected = []

    if base_arr is not None:
        selected.append(_safe_normalize(base_arr, n))

    selected.append(_safe_normalize(np.ones(n, dtype=np.float64), n))

    x = np.arange(n, dtype=np.float64)
    s1 = 1.0 + 0.8 * np.sin(2.0 * np.pi * x / float(n))
    selected.append(_safe_normalize(s1, n))

    s2 = 1.0 + 0.6 * np.sin(2.0 * np.pi * x / float(n)) + 0.25 * np.sin(4.0 * np.pi * x / float(n))
    selected.append(_safe_normalize(s2, n))

    rsmooth = np.exp(rng.normal(0.0, 0.40, n))
    selected.append(_safe_normalize(rsmooth, n))

    fourier_end = min(master_deadline, t0 + 120.0)
    fseqs = _fourier_search(n, fourier_end, rng, K=4)
    for i, seq in enumerate(fseqs):
        if i >= 3:
            break
        selected.append(_safe_normalize(seq, n))

    rand_fourier = _fourier_seq(n, rng.normal(0.0, 0.6, 8), K=4, scale=0.8)
    selected.append(_safe_normalize(rand_fourier, n))

    if len(selected) > 8:
        selected = selected[:8]

    states = []
    global_best_seq = None
    global_best_score = np.inf

    for i, seq in enumerate(selected):
        seq = _safe_normalize(seq, n)
        ex = float(evaluate_sequence(seq.tolist()))
        states.append([seq.copy(), ex])
        if np.isfinite(ex) and ex < global_best_score:
            global_best_score = ex
            global_best_seq = seq.copy()

    if global_best_seq is None:
        global_best_seq = _safe_normalize(np.ones(n, dtype=np.float64), n)
        global_best_score = float(evaluate_sequence(global_best_seq.tolist()))

    print("initial best exact:", global_best_score, "n:", n, flush=True)

    round_time = 45.0
    idx = 0
    num_states = len(states)

    while time.monotonic() < master_deadline - 40.0:
        if num_states == 0:
            break
        i = idx % num_states
        state = states[i][0]
        slice_end = min(master_deadline - 10.0, time.monotonic() + round_time)
        if slice_end <= time.monotonic():
            break

        try:
            best_seq_i, score_i, state_after = _optimize(
                state, slice_end, rng, exact_every=80, label="c%d" % i
            )
            states[i][0] = state_after
            if np.isfinite(score_i):
                states[i][1] = score_i
            if np.isfinite(score_i) and score_i < global_best_score:
                global_best_score = score_i
                global_best_seq = best_seq_i.copy()
                print("new global best:", global_best_score, "from", i, flush=True)
        except Exception:
            pass

        idx += 1
        if idx % num_states == 0:
            round_time = min(600.0, round_time * 1.5)

    if global_best_seq is None:
        global_best_seq = _safe_normalize(np.ones(n, dtype=np.float64), n)

    final_end = master_deadline - 2.0
    if time.monotonic() < final_end:
        try:
            final_best, final_score, final_state = _optimize(
                global_best_seq, final_end, rng, exact_every=80, label="final"
            )
            if np.isfinite(final_score) and final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = final_best.copy()
            if final_state is not None:
                exs = float(evaluate_sequence(final_state.tolist()))
                if np.isfinite(exs) and exs < global_best_score:
                    global_best_score = exs
                    global_best_seq = final_state.copy()
        except Exception:
            pass

    final_seq = _safe_normalize(global_best_seq, n)
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        final_score = np.inf

    if not np.isfinite(final_score):
        final_seq = _safe_normalize(np.ones(n, dtype=np.float64), n)

    return [float(v) for v in final_seq]
```
