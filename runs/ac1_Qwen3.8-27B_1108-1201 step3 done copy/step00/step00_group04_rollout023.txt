```python
import numpy as np
from scipy.fft import fft, ifft
import time

def fast_convolve(p):
    """Compute full linear convolution of p with itself using FFT."""
    n = len(p)
    # Need FFT of size at least 2n-1
    size = 1
    while size < 2*n-1:
        size *= 2
    P = fft(p, size)
    C = ifft(P * P)
    return C[:2*n-1].real

def optimize(n, alpha_init, lr, n_iter, time_budget):
    """Optimize the probability distribution to minimize max of self-convolution."""
    # Initial guess: cosine-weighted
    j = np.arange(n)
    p = 1.0 + alpha_init * np.cos(np.pi * j / (n-1))
    p = np.clip(p, 0.0, None)
    p = p / p.sum()
    
    best_p = p.copy()
    best_max = fast_convolve(p).max()
    
    for it in range(n_iter):
        c = fast_convolve(p)
        max_c = c.max()
        if max_c < best_max:
            best_max = max_c
            best_p = p.copy()
        
        # Find active peaks (within tolerance of max)
        tol = max_c * 0.01 + 1e-10
        active = np.where(c >= max_c - tol)[0]
        
        # Compute subgradient: average gradient over active peaks
        grad = np.zeros(n)
        for k in active:
            # d(c_k)/d(p_i) = 2*p_{k-i} if 0 <= k-i <= n-1
            i_range = np.arange(max(0, k-(n-1)), min(n-1, k)+1)
            j_range = k - i_range
            grad[i_range] += 2 * p[j_range]
        grad /= len(active)
        
        # Project gradient onto constraint (sum p = 1)
        grad = grad - grad.mean()
        
        # Step
        p_new = p - lr * grad
        p_new = np.clip(p_new, 0.0, None)
        p_new = p_new / p_new.sum()
        
        # Check if improved
        c_new = fast_convolve(p_new)
        if c_new.max() < max_c:
            p = p_new
            # Maybe increase learning rate slightly
        else:
            # Decrease learning rate
            lr *= 0.8
            if lr < 1e-10:
                break
    
    return best_p, best_max
```

import numpy as np
from scipy.fft import fft, ifft
import time

def fast_convolve(p):
    """Full linear self-convolution using FFT."""
    n = len(p)
    size = 1
    while size < 2*n - 1:
        size *= 2
    P = fft(p, size)
    C = ifft(P * P, size)
    return C[:2*n-1].real

def compute_gradient(c_k_indices, p, n):
    """Compute subgradient of max at the given peak indices."""
    grad = np.zeros(n)
    for k in c_k_indices:
        # d(c_k)/d(p_i) = 2*p_{k-i} for valid i
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        i_vals = np.arange(j_min, j_max + 1)
        grad[i_vals] += 2.0 * p[k - i_vals]
    return grad

def optimize_distribution(n, init_p=None, max_iter=100000, time_limit=60):
    """Optimize p to minimize max(p*p)."""
    if init_p is None:
        p = np.ones(n) / n
    else:
        p = init_p.copy()
        p = np.clip(p, 0, None)
        p /= p.sum()
    
    start_time = time.time()
    best_p = p.copy()
    best_max = fast_convolve(p).max()
    lr = 0.01
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        c = fast_convolve(p)
        max_c = c.max()
        
        if max_c < best_max:
            best_max = max_c
            best_p = p.copy()
        
        # Find active peaks (within 1% of max)
        threshold = max_c * 0.99
        active = np.where(c >= threshold)[0]
        
        # Subgradient
        grad = compute_gradient(active, p, n)
        grad /= len(active)
        
        # Project: remove mean to maintain sum(p) = 1
        grad -= grad.mean()
        
        # Step
        p_new = p - lr * grad
        p_new = np.clip(p_new, 0, None)
        s = p_new.sum()
        if s < 1e-10:
            break
        p_new /= s
        
        # Check improvement
        c_new = fast_convolve(p_new)
        if c_new.max() < max_c:
            p = p_new
        else:
            lr *= 0.9
            if lr < 1e-12:
                break
    
    return best_p, best_max

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.time() + budget_s - 10
    
    best_score = 2.0  # uniform baseline
    best_seq = None
    
    # Try multiple n values and initializations
    n_values = [2048, 4096, 8192]  # powers of 2 for FFT efficiency
    # But the sequence length n doesn't need to be a power of 2
    # The evaluate_sequence function works with any length
    
    for n in n_values:
        if time.time() > deadline:
            break
        
        # Try different initializations
        inits = []
        # 1. Uniform
        inits.append(np.ones(n) / n)
        # 2. Cosine edge-bias
        j = np.arange(n)
        p_cos = 1.0 + 0.5 * np.cos(np.pi * j / (n - 1))
        p_cos = np.clip(p_cos, 0, None)
        inits.append(p_cos / p_cos.sum())
        # 3. Stronger edge bias
        p_cos2 = 1.0 + 0.9 * np.cos(np.pi * j / (n - 1))
        p_cos2 = np.clip(p_cos2, 0, None)
        inits.append(p_cos2 / p_cos2.sum())
        # 4. Random
        rng = np.random.default_rng(seed + n)
        p_rand = rng.random(n)
        inits.append(p_rand / p_rand.sum())
        
        for init in inits:
            if time.time() > deadline:
                break
            p_opt, max_c = optimize_distribution(n, init, max_iter=50000, time_limit=30)
            score = 2 * n * max_c  # since sum(p) = 1
            if score < best_score:
                best_score = score
                best_seq = (p_opt * n).tolist()  # scale back (or any scale)
                print(f"n={n}, score={score:.6f}, max_c={max_c:.8f}")
    
    if best_seq is None:
        # Fallback: uniform
        return [1.0] * 2048

import time

def fast_self_convolve(p):
    """Compute full linear self-convolution of p using FFT."""
    n = len(p)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    P = fft(p, size)
    C = ifft(P * P, size)
    return np.real(C[:2 * n - 1])

def compute_grad_active(active, p, n):
    """Compute subgradient at active peak positions. O(n * |active|)."""
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * p[k - idx]
    return grad
