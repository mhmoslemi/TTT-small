"""
Mirror descent (multiplicative weights) with adaptive step size, momentum-free
natural-gradient updates, and periodic reheating to minimize
J(a) = 2n*max(convolve(a,a)) / (sum(a))^2.
Uses FFT-based autocorrelation for O(n log n) per-iteration cost.

import numpy as np
import time


def fft_autocorr(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    return np.fft.irfft(F * F, fft_size)[:size]


def compute_M_and_grad(f):
    """Return M=max(f*f) and the subgradient vector (same shape as f)."""
    n = len(f)
    conv = fft_autocorr(f)
    M = conv.max()
    tol = max(1e-12 * abs(M), 1e-15)
    peak_indices = np.where(conv >= M - tol)[0]
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        grad[lo:hi] += 2.0 * f[k - hi + 1 : k - lo + 1][::-1]
    grad /= len(peak_indices)
    return M, grad


def mirror_step(f, grad, alpha):
    """Multiplicative-weights (mirror descent) step on the simplex."""
    log_u = -alpha * grad
    log_u -= log_u.max()
    f_new = f * np.exp(log_u)
    s = f_new.sum()
    if s < 1e-300:

    return f_new / s


def project_simplex(f):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    f = np.maximum(f, 0.0)

    if s < 1e-15:

    # --- Final refinement ---
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"[FINAL] refinement, budget={rem - 3:.0f}s")
        ff, _ = optimize_sequence(best_f, rem - 3, seed=seed + 999, verbose=True)
        sf = float(evaluate_sequence(ff.tolist()))
        print(f"[FINAL] score={sf:.6f}")
        if sf < best_score:
            best_score, best_f = sf, ff.copy()

    # Safety: ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.sum() < 0.01:
        best_f = np.ones(n)
    print(f"[DONE] best_score={best_score:.6f}")
    return [float(x) for x in best_f]
