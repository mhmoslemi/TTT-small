from scipy.optimize import linprog

def compute_convolution(f, fft_size):
    """Compute full convolution using FFT."""
    F = np.fft.rfft(f, fft_size)
    c = np.fft.irfft(F * F, fft_size)
    return c[:2*len(f)-1]

def compute_subgradient(f, c, n, top_k):
    """Compute subgradient targeting top-k peaks."""
    # Find top-k lags
    top_idx = np.argsort(c)[-top_k:]
    
    sg = np.zeros(n)
    for k in top_idx:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        indices = np.arange(i_start, i_end + 1)
        sg[indices] += f[k - indices]
    return sg
