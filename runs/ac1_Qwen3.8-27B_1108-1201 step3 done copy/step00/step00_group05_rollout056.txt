import numpy as np
import time

def compute_gradient(a, tau):
    """Compute gradient of soft-max of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - np.max(c)  # numerical stability
    exp_c = np.exp(c_shifted / tau)
    q = exp_c / np.sum(exp_c)
    grad = 2.0 * np.correlate(q, a, 'valid') - q[0::2] * a
    return grad, c

def project(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s <= S:
        return a + (S - s) / len(a)
    # Water-filling
    lo, hi = 0.0, np.max(a)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(a - mid, 0.0)) > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.maximum(a - t, 0.0)
    # Fix numerical issues
    diff = S - np.sum(result)
    if abs(diff) > 1e-10:
        # Add to largest elements
        idx = np.argmax(result)
        result[idx] += diff
    return result

def run_gradient_descent(a_init, S, n_steps, tau_start, tau_end, lr):
    """Run projected gradient descent."""
    a = a_init.astype(np.float64).copy()
    a = project(a, S)
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad, c = compute_gradient(a, tau)
        a = a - lr * grad
        a = project(a, S)
        
        if step % 50 == 0:

            if score < best_score:

"""
Gradient-based optimization of non-negative sequences to minimize
2*n*max(conv(a,a))/sum(a)^2. Uses projected gradient descent with
soft-max smoothing and multiple restarts.
"""

import numpy as np
import time

def compute_soft_max_grad(a, tau):
    """Compute gradient of soft-max of full autocorrelation of a."""
    c = np.convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / tau)
    q = exp_c / np.sum(exp_c)
    grad = 2.0 * np.correlate(q, a, 'valid') - q[0::2] * a
    return grad

def project_onto_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = len(a)
    s = np.sum(a)
    if s <= S + 1e-12:
        if s < S:
            a += (S - s) / n
        return a
    # Water-filling: find t such that sum(max(a - t, 0)) = S
    lo, hi = 0.0, np.max(a)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(a - mid, 0.0)) > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.maximum(a - t, 0.0)
    # Correct for numerical error
    result[np.argmax(result)] += S - np.sum(result)
    return result

def make_chirp_init(n, seed=0):
    """Create a chirp-based non-negative initialization."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    # Real chirp: cos(pi * t^2 / n) shifted to be non-negative
    a = 1.0 + 0.8 * np.cos(np.pi * t * t / n)
    # Add small random perturbation
    a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    return a

def make_broadband_init(n, seed=0):
    """Create a broadband (flat-spectrum targeting) initialization."""
    rng = np.random.default_rng(seed)
    # Random non-negative sequence
    a = rng.exponential(1.0, size=n)
    return a

def make_uniform_init(n):
    """Uniform initialization."""
    return np.ones(n, dtype=np.float64)

def run_gradient_descent(a_init, n_steps=None, tau_start=None, tau_end=0.05, lr=5.0, time_budget=None, verbose=False):
    """Run projected gradient descent to minimize max autocorrelation."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a) * S
    a = project_onto_simplex(a, S)
    
    if tau_start is None:
        tau_start = 1.0
    
    best = a.copy()
    best_max_c = np.max(np.convolve(best, best))
    best_score = float(evaluate_sequence(best.tolist()))
    
    n_improvements = 0
    start_time = time.monotonic()
    
    for step in range(n_steps if n_steps else 10**9):
        if time_budget is not None and (time.monotonic() - start_time) > time_budget:
            break
        
        # Anneal tau
        if n_steps:
            frac = step / max(1, n_steps - 1)
        else:
            frac = min(1.0, step / 1000.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad = compute_soft_max_grad(a, tau)
        a = a - lr * grad
        a = project_onto_simplex(a, S)
        
        if (step + 1) % 30 == 0 or step == 0:
            c = np.convolve(a, a)
            current_max_c = np.max(c)
            if current_max_c < best_max_c - 1e-8:
                best_max_c = current_max_c
                best = a.copy()
                n_improvements += 1
                if verbose:
                    score = float(evaluate_sequence(best.tolist()))
                    print(f"  Step {step+1}: max_c={best_max_c:.6f}, score={score:.6f}, tau={tau:.4f}")
    
    # Final evaluation
    final_score = float(evaluate_sequence(best.tolist()))
    return best, final_score

def propose_candidate(budget_s=1800):

    total_deadline = time.monotonic() + budget_s - 5.0
    
    best_sequence = None
    best_score = np.inf

    lengths = [3000, 5000, 7000, 4498]

        provided = np.array(height_sequence_1, dtype=np.float64)

            provided = np.maximum(provided, 0.0)
        else:
            provided = None
    except (NameError, Exception):
        provided = None
    
    for n in lengths:
        if time.monotonic() > total_deadline:
            break
        
        time_for_n = max(60.0, (total_deadline - time.monotonic()) * 0.6)
        
        # Multiple initializations for this length
        inits = [
            ("uniform", make_uniform_init(n)),
            ("chirp0", make_chirp_init(n, seed=0)),
            ("chirp1", make_chirp_init(n, seed=1)),
            ("broadband", make_broadband_init(n, seed=42)),
        ]
        
        if provided is not None and len(provided) == n:
            inits.insert(0, ("provided", provided.copy()))
        elif provided is not None:
            # Resample to length n
            from numpy import interp
            x_old = np.linspace(0, 1, len(provided))
            x_new = np.linspace(0, 1, n)
            resampled = interp(x_new, x_old, provided)
            inits.insert(0, ("provided_resampled", np.maximum(resampled, 0.0)))
        
        n_inits = len(inits)
        time_per_init = time_for_n / n_inits
        
        for name, a_init in inits:
            if time.monotonic() > total_deadline:
                break
            if time.monotonic() + time_per_init < time.monotonic() + 30:
                break  # Not enough time
                
            print(f"Trying n={n}, init={name}, time_budget={time_per_init:.0f}s")
            
            a_result, score = run_gradient_descent(
                a_init, 
                tau_start=1.0, 
                tau_end=0.05, 
                lr=5.0,
                time_budget=time_per_init,
                verbose=True
            )
            
            if score < best_score:
                best_score = score
                best_sequence = a_result.copy()
                print(f"  NEW BEST: score={score:.6f} (n={n}, init={name})")
    
    if best_sequence is None:
        # Fallback
        a = make_uniform_init(1000)
        best_sequence = a
        best_score = float(evaluate_sequence(a.tolist()))
    
    print(f"\nFinal best score: {best_score:.6f}")
    return [float(x) for x in best_sequence]

Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses projected gradient descent with soft-max smoothing, adaptive learning
rate, tau annealing, and multiple initializations (chirp, random, provided).
Two-phase strategy: quick exploration then focused polishing.

import numpy as np
import time


def compute_soft_max_grad(a, tau):
    """Compute gradient of tau*log-sum-exp(conv(a,a)) with respect to a."""
    c = np.convolve(a, a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / tau)
    q = exp_c / exp_c.sum()
    grad = 2.0 * np.correlate(q, a, 'valid') - q[0::2] * a
    return grad, c


def project_simplex(a, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0).copy()
    n = a.size
    s = a.sum()
    if s <= S:
        return a + (S - s) / n
    lo, hi = 0.0, a.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.maximum(a - mid, 0.0).sum() > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    res = np.maximum(a - t, 0.0)
    idx = res.argmax()
    res[idx] += S - res.sum()
    return res
