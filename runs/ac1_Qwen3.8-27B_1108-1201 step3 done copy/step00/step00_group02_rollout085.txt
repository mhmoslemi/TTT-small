def _smooth_perturb(a, rng, scale):
   n=a.size; target=sum(a)
   x=np.linspace(0.0,1.0,n) if n>1 else np.array([0.0])
   noise=np.zeros(n)
   for f in range(1, 10):
      amp = float(rng.normal(0.0, 1.0 / float(f)))
      noise = noise + amp * np.sin(2.0*np.pi*float(f)*x + float(rng.uniform(0.0, 2.0*np.pi)))
   noise -= float(np.mean(noise))

```python
import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


"""
Hybrid optimizer for the nonnegative step-function objective

    score(a) = 2 * n * max(convolve(a, a)) / (sum(a))^2.

The code keeps at worst a high-quality nondegenerate fallback, but tries to
beat it using:
  1. many structured, skewed initializations (exponential/beta/sigmoid/sine),
  2. soft-subgradient descent in the sequence domain,
  3. occasional active-constraint linear-program direction searches,
  4. smooth random restarts/perturbations to escape plateaus,
  5. a small portfolio of independent starting points/lengths.
"""

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _target_for_n(n):
    n = max(1, int(n))
    t = float(np.sqrt(2.0 * float(n)))
    if not np.isfinite(t) or t <= 1e-12:
        return 1.0
    # Keeping the norm modest avoids hitting the evaluator's 1000 clip and
    # preserves better numerical behavior in the LP subroutines.
    return min(t, 900.0)


def _normalize_target(a, target=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)

    if not np.all(np.isfinite(a)):
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)

    a = np.clip(a, 1e-12, None)

    if target is None:
        target = _target_for_n(a.size)
        target = float(target)

    if not np.isfinite(target) or target <= 1e-12:
        target = _target_for_n(a.size)
        target = float(target)

    s = float(np.sum(a))
    if s <= 1e-12:
        return np.full(a.size, target / float(a.size), dtype=np.float64)

    return a * (target / s)


def _autocorr_full(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    # For short sequences, the direct convolution is usually faster and avoids
    # FFT roundoff.  The evaluator and our fast scorer agree closely in either case.
    if n < 128:
        out = np.convolve(a, a, mode="full")
        return out.astype(np.float64, copy=False)

    m = 2 * n - 1
    nfft = _next_pow2(m)
    spec = np.fft.rfft(a, n=nfft)
    out = np.fft.irfft(spec * spec, n=nfft)[:m]
    return out.astype(np.float64, copy=False)


def _score_from_conv(a, conv):
    n = int(a.size)
    if n == 0:
        return np.inf
    sa = float(np.sum(a))
    if not np.isfinite(sa) or sa < 1e-12:
        return np.inf
    mb = float(np.max(conv))
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * float(n) * mb / (sa * sa))


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    conv = _autocorr_full(a)
    return _score_from_conv(a, conv)


def _correlation_ind(a, w):
    """
    Returns c[i] = sum_m a[m] * w[i+m], i=0,...,n-1, for w of length L=2n-1.
    This is used to compute weighted sums of convolution gradients.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    w = np.asarray(w, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    if n < 128:
        c = np.convolve(a[::-1], w, mode="full")
        return c[n - 1:2 * n - 1].astype(np.float64, copy=False)

    m = n + w.size - 1
    nfft = _next_pow2(m)
    fa = np.fft.rfft(a[::-1], n=nfft)
    fw = np.fft.rfft(w, n=nfft)
    c = np.fft.irfft(fa * fw, n=nfft)
    return c[n - 1:2 * n - 1].astype(np.float64, copy=False)


def _add_candidate(arrs, a, max_count):
    if len(arrs) >= int(max_count):
        return
    b = _normalize_target(a)
    if b.size > 0 and np.all(np.isfinite(b)):
        arrs.append(b)


def _resample_sequence(seq, n):
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    n = max(1, int(n))
    if seq.size == 0:
        return np.ones(n, dtype=np.float64)
    if seq.size == n:
        return seq.copy()
    old = np.linspace(0.0, 1.0, seq.size, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return np.interp(new, old, seq).astype(np.float64, copy=False)


def _param_candidate_arrays(n, rng, max_count=30):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    arrs = []

    _add_candidate(arrs, np.ones(n, dtype=np.float64), max_count)

    for b in (0.25, 0.5, 0.8, 1.2, 1.6, 2.2, 3.0):
        _add_candidate(arrs, np.exp(-b * x), max_count)
        _add_candidate(arrs, np.exp(-b * (1.0 - x)), max_count)

    for p in (0.2, 0.4, 0.7, 1.0, 1.5, 2.5):
        _add_candidate(arrs, np.power(x + 1e-6, p), max_count)
        _add_candidate(arrs, np.power(1.0 - x + 1e-6, p), max_count)

    for c in (0.15, 0.3, 0.5, 0.7, 0.85):
        for s in (3.0, 6.0, 12.0):
            _add_candidate(arrs, 1.0 / (1.0 + np.exp(-s * (x - c))), max_count)

    for m in (1, 2, 3, 5):
        for amp in (0.3, 0.6, 0.9):
            for ph in (0.0, 0.5 * np.pi, float(rng.uniform(0.0, 2.0 * np.pi))):
                a = 1.0 + amp * np.sin(2.0 * np.pi * float(m) * x + ph)
                a = np.maximum(a, 0.05)
                _add_candidate(arrs, a, max_count)

    for p in (0.2, 0.5, 1.2):
        for q in (0.2, 0.5, 1.2):
            a = np.power(x + 1e-6, p) * np.power(1.0 - x + 1e-6, q)
            _add_candidate(arrs, a, max_count)

    for _ in range(10):
        if len(arrs) >= int(max_count):
            break
        a = np.ones(n, dtype=np.float64)
        for f in range(1, 8):
            amp = float(rng.normal(0.0, 1.0 / float(f)))
            a = a + amp * np.sin(2.0 * np.pi * float(f) * x + float(rng.uniform(0.0, 2.0 * np.pi)))
        a = np.maximum(a, 0.05)
        _add_candidate(arrs, a, max_count)

    return arrs


def _build_starts(parent, rng):
    arrs = []
    p = None

    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64).reshape(-1)
        if parent.size >= 2:
            # Avoid extremely expensive parent operations if the provided parent
            # is huge.  We still benefit from its global shape via resampling.
            if parent.size > 50000:
                parent = _resample_sequence(parent, 24576)
            p = _normalize_target(parent, _target_for_n(parent.size))

    par_len = int(p.size) if p is not None else 0
    lengths = [1024, 2048, 4096, 8192, 16384]
    if 512 <= par_len <= 32768:
        lengths.append(par_len)

    lengths = sorted(set([int(n) for n in lengths if 256 <= int(n) <= 32768]))
    if len(lengths) > 5:
        keep = []
        for n in lengths:
            if n == par_len or n in (2048, 4096, 8192, 16384):
                keep.append(n)
        lengths = keep[:5]

    if p is not None:
        _add_candidate(arrs, p, 600)
        if p.size > 0:
            x = np.linspace(0.0, 1.0, p.size, dtype=np.float64)
            for b in (0.5, 1.0, 2.0, 3.0):
                _add_candidate(arrs, p * np.exp(-b * x), 600)
                _add_candidate(arrs, p * np.exp(-b * (1.0 - x)), 600)

    for n in lengths:
        if len(arrs) >= 600:
            break

        if p is not None and int(p.size) != int(n):
            rp = _resample_sequence(p, n)
            _add_candidate(arrs, rp, 600)
            if n > 0:
                x = np.linspace(0.0, 1.0, n, dtype=np.float64)
                for b in (0.5, 1.0, 2.0):
                    _add_candidate(arrs, rp * np.exp(-b * x), 600)
                    _add_candidate(arrs, rp * np.exp(-b * (1.0 - x)), 600)

        for a in _param_candidate_arrays(n, rng, 30):
            _add_candidate(arrs, a, 600)
            if len(arrs) >= 600:
                break

    return arrs


def _score_arrays(arrs, deadline):
    scored = []
    for idx, a in enumerate(arrs):
        if time.monotonic() > float(deadline):
            break
        a = np.asarray(a, dtype=np.float64).reshape(-1)
        if a.size < 2:
            continue
        try:
            sc = _fast_score(a)
        except Exception:
            sc = np.inf
        if np.isfinite(sc):
            scored.append((float(sc), "cand%d" % int(idx), a))
    return scored


def _select_top_by_score(scored, k):
    pool = list(scored)
    top = []
    k = max(0, int(k))
    for _ in range(min(k, len(pool))):
        if not pool:
            break
        best_i = 0
        best_val = float(pool[0][0])
        for i in range(1, len(pool)):
            v = float(pool[i][0])
            if v < best_val:
                best_val = v
                best_i = i
        top.append(pool.pop(best_i))
    return top


def _best_per_length(scored):
    d = {}
    for sc, name, a in scored:
        n = int(np.asarray(a).size)
        old = d.get(n)
        if old is None or float(sc) < float(old[0]):
            d[n] = (float(sc), name, a)
    return d


def _unique_starts(overall, bylen_top, max_starts):
    starts = []
    used = set()
    max_starts = max(0, int(max_starts))

    for item in overall:
        n = int(np.asarray(item[2]).size)
        if n not in used:
            starts.append(item)
            used.add(n)
        if len(starts) >= max_starts:
            break

    for item in bylen_top:
        n = int(np.asarray(item[2]).size)
        if n not in used:
            starts.append(item)
            used.add(n)
        if len(starts) >= max_starts:
            break

    return starts


def _smooth_perturb(a, rng, scale):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    target = float(np.sum(a))
    if n < 2:
        return _normalize_target(a, target)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    noise = np.zeros(n, dtype=np.float64)

    # Low-frequency smooth component, more effective than white noise here.
    for f in range(1, 10):
        amp = float(rng.normal(0.0, 1.0 / float(f)))
        noise = noise + amp * np.sin(2.0 * np.pi * float(f) * x + float(rng.uniform(0.0, 2.0 * np.pi)))

    noise = noise - float(np.mean(noise))
    amp = float(scale) * (float(np.max(a)) + float(np.mean(a)) + 1e-300)
    trial = a + amp * noise

    # Occasional local block move to escape very flat manifolds.
    if rng.random() < 0.5:
        width = max(2, n // 100)
        hi = max(1, n - width + 1)
        start = int(rng.integers(0, hi))
        mult = float(np.exp(rng.normal(0.0, 0.5)))
        trial[start:start + width] = trial[start:start + width] * mult

    return _normalize_target(trial, target)


def _select_best_candidate(cands):
    best = None
    for cand in cands:
        sc = float(cand[0])
        if np.isfinite(sc):
            if best is None or sc < float(best[0]):
                best = cand
    return best


def _soft_grad_candidates(a, conv, tau_frac):
    """
    Gradient candidates for the smooth approximation

        logsumexp((conv - max(conv)) / tau)

    With tau small this approximates the subgradient of the true max.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    conv = np.asarray(conv, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0 or conv.size == 0:
        return []

    maxc = float(np.max(conv))
    if not np.isfinite(maxc) or maxc <= 0.0:
        return []

    tau = max(float(tau_frac) * maxc, 1e-300)
    z = (conv - maxc) / tau
    w = np.exp(z)
    ws = float(np.sum(w))
    if ws <= 0.0 or not np.isfinite(ws):
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / ws

    # Weighted subgradient: 2 * weighted sum of shifted a.
    grad = 2.0 * _correlation_ind(a, w)
    grad = grad - float(np.mean(grad))

    maxg = float(np.max(np.abs(grad)))
    maxa = float(np.max(a))
    if maxg <= 1e-300 or maxa <= 0.0:
        return []

    target = float(np.sum(a))
    trials = []
    for frac in (0.05, 0.15, 0.5):
        trial = a - (frac * maxa / maxg) * grad
        trial = _normalize_target(trial, target)
        tconv = _autocorr_full(trial)
        tsc = _score_from_conv(trial, tconv)
        trials.append((float(tsc), trial, tconv))
    return trials


def _lp_direction(a, conv, R, max_time):
    """
    Active-constraint LP direction:
        maximize 1^T g
        s.t. (a * g)[k] <= max(conv) for a selected set of active k,
             g >= 0.

    Only a small subset of the most active convolution coefficients is used,
    which makes the LP cheap and still gives a useful improvement direction.
    """
    if linprog is None:
        return None

    a = np.asarray(a, dtype=np.float64).reshape(-1)
    conv = np.asarray(conv, dtype=np.float64).reshape(-1)
    n = int(a.size)
    L = int(conv.size)
    if n < 2 or L < 2:
        return None

    R = max(1, min(int(R), L))
    if L <= R:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(conv, -R)[-R:].astype(np.int64, copy=False)

    center = int(n - 1)
    if 0 <= center < L and not np.any(idx == center):
        idx = np.append(idx, center)

    idx = np.unique(idx.astype(np.int64, copy=False))
    M = float(np.max(conv))
    if not np.isfinite(M) or M <= 0.0:
        return None

    A = np.zeros((idx.size, n), dtype=np.float64)
    for row in range(idx.size):
        k = int(idx[row])
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if hi < lo:
            A[row, 0] = float(a[0])
            continue

        start_idx = k - hi
        stop_idx = k - lo + 1
        A[row, lo:hi + 1] = a[start_idx:stop_idx][::-1]

    row_sums = A.sum(axis=1)
    for row in np.where(row_sums <= 1e-300)[0]:
        A[int(row), 0] = max(float(a[0]), 1e-12)

    c = -np.ones(n, dtype=np.float64)
    b = np.full(idx.size, M, dtype=np.float64)
    upper = 1.0e4
    bounds = [(0.0, upper)] * n
    opts = {"time_limit": max(0.3, float(max_time))}

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=opts)
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs-ds", options=opts)
        except Exception:
            return None

    if res is None or res.x is None:
        return None

    x = np.asarray(res.x, dtype=np.float64).reshape(-1)
    if x.size != n or not np.all(np.isfinite(x)):
        return None

    s = float(np.sum(x))
    if s <= 1e-12:
        return None

    return _normalize_target(x, float(np.sum(a)))


def _mix_candidates(a, g):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    g = np.asarray(g, dtype=np.float64).reshape(-1)
    if a.size != g.size or a.size == 0:
        return []

    target = float(np.sum(a))
    g = _normalize_target(g, target)
    trials = []

    for t in (0.05, 0.1, 0.25, 0.5, 0.8):
        trial = (1.0 - t) * a + t * g
        trial = _normalize_target(trial, target)
        tconv = _autocorr_full(trial)
        tsc = _score_from_conv(trial, tconv)
        trials.append((float(tsc), trial, tconv))

    # Also try the LP direction alone; it can be far from a and occasionally useful.
    tconv = _autocorr_full(g)
    tsc = _score_from_conv(g, tconv)
    trials.append((float(tsc), g.copy(), tconv))

    return trials


def _local_search(a, end_time, rng, name, verbose=True):
    a = _normalize_target(a)
    conv = _autocorr_full(a)
    score = _score_from_conv(a, conv)

    if not np.isfinite(score):
        a = _normalize_target(np.ones(1024, dtype=np.float64))
        conv = _autocorr_full(a)
        score = _score_from_conv(a, conv)

    best_a = a.copy()
    best_score = float(score)

    if verbose:
        print("[%s] initial fast score %.6f, length %d" % (name, best_score, int(a.size)))

    tau_frac = 0.01
    no_improve = 0
    iter = 0
    improved = False
    lp_enabled = (linprog is not None)
    lp_fails = 0
    last_lp = 0.0
    last_print = time.monotonic()
    tol = 1e-12

    while time.monotonic() < float(end_time):
        iter += 1
        improved = False

        grad_cands = _soft_grad_candidates(a, conv, tau_frac)
        best_cand = _select_best_candidate(grad_cands)

        if best_cand is not None and float(best_cand[0]) < float(score) - tol:
            score = float(best_cand[0])
            a = best_cand[1]
            conv = best_cand[2]
            if score < best_score:
                best_score = score
                best_a = a.copy()
            no_improve = 0
            tau_frac = max(0.002, tau_frac * 0.85)
            improved = True

        if (not improved) and lp_enabled and ((iter % 15 == 0) or (no_improve > 12)) and (time.monotonic() - last_lp) > 1.5:
            remaining = float(end_time) - time.monotonic()
            if remaining > 5.0:
                last_lp = time.monotonic()
                R = 64 if no_improve > 30 else 32
                mt = min(5.0, max(1.0, remaining / 4.0))
                try:
                    g = _lp_direction(a, conv, R, mt)
                except Exception:
                    g = None

                if g is None:
                    lp_fails += 1
                    if lp_fails > 25:
                        lp_enabled = False
                else:
                    mix_cands = _mix_candidates(a, g)
                    best_mix = _select_best_candidate(mix_cands)
                    if best_mix is not None and float(best_mix[0]) < float(score) - tol:
                        score = float(best_mix[0])
                        a = best_mix[1]
                        conv = best_mix[2]
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        no_improve = 0
                        tau_frac = max(0.004, tau_frac * 0.9)
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve % 3 == 0:
                tau_frac = min(0.30, tau_frac * 1.2)

            # Controlled non-improving move if stuck for a long time.
            if no_improve > 70 and grad_cands:
                best_all = _select_best_candidate(grad_cands)
                if best_all is not None and float(best_all[0]) < float(score) * 1.02:
                    score = float(best_all[0])
                    a = best_all[1]
                    conv = best_all[2]
                    no_improve = 20
                    improved = True

            if (not improved) and (no_improve >= 8 or no_improve % 30 == 0):
                for scale in (0.02, 0.05, 0.10, 0.20, 0.40):
                    trial = _smooth_perturb(a, rng, scale)
                    tconv = _autocorr_full(trial)
                    tsc = _score_from_conv(trial, tconv)
                    if np.isfinite(tsc) and tsc < score - tol:
                        score = float(tsc)
                        a = trial
                        conv = tconv
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break

            if (not improved) and no_improve > 150:
                for scale in (0.5, 1.0):
                    trial = _smooth_perturb(best_a, rng, scale)
                    tconv = _autocorr_full(trial)
                    tsc = _score_from_conv(trial, tconv)
                    if np.isfinite(tsc) and tsc < score - tol:
                        score = float(tsc)
                        a = trial
                        conv = tconv
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        no_improve = 40
                        improved = True
                        break
                if not improved:
                    no_improve = 40

        if verbose:
            tnow = time.monotonic()
            if tnow - last_print > 25.0 or iter == 1:
                print("[%s] iter %d, current %.6f, best %.6f, tau %.4f"
                      % (name, iter, score, best_score, tau_frac))
                last_print = tnow

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
        t0 = time.monotonic()
        budget = max(120.0, float(budget_s))
        hard_deadline = t0 + max(10.0, budget - 5.0)

        parent = None
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
            if parent.size < 2:
                parent = None
        except Exception:
            parent = None

        starts = _build_starts(parent, rng)
        if not starts:
            starts = [np.ones(2048, dtype=np.float64)]

        # Leave enough room for optimization and final exact evaluation.
        init_deadline = min(hard_deadline - 120.0, t0 + min(180.0, 0.15 * budget))
        scored = _score_arrays(starts, float(init_deadline))

        if not scored:
            fb = _normalize_target(np.ones(2048, dtype=np.float64))
            fb_sc = _fast_score(fb)
            if not np.isfinite(fb_sc):
                fb_sc = float("inf")
            scored = [(float(fb_sc), "fallback", fb)]

        top_overall = _select_top_by_score(scored, 3)
        bylen = _best_per_length(scored)
        bylen_top = _select_top_by_score(list(bylen.values()), 4)
        chosen = _unique_starts(top_overall, bylen_top, 4)
        if not chosen:
            chosen = [scored[0]]

        print("Initial candidates considered: %d, finite: %d, starting: %d"
              % (len(starts), len(scored), len(chosen)))
        for it in chosen[:4]:
            print("  start fast score %.6f length %d" % (float(it[0]), int(np.asarray(it[2]).size)))

        final_candidates = []
        reserve = max(45.0, 0.05 * budget)
        opt_deadline = hard_deadline - reserve
        now = time.monotonic()
        if opt_deadline <= now + 30.0:
            opt_deadline = hard_deadline - 15.0

        total_opt = max(30.0, opt_deadline - now)
        raw_w = [0.55, 0.25, 0.15, 0.05][:len(chosen)]
        wsum = sum(raw_w)
        if wsum <= 0.0:
            raw_w = [1.0 / max(1, len(chosen))] * len(chosen)
            wsum = 1.0
        weights = [w / wsum for w in raw_w]

        opt_start = now
        for i, item in enumerate(chosen):
            t_start = max(time.monotonic(), opt_start + total_opt * sum(weights[:i]))
            t_end = min(opt_start + total_opt * sum(weights[:i + 1]), opt_deadline)

            # If current time is already beyond this slice, start immediately
            # but cap at slice_end; if too little remains, skip.
            t_end = max(t_end, min(time.monotonic() + 20.0, opt_deadline))
            if t_end - time.monotonic() < 15.0:
                break

            name = "opt%d_%s" % (i, str(item[1]))
            try:
                opt_a, opt_sc = _local_search(item[2], t_end, rng, name, verbose=True)
            except Exception:
                opt_a = item[2]
                opt_sc = item[0]

            if opt_a is None or opt_a.size < 2:
                continue
            final_candidates.append((float(opt_sc), name, opt_a))

        if not final_candidates:
            best = top_overall[0]
            final_candidates.append((float(best[0]), "fallback_final", best[2]))

        final_candidates = _select_top_by_score(final_candidates, 6)
        best_seq = final_candidates[0][2]
        best_fast = float(final_candidates[0][0])

        # Validate the best few sequences with the authoritative evaluator.
        exact_deadline = hard_deadline - 1.0
        best_exact = float("inf")
        best_exact_seq = best_seq
        exact_count = 0

        for sc, name, a in final_candidates:
            if time.monotonic() > exact_deadline:
                break
            if exact_count >= 8:
                break
            exact_count += 1
            try:
                a = a.astype(np.float64, copy=False).reshape(-1)
                # The evaluator expects nonnegative numbers; enforce this here too.
                a = np.clip(a, 0.0, 1000.0)
                if float(np.sum(a)) < 0.01:
                    continue
                s = float(evaluate_sequence([float(v) for v in a]))
                if np.isfinite(s) and s < best_exact:
                    best_exact = s
                    best_exact_seq = a
                print("Exact validate %s: %.6f" % (name, s))
            except Exception:
                pass

        if not np.isfinite(best_exact):
            # Fall back to the best validated by our fast proxy.
            best_seq = best_fast[2] if False else best_seq
            best_fast = float(final_candidates[0][0])

        if np.isfinite(best_exact):
            print("Final exact score: %.6f" % best_exact)
            best_seq = best_exact_seq
        else:
            print("Final fast score only: %.6f" % best_fast)

        best_seq = np.asarray(best_seq, dtype=np.float64).reshape(-1)
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        if best_seq.size < 1:
            best_seq = np.ones(2048, dtype=np.float64)

        return [float(x) for x in best_seq]

    except Exception:
        # Minimal valid fallback in case anything unexpected happens.
        try:
            fb = _normalize_target(np.ones(2048, dtype=np.float64))
            return [float(x) for x in fb]
        except Exception:
            return [1.0] * 2048
```
