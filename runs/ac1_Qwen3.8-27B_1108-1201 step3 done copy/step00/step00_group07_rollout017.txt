**Algorithm: Subgradient descent with projection**

Objective: minimize `max_j c[j]` subject to `a >= 0`, `sum a = S` (fixed).

Equivalently (by homogeneity): minimize `max_j c[j] / (sum a)^2`.

Since the problem is scale-invariant, let's fix `sum a = 1` and minimize `max_j c[j]`.

Subgradient method:
1. a ← initial (normalized to sum=1)
2. Repeat:
   a. Compute c = FFT-based convolution of a with a.
   b. Find M = max(c), and the set J = {j : c[j] >= M - δ} for some small δ.
   s ← (1/|J|) * sum_{j in J} grad_j
   where grad_j[k] = a[j-k] if 0 <= j-k < n, else 0.
   Equivalently, s = (1/|J|) * (a ⊛ 1_J) where ⊛ is cross-correlation and 1_J is the indicator of J.
   c. a ← a - η * s (with η > 0 step size)
   d. Project a to {x >= 0, sum x = 1} (projection onto the simplex).
3. Return a.

The projection onto the simplex {x >= 0, sum x = 1} can be done as:
- Sort x in decreasing order: x_(1) >= x_(2) >= ... >= x_(n)
- Find the largest k such that x_(k) > (1 - sum_{i<=k} x_(i)) / k
- Set the threshold: τ = (1 - sum_{i<=k} x_(i)) / k
- Projected value: max(0, x_i - τ)

This is O(n log n) per iteration due to the sort. To speed this up, we can use `np.partition` or just accept the O(n log n) cost.

Actually, for n=6733, `np.sort` takes about 50 microseconds. So the projection adds about 50μs per iteration. That's fine.

        for j in peak_indices:
            for i in range(n):
                k = j - i
                if 0 <= k < n:
                    s[i] += a[k]
        s /= npeaks
    else:
        # FFT-based: s[k] = (1/|J|) * sum_j p[j] * a[j-k]
        # This is cross-correlation of p with a
        need = 2 * n - 1
        fft_size = 1
        while fft_size < need:
            fft_size <<= 1
        p = np.zeros(fft_size)
        p[peak_indices] = 1.0
        P = np.fft.rfft(p)
        A = np.fft.rfft(a, fft_size)
        # cross-correlation: ifft(conj(P) * A) gives sum_j p[j]*a[j-k]
        s_full = np.fft.irfft(np.conj(P) * A, fft_size).real[:n]
        s = s_full / npeaks
    return s


def _evaluate_internal(a, A, fft_size):
    """Compute score using precomputed FFT. Returns max_conv, score."""
    n = len(a)
    c = np.fft.irfft(A * A, fft_size)[:2 * n - 1].real
    max_conv = c.max()
    # sum(a) = 1 by construction
    score = 2.0 * n * max_conv
    return max_conv, score, c


def _try_two_level(n, rng):
    """Try a two-level sequence [h]*k + [l]*(n-k) optimized over parameters."""
    best_a = None
    best_score = np.inf
    for k in [n // 4, n // 3, n // 2, 2 * n // 3, 3 * n // 4]:
        for ratio in [0.1, 0.2, 0.3, 0.5, 0.7, 1.5, 2.0, 3.0, 5.0]:
            h = ratio
            l = 1.0
            a = np.zeros(n)
            a[:k] = h
            a[k:] = l
            s = a.sum()
            if s < 1e-10:
                continue
            a /= s
            c, A, fs = _fft_convolve(a)
            score = 2.0 * n * c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


def _try_geometric(n, rng):
    """Try geometric sequences a[i] = r^i normalized."""
    best_a = None
    best_score = np.inf
    for r in np.linspace(0.3, 0.95, 30):
        a = r ** np.arange(n, dtype=np.float64)
        s = a.sum()
        a /= s
        c, A, fs = _fft_convolve(a)
        score = 2.0 * n * c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # Also try reversed
        a2 = a[::-1].copy()
        c2, _, _ = _fft_convolve(a2)
        score2 = 2.0 * n * c2.max()
        if score2 < best_score:
            best_score = score2
            best_a = a2.copy()
    return best_a, best_score


def _subgradient_phase(a, n, fft_size, deadline, rng, label=""):
    """Run subgradient descent with momentum and simplex projection."""
    best_a = a.copy()
    best_score = np.inf
    c, A, _ = _fft_convolve(a)
    current_score = 2.0 * n * c.max()
    best_score = current_score
    best_a = a.copy()

    # Momentum state
    velocity = np.zeros(n)
    momentum = 0.85
    lr = 2.0 * n  # initial learning rate (compensates for small gradients)
    lr_decay = 0.9999
    min_lr = 0.001 * n

    t = 0
    no_improve = 0
    patience = 50000

    while time.monotonic() < deadline:
        t += 1

        # Compute convolution
        c = np.fft.irfft(A * A, fft_size)[:2 * n - 1].real
        max_c = c.max()
        current_score = 2.0 * n * max_c

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Check patience - if stuck, perturb
        if no_improve > patience:
            # Random perturbation to escape local minimum
            a = a + rng.normal(0, 0.01 / n, n)
            a = _project_simplex(a)
            A = np.fft.rfft(a, fft_size)
            velocity = np.zeros(n)
            lr = 2.0 * n
            no_improve = 0
            patience = int(patience * 0.8)
            continue

        # Find peak region
        threshold = max_c * 0.995
        peak_mask = c >= threshold
        peak_indices = np.where(peak_mask)[0]

        # Compute subgradient
        s = _compute_subgradient(c, a, peak_indices, n)

        # Update with momentum
        velocity = momentum * velocity - lr * s
        a = a + velocity
        a = _project_simplex(a)
        A = np.fft.rfft(a, fft_size)

        # Decrease learning rate
        lr = max(min_lr, lr * lr_decay)

        # Periodic output
        if t % 100000 == 1:
            elapsed = time.monotonic() - (deadline - float(budget_s_global))
            print(f"  [{label}] iter {t}: score={current_score:.6f}, best={best_score:.6f}")

    return best_a, best_score


def _coordinate_polish(a, n, fft_size, deadline, rng):
    """Coordinate descent polish: try adjusting individual elements."""
    best_a = a.copy()
    best_score = np.inf
    c, A, _ = _fft_convolve(a)
    best_score = 2.0 * n * c.max()
    best_a = a.copy()

    t = 0
    while time.monotonic() < deadline:
        t += 1
        # Pick a random coordinate
        idx = rng.integers(0, n)

        # Try several perturbation sizes
        for scale in [0.001, 0.01, 0.05, 0.1]:
            for direction in [-1, 1]:
                a_trial = a.copy()
                a_trial[idx] = max(0, a_trial[idx] + direction * scale * a.mean())
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                c_trial = np.fft.irfft(np.fft.rfft(a_trial, fft_size) ** 2, fft_size)[:2 * n - 1].real
                score = 2.0 * n * c_trial.max()
                if score < best_score:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    A = np.fft.rfft(a, fft_size)
                    break

        if t % 10000 == 0:
            print(f"  [polish] iter {t}: best={best_score:.6f}")

    return best_a, best_score


# Global variable to track total budget for timing
budget_s_global = 1800.0


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2*n*max(conv(a,a))/(sum a)^2 using multi-phase optimization:
    1. Exploration: try various structured initializations.
    2. Refinement: FFT-based subgradient descent with momentum.
    3. Polish: coordinate descent for final improvements.
    """
    global budget_s_global
    budget_s_global = float(budget_s)
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --- Phase 0: Get initial sequence ---
    n_target = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size >= 10:
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.sum() > 0.01:
                n_target = len(parent)
    except Exception:
        pass

    if n_target is None:
        n_target = 2000

    n = n_target
    print(f"Sequence length: {n}")

    # Compute FFT size
    need = 2 * n - 1
    fft_size = 1
    while fft_size < need:
        fft_size <<= 1
    print(f"FFT size: {fft_size}")

    # --- Phase 1: Exploration (~20% of budget) ---
    phase1_deadline = start_time + 0.2 * float(budget_s)
    candidates = []

    # Try the parent sequence
    try:
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if a0.sum() > 0.01:
            a0 = a0 / a0.sum()
            c0, _, _ = _fft_convolve(a0)
            score0 = 2.0 * n * c0.max()
            candidates.append((a0.copy(), score0))
    except Exception:
        pass

    # Try uniform
    a_unif = np.ones(n) / n
    c_unif, _, _ = _fft_convolve(a_unif)
    candidates.append((a_unif.copy(), 2.0 * n * c_unif.max()))

    # Try two-level sequences
    a_2l, sc_2l = _try_two_level(n, rng)
    if a_2l is not None:
        candidates.append((a_2l.copy(), sc_2l))

    # Try geometric sequences
    a_geo, sc_geo = _try_geometric(n, rng)
    if a_geo is not None:
        candidates.append((a_geo.copy(), sc_geo))

    # Try random sequences
    for _ in range(5):
        a_rand = rng.random(n)
        a_rand /= a_rand.sum()
        c_r, _, _ = _fft_convolve(a_rand)
        candidates.append((a_rand.copy(), 2.0 * n * c_r.max()))

    # Sort by score and pick the best
    candidates.sort(key=lambda x: x[1])
    a = candidates[0][0].copy()
    best_score = candidates[0][1]
    print(f"Phase 1 best: score={best_score:.6f}")
    for i, (a_c, s_c) in enumerate(candidates[:5]):
        print(f"  Candidate {i}: score={s_c:.6f}")

    # --- Phase 2: Subgradient optimization (~60% of budget) ---
    phase2_deadline = start_time + 0.8 * float(budget_s)
    print("Phase 2: Subgradient descent...")
    a, score = _subgradient_phase(a, n, fft_size, phase2_deadline, rng, label="grad")
    if score < best_score:
        best_score = score
        best_a = a.copy()
    else:
        # Keep the earlier best
        pass

    # Also try from the second best candidate
    if len(candidates) > 1:
        a2 = candidates[1][0].copy()
        a2, score2 = _subgradient_phase(a2, n, fft_size, phase2_deadline, rng, label="grad2")
        if score2 < best_score:
            best_score = score2
            best_a = a2.copy()

    # --- Phase 3: Polish (~10% of budget) ---
    polish_deadline = deadline
    print("Phase 3: Coordinate polish...")
    best_a, polish_score = _coordinate_polish(best_a, n, fft_size, polish_deadline, rng)
    if polish_score < best_score:
        best_score = polish_score
        best_a = polish_a if 'polish_a' in dir() else best_a

    # --- Verify with evaluate_sequence ---
    print("Verifying with evaluate_sequence...")
    final_list = [float(x) for x in best_a]
    final_eval = float(evaluate_sequence(final_list))
    print(f"Internal best score: {best_score:.6f}")
    print(f"evaluate_sequence score: {final_eval:.6f}")

    # If the internal estimate was optimistic, try the candidates too
    best_return = final_list
    best_return_score = final_eval

    for a_c, s_c in candidates[:3]:
        cl = [float(x) for x in a_c]
        ev = float(evaluate_sequence(cl))
        if np.isfinite(ev) and ev < best_return_score:
            best_return_score = ev
            best_return = cl
            print(f"  Found better candidate: {ev:.6f}")

    print(f"FINAL: {best_return_score:.6f}")
    return best_return
