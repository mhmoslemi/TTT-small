import numpy as np
import time

def fast_convolve(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fc = np.fft.irfft(fa * fa, N)
    return fc[:2*n-1]

def project_simplex(v, S):
    """Project v onto {x: sum(x) = S, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cvsum = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cvsum - S) / rho > 0
    rho_star = np.max(np.where(cond)[0]) + 1
    theta = (cvsum[rho_star-1] - S) / rho_star
    return np.maximum(v - theta, 0)

def evaluate_conv(a):
    """Return (max_conv, m_star) for the self-convolution of a."""
    conv = fast_convolve(a)
    m_star = np.argmax(conv)
    return conv[m_star], m_star

def gradient_at_peak(a, m_star):

    n = len(a)
    grad = np.zeros(n)

    hi = min(n - 1, m_star)
    grad[lo:hi+1] = 2.0 * a[m_star - hi : m_star - lo : -1]
    return grad

"""
Minimize 2n*max(convolve(a,a))/(sum a)^2 via projected gradient descent on the
simplex using analytic subgradients of the peak convolution coefficient,
multiple initializations, FFT-based convolution, and adaptive line search.
Exploits scale invariance to work on the probability simplex.

import time


def _fft_selfconv(a):
    """Full linear self-convolution a*a via rFFT (length 2n-1)."""
    n = a.shape[0]
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    return out[:total]


def _proj_simplex(v, S):
    """Project v onto {x: sum(x)=S, x>=0}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cs - S) / rho > 0.0
    if not cond.any():
        return np.full(n, S / n, dtype=np.float64)
    r = int(np.argmax(~cond)) + 1
    theta = (cs[r - 1] - S) / r
    return np.maximum(v - theta, 0.0)


def _subgrad(a, m, n):
    """Subgradient vector of conv(a,a)[m] w.r.t. a (length n)."""
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    if lo <= hi:
        idx = m - np.arange(lo, hi + 1)
        g[lo:hi + 1] = 2.0 * a[idx]
    return g


def _optimize(a0, t_limit, verbose=True):
    """Projected-gradient descent minimizing max(self-conv) on the simplex."""
    n = a0.shape[0]
    a = np.maximum(a0.astype(np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(n) / n
    else:
        a /= s

    conv = _fft_selfconv(a)
    best_max = float(conv.max())
    best_a = a.copy()
    last_improve = time.monotonic()
    lr = 0.3
    stuck = 0

    while time.monotonic() - last_improve < t_limit and time.monotonic() < last_improve + t_limit:
        m = int(np.argmax(conv))
        cur_max = float(conv[m])
        g = _subgrad(a, m, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Backtracking line search
        alpha = lr
        found = False
        for _ in range(12):
            y = a - alpha * g
            y = _proj_simplex(y, 1.0)
            nc = _fft_selfconv(y)
            nm = float(nc.max())
            if nm < cur_max - 1e-16:
                a = y
                conv = nc
                found = True
                break
            alpha *= 0.5

        if found:
            new_max = float(conv.max())
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_a = a.copy()
                last_improve = time.monotonic()
                stuck = 0
            else:
                stuck += 1
            lr = min(2.0 * alpha, 3.0)
        else:
            stuck += 1
            if stuck > 150:
                # Perturb around best
                a = best_a + 0.01 * np.random.standard_normal(n)
                a = np.maximum(a, 0.0)
                s2 = a.sum()
                if s2 > 1e-14:
                    a /= s2
                else:
                    a = best_a.copy()
                conv = _fft_selfconv(a)
                stuck = 0
                lr = 0.3
            else:
                lr *= 0.7

    score = 2.0 * n * best_max
    return best_a, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Uses projected gradient descent with analytic subgradients, multiple
    structured and random starts, FFT convolution, and adaptive step sizes.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 5.0

    # Determine length from provided sequence or default
    n = 4806
    starts = []
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if hs.size > 10 and np.all(np.isfinite(hs)):
            n = hs.size
            starts.append(np.maximum(hs, 0.0))
    except Exception:
        pass

    i = np.arange(n, dtype=np.float64)

    # Structured starts
    starts.append(np.arange(1, n + 1, dtype=np.float64))
    starts.append(np.arange(n, 0, -1, dtype=np.float64))
    starts.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i / n)))
    starts.append(np.exp(-((i - 0.3 * n) / (0.15 * n)) ** 2) + 1e-3)
    starts.append(np.exp(-((i - 0.7 * n) / (0.15 * n)) ** 2) + 1e-3)
    starts.append(np.abs(np.sin(np.pi * i / n)) ** 0.5 + 1e-3)
    starts.append(rng.random(n) + 0.01)
    starts.append(np.abs(rng.standard_normal(n)) + 0.01)

    best_seq = None
    best_score = np.inf

    n_starts = len(starts)
    for k, s in enumerate(starts):
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break
        tl = rem / (n_starts - k)
        tl = max(tl, 5.0)
        tl = min(tl, rem - 2.0)

        print(f"[start {k+1}/{n_starts}] len={n} limit={tl:.0f}s t={time.monotonic()-t0:.0f}s")
        seq, score = _optimize(s, tl)
        exact = evaluate_sequence(seq.tolist())
        if not np.isfinite(exact):
            exact = score
        if exact < best_score:
            best_score = exact
            best_seq = seq
            print(f"  IMPROVED: {exact:.6f}")
        else:
            print(f"  kept best: {best_score:.6f} (this={exact:.6f})")

    # Final refinement
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30.0:
            print(f"[refine] score={best_score:.6f} rem={rem:.0f}s")
            seq, score = _optimize(best_seq, rem - 3.0)
            exact = evaluate_sequence(seq.tolist())
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = seq
                print(f"[refine done] {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[FINAL] score={best_score:.6f} time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in best_seq]
