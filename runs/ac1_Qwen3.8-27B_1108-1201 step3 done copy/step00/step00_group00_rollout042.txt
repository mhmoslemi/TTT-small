import numpy as np
import time

def _next_power_of_2(x):
    """Return the next power of 2 >= x."""
    return int(2 ** np.ceil(np.log2(max(1, x))))

def _convolve_full(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    N = _next_power_of_2(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    return np.fft.irfft(A * A, n=N)[:2*n-1]

def _objective_value(a):
    """Compute the objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = _convolve_full(a)
    return 2.0 * n * np.max(conv) / (S * S)

def _simplex_project(v, target=1.0):
    """Project v onto the probability simplex."""
    n = len(v)
    # Bisection for the Lagrange multiplier
    v = np.asarray(v, dtype=np.float64)
    lo = float(np.min(v)) - target
    hi = float(np.max(v)) + target
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        x = np.maximum(v - mid, 0.0)
        s = float(np.sum(x))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.maximum(v - 0.5 * (lo + hi), 0.0)
    # Fix numerical drift
    diff = target - float(np.sum(x))
    if diff > 0:
        nonzero = x > 0
        x[nonzero] += diff / np.count_nonzero(nonzero)
    elif diff < 0:
        scale = target / max(float(np.sum(x)), 1e-30)
        x = x * scale
    return x

def _soft_max_gradient(a, tau):
    """
    Gradient of (1/tau)*log(sum_k exp(tau*c_k)) where c = conv(a,a).
    Assumes a has sum = 1 (normalized).
    Returns the gradient (before centering for the sum constraint).
    """
    n = len(a)
    conv = _convolve_full(a)
    
    # Softmax weights (numerically stable)
    c_shifted = conv - float(np.max(conv))
    weights = np.exp(tau * c_shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        # Fallback: only the max lag
        k_max = int(np.argmax(conv))
        grad = np.zeros(n)
        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        a_rev = a[::-1]
        idx_start = n - 1 - k_max + j_start
        idx_end = n - 1 - k_max + j_end
        grad[j_start:j_end+1] = 2.0 * a_rev[idx_start:idx_end+1]
        return grad
    weights = weights / w_sum
    
    # Find significant lags
    threshold = 1e-8
    sig_lags = np.where(weights > threshold)[0]
    if len(sig_lags) == 0:
        sig_lags = np.array([int(np.argmax(weights))])

        grad = grad - np.mean(grad)

        obj = _objective_value(a_new)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"  After parent opt: {best_obj:.6f}")

    # Phase 2: Multi-start with different initializations and lengths
    remaining = deadline - time.monotonic()
    if remaining > 60:
        configs = [
            (4806, 'bimodal', 25.0, 0.07),
            (4806, 'sinusoidal', 15.0, 0.1),
            (3600, 'chirp', 20.0, 0.08),
            (4806, 'hann', 30.0, 0.06),
            (4096, 'random', 10.0, 0.12),
            (5000, 'gaussian', 20.0, 0.08),
            (4806, 'exponential', 15.0, 0.1),
            (4806, 'uniform', 5.0, 0.15),
        ]
        time_per_trial = max(60.0, (remaining - 300.0) / max(1, len(configs)))
        for ci, (n_len, method, t_start, l_r) in enumerate(configs):
            if time.monotonic() >= deadline - 200:
                break
            a_init = _init_seq(n_len, method, rng)
            # Add small noise to escape saddle points
            a_init = a_init + rng.normal(0, 0.002, n_len)
            a_init = _proj_simplex(a_init, 1.0)
            init_obj = _obj(a_init)
            print(f"Trial {ci}: n={n_len}, method={method}, init_obj={init_obj:.6f}")
            t_deadline = time.monotonic() + time_per_trial
            t_deadline = min(t_deadline, deadline - 200.0)
            a_opt, obj_opt = _run_optimize(a_init, t_deadline,
                                           tau_start=t_start, tau_end=300.0,
                                           lr_start=l_r, rng=rng)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                print(f"  New best: {best_obj:.6f}")

    # Phase 3: Fine-tuning
    remaining = deadline - time.monotonic()
    if best_a is not None and remaining > 30:
        print(f"Phase 3 fine-tuning: current best={best_obj:.6f}, remaining={remaining:.0f}s")
        t3_deadline = min(deadline, time.monotonic() + remaining - 5.0)
        a_opt, obj_opt = _run_optimize(best_a, t3_deadline,
                                       tau_start=100.0, tau_end=500.0,
                                       lr_start=0.01, rng=rng)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    # Final output
    if best_a is not None:
        # Ensure valid: non-negative, sum >= 0.01, values <= 1000
        final = np.asarray(best_a, dtype=np.float64)
        final = np.clip(final, 0.0, 1000.0)
        s = float(np.sum(final))
        if s < 0.01:
            final = final * 1.0
        final_obj = _obj(final)
        print(f"Final: obj={final_obj:.6f}, len={len(final)}")
        return [float(x) for x in final]

    # Fallback
    a = np.ones(4806) / 4806
    print(f"Fallback: obj={_obj(a):.6f}")
    return [float(x) for x in a]
