    c = np.convolve(f, f)
    peak_idx = int(np.argmax(c))

    grad = np.zeros(n)
    lo = max(0, peak_idx - n + 1)
    hi = min(n, peak_idx + 1)
    for j in range(lo, hi):
        grad[j] = f[peak_idx - j]
    
    # Desired perturbation: reduce where grad is large, increase where grad is small
    # Normalize grad
    if grad.max() > 0:
