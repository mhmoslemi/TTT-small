import numpy as np
import time

def _convolve(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def _score_local(a):
    """Compute the score efficiently (same as evaluate_sequence but without overhead)."""
    n = len(a)
    if n == 0 or a.sum() < 0.01:
        return np.inf
    c = _convolve(a)
    max_b = c.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)

def _subgrad_step(a, k_star, eta):
    """One subgradient step assuming k_star is the peak lag."""
    n = len(a)
    # Compute a_reversed at position k_star
    a_rev = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n, k_star + 1)
    for j in range(j_start, j_end):
        a_rev[j] = a[k_star - j]
    a_new = a - eta * a_rev
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def _dig(a, k_star, eta):
    """Digging: reduce a at positions contributing to peak at k_star."""
    n = len(a)
    a_new = a.copy()
    j_start = max(0, k_star - n + 1)
    j_end = min(n, k_star + 1)
    for j in range(j_start, j_end):
        # contrib of position j to c[k_star] is a[j] * a[k_star - j]
        a_new[j] -= eta * a[j] * a[k_star - j] / (a.max() * a.max() + 1e-15)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def _fft_convolve(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def _local_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def _peak_lag(a):
    c = _fft_convolve(a)
    return int(np.argmax(c)), c

def _dig_update(a, k_star, eta):
    n = len(a)
    a_new = a.copy()
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n, k_star + 1) if k_star + 1 <= n else n
    if j_hi > j_lo:
        idx = np.arange(j_lo, j_hi)
        partner = k_star - idx
        # contrib[j] = a[j] * a[k_star - j]
        contrib = a[idx] * a[partner]
        # Normalize contrib so max is 1
        max_c = contrib.max()
        if max_c > 1e-30:

    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

def _subgrad_update(a, k_star, eta, smooth_buf=None, alpha_smooth=0.1):
    n = len(a)
    a_rev = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n, k_star + 1)
    if j_hi > j_lo:
        idx = np.arange(j_lo, j_hi)
        a_rev[j_lo:j_hi] = a[k_star - idx]

        direction = smooth_buf

def _lp_refine(a, n_constraints=300, t=0.5, max_time=15.0):
    """LP-based refinement step."""
    from scipy.optimize import linprog
    from scipy import sparse as sp
    
    n = len(a)
    c = _fft_convolve(a)
    S = a.sum()
    a_norm = a / S
    cn = _fft_convolve(a_norm)
    M = cn.max()
    
    # Select top K constraints
    K = min(n_constraints, 2 * n - 1)
    top_k = np.argsort(cn)[-K:]
    
    # Build constraint matrix (K x n)
    # Row k: A[k, i] = a_norm[k-i] if 0 <= k-i <= n-1, else 0
    A = np.zeros((K, n))
    for row_idx, k in enumerate(top_k):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        if i_hi > i_lo:
            cols = np.arange(i_lo, i_hi)
            A[row_idx, cols] = a_norm[k - cols]
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # In standard form: min -sum(g) s.t. A @ g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(K, M), bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    
    g_norm = g / g_sum
    a_new = (1 - t) * a_norm + t * g_norm
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new
