def subgradient_step(a, lr):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    K = np.where(b == max_b)[0]  # indices where max is achieved
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in K:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]
    grad /= len(K)
    
    # Step
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    return a_new
