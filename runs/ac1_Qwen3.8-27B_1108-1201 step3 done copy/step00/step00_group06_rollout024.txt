Minimizes the peak self-convolution of a non-negative sequence using
FFT-based smoothed subgradient descent with adaptive step size and
periodic diversification.
"""

import numpy as np
import time


def _fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    result = np.fft.irfft(fa * fa, N)
    return result[:2 * n - 1]


def _score_from_parts(n, M, S):
    """Compute objective from components."""
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)


def _smoothed_gradient(a, conv, tau):
    """Compute gradient of softmax-smoothed max of conv."""
    n = len(a)
    # Softmax weights
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / tau)
    w = exp_c / np.sum(exp_c)
    
    # Gradient: sum_m w[m] * grad(c[m])
    # grad(c[m])_k = 2*a[m-k] if 0<=m-k<n, else 0
    grad = np.zeros(n)
    # For efficiency, only process positions where w[m] is significant
    significant = w > 1e-10
    for m in np.where(significant)[0]:
        m = int(m)
        k_start = max(0, m - n + 1)
        k_end = min(m, n - 1)
        if k_start <= k_end:
            j_start = m - k_end
            j_end = m - k_start
            grad[k_start:k_end + 1] += w[m] * 2.0 * a[j_start:j_end + 1][::-1]
    return grad


def _project_simplex(a, target_sum=1.0):
    """Project onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _try_initializations(n, rng):
    """Generate several initial sequences of length n."""
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Gentle ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # 3. Square root ramp
    sqrt_ramp = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    inits.append(sqrt_ramp / sqrt_ramp.sum())
    
    # 4. Exponential (decaying)
    exp_seq = np.exp(np.linspace(0, 1, n))
    inits.append(exp_seq / exp_seq.sum())
    
    # 5. Sine bump
    sine = np.sin(np.pi * np.arange(n) / n)
    sine = np.maximum(sine, 0)
    inits.append(sine / sine.sum())
    
    # 6. Random
    inits.append(rng.random(n))
    inits[-1] /= inits[-1].sum()
    
    # 7. Log ramp
    log_ramp = np.log(np.arange(1, n + 1, dtype=np.float64))
    inits.append(log_ramp / log_ramp.sum())
    
    # 8. "Stretched" - concave up

    concave_up = t ** 3
    concave_up = concave_up / concave_up.sum()
    inits.append(concave_up)
    
    return inits

    """Compute gradient using only top-k positions."""
    n = len(a)
    # Get top-k positions
    if len(conv) <= top_k:
