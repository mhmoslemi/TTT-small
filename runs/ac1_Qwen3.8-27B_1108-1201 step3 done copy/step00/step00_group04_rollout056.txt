def project_simplex(y):
    n = len(y)
    # Binary search for tau such that sum(max(0, y_i - tau)) = 1
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(100):
        tau = (lo + hi) / 2
        s = np.sum(np.maximum(0, y - tau))
        if s > 1:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    return np.maximum(0, y - tau)

def project_simplex(y):
    """Project y onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(y)
    # Sort y in descending order
    u = np.sort(y)[::-1]
    # Find the largest rho such that sum_{i=1}^{rho} (u_i - tau) = 1
    # where tau = (sum_{i=1}^{rho} u_i - 1) / rho
    # Condition: u_rho > tau, i.e., u_rho > (sum_{i=1}^{rho} u_i - 1) / rho
    # i.e., rho * u_rho > sum_{i=1}^{rho} u_i - 1
    # i.e., 1 + rho * u_rho > sum_{i=1}^{rho} u_i
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho + 1 > cssv
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - 1) / rho_max
    return np.maximum(0, y - tau)

    b = np.fft.irfft(spec * spec.conj(), N_fft)[:N]
