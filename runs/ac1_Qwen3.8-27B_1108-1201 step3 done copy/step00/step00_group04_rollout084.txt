"""
Optimizes a non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses mirror descent (exponential weights) with soft-max smoothing of the max,
FFT-accelerated convolution, multiple sequence lengths, and periodic
exploration via noise to escape local minima.
"""

import numpy as np
import time


def fft_convolve(b):
    """Fast self-convolution of b using FFT."""
    n = len(b)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(B * B, fft_size)[:size]


def compute_objective(b):
    """Compute 2*n*max(conv(b,b)) for normalized b (sum=1)."""
    n = len(b)
    c = fft_convolve(b)
    return 2.0 * n * float(np.max(c))


def mirror_descent(b, rng, deadline, verbose=True):
    """Optimize b using mirror descent with soft-max approximation."""
    n = len(b)
    best_b = b.copy()
    best_obj = compute_objective(b)

    t = 8.0
    base_eta = 0.5
    iter_count = 0
    last_improve = 0
    stall_count = 0
    patience = 3000

    if verbose:
        print(f"    MD start: obj={best_obj:.6f}, n={n}")

    while time.monotonic() < deadline:
        c = fft_convolve(b)
        c_max = float(np.max(c))

        # Softmax weights over lags
        tc = t * (c - c_max)
        exp_tc = np.exp(tc)
        p = exp_tc / float(np.sum(exp_tc))

        # Gradient: grad_j = 2 * sum_k p[k] * b[k-j] for valid k-j
        grad = 2.0 * np.correlate(p, b, mode='valid')

        # Adaptive step size
        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            g_max = 1.0
        eta = base_eta / g_max

        # Mirror descent (exponential weights update)
        log_b = np.log(np.maximum(b, 1e-300))
        new_log_b = log_b - eta * grad
        new_log_b -= float(np.max(new_log_b))
        new_b = np.exp(new_log_b)
        new_b_sum = float(np.sum(new_b))
        if new_b_sum > 1e-300:
            new_b /= new_b_sum
        else:
            break

        b = new_b
        iter_count += 1

        # Periodic evaluation
        if iter_count % 100 == 0:
            obj = compute_objective(b)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_b = b.copy()
                last_improve = iter_count
                stall_count = 0
                if verbose and iter_count % 2000 == 0:
                    print(f"    MD iter={iter_count}: obj={obj:.6f}")
            else:
                stall_count += 1

            # Increase temperature
            if iter_count % 3000 == 0:
                t = min(t * 1.15, 80.0)

            # Exploration: add noise when stalled
            if stall_count > patience // 100:
                noise_scale = 0.5 * float(np.max(b))
                noise = rng.normal(0.0, noise_scale, size=n)
                b = b + noise
                b = np.maximum(b, 1e-10)
                b /= float(np.sum(b))
                stall_count = 0
                if verbose:
                    print(f"    MD iter={iter_count}: noise restart, t={t:.1f}")

        # Gradually increase base_eta for convergence
        if iter_count % 5000 == 0:
            base_eta *= 1.1

    # Final check
    obj = compute_objective(b)
    if obj < best_obj:
        best_obj = obj
        best_b = b.copy()

    return best_b, best_obj


def structured_init(n, kind, rng):
    """Create a structured initialization of length n."""
    if kind == 'uniform':
        return np.ones(n, dtype=np.float64)
    elif kind == 'gaussian':
        j = np.arange(n, dtype=np.float64)
        return np.exp(-4.0 * ((j - (n - 1) / 2.0) / (n / 3.0)) ** 2)
    elif kind == 'raised_cosine':
        j = np.arange(n, dtype=np.float64)
        return 0.5 * (1.0 + np.cos(2.0 * np.pi * j / (n - 1)))
    elif kind == 'triangle':
        j = np.arange(n, dtype=np.float64)
        return 1.0 - np.abs(2.0 * j / (n - 1) - 1.0)
    elif kind == 'random':
        return rng.exponential(1.0, size=n)
    elif kind == 'alternating_damped':
        j = np.arange(n, dtype=np.float64)
        return 1.0 + 0.3 * np.cos(2.0 * np.pi * j / 4.0) * np.exp(-2.0 * np.abs(j - n / 2.0) / n)
    elif kind == 'sinusoidal':
        j = np.arange(n, dtype=np.float64)
        return 1.0 + 0.4 * np.sin(2.0 * np.pi * 3.0 * j / n)
    else:
        return np.ones(n, dtype=np.float64)
