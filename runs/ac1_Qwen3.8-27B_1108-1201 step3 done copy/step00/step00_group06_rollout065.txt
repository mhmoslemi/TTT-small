"""
Minimizes 2n*max(convolve(a,a))/(sum(a))^2 for non-negative a.
Uses soft-max smoothed L-BFGS in log-space with annealing,
followed by projected gradient local search with hard max.
Key: n-shape initializations and iterative beta increase to approach
near-flat self-convolution, reducing score below the uniform bound of 2.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
from scipy.ndimage import gaussian_filter
import time


def _full_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A, fsize))[:size]
    return np.maximum(c, 0.0)


def _score(a):
    """Compute score = 2n * max(c) / S^2."""
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    c = _full_convolve(a)
    return 2.0 * len(a) * float(np.max(c)) / (S * S)


def _softmax_objgrad(x, beta):
    """Smooth objective and gradient in log-space. x = log(a)."""
    a = np.exp(np.clip(x, -50.0, 50.0))
    n = len(a)
    S = float(np.sum(a))
    c = _full_convolve(a)

    cmax = float(np.max(c))
    bc = beta * (c - cmax)
    maxbc = float(np.max(bc))
    exp_bc = np.exp(bc - maxbc)
    sumexp = float(np.sum(exp_bc))
    logZ = maxbc + np.log(sumexp)
    sm = cmax + logZ / beta
    w = exp_bc / sumexp

    score = 2.0 * n * sm / (S * S)

    # Cross-correlation: cross[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # = (a_rev * w)_lin[n-1+i]
    a_rev = a[::-1]
    total_len = n + (2 * n - 1) - 1
    fsize = 1 << (total_len - 1).bit_length()
    Ar = fft(a_rev, fsize)
    W = fft(w, fsize)
    cross_full = np.real(ifft(Ar * W, fsize))
    cross = cross_full[n - 1: 2 * n - 1]

    grad = 4.0 * n * a * (cross * S - sm) / (S * S * S)
    return score, grad


def _local_search(a, rng, budget_s):
    """Projected subgradient descent with backtracking line search."""
    deadline = time.monotonic() + budget_s
    n = len(a)
    best = a.copy()
    best_s = _score(a)
    no_improve_count = 0

        S = float(np.sum(a))
        c = _full_convolve(a)
        k_star = int(np.argmax(c))
        max_c = float(c[k_star])

        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)

        grad_a = np.zeros(n)
        if np.any(valid):
            grad_a[valid] = 4.0 * n * (a[idx[valid]] * S - max_c) / (S * S * S)

        gnorm = float(np.linalg.norm(grad_a))
        if gnorm < 1e-18:
            break

        # Backtracking line search in a-space
        success = False
        lr = 0.5 / (gnorm + 1e-10)
        for _ in range(25):
            a_new = np.maximum(0.0, a - lr * grad_a)
            s_new = _score(a_new)
            if s_new < best_s:
                a = a_new
                best_s = s_new
                best = a.copy()
                no_improve_count = 0
                success = True
                break
            lr *= 0.6

        if not success:
            # Try random perturbation
            a_pert = np.maximum(0.0, a * np.exp(rng.normal(0.0, 0.015, n)))
            s_pert = _score(a_pert)
            if s_pert < best_s:
                a = a_pert
                best_s = s_pert

                no_improve_count = 0
            else:

                    break
                a = best.copy()

        # Periodic progress
        if int(time.monotonic()) % 30 == 0:
            print(f"  Local search: score={best_s:.6f}")

    return best, best_s


def _refine_with_lbfgs(x_start, beta, maxiter):
    """Run L-BFGS from x_start with given beta."""
    try:
        res = minimize(
            _softmax_objgrad, x_start, args=(float(beta),),
            jac=True, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxls': 50}
        )
        return np.clip(res.x, -50.0, 50.0), res.fun
    except Exception:
        return x_start, np.inf
