def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if np.sum(v) <= s and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Minimizes the evaluation function using projected gradient descent on the simplex,
with smooth L^p approximation of the max auto-correlation. Tries multiple initializations.

import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad(a, p=8):
    """Compute smooth objective L^p norm of convolve(a,a) and its gradient."""
    b = np.convolve(a, a)
    # L^p norm
    obj = np.sum(b ** p) ** (1.0 / p)
    # Gradient: d/da[m] sum_k b[k]^p = p * sum_k b[k]^(p-1) * 2*a[k-m] * I[0<=k-m<n]
    c = b ** (p - 1)
    grad = 2.0 * p * np.correlate(c, a, mode='valid')
    return obj, grad
