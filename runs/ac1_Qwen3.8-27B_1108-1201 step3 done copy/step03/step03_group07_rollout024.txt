"""
LP-based iterative refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming at each step to find the best direction that respects
the current correlation bound, with adaptive step size, multiple starting
points, and smooth gradient polishing. The LP formulation directly optimizes
the peak constraint rather than using a smooth surrogate.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f):
    """Build the (2n-1) x n matrix A where (f*g)(k) = sum_i A[k,i]*g[i]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # g[i] appears in (f*g)(k) for k = i, i+1, ..., i+n-1
        # with coefficient f[k-i]
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = f[:k_end - i]
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k."""
    n = len(f)
    A = build_constraint_matrix(f)
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return None

    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0.0)


def lp_refine(f, n_iters=500, t_init=0.3, budget_s=300.0):
    """LP-based iterative refinement with adaptive step size."""
    f = f.copy().astype(np.float64)
    n = len(f)
    f = f / f.sum()

    best_f = f.copy()
    best_obj = compute_objective(f)
    t = t_init
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > budget_s:
            break

        M = fast_convolve(f).max()
        g_opt = solve_lp(f, M)
        if g_opt is None:
            break

        g_sum = g_opt.sum()
        if g_sum < 1e-15:
            break
        g_scaled = g_opt / g_sum

        # Adaptive step size: try several step sizes
        stepped = False
        for trial_t in [t, 0.5 * t, 0.25 * t, 0.1 * t, 0.05 * t]:
            h = (1.0 - trial_t) * f + trial_t * g_scaled
            h = np.maximum(h, 0.0)
            s = h.sum()
            if s < 1e-15:
                continue
            h = h / s
            obj = compute_objective(h)
            if obj <= best_obj + 1e-10:
                f = h
                t = max(trial_t, 0.01)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_f = h.copy()
                stepped = True
                break

        if not stepped:
            # Try a very small step

            h = np.maximum(h, 0.0)
            s = h.sum()
            if s > 1e-15:
                h = h / s
                obj = compute_objective(h)
                if obj < best_obj:
                    f = h
                    best_obj = obj
                    best_f = h.copy()
            # Reduce step size
            t *= 0.5
            if t < 0.005:
                break

        if it % 50 == 0:
            elapsed = time.monotonic() - t_start
            print("    LP iter %d: obj=%.8f, t=%.4f, elapsed=%.1fs" % (it, best_obj, t, elapsed))

    return best_f, best_obj


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    return x_max + np.log(total) / beta, w

    f = f.copy().astype(np.float64)

    f = f / f.sum()

    best_f = f.copy()
    best_obj = compute_objective(f)

            break

        conv = fast_convolve(f)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if 2.0 * n * conv.max() < best_obj:
            best_obj = 2.0 * n * conv.max()
            best_f = f.copy()

        if it % 20 == 0:
            grad = compute_smooth_gradient(f, w)
        else:
            grad = compute_smooth_gradient(f, w)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break

        # Scale learning rate by gradient norm
        step_lr = lr * 2.0 / (grad_norm + 1e-10)
        f_new = project_simplex(f - step_lr * grad)
        
        # Check if it helped
        new_obj = compute_objective(f_new)
        if new_obj < best_obj:
            f = f_new
            best_obj = new_obj
            best_f = f_new.copy()
        else:
            lr *= 0.9
            if lr < 1e-6:
                break

    return best_f, best_obj


def spectral_init(n, rng):
    """Initialize using random phases in frequency domain (flat spectrum)."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for _ in range(30):
        phases = rng.uniform(0, 2 * np.pi, fft_size)
        A = np.zeros(fft_size, dtype=complex)
        A[0] = float(n)
        for k in range(1, fft_size // 2 + 1):
            mag = 1.0 + 0.5 * rng.random()
            A[k] = mag * np.exp(1j * phases[k])
            if k < fft_size - k:
                A[fft_size - k] = mag * np.exp(-1j * phases[k])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = compute_objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def structured_init(n, rng):
    """Generate multiple structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%d' % int(sf * 100)] = g

    # Bimodal (edge concentration)
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Triangle
    tri = np.minimum(x, n - 1 - x).astype(float) + 0.1
    tri /= tri.sum()
    inits['triangle'] = tri

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Spectral
    inits['spectral'] = spectral_init(n, rng)

    return inits

                pp = np.maximum(parent, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits['parent'] = pp / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent_n), np.maximum(parent, 0))
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits['parent_interp'] = pp / ps

        # Pick top 2-3 initializations
        priority = ['parent', 'parent_interp', 'spectral', 'bimodal', 'gauss_35', 'uniform', 'random']
        init_names = [nm for nm in priority if nm in inits][:3]

        quick_budget = max(30.0, time_per_length * 0.3)
        for name in init_names:
            if time.monotonic() > deadline - 100:
                break
            a0 = inits[name]
            b = max(10.0, quick_budget / max(1, len(init_names)))
            a_opt, obj = try_lp_with_time(a0, b)
            score = score_and_update(a_opt)
            if np.isfinite(score):
                phase1_results.append((score, n, name, a_opt.copy()))
            print("  Phase1: n=%d, init=%s, obj=%.6f" % (n, name, obj))

    phase1_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best so far: %.6f" % best_score)

    # === Phase 2: Deep LP refinement on best candidates ===
    print("=== Phase 2: Deep LP Refinement ===")
    phase2_deadline = deadline - 60.0

    # Take the top candidates from phase 1 and refine further
    candidates_to_refine = phase1_results[:3]
    if not candidates_to_refine and best_sequence is not None:
        candidates_to_refine = [(best_score, len(best_sequence), 'best', best_sequence.copy())]

    for (score, n, name, a_init) in candidates_to_refine:
        if time.monotonic() > phase2_deadline:
            break

        remaining = phase2_deadline - time.monotonic()
        b = max(30.0, remaining * 0.6)
        print("  Phase2: refining n=%d, init=%s, current_obj=%.6f, budget=%.0fs" % (n, name, score, b))

        a_opt, obj = try_lp_with_time(a_init, b)
        score = score_and_update(a_opt)
        print("  Phase2 result: obj=%.8f" % obj)

    # Also try a fresh start at the best length if we have time
    if best_sequence is not None and time.monotonic() < deadline - 120:
        best_n = len(best_sequence)
        inits_fresh = structured_init(best_n, rng)
        # Use the current best as a perturbed start
        a_pert = best_sequence / best_sequence.sum()
        noise = 0.01 * rng.standard_normal(best_n)
        a_pert = project_simplex(a_pert + noise)
        remaining = deadline - time.monotonic() - 60
        b = max(30.0, remaining * 0.5)
        print("  Phase2: fresh perturbation, n=%d, budget=%.0fs" % (best_n, b))
        a_opt, obj = try_lp_with_time(a_pert, b)
        score_and_update(a_opt)

    # === Phase 3: Gradient polishing ===
    print("=== Phase 3: Gradient Polishing ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 10
        b = max(10.0, remaining * 0.7)
        a_opt, obj = gradient_polish(best_sequence, n_iters=500, beta=30.0, lr=0.05, budget_s=b)
        score = score_and_update(a_opt)
        print("  Polish result: obj=%.8f" % obj)

        # Second polishing round with different beta
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic() - 5
            b = max(5.0, remaining * 0.5)
            a_opt2, obj2 = gradient_polish(best_sequence, n_iters=300, beta=10.0, lr=0.03, budget_s=b)
            score_and_update(a_opt2)

    # === Phase 4: Final LP micro-iterations ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 2
        b = max(5.0, remaining)
        a_opt, obj = try_lp_with_time(best_sequence, b)
        score_and_update(a_opt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = compute_objective(best_sequence)

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
