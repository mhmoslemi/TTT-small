        def project_simplex(v):
            n = len(v)
            if n == 1:
                return np.array([max(v[0], 0.0)])
            u = np.sort(v)[::-1]
            cssv = np.cumsum(u) - 1
            cond = u * np.arange(1, n+1) > cssv
            if not cond.any():
                return np.ones(n) / n
            rho = np.nonzero(cond)[0][-1]
