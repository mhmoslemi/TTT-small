import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def solve_lp(a, M_factor=1.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M*conv_max, g >= 0."""
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max() * M_factor
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        return result.x
    return None

def line_search(a, direction, current_obj, n, max_steps=30):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-phase strategy:
(1) Constraint-generation LP for fast direction finding, (2) adaptive line search
with golden-section refinement, (3) multi-scale and multi-start exploration,
(4) subgradient fine-tuning with softmax-weighted peaks, (5) Fourier-domain
spectral polishing, and (6) symmetry exploitation. Targets improvement over 1.506254.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full, length 2n-1)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def build_reduced_conv_matrix(a, row_indices):
    """Build reduced constraint matrix for specified rows only."""
    n = len(a)
    m = len(row_indices)
    C = np.zeros((m, n))
    for i, row in enumerate(row_indices):
        j_start = max(0, row - n + 1)
        j_end = min(n - 1, row)
        idx = np.arange(j_start, j_end + 1)
        C[i, idx] = a[row - idx]
    return C


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def solve_lp_reduced(a, M, k_constraints=300):
    """
    Solve LP: max sum(g) s.t. C_top @ g <= M, g >= 0.
    Uses only top-k constraints for speed.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Select top-k constraint rows (most violated / most important)
    k = min(k_constraints, 2 * n - 1)
    top_idx = np.argpartition(conv, -k)[-k:]
    top_idx = np.sort(top_idx)

    C = build_reduced_conv_matrix(a, top_idx)
    c_obj = -np.ones(n)
    b_ub = np.full(k, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None

    C = build_conv_matrix(a)
    c_obj = -np.ones(n)

def line_search_adaptive(a, direction, current_obj, n, max_evals=25):
    """Adaptive line search: geometric candidates + golden section refinement."""
    best_t = 0.0
    best_obj = current_obj
    evals = 0

    candidates = [1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01]
    for t in candidates:
        if evals >= max_evals:
            break
        a_try = np.maximum(a + t * direction, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            evals += 1
            continue
        a_try = a_try / s
        obj_try = compute_obj_normalized(a_try)
        evals += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    # Golden section refinement around best_t
    if 0.01 < best_t < 0.95 and evals < max_evals:
        lo = max(0.0, best_t * 0.5)
        hi = min(1.0, best_t * 1.8)
        gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
        for _ in range(6):
            if evals >= max_evals:
                break
            d = hi - lo
            x1 = hi - gr * d
            x2 = lo + gr * d
            a1 = np.maximum(a + x1 * direction, 0.0)
            s1 = a1.sum()
            o1 = compute_obj_normalized(a1 / s1) if s1 > 1e-15 else float('inf')
            a2 = np.maximum(a + x2 * direction, 0.0)
            s2 = a2.sum()
            o2 = compute_obj_normalized(a2 / s2) if s2 > 1e-15 else float('inf')
            evals += 2
            if o1 < o2:
                hi = x2
                if o1 < best_obj:
                    best_obj = o1
                    best_t = x1
            else:
                lo = x1
                if o2 < best_obj:
                    best_obj = o2
                    best_t = x2

    return best_t, best_obj


def subgradient_step(a, conv, lr, momentum_vec, peaks_k, temp, n):
    """Compute subgradient and apply momentum update."""
    peak_idx = np.argpartition(conv, -peaks_k)[-peaks_k:]
    peak_vals = conv[peak_idx]
    scaled = peak_vals / max(temp, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peak_idx))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for ii in range(len(peak_idx)):
        if w[ii] < 1e-12:
            continue
        k = int(peak_idx[ii])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        # d(conv[k])/da[i] = 2*a[k-i] (or a[i] if 2i==k)
        contrib = a[k - idx] * (2.0 / w[ii]) * w[ii]  # = 2*a[k-idx]
        # Handle the diagonal case (2i == k)
        for jj in range(len(idx)):
            if 2 * idx[jj] == k:
                contrib[jj] = a[idx[jj]]
        sg[idx] += w[ii] * contrib

    momentum_vec = momentum_vec + lr * sg
    return momentum_vec, sg


def sg_optimize(a0, time_limit, lr_init=0.01, momentum_rate=0.92, peaks_k=50, temp_init=0.0002):
    """Subgradient optimization with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    momentum_vec = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    momentum_rate_cur = momentum_rate
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        momentum_vec, _ = subgradient_step(a, conv, lr, momentum_vec, peaks_k, temp, n)
        a_new = a - momentum_vec
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            momentum_vec = np.zeros(n)

        obj_cur = compute_obj_normalized(a)
        if obj_cur < best_obj - 1e-13:
            best_obj = obj_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.003
        momentum_rate_cur = min(0.97, momentum_rate_cur + 0.0002)

        if no_improve >= 500:
            break

    return best_a, best_obj


def fourier_polish(a, time_limit, n_freqs=5):

    Fourier-domain polishing: identify frequency components that contribute
    to the convolution peak and damp them.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    iterations = 0

    while time.monotonic() - t0 < time_limit and iterations < 200:
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))

        # Compute FFT of a
        A = rfft(a, fft_size)

        # The convolution in frequency domain is |A|^2
        # We want to reduce the peak at lag peak_k
        # This corresponds to reducing certain frequency components
        # Simple approach: slightly damp the largest frequency components
        # and re-project

        # Try: reduce the 2nd, 3rd, ... largest non-DC frequency magnitudes
        mag = np.abs(A)
        # Don't touch DC
        top_freqs = np.argsort(mag[1:])[-n_freqs:] + 1

        # Damp these frequencies slightly
        for f in top_freqs:
            A[f] *= 0.98

        # Inverse FFT
        a_new = irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                # Try smaller damping
                A2 = rfft(a, fft_size)
                for f in top_freqs:
                    A2[f] *= 0.995
                a_try = irfft(A2, fft_size)[:n]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    obj_try = compute_obj_normalized(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
        iterations += 1

    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False, k_constraints=300, use_symmetry=False):
    """
    Main LP-based optimization loop with constraint generation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    if use_symmetry:
        a = (a + a[::-1]) / 2.0
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 80
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP with reduced constraints
        M = conv.max() * 1.001
        g = solve_lp_reduced(a, M, k_constraints=k_constraints)

        if g is None:
            no_improve += 10
        else:
            g_sum = g.sum()
            if g_sum <= s * (1 + 1e-7):
                no_improve += 1
            else:
                # Direction: move towards g (normalized)
                g_norm = g / g_sum
                d = g_norm - a
                norm_d = np.linalg.norm(d)

                if norm_d < 1e-13:
                    no_improve += 1
                else:
                    t_best, obj_best = line_search_adaptive(a, d, best_obj, n)

                    if obj_best < best_obj - 1e-12:
                        a_new = np.maximum(a + t_best * d, 0.0)
                        if use_symmetry:
                            a_new = (a_new + a_new[::-1]) / 2.0
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a = a_new / s_new
                        else:
                            a = best_a.copy()

                        conv = fast_convolve(a)
                        best_obj = 2.0 * n * conv.max()
                        best_a = a.copy()
                        no_improve = 0
                        if verbose and iteration % 20 == 0:
                            print(f"    LP iter {iteration}: obj={best_obj:.8f}, "
                                  f"score={best_obj/(2.0*n)*2.0*n:.6f}, t={time.monotonic()-t0:.1f}s")
                    else:
                        no_improve += 1

        iteration += 1

        # Perturbation on stagnation
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + perturb_count)
            scale = 0.003 + 0.004 * min(perturb_count, 8)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            if use_symmetry:
                a = (a + a[::-1]) / 2.0
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(400, 80 + 40 * perturb_count)

            if perturb_count > 40:
                break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Triangle (peak at center)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Bimodal (peaks at edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian (peak at center, wider)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits['sine'] = sine / sine.sum()

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 0.01)
    inits['raised_cos'] = rc / rc.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_resample'] = pp / pps

    # Fourier-initialized: create a signal with desired spectral properties
    # Flat-top in frequency -> concentrated in time (bad for us)
    # We want flat in time -> concentrated in frequency
    # Try: signal whose Fourier transform has a specific shape
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    # Create a frequency response that's "anti-peaky"
    freqs = np.arange(fft_size // 2 + 1)
    # Damp low frequencies (they contribute to the DC/center of convolution)
    target_spec = np.ones(fft_size // 2 + 1)
    target_spec[0] = 0.5  # reduce DC
    # Add some structure
    for f in range(1, min(5, fft_size // 2)):
        target_spec[f] = 0.7
    A_target = (target_spec + 1j * rng.standard_normal(fft_size // 2 + 1) * 0.3).astype(complex)
    a_fourier = irfft(A_target, fft_size)[:n]
    a_fourier = np.maximum(a_fourier, 0)
    fs = a_fourier.sum()
    if fs > 1e-10:
        inits['fourier'] = a_fourier / fs

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Broad scan across lengths and initializations
    Phase 2: Deep LP optimization on top candidates
    Phase 3: Multi-start exploitation with perturbations
    Phase 4: Fine-tuning (subgradient + Fourier polish + final LP)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    def update_best(a_opt):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            return True
        return False

    # ==================== PHASE 1: Broad Scan ====================
    print("=== Phase 1: Broad Scan ===")
    t_start = time.monotonic()
    phase1_time = 400.0
    phase1_deadline = t_start + phase1_time

    lengths = [256, 512, 768, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = generate_initializations(n, parent, rng)
        # Priority: parent first, then structured
        priority = ['parent', 'parent_resample', 'uniform', 'sine', 'raised_cos',
                    'bimodal', 'gaussian', 'fourier', 'random', 'triangle']
        init_names = [nm for nm in priority if nm in inits][:4]

        t_per_init = max(20.0, (phase1_time - 20.0) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 20:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a0 = inits[name]
            use_sym = (name in ['uniform', 'sine', 'raised_cos', 'gaussian', 'parent', 'parent_resample'])
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, k_constraints=200,
                                         use_symmetry=use_sym)
            if update_best(a_opt):
                print(f"  n={n}, {name}: score={best_score:.6f}")
            scan_results.append((best_score if best_sequence is a_opt else obj_opt,
                                 n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ==================== PHASE 2: Deep LP Optimization ====================
    print("=== Phase 2: Deep LP ===")
    t_start2 = time.monotonic()

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = n
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    # Remove duplicates
    final_configs = []
    seen2 = set()
    for n, a in top_configs:
        if n not in seen2:
            final_configs.append((n, a))
            seen2.add(n)

    remaining = deadline - time.monotonic() - 200.0
    n_cand = max(1, len(final_configs))
    t_per_deep = max(30.0, remaining * 0.55 / n_cand)

    for i, (n, a0) in enumerate(final_configs):
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 100)
        if t_alloc < 15:
            break

        print(f"  Deep LP {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, k_constraints=400,
                                     verbose=True)
        if update_best(a_opt):
            print(f"  Deep LP {i} improved: score={best_score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ==================== PHASE 3: Multi-Start Exploitation ====================
    print("=== Phase 3: Multi-Start ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        n_perturb = max(3, int(remaining / 60))
        t_per_perturb = remaining / n_perturb

        for pi in range(n_perturb):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per_perturb, deadline - time.monotonic() - 50)
            if t_alloc < 10:
                break

            n_best = len(best_sequence)
            # Diverse perturbation strategies
            if pi % 3 == 0:
                # Gaussian noise
                scale = 0.003 * (1 + pi * 0.3)
                noise = rng.standard_normal(n_best) * scale
                a_pert = np.maximum(best_sequence + noise, 0.0)
            elif pi % 3 == 1:
                # Fourier-domain perturbation
                fft_size = 1 << (2 * n_best - 1 - 1).bit_length()
                A = rfft(best_sequence, fft_size)
                # Perturb mid frequencies
                mid_start = n_best // 4
                mid_end = 3 * n_best // 4
                if mid_end < len(A):
                    A[mid_start:mid_end] *= (1 + rng.standard_normal(mid_end - mid_start) * 0.02)
                a_pert = irfft(A, fft_size)[:n_best]
                a_pert = np.maximum(a_pert, 0.0)
            else:
                # Structural perturbation: blend with a different template
                template = rng.dirichlet(np.ones(n_best) * 0.3)
                mix = 0.1 + 0.05 * pi
                a_pert = (1 - mix) * best_sequence + mix * template

            s = a_pert.sum()
            if s > 1e-15:
                a_pert = a_pert / s
                a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc, k_constraints=300)
                if update_best(a_opt):
                    print(f"  Perturb {pi}: score={best_score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ==================== PHASE 4: Fine-Tuning ====================
    print("=== Phase 4: Fine-Tuning ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        n_best = len(best_sequence)

        # 4a: Subgradient fine-tuning
        if remaining > 60:
            t_sg = min(100.0, remaining * 0.35)
            print(f"  Subgradient: {t_sg:.0f}s")
            sg_configs = [
                {'lr_init': 0.008, 'momentum_rate': 0.93, 'peaks_k': 60, 'temp_init': 0.00015},
                {'lr_init': 0.012, 'momentum_rate': 0.9, 'peaks_k': 40, 'temp_init': 0.0003},
                {'lr_init': 0.005, 'momentum_rate': 0.95, 'peaks_k': 100, 'temp_init': 0.0001},
            ]
            t_per_sg = t_sg / len(sg_configs)
            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 30:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 20)
                if t_alloc < 5:
                    break
                a_opt, _ = sg_optimize(best_sequence, time_limit=t_alloc, **cfg)
                if update_best(a_opt):
                    print(f"  SG {ci}: score={best_score:.6f}")

        # 4b: Fourier polish
        if remaining > 40 and time.monotonic() < deadline - 20:
            t_fp = min(40.0, deadline - time.monotonic() - 10)
            if t_fp > 5:
                print(f"  Fourier polish: {t_fp:.0f}s")
                a_opt, _ = fourier_polish(best_sequence, time_limit=t_fp, n_freqs=5)
                if update_best(a_opt):
                    print(f"  Fourier: score={best_score:.6f}")

        # 4c: Final LP polish
        if time.monotonic() < deadline - 5:
            t_final = min(30.0, deadline - time.monotonic() - 2)
            if t_final > 3:
                print(f"  Final LP: {t_final:.0f}s")
                a_opt, _ = lp_optimize(best_sequence, time_limit=t_final, k_constraints=500)
                if update_best(a_opt):
                    print(f"  Final LP: score={best_score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
