def solve_lp_sparse(f, M, active_set=None, max_constraints=None):
    """Solve LP with only a subset of constraints."""
    n = len(f)
    b = fast_convolve(f)
    
    if active_set is None:
        # Use the top k constraints (where b is largest)
        if max_constraints is not None and max_constraints < len(b):
            active_set = np.argpartition(b, -max_constraints)[-max_constraints:]
        else:
            active_set = range(len(b))
    
    # Build constraint matrix for active constraints only
    A = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        # (f*g)[k] = sum_j f[k-j] * g[j] (for valid j)
        # So the row is: A[idx, j] = f[k-j] if 0 <= k-j < n, else 0
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[idx, j] = f[kj]
    
    c = -np.ones(n)
    b_ub = np.full(len(active_set), M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def solve_lp_cg(f, M, max_iterations=20):
    """Solve LP with constraint generation."""
    n = len(f)
    b_f = fast_convolve(f)
    
    # Start with the top constraints
    k_active = np.argpartition(b_f, -50)[-50:]  # Top 50 by b_f value

        A = np.zeros((len(k_active), n))
        for idx, k in enumerate(k_active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A[idx, j] = f[k - j]
        
        c = -np.ones(n)
        b_ub = np.full(len(k_active), M)

        fg = fast_convolve(f)  # This is f*f, not f*g!

    # (fallback)
