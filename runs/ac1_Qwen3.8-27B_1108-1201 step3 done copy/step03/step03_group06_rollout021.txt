    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(p, beta):
    """
    Compute gradient of log-sum-exp(beta * (p*p)) / beta w.r.t. p.
    Returns (gradient, conv).
    
    Uses the identity: d(p*p)[k]/dp[i] = 2*p[k-i] if 0<=k-i<n else 0
    So grad[i] = 2 * sum_j p[j] * w[j+i] where w = softmax(beta * (p*p))
    Computed via FFT: grad[i] = 2 * (p_rev * w)[i + n - 1]
    """
    n = len(p)
    conv = fast_convolve(p)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    needed = n + (2 * n - 1) - 1  # = 3n - 2

    P_rev = fft(p_rev, fft_size)
    W = fft(w, fft_size)

    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad, conv

def build_constraint_matrix(a, active_indices):
    """Build constraint matrix for active constraints only."""
    n = len(a)
    rows = []
    for k in active_indices:
        row = np.zeros(n)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        row[idx] = a[k - idx]
        rows.append(row)
    return np.array(rows)


def active_set_lp(a, M, active_indices, time_per_lp=2.0):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in active_indices.
    Returns direction b or None.
    """
    n = len(a)
    A_ub = build_constraint_matrix(a, active_indices)
    b_ub = np.full(len(active_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(0.1, time_per_lp),
                               'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0.7
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Identify active set: lags where conv[k] >= 0.95 * M
        threshold = 0.95 * M
        active = np.where(conv >= threshold)[0]

        # Also include the top-K peaks
        top_k = min(len(conv), max(10, n // 8))
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        active = np.union1d(active, top_indices)

        # Solve LP on active constraints
        b = active_set_lp(a, M, active, time_per_lp=min(3.0, (time_limit - (time.monotonic() - t0)) / 5))
        if b is None:
            # Fallback: try with just the top peak
            peak_k = np.argmax(conv)
            b = active_set_lp(a, M, [peak_k], time_per_lp=1.0)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        # Update
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            step = min(1.0, step * 1.05)
        else:
            no_improve += 1
            step *= 0.85
            if no_improve > 30:
                # Perturb
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                step = 0.5
                no_improve = 0

        if verbose and no_improve % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS: obj={best_obj:.6f}, active={len(active)}, step={step:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def freq_domain_init(n, rng):
    """
    Generate initialization with approximately flat power spectrum.
    Uses random phase on the DFT, takes magnitude of IDFT, squares and normalizes.
    """
    # Random flat-spectrum complex sequence
    freqs = np.arange(n)
    phases = 2.0 * np.pi * rng.random(n)
    # Make it conjugate symmetric for real output
    P = np.exp(1j * phases)
    # Ensure conjugate symmetry: P[k] = conj(P[n-k])
    P_sym = np.zeros(n, dtype=np.complex128)
    P_sym[0] = 1.0
    for k in range(1, (n + 1) // 2):
        P_sym[k] = P[k]
        P_sym[n - k] = np.conj(P[k])
    if n % 2 == 0:
        P_sym[n // 2] = 1.0

    x = np.real(ifft(P_sym))
    p = np.abs(x)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return p / s


def gauss_hermite_init(n, rng, alpha=1.0):
    """Gaussian-like initialization with optional asymmetry."""
    x = np.arange(n, dtype=np.float64)
    center = n * (0.3 + 0.4 * rng.random())
    width = n * (0.15 + 0.25 * rng.random())
    p = np.exp(-0.5 * ((x - center) / width) ** 2)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return p / s


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Cosine window
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.001)
        cos_w /= cos_w.sum()
    else:
        cos_w = np.ones(1)
    inits['cosine'] = cos_w

    # Frequency domain
    inits['freq1'] = freq_domain_init(n, rng)
    inits['freq2'] = freq_domain_init(n, rng)

    # Random Gaussians
    inits['rand_g1'] = gauss_hermite_init(n, rng)
    inits['rand_g2'] = gauss_hermite_init(n, rng)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_ds'] = pp / ps2
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_up'] = pp / ps2

    return inits


def evaluate_obj_normalized(a):
    """Evaluate objective assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining smooth gradient descent, mirror descent,
    and active-set LP methods. Explores multiple sequence lengths and uses
    frequency-domain initializations for diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except (NameError, Exception):
        pass

    # ========== Phase 1: Broad Exploration ==========
    print("=== Phase 1: Broad Exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = min(150.0, (budget_s - 45) * 0.15)

    lengths = [192, 256, 320, 384, 512, 640]
    if parent is not None and len(parent) in lengths:
        # Put parent length first
        lengths.remove(len(parent))
        lengths.insert(0, len(parent))

    scan_results = []
    init_priority = ['parent', 'parent_ds', 'parent_up', 'bimodal', 'freq1',
                     'gauss_center', 'uniform', 'cosine', 'random', 'freq2']

    for n in lengths:
        if time.monotonic() + 10 > t_phase1 + phase1_time:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = [nm for nm in init_priority if nm in inits]
        init_names += [nm for nm in inits if nm not in init_names]
        init_names = init_names[:4]

        time_per = max(5.0, (phase1_time - (time.monotonic() - t_phase1) - 5) /
                       max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() + 4 > t_phase1 + phase1_time:
                break
            t_alloc = min(time_per, t_phase1 + phase1_time - time.monotonic() - 4)
            if t_alloc < 3:
                break

            a0 = inits[name]
            # Quick gradient descent
            a_opt, obj_opt = smooth_gradient_descent(
                a0, time_limit=t_alloc, beta_init=300, lr_init=0.06,
                momentum=0.9, seed=n * 7 + hash(name) % 100)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # ========== Phase 2: Focused Deep Optimization ==========
    print("=== Phase 2: Deep Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = min(500.0, (budget_s - 45) * 0.35)

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 320

    # Collect diverse starting points
    starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:6]:
        key = (n, name)
        if key not in seen:
            starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(starts) >= 5:
            break
    if best_sequence is not None:
        starts.append((best_n, best_sequence.copy()))

    # Phase 2a: Smooth gradient descent with varied hyperparameters
    grad_configs = [
        {'beta_init': 200, 'lr_init': 0.1, 'momentum': 0.9, 'seed_offset': 0},
        {'beta_init': 500, 'lr_init': 0.05, 'momentum': 0.93, 'seed_offset': 100},
        {'beta_init': 1000, 'lr_init': 0.03, 'momentum': 0.95, 'seed_offset': 200},
        {'beta_init': 400, 'lr_init': 0.08, 'momentum': 0.92, 'seed_offset': 300},
        {'beta_init': 800, 'lr_init': 0.04, 'momentum': 0.94, 'seed_offset': 400},
    ]

    num_runs = len(starts) * len(grad_configs)
    time_per_run = max(10.0, phase2_time * 0.5 / max(1, num_runs))

    run_i = 0
    for (n_s, a_s) in starts:
        for cfg in grad_configs:
            if time.monotonic() + 8 > phase2_start + phase2_time * 0.55:
                break
            t_alloc = min(time_per_run, phase2_start + phase2_time * 0.55 - time.monotonic() - 8)
            if t_alloc < 8:
                break
            a_opt, _ = smooth_gradient_descent(
                a_s, time_limit=t_alloc, seed=run_i + cfg['seed_offset'], **{k: v for k, v in cfg.items() if k != 'seed_offset'})
            score = float(evaluate_sequence(a_opt.tolist()))
            run_i += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2a [{run_i}]: n={n_s}, score={score:.6f}")

    # Phase 2b: Mirror descent
    md_configs = [
        {'beta_init': 300, 'lr_init': 0.04, 'seed': 10},
        {'beta_init': 600, 'lr_init': 0.02, 'seed': 20},
        {'beta_init': 100, 'lr_init': 0.08, 'seed': 30},
    ]
    for cfg in md_configs:
        if time.monotonic() + 10 > phase2_start + phase2_time:
            break
        t_alloc = min(40.0, phase2_start + phase2_time - time.monotonic() - 10)
        if t_alloc < 10:
            break
        a_opt, _ = mirror_descent(best_sequence, time_limit=t_alloc, **cfg)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P2b MD: score={score:.6f}")

    # Phase 2c: Active set optimization
    if time.monotonic() < phase2_start + phase2_time - 30:
        t_alloc = min(120.0, phase2_start + phase2_time - time.monotonic() - 15)
        if t_alloc > 15:
            a_opt, _ = active_set_optimize(best_sequence, time_limit=t_alloc, seed=555)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2c AS: score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Alternating Refinement ==========
    print("=== Phase 3: Alternating Refinement ===")
    phase3_start = time.monotonic()
    phase3_time = min(350.0, (budget_s - 45) * 0.25)

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence.copy()

        # Alternate between gradient and active-set steps
        for cycle in range(3):
            if time.monotonic() + 30 > phase3_start + phase3_time:
                break
            t_alloc_grad = min(50.0, (phase3_time - (time.monotonic() - phase3_start)) / (3 - cycle) / 2 - 5)
            t_alloc_as = min(40.0, (phase3_time - (time.monotonic() - phase3_start)) / (3 - cycle) / 2 - 5)

            if t_alloc_grad > 8:
                a_opt, _ = smooth_gradient_descent(
                    a, time_limit=t_alloc_grad, beta_init=500 + cycle * 300,
                    lr_init=0.03 - cycle * 0.005, momentum=0.93 + cycle * 0.01,
                    seed=1000 + cycle * 17)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    a = a_opt.copy()
                    print(f"  P3 grad [{cycle}]: score={score:.6f}")

            if t_alloc_as > 8 and time.monotonic() < phase3_start + phase3_time - 10:
                a_opt, _ = active_set_optimize(
                    a, time_limit=t_alloc_as, seed=2000 + cycle * 13)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    a = a_opt.copy()
                    print(f"  P3 AS [{cycle}]: score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Final Polish ==========
    print("=== Phase 4: Final Polish ===")
    phase4_start = time.monotonic()
    phase4_time = max(10.0, deadline - time.monotonic())

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Very fine gradient descent
        t_p4 = phase4_time * 0.4
        if t_p4 > 8:
            a_opt, _ = smooth_gradient_descent(
                best_sequence, time_limit=t_p4, beta_init=3000,
                lr_init=0.005, momentum=0.96, seed=9999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 fine grad: score={score:.6f}")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            n_tries = min(30, int(remaining / 2))
            for i in range(n_tries):
                if time.monotonic() > deadline - 18:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  P4 perturb [{i}] s={scale}: score={score:.6f}")
                        break

        # Mirror descent polish
        remaining2 = deadline - time.monotonic() - 15
        if remaining2 > 15:
            a_opt, _ = mirror_descent(
                best_sequence, time_limit=min(30, remaining2),
                beta_init=2000, lr_init=0.01, seed=7777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 MD polish: score={score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Scale to use full range (the evaluate function clips to [0, 1000])
    # Normalizing to sum ~ sqrt(2n) helps with the sum_a^2 denominator
    # But since evaluate_sequence just clips and computes, the scale doesn't matter
    # for the ratio. Keep as is (sum=1) which is fine.

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
