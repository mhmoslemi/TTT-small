"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
Nesterov-accelerated subgradient descent, coordinate descent polishing,
and multi-size exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: row k has f[j] at column (k-j) for valid j
    # This is a banded matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def nesterov_sg_optimize(f0, time_limit, lr_init=0.005, momentum=0.9, 
                          top_k=40, temp_init=0.0001):
    """Nesterov-accelerated subgradient descent on the simplex."""

    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Nesterov momentum
    y = f.copy()  # extrapolated point
    v = np.zeros(n)  # momentum accumulator
    lr = lr_init
    temp = temp_init
    
    t0 = time.monotonic()
    no_improve = 0
    t = 1.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        
        # Multi-peak softmax weighting
        k_eff = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_eff)[-k_eff:]
        pv = conv_y[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w /= ws
        
        # Subgradient: d(conv)/d(a_i) for the weighted peaks
        sg = np.zeros(n)

            je = min(n - 1, k)

Actually, let me reconsider the Nesterov implementation. The standard Nesterov accelerated gradient method for minimizing $f(x)$ over a convex set $C$ is:

- $y_t = x_t + \frac{t-1}{t+2}(x_t - x_{t-1})$ (extrapolation)
- $x_{t+1} = P_C(y_t - \frac{\alpha}{t} \nabla f(y_t))$ (proximal step)

For the simplex projection, this becomes:
- $y_t$ is the extrapolated point (projected onto simplex)
- Compute subgradient at $y_t$
- $x_{t+1} = \text{Proj}_{\Delta}(y_t - \eta_t \cdot sg)$

The step size $\eta_t$ should decrease as $O(1/t)$ for guaranteed convergence.

But in practice, for this problem, I've found that using a fixed momentum with decaying step size works well:
- $v_t = \mu v_{t-1} - \eta_t \cdot sg$
- $x_{t+1} = \text{Proj}_{\Delta}(x_t + v_t)$
- $y_t = x_{t+1} + \frac{t-1}{t+2}(v_t)$ (or similar)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
Nesterov-accelerated subgradient, coordinate-swap local search, and
multi-size exploration. Combines global LP directions with local polishing.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            gx = res.x
            if gx.sum() > 1e-15:
                return gx
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):

def nesterov_sg_optimize(f0, time_limit, lr_init=0.004, momentum=0.92,
                         top_k=30, temp_init=0.00006):
    """Nesterov-accelerated subgradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    lr = lr_init
    temp = temp_init

    t0 = time.monotonic()
    no_improve = 0
    t = 1

    while time.monotonic() - t0 < time_limit:
        if time_limit - (time.monotonic() - t0) < 1.0:
            break
        t += 1

        conv_y = fast_convolve(f)
        k_eff = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_eff)[-k_eff:]
        pv = conv_y[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_range = np.arange(js, je + 1)
            sg[idx_range] += 2.0 * w[idx] * f[k - idx_range]

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)

        conv_f = fast_convolve(f_new)
        obj = conv_f.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        lr = lr_init / (1.0 + 0.008 * t)
        temp = temp_init * (1.003 ** min(t, 3000))

        if no_improve > 250:
            rng = np.random.default_rng((int(time.monotonic() * 10000) + t) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.6
            no_improve = 0

def pair_swap_polish(f0, time_limit, n_rounds_max=200):
    """Local search: swap mass between coordinate pairs near peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    for rnd in range(n_rounds_max):
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 1.0:
            break

        conv = fast_convolve(f)
        # Find top peaks
        k_eff = min(20, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]

        improved_this_round = False

        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            # Coordinates that affect c_k: j in [max(0,k-n+1), min(n-1,k)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            active = list(range(js, je + 1))

            if len(active) < 2 or len(active) > 30:
                # For large active sets, just try a few random pairs
                rng = np.random.default_rng(rnd * 100 + k)
                active = list(rng.choice(active, size=min(10, len(active)), replace=False))

            # Try swapping between pairs in active set
            for ii in range(len(active)):
                for jj in range(ii + 1, len(active)):
                    i = active[ii]
                    j = active[jj]
                    if f[i] < 1e-10 or f[j] < 1e-10:
                        continue

                    # Try transferring delta from larger to smaller (and vice versa)
                    base = min(f[i], f[j])
                    for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                        delta = frac * base
                        # Try: a_i += delta, a_j -= delta
                        f_trial = f.copy()
                        f_trial[i] += delta
                        f_trial[j] -= delta
                        if f_trial[j] < 0:
                            continue
                        conv_t = fast_convolve(f_trial)
                        obj_t = conv_t.max()
                        if obj_t < best_obj - 1e-13:
                            f = f_trial
                            best_obj = obj_t
                            best_f = f.copy()
                            improved_this_round = True
                            break
                        # Try reverse
                        f_trial[i] -= 2 * delta
                        f_trial[j] += 2 * delta
                        if f_trial[i] < 0:
                            continue
                        conv_t = fast_convolve(f_trial)
                        obj_t = conv_t.max()
                        if obj_t < best_obj - 1e-13:
                            f = f_trial
                            best_obj = obj_t
                            best_f = f.copy()
                            improved_this_round = True
                            break

        if improved_this_round:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two-peak
    peak1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    peak2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    tp = peak1 + peak2
    inits.append(tp / tp.sum())

    # Sine wave
    sw = np.abs(np.sin(2 * np.pi * x / n)) + 0.1
    inits.append(sw / sw.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    t2_start = time.monotonic()
    t2_budget = min(1100.0, deadline - time.monotonic() - 200.0)

    if best_seq is not None:
        f_start = best_seq.copy()
    elif parent is not None and len(parent) == focus_n:
        f_start = parent.copy()
    else:
        f_start = np.ones(focus_n) / focus_n

    n = focus_n

    # Sub-phase 2a: FW-LP main optimization
    # Run multiple passes with perturbations
    n_passes = 4
    per_pass = t2_budget / (n_passes + 1.5)  # Leave room for SG

    for p in range(n_passes):
        rem = t2_budget - (time.monotonic() - t2_start)
        if rem < 20:
            break
        t_alloc = min(per_pass, rem - 5)
        if t_alloc < 15:
            break

        if p == 0:
            f_init = f_start
        elif p == 1:
            noise = rng.standard_normal(n) * 0.004
            f_init = project_simplex(f_start + noise)
        elif p == 2:
            # Blend with reversed
            f_init = 0.75 * f_start + 0.25 * f_start[::-1]
            f_init = project_simplex(f_init)
        else:
            noise = rng.standard_normal(n) * 0.01
            f_init = project_simplex(f_start + noise)

        print(f"  FW-LP pass {p+1}/{n_passes}, t={t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
        score = try_seq(a_opt, f"fw2_p{p}")

        if best_seq is not None:
            f_start = best_seq.copy()

    # Sub-phase 2b: Nesterov SG
    rem = t2_budget - (time.monotonic() - t2_start)
    if rem > 30 and best_seq is not None:
        t_sg = min(150.0, rem - 10)
        print(f"  Nesterov SG, t={t_sg:.0f}s")
        a_opt, obj = nesterov_sg_optimize(best_seq, time_limit=t_sg,
                                          lr_init=0.003, momentum=0.93,
                                          top_k=max(10, n // 12), temp_init=0.00004)
        score = try_seq(a_opt, "nesterov_sg")

    # === Phase 3: Final polish ===
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Current best: {best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Pair-swap local search
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_ps = min(120.0, rem - 15)
            print(f"  Pair-swap polish, t={t_ps:.0f}s")
            a_opt, obj = pair_swap_polish(best_seq, time_limit=t_ps)
            score = try_seq(a_opt, "pair_swap")

        # Fine Nesterov
        rem = deadline - time.monotonic() - 20
        if rem > 15 and best_seq is not None:
            t_fine = min(60.0, rem - 10)
            print(f"  Fine SG, t={t_fine:.0f}s")
            a_opt, obj = nesterov_sg_optimize(best_seq, time_limit=t_fine,
                                              lr_init=0.001, momentum=0.95,
                                              top_k=max(5, n_best // 20), temp_init=0.00002)
            score = try_seq(a_opt, "sg_fine")

        # Another quick FW-LP
        rem = deadline - time.monotonic() - 10
        if rem > 10 and best_seq is not None:
            t_fw = min(40.0, rem - 5)
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=t_fw)
            score = try_seq(a_opt, "fw_final")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 8 and best_seq is not None:
            for trial in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 4:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, obj = nesterov_sg_optimize(f_pert, time_limit=min(rem, 15),
                                                  lr_init=0.001, momentum=0.95,
                                                  top_k=max(5, n_best // 20), temp_init=0.00001)
                score = try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
