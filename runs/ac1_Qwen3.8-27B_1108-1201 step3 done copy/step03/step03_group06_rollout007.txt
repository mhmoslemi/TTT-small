import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a, k_indices=None):
    """Build constraint matrix for LP: (a*b)[k] <= M for specified k values."""
    n = len(a)
    if k_indices is None:
        k_indices = np.arange(2 * n - 1)
    num_k = len(k_indices)
    A_ub = np.zeros((num_k, n))
    for i, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[i, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M, k_indices=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for specified k."""
    n = len(a)
    if k_indices is None:
        k_indices = np.arange(2 * n - 1)
    num_k = len(k_indices)

    A_ub = build_lp_matrix(a, k_indices)
    b_ub = np.full(num_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'dual_reduction_threshold': 1e-8})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def objective_normalized(a):
    """Compute 2*n*max(conv) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def line_search(a, direction, best_obj, n, max_iter=30):
    """Golden section line search for optimal step in a + t*(direction - a)."""
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio

    # Check if t=1 gives improvement
    a_test = (1.0 - hi) * a + hi * direction
    a_test = np.maximum(a_test, 0.0)
    s = a_test.sum()
    if s > 1e-15:
        a_test = a_test / s
    obj_hi = objective_normalized(a_test)

    if obj_hi >= best_obj:
        # Optimal step is between 0 and 1
        for _ in range(max_iter):
            m1 = lo + gr * (hi - lo)
            m2 = lo + (1.0 - gr) * (hi - lo)

            a1 = (1.0 - m1) * a + m1 * direction
            a1 = np.maximum(a1, 0.0)
            s1 = a1.sum()
            if s1 > 1e-15:
                a1 = a1 / s1
            obj1 = objective_normalized(a1)

            a2 = (1.0 - m2) * a + m2 * direction
            a2 = np.maximum(a2, 0.0)
            s2 = a2.sum()
            if s2 > 1e-15:
                a2 = a2 / s2
            obj2 = objective_normalized(a2)

            if obj1 < obj2:
                hi = m2
            else:
                lo = m1

            if hi - lo < 1e-6:
                break

        t = (lo + hi) / 2.0
    else:
        t = 1.0

    a_new = (1.0 - t) * a + t * direction
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new, objective_normalized(a_new), t


def get_peak_indices(conv, n, top_frac=0.15):

    total = len(conv)
    num_top = max(10, int(total * top_frac))
    peaks = np.argpartition(conv, -num_top)[-num_top:]
    return peaks


def peak_targeted_perturb(a, conv, rng, strength=0.02):
    """Perturb coordinates that contribute most to peak convolution values."""
    n = len(a)
    a_new = a.copy()
    # Find the global peak
    k_peak = np.argmax(conv)
    # Get contributing indices
    i_min = max(0, k_peak - n + 1)
    i_max = min(n - 1, k_peak)
    contributors = np.arange(i_min, i_max + 1)
    # Randomly pick a few to reduce and their partners to increase
    if len(contributors) > 2:
        idx1 = rng.choice(contributors)
        idx2 = (k_peak - idx1)
        if 0 <= idx2 < n and idx2 != idx1:
            delta = strength * a[idx1]
            a_new[idx1] -= delta
            a_new[idx2] += delta
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
    # Also add small random perturbation
    noise = rng.standard_normal(n) * 0.003
    a_new = project_simplex(a_new + noise)
    return a_new


def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax approximation."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)

    # grad[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]

    grad = cross + conv_wa
    return grad, conv

def gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.03,

"""
Improved hybrid LP + gradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
1. Backtracking/golden-section line search for LP step size
2. Full constraint LP (no top-k restriction) for better directions
3. Multi-scale initialization (optimize small, upsample, refine)
4. Peak-targeted perturbations to escape local minima
5. More aggressive exploration of sequence lengths
6. Adaptive relaxation and restart strategy

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a, k_indices=None):
    """Build constraint matrix for LP: (a*b)[k] <= M for specified k values."""
    n = len(a)
    if k_indices is None:
        k_indices = np.arange(2 * n - 1)
    num_k = len(k_indices)
    A_ub = np.zeros((num_k, n))
    for i, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[i, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M, k_indices=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for specified k."""
    n = len(a)
    if k_indices is None:
        k_indices = np.arange(2 * n - 1)
    num_k = len(k_indices)

    A_ub = build_lp_matrix(a, k_indices)
    b_ub = np.full(num_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def objective_normalized(a):
    """Compute 2*n*max(conv) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def line_search(a, direction, best_obj, n, max_iter=30):
    """Golden section line search for optimal step in a + t*(direction - a)."""
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    a_test = (1.0 - hi) * a + hi * direction
    a_test = np.maximum(a_test, 0.0)
    s = a_test.sum()
    if s > 1e-15:
        a_test = a_test / s
    obj_hi = objective_normalized(a_test)

    if obj_hi >= best_obj:
        for _ in range(max_iter):
            m1 = lo + gr * (hi - lo)
            m2 = lo + (1.0 - gr) * (hi - lo)

            a1 = (1.0 - m1) * a + m1 * direction
            a1 = np.maximum(a1, 0.0)
            s1 = a1.sum()
            if s1 > 1e-15:
                a1 = a1 / s1
            obj1 = objective_normalized(a1)

            a2 = (1.0 - m2) * a + m2 * direction
            a2 = np.maximum(a2, 0.0)
            s2 = a2.sum()
            if s2 > 1e-15:
                a2 = a2 / s2
            obj2 = objective_normalized(a2)

            if obj1 < obj2:
                hi = m2
            else:
                lo = m1

            if hi - lo < 1e-7:
                break

        t = (lo + hi) / 2.0
    else:
        t = 1.0

    a_new = (1.0 - t) * a + t * direction
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new, objective_normalized(a_new), t


def get_peak_indices(conv, top_frac=0.2):
    """Get indices of top convolution peaks."""
    total = len(conv)
    num_top = max(10, int(total * top_frac))
    peaks = np.argpartition(conv, -num_top)[-num_top:]
    return peaks


def peak_targeted_perturb(a, conv, rng, strength=0.02):
    """Perturb coordinates that contribute most to peak convolution values."""
    n = len(a)
    a_new = a.copy()
    k_peak = np.argmax(conv)
    i_min = max(0, k_peak - n + 1)
    i_max = min(n - 1, k_peak)
    contributors = np.arange(i_min, i_max + 1)
    if len(contributors) > 2:
        idx1 = rng.choice(contributors)
        idx2 = (k_peak - idx1)
        if 0 <= idx2 < n and idx2 != idx1:
            delta = strength * a[idx1]
            a_new[idx1] -= delta
            a_new[idx2] += delta
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
    noise = rng.standard_normal(n) * 0.003
    a_new = project_simplex(a_new + noise)
    return a_new


def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax approximation."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)

    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return grad, conv


def lp_optimize_improved(a0, time_limit, seed=0, verbose=False):
    """Improved LP-based optimization with line search and adaptive constraints."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 50
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if iter_count < 30 or no_improve > 25:
            k_indices = None
        else:
            k_indices = get_peak_indices(conv, top_frac=0.35)

        b = solve_lp_direction(a, M * 1.0, k_indices=k_indices)
        if b is None:
            b = solve_lp_direction(a, M * 1.0, k_indices=None)
            if b is None:
                b = solve_lp_direction(a, M * 1.03, k_indices=None)
                if b is None:
                    break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new, obj_new, t_opt = line_search(a, b, best_obj, n)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                a = peak_targeted_perturb(a, conv, rng, strength=0.025)
                no_improve = 0
                max_no_improve = min(120, int(max_no_improve * 1.15))

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.03,
                      momentum=0.9, seed=0, verbose=False):
    """Smooth gradient descent with adaptive parameters."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv = smooth_gradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 250:
                a = peak_targeted_perturb(a, conv, rng, strength=0.01)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = min(8000.0, beta * 1.3)
                no_improve = 0

        beta = min(10000.0, beta * 1.001)

        if verbose and no_improve % 500 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    Grad: obj={best_obj:.6f}, beta={beta:.0f}, lr={lr:.5f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc

    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            if n == len(parent):
                pp = np.maximum(parent, 0)
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(parent)), parent)
                pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps

    return inits


def multi_scale_init(n, parent, rng):
    """Create initializations using multi-scale approach."""
    inits = generate_initializations(n, parent, rng)

    n_small = max(32, n // 4)
    a_small = rng.dirichlet(np.ones(n_small) * 0.5)
    a_small_opt, _ = lp_optimize_improved(a_small, time_limit=4.0, seed=42)
    a_up = np.interp(np.linspace(0, 1, n),
                     np.linspace(0, 1, n_small), a_small_opt)
    a_up = np.maximum(a_up, 0)
    s = a_up.sum()
    if s > 1e-10:
        inits['multiscale'] = a_up / s

    a_small2, _ = lp_optimize_improved(
        np.ones(n_small) / n_small, time_limit=4.0, seed=123)
    a_up2 = np.interp(np.linspace(0, 1, n),
                      np.linspace(0, 1, n_small), a_small2)
    a_up2 = np.maximum(a_up2, 0)
    s2 = a_up2.sum()
    if s2 > 1e-10:
        inits['multiscale2'] = a_up2 / s2

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization with improved LP line search and
    peak-targeted perturbations. Explores multiple lengths and strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Exploration ==========
    print("=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()
    phase1_time = 120.0

    lengths = [128, 192, 256, 320, 384, 512]
    scan_results = []

    for li, n in enumerate(lengths):
        if time.monotonic() + 15 > t_start + phase1_time:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        priority = ['parent', 'parent_up', 'bimodal', 'uniform',
                    'random', 'gauss', 'cosine', 'ramp_inc']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:3]

        remaining = phase1_time - (time.monotonic() - t_start) - 10
        time_per_init = max(5.0, remaining / max(1, len(lengths) - li) /
                            max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() + 5 > t_start + phase1_time:
                break
            t_alloc = min(time_per_init, t_start + phase1_time - time.monotonic() - 5)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_improved(a0, time_limit=t_alloc,
                                                  seed=n + hash(name) % 1000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done at t={time.monotonic()-t_start:.1f}s. Best: {best_score:.6f}")

    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = 550.0

    deep_candidates = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results[:5]:
        key = (n, name)
        if key not in seen_keys:
            deep_candidates.append((n, a_opt.copy()))
            seen_keys.add(key)
        if len(deep_candidates) >= 4:
            break

    if best_sequence is not None:
        deep_candidates.append((len(best_sequence), best_sequence.copy()))

    if best_sequence is not None:
        best_n = len(best_sequence)
        ms_inits = multi_scale_init(best_n, parent, rng)
        for name in ['multiscale', 'multiscale2']:
            if name in ms_inits:
                deep_candidates.append((best_n, ms_inits[name].copy()))

    time_per_cand = max(30.0, phase2_time / max(1, len(deep_candidates)))

    for i, (n_cand, a_cand) in enumerate(deep_candidates):
        if time.monotonic() + 20 > phase2_start + phase2_time:
            break
        t_alloc = min(time_per_cand, phase2_start + phase2_time - time.monotonic() - 10)
        if t_alloc < 15:
            break

        a_opt, obj_opt = lp_optimize_improved(a_cand, time_limit=t_alloc,
                                              seed=i * 37 + 7)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep LP [{i}]: n={n_cand}, score={score:.6f}")

    print(f"Phase 2 done at t={time.monotonic()-t_start:.1f}s. Best: {best_score:.6f}")

    # ========== Phase 3: Gradient Fine-tuning ==========
    print("=== Phase 3: Gradient Fine-tuning ===")
    phase3_start = time.monotonic()
    phase3_time = 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        grad_configs = [
            {'beta_init': 400, 'lr_init': 0.02, 'momentum': 0.9},
            {'beta_init': 1000, 'lr_init': 0.01, 'momentum': 0.92},
            {'beta_init': 2000, 'lr_init': 0.005, 'momentum': 0.94},
            {'beta_init': 200, 'lr_init': 0.04, 'momentum': 0.85},
        ]

        time_per_grad = max(20.0, phase3_time / (len(grad_configs) + 4))

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() + 10 > phase3_start + phase3_time:
                break
            t_alloc = min(time_per_grad, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=t_alloc,
                                               seed=i * 17 + 3, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad [{i}]: score={score:.6f}")

        for i in range(4):
            if time.monotonic() + 10 > phase3_start + phase3_time:
                break
            conv_best = fast_convolve(best_sequence)
            a_pert = peak_targeted_perturb(best_sequence, conv_best, rng, 0.005)
            t_alloc = min(time_per_grad / 2, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a_opt, obj_opt = gradient_optimize(a_pert, time_limit=t_alloc,
                                               beta_init=1500, lr_init=0.01,
                                               momentum=0.93, seed=200 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad pert [{i}]: score={score:.6f}")

    print(f"Phase 3 done at t={time.monotonic()-t_start:.1f}s. Best: {best_score:.6f}")

    # ========== Phase 4: Hybrid Alternating ==========
    print("=== Phase 4: Hybrid Alternating ===")
    phase4_start = time.monotonic()
    phase4_time = 320.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence.copy()
        best_in_phase = a.copy()
        best_obj_phase = objective_normalized(a)
        v = np.zeros(n_best)

        phase4_t0 = time.monotonic()
        no_improve_h = 0
        max_no_improve_h = 70

        while time.monotonic() - phase4_t0 < phase4_time:
            conv = fast_convolve(a)
            M = conv.max()

            k_idx = get_peak_indices(conv, top_frac=0.4)
            b = solve_lp_direction(a, M, k_indices=k_idx)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    a_curr, obj_curr, t_opt = line_search(a, b, best_obj_phase, n_best)
                    if obj_curr < best_obj_phase - 1e-13:
                        a = a_curr
                        best_obj_phase = obj_curr
                        best_in_phase = a.copy()
                        no_improve_h = 0
                    else:
                        no_improve_h += 1
            else:
                no_improve_h += 1

            grad, conv = smooth_gradient(a, 800.0)
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
            v = 0.9 * v - 0.012 * grad
            a_curr = project_simplex(a + v)
            obj_curr = objective_normalized(a_curr)
            if obj_curr < best_obj_phase - 1e-14:
                a = a_curr
                best_obj_phase = obj_curr
                best_in_phase = a.copy()
                no_improve_h = 0
            else:
                no_improve_h += 1

            if no_improve_h > max_no_improve_h:
                a = peak_targeted_perturb(a, conv, rng, 0.02)
                v = np.zeros(n_best)
                no_improve_h = 0
                max_no_improve_h = min(180, int(max_no_improve_h * 1.15))

        score = float(evaluate_sequence(best_in_phase.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = best_in_phase.copy()
            print(f"  Hybrid: score={score:.6f}")

    print(f"Phase 4 done at t={time.monotonic()-t_start:.1f}s. Best: {best_score:.6f}")

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===")
    phase5_start = time.monotonic()
    phase5_time = max(10.0, deadline - time.monotonic())

    if best_sequence is not None:
        n_best = len(best_sequence)

        t_alloc = min(phase5_time * 0.3, 70.0)
        if t_alloc > 10:
            a_opt, obj_opt = lp_optimize_improved(best_sequence, time_limit=t_alloc,
                                                  seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.6f}")

        t_alloc = min(phase5_time * 0.25, 50.0)
        if t_alloc > 8:
            a_opt, obj_opt = gradient_optimize(
                best_sequence, time_limit=t_alloc,
                beta_init=4000, lr_init=0.003, momentum=0.95, seed=777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish Grad: score={score:.6f}")

        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            n_tries = min(int(remaining / 3), 30)
            conv_best = fast_convolve(best_sequence)
            for i in range(n_tries):
                if time.monotonic() > deadline - 20:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    a_pert = peak_targeted_perturb(best_sequence, conv_best, rng, scale)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        conv_best = fast_convolve(best_sequence)
                        print(f"  Tiny perturb [{i}] s={scale}: score={best_score:.6f}")
                        break

    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
