"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive softmax-subgradient
descent with LP-based direction correction, coordinate descent polishing,
and multi-length search optimized for the n~512 regime.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_sg(a, conv, top_k=20, temperature=None):
    """Compute softmax-weighted subgradient from top-k lags."""
    n = len(a)
    if temperature is None:
        pmax = conv.max()
        temperature = pmax * 0.1
    
    # Get top-k lags
    top_idx = np.argpartition(conv, -top_k)[-top_k:]
    top_vals = conv[top_idx]
    
    # Softmax weights
    scaled = top_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(top_idx))
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= w_sum
    
    # Compute weighted subgradient
    sg = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-14:
            continue
        k = int(top_idx[i])
        wi = 2.0 * w[i]
        # (f_k)_i = 2 * a[k-i] for 0 <= k-i <= n-1
        # So sg_i += wi * a[k-i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            sg[j] += wi * a[k - j]
    
    return sg


def softmax_sg_fft(a, conv, top_k=20, temperature=None):
    """Faster softmax-weighted subgradient using FFT-based convolution."""
    n = len(a)
    if temperature is None:
        pmax = conv.max()
        temperature = pmax * 0.1
    
    top_idx = np.argpartition(conv, -top_k)[-top_k:]
    top_vals = conv[top_idx]
    
    scaled = top_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(top_idx))
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= w_sum
    
    # The subgradient for lag k is 2*a[k-j] (reversed and shifted)
    # Weighted sum: sg = 2 * sum_k w_k * (reversed a, shifted by k)
    # This is equivalent to: sg = 2 * (w_padded * reversed_a) where * is cross-correlation
    # Let's compute it as: sg_j = 2 * sum_k w_k * a[k-j] (for valid k-j)
    # = 2 * sum_k w_k * a[k-j]
    # Let b = reversed a: b_i = a[n-1-i]
    # Then a[k-j] = b[n-1-(k-j)] = b[n-1-k+j]
    # sg_j = 2 * sum_k w_k * b[n-1-k+j]
    
    # This is a cross-correlation of w (on lags 0..2n-2) with b (on 0..n-1)
    # Let's just do it directly for clarity (top_k is small)
    sg = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-14:
            continue
        k = int(top_idx[i])
        wi = 2.0 * w[i]
        # sg[j] += wi * a[k-j] for j where 0 <= k-j <= n-1
        # i.e., k-(n-1) <= j <= k, and 0 <= j <= n-1
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # a[k-j] for j in [j_lo, j_hi] is a[k-j_hi], a[k-j_hi+1], ..., a[k-j_lo]
            # which is a[j_lo'], a[j_lo'+1], ..., a[j_hi'] where j' = k-j
            # So it's a[k-j_hi : k-j_lo+1] reversed... 
            # Actually a[k-j] for j=j_lo, j_lo+1, ..., j_hi gives:
            # a[k-j_lo], a[k-j_lo-1], ..., a[k-j_hi]
            # = a[j_hi'], a[j_hi'-1], ..., a[j_lo'] where j' ranges from k-j_lo down to k-j_hi
            # Let me just index directly:
            indices = np.arange(j_lo, j_hi + 1)
            sg[indices] += wi * a[k - indices]
    
    return sg


def subgrad_descent(a0, time_limit, top_k=30, temp_frac=0.05, 
                    lr_init=0.01, momentum=0.9, decay=0.9995,
                    restart_every=300, noise_scale=0.003):
    """Subgradient descent with momentum and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pmax = conv.max()
        if pmax < 1e-15:
            break
        
        temp = pmax * temp_frac
        sg = softmax_sg(a, conv, top_k, temp)
        
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        lr *= decay
        iter_count += 1
        
        # Restart if stuck
        if no_improve >= restart_every:
            rng = np.random.default_rng(iter_count)
            a = project_simplex(a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.002, lr * 1.5)
    
    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=50, delta_frac=0.1):
    """Coordinate descent: move mass between pairs of coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            if a[i] < 1e-8:
                continue
            for j in range(n):
                if j == i:
                    continue
                # Try moving delta from i to j
                delta = a[i] * delta_frac
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                if a_test[i] < -1e-12:
                    continue
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        if not improved:
            break
    
    return a


def build_lp_matrix(n):
    """Build the constraint matrix for the LP."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = A[k, j]  # We'll fill this based on f
    return A


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end+1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None
