def fast_convolve(a):

    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def golden_section_max(f, lo, hi, tol=1e-8, max_iter=100):
    """Minimize a univariate function using golden section search."""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = hi - (hi - lo) / gr
    d = lo + (hi - lo) / gr
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - (hi - lo) / gr
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + (hi - lo) / gr
            fd = f(d)
    return 0.5 * (lo + hi)


def line_search_step(conv_a, conv_ab, conv_b, n):
    """Find optimal step size t that minimizes max conv of (1-t)*a + t*b."""
    def f(t):
        val = (1.0 - t) ** 2 * conv_a + 2.0 * t * (1.0 - t) * conv_ab + t ** 2 * conv_b
        return val.max()
    
    t = golden_section_max(f, 0.01, 1.0, tol=1e-7, max_iter=60)
    return t


def solve_lp_min_peak(conv_matrix, n):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv_matrix @ b <= M' * ones."""
    num_rows = conv_matrix.shape[0]
    # Variables: b_0,...,b_{n-1}, M'
    # c = [0,...,0, 1] (minimize M')
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # A_ub: [conv_matrix, -1] @ [b, M'] <= 0
    A_ub = np.hstack([conv_matrix, -np.ones((num_rows, 1))])
    b_ub = np.zeros(num_rows)
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def solve_lp_max_sum(conv_matrix, M, n):
    """Solve: max sum(b) s.t. b>=0, conv_matrix @ b <= M * ones."""

    res = linprog(c, A_ub=conv_matrix, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        temp *= 0.999

        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            lr = max(lr_init * 0.2, lr * 0.5)

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            a /= a_sum
            inits[f'cos_{freq}'] = a

    sigma = 0.22 * n
    g = np.exp(-0.5 * ((x - 0.5 * n) / sigma) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Two-peak
    a2p = np.zeros(n)
    a2p[:n//4] = 1.0
    a2p[3*n//4:] = 1.0
    a2p /= a2p.sum()
    inits['two_peak'] = a2p

    # Incremental (secp-like)
    inc = np.arange(1, n+1, dtype=np.float64)
    inc /= inc.sum()
    inits['ramp'] = inc

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: LP exploration -> deep refinement -> subgradient polish.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")
        return score

    # === Phase 1: LP Exploration ===
    print("=== Phase 1: LP Exploration ===")
    t1_start = time.monotonic()

    n_main = 512
    lengths = [n_main]
    if parent is not None and len(parent) != n_main:
        lengths.append(len(parent))
    lengths.append(1024)
    lengths = [L for L in lengths if 64 <= L <= 2048]
    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths

    for n in lengths:
        remain_p1 = deadline - time.monotonic() - 800
        if remain_p1 < 60:
            break

        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'uniform', 'cos_1', 'random', 'gauss', 'two_peak']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits.keys() if nm not in priority]
        init_names = init_names[:5]

        time_per_init = max(20.0, remain_p1 / (len(lengths) * len(init_names)))

        for name in init_names:
            remain = deadline - time.monotonic() - 700
            if remain < 15:
                break
            t_alloc = min(time_per_init, remain)
            if t_alloc < 10:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, alloc={t_alloc:.0f}s")
            a_opt, _ = optimize_lp_linesearch(a0, time_limit=t_alloc,
                                              max_iters=1500, seed=hash(name) % (2**31))
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: Deep LP Refinement ===
    print("=== Phase 2: Deep LP Refinement ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 400
        n_best = len(best_sequence)

        for trial in range(4):
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(120.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break

            if trial == 0:
                start = best_sequence.copy()
            else:
                noise = rng.standard_normal(n_best) * (0.001 + 0.002 * trial)
                start = project_simplex(best_sequence + noise)

            print(f"  Trial {trial}, alloc={t_alloc:.0f}s")
            a_opt, _ = optimize_lp_linesearch(start, time_limit=t_alloc,
                                              max_iters=2000, seed=100 + trial)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Subgradient Polish ===
    print("=== Phase 3: Subgradient Polish ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 100
        n_best = len(best_sequence)

        for lr in [0.02, 0.01, 0.005]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(60.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            print(f"  Subgrad lr={lr}, alloc={t_alloc:.0f}s")
            a_opt, _ = optimize_subgrad(best_sequence, time_limit=t_alloc,
                                        lr_init=lr, momentum=0.91,
                                        top_k=max(20, n_best // 15),
                                        temp_init=0.0008, seed=200)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Final LP micro-refinement ===
    print("=== Phase 4: Final Micro-Refinement ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 10
        t_alloc = max(10.0, phase4_deadline - time.monotonic())
        if t_alloc > 5:
            a_opt, _ = optimize_lp_linesearch(best_sequence, time_limit=t_alloc,
                                              max_iters=500, seed=999)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    # Final verification
    final_a = np.maximum(best_sequence, 0.0)
    final_score = float(evaluate_sequence(final_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = final_a

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
