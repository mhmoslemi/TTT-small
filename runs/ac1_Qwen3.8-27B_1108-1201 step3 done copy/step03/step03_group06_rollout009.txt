def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines fast subgradient descent with backtracking, LP-based refinement,
spectral shaping heuristics, and adaptive restarts.
Uses exact max subgradients (not log-sum-exp) for speed, with momentum
and periodic perturbation to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=1):
    """Compute subgradient of max_k conv[k] w.r.t. a.
    For top_k=1: subgradient at the argmax.
    For top_k>1: average subgradient over top-k peaks.
    """
    n = len(a)
    size = 2 * n - 1
    # Get top-k peaks
    if top_k >= size:
        peak_idx = np.arange(size)
        peak_vals = conv
    else:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]
    
    # Weight peaks (more weight to higher peaks)
    weights = peak_vals ** 4  # Sharpen: focus on the very top
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(size)
        weights[np.argmax(conv)] = 1.0
    else:
        weights = weights / w_sum
    
    # Place weights in full array
    w_full = np.zeros(size)
    w_full[peak_idx] = weights
    
    # Compute gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # Use FFT for efficiency
    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:size] = w_full
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    
    # conv term: sum_k w[k] * a[i-k] = (w * a)[i]
    conv_term = np.fft.ifft(W * A).real[:n]
    # cross term: sum_k w[k] * a[k-i] = cross-correlation
    cross_term = np.fft.ifft(A * np.conj(W)).real[:n]
    
    grad = conv_term + cross_term
    return grad


def smoothed_gradient(a, beta=100.0):
    """Smooth gradient using log-sum-exp with parameter beta."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    
    conv_term = np.fft.ifft(W * A).real[:n]
    cross_term = np.fft.ifft(A * np.conj(W)).real[:n]
    
    grad = conv_term + cross_term
    return grad, conv

def build_lp_matrix_sparse(a, peak_indices):
    """Build LP constraint matrix for specific peak indices only."""
    n = len(a)
    num_k = len(peak_indices)
    A_ub = np.zeros((num_k, n))
    for i, k in enumerate(peak_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[i, idx] = a[k - idx]
    return A_ub

def spectral_shaping(a0, num_iters=100, alpha=0.5):
    """Spectral shaping: try to flatten |A(f)|^2 in frequency domain."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Zero-pad to next power of 2 >= 2n-1
    N = 1 << (2 * n - 1).bit_length()
    
    for _ in range(num_iters):
        A = np.fft.fft(a, N)
        C = A * A  # This gives the DFT of the auto-convolution
        
        # Target: flat C (corresponds to flat auto-convolution in time)
        # C_target is the DFT of a flat sequence of length 2n-1
        c_target = np.zeros(N)
        c_target[:2*n-1] = 1.0 / (2*n-1)
        C_target = np.fft.fft(c_target)
        
        # Update A to move C toward C_target
        # A_new^2 = C + alpha * (C_target - C) = (1-alpha)*C + alpha*C_target
        C_new = (1 - alpha) * C + alpha * C_target
        
        # A_new = sqrt(C_new) (preserve phase of A)
        phase_A = np.angle(A)
        mag_C_new = np.abs(C_new)
        mag_A_new = np.sqrt(mag_C_new)
        
        A_new = mag_A_new * np.exp(1j * phase_A)
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            break
        a = a_new
    
    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    return a, obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    # Triangular
    tri_edge = np.minimum(x, n - 1 - x) + 1.0
    tri_edge /= tri_edge.sum()
    inits['tri_edge'] = tri_edge
    
    tri_center = (n - 1 - np.minimum(x, n - 1 - x)).astype(np.float64) + 1.0
    tri_center /= tri_center.sum()
    inits['tri_center'] = tri_center
    
    # Gaussians
    for sf in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Sine-like
    inits['sine'] = (np.sin(np.pi * x / n) + 0.1)
    inits['sine'] = np.maximum(inits['sine'], 0.01)
    inits['sine'] /= inits['sine'].sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

            if t_alloc < 15:
                break
            
            beta_opts = [50, 200, 500, 1000]
            a_opt, obj_opt = smoothed_optimize(best_sequence, time_limit=t_alloc,
                                               beta_init=beta_opts[i], lr_init=0.02,
                                               momentum=0.92, seed=i * 23 + 7)
            try_sequence(a_opt, f"P3 smooth[{i}] beta={beta_opts[i]}")
        
        # LP refinement
        lp_time = phase3_time * 0.4
        time_per_lp = lp_time / 3
        for i in range(3):
            if time.monotonic() - t3 > phase3_time - 30:
                break
            t_alloc = min(time_per_lp, phase3_time - (time.monotonic() - t3) - 30)
            if t_alloc < 20:
                break
            
            top_k_opts = [20, 50, 100]
            step_opts = [0.5, 0.7, 1.0]
            a_opt, obj_opt = lp_optimize_fast(best_sequence, time_limit=t_alloc,
                                              top_k=top_k_opts[i], step_init=step_opts[i],
                                              verbose=False, seed=i * 31 + 11)
            try_sequence(a_opt, f"P3 LP[{i}] k={top_k_opts[i]}")
    
    # ========== Phase 4: Alternating refinement (400s) ==========
    print("=== Phase 4: Alternating Refinement ===")
    t4 = time.monotonic()
    phase4_time = max(30.0, deadline - time.monotonic() - 100.0)
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence.copy()
        
        # Alternate subgradient and LP steps
        alt_t0 = time.monotonic()
        mode = 0  # 0 = subgrad, 1 = LP
        sg_lr = 0.02
        v = np.zeros(n_best)
        no_improve = 0
        phase4_best_obj = 2.0 * n_best * fast_convolve(a).max()
        phase4_best_a = a.copy()
        
        while time.monotonic() - alt_t0 < phase4_time - 50:
            conv = fast_convolve(a)
            current_obj = 2.0 * n_best * conv.max()
            
            if current_obj < phase4_best_obj - 1e-13:
                phase4_best_obj = current_obj
                phase4_best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 100:
                noise = rng.standard_normal(n_best) * 0.003
                a = project_simplex(a + noise)
                v = np.zeros(n_best)
                sg_lr = 0.02
                no_improve = 0
                continue
            
            if mode == 0:
                # Subgradient step
                top_k_use = 5 if current_obj < 1.505 else 3
                grad = compute_subgradient(a, conv, top_k=top_k_use)
                g_norm = np.linalg.norm(grad)
                if g_norm > 1e-15:
                    grad /= g_norm
                    v = 0.9 * v - sg_lr * grad
                    a = project_simplex(a + v)
                mode = 1
            else:
                # LP step (only use top 20 constraints for speed)
                M = conv.max()
                peaks = np.argpartition(conv, -20)[-20:]
                A_ub = build_lp_matrix_sparse(a, peaks)
                b_ub = np.full(20, M)
                c = -np.ones(n_best)
                bounds = [(0, None)] * n_best
                try:
                    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True})
                    if res.success:
                        b = res.x
                        b_sum = b.sum()
                        if b_sum > 1e-15:
                            b /= b_sum
                            a = 0.7 * a + 0.3 * b
                            a = np.maximum(a, 0.0)
                            s = a.sum()
                            if s > 1e-15:
                                a /= s
                except Exception:
                    pass
                mode = 0
        
        try_sequence(phase4_best_a, f"P4 alt, obj={phase4_best_obj:.6f}")
    
    # ========== Phase 5: Final polish (100s) ==========
    print("=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    phase5_time = max(30.0, deadline - time.monotonic() - 20.0)
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        
        # Very fine subgradient
        t_alloc = phase5_time * 0.4
        if t_alloc > 10:
            a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                                  lr_init=0.01, momentum=0.95, top_k=5,
                                                  seed=999)
            try_sequence(a_opt, "P5 fine-sg")
        
        # Fine smoothed
        t_alloc = phase5_time * 0.3
        if t_alloc > 10:
            a_opt, obj_opt = smoothed_optimize(best_sequence, time_limit=t_alloc,
                                               beta_init=2000, lr_init=0.005,
                                               momentum=0.95, seed=777)
            try_sequence(a_opt, "P5 fine-smooth")
        
        # Tiny perturbations
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            n_tries = min(int(remaining / 3), 30)
            for i in range(n_tries):
                if time.monotonic() > deadline - 15:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    s = a_pert.sum()
                    if s > 0:
                        a_pert /= s
                        score_pert = float(evaluate_sequence(a_pert.tolist()))
                        if np.isfinite(score_pert) and score_pert < best_score:
                            best_score = score_pert
                            best_sequence = a_pert.copy()
                            print(f"  Perturb [{i}] s={scale}: {score_pert:.6f}")
                            break
    
    # Final check
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
