"""
Hybrid optimization for minimizing 2*n*max(autocorr(a))/(sum(a))^2 using
coordinate descent, spectral flattening, subgradient methods, and LP-based
direction finding. Multi-phase search with diverse initializations and
adaptive restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(A * A, fft_size)[:size]
    return corr

def compute_obj(a, corr=None):
    """Compute objective value 2*n*max(corr)/(sum(a))^2."""
    n = len(a)
    if corr is None:
        corr = fast_autocorr(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * corr.max() / (S * S)

def coord_pass(a, max_passes=1, order='random', seed=0):
    """Execute coordinate descent passes. Returns updated a and best obj."""
    n = len(a)
    S = a.sum()
    corr = fast_autocorr(a)
    best_obj = compute_obj(a, corr)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    
    for p in range(max_passes):
        if order == 'random':
            order_idx = rng.permutation(n)
        else:
            order_idx = np.arange(n)
        
        for j in order_idx:
            t0 = a[j]
            if t0 < 1e-15:
                # Try setting to a small positive value
                candidates_u = [1.0 / n]  # u = t - t0 = 1/n - 0
            else:
                # b[k] = d corr[k] / d a[j] = a[k-j] if 0 <= k-j < n
                idx = np.arange(2 * n - 1) - j
                mask = (idx >= 0) & (idx < n)
                b = np.zeros(2 * n - 1)
                b[mask] = a[idx[mask]]
                
                # Candidate u values
                candidates_u = [-t0, -0.5 * t0, -0.25 * t0, 0.0]
                # Also try some positive
                candidates_u.extend([0.25 * t0, 0.5 * t0])
                
                # Add breakpoints from current peak
                k_star = np.argmax(corr)
                if b[k_star] > 1e-15:
                    # Find u where corr[k_star] + u*b[k_star] = corr[k] + u*b[k]
                    for k in range(2 * n - 1):
                        if k == k_star:
                            continue
                        db = b[k_star] - b[k]
                        if abs(db) > 1e-15:
                            u_cand = (corr[k] - corr[k_star]) / db
                            if -t0 <= u_cand <= 2 * t0:
                                candidates_u.append(u_cand)
                
                # Limit candidates
                candidates_u = list(set([round(u, 10) for u in candidates_u]))[:30]
            
            # Evaluate candidates
            best_u = 0.0
            best_local_obj = best_obj
            
            for u in candidates_u:
                if u < -t0 - 1e-12:
                    continue
                t_new = t0 + u
                if t_new < 0:
                    continue
                S_new = S + u
                if S_new < 0.01:
                    continue
                corr_new = corr + u * b if t0 > 0 else fast_autocorr(np.where(np.arange(n) == j, t_new, a))
                obj_candidate = 2.0 * n * corr_new.max() / (S_new * S_new)
                if obj_candidate < best_local_obj - 1e-14:
                    best_local_obj = obj_candidate
                    best_u = u
            
            # Apply update
            if best_u != 0.0:
                a[j] = max(0.0, t0 + best_u)
                S += best_u
                corr = corr + best_u * b if t0 > 0 else fast_autocorr(a)
                if best_local_obj < best_obj - 1e-13:
                    best_obj = best_local_obj
                    best_a = a.copy()
    
    return best_a, best_obj

def spectral_flatten_step(a, beta=0.5):
    """One step of spectral flattening. Returns new a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Compress dynamic range
    max_mag = mag.max()
    if max_mag < 1e-15:
        return a
    # new_mag = mag / (1 + beta * mag / max_mag)
    # Or: new_mag = mag ** (1/(1+beta))
    new_mag = mag / (1.0 + beta * (mag / max_mag))
    
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / (a_new.sum() / 1.0)  # keep sum as is for now
        # Actually, let's normalize to sum=1
        a_new = a_new / a_new.sum()
    else:
        a_new = a.copy()
    return a_new

def subgradient_step(a, lr, momentum_vec, momentum, top_k=20):
    """One subgradient step."""
    n = len(a)
    corr = fast_autocorr(a)
    
    if top_k >= len(corr):
        peaks = np.arange(len(corr))
    else:
        peaks = np.argpartition(corr, -top_k)[-top_k:]
    
    peak_vals = corr[peaks]
    temp = 0.002 * peak_vals.max() + 1e-15
    scaled = (peak_vals - peak_vals.max()) / temp
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for i, k in enumerate(peaks):
        if w[i] < 1e-10:
            continue
        k = int(k)
        idx = np.arange(max(0, k - n + 1), min(n, k + 1))
        other = k - idx
        grad[idx] += w[i] * a[other]
    
    momentum_vec = momentum * momentum_vec - lr * grad
    a_new = a + momentum_vec
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new, momentum_vec

def lp_optimize(a0, time_limit=30.0):
    """LP-based optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    corr = fast_autocorr(a)
    M = corr.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Build constraint matrix
        C = np.zeros((2 * n - 1, n))
        for i in range(n):
            C[i:i + n, i] = a
        
        # Solve LP: max sum(g) s.t. C@g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.001)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                           options={'time_limit': 5, 'presolve': True})
            if not result.success:
                no_improve += 1
                if no_improve > 20:
                    break
                continue
        except Exception:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        g = result.x
        g_sum = g.sum()
        if g_sum < 1.0 + 1e-8:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        d = g / g_sum - a
        
        # Line search
        best_t = 0.0
        best_obj_try = best_obj
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.7, 0.8]:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try - 1e-13:
                best_obj_try = obj_try
                best_t = t
        
        if best_t > 0:
            a_new = a + best_t * d
            a_new = np.maximum(a_new, 0.0)
            a = a_new / a_new.sum()
            corr = fast_autocorr(a)
            M = corr.max()
            best_obj = 2.0 * n * M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits.append(tri / tri.sum())
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(bi / bi.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Sine
    s = np.sin(np.pi * (x + 0.5) / n) ** 2 + 0.05
    inits.append(s / s.sum())
    
    # Power law
    pw = (x + 1.0) ** 0.3
    inits.append(pw / pw.sum())
    
    # Smooth random (low-pass filtered white noise)
    for _ in range(3):
        r = rng.random(n)
        # Low-pass filter
        R = np.fft.rfft(r)
        freqs = np.fft.rfftfreq(n)
        cutoff = 0.1
        R[freqs > cutoff] *= 0.1
        r_filtered = np.fft.irfft(R, n)
        r_filtered = np.abs(r_filtered)
        if r_filtered.sum() > 1e-10:
            inits.append(r_filtered / r_filtered.sum())
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits.append(p / ps)
                # Perturbed parent
                for scale in [0.01, 0.05, 0.1]:
                    noise = rng.standard_normal(n) * scale
                    pp = p / ps + noise
                    pp = np.maximum(pp, 0)
                    pps = pp.sum()
                    if pps > 1e-10:
                        inits.append(pp / pps)
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits.append(pp / pps)
    
    return inits
