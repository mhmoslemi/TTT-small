import numpy as np

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 8.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive alpha increase
        if no_improve > 300:
            alpha_target = min(alpha * 1.3, alpha_end)
            no_improve = 0
            lr = lr * 0.8
        
        # Gradually increase alpha toward target
        alpha += alpha_step * (alpha_target - alpha)
        if abs(alpha - alpha_target) < 0.1:
            alpha = alpha_target
            # Try to push alpha higher if stuck
            if no_improve > 100:
                alpha_target = min(alpha * 1.5, alpha_end)
        
        lr *= lr_decay
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth it={it}, obj={best_obj:.6f}, alpha={alpha:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def fpw_lp_optimize(a0, time_limit, verbose=False, verbose_every=200):
    """Frank-Wolfe iterations with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = compute_conv(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit:
        b = lp_oracle(a, M)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_local = local_objective(a)
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidate = (1 - t) * a + t * b
                obj_c = local_objective(candidate)
                if obj_c < best_local:
                    best_local = obj_c
                    best_t = t
            
            if best_t < 0.05:
                no_improve += 1
            else:
                a = (1 - best_t) * a + best_t * b
                conv = compute_conv(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        # Stuck handling
        if no_improve >= max_stall:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = compute_conv(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.004, momentum=0.93):
    """Multi-peak subgradient polishing with momentum on simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = compute_conv(a)
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.04, 1e-15)
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]
        
        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
        
        cur_obj = local_objective(a)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= 0.9997
        it += 1
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    inits['ramp_rev'] = r[::-1].copy()
    
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()
    
    # Edge-concentrated
    edge = max(1, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # Bimodal
    bm = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    bm /= bm.sum()
    inits['bimodal'] = bm.copy()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.6] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()
    
    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cw /= cw.sum()
    inits['cosine'] = cw.copy()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining smoothed gradient optimization, LP Frank-Wolfe,
    and subgradient polishing to minimize the peak autocorrelation objective.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    # === Phase 1: Smoothed optimization scan ===
    print("=== Phase 1: Smoothed Gradient Scan ===")
    lengths = [256, 320, 384, 448, 512]
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.35
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = generate_inits(n, parent, rng)
        priority = ['parent', 'parent_interp', 'uniform', 'bimodal', 'cosine',
                    'edge', 'triangle', 'gauss', 'random', 'sparse', 'ramp', 'ramp_rev']
        init_names = sorted(inits.keys(), key=lambda x: priority.index(x) if x in priority else 99)
        
        time_per_init = max(5.0, 80.0 / len(init_names))
        
        for name in init_names[:7]:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(time_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 3:
                break
            
            a0 = inits[name]
            print(f"  n={n}, init={name}, t={t_alloc:.0f}s")
            a_opt, obj_opt = smoothed_optimize(a0, time_limit=t_alloc,
                                                alpha_start=3.0, alpha_end=150.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    print(f"After Phase 1: best={best_score:.6f}, n={len(best_sequence) if best_sequence is not None else 0}")
    
    # === Phase 2: LP Frank-Wolfe on best ===
    if best_sequence is not None:
        print("=== Phase 2: LP Frank-Wolfe ===")
        remaining = deadline - time.monotonic() - 30.0
        t_deep = remaining * 0.30
        if t_deep > 10:
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_deep,
                                             verbose=True, verbose_every=300)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 3: High-alpha smoothed polishing ===
    if best_sequence is not None:
        print("=== Phase 3: High-Alpha Polish ===")
        remaining = deadline - time.monotonic() - 30.0
        t_polish = remaining * 0.25
        if t_polish > 10:
            a_opt, obj_opt = smoothed_optimize(best_sequence, time_limit=t_polish,
                                                alpha_start=50.0, alpha_end=500.0,
                                                lr_decay=0.9999,
                                                verbose=True, verbose_every=500)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 4: Subgradient polishing ===
    if best_sequence is not None:
        print("=== Phase 4: Subgradient Polish ===")
        remaining = deadline - time.monotonic() - 25.0
        if remaining > 10:
            t_sg = remaining * 0.40
            a_opt, obj_opt = subgradient_polish(best_sequence, time_limit=t_sg,
                                                 lr_init=0.003, momentum=0.94)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 5: Final LP pass ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 5: Final LP ===")
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 5:
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=remaining,
                                             verbose=True, verbose_every=300)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 5 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # === Phase 6: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("=== Phase 6: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0
        n_restarts = max(1, int(remaining // 20))
        n_restarts = min(n_restarts, 6)
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.0005 * (1 + i * 0.5)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(15.0, deadline - time.monotonic() - 8)
            if t_alloc < 3:
                break
            
            # Alternate between smoothed and FPW
            if i % 2 == 0:
                a_opt, obj_opt = smoothed_optimize(a_pert, time_limit=t_alloc,
                                                    alpha_start=30.0, alpha_end=300.0)
            else:
                a_opt, obj_opt = fpw_lp_optimize(a_pert, time_limit=t_alloc)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
