def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_smooth_conv(a, w):
    n = len(a)
    N = 1 << (len(w) + n - 1 + 7).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def theoretical_init(n, rng):
    """Initialize from ideal flat auto-convolution frequency response."""
    N_conv = 2 * n - 1
    fft_size = 1 << (2 * N_conv).bit_length()

    m = np.arange(fft_size)
    A_target = np.zeros(fft_size, dtype=complex)
    A_target[0] = 1.0

    for m_idx in range(1, fft_size // 2 + 1):
        num = np.sin(np.pi * N_conv * m_idx / fft_size)
        den = np.sin(np.pi * m_idx / fft_size)
        if abs(den) < 1e-15:
            c_val = 0.0
        else:
            phase = np.exp(-1j * np.pi * (N_conv - 1) * m_idx / fft_size)
            c_val = (phase * num / den) / N_conv

        if abs(c_val) < 1e-15:
            A_target[m_idx] = 0
        else:
            # Take square root with phase alignment
            A_val = np.sqrt(c_val)
            ideal_phase = -np.pi * (N_conv - 1) * m_idx / (2.0 * fft_size)
            actual_phase = np.angle(A_val)
            if abs(np.angle(A_val * A_val) - ideal_phase) > np.pi / 2:
                A_val = -A_val
            A_target[m_idx] = A_val

        if m_idx < fft_size - m_idx:
            A_target[fft_size - m_idx] = np.conj(A_target[m_idx])

    a = np.real(ifft(A_target, fft_size))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def exponential_init(n, rate, rng):
    """Exponential growth/decay initialization."""
    a = np.exp(rate * np.arange(n) / n)
    a += rng.standard_normal(n) * 0.02
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def smooth_bump_init(n, center_frac, width, rng):
    """Gaussian bump at specified center."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    a = np.exp(-0.5 * ((x - center) / (width * n)) ** 2)
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def cos_tapered_init(n, rng):
    """Raised cosine taper (asymmetric)."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(np.pi * x / (n - 1))) ** 0.5
    a = a * np.exp(-0.3 * x / n)  # Add asymmetry
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def flat_spectrum_init(n, rng, n_trials=30):
    """Random phase flat power spectrum initialization."""
    best_a = np.ones(n) / n
    best_obj = float('inf')
    fft_size = 1 << (2 * n - 1).bit_length()

    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1])

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['theo'] = theoretical_init(n, rng)
    inits['exp_pos'] = exponential_init(n, 1.5, rng)
    inits['exp_neg'] = exponential_init(n, -1.5, rng)
    inits['exp_strong'] = exponential_init(n, 3.0, rng)
    inits['bump_left'] = smooth_bump_init(n, 0.3, 0.3, rng)
    inits['bump_right'] = smooth_bump_init(n, 0.7, 0.3, rng)
    inits['bump_center'] = smooth_bump_init(n, 0.5, 0.4, rng)
    inits['cos_taper'] = cos_tapered_init(n, rng)
    inits['flat_spec'] = flat_spectrum_init(n, rng, n_trials=20)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Power law variations
    for p in [0.5, 1.0, 2.0]:
        a = np.power(np.arange(1, n + 1), p)
        a /= a.sum()
        a += rng.standard_normal(n) * 0.01
        a = np.maximum(a, 0)
        a /= a.sum()
        inits['pow_%s' % str(p)] = a

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            elif len(parent) > n:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0)
                is_sum = interp.sum()
                if is_sum > 1e-10:
                    inits['parent_ds'] = interp / is_sum
            else:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                interp = np.maximum(interp, 0)
                is_sum = interp.sum()
                if is_sum > 1e-10:
                    inits['parent_up'] = interp / is_sum

    return inits


def obj_fun(x, n, beta):
    """Objective for L-BFGS-B (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def jac_fun(x, n, beta):
    """Jacobian for L-BFGS-B (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B optimization with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)

    result = minimize(obj_fun, x0, jac=jac_fun, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def optimize_pg(a0, beta, lr=0.05, n_iter=300):
    """Projected gradient descent with smooth max."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        grad_a = compute_grad_smooth_conv(a, w)
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-15:
            break

        t = lr
        for _ in range(12):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
        a = a_new
        lr *= 0.999

    return best_a, best_obj


def subgradient_polish(a, n_steps=200, lr=0.01):
    """Direct subgradient method targeting the actual max coefficient."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for it in range(n_steps):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        # Gradient of conv[max_k] w.r.t. a
        # conv[max_k] = sum_j a[j] * a[max_k - j] (valid indices)
        # d(conv[max_k])/d(a[i]) = a[max_k-i] + a[i-max_k] (if valid)
        grad = np.zeros(n)
        for i in range(n):
            j1 = max_k - i
            j2 = i - max_k
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n and j2 != j1:
                grad[i] += a[j2]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Step and project
        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - lr * 0.5 * grad)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

        lr *= 0.995

    return best_a, best_obj


def multi_peak_polish(a, n_rounds=50):
    """Iteratively target the top few peaks in the convolution."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for rnd in range(n_rounds):
        conv = fast_convolve(a)
        # Find top 3 peaks
        top_k = np.argsort(conv)[::-1][:3]

        # Combine gradients from top peaks
        combined_grad = np.zeros(n)
        for k in top_k:
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    combined_grad[i] += a[j1]
                if 0 <= j2 < n and j2 != j1:
                    combined_grad[i] += a[j2]

        gnorm = np.linalg.norm(combined_grad)
        if gnorm < 1e-15:
            break
        combined_grad /= gnorm

        # Backtracking line search
        lr = 0.02
        for _ in range(8):
            a_new = project_simplex(a - lr * combined_grad)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break
            lr *= 0.5

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search procedure: multi-phase optimization to minimize 2n*max(conv)/(sum)^2."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_norm.copy()
            print("  New best: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')

    # Phase 1: Quick scan over lengths and initializations
    print("=== Phase 1: Quick Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_deadline = t_start + 400.0
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break
        inits = generate_initializations(n, parent, rng)

        priority = ['parent', 'parent_ds', 'parent_up', 'theo', 'exp_pos', 'exp_neg',
                    'exp_strong', 'bump_left', 'bump_right', 'flat_spec', 'uniform',
                    'cos_taper', 'pow_1.0', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in inits if nm not in priority]
        init_names = init_names[:6]

        time_per_init = max(8.0, 350.0 / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 30:
                break
            t_alloc = min(time_per_init, phase1_deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break

            a0 = inits[name]
            # Quick smooth optimization
            a_opt, _ = optimize_pg(a0, beta=60.0, lr=0.06, n_iter=200)
            score = score_and_update(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best: %.6f" % best_score)

    # Phase 2: Deep L-BFGS-B optimization on promising candidates
    print("=== Phase 2: Deep L-BFGS-B ===")

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    # Focus on the best length and a couple nearby
    target_lengths = sorted(set([l for l in [best_n // 2, best_n * 3 // 4, best_n,
                                             best_n * 5 // 4, best_n * 2, 256, 384]
                                 if 64 <= l <= 1024]))

    phase2_deadline = deadline - 200.0
    n_total_configs = len(target_lengths) * 4
    time_per_config = max(20.0, (phase2_deadline - time.monotonic()) / max(1, n_total_configs))

    for n in target_lengths:
        if time.monotonic() > phase2_deadline - 40:
            break

        inits = generate_initializations(n, parent, rng)

        if best_sequence is not None:
            if len(best_sequence) == n:
                inits['best'] = best_sequence.copy()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)), best_sequence)
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits['best_interp'] = pp / ps

        # Also add from scan results
        for (sc, sn, nm, sa) in scan_results[:3]:
            if sn == n:
                inits['scan_%s' % nm] = sa.copy()

        priority2 = ['best', 'best_interp', 'parent', 'parent_ds', 'parent_up',
                     'theo', 'exp_pos', 'exp_strong', 'bump_right', 'flat_spec',
                     'scan_th', 'scan_exp_pos', 'uniform']
        init_names = [nm for nm in priority2 if nm in inits] + \
                     [nm for nm in inits if nm not in priority2]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() > phase2_deadline - 30:
                break
            t_alloc = min(time_per_config, phase2_deadline - time.monotonic() - 20)
            if t_alloc < 15:
                break

            a0 = inits[name].copy()

            # Multi-beta L-BFGS-B annealing
            betas = [100.0, 50.0, 25.0, 12.0, 6.0, 3.0]
            a_cur = a0.copy()
            time_per_beta = t_alloc / len(betas)

            for beta in betas:
                if time.monotonic() > phase2_deadline - 15:
                    break
                t_single = min(time_per_beta, phase2_deadline - time.monotonic() - 15)
                if t_single < 8:
                    break
                maxiter = int(min(500, max(80, t_single * 40)))
                a_opt, _ = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
                score = score_and_update(a_opt)
                a_cur = a_opt

    print("Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Fine-tuning with multiple techniques
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 30.0

        # L-BFGS-B with very low beta
        for beta in [3.0, 2.0, 1.5, 1.0]:
            if remaining < 15:
                break
            t_single = min(120.0, remaining / 5)
            if t_single < 10:
                break
            maxiter = int(min(600, max(100, t_single * 50)))
            a_opt, _ = optimize_lbfgsb(best_sequence, beta=beta, maxiter=maxiter)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 30.0

        # Subgradient polish
        if remaining > 30:
            a_sg, _ = subgradient_polish(best_sequence, n_steps=300, lr=0.008)
            score_and_update(a_sg)
            remaining = deadline - time.monotonic() - 30.0

        # Multi-peak polish
        if remaining > 30:
            a_mp, _ = multi_peak_polish(best_sequence, n_rounds=80)
            score_and_update(a_mp)
            remaining = deadline - time.monotonic() - 30.0

        # After polishing, re-run L-BFGS-B
        if remaining > 30:
            a_opt, _ = optimize_lbfgsb(best_sequence, beta=5.0, maxiter=300)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 30.0

        # Random perturbations + re-optimize
        if remaining > 40:
            n_perturb = min(8, int(remaining / 30))
            for i in range(n_perturb):
                if remaining < 20:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                t_single = min(40.0, remaining / max(1, n_perturb - i))
                if t_single < 8:
                    break
                maxiter = int(min(300, max(50, t_single * 30)))
                a_opt, _ = optimize_lbfgsb(a_pert, beta=10.0, maxiter=maxiter)
                score_and_update(a_opt)
                # Follow up with subgradient
                if remaining > 20:
                    a_sg, _ = subgradient_polish(a_opt, n_steps=100, lr=0.005)
                    score_and_update(a_sg)
                remaining = deadline - time.monotonic() - 30.0

        # Final projected gradient with low beta
        if remaining > 20:
            n_iters = int(min(400, remaining * 25))
            a_opt, _ = optimize_pg(best_sequence, beta=8.0, lr=0.015, n_iter=n_iters)
            score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 30.0

        # One more subgradient pass
        if remaining > 20:
            a_sg, _ = subgradient_polish(best_sequence, n_steps=int(min(200, remaining * 10)), lr=0.004)
            score_and_update(a_sg)

    # Try a different length if time permits
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 80:
            alt_lengths = [l for l in [192, 256, 320, 384, 512] if l != len(best_sequence)]
            for alt_n in alt_lengths[:2]:
                if remaining < 40:
                    break
                inits_alt = generate_initializations(alt_n, parent, rng)
                if best_sequence is not None:
                    pp = np.interp(np.linspace(0, 1, alt_n),
                                   np.linspace(0, 1, len(best_sequence)), best_sequence)
                    pp = np.maximum(pp, 0)
                    ps = pp.sum()
                    if ps > 1e-10:
                        inits_alt['from_best'] = pp / ps

                a0_alt = inits_alt.get('from_best', inits_alt.get('theo', inits_alt['uniform']))
                t_single = min(60.0, remaining / 2)
                maxiter = int(min(400, max(50, t_single * 3)))
                a_opt, _ = optimize_lbfgsb(a0_alt, beta=20.0, maxiter=maxiter)
                score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 15.0

    # Final verification and return
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
