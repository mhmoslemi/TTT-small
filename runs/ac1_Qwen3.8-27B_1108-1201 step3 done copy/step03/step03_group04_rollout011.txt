def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_peak_lp(a, max_time=10.0):
    """
    Solve: min t s.t. conv(a,b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns optimal b (on simplex) or None.
    """
    n = len(a)
    A_conv = build_conv_matrix(a)
    
    # Variables: [b_0, ..., b_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: conv(a,b)[k] - t <= 0 for each k
    # Row k: A_conv[k,:] @ b - t <= 0
    A_ub = np.column_stack([A_conv.toarray(), -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b >= 0, t free (but t >= 0 to help solver)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max_time})
    if res.success:
        b_opt = np.maximum(res.x[:n], 0.0)
        s = b_opt.sum()
        if s > 1e-12:
            b_opt /= s
        return b_opt, res.x[-1]
    return None, None


def solve_volume_lp(a, M, max_time=10.0):
    """
    Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b (normalized to simplex) or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n + 1)
    c_obj[-1] = 0.0
    
    A_ub = np.column_stack([A.toarray(), np.zeros(2 * n - 1)])
    b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max_time})
    if res.success:
        b_opt = np.maximum(res.x[:n], 0.0)
        s = b_opt.sum()
        if s > 1e-12:
            b_opt /= s
        return b_opt
    return None


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def backtracking_line_search(a, b, obj_a, n_trials_max=20):
    """Backtracking line search for Frank-Wolfe step."""
    best_t = 0.0
    best_obj = obj_a
    t = 0.8
    
    for _ in range(n_trials_max):
        candidate = (1 - t) * a + t * b
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_t = t
        if obj < obj_a * (1 - 1e-4 * t):
            return best_t, best_obj
        t *= 0.7
        if t < 0.01:
            break
    return best_t, best_obj


def fpw_optimize(a0, time_limit, verbose=False, verbose_every=100, use_peak_lp=True):
    """
    Frank-Wolfe optimization with LP oracle.
    a0: initial point on simplex (sum=1, >=0)
    Returns best point found.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 100
    lp_time = min(10.0, time_limit / 50.0)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        # Choose LP type
        current_lp_time = min(lp_time, remaining * 0.3)
        
        if use_peak_lp:
            b, t_opt = solve_peak_lp(a, max_time=current_lp_time)
        else:
            b = solve_volume_lp(a, M, max_time=current_lp_time)
            if b is not None:
                t_opt = None

        if b is None:
            # Fallback: try the other LP
            if use_peak_lp:
                b = solve_volume_lp(a, M, max_time=current_lp_time)
            else:
                b, t_opt = solve_peak_lp(a, max_time=current_lp_time)
        
        if b is None:
            no_improve += 10
        else:
            # Line search
            obj_a = compute_obj(a)
            t_step, obj_new = backtracking_line_search(a, b, obj_a, n_trials_max=15)
            
            if t_step < 0.005:
                no_improve += 1
            else:
                a = (1 - t_step) * a + t_step * b
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        # Stuck handling: targeted peak perturbation
        if no_improve >= max_stall:
            conv = fast_convolve(a)
            top_k = min(8, 2 * n - 1)
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]
            
            # Push elements that contribute to top peaks
            rng = np.random.default_rng(it + 42)
            for pi in range(len(peak_indices)):
                k = int(peak_indices[pi])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                for j in range(j_start, j_end + 1):
                    if a[j] > 1e-6 and a[k - j] > 1e-6:
                        a[j] -= a[j] * 0.02 * rng.random()
                        a[k - j] -= a[k - j] * 0.02 * rng.random()
            
            # Also add some noise
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.7f}, M={M:.4e}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def soft_max_polish(a0, time_limit, temperature_init=0.01):
    """
    Polish using smoothed max via log-sum-exp.
    Minimizes 2*n*T*log(sum(exp(conv/T))) as approximation to peak.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    T = temperature_init
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.01
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Soft-max gradient
        conv_shifted = conv - conv.max()
        weights = np.exp(conv_shifted / T)
        weights /= weights.sum()
        
        # Gradient of soft-max(conv) w.r.t. a
        g = np.zeros(n)
        for i in range(len(conv)):
            if weights[i] < 1e-12:
                continue
            k = i
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * weights[i] * a[k - idx]
        
        # Update with momentum
        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
        
        # Check actual objective
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Decay temperature and learning rate
        T *= 0.999
        lr *= 0.998
        it += 1
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.004, momentum=0.93):
    """Polish with projected subgradient on simplex, targeting top peaks."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    avg_lr_window = []
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Multi-peak subgradient with softmax weighting
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.03, 1e-15)
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]
        
        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        
        avg_lr_window.append(lr)
        if len(avg_lr_window) > 100:
            avg_lr_window.pop(0)
        
        lr *= 0.9992
        it += 1
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    inits['ramp_rev'] = r[::-1].copy()
    
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri = tri ** 0.5  # flatter top
    tri /= tri.sum()
    inits['triangle'] = tri.copy()
    
    # Edge-concentrated with decay
    edge = max(1, n // 4)
    ec = np.zeros(n)
    for i in range(edge):
        ec[i] = np.exp(-i / (edge * 0.5))
        ec[n - 1 - i] = np.exp(-i / (edge * 0.5))
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    
    # Asymmetric edge
    ae = np.zeros(n)
    left_w = max(1, n // 6)
    right_w = max(1, n // 3)
    for i in range(left_w):
        ae[i] = np.exp(-i / (left_w * 0.6))
    for i in range(right_w):
        ae[n - 1 - i] = np.exp(-i / (right_w * 0.6))
    ae /= ae.sum()
    inits['asym_edge'] = ae.copy()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sparse random
    sparse = rng.random(n)
    threshold = 0.4
    sparse[sparse < threshold] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()
    
    # Cosine bell
    cos_bell = np.cos(np.pi * (x - (n-1)/2) / n) ** 2
    cos_bell = np.maximum(cos_bell, 0.01)
    cos_bell /= cos_bell.sum()
    inits['cosine'] = cos_bell.copy()
    
    # Two-bump
    bump1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    bump2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two_b = bump1 + bump2
    two_b /= two_b.sum()
    inits['two_bump'] = two_b.copy()
    
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(parent)), pp)
                pp_interp = np.maximum(pp_interp, 0.0)
                pp_interp = pp_interp / pp_interp.sum()
                inits['parent_interp'] = pp_interp
    
    return inits


def coordinate_descent_polish(a0, time_limit, step_size=0.01):
    """
    Coordinate descent: try small adjustments to individual elements
    that reduce the peak convolution value.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 2:
            break
        
        # Try a few random coordinate updates
        n_trials = min(50, max(5, int((time_limit - elapsed) * 10)))
        for _ in range(n_trials):
            i = rng.integers(0, n)
            delta = step_size * rng.choice([-1, 1]) * a[i] * 0.1
            a_new = a.copy()
            a_new[i] = max(0.0, a_new[i] + delta)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                step_size = min(step_size * 1.1, 0.05)
            else:
                step_size *= 0.95
        
        it += 1
    
    return best_a, best_obj


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-phase search: LP-FW with peak-minimizing oracle, soft-max polishing,
    coordinate descent, and targeted perturbation restarts.
    """
    rng = np.random.default_rng(42)

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p

    except NameError:
        pass
    except Exception:
        pass
    
    def update_best(a, tag=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST {tag}: score={score:.7f}, n={len(a)}")
        return score
    
    # === Phase 1: Exploration across lengths and inits ===
    print("=" * 60)
    print("Phase 1: Exploration")
    print("=" * 60)
    
    lengths = [256, 384, 512, 768]
    
    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'edge', 'asym_edge', 'uniform',
                    'two_bump', 'cosine', 'triangle', 'gauss', 'random',
                    'sparse', 'ramp', 'ramp_rev']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)
        
        # Allocate time: more time to larger n (more potential)
        n_inits_to_try = min(len(init_names), 4 if n <= 512 else 3)

        for name in init_names[:n_inits_to_try]:
            if time.monotonic() > deadline - 180:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 180)
            if t_alloc < 8:
                break
            
            a0 = inits[name]
            print(f"  n={n}, init={name}, t={t_alloc:.0f}s")
            a_opt, obj_opt = fpw_optimize(a0, time_limit=t_alloc, use_peak_lp=True)
            update_best(a_opt, f"n={n},{name}")
    
    print(f"\nAfter Phase 1: best_score={best_score:.7f}")
    
    # === Phase 2: Deep LP-FW on best ===
    if best_sequence is not None and time.monotonic() < deadline - 120:
        print("\n" + "=" * 60)
        print("Phase 2: Deep LP-FW Optimization")
        print("=" * 60)
        
        n_best = len(best_sequence)
        t_deep = min(500.0, (deadline - time.monotonic() - 120) * 0.5)
        
        a_opt, obj = fpw_optimize(best_sequence, time_limit=t_deep,
                                   verbose=True, verbose_every=100, use_peak_lp=True)
        update_best(a_opt, "deep_lp")
        
        # Also try volume LP variant
        if time.monotonic() < deadline - 100:
            t_vol = min(150.0, (deadline - time.monotonic() - 100) * 0.4)
            if t_vol > 10:
                a_opt2, obj2 = fpw_optimize(best_sequence, time_limit=t_vol,
                                             verbose=False, use_peak_lp=False)
                update_best(a_opt2, "vol_lp")
    
    # === Phase 3: Soft-max polish ===
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("\n" + "=" * 60)
        print("Phase 3: Soft-Max Polish")
        print("=" * 60)
        
        t_sm = min(120.0, (deadline - time.monotonic() - 40) * 0.5)
        if t_sm > 10:
            a_opt, obj = soft_max_polish(best_sequence, time_limit=t_sm,
                                          temperature_init=0.005)
            update_best(a_opt, "softmax")
    
    # === Phase 4: Subgradient polish ===
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print("\n" + "=" * 60)
        print("Phase 4: Subgradient Polish")
        print("=" * 60)

                    a_pert[k - j] *= (1 - 0.01 * rng.random())
            
            # Add small global noise
            noise = rng.standard_normal(n_best) * 0.001
            a_pert = np.maximum(a_pert + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            
            t_alloc = min(20.0, deadline - time.monotonic() - 8)
            if t_alloc < 5:
                break
            
            a_opt, obj = fpw_optimize(a_pert, time_limit=t_alloc, use_peak_lp=True)
            update_best(a_opt, f"restart_{i}")
    
    # === Phase 7: Final LP pass if time ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("\n" + "=" * 60)
        print("Phase 7: Final LP Pass")
        print("=" * 60)
        
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:

                                       verbose=False, use_peak_lp=True)
            update_best(a_opt, "final_lp")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\n{'=' * 60}")
    print(f"FINAL: score={best_score:.7f}, n={len(best_sequence)}")
    print(f"{'=' * 60}")
    
    return [float(x) for x in best_sequence]
