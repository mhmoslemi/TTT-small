def fast_autoconv(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    conv = fast_autoconv(a)
    s = a.sum()
    if s < 1e-15:

    return 2.0 * n * conv.max() / (s * s)

    """Apply spectral shaping to reduce auto-convolution peak."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.abs(A) ** 2
    # Soft flattening
    P_new = P ** alpha * (P.sum() / len(P)) ** (1 - alpha)

    phase = A / np.abs(A)
    phase[np.abs(A) < 1e-15] = 1.0
    A_new = np.sqrt(P_new) * phase
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    return a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-n initialization with structured and random starts

3. LP-based iterative direction finding

import numpy as np
import time
from scipy.optimize import linprog

def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_value(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    conv = fast_autoconv(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf, conv
    return 2.0 * n * conv.max() / (s * s), conv

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) s.t. a*g <= M (componentwise), g >= 0.
    """
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0

def nesterov_subgradient(a0, time_limit, lr_init=0.01, momentum_init=0.9, top_k=30):

    n = len(a)

    v = np.zeros(n)
    t = 1.0
    
    _, conv = obj_value(a)

    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:

        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue

            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * y[k - idx]

        if s_y > 1e-15:

def smoothed_lbfgs_polish(a0, time_limit, tau_init=0.005):
    """
    Polish using L-BFGS-B with smoothed objective.
    """
    from scipy.optimize import minimize
    
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj, _ = obj_value(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    tau = tau_init
    
    while time.monotonic() - t0 < time_limit and tau > 1e-6:
        def smooth_obj(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10
            x = x / sx
            conv = fast_autoconv(x)
            c_max = conv.max()
            shifted = (conv - c_max) / tau
            log_sum_exp = c_max / tau + np.log(np.sum(np.exp(shifted)))
            return 2.0 * n * tau * log_sum_exp / (1.0)  # sum is 1
        
        def smooth_grad(x):
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return np.zeros(n)
            x = x / sx
            conv = fast_autoconv(x)
            c_max = conv.max()
            shifted = (conv - c_max) / tau
            w = np.exp(shifted)
            w /= w.sum()
            # Gradient of smooth_max w.r.t. a
            g_max = np.zeros(n)
            for k in range(len(conv)):
                if w[k] < 1e-12:
                    continue
                m_start = max(0, k - n + 1)
                m_end = min(n, k + 1)
                for m in range(m_start, m_end):
                    idx = k - m
                    if 0 <= idx < n:
                        g_max[m] += 2.0 * w[k] * x[idx]
            # d/dx of 2n * smooth_max (with sum=1 constraint handled by projection)
            return 2.0 * n * g_max
        
        try:
            result = minimize(
                smooth_obj, a.copy(), jac=smooth_grad,
                method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 20}
            )
            
            if result.success or result.fun < best_obj:
                a_new = np.maximum(result.x, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    obj_new, _ = obj_value(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new.copy()
        except Exception:
            pass
        
        tau *= 0.5  # Reduce smoothing
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-n structured initialization (Hann, raised cosine, bimodal, parent)
2. Nesterov-accelerated projected subgradient descent
3. LP-based iterative direction finding with golden-section line search
4. Smoothed L-BFGS polish via softmax parameterization
5. Adaptive perturbation and re-optimization
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2. Returns (value, conv)."""
    n = len(a)
    conv = fast_autoconv(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf, conv
    return 2.0 * n * conv.max() / (s * s), conv


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution of lengths 2n-1 by n)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g) <= M (componentwise), g >= 0.
    Returns (g_normalized_to_sum1, g_sum).
    """
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def nesterov_subgradient(a0, time_limit, lr_init=0.012, top_k=30, temp_init=0.0008):
    """
    Nesterov-accelerated projected subgradient descent with adaptive temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    t_nest = 1.0

    _, conv = obj_value(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    temp = temp_init
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute weighted subgradient
        conv_y = fast_autoconv(y)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * y[k - idx]

        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        lr = lr_init / (1.0 + 5.0 * np.sqrt(t_nest))

        a_new = y - lr * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        a = a_new
        t_nest = t_new
        temp *= 1.0005

        _, conv = obj_value(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0

            y = a.copy()
            t_nest = 1.0
            no_improve = 0

                best_t = 0.0
                best_obj_try = best_obj

                    obj_try, _ = obj_value(a_try)

def softmax_smoothed_obj(x, tau, n):

    x_shift = x - x.max()

def softmax_smoothed_grad(x, tau, n):
    """Gradient of smoothed objective in softmax parameterization using FFT for efficiency."""
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()

    conv = fast_autoconv(a)
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    w = np.exp(shifted)
    w /= w.sum()

    # h[i] = sum_k w[k] * a[k-i] for valid k-i in [0, n-1]
    # This is cross-correlation, compute via FFT
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A_fft = np.fft.rfft(a, fft_size)
    # Cross-correlation: ifft(conj(A) * W) gives sum_k a[k] * w[k-i]
    # We want h[i] = sum_k w[k] * a[k-i] = (w ~ a)[i]
    # (w ~ a)[i] = ifft(conj(fft(a)) * fft(w)) -- need to be careful with definitions
    # Actually: ifft(fft(w) * conj(fft(a))) gives the circular cross-correlation
    # For linear cross-correlation with zero-padding, this works directly
    cross = np.fft.irfft(W * np.conj(A_fft), fft_size)[:2 * n - 1]
    # h[i] = cross[i] for i = 0, ..., n-1 (the sum is over k where both indices are valid)
    h = cross[:n]

    # df/da_i = 4n * h[i] (from the chain rule)
    # df/dx_j = a_j * (df/da_j - sum_i df/da_i * a_i)
    grad_da = 4.0 * n * h
    inner = np.dot(a, grad_da)
    grad_x = a * (grad_da - inner)
    return grad_x


def smoothed_lbfgs_polish(a0, time_limit, tau_init=0.004, max_rounds=5):
    """
    Polish using L-BFGS-B with smoothed objective in softmax parameterization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj, _ = obj_value(a)
    best_a = a.copy()
    t0 = time.monotonic()

    tau = tau_init
    x = np.log(np.maximum(a, 1e-15))  # initial x for softmax parameterization

    for round_idx in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        def obj_fn(x_eval):
            return softmax_smoothed_obj(x_eval, tau, n)

        def grad_fn(x_eval):
            return softmax_smoothed_grad(x_eval, tau, n)

        try:
            result = minimize(
                obj_fn, x.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30}
            )

            # Convert back
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                obj_new, _ = obj_value(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = result.x.copy()
        except Exception as e:
            print(f"  L-BFGS round {round_idx} error: {e}")

        tau *= 0.6  # Reduce smoothing each round

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    inits['hann'] = hann / hann.sum()

    # Raised cosine squared
    rc = 0.5 * (1.0 + np.cos(np.pi * x / max(1, n // 2 - 1)))
    rc = rc ** 2
    rc = np.maximum(rc, 0)
    if rc.sum() > 1e-10:
        inits['raised_cos'] = rc / rc.sum()

    # Bimodal (edges weighted)
    bi = np.zeros(n)
    edge = max(1, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Sine arch
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits['sine'] = sine / sine.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Random Dirichlet (concentrated)
    inits['random_conc'] = rng.dirichlet(np.ones(n) * 0.1)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    def eval_and_update(a):
        """Evaluate candidate, update best if improved. Returns score or inf."""
        nonlocal best_sequence, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01 or not np.all(np.isfinite(a)):
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  [improved] score={score:.6f}, n={len(a)}")
        return score

    # === Phase 1: Initialization scan + quick subgradient (250s) ===
    print("=== Phase 1: Init Scan + Subgradient ===")
    t_start = time.monotonic()
    phase1_end = t_start + 250.0

    ns_to_try = [512]
    if parent is not None and len(parent) != 512:
        ns_to_try = [len(parent), 512]
    ns_to_try += [384, 640]
    # Deduplicate
    ns_unique = []
    for nn in ns_to_try:
        if nn not in ns_unique:
            ns_unique.append(nn)
    ns_to_try = ns_unique[:4]

    all_candidates = []

    for n in ns_to_try:
        if time.monotonic() > phase1_end - 60:
            break

        inits = generate_inits(n, parent, rng)

        # Evaluate all initializations quickly
        for name, a0 in inits.items():
            score = eval_and_update(a0)
            all_candidates.append((score if np.isfinite(score) else 1e10, n, name, a0.copy()))

        # Run Nesterov subgradient on top 3 initializations
        all_candidates.sort(key=lambda x: x[0])
        n_candidates_remaining = max(1, len(ns_to_try) - ns_to_try.index(n))
        t_per_subgrad = min(40.0, max(10.0, (phase1_end - time.monotonic() - 30) / max(1, n_candidates_remaining * 3)))

        for score, cn, cname, a0 in all_candidates:
            if time.monotonic() > phase1_end - 30 or cn != n:
                continue
            if len([c for c in all_candidates if c[1] == n and c[3] is not None]) > 3:
                pass  # already have 3 for this n
            if score > 2.5:  # skip very bad inits
                continue

        # Pick top 3 for this n
        n_cands = [(sc, nm, a0) for sc, cn, nm, a0 in all_candidates if cn == n][:3]
        for idx, (sc, nm, a0) in enumerate(n_cands):
            if time.monotonic() > phase1_end - 30:
                break
            t_alloc = min(35.0, phase1_end - time.monotonic() - 20)
            if t_alloc < 8:
                break
            a_opt, _ = nesterov_subgradient(a0, time_limit=t_alloc)
            eval_and_update(a_opt)
            all_candidates.append((eval_and_update(a_opt) if best_score < 1e10 else 1e10, n, nm + '_sg', a_opt.copy()))

    print(f"Phase 1 done ({time.monotonic() - t_start:.0f}s). Best: {best_score:.6f}")

    # === Phase 2: LP-based deep optimization (900s) ===
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    if best_sequence is not None:
        # Primary LP on best
        remaining = deadline - time.monotonic() - 350
        t_alloc = min(350.0, remaining * 0.45)
        if t_alloc > 20:
            print(f"  LP round 1 (t={t_alloc:.0f}s)")
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_alloc)
            eval_and_update(a_opt)

        # LP with perturbations
        remaining = deadline - time.monotonic() - 300
        n_perts = max(1, int(remaining / 80))
        for pi in range(n_perts):
            if time.monotonic() > deadline - 300:
                break
            scale = 0.008 * (pi + 1)
            noise = rng.standard_normal(len(best_sequence)) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            s = a_pert.sum()
            if s > 1e-15:
                a_pert = a_pert / s
            t_alloc = min(70.0, deadline - time.monotonic() - 250)
            if t_alloc < 10:
                break
            print(f"  LP round {pi+2} (t={t_alloc:.0f}s)")
            a_opt, _ = lp_optimize(a_pert, time_limit=t_alloc)
            eval_and_update(a_opt)

    # Also try LP on second-best candidate
    all_candidates.sort(key=lambda x: x[0])
    for score, n, name, a0 in all_candidates[1:3]:
        if time.monotonic() > deadline - 250:
            break
        if score >= best_score * 1.001:
            continue
        t_alloc = min(60.0, deadline - time.monotonic() - 200)
        if t_alloc < 10:
            break
        print(f"  LP on alt candidate ({name}, n={n}, t={t_alloc:.0f}s)")
        a_opt, _ = lp_optimize(a0, time_limit=t_alloc)
        eval_and_update(a_opt)

    print(f"Phase 2 done ({time.monotonic() - t_start2:.0f}s). Best: {best_score:.6f}")

    # === Phase 3: Smoothed L-BFGS polish (200s) ===
    print("=== Phase 3: L-BFGS Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50
        t_alloc = min(150.0, remaining)
        if t_alloc > 15:
            print(f"  L-BFGS polish (t={t_alloc:.0f}s)")
            a_opt, _ = smoothed_lbfgs_polish(best_sequence, time_limit=t_alloc, tau_init=0.003, max_rounds=4)
            eval_and_update(a_opt)

        # Another round with different tau
        if time.monotonic() < deadline - 30:
            a_opt, _ = smoothed_lbfgs_polish(best_sequence, time_limit=20, tau_init=0.001, max_rounds=2)
            eval_and_update(a_opt)

    print(f"Phase 3 done ({time.monotonic() - t_start3:.0f}s). Best: {best_score:.6f}")

    # === Phase 4: Nesterov subgradient fine-tune with small lr (100s) ===
    print("=== Phase 4: Fine-tune Subgradient ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20
        t_alloc = min(80.0, remaining)
        if t_alloc > 10:
            a_opt, _ = nesterov_subgradient(best_sequence, time_limit=t_alloc, lr_init=0.005, top_k=50, temp_init=0.0003)
            eval_and_update(a_opt)

    print(f"Phase 4 done ({time.monotonic() - t_start4:.0f}s). Best: {best_score:.6f}")

    # === Phase 5: Final short LP burst (remaining time) ===
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_opt, _ = lp_optimize(best_sequence, time_limit=remaining)
            eval_and_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
