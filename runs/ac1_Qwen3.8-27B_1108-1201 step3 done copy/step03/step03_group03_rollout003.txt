**Frequency-domain flattening algorithm:**

The peak of `conv(a,a)` at lag k is:
`c[k] = (1/N) sum_{f=0}^{N-1} |A[f]|^2 * exp(2πi f k / N)`

where A is the N-point FFT of a (zero-padded to N >= 2n-1).

To reduce the peak at a specific lag k_peak, we want to reduce the contribution of frequencies that constructively interfere at that lag. This means reducing `|A[f]|^2` for frequencies f where `exp(2πi f * k_peak / N)` has positive real part (i.e., where the phase aligns with the peak).

**Algorithm:**
1. Normalize a to sum=1
2. Compute A = FFT(a, N)
3. Compute P = |A|^2
4. Compute c = real(IFFT(P, N)[:2n-1])
5. Find k_peak = argmax(c)
6. Create a "flattening mask": for each frequency f, compute the contribution to c[k_peak]:
   `contrib[f] = P[f] * cos(2π * f * k_peak / N)`
7. Create a new power spectrum: `P_new[f] = P[f] * (1 - alpha * contrib[f] / sum(contrib))` for positive contrib, and `P_new[f] = P[f]` for negative contrib.
   Or more simply: `P_new[f] = P[f] - alpha * max(0, cos(2π f k_peak / N)) * P[f] / N`
8. Take square root to get new amplitude: `|A_new[f]| = sqrt(P_new[f])`
9. Keep the phase of A: `A_new[f] = |A_new[f]| * exp(i * angle(A[f]))`
10. Inverse FFT: `a_new = real(IFFT(A_new, N))[:n]`
11. Clip to non-negative: `a_new = max(a_new, 0)`
12. Renormalize: `a_new /= sum(a_new)`

This is similar to the "iterative amplitude modification" (IAM) algorithm from MRI, adapted for our problem.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using frequency-domain spectral flattening,
simulated annealing with structured moves, subgradient refinement, and coordinate descent.
Key novelty: iterative frequency-domain power spectrum shaping to reduce autocorrelation peaks,
combined with simulated annealing for escaping local optima.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def spectral_flatten(a, n_iter=200, alpha=0.3, n_peaks=3, n_shift=0.0):
    """
    Frequency-domain iterative spectral flattening.
    Modifies the power spectrum to reduce autocorrelation peaks.
    """
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    
    fft_size = 2 ** (2 * n - 1).bit_length()
    # Use full FFT for phase handling
    A = np.fft.fft(a, fft_size)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iter):
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Compute current autocorrelation
        c = np.real(np.fft.ifft(P, fft_size))[:2 * n - 1]
        peak_val = c.max()
        if peak_val < 1e-15:
            break
        
        # Find top peaks
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # For each peak, compute which frequencies contribute positively
        # and suppress them
        P_new = P.copy()
        for k in top_idx:
            # Contribution of each frequency to c[k]
            cos_terms = np.cos(2.0 * np.pi * np.arange(fft_size) * k / fft_size)
            # Only suppress frequencies that constructively contribute to this peak
            positive_mask = cos_terms > 0
            if positive_mask.sum() == 0:
                continue
            # Reduce power at constructively contributing frequencies
            reduction = alpha * P[positive_mask] * cos_terms[positive_mask]
            P_new[positive_mask] -= reduction
        
        P_new = np.maximum(P_new, 0.0)
        
        # New amplitude = sqrt(new power), keep original phase
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.real(np.fft.ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Update A for next iteration
        A = np.fft.fft(a_new, fft_size)
        
        cur_obj = 2.0 * n * fast_convolve(a_new).max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
    
    return best_a, best_obj


def spectral_flatten_multi(a, total_iters=500, alpha_start=0.5, alpha_end=0.05,
                           n_peaks=5, reinit_every=100):
    """Spectral flattening with adaptive alpha and periodic re-initialization."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for block in range(total_iters // reinit_every + 1):
        alpha = alpha_start * (alpha_end / alpha_start) ** (block / max(1, total_iters // reinit_every - 1))
        iters = min(reinit_every, total_iters - block * reinit_every)
        if iters <= 0:
            break
        a, obj = spectral_flatten(a, n_iter=iters, alpha=alpha, n_peaks=n_peaks)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def simulated_annealing(a_init, time_limit, rng, n_modes=8, t_start=0.05, t_end=0.001):
    """
    Simulated annealing with structured moves on the simplex.
    Moves: sinusoidal perturbations of various frequencies, mass transfer, block operations.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    t = t_start
    n_total = max(1, int(time_limit))
    x = np.arange(n, dtype=np.float64)
    
    # Precompute sinusoidal modes
    modes = []
    for m in range(1, n_modes + 1):
        for phase in [0.0, np.pi / 2]:
            mode = np.cos(2.0 * np.pi * m * x / n + phase)
            mode = mode - mode.mean()  # zero-mean
            norm = np.linalg.norm(mode)
            if norm > 1e-10:
                modes.append(mode / norm)
    
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Choose a move type
        move_type = rng.integers(0, 4)
        
        if move_type == 0 and len(modes) > 0:
            # Sinusoidal perturbation
            mode = modes[rng.integers(0, len(modes))]
            eps = t * 0.5
            a_cand = a + eps * mode
            a_cand = project_simplex(a_cand)
        elif move_type == 1:
            # Mass transfer between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            eps = t * 0.3 * min(a[i], 0.5 / n)
            a_cand = a.copy()
            a_cand[i] -= eps
            a_cand[j] += eps
            a_cand = np.maximum(a_cand, 0.0)
            sc = a_cand.sum()
            if sc > 1e-15:
                a_cand /= sc
            else:
                continue
        elif move_type == 2:
            # Block swap: swap two random blocks
            bsize = max(2, n // (2 * n_modes))
            i = rng.integers(0, n - 2 * bsize)
            j = rng.integers(0, n - 2 * bsize)
            if abs(i - j) < bsize:
                continue
            a_cand = a.copy()
            a_cand[i:i+bsize], a_cand[j:j+bsize] = a_cand[j:j+bsize].copy(), a_cand[i:i+bsize].copy()
            # Small perturbation to break symmetry
            a_cand += rng.standard_normal(n) * 0.001 * t
            a_cand = project_simplex(a_cand)
        else:
            # Gaussian perturbation
            a_cand = a + rng.standard_normal(n) * t * 0.1
            a_cand = project_simplex(a_cand)
        
        # Evaluate
        conv_cand = fast_convolve(a_cand)
        cand_obj = 2.0 * n * conv_cand.max()
        
        # Acceptance criterion
        delta = cand_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            a = a_cand
            cur_obj = cand_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Cool down
        t = max(t_end, t * 0.9995)
        
        # Restart if stuck
        if no_improve > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.02)
            cur_obj = 2.0 * n * fast_convolve(a).max()
            t = max(t_start * 0.5, t)
            no_improve = 0
    
    return best_a, best_obj


def subgradient_refine(a_init, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp=0.0001):
    """Subgradient descent with momentum, focused on top peaks."""
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        # Soft weights based on peak values
        scaled = peak_vals / (temp + pmax * 0.1)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng_local = np.random.default_rng(hash(tuple(a[:8])) % (2**31))
            a = project_simplex(a + rng_local.standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.6)
        
        lr *= 0.9998
    
    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=100, n_try=3):
    """
    Coordinate descent: try moving mass from i to j with line search.
    Uses n_try candidate splits for each pair.
    """
    a = np.maximum(a_in, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Sample pairs to try (for large n, can't try all)
        if n <= 256:
            pairs = [(i, j) for i in range(n) for j in range(n) if i != j]
            np.random.shuffle(pairs)
            pairs = pairs[:min(len(pairs), 5000)]
        else:
            pairs = []
            for _ in range(5000):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j:
                    pairs.append((i, j))
        
        for i, j in pairs:
            if a[i] < 1e-8:
                continue
            # Try several fractions of mass to transfer
            for frac in [0.1, 0.3, 0.5]:
                delta = a[i] * frac
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
                    break
        
        if not improved:
            break
    
    return a


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def lp_direction_step(a, M, pairs, n):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = a[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    edge = np.zeros(n)
    e = max(1, n // 6)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.001)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Four peaks
    g4 = sum(np.exp(-0.5 * ((x - frac * n) / (0.06 * n)) ** 2) 
             for frac in [0.15, 0.35, 0.65, 0.85])
    inits['fourpeak'] = g4 / g4.sum()
    
    # Random sinusoidal
    m1, m2 = rng.integers(1, max(2, n // 10)), rng.integers(1, max(2, n // 10))
    ph1, ph2 = rng.uniform(0, 2 * np.pi, 2)
    sig = 1.0 + 0.5 * np.cos(2 * np.pi * m1 * x / n + ph1) + 0.3 * np.cos(2 * np.pi * m2 * x / n + ph2)
    sig = np.maximum(sig, 0.01)
    inits['sines'] = sig / sig.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                prs = p_resampled.sum()
                if prs > 1e-10:
                    inits['parent_res'] = p_resampled / prs
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining spectral flattening, simulated annealing,
    subgradient, LP directions, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_score = par_score
            best_sequence = p.copy() / p.sum()
    except Exception:
        pass
    
    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST {label}: score={score:.6f}, n={len(a)}")
        return score
    
    # Phase 1: Spectral flattening scan over lengths and initializations
    print("=== Phase 1: Spectral Flattening Scan ===")
    lengths = [256, 384, 512, 640, 768]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.append(pn)
            lengths.sort()
    
    results = []
    phase1_time = min(600.0, (deadline - time.monotonic()) * 0.35)
    phase1_end = time.monotonic() + phase1_time
    
    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'parent_res', 'uniform', 'cosine', 'sines',
                'bimodal', 'fourpeak', 'triangle', 'gauss', 'edge', 'random']
        names = [nm for nm in prio if nm in inits][:4]
        
        t_per = max(10.0, phase1_time / (len(lengths) * 4))
        
        for name in names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            
            # Spectral flattening
            t_spec = min(t_alloc * 0.6, 40.0)
            a_opt, obj_opt = spectral_flatten_multi(a0, total_iters=int(t_spec * 15),
                                                     alpha_start=0.4, alpha_end=0.03,
                                                     n_peaks=4)
            # Quick subgradient polish
            t_sg = min(t_alloc * 0.3, 10.0)
            a_opt2, obj2 = subgradient_refine(a_opt, t_sg, lr_init=0.008,
                                               momentum=0.9, top_k=20, temp=0.0001)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2
            
            score = update_best(a_opt, f"spec n={n} {name}")
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
    
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Simulated annealing on top candidates
    print("=== Phase 2: Simulated Annealing ===")
    top3 = []
    seen = set()
    for score, n, name, a_opt in results[:6]:
        key = n
        if key not in seen:
            top3.append((n, a_opt))
            seen.add(key)
        if len(top3) >= 3:
            break
    if best_sequence is not None:
        if len(best_sequence) not in seen:
            top3.append((len(best_sequence), best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 400.0
    t_per_sa = max(30.0, remaining / max(1, len(top3) * 2))
    
    for n, a0 in top3:
        if time.monotonic() > deadline - 380:
            break
        t_alloc = min(t_per_sa, deadline - time.monotonic() - 350)
        if t_alloc < 15:
            break
        
        for sa_round in range(2):
            if time.monotonic() > deadline - 340:
                break
            t_sa = t_alloc / 2.0
            if t_sa < 10:
                break
            a_sa, obj_sa = simulated_annealing(a0, t_sa, rng, n_modes=10,
                                               t_start=0.04, t_end=0.0005)
            # Spectral polish after SA
            t_sp = min(15.0, t_sa * 0.3)
            a_sp, obj_sp = spectral_flatten_multi(a_sa, total_iters=int(t_sp * 10),
                                                   alpha_start=0.2, alpha_end=0.02, n_peaks=5)
            if obj_sp < obj_sa:
                a_sa, obj_sa = a_sp, obj_sp
            update_best(a_sa, f"SA n={n} r{sa_round}")
            a0 = a_sa  # Use as start for next round
    
    # Phase 3: Deep optimization (LP + subgradient + spectral cycles)
    print("=== Phase 3: Deep Optimization ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 200.0
        n_cycles = max(2, int(remaining / 60))
        
        for cyc in range(n_cycles):
            if time.monotonic() > deadline - 170:
                break
            t_cyc = min(remaining / (n_cycles - cyc), 80.0)
            if t_cyc < 15:
                break
            
            # Spectral flattening
            t1 = t_cyc * 0.4
            a1, obj1 = spectral_flatten_multi(best_sequence, total_iters=int(t1 * 12),
                                              alpha_start=0.15, alpha_end=0.01, n_peaks=6)
            
            # Subgradient
            t2 = t_cyc * 0.3
            a2, obj2 = subgradient_refine(a1, t2, lr_init=0.004, momentum=0.92,
                                          top_k=40, temp=0.00005)
            if obj2 < obj1:
                a1, obj1 = a2, obj2
            
            # LP direction
            t3 = t_cyc * 0.2
            if t3 > 5:
                try:
                    a_lp_in = a1 / a1.sum()
                    conv_lp = fast_convolve(a_lp_in)
                    M = conv_lp.max()
                    pairs = build_lp_pairs(len(a_lp_in))
                    # Do a few LP steps
                    blend = 0.15
                    for _ in range(3):
                        g = lp_direction_step(a_lp_in, M, pairs, len(a_lp_in))
                        if g is None:
                            break
                        gs = g.sum()
                        if gs < 1e-15:
                            break
                        g = g / gs
                        a_lp_in = project_simplex((1 - blend) * a_lp_in + blend * g)
                        conv_lp = fast_convolve(a_lp_in)
                        M = conv_lp.max()
                    if 2.0 * len(a_lp_in) * M < 2.0 * len(a1) * fast_convolve(a1).max():
                        a1 = a_lp_in
                except Exception:
                    pass
            
            update_best(a1, f"cyc{cyc}")
            
            # Small simulated annealing nudge
            if time.monotonic() < deadline - 160:
                t_nudge = min(10.0, t_cyc * 0.1)
                if t_nudge > 3:
                    a_nud, _ = simulated_annealing(a1, t_nudge, rng, n_modes=5,
                                                   t_start=0.01, t_end=0.002)
                    a_sp2, _ = spectral_flatten(a_nud, n_iter=50, alpha=0.1, n_peaks=4)
                    update_best(a_sp2, f"nudge{cyc}")
    
    # Phase 4: Coordinate descent polish
    print("=== Phase 4: Coordinate Descent Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0
        t_cd = min(remaining * 0.5, 120.0)
        
        n_cd = len(best_sequence)
        max_rounds = min(80, max(20, int(t_cd / (n_cd / 10))))
        a_cd = coordinate_descent(best_sequence, max_rounds=max_rounds)
        update_best(a_cd, "CD")
        
        # Final spectral + subgradient cycle
        if time.monotonic() < deadline - 40:
            t_final = min(30.0, deadline - time.monotonic() - 35)
            a_f1, _ = spectral_flatten_multi(best_sequence, total_iters=int(t_final * 8),
                                             alpha_start=0.1, alpha_end=0.005, n_peaks=8)
            a_f2, _ = subgradient_refine(a_f1, t_final * 0.5, lr_init=0.002,
                                         momentum=0.93, top_k=50, temp=0.00003)
            update_best(a_f2, "final")
            
            # Abs final coordinate descent
            if time.monotonic() < deadline - 30:
                a_cd2 = coordinate_descent(best_sequence, max_rounds=30)
                update_best(a_cd2, "CD_final")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
