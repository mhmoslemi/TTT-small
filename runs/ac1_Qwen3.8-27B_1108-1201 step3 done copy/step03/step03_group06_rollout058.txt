def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    # Standard simplex projection
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, epsilon=0.0):

Mirror descent + LP hybrid optimization for minimizing peak auto-convolution.
Uses multiplicative weight updates as the primary method, with LP-based

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = int(np.max(rho[cond]))
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, epsilon_frac=0.05):
    """Compute subgradient using active set (peaks within epsilon_frac of max)."""
    n = len(a)
    max_val = conv.max()
    threshold = max_val * (1.0 - epsilon_frac)
    active = np.where(conv >= threshold)[0]
    
    g = np.zeros(n)
    for k in active:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        g[idx] += 2.0 * a[k - idx]
    
    g /= len(active)
    return g

def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def mirror_descent_optimize(a0, time_limit, eta_init=0.5, epsilon_init=0.02,

    epsilon = epsilon_init

        g = compute_subgradient(a, conv, epsilon)
        
        # Center gradient on support

        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()

def lp_direction(a, M, top_k=None, conv=None):
    """Solve LP for Frank-Wolfe direction."""
    n = len(a)
    num_k = 2 * n - 1
    
    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        rows = []
        for k in peaks:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            row = np.zeros(n)
            idx = np.arange(i_min, i_max + 1)
            row[idx] = a[k - idx]
            rows.append(row)
        A_ub = np.array(rows)
    else:
        rows = []
        for k in range(num_k):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            row = np.zeros(n)
            idx = np.arange(i_min, i_max + 1)
            row[idx] = a[k - idx]
            rows.append(row)
        A_ub = np.array(rows)
    
    b_ub = np.full(A_ub.shape[0], M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None

def lp_optimize(a0, time_limit, step_init=0.8, top_k=None, verbose=False, seed=0):
    """LP-based Frank-Wolfe optimization."""
    rng = np.random.default_rng(seed)

    s = a.sum()
    if s < 1e-15:

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        b = lp_direction(a, M, top_k=top_k, conv=conv)
        if b is None:
            b = lp_direction(a, M, top_k=None)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if step > 0.05:
                step *= 0.9
            else:
                if no_improve > 5:
                    noise = rng.standard_normal(n) * 0.003
                    a = project_simplex(a + noise)
                    step = step_init * 0.5
                    no_improve = 0
        
        if no_improve >= 200:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            step = step_init * 0.6
            no_improve = 0
    
    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine window
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        inits['cosine'] = cos_w / cos_w.sum()
    
    # Bimodal
    edge = max(1, n // 3)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Wave-like
    for freq in [2, 3, 4, 5]:
        w = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n)
        w = np.maximum(w, 0.1)
        inits[f'wave_{freq}'] = w / w.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_interp'] = p / ps
    
    return inits
