Minimizes 2n*max(conv(a,a))/(sum(a))^2 using linearized LP-based optimization
with all-constraint inclusion, multiple initializations, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradients_all(a):
    """
    Compute gradients of all C_k w.r.t. a.
    Returns a 2D array of shape (2n-1, n) where grad[k, j] = dC_k/da[j].
    grad_k[j] = a[k-j] + a[j-k] (with a[x]=0 for x outside [0,n-1])
    """
    n = len(a)
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # a[k-j] is non-zero for j in [max(0, k-n+1), min(k, n-1)]
        j1_start = max(0, k - n + 1)
        j1_end = min(k, n - 1)
        if j1_start <= j1_end:
            js = j1_start
            # a[k-j] for j in [j1_start, j1_end]
            # When j = j1_start, k-j = k - j1_start
            # When j = j1_end, k-j = k - j1_end
            indices = k - np.arange(j1_start, j1_end+1)
            grad[k, j1_start:j1_end+1] += a[indices]
        
        # a[j-k] is non-zero for j in [max(k, 0), min(k+n-1, n-1)]
        j2_start = max(k, 0)
        j2_end = min(k + n - 1, n - 1)
        if j2_start <= j2_end:
            indices = np.arange(j2_start, j2_end+1) - k
            grad[k, j2_start:j2_end+1] += a[indices]
    
    return grad

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute the objective value 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def optimize_smoothed(a0, time_limit, beta_init=30.0, beta_end=200.0,
                      lr_init=0.015, momentum=0.92, seed=0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        
        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)
        
        # Smoothed gradient
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        
        w = conv_y * p
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]
        
        # Momentum update
        lr = max(0.0003, lr_init * (1.0 - 0.7 * frac))
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 400:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
    
    return best_a, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Exponential decay
    for beta in [0.5, 1.0, 2.0]:
        exp_d = np.exp(-beta * x / n)
        exp_d /= exp_d.sum()
        inits[f'exp_{beta}'] = exp_d
    
    # Tilted uniform
    for eps in [0.1, 0.3, 0.5]:
        tilted = 1.0 + eps * (x - n/2) / (n/2)
        tilted = np.maximum(tilted, 0.01)
        tilted /= tilted.sum()
        inits[f'tilted_{eps}'] = tilted
    
    # Bell
    bell = 1.0 / (1.0 + 0.5 * ((x - n / 2) / (n / 4)) ** 2)
    bell /= bell.sum()
    inits['bell'] = bell
    
    # Random Dirichlet
    inits['dir_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dir_1'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['dir_2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using linearized LP-based optimization
with efficient FFT-based line search, multiple initializations, and multi-length search.
Key insight: the gradient of C_k w.r.t. a is 2*a[k-j] for j in S_k, enabling
efficient constraint evaluation via cross-correlation (FFT).
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_correl(a, b):
    """Full cross-correlation (a ⊛ b)[k] = sum_m a[m]*b[k-m] via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * np.conj(B), fft_size)
    # Cross-correlation: (a ⊛ b)[k] = (a * flip(b))[k + m - 1]
    # Actually, let's use the direct definition:
    # (a ⊛ b)[k] = sum_j a[j] * b[k-j]
    # This equals conv(a, flip(b)) shifted
    # Let me just compute it properly:
    # FFT property: IDFT(FFT(a) * conj(FFT(b))) gives circular cross-correlation
    # For linear cross-correlation, we need zero-padding
    # The result of irfft(FFT(a) * conj(FFT(b))) with proper padding gives:
    # result[k] = sum_j a[j] * b[(k-j) mod fft_size]
    # Which for k in [0, size-1] with sufficient padding equals sum_j a[j]*b[k-j]
    return result[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_grad_matrix(a):
    """
    Build the gradient matrix G where G[k, j] = dC_k/da[j] = 2*a[k-j] if j in S_k else 0.
    S_k = {j : max(0, k-n+1) <= j <= min(n-1, k)}
    Shape: (2n-1, n)
    """
    n = len(a)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            indices = k - np.arange(j_start, j_end + 1)
            G[k, j_start:j_end + 1] = 2.0 * a[indices]
    return G


def optimize_lp(a0, time_limit, seed=0, use_all_constraints=True, active_frac=0.6):
    """
    Linearized LP-based optimization.
    At each step:
    1. Compute convolution C_k
    2. Set up LP: min M' s.t. 2*(a⊛d)[k] - M' <= -C_k, sum(d)=0, a+d>=0
    3. Line search with exact quadratic evaluation
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_M = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iters = 0
    lp_failures = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Determine which constraints to include
        if use_all_constraints:
            active_k = np.arange(2 * n - 1)
        else:
            threshold = active_frac * M
            active_k = np.where(conv > threshold)[0]
            if len(active_k) < 3:
                active_k = np.arange(2 * n - 1)

        # Build LP
        # Variables: d[0..n-1], M' (index n)
        # Minimize M'
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        n_active = len(active_k)
        A_ub = np.zeros((n_active, n + 1))
        for idx, k in enumerate(active_k):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                indices = k - np.arange(j_start, j_end + 1)
                A_ub[idx, j_start:j_end + 1] = 2.0 * a[indices]
            A_ub[idx, n] = -1.0
        b_ub = -conv[active_k].astype(np.float64)

        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        # Bounds: d[i] >= -a[i]
        lower = -a.astype(np.float64)
        lower = np.append(lower, -np.inf)
        bounds = list(zip(lower, [None] * (n + 1)))

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': 3})
        except Exception:
            result = None

        if result is None or not result.success or result.x is None:
            lp_failures += 1
            if lp_failures > 5:
                # Fallback to gradient step
                peak_k = np.argmax(conv)
                j_start = max(0, int(peak_k) - n + 1)
                j_end = min(n - 1, int(peak_k))
                grad = np.zeros(n)
                indices = int(peak_k) - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] = 2.0 * a[indices]
                a = project_simplex(a - 0.005 * grad)
                no_improve = max(0, no_improve - 1)
            else:
                a = project_simplex(a + rng.standard_normal(n) * 0.005)
            continue

        lp_failures = 0
        d = result.x[:n]
        M_lp = result.x[n]

        if M_lp >= M - 1e-9:
            no_improve += 1
            if no_improve > 20:
                scale = 0.003 * (1.0 + 0.5 * np.sin(iters * 0.01))
                a = project_simplex(a + rng.standard_normal(n) * scale)
                no_improve = 0
            continue

        # Line search: C_k(a + t*d) = C_k(a) + 2t*(a⊛d)[k] + t^2*(d⊛d)[k]
        # Compute (a⊛d) and (d⊛d) via FFT
        ad_corr = fast_cross_correl(a, d)  # (a⊛d)[k] = sum_j a[j]*d[k-j]
        dd_conv = fast_convolve(d)  # (d⊛d)[k] = sum_j d[j]*d[k-j]

        # We need the full length (2n-1) for both
        ad_corr = ad_corr[:2 * n - 1]
        dd_conv = dd_conv[:2 * n - 1]

        improved = False
        best_t = 0
        best_new_M = M

        for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]:
            new_conv = conv + 2.0 * t * ad_corr + t * t * dd_conv
            new_M = new_conv.max()
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
            if new_M < M - 1e-12:
                improved = True
                break

        if improved and best_t > 0:
            a_new = a + best_t * d
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a = project_simplex(a + rng.standard_normal(n) * 0.01)
            else:
                a = a_new / s_new
        else:
            # Try very small steps
            for t in [0.001, 0.0005, 0.0001]:
                new_conv = conv + 2.0 * t * ad_corr + t * t * dd_conv
                if new_conv.max() < M - 1e-14:
                    a_new = a + t * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-10:
                        a = a_new / s_new
                        improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 40:
                a = project_simplex(a + rng.standard_normal(n) * 0.008)
                no_improve = 0

    return best_a, best_M


def optimize_smoothed(a0, time_limit, beta_init=30.0, beta_end=300.0,
                      lr_init=0.012, momentum=0.92, seed=0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)

        # Smoothed gradient via log-sum-exp
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum

        w = conv_y * p
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]

        # Momentum update with adaptive LR
        lr = max(0.0002, lr_init * (1.0 - 0.8 * frac**0.5))
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Perturbation schedule
        if no_improve >= 300 + 100 * perturb_count:
            noise_scale = 0.002 * (1.0 + 0.3 * np.sin(perturb_count * 1.7))
            a = project_simplex(a + rng.standard_normal(n) * noise_scale)
            v *= 0.2
            no_improve = 0
            perturb_count += 1

    return best_a, best_obj


def optimize_peak_target(a0, time_limit, seed=0):
    """
    Peak-targeting: identify peaks and use coordinate-descent-like updates
    to reduce them specifically.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_M = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            a = project_simplex(a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            continue

        # Find the top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[conv[peak_indices] > 0.9 * M]

        improved = False
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue

            # Find the pair with maximum contribution to C_k
            indices = k - np.arange(j_start, j_end + 1)
            contribs = a[j_start:j_end + 1] * a[indices]
            if len(contribs) == 0:
                continue

            max_idx = j_start + np.argmax(contribs)

            # Try reducing a[max_idx] and adding to another position
            for _ in range(5):
                j = rng.integers(0, n)
                if j == max_idx:
                    continue

                for frac in [0.03, 0.07, 0.1, 0.15]:
                    amt = a[max_idx] * frac
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[max_idx] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < M - 1e-12:
                        a = a_try
                        M = conv_try.max()
                        best_M = M
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break

            if improved:
                break

    return best_a, best_M


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Monotonic ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    # Exponential decays
    for beta in [0.3, 0.7, 1.5, 3.0]:
        exp_d = np.exp(-beta * x / n)
        exp_d /= exp_d.sum()
        inits[f'exp_{beta}'] = exp_d

    # Tilted uniform (key for breaking symmetry)
    for eps in [0.05, 0.15, 0.3, 0.5, 0.8]:
        tilted = 1.0 + eps * (2.0 * x / (n - 1) - 1.0)
        tilted = np.maximum(tilted, 0.01)
        tilted /= tilted.sum()
        inits[f'tilted_{eps}'] = tilted

    # Quadratic tilts
    for alpha in [1.5, 2.0, 3.0]:
        q = (x + 1) ** alpha
        q /= q.sum()
        inits[f'quad_{alpha}'] = q

    # Sine-based
    sine = 1.0 + 0.3 * np.sin(2 * np.pi * x / n)
    sine = np.maximum(sine, 0.01)
    sine /= sine.sum()
    inits['sine'] = sine

    # Random Dirichlet
    inits['dir_03'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['dir_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dir_1'] = rng.dirichlet(np.ones(n) * 1.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

                best_sequence = a_c.copy()
                print(f"    Peak: {score_c:.6f}", flush=True)

    # Phase 3: Final intensive polish
    print("=== Phase 3: Final Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Extended LP
        t_lp = remaining * 0.45
        if t_lp > 10:
            a_p, obj_p = optimize_lp(best_sequence, t_lp, seed=12345,
                                     use_all_constraints=True)
            score_p = float(evaluate_sequence(a_p.tolist()))
            if np.isfinite(score_p) and score_p < best_score:
                best_score = score_p
                best_sequence = a_p.copy()
                print(f"  LP Polish: {best_score:.6f}", flush=True)

        # High-beta gradient
        remaining = deadline - time.monotonic() - 5.0
        t_g = remaining * 0.35
        if t_g > 5:
            a_g, obj_g = optimize_smoothed(best_sequence, t_g,
                                           beta_init=200, beta_end=800,
                                           lr_init=0.002, momentum=0.96,
                                           seed=67890)
            score_g = float(evaluate_sequence(a_g.tolist()))
            if np.isfinite(score_g) and score_g < best_score:
                best_score = score_g
                best_sequence = a_g.copy()
                print(f"  FineGrad: {best_score:.6f}", flush=True)

        # Final peak targeting
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 5:
            a_pk, obj_pk = optimize_peak_target(best_sequence, remaining, seed=99999)
            score_pk = float(evaluate_sequence(a_pk.tolist()))
            if np.isfinite(score_pk) and score_pk < best_score:
                best_score = score_pk
                best_sequence = a_pk.copy()
                print(f"  FinalPeak: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
