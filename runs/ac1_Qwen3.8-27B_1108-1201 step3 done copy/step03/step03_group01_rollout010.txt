Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced LP-based iterative
method with spectral initialization, adaptive constraint tightening, dual-guided
direction selection, and multi-scale exploration.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    # C has shape (2n-1, n). C[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    # So row k of C is: [a[k], a[k-1], ..., a[0], 0, ..., 0, 0, ..., a[k-n+1]]
    # More precisely: C[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # j ranges from max(0, k-n+1) to min(n-1, k)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C


def build_conv_matrix_sparse(a):
    """Build the convolution matrix as a scipy sparse matrix for efficiency."""
    from scipy.sparse import diags, eye
    # Actually, let's just use dense for n <= 1024
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C


def solve_lp(a, M, warm_start=None):
    """
    Solve LP: maximize sum(g) s.t. C@a @ g <= M, g >= 0.
    Returns optimal g and dual variables.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        return g, g_sum
    return None, 0.0

    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (S * S)

    Enhanced LP-based optimization with:
    - Adaptive M tightening
    - Dual-guided direction
    - Better line search
    - Perturbation strategy

    else:
        a = a / S
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M  # since sum(a)=1
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 100
    perturb_count = 0
    margin = 1.0  # multiplier for M in LP constraint
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        g, g_sum = solve_lp(a, M * margin)
        
        if g is None or g_sum <= 1.0 + 1e-8:

            if no_improve % 20 == 0:
                margin = min(1.05, margin + 0.005)
        else:
            # Normalize g to sum = 1
            g = g / g_sum
            d = g - a
            norm_d = np.linalg.norm(d)
            
            if norm_d < 1e-14:

            else:

                best_local_obj = best_obj

                    S = a.sum()
                    a = a / S

                    M = conv.max()

"""
Enhanced LP-based optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses HiGHS LP solver with adaptive margin, golden-section line search,
diverse structured initializations, Nesterov-accelerated subgradient polish,
and multi-phase time allocation across sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) s.t. C_a @ g <= M (componentwise), g >= 0.
    Returns (g, g_sum) or (None, 0) on failure.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-12:
            return g, g_sum
    return None, 0.0


def golden_section_line_search(a, d, best_obj, n, max_evals=15):
    """
    Golden section search along a + t*d (clipped to nonneg, normalized).
    Returns (best_t, best_obj_found).
    """
    best_t = 0.0
    best_local = best_obj
    evals = 0

    # First: find bracket [lo, hi] where function decreases
    lo, hi = 0.0, 1.0

    # Try a few points to bracket
    for t in [0.5, 1.0, 0.25, 0.75, 0.1, 0.9]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            evals += 1
            continue
        a_try = a_try / s
        obj_try = 2.0 * n * fast_convolve(a_try).max()
        evals += 1
        if obj_try < best_local - 1e-13:
            best_local = obj_try
            best_t = t

    # Golden section refinement around best_t
    if 0.0 < best_t < 1.0 and evals < max_evals:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        span = 0.15
        a_lo = max(0.01, best_t - span)
        a_hi = min(0.99, best_t + span)

        for _ in range(4):
            if evals >= max_evals:
                break
            c1 = a_hi - gr * (a_hi - a_lo)
            c2 = a_lo + gr * (a_hi - a_lo)

            obj1 = _eval_dir(a, d, c1, n)
            evals += 1
            obj2 = _eval_dir(a, d, c2, n)
            evals += 1

            if obj1 < obj2:
                a_hi = c2
                if obj1 < best_local:
                    best_local = obj1
                    best_t = c1
            else:
                a_lo = c1
                if obj2 < best_local:
                    best_local = obj2
                    best_t = c2

    return best_t, best_local


def _eval_dir(a, d, t, n):
    """Evaluate objective along direction at step t."""
    a_try = a + t * d
    a_try = np.maximum(a_try, 0.0)
    s = a_try.sum()
    if s < 1e-15:
        return np.inf
    a_try = a_try / s
    return 2.0 * n * fast_convolve(a_try).max()


def lp_optimize(a0, time_limit, verbose=True):
    """
    Main LP-based iterative optimizer with adaptive margin and perturbation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 80
    perturb_count = 0
    margin = 1.0
    consecutive_fail = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP with current margin
        g, g_sum = solve_lp_direction(a, M * margin)

        if g is None or g_sum <= 1.0 + 1e-7:
            no_improve += 1
            consecutive_fail += 1
            if consecutive_fail % 10 == 0:
                margin = min(1.08, margin + 0.004)
        else:
            consecutive_fail = 0
            g = g / g_sum
            d = g - a
            norm_d = np.linalg.norm(d)

            if norm_d < 1e-14:
                no_improve += 1
            else:
                t_best, obj_best = golden_section_line_search(a, d, best_obj, n)

                if t_best > 0 and obj_best < best_obj - 1e-12:
                    a = a + t_best * d
                    a = np.maximum(a, 0.0)
                    S = a.sum()
                    a = a / S

                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    margin = max(1.0, margin - 0.003)

                    if verbose and iteration % 30 == 0:
                        print(f"    LP iter {iteration}, obj={best_obj:.8f}, M={M:.8f}, "
                              f"margin={margin:.4f}, elapsed={time.monotonic()-t0:.1f}s")
                else:
                    no_improve += 1

        iteration += 1

        # Perturbation on stagnation
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 31 + perturb_count * 97 + 13)
            scale = 0.002 + 0.004 * min(perturb_count, 15)
            # Use structured perturbation: concentrate noise on high-conv positions
            conv = fast_convolve(best_a)
            peak_idx = np.argpartition(conv, -n // 2)[-(n // 2):]
            noise = rng.standard_normal(n) * scale
            # Extra noise on elements that contribute to peaks
            for k in peak_idx[:10]:
                j_start = max(0, int(k) - n + 1)
                j_end = min(n - 1, int(k))
                noise[j_start:j_end + 1] += rng.standard_normal(j_end - j_start + 1) * scale * 0.5

            a = best_a + noise
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S > 1e-15:
                a = a / S
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            margin = 1.0
            max_no_improve = min(250, 80 + 40 * perturb_count)

            if verbose:
                print(f"    Perturb {perturb_count}: obj={2.0*n*M:.8f}, scale={scale:.5f}")

            if perturb_count > 40:
                break

    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.008, momentum=0.93, top_k=60, temp=0.0002):
    """
    Nesterov-accelerated projected subgradient descent.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)

    # Nesterov momentum variables
    y = a.copy()
    v = np.zeros(n)
    beta = 1.0  # momentum parameter

    conv = fast_convolve(y)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    batch_size = 20

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute subgradient using top-k peaks with softmax weights
            conv = fast_convolve(y)
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum

            # Subgradient of max_k (y*y)[k]
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * y[k - idx]

            # Nesterov update
            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new > 1e-15:
                a = a_new / S_new
            else:
                a = best_a.copy()

            # Nesterov extrapolation
            beta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
            y = a + (beta_new / beta - 1.0) * (a - best_a + 0.01 * (a - y))
            y = np.maximum(y, 0.0)
            S_y = y.sum()
            if S_y > 1e-15:
                y = y / S_y
            else:
                y = a.copy()
            beta = beta_new

            conv_a = fast_convolve(a)
            cur_obj = 2.0 * n * conv_a.max()

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        lr *= 0.998
        temp *= 1.005
        momentum = min(0.97, momentum + 0.0002)

        if no_improve >= 400:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Triangular (peaked center)
    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 0.5
    inits['triangle'] = tri / tri.sum()

    # Bimodal (mass at edges)
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Add some in the middle too
    mid = max(1, n // 8)
    bi[n // 2 - mid:n // 2 + mid] = 0.5
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Sine
    s = np.sin(np.pi * (x + 1) / (n + 1)) + 0.05
    inits['sine'] = s / s.sum()

    # Power law decay
    pw = np.exp(-0.3 * x / n)
    inits['exp'] = pw / pw.sum()

    # Alternating structure (helps with autocorrelation)
    alt = np.ones(n)
    alt[::3] = 1.5
    alt[1::3] = 0.7
    inits['alt'] = alt / alt.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                # Resample
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_rs'] = pp / pps

    # Weighted random (biased toward structure)
    w = rng.random(n)
    w *= (1.0 + 0.5 * np.sin(2 * np.pi * x / (n // 4)))
    w = np.maximum(w, 0.01)
    inits['struct_rand'] = w / w.sum()

    return inits
