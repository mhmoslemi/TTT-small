import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n*max(conv)/sum^2 for normalized a (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def subgradient_dir(a, conv, top_k=30, temperature=None):
    """Compute weighted subgradient of max(conv) at current point."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    if temperature is None:
        temperature = peak_vals.max() * 0.02
    scaled = peak_vals / max(temperature, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        wi = w[idx]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            contrib = a[k - j]
            if j == k - j:
                contrib *= 2.0
            sg[j] += wi * contrib
    return sg


def line_search_simplex(a, direction, conv, f_current, max_iter=50):
    """
    Line search: find best t in [0,1] such that a_new = project(a + t*direction)
    minimizes the objective. Uses parabolic interpolation.
    """
    n = len(a)
    # Pre-compute norm of direction for scaling
    dir_norm = np.linalg.norm(direction)
    if dir_norm < 1e-15:
        return a, f_current

    direction = direction / dir_norm

    # Initial points
    t0 = 0.0
    f0 = f_current
    t1 = 0.1
    a1 = project_simplex(a + t1 * direction)
    f1 = 2.0 * n * fast_convolve(a1).max()
    t2 = 0.5
    a2 = project_simplex(a + t2 * direction)
    f2 = 2.0 * n * fast_convolve(a2).max()

    best_t = t0
    best_f = f0
    best_a = a.copy()

    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = 1.0
    c_ = hi - gr * (hi - lo)
    d_ = lo + gr * (hi - lo)

    # Evaluate at initial points
    fc = f1 if abs(c_ - t1) < 0.01 else None
    fd = f2 if abs(d_ - t2) < 0.01 else None

    for _ in range(max_iter):
        if fc is None:
            a_c = project_simplex(a + c_ * direction)
            fc = 2.0 * n * fast_convolve(a_c).max()
        if fd is None:
            a_d = project_simplex(a + d_ * direction)
            fd = 2.0 * n * fast_convolve(a_d).max()

        if fc < fd:
            hi = d_
            d_ = c_
            fd = fc
            c_ = hi - gr * (hi - lo)
            fc = None
        else:
            lo = c_
            c_ = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = None

        mid = (lo + hi) / 2.0
        a_mid = project_simplex(a + mid * direction)
        fm = 2.0 * n * fast_convolve(a_mid).max()
        if fm < best_f:
            best_f = fm
            best_t = mid
            best_a = a_mid.copy()

        if hi - lo < 0.01:
            break

    return best_a, best_f


def optimize_subgrad_linesearch(a_init, time_limit, lr_init=0.05, top_k=40,
                                 temp_factor=0.02, momentum=0.7):
    """Subgradient descent with line search and momentum."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    prev_a = a.copy()
    prev_grad = None

    while time.monotonic() - t0 < time_limit:
        sg = subgradient_dir(a, conv, top_k=top_k,
                             temperature=conv.max() * temp_factor)

        # Barzilai-Borwein step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = sg - prev_grad
            sy = s_vec @ y_vec
            yy = y_vec @ y_vec
            ss = s_vec @ s_vec
            if yy > 1e-15 and ss > 1e-15:
                if sy > 1e-15:
                    lr_bb = ss / sy
                else:
                    lr_bb = sy / yy
                lr = np.clip(lr_bb, 0.001, 0.5)

        prev_a = a.copy()
        prev_grad = sg.copy()

        v = momentum * v - lr * sg
        direction = v - v.mean()  # Remove mean to stay on simplex

        # Line search
        a_new, f_new = line_search_simplex(a, direction, conv, best_obj, max_iter=30)
        cur_obj = 2.0 * n * f_new
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            # Perturb and restart
            rng = np.random.default_rng(hash(tuple(a[:5])) % 2**31)
            a = project_simplex(a + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0

        if time.monotonic() - t0 > time_limit * 0.8:
            temp_factor *= 0.95

    return best_a, best_obj


def optimize_multiplicative(a_init, time_limit, eta_init=0.1, top_k=30,
                             temp_factor=0.03):
    """Mirror descent (multiplicative weights) on the simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sg = subgradient_dir(a, conv, top_k=top_k,
                             temperature=conv.max() * temp_factor)

        # Multiplicative update: a[i] *= exp(-eta * sg[i])
        # Then re-normalize
        update = np.exp(-eta * sg)
        a_new = a * update
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-4:
                break

        eta *= 0.999

    return best_a, best_obj


def pair_coordinate_optimize(a_init, time_limit, max_rounds=None):
    """
    Pair-coordinate optimization: for pairs (i,j), find the optimal
    redistribution of mass between them that minimizes the objective.
    Uses 1D search along the line between e_i and e_j.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    if max_rounds is None:
        max_rounds = 1000

    # Precompute: for each pair (i,j), the change in conv when moving delta from i to j
    # conv[k] changes by: delta * (a[k-j] - a[k-i]) + delta^2 * (1 if k-i==j or k-j==i)
    # For simplicity, use linear approximation and verify

    # Sort indices by value for efficiency
    order = np.argsort(a)[::-1]

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for pass_idx in range(2):
            if pass_idx == 0:
                # Forward pass
                indices = order
            else:
                indices = order[::-1]

            for i in indices:
                if a[i] < 1e-8:
                    continue
                ai = a[i]
                # Try moving to a few candidate j's
                # Best candidates: where a[|k*-i|] is largest (most impact on peak)
                k_star = int(np.argmax(conv))
                # j that minimizes a[|k*-j|]
                impact = np.array([abs(k_star - j) for j in range(n)])
                # We want j where a[impact[j]] is small (less contribution to peak)
                # But also j where moving mass doesn't create new peaks
                # Simple heuristic: try j with small a[j] and small impact on peak
                candidates = []
                for j in range(n):
                    if j == i:
                        continue
                    # Contribution to peak k* from position j
                    contrib_j = a[abs(k_star - j)] if abs(k_star - j) < n else 0
                    if j == k_star:
                        contrib_j = 2 * a[0]
                    candidates.append((contrib_j, j))
                candidates.sort()
                for contrib_j, j in candidates[:8]:
                    # Try moving delta from i to j
                    # Optimal delta: minimize max(conv + delta*(grad_j - grad_i))
                    # Linear approx: conv[k] + delta*(a[k-j] - a[k-i])
                    delta = ai * 0.3
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    if a_test[i] < 0 or a_test[j] < 0:
                        continue
                    st = a_test.sum()
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()

                    if obj_test < best_obj - 1e-14:
                        # Accept and try to find better delta via bisection
                        lo_d, hi_d = 0.0, delta
                        best_d = delta
                        best_f = obj_test
                        for _ in range(8):
                            mid_d = (lo_d + hi_d) / 2
                            a_mid = a.copy()
                            a_mid[i] -= mid_d
                            a_mid[j] += mid_d
                            if a_mid[i] < 0:
                                hi_d = mid_d
                                continue
                            a_mid /= a_mid.sum()
                            conv_mid = fast_convolve(a_mid)
                            f_mid = 2.0 * n * conv_mid.max()
                            if f_mid < best_f:
                                best_f = f_mid
                                best_d = mid_d
                                lo_d = mid_d
                            else:
                                hi_d = mid_d
                        a[i] -= best_d
                        a[j] += best_d
                        a /= a.sum()
                        conv = fast_convolve(a)
                        best_obj = 2.0 * n * conv.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_obj


def optimize_smoothed_lbfgs(a_init, time_limit, beta_init=50.0, beta_min=5.0):
    """
    Optimize the smoothed objective: (1/beta) * log(sum(exp(beta * conv[k])))
    using L-BFGS-B, then project to simplex.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    beta = beta_init
    while time.monotonic() - t0 < time_limit:
        def smoothed_obj(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10
            x = x / sx
            conv = fast_convolve(x)
            beta_conv = beta * (conv - conv.max())
            beta_conv = np.clip(beta_conv, -50, 50)
            log_sum = conv.max() + np.log(np.sum(np.exp(beta_conv))) / beta
            return 2.0 * n * log_sum

        def smoothed_grad(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-15:
                return np.zeros(len(x))
            x = x / sx
            conv = fast_convolve(x)
            beta_conv = beta * (conv - conv.max())
            beta_conv = np.clip(beta_conv, -50, 50)
            weights = np.exp(beta_conv)
            weights /= weights.sum()
            # Gradient of sum_k weights[k] * conv[k] w.r.t. x[i]
            n_x = len(x)
            grad = np.zeros(n_x)
            for k_idx in range(len(weights)):
                if weights[k_idx] < 1e-10:
                    continue
                k = k_idx
                wi = weights[k_idx]
                j_start = max(0, k - n_x + 1)
                j_end = min(n_x - 1, k)
                for j in range(j_start, j_end + 1):
                    c = x[k - j]
                    if j == k - j:
                        c *= 2.0
                    grad[j] += wi * c
            return 2.0 * n * grad

        # Run L-BFGS-B on the smoothed problem
        try:
            result = minimize(
                smoothed_obj, a, jac=smoothed_grad,
                method='L-BFGS-B',
                options={'maxiter': 30, 'gtol': 1e-4, 'maxfun': 200}
            )
            a_new = project_simplex(result.x)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

        beta = max(beta_min, beta * 0.5)
        if beta <= beta_min:
            # Do a few more iterations at minimum beta
            try:
                result = minimize(
                    smoothed_obj, a, jac=smoothed_grad,
                    method='L-BFGS-B',
                    options={'maxiter': 50, 'gtol': 1e-5, 'maxfun': 500}
                )
                a_new = project_simplex(result.x)
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
            except Exception:
                pass
            break

    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    edge = np.zeros(n)
    e = max(1, n // 6)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2
    cos_w = np.maximum(cos_w, 0.001)
    inits['cosine'] = cos_w / cos_w.sum()

    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    hann = np.maximum(hann, 0.001)
    inits['hann'] = hann / hann.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            elif len(parent) > n:
                inits['parent_ds'] = np.interp(
                    np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p / ps)
            else:
                inits['parent_up'] = np.interp(
                    np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p / ps)

    return inits


def optimize_lp_direction(f, M, n, pairs):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                  'time_limit': max(1.0, (time.monotonic() - t0_global) if 't0_global' in dir() else 0)})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def optimize_lp_combined(f_init, time_limit, blend_init=0.3, blend_min=0.005,
                         blend_halve_every=10):
    """LP-based Frank-Wolfe with line search."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    pairs = build_lp_pairs(n)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction_fast(a, M, pairs, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search along direction g - a
        direction = g - a
        a_new, f_new = line_search_simplex(a, direction, conv, best_obj, max_iter=25)
        cur_obj = 2.0 * n * f_new
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                # Switch to smaller steps
                break

    return best_a, best_obj


def solve_lp_direction_fast(f, M, pairs, n):
    """Fast LP solve with time limit."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def simulate_anneal(a_init, time_limit, T_init=0.01, T_min=0.0001,
                    cooling=0.999, n_samples=5):
    """
    Simulated annealing: try random pair swaps, accept with probability
    exp(-delta/T). Different from coordinate descent because it allows
    temporary worsening.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    T = T_init
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Amount to move
        delta = min(a[i], 0.1 * a[i]) if a[i] > 1e-8 else 0
        if delta < 1e-10:
            delta = rng.uniform(0, 0.01)
            if a[i] < delta:
                i, j = j, i
                delta = min(a[i], 0.1 * a[i])
                if delta < 1e-10:
                    continue

        a_test = a.copy()
        a_test[i] -= delta
        a_test[j] += delta
        if a_test[i] < 0 or a_test[j] < 0:
            continue
        a_test /= a_test.sum()

        conv_test = fast_convolve(a_test)
        obj_test = 2.0 * n * conv_test.max()
        delta_obj = obj_test - best_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, T_min)):
            a = a_test
            conv = conv_test
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

        T = max(T_min, T * cooling)

        if T <= T_min * 1.01 and delta_obj > 0:
            # Reheat occasionally
            if rng.random() < 0.01:
                T = T_init * 0.1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Fast scan over lengths and inits with subgradient + LP
    Phase 2: Deep optimization with multiple methods on top candidates
    Phase 3: Fine-grained polish (pair optimization + annealing + L-BFGS)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass

    # Phase 1: Quick scan
    print("=== Phase 1: Quick Scan ===")
    lengths = [384, 512, 640, 768, 1024]
    results = []
    phase1_end = time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.25)

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)

        # Priority: parent first, then diverse inits
        prio = ['parent', 'parent_ds', 'parent_up', 'uniform', 'triangle',
                'bimodal', 'cosine', 'hann', 'random', 'edge', 'gauss']
        names = [nm for nm in prio if nm in inits][:4]

        t_per = max(8.0, 200.0 / (len(lengths) * 4))

        for name in names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 5:
                break

            a0 = inits[name]

            # Use subgradient with line search (different from previous LP-only)
            a_opt, obj_opt = optimize_subgrad_linesearch(
                a0, t_alloc, lr_init=0.05, top_k=30, temp_factor=0.02)

            # Quick multiplicative refinement
            t_rem = min(3.0, t_alloc * 0.2)
            a_opt2, obj2 = optimize_multiplicative(
                a_opt, t_rem, eta_init=0.05, top_k=20)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen_keys = set()
    for score, n, name, a_opt in results[:6]:
        key = (n, name)
        if key not in seen_keys:
            top_list.append((n, a_opt))
            seen_keys.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 300.0
    t_per = max(30.0, remaining / max(1, len(top_list) * 3))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 200)
        if t_alloc < 10:
            break

        # Method A: LP with line search
        t_sub = t_alloc / 3.0
        if t_sub > 5:
            a_opt, obj_opt = optimize_lp_combined(
                a0, t_sub, blend_init=0.25, blend_min=0.005, blend_halve_every=12)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP n={n}: {score:.6f}")
                a0 = a_opt

        # Method B: Subgradient with line search (more iterations)
        t_sub = t_alloc / 3.0
        if t_sub > 5:
            a_opt, obj_opt = optimize_subgrad_linesearch(
                a0, t_sub, lr_init=0.03, top_k=50, temp_factor=0.01, momentum=0.8)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SG n={n}: {score:.6f}")
                a0 = a_opt

        # Method C: Multiplicative
        t_sub = t_alloc / 3.0
        if t_sub > 5:
            a_opt, obj_opt = optimize_multiplicative(
                a0, t_sub, eta_init=0.08, top_k=40, temp_factor=0.015)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Mult n={n}: {score:.6f}")
                a0 = a_opt

    # Phase 3: Fine-grained polish
    print("=== Phase 3: Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        n_b = len(best_sequence)

        # 3a: Pair coordinate optimization
        t_cd = min(150.0, remaining * 0.3)
        if t_cd > 5:
            a_cd, obj_cd = pair_coordinate_optimize(best_sequence, t_cd)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  PairOpt: {score_cd:.6f}")

        # 3b: L-BFGS on smoothed objective
        remaining = deadline - time.monotonic() - 40.0
        t_lbfgs = min(80.0, remaining * 0.25)
        if t_lbfgs > 5:
            a_lb, obj_lb = optimize_smoothed_lbfgs(
                best_sequence, t_lbfgs, beta_init=30.0, beta_min=3.0)
            score_lb = float(evaluate_sequence(a_lb.tolist()))
            if np.isfinite(score_lb) and score_lb < best_score:
                best_score = score_lb
                best_sequence = a_lb.copy()
                print(f"  L-BFGS: {score_lb:.6f}")

        # 3c: Simulated annealing
        remaining = deadline - time.monotonic() - 30.0
        t_sa = min(60.0, remaining * 0.2)
        if t_sa > 5:
            a_sa, obj_sa = simulate_anneal(
                best_sequence, t_sa, T_init=0.005, T_min=0.00005,
                cooling=0.9995)
            score_sa = float(evaluate_sequence(a_sa.tolist()))
            if np.isfinite(score_sa) and score_sa < best_score:
                best_score = score_sa
                best_sequence = a_sa.copy()
                print(f"  Anneal: {score_sa:.6f}")

        # 3d: Final cycles of subgradient + pair opt
        remaining = deadline - time.monotonic() - 20.0
        for cycle in range(3):
            if time.monotonic() > deadline - 15:
                break
            t_cyc = min(remaining / (3 - cycle + 1), 30.0)
            if t_cyc < 5:
                break

            t1 = t_cyc * 0.6
            a_opt, _ = optimize_subgrad_linesearch(
                best_sequence, t1, lr_init=0.02, top_k=60,
                temp_factor=0.008, momentum=0.85)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

            t2 = t_cyc * 0.4
            a_opt2, _ = pair_coordinate_optimize(best_sequence, t2)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  Cycle {cycle}: {score2:.6f}")

        # 3e: Tiny perturbations + re-optimize
        if time.monotonic() < deadline - 10:
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 8:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 6:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = optimize_subgrad_linesearch(
                        a_p, 3.0, lr_init=0.01, top_k=40, temp_factor=0.01)
                    a_opt2, _ = pair_coordinate_optimize(a_opt, 2.0)
                    score = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt2.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
