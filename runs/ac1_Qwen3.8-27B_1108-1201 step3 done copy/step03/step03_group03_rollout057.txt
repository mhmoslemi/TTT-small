Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional optimization,

Key improvements over previous: better subgradient computation, Nesterov momentum,

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Returns (gradient_vector, peak_value).

    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pmax = conv[peaks].max()
    if pmax < 1e-15:
        return np.zeros(n), pmax
    
    # Soft weighting: more weight to closer-to-peak values
    diffs = conv[peaks] - pmax  # <= 0
    weights = np.exp(diffs / (temp * pmax))
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if weights[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        w = 2.0 * weights[idx]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # sg[m] += w * a[k-m] for m in [lo, hi]
        # a[k-m] for m in [lo, hi] is a[k-hi], a[k-hi+1], ..., a[k-lo] reversed
        # = a[hi-lo]... no wait.
        # m goes from lo to hi, k-m goes from k-lo down to k-hi
        # So a[k-m] = a[k-lo], a[k-lo-1], ..., a[k-hi]
        # That's a[k-hi : k-lo+1] reversed = a[hi-lo : -1]... 
        # Let me just index directly:
        m_range = np.arange(lo, hi + 1)
        a_vals = a[k - m_range]
        sg[m_range] += w * a_vals
    
    return sg, pmax

def solve_lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)_k <= M for all k."""
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a (2n-1) x n matrix
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # A_ub[k, j] = a[k-j] for j in [lo, hi]
        # j goes from lo to hi, k-j goes from k-lo to k-hi
        A_ub[k, lo:hi+1] = a[k - hi : k - lo + 1][::-1]
    
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_lp(a_init, time_limit, blend_init=0.4, blend_min=0.005,
                blend_halve_every=10, verbose=False):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    uniform = np.ones(n) / n
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        # Try slightly different M values for diversity
        M_try = M * (1.0 + 0.01 * np.random.default_rng().integers(0, 3))
        
        g = solve_lp_direction(a, M_try, n)
        if g is None:
            g = solve_lp_direction(a, M, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # If LP solution is too sparse, blend with uniform
        if g.max() > 0.5:
            g = 0.7 * g + 0.3 * uniform
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP iter: obj={best_obj:.8f}")
        else:
            stall += 1
        
        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
        
        if stall > blend_halve_every * 4:
            rng_local = np.random.default_rng()
            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            stall = 0
    
    return best_a, best_obj


def optimize_nesterov_subgrad(a0, time_limit, lr_init=0.02, momentum_init=0.85,
                              top_k=30, temp=0.05, verbose=False):
    """
    Nesterov-accelerated subgradient descent on the simplex.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Nesterov: y_k = a_k + (t_k - 1)/t_{k+1} * (a_k - a_{k-1})
    # a_{k+1} = project(y_k - lr * grad(y_k))
    a_prev = a.copy()
    momentum = momentum_init
    lr = lr_init
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    t_k = 1.0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        t_k += 1
        alpha = (t_k - 1) / (t_k + 0.5)  # or use momentum directly
        y = a + alpha * (a - a_prev)
        y = project_simplex(y)
        
        conv_y = fast_convolve(y)
        sg, pmax = compute_subgradient(y, conv_y, top_k=top_k, temp=temp)
        
        a_prev = a.copy()
        a = project_simplex(y - lr * sg)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and int(t_k) % 100 == 0:
                print(f"    Subgrad iter {int(t_k)}: obj={best_obj:.8f}")
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng_local = np.random.default_rng()
            a = project_simplex(a + rng_local.standard_normal(n) * 0.005)
            a_prev = a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            lr = max(0.002, lr * 0.7)
        
        lr *= 0.9998
        momentum = 0.85 + 0.1 * (t_k / (t_k + 500))
    
    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle (Fejer-like)
    tri = 1.0 - np.abs(x - mid) / (mid + 0.5)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # U-shape (high at edges, low in middle)
    u_shape = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    u_shape = np.maximum(u_shape, 0.01)
    inits['u_shape'] = u_shape / u_shape.sum()
    
    # Bell (Gaussian)
    g = np.exp(-0.5 * ((x - mid) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Sine (one period)
    s = 0.5 * (1 + np.sin(np.pi * x / n * 2))
    s = np.maximum(s, 0.01)
    inits['sine'] = s / s.sum()
    
    # Cosine squared
    c = np.cos(np.pi * (x - mid) / n) ** 2
    c = np.maximum(c, 0.01)
    inits['cos2'] = c / c.sum()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10 and len(parent) != n:
            p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
            p_interp = np.maximum(p_interp, 0)
            pis = p_interp.sum()
            if pis > 1e-10:
                inits['parent'] = p_interp / pis
        elif ps > 1e-10:
            inits['parent'] = p / ps
    
    return inits


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_interp / s


def coordinate_descent_sampled(a_in, max_rounds=50, n_samples=200, pct=0.1):
    """Sampled coordinate descent: try moving mass between random pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    rng = np.random.default_rng()
    
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_samples):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-8:
                continue
            delta = a[i] * pct * rng.random()
            if delta < 1e-9:
                continue
            
            a_test = a.copy()
            a_test[i] -= delta
            a_test[j] += delta
            a_test = np.maximum(a_test, 0.0)
            st = a_test.sum()
            if st < 1e-15:
                continue
            a_test /= st
            conv_test = fast_convolve(a_test)
            obj_test = 2.0 * n * conv_test.max()
            if obj_test < best_obj - 1e-14:
                a = a_test
                best_obj = obj_test
                conv = conv_test
                improved = True
        if not improved:
            break
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional optimization,
Nesterov-accelerated subgradient descent with dual-guided weighting, multi-scale
interpolation, and adaptive random restarts. Key improvements: adaptive top-k
subgradient with LP dual information, U-shape and asymmetric initializations,
multi-scale interpolation between lengths, and more aggressive exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=30, temp=0.05):
    """
    Compute a weighted subgradient of max_k R(k) where R = conv(a,a).
    Uses soft weighting based on peak proximity.
    Returns (gradient_vector, peak_value).
    """
    n = len(a)
    if top_k >= len(conv):
        top_k = len(conv)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pmax = conv[peaks].max()
    if pmax < 1e-15:
        return np.zeros(n), pmax

    diffs = conv[peaks] - pmax
    weights = np.exp(diffs / (temp * pmax))
    weights /= weights.sum()

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if weights[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        w = 2.0 * weights[idx]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        m_range = np.arange(lo, hi + 1)
        a_vals = a[k - m_range]
        sg[m_range] += w * a_vals

    return sg, pmax


def solve_lp_direction(a, M, n, time_limit_lp=8.0):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)_k <= M for all k.
    Returns (g, dual_weights) or (None, None)."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A_ub[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit_lp})
        if result.success:
            g = np.maximum(result.x, 0.0)
            dual = np.maximum(getattr(result, 'ineqlin', None) is None and np.zeros(2 * n - 1)
                              or (result.ineqlin if hasattr(result, 'ineqlin') else np.zeros(2 * n - 1)), 0)
            if hasattr(result, 'ineqlin') and result.ineqlin is not None:
                dual = np.maximum(result.ineqlin, 0)
            else:
                dual = np.zeros(2 * n - 1)
            return g, dual
    except Exception:
        pass
    return None, None


def optimize_lp(a_init, time_limit, blend_init=0.4, blend_min=0.005,
                blend_halve_every=10, verbose=False):
    """LP-based directional optimization with adaptive blending and dual-guided subgrad."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    uniform = np.ones(n) / n

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        it_count += 1

        g, dual = solve_lp_direction(a, M, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if g.max() > 0.5:
            g = 0.7 * g + 0.3 * uniform

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            if verbose and it_count % 20 == 0:
                print(f"      LP it={it_count}: obj={best_obj:.8f}")
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                # Use subgradient step with dual info before breaking
                if dual is not None and dual.sum() > 0:
                    sg, _ = compute_subgradient(a, conv, top_k=20, temp=0.03)
                    a = project_simplex(a - 0.005 * sg)
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                break

        if stall > blend_halve_every * 5:
            rng_local = np.random.default_rng(it_count)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            stall = 0

    return best_a, best_obj


def optimize_nesterov_subgrad(a0, time_limit, lr_init=0.02,
                              top_k=30, temp=0.05, verbose=False):
    """
    Nesterov-accelerated subgradient descent on the simplex.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    a_prev = a.copy()
    lr = lr_init

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    t_k = 1.0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        t_k += 1
        it_count += 1
        alpha = min(0.95, (t_k - 1) / (t_k + 0.5))
        y = a + alpha * (a - a_prev)
        y = project_simplex(y)

        # Adapt top_k and temp based on iteration count
        adapt_k = min(100, top_k + it_count // 50)
        adapt_temp = temp * max(0.3, 1.0 - it_count / 5000)

        conv_y = fast_convolve(y)
        sg, pmax = compute_subgradient(y, conv_y, top_k=adapt_k, temp=adapt_temp)

        a_prev = a.copy()
        a = project_simplex(y - lr * sg)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it_count % 500 == 0:
                print(f"      Subgrad it={it_count}: obj={best_obj:.8f}")
        else:
            no_improve += 1

        if no_improve > 500:
            rng_local = np.random.default_rng()
            a = project_simplex(a + rng_local.standard_normal(n) * 0.004)
            a_prev = a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            lr = max(0.001, lr * 0.6)

        lr *= 0.9999

    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):

def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_interp / s


def coordinate_descent_sampled(a_in, max_rounds=50, n_samples=300, pct=0.1, seed=0):
    """Sampled coordinate descent: try moving mass between random pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_samples):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-8:
                continue
            delta = a[i] * pct * rng.random()
            if delta < 1e-9:
                continue
            a_test = a.copy()
            a_test[i] -= delta
            a_test[j] += delta
            a_test = np.maximum(a_test, 0.0)
            st = a_test.sum()
            if st < 1e-15:
                continue
            a_test /= st
            conv_test = fast_convolve(a_test)
            obj_test = 2.0 * n * conv_test.max()
            if obj_test < best_obj - 1e-14:
                a = a_test
                best_obj = obj_test
                conv = conv_test
                improved = True
        if not improved:
            break

    return a
