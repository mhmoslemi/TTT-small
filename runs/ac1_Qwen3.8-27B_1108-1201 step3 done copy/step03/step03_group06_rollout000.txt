import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a):
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def solve_lp(a, M, constraint_indices=None):
    n = len(a)
    if constraint_indices is not None:
        A_ub = build_lp_matrix(a)[constraint_indices]
        b_ub = np.full(len(constraint_indices), M)
    else:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M, constraint_indices=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for specified k."""
    n = len(a)
    if constraint_indices is None:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(2 * n - 1, M)
    else:
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[constraint_indices]
        b_ub = np.full(len(constraint_indices), M)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def get_active_constraints(conv, frac=0.92):
    """Get indices of peaks that are within frac of the max."""
    M = conv.max()
    if M <= 0:
        return np.arange(len(conv))
    threshold = frac * M
    return np.where(conv >= threshold)[0]


def lp_step_with_backtrack(a, alpha=1.0, conv=None, max_backtrack=8):
    """Single LP step with backtracking line search. Returns (new_a, new_M, improved)."""
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    M = conv.max()

    # Use all constraints for accuracy (n=320 means 639 constraints, manageable)
    b = solve_lp_direction(a, alpha * M)
    if b is None:
        # Fallback: use only active constraints
        active = get_active_constraints(conv, 0.90)
        b = solve_lp_direction(a, alpha * M, constraint_indices=active)
    if b is None:
        return a, M, False

    b_sum = b.sum()
    if b_sum < 1e-15:
        return a, M, False
    b = b / b_sum

    # Backtracking line search
    best_a = a.copy()
    best_M = M
    t = 1.0
    for _ in range(max_backtrack):
        a_try = (1.0 - t) * a + t * b
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            t *= 0.5
            continue
        a_try = a_try / s
        M_try = fast_convolve(a_try).max()
        if M_try < best_M - 1e-14:
            best_M = M_try
            best_a = a_try.copy()
            if M_try < M * (1.0 - 0.01 * t):
                # Sufficient decrease
                break
        t *= 0.5
        if t < 1e-7:
            break

    improved = best_M < M - 1e-12
    return best_a, best_M, improved


def smooth_power_gradient(a, p=20.0):
    """Compute gradient using power mean smooth approximation of max."""
    n = len(a)
    conv = fast_convolve(a)
    positive_conv = np.maximum(conv, 1e-30)
    # Power mean: (sum conv^p)^(1/p) ~ max for large p
    conv_p = positive_conv ** p
    conv_p_sum = conv_p.sum()
    if conv_p_sum < 1e-30:
        return np.zeros(n), conv
    w = conv_p / conv_p_sum

    # Gradient of sum_k w_k * (a*a)_k where w_k = (a*a)_k^p / sum_j (a*a)_j^p
    # Simplified: use weighted average of gradients
    # grad of (a*a)_k w.r.t. a[i] = a[k-i] + a[i-k] (when indices valid)
    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.rfft(w_padded, fft_size)
    A = np.fft.rfft(a_padded, fft_size)

    # term1: sum_k w[k] * a[k-i] for valid indices
    # = cross-correlation
    cross = np.fft.irfft(A * np.conj(W), fft_size)[:n]
    # term2: sum_k w[k] * a[i-k] = convolution
    conv_term = np.fft.irfft(A * W, fft_size)[:n]

    grad = cross + conv_term
    return grad, conv


def smooth_gradient_step(a, lr=0.01, momentum_arr=None, p=20.0):
    """One smooth gradient descent step on the simplex."""
    n = len(a)
    grad, conv = smooth_power_gradient(a, p)
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-15:
        return a, conv.max(), momentum_arr, False

    grad = grad / g_norm
    if momentum_arr is None:
        momentum_arr = np.zeros(n)
    else:
        momentum_arr = 0.9 * momentum_arr
    momentum_arr = momentum_arr - lr * grad
    a_new = project_simplex(a + momentum_arr)
    M_new = fast_convolve(a_new).max()
    M_old = conv.max()
    improved = M_new < M_old - 1e-12
    return a_new, M_new, momentum_arr, improved


def deep_optimize_lp(a0, time_limit, seed=0, verbose=True):
    """
    Deep LP-based optimization with adaptive relaxation and backtracking.
    Returns (best_a, best_objective).
    """
    rng = np.random.default_rng(seed)
    a = project_simplex(np.maximum(a0.astype(np.float64), 0.0))
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    alphas = [1.0, 1.01, 1.03, 1.06, 1.10, 1.15]
    alpha_ptr = 0
    no_improve = 0
    stuck_count = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Try relaxation levels starting from current pointer
        found_improvement = False
        for ai in range(alpha_ptr, len(alphas)):
            a_new, M_new, improved = lp_step_with_backtrack(a, alpha=alphas[ai])
            if improved:
                a = a_new
                M = M_new
                if M < best_M - 1e-12:
                    best_M = M
                    best_a = a.copy()
                no_improve = 0
                stuck_count = 0
                alpha_ptr = 0
                found_improvement = True
                break
            elif M_new < M - 1e-14:
                # Accept even tiny improvements
                a = a_new
                M = M_new
                if M < best_M - 1e-12:
                    best_M = M
                    best_a = a.copy()
                no_improve = 0
                alpha_ptr = 0
                found_improvement = True
                break
        if not found_improvement:
            no_improve += 1
            alpha_ptr = min(alpha_ptr + 1, len(alphas))

            if no_improve >= len(alphas):
                # Tried all relaxation levels without success
                stuck_count += 1
                no_improve = 0
                alpha_ptr = 0

                if stuck_count > 3:
                    # Perturb to escape local minimum
                    scale = 0.002 + 0.003 * min(stuck_count, 5)
                    noise = rng.standard_normal(n) * scale
                    a = project_simplex(a + noise)
                    M = fast_convolve(a).max()
                    # Reset momentum
                    stuck_count = 0
                elif stuck_count == 2:
                    # Try a gradient step
                    a_new, M_new, _, _ = smooth_gradient_step(a, lr=0.02, p=30.0)
                    if M_new < fast_convolve(a).max():
                        a = a_new
                        M = M_new

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_M
            print(f"    LP iter {iter_count}: obj={obj:.6f}, M={M:.8f}, t={elapsed:.1f}s")

    final_obj = 2.0 * n * best_M
    return best_a, final_obj


def quick_lp_optimize(a0, time_limit, seed=0):
    """Quick LP optimization for exploration phase."""
    return deep_optimize_lp(a0, time_limit, seed=seed, verbose=False)


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1)) ** 2
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def polish_gradient(a0, time_limit, seed=0):
    """Polish using smooth gradient descent with adaptive learning rate."""
    rng = np.random.default_rng(seed)
    a = project_simplex(np.maximum(a0.astype(np.float64), 0.0))
    n = len(a)
    M = fast_convolve(a).max()
    best_a = a.copy()
    best_M = M
    v = np.zeros(n)
    lr = 0.01
    no_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        a_new, M_new, v, improved = smooth_gradient_step(a, lr=lr, momentum_arr=v, p=50.0)
        if improved:
            a = a_new
            M = M_new
            if M < best_M - 1e-12:
                best_M = M
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.7
                v *= 0.5
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                M = fast_convolve(a).max()
                v = np.zeros(n)
                lr = 0.01
                no_improve = 0

    return best_a, 2.0 * n * best_M

    Multi-phase optimization:
    Phase 1: Quick exploration across lengths (60s)
    Phase 2: Deep LP optimization on best candidate (1300s)
    Phase 3: Gradient polish + perturbations (350s)
    Phase 4: Final verification (remaining)

    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Exploration ==========
    print("=== Phase 1: Quick Exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = 60.0

    lengths = [256, 320, 384]
    scan_results = []

    for n in lengths:
        if time.monotonic() + 5 > t_phase1 + phase1_time:
            break
        inits = generate_initializations(n, parent, rng)
        # Prioritize parent and diverse structures
        priority = ['parent', 'parent_ds', 'parent_up', 'bimodal', 'uniform',
                    'gauss_center', 'random', 'cosine']
        init_names = [nm for nm in priority if nm in inits][:4]

        time_per_init = max(3.0, (phase1_time - (time.monotonic() - t_phase1) - 5) /
                            max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() + 3 > t_phase1 + phase1_time:
                break
            t_alloc = min(time_per_init, t_phase1 + phase1_time - time.monotonic() - 3)
            if t_alloc < 2:
                break

            a0 = inits[name]
            a_opt, obj_opt = quick_lp_optimize(a0, time_limit=t_alloc, seed=n + hash(name) % 1000)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best so far: {best_score:.6f}")

    if best_sequence is None:
        # Fallback: use parent or uniform
        if parent is not None and len(parent) > 0:
            best_sequence = np.maximum(parent, 0)
            s = best_sequence.sum()
            if s > 1e-10:
                best_sequence = best_sequence / s
        else:
            best_sequence = np.ones(320) / 320
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = min(1300.0, (deadline - time.monotonic()) * 0.7)

    n_best = len(best_sequence)
    print(f"  Optimizing n={n_best}, starting score={best_score:.6f}")

    # Run deep LP optimization
    a_opt, obj_opt = deep_optimize_lp(best_sequence, time_limit=phase2_time,
                                      seed=12345, verbose=True)

    score = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_sequence = a_opt.copy()
        print(f"  Deep LP improved: score={score:.6f}")

    print(f"Phase 2 done. Best so far: {best_score:.6f}")

    # ========== Phase 3: Gradient Polish + Perturbations ==========
    print("=== Phase 3: Gradient Polish ===")
    phase3_start = time.monotonic()
    remaining = deadline - time.monotonic() - 30.0
    phase3_time = min(350.0, remaining * 0.6)

    if phase3_time > 20:
        # Gradient polish
        t_grad = phase3_time * 0.5
        a_opt, obj_opt = polish_gradient(best_sequence, time_limit=t_grad, seed=42)
        score = float(evaluate_sequence(a_opt.tolist()))

            t_alloc = min(t_lp, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc < 5:
                break
            a_opt, obj_opt = deep_optimize_lp(a_pert, time_limit=t_alloc,
                                              seed=100 + i, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Perturb LP [{i}]: score={score:.6f}")

        # Tiny random perturbations
        remaining2 = phase3_start + phase3_time - time.monotonic()
        if remaining2 > 10:
            n_tries = int(remaining2 / 8)
            n_tries = min(n_tries, 15)
            for i in range(n_tries):
                if time.monotonic() > phase3_start + phase3_time - 5:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.004]:
                    noise = rng.standard_normal(n_cur) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Tiny perturb [{i}] s={scale}: score={score:.6f}")
                        break

    print(f"Phase 3 done. Best so far: {best_score:.6f}")

    # ========== Phase 4: Final Polish ==========
    print("=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic() - 20.0
    if remaining > 30 and best_sequence is not None:
        n_cur = len(best_sequence)
        # One more LP round with very small steps
        t_alloc = min(remaining * 0.6, 100.0)
        if t_alloc > 10:
            a_opt, obj_opt = deep_optimize_lp(best_sequence, time_limit=t_alloc,
                                              seed=99999, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")

        # Final gradient

                best_sequence = a_opt.copy()
