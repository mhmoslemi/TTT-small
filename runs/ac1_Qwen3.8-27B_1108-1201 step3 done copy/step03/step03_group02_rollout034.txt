"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP best-response with optimal blending, greedy coordinate descent,
smoothed projected gradient, and multi-scale upscaling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(np.asarray(v, dtype=np.float64))[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(np.asarray(v, dtype=np.float64) - theta, 0.0)


def make_conv_matrix(a):
    """Build the convolution matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(num_rows - 1, j + n - 1)
        length = hi - lo + 1
        A[lo:hi + 1, j] = a[:length]
    return A

    n = len(a)
    A = make_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def optimal_blend_factor(conv_a, conv_ab, conv_b):
    """Find t in [0,1] minimizing max_k of blended convolution."""
    # f_k(t) = (1-t)^2 * conv_a[k] + 2t(1-t)*conv_ab[k] + t^2 * conv_b[k]
    # = conv_a[k] + t*(2*conv_ab[k] - 2*conv_a[k]) + t^2*(conv_b[k]-2*conv_ab[k]+conv_a[k])
    A_arr = conv_a
    B_arr = 2.0 * conv_ab - 2.0 * conv_a
    C_arr = conv_b - 2.0 * conv_ab + conv_a

    def f(t):
        return np.max(A_arr + t * B_arr + t * t * C_arr)

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    fc = f(c_pt)
    fd = f(d_pt)
    for _ in range(50):
        if fc < fd:
            a_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = a_hi - gr * (a_hi - a_lo)
            fc = f(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_lo + gr * (a_hi - a_lo)
            fd = f(d_pt)
    return (a_lo + a_hi) / 2.0


def lp_iteration(a, time_limit, max_iters=2000):
    """LP best-response iteration with optimal blending."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a), dtype=np.float64) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_peak = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        b = solve_lp_max_sum(a, M)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        if np.linalg.norm(b - a, ord=np.inf) < 1e-12:
            break

        # Compute optimal blend factor
        conv_ab = fast_cross_conv(a, b)
        conv_b = fast_convolve(b)
        t = optimal_blend_factor(conv, conv_ab, conv_b)
        t = np.clip(t, 0.01, 0.99)

        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        peak_new = conv_new.max()

        if peak_new < best_peak - 1e-12:
            best_peak = peak_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        if no_improve > 100:
            break

    return best_a, 2.0 * n * best_peak


def greedy_cd_step(a, conv, k_star):
    """One greedy coordinate descent step: find best swap and optimal amount."""
    n = len(a)
    # damage[i] = contribution of a[i] to conv[k_star]
    # conv[k_star] = sum_j a[j] * a[k_star - j]
    # d(conv[k_star])/d(a[i]) involves a[k_star - i]
    
    # For the swap from i_max to i_min:
    # We want to find i that maximizes a[k_star - i] and j that minimizes a[k_star - j]
    
    damage = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        damage[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]
    
    i_max = int(np.argmax(damage))
    i_min = int(np.argmin(damage))
    
    if damage[i_max] <= damage[i_min] + 1e-15:
        return None, None
    
    # Compute delta_v[k] = a[k - i_min] - a[k - i_max] for k in [0, 2n-2]
    delta_v = np.zeros(2 * n - 1)
    # a[k - i] is nonzero when 0 <= k-i < n, i.e., i <= k <= i+n-1
    for i, sign in [(i_max, -1.0), (i_min, 1.0)]:
        lo_k = i
        hi_k = min(2 * n - 2, i + n - 1)
        delta_v[lo_k:hi_k + 1] += sign * a[:hi_k - lo_k + 1]
    
    # f(d) = max_k (conv[k] + 2*d*delta_v[k]) for d in [0, a[i_max]]
    # This is convex in d. Find minimum.
    d_max = a[i_max]
    if d_max < 1e-15:
        return None, None
    
    # Bisection on subgradient
    # subgradient at d = 2 * delta_v[k*] where k* = argmax(conv + 2d*delta_v)
    # Find d where subgradient crosses 0
    
    lo_d = 0.0
    hi_d = d_max
    
    # Check endpoints
    f_lo = (conv + 2.0 * lo_d * delta_v).max()
    f_hi = (conv + 2.0 * hi_d * delta_v).max()
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_d = hi_d - gr * (hi_d - lo_d)
    d_d = lo_d + gr * (hi_d - lo_d)
    fc = (conv + 2.0 * c_d * delta_v).max()
    fd = (conv + 2.0 * d_d * delta_v).max()
    
    for _ in range(40):
        if fc < fd:
            hi_d = d_d
            d_d = c_d
            fd = fc
            c_d = hi_d - gr * (hi_d - lo_d)
            fc = (conv + 2.0 * c_d * delta_v).max()
        else:
            lo_d = c_d
            c_d = d_d
            fc = fd
            d_d = lo_d + gr * (hi_d - lo_d)
            fd = (conv + 2.0 * d_d * delta_v).max()
    
    d_opt = (lo_d + hi_d) / 2.0
    f_opt = (conv + 2.0 * d_opt * delta_v).max()
    
    if f_opt >= f_lo - 1e-14:
        return None, None
    
    # Apply the swap
    a_new = a.copy()
    a_new[i_max] = a[i_max] - d_opt
    a_new[i_min] = a[i_min] + d_opt
    # Renormalize (the swap preserves sum, but let's be safe)
    s = a_new.sum()
    if s < 1e-15:
        return None, None
    a_new = a_new / s
    return a_new, d_opt

                    conv_test = fast_convolve(a_test)

Hmm, the gradient computation in the smoothed gradient descent is O(n^2) because of the double loop. Let me optimize it.

Actually, the gradient of `conv(a,a)` with respect to `a` can be computed as:
`d(conv(a,a)[k])/d(a[i]) = a[k-i] + a[i-k]` (with a[m]=0 for m not in [0,n-1])

So the full gradient matrix (Jacobian) has entries `J[k,i] = a[k-i] + a[i-k]`.

And the smoothed gradient is: `grad[i] = (2n/t) * sum_k p[k] * J[k,i] = (2n/t) * sum_k p[k] * (a[k-i] + a[i-k])`

  This is a cross-correlation of p with a!

  This is a convolution of p with a!

So `grad[i] = (2n/t) * (g1[i] + g2[i])` where g1 is cross-correlation and g2 is convolution of p and a.

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP best-response with optimal blending, greedy coordinate descent
with exact line search, FFT-accelerated smoothed gradient descent,
and multi-scale upscaling to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def make_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a,b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(num_rows - 1, j + n - 1)
        length = hi - lo + 1
        A[lo:hi + 1, j] = a[:length]
    return A


def solve_lp_max_sum(a, M):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = make_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def optimal_blend_factor(conv_a, conv_ab, conv_b):
    """Find t in [0,1] minimizing max_k of (1-t)^2*conv_a + 2t(1-t)*conv_ab + t^2*conv_b."""
    A_arr = conv_a
    B_arr = 2.0 * conv_ab - 2.0 * conv_a
    C_arr = conv_b - 2.0 * conv_ab + conv_a

    def f_of_t(t):
        return np.max(A_arr + t * B_arr + t * t * C_arr)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    fc = f_of_t(c_pt)
    fd = f_of_t(d_pt)
    for _ in range(50):
        if fc < fd:
            a_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = a_hi - gr * (a_hi - a_lo)
            fc = f_of_t(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_lo + gr * (a_hi - a_lo)
            fd = f_of_t(d_pt)
    return (a_lo + a_hi) / 2.0


def lp_iteration(a, time_limit, max_iters=5000):
    """LP best-response iteration with optimal blend factor."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a), dtype=np.float64) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_peak = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        b = solve_lp_max_sum(a, M)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        if np.linalg.norm(b - a, ord=np.inf) < 1e-12:
            break

        conv_ab = fast_cross_conv(a, b)
        conv_b = fast_convolve(b)
        t = optimal_blend_factor(conv, conv_ab, conv_b)
        t = np.clip(t, 0.02, 0.98)

        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        peak_new = conv_new.max()

        if peak_new < best_peak - 1e-12:
            best_peak = peak_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        if no_improve > 150:
            break

    obj = 2.0 * n * best_peak
    return best_a, obj


def greedy_cd_single_step(a, conv, k_star):
    """One greedy coordinate descent step: find best (i,j) swap and optimal d."""
    n = len(a)
    # damage[i] = a[k_star - i] if 0 <= k_star - i < n, else 0
    damage = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        damage[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]

    i_max = int(np.argmax(damage))
    i_min = int(np.argmin(damage))

    if damage[i_max] <= damage[i_min] + 1e-15:
        return None

    # delta_v[k] = a[k - i_min] - a[k - i_max] for k in [0, 2n-2]
    delta_v = np.zeros(2 * n - 1)
    for i, sign in [(i_max, -1.0), (i_min, 1.0)]:
        lo_k = i
        hi_k = min(2 * n - 2, i + n - 1)
        length = hi_k - lo_k + 1
        delta_v[lo_k:hi_k + 1] += sign * a[:length]

    d_max = a[i_max]
    if d_max < 1e-15:
        return None

    # Minimize f(d) = max_k(conv[k] + 2d*delta_v[k]) for d in [0, d_max]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_d = 0.0
    hi_d = d_max
    c_d = hi_d - gr * (hi_d - lo_d)
    d_d = lo_d + gr * (hi_d - lo_d)
    fc = (conv + 2.0 * c_d * delta_v).max()
    fd = (conv + 2.0 * d_d * delta_v).max()
    for _ in range(40):
        if fc < fd:
            hi_d = d_d
            d_d = c_d
            fd = fc
            c_d = hi_d - gr * (hi_d - lo_d)
            fc = (conv + 2.0 * c_d * delta_v).max()
        else:
            lo_d = c_d
            c_d = d_d
            fc = fd
            d_d = lo_d + gr * (hi_d - lo_d)
            fd = (conv + 2.0 * d_d * delta_v).max()

    d_opt = (lo_d + hi_d) / 2.0
    f_opt = (conv + 2.0 * d_opt * delta_v).max()
    f_at_0 = conv.max()

    if f_opt >= f_at_0 - 1e-14:
        return None

    a_new = a.copy()
    a_new[i_max] = a[i_max] - d_opt
    a_new[i_min] = a[i_min] + d_opt
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    a_new = a_new / s
    return a_new

def compute_smoothed_grad(a, p, n, t):
    """Compute gradient of smoothed objective using FFT. O(n log n)."""
    N = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)

    # g2[i] = (a * p)[i] for i=0..n-1 (linear convolution)
    conv_ap = np.fft.irfft(A * P, N)[:n]

    # g1[i] = (a * p_rev)[2n-2-i] where p_rev = p[::-1]
    P_rev = np.fft.rfft(p[::-1], N)
    conv_aq = np.fft.irfft(A * P_rev, N)
    g1 = conv_aq[2 * n - 2:n - 1: -1]

    return (2.0 * n / t) * (g1 + g2) if False else (2.0 * n / t) * (g1 + conv_ap)
