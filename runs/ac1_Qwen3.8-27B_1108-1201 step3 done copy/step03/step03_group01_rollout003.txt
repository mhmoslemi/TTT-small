import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def binomial_init(n, rng=None):
    """Initialize with binomial coefficient sequence C(2k,k)/4^k."""
    k = np.arange(n)
    # C(2k,k)/4^k = (2k)!/(k!k!4^k)
    # Using log-gamma for numerical stability
    from scipy.special import gammaln
    log_a = gammaln(2*k+1) - 2*gammaln(k+1) - k*np.log(4)
    a = np.exp(log_a)
    a /= a.sum()
    return a

def smoothed_gradient(a, beta):
    """Compute gradient of smoothed max of conv(a,a)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    
    # Softmax weights
    scaled = beta * conv
    scaled -= scaled.max()
    b = np.exp(scaled)
    b_sum = b.sum()
    if b_sum < 1e-30:
        b = np.zeros(size)
        b[np.argmax(conv)] = 1.0
    else:
        b /= b_sum
    
    # Gradient: grad_i = 2 * sum_k b_k * a_{k-i}
    # = 2 * ifft(conj(fft(a)) * fft(b))[:n]
    # Using full FFT for cross-correlation
    N = 1 << (2 * size - 1).bit_length()  # Large enough for cross-correlation
    Af = np.fft.fft(a, N)
    Bf = np.fft.fft(b, N)
    grad = 2.0 * np.fft.ifft(np.conj(Af) * Bf).real[:n]
    
    return grad, conv

def multiplicative_update(a, grad, lr):
    """Multiplicative update: a_i *= exp(-lr * grad_i), then normalize."""
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def gradient_optimize(a0, time_limit, beta_init=10.0, beta_final=1000.0, lr_init=0.1, verbose=False):
    """Optimize using smoothed gradient with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        grad, conv = smoothed_gradient(a, beta)
        
        # Adaptive learning rate
        a_new = multiplicative_update(a, grad, lr)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.9
        
        # Increase beta gradually (make smoothing sharper)
        if no_improve < 100:
            beta *= 1.001
        else:
            beta *= 1.005
        
        if no_improve > 500:
            break
        
        if verbose and no_improve % 100 == 0:
            print(f"  Grad iter, obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}")
    
    return best_a, best_obj

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def solve_lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0

def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    
    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        M_lp = M * 1.001
        g, g_sum = solve_lp_direction(a, M_lp)
        
        if g_sum > 1.0 + 1e-7:
            d = g - a
            # Try a few step sizes
            best_t = 0.0
            best_v = best_obj
            for t in [0.5, 0.3, 0.1, 0.05, 0.01]:
                a_try = np.maximum(a + t * d, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                v_try = 2.0 * n * conv_try.max()
                if v_try < best_v:
                    best_v = v_try
                    best_t = t
            
            if best_t > 0:
                a = np.maximum(a + best_t * d, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                best_obj = 2.0 * n * M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve >= 100:
            rng = np.random.default_rng(hash(str(no_improve)))
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(best_a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            if perturb_count > 50:
                break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Binomial-coefficient initialization (approximates flat autocorrelation)
2. Fast FFT-based gradient descent with softmax-smoothed max and multiplicative updates
3. LP-based direction finding for final refinement
4. Multi-length search with adaptive time allocation and parent sequence seeding
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import gammaln


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def binomial_init(n):
    """Initialize with binomial coefficient sequence C(2k,k)/4^k, normalized."""
    k = np.arange(n, dtype=np.float64)
    log_a = gammaln(2 * k + 1) - 2 * gammaln(k + 1) - k * np.log(4.0)
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def windowed_binomial_init(n, window='hamming'):
    """Binomial coefficients with a window to reduce edge discontinuity."""
    a = binomial_init(n)
    if window == 'hamming':
        w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) if n > 1 else np.ones(1)
        a = a * w
    elif window == 'hanning':
        w = 0.5 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) if n > 1 else np.ones(1)
        a = a * w
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def smoothed_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv_k)) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]

    scaled = beta * conv
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    b_sum = exp_s.sum()
    if b_sum < 1e-30:
        b = np.zeros(size)
        b[np.argmax(conv)] = 1.0
    else:
        b = exp_s / b_sum

    # grad_i = 2 * sum_k b_k * a_{k-i} (a_j = 0 for j outside [0,n-1])
    N = 1 << (2 * size - 1).bit_length()
    Af = np.fft.fft(a, N)
    Bf = np.fft.fft(b, N)
    grad = 2.0 * np.fft.ifft(np.conj(Af) * Bf).real[:n]

    return grad, conv


def multiplicative_update(a, grad, lr):
    """Multiplicative update maintaining non-negativity, then normalize."""
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-20:
        return a.copy()
    return a_new / s


def gradient_optimize(a0, time_limit, beta_init=5.0, beta_max=5000.0, lr_init=0.05, verbose=False):
    """Optimize using smoothed gradient with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv = smoothed_gradient(a, beta)

        a_new = multiplicative_update(a, grad, lr)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            a = a_new
            conv = conv_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            # Slightly increase lr if improving consistently
            if no_improve == 0 and lr < 0.1:
                lr *= 1.001
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 0.001:
                lr = 0.001

        # Increase beta (sharpen the max approximation)
        if no_improve < 50:
            beta *= 1.0005
        elif no_improve < 200:
            beta *= 1.002
        else:
            beta *= 1.005
        beta = min(beta, beta_max)

        iter_count += 1
        if no_improve > 800:
            # Try a perturbation
            rng = np.random.default_rng(iter_count * 7)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            no_improve = 0
            lr = lr_init * 0.5
            beta = max(beta_init, beta * 0.5)

        if verbose and iter_count % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Grad iter {iter_count}, obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        M_lp = M * 1.0005
        g, g_sum = solve_lp_direction(a, M_lp)

        if g_sum > 1.0 + 1e-7:
            d = g - a
            best_t = 0.0
            best_v = best_obj
            for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
                a_try = np.maximum(a + t * d, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                v_try = 2.0 * n * conv_try.max()
                if v_try < best_v - 1e-14:
                    best_v = v_try
                    best_t = t

            if best_t > 0:
                a = np.maximum(a + best_t * d, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                best_obj = 2.0 * n * M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iter_count += 1

        if no_improve >= 80:
            rng = np.random.default_rng(iter_count * 13 + 7)
            scale = 0.003 + 0.002 * min(perturb_count, 5)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            if perturb_count > 40:
                break

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['binomial'] = binomial_init(n)
    inits['binomial_ham'] = windowed_binomial_init(n, 'hamming')
    inits['binomial_hann'] = windowed_binomial_init(n, 'hanning')
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    edge = max(1, n // 3)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    inits['sine'] = (np.sin(np.pi * (x + 1) / (n + 1)) + 0.05)
    inits['sine'] = inits['sine'] / inits['sine'].sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits
