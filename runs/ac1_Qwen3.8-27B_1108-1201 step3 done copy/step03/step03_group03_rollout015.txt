def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

homotopy on temperature, multi-length exploration, and final refinement.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, tau, n):
    """
    Compute gradient of (1/tau) * log(sum exp(tau*(a*a)_k)) w.r.t. a.
    Returns (grad, conv, actual_max).
    """
    conv = fast_convolve(a)
    actual_max = conv.max()
    
    m = conv.max()
    if m < 1e-300:
        return np.zeros(n), conv, actual_max
    
    shifted = tau * (conv - m)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-300:
        weights = np.ones(len(conv)) / len(conv)
    else:
        weights = exp_shifted / w_sum
    
    # grad_j = 2 * sum_k weights[k] * a[k-j]
    # Use FFT: grad = 2 * IFFT(FFT(weights) * conj(FFT(a)))[:n]
    N = 1
    while N < 3 * n:
        N *= 2
    
    W_f = np.fft.rfft(weights, N)
    A_f = np.fft.rfft(a, N)
    grad = 2.0 * np.fft.irfft(W_f * np.conj(A_f), N)[:n]
    
    return grad, conv, actual_max

    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def fista_smoothed(a0, n, tau, max_iter=5000, tol=1e-10):

    f_val = (1.0 / tau) * np.log(np.sum(np.exp(tau * (conv - conv.max())))) + conv.max()
    f_val = f_val * n  # Include the factor of n
    
    # Estimate Lipschitz constant
    # L <= n * (2 + 4*tau) is an upper bound, but use backtracking
    L = n * (2.0 + 4.0 * tau) / 2.0  # Conservative estimate
    
    best_a = a.copy()
    best_obj = objective_value(a, n)

        alpha = 1.0 / L
        z = y - alpha * grad
        x_new = project_simplex(z)
        
        # Update y for next iteration
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)

        if it % 100 == 99:
            # Check if we're making progress
