def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    # C is (2n-1) x n

    C = np.zeros((2 * n - 1, n))

        C[i:i + n, i] = a
    return C

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute objective and convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf, np.zeros(2 * n - 1)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (S * S), conv


def make_conv_matrix(a):
    """Build matrix C such that C @ x = full_convolution(a, x)."""
    n = len(a)

    for i in range(n):

def solve_lp(a, M, margin=1.0):
    """
    Solve LP: max sum(x) s.t. conv(a,x) <= M*margin, x >= 0.
    Returns (x, sum_x, dual_vars) or (None, 0, None).
    """
    n = len(a)
    C = make_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M * margin)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            x = np.maximum(res.x, 0)
            x_sum = x.sum()
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                if hasattr(res.ineqlin, 'marginals') and res.ineqlin.marginals is not None:
                    duals = res.ineqlin.marginals
            return x, x_sum, duals
    except Exception:
        pass
    return None, 0.0, None


def golden_line_search(a, d, current_obj, n, max_evals=12):
    """
    Golden-section style line search in [0, 1].
    Returns (best_t, best_obj_value).
    """
    best_t = 0.0
    best_val = current_obj

    # Coarse grid
    candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        S = a_try.sum()
        if S < 1e-15:
            continue
        a_try = a_try / S
        obj, _ = compute_obj(a_try)
        if obj < best_val - 1e-15:
            best_val = obj
            best_t = t

    # Refine around best_t
    if 0.05 < best_t < 0.95:
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        golden = 0.6180339887
        c1 = hi - golden * (hi - lo)
        c2 = lo + golden * (hi - lo)
        evals = len(candidates)

        for _ in range(4):
            if hi - lo < 0.01 or evals >= max_evals:
                break
            f1, _ = _eval_at(a, d, c1, n)
            f2, _ = _eval_at(a, d, c2, n)
            evals += 2
            if f1 < f2:
                hi = c2
                c2 = c1
                c1 = hi - golden * (hi - lo)
            else:
                lo = c1
                c1 = c2
                c2 = lo + golden * (hi - lo)

        mid = (lo + hi) / 2
        f_mid, _ = _eval_at(a, d, mid, n)
        evals += 1
        if f_mid < best_val:
            best_val = f_mid
            best_t = mid

    return best_t, best_val


def _eval_at(a, d, t, n):

def dual_guided_perturb(best_a, duals, conv, rng, scale=0.005):

    if duals is not None:

        # For each position i, the "pressure" is sum_k duals[k] * a[k-i]

                    pressure[idx] += duals[k] * best_a[k - idx]

        if p_max > 1e-15:

        a = a + scale * d

def subgradient_polish(a0, time_limit, lr_init=0.01, momentum=0.93, top_k=30, temp=0.0003):
    """
    Subgradient descent polish with softmax-weighted peak selection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break

        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]

        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new > 1e-15:
            a = a_new / S_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp = max(temp * 1.003, 1e-8)
        momentum = min(0.96, momentum + 0.0002)

        if no_improve >= 500:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * (n - 1))) ** 2)
    inits['gaussian'] = g / g.sum()

    # Sine modulation
    s = np.sin(np.pi * (x + 0.5) / n) ** 2 + 0.1
    inits['sine'] = s / s.sum()

    # Cosine decay (focused center)
    c = np.cos(np.pi * x / (2 * n)) ** 2 + 0.05
    inits['cosine'] = c / c.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    # Structured: alternating high/low
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
    inits['alternating'] = alt / alt.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick scan across lengths and initializations
    Phase 2: Deep LP optimization on top candidates
    Phase 3: Subgradient polish
    Phase 4: Final perturbation and re-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Scan
    print("=== Phase 1: Multi-length LP Scan ===")
    t_start = time.monotonic()
    phase1_time = 400.0
    phase1_deadline = t_start + phase1_time

    lengths = [256, 512, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 50:
            break
        if n > 1024 and time.monotonic() - t_start > phase1_time * 0.6:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize parent-based and structured inits
        priority = ['parent', 'parent_interp', 'uniform', 'sine', 'cosine',
                    'random', 'gaussian', 'alternating', 'bimodal']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        t_per = max(20.0, (phase1_time - 20.0) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a0 = inits[name]
            a_opt, obj_opt = optimize_lp(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")
            else:
                print(f"  n={n}, {name}: INVALID")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic() - t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep LP on best candidates
    print("=== Phase 2: Deep LP Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates (diverse lengths)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    # Always include the overall best
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_lengths:
            top_configs.append((bn, best_sequence.copy()))

    remaining = deadline - time.monotonic() - 200.0
    n_cand = max(1, len(top_configs))
    t_per_deep = max(30.0, remaining * 0.55 / n_cand)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break

        print(f"  Deep LP candidate {i} (n={n}), t={t_alloc:.0f}s")
        a_opt, obj_opt = optimize_lp(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    -> {score:.6f}")
        else:
            print(f"    -> {score:.6f} (no improvement)")

    print(f"Phase 2 done ({time.monotonic() - t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Subgradient polish
    print("=== Phase 3: SG Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 30:
            sg_configs = [
                {'lr_init': 0.008, 'momentum': 0.93, 'top_k': 40, 'temp': 0.0002},
                {'lr_init': 0.015, 'momentum': 0.90, 'top_k': 20, 'temp': 0.0005},
                {'lr_init': 0.005, 'momentum': 0.95, 'top_k': 60, 'temp': 0.0001},
            ]
            t_per_sg = remaining / (len(sg_configs) + 1)

            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 25:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 15)
                if t_alloc < 8:
                    break
                a_opt, obj_opt = subgradient_polish(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG {ci}: {score:.6f}")

            # Final LP polish
            if time.monotonic() < deadline - 20:
                t_alloc = min(40.0, deadline - time.monotonic() - 10)
                if t_alloc > 10:
                    a_opt, obj_opt = optimize_lp(best_sequence, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final LP: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic() - t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Perturbation + re-optimize
    print("=== Phase 4: Perturb & Re-optimize ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            n_best = len(best_sequence)
            n_perturb = max(1, int(remaining / 12))
            t_per_p = remaining / n_perturb

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 4:
                    break
                if t_per_p < 4:
                    break

                scale = 0.002 * (1 + pi * 0.5)
                noise = rng.standard_normal(n_best) * scale
                a_pert = np.maximum(best_sequence + noise, 0)
                S = a_pert.sum()
                if S > 1e-15:
                    a_pert = a_pert / S

                t_alloc = min(t_per_p, deadline - time.monotonic() - 3)
                if t_alloc < 2:
                    break
                a_opt, obj_opt = optimize_lp(a_pert, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic() - t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final check
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
