```python
    """One step of mirror descent on the simplex with entropy barrier."""
    log_a = np.where(a > 0, np.log(a), -100)
    new_log_a = log_a - eta * grad
    # Shift for numerical stability
    new_log_a -= new_log_a.max()
    new_a = np.exp(new_log_a)
    new_a /= new_a.sum()
    return new_a
```

```python
        a = a_new
        t_val = t_new

        # Decay temperature
        tau = max(tau_min, tau * tau_decay)

        # Reset momentum periodically or on stall
        if no_improve > momentum_reset or int(t_val) > 5000:
            t_val = 1.0
            x_prev = a.copy()
            no_improve = 0

        # If very stuck, try a small random perturbation
        if no_improve > momentum_reset * 3:
            rng = np.random.default_rng(hash(tuple(np.round(a, 6))) % (2**31))
            a = project_simplex(a + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            t_val = 1.0
            x_prev = a.copy()
            no_improve = 0

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, eta_init=0.5, eta_decay=0.999,
                            eta_min=0.01):
    """
    Mirror descent (exponential weighting) on the simplex.
    Update: a_new[i] ∝ a[i] * exp(-eta * grad[i])
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    tau = 0.05
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_gradient(a, conv, tau)

        # Mirror descent step
        log_a = np.log(a)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        eta = max(eta_min, eta * eta_decay)

        # Occasionally increase tau (annealing)
        if no_improve > 500:
            tau = min(0.2, tau * 1.02)
            no_improve = 0

        # Decrease tau over time
        tau = max(0.005, tau * 0.9998)

    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50, delta_frac=0.03):
    """
    Targeted coordinate descent: move mass from high-contributing positions
    to low-contributing positions to reduce the convolution peak.
    """
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        conv = fast_convolve(a)
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]

        # Find positions that contribute most to the peak
        # c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contributions = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contributions[j] = a[j] * a[k]

        # Sort by contribution (descending)
        order = np.argsort(contributions)[::-1]
        top_movers = order[:max(1, n // 10)]

        # Find low-contribution positions (potential receivers)
        bottom_movers = order[-max(1, n // 10):]

        for i in top_movers:
            if a[i] < 1e-8:
                continue
            delta = a[i] * delta_frac
            for j in bottom_movers:
                if j == i or a[j] > 0.5 * a[np.argmax(a)]:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
                    delta *= 0.5  # Reduce step after improvement
                    break

        if not improved:
            # Try smaller deltas
            delta_frac *= 0.7
            if delta_frac < 0.001:
                break

    return a


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Triangle (peaked in center)
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge-heavy
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Random Dirichlet (various concentrations)
    inits['rand_low'] = rng.dirichlet(np.ones(n) * 0.2)
    inits['rand_mid'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_high'] = rng.dirichlet(np.ones(n) * 5.0)

    # "Checkerboard" - alternating high/low
    cb = np.ones(n)
    cb[::2] = 0.3
    inits['checker'] = cb / cb.sum()

    # Power spectrum inspired - try to make |FFT|^2 flat
    # Use a "chirp" sequence
    chirp = np.cos(np.pi * x * x / n)
    chirp = np.abs(chirp)
    chirp[chirp < 0.1] = 0.1
    inits['chirp'] = chirp / chirp.sum()

    # Parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            elif len(parent) > n:
                idx = np.linspace(0, len(parent) - 1, n).astype(int)
                inits['parent_ds'] = p[idx] / p[idx].sum()
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(parent)), p)
                inits['parent_up'] = p_interp / p_interp.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimum of 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Quick scan over lengths and initializations
    Phase 2: Deep Nesterov + mirror descent optimization
    Phase 3: Coordinate descent polishing
    Phase 4: Random restart perturbations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass

    # Phase 1: Quick scan
    print("=== Phase 1: Quick Scan ===")
    lengths_to_try = [256, 384, 512, 768, 1024]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths_to_try:
            lengths_to_try.append(pn)
            lengths_to_try.sort()

    results = []
    phase1_time_per = 15.0  # seconds per (length, init) pair

    for n in lengths_to_try:
        if time.monotonic() > deadline - 200:
            break
        inits = gen_initializations(n, parent, rng)

        # Priority: parent first, then diverse inits
        prio = ['parent', 'parent_ds', 'parent_up', 'uniform', 'bimodal',
                'edge', 'checker', 'chirp', 'gauss', 'rand_low', 'rand_mid']
        names = [nm for nm in prio if nm in inits]
        # Limit to top 4 per length for speed
        names = names[:4]

        for name in names:
            if time.monotonic() > deadline - 180:
                break
            t_alloc = min(phase1_time_per, deadline - time.monotonic() - 30)
            if t_alloc < 3:
                break

            a0 = inits[name]
            # Quick Nesterov
            a_opt, obj_opt = nesterov_optimize(a0, t_alloc,
                                               tau_init=0.08, tau_min=0.005,
                                               tau_decay=0.999, lr_scale=1.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in results[:6]:
        key = (n, round(score, 4))
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
        if len(top_candidates) >= 4:
            break

    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 150.0
    t_per_candidate = max(20.0, remaining / max(1, len(top_candidates) * 2))

    for n, a0 in top_candidates:
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(t_per_candidate, deadline - time.monotonic() - 100)
        if t_alloc < 10:
            break

        # Deep Nesterov with different parameters
        for tau_i, lr_s in [(0.05, 0.7), (0.15, 1.2), (0.03, 0.5)]:
            if time.monotonic() > deadline - 100:
                break
            t_s = t_alloc / 3.0
            if t_s < 5:
                break
            a_opt, obj_opt = nesterov_optimize(a0, t_s,
                                               tau_init=tau_i, tau_min=0.002,
                                               tau_decay=0.9995, lr_scale=lr_s)
            # Mirror descent refinement
            a_md, obj_md = mirror_descent_optimize(a_opt, min(8.0, t_s * 0.3),
                                                   eta_init=0.3, eta_decay=0.998)
            if obj_md < obj_opt:
                a_opt, obj_opt = a_md, obj_md

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n} tau={tau_i}: {score:.6f}")

    # Phase 3: Coordinate descent polishing
    print("=== Phase 3: Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0

        # Coordinate descent
        t_cd = min(100.0, remaining * 0.35)
        # Use time-limited CD
        a_cd = best_sequence.copy()
        t_cd_start = time.monotonic()
        for rnd in range(100):
            if time.monotonic() - t_cd_start > t_cd:
                break
            a_cd = coordinate_descent_polish(a_cd, max_rounds=5, delta_frac=0.02)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD {score_cd:.6f}")
            elif score_cd >= best_score:
                a_cd = best_sequence.copy()
                break

        # Additional Nesterov cycles
        remaining = deadline - time.monotonic() - 40.0
        for cycle in range(4):
            if time.monotonic() > deadline - 35:
                break
            t_cyc = min(remaining / (4 - cycle), 40.0)
            if t_cyc < 5:
                break

            a_opt, obj_opt = nesterov_optimize(best_sequence, t_cyc * 0.6,
                                               tau_init=0.04, tau_min=0.001,
                                               tau_decay=0.9997, lr_scale=0.6)
            a_opt2, obj2 = mirror_descent_optimize(a_opt, t_cyc * 0.25,
                                                   eta_init=0.2, eta_decay=0.998)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Cycle {cycle}: {score:.6f}")

        # Random perturbations + quick optimize
        if time.monotonic() < deadline - 25:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01, 0.02]:
                for _ in range(3):
                    if time.monotonic() > deadline - 20:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, obj_opt = nesterov_optimize(a_p, 3.0,
                                                       tau_init=0.03, tau_min=0.002,
                                                       tau_decay=0.999, lr_scale=0.8)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
