    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
    return A

def lp_solve(a, M, method='max_sum'):
    n = len(a)
    A = build_conv_matrix(a)
    if method == 'max_sum':
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0)
    else:  # min_peak
        c = np.zeros(n+1)
        c[-1] = 1.0
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2*n-1)
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x[:n], 0)
    return None

def objective_value(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s*s)

def lp_refine(a0, time_limit, step_init=0.3, max_iters=2000):
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()

    else:
        a = a / s
    n = len(a)

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    step = step_init
    no_improve = 0
    A_cache = build_conv_matrix(a)
    A_dirty = True
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()

            if b is None:

        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        if np.linalg.norm(b - a) < 1e-12:
            break

        alpha = step
        a_trial = a
        conv_trial = conv
        obj_trial = best_obj
        
        for _ in range(10):
            a_trial = (1 - alpha) * best_a + alpha * b
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / s_t
            conv_trial = fast_convolve(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()
            if obj_trial < best_obj * (1 - 1e-10):
                break
            alpha *= 0.5
            if alpha < 0.001:
                break
        
        if obj_trial < best_obj - 1e-12:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial
            conv = conv_trial
            A_dirty = True
            no_improve = 0
            step = min(0.8, alpha * 1.2)
        else:
            no_improve += 1
            step = max(0.01, alpha * 0.7)
        
        if no_improve > 100:
            # Try min_peak LP
            b = lp_solve(a, conv.max(), method='min_peak')

                if b_sum > 1e-15:
                    b = b / b_sum

                            no_improve = 0

def soft_max_objective(a, t):
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    obj = 2.0 * n * (1.0/t) * np.log(sum_exp)
    return obj

def smooth_refine(a0, time_limit, t_init=30.0):
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t = t_init
    
    while time.monotonic() - t0 < time_limit:
        # Use L-BFGS-B with simplex projection
        def func_and_grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return 1e10, np.zeros(len(x))
            x = x / s
            conv = fast_convolve(x)
            scaled = t * conv
            scaled -= scaled.max()
            exp_vals = np.exp(scaled)
            sum_exp = exp_vals.sum()
            obj = 2.0 * n * (1.0/t) * np.log(sum_exp)
            
            # Gradient
            p = exp_vals / sum_exp
            grad = np.zeros(n)
            for j in range(n):
                k_lo = j
                k_hi = min(2*n-2, j+n-1)
                if k_lo <= k_hi:
                    grad[j] = (2.0*t*n/sum_exp) * np.dot(p[k_lo:k_hi+1], x[k_lo-j:k_hi-j+1])
            # Normalize gradient for the constraint sum(x)=1
            grad = grad - np.mean(grad)
            return obj, grad
        
        try:
            res = minimize(func_and_grad, a, method='L-BFGS-B',
                          bounds=[(0, None)]*n,
                          options={'maxiter': 100, 'gtol': 1e-8})
            if res.fun < soft_max_objective(best_a, t):
                a_new = np.maximum(res.x, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    conv_new = fast_convolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
        except Exception:
            pass
        
        t *= 1.5
        if t > 500:
            break
    
    return best_a, best_obj

def subgradient_refine(a0, time_limit, lr_init=0.05, momentum=0.9, top_k=20):
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    rng = np.random.default_rng(999)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient of max conv(a,a)
        peak_idx = np.argmax(conv)
        k = peak_idx
        sg = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += 2.0 * a[k - idx]
        
        # Also add contribution from top-k peaks
        if top_k < len(conv):
            top_peaks = np.argpartition(conv, -top_k)[-top_k:]
            for pk in top_peaks:
                if pk == k:
                    continue
                j_s = max(0, pk - n + 1)
                j_e = min(n - 1, pk)
                idx2 = np.arange(j_s, j_e + 1)
                sg[idx2] += 0.5 * a[pk - idx2]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        if no_improve > 2000:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(lr_init * 0.5, lr * 0.5)
    
    return best_a, best_obj

def generate_initialization(n, rng, parent=None):
    init_type = rng.integers(0, 8)
    x = np.arange(n, dtype=np.float64)
    
    if parent is not None and init_type == 0:
        if len(parent) >= n:
            a = np.maximum(parent[:n], 0)
        else:
            a = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, len(parent)),
                         np.maximum(parent, 0))
    elif parent is not None and init_type == 1:
        # Parent with noise
        if len(parent) >= n:
            a = np.maximum(parent[:n], 0)
        else:
            a = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, len(parent)),
                         np.maximum(parent, 0))
        a = a + rng.standard_normal(n) * 0.1 * a.mean()
        a = np.maximum(a, 0)
    elif init_type == 2:
        a = np.ones(n)
    elif init_type == 3:
        freq = rng.choice([1, 2, 3, 4])
        a = 1.0 + 0.9 * np.cos(2.0*np.pi*freq*x/n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0)
    elif init_type == 4:
        a = np.exp(-0.5 * ((x - n/2) / (n * rng.uniform(0.15, 0.4)))**2)
    elif init_type == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif init_type == 6:
        # Two-peak
        p1, p2 = rng.choice(n, 2, replace=False)
        a = np.zeros(n)
        a[max(0,p1-3):p1+4] = 1.0
        a[max(0,p2-3):p2+4] = 1.0
        a = np.maximum(a, 0)
    else:
        # Ramp
        a = np.arange(1, n+1, dtype=np.float64)
    
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
    return A


def lp_solve_max_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None


def lp_solve_min_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        return np.maximum(res.x[:n], 0.0)
    return None


def lp_refine(a0, time_limit, step_init=0.3, max_iters=5000):
    """LP-based iterative refinement with backtracking line search."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_conv = conv.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    use_min_peak = False

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = best_conv.max()

        if not use_min_peak:
            b = lp_solve_max_sum(a, M)
            if b is None:
                use_min_peak = True
                continue
        else:
            b = lp_solve_min_peak(a)
            if b is None:
                use_min_peak = False
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        if np.linalg.norm(b - a) < 1e-12:
            if no_improve > 50:
                break
            no_improve += 1
            continue

        # Backtracking line search
        alpha = step
        found = False
        for _ in range(12):
            a_trial = (1.0 - alpha) * best_a + alpha * b
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / s_t
            conv_trial = fast_convolve(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()
            if obj_trial < best_obj * (1.0 - 1e-10):
                best_obj = obj_trial
                best_a = a_trial.copy()
                best_conv = conv_trial.copy()
                a = a_trial.copy()
                conv = conv_trial
                found = True
                break
            alpha *= 0.5
            if alpha < 0.0005:
                break

        no_improve += 1
        if found:
            no_improve = 0
            step = min(0.9, alpha * 1.3)
        else:
            step = max(0.005, alpha * 0.6)

        if no_improve % 80 == 0 and no_improve > 0:
            use_min_peak = not use_min_peak
            step = max(step_init * 0.5, step)

        if no_improve > 400:
            break

    return best_a, best_obj


def soft_max_smooth_obj(a, t):
    """Compute soft-max smoothed objective value."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    return 2.0 * n * (1.0 / t) * np.log(sum_exp)

def subgradient_refine(a0, time_limit, lr_init=0.04, momentum=0.91, top_k=15):
    """Projected subgradient with momentum and periodic perturbation."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    rng = np.random.default_rng(777)
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        # Multi-peak subgradient
        peak_idx = np.argmax(conv)
        sg = np.zeros(n)
        k = peak_idx
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += 2.0 * a[k - idx]

        if top_k < len(conv):
            top_peaks = np.argpartition(conv, -top_k)[-top_k:]
            for pk in top_peaks:
                if pk == k:
                    continue
                j_s = max(0, pk - n + 1)
                j_e = min(n - 1, pk)
                idx2 = np.arange(j_s, j_e + 1)
                sg[idx2] += 0.3 * a[pk - idx2]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 1500 + perturb_count * 500:
            noise = rng.standard_normal(n) * (0.003 + 0.002 * perturb_count)
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            perturb_count += 1
            lr = max(lr_init * 0.4, lr * 0.6)

    return best_a, best_obj


def peak_shave(a0, time_limit):
    """Directly target the peak: reduce mass at peak-contributing positions."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Find current peak
        k_peak = int(np.argmax(conv))

        # Identify positions contributing to the peak
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        peak_positions = set(range(j_start, j_end + 1))
        contrib = np.abs(a[list(peak_positions)]) * np.abs(a[k_peak - list(range(j_start, j_end + 1))])

        # Find non-peak positions to receive mass
        non_peak_positions = [j for j in range(n) if j not in peak_positions]
        if not non_peak_positions:
            break

        # Reduce top contributors, add to random non-peak positions
        n_reduce = max(1, len(peak_positions) // 4)
        sorted_items = sorted(zip(contrib, range(j_start, j_end + 1)), reverse=True)
        reduce_positions = [item[1] for item in sorted_items[:n_reduce]]
        add_positions = rng.choice(non_peak_positions, size=min(n_reduce, len(non_peak_positions)), replace=False)

        # Compute how to transfer
        eps = 0.002
        for rp in reduce_positions:
            mass = min(eps, a[rp] * 0.3)
            if mass > 1e-10:
                a[rp] -= mass
                for ap in add_positions:
                    a[ap] += mass / len(add_positions)

        a = np.maximum(a, 0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


def generate_initialization(n, rng, parent=None):

    x = np.arange(n, dtype=np.float64)

    if parent is not None and init_type == 0:

            a = np.maximum(parent[:n], 0).copy()
        else:
            a = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0)).copy()
    elif parent is not None and init_type == 1:
        if len(parent) >= n:
            a = np.maximum(parent[:n], 0).copy()
        else:
            a = np.interp(np.linspace(0, 1, n),

                          np.maximum(parent, 0)).copy()
        a = a + rng.standard_normal(n) * 0.05 * max(a.mean(), 1e-10)
        a = np.maximum(a, 0)
    elif init_type == 2:
        a = np.ones(n)
    elif init_type == 3:
        freq = int(rng.choice([1, 2, 3, 4]))
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n + phase)
        a = np.maximum(a, 0)
    elif init_type == 4:
        center = rng.uniform(0.3 * n, 0.7 * n)
        width = rng.uniform(0.15, 0.4) * n
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif init_type == 5:
        a = rng.dirichlet(np.ones(n) * 0.3)
    elif init_type == 6:
        num_peaks = int(rng.choice([2, 3, 5]))
        peaks = rng.choice(n, size=num_peaks, replace=False)
        a = np.zeros(n)
        for p in peaks:
            w = max(1, int(n * 0.02))
            a[max(0, p - w):p + w + 1] += 1.0
        a = np.maximum(a, 0)
    elif init_type == 7:
        a = np.arange(1, n + 1, dtype=np.float64)
    elif init_type == 8:

            np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
        a = np.maximum(a, 0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # Phase 3: Subgradient refinement
    print("=== Phase 3: Subgradient Refinement ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 200
        n_best = len(best_sequence)
        for lr in [0.05, 0.03, 0.02]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(50.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            a_opt, _ = subgradient_refine(best_sequence, time_limit=t_alloc,
                                          lr_init=lr, momentum=0.91,
                                          top_k=max(10, n_best // 40))
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # Phase 4: Smooth optimization (BFGS on soft-max)
    print("=== Phase 4: Smooth BFGS Polish ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 100
        for t_init in [20.0, 50.0, 100.0]:
            if time.monotonic() > phase4_deadline:
                break
            t_alloc = min(35.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            a_opt, _ = smooth_refine(best_sequence, time_limit=t_alloc, t_init=t_init)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Phase 5: Peak shaving + final LP push
    print("=== Phase 5: Peak Shaving + Final Push ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        phase5_deadline = deadline - 30
        t_alloc = min(30.0, phase5_deadline - time.monotonic() - 10)
        if t_alloc > 5:
            a_opt, _ = peak_shave(best_sequence, time_limit=t_alloc)
            try_update(a_opt)

        # Final LP push on the best
        if time.monotonic() < phase5_deadline:
            t_alloc = min(30.0, phase5_deadline - time.monotonic() - 2)
            if t_alloc > 5:
                a_opt, _ = lp_refine(best_sequence, time_limit=t_alloc, step_init=0.1)
                try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Phase 6: Final perturbation + micro-optimization
    print("=== Phase 6: Final Micro-Optimization ===")
    t6_start = time.monotonic()

    if best_sequence is not None:
        phase6_deadline = deadline - 5
        n_best = len(best_sequence)
        for scale in [0.001, 0.005, 0.002]:
            if time.monotonic() > phase6_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(15.0, phase6_deadline - time.monotonic() - 1)
            if t_alloc < 3:
                break
            a_opt, _ = lp_refine(a_pert, time_limit=t_alloc, step_init=0.15)
            try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6_start:.0f}s")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
