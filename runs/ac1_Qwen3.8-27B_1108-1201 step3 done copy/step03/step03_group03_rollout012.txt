        # Shift for numerical stability

        a_sum = a.sum()
        if a_sum < 1e-300:

        else:
            a /= a_sum
        
        # Schedule
        beta = min(beta_max, beta * beta_growth)
        lr *= lr_decay

        # grad[j] += a[k-j] for each peak k where 0 <= k-j <= n-1

        # a[k-j] for j=0..n-1 is a[k], a[k-1], ..., a[k-n+1] (where valid)
        # = a[0..k] flipped and then a[k-n+1..k] reversed...

        # And for j > k, a[k-j] = 0

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad_fft(a, w):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] via FFT.
    a: length n, w: length 2n-1
    Returns: grad of length n
    """
    n = len(a)
    a_flip = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # = 3n-2
    fft_size = 2 ** (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a_flip, fft_size)
    conv = np.fft.irfft(W * A, fft_size)
    return 2.0 * conv[n - 1 : n - 1 + n]

def compute_grad_soft(a, w, peaks):
    """Compute subgradient from selected peaks. More efficient when few peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peaks:

        # Contribution: a[k-j] for valid j
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += a[k - j_hi : k - j_lo : -1]
    return grad * (2.0 / len(peaks))

    best_a = a.copy()
    itera = 0
    t0 = time.monotonic()
    beta = beta_init
    gamma_cur = gamma
    
    # Precompute FFT size for gradient
    grad_fft_size = 2 ** (3*n - 3).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution
        conv = fast_convolve(a)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        if itera % n_subgrad_every == 0 and beta > beta_init * 10:
            # Do a subgradient step on the hard max
            top_k = min(20, len(conv))
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            grad = compute_grad_soft(a, None, peaks)
            # Move from high grad to low grad
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i != j and a[i] > 1e-10:
                for alpha in [0.3, 0.1, 0.05]:
                    a_trial = a.copy()
                    a_trial[j] += alpha * a[i]
                    a_trial[i] *= (1.0 - alpha)
                    conv_t = fast_convolve(a_trial)
                    obj_t = 2.0 * n * conv_t.max()
                    if obj_t < cur_obj:
                        a = a_trial
                        cur_obj = obj_t
                        if obj_t < best_obj:
                            best_obj = obj_t
                            best_a = a.copy()
                        break
        
        # Compute softmax weights
        beta_conv = beta * conv
        mc = beta_conv.max()
        if mc > 50:  # numerical stability
            beta_conv = beta_conv - mc
        w = np.exp(beta_conv)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient via FFT
        a_flip = a[::-1]
        W_fft = np.fft.rfft(w, grad_fft_size)
        A_fft = np.fft.rfft(a_flip, grad_fft_size)
        conv_wa = np.fft.irfft(W_fft * A_fft, grad_fft_size)
        grad = 2.0 * conv_wa[n - 1 : n - 1 + n]
        
        # Mirror descent update
        # a_j <- a_j * exp(-gamma * grad_j) / Z
        log_step = -gamma_cur * grad
        # Numerical stability: shift
        ls_max = log_step.max()
        if ls_max > 50:
            log_step = log_step - ls_max
        elif ls_max < -50:
            log_step = log_step - ls_max
        mult = np.exp(log_step)
        a = a * mult
        a_sum = a.sum()
        if a_sum < 1e-300:
            a = np.ones(n) / n
        else:
            a /= a_sum
        
        # Schedule
        beta = min(beta_max, beta * beta_growth)
        gamma_cur = max(0.01, gamma_cur * gamma_decay)
        itera += 1
    
    return best_a, best_obj

def subgradient_polish(a0, time_limit, top_k=30):
    """Subgradient method with momentum on the hard max."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = float('inf')
    best_a = a.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.5 / n
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient from top-k peaks
        pk = min(top_k, len(conv))
        peaks = np.argpartition(conv, -pk)[-pk:]
        
        # Weight by how close to max each peak is
        peak_vals = conv[peaks]
        weights = np.ones(len(peaks))
        # Emphasize the actual max
        weights[np.argmax(peak_vals)] = 2.0
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for idx, k in enumerate(peaks):
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += weights[idx] * 2.0 * a[k - j_hi : k - j_lo : -1]
        
        # Update with momentum
        v = momentum * v - lr * grad
        # Project: clip negatives, normalize
        a = np.maximum(a + v, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
            v = np.zeros(n)
        else:
            a /= a_sum
        
        if no_improve > 100:
            # Perturbation
            rng = np.random.default_rng(hash(int(cur_obj * 1e6)) % (2**31))
            a = project_nonneg_sum1(a + rng.standard_normal(n) * 0.01)
            v *= 0.3
            no_improve = 0
            lr = max(lr * 0.8, 0.01 / n)
        
        lr *= 0.9998
    
    return best_a, best_obj

def coordinate_descent_polish(a0, max_rounds=30, step_fracs=None):

                break

def interpolate_seq(seq_target, n_new):
    """Interpolate a sequence to a new length."""
    if n_new == len(seq_target):
        return seq_target.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(seq_target))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, seq_target)

def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:

    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Edge-heavy (mass at edges)
    e = max(1, n // 4)
    edge = np.zeros(n)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Quadratic decay from edges
    edge2 = np.concatenate([np.arange(n//2, 0, -1), np.arange(1, n//2+1)])

    elif len(edge2) < n:

    inits['edge_quad'] = edge2 / edge2.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Lambda distribution (geometric-like)
    lam = np.exp(-np.arange(n) * 0.1)
    inits['lambda'] = lam / lam.sum()
    
    if parent is not None and len(parent) > 0:
        p = interpolate_seq(np.maximum(parent, 0), n)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = p / ps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (exponential gradient)
with adaptive temperature scheduling, FFT-based gradient computation, subgradient
refinement on the hard max, and multi-length multi-start search.
Key improvements: O(n log n) gradient via FFT, larger sequence lengths (up to 2048),
adaptive beta schedule transitioning from smooth to hard max, and targeted
Frank-Wolfe moves for final polishing.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, w, grad_fft_size):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] via FFT.
    a: length n, w: length 2n-1, grad_fft_size: precomputed FFT size.
    Returns: grad of length n.
    """
    n = len(a)
    a_flip = a[::-1]
    W = np.fft.rfft(w, grad_fft_size)
    A = np.fft.rfft(a_flip, grad_fft_size)
    conv = np.fft.irfft(W * A, grad_fft_size)
    return 2.0 * conv[n - 1: n - 1 + n]


def compute_subgrad_peaks(a, peaks, weights=None):
    """Compute weighted subgradient from selected peak positions."""
    n = len(a)
    if weights is None:
        weights = np.ones(len(peaks)) / len(peaks)
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += weights[idx] * 2.0 * a[k - j_hi: k - j_lo: -1]
    return grad


def project_nonneg_sum1(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    grad_fft_size = 2 ** (3 * n - 3).bit_length()

    while time.monotonic() - t0 < time_limit:

            wts[np.argmax(peak_vals)] = 3.0
            wts /= wts.sum()

                    if ot < cur_obj - 1e-15:
                        a = a_t
                        cur_obj = ot
                        if ot < best_obj:
                            best_obj = ot
                            best_a = a.copy()

def frank_wolfe_polish(a0, time_limit, max_iter=50000):

        if src == dst or a[src] < 1e-14:

    inits['bimodal'] = g2 / g2.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.2)

    lam = np.exp(-np.arange(n) * 0.05)
    inits['lambda'] = lam / lam.sum()

    # Asymmetric ramp
    ramp = np.arange(1, n + 1, dtype=np.float64) ** 0.5
    inits['ramp'] = ramp / ramp.sum()

    if parent is not None and len(parent) > 0:
        p = interpolate_seq(np.maximum(parent, 0), n)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = p / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining mirror descent, subgradient, and Frank-Wolfe
    across multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_score = par_score
            best_sequence = p.copy()
    except Exception:
        pass

    # === Phase 1: Multi-length mirror descent exploration ===
    print("=== Phase 1: Mirror Descent Exploration ===")
    phase1_deadline = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.3)

    lengths = [512, 768, 1024, 1536]
    candidates = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = gen_inits(n, parent, rng)
        priority = ['parent', 'uniform', 'bimodal', 'edge', 'gauss_off',
                    'random', 'random2', 'triangle', 'lambda', 'ramp']
        names = [nm for nm in priority if nm in inits][:4]

        t_per = max(15.0, (phase1_deadline - time.monotonic() - 20) / (len(lengths) * len(names)))

        for name in names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = mirror_descent_optimize(
                a0, t_alloc,
                beta_init=3.0, beta_max=20000.0,
                beta_growth=1.003, gamma_init=0.8,
                gamma_decay=0.99998, subgrad_every=150, top_k_sub=10
            )

            # Quick subgradient polish
            a_opt2, obj2 = subgradient_polish(a_opt, min(5.0, t_alloc * 0.1))
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # === Phase 2: Deep optimization on top candidates ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_deadline = time.monotonic() + min(600.0, (deadline - time.monotonic()) * 0.35)

    top_list = []
    seen = set()
    for score, n, name, a_opt in candidates[:10]:
        if n not in seen:
            top_list.append((n, a_opt, score))
            seen.add(n)
        if len(top_list) >= 5:
            break
    if best_sequence is not None and len(best_sequence) not in seen:
        top_list.append((len(best_sequence), best_sequence.copy(), best_score))

    t_per_deep = max(20.0, (phase2_deadline - time.monotonic() - 30) / max(1, len(top_list)))

    for n, a0, init_score in top_list:
        if time.monotonic() > phase2_deadline - 40:
            break
        t_alloc = min(t_per_deep, phase2_deadline - time.monotonic() - 30)
        if t_alloc < 15:
            break

        # Deep mirror descent with slower beta growth
        a_opt, obj_opt = mirror_descent_optimize(
            a0, t_alloc * 0.5,
            beta_init=50.0, beta_max=80000.0,
            beta_growth=1.002, gamma_init=0.5,
            gamma_decay=0.99999, subgrad_every=100, top_k_sub=20
        )

        # Subgradient polish
        t_sub = min(30.0, t_alloc * 0.25)
        a_opt2, obj2 = subgradient_polish(a_opt, t_sub, top_k=40, lr_scale=0.5)
        if obj2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj2

        # Frank-Wolfe polish
        t_fw = min(20.0, t_alloc * 0.15)
        a_opt3 = frank_wolfe_polish(a_opt, t_fw)
        score_fw = float(evaluate_sequence(a_opt3.tolist()))
        if np.isfinite(score_fw) and score_fw < obj_opt:
            a_opt = a_opt3
            obj_opt = score_fw

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}: {score:.6f}")

    # === Phase 3: Polish the best solution ===
    print(f"=== Phase 3: Polish (current best: {best_score:.6f}) ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        n_best = len(best_sequence)

        # Frank-Wolfe on best
        t_fw = min(120.0, remaining * 0.3)
        a_fw = frank_wolfe_polish(best_sequence, t_fw)
        score_fw = float(evaluate_sequence(a_fw.tolist()))
        if np.isfinite(score_fw) and score_fw < best_score:
            best_score = score_fw
            best_sequence = a_fw.copy()
            print(f"  FW polish: {score_fw:.6f}")

        # Subgradient with small lr
        t_sg = min(120.0, remaining * 0.25)
        a_sg, obj_sg = subgradient_polish(best_sequence, t_sg, top_k=50, lr_scale=0.3)
        score_sg = float(evaluate_sequence(a_sg.tolist()))
        if np.isfinite(score_sg) and score_sg < best_score:
            best_score = score_sg
            best_sequence = a_sg.copy()
            print(f"  Subgrad polish: {score_sg:.6f}")

        # Mirror descent with high beta (near-hard max)
        remaining = deadline - time.monotonic() - 30.0
        t_md = min(150.0, remaining * 0.4)
        if t_md > 10:
            a_md, obj_md = mirror_descent_optimize(
                best_sequence, t_md,
                beta_init=500.0, beta_max=200000.0,
                beta_growth=1.001, gamma_init=0.3,
                gamma_decay=0.99999, subgrad_every=50, top_k_sub=30
            )
            score_md = float(evaluate_sequence(a_md.tolist()))
            if np.isfinite(score_md) and score_md < best_score:
                best_score = score_md
                best_sequence = a_md.copy()
                print(f"  High-beta MD: {score_md:.6f}")

        # Multi-start perturbations around best
        remaining = deadline - time.monotonic() - 25.0
        if remaining > 30:
            n_p = len(best_sequence)
            for pert_size in [0.005, 0.015, 0.03]:
                if time.monotonic() > deadline - 20:
                    break
                for trial in range(3):
                    if time.monotonic() > deadline - 15:
                        break
                    t_trial = max(5.0, (deadline - time.monotonic() - 12) / 3.0)
                    noise = rng.standard_normal(n_p) * pert_size
                    a_p = project_nonneg_sum1(best_sequence + noise)
                    a_opt, obj_opt = mirror_descent_optimize(
                        a_p, t_trial,
                        beta_init=100.0, beta_max=100000.0,
                        beta_growth=1.002, gamma_init=0.4,
                        gamma_decay=0.99999, subgrad_every=80, top_k_sub=20
                    )
                    a_opt2, obj2 = subgradient_polish(a_opt, min(5.0, t_trial * 0.2), top_k=30)
                    if obj2 < obj_opt:
                        a_opt = a_opt2
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({pert_size})#{trial}: {score:.6f}")

    # === Phase 4: Try a few more lengths if time remains ===
    remaining = deadline - time.monotonic() - 20.0
    if remaining > 60 and best_sequence is not None:
        print(f"=== Phase 4: Extra exploration ({remaining:.0f}s left) ===")
        n_best = len(best_sequence)
        extra_lengths = [l for l in [768, 1024, 2048] if l != n_best]
        for n_extra in extra_lengths[:2]:
            if time.monotonic() > deadline - 20:
                break
            t_alloc = min(60.0, deadline - time.monotonic() - 15)
            if t_alloc < 15:
                break
            a0 = interpolate_seq(best_sequence, n_extra)
            a_opt, obj_opt = mirror_descent_optimize(
                a0, t_alloc,
                beta_init=10.0, beta_max=50000.0,
                beta_growth=1.003, gamma_init=0.7,
                gamma_decay=0.99998, subgrad_every=100, top_k_sub=15
            )
            a_opt2, obj2 = subgradient_polish(a_opt, min(10.0, t_alloc * 0.15), top_k=40)
            if obj2 < obj_opt:
                a_opt = a_opt2
                obj_opt = obj2
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Extra n={n_extra}: {score:.6f}")

    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
