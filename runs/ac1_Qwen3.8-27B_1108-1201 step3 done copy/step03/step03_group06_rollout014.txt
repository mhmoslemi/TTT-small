        if no_improve >= max_no_improve:
            # Diversify: perturb from best
            scale = 0.003 + 0.002 * np.sqrt(no_improve / max_no_improve)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(best_a + noise)
            best_obj, _ = obj_val(a)
            no_improve = 0
            max_no_improve = min(120, int(max_no_improve * 1.25))

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            print(f"    LP iter {iter_count}: obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def compute_subgradient(a, conv, threshold_frac=0.999):
    """Compute subgradient of max_k conv[k] at a using FFT-based approach.
    Uses active set where conv[k] >= threshold_frac * max(conv)."""
    n = len(a)
    max_val = conv.max()
    threshold = threshold_frac * max_val + 1e-15
    active = np.where(conv >= threshold)[0]

    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    # Build weight vector w (uniform over active set)
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)

    # Gradient of conv[k] w.r.t. a[i] = a[k-i] + a[i-k] (with bounds)
    # Sum over k: sum_k w[k] * (a[k-i] + a[i-k])
    # = cross_corr(w, a)[i] + conv(w, a)[i]

    fft_size = 1 << (4 * n - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)

    # cross_corr[i] = sum_j w[j+i] * a[j] = IFFT(FFT(a) * conj(FFT(w)))[i]
    cross = np.fft.ifft(A * np.conj(W)).real[:n]

    # conv_wa[i] = sum_j w[j] * a[i-j] = IFFT(FFT(a) * FFT(w))[i]
    conv_wa = np.fft.ifft(A * W).real[:n]

    grad = cross + conv_wa
    return grad


def subgradient_optimize(a0, time_limit, seed=0, verbose=False):
    """Nesterov-accelerated projected subgradient method on simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj, conv = obj_val(a)
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    v = np.zeros(n)
    momentum = 0.9
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_subgradient(a, conv)

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        obj_new, _ = obj_val(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 300:
                # Restart with perturbation
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(best_a + noise)
                v = np.zeros(n)
                lr = 0.005
                no_improve = 0
                best_obj, _ = obj_val(a)
                a = best_a.copy()

        iter_count += 1
        if verbose and iter_count % 500 == 0:
            print(f"    SG iter {iter_count}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations normalized to sum=1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_r'] = ramp[::-1].copy()

    # Edge-peaked (bimodal)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Gaussian center
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g

    # Cosine window
    cos_w = np.cos(np.pi * (x + 0.5) / n) ** 2
    cos_w = np.maximum(cos_w, 0.001)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand_dense'] = rng.dirichlet(np.ones(n) * 1.0)

    # Symmetric random
    half = rng.dirichlet(np.ones(n // 2 + 1) * 0.1)
    sym = np.zeros(n)
    sym[:n // 2 + 1] = half
    sym[n // 2 + 1:] = half[:max(0, n - n // 2 - 1)][::-1]
    sym = np.maximum(sym, 0)
    ss = sym.sum()
    if ss > 1e-10:
        sym /= ss
    inits['sym_rand'] = sym

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: exploration, deep LP with full constraints + line search,
    subgradient polish, and final diversification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Exploration ==========
    print("=== Phase 1: Exploration ===")
    t1 = time.monotonic()
    phase1_time = 180.0

    lengths = [256, 320, 384, 448, 512]
    scan_results = []

    for n in lengths:
        if time.monotonic() - t1 > phase1_time - 20:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize
        priority_order = ['parent', 'parent_up', 'parent_ds', 'bimodal', 'uniform',
                          'gauss_0.25', 'cosine', 'rand_sparse', 'sym_rand']
        sorted_names = [nm for nm in priority_order if nm in inits] + \
                       [nm for nm in init_names if nm not in priority_order]
        sorted_names = sorted_names[:6]

        num_remaining = len(lengths) - lengths.index(n)
        time_per_init = max(3.0, (phase1_time - (time.monotonic() - t1) - 10) /
                            max(1, num_remaining) / max(1, len(sorted_names)))

        for name in sorted_names:
            if time.monotonic() - t1 > phase1_time - 5:
                break
            t_alloc = min(time_per_init, phase1_time - (time.monotonic() - t1) - 5)
            if t_alloc < 2:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_deep(a0, time_limit=t_alloc,
                                              seed=n + hash(name) % 10000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # ========== Phase 2: Deep LP Optimization ==========
    print("=== Phase 2: Deep LP (full constraints + line search) ===")
    t2 = time.monotonic()
    phase2_time = 900.0

    # Collect starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:10]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(deep_starts) >= 6:
            break

    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy()))

    # Add fresh starts at promising lengths
    for n_fresh in [320, 384]:
        if n_fresh not in [d[0] for d in deep_starts[:3]]:
            inits_f = generate_inits(n_fresh, parent, rng)
            for nm in ['bimodal', 'gauss_0.15', 'rand_sparse']:
                if nm in inits_f:
                    deep_starts.append((n_fresh, inits_f[nm].copy()))
                    break

    # Run LP optimization with time allocation
    num_runs = len(deep_starts)
    time_per_run = phase2_time / max(1, num_runs)
    time_per_run = max(30.0, time_per_run)
    # But don't spend more than 400s on any single run
    time_per_run = min(time_per_run, 400.0)

    for i, (n_start, a_start) in enumerate(deep_starts):
        if time.monotonic() - t2 > phase2_time - 30:
            break
        t_alloc = min(time_per_run, phase2_time - (time.monotonic() - t2) - 20)
        if t_alloc < 15:
            break

        print(f"  Deep LP run {i+1}/{len(deep_starts)}: n={n_start}, alloc={t_alloc:.0f}s")
        a_opt, obj_opt = lp_optimize_deep(a_start, time_limit=t_alloc,
                                          seed=i * 1000 + 7)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    NEW BEST: score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Subgradient Polish ==========
    print("=== Phase 3: Subgradient Polish ===")
    t3 = time.monotonic()
    phase3_time = 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Multiple subgradient runs with different params
        sg_configs = [
            {'lr': 0.003, 'mom': 0.85},
            {'lr': 0.008, 'mom': 0.90},
            {'lr': 0.015, 'mom': 0.93},
            {'lr': 0.005, 'mom': 0.88},
        ]

        time_per_sg = phase3_time / (len(sg_configs) + 2)

        for i, cfg in enumerate(sg_configs):
            if time.monotonic() - t3 > phase3_time - 10:
                break
            t_alloc = min(time_per_sg, phase3_time - (time.monotonic() - t3) - 10)
            if t_alloc < 10:
                break

            a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                                  seed=i * 50 + 3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SG cfg {i}: score={score:.6f}")

        # LP from subgradient results
        if time.monotonic() - t3 < phase3_time - 30:
            t_alloc = min(60.0, phase3_time - (time.monotonic() - t3) - 20)
            if t_alloc > 15:
                a_opt, obj_opt = lp_optimize_deep(best_sequence, time_limit=t_alloc,
                                                  seed=7777)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP polish: score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Diversification + Final Polish ==========
    print("=== Phase 4: Final ===")
    t4 = time.monotonic()
    remaining = deadline - time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Try different lengths by interpolating the best sequence
        if remaining > 100:
            for n_try in [256, 384, 448]:
                if n_try == n_best or time.monotonic() > deadline - 60:
                    continue
                if time.monotonic() - t4 > remaining * 0.3:
                    break
                # Interpolate
                a_interp = np.interp(np.linspace(0, 1, n_try),
                                     np.linspace(0, 1, n_best), best_sequence)
                a_interp = np.maximum(a_interp, 0)
                s = a_interp.sum()
                if s > 1e-10:
                    a_interp = a_interp / s
                t_alloc = min(40.0, remaining * 0.3 - (time.monotonic() - t4))
                if t_alloc > 10:
                    a_opt, _ = lp_optimize_deep(a_interp, time_limit=t_alloc,
                                                seed=n_try * 3)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Length {n_try}: score={score:.6f}")

        # Final LP polish
        t_final = deadline - time.monotonic() - 30
        if t_final > 20:
            a_opt, _ = lp_optimize_deep(best_sequence, time_limit=t_final,
                                        seed=99999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")

        # Tiny perturbations
        t_remaining = deadline - time.monotonic() - 20
        if t_remaining > 10:
            for i in range(min(10, int(t_remaining / 8))):
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    sc = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_sequence = a_pert.copy()
                        print(f"  Perturb {i} s={scale}: {sc:.6f}")
                        break

    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
