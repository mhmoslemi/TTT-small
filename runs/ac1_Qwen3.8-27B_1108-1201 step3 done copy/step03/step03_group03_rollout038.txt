"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based smoothed gradient descent
with Nesterov acceleration, mirror descent, and LP polish.
Uses adaptive temperature annealing on the log-sum-exp smoothing of the max-convolution,
combined with multi-length search and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(a, p):
    """Compute g_j = sum_k p[k] * a[k-j] for j=0..n-1 using FFT.
    This is the cross-correlation (a cross p) at lags 0..n-1.
    """
    n = len(a)
    # Need N >= n + (2n-1) - 1 = 3n-2
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    result = np.fft.irfft(np.conj(A) * P, N)
    return result[:n]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute the objective 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def optimize_nesterov(a0, time_limit, T_init=10.0, T_max=500.0,
                      lr_init=0.02, momentum_init=0.85,
                      T_step=1.002, T_step_every=30):
    """Nesterov accelerated gradient descent on smoothed max-convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    momentum = momentum_init
    T = T_init
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Softmax weights on convolution (shifted for stability)
        conv_shifted = conv - conv.max()
        p = np.exp(T * conv_shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum

        # Gradient: 2 * cross_corr(a, p)
        grad = 2.0 * fast_cross_corr(a, p)

        # Nesterov step
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        iter_count += 1

        # Adaptive temperature (increase to focus on peak)
        if iter_count % T_step_every == 0 and T < T_max:
            T = min(T_max, T * T_step)

        # Adaptive momentum (slowly increase)
        if momentum < 0.97:
            momentum = min(0.97, momentum * 1.001)

        # Adaptive learning rate
        if no_improve > 80:
            lr = max(0.0005, lr * 0.6)
            v *= 0.5
            no_improve = 0
        elif no_improve > 150:
            # Perturbation
            rng_p = np.random.default_rng(iter_count * 7919)
            a = project_simplex(a + rng_p.standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.3

        # Slow decay
        lr *= 0.99995

    return best_a, best_obj

def optimize_mirror(a0, time_limit, T_init=10.0, T_max=500.0,
                    lr_init=0.05, T_step=1.003, T_step_every=20):
    """Mirror descent (exponential family) on smoothed max-convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    T = T_init
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv_shifted = conv - conv.max()
        p = np.exp(T * conv_shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum

        grad = 2.0 * fast_cross_corr(a, p)

        # Mirror descent: multiplicative update
        # a_new_j ∝ a_j * exp(-lr * grad_j)
        log_update = np.log(np.maximum(a, 1e-300)) - lr * grad
        log_update -= log_update.max()  # for numerical stability
        a_new = np.exp(log_update)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        iter_count += 1

        if iter_count % T_step_every == 0 and T < T_max:
            T = min(T_max, T * T_step)

        if no_improve > 60:
            lr = max(0.001, lr * 0.5)
            no_improve = 0
        elif no_improve > 120:
            rng_p = np.random.default_rng(iter_count * 104729)
            a = a + rng_p.standard_normal(n) * 0.005
            a = project_simplex(a)
            conv = fast_convolve(a)
            no_improve = 0
            lr = lr_init * 0.3

        lr *= 0.9999

    return best_a, best_obj

def optimize_lp_polish(a0, time_limit=30.0, max_iters=50):
    """LP-based directional optimization for final polish."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    # Pre-build constraint matrix
    # A_ub has shape (2n-1, n), A_ub[k, j] = a[k-j] if 0<=k-j<n
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[k, j] = a[k - j]

    t0 = time.monotonic()
    blend = 0.3

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        if M < 1e-15:
            break

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True})
            if not result.success:
                break
            g = np.maximum(result.x, 0.0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a_new = project_simplex(a_new)

        # Update A_ub
        for k in range(2 * n - 1):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A_ub[k, j] = a_new[k - j]

        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            blend = max(0.01, blend * 0.9)
        else:
            blend = max(0.005, blend * 0.7)

        a = a_new

    return best_a, best_obj

def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Semicircle (Wigner)
    xc = (x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 0.5)
    semi = np.sqrt(np.maximum(1.0 - xc**2, 1e-10))
    inits['semicircle'] = semi / semi.sum()

    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.35 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # U-shape (convex)
    u = (x - 0.5 * (n - 1)) ** 2 + 0.01 * (n - 1)
    inits['u_shape'] = u / u.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                p_resampled = np.interp(np.linspace(0, 1, n),
                                        np.linspace(0, 1, len(p)), p)
                p_resampled = np.maximum(p_resampled, 0)
                prs = p_resampled.sum()
                if prs > 1e-10:
                    inits['parent'] = p_resampled / prs

    return inits

    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, p):
    """Compute g[j] = sum_k p[k]*a[k-j] for j=0..n-1 (cross-correlation) via FFT."""
    n = len(a)
    N = 2 * n
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    result = np.fft.irfft(np.conj(A) * P, N)
    return result[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_and_conv(a):
    """Return (objective, convolution) assuming sum(a)=1."""
    n = len(a)

    if p_sum < 1e-300:

        a = a_new
        conv = conv_new
        iter_count += 1

        if iter_count % T_step_every == 0 and T < T_max:
            T = min(T_max, T * T_step)

        if no_improve > 50:
            lr = max(0.001, lr * 0.5)
            no_improve = 0
        elif no_improve > 100:
            rng_p = np.random.default_rng(iter_count * 104729)
            a = project_simplex(a + rng_p.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            no_improve = 0
            lr = lr_init * 0.3

        lr *= 0.9999

    return best_a, best_obj


def optimize_subgrad(a0, time_limit, k_active=8, lr_init=0.004, momentum=0.9):
    """Subgradient descent using top-k active constraints with Nesterov momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -k_active)[-k_active:]
        peak_vals = conv[peaks]

        shifted = peak_vals - peak_vals.max()
        w = np.exp(15.0 * shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = 2.0 * w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1
        lr = max(0.00005, lr * 0.9998)

        if no_improve > 150:
            a = project_simplex(a + np.random.default_rng(iter_count * 31).standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(lr_init * 0.4, 0.0005)

    return best_a, best_obj


def optimize_lp_polish(a0, time_limit=30.0, max_iters=25):
    """LP-based directional optimization for refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = a[k - j]

    t0 = time.monotonic()
    blend = 0.3

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        if M < 1e-15:
            break

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True})
            if not result.success:
                break
            g = np.maximum(result.x, 0.0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a_new = project_simplex(a_new)

        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[k, j] = a_new[k - j]

        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            blend = max(0.01, blend * 0.85)
        else:
            blend = max(0.005, blend * 0.6)

        a = a_new

    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    xc = (x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 0.5)
    semi = np.sqrt(np.maximum(1.0 - xc * xc, 1e-10))
    inits['semicircle'] = semi / semi.sum()

    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.35 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    u = (x - 0.5 * (n - 1)) ** 2 + 0.02 * (n - 1)
    inits['u_shape'] = u / u.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(p)), p)
                p_res = np.maximum(p_res, 0)
                prs = p_res.sum()
                if prs > 1e-10:
                    inits['parent'] = p_res / prs

    return inits
