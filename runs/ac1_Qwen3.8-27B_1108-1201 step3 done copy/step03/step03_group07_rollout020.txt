"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using corrected gradient computation,
iterative projection initialization, adaptive beta scheduling, and exact
subgradient fine-tuning. Key fix: correct FFT-based gradient formula.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_correct(a, w):
    """CORRECT gradient of sum_k w[k]*conv[k] w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * sum_m a[m] * w[m+i]
    Computed via: 2 * ifft(conj(fft(a)) * fft(w))[:n]
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    return 2.0 * np.real(ifft(np.conj(A) * W, N))[:n]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho >= n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj(x, n, beta):
    """Objective for scipy minimize (log-parametrized simplex)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def softmax_jac(x, n, beta):
    """Jacobian for scipy minimize (log-parametrized simplex)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_correct(a, w)
    # Chain rule for log-parametrization: da_i/dx_j = a_i(delta_ij - a_j)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B optimization with log-parametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(softmax_obj, x0, jac=softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 40})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def optimize_pg_smooth(a0, beta, lr=0.06, n_iter=500):
    """Projected gradient descent on simplex with smooth max."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        grad_a = compute_grad_correct(a, w)
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-14:
            break

        # Backtracking line search
        t = lr
        for _ in range(15):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5

        a = a_new
        lr *= 0.999

    return best_a, best_obj


def subgradient_step(a, n_steps=50, lr=0.001):
    """Exact subgradient method: use only the argmax of conv."""
    n = len(a)
    a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for _ in range(n_steps):
        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        score = 2.0 * n * peak_val

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Subgradient: grad[i] = 2 * a[peak_idx - i] (with bounds check)
        grad = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        grad[lo:hi + 1] = 2.0 * a[peak_idx - hi:peak_idx - lo + 1][::-1]

        # Normalize gradient to unit norm for stable steps
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        a_new = project_simplex(a - lr * gnorm * grad)
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()

        if score_new < score:
            a = a_new
            lr *= 1.01
        else:
            lr *= 0.7

    return best_a, best_score


def iterative_projection_init(n, rng, n_iters=200):
    """Initialize via alternating projections: flat spectrum + non-negativity."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_score = float('inf')

    for trial in range(5):
        # Random phases
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        A = np.zeros(fft_size, dtype=complex)
        A[0] = n  # DC component
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1])

        a = np.real(ifft(A, fft_size))[:n]

        # Alternating projections
        for _ in range(n_iters):
            # Project onto non-negativity
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a /= s  # Normalize sum to 1

            # Project onto flat spectrum (in the zero-padded FFT)
            A_curr = fft(a, fft_size)
            mag = np.abs(A_curr)
            mag[0] = 1.0  # Keep DC fixed (preserves sum)
            # Set all other magnitudes to target (average of non-DC magnitudes)
            if fft_size > 1:
                target_mag = mag[1:].mean()
                if target_mag < 1e-10:
                    break
                A_curr[1:] = target_mag * A_curr[1:] / np.maximum(mag[1:], 1e-15)
            # Enforce Hermitian symmetry
            for k in range(1, fft_size // 2 + 1):
                if k < fft_size - k:
                    A_curr[fft_size - k] = np.conj(A_curr[k])
                elif k == fft_size - k:
                    A_curr[k] = np.real(A_curr[k])
            a = np.real(ifft(A_curr, fft_size))[:n]

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a


def chirp_init(n, rng):
    """Initialize with a chirp-like shape."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    # Add chirp modulation
    freq = rng.uniform(2.0, 5.0)
    a = a * (1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):

    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    inits['chirp'] = chirp_init(n, rng)
    inits['itr_proj'] = iterative_projection_init(n, rng, n_iters=100)

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            elif len(parent) > n:
                pi = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                pi = np.maximum(pi, 0)
                psi = pi.sum()
                if psi > 1e-10:
                    inits['parent_ds'] = pi / psi
            else:
                pi = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                pi = np.maximum(pi, 0)
                psi = pi.sum()
                if psi > 1e-10:
                    inits['parent_up'] = pi / psi

    return inits

    print("=== Phase 2: Deep L-BFGS-B ===")
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    target_lengths = sorted(set([l for l in [
        best_n // 2, (best_n * 3) // 4, best_n, (best_n * 5) // 4, best_n * 2
    ] if 96 <= l <= 1024 and l >= 64]))

    # Also include top scan results
    for (_, n, _, _) in scan_results[:3]:
        if n not in target_lengths:
            target_lengths.append(n)
            target_lengths = sorted(target_lengths)

    phase2_end = deadline - 200.0
    total_configs = len(target_lengths) * 2
    time_per_config = max(20.0, (phase2_end - time.monotonic()) / max(1, total_configs))

    for n in target_lengths[:6]:
        if time.monotonic() > phase2_end - 30:
            break

        inits = generate_initializations(n, parent, rng)
        if best_sequence is not None:
            bs = np.maximum(best_sequence, 0)
            bs_sum = bs.sum()
            if bs_sum > 1e-10:
                bs_norm = bs / bs_sum
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(bs_norm)), bs_norm)
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits['from_best'] = pp / ps

        init_names = ['from_best', 'itr_proj', 'chirp', 'gauss_0.3', 'parent', 'random']
        init_names = [nm for nm in init_names if nm in inits]
        init_names = init_names[:2]

        for name in init_names:
            if time.monotonic() > phase2_end - 15:
                break
            a0 = inits[name]
            a_cur = a0.copy()

            # Beta schedule: start smooth, end sharp
            betas = [30.0, 15.0, 8.0, 5.0, 3.0]
            t_remaining = phase2_end - time.monotonic() - 10
            time_per_beta = t_remaining / len(betas)

            for beta in betas:
                if time.monotonic() > phase2_end - 8:
                    break
                t_single = min(time_per_beta, phase2_end - time.monotonic() - 5)
                if t_single < 5:
                    break
                maxiter = int(min(400, max(50, t_single * 25)))
                a_opt, _ = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
                score = score_and_update(a_opt)
                a_cur = a_opt

    print("Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Fine-tuning
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        n_best = len(best_sequence)
        a_norm = best_sequence / best_sequence.sum()

        # L-BFGS-B with very low beta
        for beta in [3.0, 2.0, 1.5, 1.0]:
            if remaining < 10:
                break
            t_single = min(remaining / 5, 120.0)
            if t_single < 5:
                break
            maxiter = int(min(500, max(100, t_single * 30)))
            a_opt, _ = optimize_lbfgsb(a_norm, beta=beta, maxiter=maxiter)
            score = score_and_update(a_opt)
            a_norm = best_sequence / best_sequence.sum()
            remaining = deadline - time.monotonic() - 15.0

        # Exact subgradient steps
        if remaining > 15:
            n_sub = int(min(200, remaining * 5))
            a_sub, _ = subgradient_step(a_norm, n_steps=n_sub, lr=0.005)
            score = score_and_update(a_sub)
            remaining = deadline - time.monotonic() - 15.0

        # Perturbation + re-optimize
        if remaining > 30:
            for _ in range(8):
                if remaining < 10:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(a_norm + noise)
                t_single = min(40.0, remaining / 4)
                if t_single < 5:
                    break
                maxiter = int(min(200, max(30, t_single * 20)))
                a_opt, _ = optimize_lbfgsb(a_pert, beta=5.0, maxiter=maxiter)
                score = score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 15.0

    # Phase 4: Try altern
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 80:
            # Try a nearby length
            alt_n = best_sequence.size + 64 if best_sequence.size < 512 else best_sequence.size - 64
            alt_n = max(96, alt_n)
            inits_alt = generate_initializations(alt_n, parent, rng)
            bs = np.maximum(best_sequence, 0)
            bs_sum = bs.sum()
            if bs_sum > 1e-10:
                bs_norm = bs / bs_sum
                pp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, len(bs_norm)), bs_norm)
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits_alt['from_best'] = pp / ps
            a0_alt = inits_alt.get('from_best', inits_alt.get('itr_proj', inits_alt['uniform']))
            for beta in [20.0, 5.0, 2.0]:
                if time.monotonic() > deadline - 10:
                    break
                t_single = min(60.0, deadline - time.monotonic() - 8)
                if t_single < 5:
                    break
                maxiter = int(min(300, max(50, t_single * 20)))
                a_opt, _ = optimize_lbfgsb(a0_alt, beta=beta, maxiter=maxiter)
                score_and_update(a_opt)
                a0_alt = a_opt

    # Final check
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
