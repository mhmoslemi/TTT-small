This is the problem of finding a sequence whose self-convolution is as flat as possible. This is related to the "perfect sequence" or "allpass" problem in signal processing.

For the flat case, we'd want conv(a,a)[k] = 1/(2n-1) for all k. But conv(a,a)[0] = a[0]^2, which must be non-negative. The constraint is that the convolution of a non-negative sequence with itself is non-negative (which is automatic).

Actually, this is related to the problem of finding "ideal" pulse shapes in communications. A sequence whose self-convolution is a rectangle (or as close as possible) is called a "perfect" sequence.

For real sequences, perfect sequences (where conv(a,a) is exactly a rectangle) don't exist for n > 1 in general. But we can get close.

This is also related to the "Barker sequence" problem (which minimizes the maximum sidelobe of the autocorrelation), but here we're dealing with convolution, not autocorrelation.

Key difference: autocorrelation is (a * reverse(a)) which is always symmetric, while convolution (a*a) is not necessarily symmetric (unless a is symmetric).

For the convolution to be flat, we essentially need a to be a "root" of the rectangle sequence. That is, if R is a rectangle of length 2n-1, we need a such that a*a = R. In the z-domain, this means A(z)^2 = R(z), so A(z) = sqrt(R(z)). This is only possible if R(z) is a perfect square in the polynomial ring.

For a rectangle R of length 2n-1 with value 1/(2n-1), R(z) = (1/(2n-1)) * (1 + z + ... + z^{2n-2}). This is a perfect square iff 2n-1 is odd (which it is) and the polynomial is a square. But 1 + z + ... + z^{2n-2} = (z^{2n-1} - 1)/(z-1). For this to be a perfect square polynomial, we'd need 2n-1 to be... actually this never works for n > 1 over the reals.

import time


def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        a_blend = project_simplex(a_blend)

        # Evaluate
        conv_b = fast_convolve(a_blend)
        obj_b = 2.0 * n * conv_b.max()

        if obj_b < best_obj - 1e-13:
            best_obj = obj_b
            best_a = a_blend.copy()
            stall = 0
        else:
            stall += 1
            if stall == 20:
                cur_blend = max(0.1, cur_blend * 0.7)
            elif stall == 40:
                cur_blend = max(0.05, cur_blend * 0.5)
            elif stall >= stall_limit:
                break

        a = a_blend
        it += 1

        # Adaptive: if converging, reduce blend
        if it > 50 and stall < 5:
            cur_blend = max(0.1, cur_blend * 0.995)

    return best_a, best_obj


def spectral_refine_v2(a0, time_limit, alpha=0.4, blend=0.6):
    """
    Variant: uses clipping + power combination, no rescale of c_flat sum.
    More aggressive flattening.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    N = 2 ** (2 * n - 1).bit_length()

    conv_a = fast_convolve(a)
    best_obj = 2.0 * n * conv_a.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, N)
        C = A * A
        c = np.fft.irfft(C, N)[:2 * n - 1]

        # Clip and power
        c_max = c.max()
        if c_max < 1e-15:
            break
        c_clip = np.minimum(c, c_max * 0.8)
        c_pos = np.maximum(c_clip, 1e-30)
        c_flat = c_pos ** alpha
        c_sum = c_flat.sum()
        if c_sum > 1e-15:
            c_flat *= (c.sum() / c_sum)

        C_flat = np.fft.rfft(c_flat, N)
        A_new = np.sqrt(C_flat + 0j)
        a_new = np.fft.irfft(A_new, N)[:n]

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new

        a_blend = (1.0 - blend) * a + blend * a_new
        a_blend = project_simplex(a_blend)

        conv_b = fast_convolve(a_blend)
        obj_b = 2.0 * n * conv_b.max()

        if obj_b < best_obj - 1e-13:
            best_obj = obj_b
            best_a = a_blend.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

        a = a_blend
        it += 1

    return best_a, best_obj


def subgrad_refine(a0, time_limit, lr_init=0.01, momentum=0.92,
                   top_k=30, temp=0.0001):
    """Subgradient descent with momentum and soft peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv_a = fast_convolve(a)
    best_obj = 2.0 * n * conv_a.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = conv_a
        # Find peaks with soft weighting
        if top_k >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]

        pmax = c[peaks].max()
        if pmax < 1e-15:
            break

        # Soft weights based on relative height
        scaled = c[peaks] / (temp + pmax * 0.1)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(c[peaks])] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        # Update with momentum
        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        conv_a = fast_convolve(a)
        cur_obj = 2.0 * n * conv_a.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng_loc = np.random.default_rng(hash(tuple(a[:5])) % (2**31))
            a = project_simplex(a + rng_loc.standard_normal(n) * 0.005)
            conv_a = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.5)

        lr *= 0.9995

    return best_a, best_obj


def build_lp_matrix(n):
    """Pre-build the LP constraint matrix structure."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = 1.0  # will be filled with a[idx] later
    return A_ub


def solve_lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)_k <= M for all k."""
    from scipy.optimize import linprog

    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]

    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(a0, time_limit, blend_init=0.3, blend_min=0.01):
    """LP-based directional optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv_a = fast_convolve(a)
    best_obj = 2.0 * n * conv_a.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit - 10:
        M = conv_a.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g /= g_sum

        # Check sparsity
        if g.max() > 0.4:
            uniform = np.ones(n) / n
            g = 0.7 * g + 0.3 * uniform

        a_new = (1.0 - blend) * a + blend * g
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 15:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break

        a = a_new
        conv_a = conv_new

    return best_a, best_obj


def coord_swap_polish(a_in, max_rounds=20, time_limit=30.0):
    """Coordinate swap polish: try moving mass between pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv_a = fast_convolve(a)
    best_obj = 2.0 * n * conv_a.max()

    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Focus on non-zero coordinates
        active = np.where(a > 1e-6)[0]
        if len(active) < 2:
            break

        # Try swaps between active coordinates
        rng = np.random.default_rng(rnd * 7 + n)
        n_tries = min(200, len(active) * 3)

        for _ in range(n_tries):
            i = active[rng.integers(len(active))]
            j = active[rng.integers(len(active))]
            if i == j:
                continue

            for frac in [0.1, 0.3, 0.5]:
                delta = a[i] * frac
                if delta < 1e-10:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                if a_test[i] < 0:
                    continue
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-12:
                    a = a_test
                    best_obj = obj_test
                    conv_a = conv_test
                    improved = True
                    break

        if not improved:
            break

    return a


def gen_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random with structure
    freq = rng.integers(1, n // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi)
    wave = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    wave = np.maximum(wave, 0.01)
    inits['wave'] = wave / wave.sum()

    # Parent resampled
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), p)
                p_res = np.maximum(p_res, 0)
                ps2 = p_res.sum()
                if ps2 > 1e-10:
                    inits['parent'] = p_res / ps2

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining spectral power iteration, subgradient,
    LP direction finding, and coordinate swap polish across multiple
    sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if np.isfinite(par_score):
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Spectral scan across lengths
    print("=== Phase 1: Spectral Scan ===")
    lengths = [256, 384, 512, 640, 768, 1024]

    # Prioritize: parent length first, then others
    if parent is not None:
        pl = len(parent)
        # Find closest length or add it
        closest = min(lengths, key=lambda x: abs(x - pl))
        lengths = [closest] + [l for l in lengths if l != closest]

    results = []
    phase1_budget = min(600.0, (deadline - time.monotonic()) * 0.4)
    phase1_end = time.monotonic() + phase1_budget

    inits_per_len = 4
    t_per = max(8.0, phase1_budget / (len(lengths) * inits_per_len))

    for n in lengths:
        if time.monotonic() > phase1_end - 15:
            break
        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'uniform', 'wave', 'random', 'bimodal',
                'triangle', 'edge', 'gauss']
        names = [nm for nm in prio if nm in inits][:inits_per_len]

        for name in names:
            if time.monotonic() > phase1_end - 5:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 3)
            if t_alloc < 4:
                break

            a0 = inits[name]

            # Spectral refinement (main engine)
            a_opt, obj_opt = spectral_refine(a0, t_alloc * 0.7,
                                             alpha_start=0.2,
                                             alpha_end=0.9,
                                             blend=0.5)

            # Quick subgrad polish
            t_sg = min(3.0, t_alloc * 0.2)
            if t_sg > 1:
                a_sg, obj_sg = subgrad_refine(a_opt, t_sg,
                                               lr_init=0.008, momentum=0.9,
                                               top_k=20, temp=0.0001)
                if obj_sg < obj_opt:
                    a_opt, obj_opt = a_sg, obj_sg

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:10]:
        key = (n, name)
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
        if len(top_list) >= 6:
            break
    if best_sequence is not None and (len(best_sequence), 'best') not in seen:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 200.0
    t_per_deep = max(30.0, remaining / max(1, len(top_list)))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break

        # Deep spectral with different parameters
        a_opt, obj_opt = spectral_refine(a0, t_alloc * 0.35,
                                          alpha_start=0.15,
                                          alpha_end=0.85,
                                          blend=0.4,
                                          stall_limit=60)

        # Spectral variant
        t_v2 = t_alloc * 0.2
        if t_v2 > 5:
            a_v2, obj_v2 = spectral_refine_v2(a_opt, t_v2, alpha=0.35, blend=0.5)
            if obj_v2 < obj_opt:
                a_opt, obj_opt = a_v2, obj_v2

        # Subgradient refinement
        t_sg = t_alloc * 0.2
        if t_sg > 5:
            a_sg, obj_sg = subgrad_refine(a_opt, t_sg,
                                           lr_init=0.005, momentum=0.93,
                                           top_k=40, temp=0.00005)
            if obj_sg < obj_opt:
                a_opt, obj_opt = a_sg, obj_sg

        # LP refinement
        t_lp = t_alloc * 0.15
        if t_lp > 10 and n <= 768:
            a_lp, obj_lp = lp_refine(a_opt, t_lp, blend_init=0.2, blend_min=0.01)
            if obj_lp < obj_opt:
                a_opt, obj_opt = a_lp, obj_lp

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}: {score:.6f}")

    # Phase 3: Polish
    print("=== Phase 3: Polish ===")
    if best_sequence is not None:
        n_b = len(best_sequence)

        # Coordinate swap
        t_cd = min(60.0, (deadline - time.monotonic()) * 0.2)
        if t_cd > 10:
            a_cd = coord_swap_polish(best_sequence, max_rounds=30,
                                     time_limit=t_cd)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")

        # Final spectral + subgrad cycles
        remaining = deadline - time.monotonic() - 50.0
        for cycle in range(4):
            if time.monotonic() > deadline - 40:
                break
            t_cyc = remaining / (4 - cycle)
            t_cyc = min(t_cyc, 80.0)
            if t_cyc < 10:
                break

            # Spectral with very gentle alpha
            a_s, obj_s = spectral_refine(best_sequence, t_cyc * 0.5,
                                          alpha_start=0.7, alpha_end=0.98,
                                          blend=0.3, stall_limit=40)
            if obj_s < 2.0 * n_b * fast_convolve(best_sequence).max():
                best_sequence = a_s

            # Subgrad
            a_g, obj_g = subgrad_refine(best_sequence, t_cyc * 0.3,
                                         lr_init=0.003, momentum=0.94,
                                         top_k=50, temp=0.00003)
            if obj_g < 2.0 * n_b * fast_convolve(best_sequence).max():
                best_sequence = a_g

            score = float(evaluate_sequence(best_sequence.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                print(f"  Cycle {cycle}: {score:.6f}")

            # Quick CD
            a_cd2 = coord_swap_polish(best_sequence, max_rounds=15,
                                      time_limit=15.0)
            score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(score_cd2) and score_cd2 < best_score:
                best_score = score_cd2
                best_sequence = a_cd2.copy()
                print(f"  CD cycle {cycle}: {score_cd2:.6f}")

        # Random perturbations on best
        if time.monotonic() < deadline - 30:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 25:
                    break
                for rep in range(5):
                    if time.monotonic() > deadline - 20:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = spectral_refine(a_p, 3.0,
                                               alpha_start=0.5,
                                               alpha_end=0.9,
                                               blend=0.4)
                    a_opt2, obj2 = subgrad_refine(a_opt, 2.0,
                                                  lr_init=0.004, momentum=0.9,
                                                  top_k=30, temp=0.0001)
                    if obj2 < 2.0 * n_b * fast_convolve(a_opt).max():
                        a_opt = a_opt2
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
