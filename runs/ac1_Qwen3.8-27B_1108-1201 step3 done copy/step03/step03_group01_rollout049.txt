            a[v] += reduce_amount * 2 / len(valley_positions)
    
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- LP-based direction finding with sparse peak constraints (top-k only)
- Nesterov-accelerated subgradient with adaptive temperature
- Multiple sequence lengths and diverse initializations
- Final coordinate-descent polish
- Adaptive time allocation focusing on best candidates

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a, top_k_indices):
    """Build matrix C such that C @ g = a * g (full linear convolution),
    but only for the specified rows (top_k_indices)."""
    n = len(a)
    C = np.zeros((len(top_k_indices), n))
    for row, k in enumerate(top_k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            C[row, j] = a[k - j]
    return C


def build_conv_matrix_full(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_and_conv(a, n):
    """Compute objective value and convolution for normalized a."""
    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    return obj, conv


def solve_lp_direction(a, M, top_k=30):
    """
    Solve LP: maximize sum(g) s.t. a*g <= M at top-k peak positions, g >= 0.
    Uses sparse constraints for speed.
    Returns (g, g_sum, success).
    """
    n = len(a)
    conv = fast_convolve(a)
    # Get top-k peak positions
    k = min(top_k, 2 * n - 1)
    peak_indices = np.argpartition(conv, -k)[-k:]
    
    A_ub = build_conv_matrix_sparse(a, peak_indices)
    c = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g, g_sum, True
    return a.copy(), 1.0, False

        if step_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD step {step_count}, obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sine (single mode)
    sine = np.sin(np.pi * (x + 0.5) / n) ** 2 + 0.01
    inits['sine'] = sine / sine.sum()
    
    # Two sine modes
    sine2 = (np.sin(np.pi * (x + 0.5) / n) ** 2 + 
             0.5 * np.sin(2 * np.pi * (x + 0.5) / n) ** 2) + 0.01
    inits['sine2'] = sine2 / sine2.sum()
    
    # Triangular wave
    tri = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0) + 0.01
    inits['triangle'] = tri / tri.sum()
    
    # Exponential ramp
    ramp = np.exp(-0.5 * x / (n * 0.3)) + 0.01
    inits['ramp'] = ramp / ramp.sum()
    
    # Power spectrum shaped (1/f noise-like)
    pw = 1.0 / (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2) + 0.001
    inits['gaussian'] = g / g.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    return inits


def peak_shave_perturb(a, n, rng, intensity=0.02):
    """Perturb by targeting peak regions of the convolution."""
    conv = fast_convolve(a)
    peak_pos = np.argmax(conv)
    
    # Find contributing indices
    j_start = max(0, peak_pos - n + 1)
    j_end = min(n - 1, peak_pos)
    idx = np.arange(j_start, j_end + 1)
    contributions = a[idx] * a[peak_pos - idx]
    
    # Perturb the most contributing indices more
    a_pert = a.copy()
    for i, j in zip(idx, peak_pos - idx):
        contrib = a[i] * a[j]
        noise_scale = intensity * (contrib / max(contributions.max(), 1e-15))
        a_pert[i] += rng.standard_normal() * noise_scale
        if j != i:
            a_pert[j] += rng.standard_normal() * noise_scale
    
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-15:
        return a
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: LP scan across lengths and initializations
    Phase 2: Deep optimization on top candidates (LP + subgradient)
    Phase 3: Coordinate descent polish + final perturbations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    # Phase 1: LP Scan
    print("=== Phase 1: LP Scan ===")
    t_start = time.monotonic()
    phase1_limit = min(500.0, (budget_s - 20.0) * 0.35)
    phase1_deadline = t_start + phase1_limit
    
    lengths = [256, 512, 768]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 40:
            break
        
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'random',
                    'sine', 'sine2', 'triangle', 'ramp', 'bimodal', 'gaussian', 'power']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        # Pick top 4 initializations per length
        init_names = init_names[:4]
        
        t_per_init = max(12.0, (phase1_limit - 20.0) / (len(lengths) * 4))
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, top_k=20)
            score = float(evaluate_sequence(a_opt.tolist()))
            
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()
    
    # Select top candidates (unique by length+init)
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 4:
            break
    
    # Ensure best is included
    if best_sequence is not None:
        best_n = len(best_sequence)
        found = False
        for n, a in top_configs:
            if n == best_n and np.allclose(a, best_sequence, atol=1e-6):
                found = True
                break
        if not found:
            top_configs.append((best_n, best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 80.0
    n_candidates = max(1, len(top_configs))
    t_per_deep = max(30.0, remaining * 0.6 / n_candidates)
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 70:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 50)
        if t_alloc < 10:
            break
        
        # Split time: 60% LP, 40% subgradient
        t_lp = t_alloc * 0.6
        t_sg = t_alloc * 0.4
        
        print(f"  Deep {i} (n={n}), t_lp={t_lp:.0f}s, t_sg={t_sg:.0f}s")
        
        # LP phase
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_lp, top_k=30)
        score = float(evaluate_sequence(a_opt.tolist()))
        
        # Subgradient phase starting from LP result
        a_opt2, obj_opt2 = subgradient_optimize(
            a_opt, time_limit=t_sg, lr_init=0.008, momentum=0.94,
            top_k=60, temp=0.00015
        )
        score2 = float(evaluate_sequence(a_opt2.tolist()))
        
        if score2 < score:
            a_opt = a_opt2
            score = score2
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    Improved! Deep {i}: {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # Phase 3: Polish on the best candidate
    print("=== Phase 3: Polish ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 20:
            # Subgradient fine-tune with very small lr
            t_sg_fine = remaining * 0.4
            a_opt, _ = subgradient_optimize(
                best_sequence, time_limit=t_sg_fine,
                lr_init=0.004, momentum=0.95, top_k=80, temp=0.0001
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SG fine: {score:.6f}")
            
            # Coordinate descent
            remaining = deadline - time.monotonic() - 10.0
            if remaining > 15:
                t_cd = remaining * 0.5
                a_opt, _ = coordinate_descent_polish(best_sequence, time_limit=t_cd)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CD: {score:.6f}")
            
            # Peak-shaving perturbations + quick re-opt
            remaining = deadline - time.monotonic() - 5.0
            if remaining > 20:
                n_best = len(best_sequence)
                n_perturb = max(1, int(remaining / 20))
                t_per_perturb = remaining / n_perturb
                
                for pi in range(n_perturb):
                    if time.monotonic() > deadline - 5:
                        break
                    if t_per_perturb < 5:
                        break
                    
                    a_pert = peak_shave_perturb(best_sequence, n_best, rng, 
                                                intensity=0.01 + 0.005 * pi)
                    
                    # Quick LP + SG
                    t_quick = min(t_per_perturb - 2, deadline - time.monotonic() - 3)
                    if t_quick < 3:
                        break
                    
                    t_lp_q = t_quick * 0.5
                    t_sg_q = t_quick * 0.5
                    
                    a_opt, _ = lp_optimize(a_pert, time_limit=t_lp_q, top_k=25)
                    a_opt2, _ = subgradient_optimize(a_opt, time_limit=t_sg_q,
                                                     lr_init=0.006, momentum=0.93,
                                                     top_k=50, temp=0.0002)
                    
                    score = float(evaluate_sequence(a_opt2.tolist()))
                    score_lp = float(evaluate_sequence(a_opt.tolist()))
                    if score_lp < score:
                        score = score_lp
                        a_opt2 = a_opt
                    
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt2.copy()
                        print(f"  Perturb {pi}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
