"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LP-based directional optimization (min-max formulation)
2. Simulated annealing with O(n) convolution updates
3. Smart coordinate descent for final polishing
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_minmax_lp(f, n):
    """Solve: min t s.t. (f*g)_k <= t, g>=0, sum(g)=1"""
    # Variables: g_0, ..., g_{n-1}, t
    # Total variables: n+1
    # Objective: minimize t => c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: (f*g)_k - t <= 0 for all k
    # f*g is a linear function of g
    # (f*g)_k = sum_j f[min(n-1, 2n-1-k)... ] 
    
    # Actually, (f*g)_k = sum_j f[j] * g[k-j] where 0<=k-j<n
    # So (f*g)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Let l = k-j, so j = k-l
    # (f*g)_k = sum_{l=max(0,k-n+1)}^{min(n-1,k)} f[k-l] * g[l]
    
    # A_ub has (2n-1) rows, each of length n+1
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            l = k - j  # index in g
            A_ub[k, l] = f[j]
        A_ub[k, -1] = -1.0  # coefficient of t
    
    # sum(g) = 1 => A_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g_i >= 0, t >= 0
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return np.maximum(result.x[:n], 0.0), result.fun
    except Exception:
        pass
    return None, None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP directional optimization,
simulated annealing with fast FFT evaluation, and coordinate descent polishing.
Explores multiple sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max(), conv


def solve_minmax_lp(f, n, time_limit=20):
    """
    Solve: min t s.t. (f*g)_k <= t for all k, g>=0, sum(g)=1.
    Returns (g_star, t_star) or (None, None).
    """
    # Variables: [g_0, ..., g_{n-1}, t], total n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Build A_ub: (2n-1) x (n+1)
    # Row k: sum_l f[k-l]*g[l] - t <= 0
    rows_ub = []
    cols_ub = []
    vals_ub = []
    
    for k in range(2 * n - 1):
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        if l_min <= l_max:
            l_range = np.arange(l_min, l_max + 1)
            rows_ub.extend([k] * len(l_range))
            cols_ub.extend(l_range.tolist())
            vals_ub.extend(f[k - l_range].tolist())
        rows_ub.append(k)
        cols_ub.append(n)
        vals_ub.append(-1.0)
    
    A_ub = sp.csr_matrix((np.array(vals_ub), (np.array(rows_ub), np.array(cols_ub))),
                          shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    # sum(g) = 1
    A_eq = sp.csr_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            g = np.maximum(result.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
            return g, result.fun
    except Exception:
        pass
    return None, None


def lp_directional_optimize(a_init, time_limit, max_lp_time=15, 
                            blend_init=0.5, blend_min=0.01):
    """
    LP-based directional optimization.
    At each step: solve min-max LP to get direction g, blend with current a.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g, t_star = solve_minmax_lp(a, n, time_limit=max_lp_time)
        if g is None:
            break
        
        # If the LP says we can achieve a lower peak, great
        # Blend: a_new = (1-blend)*a + blend*g
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= 8:
            blend = max(blend_min, blend * 0.6)
            stall = 0
            if blend < blend_min * 1.5:
                break
        
        if time.monotonic() - t0 > time_limit * 0.9:
            break
    
    return best_a, best_obj


def simulated_annealing(a_init, time_limit, T0=0.0005, cooling=0.99995,
                        top_k=15, max_move=0.05, min_move=0.003):
    """
    Simulated annealing on the simplex.
    Proposes moves that reduce peak contributions.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T0
    rng = np.random.default_rng(777)
    iterations = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive move size
        move_frac = rng.uniform(min_move, max(0.001, max_move * T / T0))
        
        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Choose peak weighted by value
        peak_vals = conv[peaks]
        total_pv = peak_vals.sum()
        if total_pv < 1e-15:
            break
        peak_probs = peak_vals / total_pv
        k = int(peaks[rng.choice(len(peaks), p=peak_probs)])
        
        # Source: coordinate with high contribution to peak k
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        if l_min > l_max:
            continue
        l_range = np.arange(l_min, l_max + 1)
        # Contribution of a[l] to conv[k] is a[k-l]
        contrib = a[k - l_range]  # What a[l] multiplies
        # We want to reduce a[l] where a[l]*a[k-l] is large
        weights = a[l_range] * contrib
        w_sum = weights.sum()
        if w_sum < 1e-15:
            continue
        weights /= w_sum
        i = int(l_range[rng.choice(len(l_range), p=weights)])
        
        if a[i] < move_frac * 0.5:
            continue
        
        # Destination: coordinate that doesn't contribute much to peak k
        # Use uniform sampling from coordinates not in the peak's support
        # or weighted by low contribution
        j_weights = np.ones(n)
        # Reduce weight for coordinates that contribute to the peak
        jl_min = max(0, k - n + 1)
        jl_max = min(n - 1, k)
        if jl_min <= jl_max:
            j_range = np.arange(jl_min, jl_max + 1)
            j_weights[j_range] = 0.3 / (a[k - j_range] + 1e-10)
        j_weights[i] = 0
        jw_sum = j_weights.sum()
        if jw_sum < 1e-10:
            j = int(rng.integers(n))
            if j == i:
                continue
        else:
            j = int(rng.choice(n, p=j_weights / jw_sum))
            if j == i:
                continue
        
        # Propose move
        delta = a[i] * move_frac
        if delta < 1e-12:
            continue
        
        a[i] -= delta
        a[j] += delta
        
        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()
        
        delta_obj = new_obj - cur_obj
        if delta_obj < 0 or (T > 1e-10 and rng.random() < np.exp(-delta_obj / T)):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            a[i] += delta
            a[j] -= delta
            no_improve += 1
        
        T *= cooling
        iterations += 1
        
        # Reheat if stuck
        if no_improve > 5000 and T < T0 * 0.1:
            T = T0 * 0.3
            no_improve = 0
        
        # Check time more frequently
        if iterations % 1000 == 0:
            if time.monotonic() - t0 > time_limit:
                break
    
    return best_a, best_obj


def coordinate_descent(a_init, max_rounds=30, threshold=1e-8):
    """
    Coordinate descent: try moving mass between pairs of coordinates.
    Only considers pairs where the source has significant mass.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    # Get active coordinates (those with significant mass)
    active_thresh = 1.0 / n * 0.1  # 10% of uniform
    
    for rnd in range(max_rounds):
        improved = False
        
        # Sort by mass to limit candidates
        active = np.where(a > active_thresh)[0]
        if len(active) == 0:
            break
        
        # For each active coordinate, try moving to a few others
        for i in active:
            if a[i] < active_thresh:
                continue
            
            # Try moving to n_low (low mass coordinates) and n_rand (random)
            # Try moving to coordinates that are "far" from i in the convolution sense
            candidates = np.setdiff1d(np.arange(n), i)
            # Prioritize low-mass coordinates
            candidates = candidates[candidates != i]
            if len(candidates) > 50:
                # Take top 25 by lowest mass + 25 random
                by_mass = np.argsort(a[candidates])[:25]
                rng_cd = np.random.default_rng(rnd * 31 + i)
                by_rand = rng_cd.choice(candidates, size=25, replace=False)
                candidates = np.concatenate([candidates[by_mass], candidates[by_rand]])
            
            for j in candidates:
                if j == i:
                    continue
                # Try a few delta values
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[i] * frac
                    if delta < 1e-9:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        # Don't break; try more moves from this i
        
        if not improved:
            break
    
    return best_a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

                                        np.linspace(0, 1, len(parent)), p)

                prs = p_resampled.sum()
                if prs > 1e-10:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP directional optimization,
simulated annealing with fast FFT evaluation, and coordinate descent polishing.
Explores multiple sequence lengths and initializations to find near-optimal
non-negative step function heights.
"""

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to sum 1, ensure non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max(), conv


def solve_minmax_lp(f, n, time_limit=20):
    """
    Solve: min t s.t. (f*g)_k <= t for all k=0..2n-2, g>=0, sum(g)=1.
    Returns (g_star, t_star) or (None, None).
    """
    c = np.zeros(n + 1)
    c[-1] = 1.0

    rows_ub = []
    cols_ub = []
    vals_ub = []

    for k in range(2 * n - 1):
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        if l_min <= l_max:
            l_range = np.arange(l_min, l_max + 1)
            rows_ub.extend([k] * len(l_range))
            cols_ub.extend(l_range.tolist())
            vals_ub.extend(f[k - l_range].tolist())
        rows_ub.append(k)
        cols_ub.append(n)
        vals_ub.append(-1.0)

    A_ub = sp.csr_matrix((np.array(vals_ub), (np.array(rows_ub), np.array(cols_ub))),
                          shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = sp.csr_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))),
                          shape=(1, n + 1))
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            g = np.maximum(result.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g /= gs
            return g, result.fun
    except Exception:
        pass
    return None, None


def lp_directional_optimize(a_init, time_limit, max_lp_time=15,
                            blend_init=0.5, blend_min=0.01):
    """
    LP-based directional optimization.
    At each step: solve min-max LP to get direction g, blend with current a.
    """
    a = normalize(a_init.copy())
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g, t_star = solve_minmax_lp(a, n, time_limit=max(3, max_lp_time))
        if g is None:
            break

        # Check if LP gives meaningful improvement
        if t_star is not None and t_star >= M * (1.0 - 1e-8):
            stall += 2
        else:
            stall = 0

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 10:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend <= blend_min * 1.2:
                break

        it += 1
        if time.monotonic() - t0 > time_limit * 0.92:
            break

    return best_a, best_obj


def simulated_annealing(a_init, time_limit, T0=0.0005, cooling=0.99995,
                        top_k=15, max_move=0.05, min_move=0.003,
                        reheat_threshold=8000, reheat_temp=0.3):
    """
    Simulated annealing on the simplex with peak-guided moves.
    """
    a = normalize(a_init.copy())
    n = len(a)

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    t0 = time.monotonic()
    T = T0
    rng = np.random.default_rng(42 + n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Adaptive move size based on temperature
        move_frac = rng.uniform(min_move, max(0.001, max_move * (T / T0 + 0.1)))

        # Find top-k peaks
        if top_k >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Choose peak weighted by value
        peak_vals = conv[peaks]
        total_pv = peak_vals.sum()
        if total_pv < 1e-15:
            break
        peak_probs = peak_vals / total_pv
        k = int(peaks[rng.choice(len(peaks), p=peak_probs)])

        # Source coordinate: high contribution to peak k
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        if l_min > l_max:
            continue
        l_range = np.arange(l_min, l_max + 1)
        contrib = a[k - l_range]
        weights = a[l_range] * contrib
        w_sum = weights.sum()
        if w_sum < 1e-15:
            continue
        weights /= w_sum
        i = int(l_range[rng.choice(len(l_range), p=weights)])

        if a[i] < move_frac * 0.3 * (1.0 / n):
            continue

        # Destination: low contribution to peak k
        j_weights = np.ones(n) * 0.5
        jl_min = max(0, k - n + 1)
        jl_max = min(n - 1, k)
        if jl_min <= jl_max:
            j_range = np.arange(jl_min, jl_max + 1)
            j_weights[j_range] = 0.1 / (a[k - j_range] + 1e-8)
        j_weights[i] = 0.0
        jw_sum = j_weights.sum()
        if jw_sum < 1e-10:
            j = int(rng.integers(n))
            if j == i:
                continue
        else:
            j = int(rng.choice(n, p=j_weights / jw_sum))
            if j == i:
                continue

        # Propose move
        delta = a[i] * move_frac
        if delta < 1e-12:
            continue

        a[i] -= delta
        a[j] += delta

        new_conv = fast_convolve(a)
        new_obj = 2.0 * n * new_conv.max()

        delta_obj = new_obj - cur_obj
        if delta_obj < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_obj / T)):
            cur_obj = new_obj
            conv = new_conv
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            a[i] += delta
            a[j] -= delta
            no_improve += 1

        T *= cooling
        it += 1

        # Reheat if stuck
        if no_improve > reheat_threshold and T < T0 * 0.05:
            T = T0 * reheat_temp
            no_improve = 0
            # Also perturb
            pert = rng.standard_normal(n) * min_move * 0.5
            a = project_simplex(a + pert)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()

        if it % 5000 == 0:
            if time.monotonic() - t0 > time_limit:
                break

    return best_a, best_obj


def coordinate_descent(a_init, max_rounds=15):
    """
    Coordinate descent: try moving mass between pairs of coordinates.
    Uses limited candidate sets for efficiency.
    """
    a = normalize(a_init.copy())
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    active_thresh = 0.3 / n

    for rnd in range(max_rounds):
        improved = False
        active = np.where(a > active_thresh)[0]
        if len(active) == 0:
            break

        rng_cd = np.random.default_rng(1000 + rnd)

        for i in active:
            if a[i] < active_thresh:
                continue

            # Build candidate set: low-mass + random
            others = np.setdiff1d(np.arange(n), i)
            if len(others) > 60:
                by_mass = np.argsort(a[others])[:30]
                by_rand = rng_cd.choice(others, size=30, replace=False)
                candidates = np.concatenate([others[by_mass], others[by_rand]])
            else:
                candidates = others

            for j in candidates:
                for frac in [0.05, 0.15, 0.3]:
                    delta = a[i] * frac
                    if delta < 1e-9:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        best_a = a.copy()
                        improved = True

        if not improved:
            break

    return best_a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['tri'] = tri / tri.sum()

    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    inits['rand'] = rng.dirichlet(np.ones(n) * 0.3)

    sine = np.exp(0.4 * np.sin(2 * np.pi * x / n * 3))
    inits['sine'] = sine / sine.sum()

    # Power-law (heavier edges)
    pw = 1.0 / (np.abs(x - n/2 + 0.5) + 1.0) ** 0.5
    inits['powlaw'] = pw / pw.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), p)
                p_res = np.maximum(p_res, 0)
                prs = p_res.sum()
                if prs > 1e-10:
                    inits['parent'] = p_res / prs

    return inits
