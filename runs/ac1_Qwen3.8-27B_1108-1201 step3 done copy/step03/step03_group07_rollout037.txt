"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
- Mirror descent (multiplicative weights) with Nesterov momentum
- Smooth max approximation with adaptive beta annealing
- FFT-based efficient convolution and gradient computation

- Peak-targeted coordinate refinement
- Multi-grid strategy (coarse-to-fine)

import numpy as np
import time
from scipy.fft import fft, ifft

def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def smooth_max(x, beta):
    """Smooth max approximation using log-sum-exp."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        return x_max, np.zeros_like(x)
    weights = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, weights

def compute_grad_conv(a, weights):
    """Gradient of sum_k w[k]*conv(k) with respect to a, via FFT."""
    n = len(a)
    # conv[k] = sum_i a[i]*a[k-i]
    # d conv[k]/d a[j] = a[k-j] (if 0<=k-j<n)
    # Gradient = sum_k w[k] * a[k-j] for valid k
    # This is the cross-correlation of weights with a
    N = 1 << (n + len(weights) - 1 + 63).bit_length()
    W = fft(weights, N)
    A = fft(a, N)
    # We need: grad[j] = sum_k w[k] * a[k-j] for 0<=k-j<n
    # = sum_k w[k] * a[(k-j) mod N] ... this is a correlation
    # Actually, let's think of it as: grad[j] = sum_k w[k] * a[k-j]
    # Let m = k-j, so k = m+j: grad[j] = sum_m w[m+j] * a[m]
    # This is the cross-correlation of w and a, evaluated at -j
    # In FFT terms: ifft(conj(fft(w)) * fft(a)) gives cross-correlation
    grad = np.real(ifft(np.conj(W) * A, N))[:n]
    return grad * 2.0  # factor of 2 from derivative of a*a

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def mirror_descent(a0, beta, n_iter, lr_init, lr_decay):
    """Mirror descent (multiplicative weights) on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autoconv(a).max()
    lr = lr_init
    
    for it in range(n_iter):
        conv = fast_autoconv(a)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if obj < 1.001:
            break
        
        grad = compute_grad_conv(a, w)
        
        # Multiplicative update (mirror descent with entropy)
        # a_new[i] proportional to a[i] * exp(-lr * grad[i])
        step = lr * grad
        step = np.clip(step, -50, 50)  # numerical stability
        a_new = a * np.exp(-step)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s
        
        # Ensure non-negativity (should already be positive)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        a = a_new
        lr *= lr_decay
    
    return best_a, best_obj

def nesterov_gradient(a0, beta, n_iter, lr_init, momentum):
    """Projected gradient descent with Nesterov momentum on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autoconv(a).max()
    lr = lr_init
    
    for it in range(n_iter):
        # Nesterov look-ahead
        a_look = project_simplex(a - momentum * v)
        
        conv = fast_autoconv(a_look)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_look.copy()
        
        if obj < 1.001:
            break
        
        grad = compute_grad_conv(a_look, w)
        
        # Update
        v_new = momentum * v + grad
        a_new = project_simplex(a - lr * v_new)
        
        a = a_new
        v = v_new
        lr *= 0.999
    
    return best_a, best_obj

def adaptive_beta_optimize(a0, betas, iters_per_beta, method='mirror'):
    """Run optimization with decreasing beta (annealing)."""
    a = a0.copy()
    for beta in betas:
        if method == 'mirror':
            a, _ = mirror_descent(a, beta, iters_per_beta, lr_init=0.1, lr_decay=0.995)
        else:
            a, _ = nesterov_gradient(a, beta, iters_per_beta, lr_init=0.05, momentum=0.9)
    return a

def peak_shave(a, n_steps=100, strength=0.5):
    """Iteratively shave peaks in the autocorrelation."""
    n = len(a)
    a = a.copy()
    for _ in range(n_steps):
        conv = fast_autoconv(a)
        k_max = int(np.argmax(conv))
        # Find which pairs (i, j) with i+j = k_max (mod something) contribute
        # conv[k] = sum_i a[i]*a[k-i]
        # To reduce conv[k_max], reduce pairs where both a[i] and a[k_max-i] are large
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        if i_min > i_max:
            break
        pair_vals = a[i_min:i_max+1] * a[k_max-i_max:k_max-i_min+1][::-1]
        total = pair_vals.sum()
        if total < 1e-15:
            break
        # Reduce the larger of each pair
        for i in range(i_min, i_max + 1):
            j = k_max - i
            if 0 <= j < n:
                if a[i] > a[j]:
                    a[i] -= strength * pair_vals[i - i_min] / total * a[i]
                else:
                    a[j] -= strength * pair_vals[i - i_min] / total * a[j]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a /= s
    return a

"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv(a, weights):
    """Gradient of sum_k w[k]*conv(k) w.r.t. a via FFT cross-correlation."""
    n = len(a)
    N = 1 << (n + len(weights) - 1).bit_length()
    W = fft(weights, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n]
    return grad * 2.0


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.argmax(~cond)
    if rho_idx == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho_idx - 1] - 1.0) / rho_idx
    return np.maximum(v - theta, 0.0)


def obj_exact(a):
    """Exact objective value (scale-invariant, assumes a is non-negative)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_autoconv(a_norm)
    return 2.0 * n * conv.max()


def mirror_descent(a0, beta, n_iter, lr_init, lr_decay, time_limit=None, t_start=None):
    """Mirror descent (multiplicative weights) with smooth max on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = obj_exact(a)
    lr = lr_init

    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break

        conv = fast_autoconv(a)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj * 0.9999:
            best_obj = obj
            best_a = a.copy()

        if obj < 1.0005:
            break

        grad = compute_grad_conv(a, w)

        step = lr * grad
        step = np.clip(step, -50, 50)
        a_new = a * np.exp(-step)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        a = a_new
        lr *= lr_decay

    # Final exact check
    final_obj = obj_exact(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    return best_a, best_obj


def nesterov_projection(a0, beta, n_iter, lr_init, momentum, time_limit=None, t_start=None):
    """Projected gradient descent with Nesterov momentum on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = obj_exact(a)
    lr = lr_init

    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break

        a_look = project_simplex(a - momentum * v)
        conv = fast_autoconv(a_look)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a_look.copy()

        if obj < 1.0005:
            break

        grad = compute_grad_conv(a_look, w)
        v = momentum * v + grad
        a = project_simplex(a - lr * v)
        lr *= 0.9995

    final_obj = obj_exact(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    return best_a, best_obj


def anneal_optimize(a0, betas, method='mirror', time_limit=60.0, t_start=None):
    """Run optimization with decreasing beta (annealing schedule)."""
    if t_start is None:
        t_start = time.monotonic()
    a = a0.copy()
    for beta in betas:
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 3:
            break
        if method == 'mirror':
            iters = int(min(300, max(30, remaining * 40)))
            a, _ = mirror_descent(a, beta, iters, lr_init=0.08, lr_decay=0.997,
                                  time_limit=remaining - 1, t_start=t_start)
        else:
            iters = int(min(250, max(30, remaining * 30)))
            a, _ = nesterov_projection(a, beta, iters, lr_init=0.04, momentum=0.85,
                                       time_limit=remaining - 1, t_start=t_start)
    return a


def flat_spectrum_init(n, rng, target_sum=None):
    """Initialize with approximately flat power spectrum (non-negative)."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for trial in range(30):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * rng.uniform(0, 2 * np.pi))

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = obj_exact(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def allpass_init(n, rng):
    """Initialize using allpass filter design for flat-ish autocorrelation."""
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for trial in range(20):
        # Random impulse response, make non-negative
        a = np.abs(rng.standard_normal(n)) + 0.1
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = obj_exact(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x, n - 1 - x) + 0.05
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%.2f' % sf] = g

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)

    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['allpass'] = allpass_init(n, rng)

    if parent is not None:
        pp = np.asarray(parent, dtype=np.float64)
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(pp) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(pp)), pp)
                pp_interp = np.maximum(pp_interp, 0)
                ps2 = pp_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp_interp / ps2

    return inits


def peak_shave(a, n_steps=50, strength=0.3):
    """Iteratively shave peaks in the autocorrelation."""
    n = len(a)
    a = a.copy()
    for _ in range(n_steps):
        conv = fast_autoconv(a)
        k_max = int(np.argmax(conv))
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        if i_min > i_max:
            break
        # Compute contribution of each pair
        contributions = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = k_max - i
            if 0 <= j < n:
                contributions[i] += a[i] * a[j]
                contributions[j] += a[i] * a[j]
        total = contributions.sum()
        if total < 1e-15:
            break
        # Reduce elements that contribute most to the peak
        reduction = strength * a * (contributions / max(total, 1e-15))
        a = a - reduction
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a /= s
        else:
            break
    return a


def coordinate_refine(a, n_iters=100):
    """Coordinate descent targeting the peaks of the autocorrelation."""
    n = len(a)
    a = a.copy()

        improved = False

            j = k_peak - i
            if not (0 <= j < n):

            old_i = a[i]

            for frac in [0.6, 0.8, 0.95]:
                a[i] = old_i * frac

                if s < 1e-10:

                a_norm = a / s
                obj_test = obj_exact(a_norm)
                if obj_test < best_obj * 0.9999:
                    best_obj = obj_test
                    a = a_norm
                    improved = True
                    break
                a[i] = old_i
            if improved:
                break

        if not improved:
            # Try the second peak
            conv2 = fast_autoconv(a)
            top2 = np.argsort(conv2)[::-1][:2]
            for k_p in top2[1:]:
                k_p = int(k_p)

                    old_i = a[i]

                        a[i] = old_i * frac
                        s = a.sum()
                        if s < 1e-10:

                        a_norm = a / s
                        obj_test = obj_exact(a_norm)
                        if obj_test < best_obj * 0.9999:
                            best_obj = obj_test
                            a = a_norm
                            improved = True
                            break
                        a[i] = old_i
                    if improved:
                        break
                if improved:

    return a


def multigrid_optimize(n_target, inits, time_limit, t_start, rng):
    """Multi-grid: optimize at coarser levels first, then refine."""
    # Start at coarser level
    n_coarse = max(16, n_target // 4)
    a_coarse = inits.get('flat_spec', inits.get('uniform', np.ones(n_coarse) / n_coarse))

    # If inits is for n_target, need to create coarse version
    if n_coarse != n_target:
        a_curr_full = inits.get('flat_spec', inits.get('uniform', np.ones(n_target) / n_target))
        # Downsample to coarse
        a_coarse = np.interp(np.linspace(0, 1, n_coarse),
                             np.linspace(0, 1, n_target), a_curr_full)
        a_coarse = np.maximum(a_coarse, 0)
        s = a_coarse.sum()
        if s > 1e-10:
            a_coarse /= s

    # Optimize at coarse level
    t_coarse_start = time.monotonic()
    coarse_budget = min(30.0, (time_limit - (t_coarse_start - t_start)) * 0.3)
    if coarse_budget > 5:
        a_coarse = anneal_optimize(a_coarse, [80, 40, 20], method='mirror',
                                   time_limit=coarse_budget, t_start=t_start)

    # Upsample to target and optimize
    a_full = np.interp(np.linspace(0, 1, n_target),
                       np.linspace(0, 1, len(a_coarse)), a_coarse)
    a_full = np.maximum(a_full, 0)
    s = a_full.sum()
    if s > 1e-10:
        a_full /= s

    # Also try the direct initialization
    a_direct = inits.get('flat_spec', inits.get('uniform', np.ones(n_target) / n_target))

    # Optimize both and keep the best
    remaining = time_limit - (time.monotonic() - t_start)
    if remaining > 10:
        half = remaining / 2 - 2
        a1 = anneal_optimize(a_full, [60, 30, 15, 8], method='mirror',
                             time_limit=half, t_start=t_start)
        a2 = anneal_optimize(a_direct, [60, 30, 15, 8], method='mirror',
                             time_limit=half, t_start=t_start)
        if obj_exact(a1) < obj_exact(a2):
            return a1
        return a2
    return a_full

                                  n_steps=50, strength=0.25)
            a_opt = anneal_optimize(a_shaved, [20, 10, 5], method='mirror',
                                    time_limit=min(40, rem - 5), t_start=t_start)
            score_and_update(a_opt)

        # 3c: Nesterov with very low beta
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            a_curr = np.maximum(best_sequence, 0)
            a_curr = a_curr / a_curr.sum()
            a_opt = anneal_optimize(a_curr, [10, 5, 3], method='nesterov',
                                    time_limit=min(50, rem - 5), t_start=t_start)
            score_and_update(a_opt)

        # 3d: Random perturbations + re-optimize
        rem = deadline - time.monotonic() - 20
        if rem > 60:
            n_pert = int(min(8, rem / 15))
            for _ in range(n_pert):
                rem = deadline - time.monotonic() - 20
                if rem < 15:
                    break
                scale = 0.003 + 0.004 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
                t_single = min(40.0, rem - 5)
                if t_single < 5:
                    break
                a_opt = anneal_optimize(a_pert, [20, 10], method='mirror',
                                        time_limit=t_single, t_start=t_start)
                score_and_update(a_opt)

        # 3e: Coordinate refinement
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_coord = coordinate_refine(best_sequence / best_sequence.sum(), n_iters=150)
            score_and_update(a_coord)
            if obj_exact(a_coord) < obj_exact(best_sequence):
                if rem > 10:
                    a_opt = anneal_optimize(a_coord, [10, 5], method='mirror',
                                            time_limit=min(20, rem - 5), t_start=t_start)
                    score_and_update(a_opt)

    # Phase 4: Try alternate lengths
    if best_sequence is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 60:
            alt_lengths = []
            bn = len(best_sequence)
            for mult in [0.75, 1.5, 2.0]:
                alt = int(bn * mult)
                if 64 <= alt <= 1024 and alt != bn:
                    alt_lengths.append(alt)
            alt_lengths = alt_lengths[:2]

            for alt_n in alt_lengths:
                rem = deadline - time.monotonic() - 15
                if rem < 40:
                    break
                inits_alt = generate_inits(alt_n, parent, rng)
                bs = np.maximum(best_sequence, 0)
                interp = np.interp(np.linspace(0, 1, alt_n),
                                   np.linspace(0, 1, len(bs)), bs)
                interp = np.maximum(interp, 0)
                s = interp.sum()
                if s > 1e-10:
                    inits_alt['from_best'] = interp / s

                a0 = inits_alt.get('from_best', inits_alt.get('flat_spec',
                                                              inits_alt['uniform']))
                t_alloc = min(80.0, rem - 10)
                a_opt = anneal_optimize(a0, [80, 40, 20, 10], method='mirror',
                                        time_limit=t_alloc, t_start=t_start)
                score_and_update(a_opt)

    # Final safety check
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print("\nFINAL SCORE: %.6f (n=%d, time=%.0fs)" % (
        best_score, len(best_sequence), time.monotonic() - t_start))
    return [float(x) for x in best_sequence]
