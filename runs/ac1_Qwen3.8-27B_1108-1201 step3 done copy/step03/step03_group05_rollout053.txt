def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:

"""
Minimizes 2n*max(conv(a,a))/(sum(a))² using exact subgradient descent
with top-k peak focus, coordinate descent polishing, and multi-length
exploration. Key improvements: exact (non-smoothed) subgradient for

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgradient(a, conv, top_k=10):
    """
    Compute the exact subgradient of max(conv(a,a)) using only the top-k peaks.
    Returns gradient vector g where g[i] = 2 * sum_{k in peaks} a[k-i] * 1[0<=k-i<n]
    """
    n = len(a)
    # Find top-k peaks
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    
    # For each peak k, the contribution to g[i] is 2*a[k-i] if 0 <= k-i < n
    g = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        g[i_min:i_max+1] += 2.0 * a[k - i_max : k - i_min + 1]
    
    return g

    using top-k peaks.
    """
    rng = np.random.default_rng(seed)

    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    # Lower bound: 1/(2n-1)
    lb = 1.0 / (2 * n - 1)
    
    t0 = time.monotonic()
    no_improve = 0
    iters = 0
    lr = 0.01  # initial learning rate
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        # Compute subgradient
        g = exact_subgradient(a, conv, top_k)
        g_norm = np.linalg.norm(g)
