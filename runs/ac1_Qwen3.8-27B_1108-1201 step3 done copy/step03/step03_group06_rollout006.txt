    W_full = np.fft.fft(w_padded)
    A_full = np.fft.fft(a_padded)
    
    # sum_k w[k] * a[k-i]: let j=k-i, = sum_j w[j+i]*a[j]
    # This is ifft(fft(a) * conj(fft(w)))[i modular]
    cross = np.fft.ifft(A_full * np.conj(W_full)).real[:n]
    
    # sum_k w[k] * a[i-k]: let j=i-k, = sum_j w[i-j]*a[j]  
    # This is ifft(fft(a) * fft(w))[i]
    conv_wa = np.fft.ifft(A_full * W_full).real[:n]
    
    return cross + conv_wa


def compute_subgradient_allsame(a, conv):
    """Compute subgradient when ALL peaks in a plateau are active."""
    n = len(a)
    M = conv.max()
    peak_mask = conv >= M - 1e-8 * M
    return compute_subgradient_fft(a, conv, peak_threshold=np.inf - (M - peak_mask.sum() * 0))


def lp_optimize(a0, time_limit, top_k=50, relax=1.0, step_init=0.8, seed=0, verbose=False):
    """LP-based optimization on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if top_k is not None and top_k < 2 * n - 1:
            b = solve_lp_toppk(a, M * relax, top_k, conv)
            if b is None:
                b = solve_lp_full(a, M * relax)
        else:
            b = solve_lp_full(a, M * relax)

        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        b = b / b_sum
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if step < step_init:
                step = min(step_init, step * 1.05)
        else:
            no_improve += 1
            if step > 0.05:
                step *= 0.88
            else:
                if no_improve > 8:
                    noise = rng.standard_normal(n) * 0.003
                    a = project_simplex(a + noise)
                    step = step_init * 0.6
                    no_improve = 0
                else:
                    step = max(0.01, step * 0.9)
                    continue

        if no_improve >= 60:
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            step = step_init * 0.7
            no_improve = 0

        if verbose and no_improve % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: obj={best_obj:.6f}, step={step:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def subgradient_optimize(a0, time_limit, lr_init=0.02, verbose=False, seed=0):
    """FFT-based subgradient descent on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_subgradient_fft(a, conv)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        a_new = project_simplex(a - lr * grad)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 50:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                lr = max(lr, 0.005)
                no_improve = 0

        if verbose and no_improve % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG: obj={best_obj:.6f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def peak_targeted_optimize(a0, time_limit, seed=0, verbose=False):
    """Directly target the highest peak by redistributing mass."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    eps = 0.001 / n

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        peak_k = np.argmax(conv)

        # Find the pair (i, j) with i+j=peak_k that maximizes a[i]*a[j]
        i_vals = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
        j_vals = peak_k - i_vals
        products = a[i_vals] * a[j_vals]
        top_idx = np.argmax(products)
        i_star = i_vals[top_idx]
        j_star = j_vals[top_idx]

        if i_star == j_star:
            # Single element dominates; try to reduce it
            # Transfer mass to a random position that doesn't contribute to peak
            candidates = np.where(conv < M * 0.8)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dest = rng.choice(candidates)
            transfer = min(a[i_star] * 0.01, eps * 10)
            if transfer > 0:
                a_new = a.copy()
                a_new[i_star] -= transfer
                a_new[dest] += transfer
                a_new = np.maximum(a_new, 0)
                s2 = a_new.sum()
                if s2 > 1e-15:
                    a_new /= s2
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
            continue

        # Transfer mass from both i_star and j_star
        transfer = min(a[i_star] * 0.005, a[j_star] * 0.005, eps * 10)
        if transfer <= 0:
            break

        # Find good destinations (positions that don't heavily contribute to peaks)
        contrib = np.zeros(n)
        for k in range(2 * n - 1):
            if conv[k] > M * 0.7:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        contrib[i] += conv[k] * a[j]
        
        low_contrib = np.where(contrib < np.median(contrib))[0]
        if len(low_contrib) == 0:
            low_contrib = np.arange(n)
        
        dest1 = rng.choice(low_contrib)
        dest2 = rng.choice(low_contrib)

        a_new = a.copy()
        a_new[i_star] -= transfer
        a_new[j_star] -= transfer
        a_new[dest1] += transfer
        a_new[dest2] += transfer
        a_new = np.maximum(a_new, 0)
        s2 = a_new.sum()
        if s2 > 1e-15:
            a_new /= s2
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges) - often good for low autocorrelation
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Four-modal
    q = max(1, n // 4)
    fm = np.zeros(n)
    fm[:q//2] = 1.0
    fm[-q//2:] = 1.0
    fm[q:2*q] = 0.5
    fm[3*q:4*q] = 0.5
    if fm.sum() > 0:
        fm /= fm.sum()
        inits['quadmoulal'] = fm

    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    # Alternating (checkerboard-like)
    alt = np.ones(n)
    alt[::2] = 1.0
    alt[1::2] = 0.3
    alt /= alt.sum()
    inits['alternating'] = alt

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def hybrid_optimize(a0, time_limit, seed=0, verbose=False):
    """Alternate between LP steps and subgradient steps."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0.7
    no_improve = 0
    mode = 0  # 0=LP, 1=SG

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        if mode == 0:
            # LP step
            top_k = max(30, n // 8)
            b = solve_lp_toppk(a, M, top_k, conv)
            if b is not None and b.sum() > 1e-15:
                b = b / b.sum()
                a_new = (1.0 - step) * a + step * b
                a_new = np.maximum(a_new, 0.0)
                s2 = a_new.sum()
                if s2 > 1e-15:
                    a_new = a_new / s2
            else:
                a_new = a.copy()
        else:
            # Subgradient step
            grad = compute_subgradient_fft(a, conv)
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
                a_new = project_simplex(a - 0.01 * grad)
            else:
                a_new = a.copy()

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                step = max(0.1, step * 0.9)
            if no_improve > 40:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                step = 0.7
                no_improve = 0

        mode = 1 - mode

        if verbose and no_improve % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    HYB: obj={best_obj:.6f}, mode={mode}, t={elapsed:.1f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization: LP + subgradient + peak-targeted.
    Explores multiple sequence lengths, deep-optimizes best candidates.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Explore lengths with quick LP ==========
    print("=== Phase 1: Length Exploration ===")
    t1 = time.monotonic()
    phase1_time = 150.0

    scan_results = []
    lengths = [320, 400, 480, 512, 640, 768, 896]

    for n in lengths:
        if time.monotonic() + 10 > t1 + phase1_time:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_ds', 'parent_up', 'bimodal', 'random', 'cosine']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() + 8 > t1 + phase1_time:
                break
            t_alloc = min(20.0, t1 + phase1_time - time.monotonic() - 5)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc,
                                         top_k=max(30, n // 10), step_init=0.8,
                                         seed=n + len(name))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f} (n={len(best_sequence) if best_sequence is not None else 0})")

    # ========== Phase 2: Deep LP at best length ==========
    print("=== Phase 2: Deep LP Optimization ===")
    t2 = time.monotonic()
    phase2_time = 400.0

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 320
        best_sequence = np.ones(best_n) / best_n

    # Collect starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:5]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)

    deep_starts.append((best_n, best_sequence.copy()))

    # Also add fresh starts at best_n
    inits_best = generate_initializations(best_n, parent, rng)
    for extra in ['bimodal', 'random', 'cosine']:
        if extra in inits_best:
            deep_starts.append((best_n, inits_best[extra].copy()))

    # Deduplicate and limit
    unique_starts = []
    seen2 = set()
    for n_s, a_s in deep_starts:
        key = (n_s, tuple(np.round(a_s, 4)))
        if key not in seen2:
            unique_starts.append((n_s, a_s))
            seen2.add(key)
        if len(unique_starts) >= 6:
            break

    num_runs = len(unique_starts) * 2  # 2 configurations each
    time_per_run = max(15.0, phase2_time / max(1, num_runs))

    for i, (n_s, a_s) in enumerate(unique_starts):
        for cfg_i, cfg in enumerate([
            {'top_k': max(40, n_s // 6), 'step_init': 0.9, 'relax': 1.0},
            {'top_k': None, 'step_init': 0.7, 'relax': 1.0},
        ]):
            if time.monotonic() + 15 > t2 + phase2_time:
                break
            t_alloc = min(time_per_run, t2 + phase2_time - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a_opt, obj_opt = lp_optimize(a_s, time_limit=t_alloc, seed=i * 7 + cfg_i * 13,
                                         verbose=False, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP [{i},{cfg_i}] n={n_s}: score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Hybrid LP+SG ==========
    print("=== Phase 3: Hybrid Optimization ===")
    t3 = time.monotonic()
    phase3_time = 300.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        # Run hybrid from best and from perturbed versions
        starts = [best_sequence.copy()]
        for i in range(3):
            noise = rng.standard_normal(n_best) * 0.003
            starts.append(project_simplex(best_sequence + noise))

        time_per_hybrid = phase3_time / len(starts)

        for i, a_s in enumerate(starts):
            if time.monotonic() + 15 > t3 + phase3_time:
                break
            t_alloc = min(time_per_hybrid, t3 + phase3_time - time.monotonic() - 10)
            if t_alloc < 15:
                break

            a_opt, obj_opt = hybrid_optimize(a_s, time_limit=t_alloc, seed=i * 31 + 7)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Hybrid [{i}]: score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Peak-targeted + Subgradient ==========
    print("=== Phase 4: Peak Targeting ===")
    t4 = time.monotonic()
    phase4_time = 250.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        time_per_pt = phase4_time / 3

        # Peak-targeted
        t_alloc = min(time_per_pt, phase4_time - 5)
        if t_alloc > 10:
            a_opt, obj_opt = peak_targeted_optimize(best_sequence, time_limit=t_alloc, seed=111)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  PeakTarget: score={score:.6f}")

        # Subgradient
        if time.monotonic() + 10 < t4 + phase4_time:
            t_alloc = min(time_per_pt, t4 + phase4_time - time.monotonic() - 5)
            if t_alloc > 10:
                a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                                      lr_init=0.015, seed=222)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Subgrad: score={score:.6f}")

        # Another LP pass
        if time.monotonic() + 10 < t4 + phase4_time:
            t_alloc = min(time_per_pt, t4 + phase4_time - time.monotonic() - 5)
            if t_alloc > 10:
                a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc,
                                             top_k=max(50, n_best // 5),
                                             step_init=0.5, seed=333)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP4: score={score:.6f}")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Fine-tuning ==========
    print("=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    phase5_time = max(0, deadline - time.monotonic() - 20.0)

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Very small LP steps
        if phase5_time > 30:
            t_alloc = min(phase5_time * 0.4, 80.0)
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc,
                                         top_k=max(60, n_best // 4),
                                         step_init=0.3, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.6f}")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            for i in range(min(20, int(remaining / 3))):
                if time.monotonic() > deadline - 15:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: score={best_score:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
