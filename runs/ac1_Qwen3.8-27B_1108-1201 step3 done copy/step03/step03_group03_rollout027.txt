def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not cond.any():

def solve_fast_lp(f, M, n, top_k=50):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for top-k binding constraints.
    Returns the LP solution g (normalized to sum=1) or None.
    """
    conv_f = fast_convolve(f)
    # Select top-k constraints
    top_indices = np.argpartition(conv_f, -top_k)[-top_k:]
    
    K = len(top_indices)
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(top_indices):
        j = np.arange(n)
        valid = (k - j >= 0) & (k - j < n)
        A_ub[idx, valid] = f[k - j[valid]]
    
    b_ub = np.full(K, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 5.0})
        if result.success:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

    best_a = a.copy()

    stall = 0

        if M < 1e-15:

        g = solve_fast_lp(a, M, n, top_k=top_k)

            if g is None:

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        iter_count += 1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            # Increase top_k as we make progress
            top_k = min(top_k_max, top_k + 2)
        else:
            stall += 1
        
        if stall >= 10:

            stall = 0

                a = project_simplex(a + noise)
                conv = fast_convolve(a)
                blend = max(0.1, blend_min * 5)
                stall = 0
    
    return best_a, best_obj

```python
def compute_gradient_fft(a, w):
    """Compute gradient of sum_k w[k]*(a*a)[k] using FFT."""
    n = len(a)
    N = 1 << (3*n).bit_length()  # Enough for the convolution
    
    W = np.fft.rfft(w, N)
    A_flip = a[::-1]
    AF = np.fft.rfft(A_flip, N)
    
    conv = np.fft.irfft(W * AF, N)
    
    grad = 2.0 * conv[n-1:n-1+n]
    return grad
```

This is O(N log N) per gradient evaluation, where N ~ 3n. For n=512, N=2048, so this is very fast.

def fista_smoothed(a0, time_limit, beta=100.0, lr=None):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr is None:
        # Estimate Lipschitz constant
        # L ~ beta * 2 * max |eigenvalue| of Q_k weighted by w
        # Rough bound: L ~ 4 * beta * n (very loose)
        # Use backtracking instead
        lr = 1.0 / (4.0 * beta * n)  # Conservative initial guess
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t_mom = 1.0
    y = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient at y
        conv_y = fast_convolve(y)
        scaled = beta * conv_y
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv_y))
            w[np.argmax(conv_y)] = 1.0
        else:
            w /= w_sum
        
        grad = compute_gradient_fft(y, w)
        
        # Backtracking line search for step size
        # f(y) - f(y - lr*grad) >= lr * ||grad||^2 (for convex f)
        # Simplified: just use a fixed step with decay
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
        x_new = project_simplex(y - lr * grad)
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        a = x_new
        t_mom = t_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Slowly decrease lr
        lr *= 0.9999
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LP with adaptive
constraint selection, FISTA on smoothed objective, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v -
