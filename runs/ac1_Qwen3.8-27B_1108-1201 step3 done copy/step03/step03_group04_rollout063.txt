"""
Improved LP-based Frank-Wolfe with reduced constraint sets, warm starts,
adaptive M, and multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: reduced LP size, better line search,

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a, active_indices=None):
    """Build sparse matrix for conv(a, b) at specified lag indices."""
    n = len(a)
    if active_indices is None:
        active_indices = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[k - j])
    n_rows = len(active_indices)
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n)), active_indices


def lp_oracle_reduced(a, M, active_indices, prev_b=None, time_limit=3.0):
    """Solve reduced LP: max sum(b) s.t. conv(a,b)[k] <= M for k in active_indices, b >= 0."""
    n = len(a)
    A_mat, _ = build_conv_matrix(a, active_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def compute_obj_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def line_search_brent(a, b, n_trials_first=5):
    """Hybrid line search: grid first, then Brent refinement."""
    n = len(a)
    best_t = 0.0
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    # Grid search
    ts = np.array([0.1, 0.2, 0.3, 0.5, 0.7, 0.9])
    objs = []
    for t in ts:
        candidate = (1 - t) * a + t * b
        obj = 2.0 * n * fast_convolve(candidate).max()
        objs.append(obj)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if best_t == 0.0:
        return 0.0, best_obj
    
    # Brent refinement around best_t
    idx = list(ts).index(best_t)
    lo = ts[max(0, idx - 1)]
    hi = ts[min(len(ts) - 1, idx + 1)]
    
    def func(t):
        candidate = (1 - t) * a + t * b
        return 2.0 * n * fast_convolve(candidate).max()
    
    try:
        res = minimize_scalar(func, bounds=(lo, hi), method='brent', options={'maxiter': 20})
        if res.success and res.fun < best_obj:
            best_t = res.x
            best_obj = res.fun
    except:
        pass
    
    return best_t, best_obj


def fpw_optimize(a0, time_limit, verbose=True, verbose_every=100):

    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)

    it = 0

        M = conv.max()
        active = np.where(conv > threshold * M)[0]
        if len(active) > K_max:
            # Keep the top K_max
            top_idx = np.argsort(conv[active])[-K_max:]
            active = active[top_idx]
        
        # Solve LP
        b = lp_oracle_reduced(a, M, active.tolist(), time_limit=3.0)
        if b is None:

                break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb  # Normalize to sum=1

            no_improve += 1
        else:
            t, obj_new = line_search_brent(a, b)
            if t < 0.01:
                no_improve += 1
            else:

                conv = fast_convolve(a)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        # Stall handling

            noise_scale = 0.005 + 0.003 * no_improve / max_stall
            noise = rng.standard_normal(n) * noise_scale

            a = a / a.sum()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
            # Also adjust threshold
            threshold = max(0.8, threshold - 0.05)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, |S|={len(active)}, elapsed={elapsed:.1f}s")

def subgradient_polish(a0, time_limit, lr_init=0.003, momentum=0.93, verbose=True, verbose_every=500):
    """Polish with multi-peak subgradient on simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Multi-peak subgradient with temperature
        conv = fast_convolve(a)
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.03, 1e-15)
        scaled = (peak_vals - peak_vals.max()) / temp
        w = np.exp(scaled)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += w[i] * a[k - idx]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.01 * it)
        it += 1

        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it}, obj={best_obj:.6f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    # Edge-concentrated
    edge = max(1, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    # Triangle
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.5] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()

    return inits

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix_a(a, active_indices):
    """Build sparse matrix A such that (A @ b)[row] = conv(a,b)[active_indices[row]]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(active_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[k - j])
    n_rows = len(active_indices)
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n))


def lp_oracle(a, M, active_indices, time_limit=4.0):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for k in active_indices, b >= 0."""
    n = len(a)
    A_mat = build_conv_matrix_a(a, active_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_local_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)

def mirror_descent_polish(a0, time_limit, eta_init=0.05, eta_decay=0.9995,

def subgradient_polish(a0, time_limit, lr_init=0.004, momentum=0.92,
                       verbose=False, verbose_every=300):
    """Subgradient method with heavy ball momentum on simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    a = a / a.sum()

    best_obj = compute_local_obj(a)
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        top_k = min(25, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.035, 1e-15)
        scaled = (peak_vals - peak_vals.max()) / temp
        w = np.exp(scaled)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += w[i] * a[k - idx]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        cur_obj = compute_local_obj(a)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Adaptive LR: decrease slowly
        lr = lr_init * (lr_init / (lr_init + 0.005 * it))
        it += 1

        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it}, obj={best_obj:.8f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    # Edge-concentrated
    edge = max(1, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    inits['edge'] = ec / ec.sum()

    # Triangle
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cosine'] = cos_w / cos_w.sum()

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.6] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse

    return inits

    Phases: quick length scan, deep LP-FW, mirror descent polish,
    subgradient polish, coordinate descent, LP final pass, perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(p)}, score={parent_score:.8f}")
            best_score = parent_score
            best_sequence = parent.copy()
    except Exception as e:
        print(f"Parent not available: {e}")

    # === Phase 1: Quick scan of nearby and other lengths ===
    print("=== Phase 1: Length scan ===")
    if parent is not None:
        n_parent = len(parent)
        scan_lengths = [n_parent - 16, n_parent - 8, n_parent + 8, n_parent + 16,
                        n_parent - 32, n_parent + 32]
        # Filter to reasonable range
        scan_lengths = [l for l in scan_lengths if 64 <= l <= 1024]
        # Also add some standard lengths
        scan_lengths += [256, 320, 448, 512]
        # Deduplicate, keeping parent length first
        seen = set()
        unique_lengths = []
        for l in sorted(scan_lengths, key=lambda x: abs(x - n_parent)):
            if l not in seen and l != n_parent:
                seen.add(l)
                unique_lengths.append(l)
        scan_lengths = unique_lengths[:6]
    else:
        scan_lengths = [256, 384, 512]

    for n in scan_lengths:
        if time.monotonic() > deadline - 900:
            break
        inits = generate_inits(n, parent, rng)
        # Try up to 3 inits per length
        init_priority = ['parent_interp', 'uniform', 'edge', 'triangle', 'cosine',
                         'gauss', 'random', 'sparse']
        tried = 0
        for name in init_priority:
            if tried >= 3:
                break
            if name not in inits:
                continue
            if time.monotonic() > deadline - 880:
                break
            t_alloc = min(20.0, deadline - time.monotonic() - 850)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, _ = fpw_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                tag = " *** NEW BEST ***" if score < best_score else ""
                print(f"  n={n}, {name}: {score:.8f}{tag}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
            tried += 1

    print(f"After Phase 1: best={best_score:.8f}, n={len(best_sequence) if best_sequence is not None else 0}")

    # === Phase 2: Deep LP-FW on best ===
    if best_sequence is not None:
        print("=== Phase 2: Deep LP-FW ===")
        t_deep = min(800.0, deadline - time.monotonic() - 500.0)
        if t_deep > 30:
            a_opt, _ = fpw_optimize(best_sequence, time_limit=t_deep,
                                    verbose=True, verbose_every=150)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2: {score:.8f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Mirror descent polish ===
    if best_sequence is not None:
        print("=== Phase 3: Mirror Descent Polish ===")
        remaining = deadline - time.monotonic() - 300.0
        if remaining > 20:
            t_md = min(250.0, remaining)
            a_opt, _ = mirror_descent_polish(best_sequence, time_limit=t_md,
                                             eta_init=0.03, eta_decay=0.9997,
                                             verbose=True, verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 (MD): {score:.8f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: Coordinate descent ===
    if best_sequence is not None:
        print("=== Phase 4: Coordinate Descent ===")
        a_cd, obj_cd = coordinate_descent_pass(best_sequence, n_passes=5)
        score = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(score):
            print(f"  Phase 4 (CD): {score:.8f}")
            if score < best_score:
                best_score = score
                best_sequence = a_cd.copy()

    # === Phase 5: Subgradient polish ===
    if best_sequence is not None:
        print("=== Phase 5: Subgradient Polish ===")
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 20:
            t_sg = min(200.0, remaining)
            a_opt, _ = subgradient_polish(best_sequence, time_limit=t_sg,
                                          lr_init=0.003, momentum=0.93,
                                          verbose=True, verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 5 (SubGrad): {score:.8f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 6: Final LP-FW pass ===
    if best_sequence is not None:
        print("=== Phase 6: Final LP-FW ===")
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 15:
            a_opt, _ = fpw_optimize(best_sequence, time_limit=remaining,
                                    verbose=True, verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 6 (Final LP): {score:.8f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 7: Perturbation restarts ===
    if best_sequence is not None:
        print("=== Phase 7: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 30.0
        n_restarts = max(1, min(10, int(remaining // 15)))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            scale = 0.0015 * (1.0 + 0.4 * i)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(15.0, deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break
            a_opt, _ = fpw_optimize(a_pert, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                tag = " (IMPROVED!)" if score < best_score else ""
                print(f"  Restart {i}: {score:.8f}{tag}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 8: Final coordinate descent on best ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        a_cd, _ = coordinate_descent_pass(best_sequence, n_passes=3)
        score = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_cd.copy()
            print(f"  Final CD: {score:.8f} (improved)")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
