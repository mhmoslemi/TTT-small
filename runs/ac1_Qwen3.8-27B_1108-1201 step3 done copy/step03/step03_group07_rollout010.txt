```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (cssv - 1.0) < (np.arange(1, n+1) * u)
    rho = np.max(cond.nonzero()[0]) + 1
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def pnorm_gradient(a, p):
    """Compute the p-norm of conv(a,a) and its gradient w.r.t. a."""
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-30:
        return 0.0, np.zeros_like(a)
    normalized = conv / max_c
    powered = normalized ** p
    sum_powered = powered.sum()
    F = max_c * sum_powered ** (1.0/p)
    w = powered / sum_powered
    # Gradient: F * 2 * (w ★ a)
    n = len(a)
    N = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0 * F
    return F, grad

def softmax_pnorm_obj(x, n, p):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-30:
        return 2.0 * n * 0.0, a
    normalized = conv / max_c
    powered = normalized ** p
    sum_powered = powered.sum()
    F = max_c * sum_powered ** (1.0/p)
    return 2.0 * n * F, a

def optimize_pnorm_lbfgsb(a0, p, maxiter=300):
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)
    
    def obj(x):
        val, _ = softmax_pnorm_obj(x, n, p)
        return val
    
    def jac(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        max_c = conv.max()
        if max_c < 1e-30:
            return np.zeros(n)
        normalized = conv / max_c
        powered = normalized ** p
        sum_powered = powered.sum()
        F = max_c * sum_powered ** (1.0/p)
        w = powered / sum_powered
        N = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad_F = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0 * F
        # d(score)/d(a[i]) = 2n/S^2 * grad_F[i] - 4nF/S^3
        S = a.sum()  # = 1
        grad_score = 2.0 * n / (S**2) * grad_F - 4.0 * n * F / (S**3)
        # softmax gradient
        grad_x = a * (grad_score - np.dot(a, grad_score))
        return grad_x
    
    result = minimize(obj, x0, jac=jac, method='L-BFGS-B', 
                      options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-12})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage high-power norm

import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (cssv - 1.0) < (np.arange(1, n + 1) * u)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(cond.nonzero()[0]) + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def compute_pnorm_and_weights(conv, p):
    """Compute p-norm and normalized weights stably."""
    max_c = conv.max()
    if max_c < 1e-30:
        return 0.0, np.zeros_like(conv)
    normalized = conv / max_c
    powered = normalized ** p
    sum_powered = powered.sum()
    if sum_powered < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return max_c, w
    F = max_c * sum_powered ** (1.0 / p)
    w = powered / sum_powered
    return F, w


def pnorm_grad_a(a, w, F):
    """Compute gradient of p-norm of conv(a,a) w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0 * F
    return grad


def softmax_pnorm_obj(x, n, p):
    """Objective for L-BFGS-B in softmax parameterization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    F, _ = compute_pnorm_and_weights(conv, p)
    return 2.0 * n * F


def softmax_pnorm_jac(x, n, p):
    """Jacobian for L-BFGS-B in softmax parameterization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    S = a.sum()
    conv = fast_convolve(a)
    F, w = compute_pnorm_and_weights(conv, p)
    if F < 1e-30:
        return np.zeros(n)
    grad_F = pnorm_grad_a(a, w, F)
    # d(2nF/S^2)/da[i] = 2n/S^2 * grad_F[i] - 4nF/S^3
    grad_score = 2.0 * n / (S * S) * grad_F - 4.0 * n * F / (S * S * S)
    # Softmax reparameterization gradient
    grad_x = a * (grad_score - np.dot(a, grad_score))
    return grad_x


def optimize_pnorm_lbfgsb(a0, p, maxiter=300, time_limit=60):
    """Optimize using L-BFGS-B with p-norm smooth objective."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)
    t0 = time.monotonic()

    def obj(x):
        if time.monotonic() - t0 > time_limit:
            raise RuntimeError("time limit")
        return softmax_pnorm_obj(x, n, p)

    def jac(x):
        if time.monotonic() - t0 > time_limit:
            raise RuntimeError("time limit")
        return softmax_pnorm_jac(x, n, p)

    try:
        result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-14, 'gtol': 1e-10})
    except (RuntimeError, Exception):
        x_opt = x0
    else:
        x_opt = result.x

    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def subgradient_step(a, p=0):
    """
    One subgradient step for the exact max objective.
    If p > 0, use top-p peaks for a smoother subgradient.
    Returns updated a and new objective.
    """
    n = len(a)
    S = a.sum()
    conv = fast_convolve(a)
    max_c = conv.max()

    if p > 0 and p < len(conv):
        # Use top-p peaks
        threshold = np.sort(conv)[::-1][min(p, len(conv) - 1)]
        mask = conv >= threshold * 0.99
        peaks = np.where(mask)[0]
    else:
        peaks = np.array([np.argmax(conv)])

    # Subgradient: sum over peaks of 2*(a[k-i] + a[i-k])
    g = np.zeros(n)
    for k in peaks:
        for i in range(n):
            contrib = 0.0
            if 0 <= k - i < n:
                contrib += a[k - i]
            if 0 <= i - k < n:
                contrib += a[i - k]
            g[i] += contrib
    g *= 2.0 / len(peaks)

    # Normalize gradient
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-30:
        return a.copy()

    # Step size: adaptive
    eta = 0.05 * S / g_norm
    # Backtracking
    obj_cur = 2.0 * n * max_c / (S * S)
    for trial in range(20):
        a_new = project_simplex(a - eta * g / S)
        S_new = a_new.sum()
        if S_new < 0.01:
            break
        conv_new = fast_convolve(a_new)
        max_new = conv_new.max()
        obj_new = 2.0 * n * max_new / (S_new * S_new)
        if obj_new < obj_cur:
            return a_new
        eta *= 0.5

    return a.copy()

def multi_peak_reduce(a, n_reduction_steps=50, ratio=0.95):

def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def resize_sequence(seq, target_n):
    """Resize a sequence to target_n via interpolation."""
    seq = np.maximum(np.asarray(seq, dtype=np.float64), 0)
    if len(seq) == target_n:
        return seq / seq.sum()
    x_orig = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, target_n)
    resampled = np.interp(x_new, x_orig, seq)
    resampled = np.maximum(resampled, 0)
    s = resampled.sum()
    if s > 1e-10:
        return resampled / s
    return np.ones(target_n) / target_n


def generate_inits(n, parent, rng):

    inits.append(('opt_phase', optimal_phase_init(n, rng, n_trials=20)))

    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(('gauss', g / g.sum()))

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(('bimodal', bi / bi.sum()))

    # Random Dirichlet
    inits.append(('random', rng.dirichlet(np.ones(n) * 0.3)))

    # Sparse random
    sparse = np.zeros(n)
    k = max(2, n // 4)
    idx = rng.choice(n, k, replace=False)
    sparse[idx] = 1.0
    inits.append(('sparse', sparse / sparse.sum()))

    # From parent
    if parent is not None and len(parent) > 0:
        pp = resize_sequence(parent, n)
        inits.append(('parent', pp))

    return inits

                    best_score = score
                    best_sequence = a_opt.copy()
                # Polish
                if remaining > 20:
                    a_pol = polish_exact_max(a_opt, n_iters=80, multi_peak=2)
                    score = score_and_update(a_pol)
                    if score < best_score:
                        best_score = score
                        best_sequence = a_pol.copy()
                remaining = deadline - time.monotonic() - 20.0

    # ============ Final ============
    if best_sequence is None:
        # Fallback
        best_sequence = np.ones(128) / 128
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final verification
    final_a = np.maximum(best_sequence, 0.0)
    final_s = final_a.sum()
    if final_s > 0:
        final_a = final_a / final_s
    final_score = float(evaluate_sequence([float(x) for x in final_a]))
    if np.isfinite(final_score) and final_score <= best_score:
        best_sequence = final_a
        best_score = final_score

    print("\nFINAL: score=%.8f, n=%d, time=%.0fs" % (
        best_score, len(best_sequence), time.monotonic() - t_start))

    return [float(x) for x in best_sequence]
```
