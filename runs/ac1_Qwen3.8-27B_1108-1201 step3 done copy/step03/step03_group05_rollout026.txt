"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using power-ramp initialization,
smoothed gradient descent, and peak-shaving refinements.
Key insight: monotone (power-law) ramps significantly outperform uniform
or Gaussian initializations because they reduce the center convolution
peak via the rearrangement inequality while keeping endpoint peaks manageable.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv(a,a))/beta w.r.t. a.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p

    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)

    return 2.0 * cross[:n]


def power_ramp(n, p, decreasing=False):
    """Generate a power-law ramp sequence on the simplex."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    if decreasing:
        idx = idx[::-1]
    a = idx ** p
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def sigmoid_ramp(n, steepness, decreasing=False):
    """Generate a sigmoid-shaped ramp on the simplex."""
    x = np.linspace(0, 1, n)
    if decreasing:
        x = 1 - x
    a = 1.0 / (1.0 + np.exp(-steepness * (x - 0.5)))
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def optimize_smoothed(a0, time_limit, beta_init=50.0, beta_end=500.0,
                      lr_init=0.008, momentum=0.93, perturb_every=300,
                      perturb_scale=0.002, seed=0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-15:

    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = smoothed_gradient(y, conv_y, beta)

        # Momentum update
        v = momentum * v - lr_init * grad / (1 + total_iters * 0.001)
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Periodic perturbation
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)

    return best_a, best_obj
