    """Euclidean projection onto the probability simplex."""
    n = len(v)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_constraint_rows(a, peaks):
    """Build rows of constraint matrix for specified peak indices."""
    n = len(a)
    rows = []
    for k in peaks:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        row = np.zeros(n)
        idx = np.arange(i_min, i_max + 1)
        row[idx] = a[k - idx]
        rows.append(row)
    return np.array(rows)

def obj_from_conv(n, conv):
    """Compute objective from convolution: 2*n*max(conv) (assumes sum(a)=1)."""
    return 2.0 * n * conv.max()

def solve_lp_direction(a, M, peaks):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in peaks, b >= 0."""
    n = len(a)
    A_ub = build_constraint_rows(a, peaks)
    b_ub = np.full(len(peaks), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def adaptive_step_size(a, b, M_target, max_iter=8):
    """Binary search for max step t such that max(conv((1-t)a+tb)) <= M_target."""
    lo, hi = 0.0, 1.0
    n = len(a)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        a_test = (1.0 - mid) * a + mid * b
        a_test = np.maximum(a_test, 0.0)
        s = a_test.sum()
        if s < 1e-15:
            hi = mid
            continue
        a_test = a_test / s
        conv_test = fast_convolve(a_test)
        if conv_test.max() <= M_target:
            lo = mid
        else:
            hi = mid
    return lo

def smooth_grad(a, beta=200.0):
    """Compute smooth gradient of max(conv(a,a)) via softmax."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    fft_size = 1 << (2 * n - 1).bit_length()
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    
    W = np.fft.fft(w_pad)
    A = np.fft.fft(a_pad)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return grad, conv

def resample_to_length(a, target_n):
    """Resample a sequence to a target length via linear interpolation."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(target_n) / target_n
    else:
        a_new = a_new / s
    return a_new

def diverse_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # From parent
    if parent is not None:
        inits['parent'] = resample_to_length(parent, n)
    
    # Structured
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)

    inits['gauss_center'] = g
    
    # Power law
    pw = (x + 1) ** (-0.5)
    pw /= pw.sum()
    inits['power'] = pw
    
    # Cosine
    cos_w = (1.0 + np.cos(2 * np.pi * x / n)) / 2.0
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Random
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand_unif'] = rng.dirichlet(np.ones(n) * 1.0)

            while time.monotonic() - t3 < t3 + t_hybrid and time.monotonic() < deadline - 60:
                # LP step
                conv_c = fast_convolve(a_curr)
                M_c = conv_c.max()
                top_k_c = max(20, n_best // 8)
                peaks_c = np.argpartition(conv_c, -top_k_c)[-top_k_c:]
                b_dir = solve_lp_direction(a_curr, M_c, peaks_c)
                if b_dir is not None and b_dir.sum() > 1e-15:
                    b_dir = b_dir / b_dir.sum()
                    a_curr = (1 - step) * a_curr + step * b_dir
                    a_curr = np.maximum(a_curr, 0)
                    s_c = a_curr.sum()
                    if s_c > 1e-15:
                        a_curr = a_curr / s_c
                
                # Gradient step
                grad_c, _ = smooth_grad(a_curr, 800.0)
                gn = np.linalg.norm(grad_c)
                if gn > 1e-15:
                    grad_c /= gn
                v = 0.9 * v - 0.015 * grad_c
                a_curr = project_simplex(a_curr + v)
                
                conv_c = fast_convolve(a_curr)
                obj_c = obj_from_conv(n_best, conv_c)
                if obj_c < best_hybrid_obj - 1e-11:
                    best_hybrid_obj = obj_c
                    best_hybrid = a_curr.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                    if no_imp > 50:
                        a_curr = project_simplex(a_curr + rng.standard_normal(n_best) * 0.003)
                        v = np.zeros(n_best)
                        no_imp = 0
            
            try_update(best_hybrid)
    
    print(f"Phase 3 done. Best: {best_score:.6f}")
    
    # ========== Phase 4: Final Polish (remaining time) ==========
    print("=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if best_sequence is not None and remaining > 30:
        n_best = len(best_sequence)
        
        # Very fine LP
        t_polish = min(remaining * 0.4, 120.0)
        if t_polish > 15:
            a_opt, obj_opt = lp_optimize_core(best_sequence, time_limit=t_polish,
                                              seed=9999, use_adaptive_step=True)
            try_update(a_opt)
        
        # Tiny perturbations
        remaining2 = deadline - time.monotonic() - 15
        if remaining2 > 20:
            n_pert = int(remaining2 / 3)
            n_pert = min(n_pert, 30)
            for i in range(n_pert):
                if time.monotonic() > deadline - 20:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_p = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_p) and score_p < best_score:
                        best_score = score_p
                        best_sequence = a_pert.copy()
                        print(f"  Perturb [{i}] s={scale}: score={score_p:.6f}")
                        break
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
