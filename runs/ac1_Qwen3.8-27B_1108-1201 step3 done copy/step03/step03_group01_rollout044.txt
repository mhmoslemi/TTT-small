    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear cross-convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2 and return (obj, conv, M)."""
    n = len(a)
    conv = fast_convolve(a)
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf'), conv, 0.0
    M = conv.max()
    obj = 2.0 * n * M / (sum_a ** 2)
    return obj, conv, M


def solve_lp(a, M, max_time=10.0):
    """
    Solve LP: maximize sum(g) s.t. a*g <= M (componentwise), g >= 0.
    Returns (g_normalized, g_sum) or (None, 0) on failure.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, 0.0


def line_search(a, d, current_obj, n, max_evals=12):
    """
    Adaptive line search: find best t in [0,1] minimizing obj(a + t*d).
    Uses the fact that along the path, conv has a quadratic structure.
    """
    best_t = 0.0
    best_val = current_obj

    # Quick check at t=1
    a1 = np.maximum(a + d, 0.0)
    s1 = a1.sum()
    if s1 > 1e-15:
        a1n = a1 / s1
        obj1, _, _ = compute_objective(a1n)
        if obj1 < best_val:
            best_val = obj1
            best_t = 1.0

    if best_t >= 0.99:
        return 1.0, best_val

    # Bisection refinement
    lo, hi = 0.0, 1.0
    for _ in range(8):
        if hi - lo < 1e-6:
            break
        mid = (lo + hi) / 2.0
        am = np.maximum(a + mid * d, 0.0)
        sm = am.sum()
        if sm < 1e-15:
            hi = mid
            continue
        am = am / sm
        objm, _, _ = compute_objective(am)
        if objm < best_val:
            best_val = objm
            best_t = mid
            lo = mid
        else:
            hi = mid

    # Refine around best
    if 0.01 < best_t < 0.99:
        step = (hi - lo)
        for _ in range(3):
            for dt in [-step * 0.5, step * 0.5]:
                t_try = best_t + dt
                if t_try <= 0.001 or t_try >= 0.999:
                    continue
                at = np.maximum(a + t_try * d, 0.0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at = at / st
                objt, _, _ = compute_objective(at)
                if objt < best_val:
                    best_val = objt
                    best_t = t_try
            step *= 0.5

    return best_t, best_val


def lp_optimize(a0, time_limit, verbose=True):
    """
    LP-based iterative optimization with adaptive M and line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    obj, conv, M = compute_objective(a)
    best_obj = obj
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturbation_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if verbose and iteration % 30 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, M={M:.8f}, elapsed={elapsed:.1f}s")

        # Solve LP with current M (slight margin for numerical safety)
        M_lp = M * 1.0001
        g, g_sum = solve_lp(a, M_lp, max_time=5.0)

        if g is None:
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)

            if norm_d < 1e-14:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, best_obj, n)

                if t_best > 1e-6 and obj_best < best_obj - 1e-14:
                    a_new = a + t_best * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    obj, conv, M = compute_objective(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        iteration += 1

        # Stagnation: perturb and restart
        threshold = 60 + 20 * perturbation_count
        if no_improve >= threshold:
            rng = np.random.default_rng(hash((iteration, perturbation_count)) % (2**31))
            scale = 0.003 + 0.004 * min(perturbation_count, 8)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            _, conv, M = compute_objective(a)
            no_improve = 0
            perturbation_count += 1

            if perturbation_count > 20:
                break

        # Also break if LP keeps failing
        if no_improve > 200:
            break

    return best_a, best_obj


def spectral_polish(a0, time_limit, verbose=True):
    """
    FFT-based subgradient polishing with adaptive peak weighting.
    Focuses on reducing the highest convolution peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    obj, conv, M = compute_objective(a)
    best_obj = obj
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    top_k = min(30, n // 4)
    temperature = 0.001

    batch_size = 20

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            # Find peaks
            if len(conv) > top_k:
                peaks = np.argpartition(conv, -top_k)[-top_k:]
            else:
                peaks = np.arange(len(conv))

            # Softmax weighting on peaks
            peak_vals = conv[peaks]
            scaled = (peak_vals - peak_vals.max()) / (temperature * max(peak_vals.max(), 1e-15) + 1e-15)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum

            # Compute subgradient
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]

            # Momentum update
            v = momentum * v - lr * sg
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
                v = np.zeros(n)

            obj, conv, M = compute_objective(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adapt hyperparameters
        lr *= 0.995
        momentum = min(0.96, momentum + 0.0005)
        temperature *= 1.01

        if no_improve >= 100:
            break

    return best_a, best_obj


def multiplicative_polish(a0, time_limit, verbose=True):
    """
    Multiplicative weights / mirror descent polishing.
    Uses exp(-lr * gradient) updates which naturally maintain positivity.
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s
    n = len(a)

    obj, conv, M = compute_objective(a)
    best_obj = obj
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    top_k = min(20, n // 5)

    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        if len(conv) > top_k:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        peak_vals = conv[peaks]
        # Equal weighting on top peaks
        w = np.ones(len(peaks)) / len(peaks)

        # Compute gradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]

        # Normalize gradient
        sg_mean = sg.mean()
        sg = sg - sg_mean  # Center so sum doesn't change much

        # Multiplicative update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        obj, conv, M = compute_objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.998
        if no_improve >= 80:
            break

    return best_a, best_obj


def generate_structured_init(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine-based (reduces auto-corr peaks)
    inits['sine'] = (np.sin(np.pi * x / n) + 0.05) / (np.sin(np.pi * x / n) + 0.05).sum()

    # Multi-frequency
    freq = 1 + rng.integers(3)
    inits['multi_sine'] = (np.sin(2 * np.pi * freq * x / n) ** 2 + 0.1) / \
                           (np.sin(2 * np.pi * freq * x / n) ** 2 + 0.1).sum()

    # Chirp (frequency sweep)
    inits['chirp'] = (np.sin(np.pi * x ** 2 / n) ** 2 + 0.05) / \
                      (np.sin(np.pi * x ** 2 / n) ** 2 + 0.05).sum()

    # From parent if available
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) >= n:
                # Downsample
                idx = np.linspace(0, len(p) - 1, n).astype(int)
                pp = p[idx]
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                # Upsample
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: LP optimization on parent (n=512) - primary effort
    Phase 2: LP optimization on n=1024 - explore larger sequences
    Phase 3: Spectral + multiplicative polishing
    Phase 4: Final perturb & re-optimize
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}")
            # Quick eval
            sc = float(evaluate_sequence(p.tolist()))
            if np.isfinite(sc):
                best_score = sc
                best_sequence = p.copy()
                print(f"Parent score: {sc:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    # Phase 1: LP optimization on n=512 (primary)
    print("\n=== Phase 1: LP n=512 ===")
    t_p1 = time.monotonic()
    n1 = 512
    phase1_time = max(100.0, (deadline - time.monotonic()) * 0.45)

    if parent is not None:
        inits = generate_structured_init(n1, rng, parent)
        # Try parent first, then a few others
        init_names = [nm for nm in ['parent', 'parent_ds', 'parent_up', 'sine', 'random', 'uniform'] if nm in inits]
    else:
        inits = generate_structured_init(n1, rng, None)
        init_names = list(inits.keys())[:4]

    t_per_init = phase1_time / max(1, len(init_names))

    for name in init_names:
        remaining = deadline - time.monotonic() - 150.0
        if remaining < 30:
            break
        t_alloc = min(t_per_init, remaining)
        if t_alloc < 15:
            break

        a0 = inits[name]
        print(f"  LP on n={n1}, init='{name}', time={t_alloc:.0f}s")
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  -> score={score:.6f} (init: {name})")

    print(f"Phase 1 done ({time.monotonic()-t_p1:.0f}s). Best so far: {best_score:.6f}")

    # Phase 2: Try n=1024
    print("\n=== Phase 2: LP n=1024 ===")
    t_p2 = time.monotonic()
    n2 = 1024
    remaining2 = deadline - time.monotonic() - 200.0
    if remaining2 > 120:
        phase2_time = min(remaining2 * 0.5, 400.0)

        if parent is not None:
            inits2 = generate_structured_init(n2, rng, parent)
            init_names2 = [nm for nm in ['parent_up', 'uniform', 'sine'] if nm in inits2]
        else:
            inits2 = generate_structured_init(n2, rng, None)
            init_names2 = ['uniform', 'sine'][:2]

        t_per_init2 = phase2_time / max(1, len(init_names2))

        for name in init_names2:
            remaining = deadline - time.monotonic() - 180.0
            if remaining < 60:
                break
            t_alloc = min(t_per_init2, remaining)
            if t_alloc < 30:
                break

            a0 = inits2[name]
            print(f"  LP on n={n2}, init='{name}', time={t_alloc:.0f}s")
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  -> score={score:.6f} (n={n2}, init: {name})")

    print(f"Phase 2 done ({time.monotonic()-t_p2:.0f}s). Best so far: {best_score:.6f}")

    # Phase 3: Spectral + multiplicative polishing
    print("\n=== Phase 3: Polishing ===")
    t_p3 = time.monotonic()
    if best_sequence is not None and np.isfinite(best_score):
        remaining3 = deadline - time.monotonic() - 100.0
        if remaining3 > 50:
            n_best = len(best_sequence)
            a_base = best_sequence.astype(np.float64).copy()
            s = a_base.sum()
            if s > 1e-15:
                a_base = a_base / s

            # Spectral polish
            t_sg = min(remaining3 * 0.4, 150.0)
            if t_sg > 20:
                print(f"  Spectral polish, n={n_best}, time={t_sg:.0f}s")
                a_opt, obj_opt = spectral_polish(a_base, time_limit=t_sg, verbose=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  -> spectral: {score:.6f}")

            # Multiplicative polish
            remaining = deadline - time.monotonic() - 80.0
            if remaining > 30:
                t_mp = min(remaining * 0.5, 100.0)
                if t_mp > 15:
                    a_base = best_sequence.astype(np.float64).copy()
                    s = a_base.sum()
                    if s > 1e-15:
                        a_base = a_base / s
                    print(f"  Multiplicative polish, n={n_best}, time={t_mp:.0f}s")
                    a_opt, obj_opt = multiplicative_polish(a_base, time_limit=t_mp, verbose=True)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  -> multiplicative: {score:.6f}")

            # LP re-optimize on polished result
            remaining = deadline - time.monotonic() - 50.0
            if remaining > 30 and best_sequence is not None:
                t_lp = min(remaining, 80.0)
                a_base = best_sequence.astype(np.float64).copy()
                s = a_base.sum()
                if s > 1e-15:
                    a_base = a_base / s
                print(f"  LP re-optimize, time={t_lp:.0f}s")
                a_opt, obj_opt = lp_optimize(a_base, time_limit=t_lp, verbose=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  -> LP re-opt: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_p3:.0f}s). Best so far: {best_score:.6f}")

    # Phase 4: Perturb & re-optimize
    print("\n=== Phase 4: Perturb & Reoptimize ===")
    t_p4 = time.monotonic()
    if best_sequence is not None:
        remaining4 = deadline - time.monotonic() - 10.0
        if remaining4 > 30:
            n_best = len(best_sequence)
            n_attempts = max(1, int(remaining4 / 25))
            t_per = remaining4 / n_attempts

            for pi in range(n_attempts):
                remaining = deadline - time.monotonic() - 5.0
                if remaining < 10:
                    break
                if t_per < 8:
                    break

                scale = 0.005 + 0.003 * pi
                noise = rng.standard_normal(n_best) * scale
                a_pert = best_sequence + noise
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a_pert = a_pert / s_pert

                t_alloc = min(t_per, remaining - 3.0)
                if t_alloc < 5:
                    break

                a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc, verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.6f}")
                else:
                    print(f"  Perturb {pi}: {score:.6f} (no improvement)")

    print(f"Phase 4 done ({time.monotonic()-t_p4:.0f}s). Best so far: {best_score:.6f}")

    # Fallback
    if best_sequence is None or not np.isfinite(best_score):
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
