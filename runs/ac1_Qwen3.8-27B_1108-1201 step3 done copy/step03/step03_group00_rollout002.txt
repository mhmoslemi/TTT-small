def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n):
    """Build sparse convolution matrix C where (C @ a)[k] = conv(a,a)[k] using a on one side.
    Actually we need the gradient structure. For the LP, we need B[k,j] = a[k-j] (the coefficient
    of g[j] in the k-th convolution output). But since a changes, we rebuild each time.
    For the LP max 1'g s.t. B@ g <= M, g >= 0, where B[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(1)  # placeholder, actual value = a[idx]
    return rows, cols


def solve_lp_constraint(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Here (f*g)[k] = sum_j f[k-j]*g[j] (cross-convolution).
    Returns optimal g or None."""
    c = -np.ones(n)
    # Build constraint matrix: A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # We build it row by row for efficiency
    A_rows = []
    A_cols = []
    A_vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_rows.append(k)
            A_cols.append(j)
            A_vals.append(f[k - j])
    
    A_ub = csr_matrix((A_vals, (A_rows, A_cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def _precompute_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    cols_list = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        cols_list.append((k, lo, hi))
    return cols_list


def _solve_lp_fast(f, M, n, cols_list):
    """Solve LP using precomputed structure."""
    c = -np.ones(n)
    A_rows = []
    A_cols = []
    A_vals = []
    for (k, lo, hi) in cols_list:
        for j in range(lo, hi + 1):
            A_rows.append(k)
            A_cols.append(j)
            A_vals.append(f[k - j])
    
    A_ub = csr_matrix((A_vals, (A_rows, A_cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def subgrad_optimize(f0, time_limit, lr_init=0.003, momentum=0.92, top_k=20, temp_init=0.00005):
    """Momentum subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Softmax weighting over peaks
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                # d g_k / d a[j] = a[k-j] + (1 if j != k-j else 0) * a[k-j]
                kj = k - j
                if j == kj:
                    sg[j] += w[i] * 2.0 * f[kj]
                else:
                    sg[j] += w[i] * 2.0 * f[kj]

        # Momentum update
        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        cur_lr *= 0.9997
        cur_temp *= 1.005

        # Periodic restart
        if no_improve > 300:
            rng = np.random.default_rng(it * 17 + 42)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.6, 0.0005)
            cur_temp = temp_init
            no_improve = 0

    return best_f, best_obj


def fine_polish(f0, time_limit):
    """Ultra-fine polishing with very small perturbations and local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:

    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Strategy: try small targeted perturbations around the peak
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]

        # Find which coordinates contribute most to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributions = np.zeros(n)
        for j in range(lo, hi + 1):
            kj = peak_idx - j
            contributions[j] = f[kj]

        # Try reducing high-contribution coordinates and adding to low-contribution ones
        top_contrib = np.argsort(contributions)[::-1][:n // 4]
        low_contrib = np.argsort(contributions)[:n // 4]

        if len(top_contrib) == 0 or len(low_contrib) == 0:
            break

        # Randomly pick one from each
        tc = top_contrib[rng.integers(len(top_contrib))]
        lc = low_contrib[rng.integers(len(low_contrib))]

        # Try moving mass from tc to lc
        for delta in [0.001, 0.003, 0.005, 0.01]:
            f_trial = f.copy()
            f_trial[tc] -= delta * f[tc]
            f_trial[lc] += delta * f[tc]
            if f_trial[tc] < 0:
                continue
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_trial.copy()
                f = f_trial.copy()
                break

        # Also try random micro-perturbations
        if rng.random() < 0.3:
            f_trial = f + rng.standard_normal(n) * 0.0005
            f_trial = project_simplex(f_trial)
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_trial.copy()
                f = f_trial.copy()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two-peak
    two = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits.append(two / two.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
