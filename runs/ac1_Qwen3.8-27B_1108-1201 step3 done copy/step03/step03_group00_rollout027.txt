import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_objective_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k))."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # Gradient: 2 * cross_correlation(a, w)
    # cross_correlation(a, w)[i] = sum_m a[m] * w[i+m]
    # = (a * w_flipped)[n-1-i] where w_flipped[j] = w[n-1-j]
    w_flipped = w[::-1]
    # Convolution of a and w_flipped
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_flipped, fft_size)
    conv_aw = np.fft.irfft(A * W, fft_size)[:2*n-1]
    # cross_correlation[i] = conv_aw[n-1-i]
    cross_corr = conv_aw[n-1::-1]  # This gives cross_corr[i] = conv_aw[n-1-i]
    gradient = 2.0 * cross_corr
    # Trim to length n
    gradient = gradient[:n]
    return gradient, conv

    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:

    return None

"""
Multi-phase optimizer for minimizing 2*n*max(autocorr)/(sum)^2 using:
1) FFT-based annealed gradient descent on a log-sum-exp smoothed objective
   (O(n log n) per iteration, enabling millions of steps)
2) Periodic LP-oracle steps for global direction information
3) Subgradient polishing on the exact (unsmoothed) peak objective
4) Multi-scale restarts and diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_and_conv(a, beta):
    """
    Compute gradient and autocorrelation of the smoothed objective
    L(a) = (1/beta) * log(sum_k exp(beta * conv_k[a])).
    Returns (gradient, conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # gradient[i] = 2 * sum_m a[m] * w[i+m]
    # = 2 * (a * w_flipped)[2n-2-i]
    w_flipped = w[::-1]
    fft_size = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_flipped, fft_size)
    conv_aw = np.fft.irfft(A * W, fft_size)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    return grad, conv


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def annealed_gd(a0, time_limit, beta_start=2.0, beta_end=300.0, lr_init=0.05,
                lp_every=400, verbose=False):
    """
    Annealed gradient descent on smoothed autocorrelation peak.
    Periodically injects LP-oracle steps for global direction.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = beta_start
    lr = lr_init
    no_improve = 0
    it = 0
    n_iterations = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = smoothed_grad_and_conv(a, beta)
        co = conv.max()
        n_iterations += 1

        if co < best_obj - 1e-13:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    AGD it={it}: obj={best_obj:.9f}, beta={beta:.1f}, "
                      f"lr={lr:.6f}, t={elapsed:.0f}s, it/s={n_iterations/elapsed:.0f}")
        else:
            no_improve += 1

        # Gradient step with backtracking
        step = a - lr * grad
        step = project_simplex(step)
        conv_step = fast_convolve(step)

        attempts = 0
        while conv_step.max() > co * (1 + 1e-10) + 1e-14 and attempts < 6:
            lr *= 0.5
            step = a - lr * grad
            step = project_simplex(step)
            conv_step = fast_convolve(step)
            attempts += 1

        a = step

        # Periodic LP step for global direction
        if lp_every > 0 and it % lp_every == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem > 3.0:
                g = solve_lp_oracle(a, best_obj * 1.0)
                if g is not None and g.sum() > 1e-10:
                    g = g / g.sum()
                    best_mix = a.copy()
                    best_mix_val = conv.max()
                    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                        mix = (1.0 - alpha) * a + alpha * g
                        mix = project_simplex(mix)
                        cv = fast_convolve(mix).max()
                        if cv < best_mix_val:
                            best_mix_val = cv
                            best_mix = mix
                    a = best_mix
                    conv = fast_convolve(a)
                    if conv.max() < best_obj - 1e-13:
                        best_obj = conv.max()
                        best_a = a.copy()
                        no_improve = 0

        # Anneal schedule
        if it % 300 == 0:
            progress = min(1.0, it / max(1, int(time_limit * 20)))
            beta = beta_start * (beta_end / beta_start) ** progress
            lr = lr_init * (0.7 ** (it // 300))
            lr = max(lr, 0.00005)

        # Restart if stuck
        if no_improve > 2000:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.003 * min(1.0, no_improve / 10000)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            beta = max(beta_start, beta * 0.4)
            lr = lr_init * 0.6
            no_improve = 0

    return best_a, best_obj

def subgradient_polish(a0, time_limit, lr_init=0.002, top_k=20, verbose=False):
    """
    Subgradient descent with softmax-weighted multi-peak targeting.
    Final polish on the exact (unsmoothed) peak objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    temp = 0.0001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Softmax-weighted subgradient over top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        step = a - lr * sg
        step = project_simplex(step)
        conv_new = fast_convolve(step)
        co = conv_new.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = step.copy()
            no_improve = 0
            a = step
            conv = conv_new
            if verbose and it % 5000 == 0:
                print(f"    SG it={it}: obj={best_obj:.9f}, lr={lr:.6f}, t={time.monotonic()-t0:.0f}s")
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr *= 0.95

        # Anneal temperature (focus more on the max peak)
        if it % 500 == 0:
            temp *= 0.85

        # Restart
        if no_improve > 2000:
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(a)
            lr = lr_init * 0.5
            temp = 0.0001
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    # Asymmetric (important for reducing peak)
    asym = np.exp(-x / (n * 0.4))
    inits.append(asym / asym.sum())

    # Bimodal
    bim = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits.append(bim / bim.sum())

    if parent is not None and len(parent) != n:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
