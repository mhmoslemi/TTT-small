        radii_guess = x_flat[:, 2]
        sum_r = np.sum(radii_guess)
        return -sum_r  # Minimize negative sum to maximize sum
    
    # Constraints: for each circle, x_i - r_i >= 0, etc.
    # Also, for each pair, distance >= sum of radii
    # Create constraints
    constraints = []
    # For each circle, add constraints
    for idx in range(n):
        # x_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, idx=idx: x[3*idx] - x[3*idx + 2]})
        # x_i + r_i <= 1
        constraints.append({'type': 'ineq', 'fun': lambda x, idx=idx: 1 - x[3*idx] - x[3*idx + 2]})
        # y_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, idx=idx: x[3*idx + 1] - x[3*idx + 2]})
        # y_i + r_i <= 1
        constraints.append({'type': 'ineq', 'fun': lambda x, idx=idx: 1 - x[3*idx + 1] - x[3*idx + 2]})
    
    # For each pair of circles, add distance constraint
    for i in range(n):
        for j in range(i+1, n):
            def constraint_func(x, i=i, j=j):
                # Get centers i and j
                x_i = x[3*i]
                y_i = x[3*i + 1]
                r_i = x[3*i + 2]
                x_j = x[3*j]
                y_j = x[3*j + 1]
                r_j = x[3*j + 2]
                # Distance between centers
                dist = np.sqrt( (x_i - x_j)**2 + (y_i - y_j)**2 )
                return dist - r_i - r_j
            constraints.append({'type': 'ineq', 'fun': constraint_func})
    
    # Initial guess: flatten the initial centers and radii
    x0 = np.concatenate([centers.reshape(-1, 2)[:, 0], centers.reshape(-1, 2)[:, 1], radii])
    
    # Optimization
    result = minimize(objective, x0, method='SLSQP', constraints=constraints)
    
    # Extract the result
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    # Reshape the result
    x_result = result.x.reshape(-1, 3)
    centers_result = x_result[:, :2]
    radii_result = x_result[:, 2]
    
    # Check if the result is valid
    # (But maybe the validate_packing function would catch errors)
    # However, the problem says to let it fail loudly if there's an error
    
    return (centers_result, radii_result, np.sum(radii_result))
