from scipy.optimize import minimize

def run_packing():

    centers = np.vstack([centers, [0.1, 0.1]])
    radii = np.append(radii, 0.05)
    
    # Flatten variables for optimization
    variables = np.concatenate([centers.reshape(-1), radii])
    
    # Objective function to maximize sum of radii
    def objective(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        return -np.sum(radii_flat)  # Minimize negative sum
    
    # Constraints: boundary and circle-circle distance
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(n):
        # x_i >= r_i
        def cons_x_min_r(x, i=i):
            return x[2*i] - x[2*n + i]
        constraints.append({'type': 'ineq', 'fun': cons_x_min_r})
        
        # x_i + r_i <= 1
        def cons_x_max_1(x, i=i):
            return 1 - (x[2*i] + x[2*n + i])
        constraints.append({'type': 'ineq', 'fun': cons_x_max_1})
        
        # y_i >= r_i
        def cons_y_min_r(x, i=i):
            return x[2*i + 1] - x[2*n + i]
        constraints.append({'type': 'ineq', 'fun': cons_y_min_r})

        constraints.append({'type': 'ineq', 'fun': cons_y_max_1})

    for i in range(n):

            def cons_dist(x, i=i, j=j):
