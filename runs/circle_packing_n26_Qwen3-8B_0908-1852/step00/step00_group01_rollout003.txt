def run_packing():

    n_circles = 26
    r_initial = 0.1
    positions = []
    for y in [0.1, 0.3, 0.5, 0.7, 0.9]:

    # Add the 26th circle at center
    positions.append([0.5, 0.5, r_initial])
    # Flatten the positions into variables
    variables = []
    for x, y, r in positions:
        variables.extend([x, y, r])
    # Objective function: maximize sum of radii
    def objective(vars):
        # vars is a flat list of x, y, r for each circle
        # calculate the sum of radii
        sum_r = 0.0
        for i in range(n_circles):
            sum_r += vars[3*i + 2]
        return -sum_r  # since we minimize
    # Constraints
    constraints = []
    # For each circle, constraints on x and y
    for i in range(n_circles):
        x = vars[3*i]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[3*i] - vars[3*i + 2]})
        # x + r <= 1
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: 1.0 - vars[3*i] - vars[3*i + 2]})
        # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[3*i + 1] - vars[3*i + 2]})
        # y + r <= 1
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: 1.0 - vars[3*i + 1] - vars[3*i + 2]})
    # Pairwise constraints between circles
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            # Define the constraint: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
            def constraint_func(vars, i=i, j=j):

                y_i = vars[3*i + 1]
