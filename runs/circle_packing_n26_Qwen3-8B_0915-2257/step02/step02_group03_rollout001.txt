    num_cols = int(np.ceil(n / num_rows))

    offset = 0.3  # Perturbation to move circles closer to edges

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col * (1 - 2 * offset) / (num_cols - 1) + offset
        y = row * (1 - 2 * offset) / (num_rows - 1) + offset
        centers[i] = [x, y]

    # Initial radii based on minimum distance to neighbors
    dist_matrix = cdist(centers, centers)
    np.fill_diagonal(dist_matrix, np.inf)
    min_dist = np.min(dist_matrix)
    initial_radius = min_dist / 2
    radii = np.full(n, initial_radius)

    # Objective function: maximize total radius
    def objective(x):
        # x is a vector of [x0, y0, r0, x1, y1, r1, ...]
        x_flat = x.reshape(-1, 2)
        r_flat = x[n*2 :]

        # Total radius
        total_radius = np.sum(r_flat)

        # Constraint: all circles are within the square
        for i in range(n):
            xi, yi = x_flat[i]
            ri = r_flat[i]
            if xi - ri < 0 or xi + ri > 1 or yi - ri < 0 or yi + ri > 1:
                return -np.inf  # Penalize

        for i in range(n):

                distance = np.sqrt((xi - xj)**2 + (yi - yj)**2)
