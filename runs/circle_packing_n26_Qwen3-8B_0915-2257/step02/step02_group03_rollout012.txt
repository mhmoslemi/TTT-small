import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    # Initial guess: a hexagonal packing with 5 rows
    def hexagonal_centers(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                if i % 2 == 0:
                    x = j + 0.5
                else:
                    x = j + 0.5 - 0.5
                y = i + 0.5
                x = x / (cols - 1) if cols > 1 else 0.5
                y = y / (rows - 1) if rows > 1 else 0.5
                centers.append([x, y])
        return np.array(centers)

    centers = hexagonal_centers(26)

    def objective(radii):

    def constraints(radii):
        # Constraint: All circles stay inside the unit square
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
