def generate_base_hexagonal_layout(n):
    """Generate a base hexagonal layout for n circles."""
    if n == 0:
        return np.array([]), np.array([])
    layer = int(np.sqrt(n))
    if layer * (layer + 1) < n:
        layer += 1
    positions = []
    for i in range(layer):
        for j in range(i + 1):
            positions.append((i, j))
        for j in range(i, layer):
            positions.append((j, i))
    positions = np.array(positions)
    return positions, np.full(len(positions), 1.0)
