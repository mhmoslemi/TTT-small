import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Non-overlap constraints: distance between centers >= sum of radii
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-6)

    # Boundary constraints: circles must be inside the unit square
    for i in range(n):
        # Left boundary
        constraints.append(centers[i, 0] - radii[i] >= -1e-6)
        # Right boundary
        constraints.append(centers[i, 0] + radii[i] <= 1 + 1e-6)
        # Bottom boundary
        constraints.append(centers[i, 1] - radii[i] >= -1e-6)
        # Top boundary
        constraints.append(centers[i, 1] + radii[i] <= 1 + 1e-6)

    # Non-negativity of radii
    constraints.append(radii >= 0)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, max_iter=1000)

    # Extract results
    centers_np = centers.value
    radii_np = radii.value

    # Compute the sum of radii
    sum_radii = np.sum(radii_np)
