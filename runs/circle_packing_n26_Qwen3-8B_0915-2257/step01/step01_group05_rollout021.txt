from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    num_circles = 26
    grid = np.array([
        [i / 5, j / 5] for i in range(6) for j in range(5)
    ])  # 6x5 grid = 30 points, but we'll select 26
    centers = grid[:26]
    radius = 0.1  # Initial guess for radii

    def objective(r):
        return -np.sum(r)  # We minimize -sum(r) to maximize sum(r)

    # Constraints: distance between circles must be >= sum of radii

                dy = centers[i, 1] - centers[j, 1]
                dist[i, j] = np.sqrt(dx**2 + dy**2)
        return dist - (r[:, None] + r[None, :]) + 1e-8  # Ensure non-overlapping

    # Setup optimization
    bounds = Bounds([0] * num_circles, [np.inf] * num_circles)
    x0 = np.full(num_circles, radius)

    # Run optimization
    result = minimize(
