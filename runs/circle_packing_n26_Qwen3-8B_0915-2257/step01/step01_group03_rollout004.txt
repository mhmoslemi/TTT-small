import scipy.optimize as optimize

def run_packing():

    n = 26

    rows = 6
    cols = 5
    
    # Starting positions in the grid
    centers = []

            # Hexagonal grid with horizontal and vertical spacing
            x = j + i / 2
            y = i
            # Normalize to unit square (0,1)x(0,1)
            x = x / (cols + (rows - 1) / 2)
            y = y / rows
            centers.append((x, y))
    
    centers = np.array(centers)

    # While ensuring no overlaps and circles stay within the square

    result = optimize.minimize(
