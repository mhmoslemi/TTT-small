import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial hexagonal packing for 26 circles
    def hexagonal_grid(n):
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i + 0.5 * j
                if x < 1 and y < 1:
                    centers.append([x, y])
        return np.array(centers)

    # Calculate initial radii based on spacing
    def get_initial_radii(centers):
        spacing = np.min(cdist(centers, centers))
        return np.full(centers.shape[0], spacing / np.sqrt(3))

    # Objective function: maximize sum of radii
    def objective(x, centers, n_circles):
        return -np.sum(x)  # Minimize negative of sum to maximize

    # Constraint: non-overlapping and boundary
    def constraint(x, centers, n_circles):
        r = x
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < r[i] + r[j] - 1e-8:
                    return -1
        for i in range(n_circles):
            x_i, y_i = centers[i]
            r_i = r[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        return 0

    # Initial centers and radii
    centers = hexagonal_grid(26)
    initial_radii = get_initial_radii(centers)

    # Flatten radii for optimization
    r_flat = initial_radii.flatten()

    # Set up optimization problem
    result = minimize(
        objective,
        r_flat,
        args=(centers, len(centers)),
        bounds=[(0, 1)] * len(r_flat),
        constraints=[
            {
                'type': 'ineq',
                'fun': constraint,
                'args': (centers, len(centers)),
            }
        ],
        method='SLSQP'
    )

    # Extract final result
    final_radii = result.x
    final_radii = final_radii.reshape(-1, 1)
    sum_radii = np.sum(final_radii)

    return (centers, final_radii, sum_radii)
