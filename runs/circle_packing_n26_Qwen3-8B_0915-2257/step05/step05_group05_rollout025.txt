import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, cdist
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def constraints(radii, centers):
        constraints = []
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - (radii[i] + radii[j]))
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r)
            constraints.append(1 - x - r)
            constraints.append(y - r)
            constraints.append(1 - y - r)
        return constraints

    def generate_hybrid_initial_guess():
        # Create a hybrid pattern: 2 rows of 7 circles on the top and bottom
        # with 3 rows of 6 circles in the center, and adjust for spacing
        spacing = 0.35
        row_height = spacing
        col_width = spacing

        centers = []
        radii = []

        # Top row
        for col in range(7):
            x = col * col_width + col_width / 2
            y = 1 - spacing / 2
            centers.append([x, y])
            radii.append(0.15)

        # Bottom row
        for col in range(7):
            x = col * col_width + col_width / 2
            y = spacing / 2
            centers.append([x, y])
            radii.append(0.15)

        # Middle rows
        for row in range(3):
            for col in range(6):
                x = col * col_width + col_width / 2
                y = row * row_height + spacing
                centers.append([x, y])
                radii.append(0.15)

        # Add corner circles for maximum possible radius
        for i, (x, y) in enumerate([(0.1, 0.1), (0.9, 0.1), (0.1, 0.9), (0.9, 0.9)]):
            centers.append([x, y])
            radii.append(0.05)

        # Fill in gaps with smaller circles
        gap_positions = [
            [0.5, 0.15],
            [0.15, 0.5],
            [0.85, 0.5],
            [0.5, 0.85],
            [0.3, 0.3],
            [0.7, 0.3],
            [0.3, 0.7],
            [0.7, 0.7],
        ]
        for x, y in gap_positions:
            centers.append([x, y])
            radii.append(0.03)

        # Ensure we have exactly 26 circles
        while len(centers) < 26:
            if len(centers) == 24:
                centers.append([0.1, 0.1])
                radii.append(0.05)
            elif len(centers) == 25:
                centers.append([0.9, 0.9])
                radii.append(0.05)
            else:
                centers.append([0.5, 0.5])
                radii.append(0.03)
        return np.array(centers), np.array(radii)

    def optimize_packing(centers):
        n = centers.shape[0]
        num_vars = 2 * n + n  # 2 coordinates and 1 radius per circle

        bounds = [(0, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)]

        def objective_func(x):
            centers_flat = x[:2 * n].reshape(n, 2)
            radii = x[2 * n:]
            return -np.sum(radii)

        def constraint_func(x):
            centers_flat = x[:2 * n].reshape(n, 2)
            radii = x[2 * n:]
            constraints = []
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j]) ** 2))
                    constraints.append(dist - (radii[i] + radii[j]))
            for i in range(n):
                x_i, y_i = centers_flat[i]
                r = radii[i]
                constraints.append(x_i - r)
                constraints.append(1 - x_i - r)
                constraints.append(y_i - r)
                constraints.append(1 - y_i - r)
            return constraints

        result = minimize(
            objective_func,
            x0=np.concatenate([centers.reshape(-1), np.ones(n)]),
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraint_func},
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 500, 'disp': False}
        )

        if not result.success:
            raise RuntimeError("Optimization failed")

        optimized_centers = result.x[:2 * n].reshape(n, 2)
        optimized_radii = result.x[2 * n:]
        return optimized_centers, optimized_radii

    # Generate a more optimized initial guess with hybrid packing
    initial_centers, initial_radii = generate_hybrid_initial_guess()

    # Optimize
    optimized_centers, optimized_radii = optimize_packing(initial_centers)

    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    valid, msg = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        print(f"Validation failed: {msg}")
        raise RuntimeError("Packing failed validation")

    total_sum = np.sum(optimized_radii)
    return optimized_centers, optimized_radii, total_sum
