from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    num_cols = int(np.ceil(n / num_rows))

    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // num_cols
        col = i % num_cols

        y = row / (num_rows - 1)

        if col == 0 or col == num_cols - 1:
            x = np.clip(x, 0.1, 0.9)

        centers[i] = [x, y]

    dist_matrix = cdist(centers, centers)

    min_dist = np.min(dist_matrix)
    initial_radius = min_dist / 2
    radii = np.full(n, initial_radius)

    # Objective function: maximize total radius
    def objective(x):
        x_flat = x.reshape(-1, 2)
        r_flat = x[n*2 :]

        total_radius = np.sum(r_flat)

                return -np.inf

        return -total_radius  # Minimize negative total for maximization

    # Bounds: x and y in [0, 1], r >= 0
    bounds = []

    # Initial guess: x_flat with centers and r_flat with initial radii
