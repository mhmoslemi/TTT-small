from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    n = 26
    num_rows = int(np.ceil(np.sqrt(n)))
    num_cols = int(np.ceil(n / num_rows))

    # Create a hexagonal lattice centers
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col / (num_cols - 1)
        y = row / (num_rows - 1)
        centers[i] = [x, y]

    # Initial radii based on nearest neighbor distance
    dist_matrix = cdist(centers, centers)
    np.fill_diagonal(dist_matrix, np.inf)
    min_dist = np.min(dist_matrix)
    initial_radius = min_dist / 2
    radii = np.full(n, initial_radius)

    # Define the objective (maximize total radius)
    def objective(x):
        # x is a vector of [x0, y0, r0, x1, y1, r1, ...]
        x_flat = x.reshape(-1, 2)
        r_flat = x[n*2 :]

        # Compute total radius
        total_radius = np.sum(r_flat)

        # Check if circles are within the square
        for i in range(n):
            xi, yi = x_flat[i]
            ri = r_flat[i]
            if xi - ri < 0 or xi + ri > 1 or yi - ri < 0 or yi + ri > 1:
                return -np.inf  # Penalize

        # Check if circles are non-overlapping
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = x_flat[i]
                xj, yj = x_flat[j]
                ri = r_flat[i]
                rj = r_flat[j]
                distance = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                if distance < ri + rj - 1e-8:
                    return -np.inf  # Penalize

        return -total_radius  # Minimize negative of total radius to maximize

    # Bounds: x and y in [0, 1], r >= 0
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1)])

    # Initial guess: x_flat with centers and r_flat with initial radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Optimization: Use COBYLA for constrained optimization
    result = minimize(
        objective,
        x0,
        method="COBYLA",
        bounds=bounds,
        options={"maxiter": 1000, "disp": False}
    )

    # Extract final centers and radii
    centers = result.x[:n*2].reshape(n, 2)
    radii = result.x[n*2:]

    # Return centers, radii, and sum_radii
    return centers, radii, np.sum(radii)
