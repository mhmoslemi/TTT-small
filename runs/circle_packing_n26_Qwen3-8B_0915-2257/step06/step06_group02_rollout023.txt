from scipy.optimize import minimize
from scipy.spatial.distance import pdist, cdist
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

        centers_flat = x[:2 * 26].reshape(26, 2)
        radii = x[2 * 26:]
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_func(x):
        centers_flat = x[:2 * 26].reshape(26, 2)
        radii = x[2 * 26:]
        constraints = []
        # Boundary constraints
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r = radii[i]
            constraints.append(x_i - r)  # Left boundary
            constraints.append(1 - x_i - r)  # Right boundary
            constraints.append(y_i - r)  # Bottom boundary
            constraints.append(1 - y_i - r)  # Top boundary

        # Non-overlapping constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j]) ** 2))
                constraints.append(dist - (radii[i] + radii[j]))
        return constraints

        grid_radii = []

        return np.array(grid_points), np.array(grid_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
