import cvxpy as cp

    num_circles = n

    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable(num_circles)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: each circle must be within the unit square
    constraints = [
        (centers[i, 0] - radii[i] >= 0 for i in range(num_circles)),
        (1 - centers[i, 0] - radii[i] >= 0 for i in range(num_circles)),

    # Create optimization problem with CVXPY
    problem = cp.Problem(objective, constraints)

    # Solve the optimization problem
    problem.solve(solver=cp.OSQP, verbose=False)
