    def objective(x):
        return -np.sum(x)  # we minimize -sum to maximize sum
        
    def constraint_eq(x):

                dx = x[2*i] - x[2*j]
                dy = x[2*i+1] - x[2*j+1]

        return 0

    # Optimization loop
