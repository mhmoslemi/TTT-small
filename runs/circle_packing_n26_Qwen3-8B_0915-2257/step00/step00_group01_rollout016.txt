import scipy.optimize as optimize

def run_packing():

    cols = 6

    # Objective function: maximize the sum of radii
    def obj(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints:
    # 1. Circle centers must lie within [0,1]^2
    # 2. Each circle must not extend beyond the square
    # 3. The distance between any two circles must be at least the sum of their radii

    def constraint_center(x):
        return x  # x is [centers, radii], we extract centers

    def constraint_radius(x):

    def constraint_boundary(x):
        centers = x[:len(centers)]
        radii = x[len(centers):]
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return 1  # Violates boundary
        return 0

    def constraint_overlap(x):
        centers = x[:len(centers)]
        radii = x[len(centers):]

                dx = centers[i][0] - centers[j][0]

                dist = np.sqrt(dx**2 + dy**2)

                    return 1  # Overlapping
        return 0

    # Optimization bounds:
    # Centers must be within [0,1]^2

    bounds = []
    for i in range(len(centers)):
