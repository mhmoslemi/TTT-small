from scipy.optimize import minimize

def run_packing():
    def hexagonal_grid(n, row_start, column_start, spacing):
        x = []
        y = []
        radii = []
        for i in range(n):
            row = i // 2
            col = i % 2
            x_coord = column_start + (col - 0.5) * spacing
            y_coord = row_start + row * (np.sqrt(3) / 2) * spacing
            x.append(x_coord)
            y.append(y_coord)
            radii.append(1.0)
        return np.array(x), np.array(y), np.array(radii)

    def objective(p):
        x = p[:26]
        y = p[26:52]
        r = p[52:]
        return -np.sum(r)

    def constraints(p):
        x = p[:26]
        y = p[26:52]
        r = p[52:]
        constraints = []
        for i in range(26):
            x_i = x[i]
            y_i = y[i]
            r_i = r[i]
            constraints.append((x_i - r_i >= -1e-12) & (x_i + r_i <= 1 + 1e-12) &
                               (y_i - r_i >= -1e-12) & (y_i + r_i <= 1 + 1e-12))
        
        for i in range(26):
            for j in range(i + 1, 26):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist >= r[i] + r[j] - 1e-12)
        
        return constraints

    # Initial guess from hexagonal grid

    bounds = [(0, 1) for _ in range(52)] + [(0, 1) for _ in range(26)]
    result = minimize(
