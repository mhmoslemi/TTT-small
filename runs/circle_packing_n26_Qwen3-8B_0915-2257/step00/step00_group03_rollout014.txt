import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
import matplotlib.pyplot as plt
from itertools import combinations

def generate_initial_centers(n):
    """Generate initial centers for circle packing using Voronoi diagram."""
    points = np.random.rand(n, 2)
    vor = Voronoi(points)
    regions = vor.regions
    valid_regions = []
    for region in regions:
        if not -1 in region and len(region) > 0:
            valid_regions.append(region)
    
    # Use the first valid region as a basis for extracting meaningful points
    if not valid_regions:
        return points
    
    region = valid_regions[0]
    points_in_region = points[region]
    return points_in_region

def run_packing():
    """
    Run a circle packing algorithm to maximize the total sum of radii for 26 circles
    in a unit square.
    """
    n = 26
    initial_centers = generate_initial_centers(n)
    initial_radii = np.ones(n) * 0.05
    
    # Optimization parameters
    bounds = [(0, 1)] * 2 * n
    radii_bounds = [(0, 1)] * n
    constraints = []

    # Define the objective function to maximize total sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Define constraints: non-overlapping circles and within boundary
    def constraint_func(radii):
        centers = np.array([np.random.rand(n, 2)])  # Use initial centers as placeholder
        if np.isnan(centers).any() or np.isnan(radii).any():
            return np.inf

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return np.inf

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf

        return 0

    # Wrap the constraint function
    def constraint(x):
        return constraint_func(x)

    # Run optimization with constraints
    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=radii_bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}]
    )

    if not result.success:
        raise ValueError("Optimization failed")

    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
