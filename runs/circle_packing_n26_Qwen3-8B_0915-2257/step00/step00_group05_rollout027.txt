import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def run_packing():
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            if np.isnan(centers[i, 0]) or np.isnan(centers[i, 1]) or np.isnan(radii[i]):
                return False, "NaN values present"
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={radii[i]} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    def perturb_centers(centers, sigma=0.05):
        return centers + np.random.normal(0, sigma, centers.shape)

    def check_bounds(centers, radii):
        for i in range(centers.shape[0]):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        return True

    def simulate_packing(r):
        grid = np.linspace(0 + r, 1 - r, 5)
        centers = []
        for i, x in enumerate(grid):
            for j, y in enumerate(grid):
                if i % 2 == 0:
                    centers.append([x, y])
                else:
                    centers.append([x, y + 0.5 * r])
        centers = np.array(centers)
        if len(centers) != 26:
            return None
        return centers

    def optimize_packing(r):
        centers = simulate_packing(r)
        if centers is None:
            return None
        bounds = Bounds(np.zeros(26 * 2), np.ones(26 * 2))
        constraints = []
        for i in range(26):
            for j in range(i + 1, 26):
                def constraint_func(x):
                    c = np.reshape(x, (26, 2))
                    dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                    return dist - r - r
                constraints.append(NonlinearConstraint(constraint_func, -np.inf, 0))
        constraints.append(NonlinearConstraint(lambda x: np.sum(np.sqrt(np.sum((x[i * 2 : (i + 1) * 2] - x[j * 2 : (j + 1) * 2]) ** 2)) for i in range(26)), 0, 26 * r))
        result = opt.minimize(lambda x: -np.sum(np.sqrt(np.sum((x[i * 2 : (i + 1) * 2] - x[j * 2 : (j + 1) * 2]) ** 2)) for i in range(26)), x0=centers.flatten(), bounds=bounds, constraints=constraints)
        if not result.success:
            return None
        centers_opt = np.reshape(result.x, (26, 2))
        radii = np.full(26, r)
        if not check_bounds(centers_opt, radii):
            return None
        return centers_opt, radii

    best_sum = 0.0
    best_centers = None
    best_radii = None

            continue

                    best_sum = np.sum(radii)
                    best_centers = centers_opt
                    best_radii = radii
        except:
            pass

    return best_centers, best_radii, best_sum
