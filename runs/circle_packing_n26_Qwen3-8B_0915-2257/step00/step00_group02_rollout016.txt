import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
import matplotlib.pyplot as plt

def run_packing():
    # Constants
    n = 26
    tolerance = 1e-6

    # Initial configuration: hexagonal arrangement of 24 circles
    rows = 6
    cols = 4
    dx = 1.0 / (cols - 1)
    dy = 1.0 / (rows - 1)

    centers = []
    for row in range(rows):
        for col in range(cols):
            x = col * dx
            y = row * dy
            centers.append([x, y])
    centers = np.array(centers)

    # Add two more circles to fill in potentially optimal places in the square
    extra_centers = np.array([[0.2, 0.8], [0.8, 0.2]])  # Pre-defined extra positions
    centers = np.vstack((centers, extra_centers))

    # Initial guess for radii: same for all, just a starting point
    radii = np.ones(n) * 0.05

    def fitness(radii):
        return -np.sum(radii)  # Minimize negative sum -> maximize sum

    def constraints(radii):
        # Constraints:
        # - Each circle is in [0,1]x[0,1]
        # - No overlap between circles

        # Constraint 1: Ensure circles are fully within square
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            if (x - r < -tolerance or x + r > 1 + tolerance or
                y - r < -tolerance or y + r > 1 + tolerance):
                return False

        # Constraint 2: Ensure no overlap between circles
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - tolerance:
                    return False

        return True

    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Optimization: use local search with constraints
    result = minimize(
        fitness,
        radii,
        bounds=bounds,
        constraints={"type": "successive", "fun": constraints, "args": ()},
        method="L-BFGS-B"
    )

    if not result.success:
        raise RuntimeError("Optimization failed to converge")

    best_radii = result.x
    best_centers = centers

    return best_centers, best_radii, np.sum(best_radii)
