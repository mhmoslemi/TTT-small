import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based configuration with 26 circles arranged in 5 rows
    num_circles = 26

    # Base positions for the circles in a grid layout
    # We'll place at least 5 circles in each grid row
    positions = []
    for i in range(5):
        for j in range(5 + (i % 2)):
            x = j / 5.0
            y = i / 5.0
            positions.append([x, y])
    
    # Convert to a numpy array
    centers = np.array(positions)
    radii = np.array([0.01] * num_circles)  # Initial guess with small radii

    def objective_function(params):
        # params is a 1D array of (x1, y1, r1, x2, y2, r2, ..., x26, y26, r26)
        centers_opt = params[:2 * num_circles].reshape(num_circles, 2)
        radii_opt = params[2 * num_circles :]

        # Compute total radius sum
        total_radius = np.sum(radii_opt)

        # Now, calculate penalties for overlap or boundary violation
        penalty = 0

        for i in range(num_circles):
            x, y = centers_opt[i]
            r = radii_opt[i]

            # Boundary constraints
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                penalty += 1000 * (x - r + 1e-12) if x - r < -1e-12 else 1000 * (x + r - 1 - 1e-12)
            if y - r < -1e-12 or y + r > 1 + 1e-12:
                penalty += 1000 * (y - r + 1e-12) if y - r < -1e-12 else 1000 * (y + r - 1 - 1e-12)

            # Penalty for negative radius
            if r < 0:
                penalty += 1000000 * (-r)

        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_opt[i][0] - centers_opt[j][0]
                dy = centers_opt[i][1] - centers_opt[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                required_dist = radii_opt[i] + radii_opt[j]
                if dist < required_dist - 1e-12:
                    penalty += 10000 * (required_dist - dist)

        return -total_radius + penalty  # We minimize the penalty to maximize total_radius

    # Initial parameters for Nelder-Mead
    initial_guess = np.concatenate([centers.flatten(), np.full(num_circles, 0.01)])

    # Optimization
    result = optimize.minimize(
        objective_function,
        initial_guess,
        method='Nelder-Mead',
        options={'xatol': 1e-8, 'fatol': 1e-8, 'disp': False}
    )

    centers_opt = result.x[:2 * num_circles].reshape(num_circles, 2)
    radii_opt = result.x[2 * num_circles :]

    # Validate packing
    validation_result, message = validate_packing(centers_opt, radii_opt)
    if not validation_result:
        raise ValueError(f"Packing validation failed: {message}")

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
