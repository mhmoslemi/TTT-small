from scipy.optimize import minimize

from scipy.optimize import Bounds

def run_packing():

    initial_centers = np.random.rand(num_circles, 2)
    initial_radii = np.full(num_circles, 0.02)

    def calculate_boundary_distances(radii):
        boundary_distances = []
        for i in range(num_circles):
            r = radii[i]
            x, y = initial_centers[i]
            dist_l = x - r
            dist_r = 1 - x - r
            dist_b = y - r
            dist_t = 1 - y - r
            boundary_distances.append(min(dist_l, dist_r, dist_b, dist_t))
        return boundary_distances

    def calculate_overlap_distances(radii):
        overlap_distances = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                overlap_distances.append(dist - (radii[i] + radii[j]))
        return overlap_distances

    def objective(radii):
        return -np.sum(radii)

    bounds = Bounds(0, 1)

    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': calculate_boundary_distances},
            {'type': 'ineq', 'fun': calculate_overlap_distances}
        ],
        method='SLSQP',
        options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}
    )
