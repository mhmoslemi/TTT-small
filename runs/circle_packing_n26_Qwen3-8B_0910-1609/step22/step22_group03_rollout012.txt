from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    num_circles = 26
    bounds_positions = [(0.0, 1.0) for _ in range(num_circles * 2)]  # Positions
    bounds_radii = [(0.0, 1.0) for _ in range(num_circles)]  # Radii

        """Generate a hexagonal packing configuration."""
        positions = []
        for i in range(num_circles):
            row = i // 5
            col = i % 5
            x = col / 4.5
            y = (row * np.sqrt(3)) / 4.5 + (np.sqrt(3) / 2) / 4.5
            positions.append((x, y))
        return np.array(positions)

        for i in range(num_circles):
            row, col = divmod(i, 5)
            x = col / 4.5
            y = row / 4.5
            positions.append((x, y))
        return np.array(positions)

    def generate_clustered_config():
        """Generate a clustered configuration with inner and outer circles."""
        inner_config = []
        outer_config = []
        for i in range(num_circles):
            if i < 13:
                row, col = divmod(i, 4)
                x = col / 3.5
                y = row / 3.5
                inner_config.append((x, y))

                x = col / 3.5 + 0.3
                y = row / 3.5 + 0.3
                outer_config.append((x, y))
        return np.array(inner_config + outer_config)

        """Generate a random initial configuration."""
        return np.random.rand(num_circles, 2)

        x0 = np.concatenate([centers.reshape(-1), radii])

        def objective(x):
            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            return -np.sum(radii)

        def constraint_overlap(x):
            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            dists = squareform(pdist(centers))
            constraints = []
            for i, j in combinations(range(num_circles), 2):
                constraints.append(dists[i, j] - (radii[i] + radii[j]))

            centers = x[:num_circles * 2].reshape(num_circles, 2)
