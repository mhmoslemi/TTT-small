from scipy.spatial.distance import pdist, squareform

        corners = [(0.1, 0.1), (0.9, 0.1), (0.1, 0.9), (0.9, 0.9)]

                    row_centers.append([col / 4.5 + np.random.uniform(-0.01, 0.01), row / 4.5 + np.random.uniform(-0.01, 0.01)])
            centers.extend(row_centers)
        # Trim to 26 circles
        centers = np.array(centers)[:26]
        # Ensure no overlap
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < 0.01:
                    centers[i] += 0.01 * (centers[i] - centers[j]) / dist
                    centers[j] += 0.01 * (centers[j] - centers[i]) / dist
        return centers

    # Generate several initial configurations
    best_sum = 0.0
    best_centers = np.zeros((num_circles, 2))
    best_radii = np.zeros(num_circles)

    for _ in range(3):
        centers = generate_initial_guess()
        x0 = np.concatenate([centers.reshape(-1), np.random.rand(num_circles) * 0.05])

            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            return -np.sum(radii)

        def constraint_overlap(x):
            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)

                # Bottom boundary
