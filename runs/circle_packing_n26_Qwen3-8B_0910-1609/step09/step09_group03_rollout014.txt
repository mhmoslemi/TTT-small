from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

    np.random.seed(42)

    initial_centers = np.random.rand(num_circles, 2)
    initial_radii = np.full(num_circles, 0.05)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and boundary constraints
    def non_overlap_constraint(i, j):

            return np.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2) - (radii[i] + radii[j])
        return constraint

    def boundary_constraint(i):
        def constraint(x):
            return np.min([
                x[i, 0] - radii[i],
                x[i, 1] - radii[i],
                1 - x[i, 0] - radii[i],
                1 - x[i, 1] - radii[i]
            ])
        return constraint

    # Create bounds for each radius (non-negative)
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': non_overlap_constraint(i, j)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint(i)})

    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
    )

    final_radii = result.x
