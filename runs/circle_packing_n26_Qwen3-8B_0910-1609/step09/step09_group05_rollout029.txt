from scipy.optimize import minimize

def run_packing():
    num_circles = 26

    # Define superior configuration: 4x4 grid with added circles around edge and corners
    # Create main 4x4 grid
    main_grid = np.array([
        [0.2, 0.2], [0.5, 0.2], [0.8, 0.2],
        [0.2, 0.5], [0.5, 0.5], [0.8, 0.5],
        [0.2, 0.8], [0.5, 0.8], [0.8, 0.8]
    ])
    
    # Add circles in corners (to utilize space)
    corner_circles = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]
    ])

    # Add circles along edges (vertical/bottom)
    edge_circles = np.array([
        [0.1, 0.5], [0.9, 0.5], [0.5, 0.1], [0.5, 0.9], [0.5, 0.3], [0.5, 0.7], [0.3, 0.5], [0.7, 0.5]
    ])

    # Combine them ensuring uniqueness of centers
    positions = np.vstack([main_grid, corner_circles, edge_circles])
    positions = np.unique(positions, axis=0)

    # Trim to 26 circles
    positions = positions[:26]

    # Set initial radii
    initial_radii = np.ones(num_circles) * 0.175  # A safe initial guess

    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)

    # Distance between two centers
    def distance(i, j):
        dx = positions[i, 0] - positions[j, 0]
        dy = positions[i, 1] - positions[j, 1]
        return np.sqrt(dx * dx + dy * dy)

    # Default constraints
    bounds = [(0, 1) for _ in range(num_circles)]

    # Constraint: non-overlapping
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: distance(i, j) - (x[i] + x[j])
            })

    # Constraint: each circle remains within the unit square
    for i in range(num_circles):
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: positions[i, 0] - x[i]
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: positions[i, 1] - x[i]
        })

            'fun': lambda x, i=i: 1 - positions[i, 1] - x[i]
        })

    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
    )

    final_centers = positions
    final_radii = result.x

    # Return result
    return (final_centers, final_radii, np.sum(final_radii))
