import numpy as np
import cvxpy as cp

def run_packing():
    num_circles = 26
    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable(num_circles, non_negative=True)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: All circles must be within the unit square
    for i in range(num_circles):
        x, y = centers[i]
        constraints = [
            x - radii[i] >= 0,
            y - radii[i] >= 0,
            1 - x - radii[i] >= 0,
            1 - y - radii[i] >= 0
        ]
        for constraint in constraints:
            objective.constraints.append(constraint)

    # Constraints: All circles must not overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = cp.norm(centers[i] - centers[j])
            constraints = [dist >= radii[i] + radii[j]]
            for constraint in constraints:
                objective.constraints.append(constraint)

    # Solve the optimization problem
    problem = cp.Problem(objective)
    problem.solve()

    # Extract results
    centers_val = centers.value
    radii_val = radii.value
    sum_radii = np.sum(radii_val)

    return (centers_val, radii_val, sum_radii)
