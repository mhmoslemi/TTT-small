import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    num_circles = 26

    # Initial circular arrangement in a hexagonal grid
    initial_centers = []
    angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
    radius = 0.35
    center_x, center_y = 0.5, 0.5
    for i in range(num_circles):
        theta = angles[i]
        x = center_x + radius * np.cos(theta)
        y = center_y + radius * np.sin(theta)
        initial_centers.append([x, y])
    initial_centers = np.array(initial_centers)

    # Initial radii guess based on inter-circle distances
    def initial_radii_guess(centers):
        dist_matrix = squareform(pdist(centers))
        min_dist = np.min(dist_matrix[np.triu_indices_from(dist_matrix, k=1)])
        return np.full(len(centers), min_dist / 2.0)

    initial_radii = initial_radii_guess(initial_centers)

    def objective(radii):
        return -np.sum(radii)

    bounds = [(0, 1) for _ in range(num_circles)]

    # Define constraints
    def constraint(x, i, j):
        dist = np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2))
        return dist - (x[i] + x[j])

    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: constraint(x, i, j)
            })

    for i in range(num_circles):
        def boundary(x, i=i):
            x_coord, y_coord = initial_centers[i]
            r = x[i]
            return np.min([x_coord - r, y_coord - r, 1 - x_coord - r, 1 - y_coord - r])
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: boundary(x, i)
        })

    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 800, 'ftol': 1e-12, 'disp': False}
    )

    final_centers = initial_centers
    final_radii = result.x

    # Validate the result
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, "Outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, "Overlap"
        return True, "ok"

    is_valid, msg = validate_packing(final_centers, final_radii)
    if not is_valid:
        raise ValueError(msg)

    sum_radii = np.sum(final_radii)
    return (final_centers, final_radii, sum_radii)
