import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
import matplotlib.pyplot as plt

def run_packing():
    num_circles = 26

    # Initial grid positions in a more efficient layout, covering most of the area
    positions = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],

        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ][:26])

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative of the sum

    # Define constraints
    constraints = []

    for i in range(num_circles):
        # Boundary constraint
        def boundary_constraint(radii, i=i):
            x, y = positions[i]
            return np.array([x - radii[i], y - radii[i], 1 - x - radii[i], 1 - y - radii[i]])
        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint
        })

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            # Non-overlapping constraint
            def non_overlap_constraint(radii, i=i, j=j):
                dist = np.sqrt(np.sum((positions[i] - positions[j]) ** 2))
                return dist - (radii[i] + radii[j])
            constraints.append({
                'type': 'ineq',
                'fun': non_overlap_constraint
            })

    # Use initial guess with small radii
    initial_guess = np.full(num_circles, 0.05)

    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=[(0, 1)] * num_circles,

    centers = positions
