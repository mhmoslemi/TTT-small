def run_packing():

    hex_centers = []

            return np.min([x_c - x_i, x_r - x_i, 1 - x_c - x_i, 1 - x_r - x_i])

                constraints.append(NonlinearConstraint(lambda x: constraint_dist(i, j)(x), 0, 0))
