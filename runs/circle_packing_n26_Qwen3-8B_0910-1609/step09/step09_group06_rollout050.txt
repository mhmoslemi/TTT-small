                x += 0.1

                break
        if len(centers) == num_circles:
            break

            [0.1, 0.1], [0.1, 0.9], [0.9, 0.9], [0.9, 0.1]
        ])
        centers = np.vstack((centers, corner_points[:num_circles - len(centers)]))

    initial_radii = np.full(num_circles, 0.01)

        return -np.sum(x)  # Maximize by minimizing the negative sum

            return dist - (x[i] + x[j])

            'type': 'ineq',
            'fun': constraint_func
        })

            # Ensure the circle stays within the unit square

        constraints.append({
            'type': 'ineq',
            'fun': boundary_func
        })

    try:
        result = minimize(
            objective,
            initial_radii,
            bounds=[(0, 1)] * num_circles,
            constraints=constraints,
            method='SLSQP',
