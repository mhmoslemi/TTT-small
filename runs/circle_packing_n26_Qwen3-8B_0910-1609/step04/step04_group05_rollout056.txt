import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Step 1: Generate initial seed points via a hybrid hexagonal and square packing
    def generate_initial_centers(n):
        if n < 4:
            return np.random.rand(n, 2)
        # Generate a hexagonal grid and k-means to cluster points
        grid = np.linspace(0, 1, 100)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        centroids, _ = kmeans(grid_points, n)
        return centroids

    centers = generate_initial_centers(num_circles)

    # Step 2: Define the objective to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize

    # Step 3: Define constraints for circle-circle and boundary constraints
    constraints = []

    def distance_between_centers(i, j):
        return lambda x: np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (x[i] + x[j])

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': distance_between_centers(i, j)})

    def boundary_distance(i):
        return lambda x: np.min([centers[i, 0] - x[i], centers[i, 1] - x[i], 1 - centers[i, 0] - x[i], 1 - centers[i, 1] - x[i]])

    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': boundary_distance(i)})

    # Step 4: Initial guess for radii (uniform)
    x0 = np.full(num_circles, 0.05)

    # Step 5: Define bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Step 6: Run the optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints= constraints,
        method='SLSQP',
        options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
    )

    final_radii = result.x
    final_centers = centers

    return (final_centers, final_radii, np.sum(final_radii))
