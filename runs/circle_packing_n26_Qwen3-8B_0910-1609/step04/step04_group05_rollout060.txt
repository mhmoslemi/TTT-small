def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Create evenly spaced grid for initial configuration
    grid_x = np.linspace(0, 1, 100)
    grid_y = np.linspace(0, 1, 100)
    grid_points = np.stack((grid_x, grid_y), axis=2).reshape(-1, 2)

    # Use k-means to find a good grid of initial centers for packing
    centers = kmeans(grid_points, num_circles)[0]

            centers = np.vstack((centers, new_center))

    # Define the objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative of sum to maximize

    # Define constraints for circle-circle and circle-boundary
    def constraint_circle_circle(x, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - (x[i] + x[j])

    def constraint_boundary(x, i):
        x_c, y_c = centers[i]
        r = x[i]
        return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

    # Create constraints
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_circle_circle(x, i, j)})

    x0 = np.full(num_circles, 0.05)

    bounds = [(0, 1) for _ in range(num_circles)]

    result = minimize(
