from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    sobol = Sobol(d=2)

    centers = kmeans(all_points, num_circles)[0]
    centers = np.unique(centers, axis=0)

    if len(centers) > num_circles:
        centers = centers[:num_circles]

            new_center = np.random.rand(2)

                centers = np.vstack((centers, new_center))

    # Compute radii based on distance from cluster center
    cluster_labels = np.argmin(squareform(pdist(centers)), axis=1)

        radii[cluster_mask] = 0.05 * (np.sum(dists) / len(cluster_centers))

    def objective(x):
        return -np.sum(x)  # Minimize negative of sum to maximize

    constraints = []
