def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    max_iter = 300  # Increased number of iterations for better convergence

    # Step 1: Initialize centers with a random uniform distribution
    centers = np.random.rand(num_circles, 2)

    # Function to compute distance between circles centers
    def distance(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx ** 2 + dy ** 2)

    # Function to compute the maximum possible radius for a circle at position x
    def max_possible_radius(i):
        # Distance to the nearest wall
        r = np.min([centers[i, 0] - 1e-8, 1 - centers[i, 0] - 1e-8, centers[i, 1] - 1e-8, 1 - centers[i, 1] - 1e-8])
        # Distance to other circles
        for j in range(num_circles):
            if j == i:
                continue
            r = min(r, distance(i, j) - 1e-8)
        return r

    # Objective: maximize sum of radii (minimize negative sum of radii)
    def objective(radii):
        return -np.sum(radii)

    # Constraints:

    def constraint_radius(radii, i):
        return max_possible_radius(i) - 1e-8 - radii[i]

    # 2. Distance between centers[i] and centers[j] must be >= radii[i] + radii[j]

        return dist - 1e-8 - radii[i] - radii[j]

    # List of constraints
    constraints = []
