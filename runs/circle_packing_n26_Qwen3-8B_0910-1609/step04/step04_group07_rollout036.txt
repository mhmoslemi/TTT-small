from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial import distance
from scipy.spatial.distance import squareform, pdist

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    max_iter = 1000  # Increase number of iterations for better convergence

    def max_possible_radius(x, centers, radii, i):
        r = 0

        return r

    def get_initial_centers():
        # Generate grid of points
        grid = np.linspace(0, 1, 100)

        # Use k-means clustering
        centers = kmeans(grid_points, num_circles)[0]
        centers = np.unique(centers, axis=0)
        if len(centers) > num_circles:
            centers = centers[:num_circles]

                    centers = np.vstack((centers, new_center))

    best_sum = 0
    best_centers = None
    best_radii = None
