import scipy.optimize as optimize

def run_packing():

    row_count = int(np.sqrt(n_circles))
    col_count = n_circles // row_count + 1

    for i in range(n_circles):
        row = i // col_count
        col = i % col_count

    # Start with equal radii to maximize total area
