import numpy as np
import scipy.optimize as optimize

def run_packing():
    n_circles = 26

    # Define base grid with more structured placement
    # First, we create a hexagonal-like grid near the center, then later place some circles near edges
    # For now, we'll use a grid of (10, 10) with spacing such that the initial radius is small

    # Calculate initial spacing to create a hexagonal grid
    spacing = 0.15  # Initial small spacing to avoid overlap
    centers = []
    for i in range(n_circles):
        x = i % 10 * spacing
        y = (i // 10) % 10 * spacing
        centers.append([x, y])
    centers = np.array(centers)

    # Define radii as starting with equal value (based on spacing)
    radii = np.full(n_circles, spacing / 2)

    # Optimization bounds for radii (non-negative)
    bounds = [(0, 1.0) for _ in range(n_circles)]

    # Objective: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints for non-overlapping and boundary conditions
    constraints = []

    # Function to define distance between two centers
    def distance(i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2))

    # Add non-overlapping constraints between all pairs
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: distance(i, j) - (r[i] + r[j])
            })

    # Add boundary constraints for each circle
    for i in range(n_circles):
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: centers[i, 0] - r[i]
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: centers[i, 1] - r[i]

        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: 1 - centers[i, 1] - r[i]
        })

    # Run optimization with SLSQP
    result = optimize.minimize(

        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    final_centers = centers
