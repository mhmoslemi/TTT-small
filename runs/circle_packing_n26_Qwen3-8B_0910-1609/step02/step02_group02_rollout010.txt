from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.random.rand(num_circles, 2)
    radii = np.ones(num_circles) * 0.05  # Initial small radius

    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize

    # Distance constraint between circles: distance >= sum of radii

        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (x[i] + x[j])

    # Boundary constraint: circle must fit within [0,1]^2
    def boundary(x, i):
        x_val, y_val = centers[i]
        r_val = x[i]
        return np.min([x_val - r_val, y_val - r_val, 1 - x_val - r_val, 1 - y_val - r_val])

    # Define bounds
    bounds = [(0, 1) for _ in range(num_circles)]

    # Create a list of constraints
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_dist(x, i, j)})

    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: boundary(x, i)})

    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 2000, 'ftol': 1e-8, 'disp': False}
    )

    return (final_centers, final_radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
