from scipy.optimize import minimize
from scipy.stats import beta

    alpha = 2.0  # Controls the shape of the beta distribution
    beta_ = 2.0  # Controls the shape of the beta distribution

        beta.rvs(alpha, beta_, size=num_circles),
        beta.rvs(alpha, beta_, size=num_circles)
    ])

    radii_initial = np.ones(num_circles) * 0.05

    def objective(radii):
        return -np.sum(radii)  # Minimize the negative to maximize

        dy = initial_positions[i, 1] - initial_positions[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - (radii[i] + radii[j])

    def boundary_constraint(radii, i):
        x, y = initial_positions[i]
        r = radii[i]
        return np.min([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

    bounds = [(0, 1.0) for _ in range(num_circles)]

    constraints = []

    final_centers = initial_positions
