    cols = 5
    spacing = 0.2

    y_centers = np.linspace(0.1, 0.9, rows)
    centers = np.empty((n_circles, 2))
    index = 0

                centers[index] = (x_centers[j], y_centers[i])

            radii[i] = 0.04
