    # and introduces a more optimal center cluster.

        [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],             # Bottom boundary

    bounds = [(0, 1) for _ in range(num_circles)]

    def objective(x):
        return -np.sum(x)

            return dist - (x[i] + x[j])
        return cons

            x_coord, y_coord = centers[i]
            r = x[i]
            return np.min([x_coord - r, y_coord - r, 1 - x_coord - r, 1 - y_coord - r])
        return cons

    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': constraint_func(i, j)
            })

        def validate_packing(centers, radii):
