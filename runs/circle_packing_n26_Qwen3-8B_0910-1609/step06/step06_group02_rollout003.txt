    # We will manually define an initial configuration based on a known 26-circle packing

        centers += np.random.normal(0, 0.01, centers.shape)
        return centers

    def objective(x):
        return -np.sum(x)  # Minimize negative of sum to maximize

    # Define constraints for circle-circle and boundary
    constraints = []

    for i in range(num_circles):

        r = x[i]
        return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

    x0 = np.full(num_circles, 0.01)

    bounds = [(0, 1) for _ in range(num_circles)]

    result = minimize(
