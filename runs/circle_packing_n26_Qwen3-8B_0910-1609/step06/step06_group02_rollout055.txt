import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from itertools import product

def generate_hexagonal_centers(num_points):
    """
    Generate centers in a hexagonal grid pattern of up to n points,
    ensuring they are within the unit square [0, 1] x [0, 1].
    """
    max_x = 1.0
    max_y = 1.0
    columns = int(np.sqrt(num_points)) + 1
    rows = int(np.ceil(num_points / columns))
    
    centers = []
    for i in range(rows):
        for j in range(columns):
            if i % 2 == 0:
                x = j / (columns - 1)
            else:
                x = (j + 0.5) / (columns - 1)
            y = i / (rows - 1)
            if x <= max_x and y <= max_y:
                centers.append([x, y])
    
    return np.array(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    # Initial configurations
    initial_centers = []

    hex_grid = generate_hexagonal_centers(num_circles)

    initial_centers.append(nested_grid_packing(num_circles))

    best_centers, best_radii, best_sum = None, None, -1

    for centers_initial in initial_centers:
        # Ensure exactly 26 centers

            for _ in range(num_circles - len(centers_initial)):

            centers_initial = np.vstack((centers_initial, new_centers))

        # Define objective function
        def objective(x):

        # Define constraints for circle-circle and circle-boundary
