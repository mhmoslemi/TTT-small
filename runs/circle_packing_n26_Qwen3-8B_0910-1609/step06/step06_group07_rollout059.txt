import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    best_centers, best_radii, best_sum = None, None, -1

    def generate_config_heuristic_1() -> np.ndarray:
        # Hexagonal grid in [0,1]^2
        grid = np.linspace(0, 1, 200)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.ravel(), grid_y.ravel()), axis=1)
        return kmeans(grid_points, num_circles)[0]

        # Nested square grid: cluster positions
        def cluster_centers(num_circles):
            grid = np.linspace(0, 1, int(np.sqrt(num_circles)) + 1)
            gx, gy = np.meshgrid(grid, grid)
            points = np.stack((gx.ravel(), gy.ravel()), axis=1)
            return points[points[:, 0] + points[:, 1] < 1.5]

        cluster_centers = cluster_centers(num_circles)
        if cluster_centers.shape[0] < num_circles:
            new_centers = np.random.rand(num_circles - cluster_centers.shape[0], 2)
            return np.vstack((cluster_centers, new_centers))
        return cluster_centers

    def generate_config_heuristic_3() -> np.ndarray:
        # Random but forced to be near boundary
        centers = np.random.rand(num_circles, 2)
        centers[:, 0] = np.clip(centers[:, 0], 0.1, 0.9)
        centers[:, 1] = np.clip(centers[:, 1], 0.1, 0.9)
        return centers

    def generate_config_heuristic_4() -> np.ndarray:

        centers = np.random.rand(num_circles, 2)
        centers[:, 0] = 0.5 + 0.5 * np.tanh(centers[:, 0] - 0.5)
        centers[:, 1] = 0.5 + 0.5 * np.tanh(centers[:, 1] - 0.5)
        return centers

    initial_configs = [
        generate_config_heuristic_1(),
        generate_config_heuristic_2(),
        generate_config_heuristic_3(),
        generate_config_heuristic_4(),
    ]

            centers_initial = centers_initial[:num_circles]

        radii_initial = np.zeros(num_circles)

            cluster_center = centers_initial[cluster_id]
            distances = np.linalg.norm(centers_initial[cluster_mask] - cluster_center, axis=1)
            radii_initial[cluster_mask] = 0.05 * np.min(distances)

        def objective(x):
            return -np.sum(x)

        constraints = []

            for j in range(i + 1, num_circles):
                def constraint_dist(x, i=i, j=j):
                    return np.sqrt(np.sum((centers_initial[i] - centers_initial[j]) ** 2)) - (x[i] + x[j])

                return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])
