from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Define grid-based initial configurations
    def grid_packing(num_circles):
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        centers = kmeans(grid_points, num_circles)[0]
        return centers

    # Define hexagonal grid packing
    def hexagonal_grid(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        h = np.sqrt(3) / 3
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i * h
                if i % 2 == 0:
                    x = j / (cols - 1)
                else:
                    x = (j + 0.5) / (cols - 1)
                grid.append([x, y])
        return np.array(grid)

    # Define spiral packing
    def spiral_packing(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        radii = np.linspace(0, 1, num_circles)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    # Define star-shaped packing
    def star_packing(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        points = np.array([np.cos(angles), np.sin(angles)]).T
        return points

        centers = []

                pt = np.random.rand(2)

                    centers.append(pt)
                    break
        return np.array(centers)

    # Generate diverse initial configurations
    initial_configs = []
    initial_configs.append(grid_packing(num_circles))
    initial_configs.append(hexagonal_grid(num_circles))
    initial_configs.append(spiral_packing(num_circles))
    initial_configs.append(star_packing(num_circles))
    initial_configs.append(clustered_packing(num_circles))

                centers = initial_configs[i][:num_circles]

                centers = np.vstack((centers, new_centers))
            configs.append(centers)
        return configs

    best_centers, best_radii, best_sum = None, None, -1

        return -np.sum(x)

    def dist(x, i, j):

        r = x[i]
        return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

    for centers_initial in get_initial_centers():
