import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds
import warnings
warnings.filterwarnings("ignore", category=DeprecationWarning)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate multiple initial configurations for hybrid packing
    initial_configs = []

    # Configuration 1: Hexagonal grid
    grid = np.linspace(0, 1, 200)
    grid_x, grid_y = np.meshgrid(grid, grid)
    grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
    initial_centers_1 = kmeans(grid_points, num_circles)[0]
    initial_configs.append(initial_centers_1)

    initial_configs.append(initial_centers_2)

    # Configuration 3: Nested packing
    def nest_packing(num_circles):
        num_levels = int(np.sqrt(num_circles)) + 1
        centers = []
        for i in range(num_levels):
            for j in range(num_levels):
                x = (i + 0.5) / num_levels
                y = (j + 0.5) / num_levels
                centers.append([x, y])
        if len(centers) < num_circles:
            new_centers = []
            for _ in range(num_circles - len(centers)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers - pt, axis=1) > 1e-6):
                        new_centers.append(pt)
                        break
            centers = np.vstack((centers, new_centers))
        return np.array(centers)

    initial_centers_3 = nest_packing(num_circles)
    initial_configs.append(initial_centers_3)

        centers = np.random.rand(num_circles, 2)
        # Bias towards corners
        corners = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
        for i in range(num_circles):
            if np.random.rand() < 0.3:
                centers[i] = corners[np.random.choice(4)]
            else:
                centers[i] = np.random.rand(2)
        return np.unique(centers, axis=0)[:num_circles]

    initial_centers_4 = random_perturbed(num_circles)
    initial_configs.append(initial_centers_4)

        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]
        elif len(centers_initial) < num_circles:
            new_centers = []
            for _ in range(num_circles - len(centers_initial)):

                        new_centers.append(pt)
                        break

        # Create initial radii based on minimum distance to cluster center
        cluster_labels = np.argmin(squareform(pdist(centers_initial)), axis=1)
        radii_initial = np.zeros(num_circles)

        for cluster_id in np.unique(cluster_labels):
            cluster_mask = cluster_labels == cluster_id
            cluster_center = centers_initial[cluster_id]
            distances = np.linalg.norm(centers_initial[cluster_mask] - cluster_center, axis=1)
            radii_initial[cluster_mask] = 0.05 * np.min(distances)

        # Define the objective function to maximize the sum of radii
        def objective(x):
            return -np.sum(x)  # Minimize negative of sum to maximize

        # Define constraints for circle-circle and circle-boundary
        constraints = []

        def dist(x, i, j):
            return np.sqrt(np.sum((centers_initial[i] - centers_initial[j]) ** 2)) - (x[i] + x[j])

        for i in range(num_circles):

    return (best_centers, best_radii, best_sum)
