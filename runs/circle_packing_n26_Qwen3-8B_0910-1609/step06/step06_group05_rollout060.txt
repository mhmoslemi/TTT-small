import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List
import random

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate multiple initial configurations
    initial_configs = []

    # Configuration 1: Regular grid
    grid = np.linspace(0, 1, 100)
    grid_x, grid_y = np.meshgrid(grid, grid)
    grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
    centers_1 = kmeans(grid_points, num_circles)[0]
    initial_configs.append(centers_1)

    # Configuration 2: Hexagonal packing
    def hex_grid(n, r):
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radius = 1 / (2 * r)
        points = []
        for i in range(n):
            angle = angles[i]
            x = radius * np.cos(angle)
            y = radius * np.sin(angle)
            points.append([x, y])
        return np.array(points)

    hex_points = hex_grid(num_circles, 0.6)
    initial_configs.append(hex_points)

    # Configuration 3: Random + Clustered starting points
    def random_cluster(n_clusters, n_points_per_cluster):
        clusters = []
        for _ in range(n_clusters):
            center = np.random.rand(2)
            cluster = np.random.normal(center[0], 0.05, n_points_per_cluster) * 0.9
            cluster = np.vstack((cluster, np.random.normal(center[1], 0.05, n_points_per_cluster) * 0.9)).T
            clusters.append(cluster)
        return np.vstack(clusters)

    centers_3 = random_cluster(4, 6)
    initial_configs.append(centers_3)

    # Configuration 4: Random points within the unit square
    centers_4 = np.random.rand(num_circles, 2)
    initial_configs.append(centers_4)

    best_centers, best_radii, best_sum = None, None, -1

    for centers_initial in initial_configs:
        # Ensure exactly 26 centers
        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]
        elif len(centers_initial) < num_circles:
            new_centers = []
            for _ in range(num_circles - len(centers_initial)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 1e-6):
                        new_centers.append(pt)
                        break
            centers_initial = np.vstack((centers_initial, new_centers))

        # Create initial radii based on minimum distance to cluster center
        cluster_labels = np.argmin(squareform(pdist(centers_initial)), axis=1)
        radii_initial = np.zeros(num_circles)

        for cluster_id in np.unique(cluster_labels):
            cluster_mask = cluster_labels == cluster_id
            cluster_center = centers_initial[cluster_id]
            distances = np.linalg.norm(centers_initial[cluster_mask] - cluster_center, axis=1)
            radii_initial[cluster_mask] = 0.1 * np.min(distances)

        # Define the objective function to maximize the sum of radii
        def objective(x):
            return -np.sum(x)  # Minimize negative of sum to maximize

        # Define constraints for circle-circle and circle-boundary
        def constraint_circle_circle(i, j):
            def constraint(x):
                x_i = x[i]
                x_j = x[j]
                dx = centers_initial[i, 0] - centers_initial[j, 0]
                dy = centers_initial[i, 1] - centers_initial[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                return dist - (x_i + x_j)
            return constraint

        def constraint_boundary(i):
            def constraint(x):
                x_i = x[i]
                cx, cy = centers_initial[i]
                return np.min([cx - x_i, cy - x_i, 1 - cx - x_i, 1 - cy - x_i])
            return constraint

        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append({'type': 'ineq', 'fun': constraint_circle_circle(i, j)})

        for i in range(num_circles):
            constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

        # Initial guess for radii
        x0 = radii_initial

        # Bounds for radii
        bounds = [(0, 1) for _ in range(num_circles)]

        # Run the optimization with better settings for robustness
        try:

                objective,
                x0,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={
                    'maxiter': 1000,
                    'ftol': 1e-10,
                    'eps': 1e-8,
                    'disp': False
                }
            )

                    best_centers, best_radii, best_sum = final_centers, final_radii, sum_radii

        except Exception as e:
            pass

    return (best_centers, best_radii, best_sum)
