from scipy.cluster.vq import kmeans

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

        grid = np.linspace(0, 1, 1000)

        points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        centers = kmeans(points, num_circles)[0]
        return centers

    initial_centers = generate_initial_centers()

        return -np.sum(x)  # Maximize the sum by minimizing the negative sum

            return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])
        return constraint
