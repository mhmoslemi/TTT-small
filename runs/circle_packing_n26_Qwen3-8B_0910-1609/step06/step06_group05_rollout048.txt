import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List
import random

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def generate_config(method: str) -> np.ndarray:
        if method == 'grid':
            grid = np.linspace(0, 1, 100)
            grid_x, grid_y = np.meshgrid(grid, grid)
            grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
            centers = kmeans(grid_points, num_circles)[0]
            return centers
        elif method == 'hexagonal':
            grid = np.linspace(0, 1, 100)
            grid_x, grid_y = np.meshgrid(grid, grid)
            grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
            centers = kmeans(grid_points, num_circles)[0]
            return centers
        elif method == 'nested':
            num_levels = int(np.sqrt(num_circles))
            centers = []
            for i in range(num_levels):
                for j in range(num_levels):
                    x = (i + 0.5) / num_levels
                    y = (j + 0.5) / num_levels
                    centers.append([x, y])
            if len(centers) < num_circles:
                new_centers = []
                for _ in range(num_circles - len(centers)):
                    while True:
                        pt = np.random.rand(2)
                        if np.all(np.linalg.norm(centers - pt, axis=1) > 1e-6):
                            new_centers.append(pt)
                            break
                centers = np.vstack((centers, new_centers))
            return np.array(centers)
        else:
            raise ValueError("Unsupported configuration method")

    initial_centers_list = []
    for method in ['grid', 'hexagonal', 'nested']:
        initial_centers_list.append(generate_config(method))

    best_centers, best_radii, best_sum = None, None, -1

    for centers_initial in initial_centers_list:

            centers_initial = centers_initial[:num_circles]

            new_centers = []

                        new_centers.append(pt)
                        break
            centers_initial = np.vstack((centers_initial, new_centers))

        radii_initial = np.zeros(num_circles)
        for cluster_id in np.unique(cluster_labels):
            cluster_mask = cluster_labels == cluster_id
            cluster_center = centers_initial[cluster_id]
            distances = np.linalg.norm(centers_initial[cluster_mask] - cluster_center, axis=1)
            radii_initial[cluster_mask] = 0.05 * np.min(distances)

            return -np.sum(x)  # Minimize negative of sum to maximize

        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                def constraint_func(x, i=i, j=j):
                    dx = centers_initial[i][0] - centers_initial[j][0]
                    dy = centers_initial[i][1] - centers_initial[j][1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    return dist - (x[i] + x[j])
                constraints.append({'type': 'ineq', 'fun': constraint_func})

        def boundary_func(x, i=i):
            x_c, y_c = centers_initial[i]
            r = x[i]
            return min(x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r)

        try:

                    'eps': 1e-8,
                    'disp': False

        return (best_centers, best_radii, sum_radii)
