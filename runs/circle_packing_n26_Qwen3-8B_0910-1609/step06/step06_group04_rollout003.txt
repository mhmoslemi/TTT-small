from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Initial configuration using a hybrid strategy:
    # - Hexagonal packing
    # - Boundary-optimized clusters
    # - Nested, non-overlapping placement

    def hexagonal_grid(n):
        return np.array([[(i + 0.5) / n, (j + 0.5) / n] for i in range(n) for j in range(n)])

    def boundary_refinement(n):
        points = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
        for _ in range(n - 4):
            new_point = np.random.rand(2)
            while np.any(np.linalg.norm(points - new_point, axis=1) < 1e-6):
                new_point = np.random.rand(2)
            points = np.vstack((points, new_point))
        return points

    initial_config_1 = hexagonal_grid(int(np.sqrt(num_circles)))
    initial_config_2 = boundary_refinement(num_circles)

    # Compute kmeans for hexagonal-based packing
    initial_centers_1 = kmeans(initial_config_1, num_circles)[0]
    initial_centers_2 = kmeans(initial_config_2, num_circles)[0]

    # Best configuration so far
    best_centers, best_radii, best_sum = None, None, -1

    # Iterate with multiple configurations
    for centers_initial in [initial_centers_1, initial_centers_2]:
        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]
        elif len(centers_initial) < num_circles:
            new_centers = []
            for _ in range(num_circles - len(centers_initial)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 1e-6):
                        new_centers.append(pt)
                        break
            centers_initial = np.vstack((centers_initial, new_centers))

        # Create initial radii based on distance to cluster center
        cluster_labels = np.argmin(squareform(pdist(centers_initial)), axis=1)
        radii_initial = np.zeros(num_circles)

        for cluster_id in np.unique(cluster_labels):
            cluster_mask = cluster_labels == cluster_id
            distances = np.linalg.norm(centers_initial[cluster_mask] - centers_initial[cluster_id], axis=1)

            x_c, y_c = centers_initial[i]
            r = x[i]
            return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: boundary(x, i=i)})

        x0 = radii_initial

        bounds = [(0, 1) for _ in range(num_circles)]

        result = minimize(
