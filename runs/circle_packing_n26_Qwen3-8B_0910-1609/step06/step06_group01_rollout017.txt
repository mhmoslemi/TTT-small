def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    initial_configs = []

    grid = np.linspace(0, 1, 200)

            x = r * np.cos(i * angle)
            y = r * np.sin(i * angle)

            grid_y.append(y)
        grid_x = np.array(grid_x)
        grid_y = np.array(grid_y)

        return np.array(centers)
