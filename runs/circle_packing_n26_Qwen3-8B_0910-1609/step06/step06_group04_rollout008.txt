from scipy.optimize import minimize

from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    best_centers, best_radii, best_sum = None, None, -1

    # Function to generate initial configurations
    def generate_configurations(n):
        configs = []
        # 1. Hexagonal grid packing
        grid_x = np.linspace(0, 1, 200)
        grid_y = np.linspace(0, 1, 200)
        grid = np.stack((grid_x, grid_y), axis=-1)
        cluster_centers = kmeans(grid.reshape(-1, 2), n)[0]
        configs.append(cluster_centers)

        corners = np.array([[0, 0], [1, 0], [0, 1], [1, 1]])

        seed_points = np.vstack((corners, center))
        cluster_centers = kmeans(seed_points, n)[0]
        configs.append(cluster_centers)

        # 3. Random seeding
        random_seeds = np.random.rand(n, 2)
        configs.append(random_seeds)

        return configs

    configs = generate_configurations(num_circles)

    for centers_initial in configs:
        if len(centers_initial) != num_circles:
            centers_initial = np.vstack((centers_initial, np.random.rand(num_circles - len(centers_initial), 2)))

        # Initialize radii using minimum distance to cluster
        cluster_labels = np.argmin(squareform(pdist(centers_initial)), axis=1)
        radii_initial = np.zeros(num_circles)
        for cluster_id in np.unique(cluster_labels):
            cluster_mask = cluster_labels == cluster_id
            distances = np.linalg.norm(centers_initial[cluster_mask] - centers_initial[cluster_id], axis=1)
            radii_initial[cluster_mask] = 0.05 * np.min(distances)

        # Optimization objective: maximize sum of radii
        def objective(x):
            return -np.sum(x)

        # Constraints
        constraints = []

        # Circle-circle non-overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                def dist_func(x, i=i, j=j):
                    dx = centers_initial[i, 0] - centers_initial[j, 0]
                    dy = centers_initial[i, 1] - centers_initial[j, 1]
                    dist = np.sqrt(dx ** 2 + dy ** 2)
                    return dist - (x[i] + x[j])

        for i in range(num_circles):

                x_c, y_c = centers_initial[i]
                r = x[i]
                return min(x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r)
            constraints.append({'type': 'ineq', 'fun': boundary_func})

        # Bounds
        bounds = [(0, 1) for _ in range(num_circles)]

        # Initial guess
        x0 = radii_initial

        # Run optimization
        try:

                x0,
                bounds=bounds,
                constraints=constraints,

                    'eps': 1e-8,
                    'disp': False
                }
            )

                    best_centers = centers_initial
                    best_radii = result.x
                    best_sum = sum_radii
        except:
            pass

    return (best_centers, best_radii, best_sum)
