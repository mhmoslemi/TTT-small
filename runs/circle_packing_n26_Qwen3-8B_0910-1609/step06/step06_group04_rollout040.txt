from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def generate_hexagonal(n):
        centers = []
        x = np.linspace(0, 1, n)
        y = np.linspace(0, 1, n)
        for i in range(n):
            for j in range(n):
                if (i + j) % 2 == 0:
                    centers.append([x[i], y[j]])
                else:
                    centers.append([x[i], y[j]])
        return np.array(centers)[:num_circles]

    def generate_grid(n):
        x = np.linspace(0, 1, n)
        y = np.linspace(0, 1, n)
        grid = np.array([x, y]).T
        return grid[:num_circles]

    def generate_nested(n):
        centers = []
        level = int(np.sqrt(n))
        for i in range(level):
            for j in range(level):
                x = (i + 0.5) / level
                y = (j + 0.5) / level
                centers.append([x, y])
        while len(centers) < n:
            new_centers = []
            for _ in range(n - len(centers)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers - pt, axis=1) > 1e-6):
                        new_centers.append(pt)
                        break
            centers = np.vstack((centers, new_centers))
        return centers[:n]

    def compute_initial_radii(centers):
        cluster_labels = np.argmin(squareform(pdist(centers)), axis=1)
        radii = np.zeros(len(centers))
        for cluster_id in np.unique(cluster_labels):
            cluster_mask = cluster_labels == cluster_id
            distances = np.linalg.norm(centers[cluster_mask] - centers[cluster_id], axis=1)
            radii[cluster_mask] = 0.05 * np.min(distances)
        return radii

    def objective(x, centers):
        return -np.sum(x)

    def constraint_circle_circle(x, i, j, centers):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - (x[i] + x[j])

    def constraint_bound(x, i, centers):
        x_c, y_c = centers[i]
        r = x[i]
        return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

    configs = []

    configs.append(generate_hexagonal(200))
    configs.append(generate_nested(26))

    best_centers = None
    best_radii = None

    for centers_initial in configs:

            new_centers = []
            for _ in range(num_circles - len(centers_initial)):

                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 1e-6):
                        new_centers.append(pt)
                        break
            centers_initial = np.vstack((centers_initial, new_centers))

        radii_initial = compute_initial_radii(centers_initial)

        bounds = [(0, 1) for _ in range(num_circles)]

        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_circle_circle(x, i, j, centers_initial)})

            x0,
            bounds=bounds,
            constraints=constraints,

                'ftol': 1e-10,
                'eps': 1e-8,
                'disp': False
            }
        )

            if sum_radii > best_sum:
                best_centers = centers_initial
                best_radii = result.x
                best_sum = sum_radii

    return (best_centers, best_radii, best_sum)
