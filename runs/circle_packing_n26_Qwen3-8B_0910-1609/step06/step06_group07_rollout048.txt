import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def generate_hybrid_grid(n: int, grid_points: int = 200):
        grid = np.linspace(0, 1, grid_points)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        if n <= grid_points.shape[0]:
            return kmeans(grid_points, n)[0]
        return np.vstack((generate_hybrid_grid(n // 2, grid_points=grid_points), generate_hybrid_grid(n // 2, grid_points=grid_points)))

    def generate_hex_grid(n: int, grid_points: int = 200):
        grid = np.linspace(0, 1, grid_points)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        if n <= grid_points.shape[0]:
            return kmeans(grid_points, n)[0]
        return np.vstack((generate_hex_grid(n // 2, grid_points=grid_points), generate_hex_grid(n // 2, grid_points=grid_points)))

    def generate_nested(n: int):
        centers = []
        levels = int(np.sqrt(n)) + 1
        for i in range(levels):
            for j in range(levels):
                x = (i + 0.5) / levels
                y = (j + 0.5) / levels
                if x < 1 and y < 1:
                    centers.append([x, y])
        n_remaining = n - len(centers)
        if n_remaining > 0:
            for _ in range(n_remaining):
                while True:
                    pt = np.random.rand(2)
                    if np.linalg.norm(centers - pt, axis=1).min() > 1e-6:
                        centers.append(pt)
                        break
        return np.array(centers)

    def generate_random_config(n: int):
        centers = np.random.rand(n, 2)
        return centers

    def refine_placement(centers: np.ndarray, num_circles: int) -> np.ndarray:
        n = centers.shape[0]
        if n < num_circles:
            while n < num_circles:
                pt = np.random.rand(2)
                if np.linalg.norm(centers - pt, axis=1).min() > 1e-6:
                    centers = np.vstack((centers, pt))
                    n += 1
        return centers

    def compute_initial_radii(centers: np.ndarray, num_circles: int) -> np.ndarray:
        distances = pdist(centers)
        cluster_labels = np.argmin(squareform(distances), axis=1)
        radii = np.zeros(num_circles)
        for i in range(num_circles):
            cluster_mask = cluster_labels == i
            cluster_distances = distances[cluster_mask]
            radii[i] = 0.05 * np.min(cluster_distances)
        return radii

    def objective(x: np.ndarray) -> float:
        return -np.sum(x)  # Maximize sum

    def constraint_circle_circle(x: np.ndarray, i: int, j: int) -> float:
        dist = np.linalg.norm(centers[i] - centers[j])
        return dist - (x[i] + x[j])

    def constraint_boundary(x: np.ndarray, i: int) -> float:
        x_c, y_c = centers[i]
        r = x[i]
        return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

    initial_configs = []

    # Configuration 1: Hybrid grid-based packing
    initial_configs.append(generate_hybrid_grid(num_circles))

    # Configuration 2: Hexagonal grid-based packing
    initial_configs.append(generate_hex_grid(num_circles))

    # Configuration 3: Nested approach
    initial_configs.append(generate_nested(num_circles))

    # Configuration 4: Random packing
    initial_configs.append(generate_random_config(num_circles))

    best_centers, best_radii, best_sum = None, None, -1

    for centers_initial in initial_configs:
        centers_initial = refine_placement(centers_initial, num_circles)
        radii_initial = compute_initial_radii(centers_initial, num_circles)

        constraints = []

            for j in range(i + 1, num_circles):

        for i in range(num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i=i)})

                    best_centers, best_radii, best_sum = final_centers, final_radii, sum_radii

        except Exception as e:
            pass

    return (best_centers, best_radii, best_sum)
