from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate initial configurations
    def hexagonal_grid(num_circles):
        row = int(np.sqrt(num_circles))
        row = max(1, row)
        col = (num_circles + row - 1) // row
        grid = []
        for i in range(row):
            for j in range(col):
                x = j / (col - 1) if col > 1 else 0.5
                y = i / (row - 1) if row > 1 else 0.5
                grid.append([x, y])
        return np.array(grid)

    def spiral_packing(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        radii = np.linspace(0, 1, num_circles)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def grid_points(num_points):
        grid = np.linspace(0, 1, num_points)
        grid_x, grid_y = np.meshgrid(grid, grid)
        return np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)

    def get_initial_centers():
        initial_configs = []

        # Grid-based
        initial_centers_1 = hexagonal_grid(num_circles)
        initial_configs.append(initial_centers_1)

            centers_initial = centers_initial[:num_circles]

            for _ in range(num_circles - len(centers_initial)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 0.1):
                        new_centers.append(pt)
                        break
            centers_initial = np.vstack((centers_initial, new_centers))

        # Initial guess for radii
        radii_initial = np.full(num_circles, 0.02)

            return -np.sum(x)

        constraints = []

        for i in range(num_circles):
            for j in range(i + 1, num_circles):

                    dist = np.sqrt(np.sum((centers_initial[i] - centers_initial[j]) ** 2)) - (x[i] + x[j])
