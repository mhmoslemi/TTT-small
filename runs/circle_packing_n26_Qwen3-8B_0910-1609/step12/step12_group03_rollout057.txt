from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    initial_centers = np.array([
        [0.05, 0.05], [0.95, 0.05], [0.95, 0.95], [0.05, 0.95],

    def objective(x):
        return -np.sum(x)

    def constraint_func(i, j):
        def cons(x):
            dist = np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2))
            return dist - (x[i] + x[j])
        return cons

            x_coord, y_coord = initial_centers[i]
            r = x[i]
            return np.min([x_coord - r, y_coord - r, 1 - x_coord - r, 1 - y_coord - r])
        return cons

    constraints = []
    for i in range(num_circles):

            constraints.append({'type': 'ineq', 'fun': constraint_func(i, j)})

    result = minimize(

                if dist < radii[i] + radii[j] - 1e-12:

        return True, "ok"

    return (initial_centers, result.x, sum_radii)
