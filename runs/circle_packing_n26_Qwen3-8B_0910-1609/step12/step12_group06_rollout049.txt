from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    def objective(radii):
        return -np.sum(radii)

    # Define the constraints for non-overlap and boundary
    def constraint_func(radii, centers):
        dists = cdist(centers, centers)
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append(dists[i, j] - (radii[i] + radii[j]))
        return np.array(constraints)

        min_dist_to_boundary = np.min(np.array([
            np.min(centers[:, 0] - radii),
            np.min(centers[:, 1] - radii),
            np.min(1 - centers[:, 0] - radii),
            np.min(1 - centers[:, 1] - radii)
        )), axis=0)
        return np.array(min_dist_to_boundary)

    def get_hex_grid_centers(num_circles):

        row = 0
        col = 0

            y = row / 2
            grid_centers.append([x, y])
            col += 1
            if col % 2 == 0:
                row += 1
            if i == num_circles - 1:
                break
        return np.array(grid_centers)
