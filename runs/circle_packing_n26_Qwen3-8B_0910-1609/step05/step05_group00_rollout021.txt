def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    initial_configs = []

            for j in range(num_levels):
                x = (i + 0.5) / num_levels

                centers.append([x, y])

                        break
            centers = np.vstack((centers, new_centers))

        return np.column_stack((x, y))
