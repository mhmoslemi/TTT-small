import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Create a grid of points for initialization
    grid = np.linspace(0, 1, 100)
    grid_x, grid_y = np.meshgrid(grid, grid)
    grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)

    # Use k-means++ for better initialization
    initial_centers = kmeans(grid_points, num_circles, iter=100)[0]

    # Ensure we have exactly 26 centers
    if len(initial_centers) > num_circles:
        initial_centers = initial_centers[:num_circles]
    elif len(initial_centers) < num_circles:
        new_centers = initial_centers.copy()
        for _ in range(num_circles - len(new_centers)):
            while True:
                pt = np.random.rand(2)
                if np.all(np.linalg.norm(new_centers - pt, axis=1) > 1e-6):
                    new_centers = np.vstack((new_centers, pt))
                    break
        initial_centers = new_centers

    centers = initial_centers

    # Manual placement for corner circles
    corner_centers = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9]
    ])
    corner_mask = np.zeros(num_circles, dtype=bool)
    for i in range(4):
        idx = np.argmin(np.linalg.norm(centers - corner_centers[i], axis=1))
        if not corner_mask[idx]:
            corner_mask[idx] = True

    # Assign radii for corner circles
    radii = np.zeros(num_circles)
    for i in range(4):
        if corner_mask[i]:
            radii[i] = 0.2

    # Assign radii for edge-aligned circles
    edge_centers = np.array([
        [0.5, 0.1],
        [0.5, 0.9],
        [0.1, 0.5],
        [0.9, 0.5]
    ])
    edge_mask = np.zeros(num_circles, dtype=bool)
    for i in range(4):
        idx = np.argmin(np.linalg.norm(centers - edge_centers[i], axis=1))
        if not edge_mask[idx]:
            edge_mask[idx] = True

    for i in range(4):
        if edge_mask[i]:
            radii[i] = 0.15

    # Assign radii for the rest using k-means clustering
    cluster_labels = np.argmin(squareform(pdist(centers)), axis=1)
    cluster_radii = np.zeros(num_circles)
    for cluster_id in np.unique(cluster_labels):
        cluster_mask = cluster_labels == cluster_id
        cluster_center = centers[cluster_id]
        distances = np.linalg.norm(centers[cluster_mask] - cluster_center, axis=1)
        cluster_radii[cluster_mask] = 0.05 * np.mean(distances)

    radii = np.where(corner_mask, radii, np.where(edge_mask, radii, cluster_radii))

    # Define the objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative of sum to maximize

    # Define constraints for circle-circle and circle-boundary
    constraints = []

    # Circle-circle constraints
    def dist(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (x[i] + x[j])

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: dist(x, i, j)})

    # Boundary constraints
    def boundary(x, i=i):
        x_c, y_c = centers[i]
        r = x[i]
        return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: boundary(x, i=i)})

    # Initial guess for radii
    x0 = radii

    # Bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={
            'maxiter': 1500,
            'ftol': 1e-10,
            'eps': 1e-8,
            'disp': False
        }
    )

    final_radii = result.x
    final_centers = centers

    return (final_centers, final_radii, np.sum(final_radii))
