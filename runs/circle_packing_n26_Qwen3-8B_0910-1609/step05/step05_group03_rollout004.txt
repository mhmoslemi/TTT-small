from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    centers = np.random.rand(num_circles, 2)  # Randomly initialize centers within the square

    # Define the objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize the negative sum to maximize it

    # Define constraints for non-overlapping circles
    constraints = []

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def dist_constraint(x, i=i, j=j):
                return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (x[i] + x[j])
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    # Define constraints for circles staying within the unit square
    for i in range(num_circles):
        def boundary_constraint(x, i=i):
            x_c, y_c = centers[i]
            r = x[i]
            return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])
        constraints.append({'type': 'ineq', 'fun': boundary_constraint})

    # Initial guess for radii can be set to zero or based on initial spacing
    x0 = np.zeros(num_circles)

    # Bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Run the optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}

    # Verify the result
    if not result.success:
