from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.cluster.vq import kmeans_plusplus
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate initial centers using k-means++ for better clustering
    grid = np.linspace(0, 1, 200)
    grid_x, grid_y = np.meshgrid(grid, grid)
    grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)

    # Use k-means++ for seeding
    initial_centers = kmeans_plusplus(grid_points, num_circles)[0]

    # Define the objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative of sum to maximize

    # Define constraints for circle-circle and circle-boundary
    constraints = []

    def dist(x, i, j):
        return np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2)) - (x[i] + x[j])

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: dist(x, i, j)})

    def boundary(x, i=i):
        x_c, y_c = initial_centers[i]
        r = x[i]
        return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])

    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: boundary(x, i=i)})

    x0 = np.full(num_circles, 0.05)

    # Bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Run the optimization with more sophisticated parameters
    result = minimize(

        bounds=bounds,

        }
