from scipy.cluster.vq import kmeans

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # K-means++ initialization for better spacing
    def kmeans_plus_plus(grid_points, k):
        centers = [grid_points[np.random.choice(grid_points.shape[0])]]
        for _ in range(k - 1):
            distances = np.array([np.min([np.linalg.norm(p - c) for c in centers]) for p in grid_points])
            probabilities = distances / np.sum(distances)
            next_center = grid_points[np.random.choice(grid_points.shape[0], p=probabilities)]
            centers.append(next_center)
        return np.array(centers)

    # Use k-means++ for better initial configuration
    initial_centers = kmeans_plus_plus(grid_points, num_circles)

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Define constraints for circle-circle and circle-boundary
    def constraint_circle_circle(i, j):
        def cons(x):
            dx = initial_centers[i][0] - initial_centers[j][0]
            dy = initial_centers[i][1] - initial_centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)

        return cons

    def constraint_boundary(i):

            r = x[i]
            return np.min([cx - r, cy - r, 1 - cx - r, 1 - cy - r])
        return cons

    # Set up constraints
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': constraint_circle_circle(i, j)})

    x0 = np.full(num_circles, 0.05)
