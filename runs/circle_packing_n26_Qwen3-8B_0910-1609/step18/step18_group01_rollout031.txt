from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    def generate_hex_grid(num_circles):

                y = 0.1 + i * 0.16

                    x += 0.08
                grid.append([x, y])

        return -np.sum(radii)

    def constraint_func(radii):

        return np.array([dists[i, j] - (radii[i] + radii[j]) for i in range(num_circles) for j in range(i + 1, num_circles)])

        return np.array([np.min([initial_centers[i, 0] - radii[i], initial_centers[i, 1] - radii[i], 1 - initial_centers[i, 0] - radii[i], 1 - initial_centers[i, 1] - radii[i]]) for i in range(num_circles)])

    try:
