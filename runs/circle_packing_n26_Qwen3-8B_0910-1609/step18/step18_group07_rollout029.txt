import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from scipy.optimize import Bounds

def run_packing():
    num_circles = 26

    # Create a hybrid grid: corner clusters and hexagonal grid in the center
    def make_corner_clusters(n_clusters_per_corner=3, cluster_radius=0.25):
        corners = np.array([[0.0, 0.0], [1.0, 0.0], [1.0, 1.0], [0.0, 1.0]])
        clusters = []
        for corner in corners:
            for i in range(n_clusters_per_corner):
                angle = 2 * np.pi * i / n_clusters_per_corner
                dx = cluster_radius * np.cos(angle)
                dy = cluster_radius * np.sin(angle)
                clusters.append(corner + [dx, dy])
        return np.array(clusters)

    # Hybrid packing: corner clusters and hexagonal grid in the center
    corner_clusters = make_corner_clusters(n_clusters_per_corner=3, cluster_radius=0.15)
    proposed_centers = corner_clusters

    # Generate hexagonal grid in the center
    center_density = 0.05
    center_grid = []
    for i in range(10):
        for j in range(10):
            x = i * center_density
            y = j * center_density
            if (i + j) % 2 == 0:
                center_grid.append([x, y])
            else:
                center_grid.append([x + center_density / 2, y])
    center_grid = np.array(center_grid)[:26 - len(corner_clusters)]
    proposed_centers = np.vstack([proposed_centers, center_grid])

    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    def constraint_boundary(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r)
            constraints.append(y - r)
            constraints.append(1 - x - r)
            constraints.append(1 - y - r)
        return np.array(constraints)

    # Initial guess
    initial_radii = np.full(num_circles, 0.01)

    # Optimization bounds
    bounds = Bounds(np.zeros(num_circles), np.ones(num_circles))

    # Run optimization
    result = minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}

        method='SLSQP',
        options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False}
    )

        raise ValueError(f"Invalid packing: {msg}")

    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
