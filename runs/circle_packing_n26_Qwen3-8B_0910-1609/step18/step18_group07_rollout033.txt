from scipy.spatial.distance import squareform, pdist

def run_packing():

    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col * 0.2 + 0.1
        y = row * 0.2 + 0.1
        centers[i, 0] = x
        centers[i, 1] = y

    bounds = [(0, 1) for _ in range(num_circles)]

    def objective(radii):
        return -np.sum(radii)

        return np.array([dists[i, j] - (radii[i] + radii[j]) for i in range(num_circles) for j in range(i + 1, num_circles)])

        return np.array([np.min([centers[i, 0] - radii[i], centers[i, 1] - radii[i], 1 - centers[i, 0] - radii[i], 1 - centers[i, 1] - radii[i]]) for i in range(num_circles)])

    initial_radii = np.array([0.05] * num_circles)
