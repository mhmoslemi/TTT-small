from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    # Circle centers will be placed near the boundaries to maximize potential radii
    initial_centers = np.random.rand(num_circles, 2)

    # Initial guess for radii
    initial_radii = np.random.rand(num_circles) * 0.1

    def objective(radii):
        return -np.sum(radii)

    # Constraints:
    # 1. No circle is outside the unit square
    def boundary_constraint(radii):
        dist_to_boundary = np.zeros(num_circles)
        for i in range(num_circles):
            dist_to_boundary[i] = np.min([
                initial_centers[i, 0] - radii[i],
                initial_centers[i, 1] - radii[i],
                1.0 - initial_centers[i, 0] - radii[i],
                1.0 - initial_centers[i, 1] - radii[i]
            ])
        return dist_to_boundary

    # 2. No overlapping between any two circles
    def non_overlap_constraint(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)

    # Bounds for radii
    bounds = Bounds(lb=0.0, ub=1.0)

    # Run the optimization
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=[

        options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
    )

    # Extract the final configuration
    final_centers = initial_centers
    final_radii = result.x

    def validate_packing(centers, radii):
