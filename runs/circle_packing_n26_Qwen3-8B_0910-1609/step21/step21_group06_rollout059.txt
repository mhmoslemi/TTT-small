from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial layout: hexagonal grid
    grid_spacing = 0.3
    centers = []

        x = i % 6 * grid_spacing
        y = (i // 6) * (grid_spacing * np.sqrt(3) / 2)
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii: small but not zero
    radii = np.random.rand(num_circles) * 0.1

    # Flatten centers and radii for optimization
    x0 = np.concatenate([centers.reshape(-1), radii])

    # Define bounds: positions in [0, 1] and radii in [0, 1]

        radii = x[num_circles * 2:].reshape(num_circles)
        return -np.sum(radii)  # Minimize negative sum => maximize sum

    def constraint_overlap(x):
        centers = x[:num_circles * 2].reshape(num_circles, 2)
        radii = x[num_circles * 2:].reshape(num_circles)
        dists = squareform(pdist(centers))
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append(dists[i, j] - (radii[i] + radii[j]))
        return np.array(constraints)

    def constraint_boundary(x):

            # Right boundary
            constraints.append(1.0 - x_i - r_i)
            # Bottom boundary
            constraints.append(y_i - r_i)
            # Top boundary
            constraints.append(1.0 - y_i - r_i)
        return np.array(constraints)
