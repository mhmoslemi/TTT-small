import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    num_circles = 26
    # Initial configurations
    def generate_hexagonal_config():
        """Generate a hexagonal packing configuration."""
        positions = []
        for i in range(num_circles):
            x = i / 5
            y = (i // 5) * (np.sqrt(3) / 2) * (1 / 5)
            positions.append((x, y))
        return np.array(positions)

    def generate_grid_config():
        """Generate a grid-based configuration."""
        positions = []
        for i in range(num_circles):
            row = i // 5
            col = i % 5
            x = col / 4.5
            y = row / 4.5
            positions.append((x, y))
        return np.array(positions)

    def generate_random_config():
        """Generate a random initial configuration."""
        return np.random.rand(num_circles, 2)

    def generate_dense_config():
        """Generate a densely packed configuration as a baseline."""
        positions = []
        for i in range(num_circles):
            if i % 2 == 0:
                x = i / 5
                y = (i // 5) * (np.sqrt(3) / 2) * (1 / 5)
            else:
                x = (i + 0.5) / 5
                y = (i // 5) * (np.sqrt(3) / 2) * (1 / 5)
            positions.append((x, y))
        return np.array(positions)

    def generate_corner_corner_config():
        """Generate a configuration with circles near the square corners."""
        positions = []
        for i in range(num_circles):
            x = i / (num_circles - 1)
            y = i / (num_circles - 1)
            positions.append((x, y))
        return np.array(positions)

    configs = [
        generate_hexagonal_config(),
        generate_grid_config(),
        generate_random_config(),
        generate_dense_config(),
        generate_corner_corner_config()
    ]

    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    for init_config in configs:
        centers = init_config
        radii = np.random.rand(num_circles) * 0.1

        x0 = np.concatenate([centers.reshape(-1), radii])

        def objective(x):
            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            return -np.sum(radii)

        def constraint_overlap(x):
            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            dists = squareform(pdist(centers))
            constraints = []
            for i in range(num_circles):
                for j in range(i + 1, num_circles):
                    constraints.append(dists[i, j] - (radii[i] + radii[j]))
            return np.array(constraints)

            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            constraints = []
            for i in range(num_circles):
                x_i, y_i = centers[i]
                r_i = radii[i]
                constraints.append(x_i - r_i)
                constraints.append(1.0 - x_i - r_i)
                constraints.append(y_i - r_i)
                constraints.append(1.0 - y_i - r_i)
            return np.array(constraints)

        for _ in range(num_circles * 2):
            x_bounds.append((0.0, 1.0))
        for _ in range(num_circles):
            x_bounds.append((0.0, 1.0))

            objective,
            x0,
            bounds=x_bounds,
            constraints=[
                {'type': 'ineq', 'fun': constraint_overlap},
                {'type': 'ineq', 'fun': constraint_boundary}

            method='SLSQP',
            options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}

            best_centers = centers

    return (best_centers, best_radii, best_sum_radii)
