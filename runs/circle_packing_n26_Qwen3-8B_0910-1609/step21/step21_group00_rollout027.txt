def run_packing():

        if n < 1:
            return np.zeros((n, 2)), np.zeros(n)

        x = np.linspace(0, 1, n)
        y = np.linspace(0, 1, n)

        return squareform(pdist(centers))

    def boundary_distances(centers, radii):
        dists = np.zeros(len(centers))
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            dists[i] = min(x - r, 1.0 - x - r, y - r, 1.0 - y - r)
        return dists

    # Objective function to maximize total radius
    def objective(x):
        centers = x[:num_circles * 2].reshape(num_circles, 2)

        return -np.sum(radii)

    def constraint_boundary(x):
