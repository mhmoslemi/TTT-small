from scipy.spatial.distance import pdist, squareform

                positions.append((x, y))

        """Generate a grid-based configuration with refined spacing."""
        rows = 5
        cols = 5
        spacing = 1.0 / (cols - 1)

                y = i * spacing
                positions.append((x, y))
        return np.array(positions)

    best_radii = None

    for init_config in configs:

        centers = init_config
        radii = np.random.rand(num_circles) * 0.2

        x0 = np.concatenate([centers.reshape(-1), radii])

            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            return -np.sum(radii)

            centers = x[:num_circles * 2].reshape(num_circles, 2)
            radii = x[num_circles * 2:].reshape(num_circles)
            dists = squareform(pdist(centers))
            constraints = []
            for i in range(num_circles):
                for j in range(i + 1, num_circles):
                    constraints.append(dists[i, j] - (radii[i] + radii[j]))
            return np.array(constraints)

            centers = x[:num_circles * 2].reshape(num_circles, 2)

                # Bottom boundary

                # Top boundary

            return np.array(constraints)
