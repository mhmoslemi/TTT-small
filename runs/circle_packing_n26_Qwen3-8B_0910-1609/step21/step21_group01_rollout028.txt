from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

            y = i / (num_rows - 1)
            centers[i * num_cols + j, 0] = x
            centers[i * num_cols + j, 1] = y
            radii[i * num_cols + j] = 0.05

    x0 = np.concatenate([centers.reshape(-1), radii])
