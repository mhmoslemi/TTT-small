from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    cols = 6
    rows = 5

    total_rows = rows

        raise ValueError("Not enough rows or columns to fit 26 circles in hexagonal packing")

    row_spacing = np.sqrt(3) / 2.0 / (rows - 1)  # Height between staggered rows

        row = i // cols
        x = col * col_spacing
        y = row * row_spacing

        # Apply horizontal offset for staggered rows
        if row % 2 == 1:
            x += offset

        centers[i, 0] = x
        centers[i, 1] = y

    # Flatten into a single array for optimization
    x0 = np.concatenate([centers.reshape(-1), radii])

        radii = x[num_circles * 2:].reshape(num_circles)
        return -np.sum(radii)  # Minimize negative sum => maximize sum

    def constraint_overlap(x):

        dists = squareform(pdist(centers))
