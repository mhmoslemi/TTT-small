from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate initial configurations
    def generate_hex_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                if cols > 1 or rows > 1:
                    grid.append([x, y])
        return np.array(grid)

    def generate_spiral(n):
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0, 1, n)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_kmeans(n):
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, n)[0]
        return initial_centers

    def generate_nested(n):
        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5

                centers.append(pt)
        return np.array(centers)

        def objective(x):
            return -np.sum(x)

                x_i = centers_initial[i, 0]

                r = x[i]
                return np.min([x_i - r, y_i - r, 1 - x_i - r, 1 - y_i - r])
