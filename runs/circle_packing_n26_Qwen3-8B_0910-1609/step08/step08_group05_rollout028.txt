from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    def generate_rectangular_grid(n, grid_resolution=50):
        grid = np.linspace(0, 1, grid_resolution)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, n)[0]
        perturbation = 0.05
        perturbed_centers = initial_centers + np.random.uniform(-perturbation, perturbation, initial_centers.shape)
        return perturbed_centers

    def generate_hexagonal_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                grid.append([x, y])
        return np.array(grid)

    def generate_spiral_packing(n):
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0, 1, n)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_nested_packing(n):

        base = 0.1

                x = (i + 0.5) / 5

                centers.append([x, y])

            pt = np.random.rand(2)
            if np.all(np.linalg.norm(centers - pt, axis=1) > 0.1):
                centers.append(pt)

        return generate_rectangular_grid(n)
