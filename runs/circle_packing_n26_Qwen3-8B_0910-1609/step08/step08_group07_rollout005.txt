def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Initialize a Voronoi-based grid for hexagonal packing
    def hex_grid(n):
        grid = []
        for i in range(n):
            for j in range(n):
                x = (i + 0.5) / (n - 1)
                y = (j + 0.5) / (n - 1)
                grid.append([x, y])
        return np.array(grid)
