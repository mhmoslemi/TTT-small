import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    max_radius = 1.0
    tolerance = 1e-6
    boundary_pad = 1e-8

    # Create a staggered grid packing configuration
    # https://www.researchgate.net/publication/322393144_Optimal_Packing_of_Circles_in_a_Square
    def staggered_grid(num_circles):
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = (j + 0.5) / (cols - 1)
                y = (i + 0.5) / (rows - 1)
                centers.append([x, y])
        return np.array(centers)

    # Main function to compute distance between two points
    def dist(p1, p2):
        return np.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

    # Optimize radii and positions
    def optimize_packing(centers_initial):
        n = num_circles

        # Initial guess for radii
        radii_initial = np.full(n, 0.02)

        def objective(radii):
            return -np.sum(radii)  # we minimize this to maximize the sum

        def constraint_circle_boundary(i):
            def cons(radii):
                x, y = centers_initial[i]
                r = radii[i]
                return x - r - boundary_pad  # left
            return {'type': 'ineq', 'fun': cons}

        def constraint_circle_boundary_right(i):
            def cons(radii):
                x, y = centers_initial[i]
                r = radii[i]
                return 1 - x - r - boundary_pad  # right
            return {'type': 'ineq', 'fun': cons}

        def constraint_circle_boundary_top(i):
            def cons(radii):
                x, y = centers_initial[i]
                r = radii[i]
                return 1 - y - r - boundary_pad  # top
            return {'type': 'ineq', 'fun': cons}

        def constraint_circle_boundary_bottom(i):
            def cons(radii):
                x, y = centers_initial[i]
                r = radii[i]
                return y - r - boundary_pad  # bottom
            return {'type': 'ineq', 'fun': cons}

        def constraint_circle_overlap(i, j):
            def cons(radii):
                r_i = radii[i]
                r_j = radii[j]
                d = dist(centers_initial[i], centers_initial[j])
                return d - r_i - r_j - 1e-8  # non-overlap constraint
            return {'type': 'ineq', 'fun': cons}

        # Build all constraints
        cons = []
        for i in range(n):
            cons.append(constraint_circle_boundary(i))
            cons.append(constraint_circle_boundary_right(i))
            cons.append(constraint_circle_boundary_top(i))
            cons.append(constraint_circle_boundary_bottom(i))
        for i in range(n):
            for j in range(i + 1, n):
                cons.append(constraint_circle_overlap(i, j))

        # Run the optimization
        try:
            result = minimize(
                objective,
                radii_initial,
                method='SLSQP',
                bounds=[(0, max_radius) for _ in range(n)],
                constraints=cons,
                options={
                    'ftol': 1e-8,
                    'maxiter': 300,
                    'disp': False
                }
            )
            if result.success:
                return result.x, centers_initial
            else:
                raise ValueError("Optimization failed")
        except Exception as e:
            raise e

    # Create the initial configuration using staggered grid
    initial_centers = staggered_grid(num_circles)

    # Optimize
    best_radii, best_centers = optimize_packing(initial_centers)

    # Final answer
    return best_centers, best_radii, np.sum(best_radii)
