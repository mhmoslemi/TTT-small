import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Define a function to generate initial configurations
    def generate_initial_centers(n: int) -> np.ndarray:
        # Create a low-density grid of points
        grid_x = np.linspace(0, 1, int(np.sqrt(n)) + 1)
        grid_y = np.linspace(0, 1, int(np.sqrt(n)) + 1)
        grid_points = np.array([grid_x, grid_y]).T

        # Apply k-means clustering to find optimal points
        if n < len(grid_points):
            return grid_points[:n]
        else:
            kmeans_centers, _ = kmeans(grid_points, n)
            return kmeans_centers

    # Generate a small set of candidate configurations to explore
    initial_configs = []

    # Uniform grid
    initial_configs.append(generate_initial_centers(num_circles))

    def random_perturbation(n: int) -> np.ndarray:
        return np.random.rand(n, 2) * 1.0

    initial_configs.append(random_perturbation(num_circles))

    # Nested packing (nodes arranged in concentric circles)
    def nested_packing(n: int) -> np.ndarray:
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0.1, 0.5, n)
        return np.column_stack((radii * np.cos(angles), radii * np.sin(angles)))

    initial_configs.append(nested_packing(num_circles))

    # Hexagonal grid with spacing
    def hexagonal_grid(n: int) -> np.ndarray:
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        x, y = np.meshgrid(np.linspace(0, 1, cols), np.linspace(0, 1, rows))
        grid = np.column_stack((x.flatten(), y.flatten()))
        return grid

    initial_configs.append(hexagonal_grid(num_circles))

    best_centers, best_radii, best_sum = None, None, -1

    for centers_initial in initial_configs:
        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]

        # Define initial radii estimates (small enough to not overlap)
        radii_initial = np.full(num_circles, 0.03)

        # Define the objective function
        def objective(x):
            return -np.sum(x)  # We minimize the negative sum to maximize the sum

        # Define constraints for circle-circle and circle-boundary
        constraints = []

        # Circle-circle distance constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                def dist_constraint(x, i=i, j=j):
                    dx = centers_initial[i, 0] - centers_initial[j, 0]
                    dy = centers_initial[i, 1] - centers_initial[j, 1]
                    distance = np.sqrt(dx * dx + dy * dy)
                    return distance - (x[i] + x[j])
                constraints.append({'type': 'ineq', 'fun': dist_constraint})

        # Boundary constraints
        for i in range(num_circles):
            def boundary_constraint(x, i=i):
                x_c, y_c = centers_initial[i]
                r = x[i]
                return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])
            constraints.append({'type': 'ineq', 'fun': boundary_constraint})

        # Initial guess for radii
        x0 = radii_initial

        # Bounds for radii
        bounds = [(0, 1) for _ in range(num_circles)]

        # Run the optimization with SLSQP
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',

                    'maxiter': 1000,

                    'eps': 1e-8,

                }
            )

                    best_centers, best_radii, best_sum = final_centers, final_radii, sum_radii

        except Exception:
            pass

    return (best_centers, best_radii, best_sum)
