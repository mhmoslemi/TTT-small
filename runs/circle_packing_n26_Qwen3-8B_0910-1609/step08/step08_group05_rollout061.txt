from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

        grid = np.linspace(0, 1, grid_resolution)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, n)[0]
        return np.clip(initial_centers, 0, 1)

    def generate_hexagonal_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                grid.append([x, y])
        return np.array(grid)

        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0, 1, n)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return np.clip(points, 0, 1)

        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5
                centers.append([x, y])
        while len(centers) < n:
            pt = np.random.rand(2)
            if np.all(np.linalg.norm(centers - pt, axis=1) > 0.1):
                centers.append(pt)
        return np.clip(np.array(centers), 0, 1)

    def generate_random_points(n):
        points = np.random.rand(n, 2)
        return np.clip(points, 0, 1)

    initial_configs = [
        generate_rectangular_grid(num_circles),

    best_centers, best_radii, best_sum = None, None, -1

                    return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - (x[i] + x[j])
                constraints.append({'type': 'ineq', 'fun': dist_func})
