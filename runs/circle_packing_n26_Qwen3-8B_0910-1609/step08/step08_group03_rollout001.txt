def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    initial_configs = []

    def hexagonal_grid(num_circles):
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                grid.append([x, y])
        return np.array(grid)
