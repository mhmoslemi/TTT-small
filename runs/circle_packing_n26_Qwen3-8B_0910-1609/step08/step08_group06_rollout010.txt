from scipy.optimize import minimize
from scipy.cluster.vq import kmeans

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    initial_centers_1 = kmeans(grid_points, num_circles)[0]
    initial_configs.append(initial_centers_1)

    # Hexagonal grid configuration
    def hexagonal_grid(num_points):
        angles = np.linspace(0, 2 * np.pi, num_points, endpoint=False)

        r = np.linspace(0.05, 0.1, num_points)

        y = r * np.sin(angles)
        return np.column_stack((x, y))
    initial_centers_2 = hexagonal_grid(num_circles)
    initial_configs.append(initial_centers_2)

    # Nested configuration
    def nested_packing(num_circles):
        num_levels = int(np.sqrt(num_circles))
        centers = []
        for i in range(num_levels):
            for j in range(num_levels):
                x = (i + 0.5) / num_levels
                y = (j + 0.5) / num_levels
                centers.append([x, y])
        if len(centers) < num_circles:
            for _ in range(num_circles - len(centers)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers - pt, axis=1) > 1e-6):
                        centers.append(pt)
                        break
        return np.array(centers)
    initial_centers_3 = nested_packing(num_circles)
    initial_configs.append(initial_centers_3)

    # Random configuration
    def random_packing(num_circles):
        centers = []
        for _ in range(num_circles):
            while True:
                pt = np.random.rand(2)
                if not centers or np.all(np.linalg.norm(centers - pt, axis=1) > 1e-6):
                    centers.append(pt)
                    break
        return np.array(centers)

                x = (i + 0.5) / num_levels

                        break
        return np.array(centers)

        return -np.sum(x)  # Minimize negative of sum to maximize

    def constraint_circle_circle(i, j):

            return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (x[i] + x[j])

            x_c, y_c = centers[i]
            r = x[i]
            return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])
