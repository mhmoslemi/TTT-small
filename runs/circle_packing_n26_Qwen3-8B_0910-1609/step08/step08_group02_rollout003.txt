from scipy.optimize import minimize
from scipy.spatial import ConvexHull

def run_packing():

        spacing = 0.25
        x_coords = np.arange(0.1, 1, spacing)
        y_coords = np.arange(0.1, 1, spacing)
        grid = np.array([[x, y] for y in y_coords for x in x_coords])
        # Trim to 26 circles
        grid = grid[:26]
        return grid

    def get_convex_hull(centers):
        # Use convex hull to identify clusters and suggest tighter packing
        hull = ConvexHull(centers)
        # Return the convex hull's vertices as a refined cluster
        return hull.vertices

    def objective(radii):
        return -np.sum(radii)  # We want to maximize the sum of radii

    def constraint_circle_bound(x, i, centers):
        x_coord, y_coord = centers[i]
        r = x[i]
        return x_coord - r >= -1e-12 and y_coord - r >= -1e-12 and x_coord + r <= 1 + 1e-12 and y_coord + r <= 1 + 1e-12

    def constraint_circle_spacing(x, i, j, centers):
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - (x[i] + x[j]) >= -1e-12

    centers = generate_hex_grid()
    num_circles = centers.shape[0]

    # Initial guess: uniform radii based on minimum distance to neighbors
    initial_radii = np.full(num_circles, 0.05)

    bounds = [(0, 1) for _ in range(num_circles)]

    # Create constraints
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({

                'fun': lambda x, i=i, j=j: constraint_circle_spacing(x, i, j, centers)
            })

        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: constraint_circle_bound(x, i, centers)
        })

    # Optimize
    result = minimize(

        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',

    )

    final_radii = result.x
