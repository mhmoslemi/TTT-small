def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                if cols > 1 or rows > 1:
                    grid.append([x, y])

        return points
