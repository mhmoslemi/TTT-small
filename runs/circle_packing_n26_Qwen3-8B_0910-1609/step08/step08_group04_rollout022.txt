from scipy.spatial.distance import cdist

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

        """Generate a hexagonal grid with n points."""
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                if cols > 1 or rows > 1:
                    grid.append([x, y])
        return np.array(grid)

    def generate_spiral(n):
        """Generate a spiral pattern with n points."""
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0, 1, n)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

        """Generate initial points using KMeans."""
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)

        return initial_centers
