from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate initial configurations
    def generate_hex_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        spacing = 0.5 / (cols - 1) if cols > 1 else 1.0
        for i in range(rows):
            for j in range(cols):
                x = j * spacing
                y = i * spacing
                if cols > 1 or rows > 1:
                    grid.append([x, y])
        return np.array(grid)

    def generate_spiral(n):
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0, 0.8, n)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_kmeans(n, seed=None):
        np.random.seed(seed)
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, n, min_iter=100, max_iter=500)[0]
        return initial_centers

    def generate_nested(n):
        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5
                centers.append([x, y])
        while len(centers) < n:
            pt = np.random.rand(2)
            if np.all(np.linalg.norm(centers - pt, axis=1) > 0.1):
                centers.append(pt)
        return np.array(centers)

    initial_configs = [
        generate_hex_grid(num_circles),
        generate_spiral(num_circles),
        generate_kmeans(num_circles, seed=42),
        generate_nested(num_circles),
    ]

    best_centers, best_radii, best_sum = None, None, -1

    for centers_initial in initial_configs:
        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]
        elif len(centers_initial) < num_circles:
            add_centers = []
            for _ in range(num_circles - len(centers_initial)):
                pt = np.random.rand(2)
                if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 0.1):
                    add_centers.append(pt)
            centers_initial = np.vstack((centers_initial, np.array(add_centers)))

        # Initial radii based on distance to neighbors and boundary
        dists = cdist(centers_initial, centers_initial)
        diag = np.diag(dists)
        dists[range(dists.shape[0]), range(dists.shape[0])] = np.inf
        avg_dist = np.mean(dists[dists < np.inf])
        radii_initial = np.clip(avg_dist / 2.5, 0.01, 0.2)

        # Objective function to maximize total radius
        def objective(x):
            return -np.sum(x)

        # Constraints: non-overlapping and inside unit square
        def circle_circle_constraint(i, j):
            def cons(x):
                dx = centers_initial[i, 0] - centers_initial[j, 0]
                dy = centers_initial[i, 1] - centers_initial[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                return dist - (x[i] + x[j])
            return cons

        def boundary_constraint(i):
            def cons(x):
                x_i = centers_initial[i, 0]
                y_i = centers_initial[i, 1]
                r = x[i]
                return np.min([x_i - r, 1 - x_i - r, y_i - r, 1 - y_i - r])
            return cons

        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append({'type': 'ineq', 'fun': circle_circle_constraint(i, j)})

            constraints.append({'type': 'ineq', 'fun': boundary_constraint(i)})

        bounds = [(0, 1) for _ in range(num_circles)]
        x0 = radii_initial

        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={
                    'maxiter': 5000,
                    'ftol': 1e-10,
                    'eps': 1e-8,
                    'disp': False
