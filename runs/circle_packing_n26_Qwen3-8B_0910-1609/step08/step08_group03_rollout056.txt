def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0
    best_centers = None
    best_radii = None

    initial_configs = []

    # Hexagonal packing
    def hexagonal_grid(num_circles):
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                grid.append([x, y])
        return np.array(grid)

    initial_configs.append(initial_centers_1)

    def spiral_packing(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        radii = np.linspace(0, 1, num_circles)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    initial_centers_2 = spiral_packing(num_circles)
    initial_configs.append(initial_centers_2)

    # Nested packing

        return np.array(centers)

    initial_configs.append(initial_centers_5)

    for centers_initial in initial_configs:

        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]
        elif len(centers_initial) < num_circles:
            new_centers = []
            for _ in range(num_circles - len(centers_initial)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 0.1):

            centers_initial = np.vstack((centers_initial, new_centers))

        radii_initial = np.full(num_circles, 0.05)
        for i in range(num_circles):
            x, y = centers_initial[i]
