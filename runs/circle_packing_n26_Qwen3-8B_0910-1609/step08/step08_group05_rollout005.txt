from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        grid = np.linspace(0, 1, grid_resolution)

        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        kmeans_result = kmeans(grid_points, n)
        return kmeans_result[0]

        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        for i in range(rows):

                x = j / (cols - 1)
                y = i / (rows - 1)
                grid.append([x, y])
        return np.array(grid)

        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

        corner_points = [
            [0, 0], [1, 0], [1, 1], [0, 1]

            return unique_corner_points[:n]

            grid = np.vstack((unique_corner_points, np.random.rand(remaining, 2)))
            return grid

    initial_configs = [
