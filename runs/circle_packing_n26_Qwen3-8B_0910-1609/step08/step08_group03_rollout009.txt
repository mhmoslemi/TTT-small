import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate a grid of candidate initial positions
    grid_x = np.linspace(0, 1, 100)
    grid_y = np.linspace(0, 1, 100)
    grid_points = np.array([grid_x, grid_y]).T

    # Generate initial configurations
    initial_configs = []

    # Hexagonal grid
    def hexagonal_grid(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                grid.append([x, y])
        return np.array(grid)

    initial_configs.append(hexagonal_grid(num_circles))

    # Spiral packing
    def spiral_packing(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        radii = np.linspace(0, 1, num_circles)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    initial_configs.append(spiral_packing(num_circles))

    # K-means grid
    initial_configs.append(kmeans(grid_points, num_circles)[0])

    # Corner-focused grid
    def corner_grid(num_circles):
        def get_corners():
            return np.array([
                [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]
            ])

        corners = get_corners()
        points = []
        for pt in corners:
            for _ in range(num_circles // len(corners)):
                points.append(pt)
        if num_circles % len(corners) != 0:
            for _ in range(num_circles % len(corners)):
                points.append(np.random.rand(2))
        return np.array(points)

    initial_configs.append(corner_grid(num_circles))

    # Hybrid grid (hexagonal + random)
    def hybrid_grid(num_circles):
        hex_grid = hexagonal_grid(num_circles)
        hex_grid = np.vstack((hex_grid, np.random.rand(num_circles, 2)))
        return hex_grid[:num_circles]

    initial_configs.append(hybrid_grid(num_circles))

    best_centers, best_radii, best_sum = None, None, -1

    for centers_initial in initial_configs:
        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]
        elif len(centers_initial) < num_circles:
            new_centers = []
            for _ in range(num_circles - len(centers_initial)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 0.1):
                        new_centers.append(pt)
                        break
            centers_initial = np.vstack((centers_initial, new_centers))

        # Define initial radii estimates
        radii_initial = np.full(num_circles, 0.005)

        # Define the objective function
        def objective(x):
            return -np.sum(x)  # We minimize the negative sum to maximize the sum

        # Define constraints for circle-circle and circle-boundary
        constraints = []

        # Circle-circle distance constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                def dist_constraint(x, i=i, j=j):
                    dx = centers_initial[i, 0] - centers_initial[j, 0]
                    dy = centers_initial[i, 1] - centers_initial[j, 1]
                    distance = np.sqrt(dx * dx + dy * dy)
                    return distance - (x[i] + x[j])
                constraints.append({'type': 'ineq', 'fun': dist_constraint})

        # Boundary constraints
        for i in range(num_circles):
            def boundary_constraint(x, i=i):
                x_c, y_c = centers_initial[i]
                r = x[i]
                return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])
            constraints.append({'type': 'ineq', 'fun': boundary_constraint})

        # Initial guess for radii
        x0 = radii_initial

        bounds = [(0, 1) for _ in range(num_circles)]

        # Run the optimization with SLSQP
        try:

            pass
