import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate a variety of initial configurations
    def generate_hex_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                if cols > 1 or rows > 1:
                    grid.append([x, y])
        return np.array(grid)

    def generate_spiral(n):
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0, 1, n)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_kmeans(n):
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, n)[0]
        return initial_centers

    def generate_nested(n):
        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5
                centers.append([x, y])
        while len(centers) < n:
            pt = np.random.rand(2)
            if np.all(np.linalg.norm(centers - pt, axis=1) > 0.1):
                centers.append(pt)
        return np.array(centers)

    # Adaptive optimization parameters
    def objective(x):
        return -np.sum(x)

    def circle_circle_constraint(i, j):
        def cons(x):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - (x[i] + x[j])
        return cons

    def boundary_constraint(i):
        def cons(x):
            x_i = centers[i, 0]
            y_i = centers[i, 1]
            r = x[i]
            return np.min([x_i - r, y_i - r, 1 - x_i - r, 1 - y_i - r])
        return cons

    best_centers, best_radii, best_sum = None, None, -1

    # Combine multiple initial configurations
    initial_configs = [
        generate_hex_grid(num_circles),
        generate_spiral(num_circles),
        generate_kmeans(num_circles),
        generate_nested(num_circles),
    ]

        if len(centers_initial) > num_circles:
            centers_initial = centers_initial[:num_circles]
        elif len(centers_initial) < num_circles:
            add_centers = []
            for _ in range(num_circles - len(centers_initial)):

                if np.all(np.linalg.norm(centers_initial - pt, axis=1) > 0.1):
                    add_centers.append(pt)
            centers_initial = np.vstack((centers_initial, np.array(add_centers)))

        # Initial radii based on nearest neighbor spacing
        dist_matrix = cdist(centers_initial, centers_initial)
        np.fill_diagonal(dist_matrix, np.inf)
        min_distances = np.min(dist_matrix, axis=1)
        radii_initial = np.clip(min_distances / np.sqrt(2), 0, 0.3)

        # Optimization with adaptive bounds
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append({'type': 'ineq', 'fun': circle_circle_constraint(i, j)})
            constraints.append({'type': 'ineq', 'fun': boundary_constraint(i)})

        bounds = [(0, 1) for _ in range(num_circles)]
        x0 = radii_initial

        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraints,

                    'disp': False
                }

            pass
