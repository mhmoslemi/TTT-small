import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    num_circles = 26

    # Define optimal seed positions (hexagonal cluster with edge-aligned support)
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    def objective(radii):
        return -np.sum(radii)

    def constraint_non_overlap(radii):
        dists = cdist(centers, centers)
        return np.array([dists[i, j] - (radii[i] + radii[j]) for i in range(num_circles) for j in range(i + 1, num_circles)])

    def constraint_boundary(radii):
        boundary_distances = np.array([
            centers[i, 0] - radii[i],
            centers[i, 1] - radii[i],
            1 - centers[i, 0] - radii[i],
            1 - centers[i, 1] - radii[i]
        ] for i in range(num_circles))
        return np.min(boundary_distances, axis=1)

    bounds = [(0, 1) for _ in range(num_circles)]

    initial_guess = np.array([0.02] * num_circles)

    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_non_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP',
        options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False}
    )

    final_centers = centers
    final_radii = result.x

    return (final_centers, final_radii, np.sum(final_radii))
