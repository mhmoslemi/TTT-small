from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    # First, create a partial grid of 6 columns and 5 rows to fit 26 circles
    x = np.linspace(0.1, 0.9, 6)

                offset = 0

                offset = 0.5 * (x[1] - x[0])
            centers.append([x[j] + offset, y[i]])
    centers = np.array(centers)

    # Starting radii: small values to allow optimization
    initial_radii = np.array([0.05] * num_circles)

    # Optimization objective: Maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Non-overlapping constraint using distance between circle centers
    def constraint_func(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraint = dist - (radii[i] + radii[j])
                constraints.append(constraint)
        return np.array(constraints)

    # Boundary constraint: circles must stay within [0,1]x[0,1]
    def boundary_func(radii):
        constraints = []
        for i in range(num_circles):
            x, y = centers[i]
            left = x - radii[i]
            right = 1 - x - radii[i]
            bottom = y - radii[i]
            top = 1 - y - radii[i]
            constraint = np.min([left, right, bottom, top])
            constraints.append(constraint)
        return np.array(constraints)

    # Define bounds for all radii
    bounds = Bounds(np.zeros(num_circles), np.ones(num_circles))

    # Optimization setup
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_func},
            {'type': 'ineq', 'fun': boundary_func}
        ],
        method='SLSQP',
        options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
    )

    # Extract final result
    final_centers = centers
    final_radii = result.x

    def validate_packing(centers, radii):

            return False, "NaN values present"

                return False, f"Circle {i} has negative radius {radii[i]}"

            if (x - r < -1e-12 or x + r > 1 + 1e-12

                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    if not is_valid:
        raise ValueError("Invalid packing detected: " + msg)

    sum_radii = np.sum(final_radii)
    return (final_centers, final_radii, sum_radii)
