import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from itertools import combinations
import random

def run_packing():
    num_circles = 26
    # Create initial centers in a more optimized layout, avoiding fixed grids
    # Generate random centers in the unit square to start
    centers = np.random.rand(num_circles, 2)
    
    # Random initial radii
    initial_radii = np.random.rand(num_circles)
    
    # Objective function: Maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraint functions
    def constraint_func(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraint = dist - (radii[i] + radii[j])
                constraints.append(constraint)
        return np.array(constraints)
    
    def boundary_func(radii):
        constraints = []
        for i in range(num_circles):
            x, y = centers[i]
            left = x - radii[i]
            right = 1 - x - radii[i]
            bottom = y - radii[i]
            top = 1 - y - radii[i]
            constraint = np.min([left, right, bottom, top])
            constraints.append(constraint)
        return np.array(constraints)
    
    # Define bounds for all radii
    bounds = Bounds(np.zeros(num_circles), np.ones(num_circles))
    
    # Optimization setup
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_func},
            {'type': 'ineq', 'fun': boundary_func}
        ],
        method='SLSQP',
        options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
    )
    
    # Extract final result
    final_centers = centers
    final_radii = result.x

    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"
    
    is_valid, msg = validate_packing(final_centers, final_radii)
    if not is_valid:
        raise ValueError("Invalid packing detected: " + msg)

    sum_radii = np.sum(final_radii)
    return (final_centers, final_radii, sum_radii)
