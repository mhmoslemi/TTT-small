                grid.append([x, y])

        return np.array(grid)

        for _ in range(num_circles):

        return np.array(radii)

    configurations = [
        create_hexagonal_grid()[:26],
        create_boundary_grid()[:26],
        create_nested_grid()[:26]
    ]

    for centers in configurations:
        if len(centers) != num_circles:
            continue

        # Initial guess for radii
        initial_radii = np.random.uniform(0.02, 0.1, num_circles)

        def objective(x):
            return -np.sum(x)

            return dist - (x[i] + x[j])

            return np.min([x_coord - r, y_coord - r, 1 - x_coord - r, 1 - y_coord - r])

        constraints = []

                'type': 'ineq',

            })

    return (best_centers, best_radii, best_sum)
