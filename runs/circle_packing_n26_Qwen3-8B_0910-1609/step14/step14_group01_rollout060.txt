import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import math

def run_packing():
    num_circles = 26
    
    # Define a more optimized initial configuration
    # Create a hybrid packing: 4 rows of 6, and 1 row of 2 (to fill the gaps)
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])[:26]

    # Define bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x)

    # Constraint for non-overlapping circles
    def constraint(i, j):
        def cons(x):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (x[i] + x[j])
        return cons

    # Constraint for circles not exceeding the unit square
    def boundary(i):
        def cons(x):
            x_coord, y_coord = centers[i]
            r = x[i]
            return np.min([x_coord - r, y_coord - r, 1 - x_coord - r, 1 - y_coord - r])
        return cons

    # Initial guess for radii
    initial_radii = np.array([0.02] * num_circles)

    # Create constraints and constraints dictionary
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': constraint(i, j)
            })

    for i in range(num_circles):
        constraints.append({
            'type': 'ineq',
            'fun': boundary(i)
        })

    # Run the optimizer with better settings
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False, 'eps': 1e-8}
    )

    # Validate the result
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    is_valid, msg = validate_packing(centers, result.x)
    if not is_valid:
        raise ValueError("Validation failed")

    sum_radii = np.sum(result.x)
    return (centers, result.x, sum_radii)
