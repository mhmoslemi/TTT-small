    ])[:26]

    def objective(radii):
        return -np.sum(radii)

    def constraint_circle_circle(radii, idx, jdx, centers):
        dx = centers[idx, 0] - centers[jdx, 0]
        dy = centers[idx, 1] - centers[jdx, 1]
        dist = np.sqrt(dx*dx + dy*dy)
        return dist - (radii[idx] + radii[jdx])

        dist_to_left = x - r
        dist_to_right = 1 - x - r

        dist_to_top = 1 - y - r
        return np.min([dist_to_left, dist_to_right, dist_to_bottom, dist_to_top])
