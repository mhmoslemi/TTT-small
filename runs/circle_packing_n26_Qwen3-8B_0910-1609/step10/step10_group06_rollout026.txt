import cvxpy as cp

def run_packing():
    num_circles = 26
    # Initial placement based on a hexagonal grid
    # Generate 26 points in a hexagonal pattern with some variation
    centers = []
    x = np.linspace(0.1, 0.9, 6)  # X positions
    y = np.linspace(0.1, 0.9, 6)  # Y positions
    for i in range(6):
        for j in range(6):
            if i % 2 == 0:
                x_pos = x[j]
            else:
                x_pos = x[j] + 0.25
            if i <= 2:
                y_pos = y[i]
            else:
                y_pos = y[i - 3]
            centers.append([x_pos, y_pos])
    centers = np.array(centers)[:26]

    # Define radius variables
    radii = cp.Variable(num_circles)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraints: each circle must be inside the square
    for i in range(num_circles):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-8)
        constraints.append(x + radii[i] <= 1 + 1e-8)
        constraints.append(y - radii[i] >= -1e-8)
        constraints.append(y + radii[i] <= 1 + 1e-8)

    # Non-overlapping constraints: distance between centers >= sum of radii

        for j in range(i + 1, num_circles):
            dist = cp.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Non-negative radii
    constraints.append(radii >= 0)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, max_iter=500, eps=1e-8)

    # Return results
