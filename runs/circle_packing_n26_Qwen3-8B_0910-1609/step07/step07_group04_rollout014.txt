def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    def hexagonal_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []

            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0"/2"

                grid.append([x, y])
