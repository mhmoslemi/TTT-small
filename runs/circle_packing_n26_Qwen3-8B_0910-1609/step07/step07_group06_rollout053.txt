        rows = int(np.sqrt(num_circles))

        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                if rows > 1 and cols > 1:
                    grid.append([x, y])
        return np.array(grid)

        return points
