from scipy.cluster.vq import kmeans

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    def generate_hexagonal_grid(n_points):
        grid_heuristic = np.linspace(0, 1, n_points)
        grid_x, grid_y = np.meshgrid(grid_heuristic, grid_heuristic)
        grid_points_hex = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        return grid_points_hex

        return kmeans(grid_points, n_clusters)[0]

                    break
        return np.vstack((initial_grid, added_points))
