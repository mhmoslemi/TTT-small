from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def hexagonal_initial(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                centers.append([x, y])
        return np.array(centers)

    def spiral_initial(n):
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        radii = np.linspace(0, 1, n)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def boundary_initial(n):
        points = []
        for _ in range(n):
            while True:
                x = np.random.uniform(0, 1)
                y = np.random.uniform(0, 1)
                if (x < 0.1 or x > 0.9 or y < 0.1 or y > 0.9) or (np.random.rand() < 0.5):
                    points.append([x, y])
                    break
        return np.array(points)

        cluster_centers = kmeans(data, num_clusters)[0]
        return cluster_centers

    initial_configs = []

    initial_centers = kmeans_with_constraints(grid_points, num_circles)
    initial_configs.append(initial_centers)

    best_centers, best_radii, best_sum = None, None, -1

        if len(centers_initial) > num_circles:
