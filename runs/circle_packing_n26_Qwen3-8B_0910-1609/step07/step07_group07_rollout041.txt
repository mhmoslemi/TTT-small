from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    best_radii = None
    best_sum = -1

    # Generate multiple initial configurations
    initial_configs = []

    # Configuration 1: Hexagonal grid
    def hexagonal_grid(n):
        grid_points = []
        for i in range(int(np.sqrt(n)) + 2):
            for j in range(i + 1):
                x = (i - 0.5) / (np.sqrt(n) + 1)
                y = (j - 0.5) / (np.sqrt(n) + 1)
                grid_points.append([x, y])
        if len(grid_points) < n:
            for _ in range(n - len(grid_points)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(grid_points - pt, axis=1) > 1e-6):
                        grid_points.append(pt)
                        break
        return np.array(grid_points)

    initial_centers_1 = hexagonal_grid(num_circles)
    initial_configs.append(initial_centers_1)

    # Configuration 2: Square grid
    def square_grid(n):
        grid_points = []
        for i in range(int(np.sqrt(n)) + 2):
            for j in range(i + 1):
                x = (i - 0.5) / (np.sqrt(n) + 1)
                y = (j - 0.5) / (np.sqrt(n) + 1)
                grid_points.append([x, y])

                angle = 2 * np.pi * i / 12

                y = r * np.sin(angle)
                grid_points.append([x, y])
        if len(grid_points) < n:
            for _ in range(n - len(grid_points)):

                    pt = np.random.rand(2) * 1.5  # slightly expand for pushing

                    break
