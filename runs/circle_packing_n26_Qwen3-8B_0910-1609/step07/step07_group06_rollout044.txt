from scipy.cluster.vq import kmeans

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def generate_hexagonal_grid(num_circles):
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                if rows > 1 and cols > 1:
                    grid.append([x, y])
        return np.array(grid)

    def generate_spiral_packing(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        radii = np.linspace(0, 1, num_circles)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_kmeans_grid(num_circles):
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, num_circles)[0]
        return initial_centers

    def generate_nested_packing(num_circles):
        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5
                centers.append([x, y])
        while len(centers) < num_circles:
            pt = np.random.rand(2)
            if np.all(np.linalg.norm(centers - pt, axis=1) > 0.1):
                centers.append(pt)
        return np.array(centers)

    def generate_initial_centers():
        initial_centers = [
            generate_hexagonal_grid(num_circles),
            generate_spiral_packing(num_circles),
            generate_kmeans_grid(num_circles),
            generate_nested_packing(num_circles),
        ]
        initial_centers = [np.vstack((c[:num_circles] if len(c) > num_circles else c)) for c in initial_centers]
        return initial_centers

    def constraint_func(x, i, j, centers):
        dist = np.linalg.norm(centers[i] - centers[j])
        return dist - x[i] - x[j]

    def boundary_func(x, i, centers):
        r = x[i]
        x_i, y_i = centers[i]
        return np.min([x_i - r, y_i - r, 1 - x_i - r, 1 - y_i - r])

    best_centers = None
    best_radii = None
    best_sum = -1

        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j, c=centers_initial: constraint_func(x, i, j, c)})

        for i in range(num_circles):
