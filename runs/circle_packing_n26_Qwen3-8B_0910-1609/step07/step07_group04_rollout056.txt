from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def hexagonal_grid(num_circles):
        row = int(np.sqrt(num_circles))
        col = num_circles // row + (1 if num_circles % row else 0)
        grid = []
        for i in range(row):
            for j in range(col):
                x = j / (col - 1) if col > 1 else 0.5
                y = i / (row - 1) if row > 1 else 0.5
                grid.append([x, y])

        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)

        return points
