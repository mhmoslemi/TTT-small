from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    best_centers = None
    best_radii = None
    best_sum = -1

    def generate_hexagonal_grid():
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                if rows > 1 and cols > 1:
                    grid.append([x, y])
        return np.array(grid)

        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)

        return initial_centers

                centers.append(pt)
        return np.array(centers)

            return lambda x: np.sqrt((centers_initial[i, 0] - centers_initial[j, 0])**2 + (centers_initial[i, 1] - centers_initial[j, 1])**2) - (x[i] + x[j])

        for i in range(num_circles):
