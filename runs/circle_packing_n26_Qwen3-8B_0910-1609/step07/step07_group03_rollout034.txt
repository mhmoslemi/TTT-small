import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    best_centers, best_radii, best_sum = None, None, -1

    def generate_initial_centers(config_type):
        if config_type == 'grid':
            grid = np.linspace(0, 1, 200)
            grid_x, grid_y = np.meshgrid(grid, grid)
            grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
            return kmeans(grid_points, num_circles)[0]
        elif config_type == 'hex':
            grid = np.linspace(0, 1, 200)
            grid_x, grid_y = np.meshgrid(grid, grid)
            grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
            return kmeans(grid_points, num_circles)[0]
        elif config_type == 'nested':
            num_levels = int(np.sqrt(num_circles))
            centers = []
            for i in range(num_levels):
                for j in range(num_levels):
                    x = (i + 0.5) / num_levels
                    y = (j + 0.5) / num_levels
                    centers.append([x, y])
            if len(centers) < num_circles:
                while len(centers) < num_circles:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(np.array(centers) - pt, axis=1) > 1e-6):
                        centers.append(pt)
            return np.array(centers)
        elif config_type == 'random':
            centers = []
            for _ in range(num_circles):
                while True:
                    pt = np.random.rand(2)
                    if not centers or np.all(np.linalg.norm(np.array(centers) - pt, axis=1) > 1e-6):
                        centers.append(pt)
                        break
            return np.array(centers)
        else:
            raise ValueError("Unknown config type")

    initial_configs = [
        generate_initial_centers('grid'),
        generate_initial_centers('hex'),
        generate_initial_centers('nested'),
        generate_initial_centers('random')
    ]

    def objective(x):
        return -np.sum(x)

    def constraint_circle_circle(centers, x):
        dists = np.zeros((num_circles, num_circles))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dists[i, j] = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dists - (x[:, np.newaxis] + x[np.newaxis, :]) + 1e-12

    def constraint_circle_boundary(centers, x):
        return np.minimum(centers[:, 0] - x, centers[:, 1] - x, 1 - centers[:, 0] - x, 1 - centers[:, 1] - x) + 1e-12

    for config in initial_configs:
        if len(config) > num_circles:
            config = config[:num_circles]
        elif len(config) < num_circles:
            while len(config) < num_circles:
                pt = np.random.rand(2)
                if np.all(np.linalg.norm(config - pt, axis=1) > 1e-6):
                    config = np.vstack((config, pt))
        centers = config

        def get_constraints(x):
            return np.concatenate((constraint_circle_circle(centers, x), constraint_circle_boundary(centers, x)))

        bounds = [(0, 1) for _ in range(num_circles)]
        x0 = np.full(num_circles, 0.01)
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': get_constraints},
                method='SLSQP',
                options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                sum_radii = np.sum(result.x)
                if sum_radii > best_sum:
                    best_centers = centers
                    best_radii = result.x
                    best_sum = sum_radii
        except:
            pass

    return best_centers, best_radii, best_sum
