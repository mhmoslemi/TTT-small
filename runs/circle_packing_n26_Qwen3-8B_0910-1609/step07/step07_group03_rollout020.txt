import numpy as np
from scipy.optimize import minimize
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import pdist, squareform
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    # Generate hexagonal grid points within the square
    def generate_hexagonal_grid(n):
        grid = []
        for i in range(n):
            for j in range(n):
                x = i / (n - 1)
                y = j / (n - 1)
                if (i + j) % 2 == 0:
                    x = (i + 0.5) / (n - 1)
                    y = j / (n - 1)
                grid.append([x, y])
        return np.array(grid)

    hex_grid = generate_hexagonal_grid(100)
    initial_centers = kmeans(hex_grid, num_circles)[0]
    
    # Ensure centers are within [0, 1]^2
    initial_centers = np.clip(initial_centers, 0, 1)
    
    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Define constraints
    constraints = []

    def circle_boundary_constraint(radii, i):
        x, y = initial_centers[i]
        return np.min([x - radii[i], y - radii[i], 1 - x - radii[i], 1 - y - radii[i]])

    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: circle_boundary_constraint(r, i)})

    def circle_circle_constraint(radii, i, j):
        dx = initial_centers[i, 0] - initial_centers[j, 0]
        dy = initial_centers[i, 1] - initial_centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - (radii[i] + radii[j])

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: circle_circle_constraint(r, i, j)})

    # Initial guess for radii
    x0 = np.full(num_circles, 0.01)

    # Bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Run the optimization with SLSQP
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={
            'maxiter': 1000,
            'ftol': 1e-10,
            'eps': 1e-8,
            'disp': False
        }
    )

    if result.success:
        best_centers = initial_centers
        best_radii = result.x
        best_sum = np.sum(best_radii)
    else:
        raise RuntimeError("Optimization failed with message: " + result.message)

    return (best_centers, best_radii, best_sum)
