import numpy as np
from scipy.optimize import differential_evolution
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate initial centers using k-means clustering
    def generate_centers(n):
        points = np.random.rand(n, 2)
        centers, _ = kmeans(points, n)
        return centers

    initial_centers = generate_centers(num_circles)

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints for non-overlapping and boundary conditions
    constraints = []

    def circle_boundary(radii, i):
        x, y = initial_centers[i]
        return np.min([x - radii[i], y - radii[i], 1 - x - radii[i], 1 - y - radii[i]])

    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: circle_boundary(r, i)})

    def circle_distance(radii, i, j):
        dx = initial_centers[i, 0] - initial_centers[j, 0]
        dy = initial_centers[i, 1] - initial_centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - (radii[i] + radii[j])

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: circle_distance(r, i, j)})

    # Bounds for radii: [0, 1] for all circles
    bounds = [(0, 1) for _ in range(num_circles)]

    # Run differential evolution optimization
    result = differential_evolution(
        objective,
        bounds,
        constraints=constraints,
        strategy='best1bin',
        popsize=15,
        maxiter=500,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.9,
        polish=True
    )

    final_radii = result.x
    final_centers = initial_centers

    # Calculate the sum of radii
    sum_radii = np.sum(final_radii)

    return (final_centers, final_radii, sum_radii)
