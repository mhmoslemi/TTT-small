def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    pop_size = 100
    generations = 100
    cx_prob = 0.8
    mut_prob = 0.2

    # Initial configurations - three different packing strategies
    def hexagonal_grid(num_circles):
        grid_points = []
        spacing = 1 / (np.sqrt(num_circles) + 2)
        for i in range(int(np.sqrt(num_circles)) + 2):
            for j in range(i):
                x = i * spacing
                y = j * spacing
                grid_points.append([x, y])
        if len(grid_points) < num_circles:
            for _ in range(num_circles - len(grid_points)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(grid_points - pt, axis=1) > 1e-6):
                        grid_points.append(pt)
                        break
        return np.array(grid_points)

    def nested_packing(num_circles):
        num_levels = int(np.sqrt(num_circles)) + 1
        centers = []
        for i in range(num_levels):
            for j in range(num_levels):
                x = (i + 0.5) / num_levels
                y = (j + 0.5) / num_levels
                centers.append([x, y])
        if len(centers) < num_circles:
            for _ in range(num_circles - len(centers)):
                while True:
                    pt = np.random.rand(2)
                    if np.all(np.linalg.norm(centers - pt, axis=1) > 1e-6):
                        centers.append(pt)
                        break
        return np.array(centers)

    def random_packing(num_circles):
        centers = []
        for _ in range(num_circles):
            while True:
                pt = np.random.rand(2)
                if not centers or np.all(np.linalg.norm(centers - pt, axis=1) > 1e-6):
                    centers.append(pt)
                    break
        return np.array(centers)

    # Create initial population using all three strategies
    initial_pop = []

    for _ in range(pop_size):
        centers = np.array([random.choice([hexagonal_grid(num_circles), nested_packing(num_circles), random_packing(num_circles)])])
        initial_pop.append(centers)

    def evaluate(individual):
        centers = individual["centers"]
        radii = individual["radii"]
        # Boundary check
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return (float("-inf"),)
        # Circle-circle overlap check
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return (float("-inf"),)
        # If all constraints satisfied, return the sum of radii
        return (np.sum(radii),)

    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    toolbox = base.Toolbox()

    # Define the fitness function
    def fitness_func(individual):
        return evaluate(individual)

    # Define the population and evaluation
    def create_individual():
        centers = np.array([random.choice([hexagonal_grid(num_circles), nested_packing(num_circles), random_packing(num_circles)])])
        radii = np.random.uniform(0.01, 0.1, num_circles)
        return {"centers": centers, "radii": radii}

    def create_population():
        return [create_individual() for _ in range(pop_size)]

    # Register the functions
    toolbox.register("individual", create_individual)

            if config == hexagonal_grid:
                return nested_packing
            elif config == nested_packing:
                return random_packing
            else:
                return hexagonal_grid

    best = max(population, key=lambda ind: evaluate(ind))

    sum_radii = np.sum(best_radii)
    return (best_centers, best_radii, sum_radii)
