def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    max_iter = 1000

    def generate_hex_grid(n):
        grid = []

            for col in range(int(np.sqrt(n)) + 1):
                if row % 2 == 0:
                    x = col / (np.sqrt(n) + 1)
                else:
                    x = (col + 0.5) / (np.sqrt(n) + 1)

                grid.append([x, y])
        return np.array(grid)
