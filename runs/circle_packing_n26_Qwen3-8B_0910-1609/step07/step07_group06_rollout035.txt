import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def generate_hexagonal_grid(num_circles):
        rows = int(np.sqrt(num_circles))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                grid.append([x, y])
        return np.array(grid)

    def generate_spiral(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        radii = np.linspace(0, 1, num_circles)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_kmeans(num_circles):
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, num_circles)[0]
        return initial_centers

    def generate_nested(num_circles):
        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5
                centers.append([x, y])

                centers.append(pt)
        return np.array(centers)

        bounds = [(-1, 1) for _ in range(2 * num_circles)]

        def objective(x):
            return -np.sum(x[num_circles:])

        def circle_circle_constraint(x, i, j):
            dx = x[2 * i] - x[2 * j]
            dy = x[2 * i + 1] - x[2 * j + 1]
            return np.sqrt(dx ** 2 + dy ** 2) - (x[num_circles + i] + x[num_circles + j])

        def boundary_constraint(x, i):
            x_i = x[2 * i]
            y_i = x[2 * i + 1]
            r_i = x[num_circles + i]
            return np.min([x_i - r_i, y_i - r_i, 1 - x_i - r_i, 1 - y_i - r_i])

        constraints = []

            for j in range(i + 1, num_circles):
                constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: circle_circle_constraint(x, i, j)})

            constraints=constraints,

                'disp': False
            }

            radii = x[num_circles:]
            return centers, radii, np.sum(radii)
        else:
            raise Exception(f"Optimization failed: {result.message}")

    return (best_centers, best_radii, best_sum)
