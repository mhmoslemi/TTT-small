    def hexagonal_grid(num_circles):
        row = int(np.sqrt(num_circles))
        col = num_circles // row + (1 if num_circles % row else 0)
        grid = []
        for i in range(row):
            for j in range(col):
                x = j / (col - 1) if col > 1 else 0.5
                y = i / (row - 1) if row > 1 else 0.5
                grid.append([x, y])
        return np.array(grid)

        return points

    def grid_kmeans(num_circles):

        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        centers = kmeans(grid_points, num_circles)[0]
        return centers
