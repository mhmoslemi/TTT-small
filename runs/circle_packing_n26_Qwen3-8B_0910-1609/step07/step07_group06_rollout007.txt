import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    def generate_hexagonal_grid(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                if rows > 1 and cols > 1:
                    grid.append([x, y])
        return np.array(grid)

    def generate_spiral_packing(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
        radii = np.linspace(0, 1, num_circles)
        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_kmeans_grid(num_circles):
        grid = np.linspace(0, 1, 500)
        grid_x, grid_y = np.meshgrid(grid, grid)
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, num_circles)[0]
        return initial_centers

    def generate_nested_packing(num_circles):
        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5
                centers.append([x, y])
        while len(centers) < num_circles:
            pt = np.random.rand(2)
            if np.all(np.linalg.norm(centers - pt, axis=1) > 0.1):
                centers.append(pt)
        return np.array(centers)

    # Hybrid initial configuration
    initial_centers = np.vstack([
        generate_hexagonal_grid(num_circles),
        generate_spiral_packing(num_circles),
        generate_kmeans_grid(num_circles),
        generate_nested_packing(num_circles)
    ])
    initial_centers = np.unique(initial_centers, axis=0)
    initial_centers = initial_centers[:num_circles]

    # Ensure exact number of circles
    if len(initial_centers) < num_circles:
        add_centers = []
        for _ in range(num_circles - len(initial_centers)):
            pt = np.random.rand(2)
            if np.all(np.linalg.norm(initial_centers - pt, axis=1) > 0.1):
                add_centers.append(pt)
        initial_centers = np.vstack((initial_centers, add_centers))

    radii_initial = np.ones(num_circles) * 0.05

    def objective(x):
        return -np.sum(x)

    def circle_circle_constraint(x, i, j):
        x1, y1 = initial_centers[i]
        x2, y2 = initial_centers[j]
        return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - (x[i] + x[j])

        x_i, y_i = initial_centers[i]
        r_i = x[i]
        return np.min([x_i - r_i, 1 - x_i - r_i, y_i - r_i, 1 - y_i - r_i])

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: circle_circle_constraint(x, i, j)})

        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: boundary_constraint(x, i=i)})

    return (final_centers, final_radii, sum_radii)
