import numpy as np
from scipy.optimize import differential_evolution
from scipy.cluster.vq import kmeans
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Create a dense grid of points in [0, 1] x [0, 1]
    grid_x, grid_y = np.meshgrid(np.linspace(0, 1, 200), np.linspace(0, 1, 200))
    grid_points = np.column_stack((grid_x.ravel(), grid_y.ravel()))
    
    # Use k-means to select the best centers
    centers_initial, _ = kmeans(grid_points, num_circles)
    
    # Initial radii estimate using Voronoi cell size
    dists = cdist(centers_initial, centers_initial)
    radius_initial = np.min(dists) / 2.0
    radii_initial = np.full(num_circles, radius_initial)

    # Objective function to minimize (negative of sum of radii)
    def objective(radii):
        return -np.sum(radii)

    # Constraints: Circle-circle and Circle-boundary distance constraints
    constraints = []

    # Circle-circle constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def dist_constraint(radii, i=i, j=j):
                dx = centers_initial[i, 0] - centers_initial[j, 0]
                dy = centers_initial[i, 1] - centers_initial[j, 1]
                return np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    # Circle-boundary constraints
    for i in range(num_circles):
        def boundary_constraint(radii, i=i):
            x, y = centers_initial[i]
            r = radii[i]
            return np.min([x - r, y - r, 1 - x - r, 1 - y - r])
        constraints.append({'type': 'ineq', 'fun': boundary_constraint})

    # Define bounds for radii
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    # Run differential evolution optimization
    result = differential_evolution(
        objective,
        bounds,
        constraints=constraints,
        strategy='best1bin',
        popsize=15,
        maxiter=200,
        tol=1e-10,
        polish=True,
        mutation=(0.5, 1),
        recombination=0.9,
        disp=False
    )

    # Extract the final result
    best_radii = result.x
    best_centers = centers_initial
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
