from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Generate hexagonal grid
    def generate_hexagonal_grid(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = (num_circles + rows - 1) // rows
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                grid.append([x, y])
        return np.array(grid)

    # Generate spiral arrangement
    def generate_spiral(num_circles):
        angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)

        points = np.array([radii * np.cos(angles), radii * np.sin(angles)]).T
        return points

    def generate_nested_grid(num_circles):
        centers = []
        for i in range(5):
            for j in range(5):
                x = (i + 0.5) / 5
                y = (j + 0.5) / 5
                centers.append([x, y])
        while len(centers) < num_circles:
            pt = np.random.rand(2)
            if np.all(np.linalg.norm(centers - pt, axis=1) > 0.1):
                centers.append(pt)
        return np.array(centers)

    # Generate adaptive grid
    def generate_adaptive_grid(num_circles):
        grid_x, grid_y = np.meshgrid(np.linspace(0, 1, 100), np.linspace(0, 1, 100))
        grid_points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        initial_centers = kmeans(grid_points, num_circles)[0]
        return initial_centers

    # Collect best result
    best_centers, best_radii, best_sum = None, None, -1

    # Try different initial configurations
    configurations = [
        generate_hexagonal_grid(num_circles),
        generate_spiral(num_circles),
        generate_nested_grid(num_circles),
        generate_adaptive_grid(num_circles)
    ]

    for centers_initial in configurations:
        # Ensure centers_initial has exactly num_circles points
        centers_initial = centers_initial[:num_circles]

        # Initial radii
        radii_initial = np.full(num_circles, 0.01)

        def objective(radii):

        result = minimize(
