import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.cluster.vq import kmeans
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Create a hybrid initial configuration: start with a dense grid and apply k-means
    def hybrid_initial(num_circles):
        grid = np.linspace(0, 1, 100)
        grid_x, grid_y = np.meshgrid(grid, grid)
        points = np.stack((grid_x.flatten(), grid_y.flatten()), axis=1)
        centers = kmeans(points, num_circles)[0]
        return centers

    # Initial configuration based on a grid with k-means
    centers_initial = hybrid_initial(num_circles)

    # Define the objective function for maximization
    def objective(x):
        return -np.sum(x)  # We minimize the negative sum to maximize the sum

    # Define constraints for circle-circle and circle-boundary
    constraints = []

    # Circle-circle distance constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def dist_constraint(x, i=i, j=j):
                dx = centers_initial[i, 0] - centers_initial[j, 0]
                dy = centers_initial[i, 1] - centers_initial[j, 1]
                distance = np.sqrt(dx * dx + dy * dy)
                return distance - (x[i] + x[j])
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    # Boundary constraints
    for i in range(num_circles):
        def boundary_constraint(x, i=i):
            x_c, y_c = centers_initial[i]
            r = x[i]
            return np.min([x_c - r, y_c - r, 1 - x_c - r, 1 - y_c - r])
        constraints.append({'type': 'ineq', 'fun': boundary_constraint})

    # Initial guess for radii
    radii_initial = np.full(num_circles, 0.05)

    # Bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]

    # Use Nelder-Mead for optimization
    result = minimize(
        objective,
        radii_initial,
        bounds=bounds,
        constraints=constraints,
        method='Nelder-Mead',
        options={
            'maxiter': 1000,
            'ftol': 1e-10,
            'eps': 1e-8,
            'disp': False
        }
    )

    if result.success:
        final_radii = result.x
        final_centers = centers_initial
        sum_radii = np.sum(final_radii)
        return (final_centers, final_radii, sum_radii)
    else:
        raise RuntimeError("Optimization failed: " + str(result.message))
