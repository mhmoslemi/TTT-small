import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial configuration using a hexagonal grid
    def generate_hex_grid(n_circles, side_length):
        positions = []
        for i in range(n_circles):
            row = i // 3
            col = i % 3
            x = col * (1 + side_length) / (n_circles ** 0.5)
            y = row * (1 + side_length) * np.sqrt(3) / (n_circles ** 0.5)
            positions.append([x, y])
        return np.array(positions)

    # Define objective function (to maximize sum of radii, so we minimize -sum of radii)
    def objective_function(x, n_circles):
        centers = x[:n_circles * 2].reshape(n_circles, 2)
        radii = x[n_circles * 2:].reshape(n_circles)
        return -np.sum(radii)

    # Define constraint functions
    def square_boundary_constraint(x, n_circles):
        centers = x[:n_circles * 2].reshape(n_circles, 2)
        radii = x[n_circles * 2:].reshape(n_circles)
        return np.array([max(0, center[0] - radius) for center, radius in zip(centers, radii)]) - 1e-12

    def no_overlap_constraint(x, n_circles):
        centers = x[:n_circles * 2].reshape(n_circles, 2)
        radii = x[n_circles * 2:].reshape(n_circles)
        dist_matrix = cdist(centers, centers)
        return np.array([dist_matrix[i][j] - radii[i] - radii[j] for i in range(n_circles) for j in range(i + 1)]) - 1e-12

    # Start with a hexagonal grid of 25 circles
    n_circles = 26
    centers = generate_hex_grid(n_circles, 0.1)
    radii = np.ones(n_circles) * 0.05

    # Add a big circle in the middle that we will optimize
    centers[-1] = [0.5, 0.5]
    radii[-1] = 0.2

    # Flatten centers and radii for the optimization vector
    x0 = np.concatenate([centers.reshape(-1), radii])

    # Set bounds: centers within [0,1]^2 and radii nonnegative
    bounds = []
    for _ in range(n_circles * 2):
        bounds.append((0, 1))
    for _ in range(n_circles):
        bounds.append((0, 1.0))  # Radius upper bound

    # Optimization settings
    constraints = [
        {'type': 'ineq', 'fun': square_boundary_constraint, 'args': (n_circles,)},
        {'type': 'ineq', 'fun': no_overlap_constraint, 'args': (n_circles,)},
    ]

    # Run optimization
    result = optimize.minimize(

    # Extract final result

    final_radii = result.x[n_circles * 2:].reshape(n_circles)
    total_sum = np.sum(final_radii)

    return final_centers, final_radii, total_sum
