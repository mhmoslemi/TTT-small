import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_no_overlap(radii, centers):
        n = len(radii)
        distances = squareform(pdist(centers))
        for i, j in combinations(range(n), 2):
            if distances[i, j] < radii[i] + radii[j] - 1e-8:
                return -np.inf
        return 0

    def constraint_boundary(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return -np.inf
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -np.inf
        return 0

    # Initial grid-based placement with hexagonal pattern
    rows = 5
    cols = 6
    spacing = 0.3
    centers = []

    # First row
    centers.append([0.5, 0.1 + spacing])
    for i in range(1, cols):
        centers.append([0.5 + i * spacing, 0.1 + spacing])

    # Subsequent rows
    for row in range(1, rows):
        for i in range(cols):
            x = 0.5 + i * spacing - 0.5 * spacing
            y = 0.1 + spacing * (row + 1)
            centers.append([x, y])

    centers = np.array(centers)
    radii = np.full(len(centers), 0.05)

    # Optimization
    bounds = [(0, 0.5) for _ in range(len(radii))]
    cons = [
        {'type': 'ineq', 'fun': constraint_no_overlap, 'args': (centers,)},
        {'type': 'ineq', 'fun': constraint_boundary, 'args': (centers,)}
    ]

    result = minimize(objective, radii, bounds=bounds, constraints=cons)

    # Validate the result
    if not result.success:
        raise RuntimeError("Optimization failed")

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    # Final validation
    if not validate_packing(centers, radii)[0]:
        raise RuntimeError("Invalid packing")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
