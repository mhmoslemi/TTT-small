def generate_hex_grid(n):

            centers.append([x, y])

    centers = np.array(centers[:26])
    return centers

def compute_overlap(c1, r1, c2, r2):
    """Check if two circles with centers and radii overlap."""
    dx = c1[0] - c2[0]
    dy = c1[1] - c2[1]
    dist = np.sqrt(dx**2 + dy**2)
    return dist < r1 + r2 - 1e-12

    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            if compute_overlap(centers[i], radii[i], centers[j], radii[j]):
