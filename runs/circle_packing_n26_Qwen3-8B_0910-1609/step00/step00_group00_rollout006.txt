import scipy.optimize as optimize
import cvxpy as cp

def run_packing():

    initial_centers = []

                x += 0.5 / 4.0
            initial_centers.append([x, y])
    initial_centers = np.array(initial_centers)

    # Convert to variables for optimization
    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable(num_circles, non_negative=True)

    objective = cp.Maximize(cp.sum(radii))

        constraints.append(y >= 0.0)

    # No overlap between circles
    for i in range(num_circles):

            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)

    problem = cp.Problem(objective, constraints)
