import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    n = 26
    row_count = 5
    col_count = 6

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

        row = i // col_count
        col = i % col_count
        x = col / (col_count - 1)
        y = row / (row_count - 1)

        radii[i] = 0.01  # Initial small radius

    bounds_centers = np.array([[0.0, 1.0] for _ in range(2 * n)])

        for i in range(n):

                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    overlaps.append(1)
        return np.sum(overlaps)

    cons = [{'type': 'ineq', 'fun': constraint_non_overlap}]

    x0 = np.concatenate([centers.flatten(), radii])
