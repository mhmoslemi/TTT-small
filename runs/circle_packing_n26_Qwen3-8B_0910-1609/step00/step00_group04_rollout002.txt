import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():

    grid_y = np.linspace(0.1, 0.9, 5)  # 5 rows
    grid = np.array([(x, y) for x in grid_x for y in grid_y])  # 25 points
    # Add one extra circle at the center
    centers = np.vstack((grid, [0.5, 0.5]))

    bounds = Bounds(0, np.inf)

    # Define constraints: circles must be within the unit square and must not overlap
    def constraints_func(centers, radii):

        def in_bounds(centers, radii):
            return np.all((centers[:, 0] - radii <= 1 + 1e-12) &
                         (centers[:, 0] + radii >= 0 - 1e-12) &
                         (centers[:, 1] - radii <= 1 + 1e-12) &

        # Constraint 2: No overlap between circles
        def no_overlap(centers, radii):
            for i in range(len(radii)):
                for j in range(i + 1, len(radii)):
                    dx = centers[i, 0] - centers[j, 0]
