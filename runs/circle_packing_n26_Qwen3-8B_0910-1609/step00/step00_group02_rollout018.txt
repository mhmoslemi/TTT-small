import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Define base configuration for 16 circles as 4x4 grid in the center
    grid_x = np.linspace(0.15, 0.85, 4)
    grid_y = np.linspace(0.15, 0.85, 4)
    base_centers = np.zeros((16, 2))
    for i in range(4):
        for j in range(4):
            base_centers[i * 4 + j] = [grid_x[j], grid_y[i]]
    
    # Add 10 circles in corners to maximize their radii
    corner_centers = np.array([
        [0.05, 0.05], [0.95, 0.05], [0.95, 0.95], [0.05, 0.95],
        [0.25, 0.25], [0.75, 0.25], [0.75, 0.75], [0.25, 0.75],
        [0.5, 0.5]
    ])
    
    # Combine all centers
    initial_centers = np.vstack((base_centers, corner_centers))
    
    # Initial guess for radii: uniform but small to avoid overlap
    initial_radii = np.full(26, 0.05)
    
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii
    
    def constraint_func(radii):
        # All circles must be within the unit square
        for i in range(26):
            if (initial_centers[i, 0] - radii[i] < 0 or
                initial_centers[i, 0] + radii[i] > 1 or
                initial_centers[i, 1] - radii[i] < 0 or
                initial_centers[i, 1] + radii[i] > 1):
                return False
        # All circles must be non-overlapping
        for i in range(26):
            for j in range(i + 1, 26):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    # Optimize using scipy.optimize
    result = minimize(
        objective, initial_radii,
        method='SLSQP',
        constraints={"type": "ineq", "fun": constraint_func},
        bounds=[(0, 1) for _ in range(26)]
    )
    
    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
