import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define the number of circles and square size
    n = 26
    L = 1.0  # unit square side length

    # Function to compute circle centers in a hexagonal packing
    def get_hexagonal_centers(n, offset=0.5):
        # Returns coordinates of a hexagonal grid with n circles
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        centers = []
        for r in range(rows):
            for c in range(cols):
                x = c + offset
                x = x if r % 2 == 0 else c + offset + 0.5
                y = r + 0.5
                centers.append((x, y))
        return np.array(centers)

    # Initial circle centers in hexagonal-like grid within [0, 1]^2
    centers = get_hexagonal_centers(n)
    # Scale centers to fit in [0, 1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0)) * L
    # Add small offset to avoid perfect alignment
    centers += np.random.rand(*centers.shape) * 1e-3

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # minimize negative sum to maximize actual sum

    # Inequality constraints:
    # - Circle must be fully within [0,1]^2
    # - Circles must not overlap
    def constraint1(radii):
        return np.sum(radii)  # just for constraint bounds

    def constraint2(radii):
        # Ensure circles are fully within [0,1]^2
        # Using distance from nearest boundary
        dist_to_boundary = np.minimum(centers[:, 0], 1 - centers[:, 0])
        dist_to_boundary = np.minimum(dist_to_boundary, centers[:, 1])
        dist_to_boundary = np.minimum(dist_to_boundary, 1 - centers[:, 1])
        return radii - dist_to_boundary

    def constraint3(radii):
        # Ensure circles do not overlap
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
        return radii[:, np.newaxis] + radii[np.newaxis, :] - dists

    # Bounds for radii (minimum 0, maximum based on initial guess)
    max_radius = 0.1  # This is a starting guess; it will be adjusted by the solver

    # Create bounds for each radius
    bounds = [(0, max_radius) for _ in range(n)]

    # Combine all constraints into a single vector for optimization
    def inequality_constraints(radii):
        return np.concatenate([
            constraint2(radii),
            constraint3(radii)
        ])

    # Use scipy.optimize to maximize sum of radii
    result = optimize.minimize(
        objective,
        x0=np.full(n, 0.1),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': inequality_constraints}
        ],
        method='SLSQP'
    )

    radii = result.x
    sum_radii = np.sum(radii)

    # Validate the packing
    validation_result, message = validate_packing(centers, radii)
    if not validation_result:
        raise ValueError("Packing validation failed: " + message)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
