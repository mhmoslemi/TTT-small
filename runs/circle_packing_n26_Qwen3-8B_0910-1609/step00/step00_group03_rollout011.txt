from scipy.optimize import minimize

def run_packing():

    n = 26

    # Step 2: Generate points in a hexagonal lattice of 5 rows
    num_rows = 5
    rows = np.linspace(0, 1, num_rows)
    cols_per_row = [5, 6, 6, 6, 5]  # To match 26 circles in total
    x = []
    y = []

    # Generate x and y positions for each row
    for i, row in enumerate(rows):
        row_y = row
        row_x = np.linspace(0.05, 0.95, cols_per_row[i])
        for j, x_pos in enumerate(row_x):
            x.append(x_pos)
            y.append(row_y)

    # Convert to numpy arrays
    centers = np.array([x, y]).T
    radii = np.zeros(n)

    # Step 3: Optimize radii with constraints
    def objective(radii):
        return -np.sum(radii)  # Minimization of negative sum (maximize sum)

    def constraint_overlap(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf  # Penalize overlap
        return 0

    # Setup constraints and bounds
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'ineq', 'fun': constraint_overlap}]

    # Initial guess: all radii equal
    initial_guess = np.full(n, 0.05)

    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract the optimized radii
    radii = result.x

    # Return the centers, radii, and sum of radii
    return centers, radii, np.sum(radii)
