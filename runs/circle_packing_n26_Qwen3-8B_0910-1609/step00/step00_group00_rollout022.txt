import scipy.optimize as optimize

def run_packing():

    positions = []
    radii = []

        col = i // 5
        row = i % 5
        x = 0.2 + (col + 0.5) * 0.25
        y = 0.2 + (row + 0.5) * 0.25
        # Introduce random jitter
        x += np.random.uniform(-0.05, 0.05)
        y += np.random.uniform(-0.05, 0.05)
        positions.append([x, y])
        radii.append(0.05)

    # Flatten and set as initial guess
    x0 = np.array(positions)
    r0 = np.array(radii)

    # Define bounds for positions and radii
    pos_bounds = [(0, 1) for _ in range(2 * n)]
    r_bounds = [(0, 1) for _ in range(n)]

    # Function to compute total radius sum

        r_vec = x[2 * n:]
        return -np.sum(r_vec)  # Maximize by minimizing the negative

    # Constraints: non-overlap (dist >= r_i + r_j - 1e-8)
    # Create constraints for distance between all pairs
    def constraint_func(x):
        x_vec = x[:2 * n].reshape(n, 2)
        r_vec = x[2 * n:]
        constraints = []
        for i in range(n):

                dx = x_vec[i, 0] - x_vec[j, 0]
                dy = x_vec[i, 1] - x_vec[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - r_vec[i] - r_vec[j] + 1e-8)
        return constraints

    result = optimize.minimize(
