import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial layout: 5 rows with 5 circles each
    num_circles = 26
    num_rows = 5
    circles_per_row = 5
    radius_initial = 0.1  # Initial radius guess

    # Generate initial positions with uniform spacing
    row_height = np.sqrt(3)/2 * 2 * radius_initial  # Vertical spacing between rows
    horizontal_spacing = 2 * radius_initial  # Horizontal spacing between circles
    centers = []
    radii = []

    # Calculate vertical spacing adjustment to fit all rows within [0, 1]
    total_vertical_height = (num_rows - 1) * row_height
    vertical_spacing_factor = (1 - total_vertical_height) / (num_rows + 1)
    row_positions = np.linspace(0 + vertical_spacing_factor, 1 - vertical_spacing_factor, num_rows)

    for i, y in enumerate(row_positions):
        # Calculate horizontal spacing considering boundary constraints
        if i == 0 or i == num_rows - 1:
            # Edge rows: reduce horizontal spacing
            horizontal_spacing = 2 * (1 - 2 * radius_initial) / (circles_per_row - 1)
        else:
            horizontal_spacing = 2 * radius_initial

        for j in range(circles_per_row):
            x = j * horizontal_spacing
            centers.append([x, y])
            radii.append(radius_initial)

    centers = np.array(centers)
    radii = np.array(radii)

    # Define function to compute total overlap and adjust radii
    def objective_function(radii):
        total_overlap = 0
        for i in range(len(radii)):

                dist = np.sqrt(dx**2 + dy**2)
                total_overlap += max(0, radii[i] + radii[j] - dist)
        return total_overlap

    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimize radii to minimize overlap
    result = optimize.minimize(
        objective_function,
        radii,
        bounds=bounds,
        method='L-BFGS-B',
        options={'disp': False}

    total_sum = np.sum(radii_optimized)

    return (centers, radii_optimized, total_sum)
