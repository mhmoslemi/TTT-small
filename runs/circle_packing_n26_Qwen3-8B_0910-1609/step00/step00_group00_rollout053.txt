import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Define the number of circles
    n_circles = 26
    
    # Initial guess: use a hexagonal packing with 5 rows
    # This is a hybrid approach: 5 rows with varying numbers of circles
    # Row 0: 5 circles
    # Row 1: 6 circles
    # Row 2: 5 circles
    # Row 3: 6 circles
    # Row 4: 5 circles
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    # Place circles in a hexagonal grid with varying row sizes
    row_height = np.sqrt(3)/2
    max_y = 1.0 - 0.05
    num_rows = 5
    circle_radius = 0.08
    
    # Assign rows
    row_indices = [0, 1, 2, 3, 4]
    for i, row in enumerate(row_indices):
        y = 0.05 + i * row_height
        if i % 2 == 0:
            num_circles = 5
        else:
            num_circles = 6
        for j in range(num_circles):
            x = 0.05 + j * (1.0 - 0.1) / (num_circles - 1)
            centers[i * num_circles + j] = [x, y]
            radii[i * num_circles + j] = circle_radius
    
    # Use scipy.optimize to optimize the positions and radii
    def objective(x):
        centers_flat = x[:n_circles*2].reshape((n_circles, 2))
        radii_flat = x[n_circles*2:]
        sum_radii = np.sum(radii_flat)
        return -sum_radii  # We want to maximize sum_radii, so minimize -sum_radii
    
    def constraints(x):
        centers_flat = x[:n_circles*2].reshape((n_circles, 2))
        radii_flat = x[n_circles*2:]
        constraints = []
        
        # Ensure all circles are within [0,1]^2
        for i in range(n_circles):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False, "Circle {} outside unit square".format(i)
        
        # Ensure non-overlap
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                if dist_matrix[i, j] < radii_flat[i] + radii_flat[j] - 1e-12:
                    return False, "Circles {} and {} overlap".format(i, j)
        
        return True, "ok"
    
    # Set bounds for centers: x and y between 0 and 1, radii between 0 and 0.15
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
    for _ in range(n_circles):
        bounds.append((0.0, 0.15))  # radius
    
    # Run the optimization
    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    centers_opt = result.x[:n_circles*2].reshape((n_circles, 2))
    radii_opt = result.x[n_circles*2:]
    sum_radii = np.sum(radii_opt)
    
    return (centers_opt, radii_opt, sum_radii)
