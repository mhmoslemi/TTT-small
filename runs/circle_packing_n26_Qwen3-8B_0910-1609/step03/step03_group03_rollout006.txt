from scipy.optimize import minimize

def run_packing():

    centers = np.zeros((num_circles, 2))
    radius = 0.1  # Initial guess for all radii

        centers[i, 1] = 0.5

    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize

    # Constraint for non-overlapping circles
    def constraint_dist(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (x[i] + x[j])

    def boundary(x, i):
        x_i, y_i = centers[i]
        r = x[i]
        return x_i - r >= -1e-12 and y_i - r >= -1e-12 and x_i + r <= 1 + 1e-12 and y_i + r <= 1 + 1e-12

    # Define bounds for radii (0 to 1)
    bounds = [(0, 1) for _ in range(num_circles)]

    # Create constraints dynamically
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_dist(x, i, j)})
    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: boundary(x, i)})

    # Run optimization with improved settings
    result = minimize(

        bounds=bounds,
        constraints=constraints,

    return (final_centers, final_radii, sum_radii)
