from scipy.optimize import minimize

def run_packing():

    # Initial centers with adaptive spacing
    centers = np.random.rand(num_circles, 2)
    centers[:, 0] = np.sort(centers[:, 0])  # Sort by x-coordinate for better spacing
    centers[:, 1] = np.sort(centers[:, 1])  # Sort by y-coordinate

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint: non-overlapping between all pairs
    def constraint_pair(i, j):
        return lambda r: np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j])

    # Constraint: circle fully within the square
    def boundary(i):
        return lambda r: min(centers[i, 0] - r[i], centers[i, 1] - r[i], 1 - centers[i, 0] - r[i], 1 - centers[i, 1] - r[i])

    # Initial guess: small radius to ensure validity
    x0 = np.full(num_circles, 0.05)

    # Bounds for each radius
    bounds = [(0, 1) for _ in range(num_circles)]

    # Create constraints for the non-overlapping condition
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint_pair(i, j)(r)
            })

    # Create constraints for the boundary condition
    for i in range(num_circles):
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: boundary(i)(r)
        })

    # Run the optimization using SLSQP
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
    )

    # Extract the final centers and radii
    final_centers = centers

    def validate_packing(centers, radii):

            if (x - r < -1e-12 or x + r > 1 + 1e-12

    return (final_centers, final_radii, sum_radii)
