import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

    radius_initial = 0.05

    def hexagonal_lattice(n):
        row = int(np.ceil(np.sqrt(n)))
        col = int(np.ceil(n / row))

        offset = spacing / 2
        centers = []
        for i in range(n):
            col_idx = i % col
            row_idx = i // col
            x = (col_idx + 0.5) * spacing

                x += offset
            centers.append([x, y])

    def boundary(x, i):

        r_val = x[i]
        return x_val - r_val, y_val - r_val, 1 - x_val - r_val, 1 - y_val - r_val

    constraints = []

    result = minimize(
