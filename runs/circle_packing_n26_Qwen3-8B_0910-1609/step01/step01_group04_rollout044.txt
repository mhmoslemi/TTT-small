import numpy as np
import cvxpy as cp

def run_packing():

    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # The objective is to maximize the total sum of radii

    # Constraints
    constraints = []

    # (1) All circles must be inside the square [0,1]x[0,1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # (2) No overlaps between circles

        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # (3) Radii must be non-negative
    for i in range(n):
        constraints.append(radii[i] >= 0)

    # Create and solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, eps=1e-6)

    # Output the results
