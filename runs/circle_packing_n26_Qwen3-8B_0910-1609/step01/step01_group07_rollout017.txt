from scipy.optimize import minimize

def run_packing():

    rows = 4
    cols = 7

    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    count = 0

    for i in range(rows):
        for j in range(cols):
            if count >= N:
                break
            x = i / (rows - 1)
            y = j / (cols - 1)
            centers[count, 0] = x
            centers[count, 1] = y
            count += 1

    # Ensure all circles are initialized with same starting radius (for optimization)
    initial_radius = 0.05
    radii = np.full(N, initial_radius)

    # Define the objective function (negative sum to maximize)
    def objective(radii):
        return -np.sum(radii)

    # Define constraints for non-overlap
    def non_overlap_constraints(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = centers[i]
                x_j, y_j = centers[j]
                dist = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda r, i=i, j=j: dist - (r[i] + r[j])
                })
        return constraints

    def boundary_constraints(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
