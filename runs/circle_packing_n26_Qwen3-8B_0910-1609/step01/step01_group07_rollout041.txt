from scipy.optimize import minimize

def run_packing():

    # We will use a vertical packing of circles in rows, with offset
    # and select the 26 best positions for the circles

    # Number of rows and columns
    rows = 7
    cols = 4

    # Radius for all circles (will be refined)
    initial_r = 0.1

    # Generate a grid of points
    grid = []
    for r in range(rows):
        for c in range(cols):
            x = c * (1 / (cols - 1)) + 0.5 / (cols - 1)
            y = r * (1 / (rows - 1)) + 0.5 / (rows - 1)
            grid.append([x, y])
    
    # Generate a small offset for hexagonal packing
    offset = 0.1  # distance between rows
    hex_grid = []
    for r in range(rows):
        for c in range(cols):
            x = c * (1 / (cols - 1)) + 0.5 / (cols - 1)
            y = r * (1 / (rows - 1)) + 0.5 / (rows - 1) + (r % 2) * offset
            hex_grid.append([x, y])
    
    # Use hex_grid as the base positions
    centers = np.array(hex_grid)

    # Initial radii (same for all)
    radii = np.full(len(centers), initial_r)

    # Optimization: refine radii to maximize sum while satisfying non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)  # minimize negative of sum (equivalent to maximizing sum)

    def constraints(radii):
        # Check that each circle is within the square [0,1]^2
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        # Check that no two circles overlap
        for i in range(len(radii)):

                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < radii[i] + radii[j]:
                    return False
        return True

    # We will use scipy's optimization to maximize the sum of radii
    def optimize_radii():

            objective,

            constraints=[
                {'type': 'ineq', 'fun': lambda r: constraints(r)},

            method='SLSQP'

    final_radii = optimize_radii()
