import cvxpy as cp

def run_packing():

        centers[i, 1] = y

    # Build optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Constraints: non-overlapping between circles
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-8)

    # Constraints: circles must be within the unit square

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    prob = cp.Problem(objective, constraints)
