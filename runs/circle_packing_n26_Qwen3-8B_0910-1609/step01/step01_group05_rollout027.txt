import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define initial centers using a hexagonal packing in a unit square
    num_circles = 26
    cols = 5
    rows = 6
    spacing = 1.0 / (cols - 1)  # Horizontal spacing
    vertical_spacing = np.sqrt(3) / 2 * spacing  # For hexagonal packing

    centers = np.empty((num_circles, 2))
    radii = np.ones(num_circles)  # Initial guess for radii

    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * vertical_spacing
            # Place circles in a hexagonal grid, offset for even rows
            if i % 2 == 1:
                x_offset = spacing / 2
                x = j * spacing + x_offset
            centers[i * cols + j] = [x, y]

    # Initial radius guess to match the arrangement
    radii = np.ones(num_circles) * 0.1

    # Optimization: Maximize sum of radii while ensuring no overlaps and staying within the square
    def objective(radii):
        return -np.sum(radii)  # Negative because we are minimizing

    def constraint_circle_in_square(radii, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1  # Violation of square constraint
        return 0

    def constraint_no_overlap(radii, centers):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx ** 2 + dy ** 2)
                if distance < radii[i] + radii[j] - 1e-12:
                    return -1  # Overlap
        return 0

    # Create bounds for radii: non-negative
    bounds = [(0, 1)] * num_circles

    # Define constraints as functions for optimization
    cons = [
        opt.ScalarConstraint(constraint_circle_in_square, args=(centers,), name="in_square"),
        opt.ScalarConstraint(constraint_no_overlap, args=(centers,), name="no_overlap")
    ]

    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=cons,
        method="SLSQP",
        tol=1e-6
    )

    if not result.success:
        raise RuntimeError("Optimization failed with status: " + str(result.status))

    radii = result.x
    total_radius_sum = np.sum(radii)

    return centers, radii, total_radius_sum
