import scipy.optimize as opt

def run_packing():

    n_circles = 26
    y_offsets = np.linspace(0, 1, 6)
    x_offsets = np.linspace(0, 1, 5)
    grid_centers = np.array([[x, y] for y in y_offsets for x in x_offsets])
    # Only take the first 26 points (from the grid of 6 rows x 5 columns = 30 points)
    centers = grid_centers[:26]
    
    # Radius optimization variable (start with equal radii)
    radii = np.full(n_circles, 0.05)
    
    # Objective function to maximize total sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Constraints:
    # 1. Each circle must stay within unit square
    # 2. No overlap between any two circles
    def constraint_circle_in_square(radii):
        return np.sum((centers - radii[:, np.newaxis]) ** 2) <= 1.0
    
    def constraint_no_overlap(radii):
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Use L-BFGS-B optimizer to maximize the sum of radii
    result = opt.minimize(
        fun=objective,
        x0=radii,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_circles)],
        constraints=[
            {'type': 'ineq', 'fun': constraint_circle_in_square},
            {'type': 'ineq', 'fun': constraint_no_overlap}
        ]
    )
    
    # Extract results
    final_centers = centers
