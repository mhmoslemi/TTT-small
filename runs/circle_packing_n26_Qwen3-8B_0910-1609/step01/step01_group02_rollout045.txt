from scipy.optimize import minimize

def run_packing():

    n = 26

    # Let's use a grid with 5 rows and 5 columns with some offset for the hexagonal layout
    
    # Define grid
    num_rows = 5
    num_cols = 5
    row_height = 0.9 / (num_rows - 1)  # height per row
    col_width = 0.9 / (num_cols - 1)  # width per column
    
    # Create centers without radii
    centers = []
    for i in range(num_rows):
