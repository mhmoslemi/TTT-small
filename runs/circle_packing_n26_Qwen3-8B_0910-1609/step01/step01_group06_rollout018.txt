import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial hexagonal grid in the unit square
    def initial_hexagonal_layout(n):
        radii = np.full(n, 1.0 / np.sqrt(12))  # Based on hexagonal packing density
        centers = []
        for i in range(n):
            col = i // 5  # 5 columns in a hexagonal grid
            row = i % 5
            x = col + (row % 2) * 0.5 * np.sqrt(3) / 2
            y = row * np.sqrt(3) / 2
            centers.append([x, y])
        centers = np.array(centers)
        # Normalize to fit in [0, 1]
        centers /= np.max(centers)
        return np.array(centers), radii

    # Helper function for constraints
    def constraints(x, centers, radii):
        # x is a vector of all variables: [x0, y0, r0, x1, y1, r1, ...]
        n = len(radii)
        x_vec = x[:2 * n]
        r_vec = x[2 * n:]
        centers = np.array(x_vec).reshape(n, 2)
        radii = r_vec
        constraints = []

        # Boundary constraints
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                constraints.append(1.0)
            else:
                constraints.append(0.0)

        # Circle-circle non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    constraints.append(1.0)
                else:
                    constraints.append(0.0)

        return constraints

    n = 26
    centers, radii_init = initial_hexagonal_layout(n)
    x0 = np.concatenate([centers.flatten(), radii_init])

    # Define optimization problem
    def objective(x):
        return -np.sum(x[2 * n:])  # Maximize sum of radii

    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0) for _ in range(2)])  # x, y of each circle
        bounds.extend([(0.0, 1.0) for _ in range(1)])  # radius of each circle

    # Optimize
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            opt.LinearConstraint(np.ones(len(x0)), lb=-np.inf, ub=0.0, name="non_overlap")
        ],
        method='SLSQP'
    )

    # Extract final centers and radii
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
