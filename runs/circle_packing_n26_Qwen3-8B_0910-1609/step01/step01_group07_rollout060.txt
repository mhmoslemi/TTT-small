import scipy.optimize as optimize

def run_packing():

    # Optimized for unit square [0, 1] x [0, 1]

    centers = []

        # Even rows start at (col_width * 0.5) for offset
        # Odd rows start at (col_width * 0) for alignment
        for col in range(6 if row == 0 else 5):
            x = col * col_width + (col % 2) * col_width * 0.5
            y = row * row_height
            centers.append([x, y])

    centers = np.array(centers)

    # Initial radii for all circles (small, to be optimized)
    radii = np.full(len(centers), 0.01)

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints:
    # 1. Circle centers are within the unit square
    # 2. Circle radii are non-negative
    # 3. Circles do not overlap

    def constraint_center_in_unit_square(x):
        # x is [centers, radii]
        centers = x[:len(centers), :]
        radii = x[len(centers):]
        return np.array([centers[:, 0] - radii < 1 + 1e-12,
                        centers[:, 0] + radii > -1e-12,
                        centers[:, 1] - radii < 1 + 1e-12,
                        centers[:, 1] + radii > -1e-12])

    def constraint_non_negative_radii(x):
        return x[len(centers):] >= 0

    def constraint_circles_do_not_overlap(x):
        centers = x[:len(centers), :]
        radii = x[len(centers):]
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return dist - radii[i] - radii[j] + 1e-12
        return 0

    # Combine all constraints
    def constraints(x):

        c2 = constraint_non_negative_radii(x)
        c3 = constraint_circles_do_not_overlap(x)
        return np.concatenate([c1, c2, [c3]])

    result = optimize.minimize(

    total_sum_radii = np.sum(radii_opt)
