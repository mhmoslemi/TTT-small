def run_packing():

    # We'll place the centers in a grid structure, and adjust radii based on spacing
    
    # Generate a hexagonal grid of 26 points within the unit square
    num_rows = 6
    num_cols = 5
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j / (num_cols - 1) if num_cols > 1 else 0.5
            y = i / (num_rows - 1) if num_rows > 1 else 0.5
            centers.append([x, y])
    centers = np.array(centers)

    # Get all pairs of indices to compute distances between centers
    distance_matrix = np.zeros((len(centers), len(centers)))
    for i in range(len(centers)):

            dy = centers[i, 1] - centers[j, 1]
            distance_matrix[i, j] = np.sqrt(dx ** 2 + dy ** 2)
            distance_matrix[j, i] = distance_matrix[i, j]

    # Initial guess for radii: all circles have the same initial radius
    r0 = 0.1  # Start with small radii and let the optimization grow them

    # Define objective function: maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Minimize negative sum to maximize sum

    # - The distance between any two centers >= r_i + r_j
    # - Each circle must fit within the unit square (radius <= 1 - max(x, y))
    # - r >= 0

        return distance_matrix[i, j] - r[i] - r[j]

        return (1 - x - r[i]) - (x - r[i]) - 1e-12  # Ensure the circle doesn't exceed the square

            'type': 'ineq',
            'fun': lambda r, i=i: constraint_square(r, i)

        constraints.append({
            'type': 'ineq',

    # Optimize

        x0=r0 * np.ones(len(centers)),

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
