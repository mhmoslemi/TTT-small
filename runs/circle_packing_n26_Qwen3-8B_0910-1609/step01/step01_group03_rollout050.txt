import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def create_hex_grid(n: int, square_size: float = 1.0, spacing: float = 0.3):
    """
    Creates an approximate hexagonal grid of circles inside the unit square.
    Returns: (centers, radii)
    """
    if n == 0:
        return np.array([]), np.array([])

    # Hexagonal grid parameters
    max_x, max_y = square_size, square_size
    row_count = int(np.ceil(np.sqrt(n)))
    col_count = int(np.ceil(n / row_count))
    radius = spacing * 0.5

    centers = []
    for i in range(row_count):
        for j in range(col_count):
            x = j * spacing * np.sqrt(3) / 2
            y = i * spacing * 1.5
            if x > max_x or y > max_y:
                continue
            centers.append([x, y])

    # Trim to n circles
    centers = np.array(centers[:n])
    radii = np.full(n, radius)
    return centers, radii

def compute_overlap(centers, radii, i, j):
    dx = centers[i, 0] - centers[j, 0]
    dy = centers[i, 1] - centers[j, 1]
    dist = np.sqrt(dx ** 2 + dy ** 2)
    return dist - (radii[i] + radii[j])

def optimize_positions(centers, radii, max_iter=100, tol=1e-6):
    n = centers.shape[0]
    if n == 0:
        return np.array([]), np.array([])

    # Reformulate problem: maximize sum(radii) s.t. constraints
    def objective(radii_vec):
        return -np.sum(radii_vec)

    def constraint_overlap(i, j):
        return lambda r: compute_overlap(centers, r, i, j)

    # Prepare constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})

    # Add square boundary constraints
    def left_constraint(r):
        return np.min(centers[:, 0] - r) + 1e-6

    def right_constraint(r):
        return np.max(centers[:, 0] + r) - 1.0 - 1e-6

        return np.min(centers[:, 1] - r) + 1e-6

    constraints.append({'type': 'ineq', 'fun': left_constraint})

    constraints.append({'type': 'ineq', 'fun': bottom_constraint})
    constraints.append({'type': 'ineq', 'fun': top_constraint})

    # Initial guess
    radii_vec = radii.copy()

    result = minimize(

        constraints=constraints,
        bounds=[(0, 1) for _ in range(n)]

    if not result.success:

    return centers, result.x
