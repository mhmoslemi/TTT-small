from itertools import combinations

def run_packing():

    # Start with a grid of 5x5 (25 circles) and add one more at center

    radii = []
    for i in range(5):

            x = i / 4.0

            centers.append([x, y])

    radii = [0.1] * len(centers)  # Initial radius guesses

    def objective(radii):

    def constraints(radii):

            if dist < radii[i] + radii[j] - 1e-8:
                return False
        # Boundary constraint: circle must be inside [0, 1]x[0, 1]
        for i in range(len(centers)):

                return False
        return True

        objective,
        radii,
        constraints=lambda x: constraints(x),
        method='SLSQP',
        bounds=[(0, 1)] * len(radii)

    if not result.success:
        raise ValueError("Optimization failed.")
