import numpy as np
import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26

    # Define variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Define objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Define constraints
    constraints = []

    # Ensure circles are entirely within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Non-overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Non-negative radii
    constraints.append(radii >= 0)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, eps=1e-6)

    # Extract solution
    centers = centers.value
    radii = radii.value
    sum_radii = np.sum(radii)

    # Return the centers, radii, and sum of radii
    return (centers, radii, sum_radii)
