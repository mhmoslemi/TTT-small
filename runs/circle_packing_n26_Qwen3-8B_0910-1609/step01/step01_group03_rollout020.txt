import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define initial hexagonal packing of 26 circles
    def hexagonal_grid(n, size=(1, 1), spacing=1):
        grid = []
        for i in range(n):
            row = []
            for j in range(i + 1):
                x = j * spacing / np.sqrt(3)
                y = i * spacing / 2
                row.append((x, y))
            grid.extend(row)
        return np.array(grid)

    initial_centers = hexagonal_grid(26, size=(1, 1), spacing=1)
    initial_radii = np.ones(26) * 0.05

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(radii, centers):
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def constraint_bounds(radii, centers):
        n = centers.shape[0]
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True

    # Optimization setup
    bounds = [(0, 1) for _ in range(26)]
    cons = [
        {'type': 'ineq', 'fun': lambda r: constraint_overlap(r, initial_centers)},
        {'type': 'ineq', 'fun': lambda r: constraint_bounds(r, initial_centers)},
    ]

    result = minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-10}
    )

    centers = np.copy(initial_centers)
    # Shift centers to fit within [0, 1] using linear scaling
    scale = 1 / np.max(np.abs(centers))
    centers *= scale

    # Ensure all circles lie within [0, 1]
    centers = np.clip(centers, 0, 1)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
