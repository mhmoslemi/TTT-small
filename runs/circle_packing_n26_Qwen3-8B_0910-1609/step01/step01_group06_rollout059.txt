import cvxpy as cp

    # Define variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Constraints
    constraints = []

    # Boundary constraints: circles must lie within [0,1]^2
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)
