from scipy.optimize import minimize

        [0.2, 0.2], [0.8, 0.2], [0.8, 0.8], [0.2, 0.8], [0.5, 0.5]
    ])

    # Initial guesses for the remaining 20 smaller circles
    small_centers = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9], [0.25, 0.25],
        [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.3, 0.3], [0.7, 0.3],

        [0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]
    ])

    centers = np.vstack((large_centers, small_centers))
    radii = np.zeros(26)

    # Assign an initial radius to the large circles based on corner-to-corner distance
    # For corner circles (0.2, 0.2), radius should be about 0.15 to be safe
    # Center circle can have radius about 0.1
    radii[0:5] = [0.15, 0.15, 0.15, 0.15, 0.1]

    # Define a function to compute the total area and constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(i, j, centers, radii):
        dist = np.linalg.norm(centers[i] - centers[j])
        return dist - (radii[i] + radii[j])

    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return x - r, x + r, y - r, y + r

    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=[(0, 1) for _ in range(26)],
        constraints=[
            {'type': 'ineq', 'fun': lambda r: constraint_overlap(i, j, centers, r) for i in range(26) for j in range(i+1, 26)},
            {'type': 'ineq', 'fun': lambda r: constraint_boundary(i, centers, r) for i in range(26)}
        ]
    )

    # Extract final result
    final_radii = result.x
    final_centers = centers
    total_sum = np.sum(final_radii)
    return final_centers, final_radii, total_sum
