import numpy as np
from scipy.optimize import minimize
from typing import Tuple, Any

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial approximate hexagonal packing with 5 rows
    n_circles = 26
    max_radius = 0.2

    # Initial guess for circle centers and radii
    # Create an approximate hexagonal packing
    centers = []
    radii = []

    # First row: bottom row
    for i in range(5):
        x = (i + 0.5) / 5
        y = 0.1
        centers.append([x, y])
        radii.append(0.1)
    # Second row: shift right by half the diameter of the circle
    for i in range(5):
        x = (i + 0.5) / 5
        y = 0.3
        centers.append([x, y])
        radii.append(0.1)
    # Third row: shift left by half the diameter
    for i in range(5):
        x = (i + 0.5) / 5
        y = 0.5
        centers.append([x, y])
        radii.append(0.1)
    # Fourth row: shift right by half the diameter
    for i in range(5):
        x = (i + 0.5) / 5
        y = 0.7
        centers.append([x, y])
        radii.append(0.1)

    centers = np.array(centers)
    radii = np.array(radii)

    # Define objective function: maximize the sum of radii
    def objective(x):
        # x[0::2] = centers, x[1::2] = radii
        centers_new = x[0::2].reshape(-1, 2)
        radii_new = x[1::2]
        return -np.sum(radii_new)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint1(x):
        # Constraint: circle within the unit square
        centers_new = x[0::2].reshape(-1, 2)
        radii_new = x[1::2]
        for i in range(len(centers_new)):
            x_center, y_center = centers_new[i]
            r = radii_new[i]
            if x_center - r < -1e-12 or x_center + r > 1 + 1e-12 or y_center - r < -1e-12 or y_center + r > 1 + 1e-12:
                return -1
        return 0

    def constraint2(x):
        # Constraint: no overlap between circles
        centers_new = x[0::2].reshape(-1, 2)
        radii_new = x[1::2]
        for i in range(len(centers_new)):
            for j in range(i + 1, len(centers_new)):
                dist = np.sqrt(np.sum((centers_new[i] - centers_new[j]) ** 2))
                if dist < radii_new[i] + radii_new[j] - 1e-12:
                    return -1
        return 0

    # Flatten centers and radii into a single vector
    x0 = np.concatenate([centers.flatten(), radii])

    # Set bounds for each variable
    bounds = []
    for i in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, max_radius)])  # x and y coordinates between 0 and 1, radii between 0 and max_radius

    # Set constraints
    cons = (
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    )

    # Run optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-10,
        options={'maxiter': 1000}
    )

    # Extract result
    x_result = result.x
    centers_result = x_result[0::2].reshape(-1, 2)
    radii_result = x_result[1::2]

    # Return centers, radii, and sum of radii
    return centers_result, radii_result, np.sum(radii_result)
