import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26
    
    # Define variables: centers (x_i, y_i) and radii r_i
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)
    
    # Optimization objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Constraints
    constraints = []
    
    # Constraint 1: Each circle must be inside the unit square
    for i in range(n):
        x, y = centers[i]

        constraints.append(x <= 1 - radii[i])

    constraints.append(radii >= 0)
    
    # Create and solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS)
    
    # Extract results
