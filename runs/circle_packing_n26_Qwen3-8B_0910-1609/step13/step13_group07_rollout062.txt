from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    grid_spacing = 0.5 / (np.sqrt(3) + 1)  # Spacing for hexagonal grid that fills space efficiently

    grid_points = grid_points[grid_points[:, 0] >= 1e-10]  # Remove points outside unit square
    grid_points = grid_points[grid_points[:, 1] <= 1.0 - 1e-10]
    grid_points = grid_points[grid_points[:, 1] >= 1e-10]
    grid_points = grid_points[:26]  # Take the first 26 valid points

    # Initial configuration using hexagonal grid
    centers = grid_points

    bounds = [(0, 1) for _ in range(num_circles)]

    # Objective: Maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):

        return np.array([dists[i, j] - (radii[i] + radii[j]) for i in range(num_circles) for j in range(i + 1, num_circles)])

    # Optimization
    initial_radii = np.array([0.02] * num_circles)
    try:
        result = minimize(
            objective,
            initial_radii,
            bounds=bounds,

                r = radii[i]
                if (x - r < -1e-12 or x + r > 1 + 1e-12

            return True, "ok"
