import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    num_circles = 26
    
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False, f"Circle {i} is partially outside the unit square"
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_func(radii):
        n = len(radii)
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx**2 + dy**2)
        return np.array([dists[i, j] - (radii[i] + radii[j]) for i in range(n) for j in range(i + 1, n)])
    
    def boundary_func(radii):
        return np.array([np.min([centers[i, 0] - radii[i], centers[i, 1] - radii[i], 1 - centers[i, 0] - radii[i], 1 - centers[i, 1] - radii[i]]) for i in range(len(radii))])
    
    # Create initial placement: hexagonal grid with dynamic scaling
    def create_hex_grid(n):
        grid = []
        for row in range(int(np.ceil(np.sqrt(n)))):
            for col in range(int(np.ceil(np.sqrt(n)))):
                x = col / (np.ceil(np.sqrt(n)) - 1) if np.ceil(np.sqrt(n)) - 1 != 0 else 0.5
                y = row / (np.ceil(np.sqrt(n)) - 1) if np.ceil(np.sqrt(n)) - 1 != 0 else 0.5
                grid.append([x, y])
        return np.array(grid)
    
    # Try different configurations
    configurations = [
        create_hex_grid(num_circles),
        np.random.rand(num_circles, 2),
        np.linspace(0, 1, num_circles)[:, np.newaxis] * np.ones((num_circles, 2)),
        np.vstack((np.linspace(0, 1, num_circles), np.linspace(0, 1, num_circles))).T,
        np.random.rand(num_circles, 2) * 2 - 1
    ]
    
    best_sum = -1
    best_centers = None
    best_radii = None
    
    for centers in configurations:
        initial_radii = np.ones(num_circles) * 0.01
        
        try:
            result = minimize(
                objective,
                initial_radii,
                bounds=[(0, 1) for _ in range(num_circles)],
                constraints=[
                    {'type': 'ineq', 'fun': constraint_func},
                    {'type': 'ineq', 'fun': boundary_func}
                ],
                method='SLSQP',
                options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
            )
            
            final_radii = result.x
            is_valid, msg = validate_packing(centers, final_radii)
            if not is_valid:
                continue
                
            sum_radii = np.sum(final_radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers
                best_radii = final_radii
        except Exception:
            pass
    
    return (best_centers, best_radii, best_sum)
