from scipy.optimize import minimize
from scipy.optimize import Bounds

    # This grid uses a staggered pattern (rows alternate offset)
    # Generate centers in hexagonal pattern

    y_offsets = np.linspace(0.05, 0.95, 5)

    centers = []
    for i in range(5):
        for j in range(6):
            if (i + j) % 2 == 0:
                x = x_offsets[j]
            else:
                x = x_offsets[j] - 0.5 * (x_offsets[1] - x_offsets[0])
            y = y_offsets[i]
            centers.append([x, y])
    
    # Add one more circle at the bottom-right of the grid to increase density
    centers.append([0.7, 0.7])  # Strategically placed to maximize possible radius

    # Convert to numpy array
    initial_centers = np.array(centers)

    def obj(radii):
        return -np.sum(radii)  # We minimize the negative sum to maximize the sum

    # Compute distance between two circles
    def dist(i, j):

    def non_overlap_constr(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

    # Constraint for circle staying within the square

        return cons
