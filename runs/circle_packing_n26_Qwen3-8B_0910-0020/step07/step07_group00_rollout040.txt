from scipy.optimize import Bounds

    # Let's start with a staggered hexagonal grid of 6 rows with 5 columns (to give us 26 circles)

    centers = []
    for row in range(6):
        for col in range(5):
            # Even rows (0, 2, 4) are right-aligned, odd rows (1, 3, 5) are left-aligned
            if row % 2 == 0:
                # Right-aligned row
                x = 0.1 + col * 0.2
                y = 0.1 + row * 0.2
            else:
                # Left-aligned row
                x = 0.1 + col * 0.2
                y = 0.1 + row * 0.2
            centers.append([x, y])
    
    # Add a configurable circle at the bottom-right corner for density optimization
    centers.append([0.85, 0.85])  # This will be optimized dynamically

    # Convert to numpy array
    initial_centers = np.array(centers)

    def obj(radii):
        return -np.sum(radii)  # We minimize the negative sum to maximize the sum

    # Compute distance between two circles
    def dist(i, j):
        return np.linalg.norm(initial_centers[i] - initial_centers[j])

    # Constraint for non-overlapping circles
    def non_overlap_constr(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

            ])
        return cons

    constraints = []
