from scipy.optimize import minimize

            centers.append([x, y])

    centers.append([0.5, 0.5])

        return -np.sum(radii)

            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

            return np.array([
                x_i - r - 1e-12,

        return cons

    constraints = []
