from scipy.optimize import minimize
import math

def run_packing():

    # Define a hybrid grid: dense center, with smaller circles along the edges.
    # Central hexagonal grid with rows of decreasing size
    centers = []

    # Generate a denser central grid (hexagonal pattern) with adjustable spacing
    for row in range(5):
        for col in range(5):
            x = (col + 0.5) / (5 - 0.1)  # Slight spacing shift for more dense packing
            y = (row + 0.5) / (5 - 0.1)
            centers.append([x, y])

            if row % 2 == 0:
                x = (col + 0.5) / (5 - 0.1) + 0.15
                y = (row + 0.5) / (5 - 0.1) + 0.2
                centers.append([x, y])
    # Trim to match exactly 26 circles
    centers = np.array(centers[:26])

    # Define cost function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    def calculate_initial_radii(centers):

        initial_radii = np.zeros(num_points)

                if i != j:
                    dx = centers[i, 0] - centers[j, 0]

                initial_radii[i] = (min_dist - 1e-8) / 3
        return initial_radii

    initial_guess = calculate_initial_radii(centers)

    result = minimize(

        initial_guess,

            radii_final[i] = 0

    sum_radii = np.sum(radii_final)

    return centers, radii_final, sum_radii
