import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define a more optimized hybrid packing configuration:
    # - Use hexagonal packing in the central area
    # - Add priority to corners and edges for larger circles
    # - Use a more uneven distribution to improve boundary usage

    # Initial centers arranged in a grid with adjustable spacing
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    n = len(centers)
    if n != 26:
        raise ValueError("Invalid number of circles specified.")

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative of total radii

    # Constraint for non-overlapping circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            return dist - r_i - r_j
        return cons

    # Constraint for boundary conditions
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess for radii with bias towards corners
    initial_guess = np.zeros(n)
    corners = [0, 4, 9, 14, 19, 24]
    initial_guess[corners] = 0.15
    other = [i for i in range(n) if i not in corners]
    initial_guess[other] = 0.1

    # Bounds for radii
    bounds = [(0, 1.0) for _ in range(n)]

    # Run optimization with tighter settings
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': True}
    )

    # Extract final results
    radii_final = result.x
    sum_radii = np.sum(radii_final)

    # Ensure non-negative radii
    radii_final[radii_final < 0] = 0

    return centers, radii_final, sum_radii
