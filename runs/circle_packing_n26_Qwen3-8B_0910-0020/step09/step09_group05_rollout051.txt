        # x[0, n*2) are centers, x[n*2:] are radii

        return -np.sum(radii)  # Maximize by minimizing negative sum

    def constraint_boundary(x):
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]

            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
