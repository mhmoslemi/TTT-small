            grid_positions.append([x, y])

            # Add slightly shifted and smaller positions near the center
            x = (j + 0.3) / (grid_resolution - 1)

            grid_positions.append([x, y])

    def objective(radii):

            r = x[i]

                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12

        return cons

    # Prepare constraints
    constraints = []
