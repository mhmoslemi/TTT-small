from scipy.optimize import minimize

def run_packing():

    base_centers = []

            x = col / 5.0 + 0.1

                x += 0.25

    base_centers.append([0.95, 0.95])  # top-right

    base_centers.append([0.5, 0.6])

    centers = np.array(base_centers[:26])

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):

            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    def constraint_boundary(i):

            r = x[i]
            x_i, y_i = centers[i]
            return np.array([

                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints

    for i in range(num_points):
