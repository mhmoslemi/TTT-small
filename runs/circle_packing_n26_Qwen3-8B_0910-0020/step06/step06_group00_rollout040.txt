import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import random

def run_packing():
    n = 26
    
    # Step 1: Create random cluster centers
    # Generate centers such that they are all inside the unit square
    centers = np.random.rand(n, 2)
    # Shuffle the centers for diversity
    np.random.shuffle(centers)
    
    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of the sum
    
    # Define the distance between two points
    def dist(i, j):
        return np.linalg.norm(centers[i] - centers[j])
    
    # Define constraints for non-overlapping and boundary conditions
    def constraint_non_overlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons
    
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons
    
    # Create all constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_non_overlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
    
    # Initial guess for radii
    initial_guess = np.full(n, 0.05)
    
    # Set bounds for radii (must be non-negative)
    bounds = Bounds(np.zeros(n), np.inf)
    
    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        constraints=constraints,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )
    
    # Extract and finalize the results
    radii_final = result.x
    radii_final[radii_final < 0] = 0  # Ensure non-negative radii
    
    return centers, radii_final, np.sum(radii_final)
