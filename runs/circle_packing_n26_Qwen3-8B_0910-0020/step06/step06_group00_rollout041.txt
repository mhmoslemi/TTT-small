import numpy as np
from scipy.optimize import minimize, Bounds

def run_packing():
    n = 26

    # Define initial centers in a grid-like pattern with spacing
    x_centers = np.linspace(0.1, 0.9, 5)
    y_centers = np.linspace(0.1, 0.9, 5)

    # Create a 5x5 grid with fine spacing
    centers = np.array([[x, y] for y in y_centers for x in x_centers])

    # Add one more circle near the bottom-right corner
    centers = np.vstack([centers, [0.85, 0.85]])

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Distance between two centers
    def dist(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Constraint for non-overlap between circles
    def non_overlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

    # Constraint for boundary (circle cannot exceed square bounds)
    def boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create all constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': non_overlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': boundary(i)})

    # Initial guess for radii (assign roughly equal radii to all circles)
    initial_guess = np.full(n, 0.05)

    # Set bounds for radii (must be non-negative)
    bounds = Bounds(np.zeros(n), np.inf)

    # Run the optimization
    result = minimize(
        objective,
        initial_guess,
        constraints=constraints,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )

    # Final radii
    radii_final = result.x
    radii_final[radii_final < 0] = 0  # Ensure non-negative radii

    return centers, radii_final, np.sum(radii_final)
