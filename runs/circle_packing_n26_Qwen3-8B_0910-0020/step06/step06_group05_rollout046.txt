from scipy.optimize import minimize


def run_packing():

    min_radius = 0.0
    max_radius = 1.0
    center_radius = 0.5

    # Generate random initial positions with spacing to avoid overlap
    def generate_initial_centers(n):
        centers = []
        for _ in range(n):
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            centers.append([x, y])
        return np.array(centers)

    centers = generate_initial_centers(num_points)

    # Binary search over the maximum possible radius by minimizing the sum of radii
    # with constraint that all circles stay within square and do not overlap

    def objective(radii):
        return -np.sum(radii)

    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12

        return cons

    bounds = [(min_radius, max_radius) for _ in range(num_points)]

    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
