from scipy.optimize import minimize
import random

def run_packing():
    num_points = 26

    # Generate multiple random initial layouts
    random_seeds = [i for i in range(10)]
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    for seed in random_seeds:
        random.seed(seed)
        # Generate random centers within the unit square
        centers = np.random.rand(num_points, 2)
        # Scale center positions to [0,1] and slightly perturb to avoid symmetry
        centers += np.random.normal(0, 1e-3, centers.shape)
        centers = np.clip(centers, 0, 1)

        def objective(radii):
            return -np.sum(radii)

        def constraint_nonoverlap(i, j):
            def cons(x):
                r_i = x[i]
                r_j = x[j]
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                return dist - r_i - r_j
            return cons

                r = x[i]

                    1.0 - x_i - r - 1e-12,

        result = minimize(
