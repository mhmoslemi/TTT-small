from scipy.optimize import minimize

    positions = []

    for i in range(5):
        for j in range(5):
            x = (j + 0.5) / 5
            y = (i + 0.5) / 5
            positions.append([x, y])
    
    # Add boundary circles
    positions.append([0.95, 0.95])  # Top right corner
    positions.append([0.05, 0.05])  # Bottom left corner

    # Add extra circles at corners of the grid
    positions.append([0.05, 0.95])  # Bottom right corner
    positions.append([0.95, 0.05])  # Top left corner

    positions.append([0.5, 0.5])  # Center

    # Trim to 26 circles
    centers = np.array(positions[:26])

    # Define cost function to maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([

                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12

        return cons

    # Initial guess based on inter-circle distances
    # Regular grid spacing
    initial_radii = np.full(num_points, 0.05)

    # Optimize
    result = minimize(
