from scipy.optimize import minimize

def run_packing():

    # Define hexagonal lattice with spacing parameter
    spacing = 0.2
    hex_grid = []
    for row in range(13):
        for col in range(13):
            x = (col + 0.5 * (row % 2)) * spacing
            y = row * spacing * np.sqrt(3)/2
            if 0 <= x <= 1 and 0 <= y <= 1:
                hex_grid.append([x, y])

    # Add additional circles in corners and edges
    corners = [
        [0.1, 0.1],
        [0.1, 0.9],
        [0.9, 0.9],
        [0.9, 0.1]
    ]
    for point in corners:
        hex_grid.append(point)

    centers = np.array(hex_grid)

    # Initial radius guess
    initial_guess = np.full(len(centers), 0.05)

    # Define constraints
    def constraint_non_overlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_non_overlap(i, j)})

    def objective(radii):
