import cvxpy as cp

def run_packing():

    # This layout is inspired by known best packings and optimized with known properties
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],

        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    n = len(centers)
    radii = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []
    for i in range(n):
        # Each circle must be inside the unit square
        x, y = centers[i]
        constraints.append(x - radii[i] >= 0)
        constraints.append(x + radii[i] <= 1)
        constraints.append(y - radii[i] >= 0)
        constraints.append(y + radii[i] <= 1)

    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlapping constraint
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(warm_start=True)

    # Extract the centers and radii
    optimized_radii = radii.value
    return centers, optimized_radii, np.sum(optimized_radii)
