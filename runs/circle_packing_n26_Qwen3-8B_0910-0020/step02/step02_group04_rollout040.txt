import random
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

                total_radius -= 1000
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                total_radius -= 1000

                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    total_radius -= 1000
        return total_radius

    def mutate(solution: np.ndarray) -> np.ndarray:
        """Randomly perturb circle positions and radii."""

        radii = solution[n * 2:]

        new_radii = np.copy(radii)

    num_generations = 100

            parent2 = parents[len(parents) // 2 + i]
