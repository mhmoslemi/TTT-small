import scipy.optimize as optimize
import matplotlib.pyplot as plt
from typing import Tuple

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial guess: hybrid packing of 5x5 grid of circles
    n = 26
    centers = np.random.rand(n, 2) * 0.95 + 0.025  # small perturbation to avoid degenerate starting point

    # Use bounds for each circle to be within the square

    # Define the variables: we optimize the radius and position of each circle.

    def objective(x):
        # x has shape (2*n, ) => first n for positions, last n for radii
        # Reconstruct centers and radii
        positions = x[:n*2].reshape(n, 2)
        radii = x[n*2:].reshape(n)

        # Basic objective: maximize the sum of radii
        return -np.sum(radii)  # Negative for minimization

    def constraints(x):
        positions = x[:n*2].reshape(n, 2)
        radii = x[n*2:].reshape(n)

        # Constraint 1: all circles are within the unit square (with tolerance)
        for i in range(n):
            x_center, y_center = positions[i]
            r = radii[i]
            if (x_center - r < -1e-12 or x_center + r > 1 + 1e-12
                    or y_center - r < -1e-12 or y_center + r > 1 + 1e-12):
                return np.inf  # Violation of constraint

        # Constraint 2: circles must not overlap
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((positions[i] - positions[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf  # Violation of constraint

        # No overlap and in square, so return 0
        return 0

    # Bounds: all positions between 0 and 1, all radii non-negative
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1)])  # x and y coordinates for each circle

    initial_guess = np.concatenate([centers.reshape(-1), np.ones(n) * 0.05])  # initial guess for radii

        x0=initial_guess,
        method='SLSQP',
