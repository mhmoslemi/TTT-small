import numpy as np
import scipy.optimize as opt
import warnings

def run_packing():

    # Define function to compute total sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum = maximize sum

    constraints = []

                x1, y1 = centers[i]
                x2, y2 = centers[j]
                d = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)

                return d - r1 - r2
