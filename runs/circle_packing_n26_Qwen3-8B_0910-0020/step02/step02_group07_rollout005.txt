import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():

    # We will have 5 rows (y-coordinates) and 6 columns (x-coordinates), 5 rows give 5*6=30 circles
    # But we only need 26 => So remove 4 circles from the grid
    # Grid: x = i * 1/6, y = j * 1/5
    centers = []
    radii = []

    # Create a 5x6 grid (total 30 circles) and then remove 4 from the end
    for j in range(5):  # y-direction
        for i in range(6):  # x-direction
            x = i / 6
            y = j / 5
            # for now, use the same radius for all
            r = 0.05 * (1 - (j + i) / 11)  # adjust radius based on position
            centers.append([x, y])
            radii.append(r)

    # Remove last 4 circles to make it 26
    centers = centers[:-4]
    radii = radii[:-4]

    # Optimization: Minimize energy to refine positions and radii
    def objective(x):
        # x has shape (26 * 3) => [x1, y1, r1, x2, y2, r2, ...]
        centers_opt = x[:26*2].reshape((26, 2))
        radii_opt = x[26*2:].reshape(26)

        # Energy: distance between circles minus radii (to avoid overlaps)
        # and distance to boundary to encourage edge placement
        energy = 0.0

        # Boundary constraint: distance to edge
        for i in range(26):
            x, y = centers_opt[i]
            r = radii_opt[i]
            dist_left = x - r
            dist_right = 1 - x - r
            dist_bottom = y - r
            dist_top = 1 - y - r
            energy += max(0, 1 - (dist_left + dist_right + dist_bottom + dist_top)) ** 2

        # Pairwise distance between circles
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_opt[i] + radii_opt[j] - 1e-8:
                    energy += (radii_opt[i] + radii_opt[j] - dist) ** 2

        return energy

    # Bounds: each center is in [0,1], and each radius is non-negative
    bounds = []
    for _ in range(26):
        bounds.extend([(0, 1), (0, 1), (0, 1)])  # x, y, r for each circle

    x0 = np.concatenate([np.array(centers), np.array(radii)])

    result = opt.minimize(

        x0=x0,
        bounds=bounds,
        method="L-BFGS-B",
        options={"maxiter": 100, "ftol": 1e-8}
    )

    # Extract final centers and radii

    if not success:

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
