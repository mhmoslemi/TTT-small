import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    def objective(radii, centers):
        return -np.sum(radii)  # We maximize by minimizing negative sum

    def constraint_overlap(i, j, centers, radii):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - (radii[i] + radii[j])  # >= 0 to enforce no overlap

    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r >= -1e-12) and (x + r <= 1 + 1e-12) and (y - r >= -1e-12) and (y + r <= 1 + 1e-12)

    # Initial guess: cluster of centers
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],  # Top edge
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],  # Middle of top
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],  # Bottom
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],  # Middle of bottom
        [0.5, 0.9],  # Center of bottom
    ])

    n = len(centers)
    radii = np.full(n, 0.05)

    # Define bounds for x and y
    bounds = Bounds(np.zeros(n * 2), np.ones(n * 2))

    # Define constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            def overlap(i, j):
                return {'type': 'ineq', 'fun': lambda x: constraint_overlap(i, j, centers, x[:n])}
            constraints.append(overlap(i, j))

        def inside(i):
            return {'type': 'ineq', 'fun': lambda x: constraint_boundary(i, centers, x[:n])}
        constraints.append(inside(i))

    # Optimization
    result = minimize(
        fun=lambda x: objective(x, centers),
        x0=np.concatenate((centers.flatten(), radii)),
        bounds=bounds,
        constraints=constraints
    )

    # Extract final positions and radii
    final_pos = result.x[:n * 2].reshape(n, 2)
    final_r = result.x[n * 2:]
    sum_r = np.sum(final_r)

    return final_pos, final_r, sum_r
