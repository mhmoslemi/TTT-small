from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():

    # Initial grid-based configuration for 26 circles
    # This is a 5x5 grid with an extra circle maneuvered to maximize space
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Grid-based initial placement
    for k in range(26):
        row = k // 5
        col = k % 5
        x = col / 4.0 + 0.05  # small offset to avoid boundaries
        y = row / 4.0 + 0.05
        centers[k] = [x, y]

    # Initial radius assuming equal spacing
    initial_r = 0.1  # Adjust this based on initial grid spacing
    radii[:] = initial_r

    def objective(x):
        # x contains both centers and radii in flattened format
        centers_flat = x[:26*2].reshape(26, 2)
        radii_flat = x[26*2:].reshape(26)
        return -np.sum(radii_flat)  # Minimize negative sum to maximize sum

    # Constraints:
    # 1. Each circle must be inside the unit square
    # 2. Each circle must not overlap with any other circle
    # 3. Non-negativity of radii

    # Define bounds: each center (x, y) must be in [0, 1], and each radius ≥ 0
    bounds_flat = []
    for i in range(26):
        bounds_flat.append((0.0, 1.0))  # x-coordinate
        bounds_flat.append((0.0, 1.0))  # y-coordinate
        bounds_flat.append((0.0, 1.0))  # radius
    bounds = Bounds(bounds_flat)

    constraints = []

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
