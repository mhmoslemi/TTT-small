def generate_hexagonal_grid(n, side_length, offset_x, offset_y):
    """
    Generate centers for a hexagonal grid of n circles.
    """
    rows = int(np.sqrt(n))
    cols = int(np.ceil(n / rows))
    
    centers = []
    x_start = offset_x
    y_start = offset_y
    
    for i in range(rows):
        for j in range(cols):
            x = x_start + j * (2 * side_length)
            y = y_start + i * (np.sqrt(3) * side_length)
            if i % 2 == 1:
                x += side_length
            centers.append([x, y])
    centers = np.array(centers)
    return centers
