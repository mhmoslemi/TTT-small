import numpy as np
import cvxpy as cp

def run_packing():
    # Initial grid-based placement of 26 circles forming a 5x5 grid
    N = 5  # Number of rows
    M = 5  # Number of columns
    grid_centers = np.array([[i / (N - 1), j / (M - 1)] for i in range(N) for j in range(M)])
    radii = np.array([0.05] * 25)
    # Start with a small radius to allow optimization

    # Define the problem
    num_circles = len(grid_centers)
    centers = cp.Variable((num_circles, 2))
    radii = cp.Variable((num_circles,))

    # Constraints
    # 1. All circles must be within the unit square with some tolerance
    constraints = [
        cp.norm(centers[i] - np.array([0, 0])) <= 1 + 1e-4 for i in range(num_circles)
    ]
    # 2. Taxi-cab distance to edges to ensure full inclusion
    for i in range(num_circles):
        x, y = centers[i]
        constraints.append(x - radii[i] <= 1 + 1e-4)
        constraints.append(x + radii[i] >= -1e-4)
        constraints.append(y - radii[i] <= 1 + 1e-4)
        constraints.append(y + radii[i] >= -1e-4)

    # 3. No two circles overlap
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append(cp.norm(centers[i] - centers[j]) >= radii[i] + radii[j] - 1e-4)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)

    # Extract the optimal centers and radii
    centers_opt = centers.value
    radii_opt = radii.value

    # Ensure all radii are non-negative
    radii_opt[radii_opt < 0] = 0

    # Return the results
    return (centers_opt, radii_opt, np.sum(radii_opt))
