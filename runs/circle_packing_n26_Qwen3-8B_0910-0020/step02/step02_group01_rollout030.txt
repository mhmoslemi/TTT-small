import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26
    # Define hexagonal grid with 5 rows and 5 columns
    centers = []
    for i in range(5):
        for j in range(5):
            # Base hexagonal grid
            x = i / 4
            y = j / 4
            # Add small shifts to break symmetry and allow better packing at edges
            x_shift = 0.25 * np.sin(np.pi / 2 * i)
            y_shift = 0.25 * np.sin(np.pi / 2 * j)
            x += x_shift
            y += y_shift
            centers.append([x, y])
    centers = np.array(centers)

    # Add circles in four corners
    corners = np.array([
        [0.0, 0.0],
        [1.0, 0.0],
        [1.0, 1.0],
        [0.0, 1.0]
    ])
    centers = np.vstack([centers, corners])

    # Define cost function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    # Define constraint for non-overlapping circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for circles to be within [0,1]^2
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints list
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess for radii: small radii for all circles
    initial_guess = np.full(num_points, 0.03)

    # Run optimization with adjusted parameters for better convergence
    result = minimize(
        objective,
        initial_guess,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 800, 'ftol': 1e-9, 'disp': False}
    )

    # Extract final results
    centers_final = centers
    radii_final = result.x
    sum_radii = np.sum(radii_final)

    # Ensure radii are non-negative
    radii_final[radii_final < 0] = 0

    return centers_final, radii_final, sum_radii
