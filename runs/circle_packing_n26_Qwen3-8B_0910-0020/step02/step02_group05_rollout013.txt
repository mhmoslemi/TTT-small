import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing():
    # Define initial positions for boundary and corner circles
    # 4 corners, 4 top edges, 4 bottom edges, 4 left/right edges (excluding corners)
    corner_circles = [
        (0.0, 0.0), (1.0, 0.0), (1.0, 1.0), (0.0, 1.0)
    ]
    edge_circles = [
        (0.0, 0.25), (0.0, 0.75),
        (0.25, 0.0), (0.75, 0.0),
        (1.0, 0.25), (1.0, 0.75),
        (0.25, 1.0), (0.75, 1.0)
    ]

    # Generate a hexagonal grid in the center
    hex_grid = []
    for i in range(4):
        for j in range(4):
            if (i + j) % 2 == 0:
                x = 0.25 + i * 0.25
                y = 0.25 + j * 0.25
                hex_grid.append((x, y))

    # Combine all initial positions
    initial_centers = np.array(corner_circles + edge_circles + hex_grid)
    n = len(initial_centers)

    # Optimization function to maximize the sum of radii
    def objective(x):
        # x is a flatten vector of all coordinates and radii
        coords = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        assert len(radii) == n, "Mismatch in number of radii"
        
        # Enforce boundary constraints
        for i in range(n):
            x_i, y_i = coords[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return -np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf

        # Enforce non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((coords[i] - coords[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf

        # Return negative of sum of radii (since we minimize)
        return -np.sum(radii)

    # Initial guess: start with all circles having radius 0.05
    initial_guess = np.r_[initial_centers.flatten(), np.full(n, 0.05)]

    # Perform optimization
    result = minimize(
        objective,
        initial_guess,
        method="L-BFGS-B",
        bounds=[
            (0.0, 1.0) for _ in range(2 * n)

    radii = result.x[2 * n:]

    # Validate the packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
