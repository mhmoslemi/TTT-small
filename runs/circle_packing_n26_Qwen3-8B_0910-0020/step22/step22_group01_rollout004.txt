from scipy.optimize import minimize
from scipy.optimize import Bounds
from itertools import combinations

    centers = np.zeros((n, 2))

    for i in range(n):

        col = i % 5
        x = 0.1 + 0.2 * col
        y = 0.1 + 0.2 * row
        if row % 2 == 1:
            x += 0.05
        centers[i, 0] = x
        centers[i, 1] = y

    # Objective function: maximize sum of radii
    def objective(x):

        radii = x[n * 2:]
        return -np.sum(radii)

    def boundary_constraints(x):

        radii = x[n * 2:]
        cons = []
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            cons.append(x_i - r_i)
            cons.append(1.0 - x_i - r_i)
            cons.append(y_i - r_i)
            cons.append(1.0 - y_i - r_i)
        return np.array(cons)

    def non_overlap_constraints(x):

            r_i, r_j = radii[i], radii[j]
