import numpy as np
import random
from scipy.optimize import minimize


def run_packing():
    # Initial layout: grid-based placement
    n = 26
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 4
        col = i % 4
        centers[i] = [col / 4, row / 4]

    # Function to compute objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Function to compute pairwise distance
    def pairwise_distance(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Function to compute constraints (non-overlapping and boundary)
    def constraint_overlap(i, j, radii):
        dist = pairwise_distance(i, j)
        return dist - radii[i] - radii[j]

    # Build constraints for optimization
    constraints = []
    for i in range(n):
        # Circle must be within the square [0,1]^2
        def boundary(i, radii):
            x, y = centers[i]
            return (x - radii[i] >= -1e-12) and (x + radii[i] <= 1 + 1e-12) and \
                   (y - radii[i] >= -1e-12) and (y + radii[i] <= 1 + 1e-12)
        # Add boundary constraint
        constraints.append({'type': 'ineq', 'fun': lambda radii, i=i: boundary(i, radii) * 100})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda radii, i=i, j=j: constraint_overlap(i, j, radii) + 1e-12})

    # Initial guess for radii: all equal
    init_radii = np.full(n, 0.05)

    # Run optimization
    result = minimize(
        objective,
        init_radii,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1) for _ in range(n)],
        options={'disp': False}
    )

    # Extract results
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
