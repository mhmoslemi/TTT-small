from scipy.optimize import minimize
from scipy.optimize import LinearConstraint, Bounds

def create_hexagonal_grid(n):
    """
    Generate a hexagonal grid of points within [0, 1]^2.
    """
    row = int(np.ceil(np.sqrt(n)))
    col = int(np.ceil(n / row))
    grid = []
    for i in range(row):
        for j in range(col):
            x = j / (col - 1) if col > 1 else 0.5
            y = i / (row - 1) if row > 1 else 0.5
            grid.append((x, y))
    return np.array(grid[:n])

def run_packing():

    centers = create_hexagonal_grid(n)

    bounds = Bounds(np.zeros(n), np.inf)

    # Distance constraints between all pairs of circles (non-overlapping)
    dist_constraints = []

            def dist_constraint(r):
                dist = np.linalg.norm(centers[i] - centers[j])
                return dist - (r[i] + r[j]) + 1e-12
            dist_constraints.append(dist_constraint)

    # Boundary constraints: ensure circles are within the unit square
    boundary_constraints = []
    for i in range(n):
        def left_constraint(r):
            return centers[i, 0] - r[i] + 1e-12
        boundary_constraints.append(left_constraint)

            return 1 - centers[i, 0] - r[i] + 1e-12
        boundary_constraints.append(right_constraint)
