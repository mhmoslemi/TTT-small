import scipy.optimize as optimize

def run_packing():
    # Initial grid-based approximation (hexagonal packing)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    radius_density = 0.2  # Initial estimate for trial radii

    # Generate a hexagonal grid with horizontal spacing sqrt(3)/2 * r, vertical spacing r
    row_count = int(np.sqrt(n))
    col_count = int(np.ceil(n / row_count))
    
    # Fill the grid in hexagonal fashion
    for i in range(row_count):
        for j in range(col_count):
            idx = i * col_count + j
            if idx >= n:
                break
            x = j + 0.5 * i / col_count
            y = i / row_count
            centers[idx] = [x, y]

    def objective(radii):
        return -np.sum(radii)  # Negative because we minimize

    # Define constraints:
    # 1. Each circle must be within the square
    # 2. No overlap between circles
    # 3. Radii must be non-negative

    def constraint_1(radii):
        return np.sum(radii)  # For optimization

    def constraint_2(radii):
        result = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            result[i] = x - radii[i]
            result[i] = min(result[i], y - radii[i])
            result[i] = min(result[i], 1 - x - radii[i])
            result[i] = min(result[i], 1 - y - radii[i])
        return result

        result = np.zeros(n)

                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx**2 + dy**2

        return result

    # Initial guess for radii
    initial_radii = np.full(n, radius_density)

    bounds = [(0, 1)] * n

    result = optimize.minimize(

        bounds=bounds,
        constraints=cons,

    radii = result.x
