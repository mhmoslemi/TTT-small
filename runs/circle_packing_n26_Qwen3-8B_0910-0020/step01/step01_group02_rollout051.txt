import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def compute_distance(p1, p2):
    """Compute Euclidean distance between two points."""
    return np.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

def check_overlap(centers, radii, tolerance=1e-6):
    """Check for overlaps between all pairs of circles."""
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            d = compute_distance(centers[i], centers[j])
            if d < radii[i] + radii[j] - tolerance:
                return True
    return False

def check_boundaries(centers, radii, tolerance=1e-6):
    """Check if any circle extends beyond the unit square."""
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x - r < -tolerance or x + r > 1 + tolerance or y - r < -tolerance or y + r > 1 + tolerance:
            return True
    return False

def fitness_function(positions, radii, tolerance):
    """Fitness function for optimization, minimizing the total radius to maximize the sum."""
    if check_overlap(positions, radii, tolerance):
        return -np.inf  # Penalize overlap
    if check_boundaries(positions, radii, tolerance):
        return -np.inf  # Penalize boundary violation
    return -np.sum(radii)  # Maximize sum of radii

def run_packing():

    corners = np.array([
