def generate_regular_grid(n: int, spacing: float) -> np.ndarray:
    """Generate a regular grid of n circles in a square."""
    ratio = np.sqrt(n)
    rows = int(ratio)
    cols = int(ratio)
    if rows * cols < n:
        rows += 1
        cols = n // rows + 1
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing
            centers.append([x, y])
    return np.array(centers)
