import math
from scipy.optimize import minimize

def run_packing():

    num_circles = 26

    initial_radii = np.full(num_circles, 0.1)

    def objective(radii):
        return -sum(radii)  # Minimize negative sum to maximize

    def constraint_no_overlap(radii, centers):

            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_square(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
