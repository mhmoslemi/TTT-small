import scipy.optimize as optimize

    cols = 5

    max_radius = 0.15

    centers = []
    radii = []
    for row in range(rows):
        for col in range(cols):
            x = col * spacing
            y = row * spacing + (col % 2) * spacing / 2
            if x + max_radius <= 1 and y + max_radius <= 1:
                centers.append([x, y])
                radii.append(max_radius)
    
    # Add more circles to reach 26

        centers.append([x, y])

    def objective(radii):
        return -sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append((radii[i], 0, 1))  # Radii must be ≤ 1

                cons.append((dist - (radii[i] + radii[j]), 1e-12, np.inf))  # Non-overlapping
        return cons
    
    # Optimize with constraints

    cons = constraints(radii)
