import numpy as np
import scipy.optimize as opt
import cvxpy as cp
from itertools import combinations

def run_packing():
    # Initial approximate positions for 25 circles in a 5x5 grid
    grid = np.array([
        [i/4, j/4] for i in range(5) for j in range(5)
    ])
    # Add one more circle at the end to make 26
    centers = np.vstack((grid, [0.5, 0.5]))
    radii = np.zeros(26)

    # Helper function: distance between two points
    def dist(p1, p2):
        return np.sqrt(np.sum((p1 - p2) ** 2))

    # CVXPY optimization problem
    n = 26
    x = cp.Variable((n, 2), non_negative=True)
    r = cp.Variable(n, non_negative=True)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    # All circles must be inside [0,1]^2
    for i in range(n):
        constraints.append(x[i, 0] <= 1)
        constraints.append(x[i, 0] >= 0)
        constraints.append(x[i, 1] <= 1)
        constraints.append(x[i, 1] >= 0)

    # No overlaps between circles
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(dist(x[i], x[j]) >= r[i] + r[j] - 1e-6)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, eps=1e-6)

    # Extract results
    centers = x.value
    radii = r.value

    # Final check for nan or negative values
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values found in centers or radii")

    if np.any(radii < 0):
        raise ValueError("Negative radii found")

    # Final check for boundary constraints
    for i in range(n):
        if not (0 <= centers[i, 0] <= 1) or not (0 <= centers[i, 1] <= 1):
            raise ValueError(f"Circle {i} is outside the unit square")

    # Final check for overlaps
    for i, j in combinations(range(n), 2):
        d = dist(centers[i], centers[j])
        if d < radii[i] + radii[j] - 1e-6:
            raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
