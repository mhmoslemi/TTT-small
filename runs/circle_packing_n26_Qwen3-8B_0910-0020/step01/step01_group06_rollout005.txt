import numpy as np
import scipy.optimize as optimize

def hexagonal_grid(n, r):
    """Generates a hexagonal grid of n circles with radius r"""
    centers = []
    row = 0
    for i in range(n):
        col = i % 6
        row = i // 6
        x = col * (2 * r) + r
        y = row * (np.sqrt(3) * r)
        centers.append((x, y))
    return np.array(centers)

        return r

    def constraint_R(x):

        if R < 0:
            return -np.inf
        return R

    def constraint_center(x):
        r, R = x
        return r * 2 <= 1

    # Define all constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_r},
        {'type': 'ineq', 'fun': constraint_R},
        {'type': 'ineq', 'fun': constraint_center}
    ]

    # Initial guess
    x0 = [0.06, 0.1]

    result = optimize.minimize(

    centers = np.vstack((centers_small, centers_large))
    radii = np.array([optimal_r] * 25 + [optimal_R])

    return centers, radii, total_sum
