import numpy as np
import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26

    # Define variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: maximize total sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must be entirely inside [0, 1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # No overlapping between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt(cp.sum_squares(centers[i] - centers[j]))
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='SCS', eps=1e-6)

    # Extract results
    centers_val = centers.value
    radii_val = radii.value
    sum_radii = np.sum(radii_val)

    return (centers_val, radii_val, sum_radii)
