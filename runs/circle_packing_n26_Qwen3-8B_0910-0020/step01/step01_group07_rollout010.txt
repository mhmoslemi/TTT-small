import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Step 1: Define grid-based initial positions
    n_circles = 26
    grid_size = int(np.sqrt(n_circles))  # 5x5 grid with one extra circle
    centers = np.zeros((n_circles, 2))

    # Place circles in a grid with some offset to use corners better
    offset = 0.1
    for i in range(n_circles):
        row = i // (grid_size - 1)
        col = i % (grid_size - 1)
        x = col / (grid_size - 1) + offset
        y = row / (grid_size - 1) + offset
        centers[i] = [x, y]

    # Step 2: Initial guess for radii
    radii = np.full(n_circles, 0.05)

    # Step 3: Define objective function (maximize sum of radii)
    def objective(x):
        radii = x[:n_circles]
        return -np.sum(radii)  # Negative for minimization

    # Step 4: Define constraints
    def constraint_func(x):
        radii = x[:n_circles]
        centers = x[n_circles:]
        centers = centers.reshape((n_circles, 2))
        # Check boundary constraints
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1  # Violate boundary
        # Check non-overlap
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1  # Violate non-overlap
        return 0

    # Step 5: Setup optimization
    bounds = [(0, 1) for _ in range(n_circles)] + [(0, 1) for _ in range(n_circles * 2)]

    # Initial guess
    x0 = np.concatenate((radii, centers.flatten()))

    # Run optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_func}
        ],
        method='SLSQP'
    )

    # Extract final result
    radii = result.x[:n_circles]
    centers = result.x[n_circles:].reshape((n_circles, 2))

    # Validate final packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
