import numpy as np
import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26

    # Define grid-based hexagonal packing
    rows = int(np.sqrt(n))
    cols = n // rows + 1 if n % rows != 0 else n // rows  # Ensure row count
    
    # Define spacing and starting points
    x_start = 0.1
    y_start = 0.1
    x_spacing = 0.5
    y_spacing = 0.5

    # Create a grid of centers in a hexagonal arrangement
    centers = np.zeros((n, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if r % 2 == 0:
                x = x_start + c * x_spacing
            else:
                x = x_start + (c - 0.5) * x_spacing
            y = y_start + r * y_spacing
            if x < 1 - 1e-12 and y < 1 - 1e-12:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1

    # Define optimization problem
    radii = cp.Variable(n)
    epsilon = 1e-6

    # Objective: maximize the minimum radius
    obj = cp.Maximize(cp.Minimize(radii))

    # Constraints
    constraints = []
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i] >= -epsilon)
        constraints.append(centers[i, 0] + radii[i] <= 1 + epsilon)
        constraints.append(centers[i, 1] - radii[i] >= -epsilon)
        constraints.append(centers[i, 1] + radii[i] <= 1 + epsilon)

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist >= radii[i] + radii[j] - epsilon)

    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS)

    # Extract results
    if not prob.success:
        raise ValueError("Optimization failed, check constraints")

    # Filter out unassigned indices (ensure n circles are used)
    if idx < n:
        raise ValueError(f"Only {idx} circles placed; need 26")

    # Validate the packing
    valid, msg = validate_packing(centers[:n], radii.value)
    if not valid:
        raise ValueError(f"Packing is invalid: {msg}")

    # Compute total sum of radii
    sum_radii = np.sum(radii.value)

    return centers[:n], radii.value, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
