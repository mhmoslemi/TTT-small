import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26

    # Define hexagonal grid points with multiple layers (approx 4 layers)
    max_layer = 3
    centers = []

    # Central circle
    centers.append([0.5, 0.5])

    # Layer 1: 6 surrounding circles in hexagonal arrangement
    for angle in np.linspace(0, 2*np.pi, 6, endpoint=False):
        x = 0.5 + np.cos(angle)
        y = 0.5 + np.sin(angle)
        centers.append([x, y])

    # Layer 2: 12 surrounding circles in a larger hexagonal pattern
    for layer in range(1, max_layer + 1):
        for angle in np.linspace(0, 2*np.pi, 6, endpoint=False):
            x = 0.5 + (1 + layer) * np.cos(angle)
            y = 0.5 + (1 + layer) * np.sin(angle)
            centers.append([x, y])

    # Prune to get exactly 26 points (reshape with careful selection)
    centers = np.array(centers)
    centers = centers[:26]

    # Now perform optimization
    def objective(radii):
        return -np.sum(radii)  # Maximize total sum of radii

    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess: sizes spread out
    initial_guess = np.full(num_points, 0.1)

    result = minimize(
        objective,
        initial_guess,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False}
    )

    # Extract results
    radii_final = result.x
    radii_final[radii_final < 0] = 0
    return centers, radii_final, np.sum(radii_final)
