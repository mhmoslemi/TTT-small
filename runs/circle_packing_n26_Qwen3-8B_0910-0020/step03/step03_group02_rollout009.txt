from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    # Create a grid-like arrangement with 5 rows and 5 columns
    centers = []
    for i in range(5):
        for j in range(5):
            # Create a small perturbation in x and y to avoid perfect grid overlap
            x = i / 4 + 0.05 * np.sin(np.pi * i / 4)
            y = j / 4 + 0.05 * np.sin(np.pi * j / 4)
            centers.append([x, y])
    # Add the 26th point at the center, slightly perturbed
    centers.append([0.5, 0.5])

    centers = np.array(centers)

    # Define objective function to minimize negative of total area
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)

    # Define non-overlapping constraint
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define boundary constraint
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-8,
                1.0 - x_i - r - 1e-8,
                y_i - r - 1e-8,
                1.0 - y_i - r - 1e-8
            ])
        return cons

    # Create constraints list
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):

    result = minimize(

        bounds=bounds,
