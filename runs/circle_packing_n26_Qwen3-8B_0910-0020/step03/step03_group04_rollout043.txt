from scipy.optimize import minimize

def run_packing():

    centers = []

    def hex_grid(n):
        grid = []
        for i in range(n):
            for j in range(n):
                x = i / (n - 1) * 0.9
                y = j / (n - 1) * 0.9
                grid.append((x, y))
        return np.array(grid)

    # Generate a hexagonal grid for 26 points
    # Start with a base grid of 6 rows (26 points)
    # Adjusted to fit in the unit square with a small offset for better packing
    def hexagonal_layout(num_points):
        points = []
        for row in range(6):
            for col in range(6):
                angle = 2 * np.pi / 6 * row
                x = 0.5 + 0.3 * np.cos(angle) + 0.05 * np.cos(angle + np.pi / 6) * col
                y = 0.5 + 0.3 * np.sin(angle) + 0.05 * np.sin(angle + np.pi / 6) * col
                points.append((x, y))
        points = np.array(points)
        # Trim to 26 points
        points = points[:26]
        return points

    centers = hexagonal_layout(num_points)

    # Define cost function to maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):

            return np.array([

                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):

    initial_radii = np.full(num_points, 0.1)

    result = minimize(

        constraints=constraints,
