import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26
    # Generate initial configuration using hexagonal grid with some perturbations
    centers = []
    for i in range(5):
        for j in range(5):
            x = i / 4 + np.random.uniform(-0.05, 0.05)
            y = j / 4 + np.random.uniform(-0.05, 0.05)
            centers.append([x, y])
    
    # Add one more circle at center with perturbation
    centers.append([0.5 + np.random.uniform(-0.05, 0.05), 0.5 + np.random.uniform(-0.05, 0.05)])
    centers = np.array(centers)

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum of radii

    # Define non-overlap constraint for circle pairs
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define boundary constraint for each circle
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create all constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess for radii
    initial_guess = np.full(num_points, 0.05)

    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )

    # Ensure radii are non-negative
    radii_final = result.x
    radii_final[radii_final < 0] = 0

    # Extract final results
    centers_final = centers
    sum_radii = np.sum(radii_final)

    return centers_final, radii_final, sum_radii
