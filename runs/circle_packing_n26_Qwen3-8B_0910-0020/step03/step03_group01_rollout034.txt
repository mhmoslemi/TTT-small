from scipy.optimize import minimize

def run_packing():

    # This creates a structured, spread-out initial configuration
    spacing = 0.2

            x = i * spacing
            y = j * spacing
            # Ensure no circle is placed outside the square
            if x < 1 and y < 1:
                centers.append([x, y])

    def objective(radii):

            return dist - r_i - r_j
        return cons
