from scipy.optimize import minimize
from scipy.optimize import Bounds

    centers = []
    for i in range(5):
        for j in range(5):
            # Offset rows by 0.2 to create a hexagonal packing
            if i % 2 == 0:
                x = i / 4 + 0.15 * np.sin(np.pi * i / 4)
            else:
                x = i / 4 + 0.15 * np.sin(np.pi * i / 4 - np.pi / 2)

            centers.append([x, y])

    centers.append([0.5, 0.5])
    centers = np.array(centers)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum of radii

            r_i = x[i]
            r_j = x[j]

            return dist - r_i - r_j
        return cons

    # Define constraint for boundary (circle must be within [0,1]^2)

                1.0 - x_i - r - 1e-12,
