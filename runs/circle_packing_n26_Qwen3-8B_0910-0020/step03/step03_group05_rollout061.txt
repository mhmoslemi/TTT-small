def run_packing():

    def generate_hex_grid(n):
        grid = []
        for i in range(n):
            for j in range(n):
                x = i / (n - 1)
                y = j / (n - 1)
                if i % 2 == 0:
                    x += 0.1 * np.sin(np.pi / 2 * i)
                    y += 0.1 * np.sin(np.pi / 2 * j)

        if np.isnan(centers).any() or np.isnan(radii).any():
            return -1e9  # Invalid individual

        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-4 or x + r > 1 + 1e-4:
                return -1e9
            if y - r < -1e-4 or y + r > 1 + 1e-4:
