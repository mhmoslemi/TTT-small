def run_packing():

    # We'll start with a 5x5 grid with a tiny perturbation to avoid perfect symmetry
    centers = []
    for i in range(5):
        for j in range(5):
            x = i / 4 + 0.05 * np.sin(np.pi * i / 4)
            y = j / 4 + 0.05 * np.sin(np.pi * j / 4)
            centers.append([x, y])

    centers.append([0.5, 0.5])
    centers = np.array(centers)

            r_j = x[j]

            return dist - r_i - r_j
