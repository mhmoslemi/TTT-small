from scipy.optimize import minimize

def run_packing():

    centers = []
    for i in range(5):
        for j in range(5):
            x = i / 4.0 + 0.05 * np.sin(np.pi * i)
            y = j / 4.0 + 0.05 * np.sin(np.pi * j)
            centers.append([x, y])

    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Define cost function to maximize total radius sum
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of total radius sum

            r_j = x[j]

            return dist - r_i - r_j
        return cons

        def cons(x):
