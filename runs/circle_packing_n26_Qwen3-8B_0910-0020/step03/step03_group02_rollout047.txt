import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    num_points = 26
    # Create a hexagonal grid with 5 rows and 4 columns for a total of 20 points
    # Add 6 additional points at the corners and edges
    centers = []
    
    # Hexagonal grid points
    for row in range(5):
        for col in range(4):
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            # Scale to unit square [0,1]^2
            x = x / (4 - 1)  # scale x to [0,1]
            y = y / (5 - 1)  # scale y to [0,1]
            centers.append([x, y])
    
    # Add corner and edge points
    centers.append([0, 0.5])     # left center
    centers.append([1, 0.5])     # right center
    centers.append([0.5, 0])     # bottom center
    centers.append([0.5, 1])     # top center
    centers.append([0.25, 0.75]) # top-left
    centers.append([0.75, 0.75]) # top-right
    centers.append([0.25, 0.25]) # bottom-left
    centers.append([0.75, 0.25]) # bottom-right
    
    # Keep only the first 26 points
    centers = centers[:26]
    centers = np.array(centers)

    # Define cost function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    # Define constraint for non-overlapping circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for boundary (circle must be within [0,1]^2)
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints list
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})

    # Initial guess for radii
    initial_guess = np.full(num_points, 0.05)

    bounds = Bounds(lb=0.0, ub=np.inf)

    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False}
    )

    # Extract final results
    centers_final = centers

    sum_radii = np.sum(radii_final)

    # Ensure radii are non-negative
    radii_final[radii_final < 0] = 0

    return centers_final, radii_final, sum_radii
