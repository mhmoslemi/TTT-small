import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26
    # Create a hybrid configuration: most circles in hexagonal grid, with additional circles near boundaries
    centers = []
    
    # First row
    centers.append([0.2, 0.2])
    centers.append([0.5, 0.2])
    centers.append([0.8, 0.2])
    
    # Second row
    centers.append([0.3, 0.5])
    centers.append([0.5, 0.5])
    centers.append([0.7, 0.5])
    
    # Third row
    centers.append([0.4, 0.8])
    centers.append([0.6, 0.8])
    
    # Add circles near corners
    centers.append([0.1, 0.1])
    centers.append([0.9, 0.1])
    centers.append([0.1, 0.9])
    centers.append([0.9, 0.9])
    
    # Add center circle
    centers.append([0.5, 0.5])
    
    # Define cost function to maximize total radius sum
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of total radius sum

    # Define constraint for non-overlapping circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for circle within [0,1]^2
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints list
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess for radii
    initial_guess = np.full(num_points, 0.1)

    # Run optimization
    result = minimize(

        constraints=constraints,

    centers_final = np.array(centers)

    return centers_final, radii_final, sum_radii
