from scipy.optimize import Bounds

def run_packing():

    centers = []
    for i in range(5):

            x = i / 4.0
            y = j / 4.0
            if i % 2 == 0:
                x += 0.1 * np.sin(np.pi * i / 4)
            if j % 2 == 0:
                y += 0.1 * np.sin(np.pi * j / 4)
            centers.append([x, y])

    centers.append([0.5, 0.5])
