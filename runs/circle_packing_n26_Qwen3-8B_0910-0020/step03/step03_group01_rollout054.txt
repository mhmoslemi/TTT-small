from scipy.optimize import minimize

        x = 0.5 + 0.3 * np.cos(angle)
        y = 0.5 + 0.3 * np.sin(angle)
        centers.append([x, y])

        radius = np.random.uniform(0.2, 0.4)

        y = 0.5 + radius * np.sin(angle)
        centers.append([x, y])

    def objective(radii):
        return -np.sum(radii)

            r_j = x[j]

        return cons
