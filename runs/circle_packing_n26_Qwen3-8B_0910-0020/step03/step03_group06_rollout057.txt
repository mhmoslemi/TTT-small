from scipy.optimize import minimize
from itertools import combinations

    num_points = 26

            x = col / 4.0 + 0.1 * np.random.uniform(-1, 1)

            centers.append([x, y])

    centers = np.array(centers)
    centers = np.clip(centers, 0, 1)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of total radius sum

            r_i = x[i]
            r_j = x[j]

            return dist - r_i - r_j
        return cons

    # Define boundary constraint
    def constraint_boundary(i):

            r = x[i]

            return np.array([

                1.0 - x_i - r - 1e-12,
