from scipy.optimize import minimize

def run_packing():

    num_points = 26
    centers = []

    # Define number of rows and columns for hexagonal packing
    rows = 6
    cols = 5

    row_spacing = 1.0 / (rows - 1)
    col_spacing = 1.0 / (cols - 1)

    # Generate points in a hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * col_spacing
            y = i * row_spacing
            # Introduce small perturbations to break symmetry
            perturbation = np.random.uniform(-0.05, 0.05, 2)
            centers.append([x + perturbation[0], y + perturbation[1]])

    centers = np.array(centers)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum of radii

    def constraint_nonoverlap(i, j):

            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define boundary constraint for each circle
    def constraint_boundary(i):

            r = x[i]

            return np.array([

                1.0 - x_i - r - 1e-12,

        return cons
