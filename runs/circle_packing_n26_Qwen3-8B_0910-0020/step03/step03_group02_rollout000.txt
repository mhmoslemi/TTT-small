import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    num_points = 26

    # Generate perturbed hexagonal grid with 5 rows and 5 columns, with a small bit of randomness
    centers = []
    for row in range(5):
        for col in range(5):
            x = col / 4 + np.random.normal(0, 0.05)
            y = row / 4 + np.random.normal(0, 0.05)
            # Ensure x and y are within [0, 1]
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            centers.append([x, y])
    # Add one central circle
    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Define cost function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    # Define constraint for non-overlapping circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for boundary (circle must be within [0,1]^2)
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints list
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess for radii
    initial_guess = np.full(num_points, 0.05)

    # Set bounds to ensure radii are non-negative
    bounds = Bounds(lb=0.0, ub=np.inf)

    # Use Nelder-Mead method for more robust global optimization
    result = minimize(

        initial_guess,
        bounds=bounds,
        constraints=constraints,

    centers_final = centers

    radii_final[radii_final < 0] = 0
