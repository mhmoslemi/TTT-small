    for i in range(len(centers)):

        dy = random.uniform(-0.02, 0.02)
        centers[i][0] += dx
        centers[i][1] += dy

    # Define objective function for optimization: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints: non-overlap and boundary
    def constraint_non_overlap(radii, centers):
        n = len(centers)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                r1 = radii[i]
                x2, y2 = centers[j]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                constraints.append(dist >= r1 + r2 - 1e-8)
        return constraints

    # Optimization
    opt_result = opt.minimize(

    )

    # Get final centers and radii
    final_centers = np.array(centers)
    final_radii = opt_result.x

                return False, f"Circle {i} has negative radius {radii[i]}"

            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
