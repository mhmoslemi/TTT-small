import numpy as np
import random
from scipy.optimize import minimize

def run_packing():
    # Step 1: Initial grid placement
    grid_size = 5
    grid = np.zeros((grid_size, grid_size, 2))
    for i in range(grid_size):
        for j in range(grid_size):
            grid[i, j, 0] = i / (grid_size - 1)
            grid[i, j, 1] = j / (grid_size - 1)
    
    centers = grid.reshape(-1, 2)
    radii = np.full(centers.shape[0], 0.05)
    
    # Step 2: Optimization to maximize sum of radii
    def objective(x):
        nonlocal centers, radii
        # x contains the perturbations for center positions and radii
        delta_centers = x[:len(centers)*2].reshape(-1, 2)
        delta_radii = x[len(centers)*2:]
        new_centers = centers + delta_centers
        new_radii = radii + delta_radii
        
        # Ensure radii are non-negative and within square
        new_radii = np.maximum(new_radii, 0)
        for i in range(len(new_radii)):
            x_i, y_i = new_centers[i]
            r_i = new_radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return -np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf
        
        # Ensure no overlaps
        for i in range(len(new_centers)):
            for j in range(i + 1, len(new_centers)):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < new_radii[i] + new_radii[j] - 1e-12:
                    return -np.inf
        
        # Return negative of sum to minimize
        return -np.sum(new_radii)
    
    # Initial guess: small perturbations and small radii
    x0 = np.concatenate([np.random.normal(0, 0.01, len(centers)*2), np.random.normal(0, 0.01, len(centers))])
    
    # Optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=[(-0.1, 0.1) for _ in range(len(x0))])
    
    # Get final centers and radii
    delta_centers = result.x[:len(centers)*2].reshape(-1, 2)
    centers += delta_centers
    radii = np.clip(radii + result.x[len(centers)*2:], 0, 1)
    
    # Final validation
    assert np.all(centers >= 0) and np.all(centers <= 1)
    assert np.all(radii >= 0)
    
    # Return the result
    return centers, radii, np.sum(radii)
