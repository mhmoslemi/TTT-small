import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    """Packs 26 circles in a unit square to maximize the sum of radii using a hybrid approach combining
    hexagonal packing, boundary optimization, and gradient-based refinement."""
    
    # Base grid dimensions
    grid_rows, grid_cols = 5, 5  # Allows 25 circles in a grid
    grid = np.linspace(0, 1, grid_rows + 1)
    grid_x, grid_y = np.meshgrid(grid, grid)
    grid_points = np.column_stack((grid_x.flatten(), grid_y.flatten()))
    grid_radii = np.full(grid_points.shape[0], 0.05)  # Initial guess for radii

    # Add one more circle in the center (to reach 26)
    center = np.array([0.5, 0.5])
    grid_points = np.vstack((grid_points, center))
    grid_radii = np.append(grid_radii, 0.1)

    # Function to compute distances between all pairs
    def compute_distances(points):
        n = points.shape[0]
        dist = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dist[i, j] = np.linalg.norm(points[i] - points[j])
                dist[j, i] = dist[i, j]
        return dist

    # Objective function: maximize sum of radii
    def objective(radii, points):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint: circle i cannot overlap with circle j
    def constraint_overlap(radii, points, i, j):
        return np.linalg.norm(points[i] - points[j]) - (radii[i] + radii[j])

    # Prepare initial constraint for all pairs
    n = grid_points.shape[0]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(('ineq', constraint_overlap, radii=np.full(n, 0.0), points=grid_points, i=i, j=j))
    
    # Prepare bounds: radii >= 0
    bounds = [(0, 1)] * n

    # Use Sequential Least Squares Programming (SLSQP) to optimize
    result = optimize.minimize(
        fun=objective,
        x0=grid_radii,
        args=(grid_points,),
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract the result
    radii = result.x
    centers = grid_points
    sum_radii = np.sum(radii)

    # Final validation
    assert np.all(radii >= 0), "Negative radius found"
    assert np.all(centers >= 0) and np.all(centers <= 1), "Center out of bounds"

    return centers, radii, sum_radii
