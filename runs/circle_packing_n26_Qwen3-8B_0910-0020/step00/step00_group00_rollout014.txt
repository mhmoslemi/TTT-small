import numpy as np
from scipy.optimize import minimize
import math

def compute_hexagonal_grid(n_circles, max_x=1.0, max_y=1.0, spacing_ratio=1.0):
    radius = 0.05  # Initial guess for radius
    centers = []
    for i in range(n_circles):
        row = i // 2
        col = i % 2
        x = col * (2 * radius * spacing_ratio)
        y = row * (np.sqrt(3) * radius * spacing_ratio)
        if x > max_x:
            x = max_x - 0.01
        if y > max_y:
            y = max_y - 0.01
        centers.append((x, y))
    return np.array(centers), np.full(n_circles, radius)

def cost_function(x, n_circles):
    centers = x[:n_circles * 2].reshape((n_circles, 2))
    radii = x[n_circles * 2:]
    sum_radii = np.sum(radii)
    
    # Boundary check
    for i in range(n_circles):
        x_i, y_i = centers[i]
        r_i = radii[i]
        if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
            y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
            return -np.inf
    
    # Pairwise distance check
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-12:
                return -np.inf
    
    return sum_radii

def run_packing():
    n_circles = 26
    initial_centers, initial_radii = compute_hexagonal_grid(n_circles)
    
    # Flatten centers and radii for optimization
    x0 = np.concatenate((initial_centers.ravel(), initial_radii))
    
    result = minimize(
        fun=cost_function,
        x0=x0,
        args=(n_circles,),
        method='L-BFGS-B',
        options={'ftol': 1e-12, 'maxiter': 500}
    )
    
    centers = result.x[:n_circles * 2].reshape((n_circles, 2))
    radii = result.x[n_circles * 2:]
    
    return centers, radii, np.sum(radii)
