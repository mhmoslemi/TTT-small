def run_packing():

    n = 26

    def generate_hex_grid(n):
        rows = int(np.ceil(n / 3))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i + 0.5
                if i % 2 == 0:
                    x = j + 0.5
                centers.append((x, y))
        return np.array(centers)

    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Negative because we minimize

    # Constraints: Non-overlapping and within unit square
    def constraint_nonoverlap(radii, centers):
        n = len(centers)
        def constraint(x):
            # x is an array of all radii

                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx ** 2 + dy ** 2)
                    r1 = x[i]
                    r2 = x[j]
                    if dist < r1 + r2 - 1e-8:
                        return 0
            return 1
        return constraint

    def constraint_bounds(radii):
        return np.all(radii >= 0)

    # Create bounds for the optimization variables (radii)
    bounds = [(0, 0.5) for _ in range(n)]

    # Optimization
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[
            constraint_nonoverlap(radii, centers),
            {'type': 'ineq', 'fun': constraint_bounds},

        method='SLSQP',

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
