from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial packing: 24 circles in hexagonal pattern with 4 columns and 6 rows
    # Initial row spacing, column spacing, and margin for bottom and top
    rows = 6
    cols = 4
    margin = 0.1
    R = 0.1  # Initial estimate for radius

    # Initial placement of 24 circles in hexagonal grid
    centers = []
    radii = []
    for row in range(rows):
        for col in range(cols):
            x = col * (1 - 2 * margin) / (cols - 1)
            y = row * (1 - 2 * margin) / (rows - 1)
            centers.append([x, y])
            radii.append(R)

    # Add 2 corner circles
    for corner in [(0, 0), (1, 1)]:
        x, y = corner
        centers.append([x, y])
        radii.append(0.05)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define optimization variables: centers and radii
    # Flatten the centers to a 1D vector of size 26*2 = 52
    x0 = np.concatenate((centers.flatten(), radii))

    # Objective function: maximize total radius
    def objective(x):
        flat_centers = x[:52].reshape(26, 2)
        flat_radii = x[52:]
        return -np.sum(flat_radii)  # Minimize negative sum (i.e., maximize sum)

        return np.concatenate([flat_centers[:, 0], flat_centers[:, 1]])

    # Define bounds: (x, y) ∈ [0,1], (r ≥ 0)
    bounds = []
    for _ in range(26):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius

    # Place all constraints
    def constraint(x):
        flat_centers = x[:52].reshape(26, 2)
        flat_radii = x[52:]
        # Check if all centers are within [0,1]^2
        in_bounds = np.all((flat_centers >= 0) & (flat_centers <= 1))
        # Check if all radii are non-negative
        non_negative = np.all(flat_radii >= 0)
        if not in_bounds or not non_negative:
            return np.inf  # Penalize invalid positions or radii

        distances = cdist(flat_centers, flat_centers)
        overlaps = np.triu(distances < flat_radii[:, np.newaxis] + flat_radii[np.newaxis, :])
        if overlaps.any():

        return 0  # Satisfied

    result = minimize(objective, x0, bounds=bounds, constraints={"type": "ineq", "fun": constraint})

    # Extract the result
    optimized_centers = result.x[:52].reshape(26, 2)
    optimized_radii = result.x[52:]
    total_radius = np.sum(optimized_radii)

    return (optimized_centers, optimized_radii, total_radius)
