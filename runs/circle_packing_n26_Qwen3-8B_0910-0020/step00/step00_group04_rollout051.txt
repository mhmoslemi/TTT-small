import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

    # Create a 5x5 grid of centers with spacing ~0.2
    grid_centers = []
    grid_radii = []
    for i in range(5):

            y = j * 0.2 + 0.1
            if (i + j) % 2 == 0:
                grid_centers.append((x, y))
                grid_radii.append(0.05)
    
    # Add additional circles near corners and edges
    extra_centers = []
    extra_radii = []
    
    # Add large circles in corners
    for x, y in [(0.1, 0.1), (0.9, 0.1), (0.1, 0.9), (0.9, 0.9)]:
        extra_centers.append((x, y))

    # Add more circles in corners with smaller radii
    for x, y in [(0.2, 0.2), (0.8, 0.2), (0.2, 0.8), (0.8, 0.8)]:
        extra_centers.append((x, y))
        extra_radii.append(0.1)

    # Convert to numpy arrays
    centers = np.array(grid_centers + extra_centers)
    radii = np.array(grid_radii + extra_radii)

    # Function to calculate the objective
    def objective(params):
        new_centers = params[:centers.shape[0] * 2].reshape(centers.shape[0], 2)
        new_radii = params[centers.shape[0] * 2:].reshape(radii.shape[0])
        return -np.sum(new_radii)

    # Function to calculate constraints
    def constraints_func(params):
        new_centers = params[:centers.shape[0] * 2].reshape(centers.shape[0], 2)
        new_radii = params[centers.shape[0] * 2:].reshape(radii.shape[0])

        # Check boundary constraints for each center
        for i in range(new_centers.shape[0]):
            x, y = new_centers[i]
            r = new_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                return False
            if y - r < -1e-12 or y + r > 1 + 1e-12:
                return False

        # Check overlap constraints for each pair
        for i in range(new_centers.shape[0]):
            for j in range(i + 1, new_centers.shape[0]):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < new_radii[i] + new_radii[j] - 1e-12:

        return True

    # Optimization
    initial_params = np.concatenate([centers.flatten(), radii])
