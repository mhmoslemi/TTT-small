import scipy.optimize as opt

    cols = 5

    # Add 3 more circles at corners or edges
    # For initial estimate, use uniform radius
    r_initial = 0.09
    # Initial centers in a hexagonal grid
    centers = []
    for i in range(rows):

            x = j * (1 + 1e-6) + 0.5 * (0.5 * (i % 2))
            y = i * (np.sqrt(3) * r_initial) + 0.5 * r_initial
            centers.append((x, y))
        # Add extra circles for corners and edges
    # Add 3 more circles to fill gaps
    # For example, add one at (0.2, 0.2), one at (0.8, 0.8), and one at (0.5, 0.5)
    centers.append((0.2, 0.2))
    centers.append((0.8, 0.8))
    centers.append((0.5, 0.5))
    centers = np.array(centers)
    radii = np.full(n, r_initial)

    # Define the optimization problem
    def objective(x):
        return -np.sum(x[1:])  # Maximize sum of radii by minimizing negative of sum

    def constraint1(x):
        # Ensure all circles are inside the unit square
        for i in range(n):
            if x[i] < 0 or x[i] > 1:
                return -1
        return 0

    def constraint2(x):
        # Ensure non-overlapping between all circle pairs
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i * 2] - x[j * 2]
                dy = x[i * 2 + 1] - x[j * 2 + 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < x[i + 1] + x[j + 1] - 1e-8:

        return 0

    bounds = []

    cons = [
