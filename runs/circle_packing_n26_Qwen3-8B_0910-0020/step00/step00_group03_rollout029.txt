import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def objective(radii: np.ndarray) -> float:
        return -np.sum(radii)  # Minimize negative sum (maximize sum)

    def constraint_overlap(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:
        n = centers.shape[0]
        dist_matrix = distance_matrix(centers, centers)
        min_dist = np.minimum(dist_matrix, np.finfo(np.float64).max)
        for i in range(n):
            for j in range(i + 1, n):
                if min_dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return np.array([1.0])
        return np.array([0.0])

    def constraint_boundary(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return np.array([1.0])
        return np.array([0.0])

    # Initial estimate for centers based on a hexagonal grid
    def initial_centers(n: int) -> np.ndarray:
        grid_density = 0.6
        grid_rows = int(np.ceil(np.sqrt(n)))
        grid_cols = int(np.ceil(n / grid_rows))
        centers = np.zeros((n, 2))
        idx = 0
        for r in range(grid_rows):
            for c in range(grid_cols):
                if idx >= n:
                    break
                x = c * grid_density
                y = r * grid_density
                if r % 2 == 1:
                    x += grid_density / 2
                centers[idx] = [x, y]
                idx += 1
        return centers

    n = 26
    centers = initial_centers(n)
    radii = np.full(n, 0.05)

    # Define the problem
    bounds = [(0, 1) for _ in range(2 * n)]  # for x and y of each center
    bounds += [(0, 0.5) for _ in range(n)]   # for radii

    # Optimization setup
    result = minimize(

        x0=np.concatenate([centers.flatten(), radii]),

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):

            if dist < radii[i] + radii[j] - 1e-12:

    return True, "ok"
