import numpy as np
import cvxpy as cp

def run_packing():
    # Step 1: Initial grid positions using hexagonal packing
    # We will generate 5 rows with increasing number of circles
    # Rows will alternate between even and odd patterns to fit in square

    num_rows = 5
    num_cols = 6

    # Initial positions (hexagonal grid)
    centers = []
    for i in range(num_rows):
        row = []
        for j in range(num_cols):
            x = j + 0.5
            y = i + np.sqrt(3)/2 * j
            row.append((x, y))
        centers.append(row)
    
    # Flatten the list of positions
    centers = np.array([np.array([x, y]) for row in centers for x, y in row])
    n = len(centers)

    # Create variables
    radii = cp.Variable(n)
    positions = cp.Variable((n, 2))

    # Objective: Maximize the total area
    max_sum_radii = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []
    # Boundary constraints: circles must fit within [0, 1]^2
    for i in range(n):
        constraints.append(positions[i, 0] - radii[i] >= -1e-12)
        constraints.append(positions[i, 0] + radii[i] <= 1 + 1e-12)
        constraints.append(positions[i, 1] - radii[i] >= -1e-12)
        constraints.append(positions[i, 1] + radii[i] <= 1 + 1e-12)

    # No overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((positions[i, 0] - positions[j, 0])**2 + (positions[i, 1] - positions[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Set up the problem
    prob = cp.Problem(max_sum_radii, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    # Extract final positions and radii
    final_centers = np.array(positions.value)
    final_radii = np.array(radii.value)

    # Sum the radii
    total_sum = np.sum(final_radii)

    return (final_centers, final_radii, total_sum)
