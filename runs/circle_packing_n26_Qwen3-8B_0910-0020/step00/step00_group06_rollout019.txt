import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Define initial positions for a hybrid packing (hexagonal on the interior, boundary-aligned for outer circles)
    # 24 circles are arranged in a hexagonal grid inside the square
    # 2 circles are placed at the top and bottom corners
    # 0.1 spacing between centers (to allow for some clearance)
    num_interior = 24
    num_corners = 2  # top and bottom corners
    centers = np.zeros((num_interior + num_corners, 2))
    
    # Hexagonal grid inside the square
    rows = int(np.sqrt(num_interior))
    cols = int(np.ceil(num_interior / rows))
    spacing = 0.1
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            if x + spacing / 2 > 1 or y + spacing / 2 > 1:
                continue
            centers[i * cols + j] = [x, y]
    
    # Place two circles at the top and bottom corners
    centers[-2] = [0, 1]
    centers[-1] = [1, 0]
    
    # Initial radius guess (small, to be optimized)
    radii = np.full(num_interior + num_corners, 0.05)
    
    # Define the constraints: non-overlapping and within square
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum (maximize sum)

    def constraint_non_overlapping(i, j):
        def constraint(radii):
            if i == j:
                return 0
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - (radii[i] + radii[j])
        return constraint

    cons = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            cons.append({'type': 'ineq', 'fun': constraint_non_overlapping(i, j)})
    
    # Ensure circles are inside the square
    def constraint_inside(i):
        def constraint(radii):
            x, y = centers[i]
            return (x - radii[i]) >= -1e-12 and (x + radii[i]) <= 1 + 1e-12 and (y - radii[i]) >= -1e-12 and (y + radii[i]) <= 1 + 1e-12
        return constraint

    for i in range(len(centers)):
        cons.append({'type': 'ineq', 'fun': constraint_inside(i)})

    # Run the optimization
    bounds = Bounds([0.0] * len(radii), [1.0] * len(radii))
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=cons,
        bounds=bounds
    )
    
    if not result.success:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
