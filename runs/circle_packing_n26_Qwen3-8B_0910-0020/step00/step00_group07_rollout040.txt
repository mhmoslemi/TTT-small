import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():

    # Parameters:
    # - n: Number of circles (26)
    # - r: Initial radius for all circles (initialized as 0.05)
    # - n_rows: Number of rows (based on hexagonal arrangement)
    # - column_count: Number of circles per row
    
    n = 26
    r = 0.05
    n_rows = 4
    columns = [5, 5, 6, 6]  # Distribute circles unevenly across rows to fit 26

    # Initial placement: hexagonal grid
    def generate_hex_grid(n_rows, columns):
        centers = []
        for row in range(n_rows):
            for col in range(columns[row]):
                x = col + 0.5 * (row % 2)
                y = row * np.sqrt(3) / 2
                centers.append([x, y])
        return np.array(centers)

    # Create the initial centers
    centers = generate_hex_grid(n_rows, columns)

    # Define the objective function: maximize the sum of the radii
    def objective(x):
        return -x[0]  # Negative because we minimize

    # Define constraints:
    # 1. Distance between every pair of circles must be at least the sum of their radii
    # 2. Each circle must be contained within the unit square
    # 3. Radii must be nonnegative

    def constraint_1(x):
        radii = x[1:]
        dists = np.sqrt((centers[:, 0, np.newaxis] - centers[:, 0, np.newaxis].T) ** 2 + 
                       (centers[:, 1, np.newaxis] - centers[:, 1, np.newaxis].T) ** 2)
        min_dist = np.min(dists)
        return min_dist - (radii[np.newaxis, :] + radii[np.newaxis, :].T).min() - 1e-6

    def constraint_2(x):
        radii = x[1:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1e6
        return 0

    def constraint_3(x):
        return x[1:].min() >= 0

    # Initial guess: first value is 0 (to multiply with the sum), followed by initial radii
    x0 = np.concatenate([np.array([0.0]), np.full(n, r)])

    ]

    # Run optimization
    result = minimize(objective, x0, bounds=bounds, constraints=constraints, method='SLSQP')

    # Extract packed radii and centers
    radii = result.x[1:]
    centers = np.array(centers)

    radii[radii < 0] = 0

    sum_radii = np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
