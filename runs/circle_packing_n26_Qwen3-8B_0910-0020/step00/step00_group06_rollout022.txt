from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Initial placement: use hexagonal packing with 19 circles in the center
    # and 7 circles placed near the corners

    # Hexagonal grid parameters
    grid_rows = 5
    grid_cols = 4
    spacing = 0.45  # Will be adjusted to fit within the square

    # Generate initial positions for 19 circles in hexagonal grid
    centers = []
    radii = []

    # Hexagonal grid in the center of the square
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = spacing * (j + 0.5 * i)
            y = spacing * (i + 0.5 * j)
            centers.append([x, y])
            radii.append(0.05)  # Placeholder

    # Add corner circles
    corner_positions = [
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9], 
        [0.4, 0.1], [0.6, 0.1], [0.5, 0.9]
    ]
    for pos in corner_positions:
        centers.append(pos)
        radii.append(0.05)

    # Convert to NumPy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Optimize using coordinate descent
    def objective(x):
        # x contains all parameters: first half is x positions, second half is y positions
        # Reconstruct centers and radii
        num_circles = len(centers)
        x_params = x[:num_circles]
        y_params = x[num_circles:]
        all_centers = np.vstack((x_params.reshape(-1, 1), y_params.reshape(-1, 1))).T
        all_radii = radii.copy()

        # Penalize for negative radii
        negative_penalty = np.sum(np.maximum(0, -all_radii))

        # Total penalty
        return boundary_penalty + overlap_penalty + negative_penalty
