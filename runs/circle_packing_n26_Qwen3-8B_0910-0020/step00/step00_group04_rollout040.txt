    extra_circles = np.array([
        [0.95, 0.5],
        [0.9, 0.3],
        [0.9, 0.7]
    ])

    # Combine grid and extra circles
    centers = np.vstack([grid, extra_circles])

    # Initial radii: start with small values
    radii = np.ones(len(centers)) * 0.05

    # Objective function: maximize total area (which is proportional to sum of radii^2)
    def objective(params):
        # params: flattened array of centers and radii
        # Unflatten into centers and radii
        flat_centers = params[:len(centers)*2]
        flat_radii = params[len(centers)*2:]
        centers_opt = flat_centers.reshape((len(centers), 2))
        radii_opt = flat_radii

        # Objective: maximize the sum of (pi * r^2), so we minimize -sum(r^2)
        sum_r_squared = np.sum(radii_opt ** 2)
        return -sum_r_squared

    # Constraints for non-overlap and bounding
    def non_overlap_constraint(params):
        flat_centers = params[:len(centers)*2]
