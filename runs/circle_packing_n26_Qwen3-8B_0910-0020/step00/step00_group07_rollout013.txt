import numpy as np

def run_packing():
    # Define the number of rows and columns
    rows = 5
    cols = [6, 5, 6, 5, 6]  # Number of circles in each row (hexagonal pattern)
    total_circles = sum(cols)
    
    # Initial spacing parameters
    horizontal_spacing = 0.7  # Adjust based on desired density
    vertical_spacing = 0.7
    
    # Compute centers using hexagonal grid
    centers = []
    for row in range(rows):
        col_count = cols[row]
        for col in range(col_count):
            # Determine the x and y positions based on row and column
            x = col * horizontal_spacing
            y = row * vertical_spacing
            
            # Adjust for hexagonal offset (alternate rows)
            if row % 2 == 1:
                x += horizontal_spacing / 2
            centers.append([x, y])
    
    centers = np.array(centers)
    
    # Compute radii
    radii = []
    for i in range(total_circles):
        # Compute the minimum distance to all other points (direct neighbors)
        dists = np.linalg.norm(centers[i] - centers, axis=1)
        min_dist = np.min(dists[dists != 0])
        # Compute maximum radius that fits in the unit square and keeps circles from overlapping
        max_r = min(
            (1 - centers[i, 0]) / 2,
            (centers[i, 0]) / 2,
            (1 - centers[i, 1]) / 2,
            (centers[i, 1]) / 2,
            min_dist / 2
        )
        radii.append(max_r)
    radii = np.array(radii)
    
    sum_radii = np.sum(radii)
    
    # Validate packing
    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError("Invalid packing: " + message)
    
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
