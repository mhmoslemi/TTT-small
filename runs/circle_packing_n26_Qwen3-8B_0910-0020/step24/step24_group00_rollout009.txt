def run_packing():

        return -np.sum(radii_flat)

    def boundary_constraints(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:]
        cons = []
        for i in range(n):

            r_i = radii_flat[i]

            cons.append(1.0 - y_i - r_i)  # 1 - y_i - r_i >= 0
