from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    positions = np.zeros((n, 2))

        row = i // cols

        y = 0.1 + (row * 0.2)

    initial_radii = np.full(n, 0.05)

        return -np.sum(radii)

        return constraints

    def non_overlap_constraints(positions, radii):

        for i in range(n):

                x_j, y_j = positions[j]

                r_j = radii[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)

        return constraints
