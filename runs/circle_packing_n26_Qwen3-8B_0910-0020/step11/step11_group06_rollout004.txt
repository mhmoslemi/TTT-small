                if dist < radii[i] + radii[j] - 1e-12:

        return np.sum(radii),  # Maximize this

    # Define the genetic algorithm
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    # Parameters
    POP_SIZE = 50
    CXPB = 0.5
    MUTPB = 0.2
    MAX_GENERATIONS = 100

    # Initialize toolbox
    toolbox = base.Toolbox()
    # Random float for center coordinates (0.0 to 1.0)
    toolbox.register("center", np.random.uniform, 0.0, 1.0)
    # Random float for radius (0.0 to 1.0)
    toolbox.register("radius", np.random.uniform, 0.0, 1.0)
    # Individual: list of 2*n center coordinates followed by n radii
    toolbox.register("individual", tools.initRepeat, creator.Individual,
                     lambda: [toolbox.center() for _ in range(n*2)] + [toolbox.radius() for _ in range(n)])
    # Population
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    toolbox.register("evaluate", fitness)
    # Crossover
    toolbox.register("mate", tools.cxTwoPoint)

    def mut_ga(individual, indpb):

            if np.random.random() < indpb:
                individual[i] = np.random.uniform(0.0, 1.0)
        return individual,
    toolbox.register("mutate", mut_ga, indpb=0.1)

    toolbox.register("select", tools.selTournament, tournsize=3)

    population = toolbox.population(n=POP_SIZE)
