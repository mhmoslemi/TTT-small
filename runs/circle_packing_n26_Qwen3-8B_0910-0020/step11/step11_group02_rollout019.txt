    n = 26

    # Define a hybrid layout for better packing: central hexagonal cluster with circles near edges

        x = 0.9
        y = 0.5 + i * spacing - 1.5 * spacing / 2
        initial_centers[19 + i, 0] = x
        initial_centers[19 + i, 1] = y
