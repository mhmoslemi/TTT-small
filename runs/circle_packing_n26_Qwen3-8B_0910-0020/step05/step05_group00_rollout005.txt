from scipy.optimize import minimize

def run_packing():

    centers = []

            y += 0.01 * (np.random.rand() - 0.5)
            centers.append([x, y])

    centers.append([0.5, 0.95 + 0.01 * (np.random.rand() - 0.5)])

        return -np.sum(radii)

    def constraint_nonoverlap(i, j):

            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
