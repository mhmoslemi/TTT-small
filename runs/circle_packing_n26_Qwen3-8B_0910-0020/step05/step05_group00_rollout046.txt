from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    # - 20 equally spaced circles on a grid (slightly modified to push towards corners), 
    # - 6 small circles placed in the spaces between the main grid

                main_centres.append([x, y])
        return np.array(main_centres)

    # Create main grid
    main_centres = create_optimized_grid()

    # Place 6 small circles in the gaps between the main circles
    extra_centres = []
    # Add small circles in the top-right, top-left, bottom-right, bottom-left, middle-right, middle-left
    extra_centres.append([0.8, 0.9])
    extra_centres.append([0.2, 0.9])
    extra_centres.append([0.9, 0.1])
    extra_centres.append([0.1, 0.1])
    extra_centres.append([0.7, 0.5])
    extra_centres.append([0.3, 0.5])
    extra_centres = np.array(extra_centres)

    # Combine all centres into a single array
    centers = np.vstack([main_centres, extra_centres])

    # Define objective function to maximize total radius (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Define non-overlap constraint between any two circles
    def nonoverlap_constraint(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    # Define boundary constraint for each circle
    def boundary_constraint(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': nonoverlap_constraint(i, j)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint(i)})

    # Initial radii guess: small values to allow for more fitting
    initial_radii = np.full(num_points, 0.015)

    # Optimization method with bounds, caching, and multi-start
    bounds = Bounds(lb=0, ub=np.inf)

    results = []
    for _ in range(5):  # Run from multiple initial guesses
        result = minimize(
            objective,
            initial_radii,
            constraints=constraints,
            bounds=bounds,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
        )
        results.append(result)

    best_result = max(results, key=lambda x: -x.fun)

    sum_radii = np.sum(radii_final)

    return centers, radii_final, sum_radii
