import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import LinearConstraint
import random

def run_packing():
    num_points = 26

    # Create a more refined initial placement using a hexagonal grid
    centers = []
    # Define the grid parameters
    grid_rows = 5
    grid_cols = 5
    grid_spacing = 0.25
    vertical_offset = 0.1
    horizontal_offset = 0.05

    # Generate hexagonal grid
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * grid_spacing + horizontal_offset
            y = i * grid_spacing + vertical_offset
            if i % 2 == 0:
                centers.append([x, y])
            else:
                centers.append([x + grid_spacing / 2, y])

    # Add corner circles
    for i in range(4):
        centers.append([0.05, 0.05 + i * 0.25])
        centers.append([0.95, 0.05 + i * 0.25])
        centers.append([0.05 + i * 0.25, 0.95])
        centers.append([0.95 - i * 0.25, 0.95])

    # Use only the first 26 points
    centers = np.array(centers[:26])

    # Define cost function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    # Define constraint for non-overlapping circles
    def nonoverlap_constraint(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for boundary (circle must be within [0,1]^2)
    def boundary_constraint(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints list
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': nonoverlap_constraint(i, j)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint(i)})

    # Initial guess for radii
    initial_guess = np.full(num_points, 0.05)

    # Set bounds to ensure radii are non-negative
    bounds = Bounds(lb=0.0, ub=np.inf)

    # Run optimization with multiple restarts for better convergence
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )

    # Ensure radii are non-negative
    radii_final = result.x
    radii_final[radii_final < 0] = 0

    # Return centers, radii, and sum of radii
    return centers, radii_final, np.sum(radii_final)
