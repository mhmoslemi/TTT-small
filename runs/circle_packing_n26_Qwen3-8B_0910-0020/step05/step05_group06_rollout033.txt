import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26

    # Generate a more optimal initial configuration
    # Using a spiral-like pattern for better distribution
    centers = []
    for i in range(num_points):
        angle = 2 * np.pi * i / num_points
        radius = 0.5 * (1 - 0.5 * i / num_points)
        x = 0.5 + radius * np.cos(angle)
        y = 0.5 + radius * np.sin(angle)
        centers.append([x, y])
    
    # Add boundary points
    boundary_points = [
        [0.1, 0.5], [0.5, 0.1], [0.9, 0.5], [0.5, 0.9], [0.1, 0.9], [0.9, 0.1]
    ]
    # Add boundary points to centers, removing duplicates
    unique_centers = []
    seen = set()
    for c in centers + boundary_points:
        key = (round(c[0], 4), round(c[1], 4))
        if key not in seen:
            seen.add(key)
            unique_centers.append(c)
    # Truncate or pad to 26 points
    centers = unique_centers[:26]
    
    # Create array of centers
    centers = np.array(centers)
    
    # Define objective function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area
    
    # Define non-overlapping constraint
    def nonoverlap_constraint(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons
    
    # Define boundary constraint
    def boundary_constraint(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons
    
    # Create constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': nonoverlap_constraint(i, j)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint(i)})
    
    # Initial guess for radii
    initial_guess = np.full(num_points, 0.05)
    
    # Bounds for radii
    bounds = [(0.0, np.inf) for _ in range(num_points)]
    
    # Run the optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
    )
    
    # Extract and clean results
    radii_final = result.x
    radii_final[radii_final < 0] = 0.0
    sum_radii = np.sum(radii_final)
    
    return centers, radii_final, sum_radii
