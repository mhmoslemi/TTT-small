import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26

    # Generate a hexagonal grid with 5 rows: 5, 5, 5, 5, 5
    centers = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5
            y = row + 0.5
            x_normalized = x / 5
            y_normalized = y / 5
            centers.append([x_normalized, y_normalized])

    # Add a circle at the top center to fill the space
    centers.append([0.5, 0.95])

    centers = np.array(centers)

    # Define cost function to maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess: small radii for all circles
    initial_radii = np.full(num_points, 0.05)

    # Run the optimization
    result = minimize(
        objective,
        initial_radii,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 600, 'ftol': 1e-10, 'disp': False}
    )

    # Extract and clean results
    radii_final = result.x
    radii_final[radii_final < 0] = 0  # Enforce non-negativity

    # Ensure all circles are within the unit square
    for i in range(num_points):
        x, y = centers[i]
        r = radii_final[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii_final[i] = 0  # If outside, set radius to zero

    sum_radii = np.sum(radii_final)

    return centers, radii_final, sum_radii
