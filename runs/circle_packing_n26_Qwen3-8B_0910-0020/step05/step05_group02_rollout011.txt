from scipy.optimize import minimize

    x_spacing = 0.2

    # Create centers for hexagonal packing
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * x_spacing
            y = i * y_spacing
            centers.append([x, y])
    centers = np.array(centers)

    # Add corner circles explicitly to fill the corners and edges
    # 4 corner circles and 4 edge circles (top and bottom, left and right)
    corner_circles = [
        [0.0, 0.0],  # bottom-left
        [1.0, 0.0],  # bottom-right
        [0.0, 1.0],  # top-left
        [1.0, 1.0],  # top-right
    ]
    edge_circles = [
        [0.5, 0.0],  # bottom edge center
        [0.5, 1.0],  # top edge center
        [0.0, 0.5],  # left edge center
        [1.0, 0.5],  # right edge center
    ]
    centers = np.vstack([centers, corner_circles, edge_circles])

    # Ensure the centers are adjusted so that inner circles are not overlapping with edges
    centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
    centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)

    # Define the objective function
    def objective(radii):
        return -np.sum(radii)

    # Distance between two points
    def dist(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Constraint functions
    def non_overlap_constraint(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

    def boundary_constraint(i):

                1.0 - x_i - r - 1e-12,

            ])
        return cons
