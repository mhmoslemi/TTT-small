from scipy.optimize import minimize

def run_packing():

    def generate_centers():

        centers = []
        # Horizontal spacing
        hspacing = 0.2

            for j in range(5):
                x = j * hspacing
                y = i * vspacing
                centers.append([x, y])
        # Boundary circles
        boundary = [
            [0.1, 0.5], [0.1, 0.75], [0.1, 0.9],
            [0.25, 0.1], [0.75, 0.1], [0.9, 0.1],
            [0.9, 0.25], [0.9, 0.5], [0.9, 0.75],
            [0.75, 0.9], [0.25, 0.9], [0.1, 0.9],
            [0.5, 0.1], [0.5, 0.25], [0.5, 0.4], [0.5, 0.6], [0.5, 0.75], [0.5, 0.9]
        ]
        # Remove duplicates
        seen = set()
        unique_centers = []
        for c in centers + boundary:
            key = (round(c[0], 4), round(c[1], 4))
            if key not in seen:
                seen.add(key)
                unique_centers.append(c)
        # Truncate or pad to 26 points
        centers = unique_centers[:26]
        return np.array(centers)

    centers = generate_centers()

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of total sum of radii

    # Constraints: non-overlapping and staying within square
    def nonoverlap_constraint(i, j):

            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    def boundary_constraint(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints
    constraints = []
    for i in range(num_points):

    result = minimize(

        bounds=bounds,
        constraints=constraints,

    radii_final = result.x
