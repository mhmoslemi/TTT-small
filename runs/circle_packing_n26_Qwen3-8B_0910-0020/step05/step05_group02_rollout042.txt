import numpy as np
from scipy.optimize import minimize

def run_packing():

    n_grid = 5
    n_total = n_grid * n_grid + 1  # 25 + 1 = 26

    # Initial grid spacing
    x_spacing = 0.18
    y_spacing = 0.18

    # Initialize centers in grid
    centers = []
    for i in range(n_grid):
        for j in range(n_grid):
            x = j * x_spacing
            y = i * y_spacing
            centers.append([x, y])
    # Add a central circle
    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Define the objective function (Maximize sum of radii)
    def objective(radii):
        return -np.sum(radii)

    # Distance function
    def dist(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Constraint for non-overlapping circles
    def non_overlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

    # Constraint for boundary
    def boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Build constraints
    constraints = []
    for i in range(n_total):
        for j in range(i + 1, n_total):
            constraints.append({'type': 'ineq', 'fun': non_overlap(i, j)})

    initial_guess = np.full(n_total, 0.025)

    result = minimize(
        objective,

    )

    return centers_final, radii_final, sum_radii
