from scipy.optimize import minimize
from scipy.optimize import Bounds

                x_coord = x[j] + (x[1] - x[0]) / 2
            y_coord = y[i]

    boundary_points = [

    ]

    # Combine the hexagonal and boundary points
    combined_centers = centers + boundary_points

    # Remove duplicate points
    seen = set()
    unique_centers = []
    for point in combined_centers:
        key = (round(point[0], 4), round(point[1], 4))
        if key not in seen:
            seen.add(key)
            unique_centers.append(point)

    # Truncate or pad to 26 points
    centers = np.array(unique_centers[:26])

    # Define objective function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    def nonoverlapping_constraints(radii):

        constraints = []

                    return dist - r_i - r_j
                constraints.append({'type': 'ineq', 'fun': constraint_func})

    def boundary_constraints(radii):
