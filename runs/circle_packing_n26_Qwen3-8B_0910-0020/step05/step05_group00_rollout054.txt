import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26

    # Define a more efficient hexagonal packing with 5 rows
    centers = []
    radius_density = 0.75  # for hexagonal packing
    row_height = np.sqrt(3) / 2 * radius_density  # vertical spacing in hex grid

    for row in range(5):
        if row % 2 == 0:
            num_cols = 5  # even rows have 5 columns
        else:
            num_cols = 4  # odd rows have 4 columns to fit square

        col_spacing = 1 / (num_cols - 1) if num_cols > 1 else 1
        for col in range(num_cols):
            x = col * col_spacing
            y = row * row_height
            if row == 0:
                y = 0
            centers.append([x, y])

    centers = np.array(centers)

    # Define objective to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap
    def non_overlap_constr(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def boundary_constr(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare non-overlap constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': non_overlap_constr(i, j)})
        # If not last point, add boundary constraint (each circle has its own)
        if i != num_points - 1:
            constraints.append({'type': 'ineq', 'fun': boundary_constr(i)})

    # Initial radii estimate
    initial_radii = np.full(num_points, 0.05)
    for i in range(num_points):
        x, y = centers[i]
        initial_radii[i] = 0.05 * np.min([x, 1 - x, y, 1 - y])

    # Optimization settings
    result = minimize(
        objective,
        initial_radii,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
    )

    # Extract and clean results
    radii_final = result.x
    radii_final[radii_final < 0] = 0

        r = radii_final[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or

            radii_final[i] = 0

    sum_radii = np.sum(radii_final)

    return centers, radii_final, sum_radii
