from scipy.optimize import minimize

def run_packing():

    # Random center initialization with small radius
    centers = np.random.rand(num_points, 2)
    radii = np.random.rand(num_points) * 0.1

    # Cost function to maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint function for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - r_i - r_j
        return cons

    def constraint_boundary(i):

            r = x[i]

                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12

        return cons

    constraints = []
    for i in range(num_points):

    initial_radii = np.random.rand(num_points) * 0.1
