import numpy as np
from scipy.optimize import minimize

def run_packing():

    # Define an initial set of seed points for circle centers
    # These are chosen to be spaced across the square to allow better bound proximity
    # and allow for tighter packing in the center
    initial_centers = []
    for i in range(num_points):
        x = np.random.uniform(0.1, 0.9)
        y = np.random.uniform(0.1, 0.9)
        initial_centers.append([x, y])
    centers = np.array(initial_centers)

    # Define objective function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    # Define constraint for circle overlap
    def overlap_constraint(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for boundary
    def boundary_constraint(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': overlap_constraint(i, j)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint(i)})

    # Initial guess for radii
    initial_guess = np.full(num_points, 0.05)

    # Bounds for radii
    bounds = [(0.0, np.inf) for _ in range(num_points)]

    # Run the optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
    )

    radii_final = result.x

    sum_radii = np.sum(radii_final)
