from scipy.optimize import minimize
from random import random, uniform
from itertools import combinations

def run_packing():

    max_iter = 100
    pop_size = 100
    mutation_rate = 0.1

    best_centers = None
    best_radii = None

    def fitness(centers, radii):
        # Ensure circles are within the square
        for i in range(num_points):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return 0

        # Check circle overlaps
        for i, j in combinations(range(num_points), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                return 0

        # Return the maximum possible sum of radii
        return np.sum(radii)

    # Generate a random initial population
    def generate_individual():
        centers = np.array([[
            random() * 1.0 + 1e-6,  # Ensure small margin to avoid edges
            random() * 1.0 + 1e-6
        ] for _ in range(num_points)])

            x, y = centers[i]
            if x < 0.1:
                centers[i, 0] = 0.1
            elif x > 0.9:
                centers[i, 0] = 0.9

                centers[i, 1] = 0.1
            elif y > 0.9:
                centers[i, 1] = 0.9

        # Generate radii in [0, 0.1]

                best_fitness = fit
                best_centers = centers
                best_radii = radii

        while len(next_population) < pop_size:

            next_population.append((centers, radii))

        population = next_population

    return best_centers, best_radii, best_fitness
