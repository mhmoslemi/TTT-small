from scipy.optimize import differential_evolution
from scipy.optimize import Bounds

def run_packing():
    num_points = 26

    # Generate initial points using hexagonal packing
    def hexagonal_lattice(n):
        row_count = int(np.ceil(np.sqrt(n)))
        col_count = int(np.ceil(n / row_count))
        points = []
        for i in range(row_count):
            for j in range(col_count):
                x = j + 0.5 * i
                y = i + 0.5 * j
                if x <= 1 and y <= 1:
                    points.append([x, y])
        return np.array(points)

    centers = hexagonal_lattice(num_points)

    # Define cost function to maximize total area (sum of π * r_i^2), so minimize negative of total area
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)

    def nonoverlap_constraint(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]

            return dist - r_i - r_j
        return cons

    def boundary_constraint(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r,
                1.0 - x_i - r,
                y_i - r,
                1.0 - y_i - r
            ])

    constraints = []
