from scipy.optimize import minimize

    ])

    def nonoverlap_constraint(i, j):

            r_i = x[i]
            r_j = x[j]
            dx, dy = centers[i] - centers[j]
            return np.sqrt(dx**2 + dy**2) - r_i - r_j
        return cons
