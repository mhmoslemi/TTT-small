    def objective(x):

        radii = x[n * 2:]
        return -np.sum(radii)

    def boundary_constraints(x):
        centers = x[:n * 2].reshape(n, 2)

            r_i = radii[i]
            cons.append(x_i - r_i)  # x - r >= 0
            cons.append(1.0 - x_i - r_i)  # x + r <= 1
            cons.append(y_i - r_i)  # y - r >= 0
            cons.append(1.0 - y_i - r_i)  # y + r <= 1
        return np.array(cons)
