    # Remove duplicate points to maintain exactly 26 circles
    unique_centers = []
    seen = set()
    for point in centers:
        if tuple(point) not in seen:
            seen.add(tuple(point))
            unique_centers.append(point)
    unique_centers = np.array(unique_centers)

    # Trim to 26 centers
    unique_centers = unique_centers[:26]

    # Define initial radius guess based on spacing
    initial_radius_guess = 0.05

    def obj(radii):

            r_i = x[i]

            return dist(i, j) - r_i - r_j
        return cons
