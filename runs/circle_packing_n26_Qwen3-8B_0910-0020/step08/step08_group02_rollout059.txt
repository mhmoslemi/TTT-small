from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # The spacing between points is determined based on the number of circles
    # Using a hexagonal lattice with spacing 's' that fits 26 points
    
    # Start with a square grid and calculate spacing based on sqrt(n)
    grid_size = int(np.sqrt(n)) + 1
    spacing = 1.0 / (grid_size - 1)
    
    # Create a hexagonal grid
    centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i * spacing
            y = j * spacing
            centers.append([x, y])
    
    # Remove duplicates and keep only the first 26
    centers = np.array(centers)
    centers = np.unique(centers, axis=0)
    if len(centers) > 26:
        centers = centers[:26]
    
    # If fewer than 26 points, add points around the corners and edges
    if len(centers) < 26:
        for i in range(len(centers), 26):
            if i % 2 == 0:
                centers = np.vstack([centers, [0.1, 0.1]])
            else:
                centers = np.vstack([centers, [0.9, 0.9]])
    
    # Initial radius guess
    initial_guess = np.full(n, 0.05)

    # Define the objective to maximize the sum of radii
    def obj(radii):
        return -np.sum(radii)  # Minimize the negative sum to maximize the sum

    # Compute distance between two circles
    def dist(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Constraint for non-overlapping circles
    def non_overlap_constr(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

    # Constraint for circle staying within the square
    def boundary_constr(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Collect all constraints
    constraints = []
    for i in range(n):

    # Run the optimization
    bounds = Bounds(np.zeros(n), np.inf)

    result = minimize(

        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )
