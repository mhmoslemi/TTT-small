from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds

def run_packing():

    # Generate hexagonal grid with more well-distributed positions
    def hex_grid(num_circles, density=1.0, seed=42):
        np.random.seed(seed)
        centers = []
        for i in range(num_circles):
            theta = 2 * np.pi * i / num_circles
            r = 0.5 * np.sqrt(i / num_circles) * density
            x = r * np.cos(theta)
            y = r * np.sin(theta)
            centers.append([x + 0.5, y + 0.5])
        return np.array(centers)

    centers = hex_grid(n)

    # Initial radius guess
    initial_guess = np.full(n, 0.05)

    # Define the objective to maximize the sum of radii
    def obj(radii):
        return -np.sum(radii)  # Minimize the negative sum to maximize the sum

    # Compute distance between two circles
    def dist(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Constraint for non-overlapping circles
    def non_overlap_constr(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

    # Constraint for circle staying within the square
    def boundary_constr(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Collect all constraints
    constraints = []
    for i in range(n):

    # Run the optimization
    bounds = Bounds(np.zeros(n), np.inf)

    result = minimize(
        obj,

        bounds=bounds,
        constraints=constraints,
