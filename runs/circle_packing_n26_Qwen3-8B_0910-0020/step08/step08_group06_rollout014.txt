from scipy.optimize import minimize
from scipy.optimize import Bounds

    # and evenly spaced in between

    grid_points = []
    x = np.linspace(0.1, 0.9, 5)
    y = np.linspace(0.1, 0.9, 5)

        for j in range(5):
            if (i + j) % 2 == 0:
                # Even rows: right-aligned

                # Odd rows: left-aligned
                grid_points.append([x[j], y[i]])
    
    # Add points near corners for better utilization of boundary space
    corner_points = [
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9],

    ]
    
    # Combine grid points and corner points
    centers = grid_points + corner_points
    
    # Remove duplicate points
    unique_centers = []
    seen = set()
    for point in centers:
        if tuple(point) not in seen:
            seen.add(tuple(point))
            unique_centers.append(point)
    
    # Trim to 26 unique centers
    unique_centers = np.array(unique_centers)
    unique_centers = unique_centers[:26]
    
    # Define initial radius guess based on spacing in the grid
    initial_radius_guess = 0.05

    # Constraint for non-overlapping circles
    def non_overlap_constr(i, j):

            r_i = x[i]

            return dist(i, j) - r_i - r_j

    # Constraint for circle staying within the square

        return cons
