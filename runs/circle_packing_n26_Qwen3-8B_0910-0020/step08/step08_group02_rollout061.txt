import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # Define the initial placement with strategic spacing to allow for better total packing
    # This uses a dense random point sampling with some clusters and handling for corners
    np.random.seed(42)  # Fixed seed for reproducibility

    centers = []

    # Generate initial placement with some clusters and corner points
    def cluster_points(x, y, radius, count):
        for _ in range(count):
            angle = np.random.uniform(0, 2 * np.pi)
            dist = radius * np.random.uniform(0.2, 0.6)
            x_i = x + dist * np.cos(angle)
            y_i = y + dist * np.sin(angle)
            centers.append([x_i, y_i])

    # Corners and edges
    cluster_points(0.1, 0.1, 0.2, 3)
    cluster_points(0.9, 0.1, 0.2, 3)
    cluster_points(0.1, 0.9, 0.2, 3)
    cluster_points(0.9, 0.9, 0.2, 3)

    # Central cluster
    cluster_points(0.5, 0.5, 0.2, 5)

    # Additional points scattered evenly
    for _ in range(10):
        angle = np.random.uniform(0, 2 * np.pi)
        r = np.random.uniform(0.1, 0.3)
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers.append([x, y])

    # Ensure center points are in [0, 1]^2
    centers = np.array(centers)
    centers = np.clip(centers, 0, 1)

    # Remove duplicates by using a small threshold
    def is_duplicate(p1, p2):
        return np.linalg.norm(p1 - p2) < 1e-3

    unique_centers = []
    for p in centers:
        if not any(is_duplicate(p, p2) for p2 in unique_centers):
            unique_centers.append(p)
    centers = np.array(unique_centers)

    # Trim or pad to n=26
    if len(centers) > 26:
        centers = centers[:26]
    elif len(centers) < 26:
        for _ in range(26 - len(centers)):
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            centers = np.vstack([centers, [x, y]])

    # Initial radius guess
    initial_guess = np.full(n, 0.05)

    # Define the objective to maximize the sum of radii
    def obj(radii):
        return -np.sum(radii)  # Minimize the negative sum to maximize the sum

    # Compute distance between two circles
    def dist(i, j):
        return np.linalg.norm(centers[i] - centers[j])

    # Constraint for non-overlapping circles
    def non_overlap_constr(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            return dist(i, j) - r_i - r_j
        return cons

    # Constraint for circle staying within the square
    def boundary_constr(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Collect all constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': non_overlap_constr(i, j)})
        constraints.append({'type': 'ineq', 'fun': boundary_constr(i)})

    # Run the optimization
    bounds = Bounds(np.zeros(n), np.inf)

    # Optimize with SLSQP
    result = minimize(
        obj,
        initial_guess,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
    )

    # Ensure non-negative radii
    radii_final = result.x
    radii_final[radii_final < 0] = 0

    return centers, radii_final, np.sum(radii_final)
