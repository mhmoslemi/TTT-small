from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    def generate_positions(num_points):

        positions.append([0.5, 0.5])
        # Six surrounding circles

            angle = 2 * np.pi * i / 6
            x = 0.5 + 0.25 * np.cos(angle)
            y = 0.5 + 0.25 * np.sin(angle)
            positions.append([x, y])
        # Remaining 20 circles - place them on edges and corners
        for i in range(7, num_points):
            # Randomly distribute remaining circles on boundaries
            if i % 4 == 0:
                pos = [0.1, np.random.uniform(0.1, 0.9)]
            elif i % 4 == 1:
                pos = [np.random.uniform(0.1, 0.9), 0.1]
            elif i % 4 == 2:
                pos = [0.9, np.random.uniform(0.1, 0.9)]
            elif i % 4 == 3:
                pos = [np.random.uniform(0.1, 0.9), 0.9]
            positions.append(pos)
        return np.array(positions)

    centers = generate_positions(num_points)

    # Objective: Maximize the sum of radii (Minimize -sum(radii))
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlapping between circles
    def non_overlap_constraint(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            return np.sqrt(dx**2 + dy**2) - (r_i + r_j)
        return cons

    # Constraint for circle within unit square
    def boundary_constraint(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Create constraints list
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):

        bounds=bounds,
        constraints=constraints,
        method='SLSQP',

    sum_radii = np.sum(radii_final)

    # Ensure radii are non-negative
    radii_final[radii_final < 0] = 0

    return centers_final, radii_final, sum_radii
