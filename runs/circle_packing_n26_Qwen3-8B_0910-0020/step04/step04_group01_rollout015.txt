import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26

    # Create grid-like center positions in a hexagonal grid with spacing
    centers = []
    base_x = [0.1, 0.3, 0.5, 0.7, 0.9]
    base_y = [0.2, 0.4, 0.6, 0.8]
    for x in base_x:
        for y in base_y:
            centers.append([x, y])

    # Add top-right and bottom-left corner circles
    centers.append([0.95, 0.95])  # Top-right corner
    centers.append([0.05, 0.95])  # Top-left corner
    centers = np.array(centers)

    # Define cost function to maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess based on hybrid arrangement
    initial_radii = np.zeros(num_points)
    for i in range(num_points - 2):
        initial_radii[i] = 0.06
    initial_radii[-2] = 0.04
    initial_radii[-1] = 0.04

    # Run the optimization
    result = minimize(
        objective,
        initial_radii,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
    )

    # Extract and clean results
    radii_final = result.x
    radii_final[radii_final < 0] = 0  # Enforce non-negativity

    # Ensure all circles are within the unit square
    for i in range(num_points):
        x, y = centers[i]
        r = radii_final[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii_final[i] = 0  # If outside, set radius to zero

    sum_radii = np.sum(radii_final)

    return centers, radii_final, sum_radii
