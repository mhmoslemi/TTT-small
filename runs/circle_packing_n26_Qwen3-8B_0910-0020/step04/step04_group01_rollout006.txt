from scipy.optimize import minimize

def run_packing():

    # Custom hexagonal packing for 26 circles
    centers = []

    # Hexagonal packing with 6 rows, allowing for maximum density
    # Use a staggered layout with alternating row positions
    # Top row (row 0) with 2 circles to use corners and have larger radii
    # Next rows with increasing number of circles (row 1: 5, row 2: 6, row 3: 6, row 4: 6, row 5: 6)

    # Top row: 2 circles
    offset_top = 0.95
    for i in range(2):
        x = i * 0.5
        y = offset_top
        centers.append([x, y])

    # Subsequent rows
    for row in range(5):
        if row % 2 == 0:
            offset = 0.05
        else:
            offset = 0.45
        x_positions = np.linspace(0.05, 0.95, 6)
        for i in range(6):
            x = x_positions[i]
            y = offset + row * 0.4
            centers.append([x, y])

    centers = np.array(centers)

    # Define cost function to maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess based on hexagonal packing
    # Ensure that large circles are in top corners

    result = minimize(
