import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define improved initial configuration with a hexagonal grid
    # 7 rows with varying number of circles: 4, 5, 6, 5, 4, 3, 2
    num_rows = 7
    num_cols = [4, 5, 6, 5, 4, 3, 2]
    total_circles = sum(num_cols)

    centers = []
    # Define row spacing and column spacing
    row_height = 0.25
    col_width = 0.25
    x_start = 0.25
    y_start = 0.25

    for row in range(num_rows):
        row_centers = []
        col_count = num_cols[row]
        for col in range(col_count):
            if row % 2 == 0:
                x = x_start + col * col_width
            else:
                x = x_start + (col + 0.5) * col_width
            y = y_start + row * row_height
            row_centers.append([x, y])
        centers.extend(row_centers)

    centers = np.array(centers)

    # Define cost function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    # Define constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for boundary conditions
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Build constraints
    constraints = []
    for i in range(total_circles):
        for j in range(i + 1, total_circles):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess for radii
    initial_guess = np.full(total_circles, 0.1)

    # Run the optimization
    result = minimize(
        objective,
        initial_guess,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 600, 'ftol': 1e-12, 'disp': False}
    )

    # Apply non-negativity to radii
    radii_final = result.x
    radii_final[radii_final < 0] = 0

    # Extract final results
    centers_final = centers
    sum_radii = np.sum(radii_final)

    return centers_final, radii_final, sum_radii
