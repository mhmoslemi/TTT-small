import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import Bounds

def run_packing():
    num_points = 26
    centers = []

    # Generate initial configuration with a more symmetric layout
    # We will use symmetric grid-based cluster formation
    # Place 25 circles in a "flower" pattern around the center, with one at the center
    # and others in a hexagonal pattern

    # Center circle at (0.5, 0.5)
    centers.append([0.5, 0.5])

    # Hexagonal packing for 25 circles around the center
    for ring in range(1, 6):  # 5 concentric rings around the center
        for angle in np.linspace(0, 2 * np.pi, 12, endpoint=False):  # 12 directions
            r = 0.25 * ring
            x = 0.5 + r * np.cos(angle)
            y = 0.5 + r * np.sin(angle)
            # Clip to [0.05, 0.95] to ensure boundary clearance
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            centers.append([x, y])

    centers = np.array(centers)

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):

    # Initial guess with radii based on the distance to the nearest boundary

    # Run optimization
    result = minimize(
