from scipy.optimize import minimize

def run_packing():

    # Create spring-like positions for the central circles
    centers = []

            x = 0.2 + i * 0.2
            y = 0.2 + j * 0.2
            if i % 2 == 0:
                centers.append([x, y])
            else:
                centers.append([x + 0.1, y])

    # Add corner circles, slightly perturbed
    centers.append([0.03, 0.97])  # left-top
    centers.append([0.97, 0.97])  # right-top
    centers.append([0.03, 0.03])  # left-bottom
    centers.append([0.97, 0.03])  # right-bottom

    # Fill in the remaining 10 circle positions
    for i in range(10):
        centers.append([0.5, 0.5 + i * 0.1])

    # Convert to NumPy array
    centers = np.array(centers)

    # Define the objective: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define non-overlap constraints
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define boundary constraints
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    constraints = []
    for i in range(num_points):

    result = minimize(
