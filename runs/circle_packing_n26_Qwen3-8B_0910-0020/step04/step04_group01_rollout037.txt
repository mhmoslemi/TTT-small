import numpy as np
from scipy.optimize import minimize

def run_packing():
    num_points = 26

    # Initial positions based on a hexagonal packing pattern
    centers = []
    for i in range(num_points):
        row = i // 7
        col = i % 7
        x = col / 6.5 + 0.05
        y = row / 6.5 + 0.05
        centers.append([x, y])
    centers = np.array(centers)

    # Define cost function to maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]

            return dist - r_i - r_j
        return cons

    # Constraint for square boundaries
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Prepare constraints
    constraints = []
    for i in range(num_points):
        for j in range(i + 1, num_points):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})

    # Initial guess based on hexagonal packing
    initial_radii = np.full(num_points, 0.05)

    # Run the optimization
    result = minimize(
