from scipy.optimize import minimize

def run_packing():
    # Define more efficient initial configuration with better spacing
    # This configuration places circles in a hexagonal pattern with better spacing
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    # Define cost function to maximize total area (sum of pi * r_i^2)
    def objective(radii):
        return -np.sum(np.pi * radii ** 2)  # Minimize negative of total area

    # Define constraint for non-overlap between any two circles
    def constraint_nonoverlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - r_i - r_j
        return cons

    # Define constraint for boundary conditions
    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_i, y_i = centers[i]
            return np.array([
                x_i - r - 1e-12,
                1.0 - x_i - r - 1e-12,
                y_i - r - 1e-12,
                1.0 - y_i - r - 1e-12
            ])
        return cons

    # Build constraints
    constraints = []
    for i in range(25):
        for j in range(i + 1, 26):
            constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    # Initial guess for radii
    initial_guess = np.full(26, 0.1)

    # Run the optimization with improved settings
    result = minimize(
        objective,
        initial_guess,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}

    radii_final = result.x

    centers_final = centers
