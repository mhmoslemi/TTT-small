        n_set.add(len(initial))

        n_set.add(n)
    n_list = sorted(n_set)

    # Time allocation: more time to smaller n (faster per-iteration)
    total_starts_estimate = 0
    n_starts_per_n = {}
    for n in n_list:
        n_starts_per_n[n] = 5 if n <= 100 else 3
        total_starts_estimate += n_starts_per_n[n]

    for n in n_list:
        if time.time() - start_time > budget_s - 20:
            break
        target_sum = n / 2.0
        max_iter = 400 if n <= 100 else 200

        starts = []
        if initial is not None:
            if len(initial) == n:
                starts.append(initial.copy())
            else:
                x_old = (np.arange(len(initial)) + 0.5) * 2.0 / len(initial)
                x_new = (np.arange(n) + 0.5) * 2.0 / n
                h_int = np.interp(x_new, x_old, initial)
                starts.append(_make_feasible(h_int, target_sum))

        # Random starts
        n_rand = n_starts_per_n[n] - len(starts)
        for _ in range(max(0, n_rand)):
            h_r = np.random.random(n)
            starts.append(_make_feasible(h_r, target_sum))

        # Structured: smooth oscillation
        t = np.arange(n) / float(n)
        h_s1 = 0.5 + 0.5 * np.sin(2.0 * np.pi * 3.0 * t)
        starts.append(_make_feasible(h_s1, target_sum))
        h_s2 = 0.5 + 0.5 * np.cos(2.0 * np.pi * 2.0 * t)
        starts.append(_make_feasible(h_s2, target_sum))

        # Block structure
        h_b = np.zeros(n)
        h_b[0:n // 2] = 1.0
        starts.append(_make_feasible(h_b, target_sum))

        for h0 in starts:
            if time.time() - start_time > budget_s - 10:
                break
            h_opt = _optimize_point(h0, n, target_sum, max_iter, 1e-10)
            c5_val = _c5_only(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n
            # Second pass (polish)
            if time.time() - start_time < budget_s - 15:
                h_opt2 = _optimize_point(h_opt, n, target_sum, 200, 1e-13)
                c5_val2 = _c5_only(h_opt2)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2.copy()
                    best_n = n

    # Final polish on best solution
    if best_h is not None and time.time() - start_time < budget_s - 5:
        n_best = best_n
        target_sum = n_best / 2.0
        h_final = _optimize_point(best_h, n_best, target_sum, 300, 1e-14)
        c5_final = _c5_only(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    if best_h is None:
        n = 70
        best_h = np.ones(n) * 0.5
        best_c5 = _c5_only(best_h)
        best_n = n

    return best_h, best_c5, best_n
