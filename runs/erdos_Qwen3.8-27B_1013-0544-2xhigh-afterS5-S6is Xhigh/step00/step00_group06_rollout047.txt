    exp_v = np.exp(v - v_max)
    Z = exp_v.sum()
    w = exp_v / Z
    F = (v_max + np.log(Z)) / tau
    # Add penalty for sum constraint
    s = h.sum() - n / 2.0
    F += penalty * s * s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        L = k - (n - 1)
        if L >= 0:
            if L < n:
                grad[L:] += wk * dx * (1.0 - h[:n - L])
                grad[:n - L] -= wk * dx * h[L:]
        else:
            M = -L
            if M < n:
                grad[:n - M] += wk * dx * (1.0 - h[M:])
                grad[M:] -= wk * dx * h[:n - M]
    grad += 2.0 * penalty * s
    return F, grad

def optimize_smooth(h0, n, tau, penalty, t0, budget_s):
    def wrapper(h_flat, _tau=tau, _pen=penalty, _n=n):
        return smooth_obj_grad(h_flat, _n, _tau, _pen)
    try:
        res = minimize(wrapper, h0, jac=True, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
        return repair(res.x, n)
    except Exception:
        return h0

def coord_descent_refine(h, n, t0, budget_s, max_sweeps=80):
    target = n / 2.0
    best_local = evaluate_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget_s * 0.96:
            break
        improved = False
        for j in range(n):
            old = h[j]
            cands = [0.0, 1.0]
            if 0.0 < old < 1.0:
                cands.extend([max(0, old - 0.1), max(0, old - 0.03),
                              min(1, old + 0.03), min(1, old + 0.1)])
            for nv in cands:
                if not (0.0 <= nv <= 1.0) or abs(nv - old) < 1e-9:
                    continue
                h[j] = nv
                # Repair sum
                diff = target - h.sum()
                if abs(diff) > 1e-11:
                    other = np.ones(n, dtype=bool)
                    other[j] = False
                    if diff > 0:
                        other &= (h < 1.0 - 1e-12)
                    else:
                        other &= (h > 1e-12)
                    cnt = other.sum()
                    if cnt > 0:
                        h[other] += diff / cnt
                        h = np.clip(h, 0, 1)
                new_c5 = evaluate_c5(h, n)
                if new_c5 < best_local - 1e-13:
                    best_local = new_c5
                    improved = True
                else:
                    h[j] = old
                    # Re-repair after failed move
                    h = repair(h, n)
            if time.time() - t0 > budget_s * 0.96:
                break
        if not improved:
            break
    return h, best_local

def random_feasible(n, rng):
    h = rng.random(n)
    h = repair(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = np.inf

    def try_update(h, n):
        nonlocal best_h, best_c5
        h = repair(h, n)
        c5 = evaluate_c5(h, n)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Initialize from parent if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
        try_update(init, len(init))

    # N values to try
    n_set = []
    if best_h is not None:
        n_set.append(len(best_h))
    for n in [95, 50, 64, 80, 100, 40, 25, 120, 150]:
        if n not in n_set:
            n_set.append(n)

    for n in n_set:
        if time.time() - t0 > budget_s * 0.75:
            break

        # Starting point: interpolate from best or random
        if best_h is not None:
            x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
            x_new = (np.arange(n) + 0.5) * (2.0 / n)
            h0 = np.interp(x_new, x_old, best_h)
        else:
            h0 = np.full(n, 0.5)
        h0 = repair(h0, n)

        # Smooth optimization with increasing tau
        for tau in [3.0, 8.0, 20.0, 60.0, 200.0]:
            if time.time() - t0 > budget_s * 0.85:
                break
            penalty = 50.0
            h_opt = optimize_smooth(h0, n, tau, penalty, t0, budget_s)
            try_update(h_opt, n)
            h0 = h_opt  # warm start for next tau

        # Coordinate descent refinement
        if time.time() - t0 < budget_s * 0.9:
            h_cd, c5_cd = coord_descent_refine(h0, n, t0, budget_s)
            try_update(h_cd, n)

    # Additional: try random restarts on best n
    if best_h is not None and time.time() - t0 < budget_s * 0.8:
        n_best = len(best_h)
        for attempt in range(5):
            if time.time() - t0 > budget_s * 0.88:
                break
            # Perturb best solution
            h_pert = best_h + 0.1 * rng.randn(n_best)
            h_pert = repair(h_pert, n_best)
            for tau in [10.0, 50.0]:
                h_opt = optimize_smooth(h_pert, n_best, tau, 50.0, t0, budget_s)
                try_update(h_opt, n_best)
                h_pert = h_opt
            h_cd, _ = coord_descent_refine(h_pert, n_best, t0, budget_s, max_sweeps=30)
            try_update(h_cd, n_best)

    # Final verification
    h_final = repair(best_h, len(best_h))
    c5_final = evaluate_c5(h_final, len(best_h))
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final

    return best_h, best_c5, len(best_h)
