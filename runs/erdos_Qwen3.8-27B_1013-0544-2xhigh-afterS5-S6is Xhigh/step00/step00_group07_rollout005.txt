import numpy as np
from scipy.optimize import minimize
import time

def _project(h, target, n):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _soft_obj(h, dx, t):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c = corr - np.max(corr)
    return np.log(np.sum(np.exp(t * c))) / t

def _slsqp_optimize(h, dx, target, n, t, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target,
             'jac': lambda h: np.ones(n)}]
    obj = lambda h: _soft_obj(h, dx, t)
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return _project(res.x, target, n)

def _coord_descent(h, dx, target, n, time_limit, n_passes=300, n_cand=21):
    best_c5 = _c5(h, dx)
    candidates = np.linspace(0.0, 1.0, n_cand)
    for _ in range(n_passes):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n - 1):
            old_i = h[i]
            old_last = h[n - 1]
            best_v = old_i
            best_c = best_c5
            for v in candidates:
                if abs(v - old_i) < 1e-10:
                    continue
                h[i] = v
                h[n - 1] = old_last + old_i - v
                if h[n - 1] < -1e-10 or h[n - 1] > 1.0 + 1e-10:
                    h[i] = old_i
                    h[n - 1] = old_last
                    continue
                h[n - 1] = np.clip(h[n - 1], 0.0, 1.0)
                # Re-adjust i to restore sum
                adj = target - h.sum()
                h[i] = v + adj
                if h[i] < 0 or h[i] > 1:
                    h[i] = old_i
                    h[n - 1] = old_last
                    continue
                c = _c5(h, dx)
                if c < best_c:
                    best_c = c
                    best_v = v
            h[i] = best_v
            h[n - 1] = old_last + old_i - best_v
            h[n - 1] = np.clip(h[n - 1], 0.0, 1.0)
            # Final correction
            adj = target - h.sum()
            if abs(adj) > 1e-10:
                h[i] = np.clip(h[i] + adj, 0.0, 1.0)
            if best_v != old_i:
                improved = True
        if not improved:
            break
        best_c5 = _c5(h, dx)
    return h, best_c5

def _optimize_at_n(h_start, dx, target, n, rng, time_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target,
             'jac': lambda h: np.ones(n)}]
    best_h = h_start.copy()
    best_c5 = _c5(best_h, dx)

    # SLSQP with increasing temperature

                    if best_h[n - 1] < 0 or best_h[n - 1] > 1:

                    c = _c5(best_h, dx)

            h_new = _project(res.x, target, n)
            c5_new = _c5(h_new, dx)

                best_c5 = c5_new
                best_h = h_new
        # Quick coord descent on the perturbed result
