    c_max = np.max(c)

                res = minimize(_of, h2, jac=_og, method='SLSQP',
                              bounds=bounds2, constraints=cons2,
                              options={'maxiter': 200, 'ftol': 1e-13})
                h2 = project(res.x, _ts)
                c5_c = compute_c5(h2)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h2
            except Exception:
                pass
        # Quick coord descent at n2
        if time.time() - t0 < budget_s * 0.95 and compute_c5(best_h) == best_c5:
            best_h, best_c5 = coord_descent(best_h, _ts, max_rounds=10)

    # Final verification and projection
    best_h = project(best_h, np.sum(best_h) * len(best_h) / len(best_h) / 2 * 2 / len(best_h) * len(best_h) / 2)
    # Simpler: just ensure feasibility
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project(best_h, ts_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
