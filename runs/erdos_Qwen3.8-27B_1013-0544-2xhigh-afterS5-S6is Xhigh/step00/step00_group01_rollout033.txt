def coordinate_descent(h, n, target, max_rounds=100):
    """Systematic coordinate descent for the overlap problem."""
    best = h.copy()
    best_score = evaluate(best)
    
    for round in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Try transferring from i to j
                if best[i] > 1e-8 and best[j] < 1 - 1e-8:
                    # Binary search for optimal delta
                    lo, hi = 0.0, min(best[i], 1 - best[j])
                    for _ in range(15):
                        mid = (lo + hi) / 2
                        trial = best.copy()
                        trial[i] -= mid
                        trial[j] += mid
                        s = evaluate(trial)
                        if s < best_score:
                            best = trial
                            best_score = s
                            improved = True
                            lo = mid
                        else:
                            hi = mid
                # Try transferring from j to i
                if best[j] > 1e-8 and best[i] < 1 - 1e-8:
                    lo, hi = 0.0, min(best[j], 1 - best[i])
                    for _ in range(15):
                        mid = (lo + hi) / 2
                        trial = best.copy()
                        trial[j] -= mid
                        trial[i] += mid
                        s = evaluate(trial)
                        if s < best_score:
                            best = trial
                            best_score = s
                            improved = True
                            lo = mid
                        else:
                            hi = mid
        if not improved:
            break
    
    return best, best_score
