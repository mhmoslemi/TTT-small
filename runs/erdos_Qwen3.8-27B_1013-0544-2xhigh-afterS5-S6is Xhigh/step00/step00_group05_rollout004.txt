import numpy as np
from scipy import optimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return np.max(corr) * dx

def project_onto_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Need to adjust by diff/n while keeping in [0,1]
    # Simple approach: add diff/n to all, then clip, then iterate
    for _ in range(100):
        h = h + diff / len(h)
        h = np.clip(h, 0, 1)
        new_diff = target_sum - np.sum(h)
        if abs(new_diff) < 1e-10:
            break
        diff = new_diff
    return h

def subgradient_of_c5(h):
    """Compute subgradient of C5(h) = max_k g_k(h)"""
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    best_k = np.argmax(corr)
    lag = best_k - (n - 1)
    
    # grad of g_k with respect to h
    # g_k = dx * sum_n h[n] * (1 - h[n+k])
    # d/dh[n] = dx * (1 - h[n+k]) + dx * (-h[n-k]) [from the cross term]
    grad = np.zeros(n)
    for i in range(n):
        # h[i] appears as h[n] in the sum: contributes (1-h[i+k])
        if 0 <= i + lag < n:
            grad[i] += dx * (1 - h[i + lag])
        # h[i] appears as h[n+k] where n = i-k: contributes -h[i-lag]
        if 0 <= i - lag < n:
            grad[i] += dx * (-h[i - lag])
    return grad

        can_inc = h < 1.0 - 1e-12
        can_dec = h > 1e-12

            h[can_inc] = np.minimum(1.0, h[can_inc] + adj)
        else:
            if not np.any(can_dec):
                break
            adj = diff / np.sum(can_dec)
            h[can_dec] = np.maximum(0.0, h[can_dec] + adj)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / np.sum(mask))
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / np.sum(mask))
    return h

def one_d_min(h, i):
    """Find optimal h[i] to minimize C5, given all other h fixed."""
    n = len(h)
    dx = 2.0 / n
    old_val = h[i]
    
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full") / dx  # = sum without dx factor

        lag = k_idx - (n - 1)
        coeff = 0.0
        if 0 <= i + lag < n:
            coeff += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            coeff += (-h[i - lag])
        a[k_idx] = coeff
    
    C = corr - a * old_val
    
    # Minimize max_k (a_k * x + C_k) for x in [0,1]
    # Collect breakpoints
    candidates = [0.0, 1.0]
    for i1 in range(2 * n - 1):
        for i2 in range(i1 + 1, 2 * n - 1):
            da = a[i1] - a[i2]
            if abs(da) > 1e-12:
                x_cross = (C[i2] - C[i1]) / da
                if 0.0 <= x_cross <= 1.0:
                    candidates.append(x_cross)
    
    best_x = old_val
    best_val = np.max(a * old_val + C)
    for x in candidates:
        val = np.max(a * x + C)
        if val < best_val - 1e-12:
            best_val = val
            best_x = x
    
    return best_x

    for sweep in range(max_sweeps):

        for i in range(n):
            new_val = one_d_min(h, i)
            change = abs(new_val - h[i])

                h[i] = new_val
                h = project_feasible(h, target_sum)
                max_change = max(max_change, change)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        
        if max_change < tol:
            break
    
    return h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.maximum(h, 0.0)
    return h

def one_d_min(h, i):
    n = len(h)
    dx = 2.0 / n
    old_val = h[i]
    corr = np.correlate(h, 1.0 - h, mode="full") / dx
    m = 2 * n - 1
    lags = np.arange(m) - (n - 1)
    ip = i + lags
    im = i - lags
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)
    a = np.zeros(m)
    a[valid_p] += 1.0 - h[ip[valid_p]]
    a[valid_m] += -h[im[valid_m]]
    C = corr - a * old_val
    diff_a = a[:, None] - a[None, :]
    diff_C = C[None, :] - C[:, None]
    idx = np.arange(m)
    upper = idx[:, None] < idx[None, :]
    mask = upper & (np.abs(diff_a) > 1e-12)
    with np.errstate(divide='ignore', invalid='ignore'):
        crossings = np.where(mask, diff_C / diff_a, np.inf)
    valid = (crossings >= 0.0) & (crossings <= 1.0) & np.isfinite(crossings)
    candidates = crossings[valid]
    all_x = np.unique(np.concatenate([[0.0, 1.0, old_val], candidates]))
    vals = np.max(a[None, :] * all_x[:, None] + C[None, :], axis=1)
    return all_x[np.argmin(vals)]

def coord_descent(h, target_sum, max_sweeps=200, tol=1e-9):
    n = len(h)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        max_ch = 0.0
        for i in range(n):
            nv = one_d_min(h, i)
            ch = abs(nv - h[i])
            if ch > tol:
                h[i] = nv
                h = project_feasible(h, target_sum)
                max_ch = max(max_ch, ch)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if max_ch < tol:
            break
    return h, best_c5
