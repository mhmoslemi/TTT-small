def compute_grad_for_lag(h, lag, n):
    grad = np.zeros(n)
    # (1-h[i-lag]) part: valid when 0 <= i-lag < n, i.e., lag <= i < n+lag
    if lag >= 0:
        # i ranges from lag to n-1
        grad[lag:] += (1.0 - h[:n-lag])
        # i ranges from 0 to n-1-lag for the h[i+lag] part
        if lag > 0:
            grad[:n-lag] -= h[lag:]
        else:
            grad[:] -= h[:]
    else:
        # lag < 0, let ll = -lag
        ll = -lag
        # i-lag = i+ll, valid when 0 <= i+ll < n, i.e., 0 <= i < n-ll
        grad[:n-ll] += (1.0 - h[ll:])
        # i+lag = i-ll, valid when 0 <= i-ll < n, i.e., ll <= i < n
        grad[ll:] -= h[:n-ll]
    return grad
