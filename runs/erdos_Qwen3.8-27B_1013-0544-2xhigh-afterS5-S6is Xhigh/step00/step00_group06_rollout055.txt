import time

            lo = mid
        else:
            hi = mid

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = corr >= mx - 1e-14
    nact = float(np.sum(active))

        return mx * dx, np.zeros(n)
    grad = np.zeros(n)

            jp = j + shift
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += g
