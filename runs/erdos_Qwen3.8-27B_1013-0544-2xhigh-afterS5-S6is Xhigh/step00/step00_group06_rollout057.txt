import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), corr

def project(h, n, target):
    """Project h onto {0 <= x <= 1, sum(x) = target}"""
    # Binary search on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    return h_proj

def compute_soft_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-8:
            continue
        lag = k - (n - 1)
        L = abs(lag)
        if lag >= 0:
            # grad[j] += (1-h[j-lag]) for j in [lag, n-1]
            # grad[j] -= h[j+lag] for j in [0, n-1-lag]
            if L == 0:
                grad[:] += w * (1 - h)  # (1-h[j]) for all j
                grad[:] -= w * (h)       # -h[j] for all j
                # Net: grad[j] += w * (1 - 2*h[j])
            else:
                grad[L:] += w * (1 - h[:n-L])  # (1-h[j-L]) for j in [L, n-1]
                grad[:n-L] -= w * h[L:]         # -h[j+L] for j in [0, n-1-L]
        else:
            # lag < 0, m = -lag = L
            # grad[j] += (1-h[j-m]) for j in [m, n-1]
            # grad[j] -= h[j+m] for j in [0, n-1-m]
            grad[L:] += w * (1 - h[:n-L])
            grad[:n-L] -= w * h[L:]
    
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_to_box_sum(h, n, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = float(np.max(corr))
    ex = alpha * (corr - mc)
    ex = ex - float(np.max(ex))
    w = np.exp(ex)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = float(w[k])
        if wk < 1e-8:
            continue
        lag = k - (n - 1)
        L = abs(lag)
        if L == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        else:
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] -= wk * h[L:]
    return grad

def optimize_pg(h0, n, target, max_iters, t0, budget_s, rng):
    h = h0.copy()
    alpha = 8.0
    lr = 0.004
    v = np.zeros(n)
    mu = 0.85
    best = compute_c5(h)[0]
    best_h = h.copy()
    patience = 0
    for it in range(max_iters):
        if budget_s - (time.time() - t0) < 25:
            break
        g = soft_gradient(h, n, alpha)
        gn = float(np.linalg.norm(g))
        if gn < 1e-7:
            break
        step = lr / max(gn, 0.05)
        v = mu * v - step * g
        h = project_to_box_sum(h + v, n, target)
        if it % 60 == 0:
            c5, _ = compute_c5(h)
            if c5 < best - 1e-9:
                best = c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6:
                h = best_h + rng.normal(0, 0.06, n)
                h = project_to_box_sum(h, n, target)
                v = np.zeros(n)
                patience = 0
            alpha = min(alpha * 1.18, 100.0)
            lr = max(lr * 0.92, 0.0003)
            mu = min(mu + 0.001, 0.95)
    c5, _ = compute_c5(h)
    if c5 < best:
        best = c5
        best_h = h.copy()
    return best_h, best

def try_round(h, n, target):
    order = np.argsort(h)[::-1]
    hr = np.zeros(n)
    n1 = int(target)
    hr[order[:n1]] = 1.0
    rem = target - n1
    if rem > 1e-9 and n1 < n:
        hr[order[n1]] = rem
    return hr
