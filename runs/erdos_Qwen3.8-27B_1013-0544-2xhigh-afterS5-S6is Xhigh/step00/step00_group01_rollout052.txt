        else:
            hi = mid

For the full correlation: np.correlate(h, 1-h, mode="full")[k]
The full correlation has length 2n-1, indexed 0 to 2n-2.
The lag (shift) is s = k - (n-1), ranging from -(n-1) to (n-1).

The correlation at shift s is:
c[s] = sum over valid i of h[i] * (1-h[i-s])

For s ≥ 0: i ranges from s to n-1
  c[s] = sum_{i=s}^{n-1} h[i] * (1-h[i-s])
  
For s < 0 (let t = -s > 0): i ranges from 0 to n-1-t
  c[-t] = sum_{i=0}^{n-1-t} h[i] * (1-h[i+t])

Gradient ∂c[s]/∂h[j]:
For s ≥ 0:
  h[j] appears as h[i] when i=j (if j ≥ s): contributes (1-h[j-s])
  h[j] appears as h[i-s] when i-s=j, i=j+s (if j+s ≤ n-1): contributes -h[j+s]
  
  So ∂c[s]/∂h[j] = (1-h[j-s]) * [j ≥ s] - h[j+s] * [j+s < n]

For s < 0 (t = -s):
  h[j] appears as h[i] when i=j (if j < n-t): contributes (1-h[j+t])
  h[j] appears as h[i+t] when i+t=j, i=j-t (if j-t ≥ 0): contributes -h[j-t]
  
  So ∂c[-t]/∂h[j] = (1-h[j+t]) * [j < n-t] - h[j-t] * [j ≥ t]

For l = s ≥ 0:
  grad[l:] += (1.0 - h[:n-l])  → for j ≥ l: grad[j] += (1-h[j-l])  ✓

For l = s < 0, lp = -l = t:
  grad[:n-lp] += (1.0 - h[lp:])  → for j < n-t: grad[j] += (1-h[j+t])  ✓
  grad[lp:] -= h[:n-lp]  → for j ≥ t: grad[j] -= h[j-t]  ✓

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_feasible(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_for_lag(h, lag):
    n = len(h)
    l = lag - (n - 1)
    grad = np.zeros(n)
    if l >= 0:
        grad[l:] += (1.0 - h[:n - l])
        grad[:n - l] -= h[l:]
    else:
        lp = -l
        grad[:n - lp] += (1.0 - h[lp:])
        grad[lp:] -= h[:n - lp]
    return grad

def smooth_gradient(h, corr, temp):
    n = len(h)
    corr_max = np.max(corr)
    exp_c = np.exp((corr - corr_max) / temp)
    weights = exp_c / (np.sum(exp_c) + 1e-30)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-10:
            grad += weights[k] * gradient_for_lag(h, k)
    return grad

def optimize_h(h_init, target_sum, max_time, temp_init=0.04):
    n = len(h_init)
    h = h_init.copy()
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    alpha = 0.004
    temp = temp_init
    stuck = 0
    last_improve = time.time()

    while time.time() < max_time:
        c5_val, corr = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
            last_improve = time.time()
        else:
            stuck += 1

        grad = smooth_gradient(h, corr, temp)
        grad = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-13:
            break

        h_new = h - alpha * grad / grad_norm
        h_new = project_feasible(h_new, target_sum)

        c5_new, _ = compute_c5(h_new)
        if c5_new < c5_val - 1e-13:
            h = h_new
            alpha = min(alpha * 1.008, 0.025)
            temp = max(temp * 0.9985, 0.004)
        else:
            alpha *= 0.97
            if stuck > 400:
                noise_scale = 0.02 + 0.02 * np.exp(-stuck / 500.0)
                h = project_feasible(h + np.random.randn(n) * noise_scale, target_sum)
                alpha = 0.006
                temp = max(temp, 0.015)
                stuck = 0
        if time.time() - last_improve > 30 and stuck > 100:
            h = project_feasible(h + np.random.randn(n) * 0.04, target_sum)
            alpha = 0.008
            temp = 0.03
            stuck = 0

    return best_h, best_c5

def make_bump(n, center, width):
    x = np.linspace(0, 2, n)
    h = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    s = np.sum(h)
    if s > 0:
        h = h * (n / 2.0) / s
    return np.clip(h, 0.0, 1.0)

def make_sin(n, freq, amp):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.cos(np.pi * freq * x)
    return np.clip(h, 0.0, 1.0)
