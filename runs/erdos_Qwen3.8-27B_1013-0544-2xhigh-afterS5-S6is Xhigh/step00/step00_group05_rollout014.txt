        h0 = fix_sum(h0, n)

        if it % 5 == 0:
            slsqp_step(h0)
        else:
            lbfgsb_step(h0, tau=int(100 + 300 * p))

    # Final polish
    if time.time() - t0 < budget_s - 5:
        slsqp_step(best_h, maxiter=5000)

    return best_h, best_c5, n
