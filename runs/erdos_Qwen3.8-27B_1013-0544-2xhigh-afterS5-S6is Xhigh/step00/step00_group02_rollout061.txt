def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * (2.0/N)

def project_onto_feasible(y, N):
    """Project y onto {h: 0<=h_i<=1, sum(h)=N/2}."""
    S = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0, 1)
        s = np.sum(h)
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fix sum
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        # Adjust free variables
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
        else:
            # Fallback: add to the largest elements
            idx = np.argsort(h)[-int(abs(diff))+1:]
            if diff > 0:
                h[idx] = np.minimum(h[idx] + diff/len(idx), 1.0)
            else:
                h[idx] = np.maximum(h[idx] + diff/len(idx), 0.0)
    return h

def subgradient_c5(h, epsilon=0.01):
    """Compute subgradient of max overlap at h."""
    N = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    dx = 2.0/N
    F = np.max(corr)
    
    # Soft active set
    threshold = F - epsilon
    active = corr >= threshold
    n_act = np.sum(active)
    
    if n_act == 0:
        active = np.zeros(2*N-1, dtype=bool)
        active[np.argmax(corr)] = True
        n_act = 1
    
    grad = np.zeros(N)
    for k_idx in np.where(active)[0]:
        k = k_idx - (N - 1)
        if k >= 0:
            grad[k:] += (1 - h[:N-k])
            grad[:N-k] -= h[k:]
        else:
            ka = -k
            grad[:N-ka] += (1 - h[ka:])
            grad[ka:] -= h[:N-ka]
    
    return grad / n_act

def projected_gradient_optimize(h0, N, max_iter=5000, lr_init=0.01):
    """FISTA-style projected gradient for C5 minimization."""
    h = h0.copy()
    h_prev = h0.copy()
    t_adv = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        # Compute subgradient
        g = subgradient_c5(h, epsilon=0.01)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t_adv*t_adv)) / 2
        h_y = h + (t_adv - 1) / t_new * (h - h_prev)
        
        # Line search / adaptive step

        h_new = project_onto_feasible(h_y - lr * g, N)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        t_adv = t_new
    
    return best_h, best_c5

    x0 = np.concatenate([h0, [t0]])

        return x[-1]

        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1-h, mode="full") * (2.0/N)
        return t - corr  # >= 0
    
    def ineq_jac(x):
        h = x[:-1]
        dx = 2.0/N
        n_corr = 2*N - 1
        J = np.zeros((n_corr, len(x)))

                J[k_idx, ka:] = dx * h[:N-ka]

        return np.sum(x[:-1]) - N/2.0
    
    def eq_jac(x):
        g = np.ones(len(x))
        g[-1] = 0
        return g
    
    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    bounds = [(0,1)]*N + [(0, 1.0)]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                      constraints=cons, bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-8})
        h_opt = project_feasible(res.x[:-1], N)
        c5_opt = compute_c5(h_opt)
        if c5_opt < t0:
            return h_opt, c5_opt
    except:
        pass
    
    return h0, t0

    N = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / N))

def project_feasible(y, N):
    S = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(y - mid, 0.0, 1.0)
        if np.sum(h) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = np.sum(free)
        if nf > 0:
            h[free] += diff / nf
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_grad(h, sigma):
    N = len(h)
    dx = 2.0 / N
    R = np.correlate(h, 1.0 - h, mode="full")
    f = R * dx
    fs = f / sigma
    fs -= np.max(fs)
    exp_f = np.exp(fs)
    w = exp_f / np.sum(exp_f)
    smooth_val = sigma * (np.log(np.sum(exp_f)) + np.max(fs))
    coeff = dx / sigma
    grad = np.zeros(N)
    for k_idx in range(2 * N - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (N - 1)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:N - k])
            grad[:N - k] -= wk * h[k:]
        else:
            ka = -k
            grad[:N - ka] += wk * (1.0 - h[ka:])
            grad[ka:] -= wk * h[:N - ka]
    grad *= coeff
    return smooth_val, grad

def optimize_lbfgs(h0, N, sigma, penalty, maxiter):
    def obj(h):
        sm, _ = smooth_max_grad(h, sigma)
        cv = np.sum(h) - N / 2.0
        return sm + penalty * cv * cv

    def grad(h):
        _, g = smooth_max_grad(h, sigma)
        g += 2.0 * penalty * (np.sum(h) - N / 2.0)
        return g

    res = minimize(obj, h0, method='L-BFGS-B', jac=grad,
                   bounds=[(0.0, 1.0)] * N,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    h_opt = project_feasible(res.x, N)
    return h_opt, compute_c5(h_opt)

def optimize_slsqp(h0, N, maxiter=300):
    t0 = compute_c5(h0) + 0.005
    x0 = np.concatenate([h0, [t0]])
    dx = 2.0 / N

    def obj(x):
        return x[-1]

    def obj_g(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def ineq(x):
        h = x[:-1]
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return x[-1] - corr

    def ineq_jac(x):
        h = x[:-1]
        nc = 2 * N - 1
        J = np.zeros((nc, len(x)))
        J[:, -1] = 1.0
        for k_idx in range(nc):
            k = k_idx - (N - 1)
            if k >= 0:
                J[k_idx, k:] = -dx * (1.0 - h[:N - k])
                J[k_idx, :N - k] = dx * h[k:]
            else:
                ka = -k
                J[k_idx, :N - ka] = -dx * (1.0 - h[ka:])
                J[k_idx, ka:] = dx * h[:N - ka]
        return J

    def eq(x):
        return np.sum(x[:-1]) - N / 2.0

    def eq_jac(x):
        g = np.ones(len(x))
        g[-1] = 0.0
        return g

    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_g,
                       constraints=cons, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_opt = project_feasible(res.x[:-1], N)
        c5_opt = compute_c5(h_opt)
        if c5_opt < t0 - 1e-10:
            return h_opt, c5_opt
    except Exception:
        pass
    return h0, t0
