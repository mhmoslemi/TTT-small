import numpy as np
from scipy.optimize import minimize, brentq
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(temp * (corr - m)))) / temp

def project_constraint(h, n):
    """Project h to satisfy sum(h) = n/2, 0<=h<=1."""
    target = n / 2.0
    current_sum = np.sum(h)
    diff = target - current_sum
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    # Simple uniform adjustment
    h = h + diff / n
    # Clip
    h = np.clip(h, 0, 1)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        # Distribute residual among non-saturated variables
        if residual > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += residual / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def optimize_from(h0, n, temp, penalty, maxiter=500):
    """Optimize from h0 using L-BFGS-B."""
    def objective(h):
        return smooth_c5(h, temp) + penalty * (np.sum(h) - n/2.0)**2
    
    result = minimize(
        objective, h0,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10}
    )
    h_new = np.clip(result.x, 0, 1)
    h_new = project_constraint(h_new, n)
    return h_new

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    h = h.copy()

            mask = h > 0.0
