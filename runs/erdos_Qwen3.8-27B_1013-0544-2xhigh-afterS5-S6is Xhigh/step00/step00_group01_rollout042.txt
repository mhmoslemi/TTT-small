        jm = j - k

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, beta):
    """Compute smooth max and its gradient."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Softmax weights
    corr_max = np.max(corr)
    exp_vals = np.exp(beta * (corr - corr_max))
    total = exp_vals.sum()
    weights = exp_vals / total
    obj = dx / beta * (np.log(total) + beta * corr_max)
    
    # Gradient: dx * sum_k weights[k] * grad_c[k]
    # grad_c[k] at h[j] = (1-h[j-k]) * I(0<=j-k<n) + (-h[j+k]) * I(0<=j+k<n)
    # where k = index - (n-1)
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        # Term 1: (1-h[j-k]) for j where 0<=j-k<n
        if k >= 0:
            # j-k >= 0 means j >= k, j-k < n means j < n+k (always true for j<n)
            j_range = np.arange(k, n)
            grad[j_range] += w * (1.0 - h[j_range - k])
        else:
            # j-k >= 0 means j >= -k = |k|
            j_range = np.arange(-k, n)
            grad[j_range] += w * (1.0 - h[j_range - k])

        if k >= 0:

            grad[j_range] -= w * h[j_range + k]
        else:
            j_range = np.arange(0, n + k)
            grad[j_range] -= w * h[j_range + k]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlation(h)
    cmax = np.max(corr)
    e = np.exp(beta * (corr - cmax))
    total = e.sum()
    w = e / total
    obj = dx * (cmax + np.log(total) / beta)
    
    grad = np.zeros(n)
    for idx in range(2*n-1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n-1)
        # grad contribution from shift k
        # d/dh_j of sum_i h[i+k]*(1-h[i])
        # = (1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)
        if k >= 0:
            # j-k valid: j >= k, j < n+k → j in [k, n-1] (since j<n)
            # j+k valid: j+k < n → j < n-k → j in [0, n-k-1]
            grad[k:] += wi * (1.0 - h[:n-k])  # (1-h[j-k]) for j=k..n-1
            grad[:n-k] -= wi * h[k:]  # -h[j+k] for j=0..n-k-1
        else:

            # j-k = j+kk valid: j+kk < n → j < n-kk → j in [0, n-kk-1]
            # j+k = j-kk valid: j >= kk → j in [kk, n-1]
            grad[:n-kk] += wi * (1.0 - h[kk:])  # (1-h[j+kk]) for j=0..n-kk-1
            grad[kk:] -= wi * h[:n-kk]  # -h[j-kk] for j=kk..n-1
    
    grad *= dx
    return obj, grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    e = np.exp(beta * (corr - cmax))
    total = e.sum()
    w = e / total
    obj = dx * (cmax + np.log(total) / beta)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:]
        else:
            kk = -k
            grad[:n - kk] += wi * (1.0 - h[kk:])
            grad[kk:] -= wi * h[:n - kk]
    grad *= dx
    return obj, grad

def project_osd(p, s_target):
    lo = -p.max()
    hi = 1.0 - p.min()
    if lo >= hi:
        return np.clip(p, 0.0, 1.0)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(p + mid, 0.0, 1.0).sum()
        if s > s_target:
            hi = mid
        else:
            lo = mid
    return np.clip(p + 0.5 * (lo + hi), 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_osd(h_new, n_new / 2.0)

def perturb(h, n, amount, rng):
    h_new = h.copy()
    for _ in range(n // 4):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            d = rng.uniform(-amount, amount)
            h_new[i] += d
            h_new[j] -= d
    return project_osd(h_new, n / 2.0)

def optimize_iterative(h, n, beta, t_limit, t0):
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.5
    stall = 0
    it_count = 0
    while time.time() - t0 < t_limit:
        obj, grad = smooth_max_and_grad(h, beta)
        if obj < best_val - 1e-13:
            best_val = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        accepted = False
        for _ in range(15):
            new_h = project_osd(h - step * grad, target)
            nv = compute_c5(new_h)
            if nv < obj + 1e-14:
                h = new_h
                accepted = True
                break
            step *= 0.5
        if accepted:
            step = min(step * 1.2, 3.0)
        else:
            step *= 0.5
            if step < 1e-9:
                break
        it_count += 1
        if stall > 300 or it_count > 100000:
            break
    return best_h, best_val
