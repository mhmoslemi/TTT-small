    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(10):

            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def slp_step(h0, n, dx):
    """One SLP iteration"""
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    n_var = n + 1
    c = np.zeros(n_var)
    c[-1] = 1.0
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n_var))
    b = np.zeros(2 * n - 1)
    
    for j in range(2 * n - 1):
        shift = j - (n - 1)
        for m in range(n):
            grad = 0.0
            if 0 <= m + shift < n:
                grad += (1.0 - h0[m + shift])
            if 0 <= m - shift < n:
                grad -= h0[m - shift]
            A[j, m] = grad * dx
        A[j, -1] = -1.0
        b[j] = -corr[j] * dx
    
    bounds = [(-h0[i], 1.0 - h0[i]) for i in range(n)]
    bounds.append((0.0, np.inf))
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0 - np.sum(h0)])
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
