from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def find_min_convex_pl(b, a, tol=1e-12):
    """Find min_{t in [0,1]} max_s (b_s + a_s * t). Returns (t*, val*)."""
    # Check boundaries
    f0 = np.max(b)
    f1 = np.max(b + a)
    
    if f0 <= f1:
        # Check if t=0 is the min
        active0 = (b == f0)
        if np.all(a[active0] >= -1e-15):
            return 0.0, f0
    if f1 <= f0:
        active1 = (b + a == f1)
        if np.all(a[active1] <= 1e-15):
            return 1.0, f1
    
    # Binary search in (0, 1)
    lo, hi = 0.0, 1.0
    for _ in range(200):

        vals = b + mid * a

            hi = mid
        else:
            return mid, mx
        if hi - lo < tol:
            break
    mid = (lo + hi) / 2.0
    return mid, np.max(b + mid * a)

        stagnation = 0

            corr_full = np.correlate(h, 1.0 - h, mode="full")
            current = h[i]

            valid_j1 = (j1 >= 0) & (j1 < n)
            j2 = i - s_vals

    """

    # b and a include the s=0 entry, but we handle it separately

    #           = -t^2 + t + corr0 - current + current^2

        # -t^2 + (gp_b - a_lines[idx])*t + (gp_c - b_lines[idx]) = 0

        C = b_lines[idx] - gp_c
        disc = B*B - 4*A*C
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            t1 = (-B + sqrt_disc) / (2*A)
            t2 = (-B - sqrt_disc) / (2*A)
            if 0 <= t1 <= 1: candidates.append(t1)
            if 0 <= t2 <= 1: candidates.append(t2)
    
    # Intersections of lines with each other
    for i in range(len(b_lines)):
        for j in range(i+1, len(b_lines)):

                t = (b_lines[j] - b_lines[i]) / da

        l_vals = b_lines + a_lines * t
        return max(g_val, np.max(l_vals) if len(l_vals) > 0 else -np.inf)

    # line-line intersections
    if m > 1:
        ad = al[:, None] - al[None, :]
        bd = bl[None, :] - bl[:, None]
        with np.errstate(divide='ignore', invalid='ignore'):
            tm = bd / ad
        tri = np.triu(np.ones((m, m), dtype=bool), 1)
        valid = (np.abs(ad) > 1e-15) & (tm >= 0) & (tm <= 1) & tri
        tc = tm[valid]
        candidates = np.concatenate([candidates, tc])
    g_vals = -candidates * candidates + gp_b * candidates + gp_c
    l_vals = bl[None, :] + al[None, :] * candidates[:, None]
    l_max = np.max(l_vals, axis=1)
    f_vals = np.maximum(g_vals, l_max)
    bi = np.argmin(f_vals)
    return candidates[bi], f_vals[bi]

def coord_descent(h_init, n, max_sweeps=300, tol=1e-13):
    h = np.clip(h_init.copy(), 0.0, 1.0)
    s_vals = np.arange(-(n - 1), n).astype(float)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        stag = 0
        for i in range(n):
            t_best, v_best = optimize_coord(h, i, n, s_vals)
            if v_best * dx < best_c5 - tol:
                best_c5 = v_best * dx
            if abs(t_best - h[i]) < 1e-12:
                stag += 1
            h[i] = t_best
        h = repair_sum(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if stag >= n:
            break
    return h, best_c5

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h_old, period=2.0)

def run(seed=42, budget_s=1100, **kwargs):

            if 10 <= n_try <= 300 and n_try not in ns:

    else:
        ns = [50, 75, 95, 100, 120]

    for n in ns:
        if time.time() - t0 > budget_s - 20:

        if h0 is not None and n != n0:

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

        # Random restarts
        rem = budget_s - (time.time() - t0)
        n_rest = min(3, max(1, int(rem / 40)))
        for _ in range(n_rest):
            if time.time() - t0 > budget_s - 5:
                break
            h_p = repair_sum(h_opt + rng.randn(n) * 0.03, n)
            h_r, c5_r = coord_descent(h_p, n, max_sweeps=100)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n

    # Nelder-Mead polish
    rem = budget_s - (time.time() - t0)
    if rem > 60 and best_h is not None:
        nb = best_n
        target = nb / 2.0

        def obj_nm(x):
            return compute_c5(repair_sum(x, nb))

        res = minimize(obj_nm, best_h.copy(), method='Nelder-Mead',
                       options={'maxiter': 10000, 'xatol': 1e-10,
