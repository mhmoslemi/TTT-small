                res = minimize(obj_lpf, cur_h, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-15})
                h_new = project_onto_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < cur_c5:
                    cur_c5 = c5_new
                    cur_h = h_new
                    update_best(cur_h, cur_c5, n)
            except Exception:
                pass

        # Phase 3: Local swap optimization (focused on active lags)
        n_swaps = 80000
        stagnant = 0
        for s in range(n_swaps):
            if elapsed() > budget_s - 40:
                break
            if (s + 1) % 5000 == 0 and stagnant > 30000:
                break

            # Pick j from coordinates involved in top lags with some randomness
            if rng.random() < 0.7:
                corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
                top_lags = np.argsort(corr)[-5:]
                # Pick a coordinate that appears in a top lag
                d = top_lags[rng.randint(len(top_lags))] - (n - 1)
                if d >= 0:
                    j = rng.randint(max(0, n - d - 1))
                    k = j + d
                else:
                    d2 = -d
                    j = rng.randint(d2, n)
                    k = j - d2
                if k >= n or k < 0 or k == j:
                    j = rng.randint(n)
                    k = rng.randint(n)
                    if j == k:
                        continue
            else:
                j = rng.randint(n)
                k = rng.randint(n)
                if j == k:
                    continue

            d_max = min(1.0 - cur_h[j], cur_h[k])
            d_min = max(-cur_h[j], cur_h[k] - 1.0)
            if d_max - d_min < 1e-10:
                continue

            # Try multiple delta values
            fracs = [0.1, 0.3, 0.5, 0.7, 0.9]
            deltas = []
            for f in fracs:
                deltas.append(f * (d_max - d_min) + d_min)
                deltas.append(-f * (d_max - d_min) / 2.0)

            best_delta = 0.0
            best_swap_c5 = cur_c5
            for d in deltas:
                d = max(d_min, min(d_max, d))
                if abs(d) < 1e-10:
                    continue
                cur_h[j] += d
                cur_h[k] -= d
                c5_try = compute_c5(cur_h, dx)
                if c5_try < best_swap_c5:
                    best_swap_c5 = c5_try
                    best_delta = d
                cur_h[j] -= d
                cur_h[k] += d

            if best_delta != 0.0:
                cur_h[j] += best_delta
                cur_h[k] -= best_delta
                cur_c5 = best_swap_c5
                stagnant = 0
                update_best(cur_h, cur_c5, n)
            else:
                stagnant += 1

        # Phase 4: Fine random perturbations
        for _ in range(20000):
            if elapsed() > budget_s - 30:
                break
            n_pert = rng.randint(1, 4)
            idx = rng.choice(n, n_pert, replace=False)
            h_try = cur_h.copy()
            h_try[idx] += rng.uniform(-0.05, 0.05, n_pert)
            h_try = project_onto_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < cur_c5:
                cur_c5 = c5_try
                cur_h = h_try
                update_best(cur_h, cur_c5, n)

    if best_h is None:
        best_h = np.full(80, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / 80)
        best_n = 80

    return best_h, best_c5, best_n
