import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5*n)**2
    
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    
    grad += 2.0 * beta * (np.sum(h) - 0.5*n)
    return f, grad

    h = np.clip(h, 0.0, 1.0)
    d = 0.5*n - np.sum(h)
    if abs(d) > 1e-14:

m + log(sum(exp(shifted - m))) = m + log(sum(es))

where m = max(alpha * C) and es = exp(alpha * C - m). The maximum of es is 1, and the minimum is exp(alpha * (min(C) - max(C))). Since C values are in [0, ~2], the range is at most 2, so the minimum of es is exp(-2*alpha). For alpha=2000, that's exp(-4000) ≈ 0. This is fine numerically.

        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
