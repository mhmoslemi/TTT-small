import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)


def get_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1 - h, mode='full') * dx


def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta


def smooth_max_weights(corr, beta):
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    return e / e.sum()


def compute_grad_h(h, weights):
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        j = k - (N - 1)
        w = weights[k] * dx
        if abs(w) < 1e-16:
            continue
        if j >= 0:
            if N - j > 0:
                grad[0:N - j] += w * (1.0 - h[j:N])
                grad[j:N] -= w * h[0:N - j]
        else:
            aj = -j
            if N - aj > 0:
                grad[aj:N] += w * (1.0 - h[0:N - aj])
                grad[0:N - aj] -= w * h[aj:N]
    return grad


def augmented_obj(h, beta, mu, N):
    corr = get_corr(h)
    sm = smooth_max_val(corr, beta)
    pen = mu * (np.sum(h) - N / 2.0) ** 2
    return sm + pen


def augmented_grad(h, beta, mu, N):
    corr = get_corr(h)
    w = smooth_max_weights(corr, beta)
    g = compute_grad_h(h, w)
    g += 2.0 * mu * (np.sum(h) - N / 2.0)
    return g


def enforce_constraint(h, N):
    h = h.copy()
    diff = N / 2.0 - h.sum()
    if abs(diff) < 1e-10:
        return h
    h += diff / N
    h = np.clip(h, 0.0, 1.0)
    diff2 = N / 2.0 - h.sum()
    if abs(diff2) > 1e-8:
        if diff2 > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 1e-10:
                h += diff2 * room / tr
        else:
            space = h
            ts = space.sum()
            if ts > 1e-10:
                h += diff2 * space / ts
        h = np.clip(h, 0.0, 1.0)
    return h
