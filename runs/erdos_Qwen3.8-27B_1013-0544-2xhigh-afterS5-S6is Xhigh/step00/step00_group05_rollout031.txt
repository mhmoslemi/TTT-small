def differential_evolution_custom(obj_fn, bounds, popsize=20, maxiter=100, 
                                    mutation=(0.4, 1.0), recombination=0.9,
                                    seed=42, budget_s=100):
    n = len(bounds)
    rng = np.random.default_rng(seed)
    lb = np.array([b[0] for b in bounds])
    ub = np.array([b[1] for b in bounds])
    
    # Initialize population
    pop = np.array([lb + rng.random(n) * (ub - lb) for _ in range(popsize)])
    fitness = np.array([obj_fn(ind) for ind in pop])
    
    best_idx = np.argmin(fitness)
    best_x = pop[best_idx].copy()
    best_f = fitness[best_idx]
    
    F = mutation[0]
    CR = recombination
    
    for gen in range(maxiter):
        for i in range(popsize):
            # Select three distinct individuals
            idxs = [j for j in range(popsize) if j != i]
            a, b, c = rng.choice(idxs, 3, replace=False)
            
            # Mutation
            v = pop[a] + F * (pop[b] - pop[c])
            v = np.clip(v, lb, ub)
            
            # Crossover
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial[mask] = v[mask]
            
            # Selection
            f_trial = obj_fn(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_x = trial.copy()
        
        # Check time
        if time.time() > budget_s:
            break
    
    return best_x, best_f

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

        deficit = target - s
        if abs(deficit) < 1e-14:

        if deficit > 0:

            tr = np.sum(room)

            best_h = h_nm.copy()

    # Phase 3: Try n+1 and n-1
    for n_try in [n + 1, n - 1]:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.92 or n_try < 10:
            break
        nf = len(best_h)
        xo = np.linspace(0, 2.0, nf, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        h_int = repair(np.interp(xn, xo, best_h), n_try)
        c5_int = c5_of(h_int)
        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_int.copy()
        h_nm = nm_refine(best_h, len(best_h), 3000)
        c5_nm = c5_of(h_nm)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()

    # Final NM polish
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.98:
        nf = len(best_h)
        h_fin = nm_refine(best_h, nf, 8000)
        c5_fin = c5_of(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = repair(best_h, len(best_h))
    best_c5 = c5_of(best_h)
    return (best_h, best_c5, len(best_h))
