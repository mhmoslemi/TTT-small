import numpy as np
import time

def project_onto_set(h, target_sum):
    """Project h onto {x : 0<=x_i<=1, sum(x)=target_sum} via bisection on offset."""
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_grad(h, alpha):
    """Gradient of smooth (log-sum-exp) approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        else:
            m = -s
            grad[:n - m] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:n - m]
    return dx * grad

def optimize_local(h0, n, time_limit, seed=0):

    rng = np.random.RandomState(seed)

    best_h = h.copy()

    alpha = 20.0
    mom = 0.8
    no_improve = 0
    stall_count = 0

    start = time.time()
    while time.time() - start < time_limit:
        grad = compute_grad(h, alpha)
        v = mom * v - lr * grad
        h_new = project_onto_set(h + v, target_sum)
        c5 = eval_c5(h_new)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1
            stall_count += 1

        h = h_new

            if lr < 5e-7:

            h = project_onto_set(h + rng.randn(n) * 0.015, target_sum)

        if stall_count > 2000:
            break

        alpha = min(alpha * 1.0008, 300.0)
        if stall_count > 500:
            mom = min(mom + 0.01, 0.95)

    return best_h, best_c5

def coordinate_polish(h, n, n_steps=5000, seed=0):
    """Simple coordinate descent: nudge single coordinates to reduce C5."""
    rng = np.random.RandomState(seed)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = eval_c5(h)
    for _ in range(n_steps):
        i = rng.randint(n)
        for delta in [0.01, 0.005, 0.02, 0.001]:
            h_try = h.copy()
            h_try[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h_try = project_onto_set(h_try, target_sum)
            c5 = eval_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                h = h_try
                break
    return h, best_c5
