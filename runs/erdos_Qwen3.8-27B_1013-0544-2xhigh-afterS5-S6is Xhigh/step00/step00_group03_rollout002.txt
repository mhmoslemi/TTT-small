    n = len(h)

    max_corr = np.max(corr)

    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def soft_obj_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    se = np.exp(T * (corr - mc))
    w = se / se.sum()
    obj = (mc + np.log(se.sum()) / T) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-8:
            continue
        s = k - (n - 1)
        t1 = np.zeros(n)
        if s >= 0:
            t1[s:n] = 1.0 - h[:n-s]
        else:
            t1[:n+s] = 1.0 - h[-s:]
        t2 = np.zeros(n)
        if s >= 0:
            t2[:n-s] = h[s:]
        else:
            t2[-s:] = h[:n+s]
        grad += wk * (t1 - t2)
    return obj, grad * dx

    # Final: if we still have time, do one more SLSQP on the best
    if time.time() - t0 < budget_s - 30 and best_h is not None:
        h_final, c5_final = slsqp_phase(best_h, 2.0 / best_n)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = best_n

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / 70)
        best_n = 70

    return (best_h, best_c5, best_n)
