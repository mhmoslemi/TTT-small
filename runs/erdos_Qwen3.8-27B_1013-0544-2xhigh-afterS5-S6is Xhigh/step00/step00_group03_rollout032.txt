            g[:n-kk] += 1 - h[kk:]
            g[kk:] -= h[:n-kk]
        else:
            g[:] = 1 - 2*h

    grad = dx * grad

    sum_h = np.sum(h)

    grad += 2 * penalty_weight * (sum_h - n/2)
    
    return obj + penalty, grad

    """Project h onto {0 <= h[i] <= 1, sum(h) = n/2}"""

    target = n / 2

    s = np.sum(h_clip)
    if s < 1e-10:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-10:
        return h_clip
    # Need to adjust: find t such that sum(clip(h_clip - t, 0, 1)) = target
    # If s > target, we need to decrease: find t > 0

    lo, hi = -1.0, 1.0

import numpy as np
from scipy.optimize import minimize
import time

_G_TGT = 35.0
_G_TAU = 0.1
_G_LAM = 500.0


def do_repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(len(h), target / len(h))
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - r.sum()
    if abs(d) > 1e-13:
        m = (r > 1e-9) & (r < 1.0 - 1e-9)
        if m.sum() > 0:
            r[m] += d / m.sum()
            np.clip(r, 0.0, 1.0, out=r)
    return r


def do_eval(h):
    return (2.0 / len(h)) * np.max(np.correlate(h, 1.0 - h, mode='full'))


def _sg(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = c.max()
    et = np.exp((c - mc) / _G_TAU)
    se = et.sum()
    sm = mc + _G_TAU * np.log(se)
    w = et / se
    g = np.zeros(n)
    for i in range(len(c)):
        m = i - (n - 1)
        wi = w[i]
        if wi < 1e-9:
            continue
        if m >= 0:
            g[m:] += wi * (1.0 - h[:n - m])
            g[:n - m] -= wi * h[m:]
        elif m < 0:
            k = -m
            g[:n - k] += wi * (1.0 - h[k:])
            g[k:] -= wi * h[:n - k]
        else:
            g += wi * (1.0 - 2.0 * h)
    s = h.sum()
    return dx * sm + _G_LAM * (s - _G_TGT) ** 2, dx * g + 2.0 * _G_LAM * (s - _G_TGT)


def _eg(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mi = int(c.argmax())
    m = mi - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[m:] = 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    elif m < 0:
        k = -m
        g[:n - k] = 1.0 - h[k:]
        g[k:] -= h[:n - k]
    else:
        g[:] = 1.0 - 2.0 * h
    s = h.sum()
    return dx * c[mi] + _G_LAM * (s - _G_TGT) ** 2, dx * g + 2.0 * _G_LAM * (s - _G_TGT)


def _no(h):
    return do_eval(do_repair(h, _G_TGT))


def _make_structured(n, rng, idx):
    t = np.arange(n)
    if idx == 0:
        return 0.5 + 0.15 * np.sin(np.pi * t / n)
    elif idx == 1:
        return 0.5 + 0.15 * np.cos(np.pi * t / n)
    elif idx == 2:
        return 0.5 + 0.12 * np.sin(2.0 * np.pi * t / n)
    elif idx == 3:
        return 0.5 + 0.1 * np.sin(3.0 * np.pi * t / n)
    else:
        return 0.5 + 0.2 * rng.randn(n)
