    n = len(v)

        h = np.clip(h, 0, 1)

def subgradient_c5(h):
    """Compute a subgradient of C5 at h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr) - (n - 1)  # actual lag
    if k_star >= 0:
        # corr[k_star] = sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])
        grad = np.zeros(n)
        # Term 1: (1-h[m-k]) for m in [k, n-1]
        grad[k_star:] = 1.0 - h[:n-k_star]
        # Term 2: -h[m+k] for m in [0, n-1-k]
        grad[:n-k_star] -= h[k_star:]
    else:
        k_abs = -k_star
        grad = np.zeros(n)
        # Term 1: (1-h[m+k']) for m in [0, n-1-k']
        grad[:n-k_abs] = 1.0 - h[k_abs:]
        # Term 2: -h[m-k'] for m in [k', n-1]
        grad[k_abs:] -= h[:n-k_abs]
    return grad * dx

    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(n_iters):
        t_frac = i / max(n_iters - 1, 1)
        T = T_init * (T_final / T_init) ** t_frac
        
        # Compute soft gradient
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        weights = np.exp((corr - corr.max()) / T)
        weights /= weights.sum()
        
        # Compute gradient as weighted sum of subgradients
        grad = np.zeros(n)
        # This is expensive if we compute each subgradient separately
        # Better: compute all at once
        # Actually, let's just use the top few lags
        top_k = np.argsort(corr)[-5:]
        for k_idx in top_k:
            k = k_idx - (n-1)
            if abs(weights[k_idx]) > 1e-6:
                if k >= 0:
                    g = np.zeros(n)
                    g[k:] = 1.0 - h[:n-k]
                    g[:n-k] -= h[k:]
                else:
                    ka = -k
                    g = np.zeros(n)
                    g[:n-ka] = 1.0 - h[ka:]
                    g[ka:] -= h[:n-ka]
                grad += weights[k_idx] * g
        
        # Step size: use Polyak or diminishing
        step = 1.0 / (1.0 + i * 0.1)
        h_new = h - step * grad
        h = project_box_sum(h_new, n * 0.5)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_box_sum(v, S):
    n = len(v)
    # Bisection on lambda for projection onto {0<=h<=1, sum(h)=S}
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny correction
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def topk_subgradient(h, k_count=5):
    """Compute weighted subgradient using top-k lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top k indices
    indices = np.argsort(corr)[-k_count:]
    weights = corr[indices] - corr[indices].min() + 1e-8
    weights = weights ** 3  # sharpen
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(indices, weights):
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            ka = -k
            grad[:n-ka] += w * (1.0 - h[ka:])
            grad[ka:] -= w * h[:n-ka]
    return grad * dx

def nesterov_projected(h0, n_iters, k_count=5, gamma=0.5, seed=42):
    """Nesterov accelerated projected gradient."""
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    y = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(n_iters):
        alpha = gamma / (gamma + t)
        if t > 0:
            beta = (1.0 - np.sqrt(1.0 + 4.0 * alpha * alpha)) / (2.0 * alpha)
            y = h + beta * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
        else:
            y = h.copy()
            beta = 0.0
        
        g = topk_subgradient(y, k_count)
        h_new = y - alpha * g
        h_new = project_box_sum(h_new, S)
        h_prev = h
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_round(h, n):
    """Round h to binary while maintaining sum = n/2."""
    n_half = n // 2
    # Sort by value, take top n/2 as 1
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n_half]] = 1.0
    return h_bin

def binary_local_search(h_bin, n_iters=5000, T_init=0.01, T_final=0.0001, seed=42):
    """Simulated annealing on binary solutions with pairwise flips."""
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for t in range(n_iters):
        T = T_init * (T_final / T_init) ** (t / max(n_iters - 1, 1))
        
        # Pick random pair to flip
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

        return v[-1]

        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr

        return np.sum(v[:n]) - n * 0.5
    
    def eq_jac(v):

        return j

        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

    n = len(v)
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

            h_best, c5_best = h_new, c5_new
        if tl() > 80:
            h_new, c5_new = slsqp_refine(h_best, 400)
            if c5_new < c5_best:
                h_best, c5_best = h_new, c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_box_sum(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
