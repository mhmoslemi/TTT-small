        c_current = np.convolve(a_current, a_current)  # use direct for accuracy

        # This is a banded matrix

        vals = []

            # Non-zero entries at m where 0 <= k-m < n, i.e., max(0,k-n+1) <= m <= min(k, n-1)

                km_range = k - m_range  # = k - m, should be in [0, n-1]
                rows.extend([k] * len(m_range))
                cols.extend(m_range.tolist())
                vals.extend((2.0 * a_current[km_range]).tolist())

        # Subject to: A_sparse @ a_new - t <= c_current  (i.e., A_sparse @ a_new - [0..0, 1] @ [a,t] <= c_current)
        # sum(a_new) = 1
        # a_new >= 0, t >= 0
        
        # Full constraint matrix: [A_sparse, -1] @ [a; t] <= c_current
        # Equality: [1...1, 0] @ [a; t] = 1
        # Bounds: a >= 0, t >= 0
        
        # For linprog:
        # c_vec: objective coefficients (we want to minimize t)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize t

        # where x = [a; t], A_full = [A_sparse, -1_{N}], b_full = c_current
        # A_sparse is N x n, we add a column of -1's
        # Total: N x (n+1)
        
        # For memory, let's construct this carefully
        # A_full has N rows and n+1 columns
        # Columns 0..n-1: A_sparse
        # Column n: -1
        
        # We'll construct the full sparse matrix
        # Additional non-zeros: N entries in the last column
        rows2 = rows + list(range(N))
        cols2 = cols + [n] * N
        vals2 = vals + [-1.0] * N
        
        A_full_sparse = csr_matrix((vals2, (rows2, cols2)), shape=(N, n + 1))
        
        b_full = c_current
        
        # Equality: sum(a) = 1 → [1, 1, ..., 1, 0] @ [a; t] = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        # Bounds: a >= 0, t >= 0
        bounds = [(0, None)] * (n + 1)
        
        # Solve
        result = linprog(
            c=c_vec,
            A_ub=A_full_sparse,
            b_ub=b_full,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'presolve': True}
        )
        
        if not result.success:
            return None, t_current
        
        a_lp = result.x[:n]
        t_lp = result.x[n]
        
        return a_lp, float(t_lp)
