import numpy as np
import time

DOC = """
Local-search + projected softmax-subgradient minimizer for
    2*n*max(convolve(a,a)) / (sum(a))^2  (scale invariant).

Work on the probability simplex (sum p = 1, p >= 0).  The soft-max
surrogate makes the max-of-convolution differentiable; a projected
gradient step (clip to >=0 then renormalize) keeps us on the simplex.
Random restarts from several structured seeds + block multiplicative
perturbations + simulated-annealing accepts escape shallow minima.
Convolution and the surrogate gradient are computed with FFT for speed.
"""


def _fft_convolve(a):
    """Linear convolution of a with itself via FFT."""
    n = a.size
    L = max(2 * n - 1, 1)
    A = np.fft.rfft(a, L)
    out = np.fft.irfft(A * A, L)
    return out[:2 * n - 1]


def _fft_corr_shift(w, p):
    """Return g_i = sum_j p_j * w_{i+j} for i = 0..len(p)-1, via FFT.

    w has length >= 2*len(p)-1 (the per-lag weights of the convolution).
    """
    n = p.size
    L = max(1, np.next_fast_len(3 * n))
    W = np.fft.rfft(w[:L], L) if w.size < L else np.fft.rfft(w[:L], L)
    P = np.fft.rfft(p, L)
    r = np.fft.irfft(W * np.conj(P), L)
    return r[:n]


def _score(p):
    """Objective for a sum-1 nonnegative sequence (scale-invariant form)."""
    n = p.size
    s = p.sum()
    if s <= 0:
        return np.inf
    b = _fft_convolve(p)
    return float(2.0 * n * np.max(b) / (s * s))


def _normalize(p, target=1.0):
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0)
    s = p.sum()
    if s <= 0:
        return np.ones(p.size) / p.size
    return p * (target / s)


def _seed_wave(n, kind, rng):
    j = np.arange(n, dtype=np.float64)
    if kind == 0:                      # uniform
        p = np.ones(n)
    elif kind == 1:                    # front-loaded asymmetric ramp
        p = np.exp(-(j / n) * 1.6) + 0.25
    elif kind == 2:                    # back-loaded
        p = np.exp(-((n - 1 - j) / n) * 1.6) + 0.25
    elif kind == 3:                    # two-bump
        c = n * 0.5
        p = np.exp(-((j - 0.25 * n) / (0.12 * n)) ** 2) + \
            np.exp(-((j - 0.85 * n) / (0.10 * n)) ** 2) + 0.1
    elif kind == 4:                    # cosine window (front heavy)
        p = 0.5 + 0.5 * np.cos(np.pi * j / n)
        p = np.maximum(p, 0.0) ** 1.5 + 0.1
    else:                              # random smoothed
        raw = rng.random(n)
        k = max(3, n // 64)
        p = np.convolve(raw, np.ones(k) / k, mode="same") + 0.05
    return _normalize(p)


def _block_perturb(p, rng, strength=0.15):
    n = p.size
    cand = p.copy()
    width = int(rng.integers(max(2, n // 200), max(3, n // 8)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    return np.maximum(cand, 0.0)


def _optimize(p0, deadline):
    p = _normalize(p0)
    n = p.size
    b = _fft_convolve(p)
    best = p.copy()
    best_v = _score(p)

    scale = max(1e-12, float(np.max(b)))
    tau = 0.06 * scale
    tau_floor = 0.004 * scale
    lr = 0.05
    acc_temp = 0.5 * scale
    rng = np.random.default_rng(int(sum(np.abs(p) * 1000) % (2 ** 31)))

    it = 0
    while time.monotonic() < deadline:
        b = _fft_convolve(p)
        bm = np.max(b) - np.min(b)
        w = np.exp(b / tau - np.max(b) / tau)
        sw = w.sum()
        if sw > 0:
            w = w / sw
        g = 2.0 * _fft_corr_shift(w, p)

        p = p - lr * g
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s <= 0:
            p = _normalize(p0)
        else:
            p = p * (1.0 / s)

        # occasional exploratory block move
        if it % 37 == 0:
            cand = _block_perturb(p, rng, strength=0.20)
            cand = _normalize(cand)
            cv = _score(cand)
            if cv < best_v + acc_temp:
                p = cand

        v = _score(p)
        if v < best_v:
            best_v = v
            best = p.copy()
            acc_temp = max(tau_floor, acc_temp * 0.995)
            lr = min(0.12, lr * 1.01)
        else:
            # simulated-annealing accept to wander
            if v < best_v + acc_temp and rng.random() < 0.5:
                p = p
            acc_temp = max(tau_floor, acc_temp * 0.999)

        # anneal surrogate and step size
        tau = max(tau_floor, tau * 0.9992)
        lr = 0.04 + 0.02 * (tau / (tau + scale * 0.02))

        it += 1
        if it % 3000 == 0:
            print(f"    opt iter {it}: best={best_v:.6f} tau/scale={tau/scale:.4f}")

    return best, best_v


def _run_for_length(n, deadline, rng):
    best = None
    best_v = np.inf
    kinds = [0, 1, 2, 3, 4, 5]
    # give the first two seeds (uniform + front ramp) more time

        if cv < best_v:

            best = cand
        print(f"  len={n} seed_kind={k} score={cv:.6f} (best so far {best_v:.6f})")
        if time.monotonic() >= deadline - 1.0:
            break
    return (best, best_v) if best is not None else (None, np.inf)
