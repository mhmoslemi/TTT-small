import numpy as np
from scipy.fft import fft, ifft

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # next power of 2
    A = fft(a, size)
    B = fft(b, size)

from scipy.fft import fft, ifft

    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1 << (n + m - 2).bit_length()
    A = fft(a, size)
    B = fft(b, size)

    return fft_convolve(f, f)

Gradient-based optimization of non-negative sequence to minimize

projected gradient descent with multiple restarts.

import time
from scipy.fft import fft, ifft

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1 << (n + m - 2).bit_length()
    A = fft(a, size)
    B = fft(b, size)
    result = np.real(ifft(A * B, size))
    return result[:n + m - 1]

def compute_score_fast(f):
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return float('inf')
    conv = fft_convolve(f, f)
    return float(2.0 * n * np.max(conv) / (s * s))

def gradient_step(f, k_max, step):
    n = len(f)
    i_lo = max(0, k_max - n + 1)
    i_hi = min(n, k_max + 1)
    indices = np.arange(i_lo, i_hi)
    grad = np.zeros(n)
    grad[indices] = f[k_max - indices]
    
    # Project onto sum-preserving subspace
    grad_centered = grad - grad.mean()
    
    f_new = f - step * grad_centered
    f_new = np.clip(f_new, 0.0, None)
    s = f_new.sum()
    if s < 1e-10:
        return None
    f_new = f_new * (f.sum() / s)  # maintain the same sum
    return f_new

def optimize_single(f0, time_budget, verbose=False):
    """Optimize a single starting sequence."""
    f = np.array(f0, dtype=np.float64).copy()

    n = len(f)

        f = np.ones(n) / n
    else:
        f = f / s

    step = 0.02

    while time.monotonic() - start < time_budget:

            best_f = f_new.copy()
            best_score = score_new
            f = f_new
            stall_count = 0
            step = min(step * 1.03, 0.3)
        else:
            stall_count += 1
            step *= 0.95
            if stall_count > 200:

        if verbose and iteration % 100 == 0:

    # Sine
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * x))
    
    return inits

Actually, let me reconsider the L2 approach more carefully.

We want to minimize J(f) = sum_k (conv[f])[k]^2 subject to sum(f) = 1, f >= 0.

The gradient dJ/df[i] = 2 * sum_k (conv[f])[k] * d(conv[f][k])/df[i]
= 2 * sum_k (conv[f])[k] * 2*f[k-i] (when k-i is in [0,n-1])
= 4 * sum_{k: k-i in [0,n-1]} (conv[f])[k] * f[k-i]

Let g = conv(f, f) (length 2n-1). Then:
dJ/df[i] = 4 * sum_{j=0}^{n-1} g[j+i] * f[j] (where j = k-i, so k = j+i)

This is the cross-correlation of g and f! Specifically:
dJ/df[i] = 4 * (cross_correlate(g, f))[i]

where cross_correlate(g, f)[i] = sum_j g[j+i] * f[j].

This can be computed as:
cross_correlate(g, f) = conv(g, reverse(f))[n-1 : n-1+n]

Using FFT:
- Zero-pad g to length >= 3n (since g has length 2n-1 and reverse(f) has length n, the convolution has length 3n-2)
- Zero-pad reverse(f) to the same length
- Multiply, IFFT, take the window [n-1, n-1+n)

This gives us the gradient in O(n log n).

For the projected gradient step:
1. Compute grad = 4 * cross_correlate(conv(f,f), f) [length n]
2. Center: grad_centered = grad - mean(grad)
3. Update: f_new = f - step * grad_centered
4. Project: f_new = max(f_new, 0), renormalize
