"""
Search non-negative step heights minimizing 2*n*max(convolve(a,a))/sum(a)^2.

The search uses FFT-based convolution, a smooth-logsumexp subgradient /
multiplicative update, structured and random perturbations, and a hierarchical
multi-start job scheduler over several sequence lengths.
"""

import numpy as np
import time


def normalize_nonneg(p):
    p = np.asarray(p, dtype=float)
    if p.size == 0:
        return np.array([1.0])
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.ones(p.size, dtype=float) / float(p.size)
    return p / s


def append_start(lst, p):
    append_start__p = np.asarray(p, dtype=float)
    if append_start__p.size == 0:
        return
    append_start__p = np.where(np.isfinite(append_start__p), append_start__p, 0.0)
    append_start__p = np.maximum(append_start__p, 0.0)
    s = float(append_start__p.sum())
    if s > 1e-12:
        lst.append(append_start__p / s)


def fast_convolution(a):
    a = np.asarray(a, dtype=float)
    n = a.size
    if n == 0:
        return np.array([0.0])
    L = 2 * n - 1
    N = 1
    while N < L:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.where(c > 0.0, c, 0.0)


def score_from_conv(c, n, s):
    return float(2.0 * n * float(np.max(c)) / (s * s))


def make_weights_from_c(c, temp):
    c = np.asarray(c, dtype=float)
    L = c.size
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        w = np.zeros(L, dtype=float)
        if L:
            w[0] = 1.0
        return w
    frac = c / cmax
    frac = np.minimum(frac, 1.0)
    focus = 10.0 / max(1e-4, float(temp))
    arg = focus * (frac - 1.0)
    arg = np.maximum(arg, -745.0)
    w = np.exp(arg)
    wsum = float(w.sum())
    if not np.isfinite(wsum) or wsum <= 0.0:
        w = np.zeros(L, dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / wsum
    return w


def pressure_from_weights(p, w):
    p = np.asarray(p, dtype=float)
    w = np.asarray(w, dtype=float)
    n = p.size
    if n == 1:
        return np.array([2.0 * float(w[0]) * float(p[0])])
    L = 2 * n - 1
    total_len = n + L - 1
    N = 1
    while N < total_len:
        N <<= 1
    fp = np.fft.rfft(p, N)
    fw = np.fft.rfft(w[::-1], N)
    conv = np.fft.irfft(fp * fw, N)[:total_len]
    g = 2.0 * conv[L - 1 : L - n - 1 : -1]
    return np.where(g > 0.0, g, 0.0)


def upsample_sequence(a, n):
    a = np.asarray(a, dtype=float)
    m = a.size
    if m == 0:
        return np.ones(int(n), dtype=float) / float(max(1, n))
    a = normalize_nonneg(a)
    if m == 1:
        return np.ones(int(n), dtype=float) / float(max(1, n))
    pos = np.linspace(0.0, 1.0, m)
    tgt = np.linspace(0.0, 1.0, n)
    density = float(m) * a
    new_density = np.interp(tgt, pos, density)
    new_density = np.maximum(new_density, 0.0)
    s = float(new_density.sum())
    if s <= 1e-12:
        return np.ones(int(n), dtype=float) / float(max(1, n))
    return new_density / s


def upsample_block_sequence(a, n):
    a = np.asarray(a, dtype=float)
    m = a.size
    if m == 0:
        return np.ones(int(n), dtype=float) / float(max(1, n))
    a = normalize_nonneg(a)
    if m == 1:
        return np.ones(int(n), dtype=float) / float(max(1, n))
    density = float(m) * a
    idx = np.floor(np.linspace(0.0, float(m), n)).astype(np.int64)
    idx = np.clip(idx, 0, m - 1)
    out = density[idx]
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-12:
        return np.ones(int(n), dtype=float) / float(max(1, n))
    return out / s


def spectral_starts(n):
    n = int(n)
    if n < 2:
        return []
    L = 2 * n - 1
    target = np.ones(L, dtype=float) / float(L)
    C = np.fft.fft(target)
    F = np.sqrt(C)
    f = np.real(np.fft.ifft(F))
    starts = []
    seen = set()
    shifts = (0, 1, 2, n // 2, n - 2, n - 1, 3, n // 3, 2 * n // 3)
    for sh in shifts:
        sm = int(sh) % n
        if sm in seen:
            continue
        seen.add(sm)
        arr = np.roll(f, -sm)[:n]
        append_start(starts, arr)
        append_start(starts, arr[::-1])
    return starts


def make_starts(n, rng, num, parent=None):
    n = int(n)
    num = int(num)
    if num <= 0 or n <= 0:
        return []

    important = []
    others = []

    append_start(important, np.ones(n, dtype=float))

    for p in spectral_starts(n)[:8]:
        append_start(important, p)

    x = np.linspace(0.0, 1.0, n)
    eps = 1e-7 / float(n)

    for a, b in (
        (0.5, 1.0),
        (1.0, 2.0),
        (1.0, 3.0),
        (0.5, 1.5),
        (1.0, 1.5),
        (2.0, 3.0),
        (1.0, 1.2),
        (1.2, 1.0),
    ):
        append_start(important, (x + eps) ** a * (1.0 - x + eps) ** b)

    for lam in (0.0, 0.5, 1.0, 1.3, 1.6, 2.0, 3.0, 5.0):
        append_start(important, np.exp(lam * x))

    for mu in (0.25, 0.5, 0.75):
        for k in (2.0, 5.0, 10.0, 20.0):
            append_start(important, 1.0 / (1.0 + np.exp(-k * (x - mu))))

    if parent is not None:
        append_start(important, upsample_sequence(parent, n))
        append_start(important, upsample_block_sequence(parent, n))

    for q in (0.2, 0.5, 1.0, 2.0, 4.0):
        append_start(others, (x + eps) ** q)
        append_start(others, (1.0 - x + eps) ** q)

    for a, b in (
        (0.2, 1.0),
        (0.5, 2.0),
        (0.5, 3.0),
        (1.0, 4.0),
        (2.0, 4.0),
        (0.8, 2.0),
        (1.5, 0.8),
        (2.0, 1.0),
        (0.8, 0.8),
    ):
        append_start(others, (x + eps) ** a * (1.0 - x + eps) ** b)

    for mu in (0.15, 0.35, 0.65, 0.85):
        for k in (1.0, 8.0, 30.0):
            append_start(others, 1.0 / (1.0 + np.exp(-k * (x - mu))))

    for mode in range(1, 9):
        for amp in (0.25, 0.6, 0.9):
            for phase in (0.0, 0.5 * np.pi):
                append_start(others, 1.0 + amp * np.cos(2.0 * np.pi * mode * x + phase))

    if n <= 2048:
        x3 = (3.0 - np.sqrt(2.0)) / 7.0
        patterns = [
            np.array([1.0 - 2.0 * x3, x3, x3], dtype=float),
            np.array([0.45, 0.25, 0.20, 0.10], dtype=float),
            np.array([0.40, 0.30, 0.20, 0.10], dtype=float),
            np.array([0.50, 0.20, 0.15, 0.15], dtype=float),
            np.array([0.33, 0.25, 0.21, 0.12, 0.09], dtype=float),
        ]
        for small in patterns:
            append_start(others, upsample_sequence(small, n))
            append_start(others, upsample_block_sequence(small, n))

        for m in (3, 4, 5, 8, 16):
            sm = np.linspace(0.0, 1.0, m)
            for lam in (0.5, 1.0, 1.5, 2.0):
                small = np.exp(lam * sm)
                small = small / float(small.sum())
                append_start(others, upsample_sequence(small, n))
                append_start(others, upsample_block_sequence(small, n))

    if n <= 2048:
        alphas = (0.05, 0.1, 0.25, 0.5, 1.0, 2.0, 5.0)
    else:
        alphas = (0.1, 0.5, 1.0, 3.0)
    for alpha in alphas:
        append_start(others, rng.dirichlet(np.full(n, float(alpha))))

    for _ in range(4):
        c0 = float(rng.normal(0.0, 1.0))
        c1 = float(rng.normal(0.0, 1.0))
        c2 = float(rng.normal(0.0, 1.0))
        c3 = float(rng.normal(0.0, 1.0))
        mode = int(rng.integers(1, 5))
        shape = (
            1.0 + c0 * x + c1 * np.sin(2.0 * np.pi * x)
            + c2 * np.cos(2.0 * np.pi * mode * x) + c3 * np.sin(4.0 * np.pi * mode * x)
        )
        append_start(others, shape)

    selected = []
    for p in important:
        if len(selected) >= num:
            break
        selected.append(p)

    if len(selected) < num and others:
        need = num - len(selected)
        if len(others) <= need:
            for p in others:
                selected.append(p)
        else:
            idxs = rng.choice(len(others), size=need, replace=False)
            for idx in idxs:
                selected.append(others[int(idx)])

    while len(selected) < num:
        selected.append(np.ones(n, dtype=float) / float(n))

    return selected


def structured_perturb(p, scale, rng):
    p = np.asarray(p, dtype=float)
    n = p.size
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-6 / float(n)
    style = int(rng.integers(0, 6))
    if style == 0:
        lam = float(rng.uniform(0.5, 5.0))
        shape = np.exp(lam * x)
    elif style == 1:
        center = float(rng.uniform(0.2, 0.8))
        width = float(rng.uniform(0.05, 0.30))
        shape = np.exp(-(((x - center) / width) ** 2))
    elif style == 2:
        mu = float(rng.uniform(0.2, 0.8))
        k = float(rng.uniform(2.0, 25.0))
        shape = 1.0 / (1.0 + np.exp(-k * (x - mu)))
    elif style == 3:
        mode = int(rng.integers(1, 9))
        shape = 0.5 + 0.5 * np.cos(2.0 * np.pi * mode * x)
    elif style == 4:
        a = float(rng.uniform(0.2, 3.0))
        b = float(rng.uniform(0.2, 3.0))
        shape = (x + eps) ** a * (1.0 - x + eps) ** b
    else:
        shape = rng.normal(0.0, 1.0, n)

    shape = np.where(np.isfinite(shape), shape, 0.0)
    mean = float(shape.mean())
    std = float(shape.std())
    if std > 1e-12:
        shape = (shape - mean) / std
    else:
        shape = rng.normal(0.0, 1.0, n)

    base = p + (1e-12 / float(n))
    out = base * np.exp(float(scale) * shape)
    return normalize_nonneg(out)


def optimize_local(p, n, time_limit, rng, verbose=False):
    p = np.asarray(p, dtype=float)
    n = int(n)
    if p.size != n:
        p = upsample_sequence(p, n)
    p = normalize_nonneg(p)

    c = fast_convolution(p)
    cur = p.copy()
    cur_c = c.copy()
    cur_s = float(2.0 * n * np.max(cur_c))
    best = p.copy()
    best_s = cur_s

    lower_bound = (2.0 * float(n)) / (2.0 * float(n) - 1.0)

    if time_limit <= 0.1:
        return best, best_s

    eta = 0.25
    temp = 8.0
    stagnation = 0
    t0 = time.monotonic()
    limit = t0 + max(0.0, float(time_limit))
    it = 0

    while time.monotonic() < limit:
        it += 1
        if best_s <= lower_bound * 1.0005:
            break

        cmax = float(np.max(cur_c))
        if cmax <= 0.0 or not np.isfinite(cmax):
            break

        w = make_weights_from_c(cur_c, temp)
        g = pressure_from_weights(cur, w)
        gmax = float(np.max(g))
        if not np.isfinite(gmax) or gmax <= 0.0:
            temp = min(50.0, temp * 1.2)
            continue

        d = g / gmax

        best_trial = cur
        best_trial_c = cur_c
        best_trial_s = cur_s

        etas = (0.5 * eta, eta, 1.5 * eta)
        for et in etas:
            if not np.isfinite(et) or et <= 0.0:
                continue
            trial = cur * np.exp(-et * d)
            trial = np.where(np.isfinite(trial), trial, 0.0)
            trial = np.maximum(trial, 0.0)
            ts = float(trial.sum())
            if ts <= 1e-12:
                continue
            trial = trial / ts
            tc = fast_convolution(trial)
            sc = float(2.0 * n * np.max(tc))
            if sc < best_trial_s:
                best_trial_s = sc
                best_trial = trial
                best_trial_c = tc

        if best_trial_s < cur_s:
            cur = best_trial
            cur_c = best_trial_c
            cur_s = best_trial_s
            eta = min(2.5, eta * 1.10)
        else:
            eta = max(0.005, eta * 0.70)

        if cur_s < best_s:
            best = cur.copy()
            best_s = cur_s
            stagnation = 0
            temp = max(0.05, temp * 0.90)
        else:
            stagnation += 1
            temp = min(50.0, temp * 1.05)
            if stagnation >= 6:
                stagnation = 0
                scale = 0.10 if temp < 10.0 else 0.25
                if rng.random() < 0.55:
                    trial = structured_perturb(cur, scale, rng)
                else:
                    base = cur + (1e-12 / float(n))
                    trial = base * np.exp(rng.normal(0.0, scale, n))
                    trial = normalize_nonneg(trial)
                tc = fast_convolution(trial)
                sc = float(2.0 * n * np.max(tc))
                if sc < cur_s * 1.03 and sc < best_s * 1.10:
                    cur = trial
                    cur_c = tc
                    cur_s = sc
                    eta = 0.25

        if verbose and (it % 200 == 0):
            print("  local n={} it={} best={:.9f}".format(n, it, best_s))

    if best_s < cur_s:
        return best, best_s
    return cur, cur_s


def get_parent_sequence():
    try:
        h = height_sequence_1
    except NameError:
        return None
    try:
        arr = np.asarray(h, dtype=float)
    except Exception:
        return None
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if float(arr.sum()) <= 0.0:
        return None
    return arr


def add_job(jobs, p, n, score):
    if not np.isfinite(score):
        return
    p = np.asarray(p, dtype=float)
    n = int(n)
    if p.size != n:
        return
    p = normalize_nonneg(p)
    jobs.append(
        {
            "p": p,
            "n": n,
            "score": float(score),
            "tries": 0,
            "streak": 0,
        }
    )


def job_key(job):
    s = float(job["score"])
    t = int(job["tries"])
    st = int(job["streak"])
    return s * (1.0 + 0.02 * t) + 0.002 * st


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return the best sequence found within the allowed runtime."""
    t_start = time.monotonic()
    budget = max(5.0, float(budget_s))
    search_deadline = t_start + max(5.0, budget - 45.0)
    final_deadline = t_start + max(5.0, budget - 3.0)

    rng = np.random.default_rng(seed)
    parent = get_parent_sequence()

    n_values = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384]
    if parent is not None:
        pn = int(parent.size)
        if 4 <= pn <= 20000:
            n_values.append(pn)

    seen_n = set()
    clean_n = []
    for n in n_values:
        n = int(n)
        if n >= 4 and n not in seen_n:
            seen_n.add(n)
            clean_n.append(n)
    n_values = sorted(clean_n)
    max_n = 16384

    best_p = None
    best_score = np.inf
    jobs = []

    for n in n_values:
        if time.monotonic() >= search_deadline:
            break
        if n <= 64:
            num = 56
        elif n <= 512:
            num = 40
        elif n <= 2048:
            num = 28
        elif n <= 8192:
            num = 18
        else:
            num = 10

        starts = make_starts(n, rng, num, parent)
        topk = 3 if n <= 256 else 2
        top = []

        for p in starts:
            c = fast_convolution(p)
            sc = float(2.0 * n * np.max(c))
            if np.isfinite(sc):
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                if len(top) < topk:
                    top.append([sc, p])
                else:
                    worst_idx = 0
                    worst_val = top[0][0]
                    for i in range(1, len(top)):
                        if top[i][0] > worst_val:
                            worst_val = top[i][0]
                            worst_idx = i
                    if sc < worst_val:
                        top[worst_idx] = [sc, p]

        for item in top:
            add_job(jobs, item[1], n, item[0])

    if not jobs:
        add_job(jobs, np.ones(64, dtype=float), 64, 2.0)

    print("initial best fast score = {:.9f}".format(best_score if np.isfinite(best_score) else -1.0))

    max_jobs = 30
    while time.monotonic() < search_deadline and jobs:
        if len(jobs) > max_jobs:
            jobs.sort(key=job_key)
            while len(jobs) > max_jobs:
                removed = False
                for idx in range(len(jobs) - 1, -1, -1):
                    if jobs[idx]["tries"] > 0:
                        del jobs[idx]
                        removed = True
                        break
                if not removed:
                    del jobs[-1]

        jobs.sort(key=job_key)
        job = jobs[0]
        remaining = search_deadline - time.monotonic()
        if remaining < 2.0:
            break

        tries = int(job["tries"])
        streak = int(job["streak"])

        if tries == 0:
            alloc = 30.0
        elif streak == 0:
            alloc = 60.0
        elif streak == 1:
            alloc = 40.0
        elif streak == 2:
            alloc = 20.0
        else:
            alloc = 8.0

        if job["n"] > 8192:
            alloc *= 0.85
        alloc = min(alloc, remaining, 120.0)
        if alloc < 2.0:
            alloc = min(2.0, remaining)

        old_score = float(job["score"])
        p_opt, score_opt = optimize_local(job["p"], job["n"], alloc, rng, verbose=False)

        job["p"] = p_opt
        job["score"] = float(score_opt)
        job["tries"] = tries + 1
        if score_opt < old_score * (1.0 - 1e-6):
            job["streak"] = 0
        else:
            job["streak"] = streak + 1

        if score_opt < best_score:
            best_score = float(score_opt)
            best_p = p_opt.copy()
            print(
                "improved n={} score={:.9f} tries={} elapsed={:.1f}s".format(
                    job["n"], best_score, job["tries"], time.monotonic() - t_start
                )
            )

        if time.monotonic() < search_deadline - 60.0:
            target = None
            for cand in n_values:
                if cand > job["n"] and cand <= max_n:
                    target = cand
                    break
            if target is not None:
                child1 = upsample_sequence(p_opt, target)
                c1 = fast_convolution(child1)
                s1 = float(2.0 * target * np.max(c1))
                add_job(jobs, child1, target, s1)

                if job["n"] <= 2048:
                    child2 = upsample_block_sequence(p_opt, target)
                    c2 = fast_convolution(child2)
                    s2 = float(2.0 * target * np.max(c2))
                    add_job(jobs, child2, target, s2)

            if rng.random() < 0.25 and time.monotonic() < search_deadline - 30.0:
                nn = int(job["n"])
                child = p_opt * np.exp(rng.normal(0.0, 0.25, nn))
                child = normalize_nonneg(child)
                cc = fast_convolution(child)
                sc = float(2.0 * nn * np.max(cc))
                add_job(jobs, child, nn, sc)

    if best_p is None:
        best_p = np.ones(128, dtype=float)
        best_p = normalize_nonneg(best_p)
        best_score = 2.0

    remaining = final_deadline - time.monotonic()
    if remaining > 35.0:
        polish_limit = min(90.0, remaining - 25.0)
        if polish_limit > 5.0:
            opt_p, opt_s = optimize_local(best_p, len(best_p), polish_limit, rng, verbose=False)
            if np.isfinite(opt_s) and opt_s < best_score:
                best_p = opt_p
                best_score = float(opt_s)
                print("final polish best score = {:.9f}".format(best_score))

    if time.monotonic() < final_deadline - 1.0:
        try:
            seq_list = [float(v) for v in best_p]
            direct = evaluate_sequence(seq_list)
            if np.isfinite(direct):
                print("direct check score = {:.9f}".format(direct))
            else:
                fallback = np.exp(np.linspace(0.0, 1.0, 64))
                fallback = normalize_nonneg(fallback)
                return [float(v) for v in fallback]
        except Exception:
            pass

    return [float(v) for v in best_p]
