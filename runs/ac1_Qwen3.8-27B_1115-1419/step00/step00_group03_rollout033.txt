import numpy as np
import time

def simplex_project(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_convolution(a):
    """Compute full linear convolution using FFT for efficiency."""
    n = len(a)
    size = 2 * n - 1
    # Use FFT
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    fc = fa * fa
    c = np.fft.irfft(fc, fsize)[:size]
    return c

def smoothed_max_gradient(a, tau):
    """Compute the gradient of the softmax-smoothed max of a*a."""
    c = compute_convolution(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)

    g = 2.0 * np.correlate(p, a, mode='valid')
    return g, c, p

def simplex_project(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)

The main issue is: the softmax gradient is a weighted average of the "shift gradients" $2 \cdot (P_k a)$ where $P_k$ is the shift matrix. When the temperature is high, the weights are diffuse and the effective gradient is like the average of all shifts, which is basically proportional to $a$ itself (since $\sum_k P_k a = (\sum_k 1) \cdot a$... wait no, that's not right).

Actually, $\sum_k p_k \cdot 2(P_k a)$ where $p_k$ is a probability distribution over $k=0,...,2n-2$. Each $P_k a$ is a shifted version of $a$. So the effective gradient is a "blurred" version of $a$, where the blur is determined by the softmax weights $p$.

When $p$ is uniform (high temperature), the effective gradient is $2 \cdot \bar{a}$ where $\bar{a}$ is the "average shift" of $a$, which is basically $a$ convolved with a box function of the relevant size. This would push all elements down equally, which after projection just rescales $a$. So high temperature is not useful.

When $p$ is concentrated on the max entries (low temperature), the effective gradient is close to $2 \cdot (P_k a)$ for the argmax $k$. This is the standard subgradient of the max, which is what we want.

So the temperature should start moderate and decrease. The key is: at what temperature is the gradient most informative?

If there are, say, 5 entries near the max, we want the temperature to be such that the softmax concentrates on those 5 but not just 1. This is like choosing $\tau \approx (c_{\text{max}} - c_{\text{5th}})$ where $c_{\text{5th}}$ is the 5th largest entry.

    a = simplex_project(a0.copy())
    n = a.size
    tend = time.monotonic() + tlimit
    best = a.copy()
    best_sc = obj_val(a)
    stall = 0
    stall_max = 2000
    eta = 1.0
    it = 0
    
    while time.monotonic() < tend:
        it += 1
        c = fft_conv(a)
        tau = auto_tau(c)
        cm = c.max()
        p = np.exp((c - cm) / tau)
        p /= p.sum()
        g = 2.0 * np.correlate(p, a, mode='valid')
        
        lr = eta
        accepted = False
        for _ in range(12):
            trial = simplex_project(a - lr * g)
            tc = fft_conv(trial)
            sc = 2.0 * n * tc.max()
            if sc < best_sc * (1.0 - 1e-10):
                a = trial
                best_sc = sc
                best = a.copy()
                stall = 0
                eta = min(lr * 1.1, 50.0)
                accepted = True
                break
            lr *= 0.5
        
        if not accepted:
            stall += 1
            eta = max(eta * 0.85, 1e-5)
            if stall >= stall_max:
                a = best + rng.normal(0, 0.05 / np.sqrt(n) * np.sqrt(stall_max // 200 + 1), n)
                a = simplex_project(a)
                eta = 1.0
                stall = 0
                stall_max = max(500, stall_max // 2)
        
        if it % 100000 == 0:
            print(f"        it={it} sc={best_sc:.8f}")
    
    return best, best_sc
