import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix

def build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    # A has shape (2n-1, n)
    # For each column j, non-zeros at rows j..j+n-1 (clipped to 0..2n-2)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = j + i  # k - j = i, so A[k, j] = f[i]
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_iteration(f, M, max_time=2.0):
    """One LP-based iteration. Returns new g or None."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    start = time.monotonic()
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': max_time})
    elapsed = time.monotonic() - start
    
    if res.success:
        return res.x, elapsed
    return None, elapsed

LP-based optimization for minimum autocorrelation peak sequences.
Uses iterative LP relaxations (state-of-the-art approach) combined with
local search and multiple initializations across various sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f):
    """Build sparse (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    i_idx = np.arange(n)
    # For column j, non-zeros at rows j, j+1, ..., j+n-1 with values f[0], f[1], ..., f[n-1]
    # Total non-zeros: n*(n+1) approximately, but exactly n*n since each column has n entries
    # clipped to valid row range [0, 2n-2]
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        r_start = j
        r_end = min(j + n, 2 * n - 1)
        r_count = r_end - r_start
        rows_list.extend(range(r_start, r_end))
        cols_list.extend([j] * r_count)
        vals_list.extend(f[:r_count].tolist())
    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    return A


def lp_solve(f, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            continue
        cand_score = evaluate_sequence(candidate.tolist())
        if np.isfinite(cand_score) and cand_score < score:
            f = candidate
            score = cand_score
    return f, score
