"""
Multi-phase search minimizing 2*n*max(conv(f,f))/sum(f)^2:
1. Parametric family search (linear, power-law, exponential)
2. FFT-based multiplicative-weights LP refinement
3. Block-perturbation local search
Key insight: monotone sequences give score ~4/3 for large n, far below the all-ones baseline of 2.

import numpy as np
import time


def _fft_convolve(a, b, size=None):
    """Full linear convolution via FFT."""
    n_out = len(a) + len(b) - 1
    if size is None:
        size = 1
        while size < n_out:
            size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n_out]


def _fast_score(a, fft_size=None):
    """Fast equivalent of evaluate_sequence (no security checks)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e18
    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (S * S))

    # A^T y: (A^T y)[j] = sum_m f[m]*y[j+m] = conv(f[::-1], y)[n-1+j]
    Fr = np.fft.rfft(f[::-1], fft_size)
    Y = np.fft.rfft(y, fft_size)
    Ay_full = np.fft.irfft(Fr * Y, fft_size)
    Ayj = Ay_full[n - 1:2 * n - 1]
    # Update g
    g = g * np.exp(-eta * Ayj)
    g = np.maximum(g, 1e-18)
    return g, M


def _block_perturb(base, rng, n):
    """Random multiplicative block perturbation."""
    c = base.copy()
    lo = max(1, n // 200)
    hi = max(2, n // 8)
    w = int(rng.integers(lo, hi + 1))
    s = int(rng.integers(0, max(1, n - w)))
    c[s:s + w] *= float(np.exp(rng.normal(0.0, 0.08)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the sequence minimizing 2n*max(f*f)/S^2.
    Phases: parametric generation, LP refinement via FFT MWUA, local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)

    # --- Get base sequence ---
    base_arr = np.asarray(height_sequence_1, dtype=np.float64)
    n0 = base_arr.size
    if n0 == 0:
        n0 = 6121
        base_arr = np.ones(n0)
    best = np.clip(base_arr, 0.0, 1000.0).copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Start: n={n0}, score={best_score:.6f}", flush=True)

    # --- Phase 1: Parametric families ---
    print("Phase 1: parametric search", flush=True)
    js = np.arange(n0, dtype=np.float64)
    j1 = js + 1.0

    candidates = []
    # Linear (max scale-invariant, no clipping issue since we normalize to [0,1])
    candidates.append(j1 / n0)
    # Power law
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(np.power(j1, p) / np.power(j1, p).max())
    # Exponential ramps
    for a in [0.5, 1.0, 2.0, 3.0, 5.0]:
        e = np.exp(a * js / n0)
        candidates.append(e / e.max())
    # Sine-based
    for p in [0.5, 1.0, 2.0, 4.0]:
        s = np.sin(np.pi * js / (2.0 * (n0 - 1))) ** p
        candidates.append(np.maximum(s, 1e-10))
    # Comb / sparse
    for period in [2, 3, 5]:
        c = np.ones(n0)
        c[::period] = 0.0
        candidates.append(c)
    # Ramp + plateau
    for frac in [0.1, 0.2, 0.3, 0.5]:
        k = max(1, int(n0 * frac))
        r = np.concatenate([np.linspace(0.01, 1.0, k), np.ones(n0 - k)])
        candidates.append(r)

    for ci, cand in enumerate(candidates):
        cand = np.clip(cand, 0.0, 1000.0)
        if not np.all(np.isfinite(cand)):
            continue
        fs = _fast_score(cand)
        if fs < best_score:
            rs = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(rs) and rs < best_score:
                best = cand.copy()
                best_score = rs
                print(f"  Phase1 cand {ci}: score={best_score:.6f}", flush=True)

    print(f"After Phase 1: score={best_score:.6f}", flush=True)

    # --- Phase 2: LP-based refinement (MWUA with FFT) ---
    print("Phase 2: LP refinement", flush=True)
    f = best.copy()
    S = np.sum(f)
    if S < 1e-12:
        f = np.ones(n0)
        S = n0
    target_S = np.sqrt(2.0 * n0)
    f = f * (target_S / S)
    f = np.maximum(f, 0.0)
    # Ensure no value exceeds 1000
    f = np.minimum(f, 1000.0)

    # FFT sizes
    conv_len = 2 * n0 - 1
    corr_len = 3 * n0 - 1  # for A^T y
    fft_main = 1
    while fft_main < conv_len:
        fft_main <<= 1
    fft_corr = 1
    while fft_corr < corr_len:
        fft_corr <<= 1
    # Use one size for both (the larger)
    fft_sz = max(fft_main, fft_corr)

    g = f.copy()
    n = n0

    # Adaptive parameters
    t_pow = 3.0
    eta = 0.05
    blend = 0.1

    n_iters = 0
    for it in range(2000000):
        if time.monotonic() > deadline:
            break

        # LP iteration
        g, M = _lp_iteration(f, g, fft_sz, t_pow, eta)
        Sg = np.sum(g)
        if Sg < 1e-12:
            g = f.copy()
        else:
            g = g * (target_S / Sg)
        g = np.clip(g, 0.0, 1000.0)

        # Blend
        f = (1.0 - blend) * f + blend * g
        f = np.maximum(f, 0.0)
        f = np.minimum(f, 1000.0)

        n_iters += 1

        # Periodic evaluation
        if it % 2000 == 0:
            sf = _fast_score(f, fft_sz)
            sg = _fast_score(g, fft_sz)
            if sf < best_score or sg < best_score:
                rs_f = float(evaluate_sequence(f.tolist()))
                rs_g = float(evaluate_sequence(g.tolist()))
                if np.isfinite(rs_f) and rs_f < best_score:
                    best = f.copy()
                    best_score = rs_f
                    print(f"  LP it{it} f: {best_score:.6f}", flush=True)
                if np.isfinite(rs_g) and rs_g < best_score:
                    best = g.copy()
                    best_score = rs_g
                    print(f"  LP it{it} g: {best_score:.6f}", flush=True)

        # Adapt parameters
        if it % 20000 == 0 and it > 0:
            eta *= 0.9
            blend *= 0.95
            t_pow = min(t_pow + 0.5, 10.0)

    # Final check of f and g
    sf = float(evaluate_sequence(f.tolist()))
    if np.isfinite(sf) and sf < best_score:
        best = f.copy()
        best_score = sf
    sg = float(evaluate_sequence(g.tolist()))
    if np.isfinite(sg) and sg < best_score:
        best = g.copy()
        best_score = sg
    print(f"After Phase 2 ({n_iters} iters): score={best_score:.6f}", flush=True)

    # --- Phase 3: Local search with block perturbations ---
    print("Phase 3: local search", flush=True)
    ns = 0
    while time.monotonic() < deadline:
        cand = _block_perturb(best, rng, n0)
        fs = _fast_score(cand, fft_sz)
        if fs < best_score:
            rs = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(rs) and rs < best_score:
                best = cand
                best_score = rs
                print(f"  Local {ns}: {best_score:.6f}", flush=True)
        ns += 1

    final = float(evaluate_sequence(best.tolist()))
    print(f"Done: score={final:.6f}, n={n0}, local_steps={ns}", flush=True)
    return [float(x) for x in best]
