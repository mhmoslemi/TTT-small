"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses subgradient descent with top-K averaging, momentum, and multiple restarts.
Key insight: asymmetric sequences can reduce the convolution peak below 2.0
(the all-ones baseline). The optimizer breaks symmetry and iteratively
reduces the active peak of the autocorrelation.
"""

import numpy as np
import time


def _fft_convolve(a):
    """Full linear convolution of a with itself using FFT. Output length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    # Next power of 2 for efficient FFT
    fft_size = 1
    while fft_size < 2 * size:
        fft_size *= 2
    fa = np.fft.rfft(a, n=fft_size)
    c = np.fft.irfft(fa * fa, n=fft_size)[:size]
    return c


def _subgradient_topk(c, a, K, rng):
    """
    Compute average subgradient over top-K entries of c.
    Returns (s, max_c) where s[j] = (1/K) * sum_{k in topK} 2*a[k-j] (zero-padded).
    """
    n = len(a)
    max_c = np.max(c)
    if K == 1:
        k_star = int(np.argmax(c))
        s = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            # s[j] = 2*a[k_star - j] for j in [lo, hi]
            # a indices: k_star - j for j in [lo, hi] = [k_star-hi, k_star-lo] reversed
            s[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        return s, max_c
    else:
        # Get top-K indices (handle ties by taking first K)
        # Use partial sort for efficiency
        if K < len(c) // 2:
            # Quickselect-like: get top K
            part = np.argpartition(c, -K)[-K:]
        else:
            part = np.arange(len(c))
        s = np.zeros(n)
        for k in part:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                s[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        s /= float(len(part))
        return s, max_c


def _local_search(a_init, deadline, rng, max_iter=30000):
    """
    Subgradient local search with momentum and adaptive learning rate.
    Returns (best_score, best_sequence) or (inf, None).
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf, None

    # Normalize so sum = n (scale-invariant objective)
    a = a / sum_a * n

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(best_score):
        return np.inf, None

    # Momentum buffer
    v = np.zeros(n)
    momentum = 0.85

    # Learning rate schedule parameters
    lr0 = 1.0
    lr_min = 0.01
    decay = max(1, max_iter // 5)  # decay over first 5/6 of iterations

    # K schedule: start smooth, end sharp
    K_start = 20
    K_end = 1

    no_improve = 0
    threshold = 200  # restart after this many iters without improvement

    for it in range(max_iter):
        if time.monotonic() > deadline - 5.0:
            break

        # Adaptive K
        frac = min(1.0, it / max_iter)
        K = max(1, int(round(K_start + (K_end - K_start) * frac)))

        # Learning rate
        lr = max(lr_min, lr0 / (1.0 + it / decay))

        # Compute convolution
        c = _fft_convolve(a)
        max_c = np.max(c)

        # Subgradient
        s, _ = _subgradient_topk(c, a, K, rng)

        # Full gradient of f(a) = 2*n*max_c / (sum_a)^2
        # df/da[j] = (2n/S^2)*s[j] - 4n*max_c/S^3
        # Since we normalize S = n: df/da[j] = (2/s_n)*s[j] - 4*max_c/n^2
        # where s_n = n (since we keep sum = n)
        # Actually: f = 2n*max_c/n^2 = 2*max_c/n
        # df/dmax_c = 2/n
        # df/da[j] = (2/n)*s[j] (from the chain rule, since we fixed S=n)
        # But we also need to account for the constraint S=n... 
        # Actually since the objective is scale-invariant, working on the 
        # constraint S=n: minimize g(a) = 2*max_c/n subject to S=n, a>=0.
        # The Lagrangian: L = 2*max_c/n - lambda*(S-n)
        # grad = (2/n)*s - lambda*1
        # At optimum, lambda is determined by the constraint.
        # For a projected method, we just use the gradient of max_c (since S is fixed by projection):
        grad = (2.0 / n) * s

        # Subtract mean to maintain sum (projection onto sum=n hyperplane)
        grad -= np.mean(grad)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v

        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)

        # Re-normalize to sum = n
        sum_new = np.sum(a_new)
        if sum_new < 1e-10:
            a_new = best_a.copy()
            sum_new = np.sum(a_new)
        a = a_new / sum_new * n

        # Periodic evaluation
        if it % 80 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                if best_score < 2.0:
                    print(f"    iter {it}: score={best_score:.6f}, n={n}")
            else:
                no_improve += 80

        # Restart if stuck
        if no_improve > threshold and it < max_iter - 200:
            # Perturb best and continue
            a = best_a + 0.05 * np.sum(best_a) / n * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            sum_a = np.sum(a)
            a = a / sum_a * n
            v = np.zeros(n)
            no_improve = 0
            lr = lr0  # reset lr

    # Final evaluation
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a.copy()

    return best_score, best_a


def _make_init(n, init_type, rng):

        # Exponential ramp

        a += 0.05 * rng.standard_normal(n)

        t = np.linspace(0, 2 * np.pi, n)

        a += 0.05 * rng.standard_normal(n)
    elif init_type == 4:
        # Random smooth
        a = np.abs(rng.standard_normal(n)) + 0.1
        # Smooth
        k = min(20, n // 10)
        if k > 1:
            kernel = np.ones(k) / k
            a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
        a += 0.1 * rng.standard_normal(n)

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n)
    return a
