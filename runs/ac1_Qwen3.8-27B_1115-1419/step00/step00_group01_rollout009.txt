    # Maximize: sum g
    # Constraints: for k=0..2n-2, sum_{i} f[i]*g[k-i] <= M (valid i)
    #             g[j] >= 0

    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for k in range(2*n-1):

            j = k - i
            if 0 <= j < n:
                row[j] += f[i]
        A_ub.append(row)
        b_ub.append(M)

    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

from scipy.optimize import linprog

def compute_objective(seq):
    """Fast computation of the objective (same as evaluate_sequence but numepy-based)."""
    n = len(seq)
    S = np.sum(seq)
    if S < 0.01:
        return np.inf
    c = np.convolve(seq, seq)
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix: (f*g)[k] <= M for k=0..2n-2
    # (f*g)[k] = sum_{i} f[i]*g[k-i] for valid i, k-i
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k+1, n)):
            j = k - i
            if 0 <= j < n:
                A[k, j] += f[i]
    
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:

    return None

"""
Multi-phase optimizer for step function sequence to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding, gradient descent, and local search.
"""
import numpy as np
import time
import sys
from scipy.optimize import linprog

def fast_eval(a):
    """Compute objective without security checks."""
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    return 2.0 * n * max_c / (S * S)

def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP."""
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        A[k, jlo:jhi+1] = f[k-jhi:k-jlo+1][::-1]
    return A

def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = build_lp_matrix(f, n)
    b = np.full(2*n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def gradient_step(a, alpha, n):
    """One gradient descent step on max autocorrelation."""
    c = np.convolve(a, a)
    S = float(np.sum(a))
    kstar = int(np.argmax(c))
    max_c = c[kstar]
    
    # Subgradient: d_i = (max_c/S)*a_i - a[kstar-i] (where valid)
    d = np.zeros(n)
    # a[kstar-i] is valid when 0 <= kstar-i < n, i.e., kstar-n+1 <= i <= kstar
    ilow = max(0, kstar - n + 1)
    ihigh = min(n - 1, kstar)
    # d[i] = (max_c/S)*a[i] - a[kstar-i]
    d[:] = (max_c / S) * a
    d[ilow:ihigh+1] -= a[kstar - ihigh : kstar - ilow + 1][::-1]
    
    a_new = a - alpha * d
    a_new = np.clip(a_new, 0, None)
    S_new = np.sum(a_new)
    if S_new < 0.01:
        return a
    a_new = a_new * S / S_new
    return a_new

    f = f.copy()
    S = np.sum(f)
    f = f / S  # normalize to sum = 1
    
    c = np.convolve(f, f)
    M = float(np.max(c))
    best_f = f.copy()
    best_score = 2.0 * n * M
    
    alpha = 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(f, f)
        M = float(np.max(c))
        score = 2.0 * n * M
        if score < best_score:
            best_f = f.copy()
            best_score = score
        
        # Gradient step
        kstar = int(np.argmax(c))
        S_curr = float(np.sum(f))
        
        # d[i] = (M/S)*f[i] - f[kstar-i] (valid)
        d = (M / S_curr) * f - np.zeros(n)
        # Subtract f[kstar-i] where valid
        ilow = max(0, kstar - n + 1)
        ihigh = min(n - 1, kstar)
        d[ilow:ihigh+1] -= f[kstar - ihigh : kstar - ilow + 1][::-1]
        
        # Adaptive step size
        f_new = f - alpha * d
        f_new = np.clip(f_new, 0, None)
        f_new_sum = np.sum(f_new)
        if f_new_sum < 0.01:
            alpha *= 0.5
            continue
        f_new = f_new / f_new_sum
        
        c_new = np.convolve(f_new, f_new)
        M_new = float(np.max(c_new))
        if M_new > M * 1.1:
            alpha *= 0.5
        else:
            f = f_new
            if M_new < M * 0.99:
                alpha = min(alpha * 1.1, 1.0)
    
    return best_f, best_score

def local_search(f, n, max_iters, deadline, rng):
    """Local search with block perturbations."""
    f = f.copy()
    c = np.convolve(f, f)
    M = float(np.max(c))
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = 2.0 * n * M / (S * S)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Random block perturbation
        width = int(rng.integers(2, max(3, n // 50)))
        start = int(rng.integers(0, max(1, n - width)))
        log_scale = rng.normal(0, 0.3)
        f_new = f.copy()
        f_new[start:start+width] *= np.exp(log_scale)
        f_new = np.clip(f_new, 0, None)
        S_new = np.sum(f_new)
        if S_new < 0.01:
            continue
        f_new = f_new * S / S_new  # keep same sum
        
        c_new = np.convolve(f_new, f_new)
        M_new = float(np.max(c_new))
        score_new = 2.0 * n * M_new / (S * S)
        
        if score_new < best_score:
            best_f = f_new
            best_score = score_new
            f = f_new
            M = M_new
        else:
            f = np.clip(f + 0.05 * rng.standard_normal(n), 0, None)
            f = f * S / np.sum(f)
    
    return best_f, best_score
