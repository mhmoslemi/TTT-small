def build_conv_matrix(f):
    """Build the convolution constraint matrix for the LP."""
    n = len(f)
    diagonals = [np.full(n, f[m]) for m in range(n)]
    offsets = [-m for m in range(n)]
    A = diags(diagnals, offsets, shape=(2*n - 1, n), format='csr')
    return A

def solve_lp(f, M, time_limit=30):
    """Solve the LP: max sum(b) s.t. A@b <= M, b >= 0."""
    import cvxpy as cp
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    A = build_conv_matrix(f)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except:
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            return None
    if b.value is None:
        return None
    return np.asarray(b.value, dtype=np.float64)

def gs_step(a, N=None):
    """One Gerchberg-Saxton step."""
    n = len(a)
    if N is None:
        N = 1
        while N < 2 * n:
            N *= 2
    A = np.fft.rfft(a, N)
    if abs(A[0]) < 1e-15:
        return a
    A_new = abs(A[0]) * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

def subgradient_descent(a, n_iter=50000, initial_step=0.1):
    """Subgradient descent on max(a*a)."""
    a = a.copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    step = initial_step
    for it in range(n_iter):
        b = np.convolve(a, a)
        max_b = np.max(b)
        if max_b < 1e-15:
            break
        
        # Find active set (top 10 or 99% of max)
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = slice(k - j_end, k - j_start + 1)
                grad[j_start:j_end + 1] += 2.0 * a[idx][::-1]
        
        grad /= len(active)
        
        # Update
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        # Adapt step size
        if it % 500 == 0:
            step *= 0.97
    
    return a
