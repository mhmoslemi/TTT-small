    print(f"Parent score: {parent_score:.6f}")
    
    # 1c. Linear ramp
    ramp = np.linspace(0.01, 1.0, n)
    ramp = ramp / ramp.sum()
    ramp_score = evaluate_sequence(ramp.tolist())
    print(f"Ramp score: {ramp_score:.6f}")
    
    # 1d. Sinusoidal
    x = np.arange(n)
    sine = 1.0 + 0.5 * np.cos(2 * np.pi * x / n)
    sine = np.maximum(sine, 0)
    sine = sine / sine.sum()
    sine_score = evaluate_sequence(sine.tolist())
    print(f"Sine score: {sine_score:.6f}")
    
    # Pick best starting point
    candidates = {
        'flat': (flat, flat_score),
        'parent': (parent / max(parent.sum(), 1e-10), parent_score),
        'ramp': (ramp, ramp_score),
        'sine': (sine, sine_score),
    }
    best_name = min(candidates, key=lambda k: candidates[k][1])
    a = candidates[best_name][0].copy()
    best_score = candidates[best_name][1]
    best = a.copy()
    print(f"Best starting point: {best_name} with score {best_score:.6f}")
    
    # --- Phase 2: Smoothed-max subgradient optimization ---
    t_temp = 10.0  # temperature for softmax
    lr = 0.02
    iter_count = 0
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() < deadline - 60:
        # Compute convolution
        A = np.fft.rfft(a, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2*n-1]
        
        # Peak info
        peak_val = b.max()
        peak_idx = b.argmax()
        
        # Softmax weights for smoothed max
        b_shifted = b - peak_val
        logits = t_temp * b_shifted
        p = np.exp(logits)
        p_sum = p.sum()
        if p_sum > 0:
            p /= p_sum
        else:
            p = np.zeros(2*n-1)
            p[peak_idx] = 1.0
        
        # Gradient: grad[j] = 2 * sum_m a[m] * p[j+m]
        P = np.fft.rfft(p, fft_size)
        corr = np.fft.irfft(np.conj(A) * P, fft_size)
        grad = 2.0 * corr[:n]
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        else:
            a = a  # stay
            break
        
        # Evaluate periodically
        iter_count += 1
        if iter_count % 5 == 0:
            score = evaluate_sequence(a_new.tolist())
            if np.isfinite(score) and score < best_score:
                best = a_new.copy()
                best_score = score
                no_improve = 0
                if iter_count % 100 == 0:
                    print(f"Iter {iter_count}: score={best_score:.6f}, t={t_temp:.1f}, lr={lr:.4f}")
            else:
                no_improve += 1
        
        a = a_new
        
        # Schedule updates
        if iter_count % 100 == 0:
            t_temp = min(t_temp * 1.3, 500.0)
            lr = max(lr * 0.95, 0.001)
        
        if no_improve > max_no_improve:
            # Random restart from best
            a = best + 0.01 * rng.standard_normal(n) * best.max()
            a = np.maximum(a, 0)
            a /= a.sum()
            t_temp = max(1.0, t_temp * 0.5)
