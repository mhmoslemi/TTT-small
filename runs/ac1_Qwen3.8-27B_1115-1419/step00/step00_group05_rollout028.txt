```python
def iterative_refine(a, K=10, alpha=0.5, max_iter=100000, deadline=None):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = float('inf')
    best = a.copy()
    
    for it in range(max_iter):
        if deadline and time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a, mode='full')
        M = c.max()
        score = 2 * n * M  # (sum=1, so score = 2*n*M)
        
        if score < best_score:
            best_score = score
            best = a.copy()
        
        # Find top K peaks
        top_k = np.argpartition(c, -K)[-K:]
        
        # Compute pressure
        p = np.zeros(n)
        for k in top_k:
            # p_j += a_{k-j} for valid j
            # k-j is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                # a_{k-j} for j in [j_start, j_end]
                idx = k - np.arange(j_start, j_end + 1)
                p[j_start:j_end+1] += a[idx]
        
        p /= K
        p_max = p.max()
        if p_max < 1e-15:
            break
        
        # Multiplicative update
        a = a * np.exp(-alpha * p / p_max)
        a = np.clip(a, 1e-10, None)  # prevent underflow
        a = a / a.sum()
    
    return best, best_score
```
