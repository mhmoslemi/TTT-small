def lp_direction(f, M):
    """Find direction d that maximizes sum(d) subject to conv constraints."""
    n = len(f)
    b = np.convolve(f, f)
    
    # Variables: d_0, ..., d_{n-1}
    # Objective: max sum(d) => minimize -sum(d)
    c = -np.ones(n)
    
    # Constraints: conv(f, d)[k] <= (M - b[k]) / 2 for all k
    # conv(f, d)[k] = sum_i f[i] * d[k-i] (for valid i, k-i)
    # This is a linear function of d.
    
    # For each k, the coefficient of d[j] in conv(f,d)[k] is f[k-j] (if valid)
    # So the constraint matrix A has A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0.
    
    # A is a (2n-1) x n matrix.
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    
    # For large n, this matrix is sparse (each row has at most n non-zeros, but actually
    # for row k, the non-zeros are at positions j where 0 <= k-j <= n-1, i.e.,
    # max(0, k-(n-1)) <= j <= min(n-1, k). So each row has at most min(k+1, n, 2n-1-k) non-zeros.
    
    # This is a banded matrix! The bandwidth is at most n.
    
    # A_ub is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx <= n-1:
                A[k, j] = f[idx]
    
    b_ub = (M - b) / 2
    
    # Bounds: d_i >= -f_i, d_i <= some large value (or infinity)
    bounds = [(-f[i], None) for i in range(n)]
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
