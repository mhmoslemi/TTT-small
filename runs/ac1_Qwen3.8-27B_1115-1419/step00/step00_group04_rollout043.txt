def smooth_max_gradient(x, lam=100):
    n = len(x)
    c = np.convolve(x, x)
    # Soft max
    max_c = np.max(c)
    exp_c = np.exp(lam * (c - max_c))
    weights = exp_c / np.sum(exp_c)
    # Gradient
    sg = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] > 1e-10:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            j_vals = k - np.arange(i_lo, i_hi + 1)
            sg[i_lo:i_hi+1] += weights[k] * x[j_vals]
    sg *= 2.0
    return sg, np.max(c) + np.log(np.sum(exp_c)) / lam

    def objective_and_gradient(z):

        lam = 50
        max_c = np.max(c)
        exp_c = np.exp(lam * (c - max_c))
        log_sum_exp = max_c + np.log(np.sum(exp_c)) / lam
        weights = exp_c / np.sum(exp_c)

        sg_x = np.zeros(n)
        for k in range(2*n-1):
            w = weights[k]
            if w > 1e-12:
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                j_vals = k - np.arange(i_lo, i_hi + 1)
                sg_x[i_lo:i_hi+1] += w * x[j_vals]
        sg_x *= 2.0
        
        # Gradient w.r.t. z
        # dF/dz_j = sum_i (dF/dx_i) * x_i * (δ_ij - x_j)
        # = x_j * (sum_i dF/dx_i * x_i) - x_j * (dF/dx_j) ... no

        sg_z = x * sg_x - x * sum_sgx
        
        return log_sum_exp, sg_z
    
    result = minimize(objective_and_gradient, z, method='L-BFGS-B', 
                      options={'maxiter': 10000, 'ftol': 1e-12})
    
    return softmax(result.x)
