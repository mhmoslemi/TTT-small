def fft_convolve_full(a,b):
    n = len(a)+len(b)-1
    size = 1 << (n-1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa*fb, size)
    return c[:n]
```

    s = p.sum()
    if s <=0 or not finite: p=np.ones_like(q)/len(q)

    return p

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2
by working with a normalized sum(a)=1 and a smooth log-sum-exp surrogate
for the maximum convolution coefficient.  The search uses:
  - multi-resolution initialization on smaller lengths,
  - log-domain Adam optimization of a smooth peak surrogate,
  - several structured and random asymmetric initializations,
  - a final low-pass/block local search,
  - exact evaluation of a small pool of top candidates.
"""

import numpy as np
import time
from operator import itemgetter


def _next_pow2(x):
    x = int(max(1, x))
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _p_to_q(p):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 1e-300, 1e300)
    q = np.log(p)
    if np.any(np.isfinite(q)):
        q = np.nan_to_num(q, nan=0.0, posinf=40.0, neginf=-40.0)
    q -= np.mean(q)
    return np.clip(q, -40.0, 40.0)


def _softmax(q):
    q = np.asarray(q, dtype=np.float64).reshape(-1)
    if q.size == 0:
        return q
    q = np.nan_to_num(q, nan=0.0, posinf=40.0, neginf=-40.0)
    q -= np.mean(q)
    q = np.clip(q, -40.0, 40.0)
    q -= float(np.max(q))
    p = np.exp(q)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones(q.shape, dtype=np.float64) / float(q.size)
    else:
        p = p / s
    return p


def _fft_convolve_full_size(a, b, size):
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    ab = fa * fb
    c = np.fft.irfft(ab, size)
    return c[: len(a) + len(b) - 1]


def _fft_correlate_shift_size(w, p, size):
    # Returns corr[i] = sum_{m=0}^{n-1} p[m] * w[i+m], length n.
    n = int(p.size)
    pr = p[::-1]
    fa = np.fft.rfft(pr, size)
    fb = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[n - 1:n + n - 1]


def _score_p(p, n=None, size=None):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    if n is None:
        n = p.size
    else:
        n = int(n)
    if p.size != n:
        if p.size < n:
            tmp = np.zeros(n, dtype=np.float64)
            tmp[: p.size] = p
            p = tmp
        else:
            p = p[:n]
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 0.0, None)
    s = float(np.sum(p))
    if s < 0.01:
        return float(np.inf)
    p = p / s
    if size is None:
        size = _next_pow2(2 * n - 1)
    c = _fft_convolve_full_size(p, p, size)
    mc = float(np.max(c))
    if not np.isfinite(mc) or mc < 0.0:
        mc = 0.0
    return float(2.0 * n * mc)


def _objective_grad(q, n, sizeC, sizeW, tau_target, weight_smooth):
    n = int(n)
    p = _softmax(q)
    c = _fft_convolve_full_size(p, p, sizeC)
    mc = float(np.max(c))

    if not np.isfinite(mc) or mc <= 0.0:
        score = 0.0
        grad_q = np.zeros(n, dtype=np.float64)
        return score, grad_q, p, mc

    tau = float(tau_target) / mc
    if not np.isfinite(tau) or tau <= 0.0:
        tau = 1.0
    tau = min(max(tau, 1.0), 1000000.0)

    shifted = tau * c
    shifted -= float(np.max(shifted))
    expv = np.exp(shifted)
    ss = float(np.sum(expv))
    if not np.isfinite(ss) or ss <= 0.0:
        w = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)
    else:
        w = expv / ss
        if weight_smooth > 0.0:
            w = w * (1.0 - float(weight_smooth)) + (float(weight_smooth) / float(2 * n - 1))
            ws = float(np.sum(w))
            if np.isfinite(ws) and ws > 0.0:
                w = w / ws
        else:
            ws = float(np.sum(w))
            if np.isfinite(ws) and ws > 0.0:
                w = w / ws

    corr = _fft_correlate_shift_size(w, p, sizeW)
    grad_p = 2.0 * corr
    dprev = float(np.dot(p, grad_p))
    if not np.isfinite(dprev):
        dprev = 0.0
    grad_q = p * (grad_p - dprev)
    if not np.all(np.isfinite(grad_q)):
        grad_q = np.zeros(n, dtype=np.float64)

    score = float(2.0 * n * mc)
    return score, grad_q, p, mc


def _add_init(inits, p):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 1e-12, 1e6)
    s = float(np.sum(p))
    if s <= 0.0:
        return
    p = p / s
    inits.append(_p_to_q(p))


def _generate_inits(n, rng):
    n = int(max(2, n))
    inits = []
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)

    _add_init(inits, np.ones(n, dtype=np.float64))

    # Linear asymmetric ramps (high left, low right).
    for a in (0.5, 0.9, 1.3, 2.0, 3.0):
        p = 1.0 + a * (0.5 - t)
        _add_init(inits, p)

    # Beta-family shapes: p ~ t^alpha * (1-t)^beta, high left for beta > alpha.
    ta = np.clip(t, 1e-6, 1.0 - 1e-6)
    logt = np.log(ta)
    logone = np.log1p(-ta)
    for alpha in (0.0, 0.1, 0.3, 1.0):
        for beta in (1.0, 2.0, 3.0, 5.0):
            p = np.exp(alpha * logt + beta * logone)
            _add_init(inits, p + 1e-12)

    # Smooth two-level steps.
    for alpha in (0.35, 0.45, 0.55, 0.65):
        for w in (0.03, 0.08, 0.15):
            for r in (1.0, 2.0, 3.0):
                z = (alpha - t) / w
                z = np.clip(z, -25.0, 25.0)
                p = 1.0 + r / (1.0 + np.exp(-z))
                _add_init(inits, p)

    # Deterministic decreasing block steps.
    arange_f = np.arange(n, dtype=np.float64)
    for K in (2, 3, 4, 5, 6, 8):
        idx = np.floor(arange_f * K / float(n)).astype(int)
        idx = np.clip(idx, 0, K - 1)
        for r in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
            levels = np.linspace(1.0 + r, 1.0, K, dtype=np.float64)
            p = levels[idx].astype(np.float64)
            _add_init(inits, p)

    # Random exponential-drift sequences.
    for lam in (1.0, 2.0, 3.0, 4.0):
        for sig in (0.3, 0.8, 1.2):
            p = np.exp(lam * (1.0 - t) + sig * rng.normal(0.0, 1.0, n))
            _add_init(inits, p)

    # Low-order asymmetric Fourier perturbations.
    for _ in range(10):
        a = float(rng.uniform(-0.8, 0.8))
        b = float(rng.uniform(-0.8, 0.8))
        c = float(rng.uniform(-0.5, 0.5))
        p = (
            1.0
            + a * np.cos(np.pi * t)
            + b * np.cos(2.0 * np.pi * t)
            + c * np.sin(np.pi * t)
        )
        _add_init(inits, p)

    return inits


def _upsample_q(q_small, n_big):
    q_small = np.asarray(q_small, dtype=np.float64).reshape(-1)
    q_small = np.nan_to_num(q_small, nan=0.0, posinf=40.0, neginf=-40.0)
    if q_small.size == 0:
        return np.zeros(int(n_big), dtype=np.float64)
    if q_small.size == int(n_big):
        q = q_small.copy()
        q -= np.mean(q)
        return np.clip(q, -40.0, 40.0)

    xs = np.linspace(0.0, 1.0, q_small.size, dtype=np.float64)
    xb = np.linspace(0.0, 1.0, int(n_big), dtype=np.float64)
    q = np.interp(xb, xs, q_small)
    q -= np.mean(q)
    return np.clip(q, -40.0, 40.0)


def _pool_add(pool, score, p, n, max_size=80):
    if score is None or not np.isfinite(score):
        return pool
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    if p.size == 0:
        return pool
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(np.sum(p))
    if s < 0.01:
        return pool
    p = p / s
    pool.append([float(score), int(p.size), p])
    if len(pool) > max_size:
        pool.sort(key=itemgetter(0))
        del pool[max_size:]
    return pool


def _score_inits(inits, n, sizeC):
    scored = []
    for q in inits:
        p = _softmax(q)
        sc = _score_p(p, n, sizeC)
        if np.isfinite(sc):
            scored.append([float(sc), q, p])
    if scored:
        scored.sort(key=itemgetter(0))
    return scored


def _run_optimizer(q0, n, time_limit, rng, base_lr=0.15, tau_target=12.0,
                   weight_smooth=0.015, max_lr=0.55, max_iters=1000000):
    n = int(n)
    if time_limit <= 0.0:
        p = _softmax(q0)
        sc = _score_p(p, n)
        return sc, _p_to_q(p), p

    sizeC = _next_pow2(2 * n - 1)
    sizeW = _next_pow2(3 * n - 2)

    q = np.asarray(q0, dtype=np.float64).reshape(-1).copy()
    q = np.nan_to_num(q, nan=0.0, posinf=40.0, neginf=-40.0)
    q -= np.mean(q)
    q = np.clip(q, -40.0, 40.0)

    score, grad_q, p, _ = _objective_grad(q, n, sizeC, sizeW, tau_target, weight_smooth)
    if not np.isfinite(score):
        p = np.ones(n, dtype=np.float64) / float(n)
        q = _p_to_q(p)
        score, grad_q, p, _ = _objective_grad(q, n, sizeC, sizeW, tau_target, weight_smooth)

    if not np.isfinite(score):
        p = np.ones(n, dtype=np.float64) / float(n)
        return float(2.0 * n / float(n)), _p_to_q(p), p

    best_score = float(score)
    best_q = q.copy()
    best_p = p.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    best_iter = 0
    cur_lr = float(base_lr)

    start = time.monotonic()
    while time.monotonic() - start < max(0.0, float(time_limit)):
        if score < best_score - 1e-13:
            best_score = float(score)
            best_q = q.copy()
            best_p = p.copy()
            best_iter = t
            cur_lr = min(max_lr, cur_lr * 1.01)

        no_improve = int(t - best_iter)
        if no_improve > 300:
            noise = 0.08 if no_improve < 1000 else 0.18
            q = best_q + rng.normal(0.0, noise, n)
            q -= np.mean(q)
            q = np.clip(q, -40.0, 40.0)
            score, grad_q, p, _ = _objective_grad(q, n, sizeC, sizeW, tau_target, weight_smooth)
            if not np.isfinite(score):
                q = best_q.copy()
                score = best_score
                p = best_p.copy()
                grad_q = np.zeros(n, dtype=np.float64)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            best_iter = 0
            cur_lr = max(0.05, cur_lr * 0.75)
            tau_target = min(60.0, tau_target * 1.35)
            weight_smooth = max(0.0, weight_smooth - 0.004)
            continue

        it = int(t + 1)
        step_lr = cur_lr * min(1.0, float(it) / 30.0)
        if np.all(np.isfinite(grad_q)):
            m = 0.9 * m + 0.1 * grad_q
            v = 0.999 * v + 0.001 * (grad_q * grad_q)
            mhat = m / (1.0 - 0.9 ** it)
            vhat = v / (1.0 - 0.999 ** it)
            q = q - step_lr * (mhat / (np.sqrt(vhat) + 1e-12))
            q -= np.mean(q)
            q = np.clip(q, -40.0, 40.0)
        t = it

        if it >= int(max_iters):
            break

    score, grad_q, p, _ = _objective_grad(q, n, sizeC, sizeW, tau_target, weight_smooth)
    if np.isfinite(score) and score < best_score:
        best_score = float(score)
        best_q = q.copy()
        best_p = p.copy()

    return float(best_score), best_q, best_p


def _threshold_normalize(p, thresh):
    p = np.asarray(p, dtype=np.float64).reshape(-1).copy()
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 0.0, 1000.0)
    p[p < thresh] = 0.0
    s = float(np.sum(p))
    if s < 0.01:
        p = np.ones_like(p, dtype=np.float64) / float(max(1, p.size))
    else:
        p = p / s
    return p


def _block_local_search(p, n, time_limit, rng):
    n = int(n)
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    if p.size != n:
        if p.size < n:
            tmp = np.zeros(n, dtype=np.float64)
            tmp[: p.size] = p
            p = tmp
        else:
            p = p[:n]
    p = np.clip(p, 0.0, None)
    s = float(np.sum(p))
    if s < 0.01:
        return float("inf"), np.ones(n, dtype=np.float64) / float(n)
    p = p / s

    sizeC = _next_pow2(2 * n - 1)
    best_score = _score_p(p, n, sizeC)
    best_p = p.copy()
    best_q = _p_to_q(p)

    start = time.monotonic()
    sigma = 0.25
    min_sigma = 0.02

    while time.monotonic() - start < max(0.0, float(time_limit)):
        if n <= 6:
            width = max(1, n)
            starti = 0
        else:
            min_w = max(2, n // 400)
            max_w = max(min_w + 1, min(n - 1, n // 12))
            if max_w >= n:
                max_w = n
            if max_w <= min_w:
                width = min_w
            else:
                width = int(rng.integers(min_w, max_w + 1))
            width = min(width, n)
            hi_start = max(1, n - width + 1)
            starti = int(rng.integers(0, hi_start))

        if width >= n or rng.random() < 0.15:
            delta = rng.normal(0.0, sigma * 0.25, n)
        else:
            delta = np.zeros(n, dtype=np.float64)
            delta[starti:starti + width] = rng.normal(0.0, sigma, width)

        q = best_q + delta
        q -= np.mean(q)
        q = np.clip(q, -40.0, 40.0)
        pc = _softmax(q)
        sc = _score_p(pc, n, sizeC)
        if np.isfinite(sc) and sc < best_score - 1e-13:
            best_score = float(sc)
            best_p = pc.copy()
            best_q = q.copy()
            sigma = min(0.7, sigma * 1.05)
        else:
            sigma = max(min_sigma, sigma * 0.99)

        if sigma <= min_sigma:
            break

    return float(best_score), best_p


def _exact_evaluate_pool(pool, total_start, total_budget):
    eval_fn = globals().get("evaluate_sequence", None)
    if eval_fn is None:
        return None

    pool.sort(key=itemgetter(0))
    if not pool:
        return None

    deadline = total_start + max(30.0, float(total_budget)) - 12.0
    best_score = float("inf")
    best_seq = None
    calls = 0

    # Be conservative with expensive exact direct convolutions for large n.
    max_n = max(item[1] for item in pool)
    if max_n > 12000:
        max_calls = 12
    elif max_n > 8000:
        max_calls = 24
    else:
        max_calls = 48

    for idx in range(len(pool)):
        if calls >= max_calls:
            break
        if time.monotonic() > deadline:
            break

        _, n, p = pool[idx]
        p = np.asarray(p, dtype=np.float64).reshape(-1)
        variants = [p]
        if idx < 4 and n > 8000:
            variants.append(_threshold_normalize(p, 1e-9))
        elif idx < 8:
            variants.append(_threshold_normalize(p, 1e-9))
            variants.append(_threshold_normalize(p, 1e-11))

        for var in variants:
            if calls >= max_calls:
                break
            if time.monotonic() > deadline:
                break
            try:
                seq = [float(x) for x in var.tolist()]
                sc = float(eval_fn(seq))
            except Exception:
                sc = float("inf")
            calls += 1
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq

    return best_seq


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    B = float(max(60.0, budget_s))

    if B > 600.0:
        exact_reserve = min(160.0, max(45.0, B * 0.045))
    else:
        exact_reserve = max(12.0, B * 0.15)

    search_deadline = total_start + max(1.0, B - exact_reserve - 2.0)
    pool = []

    # Parent sequence, if available, is a safe fallback and sometimes useful.
    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64).reshape(-1)
            if p.size > 0:
                sc = _score_p(p, p.size)
                _pool_add(pool, sc, p, p.size)
        except Exception:
            pass

    # ---------------- Multi-resolution warm-up ----------------
    time_left = search_deadline - time.monotonic()
    small_budget = min(160.0, max(10.0, time_left * 0.05))
    small_ns = [64, 128, 256, 512, 1024]
    stage_time = small_budget / float(len(small_ns))
    carry_q = None

    for n in small_ns:
        if time.monotonic() > search_deadline - 5.0:
            break
        inits = _generate_inits(n, rng)
        if carry_q is not None:
            inits.append(_upsample_q(carry_q, n))
        if not inits:
            continue

        sizeC = _next_pow2(2 * n - 1)
        scored = _score_inits(inits, n, sizeC)
        if not scored:
            continue

        stage_best_score = scored[0][0]
        stage_best_q = scored[0][1].copy()
        _pool_add(pool, stage_best_score, scored[0][2], n)

        top = min(3, len(scored))
        for i in range(top):
            tlim = max(0.5, stage_time / float(top))
            if time.monotonic() > search_deadline - 5.0:
                break
            sc, bq, bp = _run_optimizer(
                scored[i][1], n, tlim, rng,
                base_lr=0.18, tau_target=10.0, weight_smooth=0.025
            )
            _pool_add(pool, sc, bp, n)
            if np.isfinite(sc) and sc < stage_best_score:
                stage_best_score = float(sc)
                stage_best_q = bq.copy()

        if np.isfinite(stage_best_score):
            carry_q = stage_best_q.copy()
        print("multi-res n=", n, "stage_score=", stage_best_score,
              "elapsed=", time.monotonic() - total_start, flush=True)

    # ---------------- Main optimization at 8192 ----------------
    final_n = 8192
    time_left = search_deadline - time.monotonic()
    optional_budget = 0.0
    if time_left > 1500.0:
        optional_budget = min(700.0, time_left * 0.28)

    primary_time = time_left - optional_budget
    if primary_time > 40.0:
        inits = _generate_inits(final_n, rng)
        if carry_q is not None:
            inits.append(_upsample_q(carry_q, final_n))
        if parent is not None:
            ppar = np.asarray(parent, dtype=np.float64).reshape(-1)
            if ppar.size > 0:
                qpar = _p_to_q(ppar)
                inits.append(_upsample_q(qpar, final_n))

        sizeC = _next_pow2(2 * final_n - 1)
        scored = _score_inits(inits, final_n, sizeC)
        if not scored:
            p = np.ones(final_n, dtype=np.float64) / float(final_n)
            scored = [[float(_score_p(p, final_n, sizeC)), _p_to_q(p), p]]

        print("final init score=", scored[0][0], "n_inits=", len(scored),
              "time=", time.monotonic() - total_start, flush=True)

        K = 6
        K = min(K, len(scored))
        trial_time = min(140.0, max(5.0, primary_time / max(1.0, K * 3.5)))

        short_best_score = scored[0][0]
        short_best_q = scored[0][1].copy()
        short_best_p = scored[0][2].copy()

        for i in range(K):
            if time.monotonic() > search_deadline - optional_budget - 8.0:
                break
            tlim = min(trial_time, max(1.0, search_deadline - time.monotonic() - optional_budget - 8.0))
            sc, bq, bp = _run_optimizer(
                scored[i][1], final_n, tlim, rng,
                base_lr=0.15, tau_target=12.0, weight_smooth=0.015
            )
            _pool_add(pool, sc, bp, final_n)
            if np.isfinite(sc) and sc < short_best_score:
                short_best_score = float(sc)
                short_best_q = bq.copy()
                short_best_p = bp.copy()
            print("final trial", i, "score=", sc, "time=", time.monotonic() - total_start,
                  flush=True)

        long_time = search_deadline - optional_budget - 8.0 - time.monotonic()
        if long_time > 20.0:
            sc, bq, bp = _run_optimizer(
                short_best_q, final_n, long_time, rng,
                base_lr=0.18, tau_target=14.0, weight_smooth=0.012
            )
            _pool_add(pool, sc, bp, final_n)
            print("final long score=", sc, "time=", time.monotonic() - total_start, flush=True)
            short_best_q = bq
            short_best_p = bp
    else:
        short_best_q = None
        short_best_p = None

    # ---------------- Optional refinement at 16384 ----------------
    time_left = search_deadline - time.monotonic()
    if time_left > 650.0:
        # Get best final-n q/p from pool.
        best_final_p = None
        best_final_sc = float("inf")
        for item in pool:
            if item[1] == final_n and item[0] < best_final_sc:
                best_final_sc = float(item[0])
                best_final_p = item[2]

        if best_final_p is None and short_best_p is not None:
            best_final_p = short_best_p
            best_final_sc = float(_score_p(best_final_p, final_n))
        if best_final_p is None:
            best_final_p = np.ones(final_n, dtype=np.float64) / float(final_n)
            best_final_q = _p_to_q(best_final_p)
        else:
            best_final_q = _p_to_q(best_final_p)

        n2 = 16384
        inits2 = [_upsample_q(best_final_q, n2)]
        # A modest fresh pool; avoid too many expensive FFTs.
        fresh = _generate_inits(n2, rng)
        inits2.extend(fresh[:24])

        sizeC2 = _next_pow2(2 * n2 - 1)
        scored2 = _score_inits(inits2, n2, sizeC2)
        if scored2:
            print("optional 16384 init best=", scored2[0][0], "time=",
                  time.monotonic() - total_start, flush=True)
            K2 = min(3, len(scored2))
            trial2 = min(110.0, max(5.0, (time_left - 30.0) / max(1.0, K2 * 3.5)))
            best2_score = scored2[0][0]
            best2_q = scored2[0][1].copy()
            best2_p = scored2[0][2].copy()
            _pool_add(pool, best2_score, best2_p, n2)

            for i in range(K2):
                if time.monotonic() > search_deadline - 10.0:
                    break
                tlim = min(trial2, max(1.0, search_deadline - 10.0 - time.monotonic()))
                sc, bq, bp = _run_optimizer(
                    scored2[i][1], n2, tlim, rng,
                    base_lr=0.16, tau_target=14.0, weight_smooth=0.012
                )
                _pool_add(pool, sc, bp, n2)
                if np.isfinite(sc) and sc < best2_score:
                    best2_score = float(sc)
                    best2_q = bq.copy()
                    best2_p = bp.copy()

            long2 = search_deadline - 12.0 - time.monotonic()
            if long2 > 25.0:
                sc, bq, bp = _run_optimizer(
                    best2_q, n2, long2, rng,
                    base_lr=0.18, tau_target=16.0, weight_smooth=0.010
                )
                _pool_add(pool, sc, bp, n2)
                print("optional 16384 long score=", sc, "time=", time.monotonic() - total_start,
                      flush=True)

    # ---------------- Local block polish ----------------
    time_left = search_deadline - time.monotonic()
    if time_left > 35.0 and pool:
        pool.sort(key=itemgetter(0))
        best_score = pool[0][0]
        best_n = pool[0][1]
        best_p = pool[0][2]
        # Do not spend too much on very large n local search.
        if best_n > 12000:
            lt = min(55.0, time_left * 0.40)
        else:
            lt = min(110.0, time_left * 0.45)
        if lt > 10.0:
            sc, bp = _block_local_search(best_p, best_n, lt, rng)
            _pool_add(pool, sc, bp, best_n)
            print("local polish score=", sc, "time=", time.monotonic() - total_start, flush=True)

    # ---------------- Exact scoring of top candidates ----------------
    if not pool:
        p = np.ones(128, dtype=np.float64) / 128.0
        return [float(x) for x in p]

    pool.sort(key=itemgetter(0))
    fast_best_seq = [float(x) for x in np.asarray(pool[0][2], dtype=np.float64).tolist()]

    exact_best = _exact_evaluate_pool(pool, total_start, B)
    if exact_best is not None:
        return exact_best
    return fast_best_seq
