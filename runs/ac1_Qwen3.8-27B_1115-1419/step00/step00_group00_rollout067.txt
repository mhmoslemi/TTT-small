def compute_gradient_fft(p, a):
    """Compute corr[j] = sum_m p[j+m] * a[m] for j=0..n-1 using FFT."""
    n = len(a)
    # p has length 2n-1
    # corr[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is the cross-correlation of p and a at lag j
    # equivalently: corr = ifft(fft(p, N) * conj(fft(a, N)))[:n]
    N = 1
    while N < len(p) + n - 1:
        N *= 2
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fp * np.conj(fa), N)
    return corr[:n]

"""
FFT-based gradient descent with log-sum-exp smoothing to minimize 2*n*max(conv(a))/sum(a)^2.
Uses multiple initializations, adaptive tau, and simulated-annealing restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute linear self-convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_grad(p, a):
    """Compute grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] using FFT.
    
    p has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    N = 1
    while N < (2 * n - 1) + n - 1:
        N *= 2
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fp * np.conj(fa), N)
    return 2.0 * corr[:n]


def smooth_max_and_grad(b, a, tau):
    """Compute smooth max value and its gradient w.r.t. a."""
    n = len(a)
    b_max = np.max(b)
    shifted = tau * b - tau * b_max
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    M_tau = (tau * b_max + np.log(np.sum(exp_vals))) / tau
    grad = compute_grad(p, a)
    return M_tau, grad


def optimize_from(a_init, budget, rng, tau_start=0.5, verbose_label=""):

    n = len(a_init)

        return a
    a = a / S

    lr = 0.05

    patience = 0
    step_count = 0

            patience += 1

        # Restart logic

            a = best_a.copy()
            a += rng.normal(0, 0.03, n)
            a = np.maximum(a, 0.0)
            S = np.sum(a)
