from scipy.optimize import linprog

def build_lp_matrix(f, n):
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        A[k, jlo:jhi+1] = f[k-jhi:k-jlo+1][::-1]
    return A

    A = build_lp_matrix(f, n)
    b = np.full(2*n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    return None

"""
Multi-phase optimizer for step function sequences minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding, spectral shaping, gradient descent, and local search.
"""
import numpy as np
import time
from scipy.optimize import linprog


def fast_M(a):
    """Return max of linear autocorrelation for a (assumes sum normalized)."""
    c = np.convolve(a, a)
    return float(np.max(c))


def build_A(f, n):
    """Build constraint matrix for LP: row k gives coefficients of (f*g)[k] in g."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        A[k, jlo:jhi + 1] = f[k - jhi:k - jlo + 1][::-1]
    return A


def lp_solve(f, M, n):
    """Solve: max sum(g) s.t. (f*g)[k]<=M for all k, g>=0. Returns g or None."""
    A = build_A(f, n)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_phase(f, n, deadline):
    """LP-based iterative optimization. f has sum=1. Returns (best_f, best_M)."""
    f = f.copy()
    M = fast_M(f)
    best_f = f.copy()
    best_M = M
    t = 0.15
    stuck = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        g = lp_solve(f, M, n)
        if g is None:
            break
        gs = float(np.sum(g))
        if gs < 1e-12:
            break
        g = g / gs
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, None)
        s = float(np.sum(f_new))
        if s < 1e-8:
            t *= 0.5
            continue
        f_new = f_new / s
        M_new = fast_M(f_new)
        if M_new < best_M - 1e-14:
            best_f = f_new.copy()
            best_M = M_new
            f = f_new
            M = M_new
            stuck = 0
            t = min(t * 1.02, 0.6)
        else:
            stuck += 1
            if stuck >= 10:
                t *= 0.5
                if t < 0.005:
                    break
                stuck = 0
    return best_f, best_M


def spectrum_phase(f, n, deadline):
    """Spectral shaping: iteratively flatten the power spectrum."""
    f = f.copy()
    N = 1
    while N < 2 * n - 1:
        N *= 2
    M = fast_M(f)
    best_f = f.copy()
    best_M = M
    alpha = 0.4
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        F = np.fft.rfft(f, N)
        power = np.abs(F) ** 2
        pmean = np.mean(power)
        # Shaping: blend toward flat
        p_new = (1.0 - alpha) * power + alpha * pmean
        # Ensure non-negative
        p_new = np.clip(p_new, 0.0, None)
        new_F = np.sqrt(p_new) * np.exp(1j * np.angle(F))
        f_new = np.fft.irfft(new_F, N)[:n]
        f_new = np.clip(f_new, 0.0, None)
        s = float(np.sum(f_new))
        if s < 1e-8:
            continue
        f_new = f_new / s
        M_new = fast_M(f_new)
        if M_new < best_M - 1e-14:
            best_f = f_new.copy()
            best_M = M_new
            f = f_new
            M = M_new
        else:
            # Nudge alpha
            if it % 500 == 0:
                alpha = np.clip(alpha + 0.05, 0.1, 0.9)
    return best_f, best_M


def grad_phase(f, n, deadline):
    """Projected gradient descent on max autocorrelation."""
    f = f.copy()
    M = fast_M(f)
    best_f = f.copy()
    best_M = M
    lr = 0.05
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        c = np.convolve(f, f)
        kstar = int(np.argmax(c))
        M = float(c[kstar])
        S = 1.0
        # Subgradient: d[i] = (M/S)*f[i] - f[kstar-i]
        d = M * f
        ilow = max(0, kstar - n + 1)
        ihigh = min(n - 1, kstar)
        d[ilow:ihigh + 1] -= f[kstar - ihigh:kstar - ilow + 1][::-1]
        f_new = f - lr * d
        f_new = np.clip(f_new, 0.0, None)
        s = float(np.sum(f_new))
        if s < 1e-8:
            lr *= 0.5
            continue
        f_new = f_new / s
        M_new = fast_M(f_new)
        if M_new < best_M - 1e-14:
            best_f = f_new.copy()
            best_M = M_new
            f = f_new
            lr = min(lr * 1.05, 0.5)
        else:
            if M_new > M + 1e-10:
                lr *= 0.7
            f = f_new if M_new < M else f
    return best_f, best_M


def local_search(f, n, deadline, rng):
    """Block-jitter local search."""
    f = f.copy()
    M = fast_M(f)
    best_f = f.copy()
    best_M = M
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        f_new = f.copy()
        width = int(rng.integers(2, max(3, n // 30)))
        start = int(rng.integers(0, n - width))
        scale = np.exp(rng.normal(0.0, 0.25))
        f_new[start:start + width] *= scale
        f_new = np.clip(f_new, 0.0, None)
        s = float(np.sum(f_new))
        if s < 1e-8:
            continue
        f_new = f_new / s
        M_new = fast_M(f_new)
        if M_new < best_M - 1e-14:
            best_f = f_new.copy()
            best_M = M_new
            f = f_new
            M = M_new
    return best_f, best_M


def make_chirp(n, rng):
    """Quadratic chirp, non-negative, sum=1."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i * i / (2.0 * n)))
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def make_bell(n, rng):
    """Gaussian bell (inverted), non-negative, sum=1."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    a = 1.0 - np.exp(-4.0 * ((i - center) / (n / 3.0)) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def make_random_n(n, rng):
    """Random positive sequence, sum=1."""
    a = rng.random(n) + 0.01
    return a / np.sum(a)


def make_from_global(n):

def make_two_hump(n, rng):
    """Two-hump pattern (anti-palindromic-ish)."""
    i = np.arange(n, dtype=np.float64)
    sigma = n / 6.0
    a = np.exp(-((i - n * 0.25) / sigma) ** 2) + np.exp(-((i - n * 0.75) / sigma) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)


def make_alternating(n, rng):
    """Alternating high-low pattern."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.4 * np.cos(2.0 * np.pi * i / 4.0)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.001, None)
    return a / np.sum(a)
