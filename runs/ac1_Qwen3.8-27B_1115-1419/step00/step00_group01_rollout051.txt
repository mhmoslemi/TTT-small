```python
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
```

```python
def flat_spectrum_init(n, rng, iters=1000):
    """Generate a non-negative sequence with approximately flat spectrum."""
    L = 1
    while L < 2 * n - 1:
        L *= 2
    
    # Random phases
    phases = rng.uniform(0, 2*np.pi, L // 2 + 1)
    phases[0] = 0  # DC component must be real and positive
    
    # Target magnitude: constant
    target_mag = np.ones(L // 2 + 1) * np.sqrt(2.0 / L)  # Normalize
    
    for _ in range(iters):
        # Inverse FFT to get time domain
        A = target_mag * np.exp(1j * phases)
        # Hermitian symmetry
        A_full = np.zeros(L, dtype=complex)
        A_full[:L//2+1] = A
        A_full[L//2+1:] = np.conj(A[1:L//2][::-1])
        a = np.fft.ifft(A_full).real[:n]
        
        # Project to non-negative
        a = np.maximum(a, 0)
        
        # Forward FFT
        A_new = np.fft.rfft(np.concatenate([a, np.zeros(L - n)]))
        
        # Update phases, keep magnitude
        phases = np.angle(A_new[:L//2+1])
        phases[0] = 0  # Keep DC real
    
    a = np.maximum(a, 0)
    if np.sum(a) > 0:
        a = a / np.sum(a)
    return a
```

```python
"""
Hybrid LP + gradient optimization for minimizing autocorrelation peak-to-sum ratio.
Uses flat-spectrum (Gerchberg-Saxton) initialization, LP-based refinement,
and L-BFGS-B with smoothed objective for multiple sequence lengths.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1
    while L < size:
        L *= 2
    A = np.fft.rfft(a, n=L)
    return np.fft.irfft(A * A, n=L)[:size]


def smooth_objective_and_grad(a, tau=20.0):
    """Compute smoothed objective and its gradient."""
    a = np.maximum(a, 0)
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    # Autocorrelation
    size = 2 * n - 1
    L = 1
    while L < size:
        L *= 2
    A = np.fft.rfft(a, n=L)
    b = np.fft.irfft(A * A, n=L)[:size]
    
    # Smooth max (log-sum-exp)
    b_max = np.max(b)
    exp_b = np.exp(tau * (b - b_max))
    sum_exp = np.sum(exp_b)
    smooth_max_val = b_max + np.log(sum_exp) / tau
    
    # Weights for gradient
    w = exp_b / sum_exp
    
    # Cross-correlation (w ⊛ a)[j] = sum_k w[k] * a[k-j]
    W = np.fft.rfft(w, n=L)
    cross = np.fft.irfft(np.conj(W) * A, n=L)[:n]
    
    # Objective: 2n * smooth_max / S^2
    obj = 2.0 * n * smooth_max_val / (S * S)
    
    # Gradient
    grad = (4.0 * n / (S * S)) * (cross - smooth_max_val / S)
    
    return obj, grad


def gerchberg_saxton_init(n, rng, iters=500):
    """Generate non-negative sequence with approximately flat spectrum."""
    size = 2 * n - 1
    L = 1
    while L < size:
        L *= 2
    
    target_mag = np.ones(L // 2 + 1) * np.sqrt(2.0 / L)
    phases = rng.uniform(0, 2 * np.pi, L // 2 + 1)
    phases[0] = 0.0
    
    for it in range(iters):
        A = target_mag * np.exp(1j * phases)
        A_full = np.zeros(L, dtype=complex)
        half = L // 2
        A_full[:half + 1] = A
        A_full[half + 1:] = np.conj(A[1:half + 1][::-1]) if half > 0 else np.zeros(0, dtype=complex)
        a = np.fft.ifft(A_full).real[:n]
        a = np.maximum(a, 0)
        
        if np.sum(a) < 1e-10:
            phases = rng.uniform(0, 2 * np.pi, L // 2 + 1)
            phases[0] = 0.0
            continue
        
        A_new = np.fft.rfft(np.concatenate([a, np.zeros(L - n)]))
        phases = np.angle(A_new)
        phases[0] = 0.0
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def build_sparse_structure(n):
    """Precompute structure for LP constraint matrix."""
    n_entries = n * n
    rows = np.empty(n_entries, dtype=np.int32)
    cols = np.empty(n_entries, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        count = j_end - j_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_start, j_end + 1)
        idx += count
    return rows, cols


def lp_step(f, row_idx, col_idx, n):
    """Perform one LP step."""
    M = np.max(fast_autocorr(f))
    if M <= 0:
        return None, 0
    
    vals = f[row_idx - col_idx]
    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs',
                     options={'time_limit': 20})
    if result.status == 0:
        return np.maximum(result.x, 0), M
    return None, M


def lp_optimize(f, budget_s, verbose=True):
    """LP-based optimization."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = np.sum(f)
    
    if S < 1e-10:
        f = np.ones(n) / n
        S = 1.0
    
    f = f * (1.0 / S)  # Normalize to sum = 1
    S = 1.0
    
    best = f.copy()
    best_score = evaluate_sequence(best.tolist())
    
    row_idx, col_idx = build_sparse_structure(n)
    
    deadline = time.monotonic() + budget_s
    iter_count = 0
    stall = 0
    
    while time.monotonic() < deadline:
        g, M = lp_step(f, row_idx, col_idx, n)
        if g is None or np.sum(g) < 1e-10:
            stall += 1
            if stall > 30:
                break
            # Perturb
            f = f + 0.05 * np.random.randn(n)
            f = np.maximum(f, 0)
            S = np.sum(f)
            if S < 1e-10:
                break
            f = f / S
            S = 1.0
            continue
        
        # Rescale g to sum = 1
        g = g / np.sum(g)
        
        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            S_new = np.sum(f_new)
            if S_new < 1e-10:
                continue
            new_peak = np.max(fast_autocorr(f_new))
            if new_peak <= M * 1.001:
                f = f_new
                S = 1.0  # We'll renormalize
                f = f / S
                improved = True
                break
        
        iter_count += 1
        
        if not improved:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
        
        # Check score periodically
        if iter_count % 10 == 0:
            score = evaluate_sequence(f.tolist())
            if np.isfinite(score) and score < best_score:
                best = f.copy()
                best_score = score
                if verbose:
                    print(f"  LP iter {iter_count}: score = {best_score:.6f}")
    
    # Final score
    score = evaluate_sequence(f.tolist())
    if np.isfinite(score) and score < best_score:
        best = f.copy()
        best_score = score
    
    return best, best_score


def gradient_optimize(f, budget_s, tau=20.0, verbose=True):

    S = np.sum(f)

        S = 1.0

    best_score = evaluate_sequence(best.tolist())
    
    deadline = time.monotonic() + budget_s
```

```python
"""
Hybrid LP + gradient optimization for minimizing autocorrelation peak-to-sum ratio.
Uses flat-spectrum (Gerchberg-Saxton) initialization, LP-based refinement for small n,
and L-BFGS-B with smoothed objective for larger n. Multiple initializations are tried.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1
    while L < size:
        L *= 2
    A = np.fft.rfft(a, n=L)
    return np.fft.irfft(A * A, n=L)[:size]


def get_score(a):
    """Compute the objective value directly."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = fast_autocorr(a)
    max_b = np.max(b)
    return 2.0 * n * max_b / (S * S)


def smooth_objective_and_grad(a, tau=20.0):
    """Compute smoothed objective value and gradient."""
    a = np.maximum(a, 0)
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros(n)

    size = 2 * n - 1
    L = 1
    while L < size:
        L *= 2
    A = np.fft.rfft(a, n=L)
    b = np.fft.irfft(A * A, n=L)[:size]

    b_max = np.max(b)
    exp_b = np.exp(tau * (b - b_max))
    sum_exp = np.sum(exp_b)
    smooth_max_val = b_max + np.log(sum_exp) / tau

    w = exp_b / sum_exp

    W = np.fft.rfft(w, n=L)
    cross = np.fft.irfft(np.conj(W) * A, n=L)[:n]

    obj = 2.0 * n * smooth_max_val / (S * S)
    grad = (4.0 * n / (S * S)) * (cross - smooth_max_val / S)

    return obj, grad


def gerchberg_saxton_init(n, rng, iters=400):
    """Generate non-negative sequence with approximately flat spectrum."""
    size = 2 * n - 1
    L = 1
    while L < size:
        L *= 2

    half = L // 2
    target_mag = np.ones(half + 1) * np.sqrt(2.0 / L)
    phases = rng.uniform(0, 2 * np.pi, half + 1)
    phases[0] = 0.0

    a = np.ones(n) / n
    for it in range(iters):
        A = target_mag * np.exp(1j * phases)
        A_full = np.zeros(L, dtype=complex)
        A_full[:half + 1] = A
        if half > 0:
            A_full[half + 1:] = np.conj(A[1:half + 1][::-1])
        a = np.fft.ifft(A_full).real[:n]
        a = np.maximum(a, 0)

        s = np.sum(a)
        if s < 1e-10:
            phases = rng.uniform(0, 2 * np.pi, half + 1)
            phases[0] = 0.0
            a = np.ones(n) / n
            continue

        A_new = np.fft.rfft(np.concatenate([a, np.zeros(L - n)]))
        phases = np.angle(A_new)
        phases[0] = 0.0

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def build_sparse_structure(n):
    """Precompute row/col indices for LP constraint matrix."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        count = j_end - j_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_start, j_end + 1, dtype=np.int32)
        idx += count
    return rows, cols


def lp_step(f, row_idx, col_idx, n):
    """Solve LP to find descent direction. Returns (direction, M)."""
    M = np.max(fast_autocorr(f))
    if M <= 1e-15:
        return None, 0.0

    vals = f[row_idx - col_idx]
    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs',
                     options={'time_limit': 15, 'feasibility_tolerance': 1e-9})
    if result.status == 0:
        g = np.maximum(result.x, 0)
        if np.sum(g) > 1e-15:
            return g, M
    return None, M

        best, sc = lp_optimize(f, alloc)
        record(sc, best, "GS-256-LP")

    # --- GS init, n=1024 ---
    time_left = total_budget - (time.monotonic() - start_time)
    if time_left > 30:
        alloc = min(total_budget * 0.15, time_left - 5)
        print(f"GS init n=1024, budget={alloc:.0f}s")
        f = gerchberg_saxton_init(1024, rng, iters=300)
        sc0 = get_score(f)
        print(f"  GS-1024 initial: {sc0:.6f}")
        if alloc > 60:
            best, sc = lp_optimize(f, alloc)
            record(sc, best, "GS-1024-LP")
        else:
            best, sc = gradient_optimize(f, alloc)
            record(sc, best, "GS-1024-Grad")

    # --- Random init, n=768 ---
    time_left = total_budget - (time.monotonic() - start_time)
    if time_left > 25:
        alloc = min(total_budget * 0.10, time_left - 5)
        print(f"Random n=768, budget={alloc:.0f}s")
        f = rng.random(768) + 0.05
        f = f / np.sum(f)
        sc0 = get_score(f)
        print(f"  Rand-768 initial: {sc0:.6f}")
        best, sc = lp_optimize(f, alloc)
        record(sc, best, "Rand-768-LP")

    # --- Polish best with random local search ---
    time_left = total_budget - (time.monotonic() - start_time)
    if time_left > 10 and results:
        results.sort(key=lambda x: x[0])
        best_score, best_seq, _ = results[0]
        print(f"Polishing best (score={best_score:.6f}), budget={time_left:.0f}s")
        polished, sc_pol = random_local_search(best_seq, time_left * 0.5, verbose=True)
        record(sc_pol, polished, "Polished")

    # --- Final selection ---
    if not results:
        f = np.ones(256)
        return [float(x) for x in f]

    results.sort(key=lambda x: x[0])
    final_score, final_seq, final_label = results[0]
    print(f"\nFINAL: [{final_label}] score={final_score:.6f}, n={len(final_seq)}")

    # Verify with the actual evaluator
    final_score_verified = evaluate_sequence([float(x) for x in final_seq])
    print(f"Verified score: {final_score_verified:.6f}")

    return [float(x) for x in final_seq]
```
