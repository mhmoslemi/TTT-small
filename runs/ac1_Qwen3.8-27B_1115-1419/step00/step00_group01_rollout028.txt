            continue
        # Compute max of new convolution incrementally
        # C_new = current_conv + (x-v)*B_vec + (x^2-v^2) at position 2j
        max_val = np.max(current_conv + (x - v) * B_vec)
        if pos_2j < two_n_minus_1:
            val_2j = current_conv[pos_2j] + (x * x - v * v)
            if val_2j > max_val:
                max_val = val_2j
        f = 2.0 * n * max_val / (new_S * new_S)
        if f < best_f:
            best_f = f
            best_x = x
    # Also try x = 0 explicitly
    x = 0.0
    new_S = 1.0 - v
    if new_S > 0.001:
        max_val = np.max(current_conv - v * B_vec)
        if pos_2j < two_n_minus_1:
            val_2j = current_conv[pos_2j] - v * v
            if val_2j > max_val:
                max_val = val_2j
        f = 2.0 * n * max_val / (new_S * new_S)
        if f < best_f:
            best_f = f
            best_x = x
    return best_x, best_f


def _coordinate_descent(a, n, max_sweeps=200, time_limit=None, deadline=None, rng=None):
    """Run coordinate descent on sequence a (sum=1). Returns best sequence and objective."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    current_a = a.copy()
    current_conv = np.convolve(current_a, current_a)
    current_obj = 2.0 * n * np.max(current_conv)
    best_a = current_a.copy()
    best_obj = current_obj
    
    factors_list = [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0]
    
    sweep = 0
    stagnant = 0
    while sweep < max_sweeps:
        if time_limit is not None and time.monotonic() - time_limit > 0:
            break
        if deadline is not None and time.monotonic() > deadline:
            break
        sweep += 1
        improved = False
        for j in range(n):
            v = current_a[j]
            if v < 1e-15:
                # Try setting to a small positive value
                B_vec = _compute_bvec(j, current_a, n)
                for x_try in [0.01 / n, 0.05 / n, 0.1 / n]:
                    new_S = 1.0 + x_try
                    max_val = np.max(current_conv + x_try * B_vec)
                    pos_2j = 2 * j
                    if pos_2j < 2 * n - 1:
                        val_2j = current_conv[pos_2j] + x_try * x_try
                        if val_2j > max_val:
                            max_val = val_2j
                    f = 2.0 * n * max_val / (new_S * new_S)
                    if f < current_obj * 0.999:
                        best_x = x_try
                        break
                else:
                    continue
            else:
                B_vec = _compute_bvec(j, current_a, n)
                best_x, best_f = _eval_candidates(j, v, current_conv, B_vec, n, factors_list)
                if best_f >= current_obj * 0.9999:
                    continue
                if best_x == v:
                    continue
            
            # Apply update
            new_S = 1.0 - v + best_x
            if new_S < 0.001:
                continue
            scale = 1.0 / new_S
            current_a *= scale
            current_a[j] = best_x * scale
            current_conv = np.convolve(current_a, current_a)
            current_obj = 2.0 * n * np.max(current_conv)
            improved = True
            
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = current_a.copy()
        
        if not improved:
            stagnant += 1
            if stagnant >= 3:
                # Try perturbation to escape local minimum
                pert_scale = 0.05 + 0.05 * stagnant
                idx = rng.choice(n, size=max(1, n // 10), replace=False)
                multipliers = 1.0 + rng.normal(0, pert_scale, size=len(idx))
                trial = current_a.copy()
                trial[idx] *= np.clip(multipliers, 0.1, 10.0)
                trial_sum = np.sum(trial)
                if trial_sum > 0.01:
                    trial /= trial_sum
                    trial_conv = np.convolve(trial, trial)
                    trial_obj = 2.0 * n * np.max(trial_conv)
                    if trial_obj < current_obj * 1.3:  # accept slightly worse
                        current_a = trial
                        current_conv = trial_conv
                        current_obj = trial_obj
                stagnant = 0
            continue
        else:
            stagnant = 0
    
    return best_a, best_obj


def _make_u_shaped(n, power=2.0):
    """Create U-shaped initialization."""
    i = np.arange(n)
    a = (1.0 + np.cos(2.0 * np.pi * i / (n - 1))) ** power
    a = np.clip(a, 0, None)
    a_sum = np.sum(a)
    if a_sum < 1e-10:
        a = np.ones(n)
        a_sum = n
    return a / a_sum


def _make_bipolar(n, sigma_ratio=0.2):
    """Create two-bump initialization."""
    i = np.arange(n, dtype=np.float64)
    sigma = sigma_ratio * n
    a = np.exp(-((i - 0.25 * n) ** 2) / (2 * sigma * sigma)) + \
        np.exp(-((i - 0.75 * n) ** 2) / (2 * sigma * sigma))
    a_sum = np.sum(a)
    return a / a_sum


def _make_random_nonneg(n, rng):
    """Random non-negative initialization."""
    a = rng.random(n) + 0.01
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-start coordinate descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Uses incremental convolution updates for efficiency. Tries multiple
    initializations (given sequence, U-shaped, bipolar, random) and runs
    coordinate descent from each. Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0  # leave buffer
    
    # Get initial sequence
    try:
        init_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    except:
        init_seq = np.ones(256, dtype=np.float64)
    
    if init_seq.ndim != 1 or init_seq.size == 0 or not np.isfinite(init_seq).all():
        init_seq = np.ones(256, dtype=np.float64)
    
    # Evaluate initial
    init_score = float(evaluate_sequence(init_seq.tolist()))
    print(f"Initial sequence: n={len(init_seq)}, score={init_score:.6f}")
    
    # Build list of initializations to try
    initializations = []
    
    # 1. The given sequence
    a0 = init_seq.copy()
    a0 = a0 / np.sum(a0)
    initializations.append(("given", a0))
    
    # 2. Various U-shaped
    n_base = len(init_seq)
    for power in [1.0, 2.0, 4.0]:
        initializations.append((f"u_p{power}", _make_u_shaped(n_base, power)))
    
    # 3. Bipolar
    initializations.append(("bipolar", _make_bipolar(n_base, 0.15)))
    initializations.append(("bipolar2", _make_bipolar(n_base, 0.3)))
    
    # 4. Random
    for _ in range(3):
        initializations.append((f"rand_{_}", _make_random_nonneg(n_base, rng)))
    
    # 5. Try different lengths
    for n_try in [2000, 4000, 8000]:
        if n_try != n_base:
            initializations.append((f"u_n{n_try}", _make_u_shaped(n_try, 2.0)))
            initializations.append((f"rand_n{n_try}", _make_random_nonneg(n_try, rng)))
    
    # Run coordinate descent from each initialization
    global_best_a = None
    global_best_score = float(evaluate_sequence(init_seq.tolist()))
    global_best_n = n_base
    
    elapsed = time.monotonic() - start_time
    remaining = budget_s - 15.0 - elapsed
    num_inits = len(initializations)
    time_per_init = max(30.0, remaining / num_inits)
    
    for init_name, init_a in initializations:
        if time.monotonic() > deadline:
            break
        n_init = len(init_a)
        init_deadline = min(deadline, time.monotonic() + time_per_init)
        
        print(f"\n--- {init_name} (n={n_init}) at {time.monotonic()-start_time:.1f}s ---")
        
        best_a, best_obj = _coordinate_descent(
            init_a, n_init,
            max_sweeps=500,
            deadline=init_deadline,
            rng=rng
        )
        
        # Evaluate with exact function
        score = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(score) and score < global_best_score:
            global_best_score = score
            global_best_a = best_a.copy()
            global_best_n = n_init
            print(f"  NEW BEST: score={score:.6f} (n={n_init})")
        else:
            print(f"  score={score:.6f} (n={n_init}) [not better]")
        
        # If we have plenty of time left, do more from the best
        elapsed = time.monotonic() - start_time
        if global_best_a is not None and budget_s - elapsed > 300:
            # Re-run from global best with more time
            print(f"\n--- Extended optimization from best (n={global_best_n}) ---")
            extra_deadline = min(deadline, time.monotonic() + (budget_s - elapsed) * 0.6)
            best_a2, best_obj2 = _coordinate_descent(
                global_best_a, global_best_n,
                max_sweeps=1000,
                deadline=extra_deadline,
                rng=rng
            )
            score2 = float(evaluate_sequence(best_a2.tolist()))
            if np.isfinite(score2) and score2 < global_best_score:
                global_best_score = score2
                global_best_a = best_a2.copy()
                global_best_n = global_best_n
                print(f"  EXTENDED BEST: score={score2:.6f}")
    
    # Final: if we have time, do one more round from the global best
    if global_best_a is not None and time.monotonic() < deadline - 60:
        print(f"\n--- Final refinement (n={global_best_n}) ---")
        best_a3, _ = _coordinate_descent(
            global_best_a, global_best_n,
            max_sweeps=500,
            deadline=deadline - 5,
            rng=rng
        )
        score3 = float(evaluate_sequence(best_a3.tolist()))
        if np.isfinite(score3) and score3 < global_best_score:
            global_best_score = score3
            global_best_a = best_a3.copy()
            print(f"  FINAL BEST: score={score3:.6f}")
    
    # Return the best (scale doesn't matter due to scale invariance, but ensure valid)
    if global_best_a is None:
        global_best_a = init_seq / np.sum(init_seq)
    
    result = [float(max(0.0, x)) for x in global_best_a]
    final_score = float(evaluate_sequence(result))
    print(f"\nFinal result: n={len(result)}, score={final_score:.6f}")
    return result
