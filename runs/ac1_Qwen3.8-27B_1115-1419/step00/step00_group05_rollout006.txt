df/da_j = 2n * [ (1/T) * sum_k p_k * d(conv[k])/d(a_j) * (sum a)^(-2) + (1/T) * log(...) * (-2) * a_j / (sum a)^3 ]

= 2n * [ (1/T) * (sum_k p_k * 2*a[k-j]) / (sum a)^2 - 2*a_j * (1/T) * log(sum exp(T*conv)) / (sum a)^3 ]

where the first term uses d(conv[k])/d(a_j) = 2*a[k-j] (for valid k-j) and the sum over k gives the cross-correlation.

Let me define:
- C = conv (length 2n-1)
- p_k = exp(T*C_k) / sum_m exp(T*C_m) (softmax)
- L = (1/T) * log(sum_k exp(T*C_k)) = (1/T) * LSE(T*C)
- S = sum(a)
- xc_j = sum_k p_k * 2*a[k-j] = 2*(a ⋆ p)(j) (as computed above)

Then:
df/da_j = 2n/T * xc_j / S^2 - 2n * L * 2*a_j / S^3

= (2n/T) * (xc_j / S^2 - 2*L*a_j/S^3)

= (2n / (T*S^2)) * (xc_j - 2*L*a_j/S)

For the descent direction, we want to decrease f, so we do:
a_j <- max(0, a_j - lr * df/da_j)

But since f is scale-invariant, the 1/S^2 and 1/S^3 terms will cause issues if S changes a lot. Let me just normalize a to have sum 1 before each iteration, or better, just use the gradient of the simplified objective g(a) = max(conv) subject to sum(a)=1.

"""
Gradient-based optimization of non-negative step function coefficients to minimize
2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated convolution, log-sum-exp
smoothing for the max operator, and projected gradient descent with backtracking.
Multiple restarts with diverse initializations and sequence lengths are explored.
"""

import numpy as np
import time


def fft_convolve(a):
    """Compute linear convolution a*a via FFT. Returns real array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fc = np.fft.irfft(fa * fa, N)
    return fc[:2 * n - 1]


def softmax(x, T):
    """Stable softmax with temperature T."""
    xT = T * x
    xT -= np.max(xT)
    ex = np.exp(xT)
    s = np.sum(ex)
    if s <= 0:
        return np.full_like(x, 1.0 / len(x))
    return ex / s


def compute_gradient(a, p, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*conv[k])) w.r.t. a.
    p = softmax(T * conv)
    Returns gradient array of length n.
    """
    n = len(a)
    # gradient_j = 2 * sum_k p[k] * a[k-j] = 2 * (a cross-correlation p)[j]
    # cross-correlation: (a ⋆ p)(j) = sum_i a[i] * p[i+j]
    N = 1
    while N < n + (2 * n - 1) - 1:
        N *= 2
    fft_a = np.fft.rfft(a, N)
    fft_p = np.fft.rfft(p, N)
    xc = np.fft.irfft(np.conj(fft_a) * fft_p, N)
    # (a ⋆ p)(j) for j = 0..n-1 is xc[j]
    grad = 2.0 * xc[:n]
    return grad


def optimize_sequence(a0, deadline, seed=0, verbose=False):
    """
    Optimize non-negative sequence a to minimize 2n*max(conv)/(sum(a))^2.
    Uses projected gradient descent with log-sum-exp smoothing.
    Returns (best_a, best_score).
    """
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.clip(a, 1e-10, 1000.0)
    
    # Normalize to sum = 1
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
    
    # Initial evaluation
    conv = fft_convolve(a)
    max_conv = np.max(conv)
    score = 2.0 * n * max_conv / (1.0 ** 2)  # sum=1 so (sum)^2 = 1
    best_a = a.copy()
    best_score = score
    
    # Adaptive T for log-sum-exp
    T = 20.0
    lr = 0.1 / (2 * n * max_conv)
    momentum = np.zeros_like(a)
    mu = 0.9
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        # Compute conv and gradient
        conv = fft_convolve(a)
        # Ensure conv is non-negative (numerical)
        conv = np.maximum(conv, 0.0)
        
        # Update T adaptively
        max_c = np.max(conv)
        mean_c = np.mean(conv)
        spread = max(1e-10, max_c - mean_c)
        T_target = max(5.0, min(200.0, 20.0 * max_c / spread))
        T = 0.95 * T + 0.05 * T_target
        
        p = softmax(conv, T)
        LSE_val = (1.0 / T) * np.log(np.sum(np.exp(T * conv - np.max(T * conv))) + 1e-300) + np.max(conv)
        
        # Gradient of LSE w.r.t. a
        xc = compute_gradient(a, p, T)
        
        # Full gradient of 2n*LSE/(sum a)^2 (but we fix sum=1 via projection)
        # Since we normalize, just use gradient of max(conv) approximation:
        grad = 2.0 * n * xc  # gradient of 2n * LSE (with sum fixed = 1)
        
        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad
        a_new = a - lr * momentum
        
        # Project onto non-negative
        a_new = np.maximum(a_new, 0.0)
        
        # Renormalize to sum = 1
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / S_new
        
        # Evaluate
        conv_new = fft_convolve(a_new)
        conv_new = np.maximum(conv_new, 0.0)
        max_new = np.max(conv_new)
        score_new = 2.0 * n * max_new
        
        if score_new < score:
            # Accept
            a = a_new
            score = score_new
            if score < best_score:
                best_score = score
                best_a = a.copy()
            # Small increase in T for sharper max approximation
            lr *= 0.999
        else:
            # Reject, reduce step size
            lr *= 0.5
            a = a + 0.0  # keep old
    
    # Final evaluation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    return best_a, final_list, best_score


def make_gaussian_init(n, center_frac=0.3, sigma_frac=0.1):
    """Create a skewed Gaussian initialization."""
    x = np.arange(n)
    center = center_frac * n
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # Add a small secondary bump on the other side
    a += 0.1 * np.exp(-0.5 * ((x - (1 - center_frac) * n) / (sigma * 0.5)) ** 2)
    return a


def make_bimodal_init(n, gap_frac=0.2):
    """Two separated bumps, asymmetric."""
    x = np.arange(n)
    c1 = 0.2 * n
    c2 = 0.7 * n
    s1 = 0.12 * n
    s2 = 0.08 * n
    a = np.exp(-0.5 * ((x - c1) / s1) ** 2)
    a += 0.5 * np.exp(-0.5 * ((x - c2) / s2) ** 2)
    return a


def make_power_init(n, beta):
    """Power-law decaying sequence (asymmetric)."""
    x = np.arange(n)
    a = (1.0 - x / n) ** beta + 1e-6
    return a


def make_noise_init(n, rng):
    """Random non-negative initialization with some structure."""
    a = np.abs(rng.standard_normal(n))
    # Smooth it
    kernel = np.ones(10) / 10
    a = np.convolve(a, kernel, 'same')
    a = np.maximum(a, 0.0)
    return a


def local_search_refine(a, deadline, seed=0):
    """
    Coordinate descent + random perturbation local search on the exact objective.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S  # normalize to sum=1
    
    conv = fft_convolve(a)
    max_c = np.max(conv)
    best_score = 2.0 * n * max_c
    best_a = a.copy()
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Try random coordinate updates
        n_updates = max(1, n // 50)
        idx = rng.integers(0, n, size=n_updates)
        for j in idx:
            for scale in [0.5, 0.7, 0.9, 1.3, 1.5, 2.0]:
                a_copy = a.copy()
                a_copy[j] *= scale
                S_c = np.sum(a_copy)
                if S_c < 1e-10:
                    continue
                a_copy = a_copy / S_c
                conv_c = fft_convolve(a_copy)
                max_c_new = np.max(conv_c)
                score_c = 2.0 * n * max_c_new
                if score_c < best_score - 1e-10:
                    a = a_copy
                    best_score = score_c
                    best_a = a.copy()
                    break
        
        # Periodic block perturbation
        if it % 10 == 0:
            width = max(1, n // 50)
            start = rng.integers(0, max(1, n - width))
            a_copy = a.copy()
            a_copy[start:start+width] *= np.exp(rng.normal(0, 0.3, width))
            a_copy = np.maximum(a_copy, 0.0)
            S_c = np.sum(a_copy)
            if S_c > 1e-10:
                a_copy = a_copy / S_c
                conv_c = fft_convolve(a_copy)
                score_c = 2.0 * n * np.max(conv_c)
                if score_c < best_score - 1e-10:
                    a = a_copy
                    best_score = score_c
                    best_a = a.copy()
    
    return best_a
