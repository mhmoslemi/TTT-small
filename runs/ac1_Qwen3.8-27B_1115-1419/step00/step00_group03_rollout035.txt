```python
import time
from scipy.optimize import linprog

def solve_lp(b, M, active_ks=None):
    """Solve the LP to find the best direction g."""
    n = len(b)
    if active_ks is None:
        # Use all constraints (expensive)
        active_ks = list(range(2*n - 1))
    
    # Build constraint matrix A: A[k, j] = b[k-j] if 0 <= k-j < n, else 0
    # For each constraint k, the row is: [b[k], b[k-1], ..., b[k-n+1]] (padded with 0s)
    
    # Actually, let's build it more efficiently
    # For constraint k: sum_i b_i * g_{k-i} <= M
    # This means: for g_j, the coefficient is b_{k-j} (if 0 <= k-j < n)
    
    m = len(active_ks)
    A = np.zeros((m, n))
    for idx, k in enumerate(active_ks):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[idx, j] = b[k-j]
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # Convert to min -sum(g) s.t. A @ g <= M, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)]*n, method='highs')
    return res.x if res.success else None
```

Algorithm: Multi-phase optimization for non-negative sequences with minimal
peak autocorrelation. Uses multiplicative weight updates (MWU) for fast
convergence, LP refinement for precision, and multiple initializations to
escape local minima.

def compute_score_normalized(b):
    """Compute J = 2n * max(conv(b,b)) for b with sum=1."""
    n = len(b)
    conv = np.convolve(b, b)
    return 2.0 * n * np.max(conv)


def mwu_step(b, conv, eta=0.3, frac=0.9, max_active=8):
    """One MWU step: reduce elements contributing to top autocorrelation peaks."""
    n = len(b)
    M = np.max(conv)
    threshold = M * frac
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        return b.copy()
    
    if len(active) > max_active:
        top_idx = active[np.argsort(conv[active])[-max_active:]]
    else:
        top_idx = active
    
    penalty = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        penalty[lo:hi] += b[k - hi + 1:k - lo + 1][::-1]
    
    pmax = np.max(penalty)
    if pmax < 1e-15:
        return b.copy()
    
    penalty /= pmax
    b_new = b * np.exp(-eta * penalty)
    b_new = np.clip(b_new, 1e-15, None)
    b_new /= b_new.sum()
    return b_new


def lp_direction(b, M, n_constraints=50):

    from scipy.optimize import linprog
    n = len(b)
    conv_len = 2 * n - 1
    
    # Build constraint matrix (only top n_constraints to keep it fast)
    conv = np.convolve(b, b)
    # Use the top n_constraints positions
    k_selected = np.argsort(conv)[::-1][:n_constraints]
    k_selected = np.sort(k_selected)
    
    m = len(k_selected)
    A = np.zeros((m, n))
    for idx, k in enumerate(k_selected):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[idx, lo:hi] = b[k - hi + 1:k - lo + 1][::-1]

    c = -np.ones(n)

        s = g.sum()
        if s > 0:

    return b.copy()  # Fallback: no change
