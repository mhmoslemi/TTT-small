def solve_lp(a, M):
    """Solve: max sum(g) s.t. a*g <= M (element-wise), g >= 0."""
    n = len(a)
    # Build constraint matrix A where (A @ g)_k = sum_j a_j * g_{k-j}
    # A has shape (2n-1, n)
    # A[k, i] = a_{k-i} if 0 <= k-i < n, else 0
    
    # For efficiency, build row by row
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # c = -1 (minimize -sum(g))
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def full_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:size]

def smooth_max_objective(x, beta, topK):
    """Compute smooth approximation of max convolution and its gradient."""
    a = np.maximum(x, 0)  # ensure non-negativity
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros_like(x)
    
    c = full_convolve(a)
    
    # Get top-K values
    # For efficiency, use argpartition
    k = min(topK, len(c))
    top_indices = np.argpartition(-c, k)[:k]
    top_values = c[top_indices]
    
    # Smooth max: (1/k) * sum(top_values^beta)^(1/beta)
    # Or simpler: sum(top_values^beta) / k^beta
    # Let's use: (sum(exp(beta * (c_k - c_max))) / K)^(1/beta) + c_max
    
    c_max = np.max(c)
    # Numerically stable softmax
    shifted = beta * (c[top_indices] - c_max)
    weights = np.exp(shifted)
    weights_sum = np.sum(weights)
    
    # Value: c_max + (1/beta) * log(weights_sum / K)
    obj_val = c_max + (1.0 / beta) * np.log(weights_sum / k)
    
    # Objective: 2n * obj_val / S^2
    val = 2.0 * n * obj_val / (S * S)
    
    # Gradient
    # d(val)/d(a_i) = 2n/S^2 * d(obj_val)/d(a_i) - 4n * obj_val / S^3
    # d(obj_val)/d(a_i) = sum_j w_j/K * d(c_kj)/d(a_i)
    # where w_j = exp(beta*(c_kj - c_max)) / weights_sum
    # d(c_k)/d(a_i) = 2 * a_{k-i} (if valid)
    
    w = weights / weights_sum  # shape (K,)
    # For each of the top indices, add w_j * 2 * a_{k_j - i} to gradient[i]
    grad_obj = np.zeros(n)
    for j in range(k):
        kj = top_indices[j]
        # d(c_kj)/d(a_i) = 2 * a_{kj - i} if 0 <= kj - i < n
        i_start = max(0, kj - n + 1)
        i_end = min(n - 1, kj)
        # a[kj - i] for i in [i_start, i_end] is a[kj - i_end : kj - i_start + 1] reversed
        # i.e., a[kj-i_end], a[kj-i_end+1], ..., a[kj-i_start]
        # which is a[kj-i_end : kj-i_start+1]
        # And i goes from i_start to i_end
        # So grad_obj[i_start:i_end+1] += w[j] * 2 * a[kj-i_end : kj-i_start+1]
        segment = a[kj - i_end : kj - i_start + 1]
        grad_obj[i_start:i_end+1] += w[j] * 2.0 * segment
    
    grad_val = (2.0 * n / (S * S)) * grad_obj - (4.0 * n * obj_val / (S * S * S))
    
    # Clip gradient to zero where x < 0 (for the projection)
    grad_val = np.where(x >= 0, grad_val, 0)
    
    return val, grad_val
