def solve_lp(f, M):
    n = len(f)
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_j f_j * g_{k-j} (valid indices)
    # This is a linear constraint on g
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[j]
    
    b = np.full(2*n-1, M)
    
    # g >= 0 handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 0:
        return np.ones(n) / n
    # Simple projection: clip and normalize
    # This is not the exact Euclidean projection but works as a feasibility step
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return v / s

def exact_project_simplex(v):
    """Exact Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    pos = np.arange(1, n + 1)
    cond = u * pos > cssv
    rho = np.max(pos[cond])
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0)

    # Use log-sum-exp trick for numerical stability
    c_shifted = T * (c - np.max(c))
    p = np.exp(c_shifted)
    p /= p.sum()
    return p

    """Compute softmax probabilities for the smooth max."""
    c_shifted = T * (c - np.max(c))
    exp_c = np.exp(c_shifted)
    p = exp_c / exp_c.sum()
    return p

def exact_project_simplex(v):
    """Exact Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    pos = np.arange(1, n + 1)
    cond = u * pos > cssv
    if not np.any(cond):
        return np.ones(n) / n
    rho = pos[np.max(np.where(cond)[0])]
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, p):
    """Compute 2 * cross_correlation(a, p) at lags 0..n-1."""
    n = len(a)
    cross = np.correlate(a, p, mode='full')
    # R[i] = cross at lag -i = index (2n-2) - i in the full output
    # Full output indices 0 to 3n-3
    # R[i] is at index (2n-2)-i for i=0..n-1
    # So R = cross[n-1:2n-1][::-1]
    return 2.0 * cross[n-1 : 2*n-1][::-1]

def solve_reduced_lp(f, M, active_lags):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for k in active_lags."""
    n = len(f)
    c_vec = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix for active lags only
    A = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[idx, j_start:j_end+1] = f[j_start:j_end+1]
    
    b = np.full(len(active_lags), M)
    bounds = [(0, None)] * n
    
    result = linprog(c_vec, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None
