def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    # Clip small negative values from FFT round-off
    conv = np.maximum(conv, 0)
    return conv
