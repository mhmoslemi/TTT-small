    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)

def compute_convolution_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use FFT size that's efficient
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]

def compute_subgradient_for_k(a, k):
    """Gradient of (a*a)_k w.r.t. a. Returns array of length n."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return g

    c = compute_convolution_fft(a)

    k_star = np.argmax(c)
