def compute_conv_fft(g):
    """Compute full linear convolution using FFT (faster for large n)."""
    n = len(g)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    G = np.fft.rfft(g, fft_size)
    conv = np.fft.irfft(G * G, fft_size)[:size]
    return conv
