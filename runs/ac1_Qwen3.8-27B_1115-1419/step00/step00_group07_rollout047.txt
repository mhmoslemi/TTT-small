    2. Identifies peak positions
    3. Computes the subgradient of max(c) with respect to f
    4. Updates f via multiplicative weights: f *= exp(-eta * subgrad)
    5. Normalizes f
    """

    eta = 0.1  # initial step size
    iter_count = 0
    stalled = 0

        c = np.convolve(f, f)
        max_c = c.max()

        subgrad = np.zeros(n)
        for k in S:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)

            f_new = np.ones(n) / n
        else:
            f_new = f_new / s
        
        f = f_new
        iter_count += 1

        if iter_count % 50 == 0:

            if score < best_score:

                eta = max(0.01, eta * 2)
                if eta > 1.0:

                    f = np.clip(f, 0, None)

                    eta = 0.1
