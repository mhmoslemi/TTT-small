"""
Skew-based low-peak-Autocorrelation search.

Key idea (from analysis): the score 2*n*max(conv(a,a))/(sum(a))^2 is scale
invariant, so we fix sum(a)=1 and minimise max(conv(a,a)).  A symmetric
(palindromic) sequence such as the uniform one attains exactly 2, but a
NON-symmetric (skewed) sequence shifts the convolution peak off the centre and
can push the score well below 2 (asymptotically ~4/3 for a linear ramp).

Algorithm: start from a skewed initial profile, then run projected gradient
descent on a log-sum-exp smoothing of the convolution with the simplex
projection {a>=0, sum(a)=1}.  The temperature is annealed so the search first
moves globally and later sharpens around the few dominant lags.  An
authoritative evaluate_sequence() call is used to track the true best.
Several initial sizes/profiles are tried to avoid a single local minimum.
"""

import time
import numpy as np


# ---------------------------------------------------------------------------
# helpers
# ---------------------------------------------------------------------------
def _project_simplex(v, S):
    """Euclidean projection of vector v onto {x>=0, sum(x)=S}."""
    n = v.size
    if n == 0:
        return v
    x = np.sort(v)[::-1]
    csum = np.cumsum(x)
    rho = np.nonzero(x * np.arange(1, n + 1) > (csum - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n, dtype=np.float64)
    rho = rho[-1]
    theta = (csum[rho] - S) / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    return w


def _init_linear_skew(n, asym):
    """Monotonically decreasing profile (the known good 'skew' family)."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (1.0 - t) ** asym
    a += 1e-6
    return a / a.sum()


def _init_ramp_plus_bump(n):
    """Linear ramp plus a mild interior bulge to give the search room to move."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (1.0 - t)
    a += 0.25 * np.exp(-((t - 0.45) ** 2) / (2.0 * 0.10 ** 2))
    a += 1e-6
    return a / a.sum()


def _conv(a):
    n = a.size
    if n < 512:
        return np.convolve(a, a, mode="full")
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[: 2 * n - 1]


def _smoothed_obj_and_weights(a, T):
    c = _conv(a)
    m = c.max()
    ex = np.exp(T * (c - m))
    w = ex / ex.sum()
    val = (np.log(ex.sum()) + T * m) / T
    return val, w


def _grad_of_conv_w(a, w):
    """Gradient of (w . conv(a,a)) w.r.t. a (loosely, the smoothed max grad).

    grad_k[j] = 2*a[k-j] for max(0,k-n+1)<=j<=min(k,n-1).
    Accumulates sum_k w[k]*grad_k.
    """
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk <= 1e-12:
            continue
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if hi < lo:
            continue
        # for j in lo..hi : 2*a[k-j]
        g[lo:hi + 1] += wk * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _refine(a0, t_end, T0, T1, lr0, eval_every):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = np.ones(a.size)
        s = a.size
    a = a / s
    n = a.size

    best_score = float("inf")
    best_a = a.copy()

    it = 0
    stalled = 0
    last_score = best_score
    while time.monotonic() < t_end:
        # linearly anneal temperature
        frac = min(1.0, it / max(1, int(eval_every * 400)))
        T = T0 + (T1 - T0) * frac

        _, w = _smoothed_obj_and_weights(a, T)
        g = _grad_of_conv_w(a, w)
        a = _project_simplex(a - lr0 * g, 1.0)
        it += 1

        if it % eval_every == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled > 0 and stalled % 40 == 0:
                # gentle restart nudge using a fresh skew blended in
                a = _project_simplex(0.7 * best_a + 0.3 * _init_linear_skew(n, 1.0), 1.0)
                lr0 *= 0.85
    return best_a, best_score, it
