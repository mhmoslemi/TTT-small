    S = a.sum()

We minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences a.
By scale invariance, we fix sum(a) = 1 and minimize 2*n*max(conv(a,a)).

The main method is an LP-based iteration:
1. Compute M = max(conv(a,a))
2. Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
3. Rescale g to sum = 1
4. Update: a <- (1-t)*a + t*g with line search on t

We also use gradient descent as a complementary method.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    """Compute the objective value (scaled by sum^2 for efficiency)."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    M = b.max()
    return 2.0 * n * M / (S * S)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a, M):
    """Solve: max 1^T g s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_step(f, step_sizes=(1.0, 0.7, 0.5, 0.3, 0.1)):
    """One LP-based step with line search on step size."""
    n = len(f)
    b = np.convolve(f, f)
    M = b.max()
    S = f.sum()
    
    g = solve_lp(f, M)
    if g is None or g.sum() <= 0:
        return f, compute_obj(f)
    
    g = g * (S / g.sum())
    
    best_f = f
    best_score = compute_obj(f)
    
    for t in step_sizes:
        candidate = (1 - t) * f + t * g
        candidate = np.maximum(candidate, 0)
        score = compute_obj(candidate)
        if score < best_score:
            best_score = score
            best_f = candidate
    
    return best_f, best_score

def gradient_step(f, lr, active_frac=0.1):
    """One gradient descent step."""
    n = len(f)
    b = np.convolve(f, f)
    M = b.max()
    S = f.sum()
    
    # Active set
    threshold = M - active_frac * (M - b.min())
    active = np.where(b >= threshold)[0]
    
    # Subgradient of max(conv(a,a)) w.r.t. a
    grad = np.zeros(n)
    for tau in active:
        lo = max(0, tau - n + 1)
        hi = min(n, tau + 1)
        ks = np.arange(lo, hi)
        js = tau - ks
        grad[ks] += 2.0 * a[js]  # Wait, 'a' should be 'f'
    grad /= len(active)
    
    # Gradient of objective (with sum fixed, we project out the sum constraint)
    # Actually, let's just use the unconstrained gradient and project
    obj_grad = (S * grad - 2.0 * M) / S**2
    
    f_new = f - lr * obj_grad
    f_new = np.maximum(f_new, 0)
    
    return f_new
