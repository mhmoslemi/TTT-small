def project_simplex(v):
   n=v.size
   target=1.0
   u=np.sort(v)[::-1]
   cssv=np.cumsum(u)-target
   ind=np.arange(1,n+1,dtype=np.float64)
   cond = u - cssv/ind > 0.0
   if not np.any(cond):
      return np.full(n, target/n, dtype=np.float64)
   idx = np.nonzero(cond)[0]
   rho = cssv[int(idx[-1])]
   x = np.maximum(v-rho,0.0)
   s = float(x.sum())
   if not np.isfinite(s) or s <= 1e-15:
       return np.full(n,target/n)
   x *= target/s
   return x

```python
"""
Multi-resolution convex heuristic for minimizing the normalized peak
self-convolution of a non-negative sequence.

The objective is scale-invariant, so the search keeps sequences normalized
to sum 1.  Fast FFT surrogates are used for the whole search; only the
final returned array is expected to be scored by the authoritative
evaluate_sequence().

The method uses:
  - several smooth/deterministic/random initializations,
  - a coarse to fine continuation in sequence length,
  - exponential/probability-mirror descent on the simplex,
  - adaptive temperature over the convolution coefficients,
  - occasional exact simplex projected-gradient steps and random restarts.
"""

import numpy as np
import time

FFT_CACHE = {}


def get_fft_len(m):
    """Smallest cached power-of-two FFT length at least m."""
    m = int(abs(m))
    val = FFT_CACHE.get(m)
    if val is not None:
        return val
    L = 1
    while L < m:
        L <<= 1
    FFT_CACHE[m] = L
    return L


def normalize_sequence(a):
    """Return a non-negative finite sequence of sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        a.fill(1.0 / max(1, a.size))
        s = float(np.sum(a))
    a *= 1.0 / s
    a = np.where(np.isfinite(a), a, 0.0)
    return a


def resize_sequence(seq, n):
    """Linearly resize a sequence to length n and renormalize."""
    seq = np.asarray(seq, dtype=np.float64).ravel()
    seq = seq[np.isfinite(seq)]
    if seq.size == 0:
        return normalize_sequence(np.ones(n, dtype=np.float64))
    if seq.size == n:
        return normalize_sequence(seq)
    xs = np.linspace(0.0, 1.0, seq.size, dtype=np.float64)
    xt = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    y = np.interp(xt, xs, seq)
    y = np.clip(y, 0.0, 1000.0)
    return normalize_sequence(y)


def exponential_start(n, alpha, reverse):
    """Exponential ramp start; alpha controls skew."""
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    if reverse:
        x = 1.0 - x
    a = np.exp(-float(alpha) * x)
    a = np.maximum(a, 1e-12)
    return normalize_sequence(a)


def exp_sine_start(n, beta, phase):
    """Smooth nonnegative start from exp(beta sin)."""
    n = int(n)
    if n <= 1:
        return np.ones(1, dtype=np.float64)
    t = 2.0 * np.pi * np.arange(n, dtype=np.float64) / float(n)
    a = np.exp(float(beta) * np.sin(t + float(phase)))
    return normalize_sequence(a)


def power_start(n, p, q):
    """Asymmetric power bump start."""
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    a = np.power(np.clip(x, 1e-6, 1.0), float(p))
    a += np.power(np.clip(1.0 - x, 1e-6, 1.0), float(q))
    a += 1e-6
    return normalize_sequence(a)


def random_smooth_start(n, rng):
    """Random low-frequency smooth nonnegative start."""
    n = int(n)
    if n <= 1:
        return np.ones(1, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    t = 2.0 * np.pi * x / float(n)
    a = np.ones(n, dtype=np.float64)
    for m in range(1, 10):
        amp = abs(float(rng.normal(0.0, 0.20 / float(m))))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a += amp * np.sin(m * t + phase)
    trend = float(rng.uniform(-0.25, 0.25))
    if n > 1:
        a += trend * (2.0 * x / float(n - 1) - 1.0)
    a = np.where(np.isfinite(a), a, 0.0)
    return normalize_sequence(a)


def lowpass_start(n, rng):
    """Random low-pass noise start."""
    n = int(n)
    if n <= 1:
        return np.ones(1, dtype=np.float64)
    f = np.fft.rfft(np.zeros(n, dtype=np.float64))
    kmax = min(n // 2, 20)
    for m in range(1, kmax + 1):
        amp = 0.18 / (float(m) ** 1.2) * float(rng.uniform(0.3, 1.0))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        if (n % 2 == 0) and (m == n // 2):
            f[m] = amp * np.cos(phase)
        else:
            f[m] = amp * np.exp(1j * phase)
    sig = np.fft.irfft(f, n)
    a = np.ones(n, dtype=np.float64) + 0.9 * sig
    a = np.where(np.isfinite(a), a, 0.0)
    return normalize_sequence(a)


def convolve_fft(a):
    """Full convolution a*a via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    L = get_fft_len(m)
    A = np.fft.rfft(a, L)
    B = np.fft.irfft(A * A, L)
    return B[:m]


def score_conv(a):
    """Return objective and coef vector using the same scale-invariant score."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf, np.zeros(1, dtype=np.float64)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-10:
        return np.inf, np.zeros(2 * n - 1, dtype=np.float64)
    b = convolve_fft(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return np.inf, b
    sc = 2.0 * float(n) * mb / (s * s)
    return sc, b


def correlation_weights(a, p):
    """
    r_i = sum_l a_l p_{i+l}, i=0..n-1.
    This is the subgradient direction (without factor 2) for weighted
    convolution coefficients with weights p.
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    if n == 1:
        return np.array([float(a[0] * p[0])], dtype=np.float64)
    L = get_fft_len(m)
    rp = np.zeros(L, dtype=np.float64)
    rp[:m] = p[::-1]
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(rp, L)
    C = np.fft.irfft(A * P, L)
    return C[m - 1 : m - n - 1 : -1].copy()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    target = 1.0
    if n == 0:
        return np.ones(1, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0.0
    if not np.any(cond):
        return np.full(n, target / float(n), dtype=np.float64)
    idx = np.nonzero(cond)[0]
    rho = cssv[int(idx[-1])]
    x = np.maximum(v - rho, 0.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(n, target / float(n), dtype=np.float64)
    x *= target / s
    return x


def mirror_update(a, r, eta):
    """
    Exponential/mirror update on the simplex.
    r is the unweighted correlation gradient part, and eta is pre-scaled.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    loga = np.log(np.maximum(a, 1e-15))
    val = loga - 2.0 * float(eta) * r
    vmax = float(np.max(val))
    if not np.isfinite(vmax):
        return a.copy()
    val -= vmax
    v = np.exp(val)
    s = float(np.sum(v))
    if not np.isfinite(s) or s <= 0.0:
        return a.copy()
    out = v / s
    return out


def project_update(a, r, eta):
    """Projected gradient update using exact simplex projection."""
    a = np.asarray(a, dtype=np.float64)
    v = a - float(eta) * (2.0 * r)
    return project_simplex(v)


def random_local_perturb(a, rng):
    """Random multiplicative local perturbation on the probability simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return a.copy()
    a2 = a.copy()
    width = max(8, int(n * float(rng.uniform(0.001, 0.05))))
    if width > n:
        width = n
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    a2[start:stop] *= np.exp(float(rng.normal(0.0, 0.12)))
    m = max(1, n // 1500)
    idx = rng.integers(0, n, size=m)
    a2[idx] *= np.exp(rng.normal(0.0, 0.25, size=m))
    if rng.random() < 0.25:
        a2 *= np.exp(float(rng.normal(0.0, 0.03)))
    return normalize_sequence(a2)


def choose_start(n, rng, base_seq, extra_seq=None):
    """Build diverse starts for length n and return the best by FFT score."""
    n = int(n)
    starts = [np.full(n, 1.0 / float(max(1, n)), dtype=np.float64)]
    if extra_seq is not None:
        starts.append(resize_sequence(extra_seq, n))
    if base_seq is not None:
        starts.append(resize_sequence(base_seq, n))

    for alpha in (0.5, 1.0, 2.0, 4.0):
        starts.append(exponential_start(n, alpha, False))
        starts.append(exponential_start(n, alpha, True))

    for beta in (0.3, 0.7, 1.2):
        for phase in (0.0, 0.25 * np.pi, 0.5 * np.pi, 0.75 * np.pi):
            starts.append(exp_sine_start(n, beta, phase))

    for p, q in ((0.5, 2.0), (2.0, 0.5), (1.0, 3.0), (3.0, 1.0)):
        starts.append(power_start(n, p, q))

    for _ in range(8):
        starts.append(random_smooth_start(n, rng))
    for _ in range(4):
        starts.append(lowpass_start(n, rng))

    best_seq = None
    best_score = np.inf
    for seq in starts:
        seq = normalize_sequence(seq)
        if seq.size != n:
            seq = resize_sequence(seq, n)
        sc, _ = score_conv(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
    if best_seq is None:
        best_seq = np.full(n, 1.0 / float(max(1, n)), dtype=np.float64)
        best_score = 2.0
    return best_seq, best_score


def local_search(a, rng, deadline):
    """
    Adaptive mirror/projection descent within a deadline.
    Returns best normalised sequence and its FFT-based score.
    """
    a = normalize_sequence(np.asarray(a, dtype=np.float64))
    n = a.size
    current, b = score_conv(a)
    if not np.isfinite(current):
        a = np.full(n, 1.0 / float(max(1, n)), dtype=np.float64)
        current, b = score_conv(a)

    best = a.copy()
    best_b = b.copy()
    best_score = current

    T = 0.40
    eta_m = 0.06
    eta_p = 0.30
    fail_count = 0
    accept_count = 0
    tol = 1e-11

    while time.monotonic() < deadline:
        h = b * float(n)
        hmax = float(np.max(h))
        if (not np.isfinite(hmax)) or hmax <= 1e-200:
            break

        rn = h / hmax
        z = (rn - 1.0) / T
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        wsum = float(np.sum(w))
        if (not np.isfinite(wsum)) or wsum <= 0.0:
            w = np.full_like(h, 1.0 / float(len(h)), dtype=np.float64)
        else:
            w = w / wsum
            w += 1e-15
            w = w / float(np.sum(w))

        r = correlation_weights(a, w)
        if not np.all(np.isfinite(r)):
            a = random_local_perturb(a, rng)
            current, b = score_conv(a)
            continue

        improved = False

        # Mirror/exponentiated updates, trying a few step scales.
        for mult in (1.0, 0.5, 2.0):
            eta = mult * eta_m * float(n)
            cand = mirror_update(a, r, eta)
            sc, bc = score_conv(cand)
            if np.isfinite(sc) and sc < current - tol:
                a = cand
                b = bc
                current = sc
                if sc < best_score:
                    best = a.copy()
                    best_b = b.copy()
                    best_score = sc
                improved = True
                break

        # Exact projected-gradient fallback.
        if not improved:
            cand = project_update(a, r, eta_p)
            sc, bc = score_conv(cand)
            if np.isfinite(sc) and sc < current - tol:
                a = cand
                b = bc
                current = sc
                if sc < best_score:
                    best = a.copy()
                    best_b = b.copy()
                    best_score = sc
                improved = True

        if improved:
            accept_count += 1
            fail_count = 0
            if T > 0.02:
                T *= 0.995
            if accept_count >= 10:
                eta_m = min(0.30, eta_m * 1.02)
                eta_p = min(1.50, eta_p * 1.02)
                accept_count = 0
        else:
            fail_count += 1
            eta_m = max(0.001, eta_m * 0.85)
            eta_p = max(0.02, eta_p * 0.85)
            if fail_count % 3 == 0:
                T = min(0.80, T * 1.05)

            cand = random_local_perturb(a, rng)
            sc, bc = score_conv(cand)
            if np.isfinite(sc) and sc < current - tol:
                a = cand
                b = bc
                current = sc
                if sc < best_score:
                    best = a.copy()
                    best_b = b.copy()
                    best_score = sc
                fail_count = 0
            else:
                if fail_count > 60:
                    a = best.copy()
                    b = best_b.copy()
                    current = best_score
                    fail_count = 0
                    T = max(T, 0.15)
                    eta_m = max(eta_m, 0.05)
                    eta_p = max(eta_p, 0.20)

                    bridge = 0.80 * best + 0.20 * random_smooth_start(n, rng)
                    bridge = normalize_sequence(bridge)
                    sc2, bc2 = score_conv(bridge)
                    if np.isfinite(sc2) and sc2 < current - tol:
                        a = bridge
                        b = bc2
                        current = sc2
                        if sc2 < best_score:
                            best = a.copy()
                            best_b = b.copy()
                            best_score = sc2
                    else:
                        a = best.copy()
                        b = best_b.copy()
                        current = best_score

    best = normalize_sequence(best)
    return best, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main entry point.

    Runs a coarse-to-fine search for the best non-negative sequence within
    the supplied wall-clock budget and returns a plain Python list.
    """
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 0.0:
        budget = 2400.0

    rng = np.random.default_rng(20240521)

    base_seq = None
    try:
        if "height_sequence_1" in globals():
            arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
            arr = arr[np.isfinite(arr)]
            if arr.size > 0:
                base_seq = arr
    except Exception:
        base_seq = None

    # Very small budget: return a reasonable deterministic start.
    if budget <= 10.0:
        n = 1024
        seq, _ = choose_start(n, rng, base_seq, None)
        return [float(x) for x in seq]

    deadline = t0 + budget - 5.0
    if deadline < t0 + 3.0:
        deadline = t0 + max(1.0, budget - 1.0)

    # Multi-resolution continuation.  The final stage gets most of the time.
    if budget > 900.0:
        stages = [
            (192, 0.04),
            (2048, 0.16),
            (8192, 0.80),
        ]
    elif budget > 300.0:
        stages = [
            (192, 0.10),
            (4096, 0.90),
        ]
    else:
        stages = [
            (2048, 1.0),
        ]

    total_time = max(1.0, deadline - t0)
    best_seq = None
    best_score = np.inf
    current_seq = None
    cum_frac = 0.0

    for stage_idx, (n, frac) in enumerate(stages):
        dl = t0 + total_time * (cum_frac + float(frac))
        if time.monotonic() >= dl - 1.0:
            break

        start, s0 = choose_start(n, rng, base_seq, current_seq)
        if np.isfinite(s0) and s0 < best_score:
            best_score = s0
            best_seq = start.copy()
            current_seq = best_seq.copy()

        if time.monotonic() < dl - 1.0:
            seq, sc = local_search(start, rng, dl)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
            if best_seq is not None:
                current_seq = best_seq.copy()
        else:
            current_seq = start.copy()

        cum_frac += float(frac)
        print(
            "stage length={0} frac={1:.2f} score={2:.10f} wall={3:.1f}s".format(
                n, cum_frac, best_score, time.monotonic() - t0
            )
        )

    if best_seq is None:
        n = stages[-1][0]
        best_seq = np.full(n, 1.0 / float(max(1, n)), dtype=np.float64)
        best_score = 2.0

    best_seq = normalize_sequence(best_seq)
    print(
        "final length={0} best_fast_score={1:.10f} wall={2:.1f}s".format(
            best_seq.size, best_score, time.monotonic() - t0
        )
    )

    return [float(x) for x in best_seq]
```
