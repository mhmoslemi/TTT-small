import time

def compute_objective(a):
    """Compute the objective value directly (faster than calling evaluate_sequence)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def gradient_step(a, lr):
    """One step of gradient descent on the objective."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)

"""
Algorithm: Gradient-based local search for minimizing 2*n*max(conv(a,a))/(sum(a))^2
over non-negative sequences. Uses multiple initializations of various types

descent with adaptive step size.

def _fast_objective(a):
    """Compute objective without the overhead of evaluate_sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def _subgradient(a):
    """Compute subgradient of the objective at point a (assumes sum(a) > 0)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    
    # Soft max weighting: use exponential weighting for smoothness
    # I[k] proportional to exp(conv[k] / temp), but use hard max for simplicity
    max_tol = 1e-8 * M + 1e-15
    I = (conv >= M - max_tol).astype(np.float64)
    
    # grad_M[j] = sum_k I[k] * a[k-j] for valid (k-j)
    # This is np.correlate(I, a, 'valid') since len(I) = 2n-1 >= n = len(a)
    grad_M = np.correlate(I, a, mode='valid')
    
    # d(2nM/S^2)/da_j = 2n/S^2 * dM/da_j - 4nM/S^3
    grad_obj = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3))
    
    return grad_obj, M, S

def _gradient_descent(a, max_steps=50000, time_limit=5.0):
    """Run adaptive gradient descent. Returns (best_a, best_obj)."""
    a = np.asarray(a, dtype=np.float64).copy()

        return a, np.inf

        gnorm = np.linalg.norm(grad)

            break

        grad_normed = grad / gnorm
        
        a_try = np.maximum(a - lr * grad_normed, 0.0)
        S_try = float(np.sum(a_try))
        if S_try < 1e-10:

"""
Algorithm: Multi-start stochastic gradient descent on the probability simplex

of various lengths, with adaptive step size and random restarts.

def _compute_M(a):
    """Compute max of self-convolution."""
    conv = np.convolve(a, a)
    return float(np.max(conv))


def _stochastic_subgradient(a, rng, max_samples=3):
    """Compute stochastic subgradient of max(conv(a,a))."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    
    # Find all positions achieving (near) max
    tol = 1e-10 * M + 1e-15
    max_pos = np.where(conv >= M - tol)[0]
    
    # Stochastic selection for diversity
    if len(max_pos) > max_samples:
        chosen = rng.choice(max_pos, size=max_samples, replace=False)
    else:
        chosen = max_pos
    
    # Build indicator vector
    I = np.zeros(len(conv), dtype=np.float64)
    I[chosen] = 1.0 / len(chosen)
    
    # grad_M[j] = sum_k I[k] * a[k-j] for valid (k-j)
    grad_M = np.correlate(I, a, mode='valid')
    return grad_M, M, conv


def _gd_simplex(a, max_steps=200000, time_limit=5.0, seed=0):
    """Gradient descent on simplex to minimize max(conv(a,a))."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    S = float(np.sum(a))
    if S <= 0:
        return a, np.inf
    a = a / S
    
    best_M = _compute_M(a)
    best_a = a.copy()
    n = len(a)
    best_obj = 2.0 * n * best_M
    
    lr = 0.5
    start = time.monotonic()
    no_improve = 0
    best_lr = 0.5
    
    for step in range(max_steps):
        if time.monotonic() - start > time_limit:
            break
        
        grad_M, M, _ = _stochastic_subgradient(a, rng)
        
        # Scale gradient
        gn = np.linalg.norm(grad_M)
        if gn < 1e-15:
            break
        
        a_try = np.maximum(a - lr * grad_M / gn, 0.0)
        S_try = float(np.sum(a_try))
        if S_try < 1e-10:
            lr *= 0.5
            no_improve += 1
            continue
        a_try = a_try / S_try
        
        M_try = _compute_M(a_try)
        
        if M_try < best_M - 1e-14:
            a = a_try
            best_M = M_try
            best_a = a_try.copy()
            best_obj = 2.0 * n * best_M
            best_lr = lr
            no_improve = 0
            lr = min(lr * 1.03, 5.0)
        else:
            lr *= 0.5
            no_improve += 1
            if no_improve > 500:
                break
    
    return best_a, best_obj

            # New power-law with random params
            n = int(rng.integers(8, 1200))
            p = rng.uniform(0.3, 5.0)
            x = np.arange(n, dtype=np.float64)
            a_new = ((n - x) / n) ** p + rng.uniform(0, 0.1) * rng.random(n)
            tl = remaining * 0.15
            a_best, obj = _gd_simplex(a_new, max_steps=100000, time_limit=tl, seed=seed + 7000 + restart_count)
        else:
            # Soft gradient on perturbed best
            a_pert = best_a.copy()
            n = len(a_pert)
            block_size = max(1, n // 10)
            start_idx = int(rng.integers(0, n - block_size))
            a_pert[start_idx:start_idx + block_size] *= np.exp(rng.normal(0, 0.3, size=block_size))
            a_pert = np.maximum(a_pert, 0.0)
            tl = remaining * 0.2
            a_best, obj = _gd_smooth(a_pert, max_steps=200000, time_limit=tl, seed=seed + 8000 + restart_count)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_best
            print(f"  Restart {restart_count}: obj = {obj:.6f}, n = {len(a_best)} [NEW BEST]", flush=True)
    
    # Phase 4: Final refinement
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        print(f"Phase 4: Final refinement with {remaining:.0f}s remaining", flush=True)
        best_a, best_obj = _gd_simplex(best_a, max_steps=1000000, time_limit=remaining * 0.6, seed=seed + 9999)
        best_a, best_obj = _gd_smooth(best_a, max_steps=500000, time_limit=remaining * 0.3, seed=seed + 9998)
        # One more hard GD pass
        best_a, best_obj = _gd_simplex(best_a, max_steps=500000, time_limit=min(remaining * 0.1, 30), seed=seed + 9997)
    
    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    S = sum(final_list)
    if S < 0.01:
        final_list = [1.0] * len(final_list)
    
    verified = evaluate_sequence(final_list)
    print(f"FINAL: internal_obj={best_obj:.8f}, verified={verified:.8f}, n={len(final_list)}", flush=True)
    
    return final_list
