import numpy as np
import time

def fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)

def quick_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return float('inf')
    b = fft_convolve(a)
    return 2.0 * float(n) * b.max() / (S * S)

def make_pattern(n, rng, ptype):
    x = np.linspace(0, 1, n)
    xx = np.linspace(-1, 1, n)
    if ptype == 0:
        return np.ones(n) / n
    elif ptype == 1:

    elif ptype == 2:

    elif ptype == 3:

    elif ptype == 4:

        a[1::2] = 0.2
        return a / a.sum()
    elif ptype == 5:

    elif ptype == 6:

    elif ptype == 7:

        a = np.clip(a, 0.001, None)
        return a / a.sum()
    else:
        a = np.sin(np.pi * x) + 0.01
        return a / a.sum()

def gradient_dir(a, b, n, frac=0.85, max_peaks=50):
    S = np.sum(a)
    max_b = b.max()
    if max_b <= 0 or S <= 0:
        return np.zeros(n)
    thresh = frac * max_b
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_peaks:
        # Keep only the largest peaks
        peak_vals = b[peaks]
        top_idx = np.argsort(peak_vals)[-max_peaks:]
        peaks = peaks[top_idx]
    
    idx = np.arange(n)
    direction = np.zeros(n)
    for p in peaks:
        bp = b[p]
        kv = p - idx
        valid = (kv >= 0) & (kv < n)
        direction[~valid] += bp
        direction[valid] += (bp - a[kv[valid]] * S)
    return direction

def run_opt(a0, budget_s, rng, verbose=False):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = quick_score(a)
    deadline = time.monotonic() + budget_s
    step = 0.05
    no_improve = 0
    iters = 0
    
    while time.monotonic() < deadline:
        b = fft_convolve(a)
        direction = gradient_dir(a, b, n)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            direction = rng.normal(0, 1, n)
            dnorm = np.linalg.norm(direction)
        direction /= dnorm
        
        a_new = a + step * direction
        a_new = np.clip(a_new, 0.0, None)
        
        score_new = quick_score(a_new)
        
        if score_new < best_score:
            best_a = a_new.copy()
            best_score = score_new
            no_improve = 0
            step = min(0.12, step * 1.015)
        else:
            no_improve += 1
            step = max(0.005, step * 0.98)
        
        a = a_new
        
        if no_improve > 40:
            # Random perturbation
            k = max(1, n // 15)
            idx_p = rng.choice(n, size=k, replace=False)
            a[idx_p] *= np.exp(rng.normal(0, 0.4, k))
            a = np.clip(a, 0.0, None)
            if a.sum() < 0.01:
                a = best_a.copy()
            no_improve = 0
            step = 0.04
            a = best_a + 0.1 * rng.normal(0, 1, n)
            a = np.clip(a, 0.0, None)
        
        iters += 1
        if verbose and iters % 1000 == 0:
            print(f"  iter={iters}, best={best_score:.6f}, step={step:.4f}")
    
    return best_a, best_score
