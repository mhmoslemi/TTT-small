Optimizes non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based smoothed gradient descent followed by LP-based refinement.

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix


def build_conv_matrix(a):
    """Build sparse matrix A such that A @ g == np.convolve(a, g) for any g of same length."""
    n = len(a)
    cols = np.repeat(np.arange(n), n)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    vals = np.tile(a, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def compute_gradient_fft(a, temperature):
    """Compute gradient of logsumexp(conv(a,a)/temperature)*temperature w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_clipped = np.clip(c / temperature, -500, 500)
    log_sum_exp = c_clipped.max() + np.log(np.sum(np.exp(c_clipped - c_clipped.max())))
    P = np.exp(c_clipped - log_sum_exp)
    # P[k] = exp(c_k/tau) / sum_j exp(c_j/tau), so grad[i] = 2 * sum_k P[k] * a[k-i]
    # Equivalent to: grad[i] = 2 * sum_j P[i+j] * a[j]
    # Compute via FFT: correlation of P and a
    N = 1
    while N < len(P) + n:
        N *= 2
    P_pad = np.zeros(N)
    P_pad[:len(P)] = P
    a_pad = np.zeros(N)
    a_pad[:n] = a
    r = np.real(np.fft.ifft(np.fft.fft(P_pad) * np.conj(np.fft.fft(a_pad))))
    grad = 2.0 * r[:n]
    return grad

        constraints = [g >= 0]
        # Build the RHS vector
        rhs = c[lag_indices]
        constraints.append(A_mat @ g <= rhs)

        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                break

        if g.value is None:
            break

        g_val = np.maximum(np.array(g.value, dtype=np.float64).flatten(), 0)
        g_sum = np.sum(g_val)
        if g_sum < 1e-10:
            break

        # Normalize g to match sum of a
        g_val = g_val / g_sum * S

        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            trial = (1 - t) * a + t * g_val
            trial = np.clip(trial, 0, 1000)
            trial_S = np.sum(trial)
            if trial_S < 0.01:
                continue
            trial_c = np.convolve(trial, trial)
            trial_score = 2.0 * n * np.max(trial_c) / (trial_S ** 2)
            if trial_score < best_score:
                best_score = trial_score
                best = trial.copy()
                a = trial.copy()
                improved = True
                break

        if it % 5 == 0:
            elapsed = time.time() - start
            print(f"  LP iter {it}: best={best_score:.6f}, M={M:.4f}, elapsed={elapsed:.1f}s")

        if not improved and it > 0:
            # Try very small steps
            found = False
            for t in [0.005, 0.002, 0.001]:
                trial = (1 - t) * a + t * g_val
                trial = np.clip(trial, 0, 1000)
                trial_S = np.sum(trial)
                if trial_S < 0.01:
                    continue
                trial_c = np.convolve(trial, trial)
                trial_score = 2.0 * n * np.max(trial_c) / (trial_S ** 2)
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
                    a = trial.copy()
                    found = True
                    break
            if not found:
                # Perturb to escape
                rng = np.random.default_rng(1000 + it)
                a = a + rng.normal(0, 0.01 * max(np.mean(a), 1e-6), n)
                a = np.clip(a, 0, 1000)
                a = a / np.sum(a) * S

    return best, best_score


def block_perturbation_search(a, time_budget, seed=99):
    """Additional local search with block perturbations for final polishing."""
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    S = np.sum(a)
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a)) / S**2
    rng = np.random.default_rng(seed)
    start = time.time()

    while time.time() - start < time_budget:
        # Random block perturbation
        width = int(rng.integers(max(1, n // 50), max(2, n // 8)))
        start_idx = int(rng.integers(0, max(1, n - width)))
        end_idx = min(n, start_idx + width)

        candidate = a.copy()
        # Multiplicative perturbation
        candidate[start_idx:end_idx] *= np.exp(rng.normal(0, 0.08, end_idx - start_idx))
        # Occasionally try additive
        if rng.random() < 0.2:
            candidate[start_idx:end_idx] += rng.normal(0, 0.05 * max(np.mean(a), 1e-6), end_idx - start_idx)

        candidate = np.clip(candidate, 0, 1000)
        cand_S = np.sum(candidate)
        if cand_S < 0.01:
            continue
        cand_c = np.convolve(candidate, candidate)
        score = 2.0 * n * np.max(cand_c) / (cand_S ** 2)

        if score < best_score:
            best_score = score
            best = candidate.copy()
            a = candidate.copy()

    return best, best_score


def try_starting_point(a_init, time_budget, seed=42):
    """Try optimizing from a given starting point."""
    n = len(a_init)
    a_init = np.array(a_init, dtype=np.float64).copy()
    a_init = np.clip(a_init, 0, 1000)
    S = np.sum(a_init)
    if S < 0.01:
        return None, float('inf')

    # Normalize
    a_init = a_init / S * np.sqrt(2.0 * n)

    # Phase 1: Gradient descent (40% of budget)
    gd_time = time_budget * 0.4
    a_gd, score_gd = gradient_descent_phase(a_init, gd_time, seed=seed)
    print(f"    After GD: score={score_gd:.6f}")

    # Phase 2: LP refinement (50% of budget)
    lp_time = time_budget * 0.5
    a_lp, score_lp = lp_refine_phase(a_gd, lp_time, full_constraints=True)
    print(f"    After LP: score={score_lp:.6f}")

    # Phase 3: Block perturbation polish (10% of budget)
    blk_time = time_budget * 0.1
    a_blk, score_blk = block_perturbation_search(a_lp, blk_time, seed=seed)
    print(f"    After block polish: score={score_blk:.6f}")

    return a_blk, score_blk


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function combining gradient descent, LP refinement, and local search.
    """
    deadline = time.time() + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)

    # Gather starting points
    starting_points = []

    # Use height_sequence_1 if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        hs1 = np.array(height_sequence_1, dtype=np.float64)
        hs1 = np.clip(hs1, 0, 1000)
        if len(hs1) > 0 and np.sum(hs1) > 0.01:
            starting_points.append(hs1)
            print(f"Using height_sequence_1 with n={len(hs1)}, sum={np.sum(hs1):.2f}")

    n_base = len(starting_points[0]) if starting_points else 2048

    # Generate alternative starting points
    # 1. Uniform
    starting_points.append(np.ones(n_base))

    # 2. Raised cosine
    x = np.linspace(0, 1, n_base)
    hc = 0.5 * (1 + np.cos(2 * np.pi * x))
    starting_points.append(hc + 0.01)  # Avoid exact zeros

    # 3. Gaussian-like
    center = n_base / 2
    sigma = n_base / 6
    gauss = np.exp(-0.5 * ((np.arange(n_base) - center) / sigma) ** 2)
    starting_points.append(gauss + 0.01)

    # 4. Perturbed uniform
    perturbed = np.ones(n_base) + rng.normal(0, 0.1, n_base)
    starting_points.append(np.clip(perturbed, 0.01, None))

    # 5. Slightly asymmetric linear ramp
    ramp = np.linspace(0.5, 1.5, n_base)
    starting_points.append(ramp)

    # Allocate time among starting points
    n_starts = len(starting_points)
    # Give more time to the first (height_sequence_1) if available
    if n_starts > 1:
        time_per = (budget_s - 10.0) / n_starts
        times = [time_per * 2.0] + [time_per * (n_starts - 1) / (n_starts - 1)] * (n_starts - 1)
        # Normalize
        times = [t / sum(times) * (budget_s - 10.0) for t in times]
    else:
        times = [budget_s - 10.0]

    best_overall = None
    best_score_overall = float('inf')

    for i, (sp, t_alloc) in enumerate(zip(starting_points, times)):
        remaining = deadline - time.time()
        if remaining < 30:
            break

        t_use = min(t_alloc, remaining - 20.0)
        if t_use < 10:
            continue

        print(f"\n  Starting point {i+1}/{n_starts} (n={len(sp)}, time={t_use:.0f}s)")
        a_try, score_try = try_starting_point(sp, t_use, seed=seed + i * 7)

        if a_try is not None and score_try < best_score_overall:
            best_score_overall = score_try
            best_overall = a_try
            print(f"  ** New best: {best_score_overall:.6f}")

    # Final: if we have a good solution, do one more refinement
    if best_overall is not None and best_score_overall > 1.5027:
        remaining = deadline - time.time()
        if remaining > 60:
            print(f"\n  Final refinement (time={remaining:.0f}s)")
            a_final, score_final = lp_refine_phase(
                best_overall, min(remaining - 10, 600), full_constraints=True)
            if score_final < best_score_overall:
                best_overall = a_final
                best_score_overall = score_final
                print(f"  ** Final best: {best_score_overall:.6f}")

    if best_overall is None:
        # Ultimate fallback
        best_overall = np.ones(1024)
        best_score_overall = float(evaluate_sequence(best_overall.tolist()))

    # Verify with exact evaluator
    final_score = float(evaluate_sequence(best_overall.tolist()))
    if not np.isfinite(final_score):
        best_overall = np.ones(1024)
        final_score = float(evaluate_sequence(best_overall.tolist()))

    print(f"\n  FINAL: score={final_score:.6f}, n={len(best_overall)}")

    return [float(x) for x in best_overall]
