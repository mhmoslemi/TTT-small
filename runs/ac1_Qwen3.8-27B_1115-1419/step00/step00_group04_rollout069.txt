"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Phase 1: Entropy-smoothed gradient descent with annealing.
Phase 2: Hard active-set descent on the exact max convolution coefficient.
Phase 3: Block-perturbation restarts to escape local minima, each followed
         by short smooth + hard refinement.
All helper functions are top-level; no lambdas or closures.
"""

import numpy as np
import time


def _full_conv(a):
    """Full self-convolution of a (length 2n-1)."""
    return np.convolve(a, a)


def _softmax(w, tau):
    """Stable softmax of w / tau."""
    mx = np.max(w)
    e = np.exp((w - mx) / tau)
    s = float(e.sum())
    if s < 1e-300:
        return np.full_like(w, 1.0 / len(w), dtype=np.float64)
    return e / s


def _grad_smooth(p, a):
    """Gradient of the entropy-smoothed max of conv(a,a) w.r.t. a.
    Uses 2 * cross-correlation of softmax weights p with a."""
    return 2.0 * np.correlate(p, a, mode='valid')


def _grad_hard(a, k, n):
    """Exact gradient of conv(a,a)[k] w.r.t. a (factor of 2, ignoring diagonal)."""
    g = np.zeros(n, dtype=np.float64)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        ai = k - np.arange(js, je)
        g[js:je] = 2.0 * a[ai]
    return g


def _project(a, n):
    """Project onto {x >= 0, sum(x) = n}."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64)
    return a / s * n


def _obj_n(a, n):
    """Objective 2*max(conv)/n, valid when sum(a)=n."""
    return 2.0 * float(np.max(_full_conv(a))) / n


def _smooth_run(a, n, deadline, tag, it_limit=2000000):
    """Entropy-smoothed gradient descent. Returns (best_a, best_obj)."""
    tau = n / 4.0
    eta = 0.04
    best = a.copy()
    best_o = _obj_n(a, n)
    stall = 0
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _full_conv(a)
        mc = float(np.max(c))
        o = 2.0 * mc / n
        if o < best_o - 1e-10:
            best_o = o
            best = a.copy()
            stall = 0
            if it % 5000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} tau={tau:.1f} eta={eta:.4f}")
        else:
            stall += 1
        p = _softmax(c, tau)
        g = _grad_smooth(p, a)
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-13:
            break
        a = _project(a - eta * g / mg, n)
        tau = max(tau * 0.99997, n * 0.001)
        if stall > 8000:
            eta = max(eta * 0.5, 1e-7)
            stall = 0
    return best, best_o


def _hard_run(a, n, deadline, tag, it_limit=2000000):
    """Hard active-set descent on exact max. Returns (best_a, best_obj)."""
    best = a.copy()
    best_o = _obj_n(a, n)
    eta = 0.02
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _full_conv(a)
        k = int(np.argmax(c))
        mc = float(c[k])
        o = 2.0 * mc / n
        if o < best_o - 1e-10:
            best_o = o
            best = a.copy()
            if it % 10000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} eta={eta:.4f}")
        g = _grad_hard(a, k, n)
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-13:
            break
        a = _project(a - eta * g / mg, n)
        if it > 0 and it % 30000 == 29999:
            eta = max(eta * 0.90, 1e-8)
    return best, best_o


def _perturb_block(a, n, rng, sigma, width_frac=None):
    """Multiply a random contiguous block by exp(N(0,sigma)), then project."""
    b = a.copy()
    if width_frac is None:
        w = max(2, n // 50)
    else:
        w = max(2, int(n * width_frac))
    s = int(rng.integers(0, max(1, n - w + 1)))
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _project(b, n)

    p1_end = min(time.monotonic() + 0.40 * (budget_s - 10.0), deadline - 2.0)
