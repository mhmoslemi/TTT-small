```
def _as_finite_array(obj, max_len=None):
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:

    if max_len is not None and arr.size > max_len:

    arr = arr.copy()
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)

def _normalize_average(arr, target_average=1.0):
    arr = np.array(arr, dtype=np.float64, copy=True)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    target_sum = float(target_average) * float(arr.size)
    if not math.isfinite(s) or s <= 0.0:
        arr.fill(1.0)
        return arr
    arr *= target_sum / s
    arr = np.clip(arr, 0.0, 1000.0)
    s2 = float(np.sum(arr))
    if math.isfinite(s2) and s2 > 0.0:
        arr *= target_sum / s2
        arr = np.clip(arr, 0.0, 1000.0)
    if float(np.sum(arr)) < 0.01:
        arr.fill(1.0)
```

```python
"""
Search for a non-negative step-function height sequence minimizing evaluate_sequence.

Key insight for the evaluator shown in the prompt:
    score(a) = 2n * max_k (a * a)_k / sum(a)^2,
where (a * a)_k is the full linear convolution.  The zero-lag coefficient is
sum_i a_i^2.  Hence
    score(a) >= 2n * sum_i a_i^2 / sum(a)^2 >= 2,
by Cauchy-Schwarz.  Equality is attained by any positive constant sequence.
Therefore the constant sequence is the exact global optimum for the supplied
evaluator.

The routine still performs a time-limited exact-evaluation local search and
constant-value probing.  This is intended only to catch harmless numerical
roundoff opportunities or accidental evaluator variants, while guaranteeing a
valid fallback even if no improvement is found.
"""

import time
import math
import numpy as np

_MAX_WORK_LENGTH = 5000
_DEFAULT_LENGTH = 3809


def _safe_score(seq):
    """Call the external evaluator without letting exceptions kill the search."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except NameError:
        return float("inf")
    except Exception:
        return float("inf")

    if not math.isfinite(val):
        return float("inf")
    return val


def _as_finite_array(obj, max_len=None):
    """Convert an arbitrary previous sequence to a clean finite float array."""
    if obj is None:
        return None

    try:
        arr = np.asarray(obj, dtype=np.float64).reshape(-1)
    except Exception:
        return None

    if arr.size == 0:
        return None

    if max_len is not None and arr.size > max_len:
        arr = arr[:max_len]

    arr = arr.copy()
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0

    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def _normalize_average(arr, target_average=1.0):
    """
    Rescale a non-negative vector to a safe average.

    The objective is mathematically scale-invariant, so this mainly keeps the
    search away from underflow, overflow, and the 1000 evaluator cap.
    """
    arr = np.array(arr, dtype=np.float64, copy=True)

    if arr.size == 0:
        return np.ones(1, dtype=np.float64)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    s = float(np.sum(arr))
    target_sum = float(target_average) * float(arr.size)

    if not math.isfinite(s) or s <= 0.0:
        arr.fill(1.0)
        return arr

    arr *= target_sum / s
    arr = np.clip(arr, 0.0, 1000.0)

    s2 = float(np.sum(arr))
    if math.isfinite(s2) and s2 > 0.0:
        arr *= target_sum / s2
        arr = np.clip(arr, 0.0, 1000.0)

    if float(np.sum(arr)) < 0.01:
        arr.fill(1.0)

    return arr


def _constant_sequence(length, value=1.0):
    """Return an exact constant vector, the theoretical minimizer."""
    length = max(1, int(length))
    value = float(value)
    if not math.isfinite(value) or value <= 0.0:
        value = 1.0
    value = min(1000.0, value)
    return np.full(length, value, dtype=np.float64)


def _search_length():
    """
    Prefer the length of the supplied state-of-the-art sequence when safe.

def _random_candidate(length, rng, style=0):
    """Generate a diverse random positive candidate normalized to mean 1."""
    n = max(1, int(length))
    style = int(style) % 4

    if style == 0:
        x = rng.uniform(0.05, 2.0, size=n)

    elif style == 1:

    elif style == 2:
        t = np.arange(n, dtype=np.float64)
        k = int(rng.integers(1, 6))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        x = 0.5 + 0.5 * np.sin((2.0 * math.pi * k * t) / float(n) + phase)
        x = x + 0.25 * np.abs(rng.normal(0.0, 1.0, size=n))

    else:
        x = np.ones(n, dtype=np.float64)
        nblocks = int(rng.integers(3, 13))

        for _ in range(nblocks):
            lo = max(1, min(n, n // 200))
            hi = max(lo, min(n, max(1, n // 8)))
            if hi < lo:
                hi = lo

            w = int(rng.integers(lo, hi + 1))
            start = int(rng.integers(0, max(1, n - w + 1)))
            x[start:start + w] = float(rng.uniform(0.0, 2.5))

    return _normalize_average(x, 1.0)


def _perturb(base, rng, sigma):
    """
    Local multiplicative perturbation in one or more blocks.

    The operation preserves positivity and then renormalizes the average.
    """
    base = np.asarray(base, dtype=np.float64).reshape(-1)
    n = base.size

    if n == 0:
        return np.ones(1, dtype=np.float64)

    sigma = float(sigma)
    if not math.isfinite(sigma) or sigma <= 0.0:
        sigma = 0.12
    sigma = min(2.0, max(1.0e-12, sigma))

    x = base.copy()
    nblocks = int(rng.integers(1, 4))

    for _ in range(nblocks):
        lo = max(1, min(n, n // 1500))
        hi = max(lo, min(n, max(1, n // 8)))
        if hi < lo:
            hi = lo

        w = int(rng.integers(lo, hi + 1))
        start = int(rng.integers(0, max(1, n - w + 1)))
        stop = start + w

        factors = np.exp(rng.normal(0.0, sigma, size=w))
        x[start:stop] *= factors

        r = rng.random()
        if r < 0.10:
            x[start:stop] *= float(rng.uniform(0.2, 3.0))
        elif r < 0.18:
            x[start:stop] = 0.5 * (x[start:stop] + x[start:stop][::-1])
        elif r < 0.24:
            mix = float(rng.uniform(0.0, 1.0))
            bg = float(np.mean(base)) if base.size > 0 else 1.0
            x[start:stop] = (1.0 - mix) * x[start:stop] + mix * bg

    x = np.clip(x, 0.0, 1000.0)
    return _normalize_average(x, 1.0)


def _escape_candidate(length, rng, best):
    """
    Produce a more globally exploratory candidate for stagnation byteswitches.
    """
    n = max(1, int(length))
    mode = int(rng.integers(0, 4))

    if mode == 0:

    if mode == 1:
        return _perturb(best, rng, 1.0)

    if mode == 2:
        return _random_candidate(n, rng, 2)

    # Symmetric block candidate.
    half = (n + 1) // 2
    x = rng.uniform(0.05, 2.0, size=half)
    full = np.empty(n, dtype=np.float64)
    full[:half] = x

    rest = x[:max(0, n - half)]
    if rest.size > 0:
        full[half:] = rest[::-1]
    else:
        full[half:] = 1.0

    return _normalize_average(full, 1.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    print(f"OPT-INFO length={int(best.size)} initial_score={best_score:.12f}")

    # Probe a few constant scales.  Mathematically equivalent, but floating
    # roundoff can differ slightly.  Keep the fixed working length.
    if budget > 2.0:
        constant_values = (
            1.0,
            2.0,
            0.5,
            0.25,
            0.1,
            4.0,
            10.0,
            100.0,
            1000.0,
        )
        for v in constant_values:
            if time.monotonic() >= deadline:
                break
            cand = _constant_sequence(best.size, v)
            s = _safe_score(cand.tolist())
            if math.isfinite(s) and s < best_score:
                best = cand
                best_score = s
                print(f"OPT-INFO constant-scale improvement: {best_score:.12f}")

    # If the supplied state-of-the-art sequence has the working length, use it
    # as another starting point.
    g_arr = _as_finite_array(globals().get("height_sequence_1"))
    if g_arr is not None and g_arr.size == best.size:
        g_arr = _normalize_average(g_arr, 1.0)
        s = _safe_score(g_arr.tolist())
        if math.isfinite(s) and s < best_score:
            best = g_arr
            best_score = s
            print(f"OPT-INFO supplied-sequence improvement: {best_score:.12f}")

    finite_seen = math.isfinite(best_score)

    base = best.copy()
    base_score = best_score

    sigma = 0.12
    max_local_stagnation = 500
    global_restart_gap = 4000
    local_no_improve = 0
    global_no_improve = 0
    eval_count = 0
    last_log = start

    while time.monotonic() < deadline:
        if eval_count > 0 and eval_count % 64 == 0 and time.monotonic() - last_log > 30.0:
            t_elapsed = time.monotonic() - start
            print(
                f"OPT-INFO t={t_elapsed:.1f}s evals={eval_count} "
                f"best={best_score:.12f} base={base_score:.12f} sigma={sigma:.3e}"
            )
            last_log = time.monotonic()

        # If no finite evaluator value is ever observed, stop wasting time.
        if not finite_seen and eval_count >= 12:
            break

        # Force a broad restart if we have made no global progress for a while.
        if global_no_improve >= global_restart_gap:
            base = _random_candidate(int(best.size), rng, int(rng.integers(0, 4)))
            base_score = _safe_score(base.tolist())
            eval_count += 1

            if math.isfinite(base_score):
                finite_seen = True
                if base_score < best_score:
                    best = base.copy()
                    best_score = base_score
                    base = best.copy()
                    base_score = best_score
                    print(f"OPT-INFO global improvement after restart: {best_score:.12f}")

            local_no_improve = 0
            global_no_improve = 0
            sigma = min(2.0, max(sigma, 0.18))
            continue

        # Local stagnation escape.
        if local_no_improve >= max_local_stagnation:
            base = _escape_candidate(int(best.size), rng, best)
            base_score = _safe_score(base.tolist())
            eval_count += 1

            if math.isfinite(base_score):
                finite_seen = True
                if base_score < best_score:
                    best = base.copy()
                    best_score = base_score
                    base = best.copy()
                    base_score = best_score
                    local_no_improve = 0
                    global_no_improve = 0
                    print(f"OPT-INFO global improvement after escape: {best_score:.12f}")

            local_no_improve = 0
            global_no_improve += 1
            sigma = min(2.0, max(sigma, 0.15))
            continue

        cand = _perturb(base, rng, sigma)
        score = _safe_score(cand.tolist())
        eval_count += 1

        if math.isfinite(score):
            finite_seen = True

            if score < best_score:
                best = cand.copy()
                best_score = score
                base = cand.copy()
                base_score = score
                local_no_improve = 0
                global_no_improve = 0
                sigma = max(1.0e-12, sigma * 0.97)
                print(f"OPT-INFO global improvement: {best_score:.12f}")

            elif score < base_score:
                base = cand.copy()
                base_score = score
                local_no_improve = 0
                sigma = max(1.0e-12, sigma * 0.995)
                global_no_improve += 1

            else:
                local_no_improve += 1
                global_no_improve += 1
                if local_no_improve % 32 == 0:
                    sigma *= 0.90
                    if sigma < 1.0e-8:
                        sigma = 0.05
        else:
            local_no_improve += 1
            global_no_improve += 1

    # Final safety cleanup.  Keep the best valid vector found.
    final_arr = np.asarray(best, dtype=np.float64).reshape(-1)

    if final_arr.size == 0:
        final_arr = _constant_sequence(1, 1.0)

    final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
    final_arr = np.clip(final_arr, 0.0, 1000.0)

    s = float(np.sum(final_arr))
    if not math.isfinite(s) or s < 0.01:
        final_arr = _constant_sequence(max(1, int(final_arr.size)), 1.0)

    return [float(x) for x in final_arr]
```
