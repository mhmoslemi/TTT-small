with adaptive step size and simulated annealing. Uses FFT-accelerated convolution
for O(n log n) gradient computation per iteration.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute linear convolution of real arrays a and b using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _compute_grad_dir(a, lam):
    """
    Compute negative gradient direction for minimizing max(conv(a,a))/sum(a)^2.
    Returns (direction, B, S) where direction is the step to add to a.
    """
    n = len(a)
    c = _fft_convolve(a, a)
    B = float(c.max())
    S = float(a.sum())

    if S < 1e-15:
        return None, B, S

    # Softmax weights concentrating on near-maximal lags
    c_shifted = c - B
    weights = np.exp(lam * c_shifted)
    wsum = float(weights.sum())
    if wsum < 1e-30:
        weights = np.zeros_like(c)
        weights[int(np.argmax(c))] = 1.0
    else:
        weights /= wsum

    # g[m] = 2 * sum_k weights[k] * a[k-m]
    #      = 2 * convolve(a[::-1], weights)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_g = _fft_convolve(a_rev, weights)
    g = 2.0 * conv_g[n - 1:2 * n - 1]

    # Negative gradient direction (up to positive constant 2n/S^3):
    # d[m] = 2*B - S*g[m]
    d = 2.0 * B - S * g

    return d, B, S


def _fast_score(a):
    """Fast objective computation (no Python list overhead)."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    B = float(c.max())
    return 2.0 * n * B / (S * S)


def _anneal_restart(best, n, rng, frac=0.3):
    """Partially randomize a fraction of the sequence for restart."""
    candidate = best.copy()
    width = max(1, int(n * frac))
    start = int(rng.integers(0, n - width + 1))
    candidate[start:start + width] = rng.random(width) * (best[start:start + width].mean() + 1e-6)
    S = candidate.sum()
    if S > 0.01:
        candidate /= S
    return candidate
