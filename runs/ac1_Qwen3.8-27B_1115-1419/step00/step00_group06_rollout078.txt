def fft_full(a):
    n = len(a)
    if n == 1: return np.array([a[0]*a[0]], dtype=np.float64)
    m = 2*n - 1
    L = 1
    while L < m: L <<= 1
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    c = c[:m]
    # Clean tiny negatives
    if np.any(c < 0): c = np.maximum(c, 0.0)
    return c

def softmax_lags(C, avgC, beta):
    if avgC <=0: return np.full(len(C), 1/len(C))
    x = C / avgC
    z = beta * x
    z = z - z.max()
    e = np.exp(z)
    e = np.nan_to_num(e, nan=0.0, posinf=0.0, neginf=0.0)
    s = e.sum()
    if s <=0 or not np.isfinite(s): return np.full(len(C),1/len(C))
    return e/s

def make_block_perturbation(a, rng, sigma=0.25):
    n=len(a); cand=a.copy()
    width=int(rng.integers(max(2,n//50), max(3,n//5)))
    start=int(rng.integers(0, n-width+1)) if n>width else 0
    stop=start+width
    mult=np.exp(rng.normal(0.0,sigma))
    cand[stop] *= mult
    if rng.random()<0.2:
       # global smooth noise
       cand *= np.exp(rng.normal(0,0.05,cand.size))
    cand=np.maximum(cand,0)
    return normalize_sum(cand,n)
```

def fft_full(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([a[0]*a[0]], dtype=np.float64)
    m = 2*n - 1
    L = 1
    while L < m:
        L <<= 1
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    c = c[:m]
    # remove tiny negatives
    c = np.maximum(c, 0.0)
    return c

def softmax_lags(C, avgC, beta):
    m=C.size
    if m==0: return None
    if not np.isfinite(beta) or beta <=0: beta=1.0
    if not np.isfinite(avgC) or avgC <=0: avgC=np.mean(C) if C.size else 1.0
    if not np.isfinite(avgC) or avgC <=0: avgC=1.0
    x=np.asarray(C,float)/avgC
    z=beta*x
    zmax=np.max(z)
    if not np.isfinite(zmax):
       return np.full(m,1.0/m)
    z=z-zmax
    e=np.exp(z)
    e=np.nan_to_num(e, nan=0.0, posinf=0.0, neginf=0.0)
    s=e.sum()
    if not np.isfinite(s) or s <=1e-300:
       return np.full(m,1.0/m)
    return e/s

```python
"""
Multi-phase nonnegative step-function search.

The objective is
    2*n*max(conv(a,a)) / sum(a)^2
for a nonnegative finite sequence a.  The program uses several complementary
heuristics inside a strict time budget:

1. A set of analytic nonnegative starts (ramps, concave/convex ramps, exponentials).
2. A mirror-descent / multiplicative-update method on the full sequence, using an
   FFT-based gradient of a smooth log-sum-exp approximation to the maximum
   convolution value.
3. A small simulated-annealing block-perturbation search.
4. A local one-coordinate refinement pass with exact local updates and renormalization.
5. Final exact evaluation of a few candidate lists, returning the best finite list.

All sequences are kept nonnegative, finite, clipped to [0, 1000], and are
normalized to average 1 during search because the objective is scale invariant.
"""

import numpy as np
import time

MAX_VALUE = 1000.0


def fft_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT, cleaned of tiny negative values."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    c = c[:m]
    c = np.where(np.isfinite(c), c, 0.0)
    return np.maximum(c, 0.0)


def normalize_sum(a: np.ndarray, target: float = None) -> np.ndarray:
    """Clip to nonnegative finite values and scale to the requested sum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if target is None:
        target = float(n)
    if n == 0:
        return np.array([], dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, MAX_VALUE)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(n, dtype=np.float64)
        s = float(n)

    a = a * (target / s)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, MAX_VALUE)

    s2 = float(a.sum())
    if not np.isfinite(s2) or s2 <= 1e-12:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a * (target / s2)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, MAX_VALUE)
    return a


def fast_score(a: np.ndarray) -> float:
    """Fast internal objective using FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(a)):
        return np.inf
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-8:
        return np.inf
    c = fft_full(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (s * s))


def softmax_lags(c: np.ndarray, avg_c: float, beta: float) -> np.ndarray:
    """Stable softmax over convolution lags, optionally smoothed by average."""
    m = c.size
    if m == 0:
        return np.array([], dtype=np.float64)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0
    if not np.isfinite(avg_c) or avg_c <= 0.0:
        avg_c = float(np.mean(c)) if c.size else 1.0
    if not np.isfinite(avg_c) or avg_c <= 0.0:
        avg_c = 1.0

    x = np.asarray(c, dtype=np.float64) / avg_c
    z = beta * x
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return np.full(m, 1.0 / m, dtype=np.float64)
    z = z - zmax
    e = np.exp(z)
    e = np.where(np.isfinite(e), e, 0.0)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(m, 1.0 / m, dtype=np.float64)
    return e / s


def multi_softmax_lags(c: np.ndarray, avg_c: float, beta: float) -> np.ndarray:
    """Average a few softmax temperatures for a more stable active-set gradient."""
    p1 = softmax_lags(c, avg_c, beta)
    if p1.size == 0:
        return p1
    p2 = softmax_lags(c, avg_c, max(1e-6, 3.0 * beta))
    p3 = softmax_lags(c, avg_c, max(1e-6, 0.35 * beta))
    return (p1 + p2 + p3) / 3.0


def compute_d(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Unnormalized gradient direction:
        D_i = sum_k p_k * a_{k-i}  (a_t = 0 outside [0, n-1]).
    This is proportional to the derivative of the soft maximum of conv(a,a).
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        val = float(p[0] * a[0]) if p.size else 0.0
        return np.array([max(0.0, val)], dtype=np.float64)

    need = m + n - 1
    L = 1
    while L < need:
        L <<= 1

    fa = np.fft.rfft(a, L)
    fp = np.fft.rfft(p[::-1], L)
    conv = np.fft.irfft(fp * fa, L)

    # Desired indices are m-1, m-2, ..., m-n
    d = conv[m - 1 : m - n - 1 : -1]
    if d.size != n:
        # Very defensive fallback; should not normally be needed.
        d = np.zeros(n, dtype=np.float64)
        for i in range(n):
            lo = max(0, i)
            hi = min(m, i + n)
            ks = np.arange(lo, hi, dtype=np.int64)
            d[i] = float(np.dot(p[ks], a[ks - i]))

    d = np.where(np.isfinite(d), d, 0.0)
    return np.maximum(d, 0.0)


def mirror_update(a: np.ndarray, d: np.ndarray, eta: float, noise: float, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative / mirror update keeping nonnegativity.
    High D positions are reduced, low D positions are increased.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    d = np.asarray(d, dtype=np.float64)
    n = a.size
    if n == 0:
        return a

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-12)

    d = np.where(np.isfinite(d), d, 0.0)
    d = d - float(np.mean(d))
    sd = float(np.std(d))
    if np.isfinite(sd) and sd > 1e-12:
        d = d / sd

    if np.isfinite(eta) and eta > 0.0:
        d = np.clip(eta * d, -1.0, 1.0)
        a = a * np.exp(-d)

    if np.isfinite(noise) and noise > 0.0 and rng is not None:
        nz = rng.standard_normal(n)
        nz = np.where(np.isfinite(nz), nz, 0.0)
        nz = np.clip(nz, -3.0, 3.0)
        a = a * np.exp(noise * nz)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(n, dtype=np.float64)

    a = a * (n / s)
    a = np.clip(a, 0.0, MAX_VALUE)
    s2 = float(a.sum())
    if not np.isfinite(s2) or s2 <= 1e-12:
        return np.ones(n, dtype=np.float64)
    a = a * (n / s2)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, MAX_VALUE)
    return a


def make_block_perturbation(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.25) -> np.ndarray:
    """Random block multiplicative perturbation, nonnegative and renormalized."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    cand = a.copy()
    if n <= 1:
        return normalize_sum(cand, n)

    lo = int(max(2, min(n - 1, n // 50)))
    hi = int(min(n - 1, max(lo + 1, n // 5)))
    if hi <= lo:
        hi = lo + 1
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))

    if rng.random() < 0.35:
        lo2 = int(max(2, min(n - 1, n // 40)))
        hi2 = int(min(n - 1, max(lo2 + 1, n // 4)))
        if hi2 > lo2:
            width2 = int(rng.integers(lo2, hi2))
            start2 = int(rng.integers(0, max(1, n - width2)))
            stop2 = min(n, start2 + width2)
            cand[start2:stop2] *= float(np.exp(rng.normal(0.0, sigma)))

    return normalize_sum(cand, n)


def make_base(n: int, kind: str, rng: np.random.Generator = None) -> np.ndarray:
    """Analytic nonnegative starting shapes, normalized to sum n."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if kind == "ramp_eps":
        a = 0.05 + 0.95 * x
    elif kind == "quad_pos":
        a = 2.0 * x + 0.25 * (4.0 * x * x - 4.0 * x + 2.0 / 3.0)
    elif kind == "quad_neg":
        a = 2.0 * x - 0.25 * (4.0 * x * x - 4.0 * x + 2.0 / 3.0)
    elif kind == "sqrt":
        a = np.sqrt(np.maximum(x, 0.0))
    elif kind == "power07":
        a = np.power(np.maximum(x, 0.0), 0.7)
    elif kind == "power15":
        a = np.power(np.maximum(x, 0.0), 1.5)
    elif kind == "x2":
        a = x * x
    elif kind == "x_2x":
        a = x * (2.0 - x)
    elif kind == "exp1":
        a = np.exp(1.0 * x)
    elif kind == "exp2":
        a = np.exp(2.0 * x)
    elif kind == "expneg1":
        a = np.exp(-1.0 * x)
    elif kind == "tanh1":
        a = 0.5 + 0.5 * np.tanh(1.0 * (2.0 * x - 1.0))
    elif kind == "tanh3":
        a = 0.5 + 0.5 * np.tanh(3.0 * (2.0 * x - 1.0))
    elif kind == "saturate2":
        a = 1.0 - np.exp(-2.0 * x)
    elif kind == "saturate4":
        a = 1.0 - np.exp(-4.0 * x)
    else:  # default ramp
        a = x.copy()

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a * (n / s)
    a = np.where(np.isfinite(a), a, 0.0)
    return np.clip(a, 0.0, MAX_VALUE)


def run_mirror(a: np.ndarray, max_time: float, rng: np.random.Generator, tag: str = "mirror") -> np.ndarray:
    """Length-limited mirror-descent search."""
    a = normalize_sum(np.asarray(a, dtype=np.float64), len(np.asarray(a, dtype=np.float64)))
    n = a.size
    if n == 0:
        return a

    start = time.monotonic()
    total = max(1.0, float(max_time))
    deadline = start + total

    c = fft_full(a)
    score = float(2.0 * np.max(c) / n)
    best = a.copy()
    best_score = score

    it = 0
    no_improve = 0
    last_print = start

    while time.monotonic() < deadline:
        c = fft_full(a)
        s = float(a.sum())
        if s <= 1e-12:
            a = np.ones(n, dtype=np.float64)
            c = fft_full(a)
            s = float(n)
        score = float(2.0 * n * np.max(c) / (s * s))

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        frac = (time.monotonic() - start) / total
        if frac < 0.25:
            beta = 1.0 + 4.0 * (frac / 0.25)
        elif frac < 0.60:
            beta = 5.0 + 15.0 * ((frac - 0.25) / 0.35)
        else:
            beta = 20.0 + 80.0 * ((frac - 0.60) / 0.40)

        eta = 0.12 * (1.0 - 0.5 * frac) + 0.005
        noise = 0.05 * (1.0 - 0.6 * frac) + 0.002

        if no_improve > 2500:
            beta = min(300.0, beta * 1.5)
            eta = min(0.35, eta * 1.35)
            noise = min(0.22, noise * 1.7)

        if no_improve > 8000:
            cand = make_block_perturbation(best, rng, 0.5)
            cc = fft_full(cand)
            sc = float(2.0 * np.max(cc) / cand.size)
            if np.isfinite(sc) and sc < best_score + 0.05:
                a = cand
            else:
                a = best.copy()
            no_improve = 0

        avg_c = (s * s) / float(2 * n - 1)
        p = multi_softmax_lags(c, avg_c, beta)
        d = compute_d(a, p)
        a = mirror_update(a, d, eta, noise, rng)
        it += 1

        now = time.monotonic()
        if now - last_print > 15.0:
            print(f"{tag}: it={it} current_score={score:.6f} best={best_score:.6f} beta={beta:.3f} eta={eta:.4f}")
            last_print = now

    c = fft_full(a)
    s = float(a.sum())
    if s > 1e-12:
        score = float(2.0 * n * np.max(c) / (s * s))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

    return best


def random_local_search(a: np.ndarray, max_time: float, rng: np.random.Generator) -> np.ndarray:
    """Short simulated-annealing block perturbation search."""
    a = normalize_sum(np.asarray(a, dtype=np.float64), len(np.asarray(a, dtype=np.float64)))
    n = a.size
    if n == 0:
        return a

    start = time.monotonic()
    total = max(1.0, float(max_time))
    deadline = start + total

    c = fft_full(a)
    score = float(2.0 * np.max(c) / n)
    best = a.copy()
    best_score = score
    it = 0
    last_print = start

    while time.monotonic() < deadline:
        frac = (time.monotonic() - start) / total
        sigma = 0.15 + 0.35 * max(0.0, 1.0 - frac)

        if rng.random() < 0.10:
            cand = a * np.exp(0.20 * rng.standard_normal(n))
            cand = normalize_sum(cand, n)
        else:
            cand = make_block_perturbation(a, rng, sigma)

        cc = fft_full(cand)
        s = float(cand.sum())
        if s <= 1e-12:
            continue
        sc = float(2.0 * n * np.max(cc) / (s * s))
        if not np.isfinite(sc):
            continue

        temp = max(1e-5, 0.05 * max(0.0, best_score - 1.0) + 1e-5) * max(0.1, 1.0 - 0.5 * frac)
        if sc < score or (temp > 0.0 and rng.random() < np.exp(-(sc - score) / temp)):
            a = cand
            score = sc
        if sc < best_score:
            best = cand.copy()
            best_score = sc

        if score > best_score * 1.5:
            a = best.copy()
            score = best_score

        it += 1
        now = time.monotonic()
        if now - last_print > 10.0:
            print(f"random-search: it={it} current={score:.6f} best={best_score:.6f}")
            last_print = now

    c = fft_full(a)
    s = float(a.sum())
    if s > 1e-12:
        sc = float(2.0 * n * np.max(c) / (s * s))
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc
    return best


def coordinate_refine(a: np.ndarray, c: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:

                new_slice[i] = base[i] + x * x
                max_here = float(np.max(new_slice))
                mx = max(max_here, outside_max)
                sc = float(2.0 * n * mx / (snew * snew))
                if np.isfinite(sc) and sc < best_cand_score - 1e-12:
                    best_cand_score = sc
                    best_x = x
                    best_new = new_slice
                    best_snew = snew

            if best_new is not None:
                ccur[i:i + n] = best_new
                current[i] = best_x
                factor = float(n / best_snew)
                if abs(factor - 1.0) > 1e-8:
                    current *= factor
                    ccur *= factor * factor
                cur_score = best_cand_score
                improved_pass = True
                if cur_score < best_score:
                    best_score = cur_score
                    best = current.copy()

        if improved_pass:
            stagnant_passes = 0
        else:
            stagnant_passes += 1
            if stagnant_passes >= 2:
                break

    c = fft_full(best)
    s = float(best.sum())
    if s > 1e-12:
        sc = float(2.0 * len(best) * np.max(c) / (s * s))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
    return best


def propose_candidate(seed=42, budget_s=2394, **kwargs):
    """
    Time-bounded search for a nonnegative sequence minimizing the given objective.
    Returns the best list found before the budget expires.
    """
    try:
        rng = np.random.default_rng(seed)
        t0 = time.monotonic()
        deadline = t0 + max(10.0, float(budget_s) - 5.0)

        n = 5582  # comparable to the provided construction length

        base_kinds = (
            "ramp",
            "ramp_eps",
            "quad_pos",
            "quad_neg",
            "sqrt",
            "power07",
            "power15",
            "x2",
            "x_2x",
            "exp1",
            "exp2",
            "expneg1",
            "tanh1",
            "tanh3",
            "saturate2",
            "saturate4"
        )

        best = None
        best_score = np.inf

        for kind in base_kinds:
            try:
                a = make_base(n, kind, rng)
                sc = fast_score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = a
            except Exception:
                pass

        extra_candidate = None
        try:
            hs = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
            if hs.ndim == 1 and 64 <= hs.size <= 8192 and np.all(np.isfinite(hs)):
                hsn = normalize_sum(hs, float(hs.size))
                sc = fast_score(hsn)
                if np.isfinite(sc):
                    extra_candidate = hsn
                    if hsn.size == n and sc < best_score:
                        best_score = sc
                        best = hsn
        except Exception:
            pass

        if best is None:
            best = make_base(n, "ramp", rng)
            best_score = fast_score(best)

        print(f"initial internal best={best_score:.6f}")

        # Main mirror-descent run from the best start.
        remaining = deadline - time.monotonic()
        if remaining > 60.0:
            t1 = min(max(60.0, 0.60 * remaining), 1800.0)
            res = run_mirror(best, t1, rng, "mirror-main")
            sc = fast_score(res)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = res
            print(f"after mirror-main internal best={best_score:.6f}")

        # A few mirror restarts from potentially useful basins.
        restart_kinds = ("ramp", "sqrt", "power07", "x_2x", "exp1", "tanh1", "quad_neg")
        for kind in restart_kinds:
            if time.monotonic() >= deadline - 90.0:
                break
            remaining = deadline - time.monotonic()
            talloc = min(max(20.0, 0.12 * remaining), 240.0)
            if talloc < 20.0:
                break
            try:
                a0 = make_base(n, kind, rng)
                sc0 = fast_score(a0)
                if not np.isfinite(sc0) or sc0 > best_score + 0.25:
                    a0 = make_block_perturbation(best, rng, 0.6)
                    sc0 = fast_score(a0)
                res = run_mirror(a0, talloc, rng, f"mirror-{kind}")
                sc = fast_score(res)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = res
                    print(f"restart {kind}: internal best={best_score:.6f}")
            except Exception:
                pass

        # Short simulated-annealing phase.
        if time.monotonic() < deadline - 60.0:
            remaining = deadline - time.monotonic()
            talloc = min(max(20.0, 0.08 * remaining), 180.0)
            if talloc >= 20.0:
                try:
                    res = random_local_search(best, talloc, rng)
                    sc = fast_score(res)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best = res
                        print(f"after random-search internal best={best_score:.6f}")
                except Exception:
                    pass

        # Coordinate refinement in the remaining time, leaving a small reserve for final checks.
        coord_deadline = deadline - 25.0
        if coord_deadline > time.monotonic() + 20.0:
            try:
                best = normalize_sum(best, float(best.size))
                c = fft_full(best)
                best = coordinate_refine(best, c, coord_deadline, rng)
                sc = fast_score(best)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                print(f"after coordinate internal best={best_score:.6f}")
            except Exception:
                pass

        # Build final candidate list and score it with the official evaluator.
        candidates = []
        candidates.append(best)
        for sigma in (0.02, 0.05, 0.10):
            try:
                candidates.append(make_block_perturbation(best, rng, sigma))
            except Exception:
                pass
        try:
            candidates.append(make_base(n, "ramp", rng))
        except Exception:
            pass
        if extra_candidate is not None:
            candidates.append(extra_candidate)

        best_list = [float(x) for x in best]
        best_exact = np.inf

        try:
            seen = 0
            for cand in candidates:
                if seen >= 8:
                    break
                cand = np.asarray(cand, dtype=np.float64)
                if cand.size == 0:
                    continue
                cand = np.where(np.isfinite(cand), cand, 0.0)
                cand = np.clip(cand, 0.0, MAX_VALUE)
                if float(cand.sum()) < 0.01:
                    continue
                seen += 1
                try:
                    sc = evaluate_sequence([float(x) for x in cand])  # noqa: F821
                    if np.isfinite(sc) and sc < best_exact:
                        best_exact = float(sc)
                        best_list = [float(x) for x in cand]
                except Exception:
                    pass
        except Exception:
            pass

        # Final sanitization.
        arr = np.asarray(best_list, dtype=np.float64)
        if arr.size == 0:
            arr = np.ones(512, dtype=np.float64)
        else:
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, MAX_VALUE)
            if float(arr.sum()) < 0.01:
                arr = np.ones(512, dtype=np.float64)

        return [float(x) for x in arr]

    except Exception:
        # Safe fallback: linear nonnegative sequence.
        n = 5582
        a = np.linspace(0.0, 2.0, n, dtype=np.float64)
        return [float(x) for x in a]
