import numpy as np
import time

def propose_candidate(seed=42, budget_s=2400, **kwargs):

    reducing the peak. The score approaches 2*(p!)^2*(p+1)^2/(2p+1)! for large n.
    Performs parameter sweep and local search refinement.
    """

    best_score = np.inf
    best_seq = None
    
    # Phase 1: Parameter sweep over (p, n)
    for n in [50, 100, 200, 300, 500, 800, 1200, 2000, 3000, 5000]:
        for p in [1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0]:
            # a[i] = ((i+1)/n)^p * 1000

            score = evaluate_sequence(seq)

    # Phase 2: Local search refinement
    if best_seq is not None and time.monotonic() < deadline:
        a = np.array(best_seq, dtype=np.float64)
        n = len(a)
        rng = np.random.default_rng(seed + 1000)
        
        def fast_eval(arr):
            """Evaluate using numpy for speed."""
            n_local = len(arr)
            S = arr.sum()
            if S < 0.01:
                return np.inf
            b = np.convolve(arr, arr)
            max_b = b.max()
            return 2.0 * n_local * max_b / (S * S)
        
        current_score = fast_eval(a)
        best_local = a.copy()
        best_local_score = current_score
        
        # Try random coordinate perturbations
        for iteration in range(50000):
            if time.monotonic() > deadline:
                break
            if iteration % 5000 == 0 and iteration > 0:
                print(f"  Local search iter {iteration}: score={best_local_score:.6f}")
            
            # Pick a random coordinate
            i = rng.integers(n)
            
            # Try to reduce this element (reduces tail contribution)

            if rng.random() < 0.7:

    return [float(x) for x in best_seq]
