    c = np.convolve(f, f)
    k_star = np.argmax(c)
    M = c[k_star]
    S = np.sum(f)
    return c, k_star, M, S

def solve_lp_step(f, n_constraints=200, M=None):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for top constraints."""
    if M is None:
        c = np.convolve(f, f)
        M = np.max(c)
    
    n = len(f)

    c = np.convolve(f, f)
    n_top = min(n_constraints, len(c))
    top_idx = np.argsort(c)[-n_top:]
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    data = []

        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            data.append(f[k - j])

    return None

"""
LP-based local search with multiple restarts for minimizing the
autocorrelation peak-to-power ratio of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft
import time


def compute_convolution(f):
    """Compute full convolution using FFT for speed."""
    n = len(f)
    size = 2 * n - 1
    # Use FFT-based convolution
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = fft(f, fft_size)
    c = np.real(ifft(F * F, fft_size))[:size]
    return c


def solve_lp(f, n_constraints, M):
    """Solve the LP to find g that maximizes sum(g) subject to constraints."""
    n = len(f)
    
    # Find top constraint indices
    c = compute_convolution(f)
    top_idx = np.argsort(c)[-n_constraints:]
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    data = []
    
    for idx, k in enumerate(top_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        vals = f[k - js]
        rows.extend(np.full(len(js), idx).tolist())
        cols.extend(js.tolist())
        data.extend(vals.tolist())
    
    if len(data) == 0:
        return None
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(n_constraints, n))
    b_ub = np.full(n_constraints, M)
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'maxiter': 10000})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def random_perturbation(f, rng, strength=0.3):
    """Apply a random multiplicative perturbation."""
    n = len(f)
    candidate = f.copy()
    
    # Block perturbation
    n_blocks = rng.integers(1, 5)

        start = rng.integers(0, n - width + 1)
        candidate[start:start+width] *= np.exp(rng.normal(0, strength))
    
    # Optional: add small symmetric component
    if rng.random() < 0.3:
        candidate = 0.5 * (candidate + candidate[::-1])
    
    return np.maximum(candidate, 0)


def simulated_annealing(f, best_score, deadline, rng, verbose=False):
    """Run simulated annealing to escape local optima."""
    n = len(f)
    current = f.copy()
    current_score = best_score
    best = f.copy()
    T = 0.1  # temperature
    
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        
        candidate = random_perturbation(current, rng, strength=0.2 + T)
        score = evaluate_sequence(candidate.tolist())
        
        if not np.isfinite(score):
            continue
        
        if score < current_score:
            current = candidate
            current_score = score
            if score < best_score:
                best = candidate
                best_score = score
                if verbose:
                    print(f"  SA iter {it}: new best={best_score:.6f}")
        else:
            # Accept with probability exp(-(score - current_score) / T)
            if rng.random() < np.exp(-(score - current_score) / T):
                current = candidate
                current_score = score
        
        T = max(0.001, T * 0.9999)
    
    return best, best_score

def compute_convolution(f):

def solve_lp(f, n_constraints, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for top constraints."""
    n = len(f)
    c = compute_convolution(f)
    n_cons = min(n_constraints, len(c))

        rows.extend([idx] * len(js))
