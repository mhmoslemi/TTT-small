import numpy as np
import time

def compute_objective(a):
    """Compute 2*n*max(conv)/(sum(a))^2"""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (a.sum()**2)

def compute_gradient_logsumexp(a, T):
    """Compute gradient of logsumexp(T*convolve(a,a)) w.r.t. a"""
    n = len(a)
    b = np.convolve(a, a)
    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    p = np.exp(Tb_shifted)
    p /= p.sum()
    # grad[m] = 2 * sum_j p[m+j] * a[j]
    # Use FFT
    N = 1 << (3*n).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(P * np.conj(A), N)
    grad = 2.0 * R[:n]
    return grad, b

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

This is more targeted than pure gradient descent.
