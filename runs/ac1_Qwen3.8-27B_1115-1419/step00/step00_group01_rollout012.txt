"""
Optimality-based construction for the stated nonnegative-sequence evaluator.

For any nonnegative finite sequence a = (a_0, ..., a_{n-1}),
the linear convolution with itself has zero-lag value

    b_0 = sum_i a_i^2.

Hence

    score = 2n * max(b) / (sum a)^2
          >= 2n * sum(a_i^2) / (sum a)^2
          >= 2

by Cauchy-Schwarz.  Equality holds if and only if all entries are equal.
Since the evaluator clips negative values to zero, the admissible optimum is
therefore exactly a constant nonnegative sequence.  This program returns such
a sequence, using a modest length that keeps the evaluator fast.
"""

def _target_length(default=6121, maximum=6121):
    """Choose a safe nonzero output length, preferring a previous length if small."""
    try:
        seq = height_sequence_1
    except Exception:
        return default

    if isinstance(seq, list):
        length = len(seq)
        if 2 <= length <= maximum:
            return int(length)

    return int(default)

    """Return a constant nonnegative sequence of the requested length."""
    if length < 1:

    return [float(height)] * int(length)
