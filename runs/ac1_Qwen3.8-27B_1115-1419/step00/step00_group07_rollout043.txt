```python
import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the algorithm from "Euclidean projection on the simplex"
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_convolution(a, N_fft):
    """Compute linear convolution of a with itself using FFT."""
    A = np.fft.rfft(a, N_fft)
    c = np.fft.irfft(A * A, N_fft)
    return c[:2*len(a)-1]

def get_gradient(a, k_star, n):
    """Compute gradient of c[k_star] with respect to a."""
    # dc[k_star]/da[i] = 2 * a[k_star - i] if 0 <= k_star - i < n else 0
    idx = k_star - np.arange(n)
    grad = np.zeros(n)
    valid = (idx >= 0) & (idx < n)
    grad[valid] = 2 * a[idx[valid]]
    return grad

def optimize(n, max_time=1800, seed=42, initial=None):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    N_fft = 1 << (2*n - 1).bit_length()
    
    if initial is not None:
        a = initial.astype(np.float64).copy()
    else:
        a = rng.random(n) + 0.01
    a = a / a.sum()
    
    best_a = a.copy()
    best_max_c = compute_convolution(a, N_fft).max()
    
    # Gradient descent with active set
    lr = 1.0  # will be adaptive
    alpha = 0.5  # for active set threshold
    
    for t in range(1000000):
        if time.time() - start_time > max_time:
            break
        
        c = compute_convolution(a, N_fft)
        max_c = c.max()
        
        # Active set: peaks above threshold
        threshold = max_c * alpha  # or max_c - some_offset
        active = np.nonzero(c >= threshold)[0]
        
        # Sum gradients over active set
        grad = np.zeros(n)
        for k in active:
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += a[idx[valid]]
        grad *= 2
        
        # Step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        # Verify improvement
        c_new = compute_convolution(a_new, N_fft)
        max_c_new = c_new.max()
        
        if max_c_new < best_max_c:
            best_max_c = max_c_new
            best_a = a_new.copy()
        
        # Adaptive learning rate
        if max_c_new <= max_c:
            # Good step, maybe increase lr slightly
            lr = min(lr * 1.05, 10.0)
            a = a_new
        else:
            # Bad step, decrease lr and try again
            lr = lr * 0.7
        
    return best_a, best_max_c
```

```python
def project_simplex(v):

    n = len(v)
    if n == 0:
```

    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""

    if n == 0:

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fast_convolve_self(a, N_fft):
    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    A = np.fft.rfft(a, N_fft)
    c = np.fft.irfft(A * A, N_fft)
    return c[:2 * n - 1]


def compute_smoothed_gradient(a, c, beta, N_fft):
    """Compute gradient of smoothed max (log-sum-exp) of c with respect to a.
    
    Returns grad[i] = 2 * sum_j a[j] * p[i+j] where p = softmax(beta * c).
    """
    # Numerically stable softmax
    beta_c = beta * (c - c.max())
    p = np.exp(beta_c)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = p / p_sum

    # grad[i] = 2 * sum_j a[j] * p[i+j] = 2 * R_{pa}[i]
    # R_{pa}[i] = IFFT(conj(FFT(p)) * FFT(a))[i]
    P_fft = np.fft.rfft(p, N_fft)
    A_fft = np.fft.rfft(a, N_fft)
    R = np.fft.irfft(np.conj(P_fft) * A_fft, N_fft)
    return 2.0 * R[:len(a)]


def optimize_smoothed(n, time_budget, rng, initial=None, verbose=False):
    """Optimize a sequence of length n to minimize max(conv(a,a)) on the simplex."""
    start_time = time.time()
    N_fft = 1 << (2 * n - 1).bit_length()

    # Initialization
    if initial is not None:
        a = np.array(initial, dtype=np.float64).copy()
    else:
        alpha_param = rng.uniform(0.3, 3.0)
        a = rng.dirichlet(np.full(n, alpha_param))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    c = fast_convolve_self(a, N_fft)
    best_max_c = c.max()

    lr = 0.5
    beta = 5.0
    min_beta = 1.0
    max_beta = 500.0
    patience = 0
    max_patience = 5000

    for t in range(5000000):
        if time.time() - start_time > time_budget:
            break

        # Compute convolution
        c = fast_convolve_self(a, N_fft)
        max_c = c.max()
        score = 2.0 * n * max_c

        if max_c < best_max_c:
            best_max_c = max_c
            best_a = a.copy()
            patience = 0
            if verbose and (t % 1000 == 0):
                print(f"    t={t}, max_c={max_c:.8f}, obj={score:.6f}, lr={lr:.4f}, beta={beta:.1f}")
        else:
            patience += 1

        if patience > max_patience:
            # Restart from a perturbed version of best
            a = best_a + 0.01 * rng.normal(0, 1, n)
            a = project_simplex(a)
            lr = 0.5
            beta = max(beta * 0.5, min_beta)
            patience = 0
            continue

        # Compute smoothed gradient
        grad = compute_smoothed_gradient(a, c, beta, N_fft)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step: scale by the natural scale of the problem
        a_std = max(np.std(a), 1.0 / n)
        step = lr * a_std * grad / grad_norm

        a_new = a - step
        a_new = project_simplex(a_new)

        # Verify improvement
        c_new = fast_convolve_self(a_new, N_fft)
        max_c_new = c_new.max()

        if max_c_new <= max_c * (1 + 1e-12):
            a = a_new
            # Increase lr slightly on success
            if max_c_new < max_c * 0.999:
                lr = min(lr * 1.01, 5.0)
                # Gradually increase beta for sharper focus
                if t % 200 == 0:
                    beta = min(beta * 1.02, max_beta)
        else:
            # Decrease lr on failure
            lr = max(lr * 0.85, 0.01)
            # Occasionally reduce beta to explore more broadly
            if t % 500 == 0:
                beta = max(beta * 0.8, min_beta)

    return best_a, best_max_c


def local_refine(a, n, time_budget, rng, N_fft=None):
    """Local refinement: small random perturbations, accept only improvements."""
    if N_fft is None:
        N_fft = 1 << (2 * n - 1).bit_length()

    start_time = time.time()
    c = fast_convolve_self(a, N_fft)
    best_max_c = c.max()
    best_a = a.copy()

    for t in range(1000000):
        if time.time() - start_time > time_budget:
            break

        # Generate a small perturbation
        perturbation_type = rng.random()

        candidate = best_a.copy()
        if perturbation_type < 0.4:
            # Block perturbation
            block_size = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            scale = np.exp(rng.normal(0, 0.1, end - start))
            candidate[start:end] *= scale
        elif perturbation_type < 0.7:
            # Pair swap with scaling
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                mix = rng.uniform(0.1, 0.9)
                old_i, old_j = candidate[i], candidate[j]
                candidate[i] = mix * old_i + (1 - mix) * old_j
                candidate[j] = (1 - mix) * old_i + mix * old_j
        else:
            # Gaussian perturbation on subset
            n_pert = int(rng.integers(5, max(6, n // 50)))
            indices = rng.choice(n, size=n_pert, replace=False)
            candidate[indices] += rng.normal(0, 0.01 / n, n_pert)

        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s

        c_cand = fast_convolve_self(candidate, N_fft)
        max_c_cand = c_cand.max()

        if max_c_cand < best_max_c:
            best_max_c = max_c_cand
            best_a = candidate

    return best_a, best_max_c
