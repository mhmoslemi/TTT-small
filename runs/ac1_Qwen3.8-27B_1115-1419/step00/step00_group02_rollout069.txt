def _fast_conv(a):
    n = int(a.size)
    if n <=0: return np.array([], dtype=float)
    conv_len = 2*n -1
    size = 1 << (conv_len -1).bit_length()
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    out = out[:conv_len]
    # irfft may return length size if? yes.
    # Clamp small negative
    # To avoid allocation? Use np.maximum(out, 0.0, out=out) maybe.
    np.maximum(out, 0.0, out=out)
    return out

def _score_from_conv(a, c=None):
    n = int(a.size)
    if n <=0: return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-8:
        return np.inf
    if c is None:
        c = _fast_conv(a)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float((2.0 * n * m) / (s * s))

def _top_weights(c, k):

def _active_gradient(a, idx, w):
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    # iterate over index arrays
    for j in range(idx.size):
        k = int(idx[j])
        wk = float(w[j])
        if wk == 0.0:
            continue
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo > hi:
            continue
        start = k - hi
        stop = k - lo
        # stop < n, start >=0
        grad[lo:hi + 1] += (2.0 * wk) * a[start:stop + 1][::-1]
    return grad

"""
Search for a non-negative coefficient sequence minimizing
    2 * n * max(autoconvolution) / sum(sequence)^2.

Strategy:
- The objective is scale invariant, so all local-search iterates are kept on
  the probability simplex (nonnegative, sum 1).
- Strong initial shapes are ramp-like: for large n, an increasing ramp has
  asymptotic value 4/3, already far below the current upper bound.
- A projected-gradient method is used on a smoothed/active-set maximizer of
  the autoconvolution.  At each step, the most active autocorrelation bins are
  selected, their subgradient is averaged, a normalized step is taken, and the
  result is projected back to the simplex.
- Periodic diversification/perturbation avoids stalling, and the best
  sequence seen is always retained and returned after the time budget.
"""

import numpy as np
import time


def _choose_length():
    """Choose a working sequence length, preferring the provided construction."""
    n = 8192
    try:
        hs = height_sequence_1
        n = int(len(hs))
    except Exception:
        n = 8192
    # Keep the final direct evaluator comfortably fast.
    if n < 1024 or n > 12000:
        n = 8192
    return n


def _simplex_project(x):
    """Project a vector onto the probability simplex {v >= 0, sum(v)=1}."""
    x = np.asarray(x, dtype=np.float64)
    n = int(x.size)
    if n <= 0:
        return np.array([], dtype=np.float64)

    if not np.all(np.isfinite(x)):
        return np.full(n, 1.0 / n, dtype=np.float64)

    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    vals = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * vals) > (cssv - 1.0)
    idx = np.flatnonzero(cond)

    if idx.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = int(idx[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(x - theta, 0.0)

    s = float(np.sum(out))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)

    if abs(s - 1.0) > 1e-12:
        out = out / s
        out = np.maximum(out, 0.0)
        s2 = float(np.sum(out))
        if not np.isfinite(s2) or s2 <= 1e-300:
            return np.full(n, 1.0 / n, dtype=np.float64)
        out = out / s2

    return out


def _fast_conv(a):
    """Linear autoconvolution via FFT, with tiny numerical negatives clamped."""
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    conv_len = 2 * n - 1
    size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)[:conv_len]
    np.maximum(out, 0.0, out=out)
    return out


def _score_from_conv(a, c=None):
    """Evaluate the invariant objective from an autoconvolution array."""
    n = int(a.size)
    if n <= 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-8:
        return np.inf
    if c is None:
        c = _fast_conv(a)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float((2.0 * n * m) / (s * s))


def _top_weights(c, k):
    """
    Return indices of the largest k autoconvolution values and smooth weights.

    The weights are concentrated near the maximum, but not only at one bin,
    which makes the descent more stable.
    """
    m = int(c.size)
    if m <= 0:
        return np.array([], dtype=np.int64), np.array([], dtype=np.float64)

    k = int(max(1, min(k, m)))
    if k >= m:
        idx = np.arange(m, dtype=np.int64)
        vals = c
    else:
        idx = np.argpartition(c, m - k)[m - k:].astype(np.int64)
        vals = c[idx]

    maxv = float(np.max(vals)) if vals.size else 0.0
    if not np.isfinite(maxv) or maxv <= 0.0:
        w = np.ones(idx.size, dtype=np.float64)
        if w.size == 0:
            return idx, w
        w /= w.size
        return idx, w

    scale = 0.02 * maxv
    if not np.isfinite(scale) or scale <= 0.0:
        scale = maxv

    w = np.exp((vals - maxv) / scale)
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.ones(idx.size, dtype=np.float64)
        sw = float(w.size)
    w = w / sw
    return idx, w


def _active_gradient(a, idx, weights):
    """
    Sum of weighted exact subgradients of the selected autoconvolution bins.

    For a bin k, d(a*a)_k / da_i = 2 * a_{k-i} whenever k-i is a valid index.
    """
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)

    for j in range(idx.size):
        k = int(idx[j])
        wk = float(weights[j])
        if wk == 0.0:
            continue

        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo > hi:
            continue

        start = k - hi
        stop = k - lo
        grad[lo:hi + 1] += (2.0 * wk) * a[start:stop + 1][::-1]

    return grad


def _optimize_sequence(initial, deadline, rng, tag, k_bins=24):
    """
    Local projected-gradient search from an initial simplex point.

    Returns (best_array, best_score).  The best array is always a valid
    nonnegative, sum-1 sequence.
    """
    n = int(initial.size)
    if n <= 0:
        return np.array([], dtype=np.float64), np.inf

    current = _simplex_project(np.asarray(initial, dtype=np.float64))
    c = _fast_conv(current)
    current_score = _score_from_conv(current, c)
    if not np.isfinite(current_score):
        return current, current_score

    best = current.copy()
    best_score = float(current_score)

    t = 0.08
    no_improve = 0
    step = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        c = _fast_conv(current)
        cscore = _score_from_conv(current, c)
        if np.isfinite(cscore) and cscore < best_score:
            best_score = float(cscore)
            best = current.copy()

        maxc = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        idx, weights = _top_weights(c, k_bins)
        if idx.size == 0:
            break

        grad = _active_gradient(current, idx, weights)
        gmax = float(np.max(grad))
        if not np.isfinite(gmax) or gmax <= 1e-300:
            break

        amean = float(np.mean(current))
        if not np.isfinite(amean) or amean <= 0.0:
            break

        # Normalize the gradient so that the largest component is about
        # t * mean(a).  This gives a stable, dimensionless step.
        step_vec = grad * (amean / gmax)
        new = current - t * step_vec
        new = _simplex_project(new)

        c_new = _fast_conv(new)
        new_score = _score_from_conv(new, c_new)

        if np.isfinite(new_score):
            if new_score < current_score:
                current = new
                current_score = float(new_score)
                if new_score < best_score:
                    best_score = float(new_score)
                    best = new.copy()
                    no_improve = 0
                    t = min(0.18, t * 1.03)
                else:
                    no_improve += 1
            else:
                t = max(0.008, t * 0.86)
                no_improve += 1
        else:
            t = max(0.008, t * 0.5)
            no_improve += 50

        step += 1

        if no_improve >= 120:
            # Restart in a mildly perturbed neighborhood of the best point.
            noise_scale = 1.0 / (float(n) * 6.0)
            current = best.copy()

            n_blocks = 10
            width = max(1, n // n_blocks)
            for b in range(n_blocks):
                s0 = b * width
                s1 = min(n, s0 + width)
                if s0 < s1:
                    current[s0:s1] *= float(np.exp(rng.normal(0.0, 0.08)))

            current = current + rng.normal(0.0, noise_scale, n)
            current = _simplex_project(current)

            c_cur = _fast_conv(current)
            current_score = _score_from_conv(current, c_cur)
            if not np.isfinite(current_score) or current_score > 2.0:
                current = best.copy()
                current_score = best_score
            else:
                if current_score < best_score:
                    best = current.copy()
                    best_score = float(current_score)

            no_improve = 0
            t = 0.05

        if step % 200 == 0:
            now = time.monotonic()
            if now - last_print >= 2.0:
                print(
                    tag,
                    "best=%.10f cur=%.10f t=%.4f step=%d"
                    % (best_score, current_score, t, step),
                    flush=True
                )
                last_print = now

    c = _fast_conv(current)
    cscore = _score_from_conv(current, c)
    if np.isfinite(cscore) and cscore < best_score:
        best = current.copy()
        best_score = float(cscore)

    return best, best_score


def _add_initial(inits, arr):
    """Validate, nonnegativize, normalize and append an initial candidate."""
    if arr is None:
        return
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if np.isfinite(s) and s > 0.0:
        inits.append(a / s)


def _make_initials(n, rng):
    """Build several strong ramp-like and interpolating initial guesses."""
    inits = []
    ar = np.arange(n, dtype=np.float64)

    # Basic increasing ramp.  This alone already gives an asymptotic score
    # near 4/3, much below the target.
    _add_initial(inits, ar + 1.0)
    _add_initial(inits, ar)

    base = ar + 1.0
    for p in (0.8, 0.9, 0.95, 1.0, 1.05, 1.1, 1.2):
        _add_initial(inits, np.power(base, p))

    # Slightly superlinear endpoint emphasis.
    denom = float(max(1, n - 1))
    _add_initial(inits, base * (1.0 + 0.25 * ar / denom))
    _add_initial(inits, base * (1.0 - 0.15 * ar / denom))

    # Concave and convex variants.
    _add_initial(inits, np.sqrt(base))
    _add_initial(inits, base * base)

    # If a previously found construction exists, include and resample it.
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if hs.size == n:
            _add_initial(inits, hs)
        elif hs.size > 1:
            x_old = np.linspace(0.0, 1.0, hs.size)
            x_new = np.linspace(0.0, 1.0, n)
            y = np.interp(x_new, x_old, hs)
            _add_initial(inits, y)
    except Exception:
        pass

    # A few smooth random monotone-ish interpolants.
    for _ in range(2):
        nknots = int(5 + rng.integers(0, 6))
        knots_x = np.sort(rng.random(nknots))
        knots_x = np.unique(knots_x)
        if knots_x.size < 2:
            continue
        knots_y = rng.random(knots_x.size)
        knots_y = knots_y + 0.5 * knots_x
        y = np.interp(np.linspace(0.0, 1.0, n), knots_x, knots_y)
        _add_initial(inits, y)

    if not inits:
        inits.append(np.full(n, 1.0 / n, dtype=np.float64))

    return inits


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Main entry point.  Searches within budget_s seconds and returns a list of
    nonnegative floats with the best evaluation found.
    """
    if "budget_s" in kwargs:
        try:
            budget_s = float(kwargs["budget_s"])
        except Exception:
            pass

    try:
        budget = max(60.0, float(budget_s))
    except Exception:
        budget = 2400.0

    start = time.monotonic()
    deadline = start + max(60.0, budget - 20.0)
    n = _choose_length()
    rng = np.random.default_rng(int(seed))

    inits = _make_initials(n, rng)

    # Fast-evaluate all initial candidates and choose a few distinct basins.
    scores = []
    for a in inits:
        c = _fast_conv(a)
        scores.append(_score_from_conv(a, c))

    order = np.argsort(scores)
    chosen = []
    chosen_scores = []
    for oi in order:
        idx = int(oi)
        sc = float(scores[idx])
        if not np.isfinite(sc):
            continue
        same = False
        for x in chosen_scores:
            if abs(sc - x) < 1e-9:
                same = True
                break
        if not same:
            chosen.append(idx)
            chosen_scores.append(sc)
        if len(chosen) >= 4:
            break

    if not chosen:
        ramp = (np.arange(n, dtype=np.float64) + 1.0)
        ramp /= ramp.sum()
        inits = [ramp]
        chosen = [0]
        chosen_scores = [1.0]

    best = inits[chosen[0]].copy()
    best_score = float(chosen_scores[0])
    print("initial fast best = %.10f, n = %d" % (best_score, n), flush=True)

    limit = deadline - 15.0

    # Main run from the strongest initial point.
    now = time.monotonic()
    if now < limit:
        main_deadline = now + max(60.0, (limit - now) * 0.65)
        try:
            best, best_score = _optimize_sequence(
                best, main_deadline, rng, "main", k_bins=24
            )
            print("after main   = %.10f" % best_score, flush=True)
        except Exception as exc:
            print("main optimization failed:", exc, flush=True)

    # Optional exploration from a couple of other strong initial points.
    alt_fracs = (0.35, 0.18, 0.08)
    for pos, alt_idx in enumerate(chosen[1:]):
        now = time.monotonic()
        if now >= limit - 5.0:
            break
        alt_fast = float(scores[alt_idx])
        if not np.isfinite(alt_fast):
            continue
        # Only spend time on alternative basins if they are competitively good.
        if alt_fast > best_score * 1.05 + 1e-7:
            continue
        frac = alt_fracs[pos] if pos < len(alt_fracs) else 0.05
        alt_deadline = now + max(20.0, (limit - now) * frac)
        try:
            alt, alt_score = _optimize_sequence(
                inits[alt_idx], alt_deadline, rng, "alt%d" % pos, k_bins=24
            )
            if np.isfinite(alt_score) and alt_score < best_score:
                best = alt.copy()
                best_score = float(alt_score)
                print("alt improved = %.10f" % best_score, flush=True)
        except Exception as exc:
            print("alt optimization failed:", exc, flush=True)

    # Final polish from the global best.
    now = time.monotonic()
    if now < deadline - 5.0:
        try:
            best, best_score = _optimize_sequence(
                best, deadline - 5.0, rng, "polish", k_bins=16
            )
            print("after polish = %.10f" % best_score, flush=True)
        except Exception as exc:
            print("polish optimization failed:", exc, flush=True)

    final = _simplex_project(np.asarray(best, dtype=np.float64))
    if not np.all(np.isfinite(final)) or final.size != n or float(np.sum(final)) < 0.01:
        ramp = (np.arange(n, dtype=np.float64) + 1.0)
        ramp /= ramp.sum()
        final = ramp

    candidate = [float(x) for x in final]

    # Official check.  If something unexpected happens, fall back to the ramp.
    try:
        official = float(evaluate_sequence(candidate))
        print("official score = %.10f" % official, flush=True)
        if not np.isfinite(official) or official > 1.45:
            ramp = (np.arange(n, dtype=np.float64) + 1.0)
            ramp /= ramp.sum()
            ramp_candidate = [float(x) for x in ramp]
            try:
                ramp_official = float(evaluate_sequence(ramp_candidate))
                print("ramp official score = %.10f" % ramp_official, flush=True)
                if np.isfinite(ramp_official) and (
                    not np.isfinite(official) or ramp_official < official
                ):
                    return ramp_candidate
            except Exception:
                pass
    except Exception as exc:
        print("official evaluation failed:", exc, flush=True)
        ramp = (np.arange(n, dtype=np.float64) + 1.0)
        ramp /= ramp.sum()
        return [float(x) for x in ramp]

    return candidate
