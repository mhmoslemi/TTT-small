def project_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(np.arange(1, n+1) * (u - cssv / np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)

    n = len(v)

"""LP-based reduction with subgradient refinement to minimize the self-convolution peak score."""

import numpy as np
import time
from scipy.sparse import csr_matrix
import cvxpy as cp


def project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / idx > 0
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute the evaluation score without validation overhead."""
    n = len(a)
    sa = a.sum()
    if sa < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (sa * sa)


def build_constraint_matrix(f, top_idx):
    """Build sparse matrix A where A[ii, j] = f[top_idx[ii] - j] if valid."""
    n = len(f)
    rows = []
    cols = []
    data = []
    for ii, k in enumerate(top_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            val = f[k - j]
            if val > 1e-14:
                rows.append(ii)
                cols.append(j)
                data.append(val)
    if not data:
        return None
    return csr_matrix((data, (rows, cols)), shape=(len(top_idx), n))


def solve_reduction_lp(f, M, top_k=200):
    """Solve max sum(g) s.t. g>=0, A@g<=M, sum(g)<=2n. Returns normalized g or None."""
    n = len(f)
    conv_ff = np.convolve(f, f)
    top_idx = np.argsort(conv_ff)[::-1][:top_k]
    
    A = build_constraint_matrix(f, top_idx)
    if A is None:
        return None
    
    g_var = cp.Variable(n, nonneg=True)
    constraints = [A @ g_var <= M, cp.sum(g_var) <= 2.0 * n]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    g_val = None
    for sn in ('CLARABEL', 'ECOS', 'SCS', 'PDLP', 'GLPK'):
        try:
            prob.solve(solver=getattr(cp, sn), verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                if g_var.value is not None:
                    cand = np.maximum(0.0, np.asarray(g_var.value, dtype=np.float64))
                    if cand.sum() > 1e-10:
                        g_val = cand
                        break
        except Exception:
            continue
    
    if g_val is None:
        return None
    return g_val / g_val.sum()


def do_subgradient_step(f, active_idx, eta):
    """Perform one subgradient step on the max-convolution objective."""
    n = len(f)
    grad = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        j_arr = np.arange(j_lo, j_hi + 1)
        grad[j_arr] += f[k - j_arr]
    if len(active_idx) > 0:
        grad *= 2.0 / len(active_idx)
    new_f = f - eta * grad
    return project_simplex(new_f, 1.0)

    # top_idx[ii] - j is in [0, n-1]
    # A[top_idx[ii] - j, j]... no, it's A[ii, j] = f[top_idx[ii] - j]

    # j ranges from max(0, k-n+1) to min(k, n-1)

    # where i is an index in f and j is an index in g, the contribution
    # goes to row (k = i+j) in the full conv, but we only include rows in top_idx
    
    # Actually, the most efficient way for a sparse matrix:
    # For each ii, k = top_idx[ii], the row has nonzeros at j = max(0,k-n+1)..min(k,n-1)
    # with values f[k-j] = f[max(0,k-n+1)..min(k,n-1)] reversed...

    count = 0
    for ii, k in enumerate(top_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        length = j_hi - j_lo + 1
        rows[count:count+length] = ii
        cols[count:count+length] = np.arange(j_lo, j_hi + 1)
        data[count:count+length] = f[k - j_hi:k - j_lo + 1][::-1]
        # Wait: f[k-j] for j = j_lo..j_hi

        # So the values are f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi]
        # Which is f[j_hi' : j_lo'+1][::-1] where j_hi' = k-j_lo, j_lo' = k-j_hi
        count += length
    
    if count == 0:
        return None
    return csr_matrix((data[:count], (rows[:count], cols[:count])), shape=(len(top_idx), n))

"""
LP-based reduction with subgradient refinement to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses iterative LP to find peak-reducing directions, then subgradient polishing.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
import cvxpy as cp


def project_simplex(v, target=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    k_arr = np.arange(n, dtype=np.float64)
    cond = (k_arr + 1.0) * u > cssv - target
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 without validation overhead."""
    n = len(a)
    sa = a.sum()
    if sa < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (sa * sa)


def build_constraint_matrix(f, top_idx):
    """Build sparse A where A[ii,j] = f[top_idx[ii]-j] for valid indices."""
    n = len(f)
    rows = []
    cols = []
    data = []
    for ii, k in enumerate(top_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi < j_lo:
            continue
        j_arr = np.arange(j_lo, j_hi + 1)
        vals = f[k - j_arr]
        mask = vals > 1e-14
        if mask.any():
            rows.extend([ii] * int(mask.sum()))
            cols.extend(j_arr[mask].tolist())
            data.extend(vals[mask].tolist())
    if not data:
        return None
    return csr_matrix((np.array(data), (np.array(rows), np.array(cols))),
                      shape=(len(top_idx), n))


def solve_reduction_lp(f, M, top_k=200):
    """Solve max sum(g) s.t. g>=0, conv(f,g)[top]<=M, sum(g)<=2n. Returns normalized g or None."""
    n = len(f)
    conv_ff = np.convolve(f, f)
    top_idx = np.argsort(conv_ff)[::-1][:top_k]

    A = build_constraint_matrix(f, top_idx)
    if A is None:
        return None

    g_var = cp.Variable(n, nonneg=True)
    constraints = [A @ g_var <= M, cp.sum(g_var) <= 2.0 * n]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    g_val = None
    for sn in ('CLARABEL', 'ECOS', 'SCS', 'PDLP', 'GLPK'):
        try:
            prob.solve(solver=getattr(cp, sn), verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                if g_var.value is not None:
                    cand = np.maximum(0.0, np.asarray(g_var.value, dtype=np.float64))
                    if cand.sum() > 1e-10:
                        g_val = cand
                        break
        except Exception:
            continue

    if g_val is None:
        return None
    return g_val / g_val.sum()


def do_subgrad(f, active_idx, eta):
    """One subgradient step: project f - eta * grad onto simplex."""
    n = len(f)
    grad = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        j_arr = np.arange(j_lo, j_hi + 1)
        grad[j_arr] += f[k - j_arr]
    if len(active_idx) > 0:
        grad *= 2.0 / len(active_idx)
    return project_simplex(f - eta * grad, 1.0)


def try_convex_mix(f, g, best_score):
    """Try convex combinations (1-t)f + t*g, return best candidate and score or None."""
    best_cand = None
    best_sc = best_score
    for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0):
        cand = (1.0 - t) * f + t * g
        cs = cand.sum()
        if cs < 1e-10:
            continue
        cand = cand / cs
        sc = fast_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_cand = cand
    return best_cand, best_sc


def subgrad_phase(f, max_iters, base_eta, deadline):
    """Run subgradient iterations, return improved sequence and score."""
    best = f.copy()
    conv_cur = np.convolve(best, best)
    Mc = conv_cur.max()
    active = np.where(conv_cur >= Mc * 0.999)[0]
    best_sc = fast_score(best)
    improved_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        eta = base_eta / (1.0 + 0.05 * it)
        nf = do_subgrad(best, active, eta)
        sc = fast_score(nf)
        if sc < best_sc - 1e-13:
            best = nf
            best_sc = sc
            improved_count += 1
            if improved_count % 100 == 0:
                cc = np.convolve(best, best)
                Mc2 = cc.max()
                active = np.where(cc >= Mc2 * 0.999)[0]
        else:
            break
    return best, best_sc, improved_count


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2n*max(conv(a,a))/(sum(a))^2 via iterative LP peak-reduction
    and subgradient polishing. Starts from height_sequence_1 and refines.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0

    # --- Initialize ---
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    if parent.sum() < 0.01:
        parent = np.ones(n)
    best = parent / parent.sum()
    best_score = fast_score(best)
    print(f"Init: n={n} score={best_score:.6f}")

    # Try alternative starting shapes
    x = np.linspace(0.0, 1.0, n)
    u_val = (x - 0.5) ** 2 + 0.05
    alt1 = u_val / u_val.sum()
    tent = np.minimum(np.arange(n, dtype=np.float64),
                      n - 1.0 - np.arange(n, dtype=np.float64)) + 1.0
    alt2 = (1.0 / tent)
    alt2 = alt2 / alt2.sum()
    for alt in (alt1, alt2):
        sc = fast_score(alt)
        if sc < best_score:
            best = alt.copy()
            best_score = sc
            print(f"Alt start improved: {best_score:.6f}")

    # --- Main LP loop ---
    top_k = 200
    stall = 0
    lp_count = 0

    while time.monotonic() < deadline:
        f = best.copy()
        s = f.sum()
        if s < 0.01:
            break
        f = f / s

        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        g = solve_reduction_lp(f, M, top_k=top_k)
        improved = False

        if g is not None:
            cand, sc = try_convex_mix(f, g, best_score)
            if cand is not None:
                best = cand
                best_score = sc
                improved = True
                lp_count += 1
                print(f"LP {lp_count}: score={best_score:.6f} top_k={top_k}")

        if not improved:
            stall += 1
            # Subgradient refinement
            nf, nsc, nimp = subgrad_phase(best, 3000, 0.005 * (1.0 + 0.05 * stall), deadline)
            if nimp > 0 and nsc < best_score:
                best = nf
                best_score = nsc
                stall = max(0, stall - 5)
                print(f"SG: score={best_score:.6f} steps={nimp}")

        if stall > 25 and rng.random() < 0.12:

            cand = np.maximum(best * noise, 0.0)

            if cs > 0.01:
                cand = cand / cs
                sc = fast_score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
                    stall = 0
                    print(f"Rand: score={best_score:.6f}")

            print(f"Grow top_k -> {top_k}")

            best_score = nsc
