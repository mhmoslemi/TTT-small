import numpy as np
import time
from scipy.fft import fft, ifft

def fast_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb)[:n].real

def project_simplex(v):
    """Project v onto the probability simplex {p >= 0, sum(p) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_score_fast(p):
    """Compute 2*n*max(p*p)/sum(p)^2 efficiently."""
    n = len(p)
    s = np.sum(p)
    if s < 1e-10:
        return np.inf
    c = fast_convolve(p, p)
    return 2.0 * n * np.max(c) / (s * s)
