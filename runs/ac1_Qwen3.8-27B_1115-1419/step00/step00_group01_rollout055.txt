def compute_score(a):

    conv = np.convolve(a, a)

    sum_a = np.sum(a)

"""
Optimization of non-negative step function to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based convolution, subgradient descent with multiplicative updates,
multiple diverse initializations, and adaptive learning rates.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute np.convolve(a, a) using FFT for O(n log n) speed."""
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    size = 1
    while size < 2 * n:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa)[:2 * n - 1]
    return np.maximum(result, 0.0)  # numerical safety


def compute_score(a):
    """Compute the objective: 2*n*max(conv)/(sum)^2"""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fast_autocorr(a)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S * S)


def generate_initial(n, seed, style=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    if style == 0:
        # Random with random gaps
        a = rng.random(n)
        n_gaps = int(n * rng.uniform(0.05, 0.3))
        gap_pos = rng.choice(n, size=n_gaps, replace=False)
        a[gap_pos] = 0.0
    elif style == 1:
        # Block structure: alternating high/low blocks
        block_len = max(1, n // int(rng.integers(4, 20)))
        a = np.zeros(n)
        for i in range(n):
            block_idx = i // block_len
            if rng.random() < 0.7:
                a[i] = rng.uniform(0.3, 1.0)
            else:
                a[i] = rng.uniform(0.0, 0.1)
    elif style == 2:
        # Skewed distribution: more mass on one side
        a = rng.exponential(1.0, n)
        a *= rng.uniform(0.5, 2.0)
        n_zeros = int(n * rng.uniform(0.0, 0.15))
        if n_zeros > 0:
            zero_pos = rng.choice(n, size=n_zeros, replace=False)
            a[zero_pos] = 0.0
    elif style == 3:
        # Sine-modulated
        t = np.arange(n) / n
        freq = rng.uniform(1.0, 5.0)
        a = np.abs(np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
        a += rng.random(n) * 0.1
        a = np.maximum(a, 0)
    elif style == 4:
        # Two-scale: coarse structure with fine noise
        coarse_len = max(1, n // 64)
        coarse = rng.random(coarse_len + 1)
        coarse[coarse < 0.2] = 0
        a = np.interp(np.arange(n), np.arange(len(coarse)), coarse)
        a *= (1.0 + 0.3 * rng.standard_normal(n))
        a = np.maximum(a, 0.0)
    else:
        a = rng.random(n)
    
    a = np.clip(a, 0.0, 1000.0)
    S = np.sum(a)
    if S > 0:
        a = a * (1.0 / S)  # normalize to sum=1 for consistency
    return a


def subgradient_step(a, conv, k_max, lr):
    """One subgradient step: reduce elements that contribute to max conv."""
    n = len(a)
    pressure = np.zeros(n)
    
    # For conv[k_max] = sum_i a[i]*a[k_max-i], the derivative w.r.t. a[i] is:
    # 2*a[k_max-i] if k_max-i != i, or a[i] if k_max-i == i
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    if lo > hi:
        return a
    
    idx = np.arange(lo, hi + 1)
    j_idx = k_max - idx
    pressure[idx] = 2.0 * a[j_idx]
    
    # Fix diagonal (i == k_max - i => i = k_max/2)
    if k_max % 2 == 0:
        mid = k_max // 2
        if lo <= mid <= hi:
            pressure[mid] = a[mid]
    
    p_max = pressure.max()
    if p_max > 1e-15:
        pressure /= p_max
    
    # Multiplicative update: reduce high-pressure, boost low-pressure
    update = np.exp(-lr * (pressure - 1.0))
    a_new = a * update
    a_new = np.maximum(a_new, 0.0)
    
    # Renormalize to keep sum fixed
    S_old = np.sum(a)
    S_new = np.sum(a_new)
    if S_new > 1e-15:
        a_new *= S_old / S_new
    else:
        a_new = a.copy()
    
    return a_new


def topk_pressure_step(a, conv, lr, topk_frac=0.1):
    """Step using weighted pressure from top-K convolution values."""
    n = len(a)
    conv_len = len(conv)
    K = max(1, int(conv_len * topk_frac))
    
    # Get top K indices
    top_k = np.argpartition(conv, -K)[-K:]
    top_vals = conv[top_k]
    # Weight by value (normalized)
    weights = top_vals / (top_vals.sum() + 1e-15)
    
    pressure = np.zeros(n)
    for k_pos in top_k:
        k_pos = int(k_pos)
        w = weights[list(top_k).index(k_pos)]
        lo = max(0, k_pos - (n - 1))
        hi = min(n - 1, k_pos)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        j_idx = k_pos - idx
        contrib = 2.0 * a[j_idx]
        if k_pos % 2 == 0:
            mid = k_pos // 2
            if lo <= mid <= hi:
                contrib[mid - lo] = a[mid]
        pressure[idx] += w * contrib
    
    p_max = pressure.max()
    if p_max > 1e-15:
        pressure /= p_max
    
    update = np.exp(-lr * (pressure - 1.0))
    a_new = a * update
    a_new = np.maximum(a_new, 0.0)
    
    S_old = np.sum(a)
    S_new = np.sum(a_new)
    if S_new > 1e-15:
        a_new *= S_old / S_new
    else:
        a_new = a.copy()
    
    return a_new


def local_search(a, time_budget, seed=0, verbose=True):
    """Main local search with adaptive learning rate and diversity moves."""
    n = len(a)
    a = a.copy().astype(np.float64)
    rng = np.random.default_rng(seed + 12345)
    
    best_a = a.copy()
    best_score = compute_score(a)
    start_time = time.monotonic()
    deadline = start_time + time_budget
    
    lr = 0.03
    stuck = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        conv = fast_autocorr(a)
        S = np.sum(a)
        if S < 0.01:
            break
        max_b = np.max(conv)
        score = 2.0 * n * max_b / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stuck = 0
            lr = min(lr * 1.02, 0.3)
        else:
            stuck += 1
            if stuck > 30:
                lr *= 0.85
        
        # Choose which k positions to target
        use_topk = (iter_count % 3 == 0)
        if use_topk:
            a = topk_pressure_step(a, conv, lr, topk_frac=0.05)
        else:
            k_max = int(np.argmax(conv))
            a = subgradient_step(a, conv, k_max, lr)
        
        # Diversity move: occasionally perturb
        if stuck > 80 or (iter_count % 200 == 0 and rng.random() < 0.3):
            if rng.random() < 0.5:
                # Block scaling
                block_size = max(1, n // rng.integers(10, 100))
                start = rng.integers(0, max(1, n - block_size))
                a[start:start+block_size] *= np.exp(rng.normal(0, 0.3))
                a = np.maximum(a, 0.0)
            else:
                # Random coordinate boost
                n_boost = max(1, n // 50)
                boost_idx = rng.choice(n, size=n_boost, replace=False)
                a[boost_idx] *= np.exp(rng.normal(0.5, 0.5, n_boost))
                a = np.maximum(a, 0.0)
            
            # Re-normalize
            S_new = np.sum(a)
            if S_new > 0.01:
                a = a * (np.sum(best_a) / S_new)
            
            stuck = 0
            lr = min(lr * 1.5, 0.4)
        
        # Safety: clip
        a = np.clip(a, 0.0, 1000.0)
        
        if lr < 1e-4:
            break
    
    return best_a, best_score
