def compute_autocorr(a):
    """Compute full convolution (autocorrelation) using FFT."""
    n = len(a)
    # Use FFT for O(n log n) convolution
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.irfft(A * A, fft_size)[:size]
    return B

def co_step(a, eta, n_hot=50):

    n = len(a)
    b = compute_autocorr(a)
    max_b = b.max()

    threshold = max_b * (1 - eta * 0.1)  # or use top-k

def fft_convolve(a):
    """Full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return result

def co_iterate(a, n_hot, eta, n_steps=1000):
    """Run CO iterations, return final sequence and best score."""
    n = len(a)
    best_local = a.copy()
    best_local_score = np.inf
    current_score = np.inf
    
    for step in range(n_steps):
        b = fft_convolve(a)
        max_b = b.max()
        score = 2 * n * max_b  # with sum_a = 1
        
        if score < best_local_score:
            best_local_score = score
            best_local = a.copy()
        
        # Find hot indices
        threshold = max_b * 0.98
        hot_mask = b >= threshold
        hot = np.where(hot_mask)[0]
        
        if len(hot) == 0:
            hot = np.array([int(np.argmax(b))])
        elif len(hot) > n_hot:
            # Take the top n_hot
            top_idx = np.argsort(b)[::-1][:n_hot]
            hot = top_idx
        
        # Compute gradient
        grad = np.zeros(n)
        for k in hot:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                # a[k-i] for i in [lo, hi]
                # i goes lo to hi, k-i goes k-lo down to k-hi
                grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        
        # Normalized gradient direction (sum-preserving)
        grad_mean = np.mean(grad)
        direction = grad - grad_mean
        norm = np.linalg.norm(direction)
        if norm > 1e-12:
            direction /= norm
            a_new = a - eta * direction

            a_new = a.copy()

        s = a_new.sum()
        if s < 1e-10:

            a_new /= s

Minimize 2*n*max(convolve(a,a))/sum(a)^2 using FFT-based Correlation Optimality (CO)
algorithm with momentum, adaptive step size, random restarts, and periodic verification.

import numpy as np
import time

def fft_full_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)

def evaluate_fast(a):
    """Fast evaluation: 2*n*max(conv)/sum^2 assuming a >= 0, sum > 0."""
    n = len(a)
    b = fft_full_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def co_main_loop(a, time_limit, seed=42):
    """Core CO algorithm loop."""
    rng = np.random.default_rng(seed)
    n = len(a)
    deadline = time.monotonic() + time_limit
    
    best_a = a.copy()
    best_score = evaluate_fast(best_a)
    print(f"Initial score: {best_score:.6f}")

    momentum = np.zeros(n)
    alpha = 0.95
    n_hot = 300
    no_improve = 0
    iter_count = 0
    print_interval = 50000  # print every 50K iterations
    
    while time.monotonic() < deadline:
        # Compute autocorrelation
        b = fft_full_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b  # S=1
        
        # Track best
        if score < best_score - 1e-9:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            # Slightly increase step size on improvement
            eta = min(eta * 1.01, 0.3)
        else:
            no_improve += 1
        
        # Find hot indices (where b is near max)
        thresh = max_b * 0.96
        hot = np.where(b >= thresh)[0]
        if len(hot) > n_hot:
            hot = np.argsort(b)[::-1][:n_hot]
        elif len(hot) == 0:
            hot = np.array([int(np.argmax(b))])
        
        # Compute gradient
        grad = np.zeros(n)
        for k in hot:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]

        momentum = alpha * momentum + (1.0 - alpha) * grad_centered
        mnorm = np.linalg.norm(momentum)

            a = a - eta * momentum / mnorm
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-14:
                a /= s
            else:
                a = np.ones(n) / n
                momentum = np.zeros(n)
        
        iter_count += 1
        
        # Print progress
        if iter_count % print_interval == 0:
            print(f"iter={iter_count}, score={score:.6f}, best={best_score:.6f}, eta={eta:.4f}")
        
        # Random restart when stuck
        if no_improve > 200000:
            r = rng.random()
            if r < 0.3:
                # Small noise
                a = a + rng.normal(0, 0.005, n)
            elif r < 0.6:
                # Medium blend with random
                a = 0.8 * a + 0.2 * np.abs(rng.normal(0, 1, n))

            momentum = np.zeros(n)
            eta = 0.1
            no_improve = 0
            print(f"  -> Random restart at iter {iter_count}")
    
    return best_a, best_score
