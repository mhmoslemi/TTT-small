"""
Minimizes the evaluation score 2n*max(conv(a,a))/sum(a)^2 using:
1. Multiple parametric initializations (geometric, cosine, Gaussian)
2. Directed local search with peak-reduction heuristics
3. Random block perturbations for exploration
"""

import numpy as np
import time

def geometric_seq(n, t):
    """Geometric decay: a[i] = exp(-t*i)"""
    return np.exp(-t * np.arange(n, dtype=np.float64))

def cosine_seq(n):
    """Cosine bell: a[i] = cos^2(pi*(i+0.5)/(2n))"""
    i = np.arange(n, dtype=np.float64)
    return np.cos(np.pi * (i + 0.5) / (2.0 * n)) ** 2

def gaussian_seq(n, sigma_frac=0.3):
    """Gaussian bump centered in the sequence"""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-(i - center)**2 / (2 * sigma**2))

def eval_score(a):
    """Compute the score directly (faster than calling evaluate_sequence for internal use)"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

def block_perturb(a, rng, step=0.1):
    """Perturb a random block of a by multiplying with exp(N(0, step))"""
    n = len(a)
    width = max(2, int(rng.integers(1, max(2, n // 50))))
    start = int(rng.integers(0, n - width + 1))
    candidate = a.copy()
    candidate[start:start+width] *= np.exp(rng.normal(0, step, size=width))
    np.clip(candidate, 0, 1000, out=candidate)
    return candidate

def local_search(a, deadline, rng, step_size=0.05):
    """Directed local search with adaptive step size."""
    n = len(a)
    best = a.copy()
    best_score = eval_score(best)
    step = step_size
    no_improve_count = 0
    
    while time.monotonic() < deadline:
        # Choose perturbation type
        r = rng.random()
        if r < 0.7:
            # Block multiplication
            candidate = block_perturb(best, rng, step)
        elif r < 0.85:
            # Targeted: find peak and reduce contributing elements
            conv = np.convolve(best, best)
            peak_idx = np.argmax(conv)
            # The peak conv[peak_idx] = sum_j best[j]*best[peak_idx-j]
            # Reduce the largest contributor
            candidates_j = range(max(0, peak_idx - n + 1), min(peak_idx + 1, n))
            products = np.array([best[j] * best[peak_idx - j] for j in candidates_j])
            if len(products) > 0:
                max_j = candidates_j[np.argmax(products)]
                candidate = best.copy()
                candidate[max_j] *= (1 - step * 0.5)
                # Compensate by increasing a random element
                inc_idx = int(rng.integers(0, n))
                candidate[inc_idx] *= (1 + step * 0.3)
                np.clip(candidate, 0, 1000, out=candidate)
            else:
                candidate = block_perturb(best, rng, step)
        else:
            # Large random jump
            width = max(2, int(rng.integers(n // 20, n // 5)))
            start = int(rng.integers(0, n - width + 1))
            candidate = best.copy()
            candidate[start:start+width] = rng.random(width) * best[start:start+width].max()
            np.clip(candidate, 0, 1000, out=candidate)
        
        score = eval_score(candidate)
        if score < best_score:
            best = candidate
            best_score = score
            no_improve_count = 0
            # Occasionally increase step size on improvement
            if rng.random() < 0.1:
                step = min(step * 1.1, 0.5)
        else:
            no_improve_count += 1
            if no_improve_count > 100:
                step = max(step * 0.95, 0.01)
    
    return best, best_score

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using parametric initializations
(optimal geometric, cosine, Gaussian) and FFT-accelerated local search
with targeted peak reduction and adaptive step sizing.
"""

import numpy as np
import time
from scipy import fft


def fast_convolve(a):
    """FFT-based auto-convolution, faster than O(n^2) for large n."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = fft.rfft(a, n=fft_size)
    return fft.irfft(fa * fa, n=fft_size)[:size]


def compute_score(a):
    """Compute the objective score using FFT convolution."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def make_geometric(n, t=None):
    """Optimal geometric: a[i] = exp(-t*i) with t = 1.256/n."""
    if t is None:
        t = 1.256 / n
    return np.exp(-t * np.arange(n, dtype=np.float64))


def make_cosine(n):
    """Cosine bell shape."""
    i = np.arange(n, dtype=np.float64)
    return np.cos(np.pi * (i + 0.5) / (2.0 * n)) ** 2


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian bump centered in sequence."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((i - c) / sigma) ** 2)


def make_triangle(n):
    """Triangular shape: rises then falls."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    return 1.0 - np.abs(i - c) / c


def perturb_block(a, rng, step):
    """Multiply a random block by exp(N(0, step))."""
    n = len(a)
    w = max(2, int(rng.integers(1, max(2, n // 80))))
    s = int(rng.integers(0, n - w))
    out = a.copy()
    out[s:s + w] *= np.exp(rng.normal(0.0, step, size=w))
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def perturb_targeted(a, conv_cache, rng, step):
    """Reduce the element most contributing to the convolution peak."""
    n = len(a)
    if conv_cache is None:
        conv_cache = fast_convolve(a)
    peak_k = int(np.argmax(conv_cache))
    lo = max(0, peak_k - n + 1)
    hi = min(n - 1, peak_k)
    js = np.arange(lo, hi + 1)
    prods = a[js] * a[peak_k - js]
    if len(prods) == 0:
        return perturb_block(a, rng, step), conv_cache
    max_local = int(np.argmax(prods))
    idx = js[max_local]
    out = a.copy()
    out[idx] *= (1.0 - 0.4 * step)
    bump = int(rng.integers(0, n))
    out[bump] *= (1.0 + 0.25 * step)
    np.clip(out, 0.0, 1000.0, out=out)
    return out, conv_cache


def perturb_symmetric(a, rng, step):
    """Perturb a block and its mirror simultaneously."""
    n = len(a)
    w = max(2, int(rng.integers(1, max(2, n // 60))))
    s = int(rng.integers(0, n - w))
    out = a.copy()
    factor = np.exp(rng.normal(0.0, step, size=w))
    out[s:s + w] *= factor
    # Mirror
    ms = n - s - w
    if ms >= 0 and ms + w <= n:
        out[ms:ms + w] *= factor[::-1]
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def local_search(start, deadline, rng, verbose=False):
    """Main local search loop with adaptive step and multiple perturbation types."""
    n = len(start)
    a = start.copy()
    score = compute_score(a)
    best = a.copy()
    best_score = score
    step = 0.06
    stuck = 0
    iter_count = 0
    conv_cache = fast_convolve(a)
    conv_stale = False

    while time.monotonic() < deadline:
        iter_count += 1
        r = rng.random()

        if r < 0.45:
            cand = perturb_block(a, rng, step)
        elif r < 0.70:
            if conv_stale:
                conv_cache = fast_convolve(a)
                conv_stale = False
            cand, conv_cache = perturb_targeted(a, conv_cache, rng, step)
            conv_stale = True
        elif r < 0.85:
            cand = perturb_symmetric(a, rng, step)
        else:
            # Larger exploratory move
            w = max(5, int(rng.integers(n // 40, n // 8)))
            s = int(rng.integers(0, n - w))
            cand = a.copy()
            cand[s:s + w] *= np.exp(rng.normal(0.0, step * 4.0, size=w))
            np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = compute_score(cand)
        if cand_score < score:
            a = cand
            score = cand_score
            conv_cache = None
            conv_stale = False
            stuck = 0
            if score < best_score:
                best = a.copy()
                best_score = score
                if verbose and iter_count % 1000 == 0:
                    print(f"    iter {iter_count}: score = {best_score:.6f}")
        else:

                stuck = 0
                # Occasionally reset to best + noise

                    a = best + rng.normal(0, step, n)
                    np.clip(a, 0, 1000, out=a)
                    score = compute_score(a)
                    conv_cache = None
                    conv_stale = False

        # Periodically boost step to escape flat regions

            step = 0.06

    return best, best_score
