import numpy as np
import time
import warnings

from scipy.optimize import minimize, linprog, differential_evolution

warnings.filterwarnings('ignore')

GAMMAS = np.array([-3.0, -2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0])


def _as_valid(a):
    """Convert to a finite, non-negative, length-clipped 1D float array."""
    arr = np.asarray(a, dtype=float)
    if arr.ndim != 1:
        arr = arr.ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def _conv(a):
    """Full linear convolution a*a, matching np.convolve(a,a)."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    if n <= 300:
        return np.convolve(a, a)
    N = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:N]
    c = np.where(c > 0.0, c, 0.0)
    return c


def _max_coeff(a):
    """Maximum coefficient of a*a."""
    c = _conv(a)
    if c.size == 0:
        return 0.0
    return float(np.max(c))


def _score_seq(a):
    """Local copy of the evaluator, for fast internal search."""
    a = _as_valid(a)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = _conv(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _random_perturb(p, rng, scale):
    """Multiplicative perturbation, preserving non-negativity and sum."""
    q = np.asarray(p, dtype=float).copy()
    if q.size == 0:
        return q
    mask = rng.random(q.size) < 0.25
    if np.any(mask):
        q[mask] *= np.exp(rng.normal(0.0, scale, int(np.sum(mask))))
    q = np.clip(q, 0.0, None)
    s = float(np.sum(q))
    if s <= 1e-15:
        q = np.ones(q.size, dtype=float) / q.size
    else:
        q /= s
    return q


def _subgradient_refine(p, max_iter=200, step=0.1, deadline=None, rng=None):
    """
    Projected subgradient / multiplicative-style local search on the simplex.
    Optimizes max coefficient of p*p for sum(p)=1.
    """
    p = _as_valid(p)
    if p.size == 0:
        p = np.ones(10, dtype=float) / 10.0
    s = float(np.sum(p))
    if s <= 1e-15:
        p = np.ones(10, dtype=float) / 10.0
    else:
        p = p / s

    n = p.size
    cur = _score_seq(p)
    no_improve = 0
    step = float(step)

    for _ in range(int(max_iter)):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _conv(p)
        if c.size == 0:
            break
        maxc = float(np.max(c))

        near = np.nonzero(c >= maxc - 1e-9 * max(1.0, maxc))[0]
        top_m = min(c.size, max(5, n))
        if near.size > top_m:
            active = np.argsort(c)[-top_m:]
        elif near.size == 0:
            active = np.argsort(c)[-top_m:]
        else:
            active = near

        g = np.zeros(n, dtype=float)
        for k in active:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            if hi >= lo:
                idx = np.arange(lo, hi + 1)
                g[idx] += 2.0 * p[ki - idx]
        if active.size > 0:
            g /= float(active.size)

        d = -(g - np.mean(g))
        dn = float(np.linalg.norm(d))
        if dn <= 1e-14:
            if rng is not None:
                p = _random_perturb(p, rng, 0.25)
                cur = _score_seq(p)
            break

        base = step / dn
        best_q = None
        best_val = cur

        for mult in (0.02, 0.05, 0.1, 0.25, 1.0):
            q = p + base * mult * d
            q = np.clip(q, 0.0, None)
            qs = float(np.sum(q))
            if qs <= 1e-15:
                continue
            q = q / qs
            val = 2.0 * n * _max_coeff(q)
            if val < best_val - 1e-12:
                best_val = val
                best_q = q

        if best_q is not None:
            p = best_q
            cur = best_val
            no_improve = 0
            step = min(1.0, step * 1.08)
        else:
            step = max(1e-3, step * 0.65)
            no_improve += 1
            if no_improve > 25:
                if rng is not None:
                    p = _random_perturb(p, rng, 0.3)
                    cur = _score_seq(p)
                    no_improve = 0
                else:
                    break

    return p, _score_seq(p)


def _lp_direction(p, T, U=1.0):
    """
    Solve the SOTA-inspired LP:
      maximize sum(g)
      subject to g >= 0, g <= U, and conv(p,g) <= T.
    """
    p = _as_valid(p)
    n = len(p)
    if n == 0 or n > 220:
        return None
    T = float(max(1e-12, T))
    A = np.zeros((2 * n - 1, n), dtype=float)

    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            A[k, idx] = p[k - idx]

    c_obj = -np.ones(n, dtype=float)
    b = np.full(2 * n - 1, T, dtype=float)
    bounds = [(0.0, float(U))] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        return None

    if res.success and res.x is not None:
        return np.asarray(res.x, dtype=float)
    return None


def _lp_refine(p, max_iter=20, deadline=None):
    """Block LP refinement using the cross-convolution constraint."""
    p = _as_valid(p)
    s = float(np.sum(p))
    if p.size == 0 or s <= 1e-15:
        p = np.ones(10, dtype=float) / 10.0
    else:
        p = p / s

    n = p.size
    cur = _score_seq(p)
    if n > 160:
        return p, cur

    for _ in range(int(max_iter)):
        if deadline is not None and time.monotonic() > deadline:
            break
        T = _max_coeff(p)
        g = _lp_direction(p, T, 1.0)
        if g is None:
            break
        gs = float(np.sum(g))
        if gs <= 1e-15:
            break
        g = g / gs

        improved = False
        for lam in (0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02):
            q = (1.0 - lam) * p + lam * g
            q = np.clip(q, 0.0, None)
            qs = float(np.sum(q))
            if qs <= 1e-15:
                continue
            q = q / qs
            val = 2.0 * n * _max_coeff(q)

                p = q
                cur = val
                improved = True
                break

            improved = False
            for lam in (0.5, 0.3, 0.2, 0.1, 0.05):
                q = (1.0 - lam) * p + lam * g2
                q = np.clip(q, 0.0, None)
                qs = float(np.sum(q))
                if qs <= 1e-15:
                    continue
                q = q / qs
                val = 2.0 * n * _max_coeff(q)
                if val < cur - 1e-12:
                    p = q
                    cur = val
                    improved = True
                    break
            if not improved:
                break

    return p, _score_seq(p)


def _slsqp_obj(x):
    return float(x[-1])


def _slsqp_eq(x):
    p = np.asarray(x[:-1], dtype=float)
    return float(np.sum(p) - 1.0)


def _slsqp_ineq(x):
    t = float(x[-1])
    p = np.asarray(x[:-1], dtype=float)
    p = np.where(p > 0.0, p, 0.0)
    c = _conv(p)
    return t - c


def _slsqp_polish(p, maxiter=300):
    """
    Exact constrained polish:
      variables p>=0, sum(p)=1, t>=0
      minimize t subject to coeff(p*p) <= t.
    """
    p = _as_valid(p)
    if p.size == 0:
        return None, np.inf
    s = float(np.sum(p))
    if s <= 1e-15:
        p = np.ones(10, dtype=float) / 10.0
    else:
        p = p / s
    p = np.maximum(p, 1e-9)
    p = p / float(np.sum(p))

    n = p.size
    if n > 90:
        return p, _score_seq(p)

    t0 = _max_coeff(p) + 1e-10
    x0 = np.concatenate((p, np.array([t0], dtype=float)))
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = (
        {'type': 'eq', 'fun': _slsqp_eq},
        {'type': 'ineq', 'fun': _slsqp_ineq}
    )

    cur = _score_seq(p)
    try:
        res = minimize(
            _slsqp_obj,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-10, 'disp': 0}
        )
    except Exception:
        return p, cur

    if res.x is not None:
        p2 = _as_valid(res.x[:-1])
        s2 = float(np.sum(p2))
        if s2 > 1e-15:
            p2 = p2 / s2
            sc = _score_seq(p2)
            if sc < cur:
                return p2, sc
    return p, cur


def _de_objective(x):
    a = np.asarray(x, dtype=float)
    a = np.where(a > 0.0, a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return 1e6
    a = a / s
    return float(_max_coeff(a))


def _de_maxiter(n):
    if n <= 8:
        return 420
    if n <= 14:
        return 260
    if n <= 24:
        return 160
    return 100


def _de_popsize(n):
    if n <= 8:
        return 36
    if n <= 14:
        return 26
    if n <= 24:
        return 18
    return 12


def _de_optimize(n, rng, maxiter=None, popsize=None):
    """Global Differential Evolution search for a fixed length."""
    if n is None or n < 1:
        return None, np.inf
    if maxiter is None:
        maxiter = _de_maxiter(n)
    if popsize is None:
        popsize = _de_popsize(n)

    bounds = [(0.0, 1.0)] * int(n)
    seed = int(rng.integers(0, 2 ** 31 - 1))

    try:
        res = differential_evolution(
            _de_objective,
            bounds,
            args=(),
            maxiter=int(maxiter),
            popsize=int(popsize),
            tol=1e-8,
            seed=seed,
            polish=False,
            strategy='best1bin',
            mutation=(0.5, 1.0),
            recombination=0.8
        )
    except Exception:
        return None, np.inf

    if res.x is not None:
        p = _as_valid(res.x)
        s = float(np.sum(p))
        if s > 1e-15:
            p = p / s
            sc = _score_seq(p)
            if np.isfinite(sc):
                return p, sc
    return None, np.inf

    if n <= 60:

    if n <= 200:

def _sub_iter(n):
    if n <= 16:
        return 320
    if n <= 30:

def _generate_starts(rng, n, count, prev_arr=None):
    """Generate diverse non-negative probability starts for length n."""
    starts = []
    n = int(n)
    xs = np.arange(n, dtype=float)

    for i in range(int(count)):
        typ = i % 12
        p = None

        if typ == 0:
            p = rng.dirichlet(np.ones(n))
        elif typ == 1:
            alpha = np.exp(np.linspace(-3.0, 3.0, n))
            p = rng.dirichlet(alpha)
        elif typ == 2:
            alpha = np.exp(np.linspace(3.0, -3.0, n))
            p = rng.dirichlet(alpha)
        elif typ == 3:
            p = np.exp(rng.normal(0.0, 1.0, size=n))
        elif typ == 4:
            gamma = float(rng.choice(GAMMAS))
            p = (xs + 1.0) ** gamma
        elif typ == 5:
            gamma = float(rng.uniform(-3.0, 3.0))
            e = gamma * (xs - (n - 1) / 2.0)
            e = e - float(np.max(e))
            p = np.exp(e)
        elif typ == 6:
            fmax = max(2, n // 2 + 1)
            f = int(rng.integers(1, fmax))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            A = float(rng.uniform(0.2, 0.95))
            p = 1.0 + A * np.cos(2.0 * np.pi * f * xs / n + phase)
        elif typ == 7:
            f1max = max(2, n // 4 + 1)
            f2max = max(2, n // 2 + 1)
            f1 = int(rng.integers(1, f1max))
            f2 = int(rng.integers(1, f2max))
            p = 1.0
            p += float(rng.uniform(0.0, 0.45)) * np.cos(2.0 * np.pi * f1 * xs / n + float(rng.uniform(0.0, 2.0 * np.pi)))
            p += float(rng.uniform(0.0, 0.45)) * np.cos(2.0 * np.pi * f2 * xs / n + float(rng.uniform(0.0, 2.0 * np.pi)))
        elif typ == 8:
            c1 = float(rng.uniform(0.0, float(n)))
            c2 = float(rng.uniform(0.0, float(n)))
            w1 = float(rng.uniform(max(1.0, n * 0.05), max(1.5, n * 0.5)))
            w2 = float(rng.uniform(max(1.0, n * 0.05), max(1.5, n * 0.5)))
            wgt = float(rng.uniform(0.1, 1.0))
            p = np.exp(-0.5 * ((xs - c1) / w1) ** 2) + wgt * np.exp(-0.5 * ((xs - c2) / w2) ** 2)
        elif typ == 9:
            f1 = int(rng.integers(1, max(2, n // 4 + 1)))
            f2 = int(rng.integers(1, max(2, n // 2 + 1)))
            p = 1.0
            p += float(rng.uniform(0.0, 0.40)) * np.sin(2.0 * np.pi * f1 * xs / n + float(rng.uniform(0.0, 2.0 * np.pi)))
            p += float(rng.uniform(0.0, 0.40)) * np.cos(2.0 * np.pi * f2 * xs / n + float(rng.uniform(0.0, 2.0 * np.pi)))
            p += 0.05 * rng.normal(0.0, 1.0, size=n)
        elif typ == 10:
            gamma = float(rng.choice(GAMMAS))
            p = (xs + 1.0) ** gamma
            perm = rng.permutation(n)
            p = p[perm]
        else:
            if prev_arr is not None and len(prev_arr) >= n:
                step = max(1, int(len(prev_arr) // n))
                off = int(rng.integers(0, max(1, step)))
                idx = (np.arange(n) * step + off) % len(prev_arr)
                p = prev_arr[idx]
            else:
                p = np.ones(n)

        p = _as_valid(p)
        s = float(np.sum(p))
        if s <= 1e-15:
            p = np.full(n, 1.0 / n, dtype=float)
        else:
            p = p / s
        starts.append(p)

    return starts


def _insert_candidate_list(lst, limit, score, seq):
    """Maintain a small list of best candidates without lambda sorting."""
    if not np.isfinite(score):
        return
    if len(lst) < int(limit):
        lst.append((score, np.asarray(seq, dtype=float).copy()))
    else:
        worst_idx = 0
        worst_val = lst[0][0]
        for i in range(1, len(lst)):
            if lst[i][0] > worst_val:
                worst_val = lst[i][0]
                worst_idx = i
        if score < worst_val:
            lst[worst_idx] = (score, np.asarray(seq, dtype=float).copy())


def _topk_candidates(lst, k):
    """Return k best candidates by score, without lambda sorting."""
    items = list(lst)
    out = []
    while items and len(out) < int(k):
        idx = 0
        val = items[0][0]
        for i in range(1, len(items)):
            if items[i][0] < val:
                val = items[i][0]
                idx = i
        out.append(items.pop(idx))
    return out


def _add_candidate(candidates, best_score, best_seq, per_n, n, score, seq, candidate_limit):
    """Update best and candidate containers."""
    if not np.isfinite(score) or seq is None:
        return best_score, best_seq
    seq = np.asarray(seq, dtype=float)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
    _insert_candidate_list(candidates, candidate_limit, score, seq)
    if n not in per_n:
        per_n[int(n)] = []
    _insert_candidate_list(per_n[int(n)], 12, score, seq)
    return best_score, best_seq


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a)**2).

    Strategy:
    - Perform global Differential Evolution searches for short-to-moderate lengths.
    - Drive good basins with active-set subgradient local search.
    - Polish top basins with the SOTA-inspired LP direction and SLSQP constrained polish.
    - Use remaining budget for stochastic perturbation + polish of the global top candidates.
    """
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0
    deadline = start_time + max(10.0, budget - 5.0)

    rng = np.random.default_rng(seed)

    best_seq = np.full(10, 0.1, dtype=float)
    best_score = _score_seq(best_seq)

    prev = globals().get('height_sequence_1', None)
    prev_arr = None
    if prev is not None:
        try:
            a = _as_valid(prev)
            if a.ndim != 1:
                a = a.ravel()
            if a.size > 0 and float(np.sum(a)) >= 0.01:
                prev_arr = a
                prev_score = _score_seq(a)
                if np.isfinite(prev_score) and prev_score < best_score:
                    best_score = prev_score
                    best_seq = a.copy()
        except Exception:
            prev_arr = None
            prev_score = np.inf

    candidates = []
    per_n = {}
    slsqp_calls = 0
    max_slq = 70

    de_ns = [4, 5, 6, 7, 8, 9, 10, 11, 12, 14, 16, 18, 20, 24, 28, 32, 36, 40]
    for n in de_ns:
        if time.monotonic() > deadline - 120.0:
            break
        p, score = _de_optimize(n, rng)
        if p is None:
            continue
        q, qscore = _subgradient_refine(p, 260 if n <= 16 else 160, 0.1, deadline, rng)
        best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, score, p, 120)
        best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, qscore, q, 120)
        print("DE n=%d current_best=%g" % (n, best_score))

    start_ns = list(range(4, 31)) + [36, 40, 48, 56, 64, 80, 100, 128, 160, 200]
    seen = set()
    for n in start_ns:
        if n in seen:
            continue
        seen.add(n)
        if time.monotonic() > deadline - 60.0:
            break
        count = _start_count(n)
        starts = _generate_starts(rng, n, count, prev_arr)
        for p in starts:
            if time.monotonic() > deadline - 60.0:
                break
            q, qscore = _subgradient_refine(p, _sub_iter(n), 0.1, deadline, rng)
            best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, qscore, q, 120)

    # Polish per-length top candidates with LP/SLSQP.
    for n in sorted(per_n.keys()):
        if n > 70:
            continue
        if time.monotonic() > deadline - 180.0:
            break
        top_count = 4 if n <= 12 else (2 if n <= 30 else 1)
        topn = _topk_candidates(per_n[n], top_count)
        for sc, p in topn:
            if time.monotonic() > deadline - 120.0:
                break
            q, qscore = _lp_refine(p, 22 if n <= 30 else 12, deadline)
            best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, qscore, q, 120)

            if slsqp_calls < max_slq and time.monotonic() < deadline - 90.0:
                miter = 520 if n <= 10 else (380 if n <= 20 else 260)
                r, rscore = _slsqp_polish(q, miter)
                slsqp_calls += 1
                best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, rscore, r, 120)

        if len(candidates) > 0 and time.monotonic() < deadline - 60.0:
            print("post-polish len=%d best=%g" % (n, best_score))

    # Final global polish: perturb top candidates, then LP/subgradient/slsqp.
    topk = _topk_candidates(candidates, 12)
    for sc, p in topk:
        if time.monotonic() > deadline - 45.0:
            break
        n = len(p)

        q, qscore = _lp_refine(p, 14, deadline)
        best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, qscore, q, 120)

        q2, q2score = _subgradient_refine(p, 360 if n <= 30 else 180, 0.18, deadline, rng)
        best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, q2score, q2, 120)

        for _ in range(2):
            if slsqp_calls >= max_slq:
                break
            if time.monotonic() > deadline - 60.0:
                break
            if n > 60:
                break
            pp = _random_perturb(p, rng, 0.22)
            rr, rscore = _slsqp_polish(pp, 260 if n <= 20 else 180)
            slsqp_calls += 1
            best_score, best_seq = _add_candidate(candidates, best_score, best_seq, per_n, n, rscore, rr, 120)

    best_seq = _as_valid(best_seq)
    if best_seq.size == 0:
        best_seq = np.full(10, 0.1, dtype=float)
    s = float(np.sum(best_seq))
    if not np.isfinite(s) or s <= 0.0:
        best_seq = np.full(10, 0.1, dtype=float)
        s = 1.0
    best_seq = best_seq / s

    # A very last local check without expensive SLSQP.
    final_seq, final_score = _subgradient_refine(best_seq, 80, 0.05, deadline, rng)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_seq = final_seq

    print("final length=%d final score=%g" % (len(best_seq), best_score))
    return [float(x) for x in best_seq]
