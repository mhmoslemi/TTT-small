def optimize_single_coordinate(full_conv, f, k):
    """Find optimal value of f[k] given full_conv = convolve(f, f)."""
    n = len(f)
    old = f[k]
    
    # Compute base (with f[k] = 0)
    # full_conv(i) = base(i) + 2*old*f(i-k) + old^2*[i=2k]
    # base(i) = full_conv(i) - 2*old*f(i-k) - old^2*[i=2k]
    
    # For the binary search, we need:
    # g_i(x) = base(i) + 2*f(i-k)*x + x^2*[i=2k]
    # We want min_x max_i g_i(x)
    
    # The function g_i(x) for each i is:
    # - x^2 + base(2k) if i = 2k
    # - 2*f(i-k)*x + base(i) if i != 2k and i-k in [0,n-1]
    # - base(i) otherwise (constant, doesn't affect the optimization)
    
    # For the max, we only care about:
    # 1. The quadratic: x^2 + base(2k)
    # 2. The linear terms with positive slope: 2*f(i-k)*x + base(i) where f(i-k) > 0
    
    # Let's collect:
    # C_quad = base(2k) = full_conv(2k) - old^2
    # For linear terms: slope_i = 2*f(i-k), intercept_i = base(i) = full_conv(i) - 2*old*f(i-k)
    #   (for i where i-k in [0,n-1] and i != 2k and f(i-k) > 0)
    
    C_quad = full_conv[2*k] - old**2 if 2*k < 2*n-1 else 0
    
    # Linear terms: for i in [k, k+n-1] intersect [0, 2n-2], i != 2k
    i_vals = np.arange(max(0, k), min(2*n-1, k+n))
    i_vals = i_vals[i_vals != 2*k]  # exclude the quadratic term
    ik_vals = i_vals - k  # these are in [0, n-1] by construction
    slopes = 2.0 * f[ik_vals]
    intercepts = full_conv[i_vals] - 2.0 * old * f[ik_vals]
    
    # Filter to positive slopes
    pos_mask = slopes > 0
    slopes_pos = slopes[pos_mask]
    intercepts_pos = intercepts[pos_mask]
    
    # Also need to consider constant terms (zero slope) for the lower bound
    const_max = 0
    zero_mask = slopes <= 0
    if np.any(zero_mask):
        const_max = intercepts[zero_mask].max()
    
    # Lower bound on t: max(C_quad, const_max, intercepts_pos.max() if any)
    t_lo = max(C_quad, const_max)
    if len(intercepts_pos) > 0:
        t_lo = max(t_lo, intercepts_pos.max())
    
    # Upper bound: current max
    current_max = full_conv.max()
    t_hi = current_max
    
    if t_hi <= t_lo:
        return old, t_hi  # can't improve
    
    # Binary search
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2.0
        # Feasibility: exists x >= 0 such that:
        # x^2 + C_quad <= t_mid  →  x <= sqrt(t_mid - C_quad) if t_mid >= C_quad
        # slope_i * x + intercept_i <= t_mid  →  x <= (t_mid - intercept_i) / slope_i
        
        x_upper = np.inf
        
        if t_mid < C_quad:
            t_lo = t_mid
            continue
        x_upper = min(x_upper, np.sqrt(t_mid - C_quad))
        
        if len(slopes_pos) > 0:
            # x <= (t_mid - intercept_i) / slope_i for all i
            # Need t_mid > intercept_i for all i, else infeasible
            if t_mid <= intercepts_pos.max():
                t_lo = t_mid
                continue
            x_lims = (t_mid - intercepts_pos) / slopes_pos
            x_upper = min(x_upper, x_lims.min())
        
        if x_upper > 0:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    # At t_hi (approximately optimal), find the best x
    x_upper = np.inf
    if t_hi >= C_quad:
        x_upper = min(x_upper, np.sqrt(max(0, t_hi - C_quad)))
    if len(slopes_pos) > 0 and t_hi > intercepts_pos.max():
        x_lims = (t_hi - intercepts_pos) / slopes_pos
        x_upper = min(x_upper, x_lims.min())
    
    x_opt = max(0.0, x_upper) if np.isfinite(x_upper) else 0.0
    return x_opt, t_hi
