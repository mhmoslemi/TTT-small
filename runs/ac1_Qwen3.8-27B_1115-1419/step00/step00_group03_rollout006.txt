```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def build_conv_matrix(f):
    """Build sparse matrix A (2n-1 x n) where A @ g = convolve(f, g)."""
    n = len(f)
    diagonals = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(0, -(n), -1))
    A = sparse.diags(diagonals, offsets, shape=(2*n-1, n), format='csc')
    return A

def compute_objective(f):
    """Compute 2*n*max(f*f) / sum(f)^2."""
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * n * np.max(c) / (S * S)

def subgradient_step(f, alpha, smooth=0.0):
    """One subgradient step to reduce max(f*f)."""
    n = len(f)
    c = np.convolve(f, f)
    max_c = np.max(c)
    # Find all k where c[k] is close to max (for smoother gradient)
    if smooth > 0:
        threshold = max_c * (1 - smooth)
        active = c >= threshold
    else:
        active = np.isclose(c, max_c, atol=1e-10 * max_c)
    
    # Subgradient: sum over active k of 2*f[k-j]
    subgrad = np.zeros(n)
    for k in np.where(active)[0]:
        # subgrad[j] += 2 * f[k-j] for valid k-j
        d_min = max(0, k - (n-1))
        d_max = min(k, n-1)
        # f[k-j] where j = k-d, d from d_min to d_max
        # subgrad[k-d] += 2*f[d]
        subgrad[k - d_max : k - d_min + 1] += 2 * f[d_min : d_max + 1][::-1]
    
    f_new = f - alpha * subgrad
    f_new = np.maximum(f_new, 0)
    S = np.sum(f_new)
    if S > 1e-10:
        f_new /= S
    return f_new, max_c, active

def lp_direction(f, max_c):
    """Solve LP to find optimal direction g."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = max_c * np.ones(2*n-1)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        S_g = np.sum(g)
        if S_g > 1e-10:
            return g / S_g
    return None
```

```python
"""
Step function optimizer minimizing 2*n*max(f*f)/sum(f)^2.
Uses combined subgradient descent and LP-based direction finding.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def _build_conv_matrix(f):
    """Build sparse matrix A (2n-1 x n) where A @ g = np.convolve(f, g)."""
    n = len(f)
    diagonals = []
    for d in range(n):
        diagonals.append(np.full(n, f[d]))
    offsets = list(range(0, -(n), -1))
    A = sparse.diags(diagonals, offsets, shape=(2*n-1, n), format='csc')
    return A

def _objective(f):
    """Compute 2*n*max(f*f)/sum(f)^2."""
    n = len(f)
    S = np.sum(f)
    if S < 1e-10:
        return np.inf
    c = np.convolve(f, f, mode='full')
    return 2.0 * n * np.max(c) / (S * S)

def _subgrad_update(f, alpha, top_k=5):
    """One subgradient step. Returns updated f and max convolution value."""
    n = len(f)
    c = np.convolve(f, f, mode='full')
    max_c = np.max(c)
    
    # Get top_k indices where c is largest
    top_indices = np.argsort(c)[-top_k:]
    
    subgrad = np.zeros(n)
    for k in top_indices:
        d_min = max(0, k - (n - 1))
        d_max = min(k, n - 1)
        subgrad[k - d_max : k - d_min + 1] += 2.0 * f[d_min : d_max + 1][::-1]
    
    f_new = f - alpha * subgrad
    f_new = np.maximum(f_new, 0.0)
    S = np.sum(f_new)
    if S > 1e-10:
        f_new /= S
    else:
        f_new = f.copy()
    return f_new, max_c

def _lp_direction(f, max_c):
    """Solve LP: max sum(g) s.t. f*g <= max_c, g >= 0. Returns normalized g."""
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, max_c)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            S_g = np.sum(g)
            if S_g > 1e-10:
                return g / S_g
    except Exception as e:
        print(f"LP error: {e}")
    return None

def propose_candidate(budget_s=2400):
    """
    Minimize 2*n*max(convolve(f,f))/sum(f)^2 using subgradient descent
    with periodic LP-based direction refinement and random restarts.
    """
    deadline = time.monotonic() + budget_s - 5.0
    rng = np.random.default_rng(777)
    
    # Sequence length
    n = 3000
    t_arr = np.arange(n, dtype=np.float64)
    
    # Generate multiple initializations
    inits = []
    # Sine window
    inits.append(np.sin(np.pi * (t_arr + 1) / (n + 1)))
    # Inverse raised cosine
    inits.append((1.0 - np.cos(2.0 * np.pi * t_arr / (n - 1))) / 2.0)
    # Quadratic (parabola, small at edges)
    x = 2.0 * t_arr / (n - 1) - 1.0
    inits.append(np.maximum(1.0 - x**2, 0.0))
    # Gaussian-like
    inits.append(np.exp(-8.0 * x**2))
    # Random with sine taper
    taper = np.sin(np.pi * (t_arr + 1) / (n + 1))
    inits.append(taper * np.exp(rng.normal(0, 0.3, n)))
    
    best_f = None
    best_score = np.inf
    
    for idx, f0 in enumerate(inits):
        f0 = np.clip(f0, 0, 1000).astype(np.float64)
        S0 = np.sum(f0)
        if S0 < 1e-10:
            continue
        f0 /= S0
        sc = _objective(f0)
        print(f"Init {idx}: n={n}, score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_f = f0.copy()
    
    if best_f is None:
        best_f = np.ones(n) / n
        best_score = _objective(best_f)
    
    f = best_f.copy()
    current_score = best_score
    print(f"Starting optimization from score {best_score:.6f}")
    
    lp_interval = 30  # do LP every 30 subgradient steps
    step = 0
    
    while time.monotonic() < deadline:
        # Subgradient descent phase
        for _ in range(lp_interval):

            except Exception as e:

        # Periodic perturbation

            f_pert = np.maximum(f_pert, 0.0)
```
