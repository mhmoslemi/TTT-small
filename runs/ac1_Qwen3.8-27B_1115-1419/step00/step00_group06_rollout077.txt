```python
"""
Search for non-negative coefficient sequences minimizing
    2 * n * max(convolve(a, a)) / sum(a)^2.

The main ideas used here are:
- Work with several short "macro" lengths and directly optimize them.
- Use a smooth max proxy: tau * logsumexp(conv(a, a) / tau), with the
  sequence constrained to the simplex via a softmax-like reparameterization.
- Use L-BFGS-B for smooth local refinement from many diverse starts.
- Occasionally use a linear-programming Frank-Wolfe-style refinement.
- Keep the best exact-score sequence found and return it.
"""

import time
import numpy as np

try:
    from scipy.optimize import minimize, linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def clean_normalize(x, floor=1e-12):
    """Return a finite nonnegative float array normalized to sum 1."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)
    x = np.clip(x, floor, 1e6)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / max(1, x.size), dtype=np.float64)
    x = x / s
    return x


def interp_to_length(arr, K):
    """Interpolate a 1D array to length K on [0,1]."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.nan_to_num(arr, nan=0.0, posinf=1.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1e6)
    if arr.size == 0:
        return np.ones(max(1, K), dtype=np.float64)
    if arr.size == 1:
        v = float(arr[0])
        if not np.isfinite(v) or v <= 0.0:
            v = 1.0
        return np.full(max(1, K), v, dtype=np.float64)
    oldx = np.linspace(0.0, 1.0, arr.size)
    newx = np.linspace(0.0, 1.0, max(1, K))
    y = np.interp(newx, oldx, arr)
    return clean_normalize(y)


def convolve_aa(a):
    """Full self-convolution of a 1D array."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    size = 2 * n - 1
    L = 1 << (max(1, size - 1).bit_length())
    fa = np.fft.rfft(a, L)
    out = np.fft.irfft(fa * fa, L)
    return out[:size]


def cross_corr_r_p(r, p):
    """
    Compute h_i = sum_k r[k] * p[k-i] for i = 0,...,n-1,
    where out-of-range p indices are treated as zero.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    r = np.asarray(r, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 1024:
        H = np.convolve(r, p[::-1], mode="full")
        return H[n - 1:n - 1 + n]
    size = len(r) + n - 1
    L = 1 << (max(1, size - 1).bit_length())
    R = np.zeros(L, dtype=np.float64)
    Q = np.zeros(L, dtype=np.float64)
    R[:len(r)] = r
    Q[:n] = p[::-1]
    H = np.fft.irfft(np.fft.rfft(R, L) * np.fft.rfft(Q, L), L)
    return H[n - 1:n - 1 + n]


def exact_score_array(a):
    """Exact evaluator, matching the provided formula plus clipping."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = convolve_aa(a)
    if c.size == 0:
        return np.inf
    mb = float(c.max())
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * a.size * mb / (s * s))


def exact_max_conv(a):
    """Return max(convolve(a,a)) for a nonnegative finite array."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    c = convolve_aa(a)
    if c.size == 0:
        return np.inf
    return float(c.max())


def p_to_logits(p):
    """
    Parameterize the simplex by K-1 logits:
    theta_i = log(p_i / p_{K-1}), i=0..K-2.
    """
    p = clean_normalize(p)
    if p is None:
        return np.array([], dtype=np.float64)
    K = p.size
    if K <= 1:
        return np.array([], dtype=np.float64)
    denom = p[-1] + 1e-300
    theta = np.log(p[:K - 1] + 1e-300) - np.log(denom)
    return np.clip(theta, -45.0, 45.0)


def theta_to_p(theta):
    """Inverse of p_to_logits: return p on the simplex."""
    theta = np.asarray(theta, dtype=np.float64).ravel()
    theta = np.nan_to_num(theta, nan=0.0, posinf=45.0, neginf=-45.0)
    K = len(theta) + 1
    w = np.empty(K, dtype=np.float64)
    w[:K - 1] = theta
    w[-1] = 0.0
    w = np.clip(w, -45.0, 45.0)
    w -= w.max()
    p = np.exp(w)
    p += 1e-300
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(K, 1.0 / K, dtype=np.float64)
    return p / s


def objective_and_grad(theta, tau):
    """
    Smooth objective:
        tau * logsumexp( convolve(p,p) / tau )
    and its gradient wrt theta, where p is on the simplex.
    """
    theta = np.asarray(theta, dtype=np.float64).ravel()
    K = len(theta) + 1
    w = np.empty(K, dtype=np.float64)
    w[:K - 1] = theta
    w[-1] = 0.0
    w = np.nan_to_num(w, nan=0.0, posinf=45.0, neginf=-45.0)
    w = np.clip(w, -45.0, 45.0)
    w -= w.max()
    p = np.exp(w)
    p += 1e-300
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full(K, 1.0 / K, dtype=np.float64)
    else:
        p = p / s

    c = convolve_aa(p)
    if c.size == 0:
        return 1e30, np.zeros(max(0, K - 1), dtype=np.float64)

    maxc = float(c.max())
    if not np.isfinite(maxc) or maxc <= 0.0:
        return 1e30, np.zeros(max(0, K - 1), dtype=np.float64)

    tau = float(tau)
    if not np.isfinite(tau) or tau <= 1e-15:
        tau = max(1e-15, maxc * 1e-3)

    z = (c - maxc) / tau
    zmax = float(z.max())
    expz = np.exp(np.clip(z - zmax, -50.0, 0.0))
    rz = float(expz.sum())
    if not np.isfinite(rz) or rz <= 0.0:
        r = np.zeros_like(c)
        r[int(np.argmax(c))] = 1.0
    else:
        r = expz / rz

    soft = float(maxc + tau * (zmax + np.log(rz)))

    h = cross_corr_r_p(r, p)
    grad_p = 2.0 * h
    A = float(np.dot(grad_p, p))
    grad_theta = p[:K - 1] * (grad_p[:K - 1] - A)

    if not np.all(np.isfinite(grad_theta)):
        grad_theta = np.zeros(max(0, K - 1), dtype=np.float64)
    if not np.isfinite(soft):
        soft = 1e30
    return float(soft), grad_theta


class SoftObj:
    """Callable objective wrapper for scipy.minimize."""

    def __init__(self, tau):
        self.tau = float(tau)

    def __call__(self, theta):
        val, grad = objective_and_grad(theta, self.tau)
        grad = np.asarray(grad, dtype=np.float64)
        if grad.shape != theta.shape:
            grad = np.zeros(theta.shape, dtype=np.float64)
        if not np.isfinite(val):
            val = 1e30
        return float(val), grad


def optimize_p_lbfmgs(p0, deadline=None, maxiter=50,
                      tau_factors=(0.25, 0.08, 0.03)):
    """Refine a simplex point using L-BFGS-B on the smooth objective."""
    p0 = clean_normalize(p0)
    if p0 is None:
        return None

    bestp = p0.copy()
    bestscore = exact_score_array(bestp)

    if not HAVE_SCIPY:
        return bestp

    K = p0.size
    if K <= 1:
        return bestp

    theta = p_to_logits(p0)
    if theta.size == 0:
        return bestp

    for idx, fac in enumerate(tau_factors):
        if deadline is not None and time.monotonic() >= deadline:
            break

        c = convolve_aa(bestp)
        maxc = float(c.max()) if c.size else 1.0 / K
        if not np.isfinite(maxc) or maxc <= 0.0:
            maxc = 1.0 / K

        tau = max(1e-15, maxc * float(fac))
        it = max(8, maxiter if idx == 0 else maxiter // 2)

        obj = SoftObj(tau)
        try:
            res = minimize(
                obj,
                theta,
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": it,
                    "maxfun": max(1, 10 * it),
                    "ftol": 1e-13,
                    "gtol": 1e-6,
                    "maxcor": 20,
                },
            )
        except Exception:
            break

        if res.x is None:
            break

        theta = np.asarray(res.x, dtype=np.float64)
        p2 = theta_to_p(theta)
        sc2 = exact_score_array(p2)

        if np.isfinite(sc2) and sc2 < bestscore:
            bestscore = sc2
            bestp = p2.copy()

    return bestp


def lp_refine(p0, rng, deadline=None, max_iters=6):
    """
    Linear-programming-style refinement:
      given f, M = max(f*f), maximize sum(g)
      s.t. g >= 0 and f*g <= M.
    Then mix f with the rescaled LP solution.
    """
    p = clean_normalize(p0)
    if p is None or not HAVE_SCIPY:
        return p

    bestp = p.copy()
    bestscore = exact_score_array(bestp)
    K = p.size
    if K > 128:
        return bestp

    # Tiny floor avoids degenerate LP columns.
    p = 0.999999 * p + 1e-7 / max(1, K)
    p = clean_normalize(p)

    accept_count = 0
    no_improve = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() >= deadline:
            break

        M = exact_max_conv(p)
        if not np.isfinite(M) or M <= 0.0:
            break

        A = np.zeros((2 * K - 1, K), dtype=np.float64)
        for j in range(K):
            A[j:j + K, j] = p[j]

        b = np.full(2 * K - 1, M, dtype=np.float64)
        cobj = -np.ones(K, dtype=np.float64) + 1e-8 * p

        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b,
                bounds=[(0.0, None)] * K,
                method="highs",
                options={"disp": False},
            )
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = clean_normalize(res.x)
        if g is None:
            break

        accepted = False
        old_best = bestscore
        for t in (0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005):
            f = (1.0 - t) * p + t * g
            f = clean_normalize(f)
            if f is None:
                continue
            Mnew = exact_max_conv(f)
            if not np.isfinite(Mnew):
                continue
            if Mnew <= M * (1.0 + 1e-8):
                p = f
                score = exact_score_array(p)
                if np.isfinite(score) and score < bestscore:
                    bestscore = score
                    bestp = p.copy()
                accepted = True
                break

        if not accepted:
            p = clean_normalize(p * np.exp(rng.normal(0.0, 0.05, size=K)))
            no_improve += 1
            if no_improve >= 4:
                break
            continue

        accept_count += 1
        if bestscore < old_best - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return bestp


def add_unique(candidates, y):
    """Normalize y and append if not too close to existing candidates."""
    y = clean_normalize(y)
    if y is None:
        return
    for old in candidates:
        if old is None:
            continue
        if y.size != old.size:
            continue
        if np.max(np.abs(old - y)) < 1e-5:
            return
    candidates.append(y)


def make_starts(K, rng, base_seq, max_starts=16):
    """Generate diverse initial simplex points of length K."""
    det = []
    rnd = []
    K = max(2, int(K))
    xs = np.linspace(0.0, 1.0, K)

    add_unique(det, np.ones(K, dtype=np.float64))

    # Linear ramps, both orientations.
    for B in (0.2, 0.5, 0.8, 1.0, 1.2, 1.4, 1.6, 1.8, 1.99):
        add_unique(det, 1.0 + B * (0.5 - xs))
        add_unique(det, 1.0 - B * (0.5 - xs))

    # Two-step plateaus.
    for w in (0.25, 0.35, 0.45, 0.5, 0.55, 0.65, 0.75, 0.85):
        y = np.ones(K, dtype=np.float64)
        y[xs < w] = 1.0 / max(1e-3, float(w))
        add_unique(det, y)

        y = np.ones(K, dtype=np.float64)
        y[xs >= w] = 1.0 / max(1e-3, float(1.0 - w))
        add_unique(det, y)

    # Exponentials.
    for lam in (-3.0, -2.0, -1.0, -0.5, 0.2, 0.5, 1.0, 2.0, 3.0):
        add_unique(det, np.exp(lam * (xs - 0.5)))

    # Power curves.
    for a in (0.25, 0.5, 0.8, 1.0, 1.5, 2.0):
        y = np.maximum(1.0 - 0.9 * xs, 1e-3)
        add_unique(det, np.power(y, a))
        y = np.maximum(1.0 - 0.9 * (1.0 - xs), 1e-3)
        add_unique(det, np.power(y, a))

    # Low-frequency asymmetric bumps.
    for mu in (0.3, 0.4, 0.5, 0.6, 0.7):
        for sig in (0.1, 0.2, 0.4):
            add_unique(det, 0.1 + np.exp(-((xs - mu) / sig) ** 2))

    # Quadratic perturbations of a ramp.
    for A in (0.5, 1.0, 1.5):
        for C in (-1.0, 0.0, 1.0):
            add_unique(det, 1.0 + A * (0.5 - xs) + C * (xs - 0.5) ** 2)

    # Interpolated hand-made small patterns.
    patterns = (
        (2.0, 1.0),
        (2.0, 3.0, 1.0, 2.0),
        (1.0, 3.0, 2.0, 1.0),
        (2.0, 1.0, 3.0, 1.0),
        (3.0, 2.0, 1.0, 2.0),
        (1.0, 2.0, 1.0, 3.0),
        (2.0, 2.0, 1.0, 3.0),
        (1.0, 3.0, 1.0, 2.0),
        (3.0, 1.0, 2.0, 1.0),
        (1.0, 2.0, 3.0, 1.0),
    )
    for pat in patterns:
        arr = np.asarray(pat, dtype=np.float64)
        y = np.interp(xs, np.linspace(0.0, 1.0, arr.size), arr)
        add_unique(det, y)

    # Previous best, if available.
    if base_seq is not None:
        y = interp_to_length(base_seq, K)
        add_unique(det, y)

    # Random low-frequency smooth starts.
    random_count = max(0, max_starts)
    for _ in range(random_count):
        y = np.ones(K, dtype=np.float64)
        for h in range(1, 3):
            amp = rng.uniform(0.3, 1.2) / (h + 0.5)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            ph2 = rng.uniform(0.0, 2.0 * np.pi)
            y += amp * np.sin(2.0 * np.pi * h * xs + ph)
            y += amp * 0.7 * np.cos(2.0 * np.pi * h * xs + ph2)
        add_unique(rnd, y)

        # Random ramp-plus-noise.
        y = 1.0 + 1.5 * (0.5 - xs) + rng.normal(0.0, 0.3, size=K)
        add_unique(rnd, y)

        # Mild independent noise.
        y = 0.5 + np.abs(rng.normal(0.0, 0.5, size=K))
        add_unique(rnd, y)

        # Local plateau.
        if K >= 6:
            lo = max(2, K // 5)
            hi = max(3, K // 2)
            if lo < hi:
                width = int(rng.integers(lo, hi))
                if 0 < width < K:
                    start = int(rng.integers(0, K - width))
                    y = np.ones(K, dtype=np.float64)
                    y[start:start + width] = rng.uniform(1.4, 3.6)
                    y *= np.exp(rng.normal(0.0, 0.2, size=K))
                    add_unique(rnd, y)

    # Mix deterministic and random, preferring at least half random.
    if max_starts <= 0:
        return []

    n_det = max(1, max_starts // 2)
    if len(det) >= n_det:
        step = max(1, len(det) // n_det)
        det_sel = det[::step]
        if len(det_sel) > n_det:
            det_sel = det_sel[:n_det]
    else:
        det_sel = list(det)

    out = []
    for y in det_sel:
        add_unique(out, y)
        if len(out) >= max_starts:
            break

    for y in rnd:
        add_unique(out, y)
        if len(out) >= max_starts:
            break

    if len(out) < max_starts:
        for y in det:
            add_unique(out, y)
            if len(out) >= max_starts:
                break

    if len(out) == 0:
        out.append(np.ones(K, dtype=np.float64) / float(K))

    return out


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main entry point.  Searches within the supplied wall-clock budget and
    returns the best sequence found as a Python list of floats.
    """
    start_time = time.monotonic()
    safe_budget = max(10.0, float(budget_s))
    deadline = start_time + safe_budget - 8.0
    rng = np.random.default_rng(20240607)

    best_seq = None
    best_score = np.inf

    def _is_finite_finite(x):
        return x is not None and np.all(np.isfinite(np.asarray(x, dtype=np.float64)))

    # Try to use the provided previous construction as a base candidate.
    try:
        height_sequence_1
    except Exception:
        height_sequence_1 = None

    if height_sequence_1 is not None:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size > 0 and arr.sum() > 0.0:
            sc = exact_score_array(arr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = arr.copy()

    # Guaranteed improvement over the all-ones baseline.
    simple_patterns = (
        (2.0, 1.0),
        (2.0, 3.0, 1.0, 2.0),
        (3.0, 1.0, 2.0, 1.0),
        (1.0, 3.0, 2.0, 1.0),
        (2.0, 1.0, 3.0, 1.0),
    )
    for pat in simple_patterns:
        p = clean_normalize(np.asarray(pat, dtype=np.float64))
        if p is None:
            continue
        sc = exact_score_array(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = p.copy()

    print(f"initial best: {best_score:.6f} len={0 if best_seq is None else len(best_seq)}",
          flush=True)

    # Search a ladder of lengths.  The returned sequence may be one of these.
    K_list = (
        2, 3, 4, 5, 6, 7, 8,
        10, 12, 14, 16,
        20, 24, 28, 32,
        40, 48, 56, 64,
        80, 96, 112, 128,
        160, 192, 224, 256,
        320, 384, 512
    )

    for K in K_list:
        if time.monotonic() >= deadline:
            break

        if K <= 8:
            K_time = 15.0
        elif K <= 16:
            K_time = 25.0
        elif K <= 32:
            K_time = 35.0
        elif K <= 64:
            K_time = 45.0
        elif K <= 128:
            K_time = 55.0
        else:
            K_time = 65.0

        K_time = min(K_time, max(2.0, deadline - time.monotonic()))
        k_deadline = min(deadline, time.monotonic() + K_time)

        if K <= 16:
            max_starts = 18
        elif K <= 64:
            max_starts = 14
        else:
            max_starts = 10

        starts = make_starts(K, rng, best_seq, max_starts=max_starts)
        if len(starts) == 0:
            continue

        per_start = max(0.25, (k_deadline - time.monotonic()) / max(1, len(starts)))

        if K <= 32:
            base_iter = 70
        elif K <= 80:
            base_iter = 50
        elif K <= 160:
            base_iter = 38
        else:
            base_iter = 28

        for p0 in starts:
            if time.monotonic() >= k_deadline or time.monotonic() >= deadline:
                break

            sc0 = exact_score_array(p0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = p0.copy()
                print(f"K={K} start best={best_score:.6f}", flush=True)

            if per_start < 0.4:
                continue

            popt = optimize_p_lbfmgs(
                p0,
                deadline=k_deadline,
                maxiter=base_iter,
                tau_factors=(0.25, 0.08, 0.03),
            )
            if popt is None:
                continue

            sc = exact_score_array(popt)
            improved_here = False
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = popt.copy()
                improved_here = True
                print(f"K={K} L-BFGS best={best_score:.6f}", flush=True)

            # Optional LP polish for promising small/medium lengths.
            if (K <= 96 and np.isfinite(sc) and sc < 1.72
                    and (k_deadline - time.monotonic()) > 2.0):
                plp = lp_refine(popt, rng, deadline=k_deadline,
                                max_iters=(8 if K <= 32 else 5))
                sc2 = exact_score_array(plp)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = plp.copy()
                    print(f"K={K} LP best={best_score:.6f}", flush=True)

            # If this start is close to the global best, give it a deeper polish.
            threshold = 1.62 if best_score < 1.62 else best_score + 0.02
            if (np.isfinite(sc) and sc < threshold
                    and (k_deadline - time.monotonic()) > 4.0):
                pdeep = optimize_p_lbfmgs(
                    popt,
                    deadline=k_deadline,
                    maxiter=base_iter + 35,
                    tau_factors=(0.10, 0.035, 0.012),
                )
                sc3 = exact_score_array(pdeep)
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = sc3
                    best_seq = pdeep.copy()
                    print(f"K={K} deep best={best_score:.6f}", flush=True)

    # Final polish of the best found sequence if length is still reasonable.
    if best_seq is not None and len(best_seq) <= 512:
        remaining = deadline - time.monotonic()
        if remaining > 25.0:
            p = clean_normalize(best_seq)
            popt = optimize_p_lbfmgs(
                p,
                deadline=deadline,
                maxiter=110,
                tau_factors=(0.08, 0.025, 0.01),
            )
            sc = exact_score_array(popt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = popt.copy()
                print(f"final polish best={best_score:.6f} len={len(best_seq)}", flush=True)

            if len(best_seq) <= 128 and (deadline - time.monotonic()) > 8.0:
                plp = lp_refine(best_seq, rng, deadline=deadline, max_iters=10)
                sc2 = exact_score_array(plp)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = plp.copy()
                    print(f"final LP best={best_score:.6f} len={len(best_seq)}", flush=True)

        # A few perturbed restarts from the best point.
        remaining = deadline - time.monotonic()
        if remaining > 20.0 and len(best_seq) <= 384:
            base = clean_normalize(best_seq)
            for noise in (0.08, 0.15, 0.25, 0.40):
                if deadline - time.monotonic() < 4.0:
                    break
                pnoise = clean_normalize(base * np.exp(rng.normal(0.0, noise, size=len(base))))
                popt = optimize_p_lbfmgs(
                    pnoise,
                    deadline=deadline,
                    maxiter=(70 if len(base) <= 128 else 35),
                    tau_factors=(0.12, 0.04, 0.015),
                )
                sc = exact_score_array(popt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = popt.copy()
                    print(f"restart noise={noise:.2f} best={best_score:.6f}", flush=True)

    if best_seq is None:
        best_seq = np.array([2.0, 1.0], dtype=np.float64)
        best_score = exact_score_array(best_seq)

    # Scale to avoid the evaluator's 1000 cap, which would be harmless but unnecessary.
    seq = np.asarray(best_seq, dtype=np.float64).ravel()
    seq = np.nan_to_num(seq, nan=0.0, posinf=1000.0, neginf=0.0)
    seq = np.clip(seq, 0.0, 1e6)
    mx = float(seq.max())
    if mx > 0.0:
        seq = seq * (500.0 / mx)
    seq = np.clip(seq, 0.0, 1000.0)

    # If possible, validate using the provided evaluator.
    final_list = [float(x) for x in seq.tolist()]
    try:
        eval_score = float(evaluate_sequence(final_list))
        if not np.isfinite(eval_score):
            # Should not happen, but fall back to exact.
            eval_score = exact_score_array(final_list)
    except Exception:
        eval_score = best_score

    print(f"FINAL len={len(final_list)} score={eval_score:.10f}", flush=True)
    return final_list
```
