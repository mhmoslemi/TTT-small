def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    b = np.convolve(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * float(n) * max_b / (sum_a * sum_a)


def resample_to_length(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def subgradient_step(a, lr, top_frac=0.9):
    n = a.size
    b = np.convolve(a, a)
    threshold = top_frac * float(np.max(b))
    active = np.where(b >= threshold)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(b))])
    
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[int(k) - idx]
    grad /= float(active.size)
    
    a_new = np.maximum(0.0, a - lr * grad)
    s = float(a_new.sum())
    if s < 1e-12:
        return a.copy()
    return a_new / s


def lp_direction(a):
    n = a.size
    b = np.convolve(a, a)
    M = float(np.max(b))
    if M <= 0:
        return a.copy()
    
    # Build convolution matrix A: (2n-1) x n
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[np.clip(diff[mask], 0, n - 1)]
    
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(1e-8, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            g = np.clip(res.x, 0.0, None)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return a.copy()

"""
Multi-phase autocorrelation flattening to minimize 2*n*max(conv(a,a))/sum(a)^2.
Phase 1: Vectorized subgradient iterations from diverse initializations.
Phase 2: LP-based direction (minimize max(g) s.t. conv(a,g)<=M, sum(g)=1).
Phase 3: Local search with block perturbations and symmetrization.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    b = np.convolve(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * float(n) * max_b / (sum_a * sum_a)


def resample_to_length(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def subgradient_step(a, lr, top_frac=0.92):
    n = a.size
    b = np.convolve(a, a)
    thr = top_frac * float(np.max(b))
    active = np.where(b >= thr)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(b))])
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        kk = int(k)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[kk - idx]
    grad /= float(active.size)
    a_new = np.maximum(0.0, a - lr * grad)
    s = float(a_new.sum())
    if s < 1e-12:
        return a.copy()
    return a_new / s


def lp_direction_uniform(a):
    """Minimize max(g) s.t. conv(a,g)<=M, g>=0, sum(g)=1."""
    n = a.size
    b = np.convolve(a, a)
    M = float(np.max(b))
    if M <= 0:
        return np.ones(n) / n

    # Build convolution matrix A_conv: (2n-1) x n
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A_conv = np.zeros((2 * n - 1, n), dtype=np.float64)
    A_conv[mask] = a[np.clip(diff[mask], 0, n - 1)]

    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize t => c = [0,...,0, 1]
    nvar = n + 1
    c = np.zeros(nvar, dtype=np.float64)
    c[n] = 1.0

    # Inequality constraints: A_ub @ x <= b_ub
    # 1) g_i <= t for i=0..n-1  =>  e_i - [0..0,1] <= 0  (n constraints)
    # 2) conv(a,g)_k <= M for k=0..2n-2  (2n-1 constraints)
    # 3) t >= 0 handled by bounds
    rows = n + (2 * n - 1)
    A_ub = np.zeros((rows, nvar), dtype=np.float64)
    b_ub = np.zeros(rows, dtype=np.float64)

    # g_i - t <= 0
    for i in range(n):
        A_ub[i, i] = 1.0
        A_ub[i, n] = -1.0
        b_ub[i] = 0.0

    # A_conv @ g <= M
    A_ub[n:n + 2 * n - 1, 0:n] = A_conv
    b_ub[n:n + 2 * n - 1] = M

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, nvar), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.success:
            g = np.clip(res.x[:n], 0.0, None)
            s = float(g.sum())
            if s > 1e-9:
                return g / s
        # Fallback: try original formulation (max sum(g))
    except Exception:
        pass

    # Fallback: max sum(g) s.t. conv(a,g) <= M, g >= 0
    try:
        c2 = -np.ones(n, dtype=np.float64)
        res2 = linprog(c2, A_ub=A_conv, b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                       bounds=[(1e-9, None)] * n, method='highs',
                       options={'presolve': True})
        if res2.success:
            g = np.clip(res2.x, 0.0, None)
            s = float(g.sum())
            if s > 1e-9:
                return g / s
    except Exception:
        pass

    return np.ones(n) / n


def try_update_best(a, best_score, best_seq):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return best_score, best_seq
    sc = compute_score(a)
    if sc < best_score:
        return sc, a.copy()
    return best_score, best_seq


def make_initializations(rng, n, height_seq_1):
    inits = []
    # From provided sequence
    try:
        h = np.asarray(height_seq_1, dtype=np.float64).ravel()
        h = np.clip(h, 0.0, 1000.0)
        if h.size >= 50:
            inits.append(resample_to_length(h, n))
    except Exception:
        pass
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n, dtype=np.float64))
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * x / float(n)))
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * (x / float(n)) ** 2))
    r = rng.standard_normal(n)
    r = np.convolve(r, np.ones(31) / 31.0, 'same')
    inits.append(np.clip(1.0 + 0.6 * r / (float(np.std(r)) + 1e-9), 0.05, None))
    inits.append(rng.exponential(1.0, n))
    inits.append(1.0
                + 0.5 * np.exp(-((x - 0.25 * n) / (0.08 * n)) ** 2)
                + 0.5 * np.exp(-((x - 0.75 * n) / (0.08 * n)) ** 2)
                - 0.3 * np.exp(-((x - 0.5 * n) / (0.04 * n)) ** 2))
    r2 = rng.standard_normal(n)
    r2 = np.convolve(r2, np.ones(61) / 61.0, 'same')
    inits.append(np.clip(1.0 + 0.8 * r2 / (float(np.std(r2)) + 1e-9), 0.05, None))
    # Small random perturbation of uniform
    inits.append(np.clip(1.0 + 0.2 * rng.standard_normal(n), 0.1, None))
    # Sine with 2 periods
    inits.append(1.0 + 0.3 * np.sin(4.0 * np.pi * x / float(n)))
    # Normalize
    out = []
    for a in inits:
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s > 1e-9:
            a = a / s
        out.append(a)
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    total_time = deadline - t_start

    n = 3000
    best_score = float('inf')
    best_seq = None

    # Generate initializations
    inits = make_initializations(rng, n, height_sequence_1)

    # === Phase 1: Subgradient (35% of time) ===
    p1_deadline = t_start + 0.35 * total_time
    print(f"Phase 1: subgradient, {len(inits)} initializations, "
          f"deadline={p1_deadline - t_start:.1f}s")

    for ii in range(len(inits)):
        if time.monotonic() >= p1_deadline:
            break
        a = inits[ii].copy()
        cur_score = compute_score(a)
        best_score, best_seq = try_update_best(a, best_score, best_seq)

        lr = 0.004
        stale = 0
        for step in range(300000):
            if time.monotonic() >= p1_deadline:
                break
            a_new = subgradient_step(a, lr, top_frac=0.92)
            new_score = compute_score(a_new)
            if new_score < cur_score - 1e-15:
                a = a_new
                cur_score = new_score
                stale = 0
                best_score, best_seq = try_update_best(a, best_score, best_seq)
                lr = min(lr * 1.01, 0.04)
            else:
                lr *= 0.75
                stale += 1
            if lr < 1e-8 or stale > 300:
                break
    print(f"Phase 1 done. Best: {best_score:.6f}  t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: LP iterations (30% of time) ===
    p2_deadline = t_start + 0.65 * total_time
    if best_seq is not None:
        a = best_seq.copy()
        cur_score = compute_score(a)
        t_vals = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        lp_stale = 0
        iters = 0

        while time.monotonic() < p2_deadline and lp_stale < 200:
            g = lp_direction_uniform(a)
            if g is None or not np.all(np.isfinite(g)) or float(np.sum(g)) < 1e-9:
                lp_stale += 1
                continue
            g = g / float(g.sum())

            best_a = a.copy()
            best_s = cur_score
            for t in t_vals:
                cand = (1.0 - t) * a + t * g
                cand = np.clip(cand, 0.0, None)
                cs = float(cand.sum())
                if cs < 1e-9:
                    continue
                cand = cand / cs
                c_score = compute_score(cand)
                if c_score < best_s:
                    best_s = c_score
                    best_a = cand.copy()

            if best_s < cur_score - 1e-14:
                a = best_a
                cur_score = best_s
                lp_stale = 0
                best_score, best_seq = try_update_best(a, best_score, best_seq)
            else:
                lp_stale += 1

            a = a / float(a.sum())
            iters += 1
            if iters % 10 == 0:
                print(f"  LP iter {iters}: score={cur_score:.6f} stale={lp_stale}")
    print(f"Phase 2 done. Best: {best_score:.6f}  t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Local search (remaining time) ===
    if best_seq is not None:
        a = best_seq.copy()
        cur_score = compute_score(a)
        n_local = a.size

        # Adaptive perturbation scale
        sigma = 0.06
        while time.monotonic() < deadline:
            # Block perturbation
            width = int(rng.integers(max(1, n_local // 300), max(2, n_local // 8)))
            start = int(rng.integers(0, max(1, n_local - width + 1)))
            stop = min(n_local, start + width)
            a_pert = a.copy()
            scale = np.exp(rng.normal(0.0, sigma, width))
            a_pert[start:stop] *= scale
            a_pert = np.clip(a_pert, 0.0, None)
            s = float(a_pert.sum())
            if s > 1e-9:
                a_pert = a_pert / s
                p_score = compute_score(a_pert)
                if p_score < cur_score:
                    a = a_pert
                    cur_score = p_score
                    best_score, best_seq = try_update_best(a, best_score, best_seq)
                    sigma = min(sigma * 1.05, 0.15)
                else:
                    sigma = max(sigma * 0.97, 0.01)

            # Symmetrize
            if rng.random() < 0.2:
                a_sym = 0.5 * (a + a[::-1].copy())
                a_sym = np.clip(a_sym, 0.0, None)
                ss = float(a_sym.sum())
                if ss > 1e-9:
                    a_sym = a_sym / ss
                    sym_score = compute_score(a_sym)
                    if sym_score < cur_score:
                        a = a_sym
                        cur_score = sym_score
                        best_score, best_seq = try_update_best(a, best_score, best_seq)

            # Smoothing (blur)
            if rng.random() < 0.1:
                k = 3
                a_blur = np.convolve(a, np.ones(2 * k + 1) / (2 * k + 1), 'same')
                a_blur = np.clip(a_blur, 0.0, None)
                sb = float(a_blur.sum())
                if sb > 1e-9:
                    a_blur = a_blur / sb
                    bl_score = compute_score(a_blur)
                    if bl_score < cur_score:
                        a = a_blur
                        cur_score = bl_score
                        best_score, best_seq = try_update_best(a, best_score, best_seq)

            # Coordinate perturbation
            if rng.random() < 0.1:
                idx = int(rng.integers(0, n_local))
                a_coord = a.copy()
                a_coord[idx] *= float(np.exp(rng.normal(0.0, 0.1)))
                a_coord = np.clip(a_coord, 0.0, None)
                sc = float(a_coord.sum())
                if sc > 1e-9:
                    a_coord = a_coord / sc
                    coord_score = compute_score(a_coord)
                    if coord_score < cur_score:
                        a = a_coord
                        cur_score = coord_score
                        best_score, best_seq = try_update_best(a, best_score, best_seq)

    # Final verification
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0).tolist()
        fs = evaluate_sequence(final)
        print(f"FINAL: score={fs:.6f} n={len(final)} max={max(final):.4f} "
              f"min={min(final):.4f} sum={sum(final):.4f}")
        return final

    # Fallback
    return [1.0] * 256
