def fast_score(x):
    """Compute score using FFT-based convolution."""
    n = len(x)
    if n == 0:
        return np.inf
    size = 2 * n
    X = np.fft.rfft(x, size)
    conv = np.fft.irfft(X * X, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(x)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np
import time

def fast_score(x):
    """Compute score using FFT-based convolution (fast)."""
    n = len(x)
    if n == 0:
        return np.inf
    size = 2 * n
    X = np.fft.rfft(x, size)
    conv = np.fft.irfft(X * X, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(x)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def compute_conv(x):
    """Compute full convolution using FFT."""
    n = len(x)
    size = 2 * n
    X = np.fft.rfft(x, size)
    return np.fft.irfft(X * X, size)[:2*n-1]

def subgradient(x, conv, n_active=5):
    """Compute approximate subgradient of the score function."""
    n = len(x)
    max_b = np.max(conv)
    sum_a = np.sum(x)
    
    # Find active lags
    threshold = max_b * 0.999
    active = np.where(conv >= threshold)[0]
    # Take top n_active by value
    if len(active) > n_active:
        active = active[np.argsort(conv[active])[-n_active:]]
    
    # Gradient of max(conv) w.r.t. x
    grad_max = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j = np.arange(lo, hi + 1)
            grad_max[j] += 2.0 * x[k - j]
    
    # Gradient of sum^2
    # score = 2n * max_b / sum_a^2
    # dscore/dx[j] = (2n/sum_a^2) * dmax_b/dx[j] - (4n*max_b/sum_a^3) * 1
    
    grad = (2.0 * n / (sum_a * sum_a)) * grad_max - (4.0 * n * max_b / (sum_a ** 3)) * np.ones(n)
    return grad
