So we can compute the gradient using two FFT-based convolutions per step:
1. c = f * f (compute convolution)
2. w = softmax(beta * c)
3. g = 2 * (rev(w) * f) (compute cross-correlation)

Each convolution is O(n log n) with FFT. Very fast!

import numpy as np
import time

def compute_score(f):
    """Compute the score without calling evaluate_sequence."""
    n = len(f)
    c = np.convolve(f, f)
    max_c = np.max(c)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_f * sum_f)

def gradient_step(f, beta, lr):
    """One step of gradient descent on the smooth surrogate."""
    n = len(f)
    # Compute convolution
    c = np.convolve(f, f)
    # Compute softmax weights
    c_shifted = c - np.max(c)
    w = np.exp(beta * c_shifted)
    w /= np.sum(w)
    # Compute gradient: g_j = 2 * sum_k w_k * f_{k-j}
    # This is 2 * (rev(w) * f)[j]
