```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

Now let me think about the `optimize_simplex` function more carefully.

The key algorithm is:
1. Start with a point a on the simplex (a >= 0, sum(a) = 1)
2. Compute the soft-max of the convolution: `F(a) = (1/beta) * log(sum_k exp(beta * c[k]))`
3. Compute the gradient: `grad = 2 * conv(w, a[::-1])[n-1:2n-1]` where `w[k] = exp(beta*c[k])/sum(exp(beta*c)))`
4. Project the gradient onto the tangent space: `grad_proj = grad - mean(grad)`
5. Take a step: `a_new = a - lr * grad_proj`
6. Project back to simplex
7. Increase beta over time (annealing)

The learning rate is crucial. Too large and we diverge; too small and we're slow.

For the soft-max, the Hessian is:

```python
Uses mirror descent (multiplicative weights) on the probability simplex with
soft-max approximation of the max convolution value, beta annealing,

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_conv(a):
    """Compute linear convolution of a with itself using FFT for efficiency."""
    n = len(a)
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def softmax_conv(a, beta):
    """Compute soft-max value and weight vector of conv(a,a)."""
    c = compute_conv(a)
    mc = np.max(c)
    exp_bc = np.exp(beta * (c - mc))
    s = exp_bc.sum()
    w = exp_bc / s
    val = mc + np.log(s) / beta
    return val, w

    """Compute gradient of soft-max of conv(a,a) with respect to a.
    
    grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * (w * a[::-1])[n-1+j]

        x = np.linspace(0, 1, n)
        freq = rng.uniform(2, 8)
        return np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * freq * x ** 2) + rng.uniform(0, 0.05, n), 0.001)
    elif name == 'ramp':
        x = np.linspace(0, 1, n)
        alpha = rng.uniform(0.5, 3.0)
        return np.maximum(x ** alpha + rng.uniform(0, 0.02, n), 0.001)
    elif name == 'uniform':
        return np.ones(n) + rng.uniform(0, 0.01, n)
    else:
        return rng.uniform(0.1, 1.0, n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses mirror descent with soft-max on the probability
    simplex, trying multiple initializations and sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0

    best_seq = None
    best_score = np.inf

    def try_sequence(seq, label):
        nonlocal best_seq, best_score
        seq = np.clip(seq, 0, 1000).astype(np.float64)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [{label}] New best: {score:.6f} (n={len(seq)})")

    # Get existing sequence
    try:
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if init_seq.ndim != 1 or init_seq.size == 0 or not np.all(np.isfinite(init_seq)):
            raise ValueError
    except Exception:
        init_seq = None

    # Build list of (name, length) to try
    tasks = []

    if init_seq is not None:
        tasks.append(('given', len(init_seq), 3.0))  # weight 3

    # Various lengths and types
    for n in [512, 1024, 2048, 4096, 8192, 16384, 32768]:
        for init_type in ['random', 'sine', 'gauss', 'chirp']:
            tasks.append((init_type, n, 1.0))

    # Sort: given first, then shorter sequences (faster to evaluate)
    tasks.sort(key=lambda t: (0 if t[0] == 'given' else 1, t[1]))

    total_weight = sum(w for _, _, w in tasks)
    available_time = deadline - time.monotonic()
    # Reserve 20% for final polishing
    search_time = available_time * 0.8
    time_per_task = search_time / total_weight

    print(f"Total tasks: {len(tasks)}, available time: {available_time:.0f}s")

    task_idx = 0
    for name, n, weight in tasks:
        if time.monotonic() >= deadline - 30:
            break

        task_time = time_per_task * weight
        task_deadline = min(time.monotonic() + task_time, deadline - 30)

        # Generate initialization
        if name == 'given':
            a = np.maximum(init_seq, 0).copy()
            s = a.sum()
            if s > 0:
                a = a / s
        else:
            a = generate_initialization(name, n, rng)
            a = a / a.sum()

        # Quick initial evaluation
        try_sequence(a, f"{name}_{n}_init")

        # Optimize
        if time.monotonic() < task_deadline:
            optimized, score = optimize_mirror_descent(a, task_deadline, rng)
            try_sequence(optimized, f"{name}_{n}_opt")

            # Try a perturbation-based restart
            if time.monotonic() < task_deadline and len(optimized) <= 16384:
                restart_deadline = min(time.monotonic() + task_time * 0.2, deadline - 30)
                while time.monotonic() < restart_deadline:
                    scale = rng.uniform(0.01, 0.05)
                    perturbed = project_simplex(optimized + rng.normal(0, scale, len(optimized)))
                    opt2, _ = optimize_mirror_descent(perturbed, restart_deadline, rng)
                    try_sequence(opt2, f"{name}_{n}_restart")
                    if np.isfinite(best_score) and best_score < 1.55:
                        break

        task_idx += 1

    # Final polishing on best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"Final polishing with {remaining:.0f}s...")
            a = best_seq / best_seq.sum()
            polished, _ = optimize_mirror_descent(a, deadline, rng, verbose=True)
            try_sequence(polished, "final_polish")

            # One more round with different beta schedule
            if remaining > 60:
                a2 = project_simplex(polished + rng.normal(0, 0.005, len(polished)))
                polished2, _ = optimize_mirror_descent(a2, deadline, rng)
                try_sequence(polished2, "final_polish2")

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback: {best_score:.6f}")

    print(f"FINAL SCORE: {best_score:.6f}")
    return [float(x) for x in best_seq]
```
