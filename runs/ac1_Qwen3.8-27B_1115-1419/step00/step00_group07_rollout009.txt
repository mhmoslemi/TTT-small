**Approach: Projected Subgradient Method with Momentum**

The objective is J(a) = 2n * max_k b_k(a) / (sum a)^2.

Since J is homogeneous of degree 0, I can fix sum(a) = n (or 1) and just minimize max_k b_k(a).

Subject to: sum(a) = n, a_i >= 0.

The subgradient of max_k b_k(a) at a is:
g_m = (2/|A|) * sum_{k in A} a_{|k - (n-1) - m|}... 

Actually, ∂b_k/∂a_m = 2*a_{k-m} (as derived earlier, with boundary conditions).

So the subgradient of max_k b_k is:
g_m = (2/|A|) * sum_{k in A} a_{k-m} (for valid k-m)

where A = {k : b_k(a) = max_b}.

The projected subgradient update is:
a^{new} = P_C(a - lr * g)
where C = {a : sum(a) = n, a >= 0} and P_C is projection onto C.

The projection onto {a : sum(a) = n, a >= 0} is:
- Subtract mean from a, clip at 0, adjust to make sum = n.
- Specifically: a_proj = max(0, a - λ) where λ is chosen so sum(a_proj) = n.

This is fast (O(n) per step).

The convolution b = conv(a, a) can be computed in O(n log n) using FFT.

So each iteration is O(n log n), and we can do many iterations.

Let me implement this:

def optimize_subgradient(a, max_iter=100000):
    n = len(a)
    a = a.copy().astype(float)
    # Normalize: sum(a) = n
    a = a * n / np.sum(a)
    
    best_a = a.copy()
    best_obj = evaluate_objective(a)
    
    lr0 = 1.0
    for it in range(max_iter):
        # Compute convolution
        b = np.convolve(a, a, mode='full')
        max_b = np.max(b)
        A = np.where(b == max_b)[0]  # indices of max
        
        # Subgradient
        g = np.zeros(n)
        for k in A:
            # ∂b_k/∂a_m = 2*a_{k-m} for valid k-m
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    g[m] += 2 * a[idx]
        g /= len(A)
        g /= max_b  # normalize
        
        # Update with projection
        a_new = a - lr0 * g
        # Project onto {sum = n, a >= 0}
        a_new = project_sum_n(a_new, n)
        
        obj = evaluate_objective(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
        
        a = a_new
        lr0 *= 0.999  # decay learning rate
    
    return best_a

    """Build sparse matrix C where C @ b = np.convolve(f, b)"""
    # C[k, j] = f[k-j] if 0 <= k-j < n, else 0

    # with values f[0], f[1], ..., f[n-1]

def optimize_lp(f, max_time=30):
    """Solve the LP: max sum(b) s.t. (f*b)_k <= M, b >= 0"""
    n = len(f)
    M = np.max(np.convolve(f, f))
    
    # Build constraint matrix
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    rows = cols + np.tile(np.arange(n, dtype=np.int32), n)

    # LP: max 1^T b s.t. A_ub @ b <= M, b >= 0

    return res.x

    lo, hi = 0.0, np.max(a)
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(a - mid, 0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - (lo + hi) / 2, 0)

from scipy.sparse import csr_matrix, coo_matrix

def _eval(a):
    """Internal fast evaluation"""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    b = np.convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * np.max(b) / (s * s))

    """Build sparse matrix C such that C @ b = np.convolve(f, b)"""
    n = f.size
    # C[k, j] = f[k-j] if 0 <= k-j < n
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    offsets = np.tile(np.arange(n, dtype=np.int32), n)
    rows_col = cols + offsets
    vals = np.tile(f, n)
    mask = rows_col < (2 * n - 1)
    row = rows_col[mask]
    col = cols[mask]

    return csr_matrix((val, (row, col)), shape=(2 * n - 1, n))

def _solve_lp(f):

    res = linprog(-np.ones(n, dtype=np.float64),
                  A_ub=C, b=np.full(2 * n - 1, M, dtype=np.float64),
                  bounds=[(0.0, None)] * n,
                  method='highs',
                  options={'time_limit': 60})
    
    if res.status == 0:
        return res.x
    return None

def _line_search(a, b_star, n, n_evals=30):
    """Find best t in [0,1] for a_t = (1-t)*a + t*b_star"""
    s_a = np.sum(a)
