from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_convolution_matrix(f, n):
    """
    Build sparse matrix A (shape (2n-1, n)) such that A @ g = np.convolve(f, g, 'full')[:2n-1].
    Actually, np.convolve(f, g, 'full') has length 2n-1, so A @ g gives the full convolution.
    """
    nnz = n * n  # total non-zeros
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)

        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(i_start, i_end + 1)
        # f[k-i] for i = i_start, ..., i_end
        # = f[k-i_start], f[k-i_start-1], ..., f[k-i_end]

        idx += count

    # Build LP
    A = build_convolution_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2*n-1, M)

def _build_A_sparsity(n):
    """Precompute sparsity pattern for the convolution matrix."""
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(i_start, i_end + 1)
        idx += count

    return rows, cols

def _fill_A_data(f, rows, cols, n):
    """Fill data values for the convolution matrix given current f."""
    nnz = n * n
