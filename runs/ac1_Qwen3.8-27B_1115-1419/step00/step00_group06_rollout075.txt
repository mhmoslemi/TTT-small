def project(w, have_mean=False):
    n=w.size
    if have_mean:
        u=np.sort(w)[::-1]
        cssv=np.cumsum(u)

    cb = b * float(2*n)

def project_simplex(v):

    n=v.size
    if n==0: return v

    u=np.sort(v)[::-1]

    rho_candidates = np.arange(1,n+1)

    out = np.maximum(v - theta, 0.0)
    s=out.sum()
    if s <=0: return np.full(n,1.0/n)
    out /= s
    return out

"""Flat self-convolution search: start from inverse-square-root profiles and use
projected penalty-gradient descent with multi-start and mutations.
"""
import numpy as np
import time

_CONV_LEN_CACHE = {}
_CORR_LEN_CACHE = {}


def _next_power_two_at_least(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def convolution_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _CONV_LEN_CACHE.get(n)
    if L is None:
        L = _next_power_two_at_least(m)
        _CONV_LEN_CACHE[n] = L
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:m]


def weighted_shift_gradient_fft(a, weights):
    a = np.asarray(a, dtype=np.float64).ravel()
    weights = np.asarray(weights, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    if weights.size < m:
        weights = np.concatenate((weights, np.zeros(m - weights.size, dtype=np.float64)))
    elif weights.size > m:
        weights = weights[:m]
    L = _CORR_LEN_CACHE.get(n)
    if L is None:
        L = _next_power_two_at_least(3 * n - 2)
        _CORR_LEN_CACHE[n] = L
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(weights, L)
    c = np.fft.irfft(np.conjugate(fw) * fa, L)
    return 2.0 * c[:n]


def softmax_weights(cb, tau):
    cb = np.asarray(cb, dtype=np.float64).ravel()
    n = cb.size
    if n == 0:
        return cb.copy()
    cmax = float(cb.max())
    if not np.isfinite(cmax):
        return np.full(n, 1.0 / n, dtype=np.float64)
    z = (cb - cmax) / max(1e-12, float(tau))
    z = np.maximum(z, -50.0)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1000.0, neginf=-1000.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    mask = (u * rho_candidates) > (cssv - 1.0)
    if not np.any(mask):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = int(np.where(mask)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def score_from_b(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    if n == 0 or b.size == 0:
        return float("inf")
    max_b = float(np.max(b))
    if max_b < 0.0:
        max_b = 0.0
    suma = float(np.sum(a))
    if not np.isfinite(suma) or abs(suma) < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (suma * suma))


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    b = convolution_full(a)
    return score_from_b(a, b)


def mutate_sequence(parent, rng, strength=0.15):
    a = np.asarray(parent, dtype=np.float64).ravel().copy()
    n = a.size
    if n == 0:
        return a
    strength = float(max(0.01, min(1.0, strength)))

    if rng.random() < 0.75 and n > 10:
        t = np.linspace(0.0, 1.0, n, dtype=np.float64)
        freq = int(rng.integers(1, 5))
        amp = min(0.45, strength * float(rng.uniform(0.5, 1.5)))
        co1 = float(rng.uniform(-1.0, 1.0))
        co2 = float(rng.uniform(-1.0, 1.0))
        mod = 1.0 + amp * (co1 * np.cos(2.0 * np.pi * freq * t) + co2 * np.sin(2.0 * np.pi * freq * t))
        mod = np.maximum(0.2, mod)
        a = a * mod

    min_w = max(1, n // 5000)
    max_w = max(min_w + 1, n // 20)
    for _ in range(3):
        if max_w <= min_w:
            width = min_w
        else:
            width = int(rng.integers(min_w, max_w))
        if width > n:
            width = n
        start_high = n - width
        if start_high <= 0:
            start = 0
        else:
            start = int(rng.integers(0, start_high))
        factor = float(np.exp(rng.normal(0.0, max(0.02, strength * 0.7))))
        a[start:start + width] *= factor

    a = np.maximum(a, 0.0)
    return normalize_sequence(a)


def dense_refine(a, rng, time_limit, label=""):
    a = normalize_sequence(a)
    n = a.size
    if n == 0:
        return np.array([1.0], dtype=np.float64), 2.0
    deadline = time.monotonic() + max(0.0, float(time_limit))

    best = a.copy()
    curr_b = convolution_full(a)
    curr_score = score_from_b(a, curr_b)
    best_score = curr_score

    if not np.isfinite(best_score):
        out = np.full(n, 1.0 / float(n), dtype=np.float64)
        return out, 2.0

    label = str(label)
    print(f"{label} start score {best_score:.6f}")

    lr = 0.25
    tau_top = 0.05
    floor = 1.40
    it = 0
    fail_count = 0
    no_best_count = 0

    while time.monotonic() < deadline:
        if curr_score < best_score - 1e-12:
            best = a.copy()
            best_score = curr_score
            no_best_count = 0
        else:
            no_best_count += 1

        if not np.isfinite(curr_score):
            a = best.copy()
            curr_b = convolution_full(a)
            curr_score = best_score
            lr = 0.25
            continue

        if curr_score < 1.52:
            margin = 0.012
        else:
            margin = 0.045
        M = best_score - margin
        if M < floor:
            M = floor
        if M >= best_score - 1e-7:
            M = best_score - 1e-7

        cb = curr_b * float(2 * n)
        cb = np.maximum(cb, 0.0)

        w_top = softmax_weights(cb, tau_top)
        ex = cb - M
        w_ex = np.maximum(ex, 0.0)
        s_ex = float(w_ex.sum())
        if s_ex > 1e-12:
            w = 0.75 * (w_ex / s_ex) + 0.25 * w_top
        else:
            w = w_top

        grad = weighted_shift_gradient_fft(a, w)
        if not np.all(np.isfinite(grad)) or float(np.max(np.abs(grad))) < 1e-15:
            k = int(np.argmax(cb))
            w = np.zeros(cb.size, dtype=np.float64)
            w[k] = 1.0
            grad = weighted_shift_gradient_fft(a, w)

        if it % 71 == 0:
            grad = grad + rng.normal(0.0, 1e-6, n)

        accepted = False
        lr_trial = lr
        for _ in range(3):
            cand = project_simplex(a - lr_trial * grad)
            cand = normalize_sequence(cand)
            cand_b = convolution_full(cand)
            cand_score = score_from_b(cand, cand_b)
            if np.isfinite(cand_score) and cand_score < curr_score - 1e-12:
                a = cand
                curr_b = cand_b
                curr_score = cand_score
                accepted = True
                fail_count = 0
                lr = min(1.0, lr * 1.04)
                break
            lr_trial *= 0.45
            if lr_trial < 1e-5:
                break

        if accepted:
            if curr_score < best_score - 1e-8 and it % 500 == 0:
                print(f"{label} iter {it} score {curr_score:.6f}")
        else:
            fail_count += 1
            lr = max(0.02, lr * 0.7)
            if fail_count > 8 or no_best_count > 1200:
                strength = 0.10 + 0.18 * min(1.0, no_best_count / 6000.0)
                a = mutate_sequence(best, rng, strength)
                curr_b = convolution_full(a)
                curr_score = score_from_b(a, curr_b)
                lr = 0.35
                fail_count = 0
                no_best_count = 0

        it += 1

    print(f"{label} best score {best_score:.6f}")
    return best, best_score


def build_starts(n, rng, parent):
    starts = []
    x = np.arange(n, dtype=np.float64)
    base = 1.0 / np.sqrt(x + 1.0)

    starts.append(normalize_sequence(base))
    starts.append(normalize_sequence(base[::-1]))

    for beta in [5.0, 25.0, 100.0, 500.0]:
        starts.append(normalize_sequence(1.0 / np.sqrt(x + beta)))

    half = np.minimum(x, float(n - 1) - x)
    starts.append(normalize_sequence(1.0 / np.sqrt(half + 1.0)))

    t = x / max(1.0, float(n - 1))
    for amp in [-0.25, -0.10, 0.10, 0.25]:
        starts.append(normalize_sequence(base * (1.0 + amp * t)))
        starts.append(normalize_sequence(base * np.exp(amp * t)))

    for amp in [-0.30, -0.20, -0.12, -0.06, 0.06, 0.12, 0.20, 0.30]:
        for freq in [1, 2]:
            mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
            if np.min(mod) > 0.05:
                starts.append(normalize_sequence(base * mod))
            mod = 1.0 + amp * np.sin(2.0 * np.pi * freq * t)
            starts.append(normalize_sequence(base * mod))

    if n > 10:
        win = max(1, n // 200)
        noise = rng.normal(0.0, 1.0, n)
        sm = np.convolve(noise, np.ones(win, dtype=np.float64) / float(win), mode="same")
        mod = 1.0 + 0.18 * sm
        mod = np.maximum(0.3, mod)
        starts.append(normalize_sequence(base * mod))

    if parent is not None:
        p = np.clip(parent, 0.0, 1000.0)
        if p.ndim == 1 and p.size == n and np.all(np.isfinite(p)) and p.sum() > 0.0:
            starts.append(normalize_sequence(p))

    valid = []
    for a in starts:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == n and np.all(np.isfinite(a)) and a.sum() > 0.0:
            valid.append(a)

    if not valid:
        valid.append(np.full(n, 1.0 / float(n), dtype=np.float64))
    return valid

        if run_budget < 30.0:
            final_reserve = max(20.0, 0.30 * remaining)
            run_budget = remaining - final_reserve

        runs = selected[:3]
        per = run_budget / max(1, len(runs))
        if per < 25.0 and len(runs) > 1:
            runs = runs[:1]
            per = run_budget

        run_idx = 0
        for a in runs:
            run_idx += 1
            now = time.monotonic()
            if now > deadline - final_reserve:
                break
            alloc = min(per, deadline - now - final_reserve)
            if alloc < 15.0:
                break
            a_run, s_run = dense_refine(a, rng, alloc, label=f"run{run_idx}")
            if np.isfinite(s_run) and s_run < best_score:
                best_score = s_run
                best_a = a_run.copy()

        final_time = max(0.0, deadline - time.monotonic() - 5.0)
        if final_time > 15.0:
            a_final, s_final = dense_refine(best_a, rng, final_time, label="final")
            if np.isfinite(s_final) and s_final < best_score:
                best_score = s_final
                best_a = a_final.copy()

    try:
        official = float(evaluate_sequence(best_a.tolist()))
    except Exception:
        official = float("inf")

    out = np.clip(np.asarray(best_a, dtype=np.float64).ravel(), 0.0, 1000.0)
    print(f"final internal score {best_score:.6f}, official {official:.6f}, n={out.size}")
    return [float(x) for x in out]
