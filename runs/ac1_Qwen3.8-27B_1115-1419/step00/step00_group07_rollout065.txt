import numpy as np
from scipy import sparse
import time


def _fft_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _feval(a):
    """Fast objective evaluation using FFT convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-8:
        return np.inf
    mx = _fft_conv(a).max()
    return float(2.0 * n * mx / (s * s))


def _build_conv_mat(a):
    """Build the convolution matrix C where (C@b)[k] = (a*b)[k]."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    # Cosine tilt

    # Random
    cands.append(rng.random(n) + 0.05)
    # Uniform
    cands.append(np.ones(n))
