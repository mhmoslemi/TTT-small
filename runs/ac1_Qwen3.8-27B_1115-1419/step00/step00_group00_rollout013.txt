```python
import numpy as np
import time

def fast_convolve_max(a):
    """Compute max of full auto-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    power = fa * np.conj(fa)
    c = np.fft.irfft(power, size)
    # The auto-convolution is in c[0:2n-1]
    # But due to FFT wrapping, for linear convolution we need size >= 2n-1
    # Since size >= 2n-1, the linear convolution is in c[0:2n-1]
    return np.max(c[:2*n-1])

def fast_evaluate(a):
    """Fast evaluation using FFT-based convolution."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    max_b = fast_convolve_max(a)
    return 2.0 * n * max_b / (s * s)
```
