Minimizes 2*n*max(conv(a,a))/sum(a)^2 using mirror descent (multiplicative weight update)
with multiple initializations and simulated annealing.
"""
import numpy as np

def compute_objective(f):
    """Compute the objective function value."""
    n = len(f)
    if n == 0:
        return np.inf
    conv = np.convolve(f, f)
    max_b = np.max(conv)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def mirror_descent_step(f, alpha, top_k=3, pressure_ema=None, ema_decay=0.9):
    """One step of mirror descent with smoothed pressure."""
    n = len(f)
    conv = np.convolve(f, f)
    
    # Find top k lags by convolution value
    # We can use partial sort for efficiency
    if top_k < n:
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
    else:
        top_indices = np.arange(len(conv))
    
    # Compute pressure from top lags
    pressure = np.zeros(n)
    for k in top_indices:
        k = int(k)
        # For lag k, pressure[i] = f[k-i] for valid i
        # Valid i: 0 <= i < n and 0 <= k-i < n
        # So i ranges from max(0, k-n+1) to min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            f_idx = k - np.arange(i_lo, i_hi + 1)
            pressure[i_lo:i_hi+1] += f[f_idx]
    
    pressure /= len(top_indices)
    
    # EMA of pressure for smoothing
    if pressure_ema is None:
        pressure_ema = pressure.copy()
    else:
        pressure_ema = ema_decay * pressure_ema + (1 - ema_decay) * pressure
    
    # Multiplicative update (mirror descent with negative entropy)
    # f_new = f * exp(-alpha * p) ≈ f / (1 + alpha * p) for small alpha*p
    f_new = f * np.exp(-alpha * pressure_ema)
    
    # Renormalize to maintain sum
    sum_new = np.sum(f_new)
    if sum_new > 0:
        f_new *= np.sum(f) / sum_new
    
    # Ensure non-negative (exp should keep it positive, but clip for safety)
    f_new = np.maximum(f_new, 0.0)
    
    return f_new, compute_objective(f_new), pressure_ema

    i = np.arange(n)
    f = 1.0 + amplitude * np.cos(2 * np.pi * freq * i / n)
    return np.maximum(f, 0.0)

    """Create a random block sequence."""
    rng = np.random.default_rng(seed)
    block_sizes = rng.integers(n // (n_blocks * 2), n // n_blocks, size=n_blocks)
    # Scale to sum to n
    block_sizes = block_sizes.astype(float)
    block_sizes *= n / block_sizes.sum()
    
    f = np.zeros(n)
    pos = 0

        if bs <= 0:

        val = rng.uniform(0.5, 3.0)
        end = min(n, pos + bs)
        f[pos:end] = val
        pos = end
    
    # Fill any remaining
    if pos < n:
        f[pos:] = 1.0
    
    return np.maximum(f, 0.0)
