import numpy as np
import time

"""
Algorithm summary
-----------------
We minimize  F(x) = 2*n*max(convolve(x,x)) / (sum(x))**2  over x >= 0.
The objective is scale invariant, so we work in the sum(x)=1 gauge where
F = 2*n*M and M = max_k (x*x)[k].  The information-theoretic lower bound is
2*n/(2*n-1) -> 1.

Method: multiplicative (exponential) subgradient descent on a smoothed proxy
    G_w(x) = sum_k w[k] (x*x)[k],   w = softmax((c - cmax)/T),  c = x*x,
with T = frac * cmax (auto-scaled to the magnitude of the convolution).
The subgradient is
    sg[i] = 2 * sum_j x[j] w[j+i]  =  2 * ifft(FFT(w) * conj(FFT(x)))[i]
computed with a single pair of FFTs (padding L >= 3n-2).  The update
    x <- x * exp(-lr * sg / mean(sg))  (clipped >= 1e-15, renormalised)
shrinks exactly the coefficients that feed the hot convolution peaks, flattening
the autocorrelation.  We anneal frac (soft -> hard) and lr, inject occasional
multiplicative noise to escape local minima, try several lengths n and several
starting points, and finally verify the best candidates with the authoritative
evaluate_sequence().
"""


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _fft_objective(x, L):
    F = np.fft.rfft(x, L)
    c = np.fft.irfft(F * F, L)
    c = c[: 2 * len(x) - 1]
    M = float(c.max())
    n = len(x)
    return 2.0 * n * M, c, F


def _optimize(x, deadline, rng, tag=""):
    x = np.asarray(x, dtype=np.float64).copy()
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        x = np.ones_like(x)
        s = 1.0
    x = x / s
    n = len(x)
    L = _next_pow2(3 * n - 1)

    score, c, F = _fft_objective(x, L)
    best = x.copy()
    best_score = score

    frac = 0.5
    lr = 0.8
    it = 0
    while time.monotonic() < deadline:
        T = frac * max(score / (2.0 * n), 1e-30)
        c_shift = c - c.max()
        w = np.exp(c_shift / T)
        w /= w.sum()
        W = np.fft.rfft(w, L)
        sg_full = np.fft.irfft(W * np.conj(F), L)
        sg = sg_full[:n] * 2.0
        sg_mean = sg.mean()
        if sg_mean <= 0.0:
            lr *= 0.9
        else:
            upd = lr * (sg / sg_mean)
            np.clip(upd, -40.0, 40.0, out=upd)
            x = x * np.exp(-upd)
            x = np.maximum(x, 1e-15)
            x = x / x.sum()

        score, c, F = _fft_objective(x, L)
        if score < best_score:
            best_score = score
            best = x.copy()

        frac = max(0.01, frac * exp_decay_frac)
        lr = max(0.05, lr * exp_decay_lr)

        if (it % 6000) == 5999:
            nz = np.exp(rng.normal(0.0, 0.05, n))
            x = np.maximum(x * nz, 1e-15)
            x = x / x.sum()

        if (it % 8000) == 0:
            print(f"  [{tag}] it={it} score={score:.6f} best={best_score:.6f} "
                  f"frac={frac:.4f} lr={lr:.4f}")
        it += 1

    return float(best_score), best


exp_decay_frac = 0.996
exp_decay_lr = 0.999


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global exp_decay_frac, exp_decay_lr
    rng = np.random.default_rng(seed)
    master_deadline = time.monotonic() + max(0.0, float(budget_s)) - 30.0
    candidates = []

    def _run(n, kind):
        if kind == "uniform":
            x = np.ones(n, dtype=np.float64)
        elif kind == "random":
            x = rng.random(n)
        elif kind == "smooth":
            xs = np.linspace(0.0, 1.0, n)
            x = 0.5 + 0.4 * np.cos(2.0 * np.pi * xs) + 0.1
            x = np.abs(x) + 0.05
        else:
            x = np.ones(n, dtype=np.float64)
        sub_deadline = time.monotonic() + max(1.0, (master_deadline - time.monotonic()) / 4.0)
        sc, bx = _optimize(x, sub_deadline, rng, tag=f"{n}/{kind}")
        candidates.append((sc, bx.copy()))
        print(f"[{n}/{kind}] local best={sc:.6f}")

    # A small per-branch anneal tuned by elapsed-time share; run a few configs.
    exp_decay_frac, exp_decay_lr = 0.996, 0.999

    if time.monotonic() < master_deadline:

    if time.monotonic() < master_deadline:

    if time.monotonic() < master_deadline:
        _run(4096, "random")
    if time.monotonic() < master_deadline:
        _run(8192, "smooth")

                candidates.append((sc, bx.copy()))
                print(f"[{base.size}/hs1] local best={sc:.6f}")
    except Exception as e:
        print("height_sequence_1 warm start failed:", e)

    # Select the best candidate by the authoritative evaluator.
    candidates.sort(key=lambda t: t[0])
    best_seq = candidates[0][1] if candidates else np.ones(4096, dtype=np.float64)
    verified = None
    for sc, bx in candidates[:6]:
        res = [float(v) for v in bx]
        try:

            v = np.inf
        print(f"  verify est={sc:.6f} -> actual={v:.6f}")
        if np.isfinite(v) and (verified is None or v < verified[0]):
            verified = (v, res)

    if verified is not None:
        best_seq = verified[1]

    # Guarantee sum >= 0.01 and non-negative finite values.
    out = [float(max(0.0, v)) for v in best_seq]
    s = sum(out)
    if s < 0.01:
        out = [v * 0.05 for v in out] if s > 0 else [0.05] * len(out)
    return out
