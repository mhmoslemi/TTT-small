```python
import cvxpy as cp

def solve_lp(a, M):
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    # conv(a, g)[k] <= M for k = 0, ..., 2n-2
    # conv(a,g)[k] = sum_j a[j]*g[k-j] for valid j
    constraints = []
    for k in range(2*n - 1):
        # g index: k-j must be in [0, n-1], so j in [max(0, k-n+1), min(k, n-1)]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # constraint: sum_{j=j_start}^{j_end} a[j] * g[k-j] <= M
        # g[k-j] where j goes from j_start to j_end
        # g index goes from k-j_end to k-j_start
        g_idx_start = k - j_end
        g_idx_end = k - j_start
        constraints.append(cp.sum(a[j_start:j_end+1] * g[g_idx_start:g_idx_end+1]) <= M)
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    return g.value
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using iterative LP + gradient-based optimization.
Key insight: asymmetric sequences can beat the uniform bound of 2.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = np.convolve(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k-n+1)
        je = min(k, n-1)
        A[k, js:je+1] = a[js:je+1]
    return A

def solve_lp(a, M):
    """LP: max sum(g) s.t. g>=0, A@ g <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return np.clip(res.x, 0, None)
    return None

def smooth_obj_and_grad(a, T):
    """Compute smooth objective and gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    S = a.sum()
    
    conv_max = conv.max()
    exp_c = np.exp((conv - conv_max) / T)
    exp_sum = exp_sum_val = exp_c.sum()
    L = conv_max + T * np.log(exp_sum)
    
    obj = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a
    weights = exp_c / exp_sum
    grad_L = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-12:
            continue
        i_lo = max(0, k-n+1)
        i_hi = min(k, n-1)
        # grad_L[i] += 2*w*a[k-i] for i in [i_lo, i_hi]
        # a[k-i] for i from i_lo to i_hi: a[k-i_lo], a[k-i_lo-1], ..., a[k-i_hi]
        # = a[k-i_hi : k-i_lo+1][::-1]
        grad_L[i_lo:i_hi+1] += 2.0 * w * a[k-i_hi:k-i_lo+1][::-1]
    
    grad_obj = 2.0 * n * (S * grad_L - 2.0 * L) / (S ** 3)
    return obj, grad_obj

def lp_search(a, max_iters=30, t_start=0.5, t_end=0.05):
    """LP-based iterative search."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0, None)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.convolve(a, a).max()
    
    for it in range(max_iters):
        t = t_start * (t_end / t_start) ** (it / max_iters)
        
        conv_a = np.convolve(a, a)
        M = conv_a.max()
        
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
        
        g = solve_lp(a, M)
        if g is None or np.sum(g) < 1e-10:
            # Try with slightly relaxed constraint
            g = solve_lp(a, M * 1.05)
            if g is None or np.sum(g) < 1e-10:
                break
        
        g = np.clip(g, 0, None)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        a = (1-t) * a + t * g
        a = np.clip(a, 0, None)
        S = a.sum()
        if S < 1e-10:
            break
        a = a / S
    
    return best_a

def gradient_polish(a, T=0.01, max_iter=50, tol=1e-10):
    """Polish using gradient-based optimization on smooth objective."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0, None)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.convolve(a, a).max()
    
    def obj_grad(x):
        x = np.clip(x, 1e-15, None)
        x = x / x.sum()
        o, g = smooth_obj_and_grad(x, T)
        return o, g
    
    res = minimize(obj_grad, a, jac=True, method='L-BFGS-B',
                   bounds=[(1e-15, None)]*n,
                   options={'maxiter': max_iter, 'gtol': tol, 'maxls': 50})
    
    if res.success or res.fun < best_obj:
        res_x = np.clip(res.x, 0, None)
        S = res_x.sum()
        if S > 1e-10:
            res_x = res_x / S
            obj = 2.0 * n * np.convolve(res_x, res_x).max()
            if obj < best_obj:
                best_a = res_x
                best_obj = obj
    
    return best_a
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative LP-based search
combined with gradient-based polish. Exploits asymmetry to beat the uniform
bound of 2.0 for the objective.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize

def build_conv_matrix(a):
    """Build matrix A (shape (2n-1, n)) such that A @ g = np.convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        A[k, js:je+1] = a[js:je+1]
    return A

def solve_lp(a, M, time_limit=20):
    """Solve: max sum(g) s.t. g >= 0, conv(a, g) <= M (elementwise)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

def smooth_obj_grad(a, T):
    """Compute smooth approximation of objective and its gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    conv_max = conv.max()
    exp_c = np.exp((conv - conv_max) / T)
    exp_sum = exp_c.sum()
    L = conv_max + T * np.log(exp_sum)
    
    obj = 2.0 * n * L / (S * S)
    
    weights = exp_c / exp_sum
    grad_L = np.zeros(n)
    # Only process significant weights
    significant = weights > 1e-10
    for k in np.where(significant)[0]:
        w = weights[k]
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        # a[k-i] for i in [i_lo, i_hi]
        # When i = i_lo: a[k - i_lo]
        # When i = i_hi: a[k - i_hi]
        # So we need a[k-i_hi : k-i_lo+1] reversed
        grad_L[i_lo:i_hi+1] += 2.0 * w * a[k-i_hi : k-i_lo+1][::-1]
    
    grad_obj = 2.0 * n * (S * grad_L - 2.0 * L) / (S * S * S)
    return obj, grad_obj

    n = len(a)

    if S < 1e-10:
        return a.copy()
    a = a / S
    
    best_a = a.copy()
    best_maxb = np.convolve(a, a).max()
    
    for it in range(max_iters):
        t = t_start * (t_end / t_start) ** (it / max(1, max_iters - 1))
        
        conv_a = np.convolve(a, a)
        M = conv_a.max()
        
        if M < best_maxb:
