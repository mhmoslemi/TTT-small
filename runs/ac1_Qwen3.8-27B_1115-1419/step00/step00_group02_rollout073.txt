import numpy as np
import time


"""
Search for nonnegative step-function heights minimizing
    2 * n * max(convolve(a, a)) / sum(a)^2.

Strategy:
  - Fast FFT-based scoring and smooth-max gradients.
  - Multiple constructive starting points: interpolated previous solution,
    exponentials, endpoint-singular envelopes, smooth random Fourier shapes.
  - A spectral Newton iteration trying to realize a flat (or slightly tapered)
    target convolution as a square of a sequence; the positive part is used.
  - Adam-style optimization in softmax/log space with annealed soft-max weight
    on the active convolution lags.
  - Progressive lengths; the best sequence found is returned.
"""


def _pair_score(pair):
    return pair[0]


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj_nonneg_sum(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a.fill(1.0 / max(1, a.size))
    else:
        a /= s
    return a


def _soft(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - float(np.max(v))
    v = np.clip(v, -745.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / max(1, v.size), dtype=np.float64)
    return e / s


def _log_plus(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-300)
    return np.log(a)


def _conv_self_fft(a, m):
    a = np.asarray(a, dtype=np.float64)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * a.size - 1]
    b = np.maximum(b, 0.0)
    return b


def _score_from_F(a, F, m):
    n = int(a.size)
    L = 2 * n - 1
    b = np.fft.ifft(F * F).real[:L]
    b = np.maximum(b, 0.0)
    S = float(np.sum(a))
    if S <= 1e-300:
        return np.inf
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    b = _conv_self_fft(a, m)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _interp(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s <= 0.0:
        return np.ones(int(max(1, n)), dtype=np.float64)
    if a.size == int(n):
        return a / s
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, int(n))
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 0.0:
        out.fill(1.0 / n)
    else:
        out /= ss
    return out


def _param_inits(n, rng):
    n = int(max(3, n))
    res = []
    t = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    res.append(_proj_nonneg_sum(np.ones(n)))

    for c in (0.3, 0.7, 1.2, 2.0, 4.0, 8.0):
        res.append(_proj_nonneg_sum(np.exp(-c * t)))

    left = (i + 1.0)
    right = (n - i)
    res.append(_proj_nonneg_sum(1.0 / np.sqrt(left * right)))

    for p, q in ((0.25, 0.25), (0.5, 0.5), (1.0, 0.25), (0.25, 1.0),
                 (1.5, 0.5), (0.5, 1.5), (2.0, 0.5), (0.5, 2.0)):
        a = (left / float(n)) ** p * (np.maximum(1.0 - t, 1e-9)) ** q
        res.append(_proj_nonneg_sum(a))

    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t) + 0.3 * np.cos(4.0 * np.pi * t) \
        + 0.2 * np.cos(6.0 * np.pi * t)
    res.append(_proj_nonneg_sum(a))

    a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t) ** 2 + 0.35 * np.cos(4.0 * np.pi * t)
    res.append(_proj_nonneg_sum(a))

    kc = min(14, n)
    a = 1.0
    for k in range(1, kc + 1):
        amp = 0.35 / (float(k) ** 0.75)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t + ph)
    res.append(_proj_nonneg_sum(a))

    a = 1.0
    for k in range(1, kc + 1):
        amp = 0.30 / (float(k) ** 0.9)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.sin(2.0 * np.pi * k * t + ph)
    res.append(_proj_nonneg_sum(a))

    return res


def _spectral_sqrt_target(target, a_start, n, m, iters, blend=0.45, score_every=12):
    n = int(max(3, n))
    L = 2 * n - 1
    target = np.asarray(target, dtype=np.float64)
    if target.size < L:
        padded = np.zeros(L, dtype=np.float64)
        padded[:target.size] = target
        target = padded
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    tmax = float(target.max())
    if tmax > 0.0 and np.isfinite(tmax):
        target = target / tmax
    else:
        target = np.ones(L, dtype=np.float64)

    T = np.fft.fft(target[:min(L, m)], m)

    a = np.asarray(a_start, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    amax = float(a.max())
    if amax <= 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a / amax
    if n > 1 and a[0] < 0.25 * a[-1]:
        a = a[::-1]
    a = np.maximum(a, 1e-12)
    if a[0] < 1e-6:
        a[0] = 1e-3

    if target[0] > 1e-15 and a[0] > 1e-300:
        scale = float(np.sqrt(target[0])) / max(float(a[0]), 1e-300)
        a = a * scale

    best = a.copy()
    best_sc = np.inf

    sqrt_T = np.fft.ifft(np.sqrt(T)).real[:n]
    sp = np.maximum(sqrt_T, 0.0)
    sp_max = float(sp.max())
    if sp_max > 0.0 and np.isfinite(sp_max):
        sp = sp / sp_max
    else:
        sp = a.copy()

    a = 0.5 * a + 0.5 * sp
    a = np.maximum(a, 1e-12)
    if a[0] < 1e-9:
        a[0] = 1e-3

    T_norm = float(np.max(np.abs(T)))
    if not np.isfinite(T_norm) or T_norm <= 0.0:
        T_norm = 1.0

    iters = int(max(1, iters))
    for it in range(iters):
        F = np.fft.fft(a, m)
        if not np.all(np.isfinite(F)):
            break

        F_absmax = float(np.max(np.abs(F)))
        if not np.isfinite(F_absmax) or F_absmax <= 0.0:
            F_absmax = 1.0
        eps = 1e-8 * max(1.0, F_absmax) + 1e-12 * T_norm
        denom = F + eps
        Fnew = 0.5 * (F + T / denom)
        if not np.all(np.isfinite(Fnew)):
            break

        anew = np.fft.ifft(Fnew).real[:n]
        if not np.all(np.isfinite(anew)):
            break

        pos = np.maximum(anew, 0.0)
        psum = float(pos.sum())
        asum = float(a.sum())
        if psum < 1e-6 * max(asum, 1e-12):
            pos = a.copy()

        bld = max(0.20, blend * (1.0 - 0.55 * (float(it) / float(iters))))
        a_new = (1.0 - bld) * a + bld * pos
        a_new = np.maximum(a_new, 1e-12)
        mx = float(a_new.max())
        if mx > 1e12:
            a_new = a_new * (1e12 / mx)
        if mx <= 0.0:
            break
        a = a_new

        if (it + 1) % score_every == 0:
            Fs = np.fft.fft(a, m)
            sc = _score_from_F(a, Fs, m)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = a.copy()

    Fs = np.fft.fft(a, m)
    sc = _score_from_F(a, Fs, m)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best = a.copy()

    best = _proj_nonneg_sum(best)
    sc = score_fft(best)
    if not np.isfinite(sc):
        sc = float(best_sc)
    return best, float(sc)


def _spectral_stage(n, best_a, deadline, rng):
    out = []
    n = int(max(3, n))
    m = _next_pow2(max(64, 4 * n))
    L = 2 * n - 1

    tL = np.linspace(0.0, 1.0, L)
    edge = np.abs(tL - 0.5) * 2.0
    flat = np.ones(L, dtype=np.float64)

    b_norm = None
    if best_a is not None:
        ba = np.asarray(best_a, dtype=np.float64)
        if ba.size == n:
            b = _conv_self_fft(ba, _next_pow2(2 * ba.size - 1))
            bm = float(b.max())
            if bm > 0.0 and np.isfinite(bm):
                b_norm = b / bm
    if b_norm is None:
        b_norm = flat.copy()

    starts = []
    if best_a is not None:
        starts.append(_proj_nonneg_sum(best_a))
    starts.append(_proj_nonneg_sum(np.exp(-1.0 * np.linspace(0.0, 1.0, n))))
    i = np.arange(n, dtype=np.float64)
    starts.append(_proj_nonneg_sum(1.0 / np.sqrt((i + 1.0) * (n - i))))

    prefs = []
    for ff in (1.0, 0.85, 0.65, 0.40, 0.20):
        for eb in (0.0, 0.5, 1.0):
            prefs.append((ff, eb))

    max_runs = 6 if n <= 2048 else 4
    iters = 70 if n <= 1024 else (45 if n <= 4096 else 30)
    start_a = starts[0]

    for ff, eb in prefs:
        if len(out) >= max_runs:
            break
        if time.monotonic() >= deadline:
            break
        if out:
            start_a = out[0][0]

        base = 1.0 + eb * (edge ** 1.5)
        bmax = float(base.max())
        if bmax > 0.0:
            base = base / bmax
        if ff >= 1.0:
            target = base.copy()
        else:
            target = ff * base + (1.0 - ff) * b_norm
        target = np.maximum(target, 0.0)
        tm = float(target.max())
        if tm > 0.0:
            target = target / tm

        a, sc = _spectral_sqrt_target(target, start_a, n, m, iters,
                                      blend=0.50, score_every=15)
        out.append((a, sc))
        print("      spectral target ff=%.2f eb=%.1f score=%.6f" % (ff, eb, sc))

    return out


def optimize_v(a_init, n, max_seconds, seed):
    n = int(max(3, n))
    max_seconds = float(max(1.0, max_seconds))
    rng = np.random.default_rng(seed)

    v = _log_plus(a_init)
    v = v - float(np.mean(v))
    v = np.clip(v, -25.0, 25.0)

    m = _next_pow2(2 * n - 1)
    a = _soft(v)
    best_a = a.copy()
    best_score = score_fft(a)
    if not np.isfinite(best_score):
        best_a = np.ones(n, dtype=np.float64) / float(n)
        v = _log_plus(best_a)
        best_score = 2.0

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0

    start = time.monotonic()
    deadline = start + max_seconds
    last_improve = start
    check_interval = max(2, min(18, 1800 // max(1, n)))

    while time.monotonic() < deadline:
        elapsed = max(1e-9, time.monotonic() - start)
        progress = min(1.0, elapsed / max_seconds)

        stuck_limit = min(90.0, 0.35 * max_seconds)
        if time.monotonic() - last_improve > stuck_limit:
            sigma = 0.10 + 0.35 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            last_improve = time.monotonic()
            continue

        a = _soft(v)
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break

        tau_frac = 0.18 * (1.0 - progress) + 0.03
        if rng.random() < 0.25:
            tau_frac *= float(rng.uniform(0.3, 3.0))
        tau = max(1e-15, bmax * max(tau_frac, 1e-4))

        x = (b - bmax) / tau
        x = np.clip(x, -60.0, 60.0)
        x = x - float(x.max())
        w = np.exp(x)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            break
        w = w / ws

        blen = int(b.size)
        top_k = max(1, int(0.05 * blen))
        if top_k < blen:
            idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            idx = np.arange(blen)
        w[idx] += 0.5
        ws = float(w.sum())
        w = w / ws

        P = np.fft.fft(w, m)
        c = np.fft.ifft(P * np.conj(F)).real[:n]
        np.maximum(c, 0.0, out=c)

        cm = float(np.dot(c, a))
        gv = a * (c - cm)
        grms = float(np.sqrt(np.mean(gv * gv)))
        if not np.isfinite(grms) or grms < 1e-14:
            sigma = 0.05 + 0.20 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            continue

        gv = gv / grms
        t += 1
        mu = beta1 * mu + (1.0 - beta1) * gv
        nu = beta2 * nu + (1.0 - beta2) * (gv * gv)

        b1 = beta1 ** t
        b2 = beta2 ** t
        denom1 = max(1e-12, 1.0 - b1)
        denom2 = max(1e-12, 1.0 - b2)
        mhat = mu / denom1
        vhat = nu / denom2

        lr = 0.10 * (1.0 - 0.65 * progress)
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.5, 0.5)
        v = v - lr * step
        v = v - float(np.mean(v))
        v = np.clip(v, -25.0, 25.0)

        if t % check_interval == 0:
            an = _soft(v)
            Fn = np.fft.fft(an, m)
            sc = _score_from_F(an, Fn, m)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_a = an.copy()
                last_improve = time.monotonic()

    an = _soft(v)
    sc = score_fft(an)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = an

    return _proj_nonneg_sum(best_a), float(best_score)


def _process_stage(n, deadline, prev_best_a, rng):
    n = int(max(3, n))
    now = time.monotonic()
    if now >= deadline:
        return (prev_best_a, score_fft(prev_best_a) if prev_best_a is not None else np.inf)

    best_a = None
    best_score = np.inf

    if prev_best_a is not None:
        best_a = _interp(prev_best_a, n)
        best_score = score_fft(best_a)

    inits = []
    if best_a is not None:
        inits.append(best_a)
    inits.extend(_param_inits(n, rng))

    rem = max(0.0, deadline - time.monotonic())
    spec_rem = min(rem * 0.30, 120.0)
    spec_deadline = min(deadline, time.monotonic() + max(1.0, spec_rem))
    specs = _spectral_stage(n, best_a, spec_deadline, rng)
    for a, sc in specs:
        if not np.isfinite(sc):
            continue
        inits.append(_proj_nonneg_sum(a))
        if sc < best_score:
            best_score = sc
            best_a = _proj_nonneg_sum(a)

    scored = []
    seen = []
    for a in inits:
        a = _proj_nonneg_sum(a)
        sc = score_fft(a)
        if not np.isfinite(sc):
            continue
        diverse = True
        for b in seen:
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < 0.045:
                diverse = False
                break
        if diverse:
            scored.append((sc, a))
            seen.append(a)
            if len(seen) >= 18:
                break

    scored.sort(key=_pair_score)

    cands = []
    for sc, a in scored:
        cands.append(a)
        if len(cands) >= 4:
            break
    if not cands:
        cands.append(np.ones(n, dtype=np.float64) / float(n))

    opt_deadline = max(time.monotonic(), deadline - 8.0)
    weights = (0.52, 0.24, 0.14, 0.10)
    for i, a in enumerate(cands):
        now = time.monotonic()
        if now + 5.0 >= opt_deadline:
            break
        rem = max(0.0, opt_deadline - now)
        w = weights[i] if i < len(weights) else 0.10
        alloc = max(12.0, min(900.0, rem * w))
        seed = int(rng.integers(0, 2**31 - 1))
        a_opt, sc = optimize_v(a, n, alloc, seed)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_opt
        print("    stage n=%d cand %d score=%.6f" % (n, i, best_score))

    if best_a is not None:
        now = time.monotonic()
        rem = max(0.0, opt_deadline - now)
        if rem > 30.0:
            seed = int(rng.integers(0, 2**31 - 1))
            a_opt, sc = optimize_v(best_a, n, rem, seed)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_opt

    return best_a, float(best_score)

            break
        stage_time = base * frac
        stage_deadline = min(global_deadline - 5.0, now + stage_time)
        print("Stage n=%d budget=%.1fs current_best=%.6f" % (n, stage_time, best_score))
        a_new, sc_new = _process_stage(int(n), stage_deadline, best_a, rng)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_a = a_new
            best_score = sc_new
        print("After stage n=%d best=%.6f" % (n, best_score))

    if best_a is None:
        best_a = np.ones(512, dtype=np.float64)
    best_a = _proj_nonneg_sum(best_a)

    try:
        if len(best_a) <= 6000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Final FFT score=%.6f exact score=%.6f n=%d" %
                  (best_score, exact, len(best_a)))
            if not np.isfinite(exact):
                best_a = np.ones(512, dtype=np.float64)
        else:
            print("Final FFT score=%.6f n=%d (exact omitted for speed)" %
                  (best_score, len(best_a)))
    except Exception as e:
        print("Exact validation skipped: %s" % e)

    return [float(x) for x in best_a]
