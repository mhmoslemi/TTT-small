from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def build_conv_matrix(a, n):
    """Build the matrix A such that A @ b = a * b (convolution)."""
    # A has shape (2n-1, n)
    # A[k, i] = a[i] if 0 <= k-i <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_direction(a, M):
    """Solve the LP to find direction b."""
    n = len(a)
    A = build_conv_matrix(a, n)
    # Maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)
    # Constraints: A @ b <= M (all of them)
    # b >= 0 (handled by bounds)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None

def soft_max_objective(a, tau):
    """Compute soft-max approximation of max(a*a)."""
    conv = np.convolve(a, a)
    # For numerical stability
    max_val = np.max(conv)
    exp_vals = np.exp(tau * (conv - max_val))
    return max_val + np.log(np.sum(exp_vals)) / tau

    conv = np.convolve(a, a)
    n = len(a)
    max_val = np.max(conv)

1. Initialize a (e.g., raised cosine).
2. For each iteration:

        mask = rng.random(n) < 0.1
        if mask.sum() == 0:
            mask[rng.integers(n)] = True
        pert = np.ones(n)
        pert[mask] = 1.0 + rng.normal(0, step_size, mask.sum())
        pert = np.maximum(pert, 0.01)
        new_a = a * pert
        new_a = simplex_project(new_a)
        score = evaluate_sequence(new_a.tolist())
        if score < best_score:
            a = new_a
            best_score = score
        else:
            step_size *= 0.98

    return a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy optimizer: gradient descent on simplex + LP refinement.
    Tries multiple initializations (raised cosine, uniform, edge-peaked, etc.)
    and allocates time budget adaptively.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 10

    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)

    # Generate diverse initializations
    inits = []

    # Raised cosine (bell in middle)
    rc = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(rc / rc.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-peaked (cosine)
    ce = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(ce / ce.sum())

    # Pure sine (zero at edges)
    sn = np.sin(np.pi * x / (n - 1))
    inits.append(sn / sn.sum())

    # Given sequence
    hs = np.array(height_sequence_1, dtype=np.float64)
    hs = np.clip(hs, 0, None)
    if hs.sum() > 1e-10:
        inits.append(hs / hs.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Two-hump
    h2 = np.sin(np.pi * x / n) ** 2 + 0.5 * np.sin(2 * np.pi * x / n) ** 2
    inits.append(np.maximum(h2, 0) / np.maximum(h2, 0).sum())

    best_a = None
    best_score = np.inf

    n_inits = len(inits)
    # Allocate roughly equal time to each initialization
    per_init_time = (budget_s - 60) / n_inits

    for i, init in enumerate(inits):
        phase_deadline = min(total_deadline - 30,
                             time.monotonic() + per_init_time * 1.5)
        if time.monotonic() > phase_deadline:
            break

        print(f"Init {i+1}/{n_inits} (n={n}), time left: {total_deadline - time.monotonic():.0f}s")

        # Phase 1: Gradient optimization
        a_grad, score1 = gradient_optimize(init.copy(), phase_deadline - 30, rng)
        print(f"  Gradient: {score1:.6f} (time left: {total_deadline - time.monotonic():.0f}s)")

        # Phase 2: LP refinement (only if time permits and n is manageable)
        if time.monotonic() < total_deadline - 60 and n <= 5000:
            lp_deadline = min(total_deadline - 30, time.monotonic() + per_init_time * 0.8)
            a_lp, score2 = lp_refine(a_grad.copy(), lp_deadline)
            print(f"  LP: {score2:.6f} (time left: {total_deadline - time.monotonic():.0f}s)")
            if score2 < best_score:
                best_score = score2
                best_a = a_lp
        if score1 < best_score:
            best_score = score1
            best_a = a_grad

        print(f"  Current best: {best_score:.6f}")

    # Try a larger n if time permits
    if time.monotonic() < total_deadline - 120:
        n2 = min(5000, n * 2)
        x2 = np.arange(n2, dtype=np.float64)
        rc2 = np.sin(np.pi * (x2 + 0.5) / n2) ** 2
        init2 = rc2 / rc2.sum()
        print(f"Trying n={n2}, time left: {total_deadline - time.monotonic():.0f}s")
        a_g2, s2 = gradient_optimize(init2, total_deadline - 60, rng)
        print(f"  Gradient (n={n2}): {s2:.6f}")
        if s2 < best_score and n2 <= 5000:
            a_l2, s3 = lp_refine(a_g2, total_deadline - 20)
            print(f"  LP (n={n2}): {s3:.6f}")
            if s3 < best_score:
                best_score = s3
                best_a = a_l2
            elif s2 < best_score:
                best_score = s2
                best_a = a_g2
        elif s2 < best_score:
            best_score = s2
            best_a = a_g2

    # Final polish
    if best_a is not None and time.monotonic() < total_deadline - 5:
        print(f"Polishing... best: {best_score:.6f}")
        best_a, best_score = polish(best_a, total_deadline - 5, rng)

    print(f"FINAL SCORE: {best_score:.6f}")
    return [float(v) for v in best_a]
