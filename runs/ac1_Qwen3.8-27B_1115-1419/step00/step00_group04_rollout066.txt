"""
Flatten-the-autocorrelation optimizer for the step-function objective

    score(a) = 2*n*max(conv(a,a)) / (sum(a))^2

which is scale invariant, so we work with sum(a)=1 and minimize max(conv(a,a)).

Method: smooth the non-smooth peak with a softmax "continuation" (temperature T)
and run a mirror-descent (multiplicative, log-domain) step driven by the exact
gradient of the softmax objective.  The gradient is computed exactly in O(n log n)
with FFTs.  Softmax weights concentrate on the largest autocorrelation lags, whose
gradient is a weighted sum of shifted copies of a; the multiplicative update keeps
the sequence dense (positive) and reshapes it to push probability mass away from the
lag-0-overlap / peak lags, flattening the autocorrelation below the uniform 1/n level.

We run a short multi-start exploration phase (several initializations and one longer
length) then a long deep-optimization phase with a geometric temperature/step schedule
and periodic reheating restarts, always tracking the best candidate by exact
evaluate_sequence calls.  Everything is time-boxed to the budget with a safe fallback.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(2, x))
    return 1 << (x - 1).bit_length()


def _conv_self(a):
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2 * n - 1]


def _grad(p, a):
    n = len(a)
    L = _next_pow2(n + len(p) - 1)
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(P * np.conj(A), L)
    return 2.0 * R[:n]


def _safe_normalize(x, rng):
    x = np.clip(np.asarray(x, dtype=np.float64), 0.0, 1000.0)
    s = x.sum()
    if not np.isfinite(s) or s < 1e-9:
        x = rng.random(x.size)
        s = x.sum()
    return x / s


def _run(a0, n, e0, e1, T0, T1, deadline, half_life):
    """Mirror-descent over a geometric e (step) and T (softness) schedule."""
    a = _safe_normalize(a0, np.random.default_rng(0))
    mu, e = 0.0, e0
    c = _conv_self(a)
    best_max = float(c.max())
    best = a.copy()
    it = 0
    no_improve = 0
    while time.time() < deadline:
        T = max(1e-9, T0 * (T1 / T0) ** (it / float(half_life)))
        e = max(0.05, e0 * (e1 / e0) ** (it / float(half_life)))
        cx = (c - c.max()) / T
        p = np.exp(cx)
        ps = p.sum()
        if not np.isfinite(ps) or ps <= 0.0:
            p = np.zeros(2 * n - 1); p[n - 1] = 1.0
        else:
            p /= ps
        g = _grad(p, a)
        la = np.log(a + 1e-300) - e * g
        la -= la.max()
        a2 = np.exp(la)
        s2 = a2.sum()
        if not np.isfinite(s2) or s2 <= 0.0:
            a2 = a
        else:
            a2 /= s2
        c = _conv_self(a2)
        m = float(c.max())
        if np.isfinite(m) and m < best_max:
            best_max = m
            best = a2.copy()
            no_improve = 0

            a2 = a2 * np.exp(rt.normal(0.0, 0.15, size=n))
            a2 = np.clip(a2, 0.0, 1000.0)
            a2 /= a2.sum()
            no_improve = 0
        a = a2
        it += 1
    return best, best_max
