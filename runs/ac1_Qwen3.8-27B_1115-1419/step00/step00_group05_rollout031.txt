def fft_aconv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]

def get_gradient(conv, a, tau, top_k=32):
    """Compute approximate gradient of soft-max of conv."""
    n = len(a)
    # Get top k indices
    if top_k < len(conv):
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(conv))
    
    # Compute softmax weights
    top_vals = conv[top_idx]
    weights = np.exp((top_vals - top_vals.max()) / tau)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        k = top_idx[i]
        w = weights[i]
        if w < 1e-8:
            continue
        # Contribution: 2*w*a[k-m] for valid m
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start > m_end:
            continue
        j = k - np.arange(m_start, m_end + 1)
        grad[m_start:m_end+1] += 2.0 * w * a[j]
    
    return grad
