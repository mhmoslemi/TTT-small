import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. (f*g)(k) <= M, g >= 0."""
    n = len(f)
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)(k) <= M for k = 0, ..., 2n-2
    # Build the constraint matrix A: A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A[k, i] = f[k-i]
    
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_convolution_matrix(f, n):
    """Build the sparse matrix A such that A @ g = convolve(f, g)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_convolution_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None
