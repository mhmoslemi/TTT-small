def cross_correlation(h, a):
    """Compute h[k] * a[k-i] summed over k, for all i."""
    # This is the cross-correlation of h and a
    # = conv(h_reversed, a) [n-1 : n-1+n]
    n = len(a)
    N = 2 * n  # FFT size (at least 2n-1 for full conv)
    H = np.fft.rfft(h, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: IFFT(conj(H) * A) or IFFT(H * conj(A))
    # corr[i] = sum_k h[k] * a[k-i]
    # = sum_k h[k] * a[k-i]
    # FFT of h: H[f] = sum_k h[k] e^{-2πifk/N}
    # FFT of a: A[f] = sum_j a[j] e^{-2πifj/N}
    # IFFT(H * conj(A))[m] = (1/N) sum_f H[f] conj(A[f]) e^{2πifm/N}
    #   = (1/N) sum_f (sum_k h[k] e^{-2πifk/N}) (sum_j a[j] e^{2πifj/N}) e^{2πifm/N}
    #   = (1/N) sum_k sum_j h[k] a[j] e^{2πif(m-k+j)/N}
    #   = sum_k sum_j h[k] a[j] delta(m-k+j) (by orthogonality, for m in [0,N))
    #   = sum_k h[k] a[m+k] (where a is periodic with period N)
    # So IFFT(H * conj(A))[m] = sum_k h[k] a[m+k] (mod N)
    # We want sum_k h[k] a[k-i] = sum_k h[k] a[k-i]
    # Let j = k-i: sum_j h[j+i] a[j]
    # This is IFFT(H * conj(A))[N-i] (by the formula above, with m = N-i)
    # Actually, let me just use: result = IFFT(conj(H) * A)
    # conj(H) * A: IFFT(conj(H) * A)[m] = sum_k sum_j conj(h[k]) a[j] delta(m+k-j) = sum_k conj(h[k]) a[m+k]
    # For real h: = sum_k h[k] a[m+k]
    # We want sum_k h[k] a[k-i] = sum_j h[j+i]
