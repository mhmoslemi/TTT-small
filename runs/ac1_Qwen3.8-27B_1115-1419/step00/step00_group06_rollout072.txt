def simplex_project(v, s):
    n=v.size
    if s <= 0: return np.zeros(n)
    if v.size == 1: return np.array([s])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0]
    if len(rho)==0: return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def simplex_project(v, s):

    n=v.size
    if n==0: return v
    if s <= 0: return np.zeros(n)
    if n==1: return np.array([s])

    tot=out.sum()
    if tot <= 1e-300: out=np.full(n,s/n)
    else: out=out*(s/tot)

"""
Search for low-score non-negative step-function heights.

Strategy:
- Work with normalized, non-negative sequences (sum = 1), since the score is scale invariant.
- Use an FFT convolution to evaluate 2*n*max(conv(a,a)) efficiently.
- Use an LP-inspired direction from a random/asymmetric start:
    max sum(g) s.t. g >= 0 and (current_sequence * g)_k <= M.
  Blend the LP solution with the current sequence and accept improvements.
- Polish the best sequence with an active-set projected subgradient method on the
  largest convolution peaks.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64).reshape(-1)
    n = u.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.full(n, 1.0 / n, dtype=np.float64)
    u = u - np.max(u)
    u = np.clip(u, -50.0, 50.0)
    ex = np.exp(u)
    s = float(ex.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return ex / s


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.full(64, 1.0 / 64.0, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a / s
    return a


def fft_convolve_self(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    N = 1
    while N < L:
        N <<= 1
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)
    out = out[:L]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def normalized_score(a, conv=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.inf
    if conv is None:
        conv = fft_convolve_self(a)
    if conv.size == 0:
        return np.inf
    mx = float(np.max(conv))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * a.size * mx)


def update_best(holder, seq, score, n=None):
    if not np.isfinite(score):
        return
    arr = np.asarray(seq, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        return
    arr = np.clip(arr, 0.0, 1000.0)
    if n is None:
        n = arr.size
    val = float(score)
    if val < float(holder[0]):
        holder[0] = val
        holder[1] = arr.copy()
        holder[2] = int(n)


def top_weights(c, topK):
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    m = c.size
    if m == 0:
        return np.array([], dtype=int), np.array([], dtype=float)
    K = int(max(1, min(int(topK), m)))
    if K < m:
        idx = np.argpartition(c, m - K)[m - K:]
    else:
        idx = np.arange(m)
    vals = np.maximum(np.asarray(c[idx], dtype=np.float64), 0.0)
    s = float(vals.sum())
    if s <= 1e-300:
        w = np.full(K, 1.0 / K, dtype=np.float64)
    else:
        w = vals / s
    return idx, w


def active_gradient(a, idx, weights):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    idx = np.asarray(idx, dtype=int).reshape(-1)
    weights = np.asarray(weights, dtype=np.float64).reshape(-1)
    for p in range(idx.size):
        w = float(weights[p])
        if w <= 1e-15:
            continue
        k = int(idx[p])
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo <= hi:
            grad[lo:hi + 1] += w * (a[k - hi:k - lo + 1][::-1] + a[lo:hi + 1])
    return grad


def simplex_project(v, s):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    if s <= 0.0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(s)], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if rho.size == 0:
        return np.full(n, s / n, dtype=np.float64)
    rho = int(rho[-1])
    theta = (cssv[rho] - s) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    tot = float(out.sum())
    if tot <= 1e-300:
        out = np.full(n, s / n, dtype=np.float64)
    else:
        out = out * (s / tot)
    return out


def make_initial_sequence(n, rng, variant):
    n = int(n)
    v = int(variant) % 12
    if v == 0:
        u = rng.normal(0.0, 1.0, size=n)
    elif v == 1:
        u = rng.normal(0.0, 2.0, size=n)
    elif v == 2:
        x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
        u = 2.0 * x
    elif v == 3:
        x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
        u = -2.0 * x
    elif v == 4:
        freq = 2.0 + 5.0 * rng.random()
        x = np.linspace(0.0, 2.0 * np.pi * freq, n, dtype=np.float64)
        u = 1.5 * np.cos(x)
    elif v == 5:
        z = rng.standard_normal(n)
        u = 2.0 * 0.5 * (z - z[::-1])
    elif v == 6:
        a = rng.random(n) + 0.05
        m = max(1, n // 4)
        a[:m] *= 2.0
        a[-m:] *= 1.5
        return normalize_nonneg(a)
    elif v == 7:
        a = (rng.random(n) > 0.5).astype(np.float64)
        if a.sum() <= 0.0:
            a[0] = 1.0
        return normalize_nonneg(a)
    elif v == 8:
        a = np.zeros(n, dtype=np.float64)
        m = max(2, n // 5)
        a[:m] = 1.0
        a[n - m:] = 1.0
        return normalize_nonneg(a)
    elif v == 9:
        a = np.linspace(1.0, 0.1, n, dtype=np.float64)
        a += 0.2 * rng.random(n)
        return normalize_nonneg(a)
    elif v == 10:
        a = np.linspace(0.1, 1.0, n, dtype=np.float64)
        a += 0.2 * rng.random(n)
        return normalize_nonneg(a)
    else:
        a = rng.random(n)
        a = np.where(a > 0.6, 2.0, 0.1) * (0.5 + 0.5 * rng.random(n))
        return normalize_nonneg(a)
    u = np.clip(u, -20.0, 20.0)
    return softmax_from_u(u)


def precompute_lp_slices(n):
    rows = []
    for k in range(2 * n - 1):
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        rows.append((k, lo, hi, k - hi, k - lo + 1))
    return rows


def fill_lp_matrix(f, A, slices):
    A[:] = 0.0
    for k, lo, hi, s0, s1 in slices:
        A[k, lo:hi + 1] = f[s0:s1][::-1]


def lp_improve_start(a, deadline, n, best_holder, rng):
    if linprog is None:
        return
    n = int(n)
    if n < 32:
        return

    f = normalize_nonneg(a)
    conv = fft_convolve_self(f)
    M = float(np.max(conv))
    best_M = M
    best_f = f.copy()
    update_best(best_holder, f, 2.0 * n * M, n)

    slices = precompute_lp_slices(n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    max_iter = 110 if n <= 1280 else 55
    no_improve = 0

    for _ in range(max_iter):
        if time.monotonic() > max(time.monotonic(), deadline) - 2.0:
            break

        fill_lp_matrix(f, A, slices)
        improved = False

        for margin in (1.0, 0.97, 0.94):
            Mu = M * margin
            if Mu <= 1e-300:
                continue
            b = np.full(2 * n - 1, Mu, dtype=np.float64)

            res = None
            try:
                res = linprog(obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
            except Exception:
                try:
                    res = linprog(obj, A_ub=A, b_ub=b, bounds=bounds)
                except Exception:
                    res = None

            if res is None or not res.success:
                continue

            g = np.asarray(res.x, dtype=np.float64).reshape(-1)
            if g.size != n or not np.all(np.isfinite(g)):
                continue
            g = np.clip(g, 0.0, None)
            sg = float(g.sum())
            if sg <= 1e-12:
                continue
            g = g / sg

            local_M = M
            local_f = f

            for t in (1.0, 0.9, 0.75, 0.5, 0.35, 0.2, 0.1):
                if t >= 1.0:
                    cand = g
                else:
                    cand = (1.0 - t) * f + t * g
                cand = np.maximum(0.0, cand)
                sc = float(cand.sum())
                if sc <= 1e-12:
                    continue
                cand = cand / sc
                conv_n = fft_convolve_self(cand)
                mn = float(np.max(conv_n))
                if mn < local_M - 1e-12:
                    local_M = mn
                    local_f = cand.copy()

            if local_M < M - 1e-12:
                f = local_f
                M = local_M
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    update_best(best_holder, f, 2.0 * n * M, n)
                if best_M < 1.05:
                    return
                break

        if not improved:
            no_improve += 1
            if no_improve >= 6:
                break
            if no_improve % 2 == 0 or no_improve > 60:
                noise = 0.05 * best_f.mean() * rng.standard_normal(n)
                f = simplex_project(best_f + noise, 1.0)
                conv = fft_convolve_self(f)
                M = float(np.max(conv))
                if M > best_M * 1.1:
                    f = best_f.copy()
                    M = best_M


def run_lp_phase(n, allotted, rng, best_holder, offset=0):
    if linprog is None:
        return
    allotted = float(max(0.0, allotted))
    if allotted < 25.0:
        return
    end = time.monotonic() + allotted
    num_starts = max(1, min(4, int(allotted / 80.0) + 1))
    slice_sec = allotted / float(num_starts)
    for s in range(num_starts):
        if time.monotonic() >= end:
            break
        slice_end = min(end, time.monotonic() + slice_sec + 20.0)
        a = make_initial_sequence(n, rng, offset + 3 * s)
        lp_improve_start(a, slice_end, n, best_holder, rng)
        if float(best_holder[0]) < 1.40:
            break


def gradient_polish(a, deadline, rng, best_holder, n, topK=64):
    n = int(n)
    if n < 8:
        return
    f = normalize_nonneg(a)
    conv = fft_convolve_self(f)
    cur = normalized_score(f, conv)
    best_f = f.copy()
    best_s = cur
    update_best(best_holder, f, cur, n)

    alpha = 0.10
    T = 0.02
    no_improve = 0

    while time.monotonic() < max(time.monotonic(), deadline) - 3.0:
        conv = fft_convolve_self(f)
        cur = normalized_score(f, conv)
        if cur < best_s - 1e-12:
            best_s = cur
            best_f = f.copy()
            update_best(best_holder, f, cur, n)
            if best_s < 1.05:
                return

        idx, w = top_weights(conv, topK)
        grad = active_gradient(f, idx, w)
        var = float(np.std(grad))
        if not np.isfinite(var) or var <= 1e-18:
            no_improve += 1
        else:
            direction = grad / var
            mean_f = float(f.mean())
            prev_cur = cur
            accepted = False

            for factor in (0.5, 1.0, 2.0, 0.25):
                step = alpha * factor
                v = f - step * mean_f * direction
                cand = simplex_project(v, 1.0)
                if np.max(np.abs(cand - f)) <= 1e-15 * max(1.0, mean_f):
                    continue
                conv_c = fft_convolve_self(cand)
                sc = normalized_score(cand, conv_c)
                if not np.isfinite(sc):
                    continue
                if sc < prev_cur + T:
                    f = cand
                    cur = sc
                    accepted = True
                    if sc < best_s - 1e-12:
                        best_s = sc
                        best_f = f.copy()
                        update_best(best_holder, f, cur, n)
                        if best_s < 1.05:
                            return
                    if sc <= prev_cur + 1e-12:
                        alpha = min(3.0, alpha * 1.2)
                        no_improve = 0
                    else:
                        alpha = max(0.005, alpha * 0.9)
                        no_improve += 1
                    break

            if not accepted:
                no_improve += 1
                if no_improve % 5 == 0:
                    alpha = max(0.005, alpha * 0.7)
                    T = max(0.0005, T * 0.95)
                if no_improve % 80 == 0:
                    noise = 0.03 * best_f.mean() * rng.standard_normal(n)
                    f = simplex_project(best_f + noise, 1.0)
                    conv = fft_convolve_self(f)
                    cur = normalized_score(f, conv)
                    T = max(T, 0.02)
                    no_improve = 0
                if no_improve > 400:
                    break


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget = float(budget_s)
    total_deadline = time.monotonic() + max(2.0, budget - 8.0)
    holder = [np.inf, None, None]

    parent = None
    try:
        parent = height_sequence_1
    except NameError:
        parent = None

    if parent is not None:
        arr = np.asarray(parent, dtype=np.float64).reshape(-1)
        if arr.size > 0:
            pn = normalize_nonneg(arr)
            if pn.size > 0:
                sc = normalized_score(pn)
                update_best(holder, pn, sc, pn.size)

    quick_lengths = [1024, 1280]
    if budget > 30.0:
        quick_lengths.append(5582)

    for n in quick_lengths:
        if time.monotonic() > total_deadline - 2.0:
            break
        for v in range(4):
            if time.monotonic() > total_deadline - 2.0:
                break
            a = make_initial_sequence(n, rng, v)
            sc = normalized_score(a)
            update_best(holder, a, sc, n)

    rem = total_deadline - time.monotonic()
    if rem > 250.0:
        run_lp_phase(1024, min(250.0, rem - 100.0), rng, holder, offset=10)

    rem = total_deadline - time.monotonic()
    if rem > 350.0 and float(holder[0]) > 1.50:
        run_lp_phase(1280, min(450.0, rem - 120.0), rng, holder, offset=30)

    rem = total_deadline - time.monotonic()
    if rem > 120.0:
        if holder[1] is not None:
            seq = np.asarray(holder[1], dtype=np.float64).copy()
            n = int(holder[2]) if holder[2] is not None else seq.size
            if float(holder[0]) > 1.99:
                n_use = n if n in (1024, 1280, 5582) else 1280
                seq = make_initial_sequence(n_use, rng, 5)
                n = n_use
        else:
            seq = make_initial_sequence(1280, rng, 5)
            n = 1280
        grad_slice = min(900.0, rem * 0.6)
        gradient_polish(seq, time.monotonic() + grad_slice, rng, holder, n, topK=64)

    for n, off in ((1280, 40), (1024, 60), (5582, 80)):
        if time.monotonic() > total_deadline - 120.0:
            break
        for v in range(3):
            if time.monotonic() > total_deadline - 90.0:
                break
            a = make_initial_sequence(n, rng, off + 3 * v)
            slice_end = min(total_deadline, time.monotonic() + 240.0)
            gradient_polish(a, slice_end, rng, holder, n, topK=64)
            if float(holder[0]) < 1.45:
                break
        if float(holder[0]) < 1.45:
            break

    if holder[1] is not None:
        nbest = int(holder[2]) if holder[2] is not None else holder[1].size
        if time.monotonic() < total_deadline - 25.0:
            gradient_polish(holder[1].copy(), total_deadline - 5.0, rng, holder, nbest, topK=96)

    final = holder[1]
    if final is None:
        final = np.full(512, 1.0, dtype=np.float64)

    final = np.asarray(final, dtype=np.float64).reshape(-1)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s <= 1e-12:
        final = np.full(512, 1.0, dtype=np.float64)
    else:
        final = final / s

    try:
        actual = float(evaluate_sequence([float(x) for x in final]))
    except Exception:
        actual = np.inf

    if not np.isfinite(actual):
        final = np.full(512, 1.0, dtype=np.float64)

    return [float(x) for x in final]
