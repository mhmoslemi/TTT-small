```python
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def solve_lp_direction(f, M):
    """Find g that maximizes sum(g) subject to g >= 0 and (f*g)_k <= M."""
    n = len(f)
    # Build constraint matrix: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_j f[j] * g[k-j] (where g[k-j] = 0 if k-j not in [0,n-1])
    # So row k of A has: A[k, j] = f[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # In matrix form: A[k, j] = f[k-j] if 0 <= k-j <= n-1 else 0
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # g[j] appears in (f*g)_k for k = j, j+1, ..., j+n-1
        # (f*g)_k includes the term f[k-j] * g[j] when 0 <= k-j < n
        # So for fixed j, k ranges from j to j+n-1, and A[k, j] = f[k-j]
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # scipy linprog: min c @ x s.t. A_ub @ x <= b_ub, bounds
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    return res.x
```

    rows = []

        rows = []
        cols = []

"""
LP-directed iterative optimization to minimize 2n*max(a*a)/(sum(a))^2.
Uses active LP to find optimal descent direction, with adaptive step sizes,
gradient-based fallback, and structured perturbation for escaping local minima.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _build_conv_rows_cols(n):
    """Build row/col indices for the (2n-1) x n convolution matrix."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        cnt = j_end - j_start + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(j_start, j_end + 1)
        idx += cnt
    return rows, cols


def _fast_obj(a):
    """Compute objective without calling evaluate_sequence."""
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(np.convolve(a, a)))
    return 2.0 * n * M / (S * S)


def _gradient_smooth_max(a, lam=50.0):
    """Gradient of log-sum-exp(lambda * (a*a)/S^2) scaled to approximate gradient of max/S^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return None
    S2 = S * S
    b = np.convolve(a, a) / S2

    # log-sum-exp for stability
    b_shift = b - b.max()
    weights = np.exp(lam * b_shift)
    weights /= weights.sum()

    # grad of b_k w.r.t. a[i]:
    # b_k = sum_j a[j]*a[k-j]/S^2
    # db_k/da[i] = (a[k-i] + a[i-k]) / S^2 (when indices valid)
    # Full gradient: sum_k weights[k] * (a[k-i] + a[i-k]) / S^2 * (1/S^2 correction for S dependence)
    # Simplify: use unnormalized b and weight by a=S target
    # Actually let's just compute (a*a) gradient directly
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k] * lam / S2
        # db_k/da[i] = a[k-i]/S^2 + a[i-k]/S^2 (valid indices only)
        # But this O(n^2) loop is too slow for n=3809
        # Use FFT-based approach instead
        break

    # FFT-based gradient computation
    # b_k = (a*a)[k]
    # db_k/da[i] = a[k-i] + a[i-k] (with zero-padding)
    # grad_a[i] = sum_k w_k * (a[k-i] + a[i-k]) / S^2
    # = (w * a_reversed)[i] + (w * a)[i]  (with appropriate shifting)
    # More precisely:
    # sum_k w_k * a[k-i] = (w_conv_a_padded)[i] - this is a cross-correlation

    # Let's use a direct approach with the full convolution structure
    # grad_a[i] = (1/S^2) * sum_k w_k * (a[k-i] + a[i-k]) for valid indices
    # = (1/S^2) * [ (flip(w) * a)[i] + (w * a)[i] ]  in circular sense... not quite

    # Simplest: use np.correlate
    # sum_k w_k * a[k-i] = correlate(w, a, 'full')[n-1-i] (approximately)
    # This is getting complicated; let's use a simpler numerical gradient for n < 5000
    # Or use the analytical form with convolutions

    # grad_a[i] = (a * flip_w)[i] + (a * w_shifted)[i] where the * is convolution
    # Actually: sum_k w[k] * a[k-i] where a is defined on [0,n-1]
    # Let A = [a, 0, ..., 0] (padded to 2n-1)
    # sum_k w[k]*a[k-i] = (w * A)[i] where * is convolution... no.

    # Let me just do it the straightforward but vectorized way:
    # For each k, the contribution to grad_a is:
    #   for i in [max(0,k-n+1), min(k, n-1)]: grad_a[i] += w[k] * (a[k-i] + (a[i-k] if i>=k else 0)) / S2
    # This is O(n^2) total but vectorizable per row of the matrix

    # Use the matrix we already have:
    # grad_a = A^T @ (w * lam / S2)  ... wait, that's not quite right
    # A[k,i] = a[k-i] for valid (k,i)
    # (A @ a)[k] = b_k = sum_i a[k-i]*a[i] = (a*a)[k] ✓
    # grad of (a^T A a) / S2... no, b = A @ a (linear in a for FIXED A)
    # But A depends on a! So it's actually quadratic.
    # b_k = A(a)[k] @ a where A(a) is the matrix built from a
    # db_k/da = 2 * A(a) @ a ... no, A(a) is NOT symmetric in the right way.

    # I'll skip the analytical gradient and just use the LP approach.
    return None


def _perturb(a, rng, scale=0.1):
    """Apply structured random perturbation."""
    n = len(a)
    out = a.copy()
    n_blocks = rng.integers(3, min(n, 12))
    block_size = max(1, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + block_size)
        out[start:end] *= float(np.exp(rng.normal(0.0, scale)))
    out = np.maximum(out, 0.0)
    S = float(np.sum(out))
    if S > 0.01:
        out *= np.sqrt(2.0 * n) / S
    return out
