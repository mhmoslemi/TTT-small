def build_conv_matrix(a):
    """Build sparse convolution constraint matrix A where A[k,i] = a[i] if 0<=k-i<n."""
    n = len(a)
    i_list = np.arange(n)
    k_start = i_list
    k_end = np.minimum(2 * n - 2, i_list + n - 1)
    counts = k_end - k_start + 1
    rows = np.repeat(k_start, counts) + np.vstack([np.arange(c) for c in counts]).ravel()
    cols = np.repeat(i_list, counts)
    vals = np.repeat(a, counts)
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_improve(a, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0. Returns b or None."""

                      options={'disp': False})

    return None
