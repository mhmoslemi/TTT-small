"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based local search
and gradient descent, with multiple asymmetric starting points.
Key insight: asymmetric sequences can beat the uniform value of 2.0.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    x = np.sort(v)[::-1]
    cssv = np.cumsum(x)
    rho = np.arange(n, 0, -1)
    cond = x * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = rho[cond][0]
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(v - theta, 0)


def build_constraint_matrix(f):
    """Build sparse (2n-1) x n Toeplitz constraint matrix A where A[k,i] = f[k-i] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            rows.extend(range(k, k + 1) * len(idx))
            cols.extend(idx.tolist())
            vals.extend(f[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 20.0, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = res.x
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def soft_max_grad(f, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv_k)) w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    scaled = tau * conv
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    # grad[j] = 2 * sum_k weights[k] * f[k-j]
    # = 2 * (cross-correlation of weights and f)[j]
    N = 4 * n + 4
    Fw = np.fft.rfft(weights, N)
    Ff = np.fft.rfft(f, N)
    corr = np.fft.irfft(Fw * np.conj(Ff), N)
    grad = 2.0 * corr[:n]
    return grad


def run_lp_search(f, n, deadline):
    """LP-based local search."""
    f = np.maximum(f, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best = f.copy()
    best_score = 2.0 * n * np.max(np.convolve(f, f))
    t = 0.5

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best = f.copy()
            t = min(t * 1.1, 0.7)

        g = solve_lp_direction(f, M)
        if g is None:
            break
        g = g / g.sum()

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        new_M = float(np.max(np.convolve(f_new, f_new)))
        new_score = 2.0 * n * new_M
        if new_score < score:
            f = f_new
        else:
            t = max(0.1, t * 0.6)

    return best, best_score

    f = np.maximum(f, 0).astype(np.float64)
    f = f / f.sum()

    best = f.copy()

        configs = [configs[i] for i in idxs]
