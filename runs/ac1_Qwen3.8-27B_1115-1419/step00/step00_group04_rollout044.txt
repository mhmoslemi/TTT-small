    """Build the sparse convolution matrix for sequences of length n.
    Matrix is (2n-1) x n, where row k represents conv(a,b)[k] = sum_i mat[k,i]*b[i] with mat[k,i] = a[k-i].

    """Precompute the row and column indices for the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp_cutting_plane(f, M, n, row_idx, col_idx, active_idx):
    """Solve max sum(g) s.t. conv(f,g)[k] <= M for k in active_idx, g >= 0."""
    g_var = cp.Variable(n, nonneg=True)
    
    # Extract relevant rows
    r = row_idx[active_idx]
    c = col_idx[active_idx]

"""
LP-based optimization for minimizing the peak of self-convolution.
Uses cvxpy with SCS solver, multiple starting points, and random restarts.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix


def build_conv_structure(n):
    """Build row and column index arrays for the sparse convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_conv_matrix(f, row_idx, col_idx):
    """Build the sparse convolution matrix for a given sequence f."""
    n = len(f)
    vals = f[row_idx - col_idx]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))


def fast_convolve(a, b):
    """Fast full convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def solve_lp(f, M, n, row_idx, col_idx):
    """Solve: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    mat = build_conv_matrix(f, row_idx, col_idx)
    g_var = cp.Variable(n, nonneg=True)
    constraints = [mat @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000, eps=1e-6)
    except:
        return None
    if g_var.value is None:
        return None
    return np.clip(np.asarray(g_var.value).flatten(), 0.0, None)


def compute_subgradient(f, conv_f):
    """Compute a subgradient of max(conv_f) with respect to f."""
    n = len(f)
    M = np.max(conv_f)
    tol = 1e-8 * M
    K = np.where(conv_f >= M - tol)[0]
    if len(K) == 0:
        K = np.array([np.argmax(conv_f)])
    # Pick a random subset of maximizers for efficiency
    if len(K) > 20:
        K = K[np.random.choice(len(K), 20, replace=False)]
    grad = np.zeros(n)
    for k in K:
        # Contribution from first factor: f[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end+1] += f[k - np.arange(j_start, j_end+1)]
        # Contribution from second factor: f[j-k] for valid j
        j_start2 = max(k, 0)
        j_end2 = min(k + n - 1, n - 1)
        if j_end2 >= j_start2:
            grad[j_start2:j_end2+1] += f[np.arange(j_start2, j_end2+1) - k]
    grad /= len(K)
    return grad
