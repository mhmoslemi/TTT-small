    subject to: convolve(a, b)[k] <= M for all k
               b >= 0

    # Actually, convolve(a, b)[k] = sum_j a[j] * b[k-j] where k-j must be in [0, n-1]

    One step of the LP-based multiplicative weights algorithm.
    Returns updated a and the score.
    """
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    score = 2 * n * M / (np.sum(a) ** 2)

    c = -np.ones(n)
    C = build_conv_matrix(a)

    a_new = np.maximum(a_new, 0)

One known approach: use a sequence whose DFT has approximately constant magnitude. For a real sequence a of length n, the DFT is A_k = sum_j a_j * exp(-2πijk/n). import numpy as np
import time
from scipy.optimize import linprog


"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based multiplicative weights (follow-the-leader) iterations
2. Adaptive step size with patience-based restart
3. Multiple sequence lengths and initializations
4. Final local search polish phase
"""


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C where C @ b = convolve(a, b)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_solve(a, M):
    """Solve LP: max sum(b) s.t. convolve(a,b) <= M elementwise, b >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_score(a):
    """Compute J(a) assuming sum(a)=1."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(conv.max())
    return 2.0 * n * M


def lp_iterations(a, deadline, min_time=10.0):
    """Run LP-based multiplicative weights iterations on sequence a."""
    n = len(a)
    eta = 0.5
    stuck = 0
    best_local_score = compute_score(a)
    best_local_a = a.copy()
    iters = 0

    while time.monotonic() < deadline - min_time:
        iters += 1
        score = compute_score(a)

        if score < best_local_score - 1e-10:
            best_local_score = score
            best_local_a = a.copy()
            stuck = 0
            eta = min(0.9, eta * 1.03)
        else:
            stuck += 1
            eta = max(0.003, eta * 0.96)

        if stuck > 300:
            break

        conv = np.convolve(a, a)
        M = float(conv.max())

        b = lp_solve(a, M)
        if b is None:
            break

        sum_b = float(b.sum())
        if sum_b < 1e-14:
            break

        b = b / sum_b
        a = (1.0 - eta) * a + eta * b
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-14:
            a = best_local_a.copy()
            break
        a = a / s

    return a, best_local_a, best_local_score, iters


def local_search(a, deadline, min_time=15.0):
    """Local search polish using coordinate and block moves."""
    n = len(a)
    rng = np.random.default_rng(12345)
    score = compute_score(a)
    best_a = a.copy()
    best_score = score
    temp = 0.05
    no_improve = 0

    while time.monotonic() < deadline - min_time:
        moved = False

        # Try coordinate moves
        for _ in range(n * 2):
            i = int(rng.integers(0, n))
            a_try = a.copy()
            delta = a[i] * (rng.normal(0, temp))
            a_try[i] = max(0.0, a[i] + delta)
            s = float(a_try.sum())
            if s < 1e-12:
                continue
            a_try = a_try / s
            sc = compute_score(a_try)
            if sc < score - 1e-12:
                a = a_try
                score = sc
                moved = True
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                break

        # Try block moves
        width = int(rng.integers(max(2, n // 64), max(3, n // 8)))
        start = int(rng.integers(0, n - width + 1))
        a_try = a.copy()
        a_try[start:start + width] *= np.exp(rng.normal(0, temp, size=width))
        a_try = np.maximum(a_try, 0)
        s = float(a_try.sum())
        if s > 1e-12:
            a_try = a_try / s
            sc = compute_score(a_try)
            if sc < score - 1e-12:
                a = a_try
                score = sc
                moved = True
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
            else:
                no_improve += 1

        if not moved:
            no_improve += 1

        if no_improve > 200:
            temp = max(0.001, temp * 0.8)
            if temp <= 0.0015:
                break
            no_improve = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses LP-based multiplicative weights followed by
    local search to minimize J(a) = 2*n*max(convolve(a,a))/(sum(a))^2.
    """
    deadline = time.monotonic() + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)

    best_sequence = None
    best_score = np.inf

    # Phase 1: LP-based multiplicative weights
    n_values = [256, 512, 1024]
    lp_deadline = time.monotonic() + float(budget_s) * 0.65

    for n in n_values:
        if time.monotonic() > lp_deadline:
            break

        max_restarts = 4
        for restart in range(max_restarts):
            if time.monotonic() > lp_deadline:
                break

            if restart == 0:
                a = np.ones(n) / n
            elif restart == 1:
                a = rng.dirichlet(np.full(n, 0.3))
            elif restart == 2:
                a = rng.dirichlet(np.full(n, 3.0))
            else:
                # Frequency-domain initialization: flat magnitude spectrum
                phases = rng.uniform(0, 2 * np.pi, n)
                a = np.abs(np.fft.ifft(np.exp(1j * phases))).real
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a = np.ones(n) / n
                else:
                    a = a / s

            a, a_frozen, local_score, iters = lp_iterations(a, lp_deadline, min_time=30.0)

            if local_score < best_score:
                best_score = local_score
                best_sequence = a_frozen.copy()

            # Also check the endpoint
            end_score = compute_score(a)
            if end_score < best_score:
                best_score = end_score
                best_sequence = a.copy()

            print(f"n={n}, restart={restart}: iters={iters}, local={local_score:.6f}, global={best_score:.6f}")

    # Phase 2: Local search polish
    if best_sequence is not None:
        polish_deadline = time.monotonic() + max(60.0, float(budget_s) * 0.3 - 30.0)
        polished, polished_score = local_search(best_sequence.copy(), polish_deadline)
        if polished_score < best_score:
            best_score = polished_score
            best_sequence = polished.copy()
        print(f"After polish: {best_score:.6f}")

    # Phase 3: Try a few more LP runs from the best solution with different n
    if best_sequence is not None and time.monotonic() < deadline - 30:
        best_n = len(best_sequence)
        # Try interpolating or truncating to different lengths
        for target_n in [best_n // 2, best_n * 2]:
            if target_n < 32 or target_n > 4096:
                continue
            if time.monotonic() > deadline - 15:
                break
            # Resample best sequence to new length
            if target_n < best_n:
                a_new = np.interp(
                    np.linspace(0, 1, target_n),
                    np.linspace(0, 1, best_n),
                    best_sequence
                )
            else:
                a_new = best_sequence.copy()
                # Pad with small values
                pad = int((target_n - best_n) // 2)
                a_new = np.concatenate([np.zeros(pad), a_new, np.zeros(target_n - best_n - pad)])
                # Or extend by repeating with decay
                ext = int(target_n - best_n)
                decay = np.exp(np.linspace(0, -2, ext)) * best_sequence[-1] * 0.1
                a_new = np.concatenate([a_new, decay[:ext]])
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
            a2, a2f, sc2, _ = lp_iterations(a_new, deadline, min_time=15.0)
            if sc2 < best_score:
                best_score = sc2
                best_sequence = a2f.copy()
            print(f"Resampled n={target_n}: score={sc2:.6f}, global={best_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = compute_score(best_sequence)
    print(f"FINAL: n={len(best_sequence)}, score={final_score:.6f}")

    return [float(x) for x in best_sequence]
