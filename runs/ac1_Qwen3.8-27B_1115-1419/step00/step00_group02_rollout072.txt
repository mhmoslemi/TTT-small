def run_optimizer(x0, max_time=5.0, lr_factor=0.03, top_k=10, weight_power=4.0, momentum=0.5, rng=None, use_full=False, tau_fraction=0.05):
    x = normalize_sequence(np.asarray(x0,float), x0.size)
    n=x.size
    best_x=x.copy(); best_score=fast_score(x)
    lr=lr_factor*n
    v=np.zeros(n)
    beta=momentum
    start=time.monotonic(); deadline=start+max_time
    last_improve=start
    pert_count=0
    it=0
    while time.monotonic() < deadline:
       c=autocorr_fft(x)
       M=float(c.max())
       score=2.0*M/n
       if score < best_score - 1e-12:
          best_score=score; best_x=x.copy(); last_improve=time.monotonic()
       now=time.monotonic()
       if now - last_improve > max(1.0, 0.1*max_time):
          x = perturb_sequence(x, rng)
          v *= 0.5
          lr = max(0.005*n, lr*0.8)
          last_improve = now
          pert_count +=1
          if pert_count > 20:
             # more aggressive restart? maybe reduce lr further
             lr = max(0.001*n, lr*0.8)
       if use_full and n <= 2048:
           tau = max(M * tau_fraction, 1e-8)
           grad, M = full_smooth_grad(x, c, tau)
       else:
           grad, M = top_smooth_grad(x, c, top_k, weight_power)
       gf = (2.0/n)*grad - (4.0*M)/(n*n)
       if beta > 0:
          v = beta*v + (1-beta)*gf
          step = lr*v
       else:
          step = lr*gf
       x = x - step
       x = np.clip(x, 0.0, 1000.0)
       s=float(x.sum())
       if s < 1e-12:
          x=np.ones(n); s=n
       x *= n/s
       it +=1
    # final best already

```python
"""
Multi-scale FFT subgradient search for non-negative step functions minimizing
2*n*max(convolve(a,a))/sum(a)^2.

The method uses smooth/asymmetric initial profiles (steps, Fourier sums,
squared Fourier sums, monotone block profiles), a small-profile search phase,
medium/large refinement with FFT autodivision and smoothed subgradients,
occasional perturbations to escape stationary/uniform basin, and final exact
validation via evaluate_sequence when safe.
"""

import numpy as np
import time

MAX_RETURN_N = 8192


def first_key(item):
    return item[0]


def next_power_of_two(size):
    size = int(size)
    if size <= 1:
        return 1
    return 1 << (size - 1).bit_length()


def autocorr_fft(a):
    """Linear self-convolution using FFT: convolve(a, a)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = next_power_of_two(size)
    fa = np.fft.rfft(a, fft_size)
    ca = np.fft.irfft(fa * fa, fft_size)[:size]
    ca = np.maximum(ca, 0.0)
    ca[~np.isfinite(ca)] = 0.0
    return ca


def fast_score(x):
    """Fast score using FFT convolution. Matches evaluator for valid nonnegative x."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return np.inf
    s = float(np.sum(x))
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = autocorr_fft(x)
    if c.size == 0:
        return np.inf
    M = float(np.max(c))
    if not np.isfinite(M) or M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (s * s))


def repair_sequence_sum(x, target_sum):
    """Increase x (within [0,1000]) until sum is target_sum, if possible."""
    x = np.asarray(x, dtype=np.float64).copy()
    x = np.clip(x, 0.0, 1000.0)
    need = float(target_sum) - float(x.sum())
    if need <= 1e-12:
        return x
    for _ in range(12):
        room = 1000.0 - x
        tr = float(room.sum())
        if tr <= 0.0 or need <= 1e-12:
            break
        x += need / tr
        x = np.clip(x, 0.0, 1000.0)
        need = float(target_sum) - float(x.sum())
    if need > 1e-9:
        idx = np.argsort(x)
        for i in idx:
            if need <= 0.0:
                break
            can = 1000.0 - float(x[i])
            if can <= 0.0:
                continue
            if can >= need:
                x[i] += need
                need = 0.0
            else:
                need -= can
                x[i] = 1000.0
    return x


def normalize_sequence(x, target_sum):
    """
    Return a copy of x with values in [0, 1000], nonnegative, finite,
    and sum as close as possible to target_sum under the cap.
    """
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1:
        x = x.ravel()
    x = x.astype(np.float64).copy()
    n = x.size
    target_sum = float(target_sum)
    if n == 0:
        return x
    x[~np.isfinite(x)] = 0.0
    x = np.clip(x, 0.0, 1000.0)
    s = float(x.sum())
    if s <= 1e-14:
        x.fill(target_sum / float(n))
        return np.clip(x, 0.0, 1000.0)

    scale = target_sum / s
    if scale > 1.0:
        mx = float(x.max())
        if mx > 0.0:
            cap_scale = 1000.0 / mx
            if scale > cap_scale:
                x *= cap_scale
                x = np.clip(x, 0.0, 1000.0)
                return repair_sequence_sum(x, target_sum)

    x *= scale
    x = np.clip(x, 0.0, 1000.0)
    s = float(x.sum())
    tol = max(1e-9, 1e-9 * abs(target_sum))
    if s + tol < target_sum:
        x = repair_sequence_sum(x, target_sum)
    elif s - tol > target_sum:
        if s > 0.0:
            x *= target_sum / s
            x = np.clip(x, 0.0, 1000.0)
    return x


def interpolate_sequence(p, n):
    """Linearly interpolate a 1D sequence p to length n."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if p.size == 0:
        return np.ones(n, dtype=np.float64)
    if p.size == n:
        return p.copy()
    if p.size == 1:
        return np.full(n, float(p[0]), dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, p.size)
    x1 = np.linspace(0.0, 1.0, n)
    return np.interp(x1, x0, p).astype(np.float64)


def make_random_profile(n, rng):
    """Random smooth nonnegative profile."""
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    K = int(rng.integers(1, 9))
    amp = float(rng.uniform(0.25, 0.95))
    base = float(rng.uniform(0.55, 1.35))
    f = np.full(n, base, dtype=np.float64)
    for k in range(1, K + 1):
        a = amp / np.sqrt(float(k))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        f += a * np.cos(2.0 * np.pi * k * t + ph)
        if rng.random() < 0.65:
            ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
            f += 0.65 * a * np.sin(2.0 * np.pi * k * t + ph2)
    style = int(rng.integers(0, 3))
    if style == 0:
        f = np.clip(f, 0.0, None)
    elif style == 1:
        f = f * f + 0.05
    else:
        f = np.exp(np.clip(f, -2.0, 2.0))
    f[~np.isfinite(f)] = 0.0
    return np.clip(f, 0.0, 1000.0)


def generate_profiles(n, rng, existing=None):
    """Generate a diverse set of starting profiles of length n."""
    n = int(n)
    if n <= 0:
        return []
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    out = []

    # Two-level step profiles. These are known to beat uniform (score ~1.777).
    for alpha in (0.30, 0.40, 0.50, 0.60, 0.70):
        m = int(round(alpha * n))
        if m <= 0:
            m = 1
        if m >= n:
            m = n - 1
        for ratio in (1.5, 2.0, 2.5, 3.0):
            x = np.ones(n, dtype=np.float64)
            x[:m] = ratio
            out.append(x)

    # Cosine-like asymmetric modulations.
    for freq in (1, 2):
        for eps in (0.35, 0.65, 0.95):
            for phase in (0.0, 0.8, 1.6, 2.4, 3.14):
                f = (1.0 + eps * np.cos(np.pi * freq * t + phase)
                     + 0.35 * eps * np.sin(np.pi * freq * t + phase))
                out.append(np.clip(f, 0.0, None))

    # Squared cosine profiles, automatically nonnegative.
    for freq in (1, 2):
        for eps in (0.4, 0.7, 1.0):
            for phase in (0.0, 1.0, 2.1, 3.0):
                g = 1.0 + eps * np.cos(np.pi * freq * t + phase)
                out.append(g * g + 0.05)

    # Monotone rational/exponential profiles.
    for lam in (1, 2, 4, 8):
        out.append(1.0 / (1.0 + lam * t) + 0.1)
        out.append(np.exp(-lam * t) + 0.2)
        out.append(np.exp(lam * t) + 0.1)

    # Random smooth profiles.
    for _ in range(12):
        out.append(make_random_profile(n, rng))

    # Random monotone step/block profiles.
    for _ in range(24):
        K = int(rng.integers(2, max(3, min(n, 16)) + 1))
        if K < 2:
            K = 2
        kind = rng.random()
        lo = float(rng.uniform(0.1, 0.7))
        hi = float(rng.uniform(1.3, 3.5))
        if kind < 0.35:
            levels = np.linspace(hi, lo, K)
        elif kind < 0.6:
            levels = np.geomspace(hi, lo, K)
        else:
            levels = np.sort(rng.uniform(0.1, 3.5, K))[::-1]
        levels = np.clip(levels, 0.05, 5.0)
        base_count = n // K
        counts = np.full(K, base_count, dtype=int)
        extra = n - base_count * K
        if extra > 0:
            counts[:extra] += 1
        out.append(np.repeat(levels, counts).astype(np.float64))

    # Use the provided state-of-the-art sequence if available, mildly distorted.
    if existing is not None:
        h = np.asarray(existing, dtype=np.float64).ravel()
        if h.size > 0:
            h[~np.isfinite(h)] = 0.0
            h = np.clip(h, 0.0, 1000.0)
            if h.size != n:
                h = interpolate_sequence(h, n)
            for slope in (-1.5, -0.7, 0.7, 1.5):
                y = h * (1.0 + slope * t)
                out.append(np.clip(y, 0.0, 1000.0))
            y = h + rng.normal(0.0, 0.1 * max(1.0, float(h.max())), n)
            out.append(np.clip(y, 0.0, 1000.0))

    return out


def top_smooth_grad(x, c, top_k=10, weight_power=4.0):
    """
    Smoothed subgradient of max(c) using the top peaks.
    Returns (grad, M).
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = x.size
    L = c.size
    grad = np.zeros(n, dtype=np.float64)
    if n == 0 or L == 0:
        return grad, 0.0
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return grad, M

    k = int(top_k)
    if k < 1:
        k = 1
    if k >= L:
        idx = np.arange(L)
        vals = c
    else:
        idx = np.argpartition(c, -k)[-k:]
        vals = c[idx]

    rel = np.clip(vals / max(M, 1e-300), 0.0, 1.0)
    w = np.power(rel, float(weight_power))
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.ones(k, dtype=np.float64) / float(k)
    else:
        w = w / sw

    for pos, wj in zip(idx, w):
        pos = int(pos)
        start = pos - (n - 1)
        if start < 0:
            start = 0
        end = pos + 1
        if end > n:
            end = n
        if start >= end:
            continue
        x_start = pos - (end - 1)
        x_end = pos - start + 1
        # Align x indices decreasing with start:end increasing.
        grad[start:end] += 2.0 * float(wj) * x[x_start:x_end][::-1]

    return grad, M


def full_smooth_grad(x, c, tau_fraction=0.05):
    """
    Smaller-n smoothed gradient using a log-sum-exp style weight over all c.
    Returns (grad, M).
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = x.size
    L = c.size
    grad = np.zeros(n, dtype=np.float64)
    if n == 0 or L == 0:
        return grad, 0.0
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return grad, M

    tau = max(float(tau_fraction) * M, 1e-8)
    rel = (c - M) / tau
    w = np.exp(rel)
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        k = int(np.argmax(c))
        w = np.zeros(L, dtype=np.float64)
        w[k] = 1.0
    else:
        w = w / sw

    size = 3 * n - 2
    if size < 1:
        size = 1
    fft_size = next_power_of_two(size)
    xrev = x[::-1]
    Fx = np.fft.rfft(xrev, fft_size)
    Fw = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(Fx * Fw, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    grad[~np.isfinite(grad)] = 0.0
    return grad, M


def perturb_sequence(x, rng):
    """Random multiplicative/additive perturbation, preserving normalization."""
    x = np.asarray(x, dtype=np.float64).ravel().copy()
    n = x.size
    if n <= 0:
        return x
    if rng is None:
        rng = np.random.default_rng(12345)
    y = x.copy()

    if n > 1 and rng.random() < 0.45:
        width = max(1, min(n - 1, int(n * float(rng.uniform(0.005, 0.08)))))
        start = int(rng.integers(0, n - width))
        factor = float(np.exp(rng.normal(0.0, 0.25)))
        y[start:start + width] *= factor

    sigma = max(0.05 * float(np.std(y)), 1.0 / float(max(n, 1)))
    y = y + rng.normal(0.0, sigma, n)
    y = np.clip(y, 0.0, 1000.0)
    return normalize_sequence(y, float(n))


def run_optimizer(x0, max_time=10.0, lr_factor=0.03, top_k=12,
                  weight_power=2.0, momentum=0.5, rng=None,
                  use_full=False, tau_fraction=0.05):
    """
    Subgradient/projected/fitted optimizer for a fixed length.
    Returns (best_x, best_internal_score).
    """
    x = normalize_sequence(np.asarray(x0, dtype=np.float64).ravel(), float(x0.size))
    n = x.size
    if n == 0:
        return x, np.inf

    best_x = x.copy()
    best_score = fast_score(x)
    if not np.isfinite(best_score):
        x = np.ones(n, dtype=np.float64)
        best_score = fast_score(x)
        best_x = x.copy()

    y = np.zeros(n, dtype=np.float64)
    beta = float(momentum)
    lr = float(max(1e-8, lr_factor) * n)
    start = time.monotonic()
    deadline = start + max(0.1, float(max_time))
    last_improve = start
    broad_perturbs = 0
    side_pollute = 0

    while time.monotonic() < deadline:
        S = float(x.sum())
        if not np.isfinite(S) or S < 0.01:
            x = np.ones(n, dtype=np.float64)
            S = float(n)

        c = autocorr_fft(x)
        M = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(M) or M < 0.0:
            M = 0.0
        score = float(2.0 * n * M / (S * S))

        now = time.monotonic()
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_x = x.copy()
            last_improve = now

        if now - last_improve > max(1.0, 0.12 * float(max_time)):
            x = perturb_sequence(x, rng)
            y *= 0.5
            lr = max(0.005 * n, lr * 0.82)
            last_improve = now
            broad_perturbs += 1
            if broad_perturbs > 18:
                lr = max(0.001 * n, lr * 0.85)

        if use_full and n <= 2048:
            grad, M = full_smooth_grad(x, c, tau_fraction)
        else:
            grad, M = top_smooth_grad(x, c, top_k, weight_power)

        S3 = S * S * S
        gf = (2.0 * n / (S * S)) * grad - (4.0 * n * M) / S3
        gf[~np.isfinite(gf)] = 0.0

        if beta > 0.0:
            y = beta * y + (1.0 - beta) * gf
            step = lr * y
        else:
            step = lr * gf

        # Limit per-coordinate change for numerical stability.
        step = np.clip(step, -0.5, 0.5)
        x = x - step
        x = np.clip(x, 0.0, 1000.0)
        S = float(x.sum())
        if not np.isfinite(S) or S < 1e-12:
            x = np.ones(n, dtype=np.float64)
            S = float(n)
        x *= n / S
        x = np.clip(x, 0.0, 1000.0)

        side_pollute += 1
        if side_pollute > 250000:
            break

    # Score final iterate in case it improved on the last step.
    final_score = fast_score(x)
    if np.isfinite(final_score) and final_score < best_score - 1e-12:
        best_score = final_score
        best_x = x.copy()

    return best_x, best_score


def add_candidate(candidates, x, score, limit=60):
    if x is None:
        return candidates
    score = float(score)
    if not np.isfinite(score):
        return candidates
    x = np.asarray(x, dtype=np.float64).ravel()
    x[~np.isfinite(x)] = 0.0
    x = np.clip(x, 0.0, 1000.0)
    candidates.append([score, x.copy()])
    if len(candidates) > limit:
        candidates.sort(key=first_key)
        del candidates[limit:]
    return candidates


def search_at_scale(n, rng, existing, time_budget, candidates,
                    seed_sequences=None, top_count=6, opt_seconds=10.0,
                    lr_factor=0.03, top_k=12, weight_power=2.0,
                    momentum=0.5, use_full=False, tau_fraction=0.05,
                    extra_random=None):
    """
    Generate profiles at length n, rank them by fast score, then optimize
    priority seeds and top profiles until the local time budget expires.
    Returns best candidate at this scale.
    """
    n = int(n)
    if n <= 0:
        return None, np.inf

    start = time.monotonic()
    deadline = start + max(0.5, float(time_budget))
    best_x = None
    best_score = np.inf

    scored = []
    profs = generate_profiles(n, rng, existing)
    for p in profs:
        x = normalize_sequence(p, n)
        sc = fast_score(x)
        if np.isfinite(sc):
            scored.append([sc, x])

    if extra_random is None:
        if n <= 512:
            extra_random = 300
        elif n <= 1024:
            extra_random = 200
        elif n <= 2048:
            extra_random = 100
        elif n <= 8192:
            extra_random = 30
        else:
            extra_random = 10

    count = 0
    while count < int(extra_random) and time.monotonic() < deadline:
        p = make_random_profile(n, rng)
        x = normalize_sequence(p, n)
        sc = fast_score(x)
        if np.isfinite(sc):
            scored.append([sc, x])
        count += 1

    scored.sort(key=first_key)

    # Priority seeds from lower scales / supplied bests.
    priority = []
    if seed_sequences is not None:
        for s in seed_sequences:
            if s is None:
                continue
            arr = np.asarray(s, dtype=np.float64).ravel()
            if arr.size == 0:
                continue
            arr[~np.isfinite(arr)] = 0.0
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size != n:
                arr = interpolate_sequence(arr, n)
            arr = np.clip(arr, 0.0, 1000.0)
            ax = normalize_sequence(arr, n)
            sc = fast_score(ax)
            priority.append(ax)
            candidates = add_candidate(candidates, ax, sc, 80)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_x = ax.copy()

    # Run priority seeds first.
    if priority:
        pcount = len(priority)
        for i in range(pcount):
            if time.monotonic() >= deadline:
                break
            rem_tasks = max(1, pcount - i + min(top_count, len(scored)))
            alloc = max(0.5, (deadline - time.monotonic()) / float(rem_tasks))
            alloc = min(alloc, float(opt_seconds))
            y, ys = run_optimizer(priority[i], max_time=alloc,
                                  lr_factor=lr_factor, top_k=top_k,
                                  weight_power=weight_power,
                                  momentum=momentum, rng=rng,
                                  use_full=use_full,
                                  tau_fraction=tau_fraction)
            if np.isfinite(ys) and ys < best_score:
                best_score = ys
                best_x = y.copy()
            candidates = add_candidate(candidates, y, ys, 80)

    # Run best generated/random profiles.
    run_top = max(0, int(top_count))
    if run_top > 0 and scored:
        for i in range(min(run_top, len(scored))):
            if time.monotonic() >= deadline:
                break
            rem = max(1, min(run_top, len(scored)) - i)
            alloc = max(0.5, (deadline - time.monotonic()) / float(rem))
            alloc = min(alloc, float(opt_seconds))
            sc0, x0 = scored[i]
            # Skip obviously terrible if we already have a good one? Keep a little diversity.
            if np.isfinite(best_score) and sc0 > best_score * 1.15 and i > 2:
                continue
            y, ys = run_optimizer(x0, max_time=alloc,
                                  lr_factor=lr_factor, top_k=top_k,
                                  weight_power=weight_power,
                                  momentum=momentum, rng=rng,
                                  use_full=use_full,
                                  tau_fraction=tau_fraction)
            if np.isfinite(ys) and ys < best_score:
                best_score = ys
                best_x = y.copy()
            candidates = add_candidate(candidates, y, ys, 80)

    return best_x, best_score


def get_existing_sequence():
    obj = globals().get("height_sequence_1")
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr[~np.isfinite(arr)] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)
    if float(arr.sum()) < 0.01:
        return None
    return arr


def finalize_sequence(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    x[~np.isfinite(x)] = 0.0
    x = np.clip(x, 0.0, 1000.0)
    if float(x.sum()) < 0.01:
        x = np.ones(64, dtype=np.float64)
    if x.size > MAX_RETURN_N:
        x = interpolate_sequence(x, MAX_RETURN_N)
        x[~np.isfinite(x)] = 0.0
        x = np.clip(x, 0.0, 1000.0)
    return [float(v) for v in x]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point. Searches for a nonnegative coefficient sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2 within budget_s seconds.
    """
    rng = np.random.default_rng(int(seed))
    total = max(1.0, float(budget_s))
    start = time.monotonic()

    # Quick fallback if absurd budget.
    if total < 25.0:
        fb = np.array([2.0, 2.0, 1.0, 1.0], dtype=np.float64)
        fb = normalize_sequence(fb, 4.0)
        return finalize_sequence(fb)

    # Reserve time for final validation / exact evaluation.
    reserve = 90.0 if total < 900.0 else 120.0
    if total < reserve + 60.0:
        reserve = total * 0.2
    hard_end = start + total
    deadline = start + total - reserve
    if deadline < start + 10.0:
        deadline = start + max(5.0, total * 0.75)

    existing = get_existing_sequence()
    candidates = []

    # Guaranteed decent fallbacks.
    fb4 = np.array([2.0, 2.0, 1.0, 1.0], dtype=np.float64)
    fb4 = normalize_sequence(fb4, 4.0)
    candidates = add_candidate(candidates, fb4, fast_score(fb4), 80)

    fb2048 = np.ones(2048, dtype=np.float64)
    fb2048[:1024] = 2.0
    fb2048 = normalize_sequence(fb2048, 2048.0)
    candidates = add_candidate(candidates, fb2048, fast_score(fb2048), 80)

    best_x = fb2048.copy()
    best_score = fast_score(fb2048)

    if best_x is None or best_score > 2.0:
        best_x = fb2048
        best_score = 2.0

    # Choose scales.
    if total >= 1800.0:
        n_small, n_mid, n_main, n_extra = 512, 2048, 8192, 16384
    elif total >= 900.0:
        n_small, n_mid, n_main, n_extra = 512, 1024, 4096, 8192
    elif total >= 400.0:
        n_small, n_mid, n_main, n_extra = 256, 1024, 2048, 4096
    else:
        n_small, n_mid, n_main, n_extra = 128, 512, 1024, 2048

    main_window = max(1.0, deadline - start)
    phase_a_time = min(120.0, max(3.0, 0.055 * main_window))
    phase_b_time = min(360.0, max(4.0, 0.15 * main_window))
    phase_c_time = min(1300.0, max(10.0, 0.55 * main_window))
    phase_d_time = min(300.0, max(2.0, 0.12 * main_window))

    # Phase A: small profile search, full smooth gradient helps find shape.
    seeds = [existing] if existing is not None else None
    small_best, small_score = search_at_scale(
        n_small, rng, existing, phase_a_time, candidates, seed_sequences=seeds,
        top_count=8, opt_seconds=8.0, lr_factor=0.045, top_k=8,
        weight_power=4.0, momentum=0.5, use_full=True, tau_fraction=0.05
    )
    if small_best is not None and np.isfinite(small_score) and small_score < best_score:
        best_x = small_best.copy()
        best_score = small_score
    print(f"[small {n_small}] best={small_score:.6f}", flush=True)

    # Phase B: medium refinement.
    seeds = []
    if small_best is not None:
        seeds.append(small_best)
    if existing is not None:
        seeds.append(existing)
    mid_best, mid_score = search_at_scale(
        n_mid, rng, existing, phase_b_time, candidates, seed_sequences=seeds,
        top_count=5, opt_seconds=60.0, lr_factor=0.04, top_k=12,
        weight_power=2.0, momentum=0.5, use_full=True, tau_fraction=0.06
    )
    if mid_best is not None and np.isfinite(mid_score) and mid_score < best_score:
        best_x = mid_best.copy()
        best_score = mid_score
    print(f"[medium {n_mid}] best={mid_score:.6f}, global={best_score:.6f}", flush=True)

    # Phase C: main large refinement.
    seeds = []
    if mid_best is not None:
        seeds.append(mid_best)
    if small_best is not None:
        seeds.append(small_best)
    if existing is not None:
        seeds.append(existing)
    main_best, main_score = search_at_scale(
        n_main, rng, existing, phase_c_time, candidates, seed_sequences=seeds,
        top_count=3, opt_seconds=450.0, lr_factor=0.03, top_k=16,
        weight_power=1.5, momentum=0.5, use_full=False, tau_fraction=0.05,
        extra_random=20 if n_main <= 4096 else 12
    )
    if main_best is not None and np.isfinite(main_score) and main_score < best_score:
        best_x = main_best.copy()
        best_score = main_score
    print(f"[main {n_main}] best={main_score:.6f}, global={best_score:.6f}", flush=True)

    # Phase D: optional extra-large exploration, then downsample for return.
    if n_extra and time.monotonic() < deadline:
        dtime = min(phase_d_time, max(2.0, deadline - time.monotonic()))
        if dtime > 30.0:
            seeds = []
            if main_best is not None:
                seeds.append(main_best)
            elif mid_best is not None:
                seeds.append(mid_best)
            elif small_best is not None:
                seeds.append(small_best)
            extr_best, extr_score = search_at_scale(
                n_extra, rng, existing, dtime, candidates, seed_sequences=seeds,
                top_count=1, opt_seconds=dtime * 0.8, lr_factor=0.022,
                top_k=16, weight_power=1.2, momentum=0.45, use_full=False,
                tau_fraction=0.05, extra_random=8
            )
            if extr_best is not None and np.isfinite(extr_score) and extr_score < best_score:
                best_x = extr_best.copy()
                best_score = extr_score
                print(f"[extra {n_extra}] best={extr_score:.6f}, global={best_score:.6f}", flush=True)

    # Final polish of global best.
    while time.monotonic() < deadline - 2.0:
        chunk = min(120.0, max(2.0, deadline - time.monotonic() - 2.0))
        if best_x is None or best_x.size == 0:
            break
        y, ys = run_optimizer(best_x, max_time=chunk, lr_factor=0.012,
                              top_k=24, weight_power=1.0, momentum=0.3,
                              rng=rng, use_full=False)
        if y is not None and np.isfinite(ys) and ys < best_score:
            best_x = y.copy()
            best_score = ys
            candidates = add_candidate(candidates, y, ys, 80)
        else:
            # If no improvement, stop polishing to leave time for validation.
            break

    # If the best found is too large, make a compatible candidate.
    if best_x is not None and best_x.size > MAX_RETURN_N:
        ds = normalize_sequence(interpolate_sequence(best_x, MAX_RETURN_N), float(MAX_RETURN_N))
        dsc = fast_score(ds)
        candidates = add_candidate(candidates, ds, dsc, 80)
        if np.isfinite(dsc) and dsc < best_score:
            best_x = ds
            best_score = dsc

    # Ensure at least one return-length candidate.
    candidates.sort(key=first_key)
    return_candidate = None
    return_score = np.inf
    for item in candidates:
        arr = item[1]
        if arr.size <= MAX_RETURN_N:
            return_candidate = arr
            return_score = item[0]
            break
    if return_candidate is None:
        return_candidate = normalize_sequence(interpolate_sequence(best_x, MAX_RETURN_N),
                                              float(MAX_RETURN_N))
        return_score = fast_score(return_candidate)

    # Exact validation using the real evaluator for safe-size candidates.
    exact_best_x = return_candidate
    exact_best_score = np.inf
    exact_x = exact_best_x
    exact_score = return_score

    # Evaluate a limited set of return-length candidates.
    max_len_eval = 0
    for item in candidates:
        if item[1].size <= MAX_RETURN_N:
            max_len_eval = max(max_len_eval, item[1].size)

    if max_len_eval > 0:
        if max_len_eval > 7000:
            exact_limit = 10
        else:
            exact_limit = 18
        seen = 0
        for item in candidates:
            if seen >= exact_limit:
                break
            arr = item[1]
            if arr.size > MAX_RETURN_N:
                continue
            seq = finalize_sequence(arr)
            try:
                sc = float(evaluate_sequence(seq))
            except Exception:
                sc = np.inf
            seen += 1
            if np.isfinite(sc) and sc < exact_best_score:
                exact_best_score = sc
                exact_best_x = arr
            if time.monotonic() > hard_end - 3.0:
                break

    # If exact validation did not produce anything, fall back to internal best.
    if not np.isfinite(exact_best_score):
        exact_best_x = return_candidate
        exact_best_score = return_score

    print(f"[final] internal return best={return_score:.6f}, exact tracked={exact_best_score:.6f}", flush=True)

    # Return the exact-best if known; otherwise internal best.
    if exact_best_x is None:
        exact_best_x = return_candidate
    return finalize_sequence(exact_best_x)
```
