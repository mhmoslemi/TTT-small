def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k."""
    n = len(a)
    # Variables: g[0], ..., g[n-1]
    # Objective: minimize -sum(g) (for linprog which minimizes)
    c_obj = -np.ones(n)
    
    # Constraints: (a*g)[k] <= M for k = 0, ..., 2n-2
    # A_ub @ g <= b_ub
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]

    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)

    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c)

            g = np.clip(result.x, 0, None)
            s = g.sum()
            if s > 1e-10:

    """Fast convolution using FFT."""
    if b is None:

"""
Optimizes step-function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a hybrid approach:
  1. LP-based descent (SOTA-inspired) for global progress
  2. Subgradient steps for fast local improvement
  3. Random local search for escape from local minima
  4. Multi-start with diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _convolution_max(a):
    """Compute max of full self-convolution."""
    n = len(a)
    if n < 2:
        return a[0] ** 2 if n == 1 else 0.0
    return float(np.max(np.convolve(a, a)))


def _subgradient_step(a, k_star, eta):
    """
    One subgradient step to reduce c[k_star].
    s[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0.
    """
    n = len(a)
    s = np.zeros(n)
    # s[i] = 2*a[k_star - i] for i = max(0, k_star-n+1) to min(n-1, k_star)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    s[i_lo:i_hi+1] = 2.0 * a[k_star - i_hi : k_star - i_lo + 1][::-1]
    
    a_new = a - eta * s
    a_new = np.clip(a_new, 0, None)
    s_sum = a_new.sum()
    if s_sum > 1e-15:
        a_new /= s_sum
    return a_new


def _build_sparse_conv_matrix(a):
    """
    Build sparse matrix A (shape (2n-1, n)): (A@g)[k] = full_conv(a, g)[k].
    """
    n = len(a)
    # Total nonzeros in the convolution matrix = n^2 - 1
    total_nnz = n * n - 1
    rows = np.empty(total_nnz, dtype=np.int64)
    cols = np.empty(total_nnz, dtype=np.int64)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        cnt = i_hi - i_lo + 1
        if cnt > 0:
            rows[idx:idx + cnt] = k
            cols[idx:idx + cnt] = np.arange(i_lo, i_hi + 1)
            vals[idx:idx + cnt] = a[k - i_hi : k - i_lo + 1][::-1]
            idx += cnt
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def _lp_solve(a, M, top_k=None, time_limit=5.0):
    """
    LP: max 1@g s.t. g>=0, (a*g)[k]<=M.
    Optionally restrict to top_k most binding constraints.
    Returns g normalized to sum=1, or None on failure.
    """
    n = len(a)
    if n > 5000:
        # For very large n, use only top_k constraints
        if top_k is None:
            top_k = min(200, 2 * n - 1)
    
    if top_k is not None and top_k < 2 * n - 1:
        c = np.convolve(a, a)
        # Get the top_k lags
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        # Build only those rows of the matrix
        A = _build_sparse_conv_matrix(a)[top_idx].tocsr()
        b = c[top_idx].clip(upper=M)  # Actually should be M for all
        b = np.full(len(top_idx), M, dtype=np.float64)
    else:
        A = _build_sparse_conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'mip_rel_gap': 0.01})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def _line_search(a, g, max_m):
    """Find best t for a_new = (1-t)*a + t*g."""
    best_t = 0.0
    best_m = max_m
    for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
        a_new = (1.0 - t) * a + t * g
        m = _convolution_max(a_new)
        if m < best_m:
            best_m = m
            best_t = t
    return best_t, best_m


def _random_local_search(a, n_iter, rng, deadline):
    """Local search with random multiplicative perturbations."""
    n = len(a)
    best = a.copy()
    best_m = _convolution_max(best)
    improvements = 0
    
    for _ in range(n_iter):
        if time.monotonic() > deadline:
            break
        cand = best.copy()
        # Random block
        w = int(rng.integers(max(2, n // 100), max(3, n // 5)))
        s = int(rng.integers(0, n - w + 1))
        factor = float(np.exp(rng.normal(0, 0.15)))
        cand[s:s + w] *= factor
        cand = np.clip(cand, 0, None)
        s_sum = cand.sum()
        if s_sum < 1e-12:
            continue
        cand /= s_sum
        m = _convolution_max(cand)
        if m < best_m - 1e-12:
            best = cand
            best_m = m
            improvements += 1
    
    return best, best_m, improvements


def _subgradient_loop(a, max_iters, deadline):
    """Iterative subgradient descent on the max convolution."""
    n = len(a)
    eta = 0.1  # Initial step size
    best = a.copy()
    best_m = _convolution_max(best)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        M = c[k_star]
        
        a = _subgradient_step(a, k_star, eta)
        m = _convolution_max(a)
        
        if m < best_m:
            best = a.copy()
            best_m = m
            eta = min(eta * 1.1, 1.0)  # Slightly increase step
        else:
            eta *= 0.9  # Decrease step
    
    return best, best_m
