import numpy as np
from scipy import optimize, sparse
import time

def compute_score_normalized(a):
    """Compute the score for a non-negative sequence."""
    n = len(a)
    if n == 0 or np.sum(a) < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2 * n * np.max(c) / (np.sum(a)**2)

def subgradient_step(a, k_peak, lr):
    """One subgradient step for peak at lag k_peak."""
    n = len(a)
    # Gradient of (a*a)[k_peak] w.r.t. a[i] is 2*a[k_peak-i] for valid i
    grad = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            grad[i] = 2 * a[j]
    # Descent: a - lr * grad, projected to a >= 0
    a_new = np.maximum(0, a - lr * grad)
    # Renormalize to same sum
    if np.sum(a_new) > 0:
        a_new = a_new * np.sum(a) / np.sum(a_new)
    return a_new

def top_peaks_subgradient(a, top_k=5, lr=0.01):
    """Subgradient step considering top peaks."""
    n = len(a)
    c = np.convolve(a, a)
    peak_lags = np.argsort(c)[-top_k:][::-1]
    
    # Average gradient over top peaks
    total_grad = np.zeros(n)
    for k in peak_lags:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                total_grad[i] += a[j]
    total_grad /= top_k  # average
    
    sum_a = np.sum(a)
    a_new = np.maximum(0, a - lr * total_grad)
    if np.sum(a_new) > 0:
        a_new = a_new * sum_a / np.sum(a_new)
    return a_new

This means: at the optimum, the "mirror" values a[k*-i] are all equal for the support of a.

This is a very useful characterization! It means the optimal a (for a fixed peak lag k*) has the property that a[k*-i] is constant on the support of a.

If the support of a is {0, 1, ..., n-1} (all positive), then a[k*-i] must be constant for all i in the support. This means a is constant (a[i] = c for all i), which gives the uniform solution. But that's only optimal if the peak is at k*.

If the support is a proper subset, say {i_1, ..., i_m}, then a[k*-i_1] = a[k*-i_2] = ... = a[k*-i_m]. This constrains the values of a at positions k*-i_1, k*-i_2, etc.

This gives us a way to construct good solutions:
1. Choose a subset S of {0, ..., n-1} as the support
2. For the peak at lag k*, we need a[k*-i] to be constant for i in S
3. This means a is constant on the set {k*-i : i in S} (intersected with {0,...,n-1})

So the optimal solution for a single peak constraint has a "two-set" structure: a is constant on one set and may vary elsewhere.

But we have multiple peak constraints (all lags), so the structure is more complex.

This analysis suggests that good solutions have a "structured" form: they're piecewise constant or have some periodicity.

Let me now think about a practical algorithm that exploits this:

**Algorithm: Support-based optimization**

1. Start with a candidate support S (initially all of {0,...,n-1})
2. For the current support, solve the QP: min a^T M_{k*} a subject to sum a = S, a[i]=0 for i notin S
   - This can be done by noting that at the optimum, a[k*-i] = c for i in S (on the support)
   - This gives a system of linear equations
3. Check if this solution satisfies all peak constraints (not just k*)
4. If not, update the support or the peak lag

```python
import numpy as np
from scipy import optimize, sparse
import time

def _convolve_full(a):
    """Full convolution using FFT for efficiency."""
    n = len(a)
    size = 2 * n - 1
    # Use FFT
    fa = np.fft.rfft(a, 2 * n)
    c = np.fft.irfft(fa * fa, 2 * n)[:size]
    return c

def _score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = _convolve_full(a)
    return 2.0 * n * np.max(c) / (s * s)

def _initial_linear(n, alpha=1.0):
    """Linear ramp: a[i] = (n-i)^alpha, normalized."""
    a = np.power(np.arange(n, 0, -1, dtype=np.float64), alpha)
    return a / np.sum(a) * n  # normalize so sum = n (avg = 1)

def _initial_exponential(n, rate=0.5):
    """Exponential decay."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64) / n)
    return a / np.sum(a) * n

def _initial_power(n, beta=0.5):
    """Power law: a[i] = (1 - i/n)^beta."""
    x = 1.0 - np.arange(n, dtype=np.float64) / n
    a = np.power(x, beta)
    a = np.maximum(a, 0)
    return a / np.sum(a) * n

def _subgradient_update(a, c, top_k=10, lr=0.05, momentum=0.5, vel=None):
    """Multi-peak subgradient update."""
    n = len(a)
    # Find top k peak lags
    peak_lags = np.argsort(c)[-top_k:][::-1]
    
    # Combined gradient
    grad = np.zeros(n)
    for k in peak_lags:
        i_s = max(0, k - n + 1)
        i_e = min(n - 1, k)
        # grad[i] += 2 * a[k-i] for i in [i_s, i_e]
        # a[k-i] for i from i_s to i_e is a[k-i_s : k-i_e-1 : -1]
        grad[i_s:i_e+1] += 2.0 * a[k - i_e : k - i_s - 1 : -1]
    
    # Average over peaks
    grad /= top_k
    
    # Momentum
    if vel is None:
        vel = np.zeros(n)
    vel = momentum * vel + (1 - momentum) * grad
    
    # Update
    a_new = a - lr * vel
    
    # Project to a >= 0
    a_new = np.maximum(a_new, 0.0)
    
    # Renormalize
    s_old = np.sum(a)
    s_new = np.sum(a_new)
    if s_new > 0:
        a_new = a_new * s_old / s_new
    else:
        a_new = a.copy()
    
    return a_new, vel

def _lp_update(f, M, n):
    """Solve the LP to find the best blend direction."""
    # Maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0
    # Build sparse matrix A: A[k,j] = f[k-j] for valid (k,j)
    
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_s = max(0, k - n + 1)
        j_e = min(n - 1, k)
        c = j_e - j_s + 1
        rows.extend([k] * c)
        cols.extend(range(j_s, j_e + 1))
    
    rows = np.array(rows)
    cols = np.array(cols)
    vals = f[rows - cols]
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize 1^T g subject to A g <= M, g >= 0
    # linprog minimizes, so minimize -1^T g
    res = optimize.linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(2*n-1, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )
    
    if res.success:
        return res.x
    return None
```
