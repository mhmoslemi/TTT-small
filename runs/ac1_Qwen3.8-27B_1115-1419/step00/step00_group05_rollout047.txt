import numpy as np
import time

def _compute_convolution(a):
    """Compute full convolution efficiently using FFT for large n."""
    n = len(a)
    if n < 128:
        return np.convolve(a, a)
    # Use FFT for efficiency
    size = 1
    while size < 2*n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fc = fa * fa
    c = np.fft.irfft(fc, size)[:2*n-1]
    return c

def _softmax_weights(c, beta):
    """Compute softmax weights for the convolution values."""
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))
    w = exp_c / np.sum(exp_c)
    return w

def _gradient(c, a, w):
    """Compute gradient of sum_k w[k]*c[k] with respect to a.
    
    c = conv(a, a), so c[k] = sum_j a[j]*a[k-j]
    dc[k]/da[i] = a[k-i]*(1 + [k!=2i]) for valid k-i
    """
    n = len(a)
    L = len(c)  # 2n-1
    
    # h[i] = sum_k w[k] * a[k-i] for valid k
    # = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = cross-correlation of w and a at lag i
    # numpy.correlate(w, a, 'full')[n-1 : 2n-1]
    h = np.correlate(w, a, 'full')[n-1 : 2*n-1]
    # Verify: h[i] = sum_j a[j] * w[i+j] (need i+j < L = 2n-1, i.e., j < 2n-1-i)
    # Since j < n and i >= 0: j < n <= 2n-1-i (when i <= n-1). Yes, valid.
    
    # correction[i] = w[2i] * a[i] if 2i < L
    correction = np.zeros(n)
    idx = 2 * np.arange(n)
    valid = idx < L
    correction[valid] = w[idx[valid]] * a[valid]
    
    g = 2 * h - correction
    return g

def _multiplicative_update(a, c, lr, beta):
    """Perform one multiplicative update step."""
    w = _softmax_weights(c, beta)
    g = _gradient(c, a, w)
    # Normalized gradient (to make learning rate scale-invariant)
    g_norm = g / (np.linalg.norm(g) + 1e-12)
    a_new = a * np.exp(-lr * g_norm)
    return a_new

**SDP / SOCP approach:**

We want to minimize M subject to:
- (a*a)[k] <= M for all k
- sum(a) >= 1
- a >= 0

This is a quadratically constrained program (QCP). The constraints (a*a)[k] <= M are quadratic (non-convex since a*a is not concave). However, we can use the following trick:

For a FIXED vector b, the constraint a*b <= M (inner product) is linear in a. And (a*a)[k] = a · (R_k a) where R_k is the flip-shift matrix.

So we can use the "outer approximation" method:
1. Start with some a
2. Linearize the quadratic constraints at the current point
3. Solve the resulting LP
4. Repeat

This is essentially the cutting plane / outer approximation method for QCQPs.

At iteration t, with current point a_t:
(a*a)[k] ≈ (a_t * a_t)[k] + (a_t * (a - a_t))[k] + ((a-a_t) * (a-a_t))[k]

For a first-order approximation (dropping the last term):
(a*a)[k] ≈ (a_t*a_t)[k] + 2*(a_t * a)[k] (using symmetry)

Wait, more carefully: (a*a)[k] = sum_j a[j]*a[k-j]
At a = a_t + h: (a*a)[k] ≈ (a_t*a_t)[k] + 2*(a_t*h)[k] + (h*h)[k]

Dropping (h*h)[k]: (a*a)[k] ≈ c_t[k] + 2*(a_t*h)[k] where c_t = a_t*a_t and h = a - a_t.

So the constraint becomes: c_t[k] + 2*(a_t*(a-a_t))[k] <= M
= c_t[k] - 2*(a_t*a_t)[k] + 2*(a_t*a)[k] <= M
= c_t[k] - 2*c_t[k] + 2*(a_t*a)[k] <= M
= -c_t[k] + 2*(a_t*a)[k] <= M
= 2*(a_t*a)[k] <= M + c_t[k]

This is LINEAR in a (for fixed a_t)! So we can solve:
minimize M
subject to: 2*(a_t*a)[k] <= M + c_t[k] for all k
            sum(a) >= 1
            a >= 0
            M >= 0

This is an LP! And we can solve it with scipy or cvxpy.

The iteration is:
1. Solve LP to get a_{t+1}
2. Set a_t = a_{t+1}
3. Repeat

from scipy.optimize import linprog

def solve_lp(a_t, M_init):
    """Solve the LP relaxation at point a_t.
    
    Variables: [a[0], ..., a[n-1], M]
    Minimize: M
    Subject to: 2*(a_t*a)[k] <= M + c_t[k] for k=0,...,2n-2
                sum(a) >= 1  (i.e., -sum(a) <= -1)
                a >= 0, M >= 0
    """
    n = len(a_t)
    c_t = np.convolve(a_t, a_t)  # length 2n-1
    L = len(c_t)
    
    # Variables: a[0..n-1] (n vars) + M (1 var) = n+1 vars
    # Objective: minimize M → c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: A_ub @ x <= b_ub
    # For each k: 2*(a_t*a)[k] - M <= c_t[k]
    # 2*(a_t*a)[k] = 2 * sum_j a[j] * a_t[k-j] (with valid indices)
    # Coefficient of a[j] in constraint k: 2 * a_t[k-j] (if k-j is valid, i.e., 0 <= k-j < n)
    # Coefficient of M: -1
    
    # Build A_ub (L + 1 rows, n+1 columns) -- the +1 is for -sum(a) <= -1
    A = np.zeros((L + 1, n + 1))
    b = np.zeros(L + 1)
    
    for k in range(L):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[k, j] = 2.0 * a_t[kj]
        A[k, n] = -1.0  # coefficient of M
        b[k] = c_t[k]
    
    # -sum(a) <= -1
    A[L, :n] = -1.0
    b[L] = -1.0
    
    # Bounds: a[i] >= 0, M >= 0
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        a_new = result.x[:n]
        M_new = result.x[n]
    else:
        a_new = a_t
        M_new = M_init
    return a_new, M_new

Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Cutting-plane LP relaxation for initial refinement
2. Multiplicative gradient updates for polishing
3. Multiple lengths and initializations
"""
import numpy as np
import time
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    if n < 256:
        return np.convolve(a, a, mode='full')
    size = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def _softmax(c, beta):
    m = c.max()
    e = np.exp(beta * (c - m))
    return e / e.sum()

def _grad(a, w):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a."""
    n = len(a)
    # h[i] = sum_j a[j]*w[i+j]
    h = np.correlate(w, a, mode='full')[n-1:2*n-1]
    # correction: where k=2i is valid
    idx = 2 * np.arange(n)
    L = len(w)
    valid = idx < L
    corr = np.zeros(n)
    corr[valid] = w[idx[valid]] * a[valid]
    return 2*h - corr

def _obj(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _conv(a)
    return 2.0*n*c.max()/(s*s)

def _lp_step(a_t, n):
    """One cutting-plane LP step."""
    c_t = _conv(a_t)
    L = len(c_t)
    # Variables: a[0..n-1], M → n+1 vars
    # Min M s.t. 2*(a_t*a)[k] - M <= c_t[k], sum(a)=n, a>=0, M>=0
    # A_ub: L rows for conv constraints, plus 2 rows for sum(a)=n (as two inequalities)
    # Actually use A_eq for sum(a)=n
    
    # Build A_ub (L x (n+1))
    # A_ub[k, j] = 2*a_t[k-j] if 0<=k-j<n else 0
    # A_ub[k, n] = -1
    A_ub = np.zeros((L, n+1))
    for k in range(L):
        # valid j: max(0, k-(n-1)) <= j <= min(k, n-1)
        j_lo = max(0, k - (n-1))
        j_hi = min(k, n-1)
        if j_lo <= j_hi:
            A_ub[k, j_lo:j_hi+1] = 2.0 * a_t[k-j_hi:k-j_lo+1][::-1]
        A_ub[k, n] = -1.0
    b_ub = c_t.copy()
    
    # Equality: sum(a) = n
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n])
    
    # Objective: minimize M
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * (n+1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[n]
    except:
        pass
    return a_t.copy(), np.inf

    a = a.copy().astype(np.float64)
