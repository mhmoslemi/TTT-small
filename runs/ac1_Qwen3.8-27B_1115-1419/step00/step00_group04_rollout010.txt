Key insight: If a[i] is small where a[n-1-i] is large, then b[n-1] is small. But then b[0] = a[0]^2 and b[2n-2] = a[n-1]^2 must also be small.

What about a "sparse" sequence? If most a[i] = 0 and a few are 1:
- Suppose a has m non-zero entries, each equal to 1/m (so sum = 1)
- b[k] is sum of products of pairs that are k apart
- If the non-zero entries are "spread out" (good prime differences), b[k] is small for most k
- But b at lags matching differences of pairs will be non-zero

This is related to "difference sets" in combinatorial design theory!

A (v, k, λ) difference set: a subset D of a group G of order v, with |D| = k, such that every non-identity element is expressed as a difference d1-d2 (d1,d2 in D, d1≠d2) exactly λ times.

For a cyclic group Z_n, this gives a sequence where b[k] = λ for k ≠ 0 (mod n) and b[0] = k. For the linear convolution (not circular), it's similar but with edge effects.

For a perfect difference set (λ=1), we get a "BCH" sequence. The minimum peak (excluding b[0]) would be 1.

But we need REAL non-negative values summing to 1, not 0-1 sequences. And we need linear convolution, not circular.

def build_Qk(n, k):
    """Build the sparse matrix Q such that a^T Q a = b[k] = sum_i a[i]*a[k-i]"""
    # b[k] = sum_{i} a[i]*a[k-i] where 0<=i<n, 0<=k-i<n
    # Valid i: max(0, k-(n-1)) <= i <= min(k, n-1)
    i_lo = max(0, k - (n-1))
    i_hi = min(k, n-1)
    rows = []
    cols = []
    vals = []
    for i in range(i_lo, i_hi+1):
        j = k - i
        if i == j:
            # a[i]*a[i] → Q[i,i] += 1
            rows.append(i)
            cols.append(i)
            vals.append(1.0)
        else:
            # a[i]*a[j] + a[j]*a[i] → Q[i,j] += 0.5, Q[j,i] += 0.5
            # Wait, a^T Q a = sum_{i,j} a[i] Q[i,j] a[j]
            # We want this to equal sum_i a[i]*a[k-i]
            # For i ≠ j=k-i: contributes a[i]*a[j] which is a[i]*Q[i,j]*a[j] + a[j]*Q[j,i]*a[i]

            rows.append(i)
            cols.append(j)
            vals.append(0.5)
            rows.append(j)
            cols.append(i)
            vals.append(0.5)
    Q = sp.coo_matrix((vals, (rows, cols)), shape=(n, n)).tocsr()
    return Q

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequence a.
Uses CVXPY active-set method + projected gradient refinement.
"""

import numpy as np
import time
import cvxpy as cp
import scipy.sparse as sp

def compute_obj(a):
    """Compute the objective value (scale-invariant)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)

def build_Qk_sparse(n, k):
    """Build sparse Q such that a^T Q a = sum_i a[i]*a[k-i] (valid indices)."""
    i_lo = max(0, k - (n - 1))
    i_hi = min(k, n - 1)
    if i_lo > i_hi:
        return None
    rows = []
    cols = []
    vals = []
    for i in range(i_lo, i_hi + 1):
        j = k - i
        if j < 0 or j >= n:
            continue
        if i == j:
            rows.append(i)
            cols.append(i)
            vals.append(1.0)
        else:
            rows.append(i)
            cols.append(j)
            vals.append(0.5)
            rows.append(j)
            cols.append(i)
            vals.append(0.5)
    if not rows:
        return None
    Q = sp.coo_matrix((vals, (rows, cols)), shape=(n, n))
    # Make it properly symmetric
    Q = (Q + Q.T) * 0.5  # ensure symmetry
    return Q.tocsr()

def cvxpy_active_set(n, a_init, max_rounds=50, time_budget=1200):
    """Solve min t s.t. sum(a)=1, a>=0, a^T Q_k a <= t for active k."""
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    # Initial active set: b[0], b[n-1], b[2n-2]
    # b[0] = a[0]^2, b[2n-2] = a[n-1]^2
    # b[n-1] = sum a[i]*a[n-1-i]
    active = [0, n - 1]
    # Add a few more strategically
    if n > 1:
        active.append(2 * n - 2)
    
    obj = cp.Minimize(t)
    base_constraints = [cp.sum(a) == 1]
    
    best_a = a_init.copy()
    best_t = float('inf')
    
    deadline = time.monotonic() + time_budget
    
    prev_t = float('inf')
    stall_count = 0
    
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Build problem
        constraints = base_constraints.copy()
        Q_cache = {}
        for k in active:
            if k not in Q_cache:
                Q_cache[k] = build_Qk_sparse(n, k)
            Q = Q_cache[k]
            if Q is not None:
                constraints.append(cp.quad_form(a, Q) <= t)
        
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                try:
                    prob.solve(solver=cp.CLARABEL, verbose=False)
                except:
                    break
        
        if a.value is None or t.value is None:
            break
        
        a_sol = np.clip(a.value, 0, None)
        a_sol = a_sol / max(np.sum(a_sol), 1e-10)  # re-normalize
        t_sol = float(t.value)
        
        if t_sol < best_t:
            best_t = t_sol
            best_a = a_sol.copy()
            prev_t = t_sol
        
        # Check for new violations
        b = np.convolve(a_sol, a_sol)
        violated = np.where(b > t_sol + 1e-8)[0]
        
        if len(violated) == 0:
            print(f"  Round {round_idx}: Converged. t = {t_sol:.8f}, active = {len(active)}")
            break
        
        # Add the worst violator
        worst_k = violated[np.argsort(b[violated])[-1]]
        if worst_k not in active:
            active.append(int(worst_k))
            print(f"  Round {round_idx}: t = {t_sol:.8f}, active = {len(active)}, worst k = {worst_k}, b[worst] = {b[worst_k]:.8f}")
        else:
            # All violators already active; nudge with a tiny amount
            stall_count += 1
            if stall_count > 5:
                print(f"  Round {round_idx}: Stalling. t = {t_sol:.8f}")
                break
            # Remove the least important constraint to help? Or just stop
            break
        
        if abs(prev_t - t_sol) < 1e-10:
            print(f"  Round {round_idx}: Stalled. t = {t_sol:.8f}")
            break
        prev_t = t_sol
    
    return best_a, best_t

def projected_gradient(a, lr, n):
    """One step of projected gradient on min max(convolve(a,a)) s.t. sum=1, a>=0."""
    b = np.convolve(a, a)
    b_max = np.max(b)
    
    # Find argmax indices (top few for stability)
    top_k = 5
    idx = np.argsort(b)[-top_k:]
    
    # Subgradient: average of gradients at argmax locations
    grad = np.zeros(n)
    for k in idx:
        # d b[k] / d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        i_lo1 = max(0, k - (n - 1))
        i_hi1 = min(k, n - 1)
        for i in range(i_lo1, i_hi1 + 1):
            j = k - i
            # a[i] * a[j] contributes a[j] to grad[i] and a[i] to grad[j]
            grad[i] += a[j]
            if j != i:
                grad[j] += a[i]
    
    grad /= top_k
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    S = np.sum(a_new)
    if S > 1e-10:
        a_new /= S
    else:
        a_new = np.ones(n) / n
    return a_new, b_max

def fista_refine(a, time_budget=600):
    """FISTA (accelerated projected gradient) refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    a = a / S  # normalize to sum=1
    
    a_k = a.copy()
    a_kp1 = a.copy()
    t_k = 1.0
    
    best_a = a.copy()
    best_val = compute_obj(a)
    
    deadline = time.monotonic() + time_budget
    lr = 1.0 / n  # initial step size
    
    iteration = 0
    while time.monotonic() < deadline:
        # Compute gradient of max(convolve(a, a))
        b = np.convolve(a_k, a_k)
        b_max = b.max()
        
        # Soft-max subgradient
        tau = max(b_max * 0.01, 1e-10)
        w = np.exp((b - b_max) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        
        # Gradient: sum_k w[k] * (shifted a terms)
        # For efficiency, use the fact that:
        # sum_k w[k] * a[k-j] = cross-correlation(w, a)[j]
        # sum_k w[k] * a[j-k] = convolution(w, a_reversed)[j] ... 
        # Let me just compute it directly for the top indices
        top_idx = np.where(w > w.max() * 0.001)[0]  # significant weights
        if len(top_idx) == 0:
            top_idx = [np.argmax(b)]
        
        grad = np.zeros(n)
        for k in top_idx:
            wk = w[k]
            i_lo1 = max(0, k - (n - 1))
            i_hi1 = min(k, n - 1)
            for i in range(i_lo1, i_hi1 + 1):
                j = k - i
                grad[i] += wk * a_k[j]
                if j != i:
                    grad[j] += wk * a_k[i]
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        y_k = a_kp1 + ((t_k - 1) / t_new) * (a_kp1 - a_k)
        
        a_new = y_k - lr * grad
        a_new = np.clip(a_new, 0, None)
        S_new = np.sum(a_new)
        if S_new > 1e-10:
            a_new /= S_new
        else:
            a_new = y_k.copy()
        
        # Backtracking line search
        val = np.max(np.convolve(a_new, a_new))
        if val > b_max + 1e-10:
            lr /= 2
            a_new = y_k - lr * grad
            a_new = np.clip(a_new, 0, None)
            S_new = np.sum(a_new)
            if S_new > 1e-10:
                a_new /= S_new
            elif S_new == 0:
                a_new = y_k.copy()
        
        a_k = a_kp1
        a_kp1 = a_new
        t_k = t_new
        iteration += 1
        
        if iteration % 100 == 0:
            obj_val = compute_obj(a_kp1)
            if obj_val < best_val:
                best_val = obj_val
                best_a = a_kp1.copy()
            print(f"  FISTA iter {iteration}: val = {obj_val:.8f}, best = {best_val:.8f}")
    
    return best_a, best_val

Let me be more careful. b[k] = sum_i a[i] * a[k-i] (with a out of range = 0).

∂b[k]/∂a[j] = a[k-j] (if 0 ≤ k-j < n) + a[j-k] (if 0 ≤ j-k < n)

So the "full" gradient (for all k weighted by w) is:
grad[j] = sum_k w[k] * (a[k-j] * 1[0≤k-j<n] + a[j-k] * 1[0≤j-k<n])

First term: sum_k w[k] * a[k-j] * 1[0≤k-j<n]
  Let m = k-j: sum_m w[m+j] * a[m] * 1[0≤m<n]
  = (w_shifted cross-correlate a) where w_shifted[m] = w[m+j]
  This is the cross-correlation of w and a at lag j!
  = sum_m w[m+j] * a[m] = cross_corr(w, a)[j]

Second term: sum_k w[k] * a[j-k] * 1[0≤j-k<n]
  Let m = j-k: sum_m w[j-m] * a[m] * 1[0≤m<n]
  = sum_m a[m] * w[j-m] = (a convolve w)[j] = conv(a, w)[j]

So: grad = cross_correlate(w, a) + convolve(a, w)

Where cross_correlate(w, a)[j] = sum_m w[m+j] * a[m] (with w out of range = 0)
And convolve(a, w)[j] = sum_m a[m] * w[j-m] (with out of range = 0)

Using FFT:
- Let W = FFT(w_padded), A = FFT(a_padded) with sufficient padding (size ≥ 2n-1 + n - 1 = 3n-2, or just use next power of 2 ≥ 2n)
- cross_correlate(w, a) = IFFT(conj(W) * A)  [circular cross-correlation]
- convolve(a, w) = IFFT(A * W)  [circular convolution]

For the cross-correlation: (w ⋆ a)[j] = sum_m w[m+j] a[m]. The result has indices j from -(n-1) to n-1 (where w has indices 0 to 2n-2). With padding to size ≥ 2n + n - 1 = 3n-1, we avoid aliasing.

- w has length 2n-1 (indices 0 to 2n-2)
- a has length n (indices 0 to n-1)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 via CVXPY active-set + gradient refinement.
Exploits convexity of max of PSD quadratic forms.
"""

import numpy as np
import time
import cvxpy as cp
import scipy.sparse as sp

def compute_score(a):
    """Compute the objective: 2*n*max_b/S^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return 1e10
    b = np.convolve(a.astype(np.float64), a.astype(np.float64))
    return 2.0 * n * float(np.max(b)) / (S * S)

def build_Qk(n, k):
    """Sparse matrix Q_k with a^T Q_k a = b[k] = sum_i a[i]*a[k-i]."""
    i_lo = max(0, k - (n - 1))
    i_hi = min(k, n - 1)
    if i_lo > i_hi:
        return sp.csr_matrix((n, n))
    rows = []
    cols = []
    vals = []
    for i in range(i_lo, i_hi + 1):
        j = k - i
        if i == j:
            rows.append(i)
            cols.append(i)
            vals.append(1.0)
        else:
            rows.append(i); cols.append(j); vals.append(0.5)
            rows.append(j); cols.append(i); vals.append(0.5)
    if not rows:
        return sp.csr_matrix((n, n))
    Q = sp.coo_matrix((vals, (rows, cols)), shape=(n, n))
    Q = (Q + Q.T) * 0.5
    return Q.tocsr()

def cvxpy_solve(n, a_init, max_rounds=30, time_budget=1000):
    """Active-set CVXPY solver."""
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    obj = cp.Minimize(t_var)
    base_cons = [cp.sum(a_var) == 1]
    
    # Initial active constraints
    active_k = [0, n - 1, 2 * n - 2]
    Q_cache = {}
    
    best_a = a_init.copy()
    best_t = np.inf
    deadline = time.monotonic() + time_budget
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        cons = list(base_cons)
        for k in active_k:
            if k not in Q_cache:
                Q_cache[k] = build_Qk(n, k)
            Q = Q_cache[k]
            if Q.nnz > 0:
                cons.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
        
        prob = cp.Problem(obj, cons)
        
        t0 = time.monotonic()
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                if solver == cp.ECOS:
                    prob.solve(solver=solver, verbose=False, max_iters=5000)
                elif solver == cp.SCS:
                    prob.solve(solver=solver, verbose=False, max_iters=5000)
                else:
                    prob.solve(solver=solver, verbose=False)
                if a_var.value is not None and t_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            print(f"  CVXPY round {rnd}: Solve failed")
            break
        
        t0_solve = time.monotonic() - t0
        a_sol = np.clip(np.array(a_var.value, dtype=np.float64), 0, None)
        S = np.sum(a_sol)
        if S < 1e-10:
            break
        a_sol /= S
        t_sol = float(t_var.value)
        
        if t_sol < best_t:
            best_t = t_sol
            best_a = a_sol.copy()
        
        # Check violations
        b = np.convolve(a_sol, a_sol)
        viol = np.where(b > t_sol + 1e-7)[0]
        
        print(f"  CVXPY round {rnd}: t={t_sol:.8f}, |active|={len(active_k)}, violations={len(viol)}, solve_time={t0_solve:.1f}s")
        
        if len(viol) == 0:
            break
        
        # Add worst violator not already active
        worst = viol[np.argmax(b[viol])]
        added = False
        for v in viol:
            if v not in active_k:
                active_k.append(int(v))
                added = True
                break
        if not added:
            # All violators already active; try adding next worst
            for v in viol[np.argsort(b[viol])[::-1]]:
                if v not in active_k:
                    active_k.append(int(v))
                    break
            else:
                print(f"  CVXPY round {rnd}: No new constraint to add. Stopping.")
                break
    
    return best_a, best_t

def gradient_step(a, lr, top_k=20, tau_frac=0.05):
    """One projected gradient step. Returns new a and max_b."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = float(b.max())
    
    # Soft-max weights
    tau = max(b_max * tau_frac, 1e-14)
    w = np.exp((b - b_max) / tau)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= ws
    
    # Gradient
    top_idx = np.argsort(w)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        # First part: a[k-i] for i where 0<=k-i<n
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            i_arr = np.arange(i_lo, i_hi + 1)
            j_arr = k - i_arr
            grad[i_arr] += wk * a[j_arr]
            # For diagonal (i==j), we've already counted it once; need to handle
            # Actually no: the product a[i]*a[j] contributes a[j] to d/d a[i] and a[i] to d/d a[j]

            # But in my loop: grad[i] += wk*a[j] and if j!=i, grad[j] += wk*a[i]
            # When i==j, we only add wk*a[i] once. But the true derivative of a[i]^2 is 2*a[i].
            # So I need to add another wk*a[i] for the diagonal case.

        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            i_arr2 = np.arange(i_lo2, i_hi2 + 1)
            j_arr2 = i_arr2 - k
            grad[i_arr2] += wk * a[j_arr2]
    
    # Wait, I think I'm double-counting. Let me re-derive.

    # db[k]/da[j] = sum_{i: i=j, k-i valid} a[k-i] + sum_{i: k-i=j, i valid} a[i]
    #            = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # Note: if k-j = j-k (i.e., j=k/2), both terms give a[j], so total is 2*a[j].
    # My computation above:
    # First loop: for i = j, adds a[k-j] = a[(k)-j] ... hmm this is getting confusing.
    # Let me just verify: grad[j] should be a[k-j] + a[j-k] (with 0 for out-of-range).
    # My first loop adds to grad[i] the value a[k-i] for i in valid range. This gives grad[j] += a[k-j]. ✓
    # My second loop adds to grad[i] the value a[i-k] for i in valid range. This gives grad[j] += a[j-k]. ✓
    # Total: grad[j] = a[k-j] + a[j-k]. ✓
    # When j = k/2: grad[j] = a[k/2] + a[k/2] = 2*a[k/2]. ✓ (derivative of a[j]^2 is 2a[j])
    
    # Update
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    S_new = np.sum(a_new)
    if S_new < 1e-10:
        a_new = a.copy()
    else:
        a_new /= S_new
    return a_new, b_max

def gradient_refine(a, time_budget=800):
    """Iterated projected gradient with backtracking."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a /= np.sum(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    lr = 1.0 / n
    deadline = time.monotonic() + time_budget
    stagnation = 0
    prev_best = best_score
    
    it = 0
    while time.monotonic() < deadline:
        a_new, b_max = gradient_step(a, lr)
        
        # Check improvement
        new_score = 2.0 * n * b_max / 1.0  # since sum(a)=1, score = 2n*max_b
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        a = a_new
        
        # Adapt learning rate
        if stagnation > 100:
            lr *= 0.5
            stagnation = 0
            if lr < 1e-10:
                break
        elif it % 500 == 0 and it > 0:
            # Try to increase lr
            lr = min(lr * 1.2, 1.0)
        
        it += 1
        if it % 5000 == 0:
            print(f"  Grad iter {it}: score={best_score:.8f}, lr={lr:.6f}")
    
    return best_a, best_score

Let me also think about whether the problem is actually convex.

The objective is: minimize max_k (a^T Q_k a) subject to sum(a)=1, a>=0.

Each a^T Q_k a is a convex function of a (since Q_k is PSD). The max of convex functions is convex. The constraints define a convex set. So yes, this is a convex optimization problem!

The minimum value of max_k b[k] given sum(a)=1 is what we need. Let's call it T*. The score is 2n * T*.

For the uniform distribution, T* = 1/n = 1/3809, score = 2*3809/3809 = 2.

The theoretical lower bound: since sum b[k] = (sum a)^2 = 1 and there are 2n-1 = 7617 terms, max b[k] >= 1/7617. So T* >= 1/7617, score >= 2*3809/7617 ≈ 1.0003.

But for non-negative sequences, the achievable T* is likely much larger than 1/(2n-1). The question is how close we can get.

Actually, I just realized: for non-negative sequences, b[0] = a[0]^2. If sum(a) = 1 and a >= 0, then a[0] <= 1, so b[0] <= 1. But that's a very loose bound.

If we set a[0] = a[n-1] = 0, then b[0] = b[2n-2] = 0, and the constraint is on the interior b[k]. This is like a "tapered" sequence.

For a sequence that's 0 at the edges and concentrated in the middle, b[n-1] = sum a[i]*a[n-1-i] is the key constraint. If the sequence is symmetric, b[n-1] = sum a[i]^2 >= 1/n (by Cauchy-Schwarz, with equality for uniform). So T* >= 1/n for symmetric sequences, giving score >= 2.

Wait! This means for SYMMETRIC non-negative sequences, the score is AT LEAST 2! Because b[n-1] = sum a[i]^2 >= (sum a[i])^2/n = 1/n.

So to beat 2, we MUST use asymmetric sequences!

For asymmetric sequences, b[n-1] = sum a[i]*a[n-1-i]. By Cauchy-Schwarz:
(sum a[i]*a[n-1-i])^2 <= (sum a[i]^2)(sum a[n-1-i]^2) = (sum a[i]^2)^2

So b[n-1] <= sum a[i]^2. But we want to MINIMIZE b[n-1], not maximize.

By AM-GM or just rearrangement: if a is sorted and a reversed, the "cross" sum is at least 1/n... no, that's not right. The "cross" sum can be made small by anti-correlating a with its reverse.

Example: n=4, a = [1/4, 1/4, 1/4, 1/4]. b[3] = 1/16+1/16+1/16+1/16 = 1/4 = 1/n. Score = 2*4*1/4 = 2.

n=4, a = [1/2, 1/6, 1/6, 1/6]. b[3] = (1/2)(1/6)+(1/6)(1/6)+(1/6)(1/6)+(1/6)(1/2) = 1/12+1/36+1/36+1/12 = 1/12+1/36+1/36+1/12 = 4/36+1/36+1/36 = 6/36 = 1/6.
S^2/n = (1)^2/4 = 1/4. Is 1/6 < 1/4? Yes! So max_b might be < 1/4.
But b[0] = (1/2)^2 = 1/4. So max_b = max(1/4, 1/6, ...) = 1/4. Score = 2*4*1/4 = 2. Same!

The problem is b[0] = a[0]^2. If a[0] > 1/sqrt(n), then b[0] > 1/n.

For n=4: 1/sqrt(4) = 1/2. a[0] = 1/2 gives b[0] = 1/4 = 1/n. Exactly at the boundary.

For score < 2, we need ALL b[k] < 1/n. In particular a[0]^2 < 1/n, so a[0] < 1/sqrt(n). And by the same logic for all positions.

But sum a[i] = 1 and n * 1/sqrt(n) = sqrt(n) > 1 for n > 1. So it's feasible to have all a[i] < 1/sqrt(n).

The question is whether we can simultaneously satisfy:
- a[i]^2 < 1/n for all i (i.e., a[i] < 1/sqrt(n))
- sum a[i] = 1
- b[k] < 1/n for all k

The first two are feasible. The last is the constraint. Let's see:

If a[i] = 1/n for all i (uniform), then b[k] = (n-k+1)/n^2 for k=0..n-1 (and symmetric). Max is n/n^2 = 1/n. Score = 2.

To get strictly less than 1/n everywhere, we need to "spread out" the convolution. The issue is that near the center (k ≈ n-1), there are about n/2 terms in the sum, each of size about (1/n)^2 = 1/n^2, giving about 1/(2n). That's already < 1/n. Good.

Wait, for uniform a, b[n-1] = sum a[i]*a[n-1-i] = n*(1/n)^2 = 1/n. That's exactly 1/n, not less.

For b[n-1] < 1/n with sum a = 1 and a >= 0:
sum a[i]*a[n-1-i] < 1/n

By Cauchy-Schwarz: (sum a[i]*a[n-1-i])^2 <= (sum a[i]^2)(sum a[n-1-i]^2) = (sum a[i]^2)^2
So b[n-1] <= sum a[i]^2.

And sum a[i]^2 >= 1/n (by Cauchy-Schwarz on the sum). Equality iff all a[i] equal.

So b[n-1] can be less than sum a[i]^2 (for asymmetric sequences), and sum a[i]^2 can be greater than 1/n (for non-uniform).

The question is: can b[n-1] < 1/n? Yes, if a[i] is small where a[n-1-i] is large. But then b[0] = a[0]^2 and b[n-2] = a[0]*a[n-2] + ... need to be small too.

This is a combinatorial/number-theoretic problem in disguise. The best non-negative sequences for this purpose are related to " Sidon sequences" or "Berdyaev's conjectures" about non-negative sequences with minimal autocorrelation.

For the coordinate descent:

- We want to choose a[j] to minimize T = max_k b[k]
- This is a 1-D problem: minimize max of linear functions → bisection

This is much more tractable! Let me code this up.

Actually, let's verify: f(x) = max_k (A_k + c_k * x) / (S' + x)^2 where S' = S - a[j].
Each (A_k + c_k*x) / (S'+x)^2 is a rational function. Its derivative:
d/dx [(A_k + c_k x)/(S'+x)^2] = (c_k(S'+x)^2 - (A_k+c_k x)*2(S'+x)) / (S'+x)^4
= (S'+x)(c_k(S'+x) - 2(A_k+c_k x)) / (S'+x)^4
= (c_k S' + c_k x - 2A_k - 2c_k x) / (S'+x)^3
= (c_k S' - c_k x - 2A_k) / (S'+x)^3

So each individual function (A_k + c_k*x)/(S'+x)^2 is decreasing in x when c_k*x + 2A_k > c_k*S', i.e., when c_k > 2A_k/(S'-x) (roughly). For x near a[j] and S' + x = S, this condition is c_k > 2(A_k)/(S) = 2(b[k]-a[j]c_k)/S.

The max of these functions is not necessarily unimodal. But in practice, it should be "roughly" unimodal (decreasing then potentially increasing for very large x).

For practical purposes, I'll use a golden section search or just evaluate at a few candidate x values.

import numpy as np

def compute_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return 1e10
    b = np.convolve(np.asarray(a, dtype=np.float64), np.asarray(a, dtype=np.float64))
    return 2.0 * n * float(np.max(b)) / (S * S)

def build_Qk(n, k):
    i_lo = max(0, k - (n - 1))
    i_hi = min(k, n - 1)
    if i_lo > i_hi:
        return None
    rows, cols, vals = [], [], []
    for i in range(i_lo, i_hi + 1):
        j = k - i
        if i == j:
            rows.append(i); cols.append(i); vals.append(1.0)
        else:
            rows.append(i); cols.append(j); vals.append(0.5)
            rows.append(j); cols.append(i); vals.append(0.5)
    if not rows:
        return None
    Q = sp.coo_matrix((vals, (rows, cols)), shape=(n, n))
    return Q.tocsr()

def make_init_quadratic(n, freq=1.0, alpha=0.95, rng=None):
    """a[i] = max(0, 1 + alpha*cos(2*pi*freq*i^2/n))"""

    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
        S = n
    return a / S
