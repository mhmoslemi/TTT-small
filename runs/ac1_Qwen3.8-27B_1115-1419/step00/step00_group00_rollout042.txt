        print(f"  Using provided sequence: n={len(a_norm)}")

    # Try various lengths and types
    lengths_to_try = [n_default, 4096, 5000, 3000, 8000]
    types_to_try = ['uniform', 'gaussian', 'triangular', 'bimodal', 'random_smooth', 'logarithmic']

    # First do a quick pass to find promising candidates
    time_per_trial = 30.0  # seconds per trial in phase 1
    phase1_deadline = t0 + 300.0

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline:
            break
        for kind in types_to_try:
            if time.monotonic() > phase1_deadline:
                break
            a = _try_initialization(kind, n, rng)
            # Quick coordinate descent (limited)
            trial_limit = min(time_per_trial, deadline - time.monotonic())
            if trial_limit < 5:
                break
            a_best, max_best = _run_coord_descent(a, max_sweeps=50, time_limit=trial_limit, start_time=t0)
            candidates.append((f"{kind}_{n}", a_best, max_best))
            print(f"  {kind} n={n}: max_conv={max_best:.8f}, score_est={2*n*max_best:.6f}")

    # Sort candidates by estimated score
    candidates.sort(key=lambda x: x[2] if len(x) > 2 and x[2] < 1 else 999)
    print(f"\nPhase 1 complete. Found {len(candidates)} candidates. Best est score: {2*candidates[0][1].size*candidates[0][2]:.6f}")

    # Phase 2: Intensive optimization on top candidates
    print("\nPhase 2: Intensive coordinate descent...")
    top_k = min(5, len(candidates))

    for idx in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        name, a_start, _ = candidates[idx]
        n = len(a_start)
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        print(f"  Optimizing candidate {idx} ({name}, n={n}), time budget={remaining:.0f}s")

        a_opt, max_opt = _run_coord_descent(a_start, max_sweeps=100000,
                                            time_limit=remaining * 0.8, start_time=time.monotonic())
        score = float(evaluate_sequence(a_opt.tolist()))
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            best_n = n
            print(f"    IMPROVED! score={score:.6f}")
        else:
            print(f"    No improvement (score={score:.6f} vs best={best_score:.6f})")

    # Phase 3: Further refinement with restarts
    if best_a is not None and time.monotonic() < deadline - 30:
        print("\nPhase 3: Refinement with perturbations...")
        n = len(best_a)
        current = best_a.copy()
        current_score = best_score

        # Try perturbation + coordinate descent
        for restart in range(20):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Perturb
            sigma = 0.05 / np.sqrt(n)
            perturbed = current + rng.normal(0, sigma, n)
            perturbed = np.clip(perturbed, 0, None)
            s = perturbed.sum()
            if s < 1e-10:
                continue
            perturbed /= s

            a_opt, max_opt = _run_coord_descent(perturbed, max_sweeps=50000,
                                                time_limit=remaining * 0.4, start_time=time.monotonic())
            score = float(evaluate_sequence(a_opt.tolist()))
            if score < best_score - 1e-10:
                best_score = score
                best_a = a_opt.copy()
                current = a_opt.copy()
                current_score = score
                print(f"  Restart {restart}: IMPROVED to {score:.6f}")
            elif score < best_score:
                # Slightly worse, but might lead to further improvement
                # Accept with some probability (annealing)
                if rng.random() < 0.3:
                    current = a_opt.copy()

    # Final validation
    if best_a is not None:
        final_score = float(evaluate_sequence(best_a.tolist()))
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_a)}")
        print(f"Total time: {time.monotonic()-t0:.1f}s")
        return [float(x) for x in best_a]
    else:
        # Fallback: uniform sequence
        a = np.ones(1024)
        return [float(x) for x in a]
