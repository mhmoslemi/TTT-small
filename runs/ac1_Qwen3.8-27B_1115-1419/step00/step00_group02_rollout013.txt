def compute_score(x):
    """Fast score computation using FFT convolution."""
    n = len(x)
    if n == 0:

    x = np.maximum(0, x)
    x = np.minimum(1000, x)
    s = np.sum(x)
    if s < 0.01:
        return np.inf
    conv = np.convolve(x, x, mode='full')
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)

    x = np.clip(x, 0.0, 1000.0)

def _peak_lags(x, tol=1e-9):
    """Return indices where convolution is (near) maximum."""
    conv = np.convolve(x, x, mode='full')
    mx = np.max(conv)
    return np.where(conv >= mx * (1.0 - tol))[0]


def _subgrad_direction(x, peaks):
    """
    Compute subgradient of max_conv at peak lags.
    d conv[k]/dx[i] = 2*x[k-i] if 0<=k-i<n, else 0.
    Returns the negative gradient direction (to decrease the peak).
    """
