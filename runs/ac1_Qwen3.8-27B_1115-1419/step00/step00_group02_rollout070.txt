```
def project_simplex(v):
    v = np.asarray(v, dtype=float).copy()
    mu = 0.0
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, len(v)+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.full(len(v), 1.0/len(v))
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    # ensure sum 1 (clip numerical)
    out /= np.sum(out)
    return out
```

```python
import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

"""
Construct a low-peak autocorrelation step function.

Main idea:
- The objective is scale invariant and, for a block-stretched coarse sequence,
  is asymptotically (and essentially exactly for equal block lengths) equal to
  the objective of the coarse sequence itself.
- Search a short coarse nonnegative sequence c of length m by randomized
  global/local methods, then stretch it to the requested/default length.
- Greedily reduce the maximum convolution peak using active-lag subgradients.
"""

_RNG = np.random.default_rng(12345)
_FULL_LENGTHS = None


def project_simplex(v):
    """Project a vector onto the nonnegative simplex sum(x)=1."""
    v = np.asarray(v, dtype=float).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0])

    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    idx = np.nonzero(sv * np.arange(1, n + 1, dtype=float) > (cssv - 1.0))[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)

    rho = int(idx[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        return np.full(n, 1.0 / n)
    out /= s
    return out


def coarse_score_p(p):
    """Score for a nonnegative coarse sequence p, normalized internally."""
    p = np.asarray(p, dtype=float).ravel()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, None)
    s = float(p.sum())
    if s <= 1e-15:
        return 1e100
    p = p / s
    b = np.convolve(p, p)
    M = float(np.max(b))
    if not np.isfinite(M) or M < 0.0:
        M = 0.0
    return float(2.0 * len(p) * M)


def coarse_raw_score(x):
    """Bounded raw objective for coarse sequences."""
    x = np.asarray(x, dtype=float).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, None)
    s = float(x.sum())
    if s <= 1e-12:
        return 1e100
    c = x / s
    b = np.convolve(c, c)
    M = float(np.max(b))
    if not np.isfinite(M) or M < 0.0:
        M = 0.0
    return float(2.0 * len(c) * M)


def softmax_vector(t):
    """Stable softmax with one flat direction removed by centering."""
    t = np.asarray(t, dtype=float).ravel()
    if not np.all(np.isfinite(t)):
        return np.full(len(t), 1.0 / max(1, len(t)))
    t = t - float(np.mean(t))
    z = t - float(np.max(t))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(len(t), 1.0 / max(1, len(t)))
    return e / s


def softmax_score_t(t):
    """Softmax-parametrized coarse score."""
    return coarse_score_p(softmax_vector(t))


def next_fast_len(n):
    """Smallest power of two >= n."""
    if n <= 1:
        return 1
    return int(2 ** (n - 1).bit_length())


def fast_score_full(a):
    """Fast FFT evaluation of the full objective, mimicking the official score."""
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return 1e100
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(np.sum(a))
    if s < 0.01:
        return 1e100

    nfft = next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, nfft)
    b = np.fft.irfft(fa * fa, nfft)[:2 * n - 1]
    b = np.where(np.isfinite(b), b, 0.0)
    M = float(np.max(b))
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (s * s))


def make_lengths(N, m):
    """Nearly equal block lengths summing to N."""
    N = int(N)
    m = int(m)
    if N <= 0:
        return np.array([], dtype=int)
    if m <= 0:
        m = 1
    if m > N:
        m = N
    base, rem = divmod(N, m)
    lengths = np.full(m, base, dtype=int)
    lengths[:rem] += 1
    return lengths


def stretch_from_c(c, N):
    """Stretch a coarse sequence to length N using near-equal blocks."""
    c = np.asarray(c, dtype=float).ravel()
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.clip(c, 0.0, None)
    s = float(c.sum())
    if s <= 1e-15:
        return np.ones(max(1, int(N)), dtype=float)
    c = c / s
    lengths = make_lengths(int(N), len(c))
    return np.repeat(c, lengths)


def full_score_from_raw_global(x):
    """Objective for stretching a coarse raw vector with current global lengths."""
    lengths = _FULL_LENGTHS
    if lengths is None or len(lengths) == 0:
        return 1e100
    x = np.asarray(x, dtype=float).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, None)
    s = float(x.sum())
    if s <= 1e-12:
        return 1e100
    c = x / s
    a = np.repeat(c, lengths)
    return fast_score_full(a)


def active_refine_c(c, max_iter, deadline):
    """Local active-lag projected gradient refinement on the coarse simplex."""
    c = project_simplex(np.asarray(c, dtype=float).copy())
    best_c = c.copy()
    best_s = coarse_score_p(c)
    m = len(c)
    if m == 0:
        return best_c, best_s

    lr = 0.25
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        b = np.convolve(c, c)
        order = np.argsort(b)[::-1]

        # Start from the unique maximum, then smooth over a few top lags.
        K = 1 if it % 25 == 0 else min(m, max(2, m // 4 + 1))
        active = order[:K]

        grad = np.zeros(m, dtype=float)
        for k in active:
            k = int(k)
            start = max(0, k - (m - 1))
            stop = min(m - 1, k)
            for i in range(start, stop + 1):
                grad[i] += 2.0 * c[k - i]
        grad /= float(K)

        improved = False
        for trial in (lr, lr * 0.35, lr * 0.1, lr * 3.0):
            cand = project_simplex(c - trial * grad)
            sc = coarse_score_p(cand)
            if sc < best_s - 1e-15:
                c = cand
                best_c = cand.copy()
                best_s = sc
                improved = True
                break

        if not improved and it % 25 == 0:
            scale = 0.06 + 0.08 / (1.0 + 0.01 * it)
            cand = project_simplex(c + _RNG.normal(0.0, scale, size=m))
            sc = coarse_score_p(cand)
            if sc < best_s - 1e-15:
                c = cand
                best_c = cand.copy()
                best_s = sc

        if it % 100 == 99:
            lr = max(0.03, lr * 0.95)

    return best_c, best_s


def search_one_m(m, deadline, seed, current_best):
    """Search a coarse sequence of fixed length m."""
    m = int(m)
    best_c = None
    best_s = float(current_best)
    seed = int(seed)

    # Global random differential evolution in the bounded cube.
    try:
        if time.monotonic() < deadline:
            if m <= 8:
                maxiter = 1000
                popsize = 8
            elif m <= 16:
                maxiter = 700
                popsize = 5
            else:
                maxiter = 260
                popsize = 4

            bounds = [(0.0, 1.0) for _ in range(m)]
            de = differential_evolution(
                coarse_raw_score,
                bounds,
                popsize=popsize,
                maxiter=maxiter,
                seed=seed,
                tol=1e-9,
                polish=False,
            )
            x = np.asarray(de.x, dtype=float)
            x = np.clip(x, 0.0, None)
            sx = float(x.sum())
            if sx > 1e-12:
                c = x / sx
                sc = float(coarse_score_p(c))
                if np.isfinite(sc) and sc < best_s:
                    best_s = sc
                    best_c = c.copy()
    except Exception:
        pass

    # Cheap random fallback if DE did not beat the current best.
    if best_c is None:
        rng = np.random.default_rng(seed + 101)
        for _ in range(3000):
            c = project_simplex(rng.random(m))
            sc = float(coarse_score_p(c))
            if sc < best_s:
                best_s = sc
                best_c = c.copy()

    # Active-peak local refinement.
    if best_c is not None and time.monotonic() + 10.0 < deadline:
        c, sc = active_refine_c(
            best_c,
            5000 if m <= 12 else 2000,
            deadline,
        )
        if sc < best_s:
            best_s = sc
            best_c = c

    # Several softmax/Nelder-Mead starts to escape flat minima.
    rng = np.random.default_rng(seed + 777)
    starts = 5 if m <= 10 else 3
    maxit = 3500 if m <= 10 else 1400
    for st in range(starts):
        if time.monotonic() > deadline:
            break
        if st == 0 and best_c is not None:
            t0 = np.log(np.maximum(best_c, 1e-12))
        else:
            t0 = rng.normal(0.0, 0.65, size=m)

        try:
            res = minimize(
                softmax_score_t,
                t0,
                method="Nelder-Mead",
                options={"maxiter": maxit, "adaptive": True},
            )
            p = softmax_vector(res.x)
            sc = float(coarse_score_p(p))
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_c = p.copy()
        except Exception:
            pass

    return best_c, best_s


def run_full_refine(c, N, m, deadline):
    """Refine a coarse sequence using the actual stretched sequence objective."""
    global _FULL_LENGTHS

    c = np.asarray(c, dtype=float).ravel()
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.clip(c, 0.0, None)
    s = float(c.sum())
    if s <= 1e-15:
        c = np.full(max(1, int(m)), 1.0 / max(1, int(m)))
    else:
        c = c / s

    lengths = make_lengths(int(N), int(m))
    _FULL_LENGTHS = lengths

    seq = stretch_from_c(c, N)
    best_c = c.copy()
    best_sc = fast_score_full(seq)
    best_seq = seq.copy()

    if time.monotonic() + 30.0 < deadline:
        try:
            res = minimize(
                full_score_from_raw_global,
                c.copy(),
                method="Nelder-Mead",
                options={"maxiter": 800 if m <= 12 else 400, "adaptive": True},
            )
            xf = np.asarray(res.x, dtype=float)
            xf = np.where(np.isfinite(xf), xf, 0.0)
            xf = np.clip(xf, 0.0, None)
            sx = float(xf.sum())
            if sx > 1e-12:
                cf = xf / sx
                seq2 = stretch_from_c(cf, N)
                sc2 = fast_score_full(seq2)
                if sc2 < best_sc:
                    best_c = cf.copy()
                    best_sc = sc2
                    best_seq = seq2
        except Exception:
            pass

    # Short random recommendations around the best coarse point.
    rng = np.random.default_rng(seed=918273645)
    for i in range(40):
        if time.monotonic() > deadline:
            break
        scale = 0.05 + 0.12 / (1.0 + 0.05 * i)
        cf = project_simplex(best_c + rng.normal(0.0, scale, size=len(best_c)))
        seq2 = stretch_from_c(cf, N)
        sc2 = fast_score_full(seq2)
        if sc2 < best_sc:
            best_c = cf.copy()
            best_sc = sc2
            best_seq = seq2

    return best_c, best_sc, best_seq


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Find a nonnegative sequence minimizing:
        2 * n * max(convolve(a, a)) / sum(a)^2
    by optimizing a short coarse sequence and stretching it to the working length.
    """
    start = time.monotonic()
    deadline = start + max(1.0, float(budget_s) - 30.0)

    # Choose a working length.
    N = 6962
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        hs = hs.ravel()
        if hs.size >= 16 and np.all(np.isfinite(hs)):
            N = int(hs.size)
    except Exception:
        pass

    if N < 64:
        N = 6962
    if N > 12000:
        N = 12000

    # Uniform fallback.
    best_seq = np.ones(N, dtype=float)
    best_score = fast_score_full(best_seq)

    best_c = None
    current_best_coarse = 2.0

    print(f"[init] N={N}, uniform={best_score:.10f}, budget={budget_s:.0f}s", flush=True)

    # Short coarse lengths are efficient; increasing m gives more room to flatten
    # the convolution, while DE + refinement remains cheap for m <= 32.
    m_list = [7, 8, 9, 10, 12, 14, 16, 18, 20, 24, 28, 32]

    for m in m_list:
        if time.monotonic() > deadline - 180.0:
            break
        if m <= 1 or m > N:
            continue

        c, sc = search_one_m(m, deadline, int(seed) + 1000 * m, current_best_coarse)

        if c is not None and np.isfinite(sc) and sc < current_best_coarse:
            current_best_coarse = sc
            best_c = c.copy()
            print(f"[coarse] m={m}, score={sc:.10f}", flush=True)
        else:
            # Uniform coarse sequence as a valid fallback for this m.
            c = np.full(m, 1.0 / m)

        seq = stretch_from_c(c, N)
        fsc = fast_score_full(seq)
        if fsc < best_score:
            best_score = fsc
            best_seq = seq.copy()
            print(f"[stretched] m={m}, full={fsc:.10f}", flush=True)

        # If we are already safely below the target, polish with full-sequence
        # objective and possibly stop.
        if best_score < 1.495 and time.monotonic() + 90.0 < deadline:
            c2, fsc2, seq2 = run_full_refine(c, N, m, deadline)
            if fsc2 < best_score:
                best_score = fsc2
                best_seq = seq2.copy()
                best_c = c2.copy()
                print(f"[full-refine] m={m}, full={fsc2:.10f}", flush=True)

            if best_score < 1.495:
                break

    # Final sanity checks and return as a Python list of floats.
    final = np.asarray(best_seq, dtype=float)
    final = final.ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)

    if final.size == 0 or float(np.sum(final)) < 0.01:
        final = np.ones(6962, dtype=float)

    return [float(x) for x in final]
```
