Minimize 2*n*max(conv(a,a))/sum(a)^2 via multi-start subgradient descent.
Since the objective is scale-invariant, normalize sum(a)=1 and minimize 2*n*max(conv(a,a)).
Uses top-K active set subgradient with backtracking line search and periodic perturbations.
"""

import numpy as np
import time


def _b_of_a(a):
    return np.convolve(a, a, mode='full')


def _subgrad(a, b, topk):
    n = a.size
    m = b.size
    k = min(topk, m)
    if k < m:
        idx = np.argpartition(b, m - k)[m - k:]
    else:
        idx = np.arange(m)
    g = np.zeros(n)
    for ki in idx:
        ki = int(ki)
        lo = ki - n + 1
        if lo < 0:
            lo = 0
        hi = ki
        if hi >= n:
            hi = n - 1
        if lo <= hi:
            ss = ki - hi
            se = ki - lo
            g[lo:hi + 1] += 2.0 * a[ss:se + 1][::-1]
    g /= float(k)
    g -= g.mean()
    return g


def _proj_step(a, g, alpha):
    x = a - alpha * g
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s > 1e-15:
        x /= s
    else:
        x = a.copy()
    return x


def _run_opt(a0, tlimit, rng, topk0=10, topk1=300):
    n = a0.size
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a /= ss

    dl = time.monotonic() + tlimit
    ba = a.copy()
    bb = _b_of_a(a)
    bm = float(bb.max())
    alpha = 2.0 / n
    it = 0
    stall = 0

    while time.monotonic() < dl - 0.5:
        b = _b_of_a(a)
        mb = float(b.max())
        if mb < bm:
            bm = mb
            ba = a.copy()
            stall = 0
        else:
            stall += 1

        prog = min(1.0, 4.0 * it / max(1, tlimit))
        tk = int(topk0 + (topk1 - topk0) * prog)

        g = _subgrad(a, b, tk)

        best_alpha = alpha
        best_a = a
        best_mb = mb
        cur_alpha = alpha
        for _ in range(15):
            an = _proj_step(a, g, cur_alpha)
            mbn = float(_b_of_a(an).max())
            if mbn < mb * (1.0 - 1e-12):
                best_a = an
                best_mb = mbn
                best_alpha = cur_alpha
                break
            cur_alpha *= 0.5
        a = best_a
        if best_mb < mb:
            alpha = min(best_alpha * 1.03, 50.0 / n)
        else:
            alpha = best_alpha

        if it % 4000 == 3999:
            a = a + 0.008 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            alpha = 2.0 / n

        if stall > 8000:
            a = ba + 0.04 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            alpha = 2.0 / n
            stall = 0

        it += 1

    score = 2.0 * n * bm
    return ba, score

        starts.append(np.maximum(bi, 0.0))
    for i in range(len(starts)):
        starts[i] = np.maximum(starts[i], 1e-15)
    return starts
