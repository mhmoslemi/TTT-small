def peak_redistribute(a, b, rng):
    n = len(a)
    a_new = a.copy()
    
    # Find the peak
    k_max = np.argmax(b)
    
    # Compute the contribution of each a[j] to b[k_max]
    # db[k_max]/da[j] = 2*a[k_max - j] (if 0 <= k_max - j < n)
    contrib = np.zeros(n)
    for j in range(n):
        other = k_max - j
        if 0 <= other < n:
            contrib[j] = 2 * a[other]
    
    # Also consider other near-max peaks
    sorted_b = np.sort(b)[::-1]
    threshold = sorted_b[min(5, len(b))-1]  # top 5 peaks
    peak_indices = np.where(b >= threshold * 0.9)[0]
    for k in peak_indices:
        if k == k_max: continue
        weight = (b[k] - threshold * 0.5) / (b[k_max] - threshold * 0.5 + 1e-12)
        for j in range(n):
            other = k - j
            if 0 <= other < n:
                contrib[j] += weight * 2 * a[other]
    
    # Move mass: decrease a[j] where contrib[j] is high, increase where it's low
    total_contrib = contrib.sum()
    if total_contrib < 1e-12:
        return a_new
    
    # Fraction of mass to move
    frac = rng.uniform(0.01, 0.05)
    
    # Create a "desirable" distribution (inverse of contribution)
    # Positions with low contribution are "desirable" for receiving mass
    desirable = 1.0 / (contrib + 1e-12)
    desirable /= desirable.sum()
    
    # Remove mass from high-contribution positions
    remove_frac = frac * a / a.sum() * (contrib / contrib.max())

    a_new = np.clip(a_new, 0, 1000)
    
    # Add mass to desirable positions
    total_removed = remove_frac.sum()
    a_new += total_removed * desirable

The key observation is: if the peak of the convolution is at k*, then reducing a[j] for j where a[k*-j] is large will reduce the peak. At the same time, we should increase a[j] for j where a[k*-j] is small (to maintain or increase the sum).

def peak_shave(a, b, rng, lr=0.01):
    n = len(a)
    a_new = a.copy()
    k_max = np.argmax(b)
    
    # For each position j, the "danger" of increasing a[j] is a[k_max - j]
    danger = np.zeros(n)
    for j in range(n):
        other = k_max - j
        if 0 <= other < n:
            danger[j] = a[other]
    
    # Also consider the gradient of the objective:
    # dF/da[j] ∝ danger[j] - M/S (from the full gradient)
    S = a.sum()
    M = b.max()
    threshold = M / S  # positions with danger < threshold are "safe" to increase
    
    # Move: decrease dangerous positions, increase safe positions
    n_move = rng.integers(1, max(2, n // 50))
    for _ in range(n_move):
        # Pick a dangerous position to decrease
        danger_mask = danger > threshold
        if danger_mask.sum() == 0:
            break
        j = rng.choice(np.where(danger_mask)[0])
        # Pick a safe position to increase
        safe_mask = danger <= threshold
        if safe_mask.sum() == 0:
            break
        i = rng.choice(np.where(safe_mask)[0])
        # Transfer mass
        transfer = min(a_new[j], lr * a[j])
        a_new[j] -= transfer
        a_new[i] += transfer
    
    return np.clip(a_new, 0, 1000)
