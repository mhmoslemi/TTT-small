import numpy as np
import time

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def subgradient_step(a, learning_rate):
    """One subgradient step."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    
    if S < 0.01:
        return a, np.inf
    
    # Find active set
    tol = 1e-10 * M
    active = np.where(conv >= M - tol)[0]
    
    # Compute subgradient of M
    # ∂M/∂a_i = (1/|K*|) * sum_{k in K*} a_{k-i}
    grad_M = np.zeros(n)
    for i in range(n):
        # For each active k, a_{k-i} contributes if 0 ≤ k-i < n
        # k-i >= 0 => k >= i
        # k-i < n => k < i+n
        valid_k = active[(active >= i) & (active < i + n)]
        if len(valid_k) > 0:
            grad_M[i] = np.sum(a[valid_k - i]) / len(active)
    
    # Full gradient of score
    grad = (2.0 * n / (S * S)) * (grad_M - 2.0 * M / S)
    
    # Update
    a_new = a - learning_rate * grad
    a_new = np.clip(a_new, 0, 1000)
    
    return a_new, compute_score(a_new)

import time

def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1 << (2*n-2).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def _count_bits(x):
    return 1 << (x-1).bit_length()

def _compute_score(a):
    n = len(a)
    conv = _fft_convolve(a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S*S)

def _smooth_grad(a, eps):
