        # partner[m] = a[k-m] for m in [lo, hi]
        # k-lo, k-(lo+1), ..., k-hi (reversed)
        partner[lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    # Full gradient of F = 2n*max_c/S^2:
    # dF/da[m] = (4n/S^2) * (partner[m] - max_c/S)
    grad = (4.0 * n / (S * S)) * (partner - max_c / S)
    
    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

def multi_peak_gradient(a, p=4):
    """
    Compute a smooth gradient using the p-th power sum of the convolution.
    Returns (gradient, current_objective).
    """
    n = len(a)
    c = compute_conv(a)
    S = np.sum(a)
    max_c = np.max(c)
    
    if S < 1e-10:
        return np.zeros(n), np.inf
    
    # Use power sum for smoothness
    # Normalize to avoid overflow
    c_pos = np.maximum(c, 1e-30)
    log_c = np.log(c_pos)
    log_max = np.log(max_c) if max_c > 0 else 0
    
    # Soft max: G = (sum exp(p*(log_c - log_max)))^{1/p} * max_c
    shifted = p * (log_c - log_max)
    shifted = np.clip(shifted, -50, 50)  # Avoid overflow
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    G = max_c * sum_exp ** (1.0 / p)
    
    # Gradient: dG/da[m] = (1/sum_exp) * sum_k exp_shifted[k] * 2*a[k-m] * max_c
    # = (2*max_c/sum_exp) * sum_k exp_shifted[k] * a[k-m]
    w = exp_shifted  # weights proportional to c[k]^{p-1} (approximately)
    
    # Compute cross-correlation: for each m, sum_k w[k] * a[k-m]
    # Using FFT for efficiency
    # sum_k w[k] * a[k-m] for m = 0, ..., n-1
    # This is the "anti-diagonal" sum of the outer product w ⊗ a
    # = cross-correlation of w and a at lag m
    # = ifft(conj(fft(w, N)) * fft(a, N))[m] for appropriate N
    
    N = 2 * n  # FFT size
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: ifft(conj(W) * A) gives sum_k w[k+m] * a[k]
    # We want sum_k w[k] * a[k-m] = sum_j w[j+m] * a[j] (substituting j=k-m)
    # = (w cross-corr a)[m]
    corr = np.fft.irfft(np.conj(W) * A, N)[:n]
    
    grad_G = (2.0 * max_c / sum_exp) * corr
    
    # Full gradient of F = 2n*G/S^2
    grad_F = (2.0 * n / S**2) * grad_G - (4.0 * n * G / S**3) * S
    # Simplify: = (2n/S^2) * (grad_G - 2*G)
    # Wait: dF/da = (2n/S^2) * dG/da - (2n*G/S^3) * dS/da
    # dS/da[m] = 1
    # So dF/da[m] = (2n/S^2) * grad_G[m] - 2n*G/S^3
    grad_F = (2.0 * n / S**2) * grad_G - 2.0 * n * G / S**3
    
    score = 2.0 * n * max_c / (S * S)
    return grad_F, score

def lp_refine(a, max_time=30.0):
    """
    Solve the LP to find a better direction.
    Returns the optimal b direction (rescaled to match sum of a) or None.
    """
    n = len(a)
    c = compute_conv(a)
    M = np.max(c)
    S = np.sum(a)
    
    if M < 1e-15 or S < 1e-10:
        return None
    
    # Build sparse constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            val = a[k - j]
            if val > 1e-15:  # Skip zeros for sparsity
                rows.append(k)
                cols.append(j)
                data.append(val)
    
    if not data:
        return a.copy()
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    
    # Maximize sum(b) s.t. A_ub @ b <= M, b >= 0
    # => Minimize -sum(b) s.t. A_ub @ b <= M, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'time_limit': max_time})
    except Exception:
        return None
    
    if not result.success:
        return None
    
    b_opt = result.x
    S_b = np.sum(b_opt)
    if S_b < 1e-10:
        return None
    
    # Rescale to match sum of a
    b_rescaled = b_opt * (S / S_b)
    return b_rescaled
