Minimize 2*n*max(conv(a,a))/sum(a)^2 via subgradient descent on the
probability simplex (a>=0, sum(a)=1). The objective is scale-invariant,
so we normalize sum(a)=1 and minimize 2*n*max(conv(a,a)).
"""

import numpy as np
import time


def _compute_b(a):
    """Return conv(a,a) as float64 array."""
    return np.convolve(a, a)


def _subgrad_topk(a, b, k=50):
    """Subgradient of max(b) w.r.t. a using top-k active set."""
    n = len(a)
    m = len(b)  # 2n-1
    k = min(k, m)
    # Get indices of top k elements of b
    if k < m:
        idx = np.argpartition(b, m - k)[m - k:]
    else:
        idx = np.arange(m)
    g = np.zeros(n)
    for ki in idx:
        ki = int(ki)
        lo = ki - n + 1
        if lo < 0:
            lo = 0
        hi = ki
        if hi >= n:
            hi = n - 1
        if lo <= hi:
            seg_start = ki - hi
            seg_end = ki - lo
            g[lo:hi+1] += 2.0 * a[seg_start:seg_end+1][::-1]
    g /= float(len(idx))
    g -= g.mean()
    return g


def _opt_step(a, g, alpha):
    """One projected gradient step: a - alpha*g, project to simplex."""
    a_new = a - alpha * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def _subgrad(a, b, topk):
    n = a.size
    m = b.size
    k = min(topk, m)
    if k < m:
        idx = np.argpartition(b, m - k)[m - k:]
    else:
        idx = np.arange(m)
    g = np.zeros(n)
    for ki in idx:

        if lo <= hi:
