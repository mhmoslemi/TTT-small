```python
"""
Inequality-constructed search for low-height step-function coefficients.

This program searches for a non-negative sequence minimizing

    2 * n * max(a * a) / (sum(a))**2

where `a * a` is the linear convolution used by `evaluate_sequence`.

Algorithm:
  * Start from several strong initials:
      - the provided global `height_sequence_1` if it exists,
      - one-sided central-binomial sequences (asymptotic score pi/2 ~ 1.5708),
      - a few lengths around 4k / 8k / 16k / 20k.
  * Optimize in normalized probability-space using a soft-max subgradient of
    the maximum convolution coefficient.
  * The gradient is applied in log-variables, which keeps coefficients
    non-negative.  A small line-search over step sizes is used.
  * Adaptive temperature focuses the subgradient on the tight coefficients.
  * If the search stalls, controlled random asymmetric perturbations and
    blends with fresh central-binomial shapes are used to escape local minima.
  * The best normalized sequence is rescaled before returning so that the
    submitted list is numerically comfortable for the official evaluator.
"""

import math
import time

import numpy as np

_TARGET = 1.5027
_MAX_N = 20000


def _fft_len_for_n(n):
    """Power-of-two FFT length sufficient for both p*p and p*reversed(w)."""
    n = int(n)
    if n <= 1:
        return 1
    need = 3 * n - 2
    L = 1
    while L < need:
        L <<= 1
    return L


def _valid_normalized(p):
    """Clip to evaluator-safe values, make strictly positive, normalize sum=1."""
    p = np.asarray(p, dtype=np.float64).reshape(-1).copy()
    if p.size == 0:
        return np.ones(1, dtype=np.float64)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)

    s = float(p.sum())
    if not math.isfinite(s) or s <= 0.0:
        p.fill(1.0)
        s = float(p.size)

    p = p / s
    pmin = 1.0e-12 / float(p.size)
    p = np.maximum(p, pmin)

    s = float(p.sum())
    if not math.isfinite(s) or s <= 0.0:
        p.fill(1.0)
        s = float(p.size)
    p = p / s
    return p


def _score_with_L(p, L):
    """
    Fast score using FFT convolution.

    Returns:
      (score, max_coefficient, convolution_coefficients, rfft(p))
    """
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = int(p.size)
    if n == 0:
        return float("inf"), 0.0, None, None

    s = float(p.sum())
    if not math.isfinite(s) or s <= 0.0:
        return float("inf"), 0.0, None, None

    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)[: 2 * n - 1]
    c[c < 0.0] = 0.0

    m = float(c.max()) if c.size else 0.0
    if not math.isfinite(m) or m < 0.0:
        m = 0.0

    score = 2.0 * float(n) * m / (s * s)
    if not math.isfinite(score):
        return float("inf"), m, c, Fp
    return score, m, c, Fp


def _weights_from_convolution(c, M, beta):
    """
    Nonnegative weights over convolution coefficients, concentrating near max.

    If beta is very large, use an active-set uniform weight over exact/near max.
    """
    c = np.asarray(c, dtype=np.float64)
    n = int(c.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)

    if M <= 0.0 or not math.isfinite(M):
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    # Exact active-set for very sharp temperatures.
    if beta >= 1.0e4:
        tol = 1.0e-10 * M + 1.0e-300
        mask = c >= M - tol
        if not mask.any():
            mask = np.zeros(n, dtype=bool)
            idx = int(np.argmax(c))
            mask[idx] = True
        w = mask.astype(np.float64)
        s = float(w.sum())
        if s <= 0.0:
            w.zero()
            w[int(np.argmax(c))] = 1.0
            s = 1.0
        return w / s

    rel = (M - c) / M
    rel = np.maximum(rel, 0.0)
    z = float(beta) * rel
    w = np.exp(-z)
    s = float(w.sum())

    if not math.isfinite(s) or s <= 0.0:
        tol = 1.0e-10 * M + 1.0e-300
        mask = c >= M - tol
        if not mask.any():
            mask = np.zeros(n, dtype=bool)
            idx = int(np.argmax(c))
            mask[idx] = True
        w = mask.astype(np.float64)
        s = float(w.sum())
        if s <= 0.0:
            w.zero()
            w[int(np.argmax(c))] = 1.0
            s = 1.0

    return w / s


def _gradient_from_weights(Fp, w, n, L):
    """
    Natural gradient component G_i = sum_k w_k * p_{k-i} for convolution.

    Computed by convolution of p with reversed w.
    """
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)

    w = np.asarray(w, dtype=np.float64)
    if w.size != 2 * n - 1:
        # Fallback for unexpected sizes: uniform zero gradient.
        return np.zeros(n, dtype=np.float64)

    FW = np.fft.rfft(w[::-1], L)
    H = np.fft.irfft(Fp * FW, L)

    # G_i = H[2n-2-i], for i=0,...,n-1.
    G = H[n - 1 : 2 * n - 1][::-1].copy()
    G = np.where(np.isfinite(G), G, 0.0)
    G = np.maximum(G, 0.0)
    return G


def _update_log(p, direction, eta):
    """Multiplicative log-update p_i <- p_i * exp(-eta * direction_i)."""
    p = np.asarray(p, dtype=np.float64)
    direction = np.asarray(direction, dtype=np.float64)
    if p.size != direction.size:
        return p.copy()

    exponent = -float(eta) * direction
    exponent = np.clip(exponent, -50.0, 50.0)
    q = p * np.exp(exponent)
    return _valid_normalized(q)


def _unique_etas(vals):
    """Deduplicate positive eta candidates on a log scale and cap them."""
    out = []
    for v in vals:
        try:
            v = float(v)
        except Exception:
            continue
        if not math.isfinite(v) or v <= 0.0:
            continue
        if v < 1.0e-5:
            v = 1.0e-5
        if v > 10.0:
            v = 10.0

        duplicate = False
        for u in out:
            try:
                if abs(math.log(v) - math.log(u)) < 0.35:
                    duplicate = True
                    break
            except Exception:
                duplicate = True
                break
        if not duplicate:
            out.append(v)

    out.sort()
    return out[:7]


def _perturb(p, rng, strength=0.08):
    """Random asymmetric perturburbation in log space."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return p.copy()

    strength = max(1.0e-4, float(strength))
    noise = rng.normal(0.0, strength, size=n)

    slope = rng.normal(0.0, 0.7 * strength)
    if n > 2:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        noise = noise + slope * (x - 0.5)

    if rng.random() < 0.35:
        lo = max(1, n // 100)
        hi = max(lo + 1, n // 5)
        width = int(rng.integers(lo, hi))
        if width >= 1 and n - width > 0:
            start = int(rng.integers(0, n - width))
            noise[start : start + width] += rng.normal(0.0, 1.5 * strength)

    q = p * np.exp(np.clip(noise, -10.0, 10.0))
    return _valid_normalized(q)


def _binomial_seq(n):
    """
    Central-binomial profile: C_i = binom(2i,i) / 4^i.

    Its square has constant first-half coefficients, giving asymptotic score
    pi/2 ~= 1.5708.  This is a strong asymmetric starting point.
    """
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)

    out = np.empty(n, dtype=np.float64)
    v = 1.0
    for i in range(n):
        out[i] = v
        if i + 1 < n:
            v *= (2 * i + 1) / (2.0 * (i + 1))

    s = float(out.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64)
    return out / s


def _downsample_positive(arr, target):
    """Average contiguous blocks to reduce length while staying nonnegative."""
    arr = np.asarray(arr, dtype=np.float64).reshape(-1)
    n = int(arr.size)
    target = int(target)
    if target <= 0 or n <= target:
        return arr

    block = max(1, n // target)
    m = n // block
    if m < 256:
        # If averaging would make it too short, use a strided sample.
        step = max(1, n // target)
        arr = arr[::step]
        if arr.size > target:
            arr = arr[:target]
        return arr

    arr = arr[: m * block]
    out = arr.reshape(m, block).mean(axis=1)
    return out


def _get_global_array():
    """Safely fetch the optional global construction `height_sequence_1`."""
    try:
        return height_sequence_1
    except NameError:
        return None


def _make_initials(max_n):
    """Build a small pool of initial normalized candidate sequences."""
    max_n = int(max_n)
    candidates = []

    g = _get_global_array()
    if g is not None:
        g = np.asarray(g, dtype=np.float64).reshape(-1)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.clip(g, 0.0, 1000.0)

        if g.size >= 128:
            if g.size > max_n:
                g = _downsample_positive(g, max_n)

            if g.size >= 128 and g.sum() > 0.0:
                p = _valid_normalized(g)
                L = _fft_len_for_n(p.size)
                sc, _, _, _ = _score_with_L(p, L)
                if math.isfinite(sc):
                    candidates.append((sc, p))

    preferred = [4096, 8192, 16384, max_n]
    if g is not None and 256 <= g.size <= max_n:
        preferred.append(int(g.size))

    selected = []
    for n in preferred:
        n = int(n)
        if n < 256 or n > max_n:
            continue
        ok = True
        for m in selected:
            if m == n:
                ok = False
                break
        if ok:
            selected.append(n)

    for n in selected:
        p = _binomial_seq(n)
        L = _fft_len_for_n(n)
        sc, _, _, _ = _score_with_L(p, L)
        if math.isfinite(sc):
            candidates.append((sc, p))

    return candidates


def _scale_output(p):
    """
    Return a list that is safe for the official evaluator.

    The objective is scale-invariant, but moderate scaling helps precision.
    """
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    if p.size == 0:
        return [1.0]

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, None)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 0.0:
        p = np.ones(p.size, dtype=np.float64)
        s = float(p.size)
    p = p / s

    n = int(p.size)
    scale = math.sqrt(2.0 * float(n))
    maxp = float(p.max())
    if maxp > 0.0:
        scale = min(scale, 1000.0 / (maxp * 1.01))
    scale = max(scale, 0.05)

    out = p * scale
    out = np.clip(out, 0.0, 1000.0)

    s2 = float(out.sum())
    if not math.isfinite(s2) or s2 < 0.01:
        out = np.full(n, max(0.02 / float(n), 1.0e-12), dtype=np.float64)

    return [float(x) for x in out]


def _optimize(p_in, deadline, rng):
    """
    Main soft-max natural-gradient optimizer.

    Returns normalized best sequence and its fast score.
    """
    p = _valid_normalized(p_in)
    n = int(p.size)
    if n < 4:
        L = _fft_len_for_n(n)
        sc, _, _, _ = _score_with_L(p, L)
        return p, sc

    L = _fft_len_for_n(n)
    score, M, c, Fp = _score_with_L(p, L)
    if not math.isfinite(score):
        p = _binomial_seq(n)
        score, M, c, Fp = _score_with_L(p, L)

    best = p.copy()
    best_score = score

    beta = 80.0
    no_progress = 0
    stuck_count = 0
    iter_count = 0
    last_eta = 0.2

    print(f"[_optimize] start n={n} score={best_score:.8f}")

    while time.monotonic() < max(deadline, time.monotonic()):
        if not math.isfinite(score):
            p = best.copy()
            score, M, c, Fp = _score_with_L(p, L)
            if not math.isfinite(score):
                break

        if score < best_score - 1.0e-12:
            best_score = score
            best = p.copy()
            no_progress = 0
            stuck_count = 0
            if iter_count < 2000 or iter_count % 500 == 0:
                print(f"[_optimize] improved n={n} iter={iter_count} best={best_score:.8f}")

        if M <= 0.0 or not math.isfinite(M):

        # If the current point is far worse than the incumbent, refocus.
        if score > best_score + 0.05:
            p = best.copy()
            score, M, c, Fp = _score_with_L(p, L)

            continue

            strength = 0.08 if best_score > _TARGET else 0.03

            candidate = _perturb(best, rng, strength)

                score = sc_try
                M = M_try

            else:

            continue

        direction = direction / drange

        best_try_p = p

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """

    deadline = start + max(1.0, budget - 10.0)

    candidates = _make_initials(_MAX_N)

    primary_score = float("inf")
    primary_p = None

    for sc, p in candidates:
        if math.isfinite(sc) and sc < primary_score:
            primary_score = sc
            primary_p = p.copy()

    if primary_p is None:
        primary_p = _binomial_seq(8192)
        L = _fft_len_for_n(primary_p.size)
        primary_score, _, _, _ = _score_with_L(primary_p, L)

    best_p, best_score = _optimize(primary_p, deadline, rng)

    # The optimizer always keeps its incumbent; update in case it returns a copy.
    if math.isfinite(best_score):
        pass
    else:
        best_p = primary_p.copy()
        best_score = primary_score

    out = _scale_output(best_p)
    try:
        print(f"[final] returned_n={len(out)} fast_best_score={float(best_score):.10f}")
    except Exception:
        pass

    return out
```
