```python
    # A[k, j] = contribution of g[j] to conv(f,g)[k]

    # Total nonzeros = sum_{k=0}^{2n-2} min(k+1, n, 2n-1-k) = n^2

    n_constraints = 2 * n - 1

    rows = []
    cols = []
    vals = []

        for j in range(max(0, k - n + 1), min(n, k + 1)):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
```

```python
def build_convolution_matrix(f):
    """Build the matrix A such that conv(f,g) = A @ g"""
    n = len(f)
    n_rows = 2 * n - 1
    n_cols = n
    
    # Use scipy sparse
    from scipy.sparse import lil_matrix
    A = lil_matrix((n_rows, n_cols))
    
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        js = np.arange(j_start, j_end)
        vals = f[k - js]
        A[k, js] = vals
    
    return A.tocsr()
```

```python
from scipy.sparse import csr_matrix
import time

def build_A(f):
    """Build matrix A such that conv(f,g) = A @ g"""
    n = len(f)
    n_rows = 2 * n - 1
    n_cols = n
    
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Build row by row
    all_rows = []
    all_cols = []
    all_vals = []
    
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            all_rows.extend([k] * (j_end - j_start))
            all_cols.extend(js.tolist())
            all_vals.extend(f[k - js].tolist())
    
    return csr_matrix((all_vals, (all_rows, all_cols)), shape=(n_rows, n_cols))

def lp_iteration(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    return None

def optimize(f, max_iters=100, t=0.5):
    """LP-based iterative optimization"""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    f = f / np.sum(f)  # normalize to sum=1
    
    best = f.copy()
    best_score = 2*n*np.max(np.convolve(f,f))/np.sum(f)**2

        conv_ff = np.convolve(f, f)

        g = lp_iteration(f, M)
        if g is None:
            break

        if g_sum <= 0:
            break
        
        g = g / g_sum  # normalize to sum=1
        
        # Blend
        f_new = (1-t)*f + t*g
```

```python
from scipy.sparse import csr_matrix
import time

def build_A_matrix(f):
    """Build sparse matrix A such that conv(f, g) = A @ g for all g."""
    n = len(f)
    n_rows = 2 * n - 1
    n_cols = n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        js = np.arange(j_start, j_end)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n_cols))
```

```python
"""
LP-based iterative optimization for minimizing the peak of self-convolution.
Uses the primal LP approach: at each step, solve an LP to find a direction
that maximizes total mass subject to the convolution bound, then blend.
Multiple starting points and sequence lengths are explored.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix
import time


def build_A_matrix(f):
    """Build sparse matrix A such that conv(f, g) = A @ g."""
    n = len(f)
    n_rows = 2 * n - 1
    n_cols = n
    
    # Efficient row-by-row construction
    data = []
    rows = []
    cols = []
    
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start >= j_end:
            continue
        js = np.arange(j_start, j_end)
        data.extend(f[k - js].tolist())
        rows.extend([k] * (j_end - j_start))
        cols.extend(js.tolist())
    
    return csr_matrix((data, (rows, cols)), shape=(n_rows, n_cols))


def build_A_rows(f, row_indices):
    """Build sparse matrix A with only specified rows."""
    n = len(f)
    n_rows = len(row_indices)
    n_cols = n
    
    data = []
    rows = []
    cols = []
    
    for idx, k in enumerate(row_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start >= j_end:
            continue
        js = np.arange(j_start, j_end)
        data.extend(f[k - js].tolist())
        rows.extend([idx] * (j_end - j_start))
        cols.extend(js.tolist())
    
    return csr_matrix((data, (rows, cols)), shape=(n_rows, n_cols))


def solve_lp_full(f, M):
    """Solve full LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_A_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.status == 0 and np.sum(res.x) > 0:
        return res.x
    return None


def solve_lp_reduced(f, M, n_active=200):
    """Solve LP with only top n_active constraints."""
    n = len(f)
    conv_ff = np.convolve(f, f)
    active_idx = np.argsort(conv_ff)[-n_active:]
    
    A = build_A_rows(f, active_idx)
    c = -np.ones(n)
    b = np.full(len(active_idx), M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.status == 0 and np.sum(res.x) > 0:
        return res.x
    return None


def compute_score(f):
    """Compute the evaluation score for a sequence."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * float(n) * float(np.max(conv)) / float(s * s)


def lp_optimize(f, deadline, max_iters=200, use_reduced=False, n_active=200):
    """LP-based iterative optimization with time limit."""
    f = np.asarray(f, dtype=np.float64).copy()
    n = len(f)
    
    s = np.sum(f)
    if s <= 0:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best = f.copy()
    best_score = compute_score(f)
    t = 0.3
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        
        if use_reduced:
            g = solve_lp_reduced(f, M, n_active=n_active)
        else:
            g = solve_lp_full(f, M)
        
        if g is None:
            break
        
        g_sum = float(np.sum(g))
        if g_sum <= 1e-15:
            break
        
        g_norm = g / g_sum
        
        # Try both blending and direct replacement
        candidates = []
        
        # Blended
        f_blend = (1 - t) * f + t * g_norm
        f_blend = np.maximum(f_blend, 0)
        sb = np.sum(f_blend)
        if sb > 0:
            f_blend = f_blend / sb
            candidates.append(f_blend)
        
        # Direct (if different enough)
        if not np.allclose(g_norm, f, atol=1e-6):
            candidates.append(g_norm)
        
        # Try smaller blends
        for t_try in [0.1, 0.5, 0.8]:
            f_try = (1 - t_try) * f + t_try * g_norm
            f_try = np.maximum(f_try, 0)
            st = np.sum(f_try)
            if st > 0:
                f_try = f_try / st
                candidates.append(f_try)
        
        # Evaluate all candidates
        best_candidate = f
        best_candidate_score = best_score
        
        for cand in candidates:
            sc = compute_score(cand)
            if sc < best_candidate_score:
                best_candidate_score = sc
                best_candidate = cand.copy()
        
        if best_candidate_score < best_score:
            best = best_candidate
            best_score = best_candidate_score
            f = best_candidate
            # Increase t slightly if improving
            t = min(0.8, t * 1.1)
        else:
            f = best_candidate  # still move, but don't update best
            t = max(0.05, t * 0.8)
        
        # Convergence check
        if np.max(np.abs(f - g_norm)) < 1e-10:
            break
    
    return best, best_score


def block_perturb(base, rng, width_range=(5, 50)):
    """Apply a random block perturbation."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(width_range[0], min(width_range[1], n) + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
    candidate = np.maximum(candidate, 0)
    return candidate


def local_search(base, deadline, rng, max_iters=5000):
    """Simple local search with block perturbations."""
    f = np.asarray(base, dtype=np.float64).copy()
    best = f.copy()
    best_score = compute_score(f)
    
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        cand = block_perturb(best, rng)
        sc = compute_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    
    return best, best_score
```

```python
def multiplicative_update(f, n_steps=1000):
    """Optimize using multiplicative (exponential) updates."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = f / np.sum(f)
    n = len(f)
    
    eta = 1.0  # learning rate
    best = f.copy()
    best_score = compute_score(f)
    
    for step in range(n_steps):
        conv = np.convolve(f, f)
        max_val = np.max(conv)
        # Find all lags where conv is close to max
        threshold = 0.99 * max_val
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient: average gradient of active q_k
        grad = np.zeros(n)
        for k in active:
            # q_k(a) = sum_i a[i]*a[k-i]
            # d/d a[j] q_k = a[k-j] + a[j-k] (with bounds)
            for j in range(n):
                val = 0
                if 0 <= k - j < n:
                    val += f[k-j]
                if 0 <= j - k < n and j != k-j:
                    val += f[j-k]
                elif 0 <= j - k < n:
                    val += f[j-k]  # this is a[j] itself when j=k-j
                grad[j] += val
        
        grad /= len(active)

        f = f / np.sum(f)
        
        score = compute_score(f)
        if score < best_score:
            best = f.copy()
            best_score = score
        
        # Adaptive eta
        if step % 50 == 0 and step > 0:
            eta *= 0.9  # slowly decrease
    
    return best, best_score
```

```python
    n = len(a)
    result = np.zeros(n)
    
    # Part 1: a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    j1s = max(0, k - n + 1)
    j1e = min(n, k + 1)
    if j1s < j1e:
        result[j1s:j1e] += a[k - np.arange(j1e - j1s)][::-1]

    j2e = min(n, k + n)

        # Remove double-counted diagonal
```
