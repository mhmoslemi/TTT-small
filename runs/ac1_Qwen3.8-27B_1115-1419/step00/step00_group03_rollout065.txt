"""
Frank-Wolfe optimization with LP subproblem to minimize the peak of the
self-convolution of a non-negative sequence, thereby minimizing the score
2n * max(conv(a,a)) / (sum(a))^2. Uses multiple diverse initializations
and active-set LP reduction for efficiency.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_constraint_matrix(p, n, active_rows=None):
    """Build constraint matrix A where (A @ q)_j = conv(p, q)_j for row j."""
    if active_rows is not None:
        rows = active_rows
    else:
        rows = np.arange(2 * n - 1)
    A = np.zeros((len(rows), n))
    for idx, j in enumerate(rows):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        if k_lo <= k_hi:
            A[idx, k_lo:k_hi + 1] = p[j - k_hi:j - k_lo + 1][::-1]
    return A


def solve_lp_subproblem(p, n, M, conv_prev, use_active=True):
    """Solve LP: max sum(q) s.t. conv(p,q)_j <= M for all j, q >= 0."""
    if use_active:
        threshold = 0.4 * M
        active = np.where(conv_prev >= threshold)[0]
        peak_row = np.argmax(conv_prev)
        if peak_row not in active:
            active = np.append(active, peak_row)
    else:
        active = np.arange(2 * n - 1)

    A = build_constraint_matrix(p, n, active)
    c = -np.ones(n)
    b = np.full(len(active), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 8.0})
        if res.status != 0:
            return None
        return res.x
    except Exception:
        return None


def compute_peak_score(p):
    """Compute 2n * max(conv(p,p)) assuming sum(p)=1."""
    n = len(p)
    conv = np.convolve(p, p)
