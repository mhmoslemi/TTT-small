```python
def peak_reduction_step(a, k_peak):
    n = len(a)
    # Compute the "cost" direction
    # d[j] should be negative where a[k_peak - j] is large
    cost = np.zeros(n)
    for j in range(n):
        idx = k_peak - j
        if 0 <= idx < n:
            cost[j] = a[idx]
    
    # Center to maintain sum
    cost -= np.mean(cost)
    
    # Direction is -cost (reduce where cost is high)
    d = -cost
    
    # Find max step size
    # a + eps*d >= 0
    # For j where d[j] < 0: eps <= -a[j]/d[j]
    neg_mask = d < 0
    if np.any(neg_mask):
        eps_max = np.min(-a[neg_mask] / d[neg_mask])
    else:
        eps_max = 1.0
    
    # Also limit step size for numerical stability
    eps = min(eps_max * 0.9, np.max(np.abs(a)) * 0.1)
    
    a_new = np.maximum(0, a + eps * d)
    # Renormalize
    if np.sum(a_new) > 0:
        a_new = a_new / np.sum(a_new) * np.sum(a)
    return a_new
```
