"""
Optimize step function sequence to minimize 2*n*max(conv)/sum^2 using:
1. Central binomial coefficient initialization (flat convolution top)
2. LP-based iterative refinement (maximize sum subject to cross-conv bound)
3. Multiplicative weight updates as fallback
4. Random perturbation polish
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def central_binomial(n):
    """Generate first n central binomial coefficients C(2j,j)/4^j."""
    a = np.ones(n)
    for j in range(1, n):
        a[j] = a[j - 1] * (2 * j - 1) / (2 * j)
    return a


def build_A_rowcol(n):
    """Build row and column indices for the convolution constraint matrix (reusable)."""
    col_indices = np.repeat(np.arange(n), n)
    row_indices = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    val_indices = np.tile(np.arange(n), n)
    return row_indices, col_indices, val_indices


def build_conv_matrix_sparse(f, row_indices, col_indices, val_indices, n):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    vals = f[val_indices]
    A = sp.csr_matrix((vals, (row_indices, col_indices)), shape=(2 * n - 1, n))
    return A


def solve_lp_for_b(f, M, row_indices, col_indices, val_indices, n):
    """Solve LP: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0."""
    A = build_conv_matrix_sparse(f, row_indices, col_indices, val_indices, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x
    return None


def multiplicative_update(f, n, lr=0.05):
    """One step of multiplicative weight update to flatten convolution."""
    conv = np.convolve(f, f)
    M = np.max(conv)
    threshold = M * 0.999
    max_indices = np.where(conv >= threshold)[0]

    # Compute pressure
    n_max = len(max_indices)
    if n_max == 0:
        return f.copy()

    # Fast path: if max_indices is approximately [0, n-1]
    if n_max >= n * 0.9 and max_indices[0] <= 1 and max_indices[-1] >= n - 2:
        cumsum = np.concatenate(([0.0], np.cumsum(f)))
        pressure = np.zeros(n)
        for j in range(n):
            count = n - j
            if count > 0:
                pressure[j] = cumsum[n - j] / count
    else:
        pressure = np.zeros(n)
        for j in range(n):
            valid = (max_indices >= j) & (max_indices < j + n)
            if np.any(valid):
                idx = max_indices[valid] - j
                pressure[j] = np.mean(f[idx])

    max_p = np.max(pressure)
    if max_p <= 0:
        return f.copy()

    update = 1.0 - lr * pressure / max_p
    f_new = f * update
    f_new = np.clip(f_new, 0, None)
    S = np.sum(f_new)
    if S < 1e-10:
        return f.copy()
    f_new = f_new * np.sum(f) / S
    return f_new
