    else:
        a = np.ones(n, dtype=np.float64) / n
    return a, n_neg


def make_random_start(n, rng, spread):
    """Generate a random non-negative start with given spread."""
    a = np.abs(rng.standard_normal(n)) * spread
    a[0] = 1.0
    return normalize_nonneg(a)


def make_ramp_start(n, direction=1):
    """Linear ramp from near-zero to max (or reverse)."""
    if direction == 1:
        a = np.linspace(0.01, 1.0, n, dtype=np.float64)
    else:
        a = np.linspace(1.0, 0.01, n, dtype=np.float64)
    return normalize_nonneg(a)


def make_sine_start(n, freq=1.0, amp=0.7):
    """Sinusoidal modulation of a constant."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0.0)
    return normalize_nonneg(a)


def optimize_smoothed_max(a, fft_size, deadline, rng, verbose=True):
    """Run smoothed-max subgradient descent. Returns (best_a, best_score)."""
    n = len(a)
    a = normalize_nonneg(a).copy()
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    if verbose:
        print(f"  Start score: {best_score:.8f}")

    t_temp = 5.0
    lr = 0.01
    iter_count = 0
    no_improve = 0
    max_no_improve = 80

    while time.monotonic() < deadline:
        # Compute convolution
        b = compute_conv_fft(a, fft_size)[:2 * n - 1]

        # Softmax weights (stable)
        b_max = b.max()
        logits = t_temp * (b - b_max)
        p = np.exp(logits)
        p_sum = p.sum()
        if p_sum > 1e-30:
            p /= p_sum
        else:
            p = np.zeros(2 * n - 1, dtype=np.float64)
            p[b.argmax()] = 1.0

        # Gradient
        grad = compute_grad_fft(a, p, fft_size)

        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        else:
            break
        a = a_new
        iter_count += 1

        # Periodic exact evaluation
        if iter_count % 20 == 0:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best = a.copy()
                best_score = score
                no_improve = 0
                if verbose and iter_count % 200 == 0:
                    print(f"  iter={iter_count} score={best_score:.8f} t={t_temp:.1f} lr={lr:.5f}")
            else:
                no_improve += 1

        # Schedule
        if iter_count % 150 == 0:
            t_temp = min(t_temp * 1.4, 800.0)
            lr = max(lr * 0.93, 0.0005)

        if no_improve > max_no_improve:
            # Random restart from best with perturbation
            noise = 0.02 * best.max() * rng.standard_normal(n)
            a = normalize_nonneg(best + noise)
            t_temp = max(2.0, t_temp * 0.5)
            lr = max(0.005, lr * 1.5)
            no_improve = 0

    # Final check
    score = evaluate_sequence(a.tolist())
    if np.isfinite(score) and score < best_score:
        best = a.copy()
        best_score = score

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: try flat-convolution sequence, optimize with FFT subgradient,
    and explore alternative starts. Returns best sequence found.
    """
    deadline = time.monotonic() + budget_s - 20.0
    rng = np.random.default_rng(seed)

    # Get length from parent
    if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = len(parent)
    else:
        n = 5582
        parent = np.ones(n, dtype=np.float64)

    print(f"n = {n}")

    # FFT size: need >= 3n-1 for cross-correlation
    fft_size = 1
    while fft_size < 3 * n - 1:
        fft_size <<= 1
    print(f"fft_size = {fft_size}")

    # --- Phase 1: Generate and evaluate starting candidates ---
    t0 = time.monotonic()
    flat, n_neg = make_flat_start(n)
    flat_score = evaluate_sequence(flat.tolist())
    print(f"Flat seq: {n_neg} neg entries, score={flat_score:.8f} (time={time.monotonic()-t0:.2f}s)")

    parent_norm = normalize_nonneg(parent)
    parent_score = evaluate_sequence(parent_norm.tolist())
    print(f"Parent: score={parent_score:.8f}")

    ramp1 = make_ramp_start(n, 1)
    ramp1_score = evaluate_sequence(ramp1.tolist())
    print(f"Ramp: score={ramp1_score:.8f}")

    sine1 = make_sine_start(n, 1.0, 0.7)
    sine1_score = evaluate_sequence(sine1.tolist())
    print(f"Sine: score={sine1_score:.8f}")

    candidates = [
        (flat, flat_score, "flat"),
        (parent_norm, parent_score, "parent"),
        (ramp1, ramp1_score, "ramp"),
        (sine1, sine1_score, "sine"),
    ]
    candidates.sort(key=lambda x: x[1])
    best_name = candidates[0][2]
    best_start = candidates[0][0].copy()
    best_score = candidates[0][1]
    print(f"Best start: {best_name} ({best_score:.8f})")

    overall_best = best_start.copy()
    overall_score = best_score

    # --- Phase 2: Optimize the best start ---
    t_opt_start = time.monotonic()
    a_opt, score_opt = optimize_smoothed_max(
        best_start, fft_size, deadline - 120.0, rng, verbose=True
    )
    print(f"Optimization done in {time.monotonic()-t_opt_start:.1f}s, score={score_opt:.8f}")
    if score_opt < overall_score:
        overall_best = a_opt
        overall_score = score_opt

    # --- Phase 3: Try other starts if time remains ---
    remaining = deadline - time.monotonic()
    print(f"Remaining time: {remaining:.1f}s")

    for name, start_arr, s in candidates[1:]:
        if time.monotonic() > deadline - 30:
            break
        alloc = min(remaining * 0.3, 300.0)
        tl = time.monotonic() + alloc
        a2, s2 = optimize_smoothed_max(start_arr, fft_size, tl, rng, verbose=False)
        if s2 < overall_score:
            overall_best = a2
            overall_score = s2
            print(f"  Improved with {name}: {overall_score:.8f}")
        remaining = deadline - time.monotonic()

    # --- Phase 4: Random restarts from best ---
    while time.monotonic() < deadline - 10:
        noise_scale = 0.05 * overall_best.max()
        a_rand = normalize_nonneg(overall_best + noise_scale * rng.standard_normal(n))
        alloc_time = min(60.0, deadline - 10.0 - time.monotonic())
        tl = time.monotonic() + alloc_time
        a3, s3 = optimize_smoothed_max(a_rand, fft_size, tl, rng, verbose=False)
        if s3 < overall_score:
            overall_best = a3
            overall_score = s3
            print(f"  Random restart improved: {overall_score:.8f}")
        else:
            # Reduce noise for next attempt
            noise_scale *= 0.7

    # --- Final validation ---
    final_score = evaluate_sequence(overall_best.tolist())
    print(f"Final score: {final_score:.8f}")

    # Sanity check
    assert len(overall_best) == n
    assert np.all(overall_best >= 0)
    assert not np.any(np.isnan(overall_best))
    assert not np.any(np.isinf(overall_best))
    assert np.sum(overall_best) > 0.01

    return [float(x) for x in overall_best]
