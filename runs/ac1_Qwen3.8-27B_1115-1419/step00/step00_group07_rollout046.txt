import time
from scipy.signal import fftconvolve

def fast_convolve(a, b):
    """Fast convolution using FFT."""
    return fftconvolve(a, b, mode='full')

def compute_objective(p):
    """Compute 2*n*max(convolve(p,p)) / (sum(p))^2."""
    n = len(p)
    S = np.sum(p)
    if S < 1e-10:
        return np.inf
    b = fftconvolve(p, p, mode='full')
    return 2.0 * n * np.max(b) / (S * S)

    """Smooth approximation of max using log-sum-exp."""
    m = np.max(b)
    return m + tau * np.log(np.sum(np.exp((b - m) / tau)))

    """Smooth objective using log-sum-exp."""
    n = len(p)
    S = np.sum(p)
    if S < 1e-10:

    b = fftconvolve(p, p, mode='full')
    sm = smooth_max(b, tau)
    return 2.0 * n * sm / (S * S)

def _cross_corr_fft(w, p):
    """Compute c[j] = sum_i w[i+j]*p[i] for j=0..len(p)-1 using FFT."""
    n = len(p)
    N = 1 << (len(w) + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p, N)
    R = np.fft.irfft(W * np.conj(P), N)
    return R[:n]

import numpy as np
import time
from scipy.signal import fftconvolve


def _cross_corr_fft(w, p):
    """Compute c[j] = sum_i w[i+j]*p[i] for j=0..len(p)-1 using FFT."""
    n = len(p)
    N = 1 << (len(w) + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p, N)
    R = np.fft.irfft(W * np.conj(P), N)
    return R[:n]


def _smooth_grad_normalized(p, tau):
    """
    Compute smooth objective 2n*logsumexp(b/tau) and gradient w.r.t. p (normalized, sum=1).
    Returns (obj, grad) where grad is NOT projected.
    """
    n = len(p)
    b = fftconvolve(p, p, mode='full')
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    exp_sum = exp_b.sum()
    w = exp_b / exp_sum

    obj = 2.0 * n * lse

    grad = (4.0 * n / tau) * c
    return obj, grad
