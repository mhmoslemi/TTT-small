```
def multiplicative_one(a, beta, lr):
    n=a.size
    C=np.convolve(a,a)
    M=C.max()
    if M <= 0: return None
    w=np.exp(beta*(C-M))
    D=np.correlate(w,a,mode='valid')
    if D.size != n: return None
    D=np.nan_to_num(D, nan=0.0, posinf=0.0, neginf=0.0)
    meanD=np.mean(D)
    if meanD <= 0: return None
    tiny = max(1e-300, 1e-10*meanD)
    D_safe=np.maximum(D, tiny)
    ratio=meanD/D_safe
    # cap ratio to prevent extreme
    ratio=np.clip(ratio, 1e-4, 1e4)
    a_new = a * (ratio ** lr)
    # optionally enforce a minimum floor to recover zeros
    floor=1e-12 * np.sum(a)/n
    a_new = np.maximum(a_new, floor)
    return norm_sequence(a_new, np.sum(a))
```

```python
"""
Optimizes non-negative step-function heights to minimize
    2 * n * max(conv(a, a)) / (sum(a))^2.

Method:
- Scale-invariant normalization.
- Fast smooth surrogate optimization using softmax-weighted convolution
  gradients, implemented with FFT convolution/correlation.
- Multiplicative exponentiated-gradient updates and projected gradient
  updates as cheap descent steps.
- Periodic sparse LP direction-finding in the style of the Frank-Wolfe /
  active-set construction:
      maximize sum(b)
      subject to (a * b)[k] <= M for selected k, b >= 0, b <= ub
  The LP direction is mixed with the current iterate.
- Multiple initializations and several sequence lengths are tried if time remains.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


INF = float("inf")


def _as_1d_float(x):
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim == 0:
        arr = arr.reshape(1)
    elif arr.ndim != 1:
        arr = arr.reshape(-1)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return arr


def _norm_sequence(a, target=None):
    a = _as_1d_float(a)
    n = a.size
    if n == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    if target is None:
        target = math.sqrt(2.0 * n)
    target = float(target)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(n, target / max(1, n), dtype=np.float64)
    a = a * (target / s)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 > 0.0:
        a = a * (target / s2)
    return a


def convolution_fft_two(x, y):
    x = _as_1d_float(x)
    y = _as_1d_float(y)
    if x.size == 0 or y.size == 0:
        return np.empty(0, dtype=np.float64)
    L = x.size + y.size - 1
    m = 1
    while m < L:
        m <<= 1
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    out = np.fft.irfft(X * Y, m)[:L]
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    return out


def score_from_C(a, C):
    a = _as_1d_float(a)
    C = _as_1d_float(C)
    n = a.size
    if n == 0 or C.size == 0:
        return INF
    s = float(a.sum())
    if s < 1e-12:
        return INF
    if not np.all(np.isfinite(C)):
        return INF
    cmax = float(np.max(C))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def direct_score(a):
    a = _as_1d_float(a)
    if a.size == 0:
        return INF
    a = np.clip(a, 0.0, 1000.0)
    C = convolution_fft_two(a, a)
    return score_from_C(a, C)


def unique_preserve_array(arr):
    out = []
    seen = set()
    arr = np.asarray(arr).ravel()
    for x in arr:
        xi = int(x)
        if xi not in seen:
            seen.add(xi)
            out.append(xi)
    return np.asarray(out, dtype=np.int64)


def select_rows(n, C, K, rng):
    total = 2 * int(n) - 1
    K = int(min(max(1, int(K)), total))
    if K >= total:
        return np.arange(total, dtype=np.int64)

    top = int(min(K, max(1, K // 2)))
    C = _as_1d_float(C)
    if C.size != total:
        C = np.resize(C, total)

    if top >= total:
        top_idx = np.arange(total, dtype=np.int64)
    else:
        top_idx = np.argpartition(C, -top)[-top:]
        top_idx = np.asarray(top_idx, dtype=np.int64)

    needed = K - top_idx.size
    extra = []
    if needed > 0:
        seen = {int(r) for r in top_idx}
        even_rows = np.linspace(0, total - 1, needed * 2).astype(np.int64)
        random_rows = rng.integers(0, total, size=max(needed * 3, 16)).astype(np.int64)
        for r in np.concatenate((even_rows, random_rows)):
            ri = int(r)
            if ri not in seen:
                seen.add(ri)
                extra.append(ri)
                if len(extra) >= needed:
                    break
        if len(extra) < needed:
            for ri in range(total):
                if len(extra) >= needed:
                    break
                if ri not in seen:
                    extra.append(ri)
                    seen.add(ri)

    rows = unique_preserve_array(np.concatenate((top_idx, np.asarray(extra, dtype=np.int64))))
    if rows.size > K:
        rows = rows[:K]
    return rows


def build_conv_matrix(a, rows):
    n = int(a.size)
    rows = np.asarray(rows, dtype=np.int64)
    if rows.size == 0:
        return None

    starts = np.maximum(0, rows - n + 1).astype(np.int32, copy=False)
    ends = np.minimum(n - 1, rows).astype(np.int32, copy=False)
    lengths = (ends - starts + 1).astype(np.int32, copy=False)
    if np.any(lengths <= 0):
        return None

    cols_list = []
    for s, e in zip(starts, ends):
        cols_list.append(np.arange(int(s), int(e) + 1, dtype=np.int32))
    if not cols_list:
        return None

    cols = np.concatenate(cols_list, axis=None).astype(np.int32, copy=False)
    rows_rep = np.repeat(rows.astype(np.int32, copy=False), lengths.astype(np.int32, copy=False))
    data = a[cols.astype(np.int64, copy=False)]
    return rows_rep, cols, data


def lp_direction(a, C, M, K, time_limit, rng, tighten=1.0):
    if not HAVE_SCIPY:
        return None
    n = int(a.size)
    if n <= 1:
        return a.copy()
    M = float(M) * float(tighten)
    if not np.isfinite(M) or M <= 1e-15:
        return None

    a_use = a.copy()
    S = float(a_use.sum())
    if S <= 1e-12:
        return None

    floor = 1e-12 * S / max(1, n)
    if float(np.min(a_use)) <= floor:
        a_use = np.maximum(a_use, floor)
        S = float(a_use.sum())

    rows = select_rows(n, C, K, rng)
    if rows.size == 0:
        return None

    bm = build_conv_matrix(a_use, rows)
    if bm is None:
        return None
    rows_rep, cols, data = bm

    try:
        A = csr_matrix((data, (rows_rep, cols)), shape=(rows.size, n))
    except Exception:
        return None

    mean_a = float(np.mean(a_use))
    max_a = float(np.max(a_use))
    ub = max(1.0, 50.0 * mean_a, 5.0 * max_a)
    ub = min(1000.0, max(1.0, ub))

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows.size, float(M), dtype=np.float64)
    bounds = [(0.0, float(ub))] * n

    res = None
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(max(1.0, time_limit)), "presolve": True}
        )
    except Exception:
        res = None
    if res is None:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            res = None
    if res is None:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds)
        except Exception:
            return None

    if getattr(res, "x", None) is None:
        return None

    b = np.asarray(res.x, dtype=np.float64)
    if b.size != n or not np.all(np.isfinite(b)):
        return None
    b = np.clip(b, 0.0, ub)
    sb = float(b.sum())
    if sb <= 1e-12:
        return None
    b = b * (S / sb)
    b = np.clip(b, 0.0, 1000.0)
    sb2 = float(b.sum())
    if sb2 > 0.0:
        b = b * (S / sb2)
    return b


def project_simplex(v, s):
    v = _as_1d_float(v)
    n = v.size
    if n == 0:
        return v
    s = float(s)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (css - s)
    if not np.any(cond):
        rho = 0
    else:
        rho = int(np.nonzero(cond)[0][-1])
    theta = (css[rho] - s) / float(rho + 1)
    out = np.maximum(v - theta, 0.0)
    return out


def projected_gradient_candidate(a, C, M, beta, lr):
    a = _as_1d_float(a)
    C = _as_1d_float(C)
    n = a.size
    if n == 0 or C.size == 0:
        return None
    M = float(np.max(C)) if C.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None

    w = np.exp(float(beta) * (C - M))
    ws = float(w.sum())
    if ws <= 1e-300:
        return None
    y = w / ws

    D = convolution_fft_two(y, a[::-1])
    if D.size < 2 * n - 1:
        return None
    D = D[n - 1:n - 1 + n]
    if D.size != n or not np.all(np.isfinite(D)):
        return None
    D = np.maximum(D, 0.0)

    v = a - (2.0 * float(lr)) * D
    if not np.all(np.isfinite(v)):
        return None
    cand = project_simplex(v, float(a.sum()))
    cand = np.where(np.isfinite(cand), cand, 0.0)
    return _norm_sequence(cand, float(a.sum()))


def multiplicative_candidate(a, C, M, beta, lr):
    a = _as_1d_float(a)
    C = _as_1d_float(C)
    n = a.size
    if n == 0 or C.size == 0:
        return None
    S = float(a.sum())
    if S <= 1e-12:
        return None
    M = float(np.max(C)) if C.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None

    w = np.exp(float(beta) * (C - M))
    D = convolution_fft_two(w, a[::-1])
    if D.size < 2 * n - 1:
        return None
    D = D[n - 1:n - 1 + n]
    if D.size != n or not np.all(np.isfinite(D)):
        return None
    D = np.maximum(D, 0.0)

    meanD = float(np.mean(D))
    if meanD <= 1e-300:
        return None
    tiny = max(1e-300, 1e-10 * meanD)
    Dsafe = np.maximum(D, tiny)
    ratio = meanD / Dsafe
    ratio = np.clip(ratio, 1e-6, 1e6)
    ratio = np.where(np.isfinite(ratio), ratio, 1.0)

    a_new = a * np.power(ratio, float(lr))
    floor = 1e-12 * S / max(1, n)
    a_new = np.maximum(a_new, floor)
    a_new = np.where(np.isfinite(a_new), a_new, floor)
    return _norm_sequence(a_new, S)


def perturb_sequence(a, rng, rel_strength=0.1):
    a = _as_1d_float(a)
    n = a.size
    S = float(a.sum())
    if n == 0 or S <= 0.0:
        return a
    mean = S / max(1, n)
    sigma = max(1e-12, float(rel_strength) * mean)
    b = a + rng.normal(0.0, sigma, size=n)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    return _norm_sequence(b, S)


def multiplicative_burst(a, time_limit, rng):
    a = _as_1d_float(a)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64), INF

    a = _norm_sequence(a)
```

```python
"""
Optimizes non-negative step-function heights to minimize
    2 * n * max(conv(a, a)) / (sum(a))^2.

Approach:
- Scale-invariant normalization.
- Fast FFT-based smooth surrogate gradients.
- Multiplicative exponentiated-gradient updates.
- Projected-gradient updates as fallback.
- Periodic sparse LP direction-finding with active rows.
- Multiple initializations and a secondary length attempt if time remains.

    HAVE_SCIPY = False


INF = float("inf")


def _as_1d_float(x):
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim == 0:
        arr = arr.reshape(1)
    elif arr.ndim != 1:
        arr = arr.reshape(-1)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return arr


def _norm_sequence(a, target=None):
    a = _as_1d_float(a)
    n = a.size
    if n == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    if target is None:
        target = math.sqrt(2.0 * n)
    target = float(target)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(n, target / max(1, n), dtype=np.float64)
    a = a * (target / s)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 > 0.0:
        a = a * (target / s2)
    return a


def convolution_fft_two(x, y):
    x = _as_1d_float(x)
    y = _as_1d_float(y)
    if x.size == 0 or y.size == 0:
        return np.empty(0, dtype=np.float64)
    L = x.size + y.size - 1
    m = 1
    while m < L:
        m <<= 1
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    out = np.fft.irfft(X * Y, m)[:L]
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    return out


def score_from_C(a, C):
    a = _as_1d_float(a)
    C = _as_1d_float(C)
    n = a.size
    if n == 0 or C.size == 0:
        return INF
    s = float(a.sum())
    if s < 1e-12:
        return INF
    if not np.all(np.isfinite(C)):
        return INF
    cmax = float(np.max(C))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))

def unique_preserve_array(arr):
    out = []
    seen = set()
    arr = np.asarray(arr).ravel()
    for x in arr:
        xi = int(x)
        if xi not in seen:
            seen.add(xi)
            out.append(xi)
    return np.asarray(out, dtype=np.int64)


def select_rows(n, C, K, rng):
    total = 2 * int(n) - 1
    K = int(min(max(1, int(K)), total))
    if K >= total:
        return np.arange(total, dtype=np.int64)

    top = int(min(K, max(1, K // 2)))
    C = _as_1d_float(C)
    if C.size != total:
        C = np.resize(C, total)

    if top >= total:
        top_idx = np.arange(total, dtype=np.int64)
    else:
        top_idx = np.argpartition(C, -top)[-top:]
        top_idx = np.asarray(top_idx, dtype=np.int64)

    needed = K - top_idx.size
    extra = []
    if needed > 0:
        seen = {int(r) for r in top_idx}
        even_rows = np.linspace(0, total - 1, needed * 2).astype(np.int64)
        random_rows = rng.integers(0, total, size=max(needed * 3, 16)).astype(np.int64)
        for r in np.concatenate((even_rows, random_rows)):
            ri = int(r)
            if ri not in seen:
                seen.add(ri)
                extra.append(ri)
                if len(extra) >= needed:
                    break
        if len(extra) < needed:
            for ri in range(total):
                if len(extra) >= needed:
                    break
                if ri not in seen:
                    extra.append(ri)
                    seen.add(ri)

    rows = unique_preserve_array(np.concatenate((top_idx, np.asarray(extra, dtype=np.int64))))
    if rows.size > K:
        rows = rows[:K]
    return rows


def build_conv_matrix(a, rows):
    n = int(a.size)
    rows = np.asarray(rows, dtype=np.int64)
    if rows.size == 0:
        return None

    starts = np.maximum(0, rows - n + 1).astype(np.int32, copy=False)
    ends = np.minimum(n - 1, rows).astype(np.int32, copy=False)
    lengths = (ends - starts + 1).astype(np.int32, copy=False)
    if np.any(lengths <= 0):
        return None

    cols_list = []
    for s, e in zip(starts, ends):
        cols_list.append(np.arange(int(s), int(e) + 1, dtype=np.int32))
    if not cols_list:
        return None

    cols = np.concatenate(cols_list, axis=None).astype(np.int32, copy=False)
    rows_rep = np.repeat(rows.astype(np.int32, copy=False), lengths.astype(np.int32, copy=False))
    data = a[cols.astype(np.int64, copy=False)]
    return rows_rep, cols, data


def lp_direction(a, C, M, K, time_limit, rng, tighten=1.0):

def project_simplex(v, s):
    v = _as_1d_float(v)
    n = v.size
    if n == 0:
        return v
    s = float(s)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (css - s)
    if not np.any(cond):
        rho = 0
    else:
        rho = int(np.nonzero(cond)[0][-1])
    theta = (css[rho] - s) / float(rho + 1)
    out = np.maximum(v - theta, 0.0)
    return out

        best = a.copy()
        best_score = 2.0

    deadline = time.monotonic() + max(1.0, float(time_limit))
    no_improve = 0

    mult_steps = (
        (1.0, 1.0),
        (1.0, 0.5),
        (2.0, 0.3),
        (5.0, 0.2),
        (10.0, 0.1),
        (2.0, 0.5),
        (5.0, 0.5),
    )
    pg_steps = (
        (2.0, 0.05),
        (2.0, 0.2),
        (5.0, 0.2),
        (5.0, 1.0),
        (10.0, 1.0),
        (10.0, 5.0),
    )

    while time.monotonic() < deadline:
        C = convolution_fft_two(a, a)
        M = float(np.max(C)) if C.size else 0.0
        if not np.isfinite(M) or M <= 0.0:
            break

        improved = False

        for beta, lr in mult_steps:
            cand = multiplicative_candidate(a, C, M, beta, lr)
            if cand is None:
                continue
            sc = direct_score(cand)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best = cand.copy()
                a = cand.copy()
                best_score = sc
                no_improve = 0
                improved = True
                break

        if not improved:
            for beta, lr in pg_steps:
                cand = projected_gradient_candidate(a, C, M, beta, lr)
                if cand is None:
                    continue
                sc = direct_score(cand)
                if np.isfinite(sc) and sc < best_score - 1e-10:
                    best = cand.copy()
                    a = cand.copy()
                    best_score = sc
                    no_improve = 0
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve >= 120:
                break
            if no_improve % 8 == 0:
                strength = min(0.30, 0.015 * no_improve)
                base = best if (no_improve % 2 == 0) else a
                a = perturb_sequence(base, rng, strength)
            else:
                a = best.copy()

    return best, best_score


def structured_initial_candidates(n, rng):
    n = int(n)
    if n <= 1:
        return [np.ones(n, dtype=np.float64)]

    x = np.arange(n, dtype=np.float64) / float(n - 1)
    old = 2.0 * x - 1.0
    abs_old = np.abs(old)
    target = math.sqrt(2.0 * n)

    cands = []
    cands.append(np.ones(n, dtype=np.float64))

    alphas = (0.25, 0.6, 1.2, 2.5)
    powers = (0.5, 1.0, 2.0)

    for alpha in alphas:
        for p in powers:
            vals = 1.0 + alpha * (abs_old ** p)
            cands.append(np.maximum(vals, 1e-6))
            vals2 = 1.0 + alpha * ((1.0 - abs_old) ** p)
            cands.append(np.maximum(vals2, 1e-6))

    for _ in range(3):
        period = int(rng.integers(3, max(4, n // 4)))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        ampl = float(rng.uniform(0.1, 0.8))
        vals = 1.0 + ampl * np.cos((2.0 * math.pi * np.arange(n, dtype=np.float64) / period) + phase)
        cands.append(np.maximum(vals, 1e-6))

    vals = np.exp(rng.normal(0.0, 0.45, size=n))
    cands.append(np.maximum(vals, 1e-6))

    vals = np.exp(rng.normal(0.0, 0.45, size=n) + 0.5 * abs_old)
    cands.append(np.maximum(vals, 1e-6))

    out = []
    for c in cands:
        c = _norm_sequence(c, target)
        if c.size == n and np.all(np.isfinite(c)):
            out.append(c)
    return out


def best_initial(candidates):
    best = None
    best_score = INF
    for c in candidates:
        c = _as_1d_float(c)
        if c.size == 0:
            continue
        sc = direct_score(c)
        if np.isfinite(sc) and sc < best_score:
            best = c.copy()
            best_score = sc
    return best, best_score


def refine_sequence(a0, max_time, rng, do_lp=True, label="seq"):
    a = _as_1d_float(a0)
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64), 2.0

    a = _norm_sequence(a)
    best = a.copy()
    C = convolution_fft_two(a, a)
    best_score = score_from_C(a, C)
    if not np.isfinite(best_score):
        a = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best = a.copy()
        best_score = 2.0

    start = time.monotonic()
    deadline = start + max(1.0, float(max_time))

    print(f"[{label}] start n={n} score={best_score:.8f} time_budget={max_time:.1f}s", flush=True)

    if float(a.std()) <= 1e-7 * max(1.0, float(a.mean())):
        a = perturb_sequence(a, rng, 0.05)

    burst_time = min(max(5.0, 0.10 * float(max_time)), max(1.0, float(max_time) - 5.0))
    if burst_time > 2.0:
        tmp, tmp_score = multiplicative_burst(a, burst_time, rng)
        if np.isfinite(tmp_score) and tmp_score < best_score:
            best = tmp.copy()
            best_score = tmp_score
            print(f"[{label}] after initial burst score={best_score:.8f}", flush=True)
        else:
            a = best.copy()

    no_improve = 0
    last_burst = time.monotonic()
    last_improve = start

    if do_lp and HAVE_SCIPY and n > 1:
        total = 2 * n - 1
        if n <= 800:
            ks = [int(total)]
        else:
            k1 = max(250, int(0.12 * n))
            k2 = max(700, int(0.25 * n))
            k3 = max(1200, int(0.40 * n))
            ks = []
            for k in (k1, k2, k3):
                k = int(min(total, k))
                if k not in ks:
                    ks.append(k)
            if n <= 2500 and total <= 6000 and int(total) not in ks:
                ks.append(int(total))

        idx = 0
        lp_count = 0

        while time.monotonic() < deadline - 8.0:
            remaining = deadline - time.monotonic()
            if remaining < 8.0:
                break
            if not ks:
                break

            k = int(ks[idx % len(ks)])
            if k >= total:
                tlim = max(5.0, min(90.0, 0.20 * remaining))
            elif k > 2 * ks[0]:
                tlim = max(5.0, min(30.0, 0.12 * remaining))
            else:
                tlim = max(3.0, min(18.0, 0.07 * remaining))

            C = convolution_fft_two(a, a)
            M = float(np.max(C)) if C.size else 0.0
            if not np.isfinite(M) or M <= 0.0:
                a = best.copy()
                idx += 1
                continue

            tighten = 1.0
            if no_improve >= 8:
                tighten = 0.97
            if no_improve >= 20:
                tighten = 0.94
            if no_improve >= 35:
                tighten = 0.90

            b = lp_direction(a, C, M, k, tlim, rng, tighten=tighten)
            lp_count += 1
            improved = False

            if b is not None:
                S = float(a.sum())
                bases = [b]
                b_sym = _norm_sequence(0.5 * (b + b[::-1]), S)
                if b_sym is not None and not np.allclose(b_sym, b, rtol=1e-10, atol=1e-12):
                    bases.append(b_sym)

                for base in bases:
                    if time.monotonic() >= deadline - 8.0:
                        break
                    for t in (1.0, 0.6, 0.35, 0.18, 0.08, 0.03, 0.01):
                        cand = (1.0 - t) * a + t * base
                        cand = np.maximum(cand, 0.0)
                        cand = _norm_sequence(cand, S)
                        sc = direct_score(cand)
                        if np.isfinite(sc) and sc < best_score - 1e-10:
                            best = cand.copy()
                            a = cand.copy()
                            best_score = sc
                            no_improve = 0
                            improved = True
                            last_improve = time.monotonic()
                            if lp_count % 20 == 0 or best_score < 1.8:
                                elapsed = time.monotonic() - start
                                print(
                                    f"[{label}] LP#{lp_count} K={k} tighten={tighten:.3f} "
                                    f"t={t:.3f} score={best_score:.8f} elapsed={elapsed:.1f}s",
                                    flush=True
                                )
                            break
                    if improved:
                        break

            if not improved:
                no_improve += 1

                if time.monotonic() - last_burst > 45.0:
                    last_burst = time.monotonic()
                    tmp = best.copy()
                    btime = max(5.0, min(40.0, 0.08 * remaining))
                    tmp, sc = multiplicative_burst(tmp, btime, rng)
                    if np.isfinite(sc) and sc < best_score - 1e-10:
                        best = tmp.copy()
                        a = best.copy()
                        best_score = sc
                        no_improve = 0
                        last_improve = time.monotonic()
                        print(f"[{label}] burst improved score={best_score:.8f}", flush=True)
                else:
                    sym = _norm_sequence(0.5 * (best + best[::-1]), float(best.sum()))
                    sc = direct_score(sym)
                    if np.isfinite(sc) and sc < best_score - 1e-10:
                        best = sym.copy()
                        a = best.copy()
                        best_score = sc
                        no_improve = 0
                        last_improve = time.monotonic()
                        print(f"[{label}] sym improved score={best_score:.8f}", flush=True)
                    else:
                        a = perturb_sequence(best, rng, 0.02 + 0.01 * (no_improve % 5))

                if no_improve >= 70:
                    a = perturb_sequence(best, rng, 0.25)
                    no_improve = 0
                    last_burst = time.monotonic() - 100.0

            idx += 1

    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        tmp = best.copy()
        tmp, sc = multiplicative_burst(tmp, max(1.0, remaining - 2.0), rng)
        if np.isfinite(sc) and sc < best_score:
            best = tmp.copy()
            best_score = sc

    try:
        exact = float(evaluate_sequence([float(x) for x in best]))
        if np.isfinite(exact):
            print(f"[{label}] final exact score={exact:.8f} fft_score={best_score:.8f}", flush=True)
        else:
            print(f"[{label}] final fft score={best_score:.8f} exact invalid", flush=True)
    except Exception as exc:
        print(f"[{label}] final fft score={best_score:.8f} exact error: {exc}", flush=True)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    allow = max(120.0, float(budget_s))
    deadline = start + allow - min(35.0, 0.10 * allow)

    candidates = []

    if "height_sequence_1" in globals():
        try:
            h = _as_1d_float(globals()["height_sequence_1"])
            if h.size > 0:
                h = np.clip(h, 0.0, 1000.0)
                h = _norm_sequence(h)
                candidates.append(h)
        except Exception:
            pass

    if not candidates:
        candidates.append(np.ones(2048, dtype=np.float64))

    base = candidates[0]
    n = base.size

    if n < 64:
        base = np.ones(768, dtype=np.float64)
        n = base.size
    if n > 30000:
        base = base[:25000]
        n = base.size
    base = _norm_sequence(base)
    candidates.clear()
    candidates.append(base)
    candidates.append(perturb_sequence(base, rng, 0.10))
    candidates.append(_norm_sequence(0.5 * (base + base[::-1]), float(base.sum())))

    try:
        candidates.extend(structured_initial_candidates(n, rng))
    except Exception:
        pass

    best_global, best_global_score = best_initial(candidates)
    if best_global is None:
        best_global = _norm_sequence(base)
        best_global_score = direct_score(best_global)

    print(f"[main] best initial n={best_global.size} score={best_global_score:.8f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 120.0:
        main_time = max(60.0, 0.52 * remaining)
        res_a, res_s = refine_sequence(best_global, main_time, rng, do_lp=True, label="main")
        if res_a is not None and np.all(np.isfinite(res_a)):
            res_s2 = direct_score(res_a)
            if np.isfinite(res_s) and np.isfinite(res_s2):
                res_s = min(res_s, res_s2)
            if np.isfinite(res_s) and res_s < best_global_score:
                best_global = res_a.copy()
                best_global_score = res_s

    remaining = deadline - time.monotonic()
    if remaining > 350.0 and best_global.size > 1500:
        n2 = int(min(1152, max(384, best_global.size // 4)))
        inits2 = [np.ones(n2, dtype=np.float64)]
        try:
            inits2.extend(structured_initial_candidates(n2, rng))
        except Exception:
            pass
        init2, _ = best_initial(inits2)
        if init2 is not None:
            alt_time = max(60.0, min(1100.0, 0.45 * remaining))
            res2, res2_s = refine_sequence(init2, alt_time, rng, do_lp=True, label=f"alt{n2}")
            if res2 is not None and np.all(np.isfinite(res2)):
                res2_s2 = direct_score(res2)
                if np.isfinite(res2_s) and np.isfinite(res2_s2):
                    res2_s = min(res2_s, res2_s2)
                if np.isfinite(res2_s) and res2_s < best_global_score:
                    best_global = res2.copy()
                    best_global_score = res2_s

    remaining = deadline - time.monotonic()
    if remaining > 25.0:
        tmp = best_global.copy()
        tmp, tmp_s = multiplicative_burst(tmp, max(5.0, remaining - 5.0), rng)
        if tmp is not None and np.all(np.isfinite(tmp)):
            tmp_s2 = direct_score(tmp)
            if np.isfinite(tmp_s) and np.isfinite(tmp_s2):
                tmp_s = min(tmp_s, tmp_s2)
            if np.isfinite(tmp_s) and tmp_s < best_global_score:
                best_global = tmp.copy()
                best_global_score = tmp_s

    out = _norm_sequence(best_global)
    out = np.clip(out, 0.0, 1000.0)
    out = out.tolist()

    try:
        exact = float(evaluate_sequence(out))
        if not np.isfinite(exact):
            out = [1.0] * 1000
            exact = float(evaluate_sequence(out))
        print(f"[final] exact score={float(exact):.8f} n={len(out)}", flush=True)
    except Exception as exc:
        sc = direct_score(np.asarray(out, dtype=np.float64))
        print(f"[final] direct score={sc:.8f} exact error: {exc}", flush=True)

    return [float(x) for x in out]
```
