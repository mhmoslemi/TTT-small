                M_sg = c_sg.max()
                if M_sg < M * 0.9999:
                    a = a_sg
                    no_improve = 0
                    improved = True

        # --- Track best ---
        if M < best_M:
            best_M = M
            best_a = a.copy()
            best_score_approx = 2.0 * n * best_M

        # --- Adaptation ---
        if no_improve > 50:
            # Perturb and reset
            noise_scale = 0.01 + 0.02 * (no_improve / 100.0)
            a = simplex_project(a + rng.normal(0, noise_scale, n), 1.0)
            no_improve = 0
            t_blend = 0.5

        if no_improve > 150:
            K = max(30, K // 2)
            no_improve = 0

        # Occasionally increase K back
        if no_improve == 0 and K < K_init and it % 100 == 0:
            K = min(K_init, K * 2)

    return best_a, best_score_approx

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """LP-guided iterative optimization for minimizing 2n*max(autocorr)/sum^2.
    
    Uses linear programming to find optimal directions at each step,
    combined with subgradient descent for fine-tuning. Multiple starting
    points are explored to avoid local minima.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0

    # --- Generate starting points ---
    start_points = []

    # SP1: Given sequence
    try:
        hs = np.array(height_sequence_1, dtype=np.float64).copy()
        hs = np.clip(hs, 0.0, 1000.0)
        if len(hs) > 0 and hs.sum() > 1e-10:
            start_points.append(hs / hs.sum())
    except Exception:
        pass

    # SP2: Absolute value of random Gaussian (gives varied amplitudes)
    n_gauss = 6000
    g = np.random.default_rng(rng.integers(0, 2**31)).standard_normal(n_gauss)
    a_gauss = np.abs(g) + 0.01
    start_points.append(a_gauss / a_gauss.sum())

    # SP3: Power-law decay with noise
    n_pl = 6000
    x = np.arange(n_pl, dtype=np.float64)
    a_pl = 1.0 / (1.0 + x * 0.1) + 0.1 * np.random.default_rng(
        rng.integers(0, 2**31)).uniform(0, 1, n_pl)
    a_pl = np.maximum(a_pl, 0.0)
    start_points.append(a_pl / a_pl.sum())

    # SP4: Modulated carrier
    n_mod = 6000
    x = np.linspace(0, 10, n_mod)
    a_mod = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x) + 0.1 * np.random.default_rng(
        rng.integers(0, 2**31)).uniform(0, 1, n_mod)
    a_mod = np.maximum(a_mod, 0.01)
    start_points.append(a_mod / a_mod.sum())

    # --- Optimize from each starting point ---
    best_score = np.inf
    best_seq = None

    # Allocate time: give more time to the first (given) starting point
    n_sp = len(start_points)
    time_per_sp = (budget_s - 20) / (n_sp + 1)  # +1 for final refinement
    times = [time_per_sp * 1.5] + [time_per_sp * 0.8] * (n_sp - 1)
    # Ensure total doesn't exceed budget
    total_alloc = sum(times)
    scale = (budget_s - 20) / total_alloc
    times = [t * scale for t in times]

    for sp_idx, sp in enumerate(start_points):
        if time.monotonic() > deadline:
            break
        a_init = sp.copy()
        n_sp_len = len(a_init)
        bounds_list = [(0.0, None)] * n_sp_len
        sp_deadline = min(start_time + 10 + times[sp_idx], deadline)

        print(f"Starting point {sp_idx}: n={n_sp_len}, "
              f"alloc_time={times[sp_idx]:.0f}s")

        best_a_sp, best_M_sp = optimize_from(
            a_init, n_sp_len, bounds_list, sp_deadline, rng, K_init=200)

        score = float(evaluate_sequence(best_a_sp.tolist()))
        print(f"  SP{sp_idx} result: score={score:.6f}, max_c={best_M_sp:.2e}")

        if score < best_score:
            best_score = score
            best_seq = best_a_sp.copy()

    # --- Final refinement on the best sequence ---
    if best_seq is not None and time.monotonic() < deadline:
        print("Running final refinement...")
        a_ref = best_seq.copy()
        n_ref = len(a_ref)
        bounds_list = [(0.0, None)] * n_ref
        best_a_ref, _ = optimize_from(a_ref, n_ref, bounds_list, deadline, rng, K_init=300)
        score_ref = float(evaluate_sequence(best_a_ref.tolist()))
        print(f"  Refined: score={score_ref:.6f}")
        if score_ref < best_score:
            best_score = score_ref
            best_seq = best_a_ref.copy()

    # --- Final evaluation ---
    if best_seq is None:
        best_seq = start_points[0]
        best_score = float(evaluate_sequence(best_seq.tolist()))

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"FINAL: score={final_score:.6f}, n={len(best_seq)}")

    return best_seq.tolist()
