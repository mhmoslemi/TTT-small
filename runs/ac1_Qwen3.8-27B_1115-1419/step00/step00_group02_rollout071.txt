    v = np.asarray(v)
    u = np.sort(v)[::-1]

```
def project_simplex(v):
    v = np.asarray(v, dtype=float)
    u = np.sort(v)[::-1]
    n = v.size
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    # ensure sum exactly 1
    s = w.sum()
    if s <= 0:
        w[:] = 1.0/n
    else:
        w /= s
    return w
```

```python
"""
Search for low-peak nonnegative self-convolutions.

The objective is
    score(a) = 2 n max_k (a*a)_k / (sum a)^2 ,          a_i >= 0.
It is scale invariant, so all sequences are normalized to sum 1 during search.

Algorithm:
  * Use several cheap parametric/random starts, including decaying exponentials.
  * Optimize a smooth log-sum-exp surrogate of max(a*a) by projected gradient
    descent on the probability simplex.
  * Compute the gradient with FFT-based convolution/correlation.
  * Periodically increase the logits sharpening, restart from the best point,
    and try resampling the best shape to larger lengths.
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    """Smallest power of two >= size."""
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    """Return nonnegative a normalized to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    """Linear convolution a*a by zero-padded FFT."""
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _lse_value(b, beta):
    """Stable log-sum-exp(b)/beta."""
    if not b.size:
        return 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        return m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        return m
    return m + float(np.log(s)) / beta


def _lse_and_p(b, beta):
    """Return p_k proportional to exp(beta b_k) and lse_value(b,beta)."""
    if not b.size:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _weighted_gradient(a, weights, fft_size, n):
    """
    For b = a*a, derivative of sum_k weights_k b_k w.r.t. a_i is
        2 * sum_k weights_k a_{k-i}.
    This computes that vector using an FFT convolution.
    """
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    h = np.where(h > 0.0, h, 0.0)
    return 2.0 * h


def _project_simplex_soft(v):
    """
    Fast heuristic projection to the positive simplex:
    clip to nonnegative values and normalize.  This is not the Euclidean
    projection, but it is fast, keeps the point feasible, and works well here.
    """
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    n = w.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return w / s


def _make_uniform(n):
    return np.ones(max(int(n), 1), dtype=np.float64) / max(int(n), 1)


def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_mixture(n, alpha, gamma):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x) + float(gamma) * np.exp(-float(alpha) * (1.0 - x))
    a = a + 0.18
    return _normalize(a)


def _make_smooth_random(n, rng, base=None, amplitude=0.4):
    n = max(int(n), 2)
    if base is None:
        base = _make_uniform(n)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 16 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _score_of(a):
    """Exact score under sum-normalization, computed with FFT."""
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _block_perturb(a, rng, strength=0.22):
    n = a.size
    c = a.copy()
    width = max(2, n // 24)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.25:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)


def _perturb_mix(a, rng):
    if rng.random() < 0.6:
        return _block_perturb(a, rng, 0.18)
    noise = np.exp(rng.normal(0.0, 0.22, size=a.size))
    a2 = a * noise + 0.02 * np.ones(a.size, dtype=np.float64) / a.size * rng.random(a.size)
    return _normalize(a2)


def _refine_sequence(a, deadline, rng):
    """
    Run LSE-projected gradient descent from a.  Returns (best_a, best_score).
    """
    a = _normalize(a)
    n = a.size
    if n < 16:
        a = _make_exponential(128, 1.0)
        n = a.size

    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n - 2)

    b = _autocorr(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    alpha = 0.25
    beta_target = 12.0
    it = 0
    fail = 0
    last_improve = 0
    last_hard_improve = 0

    while time.monotonic() < deadline:
        if maxb <= 1e-300:
            maxb = 1e-300
        beta = float(min(1e7, max(1.0, beta_target / maxb)))

        p, smooth_old = _lse_and_p(b, beta)
        grad = _weighted_gradient(a, p, g_fft, n)
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale <= 1e-300:
            a = _perturb_mix(best_a, rng)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.25
            fail = 0
            it += 1
            continue

        if alpha < 0.004:
            alpha = 0.25

        v = a - alpha * grad_c
        a = _project_simplex_soft(v)

        b = _autocorr(a, b_fft)
        maxb = float(np.max(b))
        score = 2.0 * float(n) * maxb
        smooth_new = _lse_value(b, beta)

        if smooth_new < smooth_old - 1e-13:
            alpha = float(min(2.0, alpha * 1.12))
            fail = 0
            last_improve = it
        else:
            alpha *= 0.5
            fail += 1

        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score
            last_hard_improve = it

        if it % 100 == 0:
            beta_target = float(min(900.0, beta_target + 4.0))
        if it % 20 == 0 and fail > 120:
            beta_target = float(min(1000.0, beta_target * 1.25))

        if (fail > 700 or (it - last_improve > 1500 and it > 300)) and time.monotonic() < deadline - 5.0:
            a = _perturb_mix(best_a, rng)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.25
            fail = 0
            beta_target = max(18.0, beta_target * 0.4)

        if fail > 1500 and score > 1.05 * best_score + 0.03:
            a = best_a.copy()
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            beta_target = 24.0
            alpha = 0.2
            fail = 0

        it += 1

    return best_a, best_score


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or not np.isfinite(arr).all():
        return None
    s = float(arr.sum())
    if s <= 1e-300:
        return None
    return arr / s


def _generate_inits(n, given, rng, best):
    n = max(int(n), 2)
    inits = []

    base = _make_exponential(n, 1.0)
    inits.append(base)

    inits.append(_make_smooth_random(n, rng, base, 0.45))
    inits.append(_make_mixture(n, 1.0, 0.55))

    if best is not None and best.size > 4:
        inits.append(_resample_sequence(best, n))

    if given is not None and given.size > 4:
        inits.append(_resample_sequence(given, n))

    inits.append(_make_smooth_random(n, rng, _make_uniform(n), 0.75))
    inits.append(_make_exponential(n, 0.75))

    # Remove obvious duplicates by cheap fingerprint (optional but harmless).
    seen = []
    out = []
    for ic in inits:
        sig = tuple(ic[:: max(1, ic.size // 32)])
        if sig not in seen:
            seen.append(sig)
            out.append(ic)
    return out


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point.  Runs for at most `budget_s` seconds and returns the best
    nonnegative coefficient list found.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + max(budget, 60.0) - 28.0
    rng = np.random.default_rng(seed)

    given = _get_given_sequence()

    # Robust fallback: a decaying exponential already scores well below 2.
    best_a = _make_exponential(1024, 1.0)
    best_score = _score_of(best_a)
    print(f"initial fallback len={best_a.size} score={best_score:.8f}", flush=True)

    scale = float(max(0.08, min(1.0, budget / 2400.0)))

    # Staged search.  Small lengths are cheap and provide good shapes, which are
    # then resampled to larger lengths.
    configs = [
        # length, number of starts, seconds per start
        (256, 2, 25),
        (512, 2, 35),
        (1024, 3, 70),
        (2048, 4, 140),
        (4096, 3, 180),
    ]

    for n, num, sec in configs:
        sec = max(8, int(sec * scale))
        if time.monotonic() > deadline - sec - 10:
            break

        inits = _generate_inits(int(n), given, rng, best_a)
        for idx, init in enumerate(inits[:num]):
            if time.monotonic() > deadline - 10:
                break
            t_end = min(deadline - 1.0, time.monotonic() + sec)
            if time.monotonic() >= t_end:
                continue
            res_a, res_score = _refine_sequence(init.copy(), t_end, rng)
            print(
                f"n={res_a.size} start={idx}/ {num} score={res_score:.8f} best={best_score:.8f} "
                f"t={time.monotonic() - (deadline - budget + 28):.1f}s",
                flush=True,
            )
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)

    # Try the provided long sequence directly if it looks useful.
    if given is not None and given.size >= 512 and given.size <= 10000:
        if best_score > 1.52 and time.monotonic() < deadline - 120:
            sec = max(60, int(120 * scale))
            t_end = min(deadline - 60, time.monotonic() + sec)
            res_a, res_score = _refine_sequence(given.copy(), t_end, rng)
            print(f"given direct len={res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)

    # If the current best is still not near the target, try to upscale it.
    if best_score > 1.53 and time.monotonic() < deadline - 90:
        n_cur = int(best_a.size)
        while n_cur < 8192 and time.monotonic() < deadline - 70:
            n2 = min(8192, max(2048, 2 * n_cur))
            if n2 <= n_cur:
                n2 = 8192
            if n2 <= n_cur:
                break
            init = _resample_sequence(best_a, n2)
            # Mix with a low-level exponential to avoid a too degenerate resampling.
            init = _normalize(0.85 * init + 0.15 * _make_exponential(n2, 1.0))
            sec = max(60, int(120 * scale))
            t_end = min(deadline - 30, time.monotonic() + sec)
            res_a, res_score = _refine_sequence(init, t_end, rng)
            print(f"upscale {n_cur}->{res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                n_cur = int(res_a.size)
            else:
                break

    # Final deep refinement of the best sequence found so far.
    if time.monotonic() < deadline - 5:
        res_a, res_score = _refine_sequence(best_a.copy(), deadline - 2.0, rng)
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)
            print(f"final refine score={res_score:.8f}", flush=True)
        else:
            print(f"final refine no improvement best={best_score:.8f}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)

    return [float(x) for x in out]
```
