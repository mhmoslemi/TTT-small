```python
import numpy as np
import time

def simplex_project(u, S):
    """Project u onto {x >= 0, sum(x) = S}."""
    n = len(u)
    if n == 0:
        return u
    # Sort in descending order
    w = np.sort(u)[::-1]
    hw = np.cumsum(w)
    # Find rho: largest j such that w[j] > (hw[j] - S) / (j+1)
    cond = w * np.arange(1, n+1) > (hw - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.max(np.where(cond)[0])  # 0-indexed, so actual count is rho+1
    theta = (hw[rho] - S) / (rho + 1)
    return np.maximum(u - theta, 0)

def compute_conv(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a, mode='full')

def subgradient(a, c, top_m=20):
    """Compute subgradient of max(c) at a, averaging over top_m positions."""
    n = len(a)
    # Find top_m positions
    top_idx = np.argpartition(c, -top_m)[-top_m:] if len(c) >= top_m else np.arange(len(c))
    # For each top position k, gradient_j = 2 * a[k-j] (if 0 <= k-j < n)
    g = np.zeros(n)
    for k in top_idx:
        # g_j += 2 * a[k-j] for j where 0 <= k-j < n
        # i.e., j is in [k-(n-1), k] intersect [0, n-1]
        j_lo = max(0, k - (n - 1))
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            # a[k-j] for j in [j_lo, j_hi]
            # k-j ranges from k-j_lo to k-j_hi
            g[j_lo:j_hi+1] += 2.0 * a[k-j_hi:k-j_lo+1][::-1]
    g /= len(top_idx)
    return g

def gradient_descent(a_init, max_iters=5000, lr0=0.1, patience=500, verify_every=100):
    """Projected gradient descent to minimize max(a*a) subject to sum(a)=1, a>=0."""
    a = simplex_project(a_init.copy(), 1.0)
    n = len(a)
    best_score = np.inf
    best_a = a.copy()
    c = compute_conv(a)
    best_obj = c.max()
    lr = lr0
    no_improve = 0
    
    for it in range(max_iters):
        c = compute_conv(a)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            break
        
        g = subgradient(a, c, top_m=20)
        a_new = simplex_project(a - lr * g, 1.0)
        
        # Check if this is an improvement
        c_new = compute_conv(a_new)
        if c_new.max() < obj:
            a = a_new
            # Slow down learning rate slightly
            lr *= 0.999
        else:
            lr *= 0.5
    
    return best_a, best_obj, n
```
