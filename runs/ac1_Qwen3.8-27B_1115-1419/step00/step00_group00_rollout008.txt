      # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
      # Shape: (2n-1) x n
      rows = []
      cols = []
      vals = []

          j_min = max(0, k-n+1)
          j_max = min(k, n-1)
          for j in range(j_min, j_max+1):

              cols.append(j)
              vals.append(f[k-j])
