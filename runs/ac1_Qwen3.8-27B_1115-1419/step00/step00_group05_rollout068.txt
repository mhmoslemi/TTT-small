"""
Combined FFT-gradient + LP-linearization optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Phase 1: Fast FFT-based gradient descent with log-sum-exp smooth max.
Phase 2: Successive LP linearization (cutting-plane style) for refinement.
Phase 3: Random local perturbation search for final polish.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def _fft_convolution(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    N = 2 * n - 1
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:N]
    return np.maximum(c, 0.0)


def _fft_gradient(a, w):
    """
    Compute grad[m] = 2 * sum_j a[j] * w[j+m] for m=0..n-1
    using FFT cross-correlation.
    """
    n = len(a)
    N = 2 * n - 1
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # Circular cross-correlation: corr[m] = sum_j a[j] * w[(j-m) mod N]
    corr = np.fft.irfft(np.conj(A) * W, N)
    # We want R[m] = sum_j a[j] * w[j+m] = corr[-m mod N] = corr[(N-m) % N]
    R = np.empty(n)
    R[0] = corr[0]
    if n > 1:
        R[1:] = corr[N - 1 : N - n : -1]
    return 2.0 * R


def _build_lp_matrix(a):
    """
    Build the sparse matrix for the LP constraints.
    A[k, m] = 2 * a[k-m] if 0 <= k-m < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    N = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(N):
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start <= m_end:
            m_range = np.arange(m_start, m_end + 1)
            coefficients = 2.0 * a[k - m_range]
            rows.extend(np.full(len(m_range), k, dtype=np.int32).tolist())
            cols.extend(m_range.astype(np.int32).tolist())
            vals.extend(coefficients.tolist())
    return csr_matrix((vals, (rows, cols)), shape=(N, n))


def _solve_lp(a_current, time_limit=60.0):
    """
    Solve the linearized LP at point a_current.
    Returns (a_lp, t_lp) or (None, None) if failed.
    """
    n = len(a_current)
    N = 2 * n - 1

    # Compute current convolution
    c_current = np.convolve(a_current, a_current)
    c_current = np.maximum(c_current, 0.0)

    # Build constraint matrix
    A_con = _build_lp_matrix(a_current)

    # Full inequality: [A_con, -1] @ [a; t] <= c_current
    # A_con is (N, n), add column of -1's
    rows_extra = np.arange(N, dtype=np.int32)
    cols_extra = np.full(N, n, dtype=np.int32)
    vals_extra = np.full(N, -1.0)

    all_rows = np.concatenate([A_con.indptr_to_rows() if hasattr(A_con, 'indptr_to_rows') else np.repeat(np.arange(N), np.diff(A_con.indptr)), rows_extra])
    # Simpler: just densify the sparse operation
    # Actually let me build the full sparse matrix properly
    A_sparse = A_con.tocoo()
    all_rows = np.concatenate([A_sparse.row, rows_extra])
    all_cols = np.concatenate([A_sparse.col, cols_extra])
    all_vals = np.concatenate([A_sparse.data, vals_extra])
    A_full = csr_matrix((all_vals, (all_rows, all_cols)), shape=(N, n + 1))

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds
    bounds = [(0.0, None)] * (n + 1)

    # Solve
    t0 = time.monotonic()
    try:
        result = linprog(
            c=c_obj,
            A_ub=A_full,
            b_ub=c_current,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'time_limit': time_limit, 'presolve': True}
        )
    except Exception as e:
        print(f"    LP exception: {e}")
        return None, None

    if not result.success:
        return None, None

    a_lp = np.maximum(result.x[:n], 0.0)
    s = a_lp.sum()
    if s < 1e-15:
        return None, None
    a_lp /= s
    t_lp = float(result.x[n])

    elapsed = time.monotonic() - t0
    return a_lp, t_lp
