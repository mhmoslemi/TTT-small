```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def build_conv_matrix(p, n):
    """Build the (2n-1) x n matrix A where (A @ q)_j = conv(p, q)_j."""
    A = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        for k in range(n):
            idx = j - k
            if 0 <= idx < n:
                A[j, k] = p[idx]
    return A

def fw_iteration(p, n, M):
    """One Frank-Wolfe iteration: solve LP to find direction."""
    A = build_conv_matrix(p, n)
    c = -np.ones(n)  # maximize sum = minimize negative sum
    b_ub = M * np.ones(2*n-1)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return p  # no improvement
    q = res.x
    s = q.sum()
    if s < 1e-10:
        return p
    q = q / s
    return q

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10
    
    # Multiple initializations

    # Given sequence

        p0 = np.clip(p0, 0, 1000)
        if p0.size > 0 and p0.sum() > 0:
            inits.append(p0 / p0.sum())
    
    # Uniform

    inits.append(rc / rc.sum())

        r = rng.random(n)
        inits.append(r / r.sum())
    
    best = None
    best_score = np.inf
    
    for p in inits:
        if time.monotonic() > deadline:
            break
        n = len(p)
        p = p.copy()
        
        # FW iterations
        for it in range(200):
            if time.monotonic() > deadline:
                break
            M = np.max(np.convolve(p, p))
            q = fw_iteration(p, n, M)
            # Step size

            p_new = (1 - t) * p + t * q
            p_new = np.clip(p_new, 0, None)
            p_new = p_new / p_new.sum()
            p = p_new
        
        # Evaluate

        if np.isfinite(score) and score < best_score:
            best_score = score
            best = p.copy()
        print(f"Init {len(inits)}, iterations done, score: {best_score:.6f}")
    
    return [float(x) for x in best]
```

```python
"""
Frank-Wolfe optimization with LP subproblem for minimizing the peak
of the self-convolution of a non-negative sequence.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_constraint_matrix(p, n):
    """Build the (2n-1) x n constraint matrix for the LP subproblem."""
    A = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        if k_lo <= k_hi:
            # A[j, k] = p[j-k] for k in [k_lo, k_hi]
            # j-k ranges from j-k_hi to j-k_lo
            A[j, k_lo:k_hi + 1] = p[j - k_hi:j - k_lo + 1][::-1]
    return A


def solve_lp(p, n, M):
    """Solve the LP subproblem: max sum(q) s.t. A@q <= M, q >= 0."""
    A = build_constraint_matrix(p, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={"presolve": True, "dual_feasibility_tolerance": 1e-9})
    if res.status != 0:
        return None
    return res.x


def compute_score(p):
    """Compute the evaluation score for a normalized sequence."""
    n = len(p)
    conv = np.convolve(p, p)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize the peak self-convolution of a non-negative sequence using
    Frank-Wolfe iterations with an LP subproblem, combined with multiple
    diverse initializations and periodic re-evaluation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    def remaining():
        return max(0.0, deadline - time.monotonic())

    # --- Generate diverse initializations ---
    inits = []

    # 1. Given sequence (if available)
    try:
        p0 = np.asarray(height_sequence_1, dtype=np.float64)
        p0 = np.clip(p0, 0.0, 1000.0)
        if p0.size > 0 and p0.sum() > 0.01:
            inits.append(p0 / p0.sum())
    except Exception:
        pass

    n_base = 2500

    # 2. Uniform with random perturbation
    for trial in range(8):
        scale = 0.05 + 0.15 * trial / 8.0  # increasing perturbation
        p = np.ones(n_base) + scale * rng.standard_normal(n_base)
        p = np.clip(p, 0.0, None)
        if p.sum() > 0.01:
            inits.append(p / p.sum())

    x = np.arange(n_base)
    for freq in [1, 2, 3, 5]:
        for phase in [0.0, 0.7, 1.3]:
            p = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n_base + phase)
            p = np.clip(p, 0.0, None)
            if p.sum() > 0.01:

    # 4. Sparse/spiky initializations
    for trial in range(3):
        p = rng.random(n_base) ** 3  # skewed distribution

            inits.append(p / p.sum())

        p = np.ones(n_base)
        p[::period] = 2.0
        p[1::period] = 0.3 if period > 2 else 0.5
        if p.sum() > 0.01:
```

"""
Frank-Wolfe optimization with LP subproblem to minimize the peak of the
self-convolution of a non-negative sequence, thereby minimizing the score
2n * max(conv(a,a)) / (sum(a))^2. Uses multiple diverse initializations
and active-set LP reduction for efficiency.
"""

import numpy as np
from scipy.optimize import linprog

def build_constraint_matrix(p, n, active_rows=None):
    """Build constraint matrix A where (A @ q)_j = conv(p, q)_j for row j."""
    if active_rows is not None:
        rows = active_rows
    else:
        rows = np.arange(2 * n - 1)
    A = np.zeros((len(rows), n))
    for idx, j in enumerate(rows):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        if k_lo <= k_hi:
            A[idx, k_lo:k_hi + 1] = p[j - k_hi:j - k_lo + 1][::-1]
    return A


def solve_lp_subproblem(p, n, M, conv_prev, use_active=True):
    """Solve LP: max sum(q) s.t. conv(p,q)_j <= M for all j, q >= 0."""
    if use_active:
        # Keep only rows where conv_prev is at least 40% of M
        threshold = 0.4 * M
        active = np.where(conv_prev >= threshold)[0]
        # Always include the argmax row
        peak_row = np.argmax(conv_prev)
        if peak_row not in active:
            active = np.append(active, peak_row)
    else:
        active = np.arange(2 * n - 1)

    A = build_constraint_matrix(p, n, active)
    c = -np.ones(n)
    b = np.full(len(active), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 5.0})
        if res.status != 0:
            return None
        return res.x
    except Exception:
        return None
