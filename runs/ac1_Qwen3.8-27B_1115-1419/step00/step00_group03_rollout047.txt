    F = np.zeros((nc, n), dtype=np.float64)
    for i in range(nc):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            F[i, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    return F

def lp_direction(f, n, S):
    """Find b minimizing max_k (f*b)_k s.t. sum(b)=S, b>=0."""
    total = 2 * n - 1
    max_c = max(2000, min(total, 5000))
    if total <= max_c:
        idx = np.arange(total)
    else:
        idx = np.unique(np.linspace(0, total - 1, max_c, dtype=int))
    
    F = build_F(f, n, idx)
    nc = len(idx)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((nc, n + 1), dtype=np.float64)
    A_ub[:, :n] = F
    A_ub[:, n] = -1.0
    b_ub = np.zeros(nc)
    
    A_eq = np.ones((1, n), dtype=np.float64)
    b_eq = np.array([S])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if not res.success:
        return None
    return res.x[:n]

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    LP-guided local search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Uses a linear program at each step to find the optimal perturbation
    direction (minimizing max interaction with current sequence), combined
    with line search and random perturbations for global exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    
    # Get starting sequence
    try:
        seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except:
        seq = np.ones(2048, dtype=np.float64)
    
    if seq.ndim != 1 or seq.size < 10 or not np.isfinite(seq).all() or seq.sum() < 0.01:
        seq = np.ones(2048, dtype=np.float64)
    
    n = seq.size
    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Init: score={best_score:.6f}, n={n}")
    
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        f = best
        S = float(np.sum(f))
        if S < 0.01:
            break
        
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        # Get LP direction
        b = lp_direction(f, n, S)
        
        if b is not None:
            b = np.maximum(b, 0.0)
            bs = float(np.sum(b))
            if bs > 1e-12:
                b = b / bs * S
                found = False
                for mix in [0.3, 0.5, 0.1, 0.7, 1.0, 0.2, 0.8, 0.05]:
                    cand = (1.0 - mix) * f + mix * b
                    cand = np.clip(cand, 0.0, 1000.0)
                    sc = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best = cand
                        best_score = sc
                        found = True
                        it += 1
                        if it <= 5 or it % 20 == 0:
                            elapsed = time.monotonic() - start_time
                            print(f"  it={it} t={elapsed:.0f}s score={best_score:.6f} mix={mix:.2f}")
                        break
                if found:
                    stall = 0
                    continue
        
        # No improvement from LP
        stall += 1
        if stall == 1:
            pass  # First stall, just try again next iteration
        elif stall >= 10 and stall % 5 == 0:
            # Try perturbations
            for scale in [0.01, 0.05, 0.1, 0.3]:
                p = f * np.exp(scale * rng.standard_normal(n))
                p = np.clip(p, 0.0, 1000.0)
                sc = float(evaluate_sequence(p.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = p
                    best_score = sc
                    stall = 0
                    it += 1
                    print(f"  pert(s={scale:.2f}) score={best_score:.6f}")
                    break
        elif stall > 80:
            # Major restart
            new_f = 0.5 + 0.5 * rng.random(n)
            sc = float(evaluate_sequence(new_f.tolist()))
            if np.isfinite(sc):
                if sc < best_score:
                    best = new_f
                    best_score = sc
                    stall = 0
                    print(f"  random restart improved: {best_score:.6f}")
                elif stall > 150:
                    # Accept a random worse point to escape
                    best = new_f
                    best_score = sc
                    stall = 0
                    print(f"  forced restart: {best_score:.6f}")
    
    elapsed = time.monotonic() - start_time
    print(f"Final: score={best_score:.6f}, n={n}, iters={it}, time={elapsed:.0f}s")
    return [float(x) for x in best]
