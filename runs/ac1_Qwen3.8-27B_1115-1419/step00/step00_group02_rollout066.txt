def _build_conv_matrix(a):
    n = len(a)
    m = 2*n - 1
    # Compute row pointers (CSR format)
    # Row k has indices from max(0, k-n+1) to min(n-1, k)
    # Number of entries in row k: min(n-1, k) - max(0, k-n+1) + 1
    row_counts = np.array([min(n-1, k) - max(0, k-n+1) + 1 for k in range(m)])
    row_ptr = np.concatenate([[0], np.cumsum(row_counts)])
    # Build column indices and values
    nnz = row_ptr[-1]
    col_idx = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    pos = 0
    for k in range(m):

        hi = min(n - 1, k)
        for l in range(lo, hi + 1):
            col_idx[pos] = l
            vals[pos] = a[k - l]
            pos += 1
    return sp.csr_matrix((vals, col_idx, row_ptr), shape=(m, n))

def _solve_lp(a, max_time=15.0):
    """Solve max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(a)
    if n > 10000:
        return None
    
    a = a.astype(np.float64)
    c = np.convolve(a, a)
    M = float(np.max(c))
    
    if M <= 0:
        return None
    
    # Build sparse matrix
    m = 2 * n - 1
    row_counts = np.zeros(m, dtype=np.int64)
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        row_counts[k] = hi - lo + 1
    row_ptr = np.concatenate([[0], np.cumsum(row_counts)])
    nnz = int(row_ptr[-1])
    col_idx = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    
    pos = 0
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for l in range(lo, hi + 1):
            col_idx[pos] = l
            vals[pos] = a[k - l]
            pos += 1
    
    A_mat = sp.csr_matrix((vals, col_idx, row_ptr.copy()), shape=(m, n))
    
    # LP: minimize -sum(g) s.t. A_mat @ g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'time_limit': max_time})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if np.sum(g) > 1e-10:
                return g
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None
