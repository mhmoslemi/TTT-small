"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.

The program combines:
- analytic/low-frequency initializers (constant, one-sided square-root taper,
  smoothed random variants),
- an LP-inspired state-of-art step for moderate sizes,
- smoothed multiplicative projected-gradient / conditional-gradient steps using
  FFT-based convolution and its active-lag gradients,
- a final short coordinate/block polish using the exact evaluator.

The exact evaluator, when available, is always authoritative for the returned
best sequence.
"""

import numpy as np
import time
import math

try:
    import scipy.optimize as sco
    SCIPY_OPT_AVAILABLE = True
except Exception:
    SCIPY_OPT_AVAILABLE = False

TARGET = 1.5027
EPS = 1e-300


def next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def prepare_sequence(a):
    if a is None:
        return None
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= EPS:
        return None
    return arr / s


def floor_positive(a, eps=1e-12):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, float(eps))
    s = float(arr.sum())
    if not np.isfinite(s) or s <= EPS:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def exact_score_array(a):
    arr = prepare_sequence(a)
    if arr is None:
        return np.inf
    try:
        val = evaluate_sequence(arr.tolist())
    except Exception:
        val = np.inf
    if not np.isfinite(val):
        return np.inf
    return float(val)


def fast_conv(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if N is None:
        N = next_pow2(3 * n + 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = np.ascontiguousarray(c[: 2 * n - 1])
    np.maximum(c, 0.0, out=c)
    return c


def fast_score(a, N=None):
    arr = prepare_sequence(a)
    if arr is None:
        return np.inf
    s = float(arr.sum())
    if s <= EPS:
        return np.inf
    c = fast_conv(arr, N)
    M = float(np.max(c))
    return float(2.0 * arr.size * M / (s * s))


def loss_from_c(c, p):
    M = float(np.max(c))
    if M <= 0.0 or not np.isfinite(M):
        return 0.0, 0.0
    z = c / M
    val = float(np.sum(z ** p))
    if val <= 0.0 or not np.isfinite(val):
        val = 1.0
    return M * (val ** (1.0 / p)), M


def weights_from_c(c, p):
    M = float(np.max(c))
    m = c.size
    if M <= 0.0 or not np.isfinite(M):
        w = np.full(m, 1.0 / float(m), dtype=np.float64)
        return w, M
    z = np.maximum(c / M, 0.0)
    w = z ** (p - 1.0)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.full(m, 1.0 / float(m), dtype=np.float64)
    else:
        w = w / s
    return w, M


def gradient_from_weights(a, w, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if N is None:
        N = next_pow2(3 * n + 1)
    ar = np.ascontiguousarray(a[::-1])
    Ag = np.fft.rfft(ar, N)
    Wg = np.fft.rfft(w, N)
    conv = np.fft.irfft(Ag * Wg, N)
    g = np.ascontiguousarray(conv[n - 1:n - 1 + n])
    return 2.0 * g


def apply_log_shift(a, centered_gradient, scale, eta):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if scale <= 0.0 or not np.isfinite(scale):
        scale = 1.0
    g = np.asarray(centered_gradient, dtype=np.float64).ravel()
    arg = -float(eta) * g / scale
    arg = np.clip(arg, -20.0, 20.0)
    b = arr * np.exp(arg)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if not np.isfinite(s) or s <= EPS:
        return None
    return b / s


def random_perturb(a, rng, strength=0.10):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    if n <= 0:
        return None
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    vals = np.zeros(n, dtype=np.float64)
    for freq in (1, 2, 3, 5):
        amp = float(rng.uniform(0.1, 0.7))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        vals += amp * np.cos(2.0 * np.pi * float(freq) * x + phase)
    vals += 0.1 * float(strength) * rng.standard_normal(n)
    b = arr * np.exp(strength * vals)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= EPS:
        return None
    return b / s


def make_initial(n, kind=0, rng=None):
    n = int(n)
    if n <= 0:
        return None
    i = np.arange(n, dtype=np.float64)
    if kind == 0:
        a = np.ones(n, dtype=np.float64)
    elif kind == 1:
        a = 1.0 / np.sqrt(i + 1.0)
    elif kind == 2:
        if rng is None:
            delta = 0.2
        else:
            delta = float(rng.choice(np.array([0.0, 0.05, 0.15, 0.40, 1.00], dtype=np.float64)))
        a = 1.0 / np.sqrt(i + 1.0) + delta * (2.0 / np.sqrt(float(n)))
    elif kind == 3:
        a = 1.0 / np.sqrt(i + 1.0)
        if rng is not None:
            b = random_perturb(a, rng, 0.15)
            if b is not None:
                return prepare_sequence(b)
        return prepare_sequence(a)
    elif kind == 4:
        if rng is None:
            decay = 1.0 / max(1.0, float(n))
        else:
            decay = float(rng.uniform(0.0, 2.0)) / float(n)
        a = (1.0 / np.sqrt(i + 1.0)) * np.exp(-decay * i)
    elif kind == 5:
        if rng is None:
            amp = 0.5
            phase = 0.0
        else:
            amp = float(rng.uniform(0.1, 0.8))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + amp * np.sin(2.0 * np.pi * i / float(n) + phase)
    else:
        a = np.ones(n, dtype=np.float64)
    return prepare_sequence(a)


def resample_sequence(a, m):
    arr = prepare_sequence(a)
    m = int(m)
    if arr is None or m <= 0:
        return None
    n = arr.size
    if n == m:
        return arr.copy()
    if n == 1:
        return (m * arr).copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, m, endpoint=False)
    loga = np.log(np.clip(arr, 1e-300, None))
    vals = np.interp(x_new, x_old, loga)
    vals = vals - float(np.max(vals))
    b = np.exp(vals)
    return prepare_sequence(b)


def initial_for_length(n, best_arr, rng, kind=1):
    if best_arr is None:
        return make_initial(n, kind, rng)
    n = int(n)
    if best_arr.size == n:
        return best_arr.copy()
    b = resample_sequence(best_arr, n)
    if b is not None:
        return b
    return make_initial(n, kind, rng)


def block_perturb(a, rng):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = arr.size
    if n <= 0:
        return None
    width = int(rng.integers(max(1, n // 50), max(2, n // 5)))
    width = min(n, width)
    start = int(rng.integers(0, max(1, n - width + 1)))
    b = arr.copy()
    b[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.05:
        b = 0.5 * (b + b[::-1])
    return prepare_sequence(b)


def build_conv_matrix(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            T[k, lo:hi + 1] = a[k - idx]
    return T


def run_lp_start(a0, deadline, rng):
    arr = prepare_sequence(a0)
    if arr is None:
        return None, np.inf
    a = floor_positive(arr, 1e-12)
    n = a.size
    if n < 8:
        sc = exact_score_array(a)
        return a, sc
    if n > 1024:
        sc = exact_score_array(a)
        return a, sc

    N = next_pow2(3 * n + 1)
    a = floor_positive(a, 1e-12)
    c = fast_conv(a, N)
    M = float(np.max(c))
    score = float(2.0 * n * M)
    best_fast = score
    best_fast_arr = a.copy()
    best_exact = exact_score_array(a)
    if not np.isfinite(best_exact):
        best_exact = score
    best_exact_arr = a.copy()

    solves = 0
    stall = 0

    while time.monotonic() < deadline and solves < 500:
        a = floor_positive(a, 1e-12)
        c = fast_conv(a, N)
        M = float(np.max(c))
        if M <= 0.0 or not np.isfinite(M):
            break
        score = float(2.0 * n * M)
        if score < best_fast:
            best_fast = score
            best_fast_arr = a.copy()

        T = build_conv_matrix(a)
        if not np.all(np.isfinite(T)):
            break
        T /= M
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.ones(2 * n - 1, dtype=np.float64)
        bounds = [(0.0, None)] * n
        limit = max(1.0, min(25.0, float(deadline - time.monotonic() - 0.5)))

        res = None
        if SCIPY_OPT_AVAILABLE:
            try:
                res = sco.linprog(
                    c_obj,
                    A_ub=T,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options={"time_limit": limit}
                )
            except Exception:
                try:
                    res = sco.linprog(
                        c_obj,
                        A_ub=T,
                        b_ub=b_ub,
                        bounds=bounds,
                        method="interior-point",
                        options={"time_limit": limit}
                    )
                except Exception:
                    res = None

        solves += 1

        if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
            stall += 1
            if stall > 6:
                break
            continue

        b = prepare_sequence(res.x)
        if b is None:
            stall += 1
            if stall > 6:
                break
            continue

        candidate_list = [(1.0, b)]
        for t in (0.70, 0.50, 0.30, 0.15, 0.05):
            u = (1.0 - float(t)) * a + float(t) * b
            u = prepare_sequence(u)
            if u is not None:
                candidate_list.append((t, u))

        improved = False
        for _, u in candidate_list:
            cu = fast_conv(u, N)
            Mu = float(np.max(cu))
            su = float(2.0 * n * Mu)
            if np.isfinite(su) and su < score:
                a = floor_positive(u, 1e-12)
                score = su
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

        if solves % 5 == 0:
            es = exact_score_array(a)
            if np.isfinite(es) and es < best_exact:
                best_exact = es
                best_exact_arr = a.copy()

    for cand in (best_fast_arr, a):
        if cand is None:
            continue
        es = exact_score_array(cand)
        if np.isfinite(es) and es < best_exact:
            best_exact = es
            best_exact_arr = cand.copy()

    if not np.isfinite(best_exact):
        best_exact = best_fast
        best_exact_arr = best_fast_arr
    return best_exact_arr, best_exact

def consider_array(a, current_best_score, current_best_arr):

                cand_score = fast_score(cand)
            if np.isfinite(cand_score) and cand_score < best_score:
                best_score = cand_score
                best_arr = cand.copy()

    final_arr = prepare_sequence(best_arr)
    if final_arr is None:
        final_arr = np.array([1.0], dtype=np.float64)
    final_score = exact_score_array(final_arr)
    if not np.isfinite(final_score):
        final_score = fast_score(final_arr)
    if not np.isfinite(final_score):
        final_arr = np.array([1.0], dtype=np.float64)
        final_score = 2.0

    print("final score = %.10f len = %d" % (final_score, int(final_arr.size)))
    return [float(x) for x in final_arr]
