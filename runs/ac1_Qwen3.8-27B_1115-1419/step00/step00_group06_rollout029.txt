    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    S = np.sum(a)
    return 2.0 * n * max_b / (S**2)

import numpy as np
import time

def fast_eval(a):
    """Fast evaluation using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    max_b = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S**2)
