import numpy as np
import time

def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a)

def compute_score(a):
    """Compute J(a) = 2n * max(conv) / (sum a)^2."""
    n = len(a)
    c = compute_conv(a)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * np.max(c) / (s * s)

def compute_smoothed_gradient(a, t):
    """
    Compute gradient of 2n * soft_max(conv(a,a), t) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    c = compute_conv(a)
    m = np.max(c)
    exp_t = np.exp(t * (c - m))
    w = exp_t / np.sum(exp_t)
    
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2 * (a[::-1] * full_conv with w)[n-1+i]
    a_rev = a[::-1]
    full_conv = np.convolve(a_rev, w, mode='full')
    grad = 2.0 * full_conv[n-1:n-1+n]
    
    return 4.0 * n * grad  # multiply by 2n for the full objective scaling

"""
Minimize J(a) = 2n * max(conv(a,a)) / (sum(a))^2 for non-negative sequences.
Strategy: Smoothed gradient descent on the probability simplex with multiple
diverse initializations, adaptive step size, temperature annealing, and
periodic random perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.signal import fftconvolve


def simplex_project(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho_candidates = np.nonzero(mu * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    return fftconvolve(a, a, mode='full')


def compute_score_norm(a):
    """Compute J assuming sum(a)=1: returns 2n*max(conv(a,a))."""
    n = len(a)
    c = full_conv(a)
    return 2.0 * n * np.max(c)


def compute_gradient(a, t):
    """
    Compute gradient of 2n * soft_max(conv(a,a), t) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) complexity.
    """
    n = len(a)
    c = full_conv(a)
    m = np.max(c)
    exp_t = np.exp(t * (c - m))
    w = exp_t / np.sum(exp_t)
    # grad[i] = 4n * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 4n * (correlation of a with w at lag i)
    a_rev = a[::-1]
    full_cor = fftconvolve(a_rev, w, mode='full')
    grad = 4.0 * n * full_cor[n - 1:n - 1 + n]
    return grad


def make_initialization(n, init_type, rng):
    """Create an initial non-negative sequence of length n, normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'cosine':
        a = 1.0 + np.cos(2.0 * np.pi * i / n)
        a = np.maximum(a, 1e-8)
    elif init_type == 'sine_sq':
        a = np.sin(np.pi * (i + 0.5) / n) ** 2
        a = np.maximum(a, 1e-8)
    elif init_type == 'edge':
        a = (i / max(n - 1, 1) - 0.5) ** 2
        a = np.maximum(a, 1e-8)
    elif init_type == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        a = np.maximum(a, 1e-8)
    elif init_type == 'two_peak':
        c1 = n / 4.0
        c2 = 3.0 * n / 4.0
        sigma = n / 12.0
        a = np.exp(-0.5 * ((i - c1) / sigma) ** 2) + np.exp(-0.5 * ((i - c2) / sigma) ** 2)
        a = np.maximum(a, 1e-8)
    elif init_type == 'random05':
        a = rng.dirichlet(np.full(n, 0.5))
    elif init_type == 'random1':
        a = rng.dirichlet(np.full(n, 1.0))
    elif init_type == 'random3':
        a = rng.dirichlet(np.full(n, 3.0))
    elif init_type == 'chirp':
        a = 1.0 + np.cos(2.0 * np.pi * i * (i + 1) / (2.0 * n))
        a = np.maximum(a, 1e-8)
    elif init_type == 'modulated':
        a = (1.0 + np.cos(2.0 * np.pi * i / n)) * (1.0 + 0.5 * np.cos(4.0 * np.pi * i / n))
        a = np.maximum(a, 1e-8)
    else:
        a = np.ones(n)
    a = a.astype(np.float64)
    s = np.sum(a)
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def optimize_sequence(a0, deadline, rng, t_start=15.0, t_end=1.0):
    """
    Optimize a sequence using smoothed projected gradient descent.
    a0: initial sequence with sum=1.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    current_score = best_score

    t = t_start
    eta = 1.0 / (2.0 * n)
    iter_count = 0
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline:
        grad = compute_gradient(a, t)

        # Projected gradient step with backtracking
        accepted = False
        for attempt in range(5):
            a_new = simplex_project(a - eta * grad)
            a_new_score = compute_score_norm(a_new)
            if a_new_score < current_score * (1.0 - 1e-6):
                a = a_new
                current_score = a_new_score
                accepted = True
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                break
            eta *= 0.5

        if accepted:
            no_improve = 0
            eta = min(eta * 1.02, 2.0 / n)
        else:
            no_improve += 1

        # Temperature annealing (logarithmic)
        progress = min(1.0, iter_count / max(1, 5000))
        t = t_start * (t_end / t_start) ** progress

        # Periodic perturbation to escape local minima
        if no_improve >= max_no_improve:
            # Moderate perturbation
            noise_scale = 0.05 * (1.0 + no_improve / max_no_improve)
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / s
            current_score = compute_score_norm(a)
            no_improve = 0
            eta = 1.0 / (2.0 * n)

        iter_count += 1

        if iter_count % 20000 == 0:
            elapsed = time.monotonic()
            print(f"      iter={iter_count}, curr={current_score:.6f}, best={best_score:.6f}, t={t:.2f}, eta={eta:.4f}")

    return best_a, best_score
