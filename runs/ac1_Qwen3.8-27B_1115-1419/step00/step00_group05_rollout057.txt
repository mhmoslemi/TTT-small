    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)


def compute_objective(p):
    """Compute the objective 2*n*max(conv)/sum^2 for probability vector p."""
    n = len(p)
    conv = np.convolve(p, p)
    max_conv = np.max(conv)
    s = np.sum(p)
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def build_constraint_matrix(f):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_direction(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Return rescaled g or None."""
    n = len(f)
    S = np.sum(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if result.success and result.x is not None:
        g = result.x
        sum_g = np.sum(g)
        if sum_g > S * 1.00001:
            return g * S / sum_g
    return None


def subgradient_step(p, alpha):
    """One subgradient step on the exact max of convolve(p,p)."""
    n = len(p)
    conv = np.convolve(p, p)
    k_max = np.argmax(conv)
    # subgrad[i] = 2 * p[k_max - i] if 0 <= k_max - i <= n-1
    subgrad = np.zeros(n)
    for i in range(n):
        idx = k_max - i
        if 0 <= idx <= n - 1:
            subgrad[i] = 2.0 * p[idx]
    # Center the gradient (tangent space of simplex)
    subgrad = subgrad - np.mean(subgrad)
    p_new = p - alpha * subgrad
    return project_simplex(p_new)


def smooth_grad_step(p, T, lr):
    """One projected gradient step using log-sum-exp smoothing."""
    n = len(p)
    conv = np.convolve(p, p)
    m = np.max(conv)
    shifted = T * (conv - m)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    weights = exp_s / sum_exp
    # grad[i] = 2 * sum_j weights[j+i] * p[j]
    full = np.convolve(p, weights[::-1])
    cross = full[np.arange(2 * n - 2, n - 2, -1)]
    grad = 2.0 * cross
    grad_centered = grad - np.mean(grad)
    p_new = p - lr * grad_centered
    return project_simplex(p_new)


def generate_initializations(n, rng):
    """Generate diverse non-uniform initializations."""
    inits = []
    i = np.arange(n)

    # Cosine tapered (Hann-like)
    c = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    c = np.maximum(c, 1e-12)
    inits.append(c / c.sum())

    # Power law decay
    pw = (np.arange(n) + 1) ** (-0.7)
    inits.append(pw / pw.sum())

    # Bimodal Gaussian
    b = np.exp(-0.5 * ((i - n / 4) / (n / 10)) ** 2) + np.exp(-0.5 * ((i - 3 * n / 4) / (n / 10)) ** 2)
    b = np.maximum(b, 1e-12)
    inits.append(b / b.sum())

    # Triangle (inverted V)
    t = 1.0 - 2.0 * np.abs(i - (n - 1) / 2.0) / n
    t = np.maximum(t, 1e-12)
    inits.append(t / t.sum())

    # Sinc first lobe
    s = np.sinc(i / (n / 3.0))
    s = np.maximum(s, 1e-12)
    inits.append(s / s.sum())

    # Raised cosine (Blackman-like)
    x = i / (n - 1)
    r = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    r = np.maximum(r, 1e-12)
    inits.append(r / r.sum())

    # Exponential decay
    e = np.exp(-3.0 * i / n)
    inits.append(e / e.sum())

    # Random smooth (low-pass filtered random)
    r_raw = rng.standard_normal(n)
    # Simple box filter
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    r_smooth = np.convolve(r_raw, kernel, mode='same')
    r_smooth = np.abs(r_smooth) + 0.1
    inits.append(r_smooth / r_smooth.sum())

    # Alternating high-low
    al = np.ones(n)
    al[np.arange(0, n, 2)] = 2.0
    al[np.arange(1, n, 2)] = 0.5
    inits.append(al / al.sum())

    # Quarter-wave cosine
    qw = np.cos(np.pi * i / (2 * n))
    qw = np.maximum(qw, 1e-12)
    inits.append(qw / qw.sum())

    return inits


def optimize_length(n, seed, budget_s):
    """Optimize a sequence of given length within budget. Returns (best_p, best_score)."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    best_p = None
    best_score = np.inf

    # Generate and evaluate initializations
    inits = generate_initializations(n, rng)
    candidate_scores = []
    for init in inits:
        sc = compute_objective(init)
        candidate_scores.append(sc)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = init.copy()

    print(f"  n={n}: best init score = {best_score:.6f}")

    # Sort initializations by score, optimize top few
    order = np.argsort(candidate_scores)
    for idx in order:
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        p = inits[idx].copy()
        score = compute_objective(p)

        # Phase 1a: Subgradient warmup (quick escape from near-uniform)
        alpha = 1.0 / (2.0 * n)
        for it in range(200):
            if time.monotonic() > deadline - 5:
                break
            p_new = subgradient_step(p, alpha)
            s_new = compute_objective(p_new)
            if s_new < score * (1 - 1e-8):
                p = p_new
                score = s_new
            alpha *= 0.98

        # Phase 1b: LP-based improvement
        lp_budget = remaining * 0.5
        lp_deadline = time.monotonic() + lp_budget
        lp_iter = 0
        prev_score = score + 1
        stall = 0
        while time.monotonic() < lp_deadline:
            conv = np.convolve(p, p)
            M = np.max(conv)
            S = np.sum(p)

            g = lp_direction(p, M, time_limit=min(10, max(2, (lp_deadline - time.monotonic()) * 0.5)))
            if g is None:
                break

            # Line search
            best_t = 0.0
            best_t_score = score
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                p_trial = project_simplex((1 - t) * p + t * g)
                s_trial = compute_objective(p_trial)
                if s_trial < best_t_score:
                    best_t_score = s_trial
                    best_t = t

            if best_t > 0:
                p = project_simplex((1 - best_t) * p + best_t * g)
                score = best_t_score
                if score < best_score:
                    best_p = p.copy()
                    best_score = score
                if abs(prev_score - score) > 1e-9:
                    stall = 0
                else:
                    stall += 1
                prev_score = score
                lp_iter += 1
                if lp_iter % 30 == 0:
                    print(f"    n={n} LP iter {lp_iter}: score={score:.6f}")
                if stall > 10:
                    break
            else:
                break

        # Phase 2: Smooth gradient descent with annealing
        T = float(n)
        lr = 1.0 / (2.0 * n * T)
        gd_deadline = min(deadline - 5, time.monotonic() + remaining * 0.3)
        gd_iter = 0
        gd_stall = 0
        while time.monotonic() < gd_deadline:
            T = max(1.0, n * (0.95 ** gd_iter))
            lr = 0.5 / (n * T * (1.0 + 0.005 * gd_iter))
            p_new = smooth_grad_step(p, T, lr)
            s_new = compute_objective(p_new)
            if s_new < score - 1e-12:
                p = p_new
                score = s_new
                gd_stall = 0
                if score < best_score:
                    best_p = p.copy()
                    best_score = score
            else:
                gd_stall += 1
            gd_iter += 1
            if gd_stall > 200:
                break

        # Phase 3: Local block perturbations
        pert_deadline = deadline - 3
        pert_iter = 0
        while time.monotonic() < pert_deadline:
            width = max(1, int(n / rng.integers(30, 150)))
            start = int(rng.integers(0, max(1, n - width)))
            p_trial = p.copy()
            p_trial[start:start + width] *= np.exp(rng.normal(0, 0.15, size=width))
            p_trial = project_simplex(p_trial)
            s_trial = compute_objective(p_trial)
            if s_trial < score:
                p = p_trial
                score = s_trial
                if score < best_score:
                    best_p = p.copy()
                    best_score = score
            pert_iter += 1

        print(f"  n={n} init[{idx}] final: score={best_score:.6f}")

    if best_p is None:
        best_p = np.ones(n) / n
        best_score = 2.0

    return best_p, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Tries multiple sequence lengths and optimization phases
    to find the best step function minimizing 2*n*max(conv(p,p))/sum(p)^2.
    """
    print(f"Starting optimization at {time.monotonic():.1f}s, budget={budget_s}s")
    deadline = time.monotonic() + float(budget_s) - 10.0  # safety margin
    global best_overall_p
    global best_overall_score

    best_overall_p = None
    best_overall_score = np.inf

    # Try multiple lengths
    lengths_to_try = [256, 512, 1024, 2048, 4096]

    # Distribute time: give more time to medium lengths
    time_alloc = {256: 100, 512: 400, 1024: 600, 2048: 800, 4096: 400}
    total_alloc = sum(time_alloc.values())
    actual_budget = min(budget_s - 20, deadline - time.monotonic())
    # Scale allocations
    scale = actual_budget / total_alloc

    for n in lengths_to_try:
        if time.monotonic() > deadline:
            break
        alloc = time_alloc[n] * scale
        if alloc < 20:
            alloc = 20
        if time.monotonic() + alloc > deadline:
            alloc = deadline - time.monotonic()
        if alloc < 10:
            break

        print(f"\n=== Optimizing length n={n}, time budget={alloc:.0f}s ===")
        try:
            p, score = optimize_length(n, seed + n, alloc)
        except Exception as e:
            print(f"  Error for n={n}: {e}")
            continue

        if score < best_overall_score:
            best_overall_score = score
            best_overall_p = p.copy()
            print(f"  NEW BEST: n={n}, score={score:.6f}")

    # Final: try to use height_sequence_1 as inspiration if available
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, 1000)
        if hs.size > 0 and np.sum(hs) > 0:
            hs_norm = hs / np.sum(hs)
            # Try truncated versions
            for trunc_n in [512, 1024, 2048]:
                if time.monotonic() > deadline:
                    break
                if hs.size >= trunc_n:
                    trunc = hs_norm[:trunc_n]
                    trunc = trunc / trunc.sum()
                    sc = compute_objective(trunc)
                    if sc < best_overall_score:
                        best_overall_score = sc
                        best_overall_p = trunc.copy()
                        print(f"  New best from truncated hs: n={trunc_n}, score={sc:.6f}")
    except Exception:
        pass

    # Final perturbation polish on the best
    if best_overall_p is not None and time.monotonic() < deadline - 5:
        n = len(best_overall_p)
        p = best_overall_p.copy()
        score = best_overall_score
        rng = np.random.default_rng(seed + 99999)
        polish_deadline = deadline - 2
        it = 0
        while time.monotonic() < polish_deadline:
            width = max(1, int(n / rng.integers(20, 100)))
            start = int(rng.integers(0, max(1, n - width)))
            p_trial = p.copy()
            p_trial[start:start + width] *= np.exp(rng.normal(0, 0.08, size=width))
            p_trial = project_simplex(p_trial)
            s_trial = compute_objective(p_trial)
            if s_trial < score:
                p = p_trial
                score = s_trial
                best_overall_p = p.copy()
                best_overall_score = score
            it += 1
            if it % 500 == 0:
                print(f"  Polish iter {it}: score={score:.6f}")

    # Return as list of floats
    if best_overall_p is None:
        return [1.0] * 128
    return [float(x) for x in best_overall_p]
