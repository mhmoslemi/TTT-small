    L = 1
    while L < n:
        L *= 2

    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n]

    s = a.sum()
    if s < 0.01:
        return np.inf
    c = my_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

    n = len(p)
    L = 2 * n - 1
    
    # Compute convolution
    P = np.fft.rfft(p, LFFT)
    C = np.fft.irfft(P * P, LFFT)[:L]
    
    # Smoothed max weights
    C_shift = C - C.max()
    W = np.exp(C_shift / tau)
    W_sum = W.sum()
    if W_sum == 0 or not np.isfinite(W_sum):
        W = np.zeros(L)
        W[C.argmax()] = 1.0

    W_ext = np.zeros(L2)
    W_ext[:L] = W
    P2 = np.fft.rfft(p, L2)

    log_p = log_p - log_p.max()
    p_new = np.exp(log_p)
    p_new = p_new / p_new.sum()
    
    return p_new, C.max()

"""
Algorithm: Mirror descent (exponential weights) on the smoothed max-convolution
objective, followed by LP-based active set refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def my_convolve_fftp(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n]


def compute_score_normalized(p, n, LFFT):
    """Compute 2*n*max_conv assuming sum(p)=1."""
    P = np.fft.rfft(p, LFFT)
    C = np.fft.irfft(P * P, LFFT)[:2*n-1]
    return 2.0 * n * C.max()


def md_step(p, tau, eta, LFFT, L2, n):
    """One mirror descent step. Returns new p and current max_conv."""
    L = 2 * n - 1
    
    # Convolution
    P = np.fft.rfft(p, LFFT)
    C = np.fft.irfft(P * P, LFFT)[:L]
    c_max = C.max()
    
    # Softmax weights
    C_shift = C - c_max
    W = np.exp(C_shift / tau)
    Ws = W.sum()
    if Ws < 1e-300 or not np.isfinite(Ws):
        W = np.zeros(L)
        W[int(np.argmax(C))] = 1.0
    else:
        W /= Ws
    
    # Gradient = 2 * cross-correlation(W, p)
    W_ext = np.zeros(L2)
    W_ext[:L] = W
    P2 = np.fft.rfft(p, L2)
    grad = np.fft.irfft(np.fft.rfft(W_ext, L2) * np.conj(P2), L2)[:n]
    grad *= 2.0
    grad -= grad.mean()  # Project to sum-0
    
    # Exponential update
    log_p = np.log(np.maximum(p, 1e-300)) - eta * grad
    log_p -= log_p.max()
    p_new = np.exp(log_p)
    p_sum = p_new.sum()
    if p_sum < 1e-300 or not np.isfinite(p_sum):
        return p, c_max
    p_new /= p_sum
    return p_new, c_max


def run_mirror_descent(p_init, time_budget, seed=0, verbose=True):
    """Run mirror descent to minimize max convolution (with sum p = 1)."""
    rng = np.random.default_rng(seed)
    n = len(p_init)
    L = 2 * n - 1
    
    LFFT = 1
    while LFFT < L:
        LFFT *= 2
    L2 = 1
    while L2 < 3 * n:
        L2 *= 2
    
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n, dtype=np.float64)
    else:
        p /= s
    
    best_p = p.copy()
    best_mc = p @ p  # This is NOT the max conv, just a lower bound. Use actual.
    best_mc = compute_score_normalized(p, n, LFFT) / (2.0 * n)  # = max_conv
    
    t0 = time.monotonic()
    total_steps = 0
    stagnant = 0
    
    n_phases = 5
    steps_per_phase = 3000
    tau_lo, tau_hi = 0.001, 0.2
    eta_lo, eta_hi = 0.0005, 0.1
    
    for phase in range(n_phases):
        f0 = phase / n_phases
        f1 = (phase + 1) / n_phases
        tau_a = tau_hi * (tau_lo / tau_hi) ** f0
        tau_b = tau_hi * (tau_lo / tau_hi) ** f1
        eta_a = eta_hi * (eta_lo / eta_hi) ** f0
        eta_b = eta_hi * (eta_lo / eta_hi) ** f1
        
        for step in range(steps_per_phase):
            if time.monotonic() - t0 > time_budget:
                break
            frac = step / max(1, steps_per_phase)
            tau = tau_a * (tau_b / tau_a) ** frac if tau_a != tau_b else tau_a
            eta = eta_a * (eta_b / eta_a) ** frac if eta_a != eta_b else eta_a
            
            p, c_max = md_step(p, tau, eta, LFFT, L2, n)
            p = np.maximum(p, 0.0)
            ps = p.sum()
            if ps < 1e-10 or not np.isfinite(ps):
                p = best_p.copy()
                c_max = best_mc
                continue
            p /= ps
            
            total_steps += 1
            score_part = 2.0 * n * c_max
            if score_part < best_mc * 2.0 * n:
                best_mc = c_max
                best_p = p.copy()
                stagnant = 0
            else:
                stagnant += 1
            
            if stagnant > 500 and rng.random() < 0.01:
                # Random restart within neighborhood
                noise = rng.normal(0, 0.05, n)
                p = p * (1 + noise)
                p = np.maximum(p, 0)
                ps = p.sum()
                if ps > 1e-10:
                    p /= ps
                stagnant = 0
            
            if verbose and total_steps % 2000 == 0:
                el = time.monotonic() - t0
                print(f"    MD: step={total_steps} phase={phase} max_conv={c_max:.6f} "
                      f"score={2*n*c_max:.6f} t={el:.1f}s")
        
        if time.monotonic() - t0 > time_budget:
            break
    
    return best_p, 2.0 * n * best_mc


def lp_refine_step(p, K=300):
    """One LP-based refinement. Returns improved p (sum=1) or original."""
    n = len(p)
    L = 2 * n - 1
    
    # Compute current convolution to find tightest constraints
    c = my_convolve_fftp(p, p)
    M = c.max()
    
    # Top K lags
    top_k = np.argsort(c)[-K:][::-1]
    
    # Build constraint matrix
    A = np.zeros((K, n), dtype=np.float64)
    for idx in range(K):
        x = int(top_k[idx])
        i_lo = max(0, x - n + 1)
        i_hi = min(n - 1, x)
        if i_lo <= i_hi:
            A[idx, i_lo:i_hi+1] = p[x - i_hi : x - i_lo + 1][::-1]
    
    # LP: max 1'b s.t. A b <= M, b >= 0
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(K, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 60})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-10 and np.isfinite(bs):
                return b / bs
    except Exception:
        pass
    return p


def lp_refine(p, time_budget, n_lps=5, K=300):
    """Run multiple LP refinement steps."""
    t0 = time.monotonic()
    best_p = p.copy()
    best_score = compute_score_normalized(best_p, len(best_p), int(2**np.ceil(np.log2(2*len(best_p)-1))))
    
    for i in range(n_lps):
        if time.monotonic() - t0 > time_budget - 10:
            break
        print(f"  LP step {i+1}/{n_lps}, K={K}")
        p_new = lp_refine_step(p, K)
        # Blend
        t_blend = 0.5
        p_blend = (1 - t_blend) * best_p + t_blend * p_new
        p_blend = np.maximum(p_blend, 0)
        ps = p_blend.sum()
        if ps > 1e-10:
            p_blend /= ps
        
        LFFT = 1
        while LFFT < 2 * len(p_blend) - 1:
            LFFT *= 2
        sc = compute_score_normalized(p_blend, len(p_blend), LFFT)
        
        if sc < best_score:
            best_p = p_blend.copy()
            best_score = sc
            p = p_new  # Continue from the LP result
        else:
            p = p_blend
        
        print(f"    score={sc:.6f}")
    
    return best_p, best_score

```python
"""
Mirror descent + LP refinement for minimizing max-autocorrelation of
non-negative sequences. Score = 2n*max(p*p)/(sum p)^2, minimized by
finding p with flat self-convolution.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _fft_len(n):
    """Smallest power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _conv(a, b, Lft=None):
    """Full linear convolution via FFT."""
    need = len(a) + len(b) - 1
    if Lft is None:
        Lft = _fft_len(need)
    return np.fft.irfft(np.fft.rfft(a, Lft) * np.fft.rfft(b, Lft), Lft)[:need]

def _score(p):
    """Score assuming p is valid (nonneg, sum>=0.01)."""
    n = len(p)
    s = p.sum()
    if s < 0.01:
        return np.inf
    c = _conv(p, p)
    return 2.0 * n * c.max() / (s * s)

def _normalize(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s > 1e-15:
        p /= s
    return p
