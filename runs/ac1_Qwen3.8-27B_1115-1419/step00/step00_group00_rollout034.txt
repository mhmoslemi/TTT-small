    n = len(a)
    c = np.convolve(a, a, 'full')
    k_star = np.argmax(c)
    # Subgradient: ∂c[k_star]/∂a[i] = a[k_star - i] for valid i
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    # Objective gradient: 2n * grad
    # Update: a - lr * 2n * grad
    a_new = a - lr * 2 * n * grad
    # Project to simplex (sum=1, non-negative)
    a_new = project_to_simplex(a_new)
    return a_new

    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n+1)
