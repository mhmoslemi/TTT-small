from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve: max sum(b) s.t. convolve(f,b) <= M, b >= 0."""
    n = len(f)
    # Objective: maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M (for each k = 0, ..., 2n-2)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                A[k, j] = f[j]
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=np.full(n_constraints, M), bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

Minimize 2*n*max(convolve(x,x))/sum(x)^2 using LP-based and gradient-based optimization.
Strategy: For multiple sequence lengths, use LP to find dual sequences that flatten
the convolution, combined with gradient refinement on the smooth-max objective.
Key insight: Non-uniform (U-shaped) initializations break the symmetry that traps
uniform-optimal methods, and the LP naturally finds directions that reduce the peak.
"""

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time

def compute_conv(x):
    """Compute linear convolution of x with itself using FFT."""
    n = len(x)
    L = 2 * n - 1
    X = fft(x, L)
    return ifft(X * X).real

def softmax_safe(x):
    """Numerically stable softmax."""
    x = x - x.max()
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return ex / s

def get_score(x):
    """Compute the actual score 2n*max(conv)/sum^2."""
    n = len(x)
    s = x.sum()
    if s < 0.01:
        return np.inf
    c = compute_conv(x)
    return 2.0 * n * c.max() / (s * s)

def solve_lp(f, M, n):
    """Solve LP: max sum(b) s.t. convolve(f,b)[k] <= M for all k, b >= 0."""
    c_obj = -np.ones(n)
    n_cons = 2 * n - 1
    A = np.zeros((n_cons, n))
    # A[k, j] = f[j] if 0 <= k-j < n
    for j in range(n):
        k_start = j  # k-j >= 0 => k >= j
        k_end = j + n  # k-j < n => k < j+n
        k_start = max(0, k_start)
        k_end = min(n_cons, k_end)
        if k_start < k_end:
            A[k_start:k_end, j] = f[j]
    b_ub = np.full(n_cons, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

def lp_iteration(f, M, n, t_update):
    """One LP-based update iteration."""
    b = solve_lp(f, M, n)
    if b is None:
        return f, False
    b_sum = b.sum()
    if b_sum < 1e-10:
        return f, False
    f_sum = f.sum()
    g = b / b_sum * f_sum
    f_new = (1.0 - t_update) * f + t_update * g
    f_new = np.maximum(f_new, 0.0)
    f_new_sum = f_new.sum()
    if f_new_sum < 1e-10:
        return f, False
    f_new = f_new / f_new_sum * f_sum
    return f_new, True

def gradient_step(t_params, tau, n):
    """One gradient descent step in t-space. Returns new t and score."""
    x = softmax_safe(t_params)
    c = compute_conv(x)

    # Smooth max
    tc = tau * c
    tc_shift = tc - tc.max()
    exp_tc = np.exp(tc_shift)
    p = exp_tc / exp_tc.sum()
    t_smooth = c.max() + np.log(exp_tc.sum()) / tau

    # Gradient of smooth max w.r.t. x
    # grad_x[j] = 2 * sum_k p[k] * x[k-j]
    grad_x = np.zeros(n)
    for j in range(n):
        ks = max(0, j)
        ke = min(2 * n - 1, j + n)
        if ks < ke:
            grad_x[j] = 2.0 * np.dot(p[ks:ke], x[ks - j:ke - j])

    # Convert to t-space: grad_t = x * (grad_x - x . grad_x)
    xg = np.dot(x, grad_x)
    grad_t = x * (grad_x - xg)

    return grad_t, t_smooth, x

    inits = []

    x[::2] = 1.5
