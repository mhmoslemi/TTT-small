    """Find g_0 >= 0 such that f * g_0 <= M, maximizing sum(g_0)."""
    # Variables: g_0[0], ..., g_0[n-1]
    # Objective: maximize sum(g_0) → minimize -sum(g_0)
    c = -np.ones(n)
    
    # Constraints: (f * g_0)[k] <= M for k = 0, ..., 2n-2
    # (f * g_0)[k] = sum_i f[i] * g_0[k-i] (valid i)
    # This is a linear function of g_0.
    # The constraint matrix A has shape (2n-1) x n.
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # (f*g_0)[k] = sum_{j} f[j] * g_0[k-j] where 0<=j<n and 0<=k-j<n
        # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[j] * g_0[k-j]
        # Let i = k-j, so j = k-i:
        # = sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[k-i] * g_0[i]
        # So A[k, i] = f[k-i] for valid i.
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            A[k, i] = f[k - i]
    
    b = np.full(2*n-1, M)
    
    # Bounds: g_0[i] >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return result.x if result.success else None

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
