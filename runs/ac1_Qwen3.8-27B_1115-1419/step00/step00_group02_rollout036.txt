def _fft_convolve(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def _compute_subgradient(a, T):
    """Compute the subgradient of max_k conv[k] at a.
    
    g[j] = (2/|T|) * sum_{k in T, 0<=k-j<n} a[k-j]
    """
    n = len(a)
    g = np.zeros(n)
    for k in T:
        # a[k-j] is valid for j such that 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:

            # a[k-j] for j in [j_start, j_end] is a[k-j_start : k-j_end+1] reversed
            indices = np.arange(j_start, j_end + 1)
            g[indices] += a[k - indices]
    g *= 2.0 / len(T)
    return g

But wait, I need to double-check the formula. Let me re-derive:

`result = ifft(fft(a, N) * conj(fft(w, N)))`

In numpy's convention:
`ifft(X)[k] = (1/N) * sum_{m=0}^{N-1} X[m] * exp(2*pi*i*m*k/N)`

And `fft(a, N)[m] = sum_{n=0}^{N-1} a[n] * exp(-2*pi*i*n*m/N)` (with a padded to length N).

So `result[k] = (1/N) * sum_m [sum_n a[n] * exp(-2*pi*i*n*m/N)] * [conj(sum_p w[p] * exp(-2*pi*i*p*m/N))] * exp(2*pi*i*m*k/N)`

`= (1/N) * sum_m sum_n sum_p a[n] * conj(w[p]) * exp(-2*pi*i*nm/N) * exp(2*pi*i*pm/N) * exp(2*pi*i*mk/N)`

For real w: `conj(w[p]) = w[p]`.

`= (1/N) * sum_m sum_n sum_p a[n] * w[p] * exp(2*pi*i*m*(k + p - n)/N)`

The sum over m: `(1/N) * sum_m exp(2*pi*i*m*(k+p-n)/N) = 1` if `k+p ≡ n (mod N)`, else 0.

import numpy as np
import time


def _next_pow2(x):
    """Return smallest power of 2 >= x."""
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_autoconv(a):
    """Compute full auto-convolution a*a using FFT."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def _subgrad_direct(a, T):
    """Subgradient of max_k conv[k] via direct summation over argmax set T."""
    n = len(a)
    g = np.zeros(n)
    for k in T:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            idx = np.arange(js, je + 1)
            g[js:je + 1] += a[k - idx]
    g *= 2.0 / len(T)
    return g


def _subgrad_fft(a, T):
    """Subgradient of max_k conv[k] via FFT (for large |T|)."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[T] = 1.0
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fa * np.conj(fw), N)
    g = np.zeros(n)
    g[0] = result[0]
    if n > 1:
        g[1:] = result[N - 1:N - n:-1]
    g *= 2.0 / len(T)
    return g


def _local_score(a):
    """Fast local score computation (no evaluator call)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = _fft_autoconv(a)
    return 2.0 * n * np.max(conv) / (S * S)

    - FFT-based auto-convolution for O(n log n) per iteration

    - Random polishing to escape shallow local minima

    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 5.0

    # Determine sequence length
    n = 2048
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        n = len(height_sequence_1)

    best_score = np.inf

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if a0.shape == (n,) and np.sum(a0) >= 0.01:
            a0 = np.clip(a0, 0.0, 1000.0)
            starts.append(a0 / np.sum(a0))

    if n > 2:
        hanning = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
