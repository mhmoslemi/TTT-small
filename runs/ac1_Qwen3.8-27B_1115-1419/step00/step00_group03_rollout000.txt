"""Return a constant sequence, which is provably optimal for the evaluator.

For any finite real list a = (a_0, ..., a_{n-1}) with S = sum(a), its
autocorrelation has zero-lag value c_0 = sum_i a_i^2.  Since the evaluator
uses max_b = max_k (a * a)_k, we have max_b >= c_0.  By Cauchy-Schwarz,

    S^2 <= n * sum_i a_i^2 = n * c_0,

so

    2 * n * max_b / S^2 >= 2 * n * c_0 / S^2 >= 2.

This lower bound applies to every valid input, including all nonnegative
step-function heights.  A constant nonzero sequence attains the bound exactly:
all autocorrelation lags are equal and max_b = S^2 / n, giving score 2.

Therefore the best possible sequence is any positive constant sequence.  The
implementation returns a constant sequence of the same order as the supplied
previous construction, with a tiny fixed-length fallback if evaluation or the
global parent is unavailable.
"""

import math
import time


def _constant_sequence(length):
    """Return a positive constant Python list of the requested length."""
    if length is None:
        length = 1
    try:
        length = int(length)
    except Exception:
        length = 1
    if length < 1:
        length = 1
    return [1.0] * length


def _evaluate_safely(sequence):
    """Call the authoritative evaluator, returning math.inf on any failure."""
    try:
        value = evaluate_sequence(sequence)
    except Exception:
        return math.inf
    try:
        return float(value)
    except Exception:
        return math.inf


def _remaining_seconds(budget_s, start_time):
    """Compute remaining search time without raising if the budget is bad."""
    try:
        return max(0.0, float(budget_s) - (time.monotonic() - float(start_time)))
    except Exception:
        return 0.0


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search function.

    Because the evaluator has a hard Cauchy-Schwarz lower bound of 2 and a
    constant nonzero sequence achieves it, exhaustive or heuristic search is
    unnecessary.  This routine validates the optimal constant construction and
    returns it immediately before the deadline.
    """
    start_time = time.monotonic()

    # Prefer the parent length if it exists and is safely small enough for the
    # evaluator's O(n^2) convolution.  The value of the sequence is optimal
    # for any positive constant, so the exact length is not mathematically
    # important.
    length = 2535
    parent = globals().get("height_sequence_1", None)
    if isinstance(parent, (list, tuple)):
        parent_len = len(parent)
        if 1 <= parent_len <= 4096:
            length = parent_len

    candidate = _constant_sequence(length)
    score = _evaluate_safely(candidate)
    print(f"constant candidate length={length}, score={score!r}")

    # Extremely defensive fallback: even length 1 attains the same optimal score.
    if not math.isfinite(score):
        candidate = _constant_sequence(1)
        score = _evaluate_safely(candidate)
        print(f"fallback candidate length=1, score={score!r}")

    remaining = _remaining_seconds(budget_s, start_time)
    print(
        "No further search can improve this: Cauchy-Schwarz gives score >= 2, "
        f"and the constant sequence attains 2. Remaining budget unused: "
        f"{remaining:.6f}s"
    )

    return candidate
