def lp_step(f, M):
    """Solve LP to find g >= 0 with f*g <= M, maximizing sum(g)."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build convolution constraint matrix A such that A @ g = f * g
    # (f*g)[k] = sum_i f[i] * g[k-i]
    # For k = 0, ..., 2n-2:
    #   (f*g)[k] = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} f[i] * g[k-i]

    prob = cp.Problem(obj, constraints)
    prob.solve()

def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A such that A @ g = convolve(f, g)."""
    n = len(f)

    k = np.arange(2*n-1)
    j = np.arange(n)

        j_end = min(n-1, k_val)

def build_conv_matrix_sparse(f):
    """Build sparse matrix A such that A @ g = convolve(f, g)."""
    from scipy.sparse import csr_matrix
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            i = k - j
            if 0 <= i < n and f[i] != 0:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp_fast(f, M):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)<=M."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix
    # For moderate n, use dense; for large n, use sparse
    if n <= 2000:
        # Dense construction
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k-n+1)
            j_end = min(n-1, k)
            if j_start <= j_end:
                j_range = np.arange(j_start, j_end+1)
                i_range = k - j_range
                A[k, j_start:j_end+1] = f[i_range]
        constraints = [A @ g <= M]
    else:
        A = build_conv_matrix_sparse(f)
        constraints = [A @ g <= M]
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLPK]:

            break

    return np.maximum(0, g.value)
