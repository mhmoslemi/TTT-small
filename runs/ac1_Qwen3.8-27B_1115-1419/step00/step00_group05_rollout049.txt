    """Perform multiplicative (mirror) descent steps."""
    n = len(a)
    for _ in range(n_steps):
        c = _fft_convolve(a)
        S = np.sum(a)
        
        c_max = np.max(c)
        shifted = np.clip((c - c_max) / tau, -700, 700)
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp
        
        # v[m] = sum_j a[j] * w[m+j]
        v = _cross_correlate(a, w)
        
        # Load: proportional to v[m]
        mean_v = np.mean(v)
        if mean_v < 1e-15:
            break
        
        # Multiplicative update
        log_update = -eta * (v - mean_v) / mean_v  # centered to maintain sum
        a = a * np.exp(np.clip(log_update, -1, 1))
        a = np.maximum(a, 0.0)
        
        # Re-normalize
        S_new = np.sum(a)
        if S_new > 1e-10:
            a = a * (n / S_new)
    
    return a
