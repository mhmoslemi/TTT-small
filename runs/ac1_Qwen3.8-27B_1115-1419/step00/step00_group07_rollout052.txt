```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    return np.convolve(a, a)

def objective(a):
    conv = compute_conv(a)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * np.max(conv) / (s * s)

def lp_direction(a, M):
    """Find b >= 0 maximizing sum(b) s.t. conv(a,b)[k] <= M for all k."""
    n = len(a)
    # conv(a, b)[k] = sum_i a[i] * b[k-i] (with valid indices)
    # This is a linear function of b.
    # We can write it as A @ b <= M (vector of all M's)
    # where A is a (2n-1) x n matrix
    
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = a[j] if k-j is in [0, n-1], i.e., j in [k-n+1, k]
    
    # Actually, conv(a, b)[k] = sum_j a[b_index] * b[a_index] where a_index + b_index = k
    # = sum_{j: 0<=j<n, 0<=k-j<n} a[k-j] * b[j]
    
    # So conv(a,b)[k] = sum_j A[k,j] * b[j] where A[k,j] = a[k-j] if 0<=k-j<n else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    # Minimize -sum(b) s.t. A @ b <= M*1, b >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.x
    return None
```
