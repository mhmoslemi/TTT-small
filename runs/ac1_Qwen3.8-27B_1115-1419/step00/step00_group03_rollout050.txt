def smooth_max_and_weights(conv, T):
    """Compute smoothed max and softmax weights of conv."""
    m = conv.max()
    exp_vals = np.exp((conv - m) / T)
    sum_exp = exp_vals.sum()
    smooth_max = m + T * np.log(sum_exp)
    weights = exp_vals / sum_exp
    return smooth_max, weights

def compute_gradient(a, weights, M_smooth):
    """Compute gradient of J w.r.t. a using smoothed max."""
    n = len(a)
    S = a.sum()
    
    # dM/da[i] = 2 * sum_j a[j] * weights[i+j]
    grad_M = np.zeros(n)
    for j in range(n):
        if a[j] != 0:
            grad_M += a[j] * weights[j:j+n]
    grad_M *= 2
    
    # dJ/da[i] = 2n/S^2 * dM/da[i] - 4n*M/S^3
    grad_J = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M_smooth / (S * S * S))
    
    return grad_J

    n = len(a)

    # (with a[j] = 0 for j < 0 or j >= n)

    # and lowest contribution (destination)
    source = np.argmax(contributions)
    # For destination, we want a position with low contribution AND that doesn't
    # increase other conv values too much
    # Simple choice: position with lowest contribution
    destination = np.argmin(contributions)
    
    if contribution[source] <= contribution[destination]:
        return a
    
    # Move mass
    move = min(amount, a[source])
    a[source] -= move
    a[destination] += move
    
    return a

Let me also think about whether to include the LP approach as a secondary optimization. The LP gives an exact "best direction" in some sense, which might help when gradient-based methods get stuck.

For the LP:
- Variables: g[0], ..., g[n-1] (n variables)
- Objective: maximize sum(g)
- Constraints: 
  - (f * g)[k] <= M for k = 0, ..., 2n-2 (2n-1 constraints)
  - g[i] >= 0 for i = 0, ..., n-1 (n constraints, handled by variable bounds)

The constraint (f * g)[k] <= M can be written as:
sum_{j=0}^{n-1} f[j] * g[k-j] <= M (where g[m] = 0 for m < 0 or m >= n)

For a fixed k, this is:
sum_{j: max(0, k-n+1) <= j <= min(n-1, k)} f[j] * g[k-j] <= M

Let me substitute i = k-j:
sum_{i: max(0, k-n+1) <= k-i <= min(n-1, k)} f[k-i] * g[i] <= M
= sum_{i: k-min(n-1,k) <= i <= k-max(0,k-n+1)} f[k-i] * g[i] <= M
= sum_{i: max(0, k-n+1) <= i <= min(n-1, k)} f[k-i] * g[i] <= M

So the constraint for lag k is:
sum_{i=0}^{n-1} f[max(0, min(n-1, k-i))] * g[i] <= M (using the convention that f[out-of-range] = 0)

Or more precisely:
sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[k-i] * g[i] <= M

This is a linear constraint in g. The LP matrix is 2n-1 × n.

For n=256, this is a 511 × 256 LP, which is very manageable.

def solve_lp_direction(f, M):
    """Solve LP to find best direction g."""
    n = len(f)
    
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f * g)[k] <= M for k = 0, ..., 2n-2
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2*n - 1, n))
    b_ub = np.full(2*n - 1, M)
    
    for i in range(n):
        # g[i] appears in constraints k where max(0, k-n+1) <= i <= min(n-1, k)
        # i.e., k >= i and k <= n-1+i
        # For each such k, the coefficient of g[i] is f[k-i]
        for k in range(max(0, i), min(2*n-1, n+i)):
            # We need f[k-i] where 0 <= k-i < n
            A_ub[k, i] = f[k - i]
    
    # g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
