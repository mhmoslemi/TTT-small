Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses IWM (Iterative Watermark Minimization) in frequency domain combined

import numpy as np

def _fast_obj(a):
    """Compute objective = 2*n*max(convolve(a,a))/(sum(a))^2 using numpy only."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return 1e18
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)


def _iwm_run(a, n_iter=3000, beta=2.0, ratio=0.90, blend=0.5):
    """
    Run IWM optimization on a non-negative sequence.
    Returns the best sum-normalized sequence found.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)

        a = np.ones(n, dtype=np.float64)
        s = float(n)

    N = 1

    best_obj = _fast_obj(a)

    for it in range(n_iter):
        frac = it / max(1, n_iter)
        cur_ratio = ratio + 0.10 * (1.0 - frac)
        cur_beta = beta * (0.5 + 1.5 * frac)
        cur_blend = max(0.1, blend * (1.0 - 0.5 * frac))

        A = np.fft.fft(a, n=N)
        B = A * A
        r = np.fft.ifft(B).real[:2 * n - 1]

        r_max = np.max(r)
        if r_max < 1e-18:
            break
        target = cur_ratio * r_max

        excess = np.maximum(0.0, r - target)
        if np.max(excess) < 1e-18:

        r_new = r.copy()
        mask = excess > 0

            else:
                a_cand = a_new
        else:
            a_cand = a_new

        obj = _fast_obj(a_cand)
        if obj < best_obj:
            best_obj = obj
            best_a = a_cand.copy()

        a = a_cand

    return best_a

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 1e-10:

    return a
