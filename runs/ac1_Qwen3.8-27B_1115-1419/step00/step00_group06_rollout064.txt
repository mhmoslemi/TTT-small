"""
Gradient-based optimization of non-negative sequences to minimize
2*n*max(conv(f,f))/(sum(f))^2 using FFT-based convolution,
L2 and L-inf gradient descent, and multiple restarts.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1 << (n + m - 2).bit_length()
    A = fft(a, size)
    B = fft(b, size)
    return np.real(ifft(A * B, size)[:n + m - 1])


def self_convolve(f):
    """Compute f * f (linear) efficiently."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    A = fft(f, size)
    return np.real(ifft(A * A, size)[:2 * n - 1])


def compute_score(f):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return float('inf')
    conv = self_convolve(f)
    return float(2.0 * n * np.max(conv) / (s * s))


def l2_gradient(f):
    """
    Gradient of sum_k (conv(f,f)[k])^2 w.r.t. f.
    Computed as cross-correlation of g=conv(f,f) with f.
    Returns array of length n.
    """
    n = len(f)
    g = self_convolve(f)  # length 2n-1
    rf = f[::-1]  # length n
    # conv(g, rf) has length 3n-2; we need indices [n-1, 2n-2]
    size = 1 << (3 * n - 3).bit_length()
    G = fft(g, size)
    R = fft(rf, size)
    full = np.real(ifft(G * R, size))
    grad = full[n - 1:2 * n - 1]  # length n
    return grad


def linf_gradient(f):
    """
    Gradient targeting the argmax of conv(f,f).
    Returns (gradient_array, k_max).
    """
    n = len(f)
    conv = self_convolve(f)
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    i_lo = max(0, k_max - n + 1)
    i_hi = min(n, k_max + 1)
    idx = np.arange(i_lo, i_hi)
    grad[idx] = f[k_max - idx]
    return grad, k_max


def linf_gradient_topk(f, topk=3):
    """
    Gradient as convex combination of top-k peak gradients.
    Returns gradient array of length n.
    """
    n = len(f)
    conv = self_convolve(f)
    # Get top-k positions
    top_pos = np.argsort(conv)[-topk:]
    weights = conv[top_pos].astype(np.float64)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, top_pos):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        idx = np.arange(i_lo, i_hi)
        grad[idx] += w * f[k - idx]
    return grad, conv


def optimize_from(f0, budget_s, mode="auto", verbose=False):
    """
    Optimize sequence starting from f0.
    mode: 'l2', 'linf', or 'auto' (switch between them)
    Returns (best_f, best_score).
    """
    f = np.array(f0, dtype=np.float64).copy()
    f = np.clip(f, 0.0, None)
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n, dtype=np.float64) / n
    else:
        f = f / s

    best_f = f.copy()
    best_score = compute_score(f)

    start = time.monotonic()
    step = 0.01
    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.9
    stall = 0
    max_stall = 300
    use_l2 = (mode == "l2") or (mode == "auto")
    iteration = 0

    while time.monotonic() - start < budget_s:
        # Compute gradient
        if use_l2 or (mode == "auto" and use_l2):
            grad = l2_gradient(f)
        else:
            grad, _ = linf_gradient(f) if mode != "auto" else linf_gradient_topk(f, 3)

        # Center gradient (maintain sum constraint)
        grad_centered = grad - grad.mean()

        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad_centered

        # Proposed update
        f_new = f - step * momentum
        f_new = np.clip(f_new, 0.0, None)
        s_new = f_new.sum()
        if s_new < 1e-10:
            step *= 0.5
            continue
        f_new = f_new / s_new

        # Evaluate periodically (every 10 iters or on improvement check)
        if iteration % 5 == 0:
            score_new = compute_score(f_new)
            if score_new < best_score:
                best_f = f_new.copy()
                best_score = score_new
                f = f_new
                stall = 0
                step = min(step * 1.05, 0.5)
            else:
                stall += 1
                step *= 0.97
                if stall >= max_stall:
                    # Switch mode or restart
                    if mode == "auto" and use_l2:
                        use_l2 = False
                        step = 0.005
                        momentum = np.zeros(n)
                        stall = 0
                        if verbose:
                            print(f"    Switching to L-inf mode at iter {iteration}, score={best_score:.6f}")
                    elif mode == "auto" and not use_l2:
                        # Add perturbation and go back to L2
                        f = f + 0.01 * (np.random.default_rng(iteration).random(n) - 0.5)
                        f = np.clip(f, 0, None)
                        f = f / f.sum()
                        use_l2 = True
                        step = 0.01
                        momentum = np.zeros(n)
                        stall = 0
                        max_stall = 200
                        if verbose:
                            print(f"    Perturbing and restarting at iter {iteration}, score={best_score:.6f}")
                    else:
                        break
        else:
            f = f_new

        iteration += 1
        if verbose and iteration % 500 == 0:
            cur_score = compute_score(f)
            print(f"    iter={iteration}, best={best_score:.6f}, cur={cur_score:.6f}, step={step:.5f}")

    # Final check
    final_score = compute_score(best_f)
    if final_score < best_score:
        best_score = final_score

    return best_f, best_score


def make_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)

    # 1. Uniform
    inits.append(("uniform", np.ones(n)))

    # 2. Gaussian bump at center
    inits.append(("gauss_center", 1.0 + 0.8 * np.exp(-8 * (x - 0.5) ** 2)))

    # 3. Two bumps
    inits.append(("two_bumps", np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2) + 0.1))

    # 4. Cosine modulation
    inits.append(("cos1", np.clip(1.0 + 0.4 * np.cos(2 * np.pi * x), 0.01, None)))

    # 5. Higher frequency cosine
    inits.append(("cos3", np.clip(1.0 + 0.3 * np.cos(6 * np.pi * x), 0.01, None)))

    # 6. Random
    inits.append(("random1", rng.random(n) + 0.05))

    # 7. Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.5] = 0.0
    sparse[sparse > 0] += 0.1
    inits.append(("sparse", sparse))

    # 8. Linear ramp
    inits.append(("ramp", 0.3 + 0.7 * x))

    # 9. Sine wave
    inits.append(("sine", np.clip(1.0 + 0.3 * np.sin(4 * np.pi * x), 0.01, None)))

    # 10. Parabolic
    inits.append(("parab", 0.5 + 0.5 * (1 - 4 * (x - 0.5) ** 2) ** 2))

    return inits
