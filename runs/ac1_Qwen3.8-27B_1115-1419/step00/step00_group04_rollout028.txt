```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2. Scale-invariant."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def smooth_gradient_step(a, T, alpha):
    """
    One step of smoothed gradient descent.
    Minimizes max_k (a*a)(k) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    c = np.convolve(a, a)
    # Softmax
    c_max = np.max(c)
    exp_c = np.exp(T * (c - c_max))
    p = exp_c / np.sum(exp_c)
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
    a_flip = a[::-1]
    q = np.convolve(a_flip, p, mode='full')[n - 1:2 * n - 1]
    grad = 2.0 * q
    # Adaptive step size
    grad_max = np.max(np.abs(grad))
    if grad_max < 1e-15:
        return a
    eta = alpha / grad_max
    a_new = a - eta * grad
    return project_simplex(a_new)
