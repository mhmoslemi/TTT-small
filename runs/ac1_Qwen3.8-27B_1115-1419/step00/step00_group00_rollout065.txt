"""
LP-guided local search to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses the SOTA LP direction-finding method with line search, multiple
restarts, perturbations to escape local optima, and multi-length search.
Primary length n=256; extends to n=512, 1024 if time permits.
"""

import numpy as np
import time
import scipy.optimize as so


def _build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = (a*b)[k] for all k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns b array or None on failure."""
    n = len(a)
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    bnds = [(0.0, None)] * n
    cons = so.LinearConstraint(A, -np.inf, np.full(2 * n - 1, M))
    try:
        res = so.linprog(c, bounds=bnds, constraints=cons, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2n*max(convolve(a,a))."""
    conv = np.convolve(a, a)
    return 2.0 * len(a) * float(conv.max())


def _lp_step(a):
    """One LP iteration with line search.
    Input: a with sum=1, a>=0.
    Returns: (new_a, new_score, improved_flag)"""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(conv.max())
    curr_score = 2.0 * n * M

    b = _lp_solve(a, M)
    if b is None:
        return a, curr_score, False

    Sb = float(b.sum())
    if Sb <= 1.0 + 1e-9:
        return a, curr_score, False

    g = b / Sb  # g has sum=1, g >= 0

    best_a = a
    best_sc = curr_score
    improved = False

    for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01]:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-14:
            continue
        a_try /= s
        sc_try = _score_norm(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try
            improved = True

    return best_a, best_sc, improved


def _perturb(a, rng, strength=0.1):
    """Apply random perturbation to escape local optima."""
    n = len(a)
    a = a * np.exp(rng.normal(0.0, strength, n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-14:
        return np.ones(n) / n
    return a / s


def _make_initializations(n, rng, provided_seq):
    """Generate a list of initial sequences of length n (each summing to 1)."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # From provided sequence (if matches length)
    if provided_seq is not None and len(provided_seq) == n:
        a0 = np.clip(np.array(provided_seq, dtype=np.float64), 0, None)
        if a0.sum() > 0:
            inits.append(a0 / a0.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Random smooth
    for _ in range(4):
        r = rng.random(n)
        # Smooth it a bit
        kernel = np.ones(5) / 5.0
        r = np.convolve(r, kernel, mode='same')
        r = np.maximum(r, 0.01)
        inits.append(r / r.sum())

    # Pure random
    for _ in range(4):
        r = rng.random(n) + 0.001
        inits.append(r / r.sum())

    # Gaussian bumps (various widths)
    for sigma_frac in [0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-((x - n / 2) ** 2) / (2 * (sigma_frac * n) ** 2))
        inits.append(g / g.sum())

    # Two bumps
    for frac in [0.2, 0.3]:
        two = np.exp(-((x - n * 0.3) ** 2) / (2 * (frac * n) ** 2)) + \
              np.exp(-((x - n * 0.7) ** 2) / (2 * (frac * n) ** 2))
        inits.append(two / two.sum())

    # Cosine window
    cw = np.cos(np.linspace(0, np.pi, n)) ** 2
    cw = np.maximum(cw, 0.001)
    inits.append(cw / cw.sum())

    # Sine window
    sw = np.sin(np.linspace(0, np.pi, n)) ** 2
    sw = np.maximum(sw, 0.001)
    inits.append(sw / sw.sum())

    # Sparse random
    for density in [0.3, 0.5]:
        mask = rng.random(n) < density
        s = np.where(mask, 1.0, 0.0)
        if s.sum() > 1:
            inits.append(s / s.sum())

    # Alternating with noise
    alt = np.where(np.arange(n) % 2 == 0, 1.5, 0.5) + rng.normal(0, 0.1, n)
    alt = np.maximum(alt, 0.001)
    inits.append(alt / alt.sum())

    return inits


def _run_search(n, inits, rng, deadline):
    """Run LP search on sequences of length n. Returns (best_seq, best_score)."""
    best_score = float('inf')
    best_seq = None

    for init_idx, init in enumerate(inits):
        if time.monotonic() > deadline:
            break

        a = init.copy()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        stuck = 0
        max_stuck = 20
        it = 0

        while time.monotonic() < deadline:
            it += 1
            a, sc, improved = _lp_step(a)

            if improved:
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_seq = a.copy()
                    elapsed = time.monotonic()
                    print(f"  n={n} init={init_idx} it={it}: score={sc:.6f}")
                stuck = 0
            else:
                stuck += 1

            if stuck >= max_stuck:
                # Escalating perturbation
                strength = min(0.3, 0.05 + stuck * 0.005)
                a = _perturb(a, rng, strength)

                # Occasionally try aggressive restructuring
                if rng.random() < 0.25:
                    # Zero out a random fraction
                    frac = rng.uniform(0.1, 0.4)
                    mask = rng.random(n) >= frac
                    a = np.where(mask, a, 0.0)
                    s = a.sum()
                    if s < 1e-10:
                        a = np.ones(n) / n
                    else:
                        a = a / s

                # Refresh initializations occasionally
                if stuck > max_stuck * 3:
                    # Try a fresh random init
                    r = rng.random(n) + 0.01
                    a = r / r.sum()
                    stuck = 0
                    max_stuck = 20

            if it > 50000:
                break

    return best_seq, best_score


def propose_candidate(budget_s=2400, **kwargs):
