    # = ifft(fft(a, N) * conj(fft(w, N)))[:n] where N >= n + len(w) - 1
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    g = np.fft.irfft(A * np.conj(W), N)[:n]
    return g

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    deadline = time.monotonic() + budget_s - 5.0

        a0 = np.ones(2000)

    # Normalize
    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial score: {best_score:.6f}")
    
    # Softmax parameterization
    # a = softmax(x), so x = log(a) (up to a constant)
    x = np.log(a0 + 1e-30).copy()
    
    alpha = 5.0
    lr = 0.1
    total_time = 0.0
    
    it = 0
    while time.monotonic() < deadline:
        # Compute a = softmax(x)
        a = softmax(x)
        
        # Compute b = a * a
        b = fft_convolve_full(a, a)
        
        # Compute weights
        b_max = b.max()
        w = np.exp(alpha * (b - b_max))
        w_sum = w.sum()
        
        # If w_sum is too small (shouldn't happen with our normalization)
        if w_sum < 1e-30:
            break
        
        # Compute g[i] = sum_k w[k] a[k-i]
        g = cross_corr(w, a)
        
        # Compute b_bar_w = sum_k w[k] b[k] / w_sum
        b_bar_w = np.sum(w * b) / w_sum

        grad_x = a / w_sum * (g - b_bar_w)
