```python
"""
Optimal non-negative sequence for the stated evaluator.

For any non-negative vector a of length n, the full linear autocorrelation
coefficient at lag 0 is
    C_0 = sum_i a_i^2.
This coefficient is always at least every other autocorrelation coefficient,
and by Cauchy-Schwarz
    C_0 >= (sum_i a_i)^2 / n.
Therefore
    evaluate_sequence(a) = 2*n*max_k C_k / (sum_i a_i)^2 >= 2.
Equality is achieved exactly by a constant vector. Thus no finite search can
find a valid non-negative vector with a score below 2 for the evaluator given
in the prompt. This function returns a constant vector, preserving a sensible
length when available, and performs only a tiny bounded finite-precision
scale check if time is available.
"""

import numpy as np
import time

_DEFAULT_LENGTH = 5582
_MAX_EVAL_LENGTH = 6000
_MIN_SEARCH_TIME_FOR_EXACT = 0.25
_MAX_SCALE_CALLS = 8


def _base_length():
    """Return a reasonable sequence length, preferring height_sequence_1."""
    seq = globals().get("height_sequence_1", None)
    m = 0
    try:
        if isinstance(seq, np.ndarray):
            m = int(seq.size)
        elif hasattr(seq, "__len__"):
            m = int(len(seq))
    except Exception:
        m = 0

    if 1 <= m <= _MAX_EVAL_LENGTH:
        return m
    return _DEFAULT_LENGTH


def _safe_score(sequence):
    """Call the externally provided evaluator without raising exceptions."""
    try:
        value = evaluate_sequence(sequence)
    except Exception:
        return float("inf")
    try:
        value = float(value)
    except Exception:
        return float("inf")
    if not np.isfinite(value):
        return float("inf")
    return value


def _uniform_sequence(length, value=1.0):
    """Return a valid positive constant sequence."""
    try:
        m = int(length)
    except Exception:
        m = 1
    if m < 1:
        m = 1

    try:
        v = float(value)
    except Exception:
        v = 1.0
    if not np.isfinite(v) or v <= 0.0:
        v = 1.0
    if v > 1000.0:
        v = 1000.0

    return [v] * m


def _scale_sequence(sequence, factor):
    """Scale a sequence, keeping elements finite and bounded."""
    try:
        f = float(factor)
    except Exception:
        f = 1.0
    if not np.isfinite(f) or f <= 0.0:
        f = 1.0
    if f > 1000.0:
        f = 1000.0

    out = []
    for x in sequence:
        try:
            y = float(x)
        except Exception:
            y = 1.0
        if not np.isfinite(y):
            y = 1.0
        if y < 0.0:
            y = 0.0
        elif y > 1000.0:
            y = 1000.0
        out.append(y * f)
    return out


def _sanitize_sequence(sequence):
    """Ensure returned values are plain floats in [0, 1000] with nonzero sum."""
    out = []
    positive = False
    for x in sequence:
        try:
            y = float(x)
        except Exception:
            y = 1.0
        if not np.isfinite(y):
            y = 1.0
        if y < 0.0:
            y = 0.0
        elif y > 1000.0:
            y = 1000.0
        if y > 0.0:
            positive = True
        out.append(y)

    if not positive:
        return [1.0] * max(1, len(out))
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Return the best valid non-negative sequence found within the time budget.

    Under the evaluator shown in the prompt, a constant vector is the global
    minimizer for every length. The bounded scale checks are only a cautious
    finite-precision safeguard and cannot produce a meaningful improvement for
    the stated non-negative objective.
    """
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 0.0:
        budget = 2400.0

    search_time = max(0.0, budget - 5.0)
    deadline = t0 + search_time

    n = _base_length()
    best_sequence = _uniform_sequence(n, 1.0)
    best_score = float("inf")

    if n <= _MAX_EVAL_LENGTH and search_time > _MIN_SEARCH_TIME_FOR_EXACT:
        best_score = _safe_score(best_sequence)
        print(f"constant length={n} score={best_score}")

        if np.isfinite(best_score) and search_time > 0.75:
            scales = [
                float(np.nextafter(1.0, 0.0)),
                float(np.nextafter(1.0, 2.0)),
                0.5,
                2.0,
                0.25,
                4.0,
                0.1,
                10.0,
                0.01,
                100.0,
            ]

            rng = None
            try:
                rng = np.random.default_rng(seed)
            except Exception:
                rng = None

            if rng is not None:
                for _ in range(4):
                    try:
                        scales.append(float(np.exp(rng.normal(0.0, 1.0))))
                    except Exception:
                        pass

            valid_scales = []
            seen = set()
            for s in scales:
                try:
                    sf = float(s)
                except Exception:
                    continue
                if not np.isfinite(sf) or sf <= 0.0 or sf > 1000.0:
                    continue
                key = repr(sf)
                if key not in seen:
                    seen.add(key)
                    valid_scales.append(sf)

            calls = 1
            for sf in valid_scales:
                if time.monotonic() > deadline - 0.25:
                    break
                if calls >= _MAX_SCALE_CALLS:
                    break
                calls += 1

                candidate = _scale_sequence(best_sequence, sf)
                cand_score = _safe_score(candidate)
                if cand_score < best_score:
                    best_score = cand_score
                    best_sequence = candidate
                    print(f"scale={sf!r} score={cand_score}")
    else:
        print("length/time limited: returning provably optimal constant vector")

    final_sequence = _sanitize_sequence(best_sequence)
    print(f"final length={len(final_sequence)} best_known_score={best_score}")
    return final_sequence
```
