```python
"""
Minimize  2*n*max(convolve(a,a)) / (sum(a))^2  for a non‑negative step function.

Strategy
--------
The objective is scale invariant, so each iterate is normalised to sum == 1 and
we minimise  max(convolve(a,a)).  The earlier SOTA (alternating LP minimisation)
gets stuck at the uniform sequence (value 2.0) because a *uniform* parent makes
the LP's optimiser return the same uniform vector.  The fix used here is to
start from a family of strongly non‑uniform parents (geometric ramps, linear
ramps, structured randoms, quadratic‑residue difference‑set templates) and then
run the exact alternating‑LP "flatten the convolution" iteration:

    given parent a and M = max(conv(a,a)), solve
        max  sum(g)   s.t.  g >= 0, conv(a,g) <= M        (linear program in g)
    take  a <- (1-t)a + t*g  (re‑normalised).

Each LP is solved with a sparse cvxpy model (2n-1 x n convex matrix).  After the
LP iteration stalls we polish with exact‑evaluated local moves (block
multiplicative perturbations plus a peak‑lag mass‑shaving move that directly
attacks the largest convolution coefficient).  Many (length, init) combinations
are tried in parallel over the wall‑clock budget and the globally best
exact‑evaluated sequence is returned.
"""

import time
import numpy as np
import scipy.sparse as sp

try:
    import cvxpy as cp
    _HAVE_CVXPY = True
except Exception:                      # pragma: no cover - cvxpy expected present
    _HAVE_CVXPY = False


# --------------------------------------------------------------------------- #
# low level helpers (top level, no closures / lambdas)                         #
# --------------------------------------------------------------------------- #
def _convolve(a):
    """Full linear convolution of a with itself."""
    return np.convolve(a, a)


def _conv_max(a):
    return float(np.max(_convolve(a)))


def _build_conv_matrix(a):
    """Sparse (2n-1, n) matrix T with  T @ g == convolve(a, g)."""
    n = a.shape[0]
    ks = np.arange(0, 2 * n - 1)
    lo = np.maximum(0, ks - n + 1)
    hi = np.minimum(n - 1, ks)
    lengths = hi - lo + 1
    total = int(lengths.sum())
    rows = np.empty(total, dtype=np.int64)
    cols = np.empty(total, dtype=np.int64)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        i0 = lo[k]
        i1 = hi[k] + 1
        cnt = i1 - i0
        if cnt <= 0:
            continue
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(i0, i1)
        vals[idx:idx + cnt] = a[i0:i1]
        idx += cnt
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(a, M):
    """Solve  max sum(g)  s.t. g>=0, conv(a,g)<=M ; return g or None."""
    if not _HAVE_CVXPY:
        return None
    n = a.shape[0]
    T = _build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in ("CLARABEL", "ECOS", "OSQP", "SCS"):
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                gv = np.asarray(g.value, dtype=np.float64)
                if np.all(np.isfinite(gv)) and gv.sum() > 0.0:
                    return np.clip(gv, 0.0, None)
        except Exception:
            continue
    return None


def _obj(seq):
    """Authoritative objective via the supplied evaluator."""
    v = float(evaluate_sequence(seq))
    return v


def _eval_vec(a):
    return _obj([float(x) for x in a])

def _init_geometric(n, ratio):
    a = np.power(ratio, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _init_ramp(n, power):
    a = np.power(np.arange(1, n + 1, dtype=np.float64), power)
    return a / a.sum()


def _init_symmetric_ramp(n):
    a = np.power(np.arange(1, n + 1, dtype=np.float64), 1.0)
    b = a[::-1].copy()
    a = a + b
    return a / a.sum()


def _init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def _init_quadratic_residue(n):
    """Embed a quadratic‑residue (difference) set template of size ~ n."""
    # find prime p ~= n with p % 4 == 3
    def _is_prime(x):
        if x < 2:
            return False
        i = 2
        while i * i <= x:
            if x % i == 0:
                return False
            i += 1
        return True
    p = n
    guard = 0
    while not (_is_prime(p) and p % 4 == 3) and guard < 4000:
        p += 3 if (p % 4 == 1) else 2
        guard += 1
    res = set()
    for x in range(1, p):
        r = (x * x) % p
        if r != 0:
            res.add(r)
    a = np.zeros(p, dtype=np.float64)
    for r in res:
        a[r % p] = 1.0
    a = a[:n]
    if a.sum() <= 0:
        a = np.ones(n, dtype=np.float64)
    return a / a.sum()

def _lp_refine(a, budget, rng):
    """Run the alternating‑LP flattening iteration for at most ``budget`` secs."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    if a.sum() <= 0.01:
        return a, np.inf
    a = a / a.sum()
    best = a.copy()
    best_score = _eval_vec(a)
    M = _conv_max(a)
    it = 0
    deadline = time.monotonic() + budget
    stall = 0
    while time.monotonic() < deadline:
        g = _lp_step(a, M)
        if g is None:
            break
        g = g / g.sum()
        t = max(0.10, 0.5 / (1.0 + 0.05 * it))
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() <= 0.01:
            break
        a_new = a_new / a_new.sum()
        M_new = _conv_max(a_new)
        score = _eval_vec(a_new)
        if score < best_score:
            best = a_new.copy()
            best_score = score
        a = a_new
        M = M_new
        it += 1
        if abs(M_new - M) if False else abs(M_new - (best_score * (best.shape[0]))):
            pass
        # convergence test on M
        if abs(M_new - M) < 1e-7 or M_new >= M - 1e-9:
            stall += 1
        else:
            stall = 0
        if stall >= 8:
            break
        if it % 20 == 0:
            print("   lp it=%d obj=%.5f M=%.5f" % (it, best_score, M))
    return best, best_score

def _peak_shave(a, rng):
    """Move a little mass out of the largest product contributing to the peak lag."""
    n = a.shape[0]
    c = _convolve(a)
    k = int(np.argmax(c))
    i0 = max(0, k - n + 1)
    i1 = min(n - 1, k) + 1
    if i1 <= i0:
        return a
    a = a.copy()
    # choose the i inside the peak window that maximises a[i]*a[k-i]
    ii = np.arange(i0, i1)
    prods = a[ii] * a[k - ii]
    i = int(ii[int(np.argmax(prods))])
    j = k - i
    if not (0 <= j < n):
        return a
    src = i if a[i] >= a[j] else j
    other = j if src == i else i
    delta = 0.02 * a[src]
    a[src] -= delta
    a[other] += 0.5 * delta
    a[int(rng.integers(0, n))] += 0.5 * delta
    a = np.clip(a, 0.0, None)
    if a.sum() <= 0.01:
        return a
    return a / a.sum()


def _local_polish(a, budget, rng):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.sum() <= 0.01:
        return a, np.inf
    a = a / a.sum()
    best_score = _eval_vec(a)
    deadline = time.monotonic() + budget
    n = a.shape[0]
    it = 0
    while time.monotonic() < deadline:
        cand = a.copy()
        r = rng.random()
        if r < 0.35:
            w = max(1, int(a.size // 60))
            s = int(rng.integers(0, max(1, n - w)))
            cand[s:s + w] *= np.exp(rng.normal(0.0, 0.10, w))
        elif r < 0.70:
            # two‑block exchange to move mass
            w1 = max(1, int(a.size // 40))
            w2 = max(1, int(a.size // 40))
            s1 = int(rng.integers(0, max(1, n - w1)))
            s2 = int(rng.integers(0, max(1, n - w2)))
            m = 0.10 * cand[s1:s1 + w1].sum()
            cand[s1:s1 + w1] -= m / w1
            cand[s2:s2 + w2] += m / w2
            cand = np.clip(cand, 0.0, None)
        else:
            cand = _peak_shave(a, rng)
        if cand.sum() <= 0.01:
            continue
        cand = cand / cand.sum()
        sc = _eval_vec(cand)
        if np.isfinite(sc) and sc < best_score:
            a = cand
            best_score = sc
        it += 1
        if it % 50 == 0:
            print("     local it=%d obj=%.5f" % (it, best_score))
    return a, best_score
```
