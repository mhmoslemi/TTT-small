import numpy as np
import time
import sys

def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def _fft_correlation(p, a):
    """Compute corr[m] = sum_j a[j] * p[m+j] for m = 0..n-1.
    Using the identity: corr[m] = (a * flip(p))[2n-2-m] where * is linear convolution.
    """
    n = len(a)
    p_reversed = p[::-1]
    conv_result = _fft_convolve(a, p_reversed)
    # conv_result[k] = sum_j a[j] * p_reversed[k-j]
    # We want corr[m] = conv_result[2n-2-m] for m = 0..n-1
    return conv_result[n-1:2*n-1][::-1]

def _score_from_conv(C, n, S):
    """Compute the score from the convolution values."""
    return 2.0 * n * np.max(C) / (S * S)

def _smooth_objective_and_grad(a, tau):
    """Compute smoothed objective and gradient.
    a is normalized to sum=1.
    Returns (smooth_score, grad) where grad is the gradient of 2n*F with F = logsumexp(tau*C)/tau.
    """
    n = len(a)
    C = _fft_convolve(a, a)
    # Softmax weights
    C_max = C.max()
    exp_C = np.exp(tau * (C - C_max))
    Z = exp_C.sum()
    p = exp_C / Z
    F = (C_max + np.log(Z)) / tau  # = logsumexp(tau*C)/tau
    score = 2.0 * n * F  # since S=1
    # Gradient of F: dF/d a[m] = (1/tau) * sum_k p[k] * 2*a[k-m] = (2/tau) * corr[m]
    corr = _fft_correlation(p, a)
    # dF/d a[m] = (2/tau) * corr[m]
    # df/d a[m] = 2n * (2/tau) * corr[m] (since S=1 and dS/da = 1 but we're on the constraint)
    # Actually, if we don't fix S: df/da[m] = 2n*(S*dF/da[m] - 2*F*a[m])/S^3
    # With S=1: = 2n*(dF/da[m] - 2F*a[m])
    # dF/da[m] = (2/tau)*corr[m]
    grad = 2.0 * n * ((2.0/tau) * corr - 2.0 * F * a)
    return score, grad, C, p

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.full_like(v, target_sum / len(v))
    return v * (target_sum / s)

def _initialize_chirp(n):
    """Quadratic phase (chirp) initialization, non-negative."""
    i = np.arange(n)
    a = 1.0 + np.cos(np.pi * i * i / n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a

def _initialize_random(n, rng):
    """Random non-negative initialization."""
    a = rng.random(n)
    a /= a.sum()
    return a

def _initialize_two_tone(n, rng):
    """Two-tone: high on one side, low on other, with smooth transition."""
    i = np.arange(n)
    center = n * (0.3 + 0.4 * rng.random())
    width = n * (0.1 + 0.3 * rng.random())
    a = 0.1 + 0.9 / (1.0 + np.exp(-(i - center) / width))
    a /= a.sum()
    return a

def _run_smoothed_gradient_descent(a0, tau_init, tau_min, max_iters, deadline, rng, verbose=False):
    """Run smoothed gradient descent with adaptive tau."""
    a = a0.copy()
    n = len(a)
    # Normalize
    a /= a.sum()
    
    best_score = np.inf
    best_a = a.copy()
    
    tau = tau_init
    lr = 0.1
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        score, grad, C, p = _smooth_objective_and_grad(a, tau)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive tau: decrease for sharper max approximation
        if no_improve > 20:
            tau = max(tau_min, tau * 0.9)
            lr = max(0.001, lr * 0.95)
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.full(n, 1.0/n)
        else:
            a_new = a_new / s
        
        a = a_new
        
        # Periodic random perturbation (simulated annealing style)
        if verbose and it % 1000 == 0:
            print(f"  Iter {it}: smooth_score={score:.6f}, tau={tau:.4f}, lr={lr:.4f}, no_improve={no_improve}")
    
    return best_a, best_score

Minimizes 2n*max(autocorrelation)/sum^2 using smoothed gradient descent

import numpy as np
import time


def _fft_convolve(a, b):
    """Linear convolution of two 1D arrays via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _compute_conv_self(a):
    """Compute a*a (linear self-convolution) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _compute_corr(p, a):
    """Compute corr[m] = sum_j a[j] * p[m+j] for m=0..n-1 using FFT.
    Equivalent to (a * flip(p))[2n-2-m].
    """
    n = len(a)
    p_rev = p[::-1]
    conv = _fft_convolve(a, p_rev)
    # corr[m] = conv[2n-2-m] for m=0..n-1
    return conv[n - 1:2 * n - 1][::-1]


def _smoothed_gradstep(a, tau, lr):
    """One gradient step on the smoothed objective.
    a is normalized to sum=1. Returns new a and the score.
    """
    n = len(a)
    C = _compute_conv_self(a)
    
    # Log-sum-exp for numerical stability
    C_max = C.max()
    exp_C = np.exp(tau * (C - C_max))
    Z = exp_C.sum()
    p = exp_C / Z
    F = (C_max + np.log(Z)) / tau  # smoothed max
    
    if F <= 0:

    score = 2.0 * n * F  # S=1 so score = 2n*F

    corr = _compute_corr(p, a)

- FFT-based smoothed gradient descent (logsumexp approximation of max)
- Multiple diverse initializations (chirp, ramp, random, antisymmetric)
- Adaptive temperature and step size
- Simulated annealing for escaping local minima
- Periodic exact evaluation via evaluate_sequence
"""

import numpy as np
import time


def _fft_conv(self_or_a, b=None):
    """Linear convolution via FFT. If b is None, computes self-convolution."""
    if b is None:
        b = self_or_a
    a = self_or_a
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _self_conv(a):
    """Self-convolution a*a via FFT (slightly optimized)."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _cross_corr(p, a):
    """Compute corr[m] = sum_j a[j]*p[m+j] for m=0..n-1 via FFT."""
    n = len(a)
    # corr[m] = (a conv flip(p))[2n-2-m]
    conv = _fft_conv(a, p[::-1])
    return conv[n - 1:2 * n - 1][::-1]
