Smooth minimax gradient descent with FFT for minimizing peak linear autocorrelation.
"""

import numpy as np
import time


def _real_convolve_fft(x, y):
    """Fast real linear convolution via FFT."""
    n = len(x) + len(y) - 1
    L = 1 << (n - 1).bit_length()
    result = np.fft.irfft(np.fft.rfft(x, L) * np.fft.rfft(y, L), L)
    return result[:n]


def _hankel_matvec(w, x):
    """Compute D@x where D[i,j]=w[i+j] using FFT (cross-correlation)."""
    n = len(x)
    L = 1 << (len(w) + n - 1).bit_length()
    W = np.fft.rfft(w, L)
    X = np.fft.rfft(x, L)
    return np.fft.irfft(W * np.conj(X), L)[:n]


def _gradient_step(x, tau, L):
    """One gradient step for minimizing smoothed max of conv(x,x)."""
    c = _real_convolve_fft(x, x)

    return Dx / Z, c_max

    print(f"Initial: score={best_score:.6f}, n={len(best)}")
    
    # Define starting configurations
    configs = []
    
    # From given sequence
    if len(parent) >= 100:
        configs.append(("given", parent.copy()))
    
    # Uniform
    for n in [1024, 2048, 4096]:
        configs.append((f"unif_{n}", np.ones(n, dtype=np.float64)))
    
    # Raised cosine
    for n in [1024, 2048]:
        t = np.arange(n) / n
        x = (0.5 * (1 + np.cos(2 * np.pi * t))) ** 2
        x = np.maximum(x, 0.01)
        configs.append((f"cos_{n}", x))
    
    # Sine
    for n in [1024, 2048]:
        t = np.arange(n) / n
        x = np.maximum(0.5 + 0.5 * np.sin(2 * np.pi * 3 * t), 0.01)
        configs.append((f"sin_{n}", x))
    
    # Random
    for n in [1024, 2048]:
        x = np.abs(rng.normal(1.0, 0.5, n))
        configs.append((f"rand_{n}", x))
    
    best_tau = 50.0
    
    for idx, (name, x0) in enumerate(configs):
        if time.monotonic() > deadline - 120:
            break
        
        x = x0.copy().astype(np.float64)
        x = np.clip(x, 0.0, None)
        if x.sum() < 0.01:
            continue
        x = x / x.sum()
        
        n = len(x)
        L = 1 << (4 * n).bit_length()
        
        # Time budget for this config
        time_left = deadline - time.monotonic()
        time_for_config = min(300.0, time_left / max(1, len(configs) - idx))
        config_deadline = time.monotonic() + time_for_config
        
        tau = 5.0
        lr = 0.3
        n_iters = 5000
        best_local_score = float('inf')
        best_local_x = x.copy()
        stagnant = 0
        
        for it in range(n_iters):
            if time.monotonic() > config_deadline:
                break
            
            grad_dir, c_max = _gradient_step(x, tau, L)
            
            # Adaptive learning rate
            x_new = x - lr * grad_dir
            x_new = np.maximum(x_new, 0.0)
            s = x_new.sum()
            if s < 1e-10:
                x_new = x.copy()
            else:
                x_new = x_new / s
            
            # Check improvement
            c_new = _real_convolve_fft(x_new, x_new)
            new_cmax = c_new.max()
            
            if new_cmax < c_max:
                x = x_new
                stagnant = 0
                # Update local best (using fast score)
                fast_score = 2.0 * n * new_cmax  # since sum=1
                if fast_score < best_local_score:
                    best_local_score = fast_score
                    best_local_x = x.copy()
            else:
                stagnant += 1
                if stagnant > 100:
                    # Reduce learning rate or restart
                    lr *= 0.5
                    stagnant = 0
                    if lr < 1e-6:
                        break
            
            # Anneal tau
            if it % 200 == 0 and tau < 200:
                tau *= 1.1
            
            # Periodic exact evaluation
            if it % 500 == 0:
                score = float(evaluate_sequence(x.tolist()))
                if score < best_score:
                    best = x.copy()
                    best_score = score
                    best_tau = tau
                    print(f"  [{name}] iter {it}: score={score:.6f}, tau={tau:.1f}")
        
        # Final eval of this config
        score = float(evaluate_sequence(best_local_x.tolist()))
        if score < best_score:
            best = best_local_x.copy()
            best_score = score
            print(f"  [{name}] final: score={score:.6f}")
    
    # Refinement phase
    if time.monotonic() < deadline - 30:
        x = best.copy()
        x = x / x.sum()
        n = len(x)
        L = 1 << (4 * n).bit_length()
        
        tau = best_tau if best_tau > 10 else 50.0
        lr = 0.05
        refine_deadline = deadline - 5
        last_best = best_score
        stagnant = 0
        
        while time.monotonic() < refine_deadline:
            grad_dir, c_max = _gradient_step(x, tau, L)
            x_new = x - lr * grad_dir
            x_new = np.maximum(x_new, 0.0)
            s = x_new.sum()
            if s < 1e-10:
                break
            x_new = x_new / s
            
            c_new = _real_convolve_fft(x_new, x_new)
            if c_new.max() < c_max:
                x = x_new
                stagnant = 0
            else:
                stagnant += 1
                lr *= 0.95
                if lr < 1e-8:
                    break
            
            if stagnant % 200 == 0 and tau < 500:
                tau *= 1.05
            
            if stagnant > 2000:
                break
            
            # Periodic evaluation
            if stagnant % 200 == 0 or c_new.max() < 1/(2*n) * 1.05:
                score = float(evaluate_sequence(x.tolist()))
                if score < best_score:
                    best = x.copy()
                    best_score = score
