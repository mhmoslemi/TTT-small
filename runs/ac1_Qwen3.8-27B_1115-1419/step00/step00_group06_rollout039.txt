import numpy as np

def compute_full_conv(a, fft_size):
    """Compute full convolution of a with itself using FFT."""
    X = np.fft.rfft(a, fft_size)
    return np.fft.irfft(X * X)


def compute_subgradient(a, k_star, n):
    """Compute subgradient of max autocorrelation at lag k_star w.r.t. a."""
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)

    # Part 1: j in [lo, hi], contribution a[k_star - j]
    j1 = np.arange(lo, hi + 1)
    idx1 = k_star - j1
    valid1 = (idx1 >= 0) & (idx1 < n)
    g[j1[valid1]] += a[idx1[valid1]]

    # Part 2: k_star - j in [lo, hi], i.e., j in [k_star-hi, k_star-lo]
    j2_lo = max(0, k_star - hi)
    j2_hi = min(n - 1, k_star - lo)
    if j2_lo <= j2_hi:
        j2 = np.arange(j2_lo, j2_hi + 1)
        idx2 = k_star - j2
        valid2 = (idx2 >= 0) & (idx2 < n)
        g[j2[valid2]] += a[idx2[valid2]]

    return g


def normalize_symmetric(a):
    """Project to nonnegative, sum=1, symmetric."""
    a = np.maximum(a, 0.0)
    a = 0.5 * (a + a[::-1])
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
