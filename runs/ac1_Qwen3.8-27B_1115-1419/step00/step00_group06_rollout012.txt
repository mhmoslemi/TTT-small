import numpy as np
import time


"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using projected subgradient descent
with soft-max smoothing, chirp-based initializations (flat-spectrum sequences),
and FFT-accelerated gradient computation. Multiple restarts escape local minima.
"""


def fast_convolve(a):
    """Compute np.convolve(a, a) using real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_gradient(a, w):
    """
    Compute grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m] for j=0..n-1.
    w has length 2n-1 (same as the convolution output).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    a_flip = a[::-1]
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a_flip, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    return 2.0 * conv[n - 1: 2 * n - 1]


def soft_max_weights(b, tau):
    """Compute softmax weights w_k = exp((b_k - max)/tau) / sum(...)."""
    bmax = b.max()
    shifted = np.clip((b - bmax) / tau, -40.0, 0.0)
    exp_b = np.exp(shifted)
    s = exp_b.sum()
    if s < 1e-300:
        # Fallback: uniform on the max indices
        idx = np.where(b == bmax)[0]
        w = np.zeros_like(b)
        w[idx] = 1.0 / len(idx)
        return w
    return exp_b / s


def optimize_sequence(a_init, deadline, seed=0):
    """
    Run projected soft-max subgradient descent to minimize max(conv(a,a)).
    Normalizes to sum(a) = len(a). Returns (best_a, best_score).
    score = 2 * max_b / n (equivalent to 2n*max_b/S^2 with S=n).
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a *= n / s

    b = fast_convolve(a)
    best_score = 2.0 * b.max() / n
    best_a = a.copy()

    v = np.zeros(n)
    lr = 0.5
    mu = 0.7
    step = 0
    stall = 0
    lr_decay_rate = 0.9995
    tau_init = n * 0.05
    tau_min = 1.0

    while time.monotonic() < deadline:
        step += 1

        # Schedule
        tau = max(tau_min, tau_init * np.exp(-step / 200.0))
        lr = max(0.01, lr * lr_decay_rate)
        mu = min(0.92, 0.7 + 0.22 * step / 3000.0)

        b = fast_convolve(a)
        bmax = b.max()
        score = 2.0 * bmax / n

        if score < best_score * (1.0 - 1e-8):
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            # Add random perturbation to escape
            a = best_a.copy()
            a += 0.05 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a *= n / a.sum()
            v *= 0.1
            stall = 0

        # Compute soft-max gradient
        w = soft_max_weights(b, tau)
        grad = fft_gradient(a, w)

        # Momentum update
        v = mu * v + (1.0 - mu) * grad
        a_new = a - lr * v

        # Project: non-negative and renormalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            v *= 0.1
        else:
            a = a_new * (n / s_new)

    return best_a, best_score


def optimize_sequence_v2(a_init, deadline, seed=0):

    v = np.zeros(n)
    step = 0
    stall = 0
    top_k = 10

    while time.monotonic() < deadline:
        step += 1

        lr = 0.3 / (1.0 + step / 1000.0)
        mu = 0.8

        b = fast_convolve(a)
        bmax = b.max()
        score = 2.0 * bmax / n

        if score < best_score * (1.0 - 1e-8):
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            a = best_a.copy()
            a += 0.1 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a *= n / a.sum()
            v *= 0.1
            stall = 0

        # Direct gradient using top-K peaks
        if top_k < len(b):
            idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            idx = np.arange(len(b))

        grad = np.zeros(n)
        for k in idx:
            if b[k] < bmax * 0.99:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            j_range = np.arange(lo, hi + 1)
            a_idx = k - j_range
            grad[j_range] += 2.0 * a[a_idx]
        grad /= max(1, len(idx))

        v = mu * v + (1.0 - mu) * grad
        a = a - lr * v
        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            v *= 0.1
        else:
            a *= n / s_new

    return best_a, best_score
