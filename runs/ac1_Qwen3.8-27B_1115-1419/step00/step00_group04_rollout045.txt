def _compute_score(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    a_clip = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a_clip))
    if s < 0.01:
        return np.inf
    c = np.convolve(a_clip, a_clip)
    return float(2.0 * n * np.max(c) / (s * s))


def _build_conv_matrix(a):
    """Build (2n-1) x n matrix where row k is the coefficients of conv(a,.)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


def _solve_direction_lp(a, M, B=10.0):
    """Solve LP: max sum(b) s.t. A@a b <= M, 0 <= b <= B. Returns b* or None."""
    n = len(a)
    a_smooth = np.maximum(a, 1e-12)
    A = _build_conv_matrix(a_smooth)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, B)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
        # Try relaxed
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M * 1.05), bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

    c_shifted = alpha * (c - np.max(c))
    exp_c = np.exp(c_shifted)
    weights = exp_c / np.sum(exp_c)
    grad = np.correlate(weights, a, mode='valid')
    max_grad = np.max(grad)
    if max_grad < 1e-15:
        return a, False

        return a, False

    old_score = _compute_score(a)
    if new_score < old_score:
        return a_new, True
    return a, False
