The LP method:

2. Compute $M = \max(f*f)$.
3. Solve LP: max $\sum g_i$ s.t. $(f*g)_k \leq M$ for all $k$, $g_i \geq 0$.

import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    # Build constraint matrix A: (2n-1) x n
    # A[k, i] = f[k-i] if 0 <= k-i <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for i in range(n):
        for k in range(max(0, -i), min(2*n-1, n + i)):
            # k-i is in [0, n-1]
            j = k - i
            if 0 <= j <= n-1:
                A[k, i] = f[j]
    
    # Maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res
