```python
    if S < 0.01:
        return np.inf
    n = len(a)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S * S)

    Compute gradient of max(conv) w.r.t. a.
    Returns a vector of the same length as a.

    n = len(a)
    conv_len = len(conv)  # 2n-1
```

```python
import numpy as np
import time

def _convolve_full(a):
    """Full auto-convolution."""
    return np.convolve(a, a)

def _score(a, conv=None):
    """Compute the objective value."""
    if conv is None:
        conv = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * len(a) * conv.max() / (S * S)

def _gradient_single_peak(a, conv):
    """Gradient of max(conv) w.r.t. a, using the single max peak."""
    n = len(a)
    k = int(np.argmax(conv))
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = k - np.arange(i_start, i_end + 1)
        grad[i_start:i_end+1] = 2.0 * a[idx]
    return grad

def _gradient_soft_peak(a, conv, temperature):
    """Smoothed gradient using softmax-weighted peaks."""
    n = len(a)
    # Temperature-scaled softmax
    c_max = conv.max()
    w = np.exp((conv - c_max) / max(temperature, 1e-10))
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    
    # Compute gradient via sliding window
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    # Use sliding_window_view
    from numpy.lib.stride_tricks import sliding_window_view
    if n <= 1:
        return np.zeros(n)
    window = sliding_window_view(conv, n)  # shape (n, n), window[i,j] = conv[i+j]
    w_window = sliding_window_view(w, n)  # shape (n, n), w_window[i,j] = w[i+j]
    grad = 2.0 * (w_window @ a)
    return grad

def _lp_direction(a, M, max_iter=100000):
    """Solve LP to find optimal direction. Returns (direction, max_sum) or (None, None)."""
    from scipy.optimize import linprog
    
    n = len(a)
    # Variables: g[0], ..., g[n-1] >= 0
    # Maximize: sum(g) => minimize: -sum(g)
    c = -np.ones(n)
    
    # Constraints: (a * g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_i a[i] * g[k-i] (where indices must be valid)
    # For row k: sum_{i: 0<=k-i<n} a[i] * g[k-i] <= M
    # Let j = k-i: sum_{j: 0<=k-j<n and 0<=j<n} a[k-j] * g[j] <= M
    # So A[k,j] = a[k-j] if 0<=k-j<n, else 0
    
    # Build A (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k - j < n:
                A[k, j] = a[k-j]
    
    b = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x, -res.fun  # direction, max sum
    return None, None
```

```python
import time
from numpy.lib.stride_tricks import sliding_window_view

def _autoconv(a):
    return np.convolve(a, a, mode='full')

def _quick_score(a, conv):
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * a.size * conv.max() / (S * S)

def _peak_grad(a, conv, top_k=5):
    """Gradient using top-k peaks."""
    n = a.size
    if top_k == 1:
        k = int(np.argmax(conv))
        grad = np.zeros(n)
        i0, i1 = max(0, k-n+1), min(n-1, k)
        grad[i0:i1+1] = 2.0 * a[k - np.arange(i0, i1+1)]
        return grad
    # Top-k
    idx = np.argsort(conv)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        i0, i1 = max(0, k-n+1), min(n-1, k)
        grad[i0:i1+1] += 2.0 * a[k - np.arange(i0, i1+1)]
    return grad

def _soft_grad(a, conv, T):
    """Softmax-weighted gradient."""
    n = a.size
    w = np.exp((conv - conv.max()) / max(T, 1e-12))
    w /= w.sum() + 1e-300
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    # w has length 2n-1
    # window[i,j] = w[i+j] for i,j in [0,n-1]
    ws = sliding_window_view(w, n)  # (n, n)
    return 2.0 * (ws @ a)

def _lp_step(a, M):

    n = a.size
    # Build A: (2n-1) x n, A[k,j] = a[k-j] if 0<=k-j<n

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
```
