from scipy.optimize import linprog
import time

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    # Choose sequence length
    n = 4096  # Power of 2 for efficiency
    
    # Initialize: try the given sequence first
    # If it works, use it; otherwise start uniform
    parent = np.ones(n) / n  # Normalized to sum=1
    
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    deadline = time.monotonic() + budget_s - 10
    
    while time.monotonic() < deadline:
        # Compute current max convolution
        c = np.convolve(best, best)
        M = c.max()
        
        # Solve LP: maximize sum(g) s.t. (best*g)[k] <= M, g >= 0
        # (best*g)[k] = sum_j best[j] * g[k-j]
        # This is a linear constraint on g for each k
        
        # Build constraint matrix A such that A @ g <= M * ones(2n-1)
        # A[k, j] = best[j] if j <= k and k-j < n, else 0
        # Wait, (best*g)[k] = sum_j best[j] * g[k-j] where 0 <= j < n and 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(k, n-1)
        
        # This is a banded matrix! A[k,j] = best[j] if max(0,k-n+1) <= j <= min(k,n-1)
        
        # For efficiency, we can build this as a sparse matrix
        # A has (2n-1) rows and n columns
        # Row k has non-zeros in columns max(0,k-n+1) to min(k,n-1)
        
        # Let's use scipy.sparse
        from scipy.sparse import lil_matrix
        
        A = lil_matrix((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            A[k, j_start:j_end+1] = best[j_start:j_end+1]
        
        A = A.tocsr()
        
        # LP: max sum(g) s.t. A@g <= M, g >= 0
        # In scipy form: min -sum(g) s.t. A@g <= M
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 0:
                g = g / g_sum  # Normalize to sum=1
                # Interpolate
                t = 0.1  # step size
                new = (1-t)*best + t*g
                new = np.clip(new, 0, None)
                score = float(evaluate_sequence(new.tolist()))
                if score < best_score:
                    best = new
                    best_score = score
                    print(f"Improved to {score:.6f}")
        else:
            # LP failed, try a smaller step or different approach
            break
    
    return [float(x) for x in best]

    # A[k, j] = a[j] if max(0, k-n+1) <= j <= min(k, n-1), else 0

def build_conv_constraint_matrix(a):
    """Build sparse matrix A where A @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            rows.extend(range(k, k + 1) * (j_end - j_start + 1))
            cols.extend(range(j_start, j_end + 1))
            vals.extend(a[j_start:j_end + 1])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_shifted = c - c.max()  # For numerical stability

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using smooth gradient descent
with multiple initializations and LP-based refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_conv_matrix(a):
    """Build sparse matrix A where A @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs')
    return res.x if res.success else None


def init_ramp(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def init_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    return a / a.sum()
