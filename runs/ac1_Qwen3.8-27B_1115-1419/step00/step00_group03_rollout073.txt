"""
Multi-phase optimizer: FISTA (smoothed max on simplex) + coordinate descent refinement.
Minimizes 2n*max(a*a)/(sum(a))^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize_scalar


def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _auto_conv(a):
    """Linear auto-convolution."""
    return np.convolve(a, a)


def _exact_score(a):
    """Compute 2n*max(a*a)/(sum(a))^2."""
    n = len(a)
    v = _auto_conv(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf, v
    return 2.0 * n * float(v.max()) / (S * S), v

            if time.monotonic() > deadline:
                break

            v_y = _auto_conv(y)
            v_scaled = beta * v_y
            v_m = float(v_scaled.max())
            exp_vals = np.exp(v_scaled - v_m)
            p = exp_vals / exp_vals.sum()

            # grad[i] = 2 * sum_k p[k] * a[k-i]  (cross-correlation)
            grad = 2.0 * np.correlate(p, y, 'full')[n - 1:2 * n - 1]

            a_new = _project_simplex(y - eta * grad / n)

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = a_new + ((t_k - 1.0) / t_next) * (a_new - y)
            y = _project_simplex(np.maximum(y, 0))

            t_k = t_next
            a = a_new

        score, v = _exact_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"  FISTA outer={outer}, beta={beta:.1f}: score={score:.6f}, best={best_score:.6f}, t={time.monotonic():.0f}")
        beta *= 1.8

    return best_a, best_score


def _cd_sweep(a, v, deadline, rng):
    """One full sweep of coordinate descent. Returns (a, v, improved)."""
    n = len(a)
    S = float(a.sum())
    improved = False
    order = rng.permutation(n)

    for idx in range(n):
        if time.monotonic() > deadline:
            return a, v, improved

        i = int(order[idx])
        L = float(a[i])

        k_min = i
        k_max = min(2 * n - 2, i + n - 1)
        num_aff = k_max - k_min + 1
        if num_aff <= 0:
            continue

        old_vals = v[k_min:k_max + 1].copy()

        # Slopes: 2*a[k-i] for linear terms, 0 for quadratic
        j_vals = np.arange(k_min, k_max + 1) - i
        slopes = 2.0 * a[j_vals]
        quad_pos = np.where(np.arange(k_min, k_max + 1) == 2 * i)[0]
        if len(quad_pos) > 0:
            slopes[quad_pos] = 0.0

        # Unaffected max
        if k_min > 0 and k_max + 1 < 2 * n - 1:
            unaffected_max = max(float(v[:k_min].max()), float(v[k_max + 1:].max()))
        elif k_min > 0:
            unaffected_max = float(v[:k_min].max())
        elif k_max + 1 < 2 * n - 1:
            unaffected_max = float(v[k_max + 1:].max())
        else:
            unaffected_max = 0.0

        # Precompute quadratic flag for speed
        has_quad = len(quad_pos) > 0
        q_idx = quad_pos[0] if has_quad else -1
        old_q = old_vals[q_idx] if has_quad else 0.0

        def f_of_x(x):
            if x < 0:
                return np.inf
            delta = x - L
            M_new = float((old_vals + delta * slopes).max())
            if has_quad:
                qv = old_q + x * x - L * L
                if qv > M_new:
                    M_new = qv
            if M_new < unaffected_max:
                M_new = unaffected_max
            S_new = S + x - L
            if S_new < 0.01:
                return np.inf
            return 2.0 * n * M_new / (S_new * S_new)

        # Quick grid search
        x_upper = max(L * 5.0, 4.0 / n)
        best_x = L
        best_f = f_of_x(L)

        for frac in [0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0, 4.0]:
            xc = L * frac
            if xc < 0:
                xc = 0.0
            if xc > x_upper:
                continue
            fc = f_of_x(xc)
            if fc < best_f:
                best_f = fc
                best_x = xc

        # Also try x=0 explicitly
        f0 = f_of_x(0.0)
        if f0 < best_f:
            best_f = f0
            best_x = 0.0

        # Refine with bounded scalar minimization
        lo = max(0.0, best_x - 0.3 * max(L, 1.0 / n))
        hi = best_x + 0.5 * max(L, 1.0 / n)
        hi = min(hi, x_upper)
        if hi - lo > 1e-14:
            try:
                res = minimize_scalar(f_of_x, bounds=(lo, hi), method='bounded',
                                      options={'maxiter': 20, 'xatol': 1e-12})
                if res.fun < best_f:
                    best_f = res.fun
                    best_x = float(res.x)
            except Exception:
                pass

        # Apply update
        f_orig = f_of_x(L)
        if best_x != L and best_f < f_orig - 1e-14:
            delta = best_x - L
            a[i] = best_x
            S += delta
            new_seg = old_vals + delta * slopes
            if has_quad:
                new_seg[q_idx] = old_q + best_x * best_x - L * L
            v[k_min:k_max + 1] = new_seg
            improved = True

    return a, v, improved


def _make_inits(n, rng):
    """Generate diverse initial sequences on the simplex."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian windows (different widths)
    for sf in [0.15, 0.25, 0.35, 0.5]:
        sig = sf * n
        g = np.exp(-0.5 * ((x - center) / sig) ** 2)
        inits.append(g / g.sum())

    # Cosine windows
    for freq in [1, 2, 3]:
        c = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        inits.append(c / c.sum())

    # Edge-concentrated (high at boundaries)
    e = np.abs(x - center) / (n / 2.0)
        e = e ** 2 + 0.05
    inits.append(e / e.sum())

    # Random smooth
    r = rng.random(n)
    w = max(3, n // 50)
    kern = np.ones(w) / w
    rs = np.convolve(r, kern, mode='same')
    inits.append(rs / rs.sum())

    # Binary-like
    b = rng.random(n) > 0.5
    bv = b.astype(np.float64) * 0.8 + 0.2
    inits.append(bv / bv.sum())

    # Parent sequence (if available)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.all(parent >= 0) and np.all(np.isfinite(parent)):
            ps = parent / parent.sum()
            inits.append(ps)
    except Exception:
        pass

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: diverse initialization + FISTA + coordinate descent.
    Minimizes 2n*max(a*a)/(sum(a))^2 over non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    # Try multiple sequence lengths
    n_candidates = [2535, 2000, 3000]
    n = 2535  # primary

    best_overall = None
    best_score_overall = np.inf

    for n_try in n_candidates:
        if time.monotonic() > total_deadline:
            break
        phase_deadline = min(total_deadline, time.monotonic() + (budget_s - 5.0) / len(n_candidates))

        inits = _make_inits(n_try, rng)
        n_inits = min(len(inits), 6)

        for init_k in range(n_inits):
            if time.monotonic() > phase_deadline:
                break
            a = inits[init_k].copy()
            # Ensure sum = 1
            a = a / a.sum()

            # FISTA phase (time-limited)
            fista_time = min(60.0, (phase_deadline - time.monotonic()) * 0.4)
            fista_deadline = time.monotonic() + max(2.0, fista_time)
            a, sc = _fista_phase(a, fista_deadline, max_outer=8)

            print(f"n={n_try}, init={init_k}: after FISTA score={sc:.6f}")

            # CD refinement
            v = _auto_conv(a)
            cd_deadline = min(phase_deadline, time.monotonic() + (budget_s - 5.0) * 0.5)
            no_improve = 0
            for sweep in range(500):
                if time.monotonic() > cd_deadline:
                    break
                a, v, improved = _cd_sweep(a, v, cd_deadline, rng)
                if improved:
                    no_improve = 0
                else:
                    no_improve += 1
                if no_improve >= 40:
                    # Perturb to escape local minimum
                    noise = rng.normal(0, 0.02, size=n_try)
                    a = np.maximum(a + noise, 0)
                    if a.sum() < 0.01:
                        a = inits[init_k].copy()
                        a = a / a.sum()
                    else:
                        a = a / a.sum()
                        v = _auto_conv(a)
                    no_improve = 0
                    if time.monotonic() > cd_deadline:
                        break

            score, v = _exact_score(a)
            if score < best_score_overall:
                best_score_overall = score
                best_overall = a.copy()
                print(f"  NEW BEST: n={n_try}, init={init_k}, score={score:.6f}")

            print(f"  n={n_try}, init={init_k}: final score={score:.6f}, best_overall={best_score_overall:.6f}")

    if best_overall is None:
        best_overall = np.ones(n) / n
        best_score_overall, _ = _exact_score(best_overall)

    # Final verification with evaluate_sequence
    try:
        eval_sc = float(evaluate_sequence([float(x) for x in best_overall]))
        if np.isfinite(eval_sc) and eval_sc < best_score_overall:
            best_score_overall = eval_sc
    except Exception:
        pass

    print(f"FINAL: score={best_score_overall:.6f}, length={len(best_overall)}")
    return [float(x) for x in best_overall]
