def _fft_convolve(a):
    """Full linear convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def _smooth_max_and_grad(a, eps):

    n = len(a)

    # For numerical stability: M = max(conv) + (1/eps)*log(sum(exp(eps*(conv-max))))
    c_max = np.max(conv)

    log_sum = np.log(np.sum(exp_vals))
    M_smooth = c_max + log_sum / eps
