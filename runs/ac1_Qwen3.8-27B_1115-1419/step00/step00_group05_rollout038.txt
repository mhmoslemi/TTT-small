Gradient-based optimization of non-negative step function to minimize
2n * max(autocorrelation) / (sum)^2.

Uses FFT-based autocorrelation computation and smoothed max (log-sum-exp)
for differentiable optimization. Includes multiple restarts and adaptive
learning rate.
"""

import numpy as np
import time


def compute_autocorr_fft(a):
    """Compute linear autocorrelation (np.convolve(a,a)) using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft <<= 1
    A = np.fft.rfft(a, N_fft)
    power = A.real ** 2 + A.imag ** 2
    c = np.fft.irfft(power, N_fft)[:N]
    return c


def compute_smoothed_max_gradient(a, c, T):
    """
    Compute gradient of T * log(sum exp(c/T)) w.r.t. a.
    Returns (smoothed_max_value, gradient).
    """
    n = len(a)
    N = 2 * n - 1
    
    # Compute softmax weights
    c_shifted = c - c.max()  # for numerical stability
    w = np.exp(c_shifted / T)
    w /= w.sum()
    
    smoothed_max = c.max() + T * np.log(w.sum())  # This is T * log(sum exp(c/T))
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * s[i] where s[i] = sum_j w[i+j] * a[j]
    # Using convolution: s[i] = conv(w_rev, a)[2n-2-i]
    w_rev = w[::-1]
    N_fft = 1
    while N_fft < (N + n - 1):
        N_fft <<= 1
    W = np.fft.fft(w_rev, N_fft)
    A = np.fft.fft(a, N_fft)
    conv = np.fft.ifft(W * A).real
    # s[i] = conv[2n-2-i] for i=0..n-1
    # So s = conv[n-1 : 2n-1][::-1]
    s = conv[n - 1 : 2 * n - 1][::-1]
    grad = 2.0 * s
    
    return smoothed_max, grad

    n = len(best)
    a = best.copy().astype(np.float64)

    target_sum = np.sqrt(2 * n)  # This makes the objective = max_c when sum = sqrt(2n)
    a = a * (target_sum / a.sum()) if a.sum() > 0 else np.ones(n) * target_sum / n
    
    # Adaptive temperature for log-sum-exp
    T = 1.0
    T_min = 0.01
    T_decay = 0.995
    lr = 0.01

Optimizes non-negative step function coefficients to minimize
2n * max(autocorrelation) / (sum)^2 using gradient descent with
log-sum-exp smoothing, FFT-based autocorrelation, and simulated
annealing for escape from local minima.
"""

import numpy as np
import time


def _next_pow2(x):
    """Next power of 2 >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def compute_autocorr_fft(a):
    """Compute linear autocorrelation (equivalent to np.convolve(a, a)) via FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = _next_pow2(N)
    A = np.fft.rfft(a, N_fft)
    power = A.real ** 2 + A.imag ** 2
    c = np.fft.irfft(power, N_fft)[:N]
    return c


def compute_gradient(a, c, T):
    """
    Compute gradient of the smoothed max of c w.r.t. a.
    Uses: grad[i] = 2 * sum_k w_k * a[k-i] (for valid k-i)
    Computed via FFT-based convolution.
    """
    n = len(a)
    N = 2 * n - 1
    
    # Softmax weights (numerically stable)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    
    # s[i] = sum_j w[i+j] * a[j] for i=0..n-1
    # s[i] = conv(w[::-1], a)[2n-2-i]
    w_rev = w[::-1]
    conv_len = N + n - 1  # (2n-1) + n - 1 = 3n-2
    N_fft = _next_pow2(conv_len)
    W = np.fft.fft(w_rev, N_fft)
    A = np.fft.fft(a, N_fft)
    conv = np.fft.ifft(W * A).real
    s = conv[n - 1 : 2 * n - 1][::-1]
    
    return 2.0 * s


def objective_and_grad(a, T):
    """Compute approximate objective and its gradient."""
    n = len(a)
    c = compute_autocorr_fft(a)
    max_c = c.max()
    sum_a = a.sum()
    
    if sum_a < 0.01:
        return 1e10, np.zeros(n)
    
    # Smoothed max gradient
    grad_max = compute_gradient(a, c, T)
    
    # Full objective gradient:
    # f = 2n * max_c / sum_a^2
    # df/da = 2n * (grad_max * sum_a - 2 * max_c * a) / sum_a^3
    total_grad = 2.0 * n * (grad_max * sum_a - 2.0 * max_c * a) / (sum_a ** 3)
    
    score = 2.0 * n * max_c / (sum_a ** 2)
    return score, total_grad
