def _score_fast(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 using FFT."""
    S = a.sum()
    if S < 1e-15:
        return np.inf
    a = a / S
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    A = np.fft.rfft(a, L)
    r = np.fft.irfft(A * A, L)[:2 * n - 1]
