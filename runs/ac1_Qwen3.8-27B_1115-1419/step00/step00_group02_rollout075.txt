"""
Search for non-negative step-function coefficients minimizing
2*n*max(conv(a,a))/sum(a)^2.

Method:
- Start from several structured and block-parameterized initializations.
- Use a low-dimensional block-level simulated-annealing hill search.
- Refine with the linear-programming direction from the prior state-of-the-art
  method: maximize sum(g) subject to conv(f,g) <= current peak.
- Polish with subgradient / Frank-Wolfe style local search on the samples.
- Upsample the best medium-length result and polish at a larger length.
"""

import time

    import scipy.sparse as sp

    HAS_SCIPY = True
except Exception:
    HAS_SCIPY = False


def instance_score_key(item):

def sanitize_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    return np.clip(a, 0.0, 1000.0)


def normalize_sequence(a, target_sum):
    a = sanitize_array(a)
    if a.size == 0:
        return a
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        out = np.ones(a.size, dtype=np.float64) * max(0.0, float(target_sum)) / float(a.size)
        return out
    a = a * (float(target_sum) / s)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(np.sum(a))
    if s2 > 0.0 and np.isfinite(s2):
        a = a * (float(target_sum) / s2)
    return np.clip(a, 0.0, 1000.0)


def fft_convolution(a):
    a = sanitize_array(a)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    m = 2 * n - 1
    n_fft = 1 << max(0, (m - 1).bit_length())
    A = np.fft.rfft(a, n_fft)
    B = np.fft.irfft(A * A, n_fft)[:m]
    # FFT roundoff can create tiny negative values for a nonnegative convolution.
    B = np.where(B > 0.0, B, 0.0)
    return B


def convolution_max(a):
    b = fft_convolution(a)
    if b.size == 0:
        return np.inf
    v = float(np.max(b))
    return v if np.isfinite(v) else np.inf


def objective_value(a):
    a = sanitize_array(a)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    b = fft_convolution(a)
    if b.size == 0:
        return np.inf
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (s * s))

    return np.inf


def make_two_level_half(n):
    a = np.ones(max(1, int(n)), dtype=np.float64)
    half = n // 2
    if half > 0:
        a[half:] = 2.0
    return a


def make_levels_sequence(levels, n):
    levels = sanitize_array(levels).ravel()
    n = int(max(1, n))
    q = int(max(1, levels.size))
    if q == 1:
        return np.full(n, float(levels[0]), dtype=np.float64)
    base = n // q
    rem = n % q
    reps = np.full(q, base, dtype=np.int64)
    if rem > 0:
        reps[rem:] += 1
    return np.repeat(levels, reps).astype(np.float64)


def block_hill_search(n, q, time_budget, seed, monotone=True):
    if time_budget <= 1.0:
        a = make_two_level_half(n)
        return a, objective_value(a)

    rng = np.random.default_rng(seed)
    q = int(max(2, min(q, n)))

    inits = []
    inits.append(np.linspace(1.0, 2.0, q))
    inits.append(np.sqrt(np.linspace(1.0, 2.0, q)))
    inits.append(np.linspace(0.5, 1.5, q))

    half = q // 2
    two = np.ones(q, dtype=np.float64)
    if half > 0:
        two[half:] = 2.0
    inits.append(two)

    three = np.ones(q, dtype=np.float64)
    third = max(1, q // 3)
    two_third = max(third + 1, 2 * q // 3)
    if two_third < q:
        three[third:two_third] = 1.5
        three[two_third:] = 2.0
    inits.append(three)

    inits.append(np.abs(rng.standard_normal(q)) + 0.25)
    inits.append(np.abs(rng.standard_normal(q)) + 0.25)

    cur_levels = None
    best_levels = None
    cur_score = np.inf
    best_score = np.inf

    for init in inits:
        lev = sanitize_array(init).astype(np.float64)
        if lev.size != q:
            lev = lev[:q]
        if lev.size < q:
            lev = np.pad(lev, (0, q - lev.size), mode="edge")
        if monotone:
            lev = np.sort(lev)
        lev = np.clip(lev, 1e-4, 10.0)
        mean = float(np.mean(lev))
        if mean > 0.0:
            lev = lev / mean
        seq = make_levels_sequence(lev, n)
        sc = objective_value(seq)
        if sc < cur_score:
            cur_score = sc
            cur_levels = lev.copy()
        if sc < best_score:
            best_score = sc
            best_levels = lev.copy()

    if best_levels is None:
        best_levels = make_two_level_half(q)
        best_score = objective_value(make_levels_sequence(best_levels, n))
        cur_levels = best_levels.copy()
        cur_score = best_score

    T = 0.035
    stall = 0
    big_stall = 0
    deadline = time.monotonic() + max(0.0, float(time_budget) - 1.0)
    it = 0

    while time.monotonic() < deadline:
        if cur_levels is None:
            break
        cand = cur_levels.copy()
        n_pert = int(rng.integers(1, 4))
        for _ in range(n_pert):
            idx = int(rng.integers(q))
            cand[idx] *= float(np.exp(rng.normal(0.0, 0.38)))

        if monotone:
            cand = np.sort(cand)
        cand = np.clip(cand, 1e-4, 10.0)
        mean = float(np.mean(cand))
        if mean > 0.0:
            cand = cand / mean

        seq = make_levels_sequence(cand, n)
        sc = objective_value(seq)
        it += 1

        if sc < best_score:
            best_score = sc
            best_levels = cand.copy()
            cur_score = sc
            cur_levels = cand.copy()
            stall = 0
            big_stall = 0
            T = max(0.001, T * 0.99)
        else:
            stall += 1
            if sc < cur_score:
                cur_score = sc
                cur_levels = cand.copy()
                stall = 0
            elif T > 1e-6 and rng.random() < np.exp(-(sc - cur_score) / T):
                cur_levels = cand.copy()
            T = max(1e-6, T * 0.999)

        if stall > 4000:
            x = np.linspace(1.0, 2.0, q) + 0.45 * rng.standard_normal(q)
            if monotone:
                x = np.sort(x)
            x = np.clip(x, 1e-4, 10.0)
            mean = float(np.mean(x))
            if mean > 0.0:
                x = x / mean
            cur_levels = x.copy()
            cur_score = objective_value(make_levels_sequence(x, n))
            if cur_score < best_score:
                best_score = cur_score
                best_levels = x.copy()
            stall = 0
            big_stall += 1
            T = 0.035

        if big_stall > 25:
            break

        if it % 5000 == 0:
            print(f"  block nn {it} best={best_score:.6f} cur={cur_score:.6f} T={T:.5f}")

    return make_levels_sequence(best_levels, n), best_score

def precompute_conv_indices(n):
    n = int(max(1, n))
    d = np.arange(n, dtype=np.int32).reshape(n, 1)
    j = np.arange(n, dtype=np.int32).reshape(1, n)
    rows = (d + j).astype(np.int32, copy=False).ravel()
    cols = j.astype(np.int32, copy=False).ravel()
    cobj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return rows, cols, cobj, bounds


def build_conv_matrix(a, n, rows, cols):
    a = sanitize_array(a)
    data = np.repeat(a, n).astype(np.float64, copy=False)
    return sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def lp_refine(a, time_budget, pre, seed):
    if not HAS_SCIPY or pre is None:
        a = sanitize_array(a)
        n = a.size
        if n == 0:
            return a, np.inf
        a = normalize_sequence(a, np.sqrt(2.0 * n))
        return a, objective_value(a)

    rows, cols, cobj, bounds = pre
    a = sanitize_array(a)
    n = a.size
    if n == 0:
        return a, np.inf

    S_target = float(np.sqrt(2.0 * n))
    # A tiny floor keeps the LP from dealing with completely zero rows of influence.
    a = np.maximum(a, 1e-7)
    a = normalize_sequence(a, S_target)

    b = fft_convolution(a)
    best_m = float(np.max(b)) if b.size else np.inf
    best_a = a.copy()
    cur_m = best_m
    a = a.copy()

    if not np.isfinite(best_m) or best_m <= 0.0:
        return best_a, best_m

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(time_budget) - 5.0)
    no_improve = 0
    stalls = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break

        M = float(convolution_max(a))
        if not np.isfinite(M) or M <= 0.0:
            break

        if no_improve > 0:
            M_eff = M * (1.0 + min(0.20, 0.012 * no_improve))
        else:
            M_eff = M * (1.0 + 1e-9)

        try:
            A = build_conv_matrix(a, n, rows, cols)
            b_ub = np.full(2 * n - 1, float(M_eff), dtype=np.float64)
            timelimit = max(4.0, min(25.0, rem * 0.03))
            opts = {
                "presolve": True,
                "tol": 1e-8,
                "output_flag": False,
                "time_limit": timelimit
            }
            try:
                res = linprog(
                    cobj,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=list(bounds),
                    method="highs",
                    options=opts
                )
            except Exception:
                res = linprog(
                    cobj,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=list(bounds),
                    method="highs",
                    options={"presolve": True, "tol": 1e-8, "output_flag": False}
                )
        except Exception:
            no_improve += 2
            if no_improve > 8:
                break
            continue

        accepted_this_lp = False
        x = getattr(res, "x", None)
        if x is not None and np.all(np.isfinite(x)):
            g = sanitize_array(x)
            sg = float(np.sum(g))
            if sg > 1e-9:
                g = normalize_sequence(g, S_target)
                # Feasibility guard for numerical tolerance.
                try:
                    maxviol = float(np.max(A.dot(g)))
                except Exception:
                    maxviol = np.inf
                if np.isfinite(maxviol) and maxviol <= M_eff * (1.0 + 1e-5) + 1e-8:
                    for t in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35, 0.6, 1.0):
                        mix = (1.0 - t) * a + t * g
                        mix = sanitize_array(mix)
                        mix = normalize_sequence(mix, S_target)
                        mm = float(convolution_max(mix))
                        if np.isfinite(mm) and mm < cur_m - 1e-10:
                            a = mix
                            cur_m = mm
                            accepted_this_lp = True
                            if cur_m < best_m - 1e-12:
                                best_m = cur_m
                                best_a = a.copy()
                            break

        if accepted_this_lp:
            no_improve = 0
            continue

        no_improve += 1
        if no_improve >= 12:
            a0 = best_a.copy()
            if a0.size >= 8:
                width = max(2, a0.size // 64)
                for _ in range(4):
                    start = int(rng.integers(0, max(1, a0.size - width + 1)))
                    a0[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
                a0 = sanitize_array(a0)
                a0 = normalize_sequence(a0, S_target)
                pm = float(convolution_max(a0))
                if np.isfinite(pm) and pm < best_m * 1.08:
                    a = a0
                    cur_m = pm
            else:
                a = a
                cur_m = float(convolution_max(a))

            if cur_m < best_m - 1e-10:
                best_m = cur_m
                best_a = a.copy()

            no_improve = 0
            stalls += 1
            if stalls >= 6:
                break

    final_score = objective_value(best_a)
    if np.isfinite(final_score) and final_score < best_m:
        best_m = final_score
    return best_a, best_m


def gradient_polish(a, time_budget, seed):
    a = sanitize_array(a)
    n = a.size
    if n < 4:
        a = normalize_sequence(a, np.sqrt(max(1.0, 2.0 * n)))
        return a, objective_value(a)

    S_target = float(np.sqrt(2.0 * n))
    a = normalize_sequence(a, S_target)
    b = fft_convolution(a)
    best_m = float(np.max(b)) if b.size else np.inf
    best_a = a.copy()
    cur_m = best_m
    a = a.copy()

    if not np.isfinite(best_m) or best_m <= 0.0:
        return best_a, best_m

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(time_budget) - 5.0)
    eta = 0.12
    no_improve = 0
    stalls = 0
    K = 32

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        b = fft_convolution(a)
        mx = float(np.max(b)) if b.size else np.inf
        if not np.isfinite(mx) or mx <= 0.0:
            break
        cur_m = mx

        k = int(min(K, b.size))
        top_idx = np.argsort(b)[-k:]
        top_vals = b[top_idx]
        tw = np.ones(k, dtype=np.float64) / float(k)

        grad = np.zeros(n, dtype=np.float64)
        for kk, ww in zip(top_idx, tw):
            kk = int(kk)
            wm = float(ww)
            if wm == 0.0:
                continue
            jm = max(0, kk - n + 1)
            jx = min(n - 1, kk)
            if jm <= jx:
                grad[jm:jx + 1] += 2.0 * wm * a[kk - jx:kk - jm + 1][::-1]

        s = float(np.sum(a))
        if s <= 0.0 or not np.isfinite(s):
            a = normalize_sequence(np.ones(n), S_target)
            best_a = a.copy()
            best_m = float(convolution_max(a))
            break

        # Smoother active-set approximation of the objective gradient.
        m_app = float(np.mean(top_vals))
        g = grad - (2.0 * m_app / s) * a
        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn) or gn < 1e-14:
            a = best_a.copy()
            cur_m = best_m
            no_improve += 2
            if no_improve > 80:
                break
            continue

        g = g / gn
        found = False

        for _ in range(25):
            cand = a - eta * g
            cand = sanitize_array(cand)
            cand = normalize_sequence(cand, S_target)
            cm = float(convolution_max(cand))
            if np.isfinite(cm) and cm < cur_m - 1e-11:
                a = cand
                cur_m = cm
                found = True
                if cur_m < best_m - 1e-12:
                    best_m = cur_m
                    best_a = a.copy()
                break
            eta *= 0.5

        if found:
            no_improve = 0
            stalls = 0
            eta = min(0.65, eta * 1.08)
            continue

        # Frank-Wolfe style single coordinate shuffle.
        j_inc = int(np.argmin(g))
        j_dec = int(np.argmax(g))
        if j_inc != j_dec:
            for frac in (0.01, 0.03, 0.08, 0.18, 0.45):
                cand = a.copy()
                move = frac * cand[j_dec]
                cand[j_dec] -= move
                cand[j_inc] += move
                cand = sanitize_array(cand)
                cand = normalize_sequence(cand, S_target)
                cm = float(convolution_max(cand))
                if np.isfinite(cm) and cm < cur_m - 1e-11:
                    a = cand
                    cur_m = cm
                    found = True
                    if cur_m < best_m - 1e-12:
                        best_m = cur_m
                        best_a = a.copy()
                    break

        if found:
            no_improve = 0
            eta = min(0.6, max(0.02, eta * 1.05))
            continue

        no_improve += 1
        if no_improve >= 25:
            a0 = best_a.copy()
            if a0.size >= 16:
                w = max(2, a0.size // 32)
                for _ in range(6):
                    st = int(rng.integers(0, max(1, a0.size - w + 1)))
                    a0[st:st + w] *= float(np.exp(rng.normal(0.0, 0.18)))
                    if rng.random() < 0.25:
                        a0 = 0.5 * (a0 + a0[::-1])
                a0 = sanitize_array(a0)
                a0 = normalize_sequence(a0, S_target)
                pm = float(convolution_max(a0))
                if np.isfinite(pm) and pm < best_m * 1.12:
                    a = a0
                    cur_m = pm
            no_improve = 0
            stalls += 1
            if stalls >= 8:
                break

    final_score = objective_value(best_a)
    if np.isfinite(final_score) and final_score < best_m:
        best_m = final_score
    return best_a, best_m


def upsample_sequence(a, n):
    a = sanitize_array(a).ravel()
    n = int(max(1, n))
    if a.size == 0:
        return np.ones(n, dtype=np.float64)
    if a.size == n:
        return a
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, n)
    out = np.interp(new_x, old_x, np.maximum(a, 0.0))
    return np.clip(out, 0.0, 1000.0)

    candidates = make_initial_candidates(n_work)
    parent = try_parent_candidate(n_work)
    if parent is not None:
        candidates.append(parent)

    # Low-dimensional block search: cheap and often finds good structured shapes.
    block_budget = min(300.0, 0.18 * max(60.0, budget))
    if time.monotonic() < global_deadline - 240.0 and block_budget > 20.0:
        try:
            bs, bsc = block_hill_search(n_work, block_q, block_budget, int(seed) + 17, monotone=True)
            bs = normalize_sequence(bs, np.sqrt(2.0 * n_work))
            bsc = objective_value(bs)
            print(f"block search score={bsc:.6f}")
            candidates.append(bs)
            if np.isfinite(bsc) and bsc < best_score:
                best_score = bsc
                best_a = bs.copy()
        except Exception as e:
            print("block search failed:", e)

    # Score and sort initial candidates.
    scored = []
    for seq in candidates:
        seq = sanitize_array(seq)
        if seq.size != n_work:
            seq = upsample_sequence(seq, n_work)
        sc = objective_value(seq)
        if np.isfinite(sc):
            scored.append((sc, seq))
    scored.sort(key=instance_score_key)

    if len(scored) > 0:
        sc0, seq0 = scored[0]
        if sc0 < best_score:
            best_score = sc0
            best_a = seq0.copy()
        top = [seq0.copy()]
        if len(scored) > 1:
            top.append(scored[1][1].copy())
        else:
            top.append(make_two_level_half(n_work))
    else:
        top = [best_a.copy(), make_two_level_half(n_work)]

    print("top initial candidates:")
    for i, (sc, seq) in enumerate(scored[:5]):
        print(f"  {i}: {sc:.6f}")

    # LP refinement phase.
    lp_total_budget = min(1500.0, 0.52 * max(60.0, budget))
    if HAS_SCIPY and time.monotonic() < global_deadline - 300.0 and lp_total_budget > 90.0:
        pre = precompute_conv_indices(n_work)
        n_top = len(top)
        for i, seq in enumerate(n_top):
            left = global_deadline - time.monotonic() - 360.0
            if left <= 30.0:
                break
            alloc = left / float(max(1, n_top - i))
            # Leave at least a little room; cap per refinement.
            alloc = max(45.0, min(720.0, alloc))
            print(f"LP refine candidate {i}, alloc={alloc:.1f}s")
            try:
                ra, ra_s = lp_refine(seq, alloc, pre, int(seed) + 101 + i)
                ra_s = objective_value(ra)
                print(f"  after LP score={ra_s:.6f}")
                if np.isfinite(ra_s) and ra_s < best_score:
                    best_score = ra_s
                    best_a = ra.copy()
            except Exception as e:
                print("LP refinement failed:", e)

    # Gradient polish at working length.
    grad_work_budget = min(500.0, 0.18 * max(60.0, budget))
    if time.monotonic() < global_deadline - 240.0 and grad_work_budget > 30.0:
        try:
            ga, ga_s = gradient_polish(best_a, grad_work_budget, int(seed) + 701)
            ga_s = objective_value(ga)
            print(f"gradient polish work score={ga_s:.6f}")
            if np.isfinite(ga_s) and ga_s < best_score:
                best_score = ga_s
                best_a = ga.copy()
        except Exception as e:
            print("gradient polish failed:", e)

    # Upsample and polish at a longer length.
    if time.monotonic() < global_deadline - 90.0:
        final_seq = upsample_sequence(best_a, n_final)
        final_seq = normalize_sequence(final_seq, np.sqrt(2.0 * n_final))
        fs = objective_value(final_seq)
        print(f"upsampled final before polish score={fs:.6f}")
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
            best_a = final_seq.copy()

        final_grad_budget = max(20.0, global_deadline - time.monotonic() - 45.0)
        if final_grad_budget > 40.0:
            try:
                fa, fa_s = gradient_polish(final_seq, final_grad_budget, int(seed) + 9001)
                fa_s = objective_value(fa)
                print(f"final polished score={fa_s:.6f}")
                if np.isfinite(fa_s) and fa_s < best_score:
                    best_score = fa_s
                    best_a = fa.copy()
            except Exception as e:
                print("final polish failed:", e)

    # Final safety: ensure a valid nonnegative sequence and try official evaluator once.
    best_a = sanitize_array(best_a)
    if best_a.size == 0:
        best_a = make_two_level_half(n_work)
    official = evaluate_safe(best_a)
    if np.isfinite(official) and official < best_score:
        best_score = official

    print(f"proposed final internal score={best_score:.6f} official={official if np.isfinite(official) else float('inf'):.6f} n={best_a.size}")

    # If the longer polished sequence scored worse, the stored best_a is already the best.
    return [float(x) for x in best_a]
