def _next_power_of_2(n):
    """Return the smallest power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _convolve_self(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    L = _next_power_of_2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    return c[:2*n-1]

def _compute_gradient(c, a, T):
    """Compute gradient of T*log(sum(exp(c_k/T))) w.r.t. a."""
    n = len(a)
    m = len(c)  # should be 2n-1
    
    # Softmax
    c_shifted = (c - np.max(c)) / T
    p = np.exp(c_shifted)
    p /= p.sum()

    # = 2 * (p cross-correlate a)[j]

    L = _next_power_of_2(n + m - 1)

    Pp = np.fft.rfft(p, L)
    R = np.fft.irfft(Pp * np.conj(Ap), L)
    g = 2.0 * R[:n]

def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _fft_convolve(a):
    """Full linear convolution of a with itself."""
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2*n-1]

    n = len(a)
