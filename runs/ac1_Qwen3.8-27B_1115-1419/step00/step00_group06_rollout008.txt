import numpy as np
from scipy.optimize import minimize

def compute_conv_max(params, n, M):
    # params = [eps_1, eps_2, ..., eps_M]
    j = np.arange(n)
    a = np.ones(n)
    for m in range(1, M+1):
        a -= params[m-1] * np.sin(2*np.pi*m*j/(n-1))
    a = np.clip(a, 0, 1000)
    S = a.sum()
    if S < 0.01:
        return 1e10
    b = np.convolve(a, a)
    return 2*n*b.max()/S**2

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    n = 5582
    M = 10  # number of modes
    
    # Initial guess
    eps_init = np.zeros(M)
    eps_init[0] = 0.7  # dominant mode
    
    # Optimize
    result = minimize(compute_conv_max, eps_init, args=(n, M), method='Nelder-Mead', 
                      options={'maxiter': 10000, 'xatol': 1e-6})
    
    # Convert back to sequence
    j = np.arange(n)
    a = np.ones(n)
    for m in range(1, M+1):
        a -= result.x[m-1] * np.sin(2*np.pi*m*j/(n-1))
    a = np.clip(a, 0, 1000)
    
    return [float(x) for x in a]
