"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smoothed (log-sum-exp) gradient-based
optimization with L-BFGS-B. Multiple initializations including flat-spectrum (random
phase) sequences. Increasing smoothing parameter for progressive refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _compute_convolution(a):
    """Compute linear convolution of a with itself using FFT for speed."""
    n = len(a)
    size = 1
    target = 2 * n - 1
    while size < target:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:target]
    return c


def _smooth_obj_grad(a, beta):
    """
    Compute smoothed objective value and gradient.
    Objective: 2*n*smooth_max(conv(a,a)) / sum(a)^2
    where smooth_max = (1/beta)*log(sum(exp(beta*c_k)))
    """
    n = len(a)
    S = np.sum(a)
    if S < 1e-12:
        return 1e10, np.zeros_like(a)
    
    c = _compute_convolution(a)
    
    # Numerically stable log-sum-exp
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    # Clip to prevent overflow (shouldn't be needed with shift, but safe)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    
    smooth_max = (c_max + np.log(sum_exp)) / beta
    
    # Weights (normalized)
    w = exp_shifted / sum_exp
    
    # Gradient of smooth_max w.r.t. a:
    # d(smooth_max)/da[m] = sum_k w[k] * dc[k]/da[m]  (the 1/beta and beta cancel)
    # Wait: d/dc[k] of (1/beta)*log(sum exp(beta*c)) = exp(beta*c[k]/sum exp(beta*c)) = w[k]
    # So d(smooth_max)/da[m] = sum_k w[k] * dc[k]/da[m]
    # dc[k]/da[m] = 2*a[k-m] if 0<=k-m<n
    # So: d(smooth_max)/da[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    #                                       ^^^^^^^
    #                where w[m+j] is 0 if m+j >= 2n-1
    
    # Compute: grad_sm[m] = 2 * sum_j w[m+j] * a[j]
    # This is 2 * np.correlate(w, a, 'valid')[m] since:
    # np.correlate(w, a, 'valid')[m] = sum_j w[m+j] * a[j] for valid j
    
    grad_conv = np.correlate(w, a, 'valid')
    smooth_max_grad = 2.0 * grad_conv
    
    # Full objective: f = 2*n*smooth_max / S^2
    obj = 2.0 * n * smooth_max / (S * S)
    
    # Gradient: df/da[m] = 2n * (smooth_max_grad[m]/S^2 - 2*smooth_max*a[m]/S^3)
    grad = (2.0 * n / (S * S)) * (smooth_max_grad - 2.0 * smooth_max * a / S)
    
    return obj, grad


def _true_obj(a):
    """Compute exact objective value."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = _compute_convolution(a)
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def _init_flat_spectrum(n, rng):
    """Initialize with approximately flat power spectrum."""
    # Random phases → flat magnitude spectrum
    phases = rng.uniform(0, 2 * np.pi, size=n)
    A = np.exp(1j * phases)
    # IFFT gives a complex sequence; take real part's abs or use both real/imag
    a = np.abs(np.fft.ifft(A))
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    return a


def _init_random_exp(n, rng):
    """Random exponential initialization."""
    a = rng.exponential(1.0, size=n)
    a /= np.sum(a)
    return a


def _init_block(n, rng):
    """Random block structure initialization."""
    n_blocks = rng.integers(3, 10)
    a = np.zeros(n)
    positions = np.sort(rng.choice(n, n_blocks, replace=False))
    for i, p in enumerate(positions):
        height = rng.uniform(0.5, 2.0)
        a[p] = height
    a += rng.uniform(0.01, 0.1, size=n)  # small baseline
    a = np.maximum(a, 0)
    a /= np.sum(a)
    return a


def _lbfgs_step(x, beta, n, max_inner_iter):
    """Perform one L-BFGS-B optimization step."""
    def fun(xx):
        a = np.maximum(xx, 0)
        val, _ = _smooth_obj_grad(a, beta)
        return val
    
    def gfun(xx):
        a = np.maximum(xx, 0)
        _, g = _smooth_obj_grad(a, beta)
        return g
    
    result = minimize(
        fun, x, jac=gfun,
        method='L-BFGS-B',
        bounds=[(0.0, 10000.0)] * n,
        options={'maxiter': max_inner_iter, 'gtol': 1e-8, 'factr': 1e8}
    )
    return result.x
