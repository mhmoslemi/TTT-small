def minimize_1d(f, a, b, maxiter=50, tol=1e-8):
    """Minimize f on [a, b] using golden section search. f must be a function object."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2
