import numpy as np
import time
from scipy.optimize import linprog

def compute_convolution(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a)

def solve_lp(f, M):
    """
    Solve: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns the optimal g and its sum.
    """
    n = len(f)
    # Build constraint matrix A: (2n-1) x n
    # A[k, m] = f[k-m] if 0 <= k-m < n, else 0
    # This is a Toeplitz-like matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                A[k, m] = f[j]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    # Solve LP
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x, -result.fun
    else:
        return None, None

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_convolution_matrix(f, n):
    """
    Build the sparse matrix A such that (A @ g)[k] = conv(f, g)[k].
    A is (2n-1) x n.
    A[k, m] = f[k-m] if 0 <= k-m < n, else 0.
    """
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        for m in range(m_min, m_max + 1):
            rows.append(k)
            cols.append(m)
            data.append(f[k - m])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_best_response(f, n):
    """
    Solve: min t s.t. conv(f, g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns (g_opt, t_opt).
    """
    A = build_convolution_matrix(f, n)
    # Variables: g_0, ..., g_{n-1}, t (total n+1 variables)
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1
    
    # Constraints:
    # 1) conv(f,g)[k] - t <= 0 for all k (2n-1 constraints)
    #    A @ [g; 1] - t <= 0 => [A | -I] @ [g; t] <= 0
    #    Actually: A @ g - t <= 0 => A @ g + (-1) * t <= 0
    #    So the constraint matrix is [A, -e] where e is the unit vector for t.
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A.toarray()  # This is dense, might be slow for large n
    A_ub[:, n] = -1
    b_ub = np.zeros(2*n-1)
    
    # 2) sum(g) = 1 => sum(g) <= 1 and sum(g) >= 1
    #    We can handle this with bounds or equality constraints.
    #    Let's use A_eq, b_eq.
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1  # sum(g) = 1
    A_eq[0, n] = 0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        g_opt = result.x[:n]

        return g_opt, t_opt
    else:

def build_sub_matrix(f, lags, n):
    """Build matrix where row i corresponds to lag lags[i]."""
    m = len(lags)
    A = np.zeros((m, n))
    for i, k in enumerate(lags):
        m_min = max(0, k - (n - 1))
        m_max = min(n - 1, k)
        if m_min <= m_max:
            A[i, m_min:m_max+1] = f[k-m_max:k-m_min:-1]
    return A

def solve_best_response(f, n, max_cp_iter=50, initial_active=30, max_active=200):
    """Solve the best response LP using cutting plane."""
    conv_ff = np.convolve(f, f)
    active = list(np.argsort(conv_ff)[-initial_active:])

        A_sub = build_sub_matrix(f, active, n)
