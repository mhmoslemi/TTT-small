    c = c.copy()
    
    while time.monotonic() < deadline - 10:
        b = np.convolve(c, c)
        peak = b.max()
        
        # Try to reduce all peaks simultaneously
        # For each k where b[k] > threshold, reduce the contributors
        
        threshold = peak * 0.95  # target: reduce peaks by 5%
        # Find all "hot" lags
        hot_k = np.where(b > threshold)[0]
        
        # For each hot lag k, the contributors are c[j]*c[k-j]
        # We want to reduce c at positions that are "hot" for multiple lags

        # heat[j] = sum_{k in hot_k: valid} c[j]*c[k-j]

            j_max = min(n - 1, k)

        # But be careful not to create new peaks
        
        top_frac = max(1, n // 20)
        hot_positions = np.argsort(heat)[-top_frac:]  # top 5% positions
        cool_positions = np.argsort(heat)[:top_frac]  # bottom 5%
