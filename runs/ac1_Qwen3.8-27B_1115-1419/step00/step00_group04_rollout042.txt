from scipy.optimize import linprog

def solve_lp(a, M, n):
    """
    Solve: maximize sum(b) subject to (a*b)_k <= M for all k, b >= 0.
    Return the optimal b.
    """
    # Variables: b_0, ..., b_{n-1}
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (a*b)_k <= M for k = 0, ..., 2n-2
    # (a*b)_k = sum_j a_{k-j} * b_j (where a_i = 0 for i < 0 or i >= n)
    # For fixed k: (a*b)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_{k-j} * b_j
    
    # A_ub matrix: (2n-1) x n
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k-n+1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A_ub[k, j] = a[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b_j >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
