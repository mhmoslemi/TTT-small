def lp_with_constraint_generation(a, M, n, max_iter=50):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0 using constraint generation."""
    # Start with the constraint at the peak
    conv_a = np.convolve(a, a)
    active = [np.argmax(conv_a)]
    
    for it in range(max_iter):
        # Build the LP with current active constraints
        # C @ g <= M where C has rows for active constraints
        g_var = cp.Variable(n, nonneg=True)
        constraints = [g_var >= 0]
        
        for k in active:
            # (a*g)_k = sum_j a[j] * g[k-j]
            # Build as a dot product
            j_start = max(0, k - (n-1))
            j_end = min(n-1, k)
            # (a*g)_k = sum_{j=j_start}^{j_end} a[j] * g[k-j]
            # = sum_{i=k-j_end}^{k-j_start} a[k-i] * g[i]
            i_start = k - j_end
            i_end = k - j_start
            # coefficients for g[i] are a[k-i]
            coeffs = a[k-i_end:k-i_start+1][::-1]
            # constraint: coeffs @ g[i_start:i_end+1] <= M
            expr = 0
            for idx, c in enumerate(coeffs):
                expr = expr + c * g_var[i_start + idx]
            constraints.append(expr <= M)
        
        obj = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        
        if g_var.value is None:
            return None
        
        g_val = np.asarray(g_var.value)
        
        # Check for violated constraints
        conv_g = np.convolve(a, g_val)
        violations = conv_g - M
        if violations.max() < 1e-6:
            return g_val  # Feasible!
        
        # Add the most violated constraint
        new_k = int(np.argmax(violations))
        if new_k not in active:
            active.append(new_k)
    
    return np.asarray(g_var.value)  # Return best effort

def solve_lp_simple(a, M, n, max_cg_iter=100, time_limit=10.0):
    """
    Solve: max sum(g) s.t. conv(a, g) <= M (element-wise), g >= 0
    Using constraint generation.
    """
    t_start = time.monotonic()
    
    # Start with the most violated constraint (argmax of a*a)
    conv_a = np.convolve(a, a)
    active = [int(np.argmax(conv_a))]
    
    g_val = None
    c = -np.ones(n)  # minimize -sum(g)
    
    for cg_iter in range(max_cg_iter):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Build A_ub from active constraints
        A_rows = []
        for k in active:
            A_rows.append(build_convolution_row(a, k, n))
        
        A_ub = np.vstack(A_rows)
        b_ub = np.full(len(active), M)
        
        # Solve using HiGHS
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, 
                     bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            # Try with just the peak constraint
            if len(active) > 1:
                active = [active[0]]
                continue
            return g_val  # Return last feasible or None
        
        g_val = res.x
        
        # Check feasibility against ALL constraints
        conv_g = np.convolve(a, g_val)
        max_violation = conv_g.max() - M
        
        if max_violation < 1e-6 * max(1.0, M):
            return g_val  # Feasible!
        
        # Add most violated constraint
        new_k = int(np.argmax(conv_g))
        if new_k not in active:
            active.append(new_k)
    
    # Return best effort (may be infeasible)
    return g_val

def build_conv_rows(a, active_indices, n):
    """Build rows of the convolution matrix for given constraint indices."""
    m = len(active_indices)
    A = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        i_start = k - j_end
        i_end = k - j_start
        A[idx, i_start:i_end+1] = a[j_start:j_end+1][::-1]
