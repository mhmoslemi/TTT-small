import time
from scipy.fft import fft, ifft, next_fast_len

def propose_candidate(seed=42, budget_s=2400, **kwargs):

    Minimizes 2*n*max(autoconv(a))/sum(a)^2 using soft-max gradient descent.
    Uses softmax parameterization for positivity, FFT for auto-convolution
    and gradient computation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    best_score = np.inf
    best_a = None
    
    for n in [1024, 2048, 4096, 8192]:
        if time.monotonic() > deadline - 120:
            break
        
        # Initialize
        t = np.zeros(n)  # log-space
        a = np.full(n, 1.0/n)  # uniform
        
        # Adam optimizer state
        m_t = np.zeros(n)
        v_t = np.zeros(n)
        step = 0
        lr = 1.0 / (n ** 0.5)
        beta_softmax = 50.0 * n  # temperature for soft-max of autoconv
        
        for it in range(10000):
            step += 1
            if time.monotonic() > deadline - 30:
                break
            
            # Compute auto-convolution
            fft_len = next_fast_len(2 * n - 1)
            A = fft(a, fft_len)
            autoconv = np.real(ifft(A * A, fft_len))[:2*n-1]
            autoconv = np.maximum(autoconv, 0)  # numerical safety
            
            peak_val = np.max(autoconv)
            score = 2.0 * n * peak_val
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute soft-max weights
            # Scale autoconv for numerical stability
            shifted = autoconv - peak_val
            exp_shifted = np.exp(beta_softmax * shifted)
            p = exp_shifted / np.sum(exp_shifted)
            
            # Compute gradient g[j] = 2*beta * sum_k p[k] * a[k-j]
            # Using FFT: conv(p, a_flipped)
            a_flip = a[::-1]
            conv_len = next_fast_len((2*n-1) + n - 1)
            P = fft(p, conv_len)
            AF = fft(a_flip, conv_len)
            conv_result = np.real(ifft(P * AF, conv_len))
            g = 2.0 * beta_softmax * conv_result[n-1:n-1+n]
            
            # Gradient in t-space
            dot_ga = np.dot(g, a)
            grad_t = a * (g - dot_ga)
            
            # Adam update
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m_t = beta1 * m_t + (1 - beta1) * grad_t
            v_t = beta2 * v_t + (1 - beta2) * grad_t**2
            m_hat = m_t / (1 - beta1**step)
            v_hat = v_t / (1 - beta2**step)
            t -= lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.exp(t - np.max(t))
            a /= np.sum(a)
            
            # Periodically increase beta_softmax (annealing)
            if it % 500 == 499 and it < 5000:
                beta_softmax *= 1.5
            
            # Decay learning rate
            lr *= 0.999

        # Fallback
        return [1.0] * 10

import numpy as np
from scipy.fft import fft, ifft, next_fast_len
import time

"""
Minimizes 2*n*max(autoconv(a))/sum(a)^2 using:
1. Power-law initialization a[i] ∝ (i+1)^p (provably gives objective ≈ 2/p for large n)
2. Soft-max gradient descent in log-space to further reduce auto-convolution peak
The key insight: asymmetric non-negative sequences can have max auto-conv < 1/n,
beating the symmetric/uniform bound of 2.
"""


def compute_score(a):
    """Compute the objective value for a normalized non-negative sequence."""
    n = len(a)
    fft_len = next_fast_len(2 * n - 1)
    A = fft(a, fft_len)
    autoconv = np.real(ifft(A * A, fft_len))[:2 * n - 1]
    autoconv = np.maximum(autoconv, 0.0)
    return float(2.0 * n * np.max(autoconv))


def powerlaw_sequence(n, p):
    """Create a normalized power-law sequence."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, p)
    a /= np.sum(a)
    return a


def gradient_step(a, beta_sm):
    """
    Compute one gradient step using soft-max smoothing of the max auto-convolution.
    Returns gradient in log-space (t-space where a = softmax(t)).
    """
    n = len(a)
    # Compute auto-convolution
    fft_len = next_fast_len(2 * n - 1)
    A = fft(a, fft_len)
    autoconv = np.real(ifft(A * A, fft_len))[:2 * n - 1]
    autoconv = np.maximum(autoconv, 0.0)

    peak_val = np.max(autoconv)

    # Soft-max weights over auto-conv values
    shifted = autoconv - peak_val
    exp_s = np.exp(beta_sm * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        p_w = np.zeros(2 * n - 1)
        p_w[np.argmax(autoconv)] = 1.0
    else:
        p_w = exp_s / total

    # Compute g[j] = 2*beta * sum_k p_w[k] * a[k-j] via FFT correlation
    a_flip = a[::-1]
    conv_len = next_fast_len(3 * n)
    P = fft(p_w, conv_len)
    AF = fft(a_flip, conv_len)
    conv_res = np.real(ifft(P * AF, conv_len))
    g = 2.0 * beta_sm * conv_res[n - 1:n - 1 + n]

    # Gradient in t-space (log-space): grad_t = a * (g - a . g)
    dot_ga = np.dot(g, a)
    grad_t = a * (g - dot_ga)

    return grad_t, peak_val


def optimize_sequence(a_init, n, time_limit, beta_sm_start=5.0):
    """
    Optimize using Adam in log-space with annealing soft-max parameter.
    Returns the best sequence found.
    """
    deadline = time.monotonic() + time_limit
    t = np.log(a_init + 1e-300)
    a = a_init.copy()

    best_score = compute_score(a)
    best_a = a.copy()

    m_t = np.zeros(n)
    v_t = np.zeros(n)
    step = 0
    lr = 1.0
    beta_sm = beta_sm_start

    while time.monotonic() < deadline - 2.0:
        step += 1
        grad_t, peak_val = gradient_step(a, beta_sm)

        # Adam update in t-space
        b1, b2, eps = 0.85, 0.999, 1e-8
        m_t = b1 * m_t + (1 - b1) * grad_t
        v_t = b2 * v_t + (1 - b2) * grad_t ** 2
        m_hat = m_t / (1.0 - b1 ** step)
        v_hat = v_t / (1.0 - b2 ** step)
        t -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back to a-space (softmax)
        a = np.exp(t - np.max(t))
        a /= np.sum(a)
        a = np.maximum(a, 0.0)
        a /= np.sum(a)

        # Periodic score check
        if step % 100 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"  n={n}, step={step}, score={score:.6f}")

        # Anneal: increase beta_sm, decay lr
        if step % 300 == 0:
            beta_sm *= 1.8
            lr *= 0.85

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score

            if time.monotonic() > total_deadline - 120:
                break
            a = powerlaw_sequence(n, p)
            score = compute_score(a)
            if score < global_best_score:
                global_best_score = score
                global_best_a = a.copy()
                print(f"  p={p}, n={n}: score={score:.6f}")

    print(f"  Best power-law score: {global_best_score:.6f}")

    # Phase 2: Gradient optimization from best starting points
    print("Phase 2: Gradient optimization...")
    # Pick the top diverse starting points
    start_points = []
    for p in [2, 3, 4, 5, 8]:
        for n in [512, 1024, 2048]:
            a = powerlaw_sequence(n, p)
            start_points.append((a, p, n))

    # Also add the best power-law found
    if global_best_a is not None:
        start_points.append((global_best_a.copy(), -1, len(global_best_a)))

    time_per_opt = (total_deadline - time.monotonic()) / max(1, len(start_points))
    time_per_opt = min(time_per_opt, 120.0)  # cap each optimization

    for idx, (a_init, p, n) in enumerate(start_points):
        if time.monotonic() > total_deadline - 30:
            break
        if time.monotonic() + time_per_opt > total_deadline - 10:
            time_per_opt = max(5.0, total_deadline - 10 - time.monotonic())

        print(f"  Optimizing: p={p}, n={n}, time_budget={time_per_opt:.1f}s")
        a_opt, score = optimize_sequence(a_init.copy(), n, time_per_opt)
        if score < global_best_score:
            global_best_score = score
            global_best_a = a_opt.copy()
            print(f"    Improved! score={score:.6f}")

    # Phase 3: Final polish on the best sequence
    if global_best_a is not None and time.monotonic() < total_deadline - 60:
        remaining = total_deadline - 60 - time.monotonic()
        n_best = len(global_best_a)
        print(f"Phase 3: Final polish, n={n_best}, time={remaining:.1f}s")
        a_final, score = optimize_sequence(global_best_a.copy(), n_best, remaining)
        if score < global_best_score:
            global_best_score = score
            global_best_a = a_final.copy()
            print(f"  Final improvement! score={score:.6f}")

    print(f"\nFinal best score: {global_best_score:.6f}")

    if global_best_a is None:
        return [1.0] * 10

    return [float(x) for x in global_best_a]
