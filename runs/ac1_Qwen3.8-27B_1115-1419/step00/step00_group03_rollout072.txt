    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

    M = r.max()

LP-iterative + local search minimization of 2n*max(conv(a,a))/sum(a)^2.
Uses iterated LP for cross-correlation shaping and local refinement.
"""


def build_conv_mat(a):
    """Build (2n-1) x n matrix T where T @ x = np.convolve(a, x)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def evaluate_fast(a):
    """Fast local evaluation matching evaluate_sequence logic."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n < 1:
        return np.inf
    a = np.maximum(0.0, a)
    a = np.minimum(a, 1000.0)
    S = a.sum()
    if S < 0.01:
        return np.inf
    r = np.convolve(a, a)
    M = float(r.max())
    return 2.0 * n * M / (S * S)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: LP iterations followed by local refinement."""
    rng = np.random.default_rng(seed)
    t_end = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # --- Initialization: try several starting points ---
    n = 2535
    inits = []

    try:
        c = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(c) > 1:
            inits.append(c.copy())
    except Exception:
        pass

    j = np.arange(n, dtype=np.float64)
    for r in [0.9995, 0.999, 0.998, 0.995, 0.99]:
        inits.append(r ** (n - 1 - j)).copy()
    inits.append((1.0 + 2.0 * j / (n - 1)).copy())
    inits.append((1.0 + 5.0 * j / (n - 1)).copy())
    inits.append((1.0 + 3.0 * (j / (n - 1)) ** 2).copy())
    inits.append((1.0 + 8.0 * (j / (n - 1)) ** 3).copy())
    inits.append(np.exp(3.0 * j / (n - 1))).copy()
    inits.append(np.exp(5.0 * j / (n - 1))).copy())

    best = None
    best_sc = np.inf
    for c in inits:
        sc = evaluate_fast(c)
        if sc < best_sc:
            best_sc = sc
            best = c.copy()

    best_sc = float(evaluate_sequence(best.tolist()))
    print(f"Best init score: {best_sc:.6f}")
    if not np.isfinite(best_sc):
        best = np.exp(3.0 * np.arange(n, dtype=np.float64) / (n - 1))
        best_sc = float(evaluate_sequence(best.tolist()))
        print(f"Reset init: {best_sc:.6f}")

    n = len(best)
    sqrt_2n = np.sqrt(2.0 * n)

    # --- Phase 1: LP iterations ---
    t_blend = 0.12
    prev_M = np.inf
    it = 0
    no_improve = 0
    lp_end = time.monotonic() + 0.72 * (float(budget_s) - 15.0)

    while time.monotonic() < lp_end:
        S = best.sum()
        if S < 1e-10:
            break
        a = best * sqrt_2n / S

        r = np.convolve(a, a)
        M = float(r.max())

        T = build_conv_mat(a)
        b_ub = np.full(2 * n - 1, M)

        t0 = time.monotonic()
        try:
            res = linprog(-np.ones(n), A_ub=T, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs')
        except Exception as e:
            print(f"LP error: {e}")
            t_blend = max(0.02, t_blend * 0.5)
            it += 1
            continue

        dt_lp = time.monotonic() - t0

        if res.success:
            x = np.maximum(0.0, res.x)
            Sx = x.sum()
            if Sx > 1e-10:
                x = x * sqrt_2n / Sx
                new_a = (1.0 - t_blend) * a + t_blend * x
                M_cand = float(np.convolve(new_a, new_a).max())
                if M_cand < M * 1.05:
                    best = new_a
                    new_M = M_cand
                else:
                    new_M = M
                    t_blend = max(0.02, t_blend * 0.7)
            else:
                new_M = M
        else:
            new_M = M
            t_blend = max(0.02, t_blend * 0.7)

        # Check improvement
        S_b = best.sum()
        sc = 2.0 * n * new_M / (S_b * S_b) if S_b > 0.01 else np.inf
        it += 1

        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            no_improve = 0
            if it <= 10 or it % 50 == 0:
                print(f"  LP it={it}: score={best_sc:.6f}")
        else:
            no_improve += 1

        if it % 30 == 0:
            print(f"LP it={it} new_M={new_M:.6f} est_score={sc:.6f} "
                  f"best={best_sc:.6f} t={t_blend:.3f} lp={dt_lp:.1f}s no_imp={no_improve}")

        # Adapt step size
        if new_M >= prev_M:
            t_blend = max(0.01, t_blend * 0.82)
        else:
            t_blend = min(0.35, t_blend * 1.25)
        prev_M = new_M

        # Occasional diversification: small random jump if stuck
        if no_improve > 200 and it % 50 == 0:
            best *= np.exp(rng.normal(0, 0.1, n))
            best = np.maximum(0, best)
            no_improve = 0
            print(f"  Diversify at it={it}")

    print(f"LP done: {it} iters, best={best_sc:.6f}")

    # --- Phase 2: Local refinement ---
    remaining = t_end - time.monotonic()
    if remaining < 10:
        return [float(v) for v in best]

    a = best.copy()
    for scale in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]:
        if time.monotonic() >= t_end:
            break
        ndone = 0
        while time.monotonic() < t_end and ndone < 400:
            w = max(2, int(n * rng.uniform(0.003, 0.25)))
            s = int(rng.integers(0, max(1, n - w)))
            cand = a.copy()
            cand[s:s + w] *= np.exp(rng.normal(0.0, scale, size=w))
            sc = float(evaluate_sequence(cand.tolist()))
            ndone += 1
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand.copy()
                a = best.copy()
                if ndone <= 5 or ndone % 100 == 0:
                    print(f"  Local scale={scale}: score={best_sc:.6f} (try {ndone})")

    # Final check
    final_sc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(final_sc) and final_sc < best_sc:
        best_sc = final_sc
    print(f"FINAL score: {best_sc:.6f}")
    return [float(v) for v in best]
