"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequence a.
Strategy: Smoothed projected gradient descent with adaptive temperature,
multiple diverse restarts, LP-based local refinement, and periodic
length/structure exploration. Targets asymptotic flatness of the
linear self-convolution using asymmetric non-negative sequences.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def compute_conv(a):
    return np.convolve(a.astype(np.float64), a.astype(np.float64), mode='full')


def objective_value(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * conv.max() / (s * s))


def smooth_max_and_weights(conv, T):
    shift = conv.max()
    z = np.clip((conv - shift) * T, -500, 500)
    exp_z = np.exp(z)
    s = exp_z.sum()
    soft = shift + np.log(s) / T
    w = exp_z / s
    return soft, w


def grad_smooth_conv(a, T):
    n = a.size
    conv = compute_conv(a)
    soft, w = smooth_max_and_weights(conv, T)
    nw = w * T  # gradient of soft_max w.r.t. conv elements
    # Part1[m] = sum_j a[j] * nw[j+m]  => convolve(a_rev, nw)[n-1+m]
    # Part2[m] = sum_j a[j] * nw[m-j]  => convolve(a, nw)[m]
    a_rev = a[::-1]
    p1_full = np.convolve(a_rev, nw, mode='full')
    p1 = p1_full[n - 1:2 * n - 1]
    p2_full = np.convolve(a, nw, mode='full')
    p2 = p2_full[:n]
    grad = p1 + p2
    return soft, grad, conv


def norm_step(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def adam_update(params, grads, t, beta1=0.9, beta2=0.999, eps=1e-8):
    m = params['m']
    v = params['v']
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads * grads
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params['m'] = m
    params['v'] = v
    return m_hat / (np.sqrt(v_hat) + eps)


def solve_lp_refinement(f, M, max_iter=50, tol=1e-8):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = f.size
    conv_len = 2 * n - 1
    # Build sparse constraint matrix A (conv_len x n)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # This is a banded matrix
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    # For large n, build the matrix in a smarter way
    # Each row k has at most min(n, k+1, 2n-1-k) nonzeros
    from scipy.sparse import csr_matrix
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(conv_len, M)
    res = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': 30, 'presolve': True})
    if res.status == 0 and res.x is not None:
        return res.x
    return None


def generate_start_structures(n, rng):
    """Generate diverse starting structures of length n."""
    structures = []
    x = np.arange(n, dtype=np.float64)
    # 1. Uniform
    structures.append(np.ones(n))
    # 2. Linear ramp (asymmetric)
    structures.append(1.0 + 0.5 * x / max(1, n - 1))
    # 3. U-shape
    t = x / max(1, n - 1)
    structures.append(0.3 + (t - 0.5) ** 2 * 2.0)
    # 4. Cosine window (symmetric, but starting point)
    structures.append(0.5 + 0.5 * np.cos(np.pi * t))
    # 5. Two bumps
    s = n / 10.0
    structures.append(np.exp(-(x - n * 0.25) ** 2 / (2 * s * s)) +
                      np.exp(-(x - n * 0.75) ** 2 / (2 * s * s)) + 0.1)
    # 6. Sine-modulated
    structures.append(1.0 + 0.4 * np.sin(2 * np.pi * x / n))
    # 7. Quadratic (asymmetric)
    structures.append(0.5 + t * t * 1.5)
    # 8. Random
    structures.append(rng.random(n) + 0.1)
    # 9. Geometric ramp
    structures.append(np.exp(0.5 * t) - np.exp(-0.5 * (1 - t)))
    # 10. Triangle left-shifted
    peak = n * 0.3
    structures.append(np.maximum(0.1, 1.0 - np.abs(x - peak) / (n * 0.4)))
    for s in structures:
        s = np.clip(s, 0.01, None)
        structures[-1] = s / s.sum()
    return structures


def smooth_gradient_descent(a_init, n_iter, T_start, T_end, lr_base, rng, max_steps=5000):
    """Run smoothed projected gradient descent with Adam and temperature annealing."""
    a = a_init.astype(np.float64).copy()
    a = norm_step(a)
    n = a.size
    adam_params = {'m': np.zeros(n), 'v': np.zeros(n)}
    best_a = a.copy()
    best_obj = objective_value(a)

    for i in range(min(n_iter, max_steps)):
        progress = i / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** progress
        lr = lr_base * (0.98 ** (i // 100))  # decay LR periodically

        soft, grad, conv = grad_smooth_conv(a, T)
        # Project gradient: subtract mean to maintain sum=1
        grad_centered = grad - grad.mean()
        adam_grad = adam_update(adam_params, grad_centered, i + 1)

        a_new = a - lr * adam_grad
        a_new = norm_step(a_new)

        obj = objective_value(a_new)
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

        if i % 500 == 0:
            actual_obj = objective_value(a)
            if i % 2000 == 0:
                print(f"    GD iter {i}/{n_iter}, T={T:.1f}, obj={actual_obj:.6f}, best={best_obj:.6f}")

    return best_a, best_obj


def lp_refine(f, max_rounds=30, time_budget=60):
    """LP-based iterative refinement."""
    f = f.astype(np.float64).copy()
    f = norm_step(f)
    start_time = time.monotonic()
    best = f.copy()
    best_obj = objective_value(f)

    for rnd in range(max_rounds):
        if time.monotonic() - start_time > time_budget:
            break
        n = f.size
        conv = compute_conv(f)
        M = conv.max()
        # Solve LP
        g = solve_lp_refinement(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = norm_step(g)
        # Blend
        for t_blend in [0.5, 0.3, 0.7, 0.1]:
            candidate = (1 - t_blend) * f + t_blend * g
            candidate = np.clip(candidate, 0.0, None)
            cs = candidate.sum()
            if cs > 0.01:
                candidate /= cs
                obj = objective_value(candidate)
                if np.isfinite(obj) and obj < best_obj:
                    best_obj = obj
                    best = candidate.copy()
        f = best.copy()
        if rnd % 5 == 0:
            print(f"    LP round {rnd}, obj={best_obj:.6f}")

    return best, best_obj


def random_local_search(a, n_tries=200, rng=None, time_budget=30):
    """Small random perturbations + evaluation."""
    if rng is None:
        rng = np.random.default_rng()
    a = a.astype(np.float64).copy()
    n = a.size
    best = a.copy()
    best_obj = objective_value(a)
    start = time.monotonic()

    for i in range(n_tries):
        if time.monotonic() - start > time_budget:
            break
        cand = best.copy()
        # Random block perturbation
        width = rng.integers(max(1, n // 50), max(2, n // 5))
        start_i = rng.integers(0, max(1, n - width))
        scale = np.exp(rng.normal(0, 0.15))
        cand[start_i:start_i + width] *= scale
        # Occasionally flip symmetry
        if rng.random() < 0.2:
            cand = 0.7 * cand + 0.3 * cand[::-1]
        cand = np.clip(cand, 0.0, None)
        cs = cand.sum()
        if cs < 0.01:
            continue
        cand /= cs
        obj = objective_value(cand)
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best = cand.copy()

    return best, best_obj
