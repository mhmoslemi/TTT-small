```python
def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```python
and final polishing with exact subgradient steps.
Key insight: non-symmetric sequences can beat the Cauchy-Schwarz bound of 2.

import time
from scipy.optimize import minimize


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def smooth_objective(x, tau):
    """LSE-smoothed max of a*a where a = softmax(x)."""
    a = softmax(x)
    c = np.convolve(a, a)
    c_max = c.max()
    c_shifted = c - c_max
    vals = np.exp(c_shifted / tau)
    return tau * np.log(vals.sum()) + c_max


def smooth_gradient(x, tau):
    """Gradient of LSE-smoothed max w.r.t. x (log-parameterization)."""
    a = softmax(x)
    c = np.convolve(a, a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(c_shifted / tau)
    w /= w.sum()

    grad_a = 2.0 * np.correlate(w, a, mode='valid')

    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x

    # Step and project onto simplex
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def run_lbfgs(x, n, max_time=300):
    """Run L-BFGS with decreasing tau schedule. Returns best a found."""
    best_a = softmax(x).copy()
    best_score = float('inf')
    deadline = time.monotonic() + max_time

    taus = [0.08 / n, 0.04 / n, 0.02 / n, 0.01 / n, 0.005 / n]

    for tau in taus:
        if time.monotonic() > deadline:
            break

        def obj(x_):
            return smooth_objective(x_, tau)

        def grad(x_):
            return smooth_gradient(x_, tau)

        try:
            result = minimize(
                obj, x, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-12,
                         'maxfun': 800, 'maxls': 50}
            )
            x = result.x
        except Exception:
            break

        a = softmax(x)
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score, x


def propose_candidate(budget_s=2400, **kwargs):
    """Main search function: finds best sequence minimizing the evaluation."""
    # Get initial sequence
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        a0 = np.ones(256, dtype=np.float64)

    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    best_score = float('inf')
    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial: n={n}, score={best_score:.6f}")

    deadline = time.monotonic() + budget_s - 60
    rng = np.random.default_rng(12345)

    # Generate diverse initializations
    inits = [a0.copy()]

    # Pure random (Dirichlet)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Cosine-modulated
    for freq in [2, 3, 5]:
        x = np.ones(n)
        for i in range(n):
            x[i] = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * i / n)
        x = np.maximum(x, 1e-10)
        inits.append(x / x.sum())

    # Asymmetric linear ramp (breaks symmetry!)
    x = np.linspace(2.0, 0.3, n)
    inits.append(x / x.sum())

    # Exponential decay
    x = np.exp(-np.linspace(0, 3, n))
    inits.append(x / x.sum())

    # Two-block: heavy left, light right
    x = np.ones(n)
    x[:n // 2] = 1.5
    x[n // 2:] = 0.7
    inits.append(x / x.sum())

    # Random sparse
    for _ in range(2):
        x = np.where(rng.random(n) > 0.4, 1.0, 0.1)
        inits.append(x / x.sum())

    # Triangular (peaked in middle)
    mid = n // 2
    x = 1.0 - np.abs(np.arange(n) - mid) / mid
    x = np.maximum(x, 0.1)
    inits.append(x / x.sum())

    # Inverse triangular (peaked at edges)
    x = 0.3 + np.abs(np.arange(n) - mid) / mid
    inits.append(x / x.sum())

    total_inits = len(inits)
    time_per_init = min(300.0, (deadline - time.monotonic()) / max(1, total_inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > deadline:
            break

        elapsed_init = time.monotonic()
        print(f"\nRestart {idx + 1}/{total_inits}: time_per_init={time_per_init:.0f}s")

        x = np.log(init)

        # Run L-BFGS
        a, score, x = run_lbfgs(x, n, max_time=time_per_init)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  L-BFGS best: {best_score:.6f}")

        # Polish with subgradient steps
        a = a.copy()
        step = 0.03
        polish_deadline = min(
            time.monotonic() + 60,
            deadline
        )
        while time.monotonic() < polish_deadline:
            a = exact_subgradient_step(a, n_active=50, step=step)
            step *= 0.995
            if step < 0.0005:
                break
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()

        print(f"  After polish: {best_score:.6f}")
        # Use current best for next init
        a0 = best_a.copy()

    # Final extended polishing with remaining time
    print(f"\nFinal polishing phase. Best so far: {best_score:.6f}")
    a = best_a.copy()
    step = 0.02
    polish_round = 0
    while time.monotonic() < deadline - 5:
        polish_round += 1
        steps_in_round = 200
        for _ in range(steps_in_round):
            if time.monotonic() > deadline - 5:
                break
            a = exact_subgradient_step(a, n_active=60, step=step)
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            if polish_round % 10 == 1:
                print(f"  Polish round {polish_round}: {best_score:.6f} (step={step:.6f})")
        step *= 0.97
        if step < 0.0001:
            step = 0.005  # Reset for occasional larger moves
            if polish_round % 20 == 0:
                # Try a small random perturbation to escape local min
                a_perturbed = a + rng.normal(0, 0.005, n)
                a_perturbed = np.maximum(a_perturbed, 0)
                a_perturbed = a_perturbed / a_perturbed.sum()
                score_p = evaluate_sequence(a_perturbed.tolist())
                if score_p < best_score:
                    best_score = score_p
                    best_a = a_perturbed.copy()
                    a = a_perturbed.copy()
                    print(f"  Perturbation helped: {best_score:.6f}")

    print(f"\nFinal best score: {best_score:.6f}")
    return [float(val) for val in best_a]
```
