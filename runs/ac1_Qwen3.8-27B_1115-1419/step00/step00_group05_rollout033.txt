The evaluation function computes:
```

```
where `b_sequence = np.convolve(sequence, sequence)` (this is the full convolution, length 2n-1), `max_b = max(b_sequence)`, and `sum_a = sum(sequence)`.

So the objective is to minimize `2n * max(convolve(a,a)) / (sum(a))^2`.

Note that `np.convolve(a, a)` gives the linear convolution (not circular), with length 2n-1. The maximum of this convolution is what we want to minimize relative to the squared sum.

Key observations:
1. The problem is scale-invariant in a specific way. If we scale `a` by factor `c`, then `sum(a)` becomes `c * sum(a)`, `max_b` becomes `c^2 * max_b`, and the objective becomes `2n * c^2 * max_b / (c^2 * sum_a^2)` = same value. So the actual scale doesn't matter, only the relative proportions.

2. The constraint is that all values must be non-negative (the evaluator clips to [0, 1000]).

3. We want to minimize the peak of the autocorrelation relative to the total mass squared. This is related to the "peak-to-average power ratio" (PAPR) problem.

4. If all elements are equal (say all = 1), then `sum_a = n`, `max_b = n` (at the center), and the objective is `2n * n / n^2 = 2`. That matches the previous result of 2.0.

5. To beat 2, we need to make the convolution peak smaller relative to the sum squared. This means we want the sequence to be as "spread out" as possible in its convolution.

Let me think about what sequences would work well:

- Actually, for non-negative sequences, the center of the convolution always has sum(a_i * a_{k-i}) which tends to be large.
- The key insight is that we want to minimize max_k sum_i a_i * a_{k-i} / (sum_i a_i)^2

This is essentially the problem of finding a non-negative sequence whose autocorrelation is as "flat" as possible (minimizes the peak).

For the case where the sequence is uniformly distributed, the peak is at the center and equals sum(a_i^2) when the sequence is constant... wait no.

```python
import numpy as np
import time

def gradient_step(a, tau, lr):
    """One gradient step to minimize the soft-peak objective."""
    n = len(a)
    # Compute linear convolution via FFT
    conv = np.convolve(a, a)  # length 2n-1
    S = np.sum(a)
    
    # Soft max approximation
    # For numerical stability
    shift = tau * conv
    max_shift = np.max(shift)
    exp_vals = np.exp(shift - max_shift)
    log_sum_exp = max_shift + np.log(np.sum(exp_vals))
    M_soft = (1/tau) * log_sum_exp
    
    # weights for each b_k
    weights = exp_vals / np.sum(exp_vals)  # sum to 1
    
    # Gradient of M_soft w.r.t. a_i
    # dM_soft/da_i = sum_k weights[k] * db_k/da_i
    # db_k/da_i = a_{k-i} if 0 <= k-i < n, else 0
    # This is equivalent to the weighted sum of a shifted versions
    
    # More efficiently: grad_i = sum_k weights[k] * a_{k-i} (with bounds)
    # This is a "soft correlation" of weights with a
    grad_M = np.zeros(n)
    for k in range(2*n-1):
        # b_k = sum_i a_i * a_{k-i} where 0<=i<n and 0<=k-i<n
        # i ranges from max(0, k-(n-1)) to min(n-1, k)
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        if lo <= hi:
            # db_k/da_i = a_{k-i} for i in [lo, hi]
            # a_{k-i} for i in [lo, hi] means a[k-hi] to a[k-lo]
            indices = np.arange(k-hi, k-lo+1)
            grad_M[lo:hi+1] += weights[k] * a[indices]
    
    # Gradient of S^(-2) term
    # f = 2n * M / S^2
    # df/da_i = 2n * (1/S^2 * dM/da_i + M * d(S^-2)/da_i)
    # d(S^-2)/da_i = -2/S^3
    # df/da_i = 2n * (grad_M_i / S^2 - 2*M / S^3)
    #           = 2n / S^2 * (grad_M_i - 2*M/S)
    
    grad = (2.0 * n / S**2) * (grad_M - 2.0 * M_soft / S)
    
    # Projected gradient descent (non-negative constraint)
    a_new = np.maximum(0, a - lr * grad)
    return a_new, 2*n*M_soft / S**2
```

```python
import numpy as np
import time
import sys

def _soft_gradient(a, tau):
    """Compute gradient and soft-objective value using FFT."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.zeros(n), np.inf
    
    # Linear convolution via FFT
    L = 1 << (2 * n - 1).bit_length()
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)
    conv = conv[:2 * n - 1]
    # Small numerical fixes
    conv = np.maximum(conv, 0.0)
    
    # Soft max (log-sum-exp for stability)
    shifted = tau * conv
    m = float(np.max(shifted))
    exp_s = np.exp(shifted - m)
    sum_exp = float(np.sum(exp_s))
    M_soft = (m + np.log(sum_exp)) / tau
    weights = exp_s / sum_exp
    
    # Gradient of M_soft via FFT cross-correlation
    # grad_M[i] = sum_j a[j] * weights[j+i] for j=0..n-1
    L2 = 1 << (3 * n - 2).bit_length()
    af2 = np.fft.rfft(a, L2)
    wf2 = np.fft.rfft(weights, L2)
    grad_M = np.fft.irfft(np.conj(af2) * wf2, L2)[:n]
    
    # Full gradient of f = 2n * M_soft / S^2
    grad = (2.0 * n / (S * S)) * (grad_M - 2.0 * M_soft / S)
    obj = 2.0 * n * M_soft / (S * S)
    
    return grad, obj


def _gradient_optimize(a, time_budget, rng, seed_offset=0):

    # Normalize to sum = 1 for numerical stability
    a = a / S0
    
    start_time = time.monotonic()
    tau = 1.0
    lr = 0.01
    prev_obj = _soft_gradient(a, tau)[1]
    best_local = a.copy()
    best_local_obj = prev_obj
    no_improve = 0
    
    step_count = 0
    while time.monotonic() - start_time < time_budget:
        grad, obj = _soft_gradient(a, tau)
        
        if not np.isfinite(obj):
            break
        
        # Backtracking line search
        accepted = False
        for bt in range(15):
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            if np.sum(a_new) < 1e-10:
                lr *= 0.5
                continue
            # Renormalize (since objective is scale-invariant)
            a_new = a_new / np.sum(a_new)
            
            new_obj = _soft_gradient(a_new, tau)[1]
            if new_obj < prev_obj - 1e-12:
                accepted = True
                break
            lr *= 0.5

            # Try a different direction or break

            if no_improve > 100:
                break
            # Restart with higher tau

                tau = min(tau * 1.3, 1000.0)
                lr = 0.01
            continue
        
        # Update
        a = a_new
        prev_obj = new_obj
        no_improve = 0
        step_count += 1
        
        # Anneal tau periodically

            tau = min(tau * 1.5, 500.0)
            lr = max(lr * 0.8, 1e-6)
        
        # Track best
        if new_obj < best_local_obj:
            best_local_obj = new_obj
            best_local = a.copy()
    
    return best_local, best_local_obj


def _lp_step(a, M):
    """
    One LP-based refinement step.
    Solve: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Returns the optimal b or None if LP fails.
    """
    from scipy.optimize import linprog
    
    n = len(a)
    S = float(np.sum(a))
    
    # Build constraint matrix A (dense for simplicity)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Total constraints: 2n-1
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n), dtype=np.float64)
    
    for k in range(num_constraints):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            cols = np.arange(lo, hi + 1)
            A[k, cols] = a[k - cols]
    
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30})
    
    if res.success:
        return res.x
    return None
```
