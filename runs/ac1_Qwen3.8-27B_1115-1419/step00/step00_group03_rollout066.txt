"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using iterative QP-based
optimization with subgradient polishing and multiple restarts.
The QP at each step finds a low-energy direction b satisfying cross-convolution
bounds, then we blend and also try full replacement.

import numpy as np
import cvxpy as cp
import time


def _build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a,b)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            A[k, idx] = a[k - idx]
    return A


def _max_conv(a):
    """Return max of linear convolution of a with itself."""
    return float(np.max(np.convolve(a, a)))


def _solve_qp(a):
    """Solve QP: min ||b||^2 s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    M = _max_conv(a)
    if M < 1e-15:
        return a.copy()
    A = _build_conv_matrix(a)
    b = cp.Variable(n)
    cons = [cp.sum(b) == 1, b >= 0]
    cons.append(A @ b <= M)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(b)), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=10000)
    except:
        return a.copy()
    if b.value is not None and np.all(np.isfinite(b.value)):
        bv = np.asarray(b.value).ravel().astype(np.float64)
        bv = np.clip(bv, 0.0, None)
        s = bv.sum()
        if s > 1e-12:
            return bv / s
    return a.copy()


def _subgrad_step(a, lr):
    """Projected subgradient step: min max(conv(a,a)) on the probability simplex."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    m = np.arange(n)
    valid = (k_star >= m) & (k_star - m < n)
    grad[valid] = 2.0 * a[k_star - m[valid]]
    v = a - lr * grad
    # Project onto simplex: find tau s.t. sum(max(v - tau, 0)) = 1
    lo = v.min() - 1.0
    hi = v.max() + 1.0

        mid = 0.5 * (lo + hi)
        if np.sum(np.maximum(v - mid, 0.0)) > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - 0.5 * (lo + hi), 0.0)

    c = (1.0 - t) * a + t * b

    if s < 1e-15:
