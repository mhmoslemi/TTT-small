import numpy as np

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: Binary (0-1) sequence optimization via simulated annealing with
incremental convolution updates. Multiple restarts with varying n and k (density).
Key insight: binary sequences with well-distributed 1s can achieve J ~ 1.5,
well below the uniform sequence value of 2.0.
"""


def _make_offset(a, p, q, n):
    """Compute the convolution offset when moving a 1 from p to q."""
    lo = max(0, min(p, q))
    hi = min(2 * n - 2, max(p, q) + n - 1)
    s_range = np.arange(lo, hi + 1)
    idx_sq = s_range - q
    mask_sq = (idx_sq >= 0) & (idx_sq <= n - 1)
    a_sq = np.zeros(len(s_range))
    a_sq[mask_sq] = a[idx_sq[mask_sq]]
    idx_sp = s_range - p
    mask_sp = (idx_sp >= 0) & (idx_sp <= n - 1)
    a_sp = np.zeros(len(s_range))
    a_sp[mask_sp] = a[idx_sp[mask_sp]]
    offset = 2.0 * (a_sq - a_sp)
    for s_val in (2 * p, p + q, 2 * q):
        if lo <= s_val <= hi:
            idx = s_val - lo
            if s_val == p + q:
                offset[idx] += -2.0
            else:
                offset[idx] += 1.0
    return offset, lo, hi


def _sa_loop(a, conv, n, k, T0, T_min, cooling, max_steps, rng, deadline, best_state):
    """Run simulated annealing with move-a-1 operations."""
    score = 2.0 * n * conv.max() / (k * k)
    T = T0
    n_accepted = 0
    ones_pos = np.where(a == 1)[0]
    zeros_pos = np.where(a == 0)[0]

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        p = int(ones_pos[rng.integers(len(ones_pos))])
        q = int(zeros_pos[rng.integers(len(zeros_pos))])

        offset, lo, hi = _make_offset(a, p, q, n)
        conv[lo:hi + 1] += offset
        a[p] = 0
        a[q] = 1

        new_score = 2.0 * new_max / (k * k) * n
        delta = new_score - score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            score = new_score
            n_accepted += 1

        else:
            a[q] = 0
            a[p] = 1
            conv[lo:hi + 1] -= offset

        T = max(T * cooling, T_min)

    return best_state

    if score < best_state['score']:
        best_state['score'] = score
        best_state['seq'] = a.copy()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    Uses binary sequences with simulated annealing and multiple restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    best_score = float('inf')
    best_seq = None

    # Configurations: (n, k, T0, cooling, kick_freq, n_kick)
    configs = [
        (256, 100, 0.5, 0.9997, 2000, 8),
        (256, 128, 0.5, 0.9997, 2000, 8),
        (512, 200, 0.4, 0.9997, 2000, 10),
        (512, 256, 0.4, 0.9997, 2000, 10),
        (1024, 400, 0.3, 0.9998, 3000, 15),
        (1024, 512, 0.3, 0.9998, 3000, 15),
        (2048, 800, 0.2, 0.9999, 5000, 20),
        (2048, 1024, 0.2, 0.9999, 5000, 20),
    ]

    remaining_configs = len(configs)
    time_per_config = (deadline - time.monotonic()) / remaining_configs

    for cfg_idx, (n, k, T0, cooling, kick_freq, n_kick) in enumerate(configs):
        if time.monotonic() > deadline - 5:
            break

        n_configs_left = len(configs) - cfg_idx
        time_left = deadline - time.monotonic()
        time_per_restart = time_left / max(1, n_configs_left)
        n_restarts = max(1, int(time_per_restart / 2.0))

        print(f"Config n={n}, k={k}: {n_restarts} restarts, ~{time_per_restart:.1f}s each")

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 2:
                break

            config_deadline = min(deadline - 1, time.monotonic() + time_per_restart - 0.5)

            # Initialize random binary sequence
            a = np.zeros(n, dtype=np.float64)
            a[rng.choice(n, k, replace=False)] = 1.0
            conv = np.convolve(a, a).astype(np.float64)

            score = 2.0 * n * conv.max() / (k * k)
            best_state = {'score': score, 'seq': a.copy()}
            print(f"  Restart {restart}: initial score={score:.6f}")

            # Run multiple SA phases with kicks between
            n_phases = 4
            for phase in range(n_phases):
                if time.monotonic() > config_deadline:
                    break
                phase_deadline = min(config_deadline, time.monotonic() + (config_deadline - time.monotonic()) / (n_phases - phase))
                T_phase = T0 * (0.5 ** phase)
                max_steps = 200000

                _sa_loop(a, conv, n, k, T_phase, 0.01, cooling, max_steps, rng,
                         phase_deadline, best_state)

                if phase < n_phases - 1 and time.monotonic() < config_deadline:
                    _kick(a, conv, n, k, n_kick, rng, best_state)
                    print(f"  Phase {phase+1} done: best={best_state['score']:.6f}")

        print(f"  After config n={n},k={k}: best overall={best_score:.6f}")

    # Final: try to improve the best sequence found with a different approach
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = len(best_seq)
        k_best = int(np.sum(best_seq))
        if k_best > 0:
            print(f"Refining best sequence: n={n_best}, k={k_best}, score={best_score:.6f}")
            a = best_seq.copy()
            conv = np.convolve(a, a).astype(np.float64)
            # Additional SA with lower temperature
            deadline_refine = deadline - 2
            _sa_loop(a, conv, n_best, k_best, 0.05, 0.001, 0.9999, 500000,
                     rng, deadline_refine,
                     {'score': best_score, 'seq': best_seq})

    if best_seq is None:
        # Fallback: uniform sequence
        best_seq = np.ones(256, dtype=np.float64)
        best_score = 2.0

    print(f"FINAL: best_score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
