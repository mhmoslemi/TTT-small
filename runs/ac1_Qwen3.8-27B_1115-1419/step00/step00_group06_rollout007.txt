  from scipy.sparse import lil_matrix, csr_matrix

      out_len = len(a) + len(b) - 1
      N = 1 << (out_len - 1).bit_length()
      A = rfft(a, N)
      B = rfft(b, N)
      return irfft(A * B, N)[:out_len]

      s = float(np.sum(a))
