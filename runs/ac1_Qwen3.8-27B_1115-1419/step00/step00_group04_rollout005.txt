def compute_objective(a):
    """Compute the objective value and its gradient."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')  # length 2n-1
    max_val = np.max(conv)
    sum_val = np.sum(a)
    if sum_val < 0.01:
        return np.inf, None
    obj = 2.0 * n * max_val / (sum_val ** 2)
    
    # Compute gradient
    k_star = np.argmax(conv)
    # dM/da[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0
    grad_M = np.zeros(n)
    # For i in [0, n-1], k_star - i is in [k_star - (n-1), k_star]
    # We need 0 <= k_star - i < n, i.e., k_star - n + 1 <= i <= k_star
    lo = max(0, k_star - n + 1)

    return obj, grad

"""
Step function optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses gradient-based local search with simulated annealing, multiple structured
starting points (chirp, cosine-modulated, random), and adaptive perturbations.
"""

import numpy as np
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    M = conv.max()
    return 2.0 * n * M / (S * S)


def compute_grad(a):
    """Compute gradient of objective w.r.t. a, assuming max at unique lag."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.zeros(n)
    conv = np.convolve(a, a)
    M = conv.max()
    k_star = int(np.argmax(conv))
    # df/da[i] = 2n*(dM/da[i]*S - 2M)/S^3
    # dM/da[i] = 2*a[k*-i] if 0<=k*-i<n, else 0
    grad = np.full(n, -4.0 * n * M / (S ** 3))
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] += 4.0 * n * a[k_star - idx] / (S ** 2)
    return grad


def normalize(a, target_sum):
    """Normalize array to have target_sum, keeping non-negative."""
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / s)


def local_search(a, duration, rng, verbose=False):
    """
    Gradient-based local search with simulated annealing and block perturbations.
    Returns (best_sequence, best_score).
    """
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = np.clip(a.astype(np.float64), 0, 1000).copy()
    a = normalize(a, target_sum)

    best = a.copy()
    best_score = compute_obj(best)

    start_time = time.monotonic()
    lr = 0.001  # start small
    initial_temp = 0.5
    cool_rate = 0.9995
    temp = initial_temp
    no_improve_count = 0
    max_no_improve = 5000

    while time.monotonic() - start_time < duration:
        obj = compute_obj(a)
        grad = compute_grad(a)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            step_size = lr * target_sum / (grad_norm * np.sqrt(n) + 1e-8)
            a_new = a - step_size * grad
        else:
            a_new = a.copy()

        a_new = np.clip(a_new, 0, 1000)
        S_new = a_new.sum()
        if S_new < 0.01:
            a_new = normalize(a, target_sum)
        else:
            a_new = normalize(a_new, target_sum)

        new_score = compute_obj(a_new)

        # Accept/reject
        if new_score < obj:
            a = a_new
            lr = min(lr * 1.01, 0.5)
            if new_score < best_score:
                best = a.copy()
                best_score = new_score
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            # Simulated annealing acceptance
            delta = new_score - obj
            if temp > 1e-4 and rng.random() < np.exp(-delta / temp):
                a = a_new
            # Block perturbation to escape local minima
            if no_improve_count > 200 or rng.random() < 0.05:
                w = int(rng.integers(max(2, n // 80), max(3, n // 8)))
                s = int(rng.integers(0, max(1, n - w)))
                scale = np.exp(rng.normal(0, 0.3, w))
                a[s:s + w] *= scale
                a = np.clip(a, 0, 1000)
                a = normalize(a, target_sum)
            # Larger randomization if stuck
            if no_improve_count > max_no_improve:
                frac = rng.uniform(0.1, 0.4)
                mask = rng.random(n) < frac
                a[mask] = np.abs(rng.normal(1.0, 0.4, mask.sum()))
                a = np.clip(a, 0, 1000)
                a = normalize(a, target_sum)
                no_improve_count = 0

        # Cool down
        temp *= cool_rate
        # Occasionally reheat
        if rng.random() < 0.005 and temp < 0.01:
            temp = 0.1

        if verbose and no_improve_count % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"  [{elapsed:.0f}s] best={best_score:.6f} temp={temp:.4f} lr={lr:.4f}")

    return best, best_score


def make_chirp_start(n, beta, alpha, rng):
    """Create a chirp-modulated starting sequence."""
    i = np.arange(n, dtype=np.float64)
    signal = 1.0 + alpha * np.cos(2.0 * np.pi * beta * i * i / n)
    signal = np.maximum(signal, 0.01)
    # Add small noise for symmetry breaking
    signal += rng.normal(0, 0.02, n)
    signal = np.maximum(signal, 0.0)
    return signal


def make_cosine_start(n, period, alpha, rng):
    """Create a cosine-modulated starting sequence."""
    i = np.arange(n, dtype=np.float64)
    signal = 1.0 + alpha * np.cos(2.0 * np.pi * i / period)
    signal = np.maximum(signal, 0.01)
    signal += rng.normal(0, 0.02, n)
    signal = np.maximum(signal, 0.0)
    return signal


def make_random_start(n, rng):
    """Create a random starting sequence."""
    return np.abs(rng.normal(1.0, 0.3, n)) + 0.05


def make_two_level_start(n, density, rng):
    """Create a two-level starting sequence."""
    base = np.ones(n) * 0.3
    mask = rng.random(n) < density
    base[mask] = 1.0
    base += rng.normal(0, 0.05, n)
    return np.maximum(base, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the objective.
    Uses multiple starting points with gradient-based local search + annealing.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 5.0

    print(f"Starting optimization with budget={total_budget}s")

    # Build list of (name, sequence) starting points
    starts = []

    # 1. From height_sequence_1 if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        hs = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if hs.ndim == 1 and hs.size > 0 and np.all(np.isfinite(hs)):
            starts.append(("height_seq_1", hs))

    # 2. Chirp starts with different parameters
    for n in [2000, 3000]:
        for beta in [0.3, 0.7]:
            for alpha in [0.5, 0.8]:
                starts.append((f"chirp_n{n}_b{beta}_a{alpha}", make_chirp_start(n, beta, alpha, rng)))

    # 3. Cosine starts
    for n in [1500, 2500]:
        for period in [3, 5, 7, 11]:
            starts.append((f"cos_n{n}_p{period}", make_cosine_start(n, period, 0.6, rng)))

    # 4. Random starts
    for n in [1000, 2000, 3000]:
        starts.append((f"rand_n{n}", make_random_start(n, rng)))

    # 5. Two-level starts
    for n in [2000, 3500]:
        for dens in [0.3, 0.5, 0.7]:
            starts.append((f"2lvl_n{n}_d{dens}", make_two_level_start(n, dens, rng)))

    print(f"Number of starting points: {len(starts)}")

    # Phase 1: Quick screening of all starts (short search from each)
    best_overall = None
    best_score_overall = np.inf

    phase1_time = min(60.0, total_budget * 0.08)
    per_start_time = phase1_time / len(starts)
    per_start_time = max(2.0, per_start_time)

    for name, seq in starts:
        remaining = deadline - time.monotonic()
        if remaining < per_start_time + 2:
            break
        alloc = min(per_start_time, remaining - 2)
        best_local, score_local = local_search(seq, alloc, rng)
        if score_local < best_score_overall:
            best_score_overall = score_local
            best_overall = best_local.copy()
            print(f"  {name} (n={len(seq)}): {score_local:.6f}")

    print(f"Phase 1 best: {best_score_overall:.6f}")

    # Phase 2: Extended search from the best and a few close ones
    phase2_time = min(total_budget * 0.5, 1200.0)
    if best_overall is not None and deadline - time.monotonic() > phase2_time * 0.3:
        alloc2 = min(phase2_time, (deadline - time.monotonic()) * 0.5)
        best_overall, best_score_overall = local_search(
            best_overall, alloc2, rng, verbose=True)
        print(f"Phase 2 best: {best_score_overall:.6f}")

    # Phase 3: Try a few alternative starts if time permits
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Try chirp with n close to best length
        n_try = len(best_overall) if best_overall is not None else 2000
        for beta in [0.2, 0.5, 1.0, 1.5]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc3 = min(remaining * 0.3, 120.0)
            seq3 = make_chirp_start(n_try, beta, 0.7, rng)
            best3, score3 = local_search(seq3, alloc3, rng)
            if score3 < best_score_overall:
                best_score_overall = score3
                best_overall = best3.copy()
                print(f"  Phase 3 chirp b={beta}: {score3:.6f}")

    # Phase 4: Final polish on the best
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_overall is not None:
        alloc4 = max(5.0, remaining - 5)
        best_overall, best_score_overall = local_search(
            best_overall, alloc4, rng)
        print(f"Phase 4 final: {best_score_overall:.6f}")

    # Final verification with evaluate_sequence
    if best_overall is not None:
        final_seq = np.clip(best_overall, 0, 1000)
        # Ensure all values are valid
        final_seq = np.nan_to_num(final_seq, nan=0.0, posinf=1000.0, neginf=0.0)
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"Verified score from evaluate_sequence: {final_score:.6f}")
        print(f"Sequence length: {len(final_seq)}, sum: {final_seq.sum():.4f}, max: {final_seq.max():.4f}")

        if np.isfinite(final_score):
            return [float(x) for x in final_seq]

    # Fallback: return a valid sequence
    fallback = np.ones(100, dtype=np.float64)
    fb_score = evaluate_sequence(fallback.tolist())
    if np.isfinite(fb_score):
        print(f"Fallback score: {fb_score:.6f}")
        return [float(x) for x in fallback]

    return [1.0] * 10
