"""
Minimize step function score 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-accelerated gradient descent in log-space (softmax parameterization)
- Soft-max approximation of the peak autocorrelation
- Multiple initializations and sequence lengths
- LP-based directional updates for escaping local minima
"""
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    out_len = len(a) + len(b) - 1
    N = 1 << (out_len - 1).bit_length()
    A = rfft(a, N)
    B = rfft(b, N)
    return irfft(A * B, N)[:out_len]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fft_convolve(a, a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def softmax_from_b(b):
    """Compute a = softmax(b) stably."""
    b_shifted = b - np.max(b)
    e = np.exp(b_shifted)
    a = e / np.sum(e)
    return a


def gradient_of_softmax_peak(a, lam):
    """
    Compute gradient of (1/lam)*log(sum exp(lam*c[k])) w.r.t. a.
    c[k] = conv(a,a)[k], grad[m] = 2*sum_j w[m+j]*a[j]
    where w = softmax(lam*(c - max_c))
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_shifted = c - np.max(c)
    c_shifted = np.clip(c_shifted, -100.0 / max(lam, 1e-6), 0.0)
    w = np.exp(lam * c_shifted)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    # grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # = 2 * conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_val = fft_convolve(a_rev, w)
    grad = 2.0 * conv_val[n - 1:2 * n - 1]
    return grad, c


def gradient_step_logspace(b, lam, eta, deadline):
    """
    One gradient step in log-space.
    b: log-parameterization vector
    Returns: new b, a, score, max_c
    """
    a = softmax_from_b(b)
    n = len(a)
    grad_a, c = gradient_of_softmax_peak(a, lam)
    max_c = float(np.max(c))
    # grad_b[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    mean_grad = float(np.dot(a, grad_a))
    grad_b = a * (grad_a - mean_grad)
    # Adaptive step size
    norm_gb = float(np.linalg.norm(grad_b))
    if norm_gb > 1e-15:
        b_new = b - eta * grad_b / (norm_gb + 1e-10)
    else:
        b_new = b.copy()
    return b_new, a, max_c


def run_gradient_descent(b_init, n, lam_start, lam_end, eta_start, steps, deadline, label=""):
    """Run gradient descent in log-space with lambda annealing."""
    best_a = softmax_from_b(b_init)
    best_score = compute_score(best_a)
    best_max_c = float(np.max(fft_convolve(best_a, best_a)))
    b = b_init.copy()
    eta = eta_start
    t0 = time.monotonic()

    for step in range(steps):
        if time.monotonic() > deadline:
            break
        # Anneal lambda
        frac = step / max(steps - 1, 1)
        lam = lam_start + (lam_end - lam_start) * frac
        # Decrease learning rate
        eta = eta_start * (0.999 ** step)

        b, a, max_c = gradient_step_logspace(b, lam, eta, deadline)
        score = 2.0 * n * max_c  # since sum(a)=1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_max_c = max_c

        if step % 500 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"  [{label}] step={step}, score={best_score:.6f}, elapsed={elapsed:.1f}s")

    final_score = compute_score(best_a)
    print(f"  [{label}] final: score={final_score:.6f}, max_c={best_max_c:.8f}, n={n}")
    return best_a, final_score


def lp_refine(a, n_iterations, deadline, label=""):
    """
    LP-based refinement: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Mix solution with current a.
    """
    n = len(a)
    if n > 4000:
        print(f"  [{label}] LP skipped (n={n} too large)")
        return a, compute_score(a)

    best_a = a.copy()
    S = float(np.sum(a))
    best_score = compute_score(best_a)

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break
        c = fft_convolve(best_a, best_a)
        M = float(np.max(c))

        # Build constraint matrix A (2n-1 x n)
        t_build = time.monotonic()
        A = lil_matrix((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                A.row[k, j_lo:j_hi + 1] = best_a[j_lo:j_hi + 1]
        A = A.tocsr()
        print(f"  [{label}] LP it={it}, build={time.monotonic()-t_build:.2f}s, M={M:.8f}")

        # Solve LP
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        t_lp = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tol': 1e-7})
        lp_time = time.monotonic() - t_lp
        print(f"  [{label}] LP solve={lp_time:.2f}s, success={res.success}")

        if not res.success or res.x is None:
            break

        g0 = np.maximum(res.x, 0.0)
        sum_g0 = float(np.sum(g0))
        if sum_g0 < 1e-10:
            break

        # Rescale to match sum
        g = (S / sum_g0) * g0

        # Mix
        t_mix = 0.1
        a_new = (1.0 - t_mix) * best_a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            continue
        a_new /= s_new

        score_new = compute_score(a_new)
        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            best_a = a_new

        # Adaptive mixing: if score went up, reduce t
        if score_new > best_score * 1.001:
            t_mix *= 0.5

    return best_a, best_score
