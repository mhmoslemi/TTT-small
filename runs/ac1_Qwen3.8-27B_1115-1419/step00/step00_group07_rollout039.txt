def fast_cross_correlate(w, a):
    """Compute c_k = sum_n w[n+k] * a[n] for k = 0, ..., len(a)-1."""
    n = len(a)
    N = 1
    while N < len(w) + n - 1:
        N *= 2
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
