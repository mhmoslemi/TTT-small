def _fft_conv(a):
    n = a.size
    if n == 0: return np.array([], dtype=float)
    L = 2*n - 1
    N = _pow2_needed(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:L]
    # Remove roundoff negative values
    np.maximum(c, 0.0, out=c)
    return c

"""
Search for non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The implementation combines:
- multi-start initialization,
- short-cut surrogate optimization,
- smoothed gradient descent with log-sum-exp,
- exact sum-preserving swap local search with active-lag guidance,
- simulated annealing / random kicks to escape local minima.
"""

import numpy as np
import time
import math


def _pow2_needed(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _norm_nonneg(x):
    """Clip to finite, non-negative values and normalize sum to 1."""
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return None
    if not np.all(np.isfinite(arr)):
        arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-300:
        arr = np.ones_like(arr, dtype=np.float64)
        return arr / arr.size
    return arr / s


def _fft_conv(a):
    """Non-negative linear convolution by FFT."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    N = _pow2_needed(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _fast_score(a):
    """Fast scale-invariant objective using FFT convolution."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.inf
    s = float(arr.sum())
    if s <= 1e-300:
        return np.inf
    c = _fft_conv(arr)
    mx = float(c.max()) if c.size else 0.0
    return float(2.0 * n * mx / (s * s))


def _soft_weights(c, temp_factor, n):
    """Soft-max weights over convolution coefficients."""
    T = max(1e-12, float(temp_factor) / max(1, int(n)))
    mc = float(c.max())
    if mc <= 0.0:
        w = np.ones(c.size, dtype=np.float64)
        return w / w.size
    z = (c - mc) / T
    ex = np.exp(z)
    ss = float(ex.sum())
    if ss <= 0.0 or not math.isfinite(ss):
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / ss


def _grad_from_weights(a, w):
    """
    Gradient of soft-max(conv(a,a), T) with respect to a, given soft weights w.
    Returns g where g[j] = 2 * sum_m a[m] * w[m+j].
    """
    n = a.size
    M = _pow2_needed(3 * n - 1)
    fa = np.fft.rfft(a[::-1], M)
    fw = np.fft.rfft(w, M)
    conv = np.fft.irfft(fa * fw, M)
    return 2.0 * conv[n - 1:2 * n - 1]


def _accept_prob(diff, temp):
    """Simulated annealing acceptance probability."""
    if diff <= 0.0:
        return 1.0
    if temp <= 1e-300:
        return 0.0
    if diff / temp > 50.0:
        return 0.0
    return float(math.exp(-diff / temp))


def _random_kick(a, rng, strength):
    """Random multiplicative/additive perturbation."""
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return arr

    nblk = int(rng.integers(1, max(2, n // 80 + 1)))
    for _ in range(nblk):
        wmin = max(1, n // 500)
        wmax = max(wmin + 1, n // 8)
        width = int(rng.integers(wmin, wmax))
        start = int(rng.integers(0, n - width))
        arr[start:start + width] *= float(np.exp(rng.normal(0.0, strength, width)))

    if rng.random() < 0.35:
        mask = rng.random(n) < 0.04
        if mask.any():
            k = int(mask.sum())
            arr[mask] += rng.normal(0.0, strength / max(1, k) ** 0.5, k)

    out = _norm_nonneg(arr)
    return arr if out is None else out


def _apply_swap_c(c, a, i, j, delta):
    """
    Obtain new convolution after moving mass delta from i to j:
        a[i] -= delta, a[j] += delta.
    This uses O(n) updates instead of a full FFT.
    """
    cc = c.copy()
    if delta == 0.0:
        return cc
    n = a.size
    two_delta = 2.0 * delta

    cc[j:j + n] += two_delta * a
    cc[i:i + n] -= two_delta * a

    d2 = delta * delta
    cc[2 * j] += d2
    cc[2 * i] += d2
    cc[i + j] -= 2.0 * d2
    return cc


def _gradient_refine(a0, deadline, rng):
    """
    Smoothed gradient descent on sum-normalized non-negative sequence.
    Returns (best_sequence, best_fast_score).
    """
    a = _norm_nonneg(a0)
    if a is None:
        return np.ones(256, dtype=np.float64) / 256.0, np.inf

    n = a.size
    score = _fast_score(a)
    best = a.copy()
    best_score = score

    alpha = 0.08
    sa_temp = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        c = _fft_conv(a)
        s = float(a.sum())
        score = float(2.0 * n * c.max() / (s * s)) if s > 0 else np.inf
        it += 1

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            no_improve = 0
            sa_temp = max(1e-4, sa_temp * 0.985)
        else:
            no_improve += 1

        temp_factor = max(1e-4, 5.0 * math.exp(-0.008 * it))
        w = _soft_weights(c, temp_factor, n)
        g = _grad_from_weights(a, w)

        if not np.all(np.isfinite(g)):
            break

        gp = g - float(np.mean(g))
        gn = float(np.linalg.norm(gp))
        if not math.isfinite(gn) or gn < 1e-300:
            break

        a_norm = float(np.linalg.norm(a)) + 1e-300
        step = alpha * a_norm / gn
        accepted = False

        for frac in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = a - step * frac * gp
            if cand.min() < 0.0:
                if frac >= 0.5:
                    continue
                cand = np.maximum(cand, 0.0)

            cand = _norm_nonneg(cand)
            if cand is None:
                continue

            sc = _fast_score(cand)
            if not math.isfinite(sc):
                continue

            if sc < score - 1e-12:
                a = cand
                score = sc
                accepted = True
                alpha = min(0.25, alpha * 1.05)
                break

            if _accept_prob(sc - score, sa_temp) > rng.random():
                a = cand
                score = sc
                accepted = True
                break

        if not accepted:
            alpha = max(0.003, alpha * 0.85)
            if no_improve >= 200:
                a = _random_kick(a, rng, 0.8)
                score = _fast_score(a)
                no_improve = 0
                alpha = 0.05
            if no_improve > 3000:
                break

    best = np.maximum(best, 0.0)
    bs = float(best.sum())
    if bs > 0.0:
        best = best / bs
    final_score = _fast_score(best)
    if math.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    return best, best_score

def _resample_to(x, n):
    """Resample a normalized sequence to length n."""
    x = _norm_nonneg(x)
    if x is None:
        return np.ones(n, dtype=np.float64) / max(1, n)
    m = x.size
    if m == n:
        return x.copy()
    if m < 2 or n < 2:
        out = np.repeat(x, max(1, n // max(1, m)))[:n].copy()
        if out.size < n:
            out = np.pad(out, (0, n - out.size), mode="edge")
        return out
    fp = np.linspace(0.0, 1.0, n, dtype=np.float64)
    xp = np.linspace(0.0, 1.0, m, dtype=np.float64)
    out = np.interp(fp, xp, x)
    res = _norm_nonneg(out)
    return out if res is None else res


def _generate_starts(n, rng):
    """Generate a diverse list of initial non-negative normalized sequences."""
    n = max(1, int(n))
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Smooth random AR(1)-like sequence.
    x = np.zeros(n, dtype=np.float64)
    x[0] = 1.0
    for t in range(1, n):
        x[t] = 0.96 * x[t - 1] + rng.normal(0.0, 0.35)
    starts.append(np.abs(x) + 1e-3)

    # Random positive.
    starts.append(0.2 + rng.random(n))

    # Heavy-tailed random.
    starts.append(np.abs(rng.normal(0.0, 1.0, n)) ** 1.5 + 1e-3)

    # Two-level random.
    r = rng.random(n)
    starts.append(np.where(r < 0.5, 0.25 + 0.1 * rng.random(n), 1.0 + 0.1 * rng.random(n)))

    # Parametric sinusoidal families.
    for p in (1, 2, 3, 5):
        for _ in range(2):
            A = float(rng.uniform(0.25, 1.0))
            phi = float(rng.uniform(0.0, 2.0 * np.pi))
            x = 1.0 + A * np.sin(2.0 * np.pi * p * i / n + phi)
            if rng.random() < 0.5:
                q = p + int(rng.integers(1, 4))
                B = float(rng.uniform(0.0, 0.7))
                psi = float(rng.uniform(0.0, 2.0 * np.pi))
                x += B * np.sin(2.0 * np.pi * q * i / n + psi)
            x = np.maximum(x, 0.0)
            if x.sum() > 0.0:
                starts.append(x)

    # Low-dimensional smooth asymmetric trials.
    denom = max(1, n - 1)
    starts.append(1.0 + 0.9 * np.sin(np.pi * i / denom))
    starts.append(1.0 + 0.9 * np.cos(np.pi * i / denom))

    out = []
    for x in starts:
        ax = _norm_nonneg(x)
        if ax is not None:
            out.append(ax)
    return out


def _optimize_start(a0, deadline, rng):
    """Optimize one starting sequence until deadline."""
    a = _norm_nonneg(a0)
    if a is None:
        return None, np.inf

    rem = max(0.0, deadline - time.monotonic())
    if rem < 1.0:
        return a, _fast_score(a)

    grad_deadline = time.monotonic() + min(20.0, 0.20 * rem)
    if grad_deadline > time.monotonic() + 1.0:
        a, _ = _gradient_refine(a, grad_deadline, rng)

    a, sc = _swap_search(a, deadline, rng)
    return a, sc
