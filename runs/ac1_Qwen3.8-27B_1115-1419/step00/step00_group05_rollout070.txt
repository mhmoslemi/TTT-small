    time_remaining = total_deadline - time.monotonic()
    time_per_refine = max(60.0, time_remaining / (num_refine + 2))

    for rank, (init_sc, idx, ai) in enumerate(top_inits):
        if time.monotonic() > total_deadline - 30:
            break
        alloc = min(time_per_refine, total_deadline - time.monotonic() - 20)
        if alloc < 30:
            break
        d = time.monotonic() + alloc
        print(f"  Refining init {idx} (fast_score={init_sc:.6f}), budget={alloc:.0f}s")

        a_opt, opt_sc = gradient_descent(ai, d, p_power=4, verbose=True)
        actual = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(actual) and actual < best_score:
            best = a_opt.copy()
            best_score = actual
            print(f"  New best: {best_score:.6f}")

    # Phase 3: Further refinement from the best found
    if time.monotonic() < total_deadline - 60:
        remaining = total_deadline - time.monotonic() - 10
        print(f"  Final refinement, budget={remaining:.0f}s")
        a_final, _ = gradient_descent(best, total_deadline - 10, p_power=4, verbose=True)
        actual = float(evaluate_sequence(a_final.tolist()))
        if np.isfinite(actual) and actual < best_score:
            best = a_final.copy()
            best_score = actual
            print(f"  Final best: {best_score:.6f}")

    # Ensure validity
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n) / n

    final = float(evaluate_sequence([float(x) for x in best]))
    print(f"FINAL score: {final:.6f}")
    return [float(x) for x in best]
