import numpy as np
import time

def make_geometric(n, r):
    """Geometric sequence normalized to sum 1."""
    j = np.arange(n, dtype=np.float64)
    a = r ** j
    a /= a.sum()
    return a

def make_linear_tilt(n, alpha):
    """Sequence a[j] = exp(alpha * j / n) normalized."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * j / n)
    a /= a.sum()
    return a

def gradient_at_lag(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    m_lo = max(0, k - n + 1)
    m_hi = min(n - 1, k)
    if m_lo > m_hi:
        return grad
    idx = np.arange(m_lo, m_hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def max_conv_and_grad(a):
    """Compute max of self-convolution and subgradient."""
    c = np.convolve(a, a, mode='full')
    max_val = c.max()
    # Find all max indices
    max_idx = np.where(np.isclose(c, max_val, atol=1e-14))[0]
    if len(max_idx) == 0:
        max_idx = np.array([np.argmax(c)])
    grad = np.zeros(len(a))
    for k in max_idx:
        grad += gradient_at_lag(a, k)
    grad /= len(max_idx)
    return max_val, grad

def projected_step(a, grad, lr):
    """Take a gradient step and project to {a >= 0, sum = 1}."""
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, 1.0  # No change
    a_new /= s
    return a_new, 0.0

def optimize_sequence(a_init, time_limit, verbose=False):
    """Gradient descent with line search to minimize max(a*a)."""
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_val = 2.0 * n * np.convolve(a, a, mode='full').max()
    
    lr = 0.1
    stagnation = 0
    
    while time.monotonic() < time_limit:
        max_val, grad = max_conv_and_grad(a)
        current_obj = 2.0 * n * max_val
        
        # Line search
        best_lr = 0
        best_new_obj = current_obj
        for trial_lr in [lr * 0.25, lr * 0.5, lr, lr * 2, lr * 4]:
            if trial_lr < 1e-10:
                continue
            a_trial, _ = projected_step(a, grad, trial_lr)
            trial_obj = 2.0 * n * np.convolve(a_trial, a_trial, mode='full').max()
            if trial_obj < best_new_obj:
                best_new_obj = trial_obj
                best_lr = trial_lr
                a_best_trial = a_trial.copy()
        
        if best_lr > 0:
            a = a_best_trial
            lr = best_lr
            if best_new_obj < best_val - 1e-10:
                best_val = best_new_obj
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            lr *= 0.5
            stagnation += 1
        
        if stagnation > 200:
            # Try a random perturbation
            if verbose:
                print(f"  Stagnated, perturbing...")
            a = a + 0.01 * np.random.dirichlet(np.ones(n))
            a /= a.sum()
            stagnation = 0
            lr = 0.1
        
        if lr < 1e-10:
            break
    
    return best_a, best_val
