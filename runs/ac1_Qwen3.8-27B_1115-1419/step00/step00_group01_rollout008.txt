import numpy as np
import time

def compute_convolution(a):
    """Compute linear convolution using FFT for efficiency."""
    n = len(a)
    # Use FFT length that's a power of 2 for efficiency
    fft_len = 1
    while fft_len < 2*n - 1:
        fft_len *= 2
    fa = np.fft.rfft(a, fft_len)
    fc = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return fc

def objective_from_conv(c, n, S):
    """Compute objective from precomputed convolution."""
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Initialize from given sequence or uniform
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
    except:
        parent = np.ones(256)
    
    n = len(parent)
    # Normalize
    S = np.sum(parent)
    if S < 0.01:
        parent = np.ones(256)
        n = len(parent)
        S = np.sum(parent)
    parent = parent / S  # Normalize to sum 1
    
    best = parent.copy()
    best_c = compute_convolution(best)
    best_max_c = np.max(best_c)
    best_score = 2.0 * n * best_max_c  # Since S=1, obj = 2n*max_c
    
    # Optimization loop
    step_size = 0.01
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        a = best.copy()
        S_a = np.sum(a)
        c = compute_convolution(a)
        max_c = np.max(c)
        
        # Find peak index
        peak_idx = np.argmax(c)
        
        # Compute gradient of c[peak_idx] w.r.t. a
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            ki = peak_idx - j
            kj = j - peak_idx
            if 0 <= ki < n:
                val += a[ki]
            if 0 <= kj < n:
                val += a[kj]
            grad[j] = val
        
        # Update: move in direction that reduces c[peak_idx]
        # a_new = a - step * grad / S_a (to maintain normalization)
        # But we need to project to a >= 0
        
        # Actually, since the objective is scale-invariant, we don't need to maintain sum=1
        # We just need to reduce max_c / S^2
        # Gradient of max_c / S^2 w.r.t. a[i]:
        # = (grad_c[i] * S^2 - max_c * 2 * S) / S^4
        # = grad_c[i] / S^2 - 2*max_c / S^3
        # = (grad_c[i]*S - 2*max_c) / S^3
        
        grad_obj = (grad * S_a - 2.0 * max_c) / (S_a ** 3)
        
        a_new = a - step_size * grad_obj
        a_new = np.clip(a_new, 0, 1000)
        
        if np.sum(a_new) < 0.01:
            continue
        
        c_new = compute_convolution(a_new)
        max_c_new = np.max(c_new)
        S_new = np.sum(a_new)
        score_new = 2.0 * n * max_c_new / (S_new * S_new)
        
        if score_new < best_score:
            best = a_new
            best_score = score_new
            # Maybe increase step size
            step_size = min(step_size * 1.01, 0.1)
        else:
            step_size = max(step_size * 0.95, 0.001)
        
        # Random perturbation
        if it % 50 == 0:
            a_new = best * np.exp(rng.normal(0, 0.05, n))
            a_new = np.clip(a_new, 0, 1000)
            if np.sum(a_new) >= 0.01:

                score_new = 2.0 * n * np.max(c_new) / (np.sum(a_new)**2)
                if score_new < best_score:
                    best = a_new
                    best_score = score_new
    
    # Return best (must have sum >= 0.01)
    result = best.copy()
    S = np.sum(result)
    if S < 0.01:
        result = np.ones(256)
    return [float(x) for x in result]

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses gradient descent on the active peak of the autocorrelation, with
adaptive step size, randomization, and multiple restarts.
"""

import numpy as np
import time

def get_grad(k, a, n):
    """GPU-free vectorized gradient of c[k] w.r.t. a, where c=conv(a,a) and a>=0."""
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] = 2.0 * a[idx]
    return grad

def fast_convolve(a):
    """Linear convolution via FFT for speed on larger sequences."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def score(a, n):
    """Compute objective value (assumes a is already clipped to [0,1000])."""
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * np.max(c) / (S * S)

def optimize_from(a0, n, deadline, rng, label=""):
    """Run gradient descent from initial point a0 (sum-normalized to 1)."""
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-15:
        return None, None
    a /= S  # normalize to sum 1
    
    c = fast_convolve(a)
    max_c = np.max(c)
    best_a = a.copy()
    best_score = 2.0 * n * max_c  # S=1
    best_c = c.copy()
    best_max_c = max_c
    
    lr = 0.1
    no_improve = 0
    it = 0
    
    print(f"  {label}: start score={best_score:.6f}")
    
    while time.monotonic() < deadline:
        it += 1
        
        # Find peak(s)
        k = int(np.argmax(best_c))
        
        # Multi-peak: also consider nearby peaks
        threshold = best_max_c * 0.999
        peak_indices = np.where(best_c >= threshold)[0]
        
        if len(peak_indices) == 1:
            g = get_grad(k, a, n)
        else:
            # Average gradient over all near-peak indices
            g = np.zeros(n, dtype=np.float64)
            for pk in peak_indices[:10]:  # Limit to top 10
                g += get_grad(int(pk), a, n)
            g /= len(peak_indices[:10])
        
        # Step
        a_new = a - lr * g
        a_new = np.clip(a_new, 0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-15:
            lr *= 0.5
            continue
        a_new /= S_new
        
        c_new = fast_convolve(a_new)
        max_c_new = np.max(c_new)
        score_new = 2.0 * n * max_c_new
        
        if score_new < best_score - 1e-12:
            a = a_new
            c = c_new
            max_c = max_c_new
            best_a = a.copy()
            best_c = c.copy()
            best_max_c = max_c_new
            best_score = score_new
            no_improve = 0
            lr = min(lr * 1.01, 0.5)
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 0.001:
                lr = 0.01  # reset
                # Random kick

                    kick = np.clip(rng.normal(0, 0.05, n), -0.5, 0.5)
                    a_new = a * (1.0 + kick)
                    a_new = np.clip(a_new, 0, None)
                    S_new = np.sum(a_new)
                    if S_new > 1e-15:
                        a_new /= S_new
                        c_new = fast_convolve(a_new)
                        score_new = 2.0 * n * np.max(c_new)
                        if score_new < best_score:

                            c = c_new
                            best_a = a.copy()
                            best_c = c.copy()
                            best_max_c = np.max(c_new)
                            best_score = score_new
                            no_improve = 0
                no_improve = 0
        
        if it % 5000 == 0:
            print(f"  {label} iter {it}: score={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score
