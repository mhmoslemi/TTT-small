from scipy.optimize import linprog

def lp_improve(a, M=None):
    """Solve LP to find b that reduces the peak."""
    n = len(a)
    if M is None:
        c = np.convolve(a, a)
        M = np.max(c)
    
    # Constraint matrix: A[k, i] = a[i] if 0 <= k-i < n, else 0
    # This is a (2n-1) x n matrix
    # A is banded with bandwidth n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A[k, i_start:i_end+1] = a[i_start:i_end+1]
    
    # Maximize sum(b) subject to A@b <= M, b >= 0
    # linprog minimizes, so we negate the objective
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        b = res.x
        return b, M
    return None, M
