```python
def projected_subgradient_optimize(n, initial, max_iterations, beta=50, alpha0=1.0):
    f = initial.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_val = 2 * n * np.max(np.convolve(f, f))
    
    f_avg = f.copy()
    
    for t in range(1, max_iterations + 1):
        # Compute convolution
        c = np.convolve(f, f)
        val = 2 * n * np.max(c)
        if val < best_val:
            best_val = val
            best_f = f.copy()
        
        # Compute smooth gradient
        c_max = np.max(c)
        w = np.exp(beta * (c - c_max))
        w = w / w.sum()
        
        # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * f[m]
        f_rev = f[::-1]
        full_conv = np.convolve(w, f_rev, mode='full')
        grad = 2 * full_conv[n-1:2*n-1]
        
        # Step size (Polyak or diminishing)
        alpha = alpha0 / np.sqrt(t)
        
        # Update
        f_new = f - alpha * grad
        
        # Project onto simplex
        f = project_onto_simplex(f_new)
        
        # Averaging
        f_avg = f_avg * t / (t+1) + f / (t+1)
    
    # Use the average
    avg_val = 2 * n * np.max(np.convolve(f_avg, f_avg))
    if avg_val < best_val:
        best_f = f_avg
        best_val = avg_val
    
    return best_f, best_val

def project_onto_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.sum(u * np.arange(1, n+1) - (cssv - 1) > 0)
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)
```

import numpy as np
import time
from scipy.optimize import linprog

def project_onto_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    condition = u * rho_candidates - (cssv - 1) > 0
    rho = int(np.sum(condition))
    if rho == 0:
        s = v.sum()
        if s > 1e-15:
            return v / s
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)


def compute_smooth_gradient(f, beta):
    """Compute the smooth gradient of max(convolve(f,f))."""
    n = len(f)
    c = np.convolve(f, f)
    c_max = np.max(c)
    w = np.exp(beta * (c - c_max))
    w_sum = w.sum()
    w = w / w_sum
    # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * f[m]
    # = 2 * (w convolved with f[::-1])[i + n - 1]
    f_rev = f[::-1]
    full_conv = np.convolve(w, f_rev, mode='full')
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad, c


def optimize_lp_method(f, max_iters, blend=0.5):
    """Optimize using the LP-based iterative method."""
    n = len(f)
    f = f.copy()
    total = f.sum()
    if total < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / total

    best_f = f.copy()
    best_val = 2.0 * n * np.max(np.convolve(f, f))

    for it in range(max_iters):
        c = np.convolve(f, f)
        M = np.max(c)
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()

        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_s = max(0, k - n + 1)
            j_e = min(n, k + 1)
            for j in range(j_s, j_e):
                A[k, j] = f[k - j]

        # Solve LP: max sum(b) s.t. A @ b <= M, b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break

        if not result.success:
            break

        b = result.x
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        b = b / b_sum

        # Blend
        f_new = (1 - blend) * f + blend * b
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f = f_new / s

    return best_f, best_val


def optimize_subgradient(f, max_iters, beta=50.0, alpha0=0.01, momentum=0.9):
    """Optimize using projected subgradient with momentum."""
    n = len(f)
    f = f.copy()
    total = f.sum()
    if total < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / total

    best_f = f.copy()
    best_val = 2.0 * n * np.max(np.convolve(f, f))

    m = np.zeros(n)  # momentum
    f_avg = f.copy()  # Polyak averaging

    for t in range(1, max_iters + 1):
        grad, c = compute_smooth_gradient(f, beta)
        val = 2.0 * n * np.max(c)
        if val < best_val:
            best_val = val
            best_f = f.copy()

        # Momentum update
        m = momentum * m + (1 - momentum) * grad

        # Step size
        alpha = alpha0 / np.sqrt(t)

        # Update
        f_new = f - alpha * m

        # Project onto simplex
        f = project_onto_simplex(f_new)

        # Averaging
        f_avg = f_avg * (t / (t + 1.0)) + f / (t + 1.0)

    # Check the average
    avg_val = 2.0 * n * np.max(np.convolve(f_avg, f_avg))
    if avg_val < best_val:
        best_f = f_avg
        best_val = avg_val

    return best_f, best_val
