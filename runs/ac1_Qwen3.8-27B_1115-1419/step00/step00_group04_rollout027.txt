    """Reduce the peak autocorrelation of a non-negative sequence."""
    a = a.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a

Let me also consider using a smooth approximation of the max function. Instead of `max(b)`, use `logsumexp(b/τ)` for a small temperature τ. This is differentiable everywhere and gives a gradient that's a weighted average of the subgradients of the active peaks.

`f(a) = τ * log(sum_k exp(b[k]/τ))`

`∂f/∂a[m] = sum_k (exp(b[k]/τ) / sum_j exp(b[j]/τ)) * 2*a[k-m]`

This is a "soft" version of the max, and as τ → 0, it converges to the actual max. We can use a decreasing τ schedule.

This is nice because the gradient computation is:
`grad[m] = 2 * sum_k w[k] * a[k-m]` where `w[k] = exp(b[k]/τ) / sum_j exp(b[j]/τ)`

And `sum_k w[k] * a[k-m]` is again a correlation: `correlate(w, a)[m]`.

So the full gradient is `2 * correlate(w, a, 'valid')`.

import numpy as np
import time


def fft_correlate(c, a):
    """Compute np.correlate(c, a, mode='valid') using FFT for speed."""
    n_c = len(c)
    n_a = len(a)
    if n_c < n_a:
        c, a = a, c
        n_c, n_a = n_a, n_c
        return np.correlate(a, c, mode='valid')[::-1] if False else _fft_corr_inner(c, a)
    return _fft_corr_inner(c, a)


def _fft_corr_inner(c, a):
    """Inner FFT correlation: c longer, a shorter. Returns correlate(c,a,'valid')."""
    n_c = len(c)
    n_a = len(a)
    n_fft = 1
    total = n_c + n_a - 1
    while n_fft < total:
        n_fft *= 2
    d = a[::-1]
    C = np.fft.rfft(c, n_fft)
    D = np.fft.rfft(d, n_fft)
    conv = np.fft.irfft(C * D, n_fft)
    result = conv[n_a - 1: n_c]
    return result


def compute_score(a):
    """Compute the objective value directly (equiv to evaluate_sequence for valid inputs)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def smooth_peak_descent(a_init, n_iter, tau_start, tau_end, lr_scale, use_avg=True):
    """Smooth peak reduction via soft-max subgradient descent on the simplex."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
        s = 1.0
    a = a / s

    best_score = np.inf
    best_a = a.copy()
    a_avg = np.zeros(n, dtype=np.float64) if use_avg else None

    for t in range(n_iter):
        frac = t / max(1, n_iter - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        b = np.convolve(a, a)
        M = b.max()
        score = 2.0 * n * M

        if score < best_score:
            best_score = score
            best_a = a.copy()

        b_scaled = b - b.max()
        b_scaled /= tau
        exp_b = np.exp(b_scaled)
        w = exp_b / exp_b.sum()

        grad = 2.0 * fft_correlate(w, a)

        lr = lr_scale / np.sqrt(t + 1)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n, dtype=np.float64) / n
            break
        a = a / s

            a_avg += a / (t + 1)

        b_avg = np.convolve(a_avg, a_avg)
        score_avg = 2.0 * n * b_avg.max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()

    return best_a, best_score


def local_search_fine(a_init, time_budget, seed=0):
    """Refine with block multiplicative perturbations."""
    rng = np.random.default_rng(seed)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
        s = 1.0
    a = a / s

    best_score = 2.0 * n * np.convolve(a, a).max()
    best_a = a.copy()
    deadline = time.monotonic() + time_budget
    no_improve = 0

    while time.monotonic() < deadline:
        width = int(rng.integers(max(2, n // 80), max(3, n // 15)))
        start = int(rng.integers(0, n - width + 1))

        r = rng.random()
        if r < 0.5:
            noise = np.exp(rng.normal(0.0, 0.2 + 0.3 * (1 - no_improve / 200.0), size=width))
            cand = a.copy()
            cand[start:start + width] *= noise
        elif r < 0.75:
            noise = rng.normal(0.0, 0.05, size=width)
            cand = a.copy()
            cand[start:start + width] = np.maximum(cand[start:start + width] + noise, 0.0)
        else:
            width2 = int(rng.integers(max(2, n // 100), max(3, n // 20)))
            start2 = int(rng.integers(0, n - width2 + 1))
            if abs(width - width2) <= 5:
                cand = a.copy()
                tmp = cand[start:start + width].copy()
                cand[start:start + width] = cand[start2:start2 + width]
                cand[start2:start2 + width] = tmp
            else:
                cand = a.copy()
                cand[start:start + width] *= np.exp(rng.normal(0, 0.4, size=width))

        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand = cand / cs
        b_c = np.convolve(cand, cand)
        score_c = 2.0 * n * b_c.max()

        if score_c < best_score:
            best_score = score_c
            best_a = cand
            a = cand
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_score


def make_initialization(n, seed, init_type):
    """Generate various initial sequences."""
    rng = np.random.default_rng(seed)
    if init_type == 0:
        return rng.uniform(0.1, 1.0, n)
    elif init_type == 1:
        x = np.arange(n)
        return np.sin(np.pi * x / n) ** 2 + 0.01
    elif init_type == 2:
        x = np.arange(n)
        return np.exp(-4.0 * ((x - n / 2) / n) ** 2) + 0.01

        return (1.0 + 0.5 * np.cos(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t)) + 0.01
    else:
        return rng.exponential(1.0, n) + 0.01

        verified = float(evaluate_sequence(result))
        if verified < final_score:
            final_score = verified
        print(f"Verified score: {verified:.6f}")
    except:
        pass

    return result


def rng_choice(i):
    """Simple helper for choosing init type in restarts."""
    types = [0, 1, 2, 3, 4, 5]
    return types[i % len(types)]
