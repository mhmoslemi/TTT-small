def fast_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2 without the overhead of evaluate_sequence."""
    n = len(a)
    C = np.convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return float(2 * n * np.max(C) / (S ** 2))

def lp_direction(a, M, time_limit=15):
    """
    Solve the LP: maximize sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Return the solution b normalized to have the same sum as a, or None.
    """
    n = len(a)
    
    # Build constraint matrix A where (A @ b)[k] = (a * b)_conv[k]
    # A[k, i] = a[i] if 0 <= k-i < n, else 0
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            rows.extend(range(k, k + (i_max - i_min + 1)))
            cols.extend(range(i_min, i_max + 1))
    
    A = csr_matrix(
        (a[cols], (rows, cols)),
        shape=(2 * n - 1, n)
    )
    
    try:
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,

            options={'time_limit': time_limit, 'presolve': True, 'dual_feasibility_tolerance': 1e-7}
        )
