```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k"""
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: for each k in 0..2n-2, (f*g)[k] <= M
    # (f*g)[k] = sum_i f[i]*g[k-i] for valid i
    # For variable g[j], its coefficient in constraint k is f[k-j] (if k-j is in [0,n-1])
    
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
        if not np.all(np.isfinite(g)) or np.sum(g) < 1e-10:
            continue

        g = g * target_sum / np.sum(g)

        t = 0.1
        best_f_iter = f.copy()
        best_score_iter = best_score
        
        for trial_t in [0.01, 0.05, 0.1, 0.3, 0.5, 0.7, 1.0]:

                best_score = score
                best_f = f_trial.copy()
        
        f = (1 - t) * f + t * g
        f = np.maximum(f, 0)
    
    return [float(x) for x in best_f]
```

```python
def build_conv_matrix(fixed_a):
    """Build (2n-1) x n matrix for convolution constraint."""
    n = len(fixed_a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            A[k, j] = fixed_a[k - j]
    return A
```

```python
"""
LP Alternating Minimization for Minimizing Auto-Convolution Peak.

Strategy: Find non-negative sequences a with sum(a)=sqrt(2n) that minimize
max_k (a*a)[k] (equivalent to minimizing J = 2n*max_conv/sum^2).

Method: Alternating LP where we fix one copy of the sequence and solve an LP
to optimize the other, driving the auto-convolution peak down. Multiple
random restarts and sequence lengths are explored.
"""
import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(fixed_a):
    """Build (2n-1) x n matrix A where A[k,j] = fixed_a[k-j] if 0<=k-j<n."""
    n = len(fixed_a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = fixed_a
    return A


def solve_lp_step(fixed_a, S):
    """
    Solve LP: min t s.t. (fixed_a * x)[k] <= t for all k, sum(x) = S, x >= 0.
    Returns optimal x or None.
    """
    n = len(fixed_a)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    A_conv = build_conv_matrix(fixed_a)

    n_ineq = 3 * n - 1
    A_ub = np.zeros((n_ineq, n + 1))
    A_ub[:2 * n - 1, :n] = A_conv
    A_ub[:2 * n - 1, n] = -1.0
    A_ub[2 * n - 1:, :n] = -np.eye(n)
    b_ub = np.zeros(n_ineq)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)] * (n + 1), method='highs')
    if res.status != 0:
        return None
    x = res.x[:n].copy()
    x = np.maximum(x, 0.0)
    if x.sum() > 0:
        x = x * S / x.sum()
    return x


def fast_score(a):
    """Fast score computation (matches evaluate_sequence logic)."""
    n = len(a)
    conv = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def lp_alternating(a, S, max_iters, deadline, rng, tol=1e-8):
    """Run LP alternating iterations. Returns improved sequence and best score."""
    n = len(a)
    local_best = fast_score(a)
    best_a = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        x_new = solve_lp_step(a, S)
        if x_new is None:
            break

        improved = False
        for t in [1.0, 0.5, 0.3, 0.15, 0.05]:
            a_trial = np.maximum((1.0 - t) * a + t * x_new, 0.0)
            s_trial = fast_score(a_trial)
            if s_trial < local_best - tol:
                a = a_trial
                local_best = s_trial
                improved = True
                break

        if improved:
            local_best = fast_score(a)
            if local_best < local_best:
                pass
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, local_best


def polish_sequence(a, S, deadline, rng, max_iters=500):
    """Local polish using small random perturbations on the LP result."""
    n = len(a)
    best = a.copy()
    best_score = fast_score(best)
```
