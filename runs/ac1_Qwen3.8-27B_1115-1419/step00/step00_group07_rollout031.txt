    a = parent / parent.sum()
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    
    # Smoothed subgradient descent
    tau = 100  # temperature for log-sum-exp
    lr = 0.01  # learning rate
    
    for iteration in range(100000):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        conv = np.convolve(a, a)
        
        # Find top values for smoothed max
        # For efficiency, use the top L values
        L = min(50, len(conv))
        top_idx = np.argpartition(conv, -L)[-L:]
        top_vals = conv[top_idx]
        
        # Softmax weights
        shifted = top_vals - top_vals.max()
        weights = np.exp(tau * shifted)
        weights /= weights.sum()
        
        # Gradient: weighted sum of individual gradients
        grad = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            k = idx
            # grad[j] += w * 2 * a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                grad[j_start:j_end+1] += w * 2 * a[k - j_start : k - j_end][::-1]
        
        # Update
        a = a - lr * grad
        a = np.maximum(a, 0)
        if a.sum() > 0:
            a = a / a.sum()
        
        # Evaluate occasionally

            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best = a.copy()
                best_score = score
            
            # Decay learning rate
