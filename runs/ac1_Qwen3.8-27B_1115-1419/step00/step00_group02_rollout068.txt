```python
import numpy as np
import time

def ideal_seq(n):
    """Generate initial sequence approximating flat convolution."""
    C = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(C)
    for k in range(1, n):
        cross = 0.0
        for j in range(1, k):
            cross += a[j] * a[k - j]
        val = (C - cross) / (2.0 * a[0])
        a[k] = max(0.0, val)
    # Normalize
    s = a.sum()
    if s > 0:
        a /= s
    return a

def gradient_step(a, beta):
    """Compute gradient of soft-max of convolution."""
    n = len(a)
    conv_len = 2 * n - 1
    # Compute convolution via FFT
    fft_size = 1
    while fft_size < conv_len:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.real(np.fft.irfft(fa * fa, fft_size)[:conv_len])
    c = np.maximum(c, 0.0)
    
    # Softmax weights
    max_c = c.max()
    if max_c < 1e-30:
        return c, np.zeros(n)
    scores = c * beta
    scores -= scores.max()
    p = np.exp(scores)
    p /= p.sum()
    
    # Gradient: grad[i] = 2 * sum_j p[i+j] * a[j]
    grad = np.zeros(n)
    for j in range(n):
        aj = a[j]
        if aj > 1e-15:
            grad += 2.0 * aj * p[j:j+n]
    
    return c, grad, p
```

```python
        # partial_c[m] = sum_{j=0}^{min(k-1,m)} a[j] * a[m-j] (where m-j <= k-1)
        # For m = k: partial_c[k] = sum_{j=1}^{k-1} a[j]*a[k-j] (j=0 gives a[0]*a[k-0]=a[0]*a[k] which is not computed, j=k gives a[k]*a[0] not computed)
        # Actually: j must satisfy 0<=j<=k-1 AND 0<=m-j<=k-1, i.e., m-(k-1)<=j<=min(m,k-1)
        # For m=k: k-(k-1)<=j<=min(k,k-1) → 1<=j<=k-1
        # So partial_c[k] after k-1 steps = sum_{j=1}^{k-1} a[j]*a[k-j] ✓
        
        # We want: c[k] = 2*a[0]*a[k] + partial_c[k] = C (for k < n)
        # So: a[k] = (C - partial_c[k]) / (2*a[0])
        
        cross = partial_c[k]
        a[k] = (C - cross) / (2.0 * a[0])
        if a[k] < 0:
            a[k] = 0.0
        
        # Update partial_c for future use
        if a[k] > 0:
            # For m from k to 2*k-1: partial_c[m] += 2*a[k]*a[m-k]
            # For m = 2*k: partial_c[2*k] += a[k]^2
            for m in range(k, 2*k):
                partial_c[m] += 2.0 * a[k] * a[m-k]
            partial_c[2*k] += a[k] * a[k]
    
    s = a.sum()
    if s > 0:
        a /= s
    return a
```

Wait, but the inner loop `for m in range(k, 2*k)` makes this O(n²) total, and the updates are O(k) per step. Total: sum_{k=1}^{n-1} k = O(n²). For n=8192, that's about 33M operations in pure Python, which would take several minutes! Too slow.

Let me also think about whether the "ideal" sequence is a good starting point for gradient descent. The issue is that the ideal sequence tries to make ALL c[k] equal to C, but the actual minimum max(c) might be achieved by a different distribution. The ideal sequence is a good approximation of the minimizer though.

Let me also consider: after the gradient descent, should I try to "repair" the sequence to make it more symmetric? The convolution c[k] is symmetric (c[k] = c[2n-2-k]) if and only if a is symmetric (a[j] = a[n-1-j]). If I enforce symmetry, I reduce the search space by half.

Actually, c[k] = sum_j a[j]*a[k-j]. c[2n-2-k] = sum_j a[j]*a[2n-2-k-j]. Let i = n-1-j: a[j] = a[n-1-i], a[2n-2-k-j] = a[2n-2-k-(n-1-i)] = a[n-1-k+i]. So c[2n-2-k] = sum_i a[n-1-i]*a[n-1-k+i].

This equals c[k] if a is reversible (a[j] = a[n-1-j]). So symmetry of a implies symmetry of c. But the optimal a need not be symmetric.

```python
import numpy as np
import time

def _ideal_seq(n):
    """Generate initial sequence approximating flat linear convolution."""
    C = 1.0 / (2.0 * n - 1.0)
    a = np.zeros(n)
    a[0] = np.sqrt(C)
    partial_c = np.zeros(2 * n - 1)
    partial_c[0] = a[0] * a[0]
    for k in range(1, n):
        cross = partial_c[k]
        val = (C - cross) / (2.0 * a[0])
        a[k] = max(0.0, val)
        if a[k] > 0:
            partial_c[k:2*k] += 2.0 * a[k] * a[0:k]
            partial_c[2*k] += a[k] * a[k]
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def _conv_fft(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1
    while fft_size < conv_len:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    c = np.real(np.fft.irfft(fa * fa, fft_size)[:conv_len])
    return np.maximum(c, 0.0)


def _gradient_and_conv(a, beta):
    """Compute convolution and gradient of soft-max objective."""
    n = len(a)
    c = _conv_fft(a)
    max_c = c.max()
    if max_c < 1e-30:
        return c, np.zeros(n)
    scores = c * beta
    scores -= scores.max()
    p = np.exp(scores)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return c, np.zeros(n)
    p /= p_sum
    # grad[i] = 2 * sum_j p[i+j] * a[j]
    grad = np.zeros(n)
    for j in range(n):
        aj = a[j]
        if aj > 1e-12:
            grad += 2.0 * aj * p[j:j + n]
    return c, grad


def _local_refine(a, rng, budget_s, deadline, best_score, best_a):

    n = len(a)
    a = a.copy()

                best_score[0] = sc
                best_a[0] = a.copy()
                print(f"  local imp: {sc:.6f} t={time.monotonic()-_T0:.0f}s")
```
