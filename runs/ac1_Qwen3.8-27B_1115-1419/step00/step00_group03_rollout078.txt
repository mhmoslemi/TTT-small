```python
"""
Search for a nonnegative step function minimizing
    2 * n * max(np.convolve(a, a)) / (sum(a)**2).

The implementation uses:
  - multiple structured and random smooth starting points,
  - FFT-based self-convolution evaluation,
  - a mirror-descent / multiplicative gradient step driven by a blend of
    peak-subgradient and flatness gradients,
  - adaptive step size, sharpness, and stochastic restarts,
  - a final block/coordinate polishing phase.
"""

import numpy as np
import time


def next_power_of_two(m):
    m = int(max(1, m))
    size = 1
    while size < m:
        size <<= 1
    return size


def convolve_self(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    size = next_power_of_two(need)
    fx = np.fft.rfft(x, size)
    out = np.fft.irfft(fx * fx, size)
    out = out[:need]
    return np.maximum(out, 0.0)


def corr_weights_p(w, p):
    """Return g[j] = sum_l p[l] * w[j + l] for j = 0..n-1."""
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = n + w.size - 1
    size = next_power_of_two(need)
    pref = p[::-1].copy()
    F = np.fft.rfft(pref, size)
    W = np.fft.rfft(w, size)
    conv = np.fft.irfft(F * W, size)
    g = conv[n - 1:2 * n - 1]
    return np.maximum(g, 0.0)


def as_probability(q):
    if q is None:
        return None
    arr = np.asarray(q, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    arr = arr + 1e-12
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return arr / s


def score_from_p(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return np.inf
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    p = p / s
    c = convolve_self(p)
    if c.size == 0:
        return np.inf
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return np.inf
    return 2.0 * p.size * max(0.0, cmax)


def resample_sequence_to_length(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = int(max(1, n))
    if arr.size == n:
        return arr.copy()
    if arr.size <= 1:
        val = float(arr.mean()) if arr.size else (1.0 / n)
        if not np.isfinite(val):
            val = 1.0 / n
        return np.full(n, val, dtype=np.float64)
    x0 = np.linspace(0.0, arr.size - 1, arr.size)
    x1 = np.linspace(0.0, arr.size - 1, n)
    return np.interp(x1, x0, arr)


def mix_probabilities(a, b, alpha):
    a = as_probability(a)
    b = as_probability(b)
    if a is None or b is None:
        return b if a is None else a
    if b.size != a.size:
        return a
    p = (1.0 - alpha) * a + alpha * b
    return as_probability(p)


def update_best(global_best, p, score):
    if global_best is None:
        return
    if not np.isfinite(score):
        return
    if score < global_best[1] - 1e-12:
        arr = np.asarray(p, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if not np.isfinite(s) or s <= 0.0:
            arr = np.ones(arr.size if arr.size else 1, dtype=np.float64)
            s = float(arr.sum())
        arr = arr / s
        global_best[0] = arr
        global_best[1] = float(score)


def try_exact_update(global_best):
    if global_best is None or global_best[0] is None:
        return False
    seq = []
    for x in global_best[0]:
        val = float(x)
        if not np.isfinite(val) or val < 0.0:
            val = 0.0
        seq.append(val)
    try:
        sc = float(evaluate_sequence(seq))
        if np.isfinite(sc) and sc < global_best[1]:
            global_best[1] = sc
            return True
    except Exception:
        pass
    return False


def generate_start_pool(n, rng, provided_p, gen_timeout_s):
    n = int(max(1, n))
    deadline = time.monotonic() + max(0.0, float(gen_timeout_s))
    pool = []

    if provided_p is not None and provided_p.size == n:
        pool.append(provided_p.copy())

    # Uniform.
    pool.append(np.ones(n, dtype=np.float64) / float(n))

    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0

    # Asymmetric ramps.
    for B in (-0.95, -0.85, -0.7, -0.5, -0.3, 0.3, 0.5, 0.7, 0.85, 0.95):
        q = 1.0 + B * t
        p = as_probability(q)
        if p is not None:
            pool.append(p)

    # Odd sinusoidal skews.
    for B in (-0.95, -0.8, -0.6, -0.4, -0.2, 0.2, 0.4, 0.6, 0.8, 0.95):
        q = 1.0 + B * np.sin(2.0 * np.pi * x)
        p = as_probability(q)
        if p is not None:
            pool.append(p)

    # Truncated trigonometric skews.
    for amp in (0.45, 0.7, 0.9):
        phases = [0.0, np.pi / 4.0, np.pi / 2.0, 3.0 * np.pi / 4.0,
                  np.pi, 5.0 * np.pi / 4.0, 3.0 * np.pi / 2.0, 7.0 * np.pi / 4.0]
        for ph in phases:
            q = 1.0 + amp * np.cos(2.0 * np.pi * x + ph)
            p = as_probability(q)
            if p is not None:
                pool.append(p)

    # Odd cubic / quintic smooth skews.
    for B in (-0.9, -0.7, -0.5, 0.5, 0.7, 0.9):
        q = 1.0 + B * t * (1.0 - 0.25 * t * t)
        p = as_probability(q)
        if p is not None:
            pool.append(p)

        q = 1.0 + B * t * (1.0 - 0.5 * t * t) * (1.0 + 0.25 * t * t)
        p = as_probability(q)
        if p is not None:
            pool.append(p)

    # Smooth tanh steps.
    for B in (-0.9, -0.7, -0.5, 0.5, 0.7, 0.9):
        q = 1.0 + B * np.tanh(2.0 * t)
        p = as_probability(q)
        if p is not None:
            pool.append(p)

    # Piecewise constant ramps / anti-ramps.
    for k in (2, 3, 4, 6, 8, 12):
        cuts = np.floor(np.linspace(0.0, float(n), k + 1)).astype(int)
        for amp in (0.4, 0.7, 1.0):
            inc = np.linspace(1.0 - amp, 1.0 + amp, k)
            for vals in (inc, inc[::-1]):
                q = np.empty(n, dtype=np.float64)
                filled = False
                for b in range(k):
                    lo = int(cuts[b])
                    hi = int(cuts[b + 1])
                    if hi > lo:
                        q[lo:hi] = vals[b]
                        filled = True
                if not filled and n > 0:
                    q[0] = vals[0]
                p = as_probability(q)
                if p is not None:
                    pool.append(p)

    # Random low-frequency smooth shapes.
    for _ in range(160):
        if time.monotonic() >= deadline:
            break
        q = np.ones(n, dtype=np.float64)
        M = 6
        for m in range(1, M + 1):
            amp = float(rng.normal(0.0, 0.30))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            q += amp * np.cos(2.0 * np.pi * m * x + ph)
        q += float(rng.uniform(-0.75, 0.75)) * t
        p = as_probability(q)
        if p is not None:
            pool.append(p)

    # Random piecewise-linear shapes.
    for _ in range(80):
        if time.monotonic() >= deadline:
            break
        M = int(rng.integers(5, 13))
        nodes = np.linspace(0.0, 1.0, M)
        vals = rng.uniform(0.05, 2.2, M)
        q = np.interp(x, nodes, vals)
        p = as_probability(q)
        if p is not None:
            pool.append(p)

    # A few Dirichlet / smooth random vectors.
    for _ in range(50):
        if time.monotonic() >= deadline:
            break
        beta_scale = float(rng.lognormal(mean=0.0, sigma=0.9))
        if beta_scale < 0.05:
            beta_scale = 0.05
        beta = beta_scale * np.ones(n, dtype=np.float64)
        p = as_probability(rng.dirichlet(beta))
        if p is not None:
            pool.append(p)

    return pool


def mirror_optimize(p0, deadline, rng, global_best, tag,
                    S, flat_weight, lr, max_lr, max_S, restart_pool):
    p = as_probability(p0)
    if p is None:
        return
    n = p.size
    local_best = p.copy()
    local_best_score = score_from_p(p)
    update_best(global_best, local_best, local_best_score)

    min_lr = 0.018
    success = 0
    no_imm = 0
    iter_count = 0
    last_print = time.monotonic()
    last_global_print = global_best[1]

    while time.monotonic() < deadline - 2.0:
        iter_count += 1

        c = convolve_self(p)
        if c.size == 0:
            break
        cmax = float(np.max(c))
        current_score = 2.0 * n * cmax
        if not np.isfinite(current_score):
            break

        if current_score < local_best_score - 1e-12:
            local_best = p.copy()
            local_best_score = current_score

        update_best(global_best, local_best, local_best_score)
        if global_best[1] < last_global_print - 1e-6 and time.monotonic() - last_print > 5.0:
            print(f"{tag}: global improved to {global_best[1]:.6f} at iter {iter_count}", flush=True)
            last_global_print = global_best[1]
            last_print = time.monotonic()

        if iter_count % 200 == 0:
            S = min(max_S, S * 1.05)

        if cmax <= 1e-300:
            break

        c_sum = float(c.sum())
        if flat_weight >= 0.999:
            if c_sum > 1e-300:
                weights = c / c_sum
            else:
                weights = np.ones(c.size, dtype=np.float64) / c.size
        else:
            e = np.exp(S * (c - cmax))
            esum = float(e.sum())
            if not np.isfinite(esum) or esum <= 0.0:
                wmax = np.ones(c.size, dtype=np.float64) / c.size
            else:
                wmax = e / esum
            if c_sum > 1e-300:
                wflat = c / c_sum
            else:
                wflat = np.ones(c.size, dtype=np.float64) / c.size
            weights = (1.0 - flat_weight) * wmax + flat_weight * wflat
            weights = np.maximum(weights, 0.0)
            ws = float(weights.sum())
            if not np.isfinite(ws) or ws <= 0.0:
                weights = wmax
            else:
                weights = weights / ws

        # Gradient of the softened / blended peak objective, multiplied by 2.
        g_raw = 2.0 * corr_weights_p(weights, p)
        if not np.all(np.isfinite(g_raw)):
            break

        gx = g_raw - float(np.min(g_raw))
        gx_max = float(np.max(gx))
        if gx_max <= 1e-14:
            no_imm += 1
            if no_imm > 20:
                base = local_best.copy()
                if restart_pool is not None and len(restart_pool) > 0 and rng.random() < 0.6:
                    idx = int(rng.integers(0, len(restart_pool)))
                    base = mix_probabilities(local_best, restart_pool[idx], 0.2 + 0.3 * float(rng.random()))
                z = np.log(np.maximum(base, 1e-300))
                p = as_probability(np.exp(z + rng.normal(0.0, 0.04, n)))
                if p is None:
                    p = base
                no_imm = 0
                S = min(max_S, S * 1.3)
                lr = max(min_lr, lr * 0.75)
            continue

        gx = gx / gx_max

        accepted = False
        cand = None
        cand_score = np.inf

        for amp in (lr, 0.5 * lr):
            expfac = np.exp(-amp * gx)
            pc = p * expfac
            pc = as_probability(pc)
            if pc is None:
                continue
            sc = score_from_p(pc)
            if not np.isfinite(sc):
                continue
            if (sc < current_score - 1e-12) or (sc < local_best_score * 0.999):
                accepted = True
                cand = pc
                cand_score = sc
                break

        if accepted:
            p = cand
            if cand_score < local_best_score - 1e-12:
                local_best = p.copy()
                local_best_score = cand_score
            update_best(global_best, local_best, local_best_score)
            success += 1
            no_imm = 0
            if success >= 15:
                if lr < max_lr:
                    lr = min(max_lr, lr * 1.04)
                success = 0
        else:
            success = 0
            no_imm += 1
            lr = max(min_lr, lr * 0.86)

            if no_imm > 90:
                base = local_best.copy()
                if restart_pool is not None and len(restart_pool) > 0 and rng.random() < 0.55:
                    idx = int(rng.integers(0, len(restart_pool)))
                    base = mix_probabilities(local_best, restart_pool[idx], 0.15 + 0.30 * float(rng.random()))
                z = np.log(np.maximum(base, 1e-300))
                sigma_noise = 0.025 + 0.07 * float(min(1.0, no_imm / 800.0))
                p = as_probability(np.exp(z + rng.normal(0.0, sigma_noise, n)))
                if p is None:
                    p = base
                no_imm = 0
                S = min(max_S, S * 1.35)
                lr = max(min_lr, lr * 0.7)

        if iter_count % 2000 == 0:
            print(f"{tag}: iter {iter_count}, local={local_best_score:.6f}, "
                  f"global={global_best[1]:.6f}, lr={lr:.4f}, S={S:.1f}", flush=True)


def polish_sequence(p0, deadline, rng, global_best):
    p = as_probability(p0)
    if p is None:
        return
    n = p.size
    score = score_from_p(p)
    update_best(global_best, p, score)

    sigma = 0.035
    min_sigma = 0.004
    max_sigma = 0.22
    last_print = time.monotonic()

    while time.monotonic() < deadline - 1.5:
        # Random block rescale.
        min_width = max(2, n // 128)
        max_width = min(n - 1, max(min_width + 1, n // 32))
        if max_width <= min_width:
            max_width = min(n - 1, min_width + 1)
        if max_width >= n:
            max_width = max(2, n // 4)
        width = int(rng.integers(min_width, max_width + 1))
        start = int(rng.integers(0, max(1, n - width)))
        scale = float(np.exp(rng.normal(0.0, sigma)))
        p2 = p.copy()
        p2[start:start + width] *= scale
        p2 = as_probability(p2)
        if p2 is not None:
            sc = score_from_p(p2)
            if np.isfinite(sc) and sc < score - 1e-12:
                p = p2
                score = sc
                update_best(global_best, p, score)
                sigma = min(max_sigma, sigma * 1.04)
            else:
                sigma = max(min_sigma, sigma * 0.985)

        # Occasional small coordinated transfer.
        if rng.random() < 0.25:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                p2 = p.copy()
                if p[i] > p[j]:
                    d = min(p[i] * 0.25, 0.08 * (p[i] - p[j]) + 1e-9)
                    if d > 0.0 and d < p[i]:
                        p2[i] -= d
                        p2[j] += d
                else:
                    d = min(p[j] * 0.25, 0.08 * (p[j] - p[i]) + 1e-9)
                    if d > 0.0 and d < p[j]:
                        p2[j] -= d
                        p2[i] += d
                p2 = as_probability(p2)
                if p2 is not None:
                    sc = score_from_p(p2)
                    if np.isfinite(sc) and sc < score - 1e-12:
                        p = p2
                        score = sc
                        update_best(global_best, p, score)

        if time.monotonic() - last_print > 20.0:
            print(f"polish: current={score:.6f}, global={global_best[1]:.6f}", flush=True)
            last_print = time.monotonic()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start_wall = time.monotonic()
    total_budget = max(10.0, float(budget_s))
    total_deadline = start_wall + total_budget - 5.0
    rng = np.random.default_rng(int(seed))

    # Use the provided construction if it has a sane length; otherwise choose 2048.
    provided_arr = globals().get("height_sequence_1", None)
    if provided_arr is not None:
        try:
            provided_arr = np.asarray(provided_arr, dtype=np.float64).ravel()
        except Exception:
            provided_arr = None
    if provided_arr is not None:
        provided_arr = np.where(np.isfinite(provided_arr), provided_arr, 0.0)
        provided_arr = np.maximum(provided_arr, 0.0)

    default_n = 2048
    n = default_n
    provided_p = None

    if provided_arr is not None and provided_arr.size >= 256:
        if provided_arr.size <= 8192:
            n = int(provided_arr.size)
            provided_p = as_probability(provided_arr)
        else:
            n = 4096
            resampled = resample_sequence_to_length(provided_arr, n)
            provided_p = as_probability(resampled)

    if n < 256:
        n = default_n
    if n > 4096:
        n = 4096
        if provided_p is not None:
            provided_p = as_probability(resample_sequence_to_length(provided_p, n))

    # Start pool.
    gen_timeout = min(120.0, max(10.0, total_deadline - time.monotonic() - 1200.0))
    pool = generate_start_pool(n, rng, provided_p, gen_timeout)

    scored = []
    idx = 0
    score_deadline = min(time.monotonic() + 120.0, total_deadline - 900.0)
    for p in pool:
        if time.monotonic() > score_deadline and idx > 25:
            break
        sc = score_from_p(p)
        if np.isfinite(sc):
            scored.append((sc, idx, p))
        idx += 1

    if not scored:
        uniform = np.ones(n, dtype=np.float64) / float(n)
        sc = score_from_p(uniform)
        scored = [(sc, 0, uniform)]

    scored.sort()
    best_start_score = scored[0][0]
    best_start_p = scored[0][2].copy()
    global_best = [best_start_p, best_start_score]
    try_exact_update(global_best)

    restart_pool = []
    limit = min(14, len(scored))
    for i in range(limit):
        restart_pool.append(scored[i][2].copy())

    print(f"initialized length={n}, best start score={global_best[1]:.6f}", flush=True)

    # Phase plan: (S, flat_weight, lr, max_lr, fraction_of_optimization_time)
    phase_defs = [
        (2.0, 1.00, 0.60, 1.00, 0.14),
        (4.0, 0.65, 0.50, 0.85, 0.19),
        (10.0, 0.35, 0.40, 0.65, 0.24),
        (25.0, 0.15, 0.30, 0.55, 0.24),
        (70.0, 0.00, 0.22, 0.42, 0.19),
    ]

    now = time.monotonic()
    remaining = max(0.0, total_deadline - now)
    if remaining > 240.0:
        reserve = min(240.0, 0.12 * remaining)
        opt_remaining = max(0.0, remaining - reserve)
    else:
        opt_remaining = remaining * 0.8
        reserve = remaining * 0.2

    if opt_remaining > 30.0:
        for phase_index, ph in enumerate(phase_defs):
            S, fw, lr, max_lr, frac = ph
            phase_seconds = frac * opt_remaining
            if phase_seconds < 12.0:
                continue
            phase_deadline = min(total_deadline, time.monotonic() + phase_seconds)

            if phase_index < len(scored):
                phase_start = scored[phase_index][2]
            else:
                phase_start = global_best[0]

            if phase_index == 0:
                p0 = phase_start.copy()
            else:
                p0 = mix_probabilities(global_best[0], phase_start, 0.22)
                if p0 is None:
                    p0 = global_best[0].copy()

                if phase_index >= 2 and len(restart_pool) > 0 and rng.random() < 0.7:
                    ridx = int(rng.integers(0, len(restart_pool)))
                    p0 = mix_probabilities(p0, restart_pool[ridx], 0.10 + 0.10 * float(rng.random()))
                    if p0 is None:
                        p0 = global_best[0].copy()

            tag = f"phase{phase_index}"
            max_S = max(20.0, S * 4.0)
            mirror_optimize(p0, phase_deadline, rng, global_best, tag,
                            S, fw, lr, max_lr, max_S, restart_pool)
            try_exact_update(global_best)
            print(f"after {tag}: global={global_best[1]:.6f}", flush=True)

    # Final polish from the current best.
    if time.monotonic() < total_deadline - 10.0:
        polish_sequence(global_best[0], total_deadline, rng, global_best)
        try_exact_update(global_best)

    result = global_best[0]
    result = np.asarray(result, dtype=np.float64).ravel()
    result = np.where(np.isfinite(result), result, 0.0)
    result = np.maximum(result, 0.0)
    s = float(result.sum())
    if not np.isfinite(s) or s <= 0.0:
        result = np.ones(n, dtype=np.float64) / float(n)
        s = 1.0
    result = result / s

    out = []
    for x in result:
        val = float(x)
        if not np.isfinite(val) or val < 0.0:
            val = 0.0
        out.append(val)

    final_exact = np.inf
    try:
        final_exact = float(evaluate_sequence(out))
    except Exception:
        final_exact = np.inf

    if not np.isfinite(final_exact):
        uniform_out = []
        v = 1.0 / float(n)
        for _ in range(n):
            uniform_out.append(float(v))
        out = uniform_out
        try:
            final_exact = float(evaluate_sequence(out))
        except Exception:
            final_exact = np.inf

    print(f"final returned length={n}, approximate score={float(score_from_p(np.asarray(out)))}.6f"},
          flush=True)
    print(f"final evaluator score={final_exact:.6f}", flush=True)
    return out
```
