            candidate = 1.0 + 0.3 * np.sin(t + rng.uniform(0, 2 * np.pi))
        elif strategy == 1:
            # Raised cosine window
            t = np.linspace(0, 1, n)
            candidate = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
            candidate = candidate + 0.1 * rng.standard_normal(n)
        elif strategy == 2:
            # Gaussian bump
            centers = rng.uniform(0.2, 0.8, size=3)
            widths = rng.uniform(0.05, 0.2, size=3)
            candidate = np.ones(n)
            t = np.linspace(0, 1, n)
            for c, w in zip(centers, widths):
                candidate += np.exp(-0.5 * ((t - c) / w) ** 2)
        elif strategy == 3:
            # Low-frequency Fourier
            candidate = np.ones(n)
            for k in range(1, 4):
                candidate += 0.2 * rng.uniform(-1, 1) * np.sin(
                    2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 2 * np.pi)
                )
        else:
            # Block perturbation of best
            width = max(2, n // 50)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = best.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.1, size=width))

        candidate = np.clip(candidate, 0.0, 1000.0)
        if candidate.sum() < 0.01:
            continue

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"Iteration {iterations}: NEW BEST score = {best_score:.10f}")

    print(f"Verification complete after {iterations} iterations.")
    print(f"Final best score: {best_score:.10f}")

    return [float(x) for x in best]
