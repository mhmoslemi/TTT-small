```python
import numpy as np

def solve_lp_direction(f, M, n, top_k, solver='ECOS'):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for selected k."""
    b = np.convolve(f, f)
    # Select top_k positions
    if top_k >= 2*n - 1:
        positions = np.arange(2*n - 1)
    else:
        positions = np.argsort(b)[-top_k:]
    
    # Build constraint matrix
    T_sub = np.zeros((len(positions), n))
    for idx, k in enumerate(positions):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            T_sub[idx, j_min:j_max+1] = f[j_min:j_max+1]
    
    g = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ g <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=solver, max_iters=20000, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iters=20000, verbose=False)
        except:
            return None
    
    if g.value is not None and np.isfinite(g.value).all() and np.sum(g.value) > 0:
        return np.clip(g.value, 0, 1000)
    return None
```

```python
    M = max(b)
```

```python
def compute_subgradient(f, S):
    """Compute subgradient of max(f*f) at f, with active set S."""
    n = len(f)
    d = np.zeros(n)
    
    for k in S:
        # Term 1: f[k-j] for valid j
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            # d[j_min:j_max+1] += f[k-j_max : k-j_min-1 : -1] = f[j_min..j_max] mapped
            idx = np.arange(j_min, j_max + 1)
            d[idx] += f[k - idx]

        j_min2 = max(0, k)
        j_max2 = min(n - 1, k + n - 1)
        if j_max2 >= j_min2:
            idx = np.arange(j_min2, j_max2 + 1)
            d[idx] += f[idx - k]
```

```python
def project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Iterative subgradient with LP refinement for minimizing max(self-convolution)
of a non-negative sequence. Uses simplex projection, adaptive step sizes,
and periodic LP-based direction finding.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix


def project_onto_simplex(v, s):
    """Project vector v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs_rho = np.arange(1, n + 1)
    rho = np.nonzero(u * cs_rho > (cssv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, S):
    """Compute average subgradient of max_k (f*f)_k at f over active set S."""
    n = len(f)
    d = np.zeros(n)
    for k in S:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            idx = np.arange(j_lo, j_hi + 1)
            d[idx] += f[k - idx]
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_hi2 >= j_lo2:
            idx = np.arange(j_lo2, j_hi2 + 1)
            d[idx] += f[idx - k]
    if len(S) > 0:
        d /= len(S)
    return d


def solve_lp_direction(f, M, n, top_k, solver=None):
    """Solve LP: max 1^T g s.t. g >= 0, (f*g)_k <= M for top-k positions."""
    b = np.convolve(f, f)
    if top_k >= 2 * n - 1:
        positions = np.arange(2 * n - 1)
    else:
        positions = np.argsort(b)[-top_k:]

    T_sub = np.zeros((len(positions), n))
    for idx, k in enumerate(positions):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            T_sub[idx, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]

    g = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS]
    if solver is None:
        solvers_to_try = [cp.ECOS, cp.SCS]

    for slv in solvers_to_try:
        if slv is None:
            continue
        try:
            prob.solve(solver=slv, max_iters=50000, verbose=False)
            if g.value is not None and np.isfinite(g.value).all() and np.sum(g.value) > 1e-10:
                return np.clip(g.value, 0.0, 1000.0)
        except Exception:
            continue
    return None


def try_lp_refinement(f, n, s, deadline, top_k=300):
    """Do one LP-based refinement step. Returns new f or None."""
    b = np.convolve(f, f)
    M = float(np.max(b))
    if M < 1e-10:
        return None
    g = solve_lp_direction(f, M, n, top_k)
    if g is None or np.sum(g) < 1e-10:
        return None
    g = g * (s / np.sum(g))
    g = np.clip(g, 0.0, 1000.0)
    g = project_onto_simplex(g, s)
    if np.sum(g) < 1e-10:
        return None
    return g

    # --- Hyperparameters ---

    alpha_max = 1.0

    lp_interval = 80  # do LP every 80 iterations
    restart_interval = 400  # random restart every 400 iterations

    iteration = 0
    while time.monotonic() < deadline:

        # --- Compute conv and subgradient ---
        b = np.convolve(f, f)

                        best_score = score_try
                        f = f_try
                        no_improve_count = 0
                        break

        # --- Random restart if stuck ---
        if no_improve_count > restart_interval:
            no_improve_count = 0
            # Perturb from best
            f = best_f * np.exp(rng.normal(0, 0.08, n))
            f = project_onto_simplex(f, s)
            # Also try a fresh random start occasionally

                f = best_f * scale
                f = project_onto_simplex(f, s)
```
