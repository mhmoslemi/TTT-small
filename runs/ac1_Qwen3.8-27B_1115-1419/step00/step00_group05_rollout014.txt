"""
Lower-bound construction for the public evaluator.

For a sequence a of length n, let S=sum a_i and C=a*a (full linear convolution).
The zero-lag term is C_{n-1}=sum_i a_i^2, hence max(C) >= sum_i a_i^2.
For non-negative a, Cauchy-Schwarz gives (sum_i a_i)^2 <= n sum_i a_i^2.
Therefore evaluate_sequence(a) >= 2n * (S^2/n) / S^2 = 2.
Equality is attained by any constant non-zero sequence.

propose_candidate therefore returns a constant sequence of moderate length.
It verifies the score with the provided evaluator if available and falls back
to the same deterministic vector otherwise.
"""

_INF = float("inf")

def _constant_lengths():
    lengths = []
    # Prefer the length of the previously found construction when it is
    # available and not too large for the evaluator.
    try:
        prev_len = len(height_sequence_1)  # noqa: F821
        if 1 <= prev_len <= 4096:
            lengths.append(prev_len)
    except Exception:
        pass

    for m in (4096, 2048, 1024, 512, 256, 128, 64, 1):
        if m not in lengths:
            lengths.append(m)
    return lengths

def _safe_score(seq):
    try:
        val = float(evaluate_sequence(seq))  # noqa: F821
    except Exception:
        return _INF
    if val != val or val == _INF or val == -_INF:
        return _INF
    return val

def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Return a constant non-negative sequence that attains the exact
    mathematical minimum (score 2) of the public evaluator.
    """
    best = None
    best_score = _INF

    print("Using constant-sequence construction; proven lower bound is 2.")
    for length in _constant_lengths():
        cand = [1.0] * length
        score = _safe_score(cand)
        if score < best_score:
            best = cand
            best_score = score
        if score <= 2.0 + 1e-7:
            print(f"length={length} score={score!r} (at proven lower bound)")
            return cand

    if best is None:
        best = [1.0]
    print("Returning best constant vector found; best_score=", best_score)
    return best

"""
Constant-sequence construction attaining the proven lower bound.

For the supplied evaluator and any non-negative sequence a of length n:
  * the zero-lag autocorrelation term is sum_i a_i^2, so max(a*a) >= sum_i a_i^2;
  * Cauchy-Schwarz gives (sum_i a_i)^2 <= n * sum_i a_i^2.
Therefore
  2*n*max(a*a)/(sum a)^2 >= 2.
Equality is achieved by any nonzero constant sequence.

This program returns a moderate-length constant sequence and validates it with
evaluate_sequence when available.
"""

_INF = float("inf")


def _constant_lengths():
    """
    Candidate lengths for the constant vector.

    If the prior construction is accessible and its length is reasonable, use
    that length first; otherwise fall back to safe moderate lengths.
    """
    lengths = []

    try:
        prev_len = len(height_sequence_1)  # noqa: F821
        if 1 <= prev_len <= 8192:
            lengths.append(prev_len)
    except Exception:
        pass

    for m in (4096, 2048, 1024, 512, 256, 128, 64, 1):
        if m not in lengths:
            lengths.append(m)

    return lengths


def _safe_score(seq):
    """
    Evaluate a candidate safely.  Returns inf on any exception, invalid result,
    or missing evaluator.
    """
    try:
        val = float(evaluate_sequence(seq))  # noqa: F821
    except Exception:
        return _INF

    if val != val or val == _INF or val == -_INF:
        return _INF

    return val


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Return a constant non-negative sequence that attains the exact mathematical
    minimum (score 2) for the public evaluator.
    """
    best = None
    best_score = _INF

    print("Using constant-sequence construction; proven lower bound is 2.")

    for length in _constant_lengths():
        candidate = [1.0] * length
        score = _safe_score(candidate)

        if score < best_score:
            best = candidate
            best_score = score

        # For the official non-negative evaluator, anything at the bound
        # (within floating-point tolerance) is globally optimal.
        if score <= 2.0 + 1e-6:
            print(f"length={length} score={score!r} (at proven lower bound)")
            return candidate

    if best is None:
        best = [1.0]
