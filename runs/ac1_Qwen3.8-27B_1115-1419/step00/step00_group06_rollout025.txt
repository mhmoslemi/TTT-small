import numpy as np

def fast_score(x):
    """Compute objective using FFT-based convolution."""
    n = len(x)
    if n == 0:
        return float('inf')
    s = x.sum()
    if s < 0.01:
        return float('inf')
    need = 2 * n - 1
    N = 1
    while N < need:
        N *= 2
    X = np.fft.rfft(x, N)
    conv = np.fft.irfft(X * X, N)[:need]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def spectral_init(n, rng):
    """Initialize a non-negative sequence via spectral flattening."""
    # Start with random non-negative
    x = rng.random(n)
    # Iterate: flatten spectrum, project to non-negative
    for _ in range(20):
        N = 1
        need = 2 * n - 1
        while N < need:
            N *= 2
        X = np.fft.rfft(x, N)
        mag = np.abs(X)
        # Compress magnitude toward flat
        target = np.mean(mag)
        new_mag = target + 0.5 * (mag - target)
        new_mag = np.maximum(new_mag, 0.1 * target)
        phase = X / np.maximum(mag, 1e-12)
        X_new = new_mag * phase
        x = np.real(np.fft.irfft(X_new, N))[:n]
        x = np.maximum(x, 0.0)
    x = np.clip(x, 0, 1000)
    return x
