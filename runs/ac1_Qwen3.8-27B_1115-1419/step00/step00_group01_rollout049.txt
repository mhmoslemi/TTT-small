Low-peak self-convolution sequence optimizer.
Uses smoothed gradient descent with log-sum-exp, simulated annealing,
and multi-start to minimize 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time

def _fft_gradient(a, p):
    """Compute gradient of soft-max(conv(a,a))/T w.r.t. a via FFT.
    Returns grad of shape (n,) where grad[j] = 2 * sum_m a[m] * p[m+j].
    """
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    flip_a = a[::-1]
    # Linear convolution of flip_a and p via FFT
    F = np.fft.rfft(flip_a, N)
    P = np.fft.rfft(p, N)
    conv_result = np.fft.irfft(F * P, N)
    # g[j] = conv_result[j + n - 1] for j = 0, ..., n-1
    return 2.0 * conv_result[n-1 : 2*n-1]

def _soft_max_and_weights(c, T):
    """Compute soft max of c and the softmax weights."""
    max_c = c.max()
    shifted = (c - max_c) / T
    log_sum = np.log(np.sum(np.exp(shifted)))
    soft_max_val = max_c + T * log_sum
    weights = np.exp(shifted)
    weights /= weights.sum()
    return soft_max_val, weights
