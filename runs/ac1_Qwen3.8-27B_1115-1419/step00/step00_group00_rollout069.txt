def compute_score(a):
    """Compute the objective without calling evaluate_sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def build_conv_matrix(f, n):
    """Build sparse constraint matrix for conv(f, g) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            ks = np.full(len(js), k)
            vs = f[k - js]
            rows.append(ks)
            cols.append(js)
            vals.append(vs)
    if rows:
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        vals = np.concatenate(vals)
    else:
        rows = np.array([], dtype=int)
        cols = np.array([], dtype=int)
        vals = np.array([], dtype=float)
    return scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""

                      options={'time_limit': 30, 'presolve': True})
        if res.success:

    except Exception:
        pass
    return None

"""
LP-based iterative optimization to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal direction, blends via line search.
Explores multiple sequence lengths and initializations over 2400s budget.
"""

import numpy as np
import scipy.sparse
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute the objective: 2n * max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        cnt = j_end - j_start + 1
        if cnt > 0:
            rows[idx:idx+cnt] = k
            cols[idx:idx+cnt] = np.arange(j_start, j_end + 1)
            vals[idx:idx+cnt] = f[k - np.arange(j_start, j_end + 1)]
            idx += cnt
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def lp_step(f, M, n):
    """Solve LP: max sum(g) s.t. A@f_conv_g <= M, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 60, 'presolve': True})
        if res.success and res.x is not None:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None


def lp_iterate(a, max_iters=500, time_limit=300.0):
    """Run LP iterations to improve the sequence."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a[:] = 1.0
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    start = time.monotonic()
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        M = np.convolve(a, a).max()
        M_inflated = M * 1.02

        g = lp_step(a, M_inflated, n)
        if g is None or g.sum() < 1e-12:
            g = lp_step(a, M, n)
            if g is None or g.sum() < 1e-12:
                break
        g = g / g.sum()

        improved = False
        for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.03, 0.02]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, 1000.0)
            if a_new.sum() < 0.01:
                break
            a_new = a_new / a_new.sum()
            s_new = compute_score(a_new)
            if s_new < best_score * (1.0 - 1e-7):
                best_score = s_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                stagnant = 0
                break

        if not improved:
            stagnant += 1
            if stagnant >= 5:
                # Perturb to escape
                a = a * np.exp(np.random.default_rng(it).normal(0, 0.08, n))
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                stagnant = 0

    return best_a, best_score


def perturbation_search(a, time_limit=60.0, max_iters=200):
    """Simple perturbation-based local search."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    rng = np.random.default_rng(123)
    start = time.monotonic()
    step_size = 0.1

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        # Block perturbation
        width = max(2, n // 100)
        pos = rng.integers(0, n - width)
        block = a[pos:pos+width] * np.exp(rng.normal(0, step_size, width))
        new_a = a.copy()
        new_a[pos:pos+width] = block
        new_a = np.clip(new_a, 0.0, 1000.0)
        if new_a.sum() < 0.01:
            continue
        new_a = new_a / new_a.sum()
        s = compute_score(new_a)
        if s < best_score:
            best_score = s
            best_a = new_a.copy()
            a = new_a
            step_size = min(step_size * 1.1, 0.3)
        else:
            step_size *= 0.95

    return best_a, best_score


def cosine_init(n):
    """Hann/cosine window initialization."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    a = a + 1e-6
    return a / a.sum()


def gaussian_init(n):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 4.0
    a = np.exp(-(x - center)**2 / (2.0 * sigma * sigma))
    return a / a.sum()


def triangular_init(n):
    """Triangular (tent) shape - high at edges, low at center."""
    x = np.arange(n, dtype=np.float64)
    center = n / 2.0
    a = 1.0 - np.abs(x - center) / center
    a = a + 1e-6
    return a / a.sum()


def power_init(n, p=0.5):
    """Power-law taper: a[k] = (k+1)^(-p) normalized."""
    x = np.arange(1, n+1, dtype=np.float64)
    a = x.astype(np.float64) ** (-p)
    # Make it symmetric
    a = np.concatenate([a, a[::-1]])
    if len(a) > n:
        a = a[:n]
    return a / a.sum()
