"""
LP-based iterative optimization for minimizing the step function evaluation.
Uses a linear programming subproblem to find descent directions that reduce
the peak of the self-convolution while maintaining non-negativity.
"""
import numpy as np
import time
from scipy.optimize import linprog

def build_constraint_matrix(f, n):
    """Build the (2n-1) x n constraint matrix for the LP.
    Constraint k: sum_j A[k,j] * g_j <= M
    where (f*g)_k = sum_j f[k-j] * g_j for valid j.
    """
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            A[k, js] = f[k - js]
    return A

def solve_lp(f, M, n):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    A = build_constraint_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def objective_val(seq):
    """Compute the objective value (same as evaluate_sequence but without overhead)."""
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    if n == 0:
        return np.inf
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    c = np.convolve(seq, seq)
    max_c = np.max(c)
    return float(2 * n * max_c / (s * s))

"""
LP-based iterative optimization for minimizing step function self-convolution peak.
Uses linear programming subproblems to find descent directions, combined with
multiple asymmetric initializations and adaptive step sizes.
"""
import numpy as np
import time
from scipy.optimize import linprog


def build_constraint_matrix(f, n):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<=n-1 else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            A[k, js] = f[k - js]
    return A


def my_objective(seq):
    """Fast objective computation (scale-invariant form)."""
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    s = np.sum(seq)
    if s < 1e-10:
        return np.inf
    conv = np.convolve(seq, seq)
    max_c = np.max(conv)
    return float(2 * n * max_c / (s * s))


def lp_solve(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A = build_constraint_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def run_lp_iterations(f, deadline, max_iters=500):
    """Run LP-based refinement iterations on f (assumed sum=1, non-negative)."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-10:
        return f, np.inf
    f = f / s
    score = my_objective(f)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(f, f)
        M = np.max(conv)

        g = lp_solve(f, M, n)
        if g is None:
            break

        s_g = np.sum(g)
        if s_g < 1e-10:
            break
        g = g / s_g

        # If g is essentially the same as f, we're at a fixed point
        if np.allclose(g, f, atol=1e-8):
            stagnant += 1
            if stagnant > 5:
                break

        # Line search over step sizes
        improved = False
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

                score = sc
                improved = True
                break

def run_perturbation_search(f, deadline, rng, max_iters=2000):

    print(f"\nFinal result: n={n}, score={final_score:.6f}")
    return result
