def gradient_descent(a, n, S, deadline, rng):
    """Run projected gradient descent to minimize soft-max of self-convolution."""
    N = 1
    while N < 3 * n:  # Need N >= 3n-2 for cross-correlation
        N *= 2
    
    tau = 5.0  # soft-max temperature
    lr = 0.01  # learning rate
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    no_improve_count = 0
    
    for iteration in range(100000):
        if time.monotonic() > deadline:
            break
        
        # Compute C = a * a (full convolution)
        A = np.fft.fft(a, N)
        C = np.fft.ifft(A * A).real[:2*n-1]
        
        # Compute soft-max weights
        C_max = C.max()
        exp_C = np.exp(tau * (C - C_max))  # shift for numerical stability
        p = exp_C / exp_C.sum()
        
        # Compute gradient: grad[i] = 2 * sum_j p[i+j] * a[j]
        P = np.fft.fft(p, N)
        grad = 2.0 * np.fft.ifft(P * np.conj(A)).real[:n]
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project: non-negative and sum = S
        a_new = np.clip(a_new, 0.0, None)
        a_new_sum = a_new.sum()
        if a_new_sum > 0:
            a_new = a_new * S / a_new_sum
        else:
            a_new = np.ones(n) * S / n
        
        a = a_new
        
        # Evaluate periodically
        if iteration % 100 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve_count = 0
                # Increase tau as we converge (sharpen the soft-max)
                tau = min(tau * 1.05, 50.0)
            else:
                no_improve_count += 1
            
            # Random restart if stuck
            if no_improve_count > 50:
                # Perturb
                a = best_a + rng.normal(0, 0.01 * S/n, n)
                a = np.clip(a, 0, None) * S / max(a.sum(), 1e-10)
                no_improve_count = 0
                tau = 5.0
        
        # Decay learning rate
        lr = max(0.001, 0.01 * (0.999) ** (iteration // 100))
    
    final_score = float(evaluate_sequence(best_a.tolist()))
    return best_a, final_score

def compute_objective_fast(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 efficiently using FFT."""

def compute_conv_max(a):

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = np.inf
    
    # Sequence lengths to explore (larger gives more degrees of freedom)
    sizes = [6121, 4096, 8192, 2048, 10000, 16384]
    
    # Allocate time: more time to promising sizes
    time_weights = [1.5, 1.0, 1.2, 0.8, 1.0, 1.0]
    total_weight = sum(time_weights)
    available_time = deadline - time.monotonic()
    
    print(f"Starting optimization with budget {budget_s}s, n candidates: {sizes}")
    
    for n, w in zip(sizes, time_weights):
        if time.monotonic() > deadline - 20:
            break
        
        time_budget = available_time * w / total_weight
        n_deadline = time.monotonic() + time_budget
        
        S = np.sqrt(2.0 * n)
        N = 1
        while N < 3 * n:
            N *= 2
        
        inits = generate_inits(n, S, rng)
        
        # Allocate time among initializations
        n_init_deadline = n_deadline / len(inits)
        init_deadlines = [time.monotonic() + i * n_init_deadline for i in range(len(inits))]
        init_deadlines[-1] = n_deadline
        
        for idx, a0 in enumerate(inits):
            d = init_deadlines[idx] if idx < len(init_deadlines) else n_deadline
            if time.monotonic() > d - 2:
                break
            
            a, score = run_gradient_descent(a0, n, S, N, rng, d)
            
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                elapsed = time.monotonic() - start
                print(f"  n={n}, init_{idx}: score={score:.6f} (elapsed {elapsed:.0f}s)")
    
    # Final polish: run a few more iterations on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        N = 1
        while N < 3 * n:
            N *= 2
        # Polish with smaller tau for fine-tuning
        polish_deadline = deadline - 10
        a_polished, polished_score = run_gradient_descent(
            best_seq, n, S, N, rng, polish_deadline)
        if np.isfinite(polished_score) and polished_score < best_score:
            best_score = polished_score
            best_seq = a_polished
            print(f"  After polish: score={best_score:.6f}")
    
    # Final exact evaluation
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            print(f"Final verified score: {final_score:.6f} (n={len(best_seq)})")
            best_score = min(best_score, final_score)
        return [float(x) for x in best_seq]
    
    # Fallback
    n_fb = 2048
    a_fb = np.ones(n_fb) * np.sqrt(2 * n_fb) / n_fb
    score_fb = float(evaluate_sequence(a_fb.tolist()))
    print(f"Fallback: n={n_fb}, score={score_fb:.6f}")
    return [float(x) for x in a_fb]
