"""
Optimizes a non-negative step function to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses parametric initialization followed by reweighted gradient descent on the simplex.
"""

import numpy as np
import time


def _fft_convolution(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def _cross_correlation_fft(a, w):
    """
    Compute g[j] = sum_k w[k] * a[k-j] for j=0..len(a)-1
    using FFT-based cross-correlation.
    a has length n, w has length up to 2n-1.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    # Cross-correlation: ifft(conj(fa) * fw) gives sum_m a[m] * w[m+j]
    # But we want sum_k w[k] * a[k-j] = sum_m a[m] * w[m+j] (substituting k=m+j)
    # Wait: sum_k w[k] * a[k-j], let m = k-j, k = m+j:
    # = sum_m w[m+j] * a[m]
    # So it's the cross-correlation of w and a: sum_m a[m] * w[m+j]
    # FFT formula: ifft(conj(fft(a)) * fft(w))[j] = sum_m a[m] * w[m+j]
    result = np.fft.irfft(np.conj(fa) * fw, size)
    return result[:n]


def _simplex_projection(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_normalized(a):
    """Compute objective for a normalized sequence (sum=1). Returns 2n*max(conv)."""
    n = len(a)
    c = _fft_convolution(a)
    return 2 * n * np.max(c)

    Reweighted gradient descent to minimize max_k c_k.
    a is normalized (sum=1, non-negative).
    """

    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score_normalized(a)
    deadline = time.monotonic() + budget_s
    
    eta = 1.0 / n  # initial step size
    eta_min = 1e-10
    
    iter_count = 0
    stalled = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        c = _fft_convolution(a)
        current_max = np.max(c)
        current_score = 2 * n * current_max
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            stalled = 0
        else:
            stalled += 1

            # Reset to best and reduce step size
            a = best_a.copy()
            eta *= 0.5
            stalled = 0
            if eta < eta_min:

        a = a_new
