```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp_direction(a, S):
    """Solve LP to find g that minimizes max(a*g)_k subject to sum(g)=S, g>=0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (a*g)_k - t <= 0 for k = 0, ..., 2n-2
    # (a*g)_k = sum_j a_j * g_{k-j} (where g_i = 0 for i outside [0, n-1])
    # So row k of A has: A[k, j] = a_j if 0 <= k-j <= n-1 (i.e., k-n+1 <= j <= k), and A[k, n] = -1
    
    num_rows = 2 * n - 1
    A_ub = np.zeros((num_rows, n + 1))
    b_ub = np.zeros(num_rows)
    
    for k in range(num_rows):
        # (a*g)_k = sum over j where 0 <= k-j <= n-1
        # i.e., max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[k, j_start:j_end+1] = a[j_start:j_end+1]
        A_ub[k, n] = -1.0  # coefficient of t
    
    # A_ub @ x <= b_ub, where x = [g_0, ..., g_{n-1}, t]
    # b_ub = 0
    
    # Equality constraint: sum(g_j) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: g_j >= 0, t unbounded (or t >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        t_opt = result.x[n]
        return g, t_opt
    return None, None
```

```python
from scipy.fft import next_fast_len

def compute_objective_fast(a):
    n = len(a)
    s = next_fast_len(2 * n - 1)
    A = np.fft.rfft(a, s)
    conv = np.fft.irfft(A * A, s)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def solve_lp_direction(a):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    
    # Build sparse convolution matrix (2n-1) x n
    diags_list = [np.full(n, a[d]) for d in range(n)]
    A = diags(diags_list, offsets=list(range(n)), shape=(2*n-1, n), format='csr')
    
    # Add -1 column for t
    t_col = csr_matrix((-np.ones((2*n-1, 1)), ), shape=(2*n-1, 1))
    A_ub = hstack([A, t_col], format='csr')
    
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n+1)
    c[n] = 1.0
    
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

        g = np.clip(result.x[:n], 0, None)
        return g, result.x[n]
    return None, None
```

```python
def build_conv_csr(a):
    n = len(a)
    m = 2 * n - 1
    # Each row k has non-zeros at positions j where max(0, k-n+1) <= j <= min(n-1, k)
    # The value is a[k-j]
    
    # For CSR format, we need: data, indices (column indices), indptr (row pointers)
    
    # indptr: for row k, the start and end of its non-zero entries
    indptr = np.zeros(m + 1, dtype=np.int32)
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k) + 1  # exclusive
        indptr[k+1] = indptr[k] + (j_end - j_start)
    
    total_nnz = indptr[-1]
    indices = np.empty(total_nnz, dtype=np.int32)
    data = np.empty(total_nnz, dtype=np.float64)
    
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k) + 1
        row_slice = slice(indptr[k], indptr[k+1])
        indices[row_slice] = np.arange(j_start, j_end)
        data[row_slice] = a[j_start:j_end]
    
    return csr_matrix((data, indices, indptr), shape=(m, n))
```

```python
    a = np.asarray(a, dtype=np.float64)
```

```python
def lp_alternating(a, time_limit, verbose=True):
    """Alternating LP optimization with line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_objective_fast(a)
    start = time.monotonic()
    no_improve = 0
    it = 0

    if verbose:
        print(f"    LP start: obj={best_obj:.6f}, n={len(a)}")

    while time.monotonic() - start < time_limit:
        # Scale num_active with n
        na = min(max(40, len(a) // 50), 300)

        if it % 50 == 49:
            g, t_opt = solve_lp_full(a)
        else:
            g, t_opt = solve_lp_reduced(a, num_active=na)

        if g is None:
            break

        improved = False
        alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]
        for alpha in alphas:
            candidate = (1.0 - alpha) * a + alpha * g
            candidate = np.clip(candidate, 0, None)
            cs = candidate.sum()
            if cs < 0.01:
                continue
            candidate = candidate / cs
            obj = compute_objective_fast(candidate)
            if obj < best_obj:
                best_obj = obj
                best_a = candidate.copy()
                a = candidate
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        it += 1
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter {it}, obj={best_obj:.6f}, t={t_opt:.6f}, t_elapsed={elapsed:.1f}s")

    if verbose:
        print(f"    LP done: {it} iters, final obj={best_obj:.6f}")
    return best_a, best_obj


def local_search(a, time_limit, rng, verbose=True):
    """Block perturbation local search."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    if a.sum() < 0.01:
        return a, np.inf
    best_a = a.copy()
    best_obj = compute_objective_fast(a)
    n = len(a)
    start = time.monotonic()
    improvements = 0

    if verbose:
        print(f"    LS start: obj={best_obj:.6f}, n={n}")

    while time.monotonic() - start < time_limit:
        # Varying perturbation scale
        scale = 0.05 + 0.15 * rng.random()
        width = int(rng.integers(max(2, n // 500), max(3, n // 8)))
        pos = int(rng.integers(0, max(1, n - width)))
        candidate = best_a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0, scale, size=width))
        candidate = np.clip(candidate, 0, None)
        cs = candidate.sum()
        if cs < 0.01:
            continue
        obj = compute_objective_fast(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate
            improvements += 1
            if verbose and improvements % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"    LS improve #{improvements}, obj={best_obj:.6f}, t={elapsed:.1f}s")

    if verbose:
        print(f"    LS done: {improvements} improvements, final obj={best_obj:.6f}")
    return best_a, best_obj
```
