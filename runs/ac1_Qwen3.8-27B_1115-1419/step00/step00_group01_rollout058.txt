```python
import numpy as np
import time

def compute_convolution(a):
    return np.convolve(a, a)

def compute_objective(a):
    n = len(a)
    b = np.convolve(a, a)
    M = b.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2 * n * M / (S * S)

def subgradient_step(a, b, active_k, lr):
    """Compute subgradient update step."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if j == i:
                grad[i] += 2.0 * a[i]
            else:
                grad[i] += a[j]
    return grad

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"Initial score: {best_score}, n={len(best)}")
    
    n = len(best)
    S = best.sum()
    
    # Maintain a "personal best" and also try different initializations
    all_bests = [(best.copy(), best_score)]
    
    iteration = 0
    stuck_count = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        b = np.convolve(best, best)
        M = b.max()
        S = best.sum()
        
        # Active set: top values
        top_indices = np.argsort(b)[-max(5, n//100):]
        
        # Learning rate
        lr = 0.5 * S * S / (M * len(top_indices) + 1e-10)
        
        # Subgradient
        grad = subgradient_step(best, b, top_indices, lr)
        
        # Update
        new_a = best - lr * grad
        new_a = np.maximum(new_a, 0.0)
        new_S = new_a.sum()
        if new_S > 0:
```
