"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using multi-start with
analytic shapes, LP-based ping-pong refinement, and subgradient local search.
"""
import numpy as np
import time
from scipy.sparse import coo_matrix
from scipy.optimize import linprog


def _make_gaussian(n, sigma_frac=0.25):
    j = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = sigma_frac * (n - 1)
    a = np.exp(-((j - c) ** 2) / (2.0 * sigma * sigma))
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _make_cosine_power(n, power=2):
    j = np.arange(n, dtype=np.float64)
    x = np.pi * j / max(n - 1, 1)
    a = (0.5 + 0.5 * np.cos(x)) ** power
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _make_smooth_bump(n, frac=0.9, sharp=4):
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / max(n - 1, 1) - 1.0
    a = np.where(np.abs(x) < frac, (1.0 - (np.abs(x) / frac) ** sharp) ** 2, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = np.convolve(a, a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def _simplex_project(v, target=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rng > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_max_conv(a, top_k=8):
    n = len(a)
    c = np.convolve(a, a)
    mc = int(min(top_k, 2 * n - 1))
    part = np.argpartition(-c, mc - 1)[:mc] if mc < 2 * n - 1 else np.arange(2 * n - 1)
    max_c = c.max()
    active = part[c[part] >= max_c * 0.9999]
    if len(active) == 0:
        active = np.array([part[0]])
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        ja = np.arange(j_start, j_end + 1)
        idx = (k - ja).astype(np.int64)
        contrib = 2.0 * a[idx]
        if k % 2 == 0:
            mid = k // 2
            if j_start <= mid <= j_end:
                pos = mid - j_start
                contrib[pos] = a[mid]
        grad[ja] += contrib
    grad /= len(active)
    return grad


def _grad_search(a, max_iters=5000, eta0=0.05, anneal=False):
    a = np.asarray(a, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a, np.inf
    a = a / S
    n = len(a)
    best_a = a.copy()
    best_obj = _compute_obj(a)
    for it in range(max_iters):
        grad = _subgrad_max_conv(a)
        if anneal:
            eta = eta0 / (1.0 + 0.05 * it)
        else:
            eta = eta0
        a_new = a - eta * grad
        a_new = _simplex_project(a_new, 1.0)
        obj = _compute_obj(a_new)
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    return best_a, best_obj


def _build_sparse_A(f):
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(js), k, dtype=np.int32))
        cols_list.append(js.astype(np.int32))
        vals_list.append(f[k - js].astype(np.float64))
    rows = np.concatenate(rows_list) if rows_list else np.array([], dtype=np.int32)
    cols = np.concatenate(cols_list) if cols_list else np.array([], dtype=np.int32)
    vals = np.concatenate(vals_list) if vals_list else np.array([], dtype=np.float64)
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def _lp_step(f, M, time_limit=15):
    n = len(f)
    A = _build_sparse_A(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit, 'disp': 0})
        if res.success and res.x is not None:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None
