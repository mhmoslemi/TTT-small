import time

def project_simplex(v):
    """Project v onto standard simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n+1).astype(float)) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0
    return w

    """Gradient of (a*a)_k = sum_j a_j * a_{k-j} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        grad[lo:hi] = a[k - np.arange(lo, hi)]
    if k % 2 == 0:
        j = k // 2
        if 0 <= j < n:
            grad[j] += a[j]
    return grad

    """Fast score computation (same as evaluate_sequence for valid input)."""
    n = len(a)
    c = np.convolve(a, a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

    a_prev = a.copy()

    k = 0
    
    while time.monotonic() < deadline:

        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a + ((t_old - 1.0) / t) * (a - a_prev)

        a_new = project_simplex(y - step * g)
        a_prev = a

                best_a = a.copy()
                if verbose and k % 1000 == 0:

        t_old = t

        y = a + momentum * (a - a_prev)

            g /= top_k

        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        k += 1

            sc = fast_score(a)

                best_score = sc
                best_a = a.copy()
                stuck_counter = 0
            else:
                stuck_counter += 1

                a = np.maximum(a, 0)
                s = a.sum()
                if s > 0:
                    a /= s
                a_prev = a.copy()
                t = 1.0
                step = 0.4
                stuck_counter = 0

    sc = fast_score(a)

        best_score = sc
        best_a = a.copy()

"""
Minimize 2n*max(a*a)/(sum(a))^2 using FISTA (accelerated projected gradient)
on the probability simplex. The problem is convex (min of max of convex
quadratics over a convex set), so FISTA converges to the global optimum.
Uses multiple starting points, top-k peak gradients, adaptive step size,
and periodic restarts to ensure robust convergence within the time budget.
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def grad_c_k(a, k):
    """Compute gradient of the k-th full-convolution coefficient w.r.t. a."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        idx = k - np.arange(lo, hi, dtype=np.int64)
        grad[lo:hi] = a[idx]
    if k % 2 == 0:
        j = k // 2
        if 0 <= j < n:
            grad[j] += a[j]
    return grad


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2 without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

    """
    Run FISTA to minimize max convolution coefficient on the simplex.
    Uses top-k peak gradient averaging for better convergence direction.
    """
    a = a0.copy()

        t_old = t

        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        # Keep y non-negative for stability
        np.maximum(y, 0.0, out=y)
        # Renormalize y to maintain feasibility
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.1:
            y /= sy

        # Compute convolution at y
        c_y = np.convolve(y, y)

        # Gradient from top-k peaks
        if top_k >= c_y.size:
            top_idx = np.arange(c_y.size)
        else:
            top_idx = np.argpartition(c_y, -top_k)[-top_k:]

        g = np.zeros(n, dtype=np.float64)
        for mk in top_idx:
            g += grad_c_k(y, int(mk))
        g /= float(len(top_idx))

        # Gradient step + project onto simplex
        a_new = y - step * g

        a = a_new

                t = 1.0

                a_prev = a.copy()
                t = 1.0

                stuck = 0

            if it > 5000 and top_k < 10:
