```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft, next_fast_len
import time

def objective_and_grad(a):
    """Compute objective and gradient of 2n*max(conv(a,a))/sum(a)^2"""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    # Self-convolution via FFT
    N = next_fast_len(2*n - 1)
    f = fft(a, N)
    b = np.real(ifft(f * f))[:2*n-1]
    
    k_star = int(np.argmax(b))
    m = b[k_star]
    
    obj = 2.0 * n * m / (S * S)
    
    # Gradient of b_{k_star} w.r.t. a
    # db_k/da_i = a_{k-i} + a_{k-i} (if i≠k-i) = 2*a_{k-i} (if i≠k-i) or a_i (if i=k/2)
    # More carefully:
    # b_k = sum_{j} a_j * a_{k-j} (j ranges over valid indices)
    # db_k/da_i: 
    #   from j=i: a_{k-i} (if valid)
    #   from k-j=i, j=k-i: a_{k-i} (if valid and k-i≠i)
    # So db_k/da_i = 2*a_{k-i} if 0<=k-i<n and i≠k-i
    #              = a_i if 2*i==k
    #              = 0 if k-i out of range
    
    grad_b = np.zeros(n)
    # Efficient computation: for valid i, j = k_star - i must be in [0, n-1]
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    for i in range(i_min, i_max + 1):
        j = k_star - i
        if i == j:
            grad_b[i] = a[i]
        else:
            grad_b[i] = 2.0 * a[j]
    
    # Vectorized version:
    # j indices: k_star - i for i in [i_min, i_max]
    # j ranges from k_star - i_max to k_star - i_min
    # All valid since i_min, i_max ensure 0 <= j < n
    
    # Actually the loop is O(k_star) at most, which is O(n). Fine.
    
    # df/da_i = 2n/S^3 * (S * grad_b_i - 2m)
    grad_f = (2.0 * n / (S**3)) * (S * grad_b - 2.0 * m)
    
    return obj, grad_f
```

Let me also consider: maybe the optimal sequence for large n is close to a specific distribution, like a Gaussian (which has a Gaussian autocorrelation, which is flat-ish). Or maybe it's a "power spectrum" shape.

In the frequency domain: b = a * a means B(ω) = |A(ω)|^2. We want max(b) to be small, which by the uncertainty principle means A(ω) should be "spread out" in frequency. But since a ≥ 0, A(ω) has constraints.

- The peak of b is at k=0 (or k=n-1 depending on centering)
- The ratio max(b)/S^2 for a 1D Gaussian is roughly proportional to 1/(σ√n) or something

```python
"""
Optimizes a non-negative step function sequence to minimize the self-convolution
peak-to-mean ratio using gradient-based optimization (L-BFGS-B) with FFT-accelerated
convolution. Tries multiple starting points and lengths.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft, next_fast_len
import time

def _obj(x):
    """Objective: 2n * max(convolve(x,x)) / sum(x)^2"""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    S = np.sum(x)
    if S < 1e-15:
        return 1e10
    N = next_fast_len(2 * n - 1)
    F = fft(x, N)
    b = np.real(ifft(F * F))[:2 * n - 1]
    return 2.0 * n * np.max(b) / (S * S)

def _grad(x):
    """Gradient of _obj"""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    S = np.sum(x)
    if S < 1e-15:
        return np.zeros(n)
    N = next_fast_len(2 * n - 1)
    F = fft(x, N)
    b = np.real(ifft(F * F))[:2 * n - 1]
    k_star = int(np.argmax(b))
    m = b[k_star]
    
    # grad_b[i] = 2*x[k_star - i] (with bounds), halved if i == k_star - i
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    if i_lo > i_hi:
        grad_b = np.zeros(n)
    else:
        idx = np.arange(i_lo, i_hi + 1)
        j = k_star - idx
        grad_b = np.zeros(n)
        grad_b[idx] = 2.0 * x[j]
        # Fix diagonal
        if k_star % 2 == 0:
            mid = k_star // 2
            if i_lo <= mid <= i_hi:
                grad_b[mid] = x[mid]
    
    return (2.0 * n / (S * S * S)) * (S * grad_b - 2.0 * m)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-start gradient optimization for minimal self-convolution peak.
    Uses FFT-based convolution and L-BFGS-B with analytical gradients.
    """
    rng = np.random.default_rng(seed)
    t_end = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    
    best_score = float('inf')
    best_seq = None
    
    # Gather starting seeds (lengths)
    lengths = [
        64, 128, 256, 512, 1024, 2048, 4096
    ]
    
    # Add provided sequence if available
    prov = None
    try:
        prov = np.asarray(height_sequence_1, dtype=np.float64)
        prov = np.clip(prov, 0, 1000)
        if prov.size > 10 and prov.size < 100000:
            lengths.append(prov.size)
    except:
        pass
    
    # Remove duplicates and sort
    lengths = sorted(set(lengths))
    
    for n in lengths:
        if time.monotonic() > t_end:
            break
        
        # Generate initializations
        inits = []
        
        # Geometric sequences
        for r in [0.9, 0.95, 1.0, 1.02, 1.05, 1.1, 1.15, 1.2, 1.3, 1.5]:
            a = r ** np.arange(n, dtype=np.float64)
            if r > 1:
                a = a[::-1]  # Try both orientations
            inits.append(a)
            inits.append(a[::-1].copy())
        
        # Uniform
        inits.append(np.ones(n))
        
        # Random (exponential, power-law)
        for _ in range(5):
            a = rng.exponential(1.0, n)
            inits.append(a)
        for _ in range(5):
            a = rng.standard_normal(n)
            a = np.abs(a) ** 2 + 0.1
            inits.append(a)
        
        # Windowed
        inits.append(np.hanning(n))
        inits.append(np.bartlett(n))
        tri = 1.0 - np.abs(np.arange(n) - (n-1)/2) / ((n-1)/2 + 1e-10)
        inits.append(np.maximum(tri, 0))
        
        # Exponential decay (different rates)
        for rate in [0.01, 0.02, 0.05, 0.1, 0.2]:
            t = np.arange(n, dtype=np.float64) / n
            inits.append(np.exp(-rate * t * n))
        
        # If we have the provided sequence and this length matches
        if prov is not None and prov.size == n:
            inits.append(prov.copy())
        
        # Limit number of inits per length to manage time
        max_inits = 40
        if len(inits) > max_inits:
            # Sample
            idx = rng.choice(len(inits), max_inits, replace=False)
            inits = [inits[i] for i in idx]
        
        for init in inits:
            if time.monotonic() > t_end:
                break
            
            # Normalize and clip
            init = np.clip(init, 1e-6, 1000.0)
            if np.sum(init) < 1e-10:
                continue
            init = init / np.sum(init) * 1.0  # Normalize to sum=1
            
            # Set bounds
            bounds = [(0.0, 1000.0)] * n
            
            # L-BFGS-B
            maxiter = max(100, min(2000, 500000 // max(1, n)))
            try:
                res = minimize(
                    fun=_obj,
                    x0=init,
                    jac=_grad,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={
                        'maxiter': maxiter,
                        'maxfun': maxiter * 5,
                        'ftol': 1e-14,
                        'gtol': 1e-10,
                        'maxls': 50,
                    }
                )
                score = _obj(res.x)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = np.clip(res.x, 0, 1000).copy()
                    elapsed = time.monotonic()
                    remaining = t_end - elapsed
                    print(f"n={n}, score={best_score:.6f}, iters={res.nit}, time_left={remaining:.0f}s")
            except Exception as e:
                print(f"  Error for n={n}: {e}")
    
    # If no valid result, fallback
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(256, dtype=np.float64)
        best_score = _obj(best_seq)
    
    # Final verification
    final_score = _obj(best_seq)
    if not np.isfinite(final_score) or final_score > 1e6:
        best_seq = np.ones(256, dtype=np.float64)
        final_score = _obj(best_seq)
    
    print(f"Final: n={len(best_seq)}, score={final_score:.6f}")
    return [float(x) for x in best_seq]
```
