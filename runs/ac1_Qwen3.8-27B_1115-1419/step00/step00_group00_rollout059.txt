"""
Step function sequence optimization using subgradient descent and LP-based refinement.
Minimizes 2n*max(conv(a,a))/sum(a)^2 by iterative gradient and LP steps.
"""
import numpy as np
import time
from scipy.sparse import csc_matrix
from scipy.optimize import linprog

def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:n]

def _fast_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def _grad_peak(a, top_k=3):
    n = len(a)
    c = _fft_conv(a, a)
    # Get top_k peaks
    idx = np.argsort(c)[::-1][:top_k]
    weights = np.zeros(len(c))
    for j, k in enumerate(idx):
        weights[k] = 1.0 / (j + 1)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k_val, w in zip(idx, weights[idx]):
        j = k_val - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad[mask] += w * a[j[mask]]
    
    # Center the gradient (remove mean to preserve sum direction)
    grad -= grad.mean()
    return grad, c

def _next_pow2(m):
    return 1 << (m - 1).bit_length() if m > 1 else 2


def _self_conv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _score(a):
    """Fast score computation (scale-invariant)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _peak_grad(a, top_k=5):
    """Subgradient of the peak convolution term."""
    n = len(a)
    c = _self_conv(a)
    # Top-k peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    # Weights: reciprocal rank
    wts = np.array([1.0 / (r + 1) for r in range(top_k)])
    wts /= wts.sum()
    
    grad = np.zeros(n)
    for w, k in zip(wts, top_idx):
        j = k - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad[mask] += w * a[j[mask]]
    
    # Center to preserve sum (approximately)
    grad -= grad.mean()
    return grad, c

    col_parts = [np.full(n, j) for j in range(n)]

def _build_conv_matrix(a):
    n = len(a)

    # Column j has non-zeros at rows j, j+1, ..., j+n-1 with values a[0], ..., a[n-1]

        col_ptr[j + 1] = col_ptr[j] + n

    vals = np.zeros(n * n, dtype=np.float64)

        vals[j*n : (j+1)*n] = a
    
    return csc_matrix((vals, rows, col_ptr), shape=(n_rows, n))

def _lp_direction_simple(a, top_k_constraints=20):
    n = len(a)
    c = _self_conv(a)
    M = float(c.max())
    
    # Select top_k_constraints peaks
    top_idx = np.argsort(c)[::-1][:top_k_constraints]
    
    # Build constraint rows
    rows = []
    cols = []
    vals = []
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            rows.append(len(rows))  # row index
            cols.append(j)
            vals.append(a[i])
    
    A = csc_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    b = np.full(len(top_idx), M)
    obj = -np.ones(n)
    
    res = linprog(obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

```python
"""
Step function sequence optimizer: minimizes 2n*max(conv(a,a))/sum(a)^2
using multi-start subgradient descent, top-k LP refinement, and polishing.
"""

import numpy as np
import time
from scipy.sparse import csc_matrix
from scipy.optimize import linprog


def _fft_next(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _self_conv(a):
    n = len(a)
    N = _fft_next(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _fast_score(a):
    S = a.sum()
    if S < 0.01:
        return np.inf
    n = len(a)
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (S * S))
