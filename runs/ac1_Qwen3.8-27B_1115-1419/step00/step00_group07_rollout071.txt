"""
Search for a non-negative sequence minimizing  2*n*max(conv(a,a))/sum(a)^2 .

Method:
  * The objective is scale invariant, so we keep sum(a)=1 inside the search and
    minimize  2*n*max(conv(a,a))  (conv computed with FFT, O(n log n)).
  * Optimization is a mirror-descent / multiplicative-weight step driven by the
    sub-gradient at the highest autocorrelation lags (top-K), with an EMA of the
    "pressure" to smooth oscillations and an adaptive (annealed) step size.
  * Several diverse initializers (structured + random + parent) are explored in
    parallel time-slices; occasional random multi-plicative kicks help escape
    local minima. The always-authoritative evaluate_sequence() is used for the
    final (and periodic) check.
"""
import numpy as np
import time


# -----------------------------------------------------------------------------
# Fast helpers (top level, no closures, no lambdas)
# -----------------------------------------------------------------------------
def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _fast_conv_full(a):
    """Full linear convolution a * a via FFT (a is a 1-D real array)."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    fa = np.fft.rfft(a, N)
    spec = fa * fa
    c = np.fft.irfft(spec, N)[:size]
    return c


def _obj(a):
    """Internal objective = 2*n*max(conv)/sum^2, computed fast (sum not forced)."""
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _fast_conv_full(a)
    mb = float(np.max(c))
    return 2.0 * n * mb / (s * s)


def _normalize(a, target):
    s = float(np.sum(a))
    if s <= 0:
        return a
    return a * (target / s)


def _pressure(f, k, n):
    """Sub-gradient proxy for lag k: vector v[i] = f[k-i] (zero-padded)."""
    v = np.zeros(n, dtype=np.float64)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo > i_hi:
        return v
    idx = k - np.arange(i_lo, i_hi + 1)
    v[i_lo:i_hi + 1] = f[idx]
    return v


def _step(f, alpha, K, ema, decay, n):
    """One mirrored-descent step; returns (new_f, new_obj, ema)."""
    c = _fast_conv_full(f)
    if K >= c.size:
        top = np.arange(c.size)
    else:
        top = np.argpartition(c, -K)[-K:]

    p = np.zeros(n, dtype=np.float64)
    for k in top:
        p += _pressure(f, int(k), n)
    p /= float(K)

    mx = float(np.max(p))
    if mx > 0:
        p = p / mx

    if ema is None:
        ema = p
    else:
        ema = decay * ema + (1.0 - decay) * p

    f_new = f * np.exp(-alpha * ema)
    f_new = np.maximum(f_new, 0.0)
    s = float(np.sum(f_new))
    if s <= 0:
        f_new = f
        s = float(np.sum(f))
    f_new = f_new / s          # keep sum == 1 (scale invariance)
    return f_new, _obj(f_new), ema


def _make_two_level(n, ratio):
    f = np.empty(n, dtype=np.float64)
    half = n // 2
    f[:half] = ratio
    f[half:] = 1.0
    return f


def _make_cosine(n, freq, amp):
    i = np.arange(n, dtype=np.float64)
    f = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
    return np.maximum(f, 0.0)


def _make_random_blocks(n, nblocks, rng):
    f = np.empty(n, dtype=np.float64)
    bounds = sorted(rng.integers(1, n, size=nblocks - 1))
    cuts = np.concatenate(([0], bounds, [n]))
    vals = rng.uniform(0.3, 3.0, size=len(cuts) - 1)
    for j in range(len(cuts) - 1):
        f[cuts[j]:cuts[j + 1]] = vals[j]
    return f


def _make_random_smooth(n, rng):
    # low-frequency random field -> smooth non-negative profile
    freqs = rng.integers(1, 5, size=4)
    amps = rng.uniform(0.2, 1.0, size=4)
    ph = rng.uniform(0.0, 2.0 * np.pi, size=4)
    i = np.arange(n, dtype=np.float64)
    f = 1.0
    for fi, ai, pi in zip(freqs, amps, ph):
        f = f + ai * np.cos(2.0 * np.pi * fi * i / n + pi)
    return np.maximum(f, 0.0)


def _optimize_init(f0, deadline):
    """Run mirrored descent on one initialization; return (best_f, best_obj)."""
    n = f0.size
    f = _normalize(np.maximum(f0, 0.0), 1.0)
    cur = _obj(f)
    best_f = f.copy()
    best = cur

    ema = None
    decay = 0.80
    alpha = 0.08
    K = 1
    stall = 0
    it = 0
    rng = np.random.default_rng(hash(int(n)) & 0xFFFFFFFF)

    while time.monotonic() < deadline:
        f2, o2, ema = _step(f, alpha, K, ema, decay, n)
        it += 1
        if o2 < cur:
            cur = o2
            f = f2
            stall = 0
            if cur < best:
                best = cur
                best_f = f.copy()
            # reward progress: grow step a little and widen K
            alpha = min(0.30, alpha * 1.05)
            K = min(6, K + (1 if it % 400 == 0 else 0))
        else:
            stall += 1
            # keep the tried candidate only if not worse by much (anneal)
            if o2 < best * 1.02 and rng.random() < 0.3:
                f = f2
                cur = o2
            if stall >= 50:
                # anneal: shrink step, widen K, maybe kick
                alpha = max(0.01, alpha * 0.6)
                K = min(6, K + 1)
                ema = None
                stall = 0
                if rng.random() < 0.5:
                    kick = np.exp(rng.normal(0.0, 0.25, n))
                    f = _normalize(np.maximum(f * kick, 0.0), 1.0)
                    cur = _obj(f)
        if it % 2000 == 0:
            print("    iter %d  cur=%.6f  best=%.6f  alpha=%.4f K=%d" %
                  (it, cur, best, alpha, K))
    return best_f, best
