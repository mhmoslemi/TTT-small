"""

def fast_autocorr(a):
    """Compute full linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def my_score(a):
    """Compute the objective: 2n*max(a*a)/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    M = conv.max()
    return float(2.0 * n * M / (s * s))


def compute_pressure(a, S):
    """Compute pressure: p_i = sum_{k in S} a[k-i] for valid (k,i)."""
    n = len(a)
    s = np.zeros(2 * n - 1)
    s[S] = 1.0
    a_rev = a[::-1]
    conv = fftconvolve(s, a_rev, mode='full')
    p = conv[n - 1 : n - 1 + n]
    return p


def multiplicative_step(a, eta, peak_fraction, max_peaks):
    """One multiplicative pressure update step. Returns (new_a, M, S)."""
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()

    # Select near-peak positions
    threshold = M * (1.0 - peak_fraction)
    S = np.where(conv >= threshold)[0]
    if len(S) > max_peaks:
        S = np.argsort(conv)[::-1][:max_peaks]
    if len(S) == 0:
        S = np.array([np.argmax(conv)])

    p = compute_pressure(a, S)
    p = np.maximum(p, 1e-15)
    p = p / p.mean()

    a_new = a * np.exp(-eta * p)
    a_new = np.clip(a_new, 1e-15, 1000.0)
    total = a_new.sum()
    if total > 1e-15:
        a_new /= total
    else:
        a_new = a.copy()

    n = len(a)

    if len(S) == 0:
        S = np.array([np.argmax(conv)])

    p = compute_pressure(a, S)
    p = p / max(p.mean(), 1e-15)

    a_new = a - eta * p
    # Project onto simplex {x >= 0, sum(x) = 1}
    # Using the bisection method for tau
    a_new = np.maximum(a_new, 0.0)
    total = a_new.sum()
    if total > 0:
        a_new /= total
    else:
        a_new = a.copy()

    return a_new, M, S

                best = a.copy()
            else:
                a[start:stop] = old_vals
                a /= a.sum()

    # Verify with actual evaluator
    final_list = [float(max(0.0, x)) for x in best]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.6f}, n={len(final_list)}")

    return final_list
