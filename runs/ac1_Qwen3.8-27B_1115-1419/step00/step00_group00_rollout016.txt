```python
"""
LP-based alternating optimization to minimize 2*n*max(f*f)/(sum f)^2
for non-negative sequences. Each iteration solves an LP to find the most
"spread out" sequence g satisfying (f*g)(k) <= M for all lags, then blends
towards g. Combined with random restarts and multi-length exploration.
"""
import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time


def build_conv_matrix(f):
    """Build sparse (2n-1, n) matrix A such that (A@g)(k) = (f*g)(k)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        rows.extend([k] * len(j))
        cols.extend(j.tolist())
        vals.extend(f[k - j].tolist())
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_objective(f):
    """Compute 2*n*max(convolve(f,f)) / sum(f)^2."""
    n = len(f)
    s = np.sum(f)
    if s < 1e-12:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * np.max(conv) / (s * s))


def lp_iteration(f, M, lp_timeout=30.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None on failure."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': lp_timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def soft_max_approx(conv, k=1000.0):
    """Approximate max(conv) via log-sum-exp for gradient computation."""
    m = np.max(conv)
    if m == 0:
        return 0.0
    shifted = (conv - m) * k
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / k


def gradient_step(f, lr, k_soft=500.0):
    """One gradient step on smoothed objective 2*n*logsumexp(conv)/sum^2."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-12:
        return f
    # Gradient of logsumexp(conv)/sum^2 w.r.t. f
    m = np.max(conv)
    shifted = (conv - m) * k_soft
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # d(logsumexp(conv))/df(j) = sum_k weights[k] * d(conv[k])/df(j)
    # d(conv[k])/df(j) = f(k-j) if valid
    grad = np.zeros(n)
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 2, 2 * j + n - 1)
        # conv[k] involves f(j) when k-j is in [0,n-1], i.e., k in [j, j+n-1]
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            # conv[k] = sum_i f(i)*f(k-i), d/dF(j) = f(k-j) if 0<=k-j<n
            ks = np.arange(k_lo, k_hi + 1)
            idx = k - ks  # = k - j, should be in [0, n-1]
            contrib = f[idx] * weights[ks]
            grad[j] = np.sum(contrib)
    # Full gradient: d/dF of (2n/max... ) ≈ 2n * grad / s^2 - 4n * conv_based / s^3 * f
    # Simplified: just use -grad/s as descent direction on max_conv/sum^2
    grad = grad / (s * s)
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0.0)
    if f_new.sum() > 1e-12:
        f_new = f_new / f_new.sum()
    else:
        f_new = f.copy()
    return f_new

    elif shape == 'bimodal':
        f = np.exp(-((np.arange(n) - n * 0.3) ** 2) / (2 * (n * 0.15) ** 2)) + \
            np.exp(-((np.arange(n) - n * 0.7) ** 2) / (2 * (n * 0.1) ** 2))
    elif shape == 'random_smooth':
        f = np.abs(rng.normal(1.0, 0.5, n))
        # Smooth it
        kernel = np.ones(5) / 5.0
        f = np.convolve(f, kernel, mode='same')
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.0)
    fsum = f.sum()
    if fsum > 0:
        f = f / fsum
    else:
        f = np.ones(n) / n
    return f


def run_lp_search(f0, max_iters=200, t0=0.3, t_decay=0.999, lp_timeout=20.0):
    """Run LP-based alternating optimization. Returns best f found."""
    f = f0.copy()
    n = len(f)
    f = np.maximum(f, 0.0)
    fsum = f.sum()
    if fsum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / fsum
    best_f = f.copy()
    best_obj = compute_objective(f)
    t = t0
    for it in range(max_iters):
        M = float(np.max(np.convolve(f, f)))
        g = lp_iteration(f, M, lp_timeout=lp_timeout)
        if g is None:
            break
        # Blend
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        fsum = f_new.sum()
        if fsum < 1e-12:
            break
        f_new = f_new / fsum
        obj = compute_objective(f_new)
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
        f = f_new
        t = max(0.05, t * t_decay)
        if it % 20 == 0:
            print(f"  iter {it}: obj={best_obj:.6f}, M={M:.8f}, t={t:.4f}")
    return best_f, best_obj


def run_gradient_refine(f, max_iters=100, lr0=0.01, lr_decay=0.99, k_soft=500.0):
    """Fine-tune with gradient descent on smoothed objective."""
    f = f.copy()
    n = len(f)
    f = np.maximum(f, 0.0)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = compute_objective(f)
    lr = lr0
    for it in range(max_iters):
        f = gradient_step(f, lr, k_soft=k_soft)
        obj = compute_objective(f)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
        lr *= lr_decay
        if it % 30 == 0:
            print(f"  grad iter {it}: obj={best_obj:.6f}, lr={lr:.6f}")
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: LP-based optimization with multiple lengths and restarts."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Initial sequence from global
    best_f = None
    best_obj = np.inf
    best_n = 0
    
    # Try to use height_sequence_1 as starting point
    initial_seq = None
    try:
        initial_seq = np.asarray(height_sequence_1, dtype=np.float64)
        initial_seq = np.clip(initial_seq, 0.0, 1000.0)
    except NameError:
        pass
    except Exception:
        pass
    
    # Phase 1: Try multiple sequence lengths with LP search
    # Most of the time budget goes to the best-performing length
    lengths_to_try = [100, 200, 500, 1000, 2000]
    if initial_seq is not None and initial_seq.size > 0:
        orig_n = initial_seq.size
        if orig_n not in lengths_to_try:
            lengths_to_try.append(orig_n)
    
    # Time allocation: give more time to medium/large sequences
    # Quick exploration first
    alloc = {}
    for n in lengths_to_try:
        if n <= 200:
            alloc[n] = 60
        elif n <= 500:
            alloc[n] = 120
        elif n <= 1000:
            alloc[n] = 200
        elif n <= 2000:
            alloc[n] = 300
        else:
            alloc[n] = 400
    
    print(f"Searching with lengths: {lengths_to_try}")
    print(f"Time allocation: {alloc}")
    
    results = {}  # n -> (best_f, best_obj)
    
    for n in sorted(lengths_to_try):
        if time.monotonic() >= deadline:
            break
        time_left = deadline - time.monotonic()
        if time_left < 30:
            break
        
        phase_time = min(alloc[n], time_left - 10)
        print(f"\n=== Length n={n}, time budget {phase_time:.0f}s ===")
        
        # Try multiple initializations
        inits = ['decreasing', 'exp_decreasing', 'bimodal', 'random_smooth', 'uniform']
        if initial_seq is not None and n == initial_seq.size:
            inits.insert(0, 'from_sequence')
        
        rng_local = np.random.default_rng(seed + n)
        
        for init_name in inits:
            if time.monotonic() >= deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            if init_name == 'from_sequence' and initial_seq is not None:
                f0 = initial_seq.astype(np.float64).copy()
                f0 = np.clip(f0, 0, 1000)
                fsum = f0.sum()
                if fsum > 0:
                    f0 = f0 / fsum
                else:
                    f0 = np.ones(n) / n
            else:
                f0 = random_init(n, rng_local, shape=init_name)
            
            # Quick evaluation
            obj0 = compute_objective(f0)
            print(f"  Init '{init_name}': obj={obj0:.6f}")
            
            # LP search with this initialization
            # Limit iterations based on time
            max_it = int(remaining / 3.0)  # ~3s per LP iteration
            max_it = max(5, min(max_it, 300))
            
            f_best, obj_best = run_lp_search(f0, max_iters=max_it, t0=0.3, 
                                              t_decay=0.999, lp_timeout=15.0)
            print(f"  After LP: obj={obj_best:.6f}")
            
            # Gradient refinement (short)
            grad_time = min(30, remaining - 5)
            if grad_time > 5:
                max_grad = int(grad_time / 0.1)
                f_best, obj_best = run_gradient_refine(f_best, max_iters=max_grad,
                                                       lr0=0.005, lr_decay=0.995,
                                                       k_soft=300.0)
                print(f"  After grad: obj={obj_best:.6f}")
            
            # Validate with evaluate_sequence
            # Scale to avoid numerical issues
            scale = 1.0
            f_eval = f_best * scale
            valid_obj = evaluate_sequence(f_eval.tolist())
            if np.isfinite(valid_obj) and valid_obj < best_obj:
                best_obj = valid_obj
                best_f = f_best.copy()
                best_n = n
                print(f"  *** NEW BEST: obj={best_obj:.6f} (n={n}, init={init_name}) ***")
            
            # Store result for this n
            if n not in results or obj_best < results[n][1]:
                results[n] = (f_best.copy(), obj_best)
        
        # Continue refining best for this length if time remains
        if n in results and time.monotonic() < deadline - 30:
            f_cur, obj_cur = results[n]
            remaining = deadline - time.monotonic() - 10
            if remaining > 10:
                print(f"  Continuing LP for n={n} with {remaining:.0f}s remaining...")
                max_it = int(remaining / 3.0)
                f_ref, obj_ref = run_lp_search(f_cur, max_iters=max_it, t0=0.15,
                                                t_decay=0.9995, lp_timeout=10.0)
                if obj_ref < obj_cur:
                    results[n] = (f_ref, obj_ref)
                    f_eval = f_ref
                    valid_obj = evaluate_sequence(f_eval.tolist())
                    if np.isfinite(valid_obj) and valid_obj < best_obj:
                        best_obj = valid_obj
                        best_f = f_ref.copy()
                        best_n = n
                        print(f"  *** NEW BEST: obj={best_obj:.6f} (n={n}) ***")
    
    # Final phase: intensively refine the best found sequence
    if best_f is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        print(f"\n=== Final refinement for n={best_n}, {remaining:.0f}s left ===")
        # More LP iterations with smaller step
        f_cur = best_f.copy()
        for round_idx in range(5):
            if time.monotonic() >= deadline - 10:
                break
            max_it = 50
            f_ref, obj_ref = run_lp_search(f_cur, max_iters=max_it, t0=0.1,
                                            t_decay=0.9995, lp_timeout=8.0)
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_f = f_ref.copy()
                f_eval = best_f
                valid_obj = evaluate_sequence(f_eval.tolist())
                if np.isfinite(valid_obj) and valid_obj < best_obj:
                    best_obj = valid_obj
                    best_f = f_ref.copy()
                print(f"  Round {round_idx}: obj={best_obj:.6f}")
            f_cur = f_ref.copy()
        
        # Gradient polish
        if time.monotonic() < deadline - 10:
            f_ref, obj_ref = run_gradient_refine(best_f, max_iters=200, lr0=0.002,
                                                  lr_decay=0.998, k_soft=1000.0)
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_f = f_ref.copy()
            f_eval = best_f
            valid_obj = evaluate_sequence(f_eval.tolist())
            if np.isfinite(valid_obj):
                if valid_obj < best_obj:
                    best_obj = valid_obj
                print(f"  Final: obj={best_obj:.6f}")
    
    # Fallback: ensure we return a valid sequence
    if best_f is None or not np.isfinite(best_obj):
        print("WARNING: No valid solution found, using fallback")
        f_fb = np.ones(100)
        best_f = f_fb
        best_obj = compute_objective(f_fb)
    
    # Scale the sequence to a reasonable range (avoid all-zeros after evaluation)
    # The evaluator clips to [0, 1000] and requires sum > 0.01
    # Scale so max is around 500 (well within bounds)
    f_out = best_f.copy()
    f_out = np.maximum(f_out, 0.0)
    fsum = f_out.sum()
    if fsum > 0:
        # Scale so sum is moderate (e.g., 1000) to stay well within [0,1000] per element
        target_sum = 1000.0
        scale = target_sum / fsum
        f_out = f_out * scale
        # Clip to [0, 1000]
        f_out = np.clip(f_out, 0.0, 1000.0)
    
    result = [float(x) for x in f_out]
    final_score = evaluate_sequence(result)
    print(f"\nFinal sequence: n={len(result)}, evaluate_score={final_score:.6f}")
    
    return result
```
