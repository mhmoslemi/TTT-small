Optimization of step function coefficients to minimize 2n*max(a*a)/(sum(a))^2.
Uses LP-based iterative refinement (finding g that maximizes sum subject to f*g<=M)
combined with local search. Multiple starting points and sizes are explored.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix(f):
    """Build sparse convolution matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    max_k = 2 * n - 1
    # For each column j, rows k = j to min(j+n-1, 2n-2)
    row_parts = []
    col_parts = []
    val_parts = []
    for j in range(n):
        r = np.arange(j, min(max_k, j + n))
        row_parts.append(r)
        col_parts.append(np.full(len(r), j, dtype=np.int32))
        val_parts.append(np.full(len(r), f[j]))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    return csr_matrix((vals, (rows, cols)), shape=(max_k, n))

def solve_lp(f, M, time_limit=15):
    """Solve max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit})
    if res.success:
        return res.x
    return None

def simplex_project(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    v = v.copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_step(f, eta):
    """One subgradient step on the normalized objective."""
    n = len(f)
    h = np.convolve(f, f)
    k_star = int(np.argmax(h))
    # Subgradient: s[j] = 2*f[k_star - j] if 0 <= k_star - j < n
    s = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(k_star, n - 1)
    for j in range(j_lo, j_hi + 1):
        s[j] = 2 * f[k_star - j]
    # Mirror descent update on simplex
    f_new = f * np.exp(-eta * s)
    f_new = f_new / f_new.sum()
    return f_new

"""
Optimize step function coefficients to minimize 2n*max(a*a)/(sum(a))^2.
Uses LP-based iterative refinement (max sum(g) s.t. f*g<=M, g>=0),
subgradient descent, multi-start with diverse initializations, and local search.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = (f*g)_k for k=0..2n-2."""
    n = len(f)
    max_k = 2 * n - 1
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        r_end = min(max_k, j + n)
        length = r_end - j
        if length > 0:
            rows_list.append(np.arange(j, r_end, dtype=np.int32))
            cols_list.append(np.full(length, j, dtype=np.int32))
            vals_list.append(np.full(length, f[j]))
    if not rows_list:
        return csr_matrix((max_k, n), dtype=np.float64)
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(max_k, n))


def solve_lp_for_g(f, M, time_limit=12):
    """Solve max sum(g) s.t. conv_matrix(f)@g <= M, g>=0. Returns g or None."""
    n = len(f)
    try:
        A = build_conv_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def simplex_project(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_direction(f):
    """Compute subgradient of max(avg_conv) and return the direction to reduce it."""
    n = len(f)
    h = np.convolve(f, f)
    k_star = int(np.argmax(h))
    s = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(k_star, n - 1)
    if j_hi >= j_lo:
        s[j_lo:j_hi + 1] = 2.0 * f[k_star - j_hi : k_star - j_lo - 1 : -1]
    return s, k_star, h


def run_subgradient(f, n_iters, eta0, deadline, label=""):
    """Run subgradient iterations, return best found."""
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    eta = eta0
    f_use = f.copy()
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        s, k_star, h = subgradient_direction(f_use)
        f_try = f_use.copy()
        for j in range(len(f_use)):
            f_try[j] -= eta * s[j]
        f_try = np.maximum(f_try, 0.0)
        f_try = f_try / f_try.sum()
        sc = evaluate_sequence(f_try.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_try.copy()
            f_use = f_try.copy()
        eta *= 0.999
        if it > 0 and it % 500 == 0 and (it % 2000 == 0):
            eta = max(0.001, eta0 / (it + 1) * 5)
    return best_f, best_score

        best_seq, best_score = local_search(best_seq, dline, rng)
        print(f"  Final local search score: {best_score:.6f}")

    # Phase 3b: Try also with slightly different n (resize best and re-search)
    if best_seq is not None and time.monotonic() < deadline - 20:
        for target_n in [len(best_seq) - 100, len(best_seq) + 100]:
            if target_n < 100 or time.monotonic() > deadline - 15:
                continue
            x_new = np.linspace(0, 1, target_n)
            x_old = np.linspace(0, 1, len(best_seq))
            f_resized = np.interp(x_new, x_old, best_seq)
            if f_resized.sum() > 0.01:
                f_resized, sc = local_search(f_resized / f_resized.sum(),
                                              min(time.monotonic() + 20, deadline - 5), rng)
                if sc < best_score:
                    best_score = sc
                    best_seq = f_resized
                    print(f"  Resized to n={target_n}: {best_score:.6f}")

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(200, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones(200, dtype=np.float64) * 5.0

    # Scale so the values are in a comfortable range for the evaluator
    scale = 5.0 / best_seq.sum()
    result = best_seq * scale
    result = np.clip(result, 0.0, 1000.0)

    final_score = evaluate_sequence([float(x) for x in result])
    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL: score={final_score:.6f}, n={len(result)}, elapsed={elapsed:.0f}s ===")

    return [float(x) for x in result]
