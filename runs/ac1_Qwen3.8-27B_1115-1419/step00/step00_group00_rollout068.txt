Given sorted q (descending), we want to find m (number of positive elements) and tau such that:
- p_i = max(0, q_i - tau)
- sum(p) = 1
- The m largest q_i - tau > 0

So: sum_{i=1}^{m} (q_sorted[i-1] - tau) = 1
=> m * tau = sum_{i=1}^{m} q_sorted[i-1] - 1
=> tau = (cumsum[m-1] - 1) / m

Validity conditions:
- tau >= 0: cumsum[m-1] >= m (the sum of top m >= m, i.e., avg >= 1)
- tau <= q_sorted[m-1] (the m-th element): (cumsum[m-1] - 1)/m <= q_sorted[m-1]
  => cumsum[m-1] - 1 <= m * q_sorted[m-1]
  => cumsum[m-1] - m*q_sorted[m-1] <= 1
  => sum_{i=0}^{m-2} q_sorted[i] - (m-1)*q_sorted[m-1] <= 1
  (the "excess" of the top m-1 over the m-th is <= 1)
- If m < n: tau >= q_sorted[m] (the (m+1)-th element is below the water level):

Hybrid subgradient + random search for minimizing 2n*max(conv(p,p))/sum(p)^2.
Uses FFT convolution, water-filling projection, and adaptive step sizes.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

def fast_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    # Next power of 2 for efficiency
    m = 1
    while m < n:
        m *= 2
    A = fft(a, m)
    B = fft(b, m)
    result = ifft(A * B, m)[:n]
    return np.real(result)

def project_waterfill(q):
    """Project q onto {p >= 0, sum(p) = 1}."""
    n = len(q)
    if n == 0:
        return q.copy()
    if np.all(q <= 0):
        return np.ones(n) / n
    
    q_sorted = np.sort(q)[::-1]
    cumsum = np.cumsum(q_sorted)
    
    # Find m: number of positive elements in the projection
    # tau = (cumsum[m-1] - 1) / m
    # Need: tau > 0, tau >= q_sorted[m] (if m < n), tau <= q_sorted[m-1]
    
    # Vectorized approach:
    # For each m, tau_m = (cumsum[m-1] - 1) / m
    # Condition: tau_m >= q_sorted[m] (for m < n)
    # i.e., cumsum[m-1] - 1 >= m * q_sorted[m]
    # i.e., cumsum[m-1] - m*q_sorted[m] >= 1
    
    # And tau_m > 0: cumsum[m-1] > m
    # And tau_m <= q_sorted[m-1]: cumsum[m-1] - 1 <= m * q_sorted[m-1]
    # (the latter is always true since cumsum[m-1] <= m * q_sorted[0] and q_sorted[m-1] >= q_sorted[m]...)
    # Actually it's not always true. Let me just do a linear scan.
    
    best_m = n  # default: all positive
    for m in range(1, n + 1):
        tau = (cumsum[m - 1] - 1.0) / m
        if tau <= 0:
            continue
        if m < n and tau < q_sorted[m]:
            continue
        best_m = m
        break
    
    tau = (cumsum[best_m - 1] - 1.0) / best_m
    tau = max(0.0, tau)
    p = np.maximum(0.0, q - tau)
    s = p.sum()
    if s < 1e-15:

def compute_score(p):
    """Compute 2*n*max(conv(p,p)) for sum(p)=1."""
    n = len(p)
    c = fast_convolve(p, p)
    return 2.0 * n * np.max(c)

def subgradient_at_peak(p, k_star):
    """Compute gradient of (p*p)_{k_star} w.r.t. p."""
    n = len(p)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo > hi:
        return grad
    # p[k_star - i] for i in [lo, hi]
    # i = lo => k_star - lo (largest index in p)
    # i = hi => k_star - hi (smallest index in p)
    # So p_slice = p[k_star-hi : k_star-lo+1] reversed
    p_slice = p[k_star - hi : k_star - lo + 1]
    grad[lo:hi+1] = 2.0 * p_slice[::-1]
    return grad

def weighted_subgradient(p, peaks, weights):
    """Compute weighted sum of gradients at multiple peaks."""
    n = len(p)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        p_slice = p[k - hi : k - lo + 1]
        grad[lo:hi+1] += w * 2.0 * p_slice[::-1]
    return grad

Key insight: symmetric sequences have peak = sum(p^2) >= 1/n, giving score >= 2.

def fast_convolve(a, b):
    """Compute linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    m = 1
    while m < n_out:
        m *= 2
    A = fft(a, m)
    B = fft(b, m)
    result = np.real(ifft(A * B, m))[:n_out]
    return result


def project_waterfill(q):
    """Project q onto {p >= 0, sum(p) = 1} via water-filling."""
    n = len(q)
    if n == 0:
        return q.copy()
    q_sorted = np.sort(q)[::-1]
    if q_sorted[0] <= 0:
        return np.ones(n) / n
    cumsum = np.cumsum(q_sorted)
    best_m = n
    for m in range(1, n + 1):
        tau = (cumsum[m - 1] - 1.0) / m
        if tau <= 0:
            continue
        if m < n and tau < q_sorted[m]:
            continue
        best_m = m
        break
    tau = max(0.0, (cumsum[best_m - 1] - 1.0) / best_m)
    p = np.maximum(0.0, q - tau)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return p / s

    """Compute 2*n*max(conv(p,p)) where sum(p)=1."""
    n = len(p)

    """Compute gradient of (p*p)_{k_star} w.r.t. p."""

    if lo > hi:

    p_slice = p[k_star - hi: k_star - lo + 1]
    grad[lo:hi + 1] = 2.0 * p_slice[::-1]

    return grad
