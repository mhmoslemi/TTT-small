"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using LP-based iterative direction finding
and projected subgradient descent. Exploits scale invariance (sum=1 normalization).
Uses multiple starting points and sequence lengths for robustness.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_mask = u * idx > (cssv - 1)
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.where(rho_mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_convolution(a):
    """Compute full self-convolution of a."""
    return np.convolve(a, a)


def find_active_set(c, top_k=None, frac=None):
    """Find indices of near-maximal convolution values."""
    c_max = c.max()
    if top_k is not None:
        threshold = np.sort(c)[::-1][min(top_k - 1, len(c) - 1)]
    elif frac is not None:
        threshold = c_max * (1.0 - frac)
    else:
        threshold = c_max * 0.99
    return np.where(c >= threshold)[0]


def compute_subgradient(a, active_set):
    """Compute sum of dc[k]/da over k in active_set, efficiently."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_set:
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        count = m_hi - m_lo + 1
        if count <= 0:
            continue
        # First term: a[k-m] for m in [m_lo, m_hi]
        j_lo = k - m_hi
        j_hi = k - m_lo
        grad[m_lo:m_hi + 1] += a[j_lo:j_hi + 1]
        # Second term: a[m] for m in [m_lo, m_hi]
        grad[m_lo:m_hi + 1] += a[m_lo:m_hi + 1]
    return grad


def solve_lp_direction(a, M, max_constraints=300, upper=10.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for select k, g >= 0."""
    n = len(a)
    c = np.convolve(a, a)
    # Select the most constraints (top values of c as proxy for active)
    n_con = min(max_constraints, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:n_con]
    
    A = lil_matrix((n_con, n))
    for jj in range(n_con):
        k = top_idx[jj]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        A[jj, m_lo:m_hi + 1] = a[k - m_hi:k - m_lo + 1][::-1]
    A = A.tocsr()
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_con, M)
    bounds = [(0.0, upper)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(a, deadline, max_iters=500):
    """LP-based iterative optimization."""
    n = len(a)
    a = a / max(a.sum(), 1e-15)
    best_local = a.copy()
    best_score_local = np.inf
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        M = c.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        if score < best_score_local:
            best_score_local = score
            best_local = a.copy()
        
        g = solve_lp_direction(a, M, max_constraints=200)
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / max(g.sum(), 1e-15)
        
        # Line search over blending parameters
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / max(a_trial.sum(), 1e-15)
            c_trial = np.convolve(a_trial, a_trial)
            score_trial = 2.0 * n * c_trial.max() / (a_trial.sum() ** 2)
            if score_trial < score:
                a = a_trial
                improved = True
                break
        
        if not improved and it > 50:
            # Stuck, try smaller steps
            for t in [0.02, 0.01, 0.005]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / max(a_trial.sum(), 1e-15)
                c_trial = np.convolve(a_trial, a_trial)
                score_trial = 2.0 * n * c_trial.max() / (a_trial.sum() ** 2)
                if score_trial < score:
                    a = a_trial
                    break
        
    return best_local, a


def subgradient_optimize(a, deadline, max_iters=50000):
    """Projected subgradient descent on max(conv(a,a)) with sum=1 constraint."""
    n = len(a)
    a = a / max(a.sum(), 1e-15)
    best_local = a.copy()
    best_score_local = 2.0 * n * np.max(np.convolve(a, a))
    
    lr = 1.0 / n
    top_k = 20
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        c_max = c.max()
        score = 2.0 * n * c_max
        if score < best_score_local:
            best_score_local = score
            best_local = a.copy()
        
        # Find active set
        active = find_active_set(c, top_k=top_k)
        
        # Compute subgradient
        grad = compute_subgradient(a, active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        grad = grad / grad_norm
        
        # Adaptive step size
        step = lr / (1.0 + 0.01 * it)
        
        a_new = a - step * grad
        a_new = simplex_project(a_new)
        
        # Check if it improved
        c_new = np.convolve(a_new, a_new)
        if c_new.max() < c_max:
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - 0.1 * step * grad
            a_new2 = simplex_project(a_new2)
            c_new2 = np.convolve(a_new2, a_new2)
            if c_new2.max() < c_max:
                a = a_new2
            else:
                lr *= 0.9
    
    return best_local


def random_perturb_optimize(a, deadline, rng, max_iters=100000):
    """Local search with random perturbations."""
    n = len(a)
    a = a / max(a.sum(), 1e-15)
    best_local = a.copy()
    best_score_local = 2.0 * n * np.max(np.convolve(a, a))
    c_current = np.convolve(a, a)
    c_max_current = c_current.max()
    
    block_size = max(1, n // 100)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Random block perturbation
        start = rng.integers(0, n - block_size + 1)
        end = min(n, start + block_size)
        factor = np.exp(rng.normal(0.0, 0.1))
        a_trial = a.copy()
        a_trial[start:end] *= factor
        a_trial = np.maximum(a_trial, 0.0)
        if a_trial.sum() < 1e-15:
            continue
        a_trial = a_trial / a_trial.sum()
        
        c_trial = np.convolve(a_trial, a_trial)
        score_trials = c_trial.max()
        
        if score_trials < c_max_current:
            a = a_trial
            c_current = c_trial
            c_max_current = score_trials
            best_local = a.copy()
            best_score_local = 2.0 * n * c_max_current
            
            # Occasional symmetry
            if rng.random() < 0.1:
                a_sym = 0.5 * (a + a[::-1])
                a_sym = a_sym / a_sym.sum()
                c_sym = np.convolve(a_sym, a_sym)
                if c_sym.max() < c_max_current:
                    a = a_sym
                    c_current = c_sym
                    c_max_current = c_sym.max()
                    best_local = a.copy()
                    best_score_local = 2.0 * n * c_max_current
    
    return best_local


def make_structured_init(n, rng):
    """Create a structured initial sequence with some zeros."""
    a = np.ones(n)
    # Add periodic zeros
    gap = max(2, n // 16)
    for i in range(gap - 1, n, gap):
        a[i] = 0.0
    # Add some randomness
    a *= np.exp(rng.normal(0.0, 0.3, n))
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization: LP direction + subgradient + random perturbation.
    Tries multiple sequence lengths and initializations.
    """
    deadline = time.monotonic() + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    # Gather candidate initial sequences
    candidates = []
    
    # Parent sequence
    parent = None
    if 'height_sequence_1' in kwargs and kwargs['height_sequence_1']:
        parent = np.array(kwargs['height_sequence_1'], dtype=np.float64)
    else:
        try:
            parent = np.array(height_sequence_1, dtype=np.float64)
        except Exception:
            parent = None
    
    if parent is not None and len(parent) > 0:
        parent = np.clip(parent, 0, 1000)
        if parent.sum() > 0.01:
            candidates.append(('parent', parent.copy()))
    
    # Try various lengths
    for n_try in [128, 256, 512, 1024, 2048]:
        candidates.append((f'uniform_{n_try}', np.ones(n_try)))
        candidates.append((f'struct_{n_try}', make_structured_init(n_try, rng)))
    
    # Also try parent's length if reasonable
    if parent is not None:
        n_p = len(parent)
        if n_p != 128 and n_p != 256 and n_p != 512 and n_p != 1024 and n_p != 2048:
            if n_p <= 5000:
                candidates.append((f'struct_{n_p}', make_structured_init(n_p, rng)))
    
    best_overall = None
    best_overall_score = np.inf
    
    for name, seq in candidates:
        if time.monotonic() > deadline:
            break
        
        n = len(seq)
        seq = np.clip(seq, 0, 1000).astype(np.float64)
        if seq.sum() < 0.01:
            seq = np.ones(n)
        seq = seq / seq.sum()
        
        # Quick score
        c_init = np.convolve(seq, seq)
        init_score = 2.0 * n * c_init.max()
        
        # Phase 1: LP optimization (for smaller n)
        if n <= 2500:
            remaining = deadline - time.monotonic()
            if remaining > 10.0:
                lp_end = min(deadline, time.monotonic() + remaining * 0.4)
                seq_lp, _ = lp_optimize(seq, lp_end, max_iters=200)
                c_lp = np.convolve(seq_lp, seq_lp)
                lp_score = 2.0 * n * c_lp.max()
                if lp_score < init_score:
                    seq = seq_lp
                    init_score = lp_score
        
        # Phase 2: Subgradient optimization
        remaining = deadline - time.monotonic()
        if remaining > 5.0:
            sg_end = min(deadline, time.monotonic() + remaining * 0.4)
            seq_sg = subgradient_optimize(seq, sg_end, max_iters=30000)
            c_sg = np.convolve(seq_sg, seq_sg)
            sg_score = 2.0 * n * c_sg.max()
            if sg_score < init_score:
                seq = seq_sg
                init_score = sg_score
        
        # Phase 3: Random perturbation fine-tuning
        remaining = deadline - time.monotonic()
        if remaining > 3.0:
            rp_end = min(deadline, time.monotonic() + remaining * 0.5)
            seq_rp = random_perturb_optimize(seq, rp_end, rng, max_iters=50000)
            c_rp = np.convolve(seq_rp, seq_rp)
            rp_score = 2.0 * n * c_rp.max()
            if rp_score < init_score:
                seq = seq_rp
                init_score = rp_score
        
        # Evaluate with the actual evaluator
        seq_final = np.clip(seq, 0, 1000)
        if seq_final.sum() < 0.01:
            continue
        score = evaluate_sequence([float(x) for x in seq_final])
        
        if np.isfinite(score) and score < best_overall_score:
            best_overall_score = score
            best_overall = seq_final.copy()
            print(f"[{name}] n={n}, score={score:.6f}")
    
    # If no valid result, fallback to uniform
    if best_overall is None:
        best_overall = np.ones(256)
        best_overall_score = evaluate_sequence(best_overall.tolist())
        print(f"[fallback] uniform, score={best_overall_score:.6f}")
    
    # Final: try to improve best_overall with more subgradient iterations
    if time.monotonic() < deadline - 10.0:
        n_best = len(best_overall)
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            sg_end = min(deadline - 5.0, time.monotonic() + remaining * 0.6)
            seq_final_sg = subgradient_optimize(best_overall, sg_end, max_iters=100000)
            score_final = evaluate_sequence([float(max(0, min(1000, x))) for x in seq_final_sg])
            if np.isfinite(score_final) and score_final < best_overall_score:
                best_overall = seq_final_sg
                best_overall_score = score_final
                print(f"[final_sg] n={n_best}, score={score_final:.6f}")
            
            if time.monotonic() < deadline - 10.0:
                rp_end = min(deadline - 3.0, time.monotonic() + 100.0)
                seq_final_rp = random_perturb_optimize(best_overall, rp_end, rng, max_iters=50000)
                score_rpf = evaluate_sequence([float(max(0, min(1000, x))) for x in seq_final_rp])
                if np.isfinite(score_rpf) and score_rpf < best_overall_score:
                    best_overall = seq_final_rp
                    best_overall_score = score_rpf
                    print(f"[final_rp] n={n_best}, score={score_rpf:.6f}")
    
    print(f"BEST: n={len(best_overall)}, score={best_overall_score:.6f}")
    return [float(x) for x in best_overall]
