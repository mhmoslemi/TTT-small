        else:
            cand = a_cur * np.exp(rng.normal(0, float(rng.uniform(0.01, 0.04)), size=n))
            np.clip(cand, 0, 1000, out=cand)

        sc = _fast_score(cand)
        if np.isfinite(sc) and sc < best_local - 1e-12:
            best_local = sc
            best_seq_final = cand.copy()
            a_cur = cand
            stall = 0
            if int(best_local * 1e6) % 200000 == 0:
                print(f"  local {best_local:.6f} t={time.monotonic()-t_start:.0f}s")
        else:
            stall += 1

        if stall > 300:
            stall = 0
            a_cur = best_seq_final * np.exp(rng.normal(0, 0.03, size=n))
            np.clip(a_cur, 0, 1000, out=a_cur)
            amps_jit = best_amps + rng.normal(0, 0.015, size=M_modes)
            a_alt = _make_sequence(n, amps_jit)
            if _fast_score(a_alt) < _fast_score(a_cur):
                a_cur = a_alt
            b_cur = np.convolve(a_cur, a_cur)

    final_seq = best_seq_final
    final_ev = float(evaluate_sequence(final_seq.tolist()))
    if final_ev > best_local + 0.01:
        alt = [float(x) for x in best_seq_final]
        final_ev = float(evaluate_sequence(alt))
    print(f"FINAL: {final_ev:.6f}")
    return [float(x) for x in final_seq]
