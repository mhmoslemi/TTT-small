import numpy as np
import scipy.optimize as sp
import time

def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix: row k has a[i] at position (k-i) for valid i
    # (a * g)[k] = sum_i a[i] * g[k-i] <= M
    # Let c = cumsum of a, then the constraint matrix entry at (k, j) is a[k-j] if 0 <= k-j <= n-1
    
    # For efficiency, use sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    data = []
    
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            # Entry at (k, j) is a[k-j]
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(a[idx])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    res = sp.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return a.copy()  # fallback
