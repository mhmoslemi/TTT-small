def lp_update(a, rows, cols, data_indices, n, rng):

    c = np.convolve(a, a, 'full')
    M = c.max()

    data = a[data_indices]
    
    # Build sparse matrix
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # LP: maximize sum(g) subject to A@g <= M, g >= 0
    c_obj = -np.ones(n)  # Negate for minimization
    b = M * np.ones(2*n-1)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, 
                  method='highs', options={'time_limit': 120})
    
    if res.status == 0:
        g = res.x
        S_g = g.sum()
        if S_g > a.sum() * 1.001:  # Must be strictly better

            # Blend

            a_new = (1-t) * a + t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new * (n / a_new.sum())
