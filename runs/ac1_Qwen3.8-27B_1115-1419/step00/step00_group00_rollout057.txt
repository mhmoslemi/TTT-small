```python
"""
LP-based optimization for minimizing peak autocorrelation of non-negative sequences.
Uses linear programming to find optimal update directions, combined with
multiple initializations and gradient-based refinement.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_conv_mat(f, n):
    """Build sparse convolution matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            rows.extend(np.full(len(js), k, dtype=np.int32))
            cols.extend(js.astype(np.int32))
            vals.extend(f[k - hi:k - lo + 1][::-1])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_step(f, n, time_limit=5.0):
    """Perform one LP-based update step. Returns new f or None if infeasible."""
    autocorr = np.convolve(f, f)
    M = autocorr.max()
    if M < 1e-15:
        return None
    
    A = build_conv_mat(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res.status != 0:
        return None
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None
    
    return g / g_sum, M

def gradient_step(f, n, eta=0.5):
    """One projected subgradient step to reduce peak autocorrelation."""
    autocorr = np.convolve(f, f, mode='full')
    M = autocorr.max()
    peak_set = np.where(np.isclose(autocorr, M, atol=M * 1e-8))[0]
    
    # Subgradient
    grad = np.zeros(n)
    for k in peak_set:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * f[k - j]
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return f
    
    # Step
    f_new = f - eta * grad / grad_norm
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    f_new /= s
    return f_new

def random_perturb(f, n, rng, scale=0.1):
    """Add random perturbation and renormalize."""
    f_new = f * np.exp(rng.normal(0, scale, n))
    # Occasionally zero out some elements
    if rng.random() < 0.3:
        mask = rng.random(n) < 0.1
        f_new[mask] = 0
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    return f_new / s

def run_lp_optimization(f, deadline, rng, verbose=False):
    """Run LP-based optimization loop."""
    n = len(f)
    f = f / f.sum()
    best_local = f.copy()
    best_local_score = float('inf')
    
    t = 0.3  # blend parameter
    no_improve_count = 0
    
    while time.monotonic() < deadline - 60:
        result = lp_step(f, n, time_limit=5.0)
        
        if result is None:
            # LP failed, try gradient step
            f = gradient_step(f, n, eta=0.3)
            no_improve_count += 1
            if no_improve_count > 10:
                # Perturb and restart
                f = random_perturb(f, n, rng, scale=0.3)
                no_improve_count = 0
            continue
        
        g, M = result
        g_sum = g.sum()
        
        # Check if g is proportional to f (no progress)
        if g_sum > 0:
            cos_sim = np.dot(f, g) / (np.linalg.norm(f) * np.linalg.norm(g) + 1e-15)
            if cos_sim > 0.9999:
                # No progress from LP
                f = random_perturb(f, n, rng, scale=0.2)
                no_improve_count += 1
                if no_improve_count > 5:
                    t = max(0.05, t * 0.5)
                    no_improve_count = 0
                continue
        
        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s
        
        # Evaluate
        score = evaluate_sequence(f_new.tolist())
        if np.isfinite(score) and score < best_local_score:
            best_local_score = score
            best_local = f_new.copy()
            no_improve_count = 0
            if verbose:
                print(f"  LP iter: score={score:.6f}, M={M:.6f}")
        else:
            no_improve_count += 1
            if no_improve_count % 10 == 0:
                t = max(0.05, t * 0.7)
        
        f = f_new
    
    return best_local, best_local_score

def make_initialization(n, init_type, rng):
    """Create an initial sequence of length n."""
    if init_type == 'uniform':
        f = np.ones(n)
    elif init_type == 'ramp':
        f = np.arange(1, n + 1, dtype=np.float64)
    elif init_type == 'ramp_down':
        f = np.arange(n, 0, -1, dtype=np.float64)
    elif init_type == 'bimodal':
        f = np.ones(n)
        f[:n//4] = 2.0
        f[3*n//4:] = 2.0
    elif init_type == 'two_level':
        f = np.where(np.arange(n) % 2 == 0, 1.0, 2.0)
    elif init_type == 'random':
        f = rng.uniform(0, 1, n)
    elif init_type == 'random_heavy':
        f = rng.exponential(1, n)
    elif init_type == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = np.cos(x/2)**2
        f = np.maximum(f, 0)
    else:
        f = np.ones(n)
    
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n)
        s = n
    return f / s
```
