"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using local search on sequences.
Strategy: explore various lengths/densities with binary sequences, then deep
local search with peak-targeted bit flips, then non-binary polishing.

import numpy as np
import time


def compute_score_from_conv(c, n, k):
    """Compute score from convolution max, length, and sum."""
    if k <= 0:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (k * k)


def gen_random_binary(n, p, rng):
    """Generate random binary sequence with density p."""
    return (rng.random(n) < p).astype(np.float64)


def gen_quadratic_residue(n, rng):
    """Generate sequence based on quadratic residues for prime near n."""
    # Find a prime near n
    def is_prime(x):
        if x < 2:
            return False
        if x < 4:
            return True
        if x % 2 == 0 or x % 3 == 0:
            return False
        i = 5
        while i * i <= x:
            if x % i == 0 or x % (i + 2) == 0:
                return False
            i += 6
        return True

    p = n
    while not is_prime(p):
        p -= 1
    if p < 7:
        return gen_random_binary(n, 0.5, rng)

    # Quadratic residues mod p
    residues = set()
    for x in range(p):
        residues.add((x * x) % p)
    seq = np.zeros(n, dtype=np.float64)
    for r in residues:
        if r < n:
            seq[r] = 1.0
    return seq


def gen_block_pattern(n, rng):
    """Generate a block pattern: alternating blocks of 1s and 0s."""
    n_blocks = int(n // rng.integers(20, 80))
    if n_blocks < 1:
        n_blocks = 1
    block_size = n // n_blocks
    seq = np.zeros(n, dtype=np.float64)
    for i in range(n_blocks):
        start = i * block_size
        end = min(n, start + block_size)
        fill_len = max(1, block_size // int(rng.integers(2, 4)))
        seq[start:start + fill_len] = 1.0
    return seq


def local_search(a, max_time, rng, allow_growth=True):
    """
    Deep local search using peak-targeted flips.
    Maintains convolution incrementally for efficiency.
    Returns (best_score, best_sequence).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    c = np.convolve(a, a).copy()
    k = float(np.sum(a))
    current_score = compute_score_from_conv(c, n, k) if k > 0 else np.inf

    best_score = current_score
    best_a = a.copy()

    start = time.monotonic()
    no_improve = 0
    max_no_improve = 2000
    iteration = 0

    while time.monotonic() - start < max_time:
        # Random restart if stuck
        if no_improve > max_no_improve:
            a = best_a.copy()
            c = np.convolve(a, a).copy()
            k = float(np.sum(a))
            current_score = compute_score_from_conv(c, n, k) if k > 0 else np.inf
            # Perturb: flip some random bits
            n_flips = max(5, n // 40)
            indices = rng.choice(n, size=min(n_flips, n), replace=False)
            for idx in indices:
                a[int(idx)] = 1.0 - a[int(idx)]
            c = np.convolve(a, a)
            k = float(np.sum(a))
            current_score = compute_score_from_conv(c, n, k) if k > 0 else np.inf
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)
            iteration = 0
            continue

        # Find peak position(s)
        s_star = int(np.argmax(c))
        peak_val = c[s_star]

        # Generate candidate flips
        candidates_remove = []
        candidates_add = []

        # Positions that contribute to the peak
        lo = max(0, s_star - (n - 1))
        hi = min(n - 1, s_star)

        # Also check second and third peaks for multi-peak reduction
        sorted_decreasing = np.argsort(c)[::-1]
        top_peaks = sorted_decreasing[:min(5, 2 * n - 1)]

        for s_peak in top_peaks:
            s_peak = int(s_peak)
            lo_p = max(0, s_peak - (n - 1))
            hi_p = min(n - 1, s_peak)
            for p in range(lo_p, hi_p + 1):
                if a[p] > 0 and a[s_peak - p] > 0:
                    val = 1.0 if a[p] <= 1.5 else 0.5
                    candidates_remove.append((p, val))

        # Deduplicate remove candidates
        candidates_remove = list(set(candidates_remove))

        # Random removals
        ones_idx = np.where(a > 0.01)[0]

            n_rand = min(15, len(ones_idx))
            for idx in rng.choice(ones_idx, size=n_rand, replace=False):
                val = 1.0 if a[int(idx)] <= 1.5 else 0.5
                candidates_remove.append((int(idx), val))

        # Random additions (if allowed)
        if allow_growth:
            zeros_idx = np.where(a <= 0.01)[0]
            if len(zeros_idx) > 5:
                n_rand = min(15, len(zeros_idx))
                for idx in rng.choice(zeros_idx, size=n_rand, replace=False):
                    candidates_add.append(int(idx))

        # Evaluate candidates
        best_move = None
        best_move_score = current_score

        for p, val in candidates_remove:

def polish_nonbinary(a, max_time, rng):
    """Try to improve score by adjusting values slightly (non-binary polish)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    k = float(np.sum(a))
    c = np.convolve(a, a)
    current_score = compute_score_from_conv(c, n, k)
    best_score = current_score
    best_a = a.copy()

    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < max_time:
        if no_improve > 500:
            break

        # Pick a random position to adjust
        p = int(rng.integers(0, n))
        if a[p] <= 0.01:
            continue

        old_val = a[p]
        # Try several multiplier values
        found_improvement = False
        for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 0.5, 1.5]:
            new_val = old_val * mult
            if new_val < 0 or new_val > 1000:
                continue
            a[p] = new_val
            new_k = k + (new_val - old_val)
            if new_k <= 0:
                a[p] = old_val
                continue
            c = np.convolve(a, a)
            new_score = compute_score_from_conv(c, n, new_k)
            if new_score < best_score - 1e-10:
                best_score = new_score
                best_a = a.copy()
                k = new_k
                c = np.convolve(a, a)
                current_score = new_score
                no_improve = 0
                found_improvement = True
                break
            a[p] = old_val

        if not found_improvement:
            no_improve += 1

        if time.monotonic() - start > max_time * 0.8:
            break

    return best_score, best_a

    Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using multi-phase search.
    Phase 1: Explore various (length, density) with short local searches.
    Phase 2: Deep local search on best configuration.
    Phase 3: Non-binary polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 10.0)

    best_score = np.inf
    best_seq = None

    # Check if height_sequence_1 is available
    parent_seq = None
    try:
        hs1 = height_sequence_1  # noqa: F821
        if isinstance(hs1, (list, np.ndarray)) and len(hs1) > 0:
            parent_seq = np.clip(np.asarray(hs1, dtype=np.float64), 0.0, 1000.0)
            if np.isfinite(parent_seq).all() and np.sum(parent_seq) > 0.01:
                s = float(evaluate_sequence(parent_seq.tolist()))
                if np.isfinite(s):
                    best_score = s
                    best_seq = parent_seq.copy()
                    print(f"Parent sequence: score={s:.6f}, n={len(parent_seq)}")
    except Exception:
        pass

    # Phase 1: Exploration
    print("=== Phase 1: Exploration ===")
    configs = []
    # Various lengths and densities
    for n in [64, 100, 150, 200, 300, 500, 800, 1200, 2000, 3000, 5000]:
        for p in [0.25, 0.35, 0.45, 0.55, 0.65, 0.75]:
            configs.append((n, p, 'random'))
        configs.append((n, 0, 'quadratic'))
        configs.append((n, 0, 'block'))

    # Interleave with parent if available
    if parent_seq is not None:
        configs.insert(0, (len(parent_seq), 0, 'parent'))

    time_per_try = 15  # seconds per configuration
    max_phase1_time = 400

    for n, p, gen_type in configs:
        if time.monotonic() > deadline - max_phase1_time:
            break
        if time.monotonic() - start_time > 10 and (time.monotonic() - start_time) % (time_per_try * 10) < 0.1:
            pass  # timing check

        elapsed = time.monotonic() - start_time
        remaining_exploration = deadline - max_phase1_time - elapsed
        if remaining_exploration < time_per_try:
            break

        if gen_type == 'parent' and parent_seq is not None:
            a = parent_seq.copy()
        elif gen_type == 'quadratic':
            a = gen_quadratic_residue(n, rng)
        elif gen_type == 'block':
            a = gen_block_pattern(n, rng)
        else:
            a = gen_random_binary(n, p, rng)

        if np.sum(a) < 5:
            continue

        # Quick score check before investing in local search
        s0 = float(evaluate_sequence(a.tolist()))
        if not np.isfinite(s0):
            continue

        # Short local search
        t_start = time.monotonic()
        max_t = min(time_per_try, deadline - max_phase1_time - t_start)
        if max_t < 2:
            break
        score, seq = local_search(a, max_t, rng, allow_growth=True)

        # Verify with actual evaluate_sequence
        s_final = float(evaluate_sequence(seq.tolist()))
        if not np.isfinite(s_final):
            s_final = score

        if s_final < best_score:
            best_score = s_final
            best_seq = seq.copy()
            gen_str = gen_type + (f" p={p}" if gen_type == 'random' else "")
            print(f"  n={n}, gen={gen_str}: score={s_final:.6f} (k={np.sum(seq):.0f})")

    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep search
    print("=== Phase 2: Deep Search ===")
    if best_seq is None:
        best_seq = np.ones(500, dtype=np.float64)
        best_score = 2.0

    n_best = len(best_seq)
    # Deep search on best
    remaining2 = deadline - time.monotonic() - 120
    if remaining2 > 30:
        score2, seq2 = local_search(best_seq, remaining2 * 0.85, rng, allow_growth=True)
        s2 = float(evaluate_sequence(seq2.tolist()))
        if not np.isfinite(s2):
            s2 = score2
        if s2 < best_score:
            best_score = s2
            best_seq = seq2.copy()
            print(f"  Deep search improved: {best_score:.6f}")

        # Also try a different length near the best
        remaining2b = deadline - time.monotonic() - 60
        if remaining2b > 60:
            for n_try in [n_best + 100, n_best - 100, n_best + 200]:
                if n_try < 20 or time.monotonic() > deadline - 60:
                    continue
                a_try = gen_random_binary(n_try, max(0.2, np.sum(best_seq) / n_best), rng)
                s_try = float(evaluate_sequence(a_try.tolist()))
                if not np.isfinite(s_try) or s_try > best_score * 1.5:
                    continue
                max_t3 = min(remaining2b * 0.3, deadline - time.monotonic() - 30)
                if max_t3 < 10:
                    continue
                score3, seq3 = local_search(a_try, max_t3, rng, allow_growth=True)
                s3 = float(evaluate_sequence(seq3.tolist()))
                if not np.isfinite(s3):
                    s3 = score3
                if s3 < best_score:
                    best_score = s3
                    best_seq = seq3.copy()
                    print(f"  Alt length n={n_try}: score={s3:.6f}")
                    remaining2b = deadline - time.monotonic() - 30

    # Phase 3: Non-binary polish
    print("=== Phase 3: Polishing ===")
    remaining3 = deadline - time.monotonic() - 20
    if remaining3 > 30 and best_seq is not None:
        score3, seq3 = polish_nonbinary(best_seq, remaining3, rng)
        s3 = float(evaluate_sequence(seq3.tolist()))
        if not np.isfinite(s3):
            s3 = score3
        if s3 < best_score:
            best_score = s3
            best_seq = seq3.copy()
            print(f"  Polish improved: {best_score:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)

    # Ensure valid output
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    result_score = float(evaluate_sequence([float(x) for x in best_seq]))
    print(f"\nFinal score: {result_score:.6f}, length: {len(best_seq)}")

    return [float(x) for x in best_seq]
