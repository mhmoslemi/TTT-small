import numpy as np
from scipy.signal import fftconvolve
import time

"""
Minimizes 2*n*max(conv(f,f))/sum(f)^2 using smooth gradient descent.
The smooth gradient is a softmax-weighted average of individual subgradients,
giving a well-defined descent direction that automatically handles the
"chasing the max" problem. Uses multiplicative updates for natural
non-negativity, with multiple initializations and adaptive step sizes.
"""


def _compute_smooth_gradient(f, n, beta):
    """Compute the gradient of log-sum-exp(beta * C[k]) / beta as smooth max approximation."""
    C = fftconvolve(f, f)
    C_shifted = C - C.max()
    weights = np.exp(beta * C_shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        k_star = C.argmax()
        weights = np.zeros_like(C)
        weights[k_star] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_k weights[k] * f[k-i] (cross-correlation)
    L = 1
    while L < 3 * n:
        L *= 2

    w_pad = np.zeros(L)
    w_pad[:len(C)] = weights
    f_pad = np.zeros(L)
    f_pad[:n] = f

    W = np.fft.rfft(w_pad)
    F = np.fft.rfft(f_pad)
    cross = np.fft.irfft(W * np.conj(F), n=L)[:n]

    grad = 2.0 * cross
    return grad, C.max()
