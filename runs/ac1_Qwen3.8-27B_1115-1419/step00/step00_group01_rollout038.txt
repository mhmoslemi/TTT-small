def trust_region_step(f, active_set, delta):
    """Solve the trust-region subproblem."""
    n = len(f)
    # Linearization: (f_new*f_new)[k] ≈ c_k + 2*sum_j (f_new[j]-f[j])*f[k-j]
    # We want to minimize max_k (c_k + 2*sum_j (f_new[j]-f[j])*f[k-j])
    # subject to f_new >= 0, sum f_new = 1, ||f_new - f||_inf <= delta
    
    # Using scipy.optimize.linprog:
    # minimize t
    # subject to: c_k + 2*sum_j (f_new[j]-f[j])*f[k-j] <= t for k in active_set
    #             f_new[j] >= 0
    #             sum f_new = 1
    #             -delta <= f_new[j] - f[j] <= delta
    
    # Variables: [f_new[0], ..., f_new[n-1], t]
    # Total: n+1 variables
    
    from scipy.optimize import linprog
    
    c = np.convolve(f, f)
    
    # Objective: minimize t (index n)
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: c_k + 2*sum_j (f_new[j]-f[j])*f[k-j] - t <= 0
    # => 2*sum_j f[k-j]*f_new[j] - t <= -c_k - 2*sum_j f[k-j]*f[j] = -c_k - c_k = -2c_k
    # Wait: sum_j f[k-j]*f[j] = (f*f)[k] = c_k
    # So: 2*sum_j f[k-j]*f_new[j] - t <= -c_k
    
    A_rows = []
    b_rows = []
    
    for k in active_set:
        row = np.zeros(n+1)
        # 2*sum_j f[k-j]*f_new[j] - t <= -c_k
        for j in range(n):
            if 0 <= k-j < n:
                row[j] = 2.0 * f[k-j]
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-c_k)
    
    # Trust region: -delta <= f_new[j] - f[j] <= delta
    # f_new[j] - f[j] <= delta => f_new[j] <= f[j] + delta
    # -(f_new[j] - f[j]) <= delta => -f_new[j] <= -f[j] + delta => f_new[j] >= f[j] - delta
    
    for j in range(n):
        row1 = np.zeros(n+1)
        row1[j] = 1.0
        A_rows.append(row1)
        b_rows.append(f[j] + delta)
        
        row2 = np.zeros(n+1)
        row2[j] = -1.0
        A_rows.append(row2)
        b_rows.append(-f[j] + delta)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality constraint: sum f_new = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: f_new[j] >= 0, t unbounded below (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    else:
        return f

from scipy.optimize import linprog

def compute_convolution(f, fft_size):
    """Compute linear convolution of f with f using FFT."""
    F = np.fft.fft(f, n=fft_size)
    return np.real(np.fft.ifft(F * F))[:2*len(f)-1]

def trust_region_lp(f, active_set, delta):
    """Solve the trust-region LP subproblem."""
    n = len(f)
    c = np.convolve(f, f)
    
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    A_rows = []
    b_rows = []
    
    for k in active_set:
        row = np.zeros(n+1)
        for j in range(n):
            if 0 <= k-j < n:
                row[j] = 2.0 * f[k-j]
        row[n] = -1.0

        b_rows.append(-c[k])

        row[j] = 1.0

        row[j] = -1.0

    b_ub = np.array(b_rows)

    b_eq = np.array([1.0])

    return None, None
