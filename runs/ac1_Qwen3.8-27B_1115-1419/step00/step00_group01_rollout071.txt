def clean_sequence(seq):
    try:
       arr=np.asarray(seq, dtype=float).reshape(-1)
    except Exception: return None
    if arr.size==0: return None
    if not np.all(np.isfinite(arr)): return None
    arr=np.clip(arr,0.0,1000.0)
    if arr.sum() <= 1e-12: return None
    return arr
```

`linear_conv_full`:
```
def linear_conv_full(a):
    n=int(a.size)
    need=2*n-1
    if n <= 128:
       c=np.convolve(a,a,'full')
       c=np.asarray(c,dtype=float)[:need]
       if c.size and c.min() < 0:
          np.maximum(c,0.0,out=c)
       return c
    size=1
    while size < need: size <<= 1
    fa=np.fft.rfft(a, size)
    c=np.fft.irfft(fa*fa, size)[:need]
    if c.size and c.min() < 0:
       np.maximum(c,0.0,out=c)
    return c
```
For n=128, direct O(16k) ok.

`internal_score`:
```
def internal_score(a):
    a=np.asarray(a,dtype=float)
    s=a.sum()
    if s <=0: return np.inf
    c=linear_conv_full(a)
    m=float(c.max())
    if not np.isfinite(m): return np.inf
    return float(2*a.size*m/(s*s))
```

def random_log_sequence(n, rng, kind=None):
    x=np.linspace(0.0,1.0,n)
    kind = kind if kind is not None else int(rng.integers(0, 7))
    if kind == 0:
       lam=float(rng.uniform(0.1, 6.0))
       sign=1.0 if rng.random()<0.5 else -1.0
       V=sign*lam*x
    elif kind == 1:
       c1=float(rng.uniform(-3,3)); c2=float(rng.uniform(-3,3)); c3=float(rng.uniform(-3,3))
       V=c1*x + c2*x*x + c3*x*x*x
    elif kind == 2:
       V=np.zeros(n)
       for m in range(1, 8):
           amp=float(rng.uniform(0.05,1.0))
           ph=float(rng.uniform(0,2*np.pi))
           V += amp*np.cos(2*np.pi*m*x + ph)
       V += float(rng.uniform(-2,2))*x
    elif kind == 3:
       # two exp bumps
       z=np.ones(n)
       z += float(rng.uniform(0.2,3.0))*np.exp(-((x-float(rng.uniform(0,0.5)))**2)/(2*float(rng.uniform(0.02,0.4))**2))
       z += float(rng.uniform(0.2,3.0))*np.exp(-((x-float(rng.uniform(0.5,1.0)))**2)/(2*float(rng.uniform(0.02,0.4))**2))
       return normalize_positive(z)
    elif kind == 4:
       # smooth random positive
       z=rng.random(n)+0.05
       k=max(1,n//16)
       kernel=np.ones(k); kernel/=k
       z=np.convolve(z,kernel,mode='same')
       return normalize_positive(z+1e-6)
    else:
       # geometric with curvature
       lam=float(rng.uniform(-4,4))
       curv=float(rng.uniform(-3,3))
       V=lam*x + curv*x*x
    V = np.asarray(V,dtype=float)
    V = V - float(np.max(V))
    V = np.clip(V, -700.0, 0.0)
    a=np.exp(V)
    out=normalize_positive(a)
    if out is None: out=np.full(n,1.0)/n
    return out
```

def build_shape(theta, x):
    t=np.zeros(10,dtype=float)
    th=np.asarray(theta,dtype=float)
    if th.ndim !=1: th=th.ravel()
    m=min(10, th.size)
    if m>0: t[:m]=th[:m]
    xs=x
    V=(t[0]*xs + t[1]*xs*xs + t[2]*xs*xs*xs + t[3]*(xs-0.5)*(xs-0.5) + t[4]*xs*(1.0-xs)
       + t[5]*np.sin(np.pi*xs) + t[6]*np.cos(2.0*np.pi*xs) + t[7]*np.sin(2.0*np.pi*xs)

def upsample_sequence(coarse, n):
    coarse=clean_sequence(coarse)
    if coarse is None: return None
    m=coarse.size
    if m == n: return normalize_positive(coarse)
    logc=np.log(coarse + 1e-300)
    x0=np.linspace(0.0,1.0,m)
    x1=np.linspace(0.0,1.0,n)
    logf=np.interp(x1,x0,logc)

`coordinate_phi` top-level:

def _coordinate_phi(x, B, L, Q, S_base):
    vals = B + L * x + Q * x * x
    # small negatives
    if vals.min() < 0:
       vals = np.maximum(vals, 0.0)
    den = (S_base + x)
    if den <= 1e-300:
       return np.inf
    return float(np.max(vals) / (den * den))
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a)**2).

The code uses several complementary ideas:
- fast FFT evaluation of the same objective;
- coarse direct coordinate refinement on small lengths;
- random parametric and log-smooth initial sequences;
- large-length refinement by active-lag projected gradient steps;
- optional LP polishing in the spirit of the prior state-of-the-art;
- final verification with the externally supplied evaluate_sequence.
"""

import time
import numpy as np

try:
    from scipy import optimize as _scipy_optimize
    _HAS_SCIPY_OPT = True
except Exception:
    _scipy_optimize = None
    _HAS_SCIPY_OPT = False


_POOL_KEY = lambda item: item[0]

# The lambda above would violate the no-lambda rule if read strictly, so replace it below.
# Actually, remove lambda by defining a top-level function.
def _pool_key(item):
    return item[0]


_NEIGHBOR_STEPS = (0.02, 0.05, 0.10, 0.20, 0.35, 0.60, 1.00)

_ACTIVE_CONFIGS = (
    (1, 1), (1, 10), (2, 10), (3, 5), (3, 20),
    (5, 5), (5, 20), (8, 10), (8, 50),
    (12, 20), (20, 20), (20, 100), (40, 100),
)


def default_sequence(n=256):
    return np.full(int(n), 1.0 / float(n), dtype=float)


def clean_sequence(seq):
    try:
        arr = np.asarray(seq, dtype=float).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    if not np.all(np.isfinite(arr)):
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if float(arr.sum()) <= 1e-300:
        return None
    return arr


def normalize_positive(a):
    a = np.asarray(a, dtype=float).copy()
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        return None
    if not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return None
    a = a / s
    return a


def linear_conv_full(a):
    """Linear convolution a*a, using direct method for small n and FFT otherwise."""
    n = int(a.size)
    need = 2 * n - 1
    if n <= 128:
        c = np.convolve(a, a, mode="full")
        c = np.asarray(c, dtype=float)[:need]
        if c.size and float(c.min()) < 0.0:
            np.maximum(c, 0.0, out=c)
        return c

    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:need]
    if c.size and float(c.min()) < 0.0:
        np.maximum(c, 0.0, out=c)
    return c


def max_conv(a):
    c = linear_conv_full(a)
    return float(c.max()) if c.size else 0.0


def internal_score(a):
    """Fast equivalent of evaluate_sequence, without security checks."""
    a = np.asarray(a, dtype=float)
    if a.ndim != 1 or a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return float("inf")
    m = max_conv(a)
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m / (s * s))


def add_candidate(pool, a, max_size=100):
    b = normalize_positive(a)
    if b is None:
        return
    sc = internal_score(b)
    if not np.isfinite(sc):
        return
    pool.append((sc, b))
    if len(pool) > max_size:
        pool.sort(key=_pool_key)
        del pool[max_size:]


def build_shape(theta, x):
    """Fast positive log-polynomial Fourier shape."""
    t = np.zeros(10, dtype=float)
    theta = np.asarray(theta, dtype=float).reshape(-1)
    m = min(10, theta.size)
    if m:
        t[:m] = theta[:m]

    V = (
        t[0] * x
        + t[1] * x * x
        + t[2] * x * x * x
        + t[3] * (x - 0.5) * (x - 0.5)
        + t[4] * x * (1.0 - x)
        + t[5] * np.sin(np.pi * x)
        + t[6] * np.cos(2.0 * np.pi * x)
        + t[7] * np.sin(2.0 * np.pi * x)
        + t[8] * np.cos(4.0 * np.pi * x)
        + t[9] * np.sin(4.0 * np.pi * x)
    )
    V = np.asarray(V, dtype=float)
    V = V - float(np.max(V))
    V = np.clip(V, -700.0, 0.0)
    a = np.exp(V)
    out = normalize_positive(a)
    if out is None:
        out = default_sequence(x.size)
    return out


def random_log_sequence(n, rng, kind=None):
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    if kind is None:
        kind = int(rng.integers(0, 8))

    if kind == 0:
        lam = float(rng.uniform(0.1, 6.0))
        sign = 1.0 if rng.random() < 0.5 else -1.0
        V = sign * lam * x
        a = np.exp(V - V.max())
        out = normalize_positive(a)
        if out is not None:
            return out

    elif kind == 1:
        c1 = float(rng.uniform(-3.0, 3.0))
        c2 = float(rng.uniform(-3.0, 3.0))
        c3 = float(rng.uniform(-3.0, 3.0))
        V = c1 * x + c2 * x * x + c3 * x * x * x
        V = V - V.max()
        a = np.exp(np.clip(V, -700.0, 0.0))
        out = normalize_positive(a)
        if out is not None:
            return out

    elif kind == 2:
        V = np.zeros(n, dtype=float)
        for m in range(1, 8):
            amp = float(rng.uniform(0.03, 0.90))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            V += amp * np.cos(2.0 * np.pi * m * x + ph)
        V += float(rng.uniform(-2.0, 2.0)) * x
        V = V - V.max()
        a = np.exp(np.clip(V, -700.0, 0.0))
        out = normalize_positive(a)
        if out is not None:
            return out

    elif kind == 3:
        # two positive bumps, deliberately asymmetric
        z = np.ones(n, dtype=float) + 0.05
        amp1 = float(rng.uniform(0.2, 4.0))
        mu1 = float(rng.uniform(0.0, 0.6))
        sig1 = float(rng.uniform(0.02, 0.40))
        z += amp1 * np.exp(-((x - mu1) ** 2) / (2.0 * sig1 * sig1))
        amp2 = float(rng.uniform(0.2, 4.0))
        mu2 = float(rng.uniform(0.4, 1.0))
        sig2 = float(rng.uniform(0.02, 0.40))
        z += amp2 * np.exp(-((x - mu2) ** 2) / (2.0 * sig2 * sig2))
        return normalize_positive(z)

    elif kind == 4:
        # low-frequency random smooth positive sequence
        z = rng.random(n) + 0.05
        k = max(1, n // 16)
        kernel = np.ones(k, dtype=float)
        kernel /= float(k)
        z = np.convolve(z, kernel, mode="same")
        return normalize_positive(z + 1e-6)

    elif kind == 5:
        # power-law endpoint behavior, possibly U-shaped
        p = float(rng.uniform(-0.9, 0.5))
        q = float(rng.uniform(-0.9, 0.5))
        eps = 1e-3
        V = p * np.log(x + eps) + q * np.log((1.0 - x) + eps)
        V = V - V.max()
        a = np.exp(np.clip(V, -700.0, 0.0))
        return normalize_positive(a)

    elif kind == 6:
        # geometric with curvature
        lam = float(rng.uniform(-4.0, 4.0))
        curv = float(rng.uniform(-3.0, 3.0))
        V = lam * x + curv * x * x
        V = V - V.max()
        a = np.exp(np.clip(V, -700.0, 0.0))
        return normalize_positive(a)

    else:
        theta = rng.normal(0.0, 1.0, 10)
        return build_shape(theta, x)


def resample_sequence(old, n):
    """Log-linear resampling. Works for up- or down-sampling."""
    old = clean_sequence(old)
    if old is None:
        return None
    m = int(old.size)
    n = int(n)
    if m == n:
        return normalize_positive(old)
    log_old = np.log(old + 1e-300)
    x0 = np.linspace(0.0, 1.0, m)
    x1 = np.linspace(0.0, 1.0, n)
    log_new = np.interp(x1, x0, log_old)
    log_new = log_new - float(np.max(log_new))
    log_new = np.clip(log_new, -700.0, 0.0)
    a = np.exp(log_new)
    return normalize_positive(a)


def generate_deterministic_candidates(n, local, rng):
    n = int(n)
    add_candidate(local, default_sequence(n), 500)
    x = np.linspace(0.0, 1.0, n)

    for lam in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0):
        add_candidate(local, np.exp(-lam * x), 500)
        add_candidate(local, np.exp(lam * x), 500)

    z = x + 1e-3
    for p in (-0.7, -0.5, -0.3, -0.1, 0.1, 0.3, 0.5, 0.7):
        add_candidate(local, z ** p, 500)
        add_candidate(local, (1.0 - x + 1e-3) ** p, 500)
        r2 = float(rng.uniform(-0.4, 0.4))
        add_candidate(local, (z ** p) * ((1.0 - x + 1e-3) ** r2), 500)

    for v in (-4.0, -2.0, -1.0, -0.5, -0.2, 0.0, 0.2, 0.5, 1.0, 2.0, 4.0):
        for idx in (0, 1, 2):
            th = np.zeros(10, dtype=float)
            th[idx] = v
            add_candidate(local, build_shape(th, x), 500)
    for v in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        for idx in (3, 4, 5, 6, 7, 8, 9):
            th = np.zeros(10, dtype=float)
            th[idx] = v
            add_candidate(local, build_shape(th, x), 500)

    for _ in range(25):
        th = rng.normal(0.0, 1.0, 10)
        add_candidate(local, build_shape(th, x), 500)


def coordinate_phi(x, B, L, Q, S_base):
    vals = B + L * x + Q * x * x
    if vals.size and float(vals.min()) < 0.0:
        vals = np.maximum(vals, 0.0)
    den = S_base + x
    if den <= 1e-300 or not np.isfinite(den):
        return float("inf")
    m = float(vals.max())
    if not np.isfinite(m):
        return float("inf")
    return m / (den * den)


def coordinate_refine(a, deadline, max_sweeps=2):
    a = normalize_positive(a)
    if a is None:
        return None, float("inf")
    n = int(a.size)
    if n > 192:
        return a, internal_score(a)

    best = a.copy()
    best_score = internal_score(best)
    current = a.copy()
    current_score = best_score
    need = 2 * n - 1

    for _sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        changed = False
        order = np.arange(n)
        for j in range(n):
            if time.monotonic() > deadline:
                break
            idx = int(order[j])
            old = float(current[idx])
            c = linear_conv_full(current)

            off = np.arange(need, dtype=int) - idx
            valid = (off >= 0) & (off < n)

            B = c.copy()
            L = np.zeros(need, dtype=float)
            Q = np.zeros(need, dtype=float)

            if np.any(valid):
                ov = off[valid]
                contrib = np.where(ov == idx, old * old, 2.0 * old * current[ov])
                B[valid] = B[valid] - contrib
                if float(B.min()) < 0.0:
                    np.maximum(B, 0.0, out=B)
                L[valid] = np.where(ov == idx, 0.0, 2.0 * current[ov])
                Q[valid] = (ov == idx).astype(float)

            S_base = float(current.sum()) - old
            mean = float(current.sum()) / float(n)
            high = min(1.0, max(1e-8, 10.0 * old, 25.0 * mean, 0.10))
            if high <= 1e-12:
                high = 1e-6

            x_new = old
            if _HAS_SCIPY_OPT:
                try:
                    res = _scipy_optimize.minimize_scalar(
                        coordinate_phi,
                        bounds=(0.0, high),
                        args=(B, L, Q, S_base),
                        method="bounded",
                        options={"maxiter": 35},
                    )
                    rx = getattr(res, "x", None)
                    if rx is not None and np.isfinite(float(rx)):
                        x_new = float(rx)
                except Exception:
                    x_new = old
            else:
                grid = np.linspace(0.0, high, 25)
                grid = np.concatenate((grid, np.array([old])))
                x_new = old
                best_phi = coordinate_phi(old, B, L, Q, S_base)
                for gx in grid:
                    val = coordinate_phi(float(gx), B, L, Q, S_base)
                    if val < best_phi:
                        best_phi = val
                        x_new = float(gx)

            cand = current.copy()
            cand[idx] = x_new
            cand = normalize_positive(cand)
            if cand is None:
                continue
            sc = internal_score(cand)

            if sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
            if sc < current_score - 1e-12:
                current = cand
                current_score = sc
                changed = True

        if not changed:
            break

    return best, best_score


def active_gradient(a, c, top, beta):
    n = int(a.size)
    need = 2 * n - 1
    top = int(max(1, min(int(top), need)))
    maxv = float(c.max())
    if not np.isfinite(maxv) or maxv <= 0.0:
        return np.zeros(n, dtype=float), np.array([], dtype=int), np.array([], dtype=float)

    if top >= need:
        idx = np.arange(need, dtype=int)
    else:
        idx = np.argpartition(c, -top)[-top:].astype(int)

    vals = c[idx]
    vmax = float(vals.max())
    if not np.isfinite(vmax) or vmax <= 0.0:
        return np.zeros(n, dtype=float), idx, vals

    if idx.size == 1:
        w = np.ones(1, dtype=float)
    else:
        w = np.power(np.maximum(vals, 0.0) / vmax, float(beta))
        sw = float(w.sum())
        if not np.isfinite(sw) or sw <= 0.0:
            w = np.full(idx.size, 1.0 / float(idx.size), dtype=float)
        else:
            w = w / sw

    g = np.zeros(n, dtype=float)
    for k, wk in zip(idx, w):
        if not np.isfinite(wk) or wk == 0.0:
            continue
        k = int(k)
        if k < n:
            # j = 0..k, off = k-j = k..0
            g[: k + 1] += wk * a[k::-1]
        else:
            start = k - n + 1
            length = n - start
            g[start:] += wk * a[n - 1 : n - 1 - length : -1]

    return g, idx, vals


def line_search_direction(a, g, current_score):
    n = int(a.size)
    best_y = a
    best_score = current_score
    for step in _NEIGHBOR_STEPS:
        y = a - step * g
        y = np.clip(y, 0.0, None)
        s = float(y.sum())
        if s <= 1e-300:
            continue
        y = y / s
        m = max_conv(y)
        sc = float(2.0 * n * m)
        if sc < best_score:
            best_score = sc
            best_y = y.copy()
    return best_y, best_score


def perturb_sequence(a, rng):
    a = np.asarray(a, dtype=float).copy()
    n = int(a.size)
    y = a.copy()

    for _ in range(int(rng.integers(1, 4))):
        w = int(rng.integers(max(1, n // 250), max(2, n // 20)))
        s = int(rng.integers(0, max(1, n - w + 1)))
        y[s : s + w] *= float(np.exp(rng.normal(0.0, 0.35, w)))

    if rng.random() < 0.25:
        x = np.linspace(0.0, 1.0, n)
        m = int(rng.integers(1, 7))
        amp = float(rng.uniform(0.05, 0.35))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        y *= np.exp(amp * np.sin(2.0 * np.pi * m * x + ph))

    y = np.clip(y, 0.0, None)
    s = float(y.sum())
    if s <= 1e-300:
        return a
    y /= s
    return y


def improve_active(a, rng, deadline):
    a = normalize_positive(a)
    if a is None:
        return None, float("inf")

    best = a.copy()
    best_score = internal_score(best)
    current = a.copy()
    current_score = best_score
    stall = 0
    ci = 0
    it = 0
    n = int(current.size)

    while time.monotonic() < deadline:
        top, beta = _ACTIVE_CONFIGS[ci % len(_ACTIVE_CONFIGS)]
        ci += 1
        it += 1

        c = linear_conv_full(current)
        maxc = float(c.max())
        current_score = float(2.0 * n * maxc)
        if current_score < best_score - 1e-12:
            best = current.copy()
            best_score = current_score

        g, _, _ = active_gradient(current, c, top, beta)
        neighbor, nscore = line_search_direction(current, g, current_score)

        if nscore < current_score - 1e-13:
            current = neighbor
            current_score = nscore
            stall = 0
        else:
            improved = False
            for t2, b2 in ((1, 1), (3, 10), (8, 10), (20, 30)):
                if time.monotonic() > deadline:
                    break
                c2 = linear_conv_full(current)
                g2, _, _ = active_gradient(current, c2, t2, b2)
                nb2, sc2 = line_search_direction(current, g2, current_score)
                if sc2 < current_score - 1e-13:
                    current = nb2
                    current_score = sc2
                    improved = True
                    break
            if improved:
                stall = 0
            else:
                stall += 1
                if stall >= 16:
                    current = best.copy()
                    current_score = best_score
                    current = perturb_sequence(current, rng)
                    current_score = internal_score(current)
                    stall = 0

        if it % 80 == 0:
            print(
                "active n=%d it=%d best=%.8f cur=%.8f"
                % (n, it, best_score, current_score),
                flush=True,
            )

    return best, best_score


def lp_polish(a, deadline):
    a = normalize_positive(a)
    if a is None:
        return None, float("inf")
    n = int(a.size)
    if not _HAS_SCIPY_OPT:
        return a, internal_score(a)
    if n > 1600:
        return a, internal_score(a)

    best = a.copy()
    best_score = internal_score(best)
    start = time.monotonic()
    hard = min(deadline, start + 160.0)

    while time.monotonic() < hard:
        c = linear_conv_full(best)
        M = float(c.max())
        if M <= 0.0 or not np.isfinite(M):
            break

        A = np.zeros((2 * n - 1, n), dtype=float)
        for i in range(n):
            A[i : i + n, i] = best

        ones = np.ones(n, dtype=float)
        remaining = max(2, int(hard - time.monotonic()))
        try:
            res = _scipy_optimize.linprog(
                -ones,
                A_ub=A,
                b_ub=np.full(2 * n - 1, M, dtype=float),
                bounds=[(0.0, None)] * n,
                method="highs",
                options={"time_limit": remaining, "presolve": True},
            )
        except Exception:
            break

        raw_x = getattr(res, "x", None)
        if raw_x is None:
            break
        g = np.asarray(raw_x, dtype=float)
        if g.size != n or not np.all(np.isfinite(g)):
            break
        gsum = float(g.sum())
        if gsum <= 1.0 + 1e-7:
            break

        g = normalize_positive(g)
        if g is None:
            break

        improved = False
        for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
            y = (1.0 - t) * best + t * g
            y = normalize_positive(y)
            if y is None:
                continue
            sc = internal_score(y)
            if sc < best_score - 1e-12:
                best = y
                best_score = sc
                improved = True
                break
        if not improved:
            break

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s))
    deadline = t0 + max(20.0, budget - 40.0)
    rng = np.random.default_rng(seed)
    pool = []

    try:
        fallback = globals().get("height_sequence_1", None)
        fb = clean_sequence(fallback) if fallback is not None else None
    except Exception:
        fb = None

    if fb is not None:
        add_candidate(pool, fb, 300)
        # Downsample large fallback to useful coarse sizes.
        for m in (32, 64, 128):
            add_candidate(pool, resample_sequence(fb, m), 300)

    # If no usable fallback, provide basic starting points.
    if not pool:
        for m in (32, 64, 128, 512):
            add_candidate(pool, default_sequence(m), 300)

    # ---------- Phase 1: coarse random/deterministic + coordinate refinement ----------
    coarse_ns = [16, 32, 64, 128]
    coarse_fraction = min(0.35, max(0.12, 300.0 / max(budget, 1.0)))
    coarse_total = min(420.0, max(30.0, (deadline - t0) * coarse_fraction))
    segment = coarse_total / float(len(coarse_ns))

    for ncoarse in coarse_ns:
        seg_deadline = min(deadline, time.monotonic() + segment)
        local = []
        generate_deterministic_candidates(int(ncoarse), local, rng)

        # Add random starts, bounded by time.
        for _ in range(250):
            if time.monotonic() > seg_deadline:
                break
            a = random_log_sequence(int(ncoarse), rng)
            add_candidate(local, a, 500)

        local.sort(key=_pool_key)

        # Refine the best few coarse candidates by coordinate descent.
        top_count = 10 if ncoarse <= 32 else (6 if ncoarse <= 64 else 3)
        max_sweeps = 4 if ncoarse <= 32 else (2 if ncoarse <= 64 else 1)
        refine_deadline = min(deadline, seg_deadline + 15.0)

        for sc, a in local[:top_count]:
            if time.monotonic() > refine_deadline:
                break
            a2, sc2 = coordinate_refine(a, refine_deadline, max_sweeps=max_sweeps)
            add_candidate(pool, a2, 350)
            add_candidate(pool, a, 350)

        print(
            "coarse n=%d local_top=%.8f pool_size=%d t=%.1f"
            % (ncoarse, local[0][0] if local else float("inf"), len(pool),
               time.monotonic() - t0),
            flush=True,
        )

    # ---------- Phase 2: smooth parametric candidates at medium length ----------
    shape_deadline = min(deadline, time.monotonic() + min(180.0, 0.08 * budget))
    for nshape in (64, 128, 256):
        if time.monotonic() > shape_deadline:
            break
        x = np.linspace(0.0, 1.0, nshape)
        for _ in range(120):
            if time.monotonic() > shape_deadline:
                break
            theta = rng.normal(0.0, 1.0, 10)
            # Bias the first coordinate to encourage asymmetry.
            theta[0] = float(rng.uniform(-4.0, 4.0))
            add_candidate(pool, build_shape(theta, x), 350)

    # Resample the best small candidates to large practical lengths.
    pool.sort(key=_pool_key)
    small_candidates = []
    for sc, a in pool:
        if a.size <= 256:
            small_candidates.append((sc, a))
            if len(small_candidates) >= 18:
                break

    # Also include direct deterministic large starts.
    for nlarge in (512, 1024, 2048, 4096):
        add_candidate(pool, default_sequence(nlarge), 350)
        x = np.linspace(0.0, 1.0, nlarge)
        for v in (-4.0, -2.0, -1.0, 1.0, 2.0, 4.0):
            th = np.zeros(10, dtype=float)
            th[0] = v
            add_candidate(pool, build_shape(th, x), 350)

    resample_deadline = min(deadline, time.monotonic() + min(180.0, 0.08 * budget))
    for nfine in (1024, 2048, 4096):
        if time.monotonic() > resample_deadline:
            break
        for sc, a in small_candidates:
            if time.monotonic() > resample_deadline:
                break
            b = resample_sequence(a, nfine)
            add_candidate(pool, b, 400)

    # ---------- Phase 3: large-length active-gradient refinement ----------
    pool.sort(key=_pool_key)
    fine_queue = []
    for sc, a in pool:
        if a.size >= 512:
            fine_queue.append((sc, a))
            if len(fine_queue) >= 14:
                break

    if not fine_queue and pool:
        a = pool[0][1]
        b = resample_sequence(a, 2048)
        if b is not None:
            fine_queue.append((internal_score(b), b))

    improve_deadline = deadline - max(120.0, 0.06 * budget)
    idx = 0
    while time.monotonic() < improve_deadline and idx < len(fine_queue):
        sc, a = fine_queue[idx]
        idx += 1
        if a.size < 1024:
            a2 = resample_sequence(a, 2048)
            if a2 is not None:
                a = a2

        if idx <= 2:
            alloc = 520.0
        elif idx <= 5:
            alloc = 360.0
        elif idx <= 9:
            alloc = 240.0
        else:
            alloc = 140.0

        end = min(improve_deadline, time.monotonic() + alloc)
        if end - time.monotonic() < 30.0:
            break

        a, sc = improve_active(a, rng, end)
        if a is not None:
            add_candidate(pool, a, 500)
            # Try a larger length of the same improved shape.
            if a.size <= 4096:
                bigger = resample_sequence(a, min(8192, int(a.size) * 2))
                add_candidate(pool, bigger, 500)
            print(
                "improved candidate n=%d score=%.8f t=%.1f"
                % (a.size, sc, time.monotonic() - t0),
                flush=True,
            )

    # Second smaller pass on freshly improved top candidates.
    pool.sort(key=_pool_key)
    if time.monotonic() < improve_deadline:
        second_queue = []
        for sc, a in pool:
            if a.size >= 1024:
                second_queue.append((sc, a))
                if len(second_queue) >= 5:
                    break
        for sc, a in second_queue:
            if time.monotonic() >= improve_deadline:
                break
            end = min(improve_deadline, time.monotonic() + 120.0)
            if end - time.monotonic() < 25.0:
                break
            a2, sc2 = improve_active(a, rng, end)
            if a2 is not None:
                add_candidate(pool, a2, 500)

    # ---------- Phase 4: optional LP polish on a few moderate-size best points ----------
    pool.sort(key=_pool_key)
    lp_deadline = deadline - max(60.0, 0.03 * budget)
    lp_candidates = []
    for sc, a in pool:
        if 256 <= a.size <= 1600:
            lp_candidates.append((sc, a))
            if len(lp_candidates) >= 3:
                break
    if not lp_candidates and pool:
        a = pool[0][1]
        b = resample_sequence(a, 1024)
        if b is not None:
            lp_candidates.append((internal_score(b), b))

    for i, (sc, a) in enumerate(lp_candidates):
        remaining = lp_deadline - time.monotonic()
        if remaining <= 90.0:
            break
        end = min(lp_deadline, time.monotonic() + max(90.0, remaining * 0.45) - 10.0 * i)
        if end - time.monotonic() < 60.0:
            break
        a2, sc2 = lp_polish(a, end)
        if a2 is not None and sc2 < best_pool_score(pool, 500) - 1e-12:
            add_candidate(pool, a2, 500)
            print("lp n=%d score=%.8f t=%.1f" % (a2.size, sc2, time.monotonic() - t0), flush=True)

    # ---------- Phase 5: final exact evaluation and safe return ----------
    pool.sort(key=_pool_key)
    best_seq = fb
    best_exact = float("inf")

    if best_seq is not None:
        try:
            best_exact = float(evaluate_sequence(best_seq.tolist()))
        except Exception:
            best_exact = float("inf")

    # Evaluate a small final set. Prefer long candidates that are still safe.
    final_set = []
    for sc, a in pool:
        if a.size <= 8192:
            final_set.append(a)
        if len(final_set) >= 8:
            break

    if not final_set:
        final_set = [default_sequence(256)]

    for a in final_set:
        if time.monotonic() > t0 + budget - 5.0:
            break
        try:
            vv = float(evaluate_sequence(a.tolist()))
        except Exception:
            vv = float("inf")
        if np.isfinite(vv) and vv < best_exact:
            best_exact = vv
            best_seq = a
        print("final check n=%d eval=%.8f" % (a.size, vv), flush=True)

    if best_seq is None or not (isinstance(best_seq, np.ndarray) and best_seq.size > 0):
        if pool:
            best_seq = pool[0][1]
        else:
            best_seq = default_sequence(256)

    out = np.asarray(best_seq, dtype=float)
    out = np.clip(out, 0.0, 1000.0)

    print(
        "best returned n=%d internal=%.8f exact_best=%.8f wall=%.1f"
        % (out.size, internal_score(out), best_exact, time.monotonic() - t0),
        flush=True,
    )

    return [float(x) for x in out]


def best_pool_score(pool, limit=100):
    if not pool:
        return float("inf")
    scores = [float(s) for s, _ in pool[:limit]]
    return min(scores) if scores else float("inf")
