import numpy as np
import time


def make_catalan_sequence(n):
    """Generate a_k = C(2k,k)/4^k for k=0,...,n-1 using log-space cumprod."""
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    k = np.arange(1, n, dtype=np.float64)
    # a_k / a_{k-1} = (2(k-1)+1) / (2k) = (2k-1)/(2k)
    log_ratios = np.log(2.0 * k - 1.0) - np.log(2.0 * k)
    log_a = np.empty(n, dtype=np.float64)
    log_a[0] = 0.0
    log_a[1:] = np.cumsum(log_ratios)
    return np.exp(log_a)
