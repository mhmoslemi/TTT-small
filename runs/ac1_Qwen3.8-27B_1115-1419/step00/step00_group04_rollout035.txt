```python
import numpy as np
import time
import cvxpy as cp

def optimize_with_lp(a, max_iterations=100, time_budget=100):
    """Optimize using the LP-based approach."""
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    S = np.sum(a)
    if S < 0.01:
        return a
    a = a / S * np.sqrt(2*n)  # Normalize so sum = sqrt(2n)
    
    best = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a)) / (np.sum(a)**2)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > time_budget:
            break
        
        c = np.convolve(a, a)
        M = np.max(c)
        
        # Set up LP: maximize sum(g) subject to (a*g)_k <= M for all k, g >= 0
        g = cp.Variable(n)
        constraints = [g >= 0]
        for k in range(2*n - 1):
            # (a*g)_k = sum_m a[m]*g[k-m] where valid m and k-m
            terms = []
            for m in range(max(0, k-n+1), min(n, k+1)):
                j = k - m
                if 0 <= j < n:
                    terms.append(a[m] * g[j])
            if terms:
                constraints.append(cp.sum(terms) <= M)
        
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)
        
        if g.value is None:
            break
        
        g_val = np.array(g.value).flatten()
        g_val = np.clip(g_val, 0, 1000)
        
        if np.sum(g_val) < 1e-10:
            break
        
        # Normalize g to have same sum as a
        g_val = g_val / np.sum(g_val) * np.sum(a)
        
        # Try different step sizes
        best_t = 0
        best_new_score = best_score
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_trial = (1-t)*a + t*g_val
            a_trial = np.clip(a_trial, 0, 1000)
            score = 2*n*np.max(np.convolve(a_trial, a_trial))/(np.sum(a_trial)**2) if np.sum(a_trial) > 0.01 else float('inf')
            if score < best_new_score:
                best_new_score = score
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*g_val
            a = np.clip(a, 0, 1000)
            score = 2*n*np.max(np.convolve(a, a))/(np.sum(a)**2)
            if score < best_score:
                best_score = score
                best = a.copy()
    
    return best
```
