import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix


def _fast_score(a):
    """Fast computation of 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    b = np.convolve(a, a)
    M = np.max(b)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)


def _compute_gradient(a, tau, N_fft):
    """Compute gradient of smoothed max(conv(a,a)) using softmax approximation."""
    n = len(a)
    b = np.convolve(a, a)
    bmax = np.max(b)
    x = tau * (b - bmax)
    p = np.exp(x)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p /= ps
    P = np.fft.rfft(p, N_fft)
    A = np.fft.rfft(a, N_fft)
    r = np.fft.irfft(P * np.conj(A), N_fft)
    g = 2.0 * r[:n]
    return g


def _build_lp_matrix(a):
    """Build sparse constraint matrix for conv(a,b) <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            rows.extend([k] * (j_hi - j_lo + 1))
            cols.extend(js.tolist())
            vals.extend((a[k - js]).tolist())
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    A = _build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _mirror_step(a, g, eta):
    """One mirror descent step: multiplicative update + normalize."""
    # Center gradient for numerical stability
    g_centered = g - np.mean(g)
    # Clip to prevent overflow
    g_centered = np.clip(g_centered, -10.0 / max(eta, 1e-10), 10.0 / max(eta, 1e-10))
    a_new = a * np.exp(-eta * g_centered)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-15:
        return a.copy()
    a_new = a_new / S
    return a_new


def _additive_step(a, g, eta):
    """One additive gradient step: subtract, clip, normalize."""
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-15:
        return a.copy()
    a_new = a_new / S
    return a_new

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = evaluate_sequence(final_list)
    print(f"Final: score={final_score:.6f}, n={len(final_list)}")
    
    if final_score < best_score:
        best_score = final_score
    
    return final_list
