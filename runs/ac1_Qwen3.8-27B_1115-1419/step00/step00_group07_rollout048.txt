    """Solve the LP to find g that maximizes sum(g) subject to f*g <= M elementwise and g >= 0."""

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

def subgradient_step(f, alpha):
    n = len(f)
    c = np.convolve(f, f)
    k_star = np.argmax(c)
    
    # Gradient: grad[i] = 2*f[k_star - i] if 0 <= k_star - i < n
    grad = np.zeros(n)
    valid = np.arange(n)
    idx = k_star - valid  # k_star - i for i = 0, ..., n-1
    mask = (idx >= 0) & (idx < n)
    grad[mask] = 2 * f[idx[mask]]
    
    # Subgradient step
    f_new = f - alpha * grad
    f_new = project_simplex(f_new)

    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    """Compute gradient of smooth max of convolve(f,f) using FFT. O(n log n)."""
    n = len(f)
    c = np.convolve(f, f)  # length 2n-1
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()  # length 2n-1

    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)

    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
