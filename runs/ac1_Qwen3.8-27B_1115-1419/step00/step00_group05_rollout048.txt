"""
Algorithm: FFT-based gradient descent on the probability distribution,
with multiple initializations, optional LP steps, and time-bounded search.

The objective is score = 2*n*max(convolve(p,p)) where p is a probability
distribution (sum=1). We minimize this via sub-gradient descent targeting
the "hottest" convolution coefficients, using FFT for O(n log n) per step.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution of two 1D arrays using FFT."""
    n = len(a) + len(b) - 1
    fft_len = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    return np.fft.irfft(fa * fb, fft_len)[:n]

def compute_gradient(p, w, n):
    """
    Compute grad[i] = 2 * sum_k w[k] * p[k-i] (for valid k-i in [0,n-1]).
    Uses FFT for efficiency.
    """
    p_rev = p[::-1]
    conv = fft_convolve(w, p_rev)
    # grad[i] = conv[n-1+i]
    grad = 2.0 * conv[n-1 : n-1+n]
    return grad

def score_from_distribution(p, n):
    """Compute the score equivalent to evaluate_sequence for a normalized distribution."""
    c = fft_convolve(p, p)
    max_c = np.max(c)
    return 2.0 * n * max_c

def optimize_one(p, n, rng, deadline, symmetric=False, verbose=False):
    """
    Optimize a probability distribution p to minimize 2*n*max(p*p).
    Uses FFT-based sub-gradient descent.
    Returns the best p found and its score.
    """
    p = p.astype(np.float64).copy()
    p = np.clip(p, 0, None)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p /= s
    
    best_p = p.copy()
    best_score = score_from_distribution(p, n)
    
    momentum = np.zeros(n)
    beta = 0.9
    
    # Adaptive learning rate
    lr = 0.1
    lr_decay = 0.999
    lr_min = 0.001
    
    for iteration in range(100000):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = fft_convolve(p, p)
        
        # Compute weights for the gradient
        # Use soft weighting: w[k] = c[k]^alpha, emphasized for peaks
        c_max = c.max()
        if c_max < 1e-15:
            break
        alpha = 4.0  # emphasis factor
        w = (c / c_max) ** alpha
        # Normalize w so that the gradient has consistent scale
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        
        # Compute gradient
        grad = compute_gradient(p, w, n)
        
        # Subtract mean to preserve sum (approximately)
        grad -= grad.mean()
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        p -= lr * momentum
        
        # Project to simplex (p >= 0, sum(p) = 1)
        p = np.clip(p, 0, None)
        s = p.sum()
        if s < 1e-10:
            p = best_p.copy()
            momentum = np.zeros(n)
            lr = lr * 2  # restart with larger lr
        else:
            p /= s
        
        if symmetric:
            p = 0.5 * (p + p[::-1])
            p /= p.sum()
        
        # Decay learning rate
        lr = max(lr_min, lr * lr_decay)
        
        # Evaluate periodically
        if (iteration + 1) % 50 == 0:
            current_score = score_from_distribution(p, n)
            if current_score < best_score:

                best_p = p.copy()
            if verbose and (iteration + 1) % 500 == 0:
                print(f"  iter {iteration+1}: score={current_score:.6f}, best={best_score:.6f}, lr={lr:.4f}")

def lp_step(p, n, deadline, max_time=30.0):
    """
    Perform one LP-based step.
    Given p (normalized), M = max(p*p), find g >= 0 maximizing sum(g)
    subject to (p*g)(k) <= M for all k.
    Returns the mixed sequence or None if LP fails.

    start_time = time.monotonic()
    
    c = fft_convolve(p, p)
    M = c.max()
    
    # Build constraint matrix A such that A @ g <= M
    # (p*g)(k) = sum_j p[j] * g[k-j] <= M
    # For each k, the constraint is: sum_{j: 0<=k-j<n} p[j] * g[k-j] <= M
    # Let i = k-j, so j = k-i, and i ranges over [max(0,k-n+1), min(n-1,k)]
    # Constraint k: sum_{i=max(0,k-n+1)}^{min(n-1,k)} p[k-i] * g[i] <= M
    
    # Build the (2n-1) x n constraint matrix
    # This is large, so let's use a sparse representation
    # Actually, for n up to ~5000, let's try dense first
    
    if n > 6000:
        # Too large for dense LP, use subset of constraints
        # Use only constraints near the center
        k_center = n - 1
        k_range = range(max(0, k_center - n//2), min(2*n-1, k_center + n//2 + 1))
    else:
        k_range = range(2*n - 1)
    
    n_constraints = len(k_range)
    A = np.zeros((n_constraints, n))
    for idx, k in enumerate(k_range):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[idx, k - j] = p[j]
    
    # Objective: minimize -sum(g) (i.e., maximize sum(g))
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=np.full(n_constraints, M),
                         bounds=bounds, method='highs',
                         options={'time_limit': max_time})
        
        if time.monotonic() - start_time > max_time:
            return None
        
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum <= 0:
                return None
            # Rescale g to match p's normalization (sum=1)
            g = g / g_sum
            # Mix
            t = 0.3  # mixing parameter
            new_p = (1 - t) * p + t * g
            new_p = np.clip(new_p, 0, None)
            new_sum = new_p.sum()
            if new_sum > 0:
                new_p /= new_sum
                return new_p
    except Exception as e:

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    fft_len = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    return np.fft.irfft(fa * fb, fft_len)[:n]

def _cross_corr_with_weights(p, w, n):
    """Compute grad[i] = sum_k w[k] * p[k-i] for i in [0, n-1]."""
    p_rev = p[::-1]
    conv = _fft_convolve(w, p_rev)
    return conv[n-1 : n-1+n].copy()

def _score(p, n):
    c = _fft_convolve(p, p)
    return 2.0 * n * c.max()

"""
FFT-based multiplicative gradient descent for minimum peak autocorrelation sequences.
Minimizes score = 2*n*max(convolve(p,p)) where p is a probability distribution (sum=1).
Uses soft-weighted sub-gradients computed via FFT, multiple diverse initializations,
symmetry exploitation, and adaptive learning rates with stall detection.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute linear convolution of two 1D arrays using FFT."""
    n = len(a) + len(b) - 1
    fft_len = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    return np.fft.irfft(fa * fb, fft_len)[:n]


def _cross_corr(p, w, n):
    """Compute grad[i] = sum_k w[k] * p[k-i] for i in [0..n-1] using FFT."""
    p_rev = p[::-1]
    conv = _fft_convolve(w, p_rev)
    return conv[n - 1: n - 1 + n].copy()


def _score_p(p, n):
    """Compute 2*n*max(p*p) for normalized p (sum=1)."""
    c = _fft_convolve(p, p)
    return 2.0 * n * float(c.max())


def _optimize_mp(p0, n, rng, deadline, symmetric=False, verbose=False, label=""):
    """
    Multiplicative-update optimizer for minimizing 2*n*max(p*p).
    Uses soft-weighted sub-gradient via FFT.
    """
    p = np.clip(p0.astype(np.float64), 0.0, None).copy()
    s = p.sum()
    if s < 1e-12:
        p = np.ones(n, dtype=np.float64)
        s = float(n)
    p /= s

    best_p = p.copy()
    best_score = _score_p(p, n)

    if verbose:
        print(f"  [{label}] init score = {best_score:.6f}")

    alpha = 3.0
    lr = 0.12
    lr_decay = 0.9997
    lr_min = 0.003
    no_improve = 0
    stall_threshold = 300
    perturb_scale = 0.005
    iter_count = 0
    batch_size = 100

    while time.monotonic() < deadline:
        for _ in range(batch_size):
            if time.monotonic() > deadline:
                break

            c = _fft_convolve(p, p)
            c_max = c.max()
            if c_max < 1e-18:
                break

            w = (c / c_max) ** alpha
            pressure = _cross_corr(p, w, n)

            # Multiplicative update
            p = p * np.exp(-lr * pressure)
            p = np.clip(p, 1e-18, None)
            s_new = p.sum()
            if s_new < 1e-10:
                p = best_p.copy()
                lr = min(lr * 2.0, 0.8)
            else:
                p = p / s_new

            if symmetric:
                p = 0.5 * (p + p[::-1])
                p /= p.sum()

            lr = max(lr_min, lr * lr_decay)
            iter_count += 1

        # Evaluate
        current_score = _score_p(p, n)
        if current_score < best_score - 1e-9:
            best_score = current_score
            best_p = p.copy()
            no_improve = 0
            if verbose and iter_count % 1000 < batch_size:
                print(f"  [{label}] iter {iter_count}: {current_score:.6f} (new best)")
        else:
            no_improve += 1

        if no_improve >= stall_threshold:
            if verbose:
                print(f"  [{label}] stall at iter {iter_count}, best={best_score:.6f}")
            p = p + perturb_scale * rng.standard_normal(n)
            p = np.clip(p, 0.0, None)
            s2 = p.sum()
            if s2 > 0:
                p /= s2
            if symmetric:
                p = 0.5 * (p + p[::-1])
                p /= p.sum()
            lr = min(lr * 3.0, 0.4)
            perturb_scale *= 1.5
            no_improve = 0

    return best_p, best_score


def _lp_mix(p, n, max_time=20.0):
    """LP-based mixing step. Returns new p or None."""
    from scipy.optimize import linprog

    c = _fft_convolve(p, p)
    M = float(c.max())

    # Use central subset of constraints for large n
    if n > 4000:
        center = n - 1
        half = n // 3
        k_lo = max(0, center - half)
        k_hi = min(2 * n - 2, center + half)
    else:
        k_lo, k_hi = 0, 2 * n - 2

    n_cons = k_hi - k_lo + 1
    A = np.zeros((n_cons, n), dtype=np.float64)
    for idx in range(n_cons):
        k = k_lo + idx
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[idx, k - j] = p[j]

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=np.full(n_cons, M),
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time})
        if time.monotonic() - t0 > max_time:
            return None
        if res.success and res.x is not None:
            g = res.x
            gsum = g.sum()
            if gsum > 1e-10:
                g /= gsum
                t_mix = 0.4
                new_p = (1.0 - t_mix) * p + t_mix * g
                new_p = np.clip(new_p, 0.0, None)
                ns = new_p.sum()
                if ns > 1e-10:
                    return new_p / ns
    except Exception:
        pass
    return None


def _make_inits(n, rng, initial_seq, n_init):
    """Generate diverse initial probability distributions of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    # 1. Resample from provided sequence
    if n_init > 1:
        x_orig = np.linspace(0.0, 1.0, n_init)
        resampled = np.interp(x, x_orig, initial_seq.astype(np.float64))
        resampled = np.clip(resampled, 0.0, None)
        rs = resampled.sum()
        if rs > 0:
            inits.append(("resamp", resampled / rs))

    # 2. Cosine U-shape
    p = (1.0 + np.cos(2.0 * np.pi * x)) ** 0.7
    p = np.clip(p, 1e-8, None)
    inits.append(("cosU", p / p.sum()))

    # 3. Sharper U
    p = (1.0 + np.cos(2.0 * np.pi * x)) ** 3.0
    p = np.clip(p, 1e-8, None)
    inits.append ("cosU3", p / p.sum())

    # 4. Inverse-Gaussian (U-shaped)
    p = 1.0 - 0.85 * np.exp(-((x - 0.5) / 0.18) ** 2)
    p = np.clip(p, 0.005, None)
    inits.append(("invGauss", p / p.sum()))

    # 5. Two humps
    p = np.exp(-((x - 0.2) / 0.12) ** 2) + np.exp(-((x - 0.8) / 0.12) ** 2) + 0.01
    inits.append(("twoHump", p / p.sum()))

    # 6. Asymmetric ramp
    p = np.linspace(0.05, 1.0, n)
    inits.append(("ramp", p / p.sum()))

    # 7. Sine squared (U-shaped, zero at center)
    p = np.sin(np.pi * x) ** 2 + 0.05
    inits.append(("sinU", p / p.sum()))

    # 8. Random
    p = rng.random(n)
    inits.append(("rand", p / p.sum()))

    # 9. Power-law edges
    p = np.power(np.minimum(x, 1.0 - x) + 0.01, -0.3)
    p = np.clip(p, 0.0, None)
    inits.append(("powEdge", p / p.sum()))

    # 10. Triangular (inverted)
    p = np.minimum(x, 1.0 - x)
    p = np.clip(p, 0.01, None)
    inits.append(("invTri", p / p.sum()))

    return inits

    total_time = budget_s - 12.0

    t_polish = total_time * 0.15

    # === PHASE 1: Exploration ===

    phase1_deadline = time.monotonic() + t_exploration

    time_per_n = t_exploration / len(n_list)

    for n in n_list:
        n_start = time.monotonic()
        n_deadline = min(phase1_deadline, n_start + time_per_n)

        print(f"\n  n={n}, time budget={time_per_n:.0f}s")

        inits = _make_inits(n, rng, initial, n_init)

        # Shuffle inits for diversity
        rng.shuffle(inits)

        # Try each init with limited time
        time_per_init = (n_deadline - time.monotonic()) / len(inits)

        for label, p_init in inits:
            if time.monotonic() > n_deadline:
                break
            init_deadline = min(n_deadline, time.monotonic() + time_per_init)

            # Try symmetric version
            p_opt, score = _optimize_mp(
                p_init, n, rng, init_deadline,
                symmetric=True, verbose=False, label=f"{label}_s"
            )

            if score < best_score:
                best_score = score
                best_p = p_opt.copy()
                best_n = n
                print(f"    [{label}_s] new best: {score:.6f} (n={n})")

            # Try asymmetric version with remaining time
            if time.monotonic() < n_deadline:
                asym_deadline = min(n_deadline, time.monotonic() + time_per_init * 0.6)
                p_opt2, score2 = _optimize_mp(
                    p_init, n, rng, asym_deadline,
                    symmetric=False, verbose=False, label=f"{label}_a"
                )
                if score2 < best_score:
                    best_score = score2
                    best_p = p_opt2.copy()
                    best_n = n
                    print(f"    [{label}_a] new best: {score2:.6f} (n={n})")
