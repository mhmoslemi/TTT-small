from scipy.optimize import linprog

def solve_lp_direction(a, M):
    """Solve LP to find direction g that maximizes sum(g) s.t. conv(a*g) ≤ M, g ≥ 0."""
    n = len(a)
    
    # Build constraint matrix A: (2n-1) × n
    # A[k, j] = a[k-j] if 0 ≤ k-j ≤ n-1, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # a[k-j] for k = j, j+1, ..., j+n-1
        # But k max is 2n-2
        end = min(2*n-1, j+n)
        A[j:end, j] = a[:end-j]
    
    # c = -1 (we minimize -sum(g))
    c = -np.ones(n)
    
    # Constraints: A @ g ≤ M
    b = np.full(2*n-1, M)
    
    # g ≥ 0 (bounds)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None
