import numpy as np
import time
from scipy.optimize import linprog

def solve_lp_direction(f, M):
    """Solve LP to find direction g that maximizes sum(g) subject to (f*g)_k <= M and g >= 0."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_i f_i * g_{k-i} where 0 <= k-i < n
    # For each k, the constraint is: sum_{i: 0<=k-i<n} f_i * g_{k-i} <= M
    # Let j = k-i, so i = k-j, and we need 0 <= k-j < n, i.e., k-n < j <= k
    # Also 0 <= j < n
    # So j ranges from max(0, k-n+1) to min(n-1, k)
    
    A_ub = np.zeros((2*n - 1, n))
    b_ub = np.full(2*n - 1, M)
    
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            A_ub[k, j] = f[i]
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

    n = len(f)
    c = -np.ones(n)
    
    # Build sparse constraint matrix
    # For each k in 0..2n-2: sum_{j} f[k-j] * g[j] <= M
    # Row k has non-zeros at columns j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)

    cols = []
    vals = []

            cols.append(j)

"""
LP-based alternating direction method for minimizing the self-convolution ratio.
Uses the linearization trick: fix current sequence f, solve LP for best direction g
such that (f*g)_k <= M for all k, then move toward g via line search.
Also explores multiple starting points and sequence lengths.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def build_sparsity_pattern(n):
    """Precompute the sparse pattern indices for the convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_direction_lp(f, M, row_idx, col_idx, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    vals = np.empty(len(col_idx), dtype=np.float64)
    for idx, j in enumerate(col_idx):
        k = row_idx[idx]
        i = k - j
        vals[idx] = f[i]
    A_ub = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None


def objective_normalized(f, n):
    """Compute 2*n*max(f*f) assuming sum(f)=1."""
    M = np.max(np.convolve(f, f))
    return 2.0 * n * M


def run_lp_search(f0, n, row_idx, col_idx, deadline, max_iters=500):
    """Run LP-based iterations starting from f0 (normalized to sum=1)."""
    f = f0.copy()
    current_obj = objective_normalized(f, n)
    best_local = f.copy()
    best_local_obj = current_obj
    alphas = [1.0, 0.5, 0.25, 0.15, 0.08, 0.04]
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()

        M = np.max(np.convolve(f, f))
        g = solve_direction_lp(f, M, row_idx, col_idx, n)
        if g is None:
            break

        g_sum = np.sum(g)
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Line search
        best_alpha = None
        best_new_obj = current_obj
        for alpha in alphas:
            new_f = (1.0 - alpha) * f + alpha * g
            new_M = np.max(np.convolve(new_f, new_f))
            new_obj = 2.0 * n * new_M
            if new_obj < best_new_obj * (1.0 - 1e-10):
                best_new_obj = new_obj
                best_alpha = alpha

        if best_alpha is None:
            # Try very small alphas
            found = False
            for alpha in [0.01, 0.005, 0.002, 0.001]:
                new_f = (1.0 - alpha) * f + alpha * g
                new_M = np.max(np.convolve(new_f, new_f))
                new_obj = 2.0 * n * new_M
                if new_obj < current_obj:
                    best_alpha = alpha
                    best_new_obj = new_obj
                    found = True
                    break
            if not found:

                    break
                continue

        f = (1.0 - best_alpha) * f + best_alpha * g
        f = np.clip(f, 0, None)
        # Renormalize to handle numerical drift
        f_sum = np.sum(f)
        if f_sum > 0:
            f = f / f_sum
        else:
            break

        current_obj = objective_normalized(f, n)

        if current_obj < best_local_obj:
            best_local_obj = current_obj
            best_local = f.copy()
            stall = 0
        else:

            print(f"  LP it={it}, obj={current_obj:.6f}, best={best_local_obj:.6f}, it_time={elapsed:.3f}s")

    return best_local, best_local_obj


def make_structured_start(n, rng):
    """Create a structured starting sequence for diversity."""
    x = np.arange(n, dtype=np.float64)

    a += 0.1 * rng.random(n)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a
