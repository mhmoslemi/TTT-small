import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""
LP-based iterative minimization of 2*n*max(conv(a,a))/(sum(a))^2.
At each step, solve an LP to find non-negative g maximizing sum(g)
subject to conv(f,g) <= max(conv(f,f)), then update f toward g.
Uses precomputed sparsity pattern, adaptive stepping, and restarts.
"""


def _build_sparsity(n):
    """Precompute sparsity pattern (rows, cols) for convolution constraint matrix."""
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(i_start, i_end + 1)
        idx += count
    return rows, cols


def _fill_data(f, n):
    """Fill data values for the convolution constraint matrix given current f."""
    nnz = n * n
    data = np.empty(nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        data[idx:idx + count] = f[k - i_end: k - i_start + 1][::-1]
        idx += count
    return data


def _fast_obj(f):
    """Fast objective computation (same formula as evaluate_sequence)."""
    n = len(f)
    s = float(np.sum(f))
    if s < 0.01:
        return float('inf')
    conv_full = np.convolve(f, f, mode='full')
    max_b = float(np.max(conv_full))
    return float(2.0 * n * max_b / (s * s))


def _lp_solve(f, n, rows, cols):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns (g, M) or (None, M) on failure.
    """
    data = _fill_data(f, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    conv_f = np.convolve(f, f, mode='full')
    M = float(np.max(conv_f))

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    result = linprog(
        c, A_ub=A, b_ub=b, bounds=bounds,
        method='highs',
        options={"presolve": True, "time_limit": 60}
    )

    if not result.success:
        return None, M

    return result.x, M


def _resample_to_length(h, n):
    """Resample a sequence h to length n via nearest-neighbor."""
    if h.size == n:
        return h.copy()
    idx = (np.arange(n) * h.size / n).astype(int)
    idx = np.clip(idx, 0, h.size - 1)
    return h[idx].copy()
