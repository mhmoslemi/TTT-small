        i_hi = min(k, n - 1)
        if i_hi >= i_lo:
            g[i_lo:i_hi+1] += a[k-i_hi:k-i_lo+1][::-1]
    g /= max(1, len(active))
    return g, M

def run_gradient_descent(a, max_iters=10000, eta0=1.0, eta_decay=0.999, deadline=None):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = fast_score(a)
    eta = eta0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g, M = compute_subgrad(a)
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g
        # Normalize (work in log domain for stability)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        # Check progress
        score = fast_score(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
        
        a = a_new
        eta *= eta_decay
        if eta < 0.001:
            eta = 0.001
    
    return best_a, best_score
