def project_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_objective_and_grad(a, tau):
    """Smooth approximation of max(convolve(a,a)) and its gradient via FFT."""
    n = len(a)
    b = np.convolve(a, a, mode='full')

    b_max = np.max(b)
    b_scaled = (b - b_max) / tau
    exp_b = np.exp(b_scaled)
    sum_exp = np.sum(exp_b)
    obj = b_max + tau * np.log(sum_exp)
    p = exp_b / sum_exp

    # Gradient: grad[i] = (2/tau)*s[i] - (1/tau)*p[2i]*a[i] (if 2i < 2n-1)
    # where s[i] = sum_j p[i+j]*a[j] = scipy.signal.correlate(p, a, mode='valid')[i]
    s = scipy.signal.correlate(p, a, mode='valid')
    grad = (2.0 / tau) * s

    # Diagonal correction: subtract (1/tau)*a[i]*p[2i] for valid 2i
    valid_diag = 2 * np.arange(n) < (2 * n - 1)
    diag_indices = np.arange(n)[valid_diag]
    grad[diag_indices] -= (1.0 / tau) * a[diag_indices] * p[2 * diag_indices]

    return obj, grad

def block_perturbation(f, rng, n_perturb=3, scale=0.08):
    """Apply random multiplicative block perturbations."""
    n = len(f)
    candidate = f.copy()
    block_size = max(1, n // 20)
    for _ in range(rng.integers(1, n_perturb + 1)):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + block_size)
        candidate[start:end] *= np.exp(rng.normal(0, scale))
    candidate = np.maximum(candidate, 0)
    s = np.sum(candidate)
    if s > 0:
        candidate = candidate / s
    return candidate
