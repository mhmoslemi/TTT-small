import numpy as np
from scipy.optimize import linprog
import time

def build_conv_matrix(a):
    """Build the convolution constraint matrix A where (A@g)_k = (a*g)_k."""
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, a[diff.clip(0, n-1)], 0.0)
    return A

def solve_lp_direction(a):
    """Solve LP: max sum(g) s.t. g>=0, conv(a,g) <= max(conv(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    M = float(max(b))
    if M <= 0:
        return a.copy()
    
    A = build_conv_matrix(a)
    b_vec = np.full(A.shape[0], M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 0:
            return g / s
    return a.copy()

    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))

        candidate = np.clip(candidate, 0, 1000)
        s = score_sequence(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
    return best_a, best_score

    max_b = float(np.max(b))

def subgradient_step(a, lr, top_k=1):
