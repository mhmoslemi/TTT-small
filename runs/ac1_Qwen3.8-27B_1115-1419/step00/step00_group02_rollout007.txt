```python
import numpy as np
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog

def lp_update(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix: (2n-1) x n
    # Row k: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    A = A.tocsr()
    
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (all ones times M)
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x
```
