def compute_objective(a):
    """Compute 2n*max(conv)/sum^2 efficiently."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = np.convolve(a, a, mode='full')
    M = np.max(conv)
    return 2.0 * n * M / (S * S)

def compute_gradient(a):
    """Compute subgradient of the objective."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.zeros(n)
    conv = np.convolve(a, a, mode='full')
    M = np.max(conv)
    # Find all positions achieving the max (or near max)
    threshold = M - 1e-10
    active = np.where(conv >= threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        # d(a*a)[k]/a[m] = 2*a[k-m] if 0 <= k-m < n

        valid = (idx >= 0) & (idx < n)

    grad /= len(active)

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # Binary search for the threshold
    v = np.asarray(v, dtype=np.float64)
    # If all non-negative and correct sum, return as-is
    if np.all(v >= 0) and abs(np.sum(v) - target_sum) < 1e-12:

    # Binary search for t such that sum(max(v-t, 0)) = target_sum
    lo = np.min(v) - target_sum

def _obj_grad_topk(a, k=5):
    n = len(a)
    S = float(np.sum(a))
    conv = np.convolve(a, a, 'full')
    # Top-k max
    idx = np.argsort(conv)[::-1][:k]
    topk_val = conv[idx].mean()
    
    # Subgradient: average of shifted a's
    grad_conv = np.zeros(n)
    for ki in idx:
        ii = ki - np.arange(n)
        valid = (ii >= 0) & (ii < n)
        grad_conv[valid] += (2.0 / k) * a[ii[valid]]
    
    obj = 2.0 * n * topk_val / (S * S)
    grad = (2.0 * n / (S * S)) * grad_conv - (4.0 * n * topk_val / (S * S * S))
    return obj, grad

"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using smooth subgradient optimization
on the non-negative simplex with multiple initializations, restarts, and
length exploration. Uses top-k smoothed max for stable gradients and
backtracking line search for convergence.
"""

import numpy as np
import time


def _simplex_project(v, target):
    """Project v onto {x >= 0, sum(x) = target} via bisection."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if np.all(v >= 0) and abs(np.sum(v) - target) < 1e-10 * max(target, 1e-15):
        return v.copy()
    lo = float(np.min(v)) - target
    hi = float(np.max(v))
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    result = np.maximum(v - t, 0.0)
    s = np.sum(result)
    if s > 1e-30:
        result *= target / s
    return result


def _convolve(a):
    """Full convolution of a with itself."""
    return np.convolve(a, a, mode='full')


def _obj_grad_topk(a, k=5):
    """
    Compute objective 2n*topk_mean(conv)/sum^2 and subgradient.
    topk_mean approximates max while being more stable for optimization.
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    conv = _convolve(a)
    # Top-k values for smoothed max
    kk = min(k, len(conv))
    idx = np.argpartition(conv, -kk)[-kk:]
    topk_val = float(np.mean(conv[idx]))

    # Subgradient: average of per-lag gradients
    grad_conv = np.zeros(n)
    for ki in idx:
        ii = ki - np.arange(n)
        valid = (ii >= 0) & (ii < n)
        grad_conv[valid] += (2.0 / kk) * a[ii[valid]]

    obj = 2.0 * n * topk_val / (S * S)
    grad = (2.0 * n / (S * S)) * grad_conv - (4.0 * n * topk_val / (S * S * S))
    return obj, grad


def _obj_grad_smooth(a, beta=40.0):
    """Compute objective using log-sum-exp smoothed max and its gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    conv = _convolve(a)
    max_c = float(np.max(conv))
    shifted = beta * (conv - max_c)
    exp_s = np.exp(shifted)
    logsum = max_c + np.log(exp_s.sum()) / beta

    # Weights (softmax)
    w = exp_s / exp_s.sum()

    # Only use lags with significant weight
    significant = w > 1e-8
    n_sig = int(np.sum(significant))
    if n_sig == 0:
        sig_idx = np.array([int(np.argmax(conv))])
    else:
        sig_idx = np.where(significant)[0]

    grad_conv = np.zeros(n)
    for ki in sig_idx:
        wk = w[ki]
        ii = ki - np.arange(n)
        valid = (ii >= 0) & (ii < n)
        grad_conv[valid] += wk * 2.0 * a[ii[valid]]

    obj = 2.0 * n * logsum / (S * S)
    grad = (2.0 * n / (S * S)) * grad_conv - (4.0 * n * logsum / (S * S * S))
    return obj, grad


def _optimize(a0, time_limit=60.0, method='topk', k=5, beta=40.0,
              max_iter=50000, verbose=False, seed=0):
    """
    Optimize using projected (sub)gradient with backtracking line search.
    Preserves the sum of the sequence.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    target = float(np.sum(a))

        a = np.ones(n)

    else:
        a *= target / max(np.sum(a), 1e-30)

    else:

    obj, grad = obj_fn(a)

    stall = 0

    it = 0

            if verbose and it % 5000 == 0:

    return best_a, best_obj


def _make_hanning(n):
    x = np.arange(n)
    return np.sin(np.pi * x / max(n - 1, 1)) ** 2


def _make_gaussian(n):
    x = np.arange(n) - n / 2
    sigma = n / 4
    return np.exp(-0.5 * (x / sigma) ** 2)


def _make_triangle(n):
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(x - mid) / (mid + 1e-10))


def _make_arch(n):
    """High at edges, low in middle."""
    x = np.arange(n)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(n - 1, 1)))


def _make_parabola(n):
    x = np.arange(n, dtype=np.float64)
    return x * (n - 1 - x)


def _resample_seq(a, new_n):
    """Resample sequence a to new length new_n using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)

            a_res = np.maximum(a_res, 0)
            if np.sum(a_res) < 1e-10:
                continue
            time_left = deadline - time.monotonic()
            if time_left > 15:
                a_r, _ = _optimize(
                    a_res, time_limit=max(10.0, time_left * 0.4),
                    method='topk', k=5, seed=8000 + new_n)
                try_sequence(a_r, f"resample_{new_n}")

    # Final fallback
    if best_seq is None or best_score >= 2.0:
        # Use uniform as safety net
        best_seq = np.ones(256)
        best_score = 2.0
        print("  WARNING: Falling back to uniform (score=2.0)")

    print(f"\nFinal score: {best_score:.8f}, sequence length: {len(best_seq)}")
    result = [float(x) for x in best_seq]

    # Final verification
    try:
        final_score = float(evaluate_sequence(result))
        print(f"Verified final score: {final_score:.8f}")
        if not np.isfinite(final_score):
            result = [1.0] * 256
            print("  Verification failed, using uniform fallback")
    except Exception:
        result = [1.0] * 256

    return result
