```python
import numpy as np
import cvxpy as cp
import scipy.sparse as sp
import time

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T

def lp_iteration(a, M, solver='CLARABEL'):
    """Solve the LP: max sum(g) s.t. g>=0, convolve(a,g) <= M."""
    n = len(a)
    T = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    if g.value is None:
        return None
    return g.value

def local_search(a, budget_s, rng):
    """Local search with block perturbations."""
    n = len(a)
    best = a.copy()
    best_score = 2 * n * max(np.convolve(best, best)) / (np.sum(best)**2)
    deadline = time.monotonic() + budget_s
    
    while time.monotonic() < deadline:
        # Block perturbation
        candidate = best.copy()
        width = max(1, n // 50)
        start = rng.integers(0, n - width)
        candidate[start:start+width] *= np.exp(rng.normal(0, 0.1))
        candidate = np.clip(candidate, 0, None)
        if np.sum(candidate) < 0.01:
            continue
        score = 2 * n * max(np.convolve(candidate, candidate)) / (np.sum(candidate)**2)
        if score < best_score:
            best = candidate
            best_score = score
    
    return best, best_score
```
