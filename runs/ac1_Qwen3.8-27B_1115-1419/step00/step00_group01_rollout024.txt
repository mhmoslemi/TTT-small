        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def find_peak_lag(a):
    """Find the lag where autocorrelation is maximized."""
    b = np.convolve(a, a)
    k = np.argmax(b)
    # Convert to offset from center
    n = len(a)
    offset = k - (n - 1)
    return offset, b[k]

def targeted_reduction(a, k, eta):
    """Reduce autocorrelation at lag k (offset from center)."""
    n = len(a)
    a_new = a.copy()
    if k >= 0:
        a_new[:-k] -= eta * a[k:]
        a_new[k:] -= eta * a[:-k]
    else:
        k = -k
        a_new[k:] -= eta * a[:-k]
        a_new[:-k] -= eta * a[k:]
    return np.clip(a_new, 0, 1000)

def random_perturbation(a, eta, rng):
    """Apply random perturbation."""
    n = len(a)
    # Block perturbation
    width = max(1, int(rng.integers(1, max(2, n // 50))))
    start = int(rng.integers(0, n - width))
    a_new = a.copy()
    a_new[start:start+width] *= np.exp(rng.normal(0, eta, size=width))
    return np.clip(a_new, 0, 1000)

def generate_periodic(n, period, rng):
    """Generate a periodic-like sequence."""
    base = np.ones(period)
    base[rng.choice(period, size=period//3, replace=False)] = 0.5
    seq = np.tile(base, n // period + 1)[:n]
    return seq

def generate_random(n, rng):
    """Generate a random non-negative sequence."""
    return rng.uniform(0, 1, size=n)

def generate_block(n, num_blocks, rng):
    """Generate a block-structured sequence."""
    block_size = n // num_blocks
    blocks = rng.uniform(0.1, 1.0, size=num_blocks)
    seq = np.repeat(blocks, block_size)[:n]
    return seq
