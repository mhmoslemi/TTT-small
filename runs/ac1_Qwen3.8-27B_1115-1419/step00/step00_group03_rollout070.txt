"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using smoothed multiplicative updates
with optional LP corrections. Multiple initializations and sequence lengths are tried.
The multiplicative update iteratively flattens the convolution by reducing coordinates
that contribute most to peak convolution values.
"""

import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import linprog


def _obj(f):
    """Compute the objective 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 1e-10:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _pressure(f, w):
    """
    Compute pressure[i] = sum_j f[j] * w[i+j] for valid indices.
    f has length n, w has length 2n-1.
    Uses FFT convolution for O(n log n) complexity.
    """
    n = len(f)
    w_rev = w[::-1]
    conv_full = fftconvolve(f, w_rev, mode='full')
    # pressure[i] = conv_full[2n-2-i] for i = 0..n-1
    pressure = conv_full[n - 1:2 * n - 1][::-1]
    return pressure


def _multi_step(f, lr, beta):
    """One smoothed multiplicative update step to flatten the convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    max_conv = conv.max()
    if max_conv <= 1e-30:
        return f

    # Soft-max weights emphasizing convolution peaks
    scaled = beta * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum

    # Compute pressure on each coordinate
    pressure = _pressure(f, w)

    # Multiplicative update: reduce high-pressure coordinates
    max_p = pressure.max()
    if max_p <= 1e-30:
        return f

    log_factor = -lr * pressure / max_p
    log_factor = np.maximum(log_factor, -50.0)
    f_new = f * np.exp(log_factor)

    # Preserve sum (scale invariance of objective)
    s = f_new.sum()
    if s > 1e-30:
        f_new *= f.sum() / s

    f_new = np.clip(f_new, 1e-12, 1000.0)
    return f_new


def _lp_correction(f, max_time=5.0):
    """
    LP-based correction: find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend f with the LP solution. For n <= 3000.

    if n > 3000:
        return f

    t0 = time.monotonic()
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    if M <= 1e-30:
        return f

    # Build constraint matrix: A[k,i] = f[i] if 0 <= k-i < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        A[k, i_lo:i_hi + 1] = f[i_lo:i_hi + 1]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return f

    if time.monotonic() - t0 > max_time:
        return f

    if res.success and res.x is not None:
        g = res.x
        sum_g = g.sum()
        if sum_g > 1e-10:
            g *= f.sum() / sum_g
            t_blend = 0.4
            f_new = (1.0 - t_blend) * f + t_blend * g
            f_new = np.clip(f_new, 1e-12, 1000.0)
            return f_new
    return f
