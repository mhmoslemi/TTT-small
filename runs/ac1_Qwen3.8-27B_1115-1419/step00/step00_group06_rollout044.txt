import numpy as np
import time

def _next_pow2(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.max() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _autocorr_fft(p, N):
    """Compute the full autocorrelation of p using FFT."""
    P = np.fft.rfft(p, n=N)
    C = np.fft.irfft(P * P.conjugate(), n=N)
    return C[:2 * len(p) - 1]

def _softmax_gradient(p, w, N):
    """Compute 2 * cross_correlation(w, p) using FFT."""
    W = np.fft.rfft(w, n=N)
    P = np.fft.rfft(p, n=N)
    CC = np.fft.irfft(W * P.conjugate(), n=N)
    return 2.0 * CC[:len(p)]

        return rng.dirichlet(np.ones(n) * 0.5)
    elif kind == 'given':
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0, 1000)
        if base.sum() < 0.01:
            return np.ones(n) / n
        return base / base.sum()
    else:
        return np.ones(n) / n

def _gradient_descent(p, n, N, deadline, seed, T_init=50.0, alpha_init=0.05, momentum=0.9):
    """Run softmax projected gradient descent. Returns best p found."""
    rng = np.random.default_rng(seed)
    p = p.copy()
    best_p = p.copy()
    best_score = _evaluate_safe(p)
    
    T = T_init
    alpha = alpha_init
    v = np.zeros(n)
    iteration = 0
    last_improve = 0
    stagnation = 0
    
    while time.monotonic() < deadline:
        # Compute autocorrelation
        c = _autocorr_fft(p, N)
        
        # Softmax weights
        c_shifted = c - c.max()
        exp_c = np.exp(T * c_shifted)
        w = exp_c / exp_c.sum()
        
        # Gradient
        grad = _softmax_gradient(p, w, N)
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        p = p - alpha * v
        
        # Project onto simplex
        p = _project_simplex(p)
        
        iteration += 1
        
        # Evaluate periodically
        if iteration % 200 == 0:
            score = _evaluate_safe(p)
            if score < best_score:
                best_score = score
                best_p = p.copy()
                last_improve = iteration
                stagnation = 0
                if iteration % 2000 == 0:
                    print(f"  GD iter {iteration}: score={score:.6f}, T={T:.1f}, alpha={alpha:.5f}")
            else:
                stagnation += 1
            
            # Adapt parameters
            T = max(0.5, T * 0.99)
            alpha *= 0.999
            
            # Restart if stagnant
            if stagnation > 20:  # 4000 iterations without improvement
                # Try a random perturbation
                noise = rng.normal(0, 0.1, n)
                p = best_p + 0.05 * noise
                p = _project_simplex(p)
                v = np.zeros(n)
                T = max(T, 10.0)
                alpha = max(alpha, 0.01)
                stagnation = 0
    
    # Final check
    score = _evaluate_safe(p)
    if score < best_score:
        best_p = p.copy()
    
    return best_p, best_score

def _local_refinement(p, n, deadline):
    """Coordinate-wise local refinement."""
    best_p = p.copy()
    best_score = _evaluate_safe(p)
    N = _next_pow2(3 * n)
    
    for i in range(len(p)):
        if time.monotonic() > deadline:
            break
        
        # Try increasing and decreasing p[i]
        for factor in [0.7, 0.85, 1.15, 1.3]:
            p_trial = best_p.copy()
            p_trial[i] *= factor
            p_trial = np.maximum(p_trial, 0)
            if p_trial.sum() < 1e-10:
                continue
            p_trial = p_trial / p_trial.sum()
            score = _evaluate_safe(p_trial)
            if score < best_score:
                best_score = score
                best_p = p_trial.copy()
    
    return best_p, best_score

Using FFT:
- Zero-pad p to length N: p_pad = [p[0], ..., p[n-1], 0, ..., 0] (length N)
- Zero-pad w to length N: w_pad = [w[0], ..., w[2n-2], 0, ..., 0] (length N)
- Compute: R_fft = IFFT(FFT(w_pad) * conj(FFT(p_pad)))

This gives the circular cross-correlation: R_fft[j] = sum_{k=0}^{N-1} w_pad[k] * p_pad[(k-j) mod N]

For this to equal the linear cross-correlation, we need no wrap-around, i.e., for all k where w_pad[k] ≠ 0 (i.e., k < 2n-1) and all j where we care (j < n), we need (k-j) mod N to be in [0, n-1] or the product to be 0.

(k-j) mod N: if k-j >= 0 and k-j < n, then (k-j) mod N = k-j (valid p index).
