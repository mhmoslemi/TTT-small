"""
Smooth gradient descent with FFT-based gradient for minimizing autocorrelation peak.
Uses softmax smoothing of the max, simplex projection, beta continuation,
and random restarts to escape local minima.
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(f, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c[k])) where c = convolve(f,f).
    The gradient is: grad[i] = 2 * sum_k w[k] * f[k-i] (for valid indices).
    Computed in O(n log n) via FFT.
    """
    n = len(f)
    c = np.convolve(f, f)
    c_max = c.max()
    exp_c = np.exp(np.clip(beta * (c - c_max), -500.0, 0.0))
    s = exp_c.sum()
    if s < 1e-300:
        # Degenerate case: all mass on argmax
        k_star = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[k_star] = 1.0
    else:
        w = exp_c / s

    w_rev = w[::-1]
    # Linear convolution of f (len n) and w_rev (len 2n-1): result len 3n-2
    N = 4 * n  # Always >= 3n-2 and even
    F = np.fft.rfft(f, N)
    W = np.fft.rfft(w_rev, N)
    conv_full = np.fft.irfft(F * W, N)
    # grad[i] = 2 * conv_full[2n-2-i] for i=0..n-1
    grad = 2.0 * conv_full[2 * n - 2 : n - 2 : -1]
    return grad


def quick_objective(f):
    """Fast computation of 2*n*M/S^2 without type checking."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(f, f)
    M = c.max()
    return 2.0 * n * M / (S * S)


def smooth_descent(f, n_steps, beta_start, beta_end, alpha_init, eval_interval, deadline, rng):
    """
    Run smooth gradient descent with beta continuation.
    Returns best sequence found (normalized to sum=1).
    """
    n = len(f)
    best_f = f.copy()
    best_obj = quick_objective(best_f)

    beta = beta_start
    alpha = alpha_init
    no_improve = 0
    total_steps = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        grad = compute_gradient_fft(f, beta)
        f_new = f - alpha * grad
        f_new = project_simplex(f_new)
        f = f_new
        alpha *= 0.9999
        total_steps += 1

        if step % eval_interval == 0:
            obj = quick_objective(f)
            if obj < best_obj:
                best_f = f.copy()
                best_obj = obj
                no_improve = 0
            else:
                no_improve += 1

            # Beta continuation: gradually increase beta
            progress = step / max(1, n_steps)
            beta = beta_start * (beta_end / beta_start) ** progress

            # Restart if stuck
            if no_improve > 5:
                # Add perturbation and reduce alpha
                f = best_f + 0.003 * rng.standard_normal(n)
                f = project_simplex(f)
                alpha = 0.3 / max(beta, 1.0)
                no_improve = 0

    return best_f, best_obj


def lp_refine(f, M, max_iters=50, deadline=np.inf):
    """
    LP-based refinement: maximize sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Uses scipy linprog. Only feasible for small n due to O(n^2) constraints.
    """
    from scipy.optimize import linprog
    n = len(f)
    if n > 2000:
        return f  # Too large for LP

    # A matrix: A[i, j] = f[i-j] if 0<=i-j<n, else 0 (for row i of f*g)
    # f*g has length 2n-1
    c_len = 2 * n - 1
    A = np.zeros((c_len, n))
    for j in range(n):
        A[j:j+n, j] = f  # A[i,j] = f[i-j] for valid i-j

    c_vec = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
    bounds = [(0.0, None)] * n

    res = linprog(c_vec, A_ub=A, b_ub=np.full(c_len, M), bounds=bounds, method='highs')
    if res.status != 0:
        return f

    g = res.x
    S_g = g.sum()
    if S_g < 1e-10:
        return f

    # Rescale g to match sum of f
    S_f = f.sum()
    g = (S_f / S_g) * g

    # Blend
    t = 0.1
    f_new = (1 - t) * f + t * g
    return f_new
