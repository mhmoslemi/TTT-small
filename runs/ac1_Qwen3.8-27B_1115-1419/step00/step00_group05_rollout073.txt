def upsample_array(a,n):
    a=np.asarray(a,float).ravel()
    if a.size ==0 or n<=0: return np.ones(n)
    if a.size == n: return a
    if a.sum() <=0: return np.ones(n)
    old=np.linspace(0,1,a.size)

    b=np.interp(new, old, a)

def qp_direction(a,M,bias,mu,eps,dual_iters,dual_step,N,lambda0):
    n=int(a.size); L=2*n-1
    a=np.asarray(a,float)
    bias=np.asarray(bias,float)

       bias=upsample_array(bias,n)
    bias=normalize_array(bias)

    if lambda0 is None:
       lam=np.zeros(L)
    else:
       lam=np.asarray(lambda0,float).ravel()
       if lam.size != L:
           lam=np.zeros(L)
       lam=np.maximum(lam,0.0)
    fa= np.fft.rfft(a,N)
    g=None

       prefix=np.empty(L+1)
       prefix[0]=0.0
       prefix[1:]=np.cumsum(lam)
       wsum=prefix[n:n+n] - prefix[:n]
       t=c - a*wsum
       g=np.clip(t/(2.0*mu), 0.0, None)
       fg=np.fft.irfft(fa*np.fft.rfft(g,N), N)[:L]
       res=fg - M
       # if res all nan? break
       if not np.all(np.isfinite(res)):
           break
       lam=np.maximum(lam + float(dual_step)*res, 0.0)
    if g is None:
       return None, lam
    s=float(np.sum(g))
    if s <= 1e-300 or not np.isfinite(s):
       return None, lam
    return g/s, lam

import numpy as np
import time
import math

def next_fft_len(length):
    if length <= 1:
        return 1
    return 1 << (length - 1).bit_length()

def normalize_array(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return np.ones(1, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not math.isfinite(s):
        a = np.ones(a.size, dtype=float)
        s = float(a.sum())
    a = a / s
    a = np.maximum(a, 1e-300)
    a = a / float(a.sum())
    return a

def upsample_array(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if n <= 0:
        return np.ones(0, dtype=float)
    if a.size == 0 or float(a.sum()) <= 0.0:
        return np.ones(n, dtype=float)
    if a.size == n:
        return a
    old = np.linspace(0.0, 1.0, a.size)
    new = np.linspace(0.0, 1.0, n)
    return np.interp(new, old, a)

def conv_scaled_with_len(a, N, n):
    if n == 1:
        return np.array([float(a[0]) * float(a[0])], dtype=float)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c * float(n)

def objective_from_scaled_with_len(a, N, n):
    u = conv_scaled_with_len(a, N, n)
    return float(2.0 * np.max(u))

def objective_from_array(a):
    a = normalize_array(a)
    n = a.size
    if n == 0:
        return math.inf
    return objective_from_scaled_with_len(a, next_fft_len(2 * n - 1), n)

def softmax_weights(u, T):
    u = np.asarray(u, dtype=float)
    m = float(np.max(u))
    x = np.exp(-float(T) * (u - m))
    s = float(x.sum())
    if not math.isfinite(s) or s <= 1e-300:
        p = np.zeros_like(u)
        p[int(np.argmax(u))] = 1.0
        return p
    return x / s

def hard_top_weights(u, K):
    u = np.asarray(u, dtype=float)
    L = u.size
    K = int(max(1, min(K, L)))
    w = np.zeros(L, dtype=float)
    if K == L:
        w.fill(1.0 / L)
        return w
    idx = np.argpartition(u, K - 1)[:K]
    w[idx] = 1.0 / K
    return w

def correlation_from_weights_with_len(w, a, N, n):
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(np.conj(W) * A, N)[:n]
    return R

def candidate_from_direction(a, direction, step_scale, n):
    step = float(step_scale) * float(n) * direction
    np.clip(step, -10.0, 10.0, out=step)
    b = a * np.exp(-step)
    s = float(b.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return None
    return b / s

def select_bias_vector(a, pool, preferred, rng):
    n = a.size
    choices = []
    if pool is not None:
        for p in pool:
            if np.size(p) == n:
                choices.append(normalize_array(p))
    if preferred is not None and np.size(preferred) == n:
        choices.append(normalize_array(preferred))

    x = np.linspace(0.0, 1.0, n)
    left = normalize_array(np.exp(-2.0 * x) + 0.05)
    right = normalize_array(np.exp(-2.0 * (1.0 - x)) + 0.05)

    if not choices:
        return left

    for _ in range(min(8, len(choices))):
        b = choices[int(rng.integers(0, len(choices)))]
        if float(np.dot(a, b)) < 0.97:
            return b

    if float(np.dot(a, left)) < 0.97:
        return left
    if float(np.dot(a, right)) < 0.97:
        return right
    return normalize_array(np.abs(rng.normal(0.0, 1.0, n)) + 0.1)

def qp_direction(a, M, bias, mu, eps, dual_iters, dual_step, N, lambda0):
    n = int(a.size)
    L = 2 * n - 1
    a = np.asarray(a, dtype=float)
    bias = np.asarray(bias, dtype=float)
    if bias.size != n:
        bias = upsample_array(bias, n)
    bias = normalize_array(bias)

    c = 1.0 + float(eps) * bias

    if lambda0 is None:
        lam = np.zeros(L, dtype=float)
    else:
        lam = np.asarray(lambda0, dtype=float).ravel()
        if lam.size != L:
            lam = np.zeros(L, dtype=float)
        lam = np.maximum(lam, 0.0)

    fa = np.fft.rfft(a, N)
    g = None

    for _ in range(max(1, int(dual_iters))):
        prefix = np.empty(L + 1, dtype=float)
        prefix[0] = 0.0
        prefix[1:] = np.cumsum(lam)
        wsum = prefix[n:n + n] - prefix[:n]

        t = c - a * wsum
        g = np.clip(t / (2.0 * float(mu)), 0.0, None)

        fg = np.fft.irfft(fa * np.fft.rfft(g, N), N)[:L]
        res = fg - float(M)
        if not np.all(np.isfinite(res)):
            break
        lam = np.maximum(lam + float(dual_step) * res, 0.0)

    if g is None:
        return None, lam
    s = float(g.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return None, lam
    return g / s, lam

def generate_seed_arrays(n, rng, preferred=None):
    if n <= 0:
        return [np.ones(0, dtype=float)]
    out = []

    if preferred is not None and np.size(preferred) > 0:
        b = upsample_array(preferred, n)
        if b.size == n:
            out.append(normalize_array(b))

    out.append(np.ones(n, dtype=float))

    if n == 1:
        return [normalize_array(np.array([1.0]))]

    j = np.arange(n, dtype=float)
    x = j / float(n - 1)
    eps = max(1.0 / float(n), 1e-6)

    for p in (0.25, 0.5, 1.0, 2.0, 4.0):
        out.append(np.exp(-p * x))
        out.append(np.exp(-p * (1.0 - x)))

    for q in (0.25, 0.5, 1.0, 2.0):
        out.append((x + eps) ** q)
        out.append((1.0 - x + eps) ** q)

    for q in (-0.25, -0.5):
        out.append((x + eps) ** q)
        out.append((1.0 - x + eps) ** q)

    out.append((x + eps) ** 0.25 * (1.0 - x + eps) ** -0.5)
    out.append((1.0 - x + eps) ** 0.25 * (x + eps) ** -0.5)

    cos_edge = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    cos_mid = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    out.append(cos_edge)
    out.append(cos_mid)

    for _ in range(5):
        out.append(np.abs(rng.normal(0.0, 1.0, n)) + 0.1)

    for _ in range(3):
        p1 = rng.uniform(-0.6, 0.6)
        p2 = rng.uniform(-0.8, 0.6)
        b = (x + eps) ** p1 * (1.0 - x + eps) ** p2
        out.append(b)

    if n > 1:
        pat1 = np.empty(n, dtype=float)
        pat1[::2] = 1.0
        pat1[1::2] = 2.0
        out.append(pat1)

        pat2 = np.empty(n, dtype=float)
        pat2[::2] = 2.0
        pat2[1::2] = 1.0
        out.append(pat2)

    res = []
    for a in out:
        if a is None or np.size(a) != n:
            continue
        b = normalize_array(a)
        if np.all(np.isfinite(b)) and float(b.sum()) > 0.0:
            res.append(b)

    if not res:
        res.append(np.ones(n, dtype=float) / float(n))
    return res

    return best, best_obj

def search_length(n, time_budget, preferred, rng):
    if n <= 0 or time_budget <= 0.0:
        return None, math.inf

    seeds = generate_seed_arrays(n, rng, preferred)
    best_seed = seeds[0]
    best_seed_obj = math.inf
    for s in seeds:
        ob = objective_from_array(s)
        if ob < best_seed_obj:
            best_seed_obj = ob
            best_seed = s

    print(f"search_length n={n} best_seed_obj={best_seed_obj:.6f}")
    return local_search(best_seed, time_budget, rng, restart_pool=seeds, preferred=preferred)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = max(60.0, float(budget_s))
    deadline = time.perf_counter() + (budget_s - 25.0)

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=float).ravel()
        parent = normalize_array(parent)
    except Exception:
        parent = np.ones(64, dtype=float)

    best_arr = None
    best_score = math.inf

    if parent is not None and parent.size > 0:
        pobj = objective_from_array(parent)
        if np.isfinite(pobj):
            best_arr = parent.copy()
            best_score = pobj
        print(f"parent score={parent.size and best_score: .6f}")

    # Short preliminary search.  This is useful both as a direct candidate and
    # as a smoothed starting point for longer searches.
    short_n = 1024
    short_time = min(70.0, max(20.0, budget_s * 0.03))
    remaining = deadline - time.perf_counter()
    if remaining > short_time + 120.0:
        arr, sc = search_length(short_n, short_time, best_arr, rng)
        if arr is not None and np.isfinite(sc):
            if sc < best_score:
                best_arr = arr.copy()
                best_score = sc
            print(f"after short n={short_n} best={best_score:.6f}")

    # Main search at a larger length.  4096 is long enough to allow a very flat
    # self-convolution while still allowing many FFT-based iterations.
    main_n = 4096
    reserve = 90.0 if budget_s > 1800.0 else 0.0
    remaining = deadline - time.perf_counter()
    main_time = max(120.0, remaining - reserve - 20.0)
    if remaining > main_time + 10.0:
        arr, sc = search_length(main_n, main_time, best_arr, rng)
        if arr is not None and np.isfinite(sc):
            if sc < best_score:
                best_arr = arr.copy()
                best_score = sc
            print(f"after main n={main_n} best={best_score:.6f}")

    # Optional extension to 8192 when the main search has already gotten close
    # to the target and some time remains.
    remaining = deadline - time.perf_counter()
    if remaining > 110.0 and best_score > 1.52:
        long_n = 8192
        long_time = min(remaining - 25.0, 160.0)
        arr, sc = search_length(long_n, long_time, best_arr, rng)
        if arr is not None and np.isfinite(sc):
            if sc < best_score:
                best_arr = arr.copy()
                best_score = sc
            print(f"after long n={long_n} best={best_score:.6f}")

    if best_arr is None:
        best_arr = np.ones(1024, dtype=float) / 1024.0

    out = normalize_array(best_arr)

    # Final sanity fallback.  The immediate objective is internal; avoid another
    # full direct convolution if the clock is close.
    if time.perf_counter() < deadline - 5.0 and out.size <= 4096:
        try:
            real_score = evaluate_sequence([float(x) for x in out])
            print(f"final objective check={float(real_score):.6f}")
            if not np.isfinite(real_score):
                out = normalize_array(np.ones(1024, dtype=float))
        except Exception:
            pass

    out = np.clip(np.asarray(out, dtype=float), 0.0, 1000.0)
    if float(out.sum()) < 0.01:
        out = out * (0.01 / max(float(out.sum()), 1e-300))
    return [float(x) for x in out]
