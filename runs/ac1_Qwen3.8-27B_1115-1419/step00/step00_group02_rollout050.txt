def solve_lp(f, M):
    """Solve: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    # Build sparse constraint matrix
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # A is (2n-1) x n
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[j])

    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix for f."""
    n = len(f)
    # For each k, the valid j range is [max(0, k-n+1), min(n-1, k)]
    k_idx = np.arange(2*n - 1)
    j_min = np.maximum(0, k_idx - n + 1)
    j_max = np.minimum(n - 1, k_idx)
    
    # Count non-zeros per row
    counts = j_max - j_min + 1
    
    # Build row pointers
    row_ptr = np.zeros(2*n, dtype=np.int32)
    row_ptr[1:] = np.cumsum(counts)
    
    # Build column indices
    total_nz = int(row_ptr[-1])
    col_idx = np.zeros(total_nz, dtype=np.int32)
    for k in range(2*n - 1):
        start = row_ptr[k]
        end = row_ptr[k+1]
        col_idx[start:end] = np.arange(j_min[k], j_max[k]+1)
    
    # Build data
    data = np.zeros(total_nz)
    for k in range(2*n - 1):
        start = row_ptr[k]
        end = row_ptr[k+1]
        data[start:end] = f[col_idx[start:end]]
    
    return csr_matrix((data, col_idx, row_ptr), shape=(2*n-1, n))

import numpy as np
import time
import scipy.sparse
import scipy.optimize

def stable_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def project_simplex(v):
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def obj_from_normalized(f):
    """Compute objective for f with sum=1."""
    n = len(f)
    b = stable_convolve(f, f)
    return 2.0 * n * np.max(b)

def softmax_gd(f0, n_iters, beta_start, beta_end, eta_base, seed=0, noise_interval=100, noise_scale=0.01):
    """Softmax-approximated gradient descent on the simplex."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.maximum(f0, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = obj_from_normalized(f)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    no_improve = 0
    for it in range(n_iters):
        t_frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac
        
        # Compute b = f * f using FFT
        fa = np.fft.rfft(f, fft_size)
        b = np.fft.irfft(fa * fa, fft_size)[:2*n-1]
        
        # Softmax weights
        b_max = np.max(b)
        exp_b = np.exp(beta * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: 2 * conv(p, f)
        fp = np.fft.rfft(p, fft_size)
        grad = 2.0 * np.fft.irfft(fp * fa, fft_size)[:n]
        
        # Gradient descent with backtracking line search
        f_new = f.copy()
        f_new_obj = best_obj
        eta = eta_base / max(beta, 1.0)
        improved = False
        for _ in range(10):
            trial = f - eta * grad
            trial = project_simplex(trial)
            # Compute objective
            fa_trial = np.fft.rfft(trial, fft_size)
            b_trial = np.fft.irfft(fa_trial * fa_trial, fft_size)[:2*n-1]
            trial_obj = 2.0 * n * np.max(b_trial)
            if trial_obj < f_new_obj:
                f_new = trial
                f_new_obj = trial_obj
                improved = True
            eta *= 0.5
        
        if improved:
            f = f_new
            if f_new_obj < best_obj:
                best_f = f.copy()
                best_obj = f_new_obj
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Add noise periodically
        if noise_interval > 0 and it % noise_interval == 0 and t_frac > 0.2:
            noise = rng.normal(0, noise_scale, n)
            f = np.maximum(f + noise, 0)
            f = f / f.sum()
        
        # Early stopping if stuck
        if no_improve > 500:
            break
    
    return best_f, best_obj

def lp_refine(f, M, n_max_iter=20, t_step=0.5):
    """LP-based refinement (Frank-Wolfe style)."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = obj_from_normalized(f)
    
    for i in range(n_max_iter):
        M = np.max(stable_convolve(f, f))
        
        # Build LP
        # A @ g <= M * ones, g >= 0, max sum(g)
        # A[k, j] = f[j] if 0 <= k-j < n
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                rows.extend([k] * (j_hi - j_lo + 1))
                cols.extend(range(j_lo, j_hi + 1))
                data.extend(f[j_lo:j_hi+1].tolist())
        
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        
        result = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        
        if not result.success or result.x is None:
            break
        
        g = result.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Frank-Wolfe step
        t = t_step / (i + 1)  # or 2/(i+2)
        f_new = (1 - t) * f + t * g
        f_new = project_simplex(f_new)
        
        f_new_obj = obj_from_normalized(f_new)
        if f_new_obj < best_obj:
            best_f = f_new.copy()
            best_obj = f_new_obj
        f = f_new
    
    return best_f, best_obj

"""
Minimize 2*n*max(conv(f,f))/(sum(f))^2 using softmax-approximated gradient descent
on the simplex, with multiple random restarts and LP-based polishing.
The softmax approximates the non-smooth max, enabling efficient FFT-based iterations.
"""

import numpy as np
import time
import scipy.sparse
import scipy.optimize


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def fft_convolve_self(f, fft_size):
    """Self-convolution f*f using precomputed FFT size."""
    fa = np.fft.rfft(f, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:, :2 * len(f) - 1] if len(f) > 0 else np.array([])


def compute_self_conv(f, fft_size):
    """Compute f*f using given FFT size."""
    fa = np.fft.rfft(f, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:2 * len(f) - 1]


def compute_cross_conv(p, f, fft_size):
    """Compute conv(p, f) where p has length 2n-1 and f has length n."""
    fp = np.fft.rfft(p, fft_size)
    ff = np.fft.rfft(f, fft_size)
    return np.fft.irfft(fp * ff, fft_size)[:len(f)]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def normalized_objective(f):
    """Compute 2*n*max(f*f) for f with sum=1."""
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()
    b = compute_self_conv(f, fft_size)
    return 2.0 * n * np.max(b)


def softmax_gd(f0, n_iters, beta_start, beta_end, eta_base, seed=0,
               noise_every=200, noise_scale=0.02, patience=800):
    """
    Softmax-approximated gradient descent on the simplex.
    Minimizes max(f*f) subject to f>=0, sum(f)=1.
    """
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.maximum(f0, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    fft_size = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_obj = normalized_objective(f)
    current_obj = best_obj
    no_improve = 0

    for it in range(n_iters):
        t_frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # Compute b = f*f
        fa = np.fft.rfft(f, fft_size)
        b = np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]

        # Softmax weights p_k proportional to exp(beta * b_k)
        b_max = np.max(b)
        exp_b = np.exp(beta * (b - b_max))
        p_sum = exp_b.sum()
        if p_sum < 1e-300:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p = exp_b / p_sum

        # Gradient: 2 * conv(p, f)
        fp = np.fft.rfft(p, fft_size)
        grad = 2.0 * np.fft.irfft(fp * fa, fft_size)[:n]

        # Gradient descent step with backtracking
        eta = eta_base / (1.0 + beta * 0.01)
        f_new = f.copy()
        f_new_obj = current_obj
        improved = False

        for _ in range(6):
            trial = f - eta * grad
            trial = project_simplex(trial)
            fa_t = np.fft.rfft(trial, fft_size)
            b_t = np.fft.irfft(fa_t * fa_t, fft_size)[:2 * n - 1]
            trial_obj = 2.0 * n * np.max(b_t)
            if trial_obj < f_new_obj:
                f_new = trial
                f_new_obj = trial_obj
                improved = True
            eta *= 0.5

        if improved:
            f = f_new
            current_obj = f_new_obj
            if f_new_obj < best_obj:
                best_f = f.copy()
                best_obj = f_new_obj
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Periodic noise to escape local minima
        if noise_every > 0 and it % noise_every == 0 and t_frac > 0.15 and t_frac < 0.95:
            scale = noise_scale * (1.0 - t_frac * 0.5)
            noise = rng.normal(0, scale, n)
            f = np.maximum(f + noise, 0)
            f = f / f.sum()

        # If stuck for a long time, add larger perturbation
        if no_improve > patience:
            noise = rng.normal(0, 0.05, n)
            f = np.maximum(f + noise, 0)
            f = f / f.sum()
            no_improve = 0

    return best_f, best_obj


def lp_refine(f_init, max_lp_iters=15, timeout_s=180):
    """
    Frank-Wolfe style LP refinement.
    At each step: solve LP max sum(g) s.t. (f*g)_k <= M, g>=0.
    Update f = (1-t)f + t*g.
    """
    start_t = time.monotonic()
    n = len(f_init)
    f = np.maximum(f_init, 0).copy()
    f = f / f.sum()
    fft_size = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_obj = normalized_objective(f)

    for i in range(max_lp_iters):
        if time.monotonic() - start_t > timeout_s:
            break

        # Compute current peak
        fa = np.fft.rfft(f, fft_size)
        b = np.fft.irfft(fa * fa, fft_size)[:2 * n - 1]
        M = np.max(b)
        if M < 1e-15:
            break

        # Build sparse constraint matrix A where A[k,j] = f[j] if 0<=k-j<n
        # Use efficient construction
        row_list = []
        col_list = []
        data_list = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            cnt = j_hi - j_lo + 1
            row_list.extend([k] * cnt)
            col_list.extend(range(j_lo, j_hi + 1))
            data_list.extend(f[j_lo:j_hi + 1].tolist())

        A = scipy.sparse.csr_matrix((data_list, (row_list, col_list)), shape=(2 * n - 1, n))

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        result = scipy.optimize.linprog(
            c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
            options={'time_limit': max(1, int(timeout_s - (time.monotonic() - start_t)))}
        )

        if not result.success or result.x is None:
            break

        g = result.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Frank-Wolfe step size
        t_step = 2.0 / (i + 2)
        f_new = (1.0 - t_step) * f + t_step * g
        f_new = project_simplex(f_new)

        # Evaluate
        fa_new = np.fft.rfft(f_new, fft_size)
        b_new = np.fft.irfft(fa_new * fa_new, fft_size)[:2 * n - 1]
        new_obj = 2.0 * n * np.max(b_new)

        if new_obj < best_obj:
            best_f = f_new.copy()
            best_obj = new_obj

        f = f_new

    return best_f, best_obj

        if p_sum < 1e-300:
