        # And the coefficient is f[k-j]... wait no.
        
        # Let me re-derive: convolve(f, b)[k] = sum_i f[i] * b[k-i]
        # b[j] appears when k-i = j, i.e., i = k-j
        # So coeff of b[j] in position k is f[k-j] (if k-j is in [0, n-1])
        # i.e., j <= k <= j+n-1
        
        start_k = j
        end_k = min(j + n - 1, 2*n - 2)
        for k in range(start_k, end_k + 1):
            A[k, j] = f[k - j]
    
    A = A.tocsr()
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.success:
        return res.x
    return None
