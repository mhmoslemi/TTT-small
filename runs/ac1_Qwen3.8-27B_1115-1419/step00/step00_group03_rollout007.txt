        # A_ub[k, i] = f[i] if 0 <= k-i <= n-1, else 0

            i_hi = min(n-1, k)
            A_ub[k, i_lo:i_hi+1] = f[i_lo:i_hi+1]

                          bounds=[(0, None)]*n, method='highs')

                g = g * (sum_f / sum_g)
                # Blend

                f_new = np.clip(f_new, 0, 1000)

                if new_score < score:

                    score = new_score
                    t_lr = min(1.0, t_lr * 1.05)  # gradually increase
                else:
                    t_lr = max(0.01, t_lr * 0.8)

import time

def _compute_objective(f):

    conv = np.convolve(f, f)
    max_b = np.max(conv)
    sum_a = np.sum(f)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _pressure(f, w):
    """Compute pressure[i] = sum_j f[j] * w[i+j] using FFT."""
    n = len(f)

    # pressure[i] = sum_j f[j] * w[i+j] for j=0..n-1 (valid indices)
    # = convolve(f, w[::-1])[n-1 : 2n-1][::-1]

        return f

def _compute_objective(f):

    sum_a = f.sum()

    Compute pressure[i] = sum_j f[j] * w[i+j] for j=0..n-1 (valid range).
    w has length 2n-1.

def _multiplicative_step(f, lr, beta):
    """One smoothed multiplicative update."""
    n = len(f)
    conv = np.convolve(f, f)
    max_conv = conv.max()
    if max_conv <= 1e-30:
        return f
    
    # Soft-max weights on convolution coefficients
    scaled = beta * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    
    # Compute pressure
    pressure = _pressure(f, w)
    
    # Multiplicative update
    max_p = pressure.max()
    if max_p <= 1e-30:

    # Clip the exponent for stability

    exp_arg = np.maximum(exp_arg, -50.0)

    if s > 1e-30:

    f_new = np.minimum(f_new, 1000.0)
    return f_new


def _lp_correction(f, M_factor=1.0):
    """
    LP-based correction: find g maximizing sum(g) s.t. conv(f,g) <= M.
    Blend f with g.
    """
    n = len(f)
    if n > 2500:
        return f  # Too slow for large n
    
    conv_f = np.convolve(f, f)
    M = conv_f.max() * M_factor
    if M <= 0:
        return f
    
    # Build constraint matrix: A[k, i] = f[i] if 0 <= k-i <= n-1
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n-1, k)
        A[k, i_lo:i_hi+1] = f[i_lo:i_hi+1]
    
    # LP: max 1^T g s.t. A@g <= M, g >= 0
    c = -np.ones(n)
    try:
        res = _linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                       bounds=[(0, None)]*n, method='highs')
        if res.success:
            g = res.x
            sum_g = g.sum()
            if sum_g > 0:
                # Rescale to same sum as f
                g *= f.sum() / sum_g
                # Blend with existing best
                t = 0.3
                f_new = (1-t)*f + t*g
                f_new = np.clip(f_new, 1e-12, 1000)
                return f_new
    except Exception:
        pass
    return f
