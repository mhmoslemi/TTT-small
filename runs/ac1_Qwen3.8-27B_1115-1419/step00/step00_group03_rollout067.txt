"""
Smooth mirror-descent search for step sequences minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
combined with multi-start (asymmetric) initializations and a local
block-perturbation search.  Asymmetry is essential: for a symmetric
sequence the convolution peak equals sum(a_i^2) and the objective is at
least 2 (uniform).  Off-centre / ramp-like starts let the peak pair large
values with small ones, pushing the ratio below 2.
"""

import numpy as np
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    M = np.max(c)
    S = np.sum(a)
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * M / (S * S))


def cross_corr(w, a):
    """p[i] = sum_k w[k]*a[k-i] (linear, no wrap) via FFT, i in [0,n)."""
    n = len(a)
    N = _next_pow2((2 * n - 1) + (n - 1))
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(W * np.conj(A), N)[:n]


def mirror_descend(a, eps0=3.0, eps1=0.05, iters=1500, double=8.0,
                   eta_factor=0.7, deadline=None):
    """Smoothed-logsumexp mirror descent on the simplex, temperature anneal."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-12:
        return a
    a = a / s
    n = len(a)
    best = a.copy()
    best_obj = compute_obj(a)

    log_eps0 = np.log(eps0)
    log_eps1 = np.log(eps1)

    for it in range(iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        frac = it / max(1, iters - 1)
        eps = np.exp(log_eps0 + (log_eps1 - log_eps0) * frac)

        c = np.convolve(a, a)
        M = np.max(c)
        ex = np.exp((c - M) / eps)
        sm = ex / np.sum(ex)

        grad = 2.0 * cross_corr(sm, a)

        # Adaptive step in mirror geometry.
        env = np.abs(grad)
        env_max = np.max(env)
        if env_max < 1e-14:
            break
        eta = eta_factor / env_max

        # Try a couple of step sizes, keep the best (line search).
        cand = None
        cand_obj = best_obj
        for factor in (1.0, 0.5, 0.25):
            step = eta * factor
            trial = a * np.exp(-step * grad)
            trial = np.maximum(trial, 0.0)
            tsum = np.sum(trial)
            if tsum < 1e-12:
                continue
            trial = trial / tsum
            obj = compute_obj(trial)
            if obj < cand_obj:
                cand_obj = obj
                cand = trial
        if cand is None:
            # Fall back to a single gentle step.
            trial = a * np.exp(-eta * 0.25 * grad)
            trial = np.maximum(trial, 0.0)
            tsum = np.sum(trial)
            if tsum < 1e-12:
                break
            trial = trial / tsum
            cand_obj = compute_obj(trial)
            cand = trial

        a = cand
        if cand_obj < best_obj:
            best_obj = cand_obj
            best = a.copy()

    return best


def local_search(a, rng, max_iter=2000, deadline=None):
    """Random multiplicative block perturbations, accept improving moves."""
    a = a.copy()
    n = len(a)
    cur_obj = compute_obj(a)
    best = a.copy()
    best_obj = cur_obj

    w_lo = max(2, n // 250)
    w_hi = max(w_lo + 1, n // 25)

    for _ in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        trial = a
        r = rng.random()
        if r < 0.7:
            # local block multiplier
            width = int(rng.integers(w_lo, w_hi + 1))
            start = int(rng.integers(0, n - width))
            trial = a.copy()
            sig = float(rng.uniform(0.1, 0.35))
            trial[start:start + width] *= np.exp(rng.normal(0.0, sig, width))
        elif r < 0.9:
            # global small noise
            sig = float(rng.uniform(0.02, 0.08))
            trial = a * np.exp(rng.normal(0.0, sig, n))
        else:
            # swap-mass: move some mass from a high block to a low block
            trial = a.copy()
            w1 = int(rng.integers(w_lo, w_hi + 1))
            w2 = int(rng.integers(w_lo, w_hi + 1))
            i1 = int(rng.integers(0, n - w1))
            i2 = int(rng.integers(0, n - w2))
            if i1 + w1 <= i2 or i2 + w2 <= i1:  # disjoint
                amt = 0.5 * np.sum(a[i1:i1 + w1])
                trial[i1:i1 + w1] -= amt / w1
                trial[i2:i2 + w2] += amt / w2
                trial = np.maximum(trial, 0.0)

        trial = np.maximum(trial, 0.0)
        tsum = np.sum(trial)
        if tsum < 1e-12:
            continue
        trial = trial / tsum
        obj = compute_obj(trial)
        if obj < cur_obj - 1e-12:
            a = trial
            cur_obj = obj
            if obj < best_obj:
                best_obj = obj
                best = a.copy()

    return best, best_obj


def _resample_to(h, n):
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1000.0)
    if h.size < 2:
        return np.ones(n)
    idx = np.linspace(0.0, h.size - 1.0, n)
    return np.interp(idx, np.arange(h.size), h)


def _starting_points(n, rng, lists):
    x = np.arange(n, dtype=np.float64)
    pts = [
        np.maximum(1.0 - x / n, 0.02),                       # decreasing ramp
        np.exp(-0.5 * ((x - 0.30 * n) / (0.10 * n)) ** 2),  # left bump
        np.exp(-0.5 * ((x - 0.70 * n) / (0.08 * n)) ** 2),  # right bump
        (n - x + 1.0) / float(n + 1.0),                       # power-ish decay
        rng.random(n) + 1e-3,                                  # random
        np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2)
        + 0.6 * np.exp(-0.5 * ((x - 0.78 * n) / (0.05 * n)) ** 2),  # 2 bumps
    ]
    for h in lists:
        try:
            r = _resample_to(h, n)
            if np.sum(r) > 0:
                pts.append(r)
        except Exception:
            pass
    # Normalise each to sum=1, clip nonneg.
    out = []
    for p in pts:
        p = np.maximum(np.asarray(p, dtype=np.float64), 0.0)
        s = np.sum(p)
        if s > 1e-12:
            out.append(p / s)
    return out
