from scipy import sparse
from scipy.optimize import linprog
import time

def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix C such that C @ b = convolve(a, b, 'full')"""
    n = len(a)
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For row k, j ranges from max(0, k-n+1) to min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C

Minimize 2n*max(conv(a,a))/sum(a)^2 using FFT-accelerated local search
with simulated annealing, multiple restarts, and LP-based polishing.
"""
import numpy as np
from scipy import sparse
import time


def _fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _fast_eval(a):
    """Fast evaluation using FFT convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-8:
        return np.inf
    conv = _fft_conv(a)
    return float(2.0 * n * conv.max() / (s * s))

        return a

    M = _fft_conv(a).max()

                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

"""
Minimize 2n*max(conv(a,a))/sum(a)^2 via FFT local search + LP polish.
Uses simulated annealing, multiple restarts, and structured initializations.
"""
import numpy as np
from scipy import sparse
import time


def _fft_conv(a):
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A*A, N)[:2*n-1]


def _feval(a):
    n = len(a)
    s = a.sum()
    if s < 1e-8:
        return np.inf
    mx = _fft_conv(a).max()
    return float(2.0*n*mx/(s*s))


def _build_conv_mat(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _lp_step(a, rng, deadline):
    n = len(a)
    M = _fft_conv(a).max()
    C = _build_conv_mat(a)
    from scipy.optimize import linprog
    t0 = time.monotonic()
    try:
        res = linprog(-np.ones(n), A_ub=C, b_ub=np.full(2*n-1, M),
                      bounds=[(0,None)]*n, method='highs')
    except Exception:
        return a, M, False
    if not res.success:
        return a, M, False
    b = res.x
    sb = b.sum()
    if sb <= 1.0 + 1e-9:
        return a, M, False
    g = b / sb
    for t in [0.5, 0.3, 0.1, 0.05]:
        an = np.clip((1-t)*a + t*g, 0, None)
        ns = an.sum()
        if ns < 1e-8: continue
        an /= ns
        nM = _fft_conv(an).max()
        if nM < M:
            print(f"  LP: M {M:.8f} -> {nM:.8f} (t={t}, sum_b={sb:.4f}, {time.monotonic()-t0:.1f}s)")
            return an, nM, True
    return a, M, False
