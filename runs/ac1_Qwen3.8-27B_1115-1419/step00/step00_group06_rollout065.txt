import numpy as np

def fast_convolution(a):
    """Compute linear convolution a*a using FFT. Returns full convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)
    return conv[:2 * n - 1]


def compute_score(a):
    """Compute the objective value for sequence a."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fast_convolution(a)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S * S)


def alternating_pattern(n, ratio, period=2):
    """Create alternating pattern with given ratio and period."""
    seq = np.ones(n)
    for i in range(1, period):
        seq[i::period] = ratio ** (1.0 - i / max(1, period - 1))
    seq[0::period] = ratio
    return seq


def block_pattern(n, block_size, ratio):
    """Create block pattern: alternating blocks of ratio and 1."""
    seq = np.ones(n)
    period = 2 * block_size
    for j in range(0, n, period):
        end = min(j + block_size, n)
        seq[j:end] = ratio
    return seq

def local_search_refined(seq, deadline, rng):
    """
    More refined local search using gradient-informed direction selection.
    Focuses on reducing the active peak while monitoring other peaks.
    """
    n = len(seq)

    current_score = compute_score(seq)
    best_local = seq.copy()
    best_local_score = current_score

    size = 1
    while size < 2 * n - 1:
        size <<= 1

        iteration += 1

        A = np.fft.rfft(seq, size)
        conv = np.fft.irfft(A * A, size)[:2 * n - 1]
        S = np.sum(seq)
        k_star = int(np.argmax(conv))
        current_score = 2.0 * n * conv[k_star] / (S * S)

        if current_score < best_local_score:
            best_local_score = current_score
            best_local = seq.copy()

        # Compute subgradient: direction that reduces conv[k_star]
        # d(conv[k_star])/d(seq[i]) involves seq[k_star - i]
        # We want to decrease seq[i] where seq[k_star - i] is large
        # and increase seq[i] where it contributes less to the peak

        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)

        # Build "pressure" on each element based on peak contribution
        pressure = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            other = k_star - j
            if 0 <= other < n:
                pressure[j] += seq[other]
                if other != j:
                    pressure[other] += seq[j]
                else:
                    pressure[j] += 0  # already counted

        # Normalize pressure
        max_pressure = np.max(pressure)
        if max_pressure > 0:
            pressure = pressure / max_pressure

        # Try moving elements with high pressure downward
        # and elements with low pressure upward
        n_tries = min(20, max(5, n // 50))

        # High pressure candidates (reduce these)
        high_pressure_idx = np.where(pressure > 0.5)[0]
        if len(high_pressure_idx) > 0:
            for _ in range(n_tries // 2):
                idx = high_pressure_idx[rng.integers(len(high_pressure_idx))]
                old_val = seq[idx]
                if old_val < 1e-10:
                    continue
                for factor in [0.9, 0.8, 0.7, 0.5]:
                    seq[idx] = old_val * factor
                    new_score = compute_score(seq)
                    if new_score < current_score:
                        current_score = new_score
                        break
                    seq[idx] = old_val

        # Low pressure candidates (increase these to spread energy)
        low_pressure_idx = np.where((pressure < 0.2) & (seq > 0.01))[0]
        if len(low_pressure_idx) > 0:
            for _ in range(n_tries // 2):
                idx = low_pressure_idx[rng.integers(len(low_pressure_idx))]
                old_val = seq[idx]
                for factor in [1.3, 1.7, 2.5]:
                    seq[idx] = old_val * factor
                    new_score = compute_score(seq)
                    if new_score < current_score:
                        current_score = new_score
                        break
                    seq[idx] = old_val

        # Periodic block perturbation to escape local minima
        if iteration % 100 == 0:
            width = max(1, n // rng.integers(10, 50))
            start = rng.integers(0, max(1, n - width))
            old_block = seq[start:start + width].copy()

            new_score = compute_score(seq)
            if new_score < current_score:
                current_score = new_score
            else:
                seq[start:start + width] = old_block

        # Check convergence
        if current_score < best_local_score:
            best_local_score = current_score
            best_local = seq.copy()

    return best_local, best_local_score
