def fast_convolve(f):
    """Compute auto-convolution f*f using FFT."""
    n = len(f)
    target = 2 * n - 1
    N = 1
    while N < target:
        N <<= 1
    F = fft(f, N)

def fast_eval(f):
    """Fast evaluation of the objective function."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    conv = fast_convolve(f)
    return 2.0 * n * np.max(conv) / (S * S)


def subgradient_descent(f0, alpha_init=0.5, n_steps=1000, top_k=20, decay=0.9995):
    """Subgradient descent to minimize peak autocorrelation."""
    n = len(f0)
    f = f0.copy().astype(np.float64)
    S = np.sum(f)
    if S < 1e-15:
        return f0.copy(), float('inf')
    f = f / S
    alpha = alpha_init
    best_f = f.copy()
    best_obj = fast_eval(f)
    
    for step in range(n_steps):
        conv = fast_convolve(f)
        
        # Find top_k peaks
        # Use a threshold to include all "significant" peaks
        peak_indices = np.argsort(conv)[-top_k:]
        
        # Weight peaks by their value (sharpened)
        peak_vals = conv[peak_indices]
        weights = peak_vals.astype(np.float64)
        weights[weights < 0] = 0
        weight_sum = np.sum(weights)
        if weight_sum < 1e-15:
            break
        weights /= weight_sum
        
        # Compute gradient
        grad = np.zeros(n, dtype=np.float64)
        for idx in range(len(peak_indices)):
            k = peak_indices[idx]
            w = weights[idx]
            if w < 1e-15:
                continue
            # grad[j] += w * f[k-j] for valid j
            j_lo = max(0, k - n + 1)
            j_hi = min(k + 1, n)
            # f[k-j] for j in [j_lo, j_hi-1]
            # k-j ranges from k-j_lo to k-(j_hi-1) = k-j_hi+1
            # k-j_lo = k - max(0, k-n+1) = min(k, n-1)

            # So we need f[max(0,k-n+1) : min(k,n-1)+1] reversed
            f_slice = f[max(0, k-n+1) : min(k, n-1)+1]
            grad[j_lo:j_hi] += w * f_slice[::-1]
        
        # Normalize gradient
        mean_grad = np.mean(grad)
        if mean_grad < 1e-15:

        grad = grad / mean_grad - 1.0  # center it

        f_new = f * np.exp(-alpha * grad)

    """Two Gaussian bumps."""
    j = np.arange(n, dtype=np.float64)
    c1 = n * (0.3 + 0.2 * rng.random())
    c2 = n * (0.6 + 0.2 * rng.random())
    w = n * 0.1
    f = np.exp(-0.5 * ((j - c1) / w) ** 2) + np.exp(-0.5 * ((j - c2) / w) ** 2)
    f = np.clip(f, 1e-10, 1000.0)
    f /= np.sum(f)
    return f


def _init_chirp_abs(n):
    """Absolute value of a chirp-like sequence."""
    j = np.arange(n, dtype=np.float64)
    f = 1.0 + np.cos(2.0 * np.pi * j * j / (3.0 * n))
    f = np.clip(f, 0.0, 1000.0)
    f = np.maximum(f, 1e-10)
    f /= np.sum(f)
    return f


def _local_refine(f, rng, deadline, obj_thresh=1e-9):
    """Local search with block perturbations and occasional subgradient restarts."""
    n = len(f)
    best = f.copy()
    best_obj = fast_eval(f)
    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() < deadline:
        # Random block perturbation
        f_try = best.copy()
        n_blocks = rng.integers(1, 5)
        for _ in range(n_blocks):
            bs = int(rng.integers(max(2, n // 200), max(3, n // 15)))
            bs = min(bs, n)
            start = int(rng.integers(0, max(1, n - bs)))
            f_try[start:start + bs] *= float(np.exp(rng.normal(0, 0.08)))
        f_try = np.clip(f_try, 0.0, 1000.0)
        S = np.sum(f_try)
        if S < 1e-15:
            continue
        f_try /= S

        obj = fast_eval(f_try)
        if np.isfinite(obj) and obj < best_obj - obj_thresh:
            best_obj = obj
            best = f_try.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodically try a short subgradient burst
        if no_improve > 200:
            f_sub, obj_sub = _run_subgradient(
                best, alpha_init=0.2, n_steps=200,
                top_k=min(15, max(5, n // 50)), decay=0.999,
                deadline=deadline
            )
            if np.isfinite(obj_sub) and obj_sub < best_obj:
                best_obj = obj_sub
                best = f_sub.copy()
            no_improve = 0

        if no_improve > max_no_improve:
            break

    return best, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.
    Uses subgradient descent + local refinement across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s)) - 15.0

    best_seq = None
    best_score = float('inf')

    # Try a range of lengths
    lengths = [200, 350, 500, 750, 1000, 1500, 2000]

    # Order: try medium lengths first (good balance of DOF vs speed)
    lengths.sort(key=lambda x: abs(x - 800))

    for n in lengths:
        if time.monotonic() > deadline:
            break

        # Get starting points
        inits = [
            _init_uniform_noise(n, rng, 0.3),
            _init_cosine(n),
            _init_bimodal(n, rng),
            _init_chirp_abs(n),
            _init_uniform_noise(n, rng, 0.8),
        ]

        # If we have a global sequence available, also try it
        try:
            if 'height_sequence_1' in globals() and len(height_sequence_1) == n:
                f_g = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
                S = np.sum(f_g)
                if S > 0.01:
                    inits.append(f_g / S)
        except Exception:
            pass

        for f0 in inits:
            if time.monotonic() > deadline:
                break
            if not np.all(np.isfinite(f0)):
                continue
            if np.sum(f0) < 1e-10:
                continue

            f_norm = f0 / np.sum(f0)

            # Determine steps
            obj = _run_subgradient(
                f_norm,
                alpha_init=0.4,
                n_steps=3000,
                top_k=min(25, max(8, n // 40)),
                decay=0.9998,
                deadline=deadline
            )
            fb, ob = obj
            if np.isfinite(ob) and ob < best_score:
                best_score = ob
                best_seq = fb.copy()
                elapsed = time.monotonic() - t0
                print(f"n={n}, obj={best_score:.6f}, elapsed={elapsed:.1f}s")

    # Phase 2: Local refinement on the best
    if best_seq is not None and time.monotonic() < deadline:
        print(f"Before refinement: {best_score:.6f}")
        refined, ref_obj = _local_refine(best_seq, rng, deadline)
        if np.isfinite(ref_obj) and ref_obj < best_score:
            best_score = ref_obj
            best_seq = refined
            print(f"After refinement: {best_score:.6f}")

    # Phase 3: Try to squeeze more from the best with fresh subgradient
    if best_seq is not None and time.monotonic() < deadline:
        n = len(best_seq)
        fb, ob = _run_subgradient(
            best_seq,
            alpha_init=0.1,
            n_steps=3000,
            top_k=min(30, max(10, n // 30)),
            decay=0.9999,
            deadline=deadline
        )
        if np.isfinite(ob) and ob < best_score:
            best_score = ob
            best_seq = fb
            print(f"After final subgradient: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64) / 256.0
        best_score = evaluate_sequence(best_seq.tolist())

    # Ensure valid output
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    final_val = evaluate_sequence(result)
    if not np.isfinite(final_val):
        # Fallback
        result = [1.0 / 256.0] * 256
        final_val = evaluate_sequence(result)

    print(f"FINAL: score={final_val:.6f}, length={len(result)}")
    return result
