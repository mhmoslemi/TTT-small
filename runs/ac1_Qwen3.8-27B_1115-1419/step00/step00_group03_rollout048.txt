def solve_lp(a, M):
    """Solve LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Convolution matrix: A[k, j] such that (A @ b)[k] = conv(a, b)[k]
    # conv(a, b)[k] = sum_i a[i] * b[k-i]
    # For k in range(0, 2n-1), the valid range of i is max(0, k-n+1) to min(n-1, k)
    # Let j = k-i, so j ranges from max(0, k-(n-1)) to min(k, n-1)
    # A[k, j] = a[k-j] if k-j is in [0, n-1], else 0
    
    # Actually, let's just build the matrix directly
    # A is (2n-1) x n
    
    # For efficiency, we can note that A is a convolution matrix
    # A[k, j] = a[k - j] if 0 <= k-j <= n-1, else 0
    
    # Let's build it
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    # Maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M
    b_ub = np.full(2*n - 1, M)
    
    # b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
