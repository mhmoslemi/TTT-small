def solve_lp(f, M):
    n = len(f)
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f⊛g)[k] <= M for k = 0, ..., 2n-2
    # (f⊛g)[k] = sum_i f[i]*g[k-i] where 0<=i<n and 0<=k-i<n
    # For each k, the constraint is: sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[i]*g[k-i] <= M
    # Let j = k-i, so i = k-j: sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j]*g[j] <= M
    # For a fixed k and variable g[j], the coefficient is f[k-j] if 0<=k-j<n, else 0.
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    b_ub = np.full(2*n-1, M)
    
    # g >= 0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x
