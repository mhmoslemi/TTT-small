import numpy as np
import time

def projected_subgradient_descent(a, lr=0.01, n_iter=1000, verbose=False):
    """Projected subgradient descent for min max(a*a) s.t. sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()  # normalize to sum=1
    
    for it in range(n_iter):
        c = np.convolve(a, a)
        max_c = c.max()
        # Active set: terms within 1% of max
        active = c >= 0.99 * max_c
        # Compute subgradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if active[k]:
                # grad[j] += a[k-j] for valid j
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                for j in range(lo, hi + 1):
                    if k - j < n:
                        if k == 2 * j:
                            grad[j] += 2.0 * a[j]
                        else:
                            grad[j] += a[k - j]
        grad /= active.sum()
        # Subtract mean (for equality constraint)
        grad -= grad.mean()
        # Update
        a -= lr * grad
        # Project to simplex {a >= 0, sum(a) = 1}
        a = project_to_simplex(a, 1.0)
    
    return a

def project_to_simplex(v, s):
    """Project v onto {x: x >= 0, sum(x) = s}."""
    n = len(v)
    # Binary search for threshold
    lo, hi = 0.0, v.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.maximum(v - mid, 0)
        if proj.sum() > s:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2, 0)

Minimizes 2*n*max(a*a)/(sum(a))^2 for non-negative sequence a.
Uses softmax-smoothed projected gradient descent on the probability simplex,
with FFT-accelerated convolution and gradient computation.
Includes coordinate descent sweeps and adaptive step-size control.
"""

import numpy as np
import time


def project_to_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} using bisection."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    if np.max(v) <= 0:
        return np.full(n, target_sum / n)
    lo = 0.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.maximum(v - mid, 0.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2.0, 0.0)


def fast_convolve(a):
    """Linear convolution a*a using FFT for speed."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N *= 2
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def compute_gradient_fft(a, p, N_fft):
    """
    Compute grad_j = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] for j=0,...,n-1.
    Uses FFT: place reversed a, convolve with p.
    """
    n = len(a)
    p_padded = np.zeros(N_fft)
    p_padded[:2 * n - 1] = p
    a_rev = np.zeros(N_fft)
    a_rev[:n] = a[::-1]
    P = np.fft.rfft(p_padded)
    AR = np.fft.rfft(a_rev)
    result = np.fft.irfft(P * AR, N_fft)
    return 2.0 * result[n - 1:n - 1 + n]


def softmax_weights(c, epsilon):
    """Compute normalized softmax weights over convolution values."""
    c_shifted = c - np.max(c)
    exp_c = np.exp(c_shifted / epsilon)
    s = exp_c.sum()
    if s < 1e-300:
        # Uniform fallback
        return np.ones(len(c)) / len(c)
    return exp_c / s


def coordinate_descent_sweep(a, n_passes=1):
    """
    Coordinate descent: for each j, find optimal a_j given all others fixed.
    The objective (with sum=1) becomes max_k (c_k^0 + 2*a_j*a_{k-j}) or c_{2j}^0 + a_j^2.
    We find a_j that minimizes the max, subject to a_j >= 0 and sum constraint.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    for _ in range(n_passes):
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-15:
                continue
            # Remove a_j from a
            a[j] = 0.0
            c_base = fast_convolve(a)  # convolution without a_j
            # Now c_k with a_j: c_k_new = c_k_base + a_j * (a_{k-j} + a_{j-k}) for k != 2j
            # and c_{2j}_new = c_{2j}_base + a_j^2
            # We want to find a_j >= 0 that minimizes max_k (c_k_base + contribution_k)
            # subject to sum(a) = 1 (so adjusting a_j changes the sum)
            # For simplicity, just compute current max and nudge a_j toward reducing it
            
            # Contribution of a_j to c_k:
            # For k in [0, 2n-2]: contrib_k = a_j * (a_{k-j} if 0<=k-j<n else 0) 
            #                      + a_j * (a_{j-(k - (n-1))} ...) 
            # Actually: c_k = c_k_base + a_j*a_{k-j} (if 0<=k-j<n) + a_j*a_{j-k} (if 0<=j-k<n)
            # But since a[j]=0 already, a_{k-j} and a_{j-k} don't include a_j itself.
            # However a_{k-j} = 0 if k-j == j, i.e., k = 2j (since we set a[j]=0).
            # So: c_k_with = c_k_base + 2*a_j*a_{k-j} for k != 2j (0<=k-j<n)
            #       c_{2j}_with = c_{2j}_base + a_j^2
            
            # For optimization, let's just compute the new max for a few candidate a_j values
            # and pick the best. This is approximate but effective.
            
            target = old_val
            best_obj = np.inf
            best_aj = old_val
            # Try several fractions of old_val
            for frac in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6]:
                aj = target * frac
                if aj < 0:
                    aj = 0.0
                # Compute new max
                # c_new_k = c_base_k + 2*aj*a[k-j] (for 0<=k-j<n, k!=2j)
                #          c_base_2j + aj^2 (for k=2j)
                # Approximate: just check the top few c_k values
                # For speed, estimate the new max
                idx = max(0, j)
                contrib = np.zeros(2 * n - 1)
                lo = max(0, j)
                hi = min(2 * n - 2, j + n - 1)
                for k in range(lo, hi + 1):
                    km = k - j
                    if 0 <= km < n:
                        if k == 2 * j:
                            contrib[k] = aj * aj
                        else:
                            contrib[k] = 2.0 * aj * a[km]
                c_new = c_base + contrib
                # Renormalize: sum(a) changes, so we need to account for that
                new_sum = a.sum() + aj
                if new_sum < 1e-10:
                    continue
                obj = 2.0 * n * c_new.max() / (new_sum * new_sum)
                if obj < best_obj:
                    best_obj = obj
                    best_aj = aj
            a[j] = best_aj
        # Renormalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
    return a
