def _smooth_gradient(a, beta):
    """
    Compute smoothed gradient of max(a*a) using soft-max weights.
    Returns (gradient vector of length n, c array, c_max).
    """
    c = _fft_convolve(a)
    c_max = c.max()
    # Soft-max weights (numerically stable)
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    n = len(a)

    # b_i = (w * a_rev)_{n-1+i}

    conv = np.fft.irfft(W * A, size)
    b = conv[n - 1:n - 1 + n]
    grad = (2.0 / beta) * b
    return grad, c, c_max
