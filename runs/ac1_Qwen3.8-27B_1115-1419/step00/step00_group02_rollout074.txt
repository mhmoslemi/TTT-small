"""
Search for a nonnegative sequence minimizing
    2 * n * max(convolve(a, a)) / sum(a)^2.

Algorithm summary:
- Start from strong analytic initializers, primarily a-1/2 endpoint singular
  discretization, plus a small parametrized sweep.
- Polish with a mirror-descent / subgradient method on the soft maximum of the
  self-convolution.  Gradients are computed matrix-free using FFTs.
- Run several sequence lengths, keep the best internal score, and optionally do
  a final exact verification with evaluate_sequence if time permits.
"""

import time
import numpy as np


def next_pow2(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def autocorr_fft(x, size):
    """Full linear autocorrelation x*x by zero-padded FFT."""
    fa = np.fft.rfft(x, size)
    return np.fft.irfft(fa * fa, size)


def score_normalized_fft(x, n, size):
    """Score assuming sum(x) == 1."""
    fa = np.fft.rfft(x, size)
    c = np.fft.irfft(fa * fa, size)
    return float(2.0 * n * np.max(c))


def score_array(x):
    """Internal fast FFT score for a sequence, after normalization."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.inf
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    x = x / s
    n = x.size
    size = next_pow2(2 * n - 1)
    return score_normalized_fft(x, n, size)


def make_sequence_from_params(n, alpha, shift, endpoint, exp1, exp2):
    """
    Parametric initializer:
        x_i proportional to (i + shift)^(-alpha) * (1 - i/(n-1))^endpoint
              * exp(exp1*u + exp2*u^2),  u = t - 0.5.
    """
    n = int(n)
    if n <= 0:
        return []
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = i / float(n - 1)
    else:
        t = np.zeros(1, dtype=np.float64)

    shift = float(shift)
    if shift <= 0.1:
        shift = 1.0

    x = np.power(i + shift, -float(alpha))

    if float(endpoint) != 0.0:
        tail = np.maximum(0.0, 1.0 - t)
        x = x * np.power(tail, float(endpoint))

    e1 = float(exp1)
    e2 = float(exp2)
    if e1 != 0.0 or e2 != 0.0:
        u = t - 0.5
        x = x * np.exp(e1 * u + e2 * u * u)

    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return x / s


def sweep_initial(n=4096, time_limit=60.0, seed=0):
    """
    Coarse sweep over a small family of initial sequences.
    Returns (params, internal_score).
    """
    start = time.monotonic()
    lim = max(0.0, float(time_limit))
    n = int(n)
    if n <= 0:
        return (0.5, 1.0, 0.0, 0.0, 0.0), np.inf

    pure = make_sequence_from_params(n, 0.5, 1.0, 0.0, 0.0, 0.0)
    size = next_pow2(2 * n - 1)
    best_score = score_normalized_fft(pure, n, size)
    best_params = (0.5, 1.0, 0.0, 0.0, 0.0)

    if lim <= 0.0:
        return best_params, best_score

    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1)
    u = t - 0.5
    tail = np.maximum(0.0, 1.0 - t)

    alphas = (0.45, 0.48, 0.50, 0.52, 0.55, 0.60)
    shifts = (0.5, 1.0, 2.0, 4.0)
    endpoints = (0.0, 0.2, 0.5, 1.0)
    exp1s = (-1.0, -0.5, 0.0, 0.5, 1.0)
    exp2s = (-1.0, 0.0, 1.0)

    for alpha in alphas:
        if time.monotonic() - start > lim:
            break
        for shift in shifts:
            if time.monotonic() - start > lim:
                break
            base = np.power(i + float(shift), -float(alpha))
            for endpoint in endpoints:
                if time.monotonic() - start > lim:
                    break
                if endpoint == 0.0:
                    baseep = base
                else:
                    baseep = base * np.power(tail, float(endpoint))
                for e1 in exp1s:
                    if time.monotonic() - start > lim:
                        break
                    for e2 in exp2s:
                        x = baseep
                        if e1 != 0.0 or e2 != 0.0:
                            x = x * np.exp(float(e1) * u + float(e2) * u * u)
                        x = np.maximum(x, 0.0)
                        s = float(x.sum())
                        if np.isfinite(s) and s > 1e-300:
                            x = x / s
                            sc = score_normalized_fft(x, n, size)
                            if np.isfinite(sc) and sc < best_score:
                                best_score = sc
                                best_params = (alpha, shift, endpoint, e1, e2)
                        if time.monotonic() - start > lim:
                            return best_params, best_score
    return best_params, best_score


def make_p_from_c(c, gamma):
    """
    Smooth subgradient loading proportional to c^gamma.
    c is the trimmed self-convolution of length 2n-1.
    """
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.ones(c.size, dtype=np.float64) / float(c.size)

    r = np.maximum(0.0, c / cmax)
    p = np.power(r, float(gamma))
    ps = float(p.sum())
    if not np.isfinite(ps) or ps <= 0.0:
        mask = r > 0.0
        if not mask.any():
            return np.ones(c.size, dtype=np.float64) / float(c.size)
        p = mask.astype(np.float64)
        ps = float(p.sum())
    return p / ps


def lowpass_noise(noise, level=20):
    """Low-pass filter a random noise vector (for stable restarts)."""
    noise = np.asarray(noise, dtype=np.float64)
    n = noise.size
    if n == 0:
        return noise
    L = next_pow2(n)
    f = np.fft.rfft(noise, L)
    keep = max(2, f.size // max(2, int(level)))
    f[keep:] = 0.0
    return np.fft.irfft(f, L)[:n]


def gradient_optimize(x0, n, time_limit_s, seed=42, gamma_start=20.0):
    """
    FFT-based mirror-descent/local optimization.
    Returns (best_sequence, best_internal_score).
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64), np.inf

    x = np.asarray(x0, dtype=np.float64).copy()
    if x.size != n:
        x = x[:n]
        if x.size < n:
            x = np.resize(x, n)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        x = np.ones(n, dtype=np.float64) / float(n)
    else:
        x = x / s

    m = 2 * n - 1
    Lc = next_pow2(m)
    Lh = next_pow2(m + n - 1)

    c = autocorr_fft(x, Lc)[:m]
    score = float(2.0 * n * np.max(c))

    best = x.copy()
    best_c = c.copy()
    best_score = score

    gamma = float(gamma_start)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    limit = max(0.0, float(time_limit_s) - 2.0)

    while time.monotonic() - start < limit:
        it += 1

        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax <= 0.0:
            break

        p = make_p_from_c(c, gamma)

        # P_range[i] = sum_{s=i}^{i+n-1} p_s
        P = np.cumsum(p)
        P_range = np.empty(n, dtype=np.float64)
        P_range[0] = P[n - 1]
        if n > 1:
            P_range[1:] = P[n:] - P[:n - 1]

        # h[i] = sum_j p_{i+j} x_j
        xr = x[::-1]
        fp = np.fft.rfft(p, Lh)
        fx = np.fft.rfft(xr, Lh)
        convp = np.fft.irfft(fp * fx, Lh)
        h = convp[n - 1:n + n - 1]

        # Gradient of the soft-max of convolution entries w.r.t. x.
        g = P_range * x + h
        g = g - float(np.mean(g))

        med_g = float(np.median(np.abs(g)))
        med_x = float(np.median(x))

        accepted = False
        best_cand_x = None
        best_cand_c = None
        best_cand_sc = score

        if np.isfinite(med_g) and med_g > 1e-300:
            # Multiplicative candidates.
            for alpha in (0.5, 0.2, 0.08):
                step = alpha / med_g
                bound = 2.0 * alpha
                rel = np.clip(-step * g, -bound, bound)
                xn = x * np.exp(rel)
                sx = float(xn.sum())
                if not np.isfinite(sx) or sx <= 1e-300:
                    continue
                xn = xn / sx
                cn = autocorr_fft(xn, Lc)[:m]
                sc = float(2.0 * n * np.max(cn))
                if np.isfinite(sc) and sc < best_cand_sc:
                    best_cand_x = xn
                    best_cand_c = cn
                    best_cand_sc = sc

            # Additive candidates.
            for alpha in (0.10, 0.04):
                step = alpha * med_x / med_g
                bound = 3.0 * alpha * med_x
                delta = np.clip(step * g, -bound, bound)
                xn = np.maximum(x - delta, 0.0)
                sx = float(xn.sum())
                if not np.isfinite(sx) or sx <= 1e-300:
                    continue
                xn = xn / sx
                cn = autocorr_fft(xn, Lc)[:m]
                sc = float(2.0 * n * np.max(cn))
                if np.isfinite(sc) and sc < best_cand_sc:
                    best_cand_x = xn
                    best_cand_c = cn
                    best_cand_sc = sc

            if best_cand_x is not None:
                x = best_cand_x
                c = best_cand_c
                score = best_cand_sc
                accepted = True

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best = x.copy()
            best_c = c.copy()

        if not accepted:
            no_improve += 1

            if no_improve == 5:
                gamma = min(gamma * 1.5, 80.0)
            elif no_improve == 15:
                gamma = min(gamma * 1.5, 300.0)
            elif no_improve == 30:
                gamma = float(gamma_start)
            elif no_improve == 60:
                # Small smooth perturbation from current point.
                old_score = score
                noise = lowpass_noise(rng.normal(0.0, 0.05, n), 20)
                xn = x * np.exp(noise)
                sx = float(xn.sum())
                if np.isfinite(sx) and sx > 1e-300:
                    x = xn / sx
                    c = autocorr_fft(x, Lc)[:m]
                    score = float(2.0 * n * np.max(c))
                if (not np.isfinite(score)) or score > best_score * 1.2 + 1e-6:
                    x = best.copy()
                    c = best_c.copy()
                    score = best_score
                no_improve = 0
            elif no_improve == 120:
                # Larger smooth restart from the best point.
                noise = lowpass_noise(rng.normal(0.0, 0.15, n), 10)
                xn = best.copy() * np.exp(noise)
                sx = float(xn.sum())
                if np.isfinite(sx) and sx > 1e-300:
                    x = xn / sx
                    c = autocorr_fft(x, Lc)[:m]
                    score = float(2.0 * n * np.max(c))
                if (not np.isfinite(score)) or score > best_score * 1.2 + 1e-6:
                    x = best.copy()
                    c = best_c.copy()
                    score = best_score
                no_improve = 0
            elif no_improve >= 200:
                # Hard reset to best and re-open search.
                x = best.copy()
                c = best_c.copy()
                score = best_score
                gamma = float(gamma_start)
                no_improve = 0

        if it % 200 == 0:
            print(
                "gradient_optimize n={} it={} best={:.8f} cur={:.8f} gamma={:.2f}".format(
                    n, it, best_score, score, gamma
                ),
                flush=True,
            )

    if np.isfinite(score) and score < best_score:
        best_score = score
        best = x.copy()

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point.  Returns the best nonnegative sequence found.
    """
    start_time = time.monotonic()
    budget = max(120.0, float(budget_s))
    # Reserve time for final exact check / fallback.
    deadline = start_time + budget - 30.0

    rng_seed = int(seed) if seed is not None else 42

    # Safe fallback: a-1/2 initializer at a length that is cheap to re-evaluate.
    fallback_n = 6962
    fallback_seq = make_sequence_from_params(fallback_n, 0.5, 1.0, 0.0, 0.0, 0.0)
    best_seq = fallback_seq
    best_score = score_array(fallback_seq)
    best_n = fallback_n

    print("initial fallback score={:.8f}".format(best_score), flush=True)

    # Coarse parametric sweep.
    sweep_params = (0.5, 1.0, 0.0, 0.0, 0.0)
    if time.monotonic() < deadline - 320:
        t_sweep = min(80.0, max(0.0, deadline - time.monotonic() - 260.0))
        if t_sweep > 8.0:
            sweep_params, sweep_score = sweep_initial(4096, t_sweep, seed=rng_seed)
            print("sweep best params={} score={:.8f}".format(sweep_params, sweep_score), flush=True)

    # Main optimization at several lengths.
    # The allocations are conservative; remaining time later polishes the global best.
    phases = (
        (fallback_n, 700.0),
        (10000, 650.0),
        (16384, 750.0),
    )

    for phase_idx, (n, alloc) in enumerate(phases):
        rem = deadline - time.monotonic()
        if rem < 80.0:
            break

        inits = []

        pure = make_sequence_from_params(n, 0.5, 1.0, 0.0, 0.0, 0.0)
        inits.append((score_array(pure), 0, pure))

        param_seq = make_sequence_from_params(n, *sweep_params)
        param_score = score_array(param_seq)
        inits.append((param_score, 1, param_seq))

        # If the supplied global sequence matches this length and is viable, include it.
        hs = globals().get("height_sequence_1", None)
        if hs is not None:
            try:
                arr = np.asarray(hs, dtype=np.float64)
                if arr.ndim == 1 and arr.size == n:
                    arr = np.maximum(arr, 0.0)
                    s = float(arr.sum())
                    if np.isfinite(s) and s > 1e-300:
                        arr = arr / s
                        hs_score = score_array(arr)
                        inits.append((hs_score, 2, arr))
            except Exception:
                pass

        # Sort by score, then by insert index.  No lambda needed.
        inits.sort()
        init_seq = inits[0][2]

        t_opt = min(alloc, rem - 80.0)
        if t_opt > 40.0:
            seq, sc = gradient_optimize(init_seq, n, t_opt, seed=rng_seed + 1000 * phase_idx)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_seq = seq
                best_n = n
            print(
                "phase length={} init_score={:.8f} result_score={:.8f} global={:.8f}".format(
                    n, inits[0][0], sc if np.isfinite(sc) else float("inf"), best_score
                ),
                flush=True,
            )

    # Final polish on the global best if time allows.
    rem = deadline - time.monotonic()
    if rem > 180.0:
        t_polish = rem - 80.0
        if t_polish > 50.0:
            seq, sc = gradient_optimize(best_seq, best_n, t_polish, seed=rng_seed + 99999)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_seq = seq
                best_n = best_n
            print("polish global={:.8f}".format(best_score), flush=True)

    # Optional exact verification with the official evaluator if safe/fast enough.
    rem = deadline - time.monotonic()
    if rem > 8.0 and 0 < best_n <= 20000:
        try:
            exact = evaluate_sequence(best_seq.tolist())
            if np.isfinite(exact):
                best_score = float(exact)
            else:
                best_seq = make_sequence_from_params(fallback_n, 0.5, 1.0, 0.0, 0.0, 0.0)
                best_n = fallback_n
                best_score = score_array(best_seq)
        except Exception:
            pass

    # Final safety checks.
    if (
        not np.isfinite(best_score)
        or best_seq is None
        or len(best_seq) == 0
    ):
        best_seq = make_sequence_from_params(fallback_n, 0.5, 1.0, 0.0, 0.0, 0.0)
        best_n = fallback_n
        best_score = score_array(best_seq)

    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        arr = make_sequence_from_params(fallback_n, 0.5, 1.0, 0.0, 0.0, 0.0)

    return [float(v) for v in arr]
