Let j = k - i, so k = i + j:

This is equivalent to: for each i, sum over j from max(0, -i) to min(n-1, 2n-2-i) of p[i+j] * a[j].

This can be computed as a cross-correlation, which is a convolution with one signal reversed.

Since p has length 2n-1 and a_reversed has length n, the convolution has length 2n-1+n-1 = 3n-2.
The index i + n - 1 ranges from n-1 to 2n-2, which is within [0, 3n-3]. Good.

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def cross_correlate_fft(p, a):
    """Compute (p ⋆ a)[i] = sum_k p[k] * a[k-i] for i = 0, ..., len(a)-1."""
    a_rev = a[::-1]
    conv = fft_convolve(p, a_rev)
    n_a = len(a)
    return conv[n_a - 1: 2 * n_a - 1]

def smooth_max_and_weights(c, T):
    """Return (f_T, p) where f_T = T*log(sum exp(c/T)) and p = softmax(c/T)."""
    c_max = np.max(c)
    shifted = (c - c_max) / T
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    f_T = c_max + T * np.log(sum_exp)
    p = exp_shifted / sum_exp
    return f_T, p
