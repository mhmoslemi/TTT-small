Let me implement a cleaner version of this idea.

Algorithm:
1. Initialize a = uniform(1/n), M = max(a*a) = 1/n.
2. Repeat:

3. Return a.

The LP is:
max sum(b)
subject to: sum_i a[i]*b[k-i] ≤ M for k = 0, ..., 2n-2
            b[j] ≥ 0 for j = 0, ..., n-1

def project_simplex(v, cap=1.0):
    """Project v onto {x >= 0, sum(x) = cap} using the water-filling algorithm."""
    n = len(v)
    u = np.sort(v)[::-1]  # Sort in descending order

    if not np.any(cond):
        return np.full(n, cap/n)
    k = np.max(np.where(cond)[0]) + 1
    theta = (cssv[k-1] - cap) / k
    return np.maximum(v - theta, 0)

    from scipy.optimize import linprog
    n = len(a)
    a = a / a.sum()  # Normalize
    M = np.convolve(a, a).max()
    best_a = a.copy()
    best_M = M

        # (a*b)[k] = sum_j A[k,j] * b[j] <= M
        # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
