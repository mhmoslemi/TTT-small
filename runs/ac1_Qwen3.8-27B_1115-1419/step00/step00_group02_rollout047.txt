```python
import time

def compute_convolution(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    size = 2 * n
    F = np.fft.rfft(f, size=size)
    return np.fft.irfft(F * F, size=size)[:2*n-1]

def subgradient_descent(f, lr, n_iters, maintain_sum=True):
    """Subgradient descent to minimize max(f*f) subject to f >= 0, sum(f) = S."""
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_obj = 2 * n * np.max(compute_convolution(f)) / (S**2)
    
    for it in range(n_iters):
        c = compute_convolution(f)
        k_star = np.argmax(c)
        # Subgradient of max(f*f) wrt f[i]
        # sg[i] = 2 * f[k_star - i] if 0 <= k_star - i < n, else 0
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            sg[i] = 2 * f[k_star - i]
        
        # Also add gradient of constraint (to maintain sum)
        # Use a simple approach: after update, rescale
        f = f - lr * sg
        f = np.maximum(f, 0)
        if maintain_sum:
            f = f * (S / np.sum(f)) if np.sum(f) > 0 else f
        
        obj = 2 * n * np.max(compute_convolution(f)) / (S**2)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
    
    return best_f, best_obj
```

```python
import numpy as np
import time

def fft_convolve(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 2 * n
    F = np.fft.rfft(f, size=size)
    result = np.fft.irfft(F * F, size=size)
    return result[:2*n-1]

def objective(f):
    """Compute the evaluation objective (assuming f is already valid)."""
    n = len(f)
    c = fft_convolve(f)
    max_b = np.max(c)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgrad_step(f, lr, step_type='sgd'):
    """One subgradient descent step."""
    n = len(f)
    S = np.sum(f)
    c = fft_convolve(f)
    k_star = np.argmax(c)
    
    # Compute subgradient
    sg = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    if lo < hi:
        sg[lo:hi] = 2.0 * f[k_star - (hi-1) : k_star - lo + 1][::-1]
    
    # Update
    f_new = f - lr * sg
    f_new = np.maximum(f_new, 0.0)
    
    # Rescale to maintain sum
    s_new = np.sum(f_new)
    if s_new > 0:
        f_new = f_new * (S / s_new)
    
    return f_new, c, k_star

def optimize_from_init(f_init, budget_s, seed=42):
    """Optimize starting from f_init with given time budget."""
    rng = np.random.default_rng(seed)
    f = f_init.copy()
    S = np.sum(f)
    best_f = f.copy()
    best_obj = objective(f)
    
    # Adaptive learning rate
    base_lr = 0.1
    lr = base_lr
    
    start_time = time.monotonic()
    no_improve_count = 0
    max_no_improve = 500
    
    while time.monotonic() - start_time < budget_s:
        f_new, c, k_star = subgrad_step(f, lr)
        obj = objective(f_new)
        
        if obj < best_obj:
            best_f = f_new.copy()
            best_obj = obj
            no_improve_count = 0
            # Maybe increase lr slightly
            lr = np.min(0.5, lr * 1.01)
        else:
            no_improve_count += 1
            # Decrease lr
            lr = np.max(0.001, lr * 0.99)
        
        f = f_new
        
        if no_improve_count > max_no_improve:
            # Restart from best with random perturbation
            f = best_f + rng.normal(0, 0.01, size=best_f.shape)
            f = np.maximum(f, 0)
            f = f * (S / np.sum(f))
            no_improve_count = 0
            lr = base_lr
    
    return best_f, best_obj
```

```python
import numpy as np
import time

def fft_convolve(f):
    n = len(f)
    F = np.fft.rfft(f, 2*n)
    return np.fft.irfft(F * F, 2*n)[:2*n-1]

def compute_grad_smooth(f, alpha):
    """Compute smooth gradient of soft-max objective."""
    n = len(f)
    c = fft_convolve(f)
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    soft_max_val = c_max + np.log(np.sum(exp_shifted)) / alpha
    
    # Cross-correlation: grad[i] = 2 * sum_j w[i+j] * f[j]
    # = 2 * (f_rev * w)[n-1+i]
    f_rev = f[::-1]
    N = 4 * n
    F = np.fft.rfft(f_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(F * W, N)
    grad = 2.0 * conv[n-1 : 2*n-1]
    
    return soft_max_val, grad

def soft_max_optimize(f_init, budget_s, seed=42):
    """Optimize using soft-max gradient descent."""
    f = f_init.copy()
    n = len(f)
    S = np.sum(f)
    
    best_f = f.copy()
    best_obj = 2.0 * n * np.max(fft_convolve(f)) / (S*S)
    
    alpha = 10.0  # Start with moderate alpha
    lr = 0.5 / alpha  # Learning rate scaled with alpha
    
    start = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.monotonic() - start < budget_s:
        val, grad = compute_grad_smooth(f, alpha)
        
        # Gradient of the normalized objective
        # We minimize max(f*f) subject to sum(f) = S, f >= 0
        # Use simple approach: subtract gradient, project, rescale
        momentum = mu * momentum + (1-mu) * grad
        f = f - lr * momentum
        f = np.maximum(f, 0)
        
        s = np.sum(f)
        if s > 0:
            f = f * (S / s)
        
        # Compute true objective
        obj = 2.0 * n * np.max(fft_convolve(f)) / (S*S)
        
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
        
        # Adapt alpha (increase over time for sharper max approximation)
        alpha = np.min(1000.0, alpha * 1.001)
        lr = 0.5 / alpha
    
    return best_f, best_obj
```

```python
"""
Optimize non-negative sequence to minimize 2*n*max(f*f)/(sum(f))^2 using:
- Multi-start diverse initializations (geometric, random, structured)
- Soft-max gradient descent with FFT-based convolution and momentum
- Periodic restarts from best solution with perturbation
- Final evaluation with exact evaluate_sequence

import numpy as np
import time


def fft_convolve_square(f):
    """Compute f * f (full linear convolution) using FFT."""
    n = len(f)
    size = 2 * n
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def eval_fast(f):
    """Fast evaluation assuming f is valid non-negative."""
    n = len(f)
    c = fft_convolve_square(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf, c
    return 2.0 * n * c.max() / (s * s), c


def compute_soft_gradient(f, alpha):
    """Compute soft-max gradient of max(f*f) using FFT cross-correlation."""
    n = len(f)
    c = fft_convolve_square(f)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n), c_max
    shifted = alpha * (c - c_max)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), c_max
    w /= w_sum

    # grad[i] = 2 * sum_j w[i+j] * f[j]
    # = 2 * (f_rev * w)[n-1+i]  where f_rev = f[::-1]
    f_rev = f[::-1]
    N = 4 * n
    Ff = np.fft.rfft(f_rev, N)
    Fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(Ff * Fw, N)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, c_max

    # Symmetric random
    for _ in range(6):
        half = rng.exponential(1.0, n // 2)
        if n % 2 == 0:
            f = np.concatenate([half, half[::-1]])
        else:
            f = np.concatenate([half, [half[n // 2 // 2]], half[::-1]])
            f = f[:n]
        inits.append(f / f.sum())

    # Two-hump
    f = np.zeros(n)
    f[:n // 3] = 1.0
    f[2 * n // 3:] = 1.0
    inits.append(f / f.sum())

    # Three-hump
    f = np.zeros(n)
    f[:n // 4] = 1.0
    f[n // 2 - n // 8:n // 2 + n // 8] = 1.0
    f[3 * n // 4:] = 1.0
    inits.append(f / f.sum())

    # Random binary-like
    for _ in range(6):
        f = (rng.random(n) > 0.3).astype(np.float64)
        if f.sum() > 0:
            inits.append(f / f.sum())

    return inits


def optimize_softmax(f_init, budget_s, seed=42):
    """
    Soft-max gradient descent with momentum to minimize max(f*f)
    subject to f >= 0, sum(f) = S.
    """
    rng = np.random.default_rng(seed)
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = f.sum()

    best_f = f.copy()
    best_obj, _ = eval_fast(f)

    # Parameters
    alpha = 5.0
    lr = 0.8
    beta = 0.85
    momentum_buf = np.zeros(n, dtype=np.float64)

    t0 = time.monotonic()
    stall_count = 0
    max_stall = 300
    it = 0

    while time.monotonic() - t0 < budget_s:
        # Compute gradient
        grad, c_max = compute_soft_gradient(f, alpha)

        # Momentum update
        momentum_buf = beta * momentum_buf + (1 - beta) * grad
        f = f - lr * momentum_buf
        f = project_simplex(f, S)

        # Evaluate true objective
        obj, _ = eval_fast(f)

        if np.isfinite(obj) and obj < best_obj:
            best_f = f.copy()
            best_obj = obj
            stall_count = 0
        else:
            stall_count += 1

        # Adapt learning rate and alpha
        if stall_count > 50:
            lr *= 0.9
            alpha = min(500.0, alpha * 1.05)
        elif stall_count > max_stall:
            # Restart from best with perturbation
            f = best_f + rng.normal(0, 0.005, n)
            f = project_simplex(f, S)
            momentum_buf = np.zeros(n)
            lr = 0.8
            alpha = 5.0
            stall_count = 0

        it += 1

        # Time check every 1000 iterations
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 0:
                print(f"  iter={it}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s, lr={lr:.4f}, alpha={alpha:.1f}")

    return best_f, best_obj


def optimize_subgrad_direct(f_init, budget_s, seed=42):
    """
    Direct subgradient descent (no soft-max) with adaptive step size.
    More aggressive but potentially less stable.
    """
    rng = np.random.default_rng(seed)
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = f.sum()

    best_f = f.copy()
    best_obj, _ = eval_fast(f)

    lr = 0.5
    beta = 0.9
    momentum_buf = np.zeros(n, dtype=np.float64)

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < budget_s:
        c = fft_convolve_square(f)
        k_star = int(np.argmax(c))

        # Subgradient
        sg = np.zeros(n, dtype=np.float64)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        if lo < hi:
            sg[lo:hi] = 2.0 * f[k_star - (hi - 1):k_star - lo + 1][::-1]

        # Momentum
        momentum_buf = beta * momentum_buf + (1 - beta) * sg
        f = f - lr * momentum_buf
        f = project_simplex(f, S)

        obj, _ = eval_fast(f)
        if np.isfinite(obj) and obj < best_obj:
            best_f = f.copy()
            best_obj = obj
            stall = 0
        else:
            stall += 1

        if stall > 100:
            lr *= 0.95
        if stall > 500:
            f = best_f + rng.normal(0, 0.003, n)
            f = project_simplex(f, S)
            momentum_buf *= 0.3
            lr = 0.5
            stall = 0

        it += 1
        if it % 2000 == 0:
            print(f"  [subgrad] iter={it}, obj={best_obj:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def try_parent_sequence(budget_s):
    """Try optimizing from the given parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            return None
        parent = np.clip(parent, 0, 1000)
        if not np.isfinite(parent).all() or parent.sum() < 0.01:
            return None
        # Normalize
        S = parent.sum()
        f = parent / S
        best_f, best_obj = optimize_softmax(f, budget_s, seed=99)
        return best_f, best_obj
    except (NameError, AttributeError, Exception):
        return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-start optimization for minimizing 2*n*max(f*f)/(sum(f))^2.
    
    Strategy:
    1. Try diverse initializations (geometric, random, structured)
    2. For each, run soft-max gradient descent with momentum
    3. Track global best across all runs
    4. Allocate remaining time to polishing the best solution
    5. Return best sequence (validated with evaluate_sequence)
    """
    total_start = time.monotonic()
    total_budget = budget_s - 5.0  # Leave 5s for final validation

    # Determine sequence length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
        else:
            n = 2048
    except (NameError, AttributeError, Exception):
        n = 2048

    # Cap n for feasibility
    n = min(n, 20480)
    n = max(n, 256)

    print(f"Sequence length: {n}, total budget: {total_budget:.1f}s")

    global_best_f = None
    global_best_obj = np.inf

    # Phase 1: Quick scan of initializations (10% of budget)
    phase1_budget = total_budget * 0.10
    inits = generate_initializations(n)
    print(f"Generated {len(inits)} initializations")

    # Quick evaluate all inits
    for i, f_init in enumerate(inits):
        S = f_init.sum()
        f_norm = f_init / S
        obj, _ = eval_fast(f_norm)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_f = f_norm.copy()

    # Try the parent sequence if available
    parent_result = try_parent_sequence(min(10.0, total_budget * 0.05))
    if parent_result is not None:
        pf, pobj = parent_result
        if pobj < global_best_obj:
            global_best_obj = pobj
            global_best_f = pf.copy()
        print(f"Parent sequence optimized to: {pobj:.6f}")

    print(f"Best after initialization scan: {global_best_obj:.6f}")

    remaining = total_budget - (time.monotonic() - total_start)

    # Phase 2: Optimize top initializations (70% of budget)
    phase2_budget = remaining * 0.70
    # Sort inits by initial objective
    init_objs = []
    for i, f_init in enumerate(inits):
        f_norm = f_init / f_init.sum()
        obj, _ = eval_fast(f_norm)
        init_objs.append((obj, i))
    init_objs.sort()

    # Deduplicate (skip inits with similar objective to already-tried ones)
    tried_count = 0
    n_to_try = min(8, len(init_objs))
    time_per_init = phase2_budget / n_to_try

    for rank in range(n_to_try):
        elapsed_total = time.monotonic() - total_start
        if elapsed_total > total_budget - 30:
            break
        _, idx = init_objs[rank]
        f_init = inits[idx] / inits[idx].sum()
        init_obj, _ = eval_fast(f_init)
        if init_obj >= global_best_obj * 1.5:
            continue  # Skip obviously bad inits

        allocated = min(time_per_init, total_budget - 30 - (time.monotonic() - total_start))
        if allocated < 2.0:
            break

        print(f"Optimizing init {rank} (obj={init_obj:.6f}) with {allocated:.1f}s budget...")
        result_f, result_obj = optimize_softmax(f_init, allocated, seed=seed + rank)

        if result_obj < global_best_obj:
            global_best_obj = result_obj
            global_best_f = result_f.copy()
            print(f"  New best: {global_best_obj:.6f}")

    remaining = total_budget - (time.monotonic() - total_start)

    # Phase 3: Polish the best solution (remaining time minus safety margin)
    if global_best_f is not None and remaining > 10:
        phase3_budget = remaining - 10
        print(f"Phase 3: Polishing best (obj={global_best_obj:.6f}) with {phase3_budget:.1f}s")

        # Use subgradient direct method for polishing
        result_f, result_obj = optimize_subgrad_direct(global_best_f, phase3_budget * 0.5, seed=777)
        if result_obj < global_best_obj:
            global_best_obj = result_obj
            global_best_f = result_f.copy()

        remaining2 = phase3_budget - (time.monotonic() - (total_start + total_budget - phase3_budget * 0.5 - phase3_budget))
        # Simpler: just use remaining time
        elapsed_total = time.monotonic() - total_start
        if total_budget - elapsed_total > 5:
            result_f, result_obj = optimize_softmax(global_best_f, total_budget - elapsed_total - 3, seed=888)
            if result_obj < global_best_obj:
                global_best_obj = result_obj
                global_best_f = result_f.copy()

    # Final validation with evaluate_sequence
    if global_best_f is not None:
        final_seq = [float(x) for x in global_best_f]
        final_score = float(evaluate_sequence(final_seq))
        print(f"Final score (evaluate_sequence): {final_score:.6f}")

        # If evaluate_sequence gives a worse score (due to clipping etc),
        # try the raw best
        if not np.isfinite(final_score):
            # Fallback to uniform
            final_seq = [1.0 / n] * n
            final_score = float(evaluate_sequence(final_seq))
            print(f"Fallback to uniform: {final_score:.6f}")
    else:
        final_seq = [1.0 / n] * n
        final_score = float(evaluate_sequence(final_seq))
        print(f"Default uniform: {final_score:.6f}")

    elapsed = time.monotonic() - total_start
    print(f"Total time: {elapsed:.1f}s, final score: {final_score:.6f}")

    return final_seq
```
