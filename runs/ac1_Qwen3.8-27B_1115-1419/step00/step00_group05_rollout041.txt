def simplex_project(v, target=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - target) / rho
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val-1] - target) / rho_val
    return np.maximum(v - theta, 0)

    # Numerically stable softmax
    max_conv = np.max(conv)
    exp_conv = np.exp(alpha * (conv - max_conv))
    weights = exp_conv / np.sum(exp_conv)
    return weights

import numpy as np
import time

def fast_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]

def simplex_project(v, target=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - target) / rho
    if not np.any(cond):
        return np.full(n, target / n)
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - theta, 0.0)

def compute_score_fast(a):
    """Compute score without calling evaluate_sequence (for fast local evaluation)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(a, a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (s * s)

def compute_gradient(a, alpha):
    """Compute gradient of (1/alpha)*log(sum(exp(alpha*conv(a,a)))) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a, a)
    # Softmax weights (numerically stable)
    conv_shifted = conv - np.max(conv)
    exp_conv = np.exp(alpha * conv_shifted)
    weights = exp_conv / np.sum(exp_conv)
    # Gradient: grad[i] = sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Using FFT as derived above
    w_rev = weights[::-1]
    conv2 = fast_convolve(a, w_rev)

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using smooth gradient descent
on the probability simplex. Uses FFT-based convolution for efficiency,
log-sum-exp smoothing of the max function, and multiple shape initializations.
"""


def fast_convolve(a, b):
    """FFT-based linear convolution of two 1D arrays."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    if size < 32:
        return np.convolve(a, b)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]


def simplex_project(v, target=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - target) / rho
    if not np.any(cond):
        return np.full(n, target / n)
    rho_val = int(np.max(rho[cond]))
    theta = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_score_fast(a):
    """Compute the score 2*n*max(conv)/(sum)^2 efficiently."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(a, a)
    max_conv = float(np.max(conv))
    return 2.0 * n * max_conv / (s * s)


def compute_gradient(a, alpha):
    """
    Compute gradient of (1/alpha)*log(sum(exp(alpha * conv(a,a)))) w.r.t. a.
    Uses FFT for both the convolution and the cross-correlation.
    """
    n = len(a)
    conv = fast_convolve(a, a)
    conv_max = float(np.max(conv))
    exp_conv = np.exp(alpha * (conv - conv_max))
    weights = exp_conv / float(np.sum(exp_conv))
    # Gradient: grad[i] = sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Computed via FFT: grad[i] = conv2[2n-2-i] where conv2 = convolve(a, weights[::-1])
    w_rev = weights[::-1]
    conv2 = fast_convolve(a, w_rev)
    grad = conv2[n - 1: 2 * n - 1][::-1]
    return grad / alpha

    """Create a normalized non-negative starting sequence of length n."""

        a = np.arange(1, n + 1, dtype=np.float64)
    elif shape == 'ramp2':
        a = np.arange(1, n + 1, dtype=np.float64) ** 2
    elif shape == 'exp':
        a = np.exp(np.linspace(0, 2.0, n))
    elif shape == 'cos_u':
        a = 1.0 + np.cos(2.0 * np.pi * np.arange(n) / n)
        a = np.maximum(a, 0.0)
    elif shape == 'sin':
        a = 1.0 + 0.8 * np.sin(np.pi * np.arange(n) / (n - 1))
    elif shape == 'vshape':
        mid = (n - 1) / 2.0
        a = np.abs(np.arange(n) - mid) + 1.0
    elif shape == 'random':
        a = rng.exponential(1.0, n)
    elif shape == 'bimodal':
        a = np.exp(-((np.arange(n) - n * 0.2) ** 2) / (2 * (n * 0.08) ** 2)) + \
            np.exp(-((np.arange(n) - n * 0.8) ** 2) / (2 * (n * 0.08) ** 2))
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def gradient_descent(a, deadline, rng, alpha_init=20.0, lr_init=0.1):
    """
    Run smooth gradient descent to minimize the score.
    Returns the best sequence and score found.
    """
    n = len(a)
    alpha = alpha_init
    lr = lr_init
    best_a = a.copy()
    best_score = compute_score_fast(a)
    current = a.copy()
    stall_count = 0
    max_stall = 200

        step = current - lr * grad
        step = simplex_project(step, 1.0)

            step = np.maximum(step + pert, 0)
            ss = step.sum()
            if ss > 0.01:
                step = step / ss

        score = compute_score_fast(step)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = step.copy()
            current = step
            stall_count = 0
            lr = max(0.001, lr * 0.997)
            alpha = min(alpha * 1.003, 500.0)
        else:
            stall_count += 1
            lr = max(0.0005, lr * 0.95)
