import numpy as np
import time

def fast_evaluate(seq):
    """Fast evaluation mimicking evaluate_sequence logic."""
    seq = np.array(seq, dtype=np.float64)
    seq = np.clip(seq, 0, 1000)
    n = len(seq)
    if n == 0:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def peak_reduction_perturb(seq, rng, intensity=0.1):
    """Reduce the peak by targeting its contributors."""
    seq = seq.copy()
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_star = np.argmax(conv)
    
    # Find contributors to the peak
    # c[k*] = sum_j a[j] * a[k*-j] for valid j
    j_start = max(0, k_star - (n - 1))
    j_end = min(k_star, n - 1)
    
    products = np.zeros(n)
    for j in range(j_start, j_end + 1):
        products[j] = seq[j] * seq[k_star - j]
    
    total_product = products.sum()
    if total_product < 1e-15:
        return seq
    
    # Reduce top contributors
    top_k = max(1, n // 10)
    top_indices = np.argsort(products)[-top_k:]
    
    # Reduce these and add to random other positions
    mass = 0.0
    for j in top_indices:
        if products[j] > 0:
            reduction = seq[j] * intensity * (products[j] / total_product)
            seq[j] -= reduction
            mass += reduction
    
    # Distribute mass to random positions
    if mass > 0:
        target_indices = rng.integers(0, n, size=min(5, n))
        for idx in target_indices:
            seq[idx] += mass / len(target_indices)
    
    return np.clip(seq, 0, 1000)

"""
Step-function minimizer via peak-directed simulated annealing.
Optimizes the 2*n*max(auto-conv)/sum^2 objective by iteratively
reducing convolution peaks through targeted mass redistribution,
with multiple random restarts and adaptive temperature cooling.
"""

import numpy as np


def _score(seq):
    """Fast equivalent of evaluate_sequence on a numpy array."""
    s = np.clip(seq, 0.0, 1000.0)
    n = len(s)
    if n < 1:
        return np.inf
    conv = np.convolve(s, s)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(s))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _initial_candidates(n, rng):
    """Generate diverse initial sequences of length n."""
    cands = []
    u = np.ones(n)
    cands.append(u.copy())
    cands.append(np.minimum(1000.0, np.abs(rng.normal(1.0, 0.5, n))).copy())
    for _ in range(8):
        cands.append(np.clip(rng.random(n) * 1.5, 0, 1000.0).copy())
        cands.append(np.clip(0.5 + rng.random(n), 0, 1000.0).copy())
    return cands


def _peak_contrib(seq, k):
    """Computationally efficient peak contribution extraction."""
    n = len(seq)
    js = range(max(0, k - n + 1), min(k, n - 1) + 1)
    pairs = [(j, min(k - j, n - 1 - (k - j)) if False else k - j) for j in js]
    prods = np.array([seq[j] * seq[k - j] for j, _ in pairs], dtype=np.float64)
    return prods


def _perturb(seq, rng, temp):
    """Combined perturbation: peak-directed + random block noise."""
    n = len(seq)
    s = seq.copy()
    conv = np.convolve(s, s)
    k_star = int(np.argmax(conv))
    pk = float(conv[k_star])

    # Directed peak reduction
    js = range(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
    prods = [(j, s[j] * s[k_star - j]) for j in js]
    total = sum(p for _, p in prods)
    if total > 1e-18 and pk > 0:
        n_red = max(1, min(6, len(prods) // 2))
        prods_sorted = sorted(range(len(prods)), key=lambda ii: -prods[ii][1])
        mass = 0.0
        for ii in prods_sorted[:n_red]:
            j = prods[ii][0]
            red = s[j] * 0.15 * (prods[ii][1] / total)
            if s[j] - red > 1e-15:
                s[j] -= red
                mass += red
        if mass > 0:
            nz = np.where(s < 1000)[0]
            if len(nz) > 0:
                tidx = rng.choice(nz, size=min(6, len(nz)), replace=False)
                per = mass / len(tidx)
                for idx in tidx:
                    s[idx] = min(s[idx] + per, 1000.0)

    # Random block multiplicative perturbation
    bw = max(1, n // (4 + int(2 * temp)))
    st = int(rng.integers(0, max(1, n - bw)))
    s[st:st + bw] *= np.exp(rng.normal(0, 0.08 * temp, min(bw, n - st)))

    s = np.clip(s, 0.0, 1000.0)
    s = s / max(np.sum(s), 1e-18)
    return s


def _anneal(seq, rng, t_alloc, t_start):
    """Simulated annealing on a single starting sequence."""
    n = len(seq)
    seq = seq.astype(np.float64).copy()
    cur_sc = _score(seq)
    best = seq.copy()
    best_sc = cur_sc
    if not np.isfinite(cur_sc):
        cur_sc = 100.0

    acceptance = 0
    evals = 0
    nrecent = 0
    t = t_start
    t_min = 0.01
    cooling = 230.0 / t_alloc
    elapsed = 0.0

    while elapsed < t_alloc - 0.5:

        trial_sc = _score(trial)
        evals += 1

            best_sc = trial_sc
            seq = trial.copy()
            cur_sc = trial_sc
            acceptance += 1
        elif np.isfinite(trial_sc) and trial_sc < cur_sc:

            acceptance += 1

                    cur_sc = trial_sc
