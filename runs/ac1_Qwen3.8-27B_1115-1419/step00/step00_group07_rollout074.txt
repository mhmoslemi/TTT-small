    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fc = fa * fa
    c = np.fft.irfft(fc, size)[:2*n-1]
    return c

    M = c.max()

def _fft_conv(a):
    """Linear convolution of a with itself via real FFT."""
    n = a.size
    m = 2 * n - 1

    F = np.fft.rfft(a, sz)

    """Transfer delta from smaller to larger in each pair (i, k-i) for valid pairs."""
    n = len(a)

def fft_conv_self(a):
    """Linear autocovolution: c[k] = sum_i a[i]*a[k-i], length 2n-1."""
    n = a.size
    m = 2 * n - 1
    sz = 1 << (m - 1).bit_length()
    F = np.fft.rfft(a, sz)
    return np.fft.irfft(F * F, sz)[:m]

def fast_eval(a):
    """2n*max(c)/S^2 without guards."""
    S = a.sum()

    return 2.0 * a.size * cmax / (S * S)
