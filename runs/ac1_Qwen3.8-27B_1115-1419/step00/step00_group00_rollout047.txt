Minimizes the peak-to-average autocorrelation ratio using an LP-based
iterative method combined with subgradient refinement.

The key idea: at each step, solve an LP to find a direction g that
maximizes total energy subject to the cross-correlation (f*g) being
bounded by the current peak M. This gives a descent direction for
the peak autocorrelation.

import numpy as np
from scipy.sparse import csc_matrix
from scipy.optimize import linprog
import time


def build_matrix_indices(n):
    """Precompute row and column indices for the banded convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        rows.extend(range(k, k + 1) * (j_max - j_min + 1))
        cols.extend(range(j_min, j_max + 1))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def compute_objective(a):
    """Compute the objective function value."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

    """

    Returns the optimal g or None if infeasible.
    """
    n = len(f)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method=solver)

    return None

the peak-to-average autocorrelation ratio of a non-negative sequence.

Algorithm:
1. Initialize with multiple candidate sequences (flat, given, random)

3. Supplement with subgradient steps for fine-tuning.

import numpy as np
from scipy.sparse import csc_matrix

def build_sparse_structure(n):
    """Build the row/col indices for the banded gradient matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        rows.extend([k] * (j_max - j_min + 1))
        cols.extend(range(j_min, j_max + 1))
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)

def compute_b(a):
    """Compute the full self-convolution."""
    return np.convolve(a, a)

def compute_obj(a):
    """Compute the objective."""
    n = len(a)
    b = compute_b(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(b) / (s * s)

def solve_slp_lp(a, row_idx, col_idx, c_vec):
    """
    Solve the SLP subproblem.
    a: current point (sum to 1, non-negative)
    row_idx, col_idx: sparsity structure of the gradient matrix
    c_vec: constant vector for the constraints
    
    Returns (a_new, t_new) or None.
    """
    n = len(a)
    # Build gradient matrix G
    # G[k,j] = (1 + delta_{k,2j}) * a[k-j] if 0<=k-j<n
    data = a[col_idx].copy()
    # Apply the (1 + delta) factor for even k where k=2j
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        j = col_idx[idx]
        if k == 2 * j:
            data[idx] *= 2.0
    
    G = csc_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    
    # LP: minimize t

    #             sum(a) = 1
    #             a >= 0
    #             t free (but will be positive)
    
    # Variables: [a_0, ..., a_{n-1}, t]
    # Objective: [0, ..., 0, 1] (minimize t)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: G*a - t <= -c_vec
    # A_ub has shape (2n-1, n+1)
    # A_ub[k, j] = G[k,j] for j < n, A_ub[k, n] = -1
    A_ub_data = np.concatenate([data, -np.ones(2*n-1)])
    A_ub_row = np.concatenate([row_idx, np.arange(2*n-1)])
    A_ub_col = np.concatenate([col_idx, np.full(2*n-1, n)])
    A_ub = csc_matrix((A_ub_data, (A_ub_row, A_ub_col)), shape=(2*n-1, n+1))
    b_ub = -c_vec
    
    # Equality constraint: sum(a) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a_j >= 0 for j < n, t free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:

    return None

SLP (Sequential Linear Programming) optimizer for minimizing the peak
auto-correlation-to-average ratio of a non-negative sequence.

Primary method: At each step, linearize the autocorrelation constraints
at the current point, solve the resulting LP to find a descent direction,
and use backtracking to ensure the objective decreases.
Supplemented with subgradient steps and periodic perturbations.
"""

import numpy as np

def build_banded_indices(n):
    """Build row/col indices for the (2n-1) x n banded convolution matrix."""
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int64)
    cols = np.empty(total_nnz, dtype=np.int64)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_start, j_end + 1)
        idx += count
    return rows, cols


def compute_conv(a):
    """Full self-convolution of a."""
    return np.convolve(a, a)


def compute_obj(a):
    """Compute the objective: 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    b = compute_conv(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(b) / (s * s)


def solve_slp(a, row_idx, col_idx):
    """
    Solve the SLP subproblem at point a (assumed sum to 1, non-negative).
    Returns (a_new, t_new) or None.
    """
    n = len(a)
    
    # Build gradient matrix data: G[k,j] = (1+delta_{k,2j}) * a[k-j]
    data = a[col_idx].copy()
    # Apply doubling factor where k == 2j
    mask = (row_idx == 2 * col_idx)
    data[mask] *= 2.0
    
    # Build A_ub: [G | -1] with shape (2n-1, n+1)
    A_ub_data = np.empty(len(data) + (2 * n - 1), dtype=np.float64)
    A_ub_data[:len(data)] = data
    A_ub_data[len(data):] = -1.0
    
    A_ub_row = np.empty(len(data) + (2 * n - 1), dtype=np.int64)
    A_ub_row[:len(data)] = row_idx
    A_ub_row[len(data):] = np.arange(2 * n - 1)
    
    A_ub_col = np.empty(len(data) + (2 * n - 1), dtype=np.int64)
    A_ub_col[:len(data)] = col_idx
    A_ub_col[len(data):] = n  # column index for t variable
    
    A_ub = csc_matrix((A_ub_data, (A_ub_row, A_ub_col)), shape=(2 * n - 1, n + 1))
    
    # Constants: c_k = b_k - sum_j G[k,j]*a[j]
    # For k != 2j for all valid j: c_k = 0
    # For k = 2*j*: c_k = a[j*]^2
    b = compute_conv(a)
    c_vec = np.zeros(2 * n - 1)
    for j in range(n):
        k = 2 * j
        if k < 2 * n - 1:
            c_vec[k] = a[j] * a[j]
    
    # b_ub = -c_vec
    b_ub = -c_vec
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(a) = 1
    A_eq = np.hstack([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0.0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                  options={'presolve': True, 'time_limit': 30})
    
    if res.status == 0:

        s = np.sum(a_new)

def subgradient_update(a, b, alpha):
    """One subgradient step targeting the peak of the autocorrelation."""
    n = len(a)
    k_star = int(np.argmax(b))
    s = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    for j in range(j_min, j_max + 1):
        val = a[k_star - j]
        if k_star == 2 * j:
            val *= 2.0
        s[j] = val
    
    a_new = a - alpha * s
    a_new = np.maximum(a_new, 0.0)
    s_sum = np.sum(a_new)
    if s_sum > 1e-10:
        a_new = a_new / s_sum
    else:
        a_new = np.ones(n) / n
    return a_new
