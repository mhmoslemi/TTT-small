def _fast_convolve(a):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _fast_xcorr(w, a):
    """Cross-correlation: result[j] = sum_k w[k+j]*a[k] for j=0..len(a)-1."""
    n = len(a)
    N = 1
    while N < len(w) + n - 1:
        N *= 2
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(W * np.conj(A), N)[:n]
