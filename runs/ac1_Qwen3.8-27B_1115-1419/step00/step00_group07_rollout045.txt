import numpy as np
import time

def compute_autocorrelation(a):
    """Compute a * a using FFT for efficiency."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2*n-1]

def gradient_step(a, lr, peak_fraction=0.1):
    """One subgradient step to reduce max autocorrelation peak."""
    c = compute_autocorrelation(a)
    n = len(a)
    # Find peaks above certain threshold
    threshold = np.percentile(c, 100*(1-peak_fraction))
    peak_indices = np.where(c >= threshold)[0]
    
    # Compute subgradient as average of gradients at peak positions
    g = np.zeros(n)
    for k in peak_indices:
        # d(c_k)/d(a_i) = 2*a[k-i] if 0 <= k-i < n
        shift = k - np.arange(n)
        valid = (shift >= 0) & (shift < n)
        g[valid] += 2.0 * a[shift[valid]]
    g /= len(peak_indices)
    
    # Projected gradient step
    a_new = a - lr * g
    a_new = np.clip(a_new, 0, 1000)
    
    # Renormalize
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new * (np.sum(a) / s)
    
    return a_new

    """Compute a * a via FFT."""
    n = len(a)
    size = 1

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

    """Compute b_i = sum_k w_k * a_{k-i} for i=0..n-1 using FFT."""
