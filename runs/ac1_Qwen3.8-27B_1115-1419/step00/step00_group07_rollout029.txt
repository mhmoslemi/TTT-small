3. Multiple IWM restarts with different parameters

def fast_objective(a):

            r_new = r - excess * (1.0 - np.exp(-current_beta * excess / r_max))

        r_new = np.maximum(r_new, 0.0)

def local_search(a, time_limit=60.0, n_trials_per_iter=50):
    """
    Local search: random coordinate descent to fine-tune the sequence.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    best_a = a.copy()
    best_obj = float(evaluate_sequence(best_a.tolist()))
    
    rng = np.random.default_rng(12345)
    start = time.monotonic()
    
    while time.monotonic() - start < time_limit:
        # Choose perturbation type
        mode = rng.random()
        
        if mode < 0.3:
            # Single coordinate scaling
            i = int(rng.integers(0, n))
            factor = np.exp(rng.normal(0, 0.1))
            candidate = a.copy()
            candidate[i] *= factor
            candidate = np.maximum(candidate, 0.0)
            
        elif mode < 0.6:
            # Two-coordinate exchange
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            amount = rng.uniform(-0.1, 0.1) * a[i]
            amount = max(amount, -a[i] * 0.5)  # Don't go negative
            candidate = a.copy()
            candidate[i] -= amount
            candidate[j] += amount
            candidate = np.maximum(candidate, 0.0)
            
        else:
            # Block perturbation
            width = int(rng.integers(2, max(3, n // 100)))
            start_i = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start_i:start_i+width] *= np.exp(rng.normal(0, 0.05, size=width))
            candidate = np.maximum(candidate, 0.0)
        
        # Evaluate
        obj = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_a = candidate.copy()
            a = candidate.copy()
    
    return best_a, best_obj

def fast_obj(a):
    """Fast objective computation (no Python list overhead)."""
    n = len(a)
    s = np.sum(a)

    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)
