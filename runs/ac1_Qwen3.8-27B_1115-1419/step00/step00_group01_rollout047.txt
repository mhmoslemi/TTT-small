Let me think about the LP approach more carefully. The LP is:
maximize sum(g)
subject to (f*g)(k) <= M for all k = 0, ..., 2n-2
g >= 0

Where f is the current sequence and M = max(f*f).

This LP has 2n-1 constraints (one for each lag) and n variables. The constraint matrix is a Toeplitz matrix (the convolution matrix of f).

If we solve this LP, we get a direction g. The idea is that g "spreads out" more than f while being "compatible" with f (in the sense that f*g is bounded).

When we update f → (1-t)f + tg, the new self-convolution is:
((1-t)f + tg)^2 = (1-t)^2 f^2 + 2t(1-t) f*g + t^2 g^2

We know f*g <= M = max(f^2). For small t:
max of the new convolution ≈ (1-t)^2 * max(f^2) + 2t(1-t) * M
= (1-t)^2 * M + 2t(1-t) * M
= M * (1-t)^2 + 2t(1-t) * M
= M * (1 - 2t + t^2 + 2t - 2t^2)
= M * (1 - t^2)

So for small t, the max decreases quadratically! That's a nice property.

But we also need g^2 to not blow up. The LP doesn't constrain g^2 directly. So we need to be careful with the step size.

import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimizes 2*n*max(convolve(p,p))/sum(p)^2 by finding a probability vector p
    with minimal self-convolution peak. Uses LP-based direction finding,
    gradient descent, and random restarts across multiple sequence lengths.
    """
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(seed)
    best_score = np.inf
    best_p = None
    best_n = 0
    
    def obj(p):
        """Objective: 2*n*max(conv(p,p)) for p with sum=1."""
        return 2.0 * len(p) * np.max(np.convolve(p, p))
    
    def lp_direction(p, M):
        """Solve LP to find best direction g."""
        n = len(p)
        A = np.zeros((2*n - 1, n))
        for j in range(n):
            A[j:j+n, j] = p
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True, 'time_limit': 2.0})
            if res.success:
                return res.x
        except:
            pass
        return None
    
    def gradient_direction(p, M, top_k=5):
        """Compute a gradient-based direction."""
        n = len(p)
        conv = np.convolve(p, p)
        # Top k lags
        top_idx = np.argsort(conv)[-top_k:]
        # Weight by how much above the median
        weights = np.maximum(conv[top_idx] - np.median(conv), 1e-10)
        weights = weights / weights.sum()
        # Accumulate gradient
        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min <= i_max:
                grad[i_min:i_max+1] += w * 2.0 * p[k - np.arange(i_min, i_max + 1)]
        return grad
    
    def optimize_lp(p, max_iters=500):
        """LP-based optimization loop."""
        n = len(p)
        score = obj(p)
        no_improve = 0
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            conv = np.convolve(p, p)
            M = np.max(conv)
            g = lp_direction(p, M)
            if g is None:
                # Fallback to gradient
                grad = gradient_direction(p, M)
                eta = 0.1 / (1 + 0.01 * it)
                p_new = p * np.exp(-eta * grad)
                p_new = np.maximum(p_new, 1e-15)
                s = p_new.sum()
                if s < 1e-10:
                    break
                p_new /= s
                new_score = obj(p_new)
                if new_score < score:
                    p = p_new
                    score = new_score
                    no_improve = 0
                else:

                    if no_improve > 50:

import numpy as np
import time
from scipy.optimize import linprog

def convolve_peak(p):
    """Return max of linear convolution of p with itself."""
    return float(np.max(np.convolve(p, p)))

def objective(p):
    """Compute the objective: 2*n*max(conv(p,p)) for sum(p)=1."""
    return 2.0 * len(p) * convolve_peak(p)

def lp_solve(p, M):
    """Solve the LP: max sum(g) s.t. (p*g)(k) <= M, g >= 0."""
    n = len(p)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = p
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                     bounds=[(0, None)]*n, method='highs',
                     options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def subgrad_compute(p, top_k=3):
    """Compute weighted subgradient at top-k convolution peaks."""
    n = len(p)
    conv = np.convolve(p, p)
    # Find top-k indices
    top_idx = np.argsort(conv)[-top_k:]
    # Weight: proportional to excess over median
    med = np.median(conv)
    w = np.maximum(conv[top_idx] - med, 1e-12)
    w = w / w.sum()
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        k = int(idx)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += wi * 2.0 * p[k - indices]
    return grad, conv
