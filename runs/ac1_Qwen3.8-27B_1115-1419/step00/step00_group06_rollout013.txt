def fft_lag_corr(a, w):

def make_lag_weights(b, p, top_count, mix_top):
    b = np.asarray(b, dtype=np.float64)
    L = b.size
    if L ==0: return np.array([])
    b = np.where(np.isfinite(b), b, 0.0)
    maxb = float(np.max(b))
    if not np.isfinite(maxb) or maxb <= 0.0:
        w = np.ones(L, dtype=np.float64); w /= L; return w
    x = b / maxb
    if p < 50.0:
        soft = np.power(np.maximum(x, 0.0), p)
    else:
        tau = maxb / max(1.0, p)
        soft = np.exp((b - maxb) / tau)
    soft = np.where(np.isfinite(soft), soft, 0.0)
    soft += 1e-8
    s = float(np.sum(soft, dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        soft = np.ones(L); s=float(L)
    soft /= s
    k = int(top_count)
    if k <=0 or k >= L:
        top = np.ones(L)/float(L)
    else:
        # argpartition if k small; if k > L//2 maybe use complement?
        if k > L//2:
            idx = np.argsort(b)[-k:]  # but k large; not expected
        else:
            idx = np.argpartition(b, L - k)[-k:]
        top = np.zeros(L, dtype=np.float64)
        top[idx] = 1.0 / float(k)
    mix = float(np.clip(mix_top, 0.0, 1.0))
    w = mix * top + (1.0 - mix) * soft
    s = float(np.sum(w, dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        w = np.ones(L)/float(L)

def random_block_perturb(a, rng, strength=0.1, nblocks=1):
    a = normalize_positive(a)
    if a is None: return None
    n = a.size
    cand = a.copy()
    for _ in range(max(1,int(nblocks))):

def interpolate_sequence(a, n):
    a=normalize_positive(a)

    new=np.linspace(0.0,1.0,n,endpoint=False)
    b=np.interp(new, old, a)
    b=np.where(np.isfinite(b), b, 0.0)

import numpy as np
import time


"""
Searches for nonnegative step-function heights whose linear convolution has a
small peak.  The objective is scale-invariant, so all sequences are maintained
with sum 1 and score = 2*n*max(conv(a,a)).

Algorithm summary:
  - Use FFT quadratic/correlation helpers for fast objective and gradient
    computations.
  - For a current sequence, weight the largest convolution coefficients
    (soft power/logsum weights plus a top-K active set).
  - Form the log-space subgradient h_i = a_i * sum_k W_k a_{k-i}.
  - Apply multiplicative updates a_i <- a_i exp(-eta * standardized h_i)
    with an adaptive FFT line search, accepting only exact-score decreases.
  - Periodic random block perturbations and restarts escape local basins.
  - A few lengths are tried, including the provided construction when present,
    and the best FFT-score sequence is verified with evaluate_sequence().
"""


def next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def normalize_positive(x, target=1.0):
    try:
        y = np.asarray(x, dtype=np.float64)
    except Exception:
        return None
    if y.ndim != 1 or y.size == 0:
        return None
    y = y.copy()
    y[~np.isfinite(y)] = 0.0
    y = np.clip(y, 0.0, 1000.0)
    s = float(np.sum(y, dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        return None
    y = y * (float(target) / s)
    y = np.clip(y, 0.0, 1000.0)
    s2 = float(np.sum(y, dtype=np.float64))
    if not np.isfinite(s2) or s2 <= 0.0:
        return None
    y = y * (float(target) / s2)
    return y


def fft_convolution(x):
    n = int(x.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    N = next_pow2(m)
    fx = np.fft.rfft(x, N)
    c = np.fft.irfft(fx * fx, N)
    return c[:m]


def score_array(a):
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(np.sum(a, dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    b = fft_convolution(a)
    if b.size == 0:
        return float("inf")
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * n * max(0.0, mx) / (s * s))


def fft_lag_corr(a, w):
    """
    Returns c_i = sum_k w_k * a_{k-i} for i = 0...n-1, with a_j = 0 outside.
    """
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    w = np.asarray(w, dtype=np.float64)
    if w.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:min(L, w.size)] = w
        w = tmp
    else:
        w = w[:L]
    N = next_pow2(L)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    c = np.fft.irfft(fa * fw.conjugate(), N)
    return c[:n]


def make_lag_weights(b, p, top_count, mix_top):
    b = np.asarray(b, dtype=np.float64)
    L = int(b.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    b = np.where(np.isfinite(b), b, 0.0)
    maxb = float(np.max(b))
    if not np.isfinite(maxb) or maxb <= 0.0:
        w = np.ones(L, dtype=np.float64)
        w /= float(L)
        return w

    x = np.maximum(b / maxb, 0.0)
    p = float(p)
    if p < 50.0:
        soft = np.power(x, p)
    else:
        tau = maxb / max(1.0, p)
        soft = np.exp((b - maxb) / tau)
    soft = np.where(np.isfinite(soft), soft, 0.0)
    soft += 1e-8
    ss = float(np.sum(soft, dtype=np.float64))
    if not np.isfinite(ss) or ss <= 0.0:
        soft = np.ones(L, dtype=np.float64)
        ss = float(L)
    soft /= ss

    k = int(top_count)
    if k <= 0 or k >= L:
        top = np.ones(L, dtype=np.float64) / float(L)
    else:
        if k > L // 2:
            idx = np.argsort(b)[-k:]
        else:
            idx = np.argpartition(b, L - k)[-k:]
        top = np.zeros(L, dtype=np.float64)
        top[idx] = 1.0 / float(k)

    mix = float(np.clip(mix_top, 0.0, 1.0))
    w = mix * top + (1.0 - mix) * soft
    sw = float(np.sum(w, dtype=np.float64))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.ones(L, dtype=np.float64) / float(L)
    else:
        w /= sw
    return w


def apply_multiplicative(a, direction, eta, clip=4.0):
    if a is None:
        return None
    if not np.isfinite(eta) or eta <= 0.0:
        return None
    x = -(float(eta)) * np.asarray(direction, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -clip, clip)
    y = a * np.exp(x)
    return normalize_positive(y)


def random_block_perturb(a, rng, strength=0.1, nblocks=1):
    a = normalize_positive(a)
    if a is None:
        return None
    n = int(a.size)
    cand = a.copy()
    for _ in range(max(1, int(nblocks))):
        if n <= 1:
            break
        wmin = max(1, n // 300)
        wmax = max(2, n // 12)
        if wmin >= n:
            width = max(1, n // 2)
        else:
            hi = min(n, wmax + 1)
            if hi <= wmin:
                width = max(1, n // 4)
            else:
                width = int(rng.integers(wmin, hi))
                if width <= 0:
                    width = 1
        start = int(rng.integers(0, n - width + 1))
        stop = min(n, start + width)
        cand[start:stop] *= float(np.exp(rng.normal(0.0, max(1e-6, float(strength)))))
    return normalize_positive(cand)


def make_random_initial(n, rng):
    n = max(8, int(n))
    t = np.arange(n, dtype=np.float64) / float(n)
    y = np.zeros(n, dtype=np.float64)
    max_m = min(24, max(1, n // 16))
    for m in range(1, max_m + 1):
        amp = (0.7 * rng.random() + 0.05) / (float(m) ** 0.55)
        if rng.random() < 0.5:
            amp *= -1.0
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        y += amp * np.sin(2.0 * np.pi * float(m) * t + ph)
    y += 0.04 * rng.standard_normal(n)
    y -= float(np.mean(y))
    a = np.exp(0.8 * y)
    a = random_block_perturb(a, rng, 0.25, 3)
    if a is None:
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
    return a


def interpolate_sequence(a, n):
    a = normalize_positive(a)
    if a is None:
        return np.full(max(2, int(n)), 1.0 / max(2, int(n)), dtype=np.float64)
    n = max(2, int(n))
    if a.size == n:
        return a
    old = np.linspace(0.0, 1.0, a.size, endpoint=False)
    new = np.linspace(0.0, 1.0, n, endpoint=False)
    b = np.interp(new, old, a)
    b = np.where(np.isfinite(b), b, 0.0)
    return normalize_positive(b)


def optimize_sequence(a_init, budget, rng, tag):
    budget = max(0.0, float(budget))
    a = normalize_positive(a_init)
    if a is None:
        a = np.full(256, 1.0 / 256.0, dtype=np.float64)
    n = int(a.size)

    current = a.copy()
    current_score = score_array(current)
    best = current.copy()
    best_score = current_score
    if not np.isfinite(best_score):
        best_score = float("inf")

    if budget <= 0.0:
        return best, best_score

    eta = 0.05
    no = 0
    it = 0
    restarts = 0
    deadline = time.monotonic() + budget
    print(f"{tag}: start n={n} score={current_score:.10f}")

    while time.monotonic() < deadline:
        if no > 150:
            if restarts % 2 == 0:
                cand = random_block_perturb(best, rng, 0.35, 4)
            else:
                cand = make_random_initial(n, rng)
            if cand is not None:
                current = cand
                current_score = score_array(current)
                if current_score < best_score - 1e-12:
                    best = current.copy()
                    best_score = current_score
            no = 0
            restarts += 1
            it += 1
            eta = 0.05
            continue

        if no > 30 and (no % 5 == 0):
            shake_best = current.copy()
            shake_score = current_score
            for _ in range(8):
                cand = random_block_perturb(shake_best, rng, 0.08, 1)
                if cand is None:
                    break
                sc = score_array(cand)
                if sc < shake_score - 1e-12:
                    shake_best = cand
                    shake_score = sc
            if shake_score < current_score - 1e-12:
                current = shake_best
                current_score = shake_score
                no = max(0, no - 10)
                if current_score < best_score - 1e-12:
                    best = current.copy()
                    best_score = current_score
            else:
                no += 1
            it += 1
            if time.monotonic() >= deadline:
                break
            continue

        b = fft_convolution(current)
        if b.size == 0:
            break
        mx = float(np.max(b))
        if not np.isfinite(mx):
            break

        phase = min(12, it // 150)
        p = 2.0 * (1.3 ** phase)
        mix_top = min(0.92, 0.35 + 0.05 * phase)
        top_count = max(96, min(4096, int(n * (0.05 + 0.005 * phase))))

        W = make_lag_weights(b, p, top_count, mix_top)
        g = fft_lag_corr(current, W)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)

        h = current * g
        h -= float(np.mean(h))
        scale = float(np.mean(np.abs(h)))
        if not np.isfinite(scale) or scale < 1e-15:
            cand = random_block_perturb(current, rng, 0.05, 1)
            if cand is not None:
                sc = score_array(cand)
                if sc < current_score - 1e-12:
                    current = cand
                    current_score = sc
                    if sc < best_score - 1e-12:
                        best = current.copy()
                        best_score = sc
                    no = 0
                else:
                    no += 1
            else:
                no += 1
            it += 1
            continue

        h = h / scale
        if no < 25:
            factors = (0.5, 1.0, 1.7, 3.0)
        else:
            factors = (0.25, 0.5, 1.0, 2.0)

        local_best = current_score
        local_cand = None
        best_f = 1.0
        for f in factors:
            et = eta * f
            if not np.isfinite(et) or et <= 0.0:
                continue
            cand = apply_multiplicative(current, h, et, 4.0)
            if cand is None:
                continue
            sc = score_array(cand)
            if sc < local_best - 1e-12:
                local_best = sc
                local_cand = cand
                best_f = f

        if local_cand is not None:
            current = local_cand
            current_score = local_best
            if current_score < best_score - 1e-12:
                best = current.copy()
                best_score = current_score
                no = 0
            else:
                no = max(0, no - 2)
            if best_f >= 1.5:
                eta = min(1.5, eta * 1.25)
            elif best_f <= 0.5:
                eta = max(0.005, eta * 0.7)
        else:
            no += 1
            eta = max(0.005, eta * 0.65)

        it += 1
        if it % 500 == 0:
            print(f"{tag}: it={it} best={best_score:.10f} cur={current_score:.10f}")

    print(f"{tag}: end best={best_score:.10f}")
    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    if work <= 0.0:
        fb = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
        return [float(x) for x in fb]

    rng = np.random.default_rng(seed)
    candidates = []

    uniform_fb = np.full(4096, 1.0 / 4096.0, dtype=np.float64)
    uniform_fb = normalize_positive(uniform_fb)
    candidates.append((uniform_fb, 2.0))

    raw = globals().get("height_sequence_1", None)
    provided = normalize_positive(raw)
    use_provided = (
        provided is not None
        and 512 <= provided.size <= 20000

    alloc = 0.0

    if use_provided:
        frac = 0.18
        try:
            seed_a = random_block_perturb(provided, rng, 0.07, 3)
            if seed_a is None:
                seed_a = provided
            a, sc = optimize_sequence(seed_a, work * frac, rng, "provided")
            if a is not None:
                candidates.append((a, sc))
            else:
                candidates.append((provided, score_array(provided)))
        except Exception:
            candidates.append((provided, score_array(provided)))
        alloc += frac

    # Quick 8192 search.
    frac81 = 0.22
    a81 = None
    sc81 = float("inf")
    try:
        a81 = make_random_initial(8192, rng)
        a81, sc81 = optimize_sequence(a81, work * frac81, rng, "rand81")
    except Exception:
        a81, sc81 = None, float("inf")
    if a81 is not None:
        candidates.append((a81, sc81))
    alloc += frac81

    # Seed a 16384 search from the best short/nontrivial object found so far.
    base16 = None
    if a81 is not None and np.isfinite(sc81):
        base16 = a81
    if use_provided and candidates:
        for a, sc in candidates:
            if a is not None and np.isfinite(sc) and sc < sc81:

                    break

    if base16 is None:

    frac16a = 0.34

        seed16a = interpolate_sequence(base16, 16384)

    except Exception:

    if a16a is not None:
        candidates.append((a16a, sc16a))
    alloc += frac16a

    # One more independent 16384 search with the remaining budget.
    frac16b = max(0.05, 1.0 - alloc)
    a16b = None
    sc16b = float("inf")
    try:
        seed16b = make_random_initial(16384, rng)
        a16b, sc16b = optimize_sequence(seed16b, work * frac16b, rng, "rand16")
    except Exception:
        a16b, sc16b = None, float("inf")
    if a16b is not None:
        candidates.append((a16b, sc16b))

    best_idx = 0
    best_val = float("inf")
    for i, (a, sc) in enumerate(candidates):
        if a is not None and np.isfinite(sc) and sc < best_val:
            best_val = sc
            best_idx = i

    best_seq = candidates[best_idx][0]
    best_seq = normalize_positive(best_seq)
    if best_seq is None:
        best_seq = uniform_fb

    try:
        exact = evaluate_sequence([float(x) for x in best_seq])
    except Exception:
        exact = float("inf")

    if not np.isfinite(exact):
        try:
            exact_u = evaluate_sequence([float(x) for x in uniform_fb])
        except Exception:
            exact_u = float("inf")
        if np.isfinite(exact_u):
            best_seq = uniform_fb

    final = normalize_positive(best_seq)
    if final is None:
        final = uniform_fb
    return [float(x) for x in final]
