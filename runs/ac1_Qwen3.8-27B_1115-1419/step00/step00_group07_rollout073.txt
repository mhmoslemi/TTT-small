import numpy as np
import time

def compute_obj(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)

def optimize_from(a0, time_budget, seed=42):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget
    n = len(a0)
    a = np.maximum(0, a0.astype(np.float64)).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    best_a = a.copy()
    best_score = compute_obj(a)
    
    # Temperature schedule
    tau0 = max(1.0, n * 0.05)
    tau_min = max(0.1, n * 0.001)
    
    it = 0
    plateau_count = 0
    
    while time.monotonic() < deadline:
        it += 1
        tau = tau0 * (tau_min / tau0) ** (it / 1000.0)
        tau = max(tau, tau_min)
        
        b = np.convolve(a, a)
        b_max = b.max()
        
        # Softmax weights (numerically stable)
        w = np.exp((b - b_max) / tau)
        w /= w.sum()
        
        # Gradient: 2 * correlate(w, a)[:n]
        v = np.correlate(w, a, mode='full')[:n]
        grad = 2.0 * v
        
        # Adaptive step size
        # Grad has typical magnitude ~ 2*mean(a) ~ 2/n
        # We want eta * grad to be comparable to a
        # So eta ~ n/2
        eta = 1.0 / (np.max(grad) + 1e-10)
        # Cap the step
        eta = min(eta, 1.0)
        
        # Update
        a_new = a - eta * grad
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < 1e-10:
            # Reset
            a_new = a.copy()
            break
        a_new /= s
        
        new_score = compute_obj(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            plateau_count = 0
        else:
            plateau_count += 1
        
        a = a_new
        
        # If stuck for too long, perturb
        if plateau_count > 50:
            # Add random perturbation
            a = a + rng.normal(0, 0.01/n, size=n)
            a = np.maximum(0, a)
            a /= a.sum()
            plateau_count = 0
        
        # If score hasn't improved in 200 iterations, try a different perturbation
        if it % 200 == 0:
            print(f"  iter {it}, score {best_score:.6f}, tau {tau:.4f}")
    
    return best_a

import time


def compute_obj(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()

    """Compute 2 * sum_k w[k] * a[k-j] for j=0..n-1."""
    # v[j] = sum_m w[j+m] * a[m] for valid m
    # = np.correlate(w, a, 'full')[j]
    v = np.correlate(w, a, mode='full')[:n]
    return 2.0 * v

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 via smoothed-max gradient descent.
Uses softmax-weighted convolution gradient, temperature annealing,
line search, multiple initializations (ramp, cubic, two-block, trig, given),
and random restarts to escape local minima.
"""
import numpy as np
import time


def compute_obj(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def make_init(n, kind, rng):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'ramp':
        return 0.05 + x
    elif kind == 'cubic':
        return 0.01 + x ** 3
    elif kind == 'quartic':
        return 0.01 + x ** 4
    elif kind == 'two_block':
        a = np.ones(n)
        a[:n // 3] = 3.0
        a[2 * n // 3:] = 3.0
        return a
    elif kind == 'trig':
        return 1.0 + 0.7 * np.sin(2.0 * np.pi * 2.5 * x)
    elif kind == 'trig2':
        return 1.0 + 0.8 * np.sin(2.0 * np.pi * 5.0 * x)
    elif kind == 'random':
        return rng.random(n) + 0.01
    else:
        return 0.05 + x


def smooth_gradient(a, w, n):
    """Return 2*sum_k w[k]*a[k-j] for j=0..n-1 via correlate."""
    v = np.correlate(w, a, mode='full')[:n]
    return 2.0 * v


def optimize(a0, deadline, rng, verbose=False):
    n = len(a0)
    a = np.maximum(0, a0.astype(np.float64)).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    best_a = a.copy()
    best_score = compute_obj(a)
    if verbose:
        print(f"    init score: {best_score:.6f}")

    tau0 = max(1.0, n * 0.1)
    tau_min = max(0.05, n * 0.001)
    sched_len = 4000.0

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        frac = min(float(it), sched_len) / sched_len
        tau = tau0 * (tau_min / tau0) ** frac

        b = np.convolve(a, a)
        b_max = b.max()

        exps = (b - b_max) / tau
        exps = np.clip(exps, -700.0, 0.0)
        w = np.exp(exps)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        grad = smooth_gradient(a, w, n)

        # Line search with 2 step sizes
        best_step = 0.0
        best_trial = best_score
        for eta in (0.15, 0.4):
            a_t = a - eta * grad
            a_t = np.maximum(0.0, a_t)
            st = a_t.sum()
            if st < 1e-10:
                continue
            a_t /= st
            sc = compute_obj(a_t)
            if sc < best_trial:
                best_trial = sc
                best_step = eta

        if best_step > 0.0:
            a = a - best_step * grad
            a = np.maximum(0.0, a)
            a /= a.sum()
            no_improve = 0
        else:
            # Perturb to escape
            scale = 0.02 / n
            a = a + rng.normal(0.0, scale, n)
            a = np.maximum(0.0, a)
            a /= a.sum()
            no_improve += 1
            if no_improve > 300:
                a = a + rng.normal(0.0, 0.1 / n, n)
                a = np.maximum(0.0, a)
                a /= a.sum()
                no_improve = 0
            if no_improve > 1500:
                break

        sc = compute_obj(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 2000 == 0:
            print(f"      iter {it}, score={best_score:.6f}, tau={tau:.4f}")

    return best_a


def single_peak_refine(a0, deadline, rng):
    """Focus gradient descent on the single maximum lag."""
    n = len(a0)
    a = np.maximum(0, a0.astype(np.float64)).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    best_a = a.copy()
    best_score = compute_obj(a)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = np.convolve(a, a)
        k_star = int(b.argmax())

        # grad[j] = 2*a[k_star - j] for valid indices
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[idx[valid]]

        best_step = 0.0
        best_trial = best_score
        for eta in (0.02, 0.05, 0.1):
            a_t = a - eta * grad
            a_t = np.maximum(0.0, a_t)
            st = a_t.sum()
            if st < 1e-10:
                continue
            a_t /= st
            sc = compute_obj(a_t)
            if sc < best_trial:
                best_trial = sc
                best_step = eta

        if best_step > 0.0:
            a = a - best_step * grad
            a = np.maximum(0.0, a)
            a /= a.sum()
            sc = compute_obj(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            a = a + rng.normal(0.0, 0.01 / n, n)
            a = np.maximum(0.0, a)
            a /= a.sum()

        if it % 500 == 0:
            print(f"      refine iter {it}, score={best_score:.6f}, k*={k_star}")

    return best_a
