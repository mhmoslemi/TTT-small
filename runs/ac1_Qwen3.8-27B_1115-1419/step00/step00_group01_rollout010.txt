def fast_correlate(p, a):
    """Compute linear cross-correlation (p ⋆ a)(i) = sum_j p[j+i]*a[j] for i=0..len(a)-1."""
    n = len(a)
    m = len(p)
    N = 1

    A = np.fft.rfft(a, N)

"""
Gradient-based soft-peak minimization for non-negative step function heights.
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using FFT-accelerated gradient descent
on a softmax surrogate of the peak, with multiple restarts and adaptive schedules.
"""

import numpy as np
from scipy.signal import fftconvolve
import time


def _fast_correlate(p, a):
    """Compute linear cross-correlation (p ⋆ a)(i) = sum_j p[j+i]*a[j] for i=0..len(a)-1."""
    n = len(a)
    m = len(p)
    N = 1
    while N < m + n - 1:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(P * np.conj(A), N)
    return corr[:n]


def _soft_peak_grad(a, beta):
    """Gradient of (1/beta)*log(sum exp(beta*convolve(a,a))) w.r.t. a."""
    c = fftconvolve(a, a, mode='full')
    bc = beta * c
    bc -= bc.max()
    e = np.exp(bc)
    s = e.sum()
    if s < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = e / s
    return 2.0 * _fast_correlate(p, a)


def _proxy_score(a):
    """Fast proxy for the objective (avoids list conversion)."""
    n = len(a)
    c = fftconvolve(a, a, mode='full')
    return 2.0 * n * c.max() / (a.sum() ** 2)


def _make_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    # Uniform + small noise
    inits.append(np.ones(n) / n + 0.03 * rng.standard_normal(n))
    # Linear ramp
    ramp = np.linspace(0.4, 1.6, n)
    inits.append(ramp / ramp.sum() + 0.03 * rng.standard_normal(n))
    # Random
    inits.append(np.maximum(rng.random(n), 0.01) + 0.02 * rng.standard_normal(n))
    # Cosine bell
    cos = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    inits.append(cos / cos.sum() + 0.02 * rng.standard_normal(n))
    # Two-bump (asymmetric)
    x = np.arange(n)
    bump1 = np.exp(-((x - 0.3 * n) ** 2) / (2 * (0.1 * n) ** 2))
    bump2 = np.exp(-((x - 0.7 * n) ** 2) / (2 * (0.08 * n) ** 2))
    two_bump = bump1 + 0.7 * bump2
    inits.append(two_bump / two_bump.sum() + 0.02 * rng.standard_normal(n))
    return inits


def _optimize_one(n, rng, time_limit):
    """Run gradient optimization for a single sequence length."""
    deadline = time.monotonic() + time_limit
    inits = _make_inits(n, rng)
    best_a = None
    best_score = float('inf')

    time_per_init = time_limit / len(inits)

    for idx in range(len(inits)):
        if time.monotonic() > deadline - 5:
            break

        a = np.maximum(inits[idx], 0.0)
        if a.sum() < 1e-6:
            a = np.ones(n) / n
        a = a / a.sum()

        local_best = a.copy()
        local_score = _proxy_score(a)

        beta = 5.0
        lr = 0.4
        momentum = np.zeros(n)
        mu = 0.7
        stall = 0
        sub_deadline = time.monotonic() + time_per_init

        it = 0
        while time.monotonic() < sub_deadline:
            grad = _soft_peak_grad(a, beta)
            # Center for sum constraint
            centered = grad - grad.mean()
            # Momentum
            momentum = mu * momentum + (1 - mu) * centered
            # Multiplicative update
            step = lr * momentum
            step = np.clip(step, -2.0, 2.0)
            a = a * np.exp(-step)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

            # Proxy check
            ps = _proxy_score(a)
            if ps < local_score - 1e-7:
                local_score = ps
                local_best = a.copy()
                stall = 0
            else:
                stall += 1

            it += 1
            if stall > 30:
                lr *= 0.85
                beta = min(beta * 1.3, 80.0)
                # Occasional noise injection to escape local minima
                if stall > 80 and rng.random() < 0.3:
                    a = a + 0.02 * rng.standard_normal(n)
                    a = np.maximum(a, 1e-12)
                    a = a / a.sum()
                stall = 0

            if lr < 1e-5:
                break

        # Validate with actual evaluator
        real = float(evaluate_sequence(local_best.tolist()))
        if np.isfinite(real) and real < best_score:
            best_score = real
            best_a = local_best.copy()

    return best_a, best_score


def _polish(a, budget_s):
    """Refine a sequence with smaller steps and higher beta."""
    deadline = time.monotonic() + budget_s
    n = len(a)
    current_score = _proxy_score(a)
    best_a = a.copy()
    best_score = current_score
    lr = 0.1
    beta = 15.0
    momentum = np.zeros(n)
    mu = 0.8

    while time.monotonic() < deadline:
        grad = _soft_peak_grad(a, beta)
        centered = grad - grad.mean()
        momentum = mu * momentum + (1 - mu) * centered
        step = lr * momentum
        step = np.clip(step, -1.0, 1.0)
        a = a * np.exp(-step)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()

        ps = _proxy_score(a)
        if ps < best_score - 1e-8:
            best_score = ps
            best_a = a.copy()
            lr *= 1.005
            if lr > 0.3:
                lr = 0.3
        else:
            lr *= 0.97
            beta = min(beta * 1.02, 60.0)

        if lr < 1e-5:
            break

    return best_a, best_score
