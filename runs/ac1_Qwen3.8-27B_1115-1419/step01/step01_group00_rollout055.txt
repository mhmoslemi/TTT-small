"""
Optimized step-function search using L-BFGS-B with log-sum-exp smoothing,
FFT-based convolution, multiple initializations, and structured perturbations.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences a.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    m = _next_pow2(size)
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)[:size]
    return C


def _logsumexp_smooth_obj_grad(w, n, tau):
    s = np.exp(w - np.max(w))
    s /= s.sum()
    a = s * n
    c = _fft_conv(a)
    cs = c / tau
    ls = logsumexp(cs)
    obj = 2.0 * tau * ls / n
    p = np.exp(cs - ls)
    grad_a = 2.0 * np.correlate(p, a, mode='valid')
    g = (4.0 * tau / n) * grad_a
    grad_w = a * g - s * np.dot(a, g)
    return obj, grad_w


def _hard_obj_grad_w(w, n):
    """Objective and gradient using exact max (subgradient)."""
    s = np.exp(w - np.max(w))
    s /= s.sum()
    a = s * n
    c = _fft_conv(a)
    k = int(np.argmax(c))
    mc = c[k]
    obj = 2.0 * mc / n
    # p is delta at k
    p = np.zeros(2 * n - 1)
    p[k] = 1.0
    grad_a = 2.0 * np.correlate(p, a, mode='valid')
    g = grad_a  # tau=1 effectively for the subgradient
    grad_w = a * g - s * np.dot(a, g)
    return obj, grad_w


def _window_init(n, kind='hann'):
    x = np.linspace(0, 1, n)
    if kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif kind == 'tukey':
        r = 0.3
        a = np.ones(n)
        a[:int(r*n)] = 0.5*(1+np.cos(np.pi*(np.arange(int(r*n))/int(r*n)*2-1)))
        a[-int(r*n):] = 0.5*(1+np.cos(np.pi*(np.arange(int(r*n))/int(r*n)*2-1)))
    elif kind == 'raised_cos':
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.25)))
        a = np.clip(a, 0, None)
    elif kind == 'gauss':
        a = np.exp(-8 * (x - 0.5)**2)
    elif kind == 'sqrt_tri':
        tri = np.minimum(x, 1-x) * 2
        a = np.sqrt(tri)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a


def _to_w(a):
    """Convert a (sum=n, all positive) to w-space: w = log(a)."""
    return np.log(np.maximum(a, 1e-300))


def _lbgfss(a, n, deadline, tau_start, tau_end, iters_per_tau, tag):
    """
    Run L-BFGS-B with decreasing tau.
    a: initial (sum=n, non-negative)
    Returns best (a, obj).
    """
    w = _to_w(a)
    best_a = a.copy()
    best_o = 2.0 * float(np.max(_fft_conv(a))) / n
    
    n_tau = 12
    taus = np.geomspace(tau_start, tau_end, n_tau)
    
    for ti, tau in enumerate(taus):
        if time.monotonic() >= deadline:
            break
        res = minimize(
            lambda ww: _logsumexp_smooth_obj_grad(ww, n, tau),
            w,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': iters_per_tau,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxfun': iters_per_tau * 3,
            }
        )
        w = res.x
        s = np.exp(w - np.max(w))
        s /= s.sum()
        a = s * n
        c = _fft_conv(a)
        o = 2.0 * float(np.max(c)) / n
        if o < best_o:
            best_o = o
            best_a = a.copy()
            print(f"  [{tag}] tau={tau:.1f} obj={o:.6f} iters={res.nit}")
    
    return best_a, best_o


def _hard_refine(a, n, deadline, tag, it_limit=50000):
    """Subgradient / active-set hard refinement."""
    w = _to_w(a)
    best_a = a.copy()
    best_o = 2.0 * float(np.max(_fft_conv(a))) / n
    eta = 0.5
    
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        s = np.exp(w - np.max(w))
        s /= s.sum()
        aa = s * n
        c = _fft_conv(aa)
        # Find top few peaks (active set)
        top_k = 5
        idx = np.argsort(c)[-top_k:]
        mc = c[idx[0]]
        o = 2.0 * mc / n
        if o < best_o - 1e-10:
            best_o = o
            best_a = aa.copy()
        # Build subgradient (average of top-k subgradients)
        p = np.zeros(2 * n - 1)
        for k in idx:
            p[k] = 1.0 / top_k
        grad_a = 2.0 * np.correlate(p, aa, mode='valid')
        g = grad_a
        grad_w = aa * g - s * np.dot(aa, g)
        # Step
        mg = np.max(np.abs(grad_w))
        if mg < 1e-14:
            break
        w = w - eta * grad_w / mg
        if it % 1000 == 0 and it > 0:
            eta *= 0.7
    return best_a, best_o

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp
from scipy.ndimage import uniform_filter1d, gaussian_filter1d
from scipy.interpolate import interp1d

# Helper functions (all top-level, no closures)

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    m = _next_pow2(size)
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)[:size]
    # Numerical cleanup
    np.maximum(C, 0, out=C)
    return C

def _softmax(w):
    mx = np.max(w)
    e = np.exp(w - mx)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(w, 1.0/len(w))
    return e / s

def _obj_w(w, n, tau):
    """Return (obj, grad) for L-BFGS-B in w-space."""
    s = _softmax(w)
    a = s * n
    c = _fft_conv(a)
    cs = c / tau
    ls = logsumexp(cs)
    obj = 2.0 * tau * ls / n
    p = np.exp(cs - ls)
    grad_a = 2.0 * np.correlate(p, a, mode='valid')
    g = (4.0 * tau / n) * grad_a
    grad_w = a * g - s * np.dot(a, g)
    return obj, grad_w

def _hard_obj_w(w, n, topk=1):
    """Return (obj, grad) using top-k subgradient."""
    s = _softmax(w)
    a = s * n
    c = _fft_conv(a)
    idx = np.argsort(c)[-topk:]
    mc = c[idx[0]]
    obj = 2.0 * mc / n
    p = np.zeros(2*n-1)
    for k in idx:
        p[k] = 1.0/topk
    grad_a = 2.0 * np.correlate(p, a, mode='valid')
    g = grad_a  # no tau factor
    grad_w = a * g - s * np.dot(a, g)
    return obj, grad_w

def _window(n, kind):
    x = np.linspace(0, 1, n)
    if kind == 'hann':
        return 0.5*(1+np.cos(2*np.pi*x))
    elif kind == 'hamming':
        return 0.54 - 0.46*np.cos(2*np.pi*x)
    elif kind == 'blackman':
        return 0.42 - 0.5*np.cos(2*np.pi*x) + 0.08*np.cos(4*np.pi*x)
    elif kind == 'tukey':
        r = 0.4
        out = np.ones(n)
        m = int(r*n//2)
        if m > 0:
            for i in range(m):
                out[i] = 0.5*(1+np.cos(np.pi*(i/m*2-1)))
                out[n-1-i] = out[i]
        return out
