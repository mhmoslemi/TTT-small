"""

    2 * n * max(convolve(a, a)) / sum(a)^2

Strategy:

  - Multiple diverse initializations (spectral, bump, asymmetric)

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _score_and_grad(a, m, tau):
    """Returns (score, gradient of soft-max objective w.r.t. a)."""
    n = int(a.size)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 1e-300:
        return np.inf, np.zeros(n)
    S = float(a.sum())
    score = 2.0 * n * bmax / (S * S)

    x = (b - bmax) / max(tau, 1e-15)
    x = np.clip(x, -80.0, 80.0)
    x = x - float(x.max())
    w = np.exp(x)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= ws

    # Gradient of soft_max(b, tau) w.r.t. a:
    # d(soft_max)/d(a[i]) = tau * sum_k w[k] * db[k]/d(a[i])
    # db[k]/d(a[i]) = a[i-k] + a[i+k] (with bounds)
    # Efficiently: grad = ifft(fft(w) * conj(F)) gives the cross-correlation
    P = np.fft.fft(w, m)
    c = np.fft.ifft(P * np.conj(F)).real[:n]
    np.maximum(c, 0.0, out=c)

    # Scale: the soft-max objective is tau * log(sum exp(b/tau))
    # d(obj)/d(a[i]) = tau * (1/tau) * sum_k w[k] * db[k]/d(a[i]) = sum_k w[k] * db[k]/d(a[i])
    # But we want the gradient of the full score: 2n * soft_max / S^2
    # d(score)/d(a[i]) = (2n/S^2) * (soft_max_grad[i] - 2*soft_max*a[i]/S)
    # For simplicity, we use just the soft_max gradient (the S^2 term is ~constant)
    grad = c
    return score, grad


def _soft(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - float(np.max(v))
    v = np.clip(v, -745.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / max(1, v.size), dtype=np.float64)
    return e / s


def _log_plus(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-300)
    return np.log(a)


def _proj_nonneg_sum(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a.fill(1.0 / max(1, a.size))
    else:
        a /= s
    return a


def _interp(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = int(max(1, n))
    if a.size == 0 or s <= 0.0:
        return np.ones(n, dtype=np.float64)
    if a.size == n:
        return a / s
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 0.0:
        out.fill(1.0 / n)
    else:
        out /= ss
    return out


def _init_candidates(n, rng):

    n = int(max(3, n))

    t = np.linspace(0.0, 1.0, n)

    # Uniform
    res.append(_proj_nonneg_sum(np.ones(n)))

    # Raised cosine (Hann window)
    res.append(_proj_nonneg_sum(0.5 * (1.0 + np.cos(2.0 * np.pi * t))))

    # Bartlett (triangular)

    # Sine bump
    res.append(_proj_nonneg_sum(np.sin(np.pi * t) ** 2))

    # Power-law sine bumps
    for p in (1.0, 1.5, 2.0, 3.0, 4.0):
        res.append(_proj_nonneg_sum(np.sin(np.pi * t) ** p))

    # Asymmetric: concentrated left
    for alpha in (0.5, 1.0, 2.0, 3.0):
        res.append(_proj_nonneg_sum(np.exp(-alpha * t)))

    # Asymmetric: concentrated right
    for alpha in (0.5, 1.0, 2.0, 3.0):
        res.append(_proj_nonneg_sum(np.exp(-alpha * (1.0 - t))))

    # Endpoint singular (1/sqrt)
    left = np.maximum(i + 1.0, 0.5)
    right = np.maximum(n - i, 0.5)
    res.append(_proj_nonneg_sum(1.0 / np.sqrt(left * right)))
    res.append(_proj_nonneg_sum(1.0 / np.sqrt(left)))
    res.append(_proj_nonneg_sum(1.0 / np.sqrt(right)))

    # Two-sided asymmetric
    for p, q in ((0.3, 0.7), (0.7, 0.3), (0.2, 0.8), (0.8, 0.2), (0.5, 0.5)):
        a = (t + 1e-9) ** p * (1.0 - t + 1e-9) ** q
        res.append(_proj_nonneg_sum(a))

    # Cosine series (random)
    kc = min(20, n // 2)
    a = 1.0
    for k in range(1, kc + 1):
        amp = 0.4 / (float(k) ** 0.7)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t + ph)
    res.append(_proj_nonneg_sum(a))

    # Random smooth
    a = 1.0
    for k in range(1, min(15, n // 2) + 1):
        amp = 0.3 / (float(k) ** 0.8)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t + ph)
    res.append(_proj_nonneg_sum(a))

    return res


def _spectral_init(n, best_a, rng):
    """Spectral factorization-based initialization."""
    n = int(max(16, n))
    m = _next_pow2(max(64, 4 * n))
    L = 2 * n - 1

    t = np.linspace(0.0, 1.0, L)
    edge = np.abs(t - 0.5) * 2.0

    # Target: slightly tapered (lower at edges)
    targets = []
    for taper in (0.0, 0.2, 0.4, 0.6):
        target = 1.0 - taper * edge ** 1.5
        target = np.maximum(target, 0.1)
        target = target / float(target.max())
        targets.append(target)

    # Also try the current best's normalized autocorrelation
    if best_a is not None:
        ba = np.asarray(best_a, dtype=np.float64)
        if ba.size == n:
            b = _conv_self(ba, _next_pow2(2 * ba.size - 1))
            bm = float(b.max())
            if bm > 0.0:
                targets.append(b / bm)

    results = []
    for target in targets[:4]:
        a = _spectral_factor(target, n, m, best_a, rng)
        if a is not None:
            results.append(a)

    return results


def _conv_self(a, m):
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * a.size - 1]
    return np.maximum(b, 0.0)


def _spectral_factor(target, n, m, a_start, rng, iters=60):
    """Compute spectral factor of target (approximate sqrt in frequency domain)."""
    n = int(max(16, n))
    L = 2 * n - 1
    target = np.asarray(target, dtype=np.float64)
    if target.size < L:
        padded = np.zeros(L, dtype=np.float64)
        padded[:target.size] = target
        target = padded
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    tmax = float(target.max())
    if tmax > 0.0:
        target = target / tmax
    else:
        target = np.ones(L, dtype=np.float64)

    T = np.fft.fft(target[:min(L, m)], m)
    sqrt_T = np.fft.ifft(np.sqrt(np.maximum(T.real, 0.0)) + 1j * np.sqrt(np.maximum(-T.real, 0.0))).real[:n]

    # Also try ifft of sqrt of |T|
    sqrt_T2 = np.fft.ifft(np.sqrt(np.abs(T))).real[:n]

    candidates = [np.maximum(sqrt_T, 0.0), np.maximum(sqrt_T2, 0.0)]

    best_a = None
    best_sc = np.inf
    for cand in candidates:
        cmax = float(cand.max())
        if cmax <= 0.0 or not np.isfinite(cmax):
            continue
        cand = cand / cmax
        if cand[0] < 1e-6:
            cand = cand + 1e-4
        cand = _proj_nonneg_sum(cand)
        sc = _score_fft(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()

    # Iterate: alternate between time-domain projection and frequency-domain matching
    if best_a is not None:
        a = best_a.copy()
        for it in range(iters):
            F = np.fft.fft(a, m)
            F_abs = np.abs(F)
            phase = np.angle(F)
            # We want |F|^2 ≈ T, so |F| ≈ sqrt(T)
            target_mag = np.sqrt(np.abs(T))
            tmax_f = float(target_mag.max())
            if tmax_f > 0:
                target_mag = target_mag / tmax_f
            # Normalize current
            fmax = float(F_abs.max())
            if fmax > 0:
                F_abs = F_abs / fmax
            # Blend
            F_abs_new = 0.6 * F_abs + 0.4 * target_mag
            F_new = F_abs_new * np.exp(1j * phase)
            anew = np.fft.ifft(F_new).real[:n]
            anew = np.maximum(anew, 0.0)
            as = float(anew.sum())
            if as < 1e-6 * float(a.sum()):
                break
            anew = anew / as
            a = 0.7 * a + 0.3 * anew
            a = np.maximum(a, 1e-12)
            a = _proj_nonneg_sum(a)
            sc = _score_fft(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def _optimize(a_init, n, max_seconds, seed, verbose=True):

    rng = np.random.default_rng(seed)

    if not np.isfinite(best_score):
        best_a = np.ones(n, dtype=np.float64) / n
        v = _log_plus(best_a)
        best_score = 2.0

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t_step = 0

    start = time.monotonic()
    deadline = start + max_seconds
    last_improve = start

    while time.monotonic() < deadline:
        elapsed = max(1e-9, time.monotonic() - start)
        progress = min(1.0, elapsed / max_seconds)

        # Stuck detection: add noise if no improvement
        stuck_limit = min(60.0, 0.25 * max_seconds)
        if time.monotonic() - last_improve > stuck_limit:
            sigma = 0.08 + 0.25 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -30.0, 30.0)
            last_improve = time.monotonic()
            continue

        a = _soft(v)
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break

        # Annealed temperature (cosine schedule)
        tau_frac = 0.04 + 0.20 * (0.5 + 0.5 * np.cos(np.pi * progress))
        tau = max(1e-12, bmax * tau_frac)

        x = (b - bmax) / tau
        x = np.clip(x, -80.0, 80.0)
        x = x - float(x.max())
        w = np.exp(x)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:

        # Boost top-K lags
        blen = int(b.size)

            ws = float(w.sum())
            w /= ws

        P = np.fft.fft(w, m)
        c = np.fft.ifft(P * np.conj(F)).real[:n]

        # Center the gradient (remove mean to maintain zero-sum in log-space)
        cm = float(np.dot(c, a))
        gv = a * (c - cm)
        grms = float(np.sqrt(np.mean(gv * gv)))
        if not np.isfinite(grms) or grms < 1e-14:
            sigma = 0.03 + 0.15 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -30.0, 30.0)
            continue

        gv = gv / grms
        t_step += 1
        mu = beta1 * mu + (1.0 - beta1) * gv
        nu = beta2 * nu + (1.0 - beta2) * (gv * gv)

        b1 = beta1 ** t_step
        b2 = beta2 ** t_step
        mu_hat = mu / max(1e-12, 1.0 - b1)
        nu_hat = nu / max(1e-12, 1.0 - b2)

        lr = 0.15 * (0.5 + 0.5 * np.cos(np.pi * progress)) + 0.01
        step = mu_hat / (np.sqrt(nu_hat) + eps)
        step = np.clip(step, -1.0, 1.0)

        v = v - float(np.mean(v))

def _process_length(n, best_a, deadline, rng, verbose=True):
    """Process a single length: generate candidates, optimize, polish."""
    n = int(max(16, n))
    now = time.monotonic()
    if now >= deadline:
        return (best_a, _score_fft(best_a) if best_a is not None else np.inf)

    # Build candidate list
    candidates = []

    # From previous best (interpolated)
    if best_a is not None:
        candidates.append(_interp(best_a, n))

    # Spectral inits
    try:
        specs = _spectral_init(n, best_a, rng)
        candidates.extend(specs)
    except Exception:
        pass

    # Standard inits
    try:
        inits = _init_candidates(n, rng)
        candidates.extend(inits)
    except Exception:
        pass

    # Score and select top candidates
    scored = []
    for a in candidates:
        a = _proj_nonneg_sum(a)
        if a.size != n:
            continue
        sc = _score_fft(a)
        if np.isfinite(sc):
            scored.append((sc, a))

    scored.sort(key=lambda x: x[0])

    # Select diverse top candidates
    selected = []
    for sc, a in scored:
        diverse = True
        for _, b in selected:
            d = float(np.sqrt(np.sum((a - b) ** 2)))
            if d < 0.04:
                diverse = False
                break
        if diverse:
            selected.append((sc, a))
        if len(selected) >= 5:
            break

    if not selected:
        selected.append((2.0, np.ones(n, dtype=np.float64) / n))

    # Allocate time to optimization
    rem = max(1.0, deadline - time.monotonic())
    opt_time_each = max(10.0, rem * 0.6 / len(selected))
    polish_time = rem * 0.25

    best_local_a = selected[0][1]
    best_local_score = selected[0][0]

    for i, (sc, a) in enumerate(selected):
        now = time.monotonic()
        if now + 10.0 >= deadline:
            break
        alloc = min(opt_time_each, deadline - now - 5.0)
        if alloc < 10.0:
            break
        seed = int(rng.integers(0, 2**31 - 1))
        if verbose:
            print("    n=%d cand %d init_score=%.6f time=%.1fs" % (n, i, sc, alloc))
        a_opt, sc_opt = _optimize(a, n, alloc, seed, verbose=False)
        if np.isfinite(sc_opt) and sc_opt < best_local_score:
            best_local_score = sc_opt
            best_local_a = a_opt
            if verbose:
                print("      improved to %.6f" % best_local_score)

    # Polish the best
    now = time.monotonic()
    rem = max(0.0, deadline - now - 5.0)
    if rem > 10.0:
        seed = int(rng.integers(0, 2**31 - 1))
        a_pol, sc_pol = _polish(best_local_a, n, rem, seed)
        if np.isfinite(sc_pol) and sc_pol < best_local_score:
            best_local_score = sc_pol
            best_local_a = a_pol
            if verbose:
                print("    polished to %.6f" % best_local_score)

    return best_local_a, float(best_local_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    """
    seed = int(seed)
    budget_s = float(max(60.0, budget_s))
    start = time.monotonic()
    global_deadline = start + max(60.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    # Try to load previous best
    prev = None
    try:
        cand = height_sequence_1
        if isinstance(cand, list):
            arr = np.asarray(cand, dtype=np.float64)
        else:
            arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0 and np.isfinite(arr).all():
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 0.0:
                prev = arr / float(arr.sum())
    except Exception:
        prev = None

    best_a = prev
    best_score = _score_fft(prev) if prev is not None else np.inf
    if verbose_check():
        print("Initial score: %.6f (n=%d)" % (best_score, len(prev) if prev is not None else 0))

    # Stage definitions: (n, time_fraction)
    # Prioritize the length that the previous best used, then try larger
    if prev is not None and prev.size >= 1000:
        base_n = prev.size
        if base_n > 2048:
            stage_defs = [(2048, 0.10), (4096, 0.40), (8192, 0.45)]
        elif base_n > 1024:
            stage_defs = [(2048, 0.35), (4096, 0.55)]
        else:
            stage_defs = [(1024, 0.15), (2048, 0.40), (4096, 0.40)]
    else:
        stage_defs = [(1024, 0.15), (2048, 0.35), (4096, 0.40), (8192, 0.10)]

    for n, frac in stage_defs:
        now = time.monotonic()
        if now + 20.0 >= global_deadline:
            break
        stage_time = max(20.0, (budget_s - 30.0) * frac)
        stage_deadline = min(global_deadline - 5.0, now + stage_time)
        if verbose_check():
            print("Stage n=%d budget=%.1fs current_best=%.6f" % (n, stage_time, best_score))
        a_new, sc_new = _process_length(n, best_a, stage_deadline, rng, verbose=True)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_a = a_new
            best_score = sc_new
        if verbose_check():
            print("After stage n=%d best=%.6f" % (n, best_score))

    # Final polishing on the best sequence
    if best_a is not None:
        now = time.monotonic()
        rem = max(0.0, global_deadline - 10.0 - now)
        if rem > 30.0:
            if verbose_check():
                print("Final polish: n=%d, time=%.1fs, score=%.6f" % (len(best_a), rem, best_score))
            seed = int(rng.integers(0, 2**31 - 1))
            a_pol, sc_pol = _polish(best_a, len(best_a), rem, seed)
            if np.isfinite(sc_pol) and sc_pol < best_score:
                best_a = a_pol
                best_score = sc_pol
            # Second polish with different seed
            now = time.monotonic()
            rem2 = max(0.0, global_deadline - 10.0 - now)
            if rem2 > 20.0:
                seed2 = int(rng.integers(0, 2**31 - 1))
                a_pol2, sc_pol2 = _polish(best_a, len(best_a), rem2, seed2)
                if np.isfinite(sc_pol2) and sc_pol2 < best_score:
                    best_a = a_pol2
                    best_score = sc_pol2

    if best_a is None:
        best_a = np.ones(512, dtype=np.float64)
    best_a = _proj_nonneg_sum(best_a)

    # Validate with exact evaluation if feasible
    try:
        if len(best_a) <= 10000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Final: FFT score=%.6f, exact score=%.6f, n=%d" %
                  (best_score, exact, len(best_a)))
            if not np.isfinite(exact):
                best_a = np.ones(512, dtype=np.float64)
        else:
            print("Final FFT score=%.6f, n=%d" % (best_score, len(best_a)))
    except Exception as e:
        print("Exact validation skipped: %s" % e)

    return [float(x) for x in best_a]


def verbose_check():
    return True
