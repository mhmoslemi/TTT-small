```python
def _lp_direction(f, M, max_constraints=200):
    """Find direction g that maximizes sum(g) subject to f*g <= M, g >= 0."""
    n = len(f)
    # Find the most binding constraints
    fg = _fft_conv(f, np.ones(n))  # This gives all (f*g)[k] when g = 1
    # Actually we need to find which k have the largest (f*g)[k] for the current f
    # The binding constraints are where (f*f)[k] is closest to M
    ff = _fft_conv(f, f)
    # Sort by how close to M they are (most binding first)
    order = np.argsort(-ff)[:max_constraints]
    
    # Build constraint matrix A (sparse)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in order:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)  # constraint index
            cols.append(i)  # variable index (g[i])
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(order), n))
    b_vec = np.full(len(order), M)
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize array to target sum."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _compute_subgrad(a, b, top_k=20):
    """Compute smoothed subgradient of max_b w.r.t. a."""
    n = len(a)
    max_b = float(np.max(b))
    
    k = min(top_k, len(b))
    top_idx = np.argsort(b)[-k:]
    top_vals = b[top_idx]
    
    # Softmax weights
    spread = max_b - top_vals[0] + 1e-10
    if spread < 1e-10:
        weights = np.full(k, 1.0 / k)
    else:
        # Use a temperature that gives reasonable spread
        temp = spread * 0.5
        weights = np.exp((top_vals - max_b) / temp)
        weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, m in zip(weights, top_idx):
        m = int(m)
        i_lo = max(0, m - n + 1)
        i_hi = min(m, n - 1)
        grad[i_lo:i_hi + 1] += w * a[m - i_hi:m - i_lo + 1][::-1]
    
    return grad, max_b


def _adam_loop(a, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8,
               top_k=20, restart_patience=500, noise_scale=0.05):
    """Run Adam subgradient descent until deadline."""
    n = len(a)
    target_sum = float(np.sum(a))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = _score_fast(a)
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Compute convolution and subgradient
        b = _fft_conv(a, a)
        max_b = float(np.max(b))
        sum_a = float(np.sum(a))
        if sum_a < 0.01:
            break
        score = 2.0 * n * max_b / (sum_a * sum_a)
        
        # Check improvement
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > restart_patience:
            a = best_a + noise_scale * np.random.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = _normalize(a, target_sum)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue
        
        # Compute subgradient
        grad, _ = _compute_subgrad(a, b, top_k)
        
        # Project onto sum-preserving subspace
        grad = grad - np.mean(grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = _normalize(a, target_sum)

def _lp_correction(f, max_constraints=300):
    """Use LP to find a direction that reduces the peak autocorrelation."""
    n = len(f)
    ff = _fft_conv(f, f)
    M = float(np.max(ff))
    
    # Get most binding constraints

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(max_constraints, n))
    b_vec = np.full(max_constraints, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n,

    return None
```

The Hessian of `max_k (x*x)[k]` at the maximizing position k* is:
`H = x[k* - i] * e_i e_i^T` (diagonal matrix with entries x[k*-i])

So a natural preconditioner is `diag(x[k* - i])`. This is similar to what Adam does automatically, but we can use it more directly.

Actually, for the log-sum-exp approximation:
`LSE(x) = (1/T) * log(sum_k exp(T * (x*x)[k]))`

The gradient is:
`grad[i] = sum_k p(k) * x[k-i]` where `p(k) = exp(T*(x*x)[k]) / sum_j exp(T*(x*x)[j])`

This is a smooth approximation that gives a better gradient when multiple peaks are close to the max.

```python
def _lse_subgrad(a, T=1.0):
    """Compute subgradient using log-sum-exp approximation."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    
    # Compute softmax weights (truncated for efficiency)
    # Only use top-100 peaks
    top_k = min(100, len(b))
    top_idx = np.argsort(b)[-top_k:]
    top_vals = b[top_idx]
    
    # Softmax with temperature
    shifted = T * (top_vals - max_b)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, m in zip(weights, top_idx):
        m = int(m)
        i_lo = max(0, m - n + 1)
        i_hi = min(m, n - 1)
        grad[i_lo:i_hi + 1] += w * a[m - i_hi:m - i_lo + 1][::-1]
    
    return grad, max_b
```

```python
"""
Minimizes 2n*max(autocorr)/(sum)^2 using Adam-based subgradient with log-sum-exp
smoothing, asymmetric initializations, LP corrections, and coordinate descent.
Exploits the key insight that asymmetric sequences can achieve max_autocorr < (sum)^2/n.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize array to target sum, preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _lse_subgrad(a, b, top_k=50, temp_scale=1.0):
    """Compute smoothed subgradient using log-sum-exp over top-k peaks."""
    n = len(a)
    max_b = float(np.max(b))
    k = min(top_k, len(b))
    top_idx = np.argsort(b)[-k:]
    top_vals = b[top_idx]

    spread = max_b - float(top_vals[0]) + 1e-12
    if spread < 1e-10:
        weights = np.full(k, 1.0 / k)
    else:
        temp = spread * temp_scale
        weights = np.exp((top_vals - max_b) / temp)
        weights /= weights.sum()

    grad = np.zeros(n)
    for w, m in zip(weights, top_idx):
        m = int(m)
        i_lo = max(0, m - n + 1)
        i_hi = min(m, n - 1)
        seg = a[m - i_hi:m - i_lo + 1][::-1]
        grad[i_lo:i_hi + 1] += w * seg

    return grad, max_b


def _adam_subgrad_loop(a, deadline, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8,
                       top_k=30, temp_scale=0.5, restart_patience=400,
                       noise_scale=0.03, eval_every=50):
    """Run Adam subgradient descent until deadline."""
    n = len(a)
    target_sum = float(np.sum(a))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = _score_fast(a)
    no_improve = 0
    step_count = 0

    while time.monotonic() < deadline:
        b = _fft_conv(a, a)
        max_b = float(np.max(b))
        sum_a = float(np.sum(a))
        if sum_a < 0.01:
            break
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= restart_patience:
            a = best_a + noise_scale * np.random.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = _normalize(a, target_sum)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            step_count = 0
            continue

        grad, _ = _lse_subgrad(a, b, top_k=top_k, temp_scale=temp_scale)
        grad = grad - np.mean(grad)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = _normalize(a, target_sum)
        step_count += 1

    return best_a, best_score


def _coord_descent_step(a, n_subset, rng, b=None):
    """One coordinate descent step: optimize a random subset of n_subset coordinates."""
    n = len(a)
    if b is None:
        b = _fft_conv(a, a)

    idx = rng.choice(n, size=n_subset, replace=False)
    idx_sorted = np.sort(idx)
    rest_sum = float(np.sum(a)) - float(np.sum(a[idx]))
    C = max(rest_sum, 0.0)
    if C < 1e-15:
        return a, False

    # Precompute the constant part of each convolution sum (terms not involving idx)
    # L_k(x) = sum_i x[i]*x[k-i], we split into terms with i in idx and not
    # constant_k = sum_{i not in idx} a[i] * a[k-i] (with bounds)
    # variable_k(x_idx) = sum_{i in idx} x[i] * a[k-i] + sum_{i' in idx, i != i'} x[i']*a[k-i] ... 
    # Actually it's quadratic in x_idx: sum_{i in idx} x[i]*x[k-i] includes cross terms
    # Let me just build the full LP

    # The constraint is: for each k, sum_{i in idx} x[i]*a[k-i] + sum_{i in idx, k-i in idx} ... 
    # This is getting complicated. Let me use a simpler approach:
    # Just do a local optimization by trying to reduce the max coordinate-wise.

    # Simpler approach: for each coordinate in the subset, try to adjust it
    # to reduce the max convolution value.
    max_b = float(np.max(b))
    peak_pos = int(np.argmax(b))

    improved = False
    for i in idx:
        # Try reducing a[i] slightly and redistributing to other elements
        # The subgradient at position peak_pos tells us: reducing a[i] helps if a[peak_pos - i] > 0
        if 0 <= peak_pos - i < n:
            contribution = a[peak_pos - i]
            if contribution > 0:
                # Reduce a[i] and increase a random other element
                delta = 0.01 * a[i]
                if delta > 1e-10:
                    a_new = a.copy()
                    a_new[i] -= delta
                    # Distribute delta to a random position that doesn't contribute to the peak
                    other_candidates = np.arange(n)
                    other_candidates = other_candidates[(other_candidates != i)]
                    # Prefer positions where a[peak_pos - j] is small
                    if 0 <= peak_pos - other_candidates.min() < n:
                        pass
                    j = rng.choice(other_candidates)
                    a_new[j] += delta
                    if a_new[i] >= 0:
                        b_new = _fft_conv(a_new, a_new)
                        if float(np.max(b_new)) < max_b:
                            a = a_new
                            b = b_new
                            max_b = float(np.max(b))
                            improved = True

    return a, improved


def _lp_direction(f, max_constraints=200):
    """Use LP to find direction g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    ff = _fft_conv(f, f)
    M = float(np.max(ff))
    M_pad = M * 1.001  # Slight padding for numerical safety

    top_idx = np.argsort(ff)[-max_constraints:]

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(f[i])

    A = csr_matrix((vals, (rows, cols)), shape=(max_constraints, n))
    b_vec = np.full(max_constraints, M_pad)
    c = -np.ones(n)

    try:
        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-7})
        if res.success:
            return res.x
    except Exception:
        pass
    return None
```
