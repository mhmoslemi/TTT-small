"""
Minimize 2n*max(conv(a))/(sum(a))^2 using accelerated gradient methods
with improved top-K subgradients, adaptive restarts, and multi-scale search.
Key improvements over FISTA baseline:
- Larger adaptive top-K (up to all peaks in fine-tuning)

import numpy as np
import time
from scipy.fft import fft, ifft

# Global state
_best_sequence = None
_best_score = np.inf


def fast_conv(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = fft(a, fft_size)
    C = ifft(A * A).real
    return C[:size]


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    """Project v onto the standard probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def frank_wolfe_grad_step(g, x, step):
    """Frank-Wolfe update: move toward the simplex vertex with min g·e_i."""
    # Linear minimization over simplex: pick argmin g_i
    j = np.argmin(g)
    # Find best alpha in [0, step] for (1-alpha)*x + alpha*e_j
    # Minimize f along this line. Approximate with step.
    e_j = np.zeros_like(x)
    e_j[j] = 1.0
    v = e_j - x  # direction
    # Line search: minimize (x + alpha*v) - step * (g · (x + alpha*v))
    # Approximate: alpha = step (simple)
    alpha = min(step, 1.0)
    x_new = (1 - alpha) * x + alpha * e_j
    return x_new


def compute_subgradient(a, c, top_k):
    """Compute subgradient of max(c) using top-k peaks."""
    n = len(a)
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]
    
    g = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            # d(c_k)/d(a[j]) = 2*a[k-j] for j in [lo, hi)
            # k-j ranges from k-lo to k-hi+1
            j = np.arange(lo, hi)
            g[j] += 2.0 * a[k - j]
    g /= float(len(idx))
    return g


def run_optimization(a0, n, duration, rng, top_k_start=5, top_k_end=50):
    """
    Run accelerated gradient with adaptive top-K, backtracking,
    and Nesterov restarts on the probability simplex.
    """
    global _best_sequence, _best_score
    
    a = a0.copy()
    a = project_simplex(a)
    a_prev = a0.copy()
    t_k = 1.0
    step = 0.5  # initial step size
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    
    while time.monotonic() < deadline:
        # Nesterov momentum
        t_old = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_k
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        # Renormalize to simplex
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
        else:
            y /= sy
        
        # Compute convolution and find peaks
        c_y = fast_conv(y)
        
        # Adaptive top-K
        progress = min(1.0, it / max(1, int(duration * 100)))  # rough progress
        current_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        current_k = max(3, min(current_k, len(c_y)))
        
        # Subgradient
        g = compute_subgradient(y, c_y, current_k)
        
        # Gradient step + project
        a_candidate = y - step * g
        a_new = project_simplex(a_candidate)
        
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation
        if it % 50 == 0 or it < 10:
            sc = fast_score(a)
            if sc < best_score:
                # Improvement: reset stuck counter, possibly increase step
                stuck = 0
                best_score = sc
                best_a = a.copy()
                _best_score = min(_best_score, sc)
                _best_sequence = best_a.copy()
            else:
                stuck += 1
            
            # Adaptive step size
            if stuck == 30:
                step *= 0.8
            elif stuck == 80:
                step *= 0.7
                # Nesterov restart
                a_prev = a.copy()
                t_k = 1.0
            elif stuck > 200:
                # Perturb and restart
                a = best_a + rng.normal(0.0, 0.005 / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_k = 1.0
                step = 0.3
                stuck = 0
            
            # Very fine-tuning: use all peaks
            if stuck > 100 and current_k < len(c_y):
                current_k = len(c_y)
        
        if it % 10000 == 0 and it > 0:
            print(f"    it={it}, score={best_score:.8f}, step={step:.4f}, K={current_k}")
    
    # Final

    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

subgradients, backtracking, Nesterov restarts, and multi-scale search.

import numpy as np

from scipy.fft import fft, ifft


def _fft_conv(a):
    """Compute full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize <<= 1
    A = fft(a, fsize)
    C = ifft(A * A).real
    return C[:size]


def _fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2 without calling evaluate_sequence."""
    n = len(a)
    c = _fft_conv(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def _project_simplex(v):
    """Project v onto the probability simplex {a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad(a, c, K):
    """Subgradient of max(c) using top-K peaks. Returns gradient w.r.t. a."""
    n = len(a)
    m = len(c)
    if K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, -K)[-K:]
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            j = np.arange(lo, hi)
            g[lo:hi] += 2.0 * a[k - j]
    g /= float(len(idx))
    return g

    stuck = 0

        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        mu = (t_old - 1.0) / t_k
        y = a + mu * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
        else:
            y /= sy

        c_y = _fft_conv(y)

        # Adaptive K: grow over time

        K = max(K_min, int(K_min + (K_max - K_min) * frac))
        K = min(K, len(c_y))

        g = _subgrad(y, c_y, K)

        a_new = _project_simplex(y - step * g)
        a_prev = a
        a = a_new
