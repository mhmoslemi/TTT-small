"""
Minimize 2*n*max(a*a)/(sum(a))^2 using IRLS with FFT-based
projected gradient, multiple diverse initializations, and
adaptive smoothing.
"""

import numpy as np
import time

def simplex_project(x, target_sum=1.0):
    """Project x onto {a >= 0, sum(a) = target_sum}."""
    n = len(x)
    if n == 0:
        return x
    # Algorithm from Duchi et al. (2011)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target_sum) / rho > 0
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(x - theta, 0)

def convolve_fft(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def hankel_vec(w, a):
    """Compute v[i] = sum_j w[i+j] * a[j] using FFT.
    w has length 2n-1, a has length n, result has length n."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size *= 2
    w_pad = np.zeros(size)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(size)
    a_pad[:n] = a
    result = np.fft.irfft(np.fft.rfft(w_pad) * np.conj(np.fft.rfft(a_pad)), size)
    return result[:n]

def irls_optimize(a, n, conv_size, hankel_size, max_iters=5000, time_limit=60.0, tau_init=0.05, verbose=False):
    """Run IRLS optimization. Returns (best_a, best_score, final_a)."""
    start_time = time.monotonic()
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a /= s
    
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    tau = tau_init
    step = 1.0  # Will be adapted
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Compute convolution
        c = convolve_fft(a)
        
        # Softmax weights
        c_max = c.max()
        log_w = (c - c_max) / tau
        log_w_max = log_w.max()
        w = np.exp(log_w - log_w_max)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum
        
        # Gradient: 2 * W * a (W is Hankel matrix defined by w)
        grad = 2.0 * hankel_vec(w, a)
        
        # Adaptive step size: use 1 / max(grad) as a rough Lipschitz estimate
        grad_max = np.max(np.abs(grad))
        if grad_max > 1e-10:
            step = min(step * 1.05, 0.5 / grad_max)  # Conservative step
        else:
            step = 0.1
        
        # Gradient step
        a_new = a - step * grad
        
        # Project onto simplex
        a_new = simplex_project(a_new, 1.0)
        
        a = a_new
        
        # Evaluate periodically

            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()

            if np.isfinite(score) and score < best_score:

                if verbose:

    score = evaluate_sequence(a.tolist())
    if np.isfinite(score) and score < best_score:

        best_a = a.copy()
    
    return best_a, best_score, a

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using IRLS (Iterative Reweighted Least Squares)
with FFT-based projected gradient descent, multiple diverse initializations,
adaptive smoothing, and structured local search.
Key idea: approximate max(convolution) via softmax-weighted sum, solve resulting
quadratic subproblem with projected gradient, iterate with decreasing temperature.
"""

import numpy as np
import time


def simplex_project(x, target_sum=1.0):
    """Project x onto {a >= 0, sum(a) = target_sum} using Duchi et al. algorithm."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(x - theta, 0)


def convolve_fft(a, conv_size):
    """Compute full convolution of a with itself using precomputed FFT size."""
    n = len(a)
    fa = np.fft.rfft(a, conv_size)
    return np.fft.irfft(fa * fa, conv_size)[:2 * n - 1]


def hankel_vec(w, a, hankel_size):
    """Compute v[i] = sum_j w[i+j] * a[j] using FFT.
    w has length 2n-1, a has length n, result has length n."""
    n = len(a)
    w_pad = np.zeros(hankel_size)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(hankel_size)
    a_pad[:n] = a
    result = np.fft.irfft(np.fft.rfft(w_pad) * np.conj(np.fft.rfft(a_pad)), hankel_size)
    return result[:n]


def get_score(a):
    """Evaluate the sequence safely."""
    try:
        s = evaluate_sequence(a.tolist())
        return s
    except Exception:
        return float('inf')


def irls_optimize(a, n, conv_size, hankel_size, time_limit=60.0, tau_init=0.05,
                  tau_min=0.002, verbose=False, tag=""):
    """Run IRLS optimization with adaptive parameters. Returns (best_a, best_score)."""
    start_time = time.monotonic()
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_score = get_score(best_a)
    tau = tau_init
    step = 0.4
    patience = 0
    max_patience = 2000
    eval_interval = max(20, int(time_limit / 120))

    for it in range(1000000):
        if time.monotonic() - start_time > time_limit:
            break

        # Compute convolution
        c = convolve_fft(a, conv_size)

        # Softmax weights (numerically stable)
        c_max = c.max()
        log_w = (c - c_max) / tau
        lw_max = log_w.max()
        w = np.exp(log_w - lw_max)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum

        # Gradient: 2 * W * a via FFT
        grad = 2.0 * hankel_vec(w, a, hankel_size)

        # Adaptive step size
        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm > 1e-12:
            step = min(step * 1.02, 0.8 / max(grad_norm, 1e-10))
        step = max(step, 0.01)

        # Gradient step + project
        a_new = a - step * grad
        a_new = simplex_project(a_new, 1.0)
        a = a_new

        # Evaluate periodically
        if it % eval_interval == 0:
            score = get_score(a)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                patience = 0
                if verbose:
                    elapsed = time.monotonic() - start_time
                    print(f"    {tag} iter {it}: score={score:.6f}, tau={tau:.5f}, t={elapsed:.0f}s")
            else:
                patience += eval_interval

        # Decrease tau with schedule
        if it % 500 == 0 and tau > tau_min:
            tau *= 0.92

        # Stagnation handling: increase tau to explore
        if patience >= max_patience:
            tau = max(tau * 2.0, 0.02)
            step = max(step * 1.5, 0.1)
            patience = 0
            if verbose:
                print(f"    {tag} iter {it}: restarted with tau={tau:.5f}")

    # Final check
    score = get_score(a)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def subgradient_polish(a, n, conv_size, time_limit=60.0, verbose=False):
    """Polish using exact subgradient steps on the true max."""
    start_time = time.monotonic()
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_score = get_score(best_a)
    step = 0.15

    for it in range(1000000):
        if time.monotonic() - start_time > time_limit:
            break

        c = convolve_fft(a, conv_size)
        max_c = c.max()
        if max_c < 1e-15:
            break

        # Find active set (peaks close to max)
        threshold = max_c * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Subgradient: average of contributions from active peaks
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        a_new = a - step * g
        a_new = simplex_project(a_new, 1.0)
        a = a_new
        step *= 0.9995
        if step < 0.005:
            step = 0.005

        if it % 200 == 0:
            score = get_score(a)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose and it % 2000 == 0:
                    print(f"    SG polish iter {it}: score={score:.6f}")

    score = get_score(a)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def generate_diverse_starts(n, rng, existing_best=None):
    """Generate diverse starting sequences."""
    starts = []
    if existing_best is not None:
        starts.append(existing_best.copy())

    # Flat
    starts.append(np.ones(n) / n)

    # Linear ramps (asymmetric)
    for a0, a1 in [(2.0, 0.3), (0.3, 2.0), (3.0, 0.5), (1.5, 0.2)]:
        x = np.linspace(a0, a1, n)
        starts.append(x / x.sum())

    # Exponential
    for rate in [2.0, 4.0, 6.0]:
        x = np.exp(-rate * np.arange(n) / n)
        starts.append(x / x.sum())
        x2 = np.exp(-rate * (n - 1 - np.arange(n)) / n)
        starts.append(x2 / x2.sum())

    # Cosine modulated
    for freq in [3, 7, 13, 21]:
        x = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        x = np.maximum(x, 0.05)
        starts.append(x / x.sum())

    # Sine (phase-shifted, asymmetric)
    for freq in [5, 11, 17]:
        for phase in [0, np.pi / 3, np.pi / 2]:
            x = 1.0 + 0.4 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
            x = np.maximum(x, 0.05)
            starts.append(x / x.sum())

    # Multi-scale: low + high frequency
    x = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * np.arange(n) / n) + 0.2 * np.sin(2 * np.pi * 19 * np.arange(n) / n)
    x = np.maximum(x, 0.1)
    starts.append(x / x.sum())

    # Two-block
    x = np.ones(n)
    x[:n // 3] = 2.0
    x[n // 3:2 * n // 3] = 1.0
    x[2 * n // 3:] = 0.4
    starts.append(x / x.sum())

    # Three-block
    x = np.ones(n)
    x[:n // 4] = 1.5
    x[n // 4:n // 2] = 0.5
    x[n // 2:3 * n // 4] = 1.8
    x[3 * n // 4:] = 0.7
    starts.append(x / x.sum())

    # Random Dirichlet
    for alpha in [0.3, 1.0, 2.0]:
        starts.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse random
    for p in [0.3, 0.6]:
        x = np.where(rng.random(n) > p, 1.0, 0.15)
        starts.append(x / x.sum())

    # Triangular (peaked middle)
    mid = n // 2
    x = 1.0 - np.abs(np.arange(n) - mid) / (mid * 0.8)
    x = np.maximum(x, 0.1)
    starts.append(x / x.sum())

    # Asymmetric triangular
    x = 1.0 - np.abs(np.arange(n) - n * 0.3) / (n * 0.4)
    x = np.maximum(x, 0.1)
    starts.append(x / x.sum())

    return starts


def local_search(a, n, conv_size, time_limit=120.0, rng=None):
    """Local search with structured perturbations and acceptance."""
    if rng is None:
        rng = np.random.default_rng()

    start_time = time.monotonic()
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_score = get_score(best_a)
    temperature = 0.003

    while time.monotonic() - start_time < time_limit:
        pert_type = rng.integers(0, 6)

        if pert_type == 0:
            # Swap pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                a_new = a.copy()
                a_new[i], a_new[j] = a[j], a[i]
            else:
                a_new = a

        elif pert_type == 1:
            # Gaussian noise
            level = rng.uniform(0.001, temperature)
            a_new = a + rng.normal(0, level, n)

        elif pert_type == 2:
            # Scale random subset
            frac = rng.uniform(0.1, 0.4)
            mask = rng.random(n) < frac
            scale = rng.uniform(0.6, 1.5)
            a_new = a.copy()
            a_new[mask] *= scale

        elif pert_type == 3:
            # Reverse random segment
            length = rng.integers(20, min(300, n // 10))
            i = rng.integers(0, max(1, n - length))
            a_new = a.copy()
            a_new[i:i + length] = a[i:i + length][::-1]

        elif pert_type == 4:
            # Blend with smooth random
            t = rng.uniform(0.05, 0.25)
            smooth = np.ones(n)
            k = rng.integers(2, 20)
            for i in range(n):
                smooth[i] = 1.0 + 0.3 * np.sin(2 * np.pi * k * i / n + rng.uniform(0, 2 * np.pi))
            smooth = np.maximum(smooth, 0.1)
            smooth /= smooth.sum()
            a_new = (1 - t) * a + t * smooth

        else:
            # Move mass from low to high region
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                a_new = a.copy()
                amount = a_new[i] * rng.uniform(0.1, 0.4)
                a_new[i] -= amount
                a_new[j] += amount

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new /= s

        new_score = get_score(a_new)
        if np.isfinite(new_score):
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new.copy()
                temperature = max(temperature * 0.97, 0.001)
            elif rng.random() < np.exp(-(new_score - best_score) / (temperature * 0.01 + 1e-10)):
                a = a_new.copy()

    return best_a, best_score
