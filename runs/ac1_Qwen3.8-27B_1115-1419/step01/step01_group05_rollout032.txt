def _lse_and_p(b, beta):
    if not b.size:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _weighted_gradient(a, weights, fft_size, n):
    """
    Gradient of sum_k weights_k * (a*a)_k with respect to a.
    (a*a)_k = sum_j a_j * a_{k-j}
    d/da_i [(a*a)_k] = 2*a_{k-i} (for 0 <= k-i < n, k != 2i)
                       = a_i (for k = 2i)
    Approximation: 2 * sum_k weights_k * a_{k-i} (close enough for optimization)
    """
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h

def _make_smooth_random(n, rng, base=None, amplitude=0.4):

def _peak_shave(a, b, n, num_peaks=5, strength=0.3, rng=None):
    """
    Direct peak shaving: find the top positions in b=conv(a,a),
    and reduce mass at elements that contribute most to those peaks.
    """
    a = a.copy()
    top_idx = np.argsort(b)[-num_peaks:]
    # For each top peak at position k, the elements a_i that contribute are
    # those where a_{k-i} is large, i.e., i near k (for small i) or i near 0
    # (for k near 0).
    # Simple approach: for each top peak k, find the i that maximizes a_{k-i},
    # and reduce a_i there.
    pressures = np.zeros(n, dtype=np.float64)
    for k in top_idx:
        # a_i contributes to (a*a)_k through a_{k-i}
        # The "contribution" of a_i to peak k is a_i * a_{k-i}
        # To reduce peak k, we want to reduce a_i where a_{k-i} is large
        # This means reducing a_i at positions where a_{k-i} is large
        # i.e., reducing a at positions i where a_{k-i} is in the top values
        
        if k < n:
            # k-i ranges from 0 to k-1 (when i from k to 1)
            # and k-i ranges from k+1 to n-1 (when i from -1 to k-n+1, invalid)
            # Valid: 0 <= k-i < n means max(0, k-n+1) <= i <= min(n-1, k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                segment = a[lo:hi+1]
                # The contribution of a[lo:hi+1] to peak k involves a[k-i]
                # For i in [lo, hi], k-i ranges from k-lo to k-hi (reversed)
                contrib = segment * a[k - lo:k - hi:-1] if hi > lo else segment * a[k-lo:k-hi:-1]
                # Actually let me just use: pressure at position i is a_{k-i}
                # So pressure[lo:hi+1] += a[k-lo : k-hi : -1]
                pressures[lo:hi+1] += a[n-1-hi:n-1-lo-1:-1]  # this is getting complicated
        # Simpler: just use the gradient direction
    # Use the gradient-based approach as fallback
    top_p = np.zeros(2*n-1, dtype=np.float64)
    top_p[top_idx] = 1.0
    # Normalize
    top_p = top_p / top_p.sum()
    g_fft = _next_pow2(3 * n - 2)
    grad = _weighted_gradient(a, top_p, g_fft, n)
    # Multiplicative reduction where gradient is high
    grad_mean = float(np.mean(grad))
    grad_centered = grad - grad_mean
    a = a * np.exp(-strength * grad_centered)
    return _normalize(a)

"""
Search for low-peak nonnegative self-convolutions using multiplicative
(exponential family) gradient descent with log-sum-exp smoothing,
combined with direct peak-shaving moves and structured restarts.

The objective is score(a) = 2*n*max(conv(a,a))/(sum(a))^2, which is
scale-invariant so we normalize sum(a)=1 and minimize 2*n*max(conv(a,a)).

Key innovations:
- Multiplicative (exponential) updates instead of additive projected gradient
- Direct "peak shaving" moves targeting top convolution peaks
- Adaptive beta (temperature) annealing with periodic sharpening
- Diverse structured restarts (symmetry breaking, component swapping)
- Multi-scale search: optimize at lower resolution then upsample
- Conjugate momentum for faster convergence

def _conjugate_step(a, grad_prev, prev_direction, n):
    """Ried Reisen style: new_dir = -grad + beta * prev_dir."""
    if prev_direction is None:
        return -grad_prev.copy()
    dg = float(np.dot(grad_prev, prev_direction))
    if not np.isfinite(dg) or abs(dg) < 1e-300:
        return -grad_prev.copy()
    gm = float(np.dot(grad_prev, grad_prev))
    if not np.isfinite(gm) or gm < 1e-300:
        return -grad_prev.copy()
    beta = float(np.clip(dg / gm, 0.0, 1.5))
    return -grad_prev + beta * prev_direction

    if n < 8:
        return a, 1e100

    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n - 2)

    b = _autocorr(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    beta = beta_start
    alpha = 0.15
    it = 0
    fail = 0
    last_improve = 0
    patience = 2500

    while time.monotonic() < deadline:
        if maxb <= 1e-300:
            maxb = 1e-300
        eff_beta = min(1e6, max(1.0, beta / max(maxb, 1e-10)))
        p, smooth_old = _lse_and_p(b, eff_beta)
        grad = _weighted_gradient(a, p, g_fft, n)
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale <= 1e-300:
            a = _make_smooth_random(n, rng, best_a, 0.3)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.15
            fail = 0
            it += 1
            continue

        if alpha < 0.002:
            alpha = 0.15

        v = a - alpha * grad_c / scale
        v = np.maximum(v, 0.0)
        v_s = float(v.sum())
        if v_s > 1e-300:
            a_new = v / v_s
        else:
            a_new = _make_uniform(n)

        b_new = _autocorr(a_new, b_fft)
        new_maxb = float(np.max(b_new))
        new_score = 2.0 * float(n) * new_maxb
        smooth_new = _lse_value(b_new, eff_beta)

        if smooth_new < smooth_old - 1e-15:
            a = a_new
            b = b_new
            maxb = new_maxb
            score = new_score
            alpha = min(2.0, alpha * 1.1)
            fail = 0
            last_improve = it
        else:
            alpha *= 0.5
            fail += 1

        if score < best_score - 1e-15:
            best_a = a.copy()
            best_score = score

        if it % 60 == 59:
            beta = min(beta_max, beta * 1.12)

        if (fail > 600 or (it - last_improve > patience and it > 300)):
            if time.monotonic() < deadline - 5:
                r = rng.random()
                if r < 0.4:
                    a = best_a * np.exp(rng.normal(0, 0.18, size=n))
                    a = _normalize(a)
                else:
                    a = _make_smooth_random(n, rng, best_a, 0.4)
                b = _autocorr(a, b_fft)
                maxb = float(np.max(b))
                score = 2.0 * float(n) * maxb
                alpha = 0.15
                fail = 0
                beta = max(beta_start, beta * 0.5)

        it += 1

    return best_a, best_score


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or not np.isfinite(arr).all():
        return None
    s = float(arr.sum())
    if s <= 1e-300:
        return None
    return arr / s


def _generate_inits(n, given, rng, best):
    n = max(int(n), 2)
    inits = []

    inits.append(_make_exponential(n, 1.0))
    inits.append(_make_exponential(n, 0.6))
    inits.append(_make_ramp(n, 1.5))
    inits.append(_make_ramp(n, 3.0))
    inits.append(_make_piecewise(n, 0.4, 0.5))
    inits.append(_make_mixture(n, 1.0, 0.5))
    inits.append(_make_smooth_random(n, rng, _make_uniform(n), 0.6))
    inits.append(_make_smooth_random(n, rng, _make_exponential(n, 0.8), 0.4))

    if best is not None and best.size > 4:
        inits.append(_resample_sequence(best, n))
        inits.append(_make_smooth_random(n, rng, _resample_sequence(best, n), 0.3))

    if given is not None and given.size > 4:
        inits.append(_resample_sequence(given, n))

    seen = []
    out = []
    for ic in inits:
        sig = tuple(ic[::max(1, ic.size // 32)])
        if sig not in seen:
            seen.append(sig)
            out.append(ic)
    return out


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + max(budget, 60.0) - 25.0
    rng = np.random.default_rng(seed)

    given = _get_given_sequence()

    best_a = _make_exponential(256, 1.0)
    best_score = _score_of(best_a)
    print(f"initial fallback len={best_a.size} score={best_score:.8f}", flush=True)

    scale = float(max(0.08, min(1.0, budget / 2400.0)))

    # Phase 1: Multi-scale optimization
    configs = [
        (128, 3, 20, 'mul'),
        (256, 4, 50, 'mul'),
        (256, 3, 50, 'add'),
        (512, 3, 80, 'mul'),
        (512, 2, 60, 'add'),
        (1024, 2, 120, 'mul'),
        (2048, 2, 180, 'mul'),
    ]

    for n, num_starts, sec_per, method in configs:
        sec = max(10, int(sec_per * scale))
        if time.monotonic() > deadline - sec - 15:
            break

        inits = _generate_inits(int(n), given, rng, best_a)
        for idx, init in enumerate(inits[:num_starts]):
            if time.monotonic() > deadline - 15:
                break
            t_end = min(deadline - 5.0, time.monotonic() + sec)
            if time.monotonic() >= t_end:
                continue

            if method == 'mul':
                res_a, res_score = _refine_mult(init.copy(), t_end, rng)
            else:
                res_a, res_score = _refine_additive(init.copy(), t_end, rng)

            print(
                f"n={res_a.size} method={method} start={idx}/{num_starts} "
                f"score={res_score:.8f} best={best_score:.8f} "
                f"t={time.monotonic() - (deadline - budget + 25):.1f}s",
                flush=True,
            )
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)

    # Phase 2: If given sequence is available, try direct refinement
    if given is not None and given.size >= 64 and given.size <= 2000:
        if time.monotonic() < deadline - 100:
            sec = max(60, int(100 * scale))
            t_end = min(deadline - 40, time.monotonic() + sec)
            res_a, res_score = _refine_mult(given.copy(), t_end, rng)
            print(f"given direct mul len={res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)

    # Phase 3: Upscale if needed
    if best_score > 1.51 and time.monotonic() < deadline - 120:
        n_cur = int(best_a.size)
        attempts = 0
        while n_cur < 4096 and time.monotonic() < deadline - 60 and attempts < 3:
            n2 = min(4096, max(1024, 2 * n_cur))
            if n2 <= n_cur:
                n2 = 4096
            if n2 <= n_cur:
                break
            init = _resample_sequence(best_a, n2)
            # Mix with exponential to add structure
            init = _normalize(0.8 * init + 0.2 * _make_exponential(n2, 0.8))
            sec = max(40, int(80 * scale))
            t_end = min(deadline - 30, time.monotonic() + sec)
            res_a, res_score = _refine_mult(init, t_end, rng)
            print(f"upscale {n_cur}->{res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                n_cur = int(res_a.size)
                attempts = 0
            else:
                attempts += 1

    # Phase 4: Final deep refinement with both methods
    if time.monotonic() < deadline - 10:
        t_end = min(deadline - 5, time.monotonic() + max(30, int(40 * scale)))
        res_a, res_score = _refine_mult(best_a.copy(), t_end, rng)
        print(f"final mul score={res_score:.8f} best={best_score:.8f}", flush=True)
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)

    if time.monotonic() < deadline - 5:
        t_end = min(deadline - 2, time.monotonic() + max(20, int(25 * scale)))
        res_a, res_score = _refine_additive(best_a.copy(), t_end, rng)
        print(f"final add score={res_score:.8f} best={best_score:.8f}", flush=True)
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)

    # Final sanity check with evaluate_sequence
    try:
        final_score = evaluate_sequence([float(x) for x in out])
        print(f"FINAL: len={out.size} evaluate_score={final_score:.8f}", flush=True)
    except Exception as e:
        print(f"FINAL: len={out.size} error checking: {e}", flush=True)

    return [float(x) for x in out]
