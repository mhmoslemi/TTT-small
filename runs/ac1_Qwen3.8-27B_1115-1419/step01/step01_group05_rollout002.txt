"""
Minimize the peak of the self-convolution of a non-negative sequence.
Score = 2n * max(conv(a,a)) / (sum a)^2.

Uses log-space optimization with LSE-smoothed max, Nesterov momentum,
and multi-start strategy with FFT-based gradient computation.
"""

import numpy as np
import time

def _fft_size(n):
    size = 2*n - 1
    # Find next power of 2 >= size
    p = 1
    while p < size:
        p *= 2
    return p

def _softmax(x):
    m = np.max(x)
    e = np.exp(x - m)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def _score(a):
    """Compute exact score."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    fs = _fft_size(n)
    A = np.fft.rfft(a, fs)
    c = np.fft.irfft(A*A, fs)[:2*n-1]
    return 2.0 * n * np.max(c) / (s*s)

def _compute_c(a, fs):
    """Compute linear convolution a*a via FFT."""
    A = np.fft.rfft(a, fs)
    c = np.fft.irfft(A*A, fs)
    return c[:2*len(a)-1]

def _autocorr_weighted(p, a, fs):
    """Compute h[m] = sum_k p_k * a[k-m] via FFT.
    This is the cross-correlation of p (length 2n-1) with a (length n).
    h[m] for m=0..n-1.
    """
    n = len(a)
    # h[m] = sum_k p[k] * a[k-m] = (p * a_flipped)[m + n - 1]
    # where * is linear convolution
    # Alternatively: h[m] = sum_{k=m}^{m+n-1} p[k] * a[k-m]  (with p padded)
    # Let's use: h = ifft(fft(p)[:n] * conj(fft(a))) ... no this isn't right.
    
    # Direct approach: h[m] = sum_{j=0}^{n-1} p[m+j] * a[j]
    # This is a sliding dot product.
    # Using FFT: create P = fft(p, fs), A = fft(a, fs)
    # h[m] = ifft(P * conj(A))[m + something]
    
    # Actually let's think of it as: h = cross_correlation(p, a) where we take the first n values
    # cross_correlation(p, a)[m] = sum_k p[k+m] * a[k] for valid k
    # = sum_k p[k] * a[k-m] = sum_k p[k] * flip(a)[n-1-k+m] ... 
    
    # Simplest: just do it with the full convolution
    # h[m] = sum_j p[m+j] * a[j] for j=0..n-1 (p is length 2n-1)
    # Flip a: af = a[::-1], length n
    # conv(p, af)[m + n - 1] = sum_j p[m+j] * af[n-1-j] = sum_j p[m+j] * a[j]  ✓
    
    af = a[::-1]
    fs2 = _fft_size(max(len(p), n) + n - 1)  # need size for conv of p (len 2n-1) and af (len n)
    # Actually p has length 2n-1, af has length n, so conv has length 3n-2
    # We only need indices n-1 to 2n-2 (n values)
    
    # For efficiency, use the same fs if it's large enough
    needed = (2*n-1) + n - 1  # = 3n-2
    fs2 = fs  # fs = next_pow2(2n-1), which might not be enough for 3n-2
    # Actually fs = next_pow2(2n-1) >= 2n-1. We need >= 3n-2.
    # For n=256: 2n-1=511, fs=512. 3n-2=766 > 512. Not enough!
    
    # Let's use a larger size
    while fs2 < needed:
        fs2 *= 2
    
    P = np.fft.rfft(p, fs2)
    AF = np.fft.rfft(af, fs2)
    conv = np.fft.irfft(P * AF, fs2)
    h = conv[n-1:2*n-1]  # n values
    return h

def _optimize_logspace(x0, beta0, deadline, rng):
    """
    Optimize in log-space. x = log(a) (unnormalized), a = softmax(x).
    Minimize smoothed max of c = conv(a,a).
    """
    n = len(x0)
    fs = _fft_size(n)
    fs_corr = fs
    while fs_corr < 3*n - 2:
        fs_corr *= 2
    
    x = x0.copy().astype(np.float64)
    # Center x for numerical stability
    x -= np.mean(x)
    
    a = _softmax(x)
    c = _compute_c(a, fs)
    current_score = 2.0 * n * np.max(c)  # with sum a = 1
    
    best_a = a.copy()
    best_score = current_score
    
    beta = beta0
    alpha = 0.01  # step size
    momentum = 0.9  # Nesterov momentum
    
    v = np.zeros(n)  # velocity for momentum
    consecutive_fail = 0
    last_improve_iter = 0
    it = 0
    
    while time.monotonic() < deadline:
        # Compute LSE-smoothed max
        m_c = np.max(c)
        beta_c = beta * m_c if m_c > 0 else beta
        # Actually, let's use a fixed beta and just work with c directly
        # smoothed = (1/beta) * log(sum exp(beta*c))
        
        arg = beta * (c - m_c)
        arg = np.clip(arg, -700, 700)
        p = np.exp(arg)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones_like(c) / len(c)
            p_sum = len(c)
        else:
            p = p / p_sum
        
        smoothed_max = m_c + np.log(p_sum) / beta
        
        # Compute gradient
        # grad[m] = 2*a[m]/beta * (h[m] - g_val)
        # where h[m] = sum_k p[k]*a[k-m], g_val = sum_k p[k]*c[k]
        
        h = _autocorr_weighted(p, a, fs_corr)  # h has length n
        g_val = np.dot(p, c)
        
        grad = 2.0 * a / beta * (h - g_val)
        
        # Center the gradient (invariance to adding constant to x)
        grad -= np.mean(grad)
        
        # Nesterov step
        v = momentum * v + grad
        x_new = x - alpha * v
        
        # Compute new objective
        a_new = _softmax(x_new)
        c_new = _compute_c(a_new, fs)
        m_new = np.max(c_new)
        score_new = 2.0 * n * m_new
        
        # Adaptive step size
        if score_new < current_score:
            x = x_new
            a = a_new
            c = c_new
            current_score = score_new
            consecutive_fail = 0
            if alpha < 0.5:
                alpha *= 1.05
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
                last_improve_iter = it
        else:
            # Reject step, reduce step size
            alpha *= 0.7
            consecutive_fail += 1
            # Don't update x or v; just reduce momentum effect
            v *= 0.5
        
        # Increase beta periodically
        if it % 200 == 0 and it > 0:
            beta *= 1.3
            alpha *= 0.5
        
        # Momentum decay in plateau
        if it - last_improve_iter > 1000 and it > 200:
            momentum = max(0.0, momentum - 0.05)
            alpha *= 0.5
        
        if consecutive_fail > 500:
            # Restart from best with perturbation
            x = np.log(np.maximum(best_a, 1e-30))
            x -= np.mean(x)
            x += rng.normal(0, 0.3, size=n)
            v = np.zeros(n)
            alpha = 0.01
            beta = max(10.0, beta * 0.5)
            consecutive_fail = 0
        
        it += 1
    
    return best_a, best_score

def _optimize_simplex(a0, deadline, rng):

    n = len(a0)
    fs = _fft_size(n)

    a = a / a.sum()
    
    current_score = 2.0 * n * np.max(_compute_c(a, fs))
    best_a = a.copy()
    best_score = current_score
    
    c = _compute_c(a, fs)
    beta = 20.0
    alpha = 0.1
    momentum = 0.85
    v = np.zeros(n)
    consecutive_fail = 0
    last_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        m_c = np.max(c)
        arg = beta * (c - m_c)
        arg = np.clip(arg, -700, 700)
        p = np.exp(arg)
        ps = p.sum()
        if ps < 1e-300:
            p = np.ones_like(c) / len(c)
        else:
            p /= ps
        
        # Gradient of LSE(beta*c)/beta w.r.t. a:
        # c_k = sum_i a[i]*a[k-i]
        # dc_k/da_m = a[k-m] + a[m-k] (both terms contribute, with 0 if out of range)
        # Wait: c_k = sum_i a[i]*a[k-i]
        # dc_k/da_m: terms where i=m: a[k-m]; terms where k-i=m, i=k-m: a[k-m]
        # So dc_k/da_m = a[k-m] + (a[m-k] if m!=k-m else a[k-m])

            v *= 0.5
            consecutive_fail += 1
        
        if it % 300 == 0 and it > 0:

            alpha *= 0.5

            a = best_a.copy() + rng.normal(0, 0.05, n) * best_a
            a = np.maximum(a, 0)
            a /= a.sum()
            c = _compute_c(a, fs)
            current_score = 2.0 * n * np.max(c)
            v = np.zeros(n)

def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Uses log-space and simplex-space optimization with LSE-smoothed max,
    Nesterov momentum, multi-start, and multi-scale strategy.
    """
    deadline = time.monotonic() + budget_s - 30
    rng = np.random.default_rng(seed)
    
    # Get initial sequence if available
    init_seq = None
    try:
        raw = globals().get('height_sequence_1', None)
        if raw is not None:
            init_seq = np.array(raw, dtype=np.float64).ravel()
            init_seq = np.maximum(init_seq, 0)
            if init_seq.sum() > 0.01:
                init_seq = init_seq / init_seq.sum()
    except:
        pass
    
    best_a = None
    best_score = np.inf
    
    def try_sequence(a, label=""):
        nonlocal best_a, best_score
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        sc = 2.0 * len(a) * np.max(_compute_c(a, _fft_size(len(a))))
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  NEW BEST [{label}] n={len(a)} score={sc:.8f}", flush=True)
        return sc
    
    # --- Phase 1: Try different starts and lengths ---
    print(f"Phase 1: Multi-start optimization", flush=True)
    
    # Initial candidates
    inits = []
    
    # From given sequence
    if init_seq is not None:
        inits.append(('given', init_seq.copy()))
    
    # Exponential
    for n in [128, 256, 512, 1024, 2048]:
        x = np.arange(n) / n
        a = np.exp(-2.0 * x)
        inits.append((f'exp_{n}', a / a.sum()))
    
    # Symmetric exponential
    for n in [256, 512, 1024]:
        x = np.arange(n)
        xm = (x - n/2) / (n/2)
        a = np.exp(-1.5 * xm**2)
        inits.append((f'gauss_{n}', a / a.sum()))
    
    # Smooth random
    for n in [256, 512]:
        z = rng.normal(size=n)
        F = np.fft.rfft(z)
        F[2:] *= 0.1
        F[4:] *= 0.01
        a = np.fft.irfft(F, n)
        a = np.abs(a) + 0.1
        inits.append((f'rand_{n}', a / a.sum()))
    
    # Time budget allocation
    total_starts = len(inits)
    time_per_start = max(30, min(300, (budget_s - 300) / max(1, total_starts)))
    
    for label, a_init in inits:
        if time.monotonic() > deadline - 60:
            break
        if best_score < 1.505:  # Already at target
            break
        t_end = min(deadline - 30, time.monotonic() + time_per_start)
        
        sc = try_sequence(a_init, label)
        
        # Simplex optimization
        a_opt, sc_opt = _optimize_simplex(a_init.copy(), t_end, rng)
        sc_opt = try_sequence(a_opt, label+'_opt')
    
    # --- Phase 2: Upscale best solution ---
    print(f"Phase 2: Upscaling. Current best: {best_score:.8f}", flush=True)
    
    if best_a is not None and best_score > 1.505:
        cur_n = len(best_a)
        # Try upscaling to larger n
        for target_n in [512, 1024, 2048, 4096, 8192]:
            if target_n <= cur_n:
                continue
            if time.monotonic() > deadline - 120:
