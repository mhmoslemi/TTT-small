    cols = []

        # Perturb and smooth
        rng = np.random.default_rng(seed + attempt * 1000)
        sigma = 0.05 + 0.05 * min(attempt, 10)
        a_p = _project(best_a * np.exp(rng.normal(0, sigma, n)), n)
        sub2 = min(time.monotonic() + 10.0, deadline - 3.0)
        a4, M4 = _smooth_run(a_p, n, sub2, f"s{attempt}", max_iter=30000)
        if M4 < best_M:
            best_M = M4
            best_a = a4
            print(f"  Pert+smooth {attempt}: M={best_M:.4f} obj={best_M*2/n:.6f}")
        
        # Occasionally try LP from perturbed point
        if attempt % 3 == 0 and time.monotonic() < deadline - 20:
            for _ in range(5):
                M_cur = float(np.max(_full_conv(a4)))
                g = _lp_direction(a4, n, row_idx, col_idx, M_cur)
                if g is not None and np.sum(g) > 1e-10:
                    g = g / np.sum(g) * n
                    for tt in [0.1, 0.3, 0.5]:
                        a_try = np.maximum((1-tt)*a4 + tt*g, 0)
                        S = float(a_try.sum())
                        if S > 1e-10:
                            a_try = a_try / S * n
                            M_try = float(np.max(_full_conv(a_try)))
                            if M_try < M_cur:
                                a4 = a_try
                                M_cur = M_try
            if float(np.max(_full_conv(a4))) < best_M:
                best_M = float(np.max(_full_conv(a4)))
                best_a = a4
                print(f"  LP from pert {attempt}: M={best_M:.4f}")
    
    # Final validation
    result = [float(x) for x in best_a]
    sc = float(evaluate_sequence(result))
    print(f"FINAL: obj_internal={2*best_M/n:.6f} evaluated={sc:.6f} n={n}")
    return result
