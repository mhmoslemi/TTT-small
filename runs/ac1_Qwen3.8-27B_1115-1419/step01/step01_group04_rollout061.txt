```
def fast_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.empty(0, dtype=np.float64)
    L = 2*n - 1
    N = next_power_of_two(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c
```

```
def make_harmonic_start(n, rng, max_freq=6, max_amp=2.0):
    n=int(n)
    if n <2: return normalize_to_sum(np.ones(n),1.0)
    t=np.linspace(0.0,1.0,n)
    z=np.zeros(n,float)
    for m in range(1,max_freq+1):
       amp=rng.uniform(-max_amp,max_amp)/( (m+1.0)**0.7)
       ph=rng.uniform(0.0,2.0*np.pi)
       z += amp*np.cos(2.0*np.pi*m*t + ph)
    amp=rng.uniform(-max_amp,max_amp)
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a)^2) over non-negative finite coefficient lists.

The objective is homogeneous, so the search keeps probabilities p = a / sum(a)
and returns an arbitrary valid scaling at the end.

Main efficient optimizer:
- A dense exponential-weighted subgradient / annealed log-sum-exp natural gradient
  on the log-probability simplex.
- Peaks c_k are weighted by softmax(beta*c_k) with beta made adaptively large.
- The gradient is computed with FFT in O(n log n).
- Line searches, annealing emphasis, and smooth random restarts are used.

Starts are generated from:
- height_sequence_1 if available,
- one-sided fractional power-law tails, e.g. (i+gamma)^(-alpha)*(1-x)^beta,
- random low-frequency exponential/harmonic shapes,
- upsamples of previously best sequences in a small pool.

A sparse top-peak search is used as a final polish. Exact evaluator calls are
reserved for the last few-best candidates.
"""

import numpy as np
import time


def _pair_score(pair):
    return pair[0]


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_to_sum(a, target=1.0):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = float(a.sum())
    if s <= 1e-300:
        a.fill(float(target) / float(max(1, a.size)))
        return a

    a *= float(target) / s
    return a


def fast_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    L = 2 * n - 1
    N = next_power_of_two(L)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    n = a.size
    if n == 0:
        return float("inf")

    s = float(a.sum())
    if s < 1e-300:
        return float("inf")

    p = a / s
    c = fast_convolution(p)
    if c.size == 0:
        return float("inf")

    return float(2.0 * n * float(np.max(c)))


def append_start(starts, a):
    try:
        arr = normalize_to_sum(a, 1.0)
        sc = fast_score(arr)
        if np.isfinite(sc):
            starts.append((float(sc), arr))
    except Exception:
        pass


def make_power_start(n, alpha, gamma, tail=0.0, side=1):
    n = int(n)
    idx = np.arange(n, dtype=np.float64)
    if n > 1:
        x = idx / float(n - 1)
    else:
        x = np.zeros(n, dtype=np.float64)

    gamma = float(max(1e-3, gamma))
    alpha = float(alpha)

    if side > 0:
        base = (idx + gamma) ** (-alpha)
        tail_pos = 1.0 - x
    else:
        base = (float(n - 1) - idx + gamma) ** (-alpha)
        tail_pos = x

    if abs(float(tail)) > 1e-12:
        if float(tail) < 0.0:
            base = base * (tail_pos + 1e-7) ** float(tail)
        else:
            base = base * (tail_pos + 1e-15) ** float(tail)

    base = np.where(np.isfinite(base), base, 0.0)
    base = np.maximum(base, 0.0)
    return normalize_to_sum(base, 1.0)


def make_exp_start(n, rng):
    n = int(n)
    if n <= 2:
        return normalize_to_sum(np.ones(n, dtype=np.float64), 1.0)

    t = np.linspace(0.0, 1.0, n)
    eps = 1e-7
    z = (
        rng.normal(0.0, 0.9) * np.log(t + eps)
        + rng.normal(0.0, 0.9) * np.log(1.0 - t + eps)
    )

    for m in range(1, 6):
        amp = rng.normal(0.0, 1.0 / (m + 0.5))
        ph = rng.normal(0.0, 1.0)
        z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)

    z -= float(np.max(z))
    a = np.exp(z)
    return normalize_to_sum(a, 1.0)


def make_harmonic_start(n, rng, max_freq=7, max_amp=2.2):
    n = int(n)
    if n <= 2:
        return normalize_to_sum(np.ones(n, dtype=np.float64), 1.0)

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)

    for m in range(1, max_freq + 1):
        amp = rng.uniform(-max_amp, max_amp) / ((float(m) + 1.0) ** 0.7)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)

    amp = rng.uniform(-max_amp, max_amp)
    z += amp * np.cos(np.pi * (2.0 * t - 1.0))

    z -= float(np.max(z))
    a = np.exp(z)
    return normalize_to_sum(a, 1.0)


def smooth_perturb(a, rng, strength=0.25):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n == 0:
        return a

    if n > 12:
        t = np.linspace(0.0, 1.0, n)
        z = np.zeros(n, dtype=np.float64)
        for m in range(1, 7):
            amp = float(strength) / (0.5 + 0.7 * float(m))
            ph = rng.uniform(0.0, 2.0 * np.pi)
            z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)
        a *= np.exp(z)

    if n > 24:
        lo = max(1, n // 500)
        hi = max(lo + 1, n // 5 + 1)
        if hi > n:
            hi = n
        width = int(rng.integers(lo, hi))
        start_max = max(1, n - width + 1)
        start = int(rng.integers(0, start_max))
        stop = min(n, start + width)
        if stop > start:
            a[start:stop] *= np.exp(rng.normal(0.0, 0.5 * float(strength), stop - start))

    return normalize_to_sum(a, 1.0)


def upsample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64)
    n = int(n)
    if a.size == 0:
        return normalize_to_sum(np.ones(n, dtype=np.float64), 1.0)

    m = a.size
    if m == n:
        return normalize_to_sum(a, 1.0)

    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)

    if np.all(a > 1e-12):
        yold = np.log(a + 1e-300)
        ynew = np.exp(np.interp(xnew, xold, yold))
    else:
        ynew = np.interp(xnew, xold, a)

    return normalize_to_sum(ynew, 1.0)


def generate_starts(n, parent, pool, rng, max_starts, random_count):
    n = int(n)
    starts = []

    if max_starts <= 0:
        return starts

    parents = []
    if parent is not None:
        parents.append(parent)

    pool_top = min(3, len(pool))
    for i in range(pool_top):
        parents.append(pool[i][1])

    for i in range(len(parents)):
        if len(starts) >= max_starts:
            break
        arr = parents[i]
        if arr is None:
            continue
        a = upsample_sequence(arr, n)
        append_start(starts, a)
        if len(starts) < max_starts:
            a_rev = upsample_sequence(arr[::-1], n)
            append_start(starts, a_rev)

    if len(starts) < max_starts:
        append_start(starts, np.ones(n, dtype=np.float64))

    # One-sided fractional power tails are the main analytic family.
    # Reversal is score-symmetric, so side=1 is sufficient for coverage.
    alphas = (0.30, 0.35, 0.40, 0.45, 0.50, 0.20, 0.25, 0.55, 0.00, -0.25, 0.60)
    tails = (1.0, 0.8, 1.2, 0.5, 1.5, 0.0, 2.0, 0.3, 1.8)
    gammas = (0.7, 1.2, 2.5, float(np.sqrt(float(n))), float(n) / 50.0)

    for alpha in alphas:
        if len(starts) >= max_starts:
            break
        for tail in tails:
            if len(starts) >= max_starts:
                break
            for gamma in gammas:
                if len(starts) >= max_starts:
                    break
                a = make_power_start(n, alpha, gamma, tail, 1)
                append_start(starts, a)

    random_left = max(0, min(int(random_count), max_starts - len(starts)))
    for _ in range(random_left):
        if rng.random() < 0.6:
            a = make_exp_start(n, rng)
        else:
            a = make_harmonic_start(n, rng)
        append_start(starts, a)

    starts.sort(key=_pair_score)
    return starts


def add_candidate_pool(pool, seq, score):
    try:
        sc = float(score)
    except Exception:
        return
    if seq is None or not np.isfinite(sc):
        return

    arr = np.asarray(seq, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)

    pool.append((sc, arr))
    pool.sort(key=_pair_score)
    if len(pool) > 10:
        pool.pop()


def soft_dense_search(a, budget_s, seed=0, print_label=""):
    a = normalize_to_sum(a, 1.0)
    n = a.size
    if n < 2 or budget_s <= 0.0:
        return a, fast_score(a)

    N = next_power_of_two(3 * n - 2)
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))

    current = a.copy()
    best = a.copy()
    cur_score = fast_score(current)
    best_score = cur_score

    emph = 8.0
    emph_min = 3.0
    emph_max = 1000.0

    target = 0.35
    no_improve = 0
    last_print = t0
    it = 0

    while time.monotonic() < deadline - 1e-6:
        it += 1
        if time.monotonic() < deadline - 0.25:
            break

        try:
            P = np.fft.rfft(current, N)
            c = np.fft.irfft(P * P, N)[:2 * n - 1]
            np.maximum(c, 0.0, out=c)

            cmax = float(np.max(c))
            if cmax <= 1e-300:
                break

            # Softmax weights over convolution peaks, stabilized by subtracting cmax.
            z = emph * (c - cmax) / cmax
            w = np.exp(z)
            Z = float(np.sum(w))
            if not np.isfinite(Z) or Z <= 0.0:
                break
            w /= Z

            # Gradient h_i = 2 sum_k w_k p_{k-i}.
            wR = w[::-1]
            W = np.fft.rfft(wR, N)
            Hconv = np.fft.irfft(P * W, N)
            h = 2.0 * Hconv[n - 1:2 * n - 1][::-1]
            h = np.where(np.isfinite(h), h, 0.0)

            center = float(np.mean(h))
            h = h - center
            scale = float(np.mean(np.abs(h)))
            if scale < 1e-12:
                no_improve += 1
                if no_improve > 200:
                    current = smooth_perturb(best, rng, 0.25 + 0.30 * rng.random())
                    cur_score = fast_score(current)
                    no_improve = 0
                    target = 0.40
                    emph = max(emph_min, min(emph_max, 8.0))
                continue

            old_score = cur_score
            accepted = False

            for mult in (1.0, 0.5, 0.25, 0.125):
                eff_target = target * mult
                log_shift = -(eff_target / scale) * h
                max_shift = 2.0 * eff_target + 1e-12
                np.clip(log_shift, -max_shift, max_shift, out=log_shift)

                cand = current * np.exp(log_shift)
                cand = np.where(np.isfinite(cand), cand, 0.0)
                cand = np.maximum(cand, 0.0)
                cand = cand + 1e-18
                cand = normalize_to_sum(cand, 1.0)

                cand_score = fast_score(cand)

                if np.isfinite(cand_score) and cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score

                if np.isfinite(cand_score) and cand_score <= old_score + 1e-12:
                    current = cand
                    cur_score = cand_score
                    accepted = True
                    no_improve = 0

                    if cur_score < old_score - 1e-13:
                        target = min(1.0, target * 1.10)
                        emph = min(emph_max, emph * 1.07)
                    else:
                        target = max(0.08, target * 0.95)
                        emph = max(emph_min, emph * 0.995)

                    if time.monotonic() - last_print > 20.0:
                        print(
                            "soft_dense",
                            print_label,
                            "it=", it,
                            "score=%.7f" % cur_score,
                            "best=%.7f" % best_score,
                            "emph=%.1f" % emph,
                        )
                        last_print = time.monotonic()
                    break

            if not accepted:
                target = max(0.05, target * 0.75)
                emph = max(emph_min, emph * 0.97)
                no_improve += 1
                if no_improve > 250:
                    current = smooth_perturb(best, rng, 0.20 + 0.40 * rng.random())
                    cur_score = fast_score(current)
                    no_improve = 0
                    target = 0.40
                    emph = max(emph_min, min(emph_max, 6.0))

        except Exception:
            break

    return best, best_score


def gradient_topk(p, c, K):
    p = np.asarray(p, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = p.size
    L = c.size
    K = int(max(1, min(K, L)))

    grad = np.zeros(n, dtype=np.float64)

    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]

    inv = 1.0 / float(K)
    for kk in idx:
        k = int(kk)
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        src_start = k - (stop - 1)
        src_end = k - start + 1
        grad[start:stop] += inv * p[src_start:src_end][::-1]

    return 2.0 * grad


def topk_search(a, budget_s, seed=0, print_label=""):
    a = normalize_to_sum(a, 1.0)
    n = a.size
    if n < 2 or budget_s <= 0.0:
        return a, fast_score(a)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))

    current = a.copy()
    best = a.copy()
    cur_score = fast_score(current)
    best_score = cur_score

    K = max(4, min(96, n // 16))
    target = 0.40
    mom_scale = 0.45
    mom = np.zeros(n, dtype=np.float64)
    no_improve = 0
    acc_streak = 0
    it = 0

    while time.monotonic() < deadline - 1e-6:
        it += 1
        if time.monotonic() < deadline - 0.25:
            break

        try:
            c = fast_convolution(current)
            grad = gradient_topk(current, c, K)

            g = grad - float(np.mean(grad))
            scale = float(np.mean(np.abs(g)))
            if scale < 1e-12:
                no_improve += 1
                if no_improve > 120:
                    current = smooth_perturb(best, rng, 0.25 + 0.30 * rng.random())
                    cur_score = fast_score(current)
                    no_improve = 0
                    target = 0.45
                    K = max(4, min(96, n // 16))
                continue

            g /= scale
            mom = mom_scale * mom + (1.0 - mom_scale) * g

            old_score = cur_score
            accepted = False

            for mult in (1.0, 0.5, 0.25):
                eff_target = target * mult
                log_shift = -eff_target * mom
                max_shift = 2.0 * eff_target + 1e-12
                np.clip(log_shift, -max_shift, max_shift, out=log_shift)

                cand = current * np.exp(log_shift)
                cand = np.where(np.isfinite(cand), cand, 0.0)
                cand = np.maximum(cand, 0.0)
                cand = cand + 1e-18
                cand = normalize_to_sum(cand, 1.0)

                cand_score = fast_score(cand)

                if np.isfinite(cand_score) and cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score

                if np.isfinite(cand_score) and cand_score <= old_score + 1e-12:
                    current = cand
                    cur_score = cand_score
                    accepted = True
                    no_improve = 0
                    acc_streak += 1

                    if cur_score < old_score - 1e-13:
                        target = min(1.0, target * 1.08)
                        mom_scale = min(0.75, mom_scale + 0.02)
                    else:
                        target = max(0.10, target * 0.95)
                        mom_scale = max(0.15, mom_scale * 0.98)

                    if acc_streak % 250 == 0:
                        K = max(4, int(float(K) * 0.90))

                    if time.monotonic() - t0 > 1.0 and it % 200 == 0:
                        print(
                            "topk",
                            print_label,
                            "it=", it,
                            "score=%.7f" % cur_score,
                            "best=%.7f" % best_score,
                            "K=", K,
                        )
                    break

            if not accepted:
                target = max(0.06, target * 0.75)
                mom_scale = max(0.15, mom_scale * 0.85)
                no_improve += 1
                if no_improve > 120:
                    current = smooth_perturb(best, rng, 0.20 + 0.40 * rng.random())
                    cur_score = fast_score(current)
                    no_improve = 0
                    target = 0.45
                    mom_scale = 0.45
                    if rng.random() < 0.5:
                        K = max(4, K // 2)
                    else:
                        K = max(4, min(96, n // 16))

        except Exception:
            break

    return best, best_score


def run_level(n, budget_s, parent, pool, seed, label):
    n = int(n)
    budget_s = float(max(0.0, budget_s))
    if n <= 0:
        return parent, float("inf")
    if budget_s <= 0.0:
        return parent, fast_score(parent)

    t0 = time.monotonic()
    deadline = t0 + budget_s

    rng = np.random.default_rng(seed)

    if n <= 1024:
        max_starts = 180
        rand_count = 30
    elif n <= 2048:
        max_starts = 140
        rand_count = 22
    elif n <= 4096:
        max_starts = 100
        rand_count = 14
    else:
        max_starts = 55
        rand_count = 8

    starts = generate_starts(n, parent, pool, rng, max_starts, rand_count)
    if not starts:
        starts = [(2.0, normalize_to_sum(np.ones(n, dtype=np.float64), 1.0))]

    starts.sort(key=_pair_score)
    best_seq = starts[0][1].copy()
    best_score = float(starts[0][0])
    add_candidate_pool(pool, best_seq, best_score)

    if budget_s < 25.0:
        return best_seq, best_score

    if budget_s < 100.0:
        k = min(len(starts), 2)
    elif n >= 4096:
        k = min(len(starts), 4)
    else:
        k = min(len(starts), 5)

    if k == 1:
        weights = [1.0]
    elif k == 2:
        weights = [0.55, 0.45]
    elif k == 3:
        weights = [0.50, 0.30, 0.20]
    elif k == 4:
        weights = [0.45, 0.27, 0.18, 0.10]
    else:
        weights = [0.40, 0.25, 0.18, 0.12, 0.05]

    for i in range(k):
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break

        alloc = weights[i] * budget_s
        time_this = min(alloc, rem - 2.0)
        if time_this < 5.0:
            break

        try:
            seq, sc = soft_dense_search(
                starts[i][1],
                time_this,
                seed=int(seed * 1000 + i * 17 + 1),
                print_label=label,
            )
            add_candidate_pool(pool, seq, sc)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_seq = seq
        except Exception as e:
            print("run_level search error", label, i, e)

    return best_seq, best_score


def sanitize_for_return(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return [1.0] * 1024

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = float(a.sum())
    if s < 1e-300:
        return [1.0] * 1024

    n = a.size
    scale = float(n) / s
    maxv = float(a.max())

    if maxv > 0.0 and maxv * scale > 900.0:
        scale = 900.0 / maxv

    a = a * scale
    if float(a.sum()) < 0.01:
        return [1.0] * 1024

    a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]
```
