import numpy as np
import time

def compute_subgradient(a, topk=10):
    """Compute the (average) subgradient of max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    # Find top-k peaks
    if topk >= len(c):
        active = np.arange(len(c))
    else:
        active = np.argsort(c)[::-1][:topk]
    
    g = np.zeros(n)
    for k in active:
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d/d a[i] of (a*a)[k] = a[k-i] (for valid k-i)
        lo = max(0, k - (n-1))  # minimum i such that k-i <= n-1
        hi = min(n-1, k)        # maximum i such that k-i >= 0
        if lo <= hi:
            # a[k-i] for i in [lo, hi]
            # k-i ranges from k-lo to k-hi
            g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    
    g /= len(active)
    return g, c.max()

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def bb_step_size(a_k, a_k1, g_k, g_k1):
    """Barzilai-Borwein step size."""
    s = a_k - a_k1
    y = g_k - g_k1
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    if yy < 1e-30:
        return 0.01
    if sy <= 0:
        return 0.01  # Fallback
    step = sy / yy
    return np.clip(step, 0.001, 0.1)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]

    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    c_max = c.max()
    c_shifted = c - c_max
    vals = np.exp(c_shifted / tau)

    topk_end = min(100, n)

        x_new = project_simplex(x_new)

        x = x_new

        a0 = np.exp(-2.0 * np.arange(n) / n)

    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    best_score = evaluate_sequence(a0.tolist())
    best_a = a0.copy()
    print(f"Initial: n={n}, score={best_score:.6f}")

    deadline = time.monotonic() + budget_s - 60
    rng = np.random.default_rng(777)

    # Phase 1: Generate diverse initializations (quick)
    print("\nPhase 1: Generating initializations...")
    inits = [a0.copy()]

    # Various parametric families
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        x = np.exp(-alpha * np.arange(n) / n)
        inits.append(x / x.sum())

    for beta in [1.5, 2.0, 3.0]:
        x = (1.0 + np.arange(n) / n) ** (-beta)
        inits.append(x / x.sum())

    # Asymmetric with oscillation
    for freq in [3, 7, 15]:
        for phase in [0, 0.5]:
            x = np.exp(-1.5 * np.arange(n) / n) * (1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase * np.pi))
            x = np.maximum(x, 1e-10)
            inits.append(x / x.sum())

    # Linear ramp
    x = np.linspace(3.0, 0.1, n)
    inits.append(x / x.sum())

    # Two-block
    x = np.ones(n)
    x[:n // 3] = 2.0
    x[n // 3:2 * n // 3] = 1.0
    x[2 * n // 3:] = 0.3
    inits.append(x / x.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Evaluate all inits quickly
    print(f"  Evaluating {len(inits)} initializations...")
    for i, init in enumerate(inits):
        s = evaluate_sequence(init.tolist())
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = init.copy()
    print(f"  Best after init evaluation: {best_score:.6f}")

    # Phase 2: Main optimization - Nesterov subgradient on best init
    print(f"\nPhase 2: Nesterov subgradient optimization...")
    time_for_nesterov = min(900, (deadline - time.monotonic()) * 0.4)
    a, score = nesterov_subgradient(best_a, n, max_time=time_for_nesterov, rng=rng)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    print(f"  After Nesterov: {best_score:.6f}")

    # Phase 3: L-BFGS restarts from current best
    print(f"\nPhase 3: L-BFGS restarts...")
    time_remaining = deadline - time.monotonic()
    time_per_lbfgs = min(120, time_remaining * 0.6 / 3)

    for restart in range(3):
        if time.monotonic() > deadline - 300:
            break
        print(f"  L-BFGS restart {restart + 1} (time={time_per_lbfgs:.0f}s)")
        a, score = run_smooth_lbfgs(best_a, n, max_time=time_per_lbfgs, rng=rng)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    Improved to: {best_score:.6f}")
        else:
            print(f"    No improvement (score={score:.6f})")
            # Try from a perturbed version
            a_pert = best_a + rng.normal(0, 0.01, n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a, score = run_smooth_lbfgs(a_pert, n, max_time=time_per_lbfgs / 2, rng=rng)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"    Perturbed version improved to: {best_score:.6f}")

    # Phase 4: Fine polishing
    print(f"\nPhase 4: Fine polishing...")
    time_remaining = deadline - time.monotonic()
    time_for_polish = min(400, time_remaining * 0.7)
    a, score = fine_polish(best_a, n, max_time=time_for_polish, rng=rng)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    print(f"  After polishing: {best_score:.6f}")

    # Phase 5: Final Nesterov from polished result
    if time.monotonic() < deadline - 30:
        print(f"\nPhase 5: Final Nesterov from polished...")
        time_for_final = min(200, deadline - time.monotonic() - 20)
        a, score = nesterov_subgradient(best_a, n, max_time=time_for_final,
                                        rng=np.random.default_rng(999))
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  Final Nesterov improved to: {best_score:.6f}")

    # Final verification
    final_score = evaluate_sequence(best_a.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL BEST SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_a)}")
    print(f"{'='*50}")

    return [float(val) for val in best_a]
