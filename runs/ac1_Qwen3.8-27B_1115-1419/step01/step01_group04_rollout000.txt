- Pad I to length 2L-1 (or next power of 2)
- Pad a to same length
- `(I ⊛ a) = ifft(conj(fft(I)) * fft(a))`? 

Actually, the DFT of cross-correlation:
If `r_{xy}[k] = sum_n x[n+k] * conj(y[n])` (cross-correlation), then `DFT(r) = conj(X) * Y`.

In our case, everything is real, so conj = identity.

Let me define:
- `x = I` (length L, padded to N)
- `y = a` (length n, padded to N)
- We want `r[k] = sum_n x[n+k] * y[n]` for k = 0, ..., n-1

This is `r = ifft(fft(x) * fft(y))`? No, that gives the convolution, not cross-correlation.

Convolution: `(x * y)[k] = sum_n x[n] * y[k-n]`
Cross-correlation: `(x ⊛ y)[k] = sum_n x[n+k] * y[n]`

Note: `(x ⊛ y)[k] = (x * y_rev)[k + len(y) - 1]` where y_rev = y reversed.

Or: `fft(x ⊛ y) = conj(fft(x)) * fft(y)` (for the non-reversed version).

Let me just use numpy's correlate or compute it directly for small K.

Actually, for K up to ~1000 and n=4096, the direct computation is:
- For each of K indices: add up to n values
- Total: K * n = 1000 * 4096 = 4M floating point additions

At ~1 GB/s memory bandwidth, this is ~30ms. That's too slow for many iterations.

With FFT:
- One FFT of I (length ~8192): ~100μs
- One FFT of a (length ~8192): ~100μs (can precompute)
- Pointwise multiply: ~10μs
- One IFFT: ~100μs
- Total: ~300μs per subgradient

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 via multiplicative gradient descent
with adaptive top-K subgradient, momentum, and periodic smooth restarts.
Uses FFT convolution for speed. Multi-resolution search from n=256 to n=4096.
"""

import numpy as np
import time
import math


def _norm(a, target):
    a = np.clip(np.nan_to_num(a), 0.0, 1000.0)
    for _ in range(4):
        s = float(a.sum())
        if s < 1e-15:
            a.fill(target / len(a))
            return a
        a *= target / s
        np.clip(a, 0.0, 1000.0, out=a)
    return a


def _convolve(a):
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_topK(a, c, K):
    n = len(a)
    L = len(c)
    K = max(1, min(K, L))
    d = np.zeros(n)
    if K < L:
        idx = np.argpartition(c, -K)[-K:]
    else:
        idx = np.arange(L)
    w = 1.0 / K
    for k in idx:
        k = int(k)
        s0 = max(0, k - n + 1)
        s1 = min(n, k + 1)
        j0 = k - s1 + 1
        j1 = k - s0 + 1
        d[s0:s1] += w * a[j0:j1][::-1]
    return 2.0 * d


def _smooth_perturb(a, rng, strength=0.3):
    a = a.copy()
    n = len(a)
    t = np.linspace(0, 1, n)
    z = np.zeros(n)
    for m in range(1, 5):
        amp = strength / (0.5 * m)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * m * t + ph)
    a *= np.exp(z)
    if n > 8 and rng.random() < 0.3:
        wdt = int(rng.integers(1, max(2, n // 10 + 1)))
        st = int(rng.integers(0, n - wdt))
        a[st:st+wdt] *= np.exp(rng.normal(0, strength, wdt))
    return _norm(a, float(n))


def _upsample(seq, n):
    m = len(seq)
    if m == n:
        return _norm(seq.astype(np.float64), float(n))
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    out = np.interp(x1, x0, seq)
    return _norm(out, float(n))


def _gen_starts(n, rng, parent=None):
    starts = []
    
    def add(a):
        a = _norm(a.astype(np.float64), float(n))
        sc = _score_fast(a)
        if np.isfinite(sc):
            starts.append((sc, a))
    
    # Uniform
    add(np.ones(n))
    
    # Two-block
    for frac in (0.1, 0.2, 0.3, 0.4):
        for ratio in (0.5, 1.5, 2.0, 3.0):
            m = max(1, min(n-1, int(frac * n)))
            denom = m * ratio + (n - m)
            B = n / denom
            A = ratio * B
            a = np.zeros(n)
            a[:m] = A
            a[m:] = B
            add(a)
    
    # Smooth shapes
    if n > 4:
        t = np.linspace(0, 1, n)
        for f in range(1, 8):
            add(1.0 + 0.5 * np.cos(2 * np.pi * f * t))
            add(1.0 + 0.5 * np.sin(np.pi * f * t))
    
    # Random smooth
    for _ in range(20):
        t = np.linspace(0, 1, n)
        z = np.zeros(n)
        for m in range(1, 10):
            amp = rng.uniform(-1, 1) / (m ** 0.6)
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * m * t + ph)
        add(np.exp(z - z.max()))
    
    # Random blocks
    for _ in range(10):
        nb = int(rng.integers(3, 13))
        edges = np.sort(rng.integers(1, n, nb - 1))
        edges = np.concatenate([[0], edges, [n]])
        vals = rng.uniform(0.2, 3.0, nb)
        a = np.repeat(vals, np.diff(edges))
        add(a)
    
    # Parent upsample
    if parent is not None and len(parent) > 0:
        add(parent)
    
    if not starts:
        starts.append((2.0, np.ones(n, dtype=np.float64)))
    
    starts.sort(key=lambda x: x[0])
    return starts
