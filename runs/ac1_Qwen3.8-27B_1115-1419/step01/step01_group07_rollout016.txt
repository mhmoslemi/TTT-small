def fft_conv(a):
    n=len(a); L=2*n-1; m=1
    while m < L: m <<=1
    f=np.fft.rfft(a,m)
    c=np.fft.irfft(f*f,m)
    # Remove tiny negative
    c[:L] = np.where(c[:L] < 0, 0, c[:L])

```
def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.array([0.0])
    L = 2*n - 1
    m = 1
    while m < L:
        m <<= 1
    f = np.fft.rfft(a, m)
    c = np.fft.irfft(f*f, m, n=L)
    if c.size < L:
       c = np.concatenate([c, np.zeros(L-c.size)])
    c = np.maximum(c, 0.0)
    return c[:L]
```

```
def project_simplex(a):
    a=np.asarray(a,float)
    n=a.size
    if n==0: return a
    b=1.0
    u=np.sort(a)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u * np.arange(1,n+1) > (cssv - b))[0]
    if rho.size ==0:
       return np.zeros(n)
    rho=rho[-1]
    theta=(cssv[rho] - b)/(rho+1.0)
    return np.maximum(a-theta,0.0)
```

```python
"""
Step-function optimizer for minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
for non-negative sequences a.

The construction uses:
  - FFT-based convolution and gradient evaluations,
  - active-set cutting-plane LPs of the form min T s.t. a*b <= T, sum b = 1,
  - LSE-smoothed self-convolution optimization with L-BFGS in log-softmax parameters,
  - mirror/dual-type response updates,
  - projection subgradient polishing,
  - multi-resolution restarts and final local polishing.

The returned sequence is scaled to sum 1, which is safe under the provided
evaluation function and keeps all entries in [0, 1].
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog, Bounds


_CURRENT_TAU = 1e-12


def softmax_vec(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return e / s


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1e300)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def resample_sequence(a, n):
    a = normalize_sequence(a)
    old_n = a.size
    if n <= 0:
        return a
    if old_n == int(n):
        return a
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / float(old_n)
    x_new = (np.arange(int(n), dtype=np.float64) + 0.5) / float(int(n))
    out = np.interp(x_new, x_old, a, left=a[0], right=a[-1])
    return normalize_sequence(out)


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    m = 1
    while m < L:
        m <<= 1
    f = np.fft.rfft(a, m)
    c = np.fft.irfft(f * f, m, n=L)
    if c.size < L:
        c = np.concatenate([c, np.zeros(L - c.size, dtype=np.float64)])
    c = c[:L]
    c = np.where(c > 0.0, c, 0.0)
    return c


def conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    L = n + m - 1
    size = 1
    while size < L:
        size <<= 1
    z = np.fft.rfft(a, size) * np.fft.rfft(b, size)
    c = np.fft.irfft(z, size, n=L)
    if c.size < L:
        c = np.concatenate([c, np.zeros(L - c.size, dtype=np.float64)])
    c = c[:L]
    c = np.where(c > 0.0, c, 0.0)
    return c


def correlate_fft(weights, a):
    weights = np.asarray(weights, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = weights.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    need = L + n - 1
    m = 1
    while m < need:
        m <<= 1
    ar = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(weights, m) * np.fft.rfft(ar, m), m)
    out = conv[n - 1:n - 1 + n]
    out = np.where(out > 0.0, out, 0.0)
    return out


def quick_score(a):
    a = normalize_sequence(a)
    n = a.size
    if n == 0:
        return float("inf")
    c = fft_conv(a)
    if c.size == 0:
        return float("inf")
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float("inf")
    return 2.0 * float(n) * mx

def peak_weights(c, tau):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 1e-300:
        return np.ones(c.size, dtype=np.float64) / float(c.size)
    tau = float(tau)
    if tau <= 1e-300:
        tau = 1e-300
    shifted = (c - cmax) / tau
    shifted = np.minimum(shifted, 0.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s <= 1e-300 or not np.isfinite(s):
        w = np.ones(c.size, dtype=np.float64)
        s = float(c.size)
    return w / s


def uniform_top_weights(c, k):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    k = int(k)
    if L == 0 or k <= 0:
        return c.astype(np.float64, copy=False)
    if k >= L:
        return np.ones(L, dtype=np.float64) / float(L)
    if k == 1:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(L, dtype=np.float64)
    w[idx] = 1.0 / float(k)
    return w


def project_simplex(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    b = 1.0
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1, dtype=np.float64) > (cssv - b))[0]
    if rho.size == 0:
        return np.zeros(n, dtype=np.float64)
    rho = int(rho[-1])
    theta = (cssv[rho] - b) / float(rho + 1)
    return np.maximum(a - theta, 0.0)


def select_top_lags(c, p, include_center=True):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.zeros(0, dtype=np.int64)
    center = (L - 1) // 2
    p = int(p)
    if p <= 0:
        if include_center:
            return np.array([center], dtype=np.int64)
        return np.zeros(0, dtype=np.int64)
    if p >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(c, -p)[-p:]
        idx = idx[np.argsort(c[idx])[::-1]]
        idx = idx.astype(np.int64)
    if include_center:
        if center not in idx:
            idx = np.unique(np.concatenate([idx, np.array([center], dtype=np.int64)]))
            idx = idx[np.argsort(c[idx])[::-1]].astype(np.int64)
    return idx


def build_rows(a, lags):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    q = len(lags)
    A = np.zeros((q, n), dtype=np.float64)
    for r, kk in enumerate(lags):
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if hi >= lo:
            A[r, lo:hi + 1] = a[kk - hi:kk - lo + 1][::-1]
    return A


def solve_active_lp(a, lags):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    lags = np.asarray(lags, dtype=np.int64)
    if n == 0 or lags.size == 0:
        return None, None
    try:
        A = build_rows(a, lags)
        A = np.where(np.isfinite(A), A, 0.0)
        q = A.shape[0]

        cobj = np.zeros(n + 1, dtype=np.float64)
        cobj[-1] = 1.0

        A_ub = np.hstack((A, -np.ones((q, 1), dtype=np.float64)))
        b_ub = np.zeros(q, dtype=np.float64)

        A_eq = np.ones(n + 1, dtype=np.float64)
        A_eq[-1] = 0.0
        b_eq = np.array([1.0])

        lb = np.zeros(n + 1, dtype=np.float64)
        ub = np.ones(n + 1, dtype=np.float64)
        ub[-1] = np.inf
        bounds = Bounds(lb, ub)

        try:
            options = {"presolve": True, "time_limit": 25.0}
            res = linprog(
                cobj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options=options,
            )
        except Exception:
            try:
                res = linprog(
                    cobj,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True},
                )
            except Exception:
                return None, None

        if res is not None and getattr(res, "success", False):
            x = getattr(res, "x", None)
            if x is not None and x.size == n + 1 and np.all(np.isfinite(x)):
                b = np.maximum(x[:-1], 0.0)
                s = float(np.sum(b))
                T = float(x[-1])
                if s > 1e-15:
                    b /= s
                    if not np.isfinite(T):
                        T = 0.0
                    return b, max(0.0, T)
    except Exception:
        return None, None
    return None, None


def active_lp_better(a, p, max_adds=3):
    a = normalize_sequence(a)
    n = a.size
    c = fft_conv(a)
    active = select_top_lags(c, p, include_center=True)
    b = None
    T = None
    for _ in range(max(1, int(max_adds))):
        if b is None:
            b, T = solve_active_lp(a, active)
            if b is None:
                return None, None
        mixed = conv2(a, b)
        if mixed.size == 0:
            return b, T
        maxm = float(np.max(mixed))
        if maxm <= (1.0005 * float(T) if T is not None else np.inf) + 1e-12:
            return b, T
        viol_n = max(5, min(24, int(p) // 4))
        viol = select_top_lags(mixed, viol_n, include_center=False)
        if viol.size == 0:
            break
        new_active = np.unique(np.concatenate([active, viol]))
        if new_active.size == active.size:

def lse_objective(x):
    a = softmax_vec(x)
    c = fft_conv(a)
    if c.size == 0:
        return 0.0
    cmax = float(np.max(c))
    tau = float(_CURRENT_TAU)
    if tau <= 1e-300:
        tau = 1e-300
    shifted = (c - cmax) / tau
    shifted = np.minimum(shifted, 0.0)
    return float(cmax + tau * np.log(np.sum(np.exp(shifted))))


def lse_gradient(x):
    a = softmax_vec(x)
    c = fft_conv(a)
    if c.size == 0:
        return np.zeros_like(x)
    cmax = float(np.max(c))
    tau = float(_CURRENT_TAU)
    if tau <= 1e-300:
        tau = 1e-300
    shifted = (c - cmax) / tau
    shifted = np.minimum(shifted, 0.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s <= 1e-300 or not np.isfinite(s):
        w = np.ones(c.size, dtype=np.float64)
        s = float(c.size)
    w /= s
    grad_a = 2.0 * correlate_fft(w, a)
    if not np.all(np.isfinite(grad_a)):
        return np.zeros_like(x)
    dot = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - dot)
    grad_x = np.where(np.isfinite(grad_x), grad_x, 0.0)
    return grad_x


def run_lse_lbfgs(a, max_time):
    a = normalize_sequence(a)
    if max_time <= 0:
        return a, quick_score(a)

    x = np.log(a + 1e-300)
    best_a = a.copy()
    best_score = quick_score(a)

    c = fft_conv(a)
    M = float(np.max(c)) if c.size else 1.0
    if not np.isfinite(M) or M <= 1e-300:
        M = 1.0 / max(1, 2 * a.size - 1)

    n = a.size
    if n > 6000:
        maxiter = 80
        maxfun = 350
    elif n > 2000:
        maxiter = 120
        maxfun = 500
    else:
        maxiter = 220
        maxfun = 900

    deadline = time.monotonic() + max_time
    ratios = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03]
    global _CURRENT_TAU

    for ratio in ratios:
        if time.monotonic() > deadline:
            break
        _CURRENT_TAU = max(1e-14, ratio * M)
        try:
            res = minimize(
                lse_objective,
                x,
                jac=lse_gradient,
                method="L-BFGS-B",
                options={
                    "maxiter": maxiter,
                    "maxfun": maxfun,
                    "ftol": 1e-15,
                    "gtol": 1e-11,
                    "maxls": 20,
                },
            )
            if res is not None and getattr(res, "x", None) is not None:
                x = np.asarray(res.x, dtype=np.float64)
                if not np.all(np.isfinite(x)):
                    break
        except Exception:
            break

        a_new = softmax_vec(x)
        s_new = quick_score(a_new)
        if np.isfinite(s_new) and s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            x = np.log(a_new + 1e-300)

    return best_a, best_score

    if not np.isfinite(s0):
        s0 = float("inf")
    if s0 < best_score:
        best_score = s0
        best_a = a.copy()

    deadline = time.monotonic() + time_budget
    it = 0
    no_improve = 0
    last_improve = time.monotonic()
    tau_factors = [4.0, 2.0, 1.0, 0.5, 0.25]
    steps = [0.45, 0.25, 0.12, 0.06, 0.03]

    while time.monotonic() < deadline:
        it += 1
        if time.monotonic() > deadline - 5:
            break
        c = fft_conv(a)
        cmax = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(cmax) or cmax <= 0:
            break

        if it % 4 == 0:
            top_k = [5, 15, 40, 120]
            w = uniform_top_weights(c, top_k[it % len(top_k)])
        else:
            w = peak_weights(c, cmax * tau_factors[it % len(tau_factors)])

        grad = 2.0 * correlate_fft(w, a)
        if not np.all(np.isfinite(grad)):
            no_improve += 1
            if no_improve > 80:
                break
            continue

        improved = False
        for step in steps:
            a_new = project_simplex(a - step * grad)
            s_new = quick_score(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                no_improve = 0
                last_improve = time.monotonic()
                print(f"  subgradient improvement: {best_score:.6f}", flush=True)
                break

        if not improved:
            no_improve += 1
            if it % 12 == 0:
                a = normalize_sequence(a * np.exp(rng.normal(0.0, 0.04, a.size)))
                s_a = quick_score(a)
                if np.isfinite(s_a) and s_a < best_score:
                    best_score = s_a
                    best_a = a.copy()
                    no_improve = 0
                    last_improve = time.monotonic()
            if (time.monotonic() - last_improve) > max(35.0, time_budget / 4.0) and no_improve > 30:
                break
            if no_improve > 90:
                break

    return best_a, best_score


def mirror_phase(a, best_score, time_budget):
    a = normalize_sequence(a)
    if time_budget <= 0:
        return a, best_score
    best_a = a.copy()
    s0 = quick_score(a)
    if not np.isfinite(s0):
        s0 = float("inf")
    if s0 < best_score:
        best_score = s0
        best_a = a.copy()

    deadline = time.monotonic() + time_budget
    it = 0
    no_improve = 0
    last_improve = time.monotonic()
    lambda_factors = [6.0, 3.0, 1.5, 0.8, 0.4]
    b_factors = [4.0, 2.2, 1.2, 0.7]

    while time.monotonic() < deadline:
        it += 1
        if time.monotonic() > deadline - 5:
            break
        c = fft_conv(a)
        cmax = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(cmax) or cmax <= 0:
            break

        w_l = peak_weights(c, cmax * lambda_factors[it % len(lambda_factors)])
        h = correlate_fft(w_l, a)
        h = np.where(h > 0.0, h, 0.0)
        hs = float(np.sum(h))
        if hs <= 1e-300 or not np.isfinite(hs):
            no_improve += 1
            if no_improve > 120:
                break
            continue
        h /= hs
        h_mean = float(np.mean(h))
        h_std = float(np.std(h))
        denom = max(h_std, abs(h_mean) * 1e-3, 1e-15)

        improved = False
        for beta in b_factors:
            z = (h - h_mean) / (denom * beta)
            z = -z
            b = softmax_vec(z)
            for t in [0.8, 0.55, 0.3, 0.12]:
                cand = normalize_sequence((1.0 - t) * a + t * b)
                s_new = quick_score(cand)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = cand.copy()
                    a = cand.copy()
                    improved = True
                    no_improve = 0
                    last_improve = time.monotonic()
                    print(f"  mirror improvement: {best_score:.6f}", flush=True)
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if (time.monotonic() - last_improve) > max(30.0, time_budget / 4.0) and no_improve > 80:
                break
            if no_improve > 180:
                break

    return best_a, best_score


def beta_shape(n, alpha, beta):
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    t = np.clip(t, 1e-8, 1.0 - 1e-8)
    val = np.power(t, alpha) * np.power(1.0 - t, beta)
    val = np.maximum(val, 1e-12)
    return normalize_sequence(val)


def generate_initial_candidates(parent, n, rng):
    n = int(n)
    cand = []

    if parent is not None:
        p = resample_sequence(parent, n)
        cand.append(p)

    cand.append(np.ones(n, dtype=np.float64) / float(n))

    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    t = np.clip(t, 1e-8, 1.0 - 1e-8)

    for al, be in [
        (0.25, 0.25),
        (0.5, 0.5),
        (0.5, 1.0),
        (1.0, 0.5),
        (1.0, 1.0),
        (0.5, 2.0),
        (2.0, 0.5),
        (1.0, 2.0),
        (2.0, 1.0),
        (0.35, 0.9),
        (0.9, 0.35),
    ]:
        cand.append(beta_shape(n, al, be))

    for eps in [1e-3, 1e-2]:
        val = 1.0 / (np.sqrt(np.clip(t, eps, 1.0)) * np.sqrt(np.clip(1.0 - t, eps, 1.0)))
        cand.append(normalize_sequence(val))

    for m in [1, 2, 3, 5]:
        val = 1.0 + 0.35 * np.cos(2.0 * np.pi * float(m) * t)
        val = np.maximum(val, 1e-8)
        cand.append(normalize_sequence(val))

    try:
        for alpha in [0.3, 0.7, 1.3, 3.0]:
            cand.append(normalize_sequence(rng.dirichlet(np.full(n, alpha))))
    except Exception:
        pass

    if parent is not None:
        p = resample_sequence(parent, n)
        if p.size == n:
            cand.append(normalize_sequence(p * np.exp(0.7 * (t - 0.5))))
            cand.append(normalize_sequence(p * np.exp(-0.7 * (t - 0.5))))
            cand.append(normalize_sequence(p * np.exp(1.5 * (t - 0.5))))
            cand.append(normalize_sequence(p * np.exp(-1.5 * (t - 0.5))))
            cand.append(normalize_sequence(p * np.exp(0.25 * np.cos(4.0 * np.pi * t))))

    return cand


def multi_resolution_phase(parent, best_a, best_score, time_budget, rng):
    best_a = normalize_sequence(best_a)
    if time_budget <= 0:
        return best_a, best_score
    final_n = best_a.size
    deadline = time.monotonic() + time_budget

    small_ns = [128, 256, 512]
    for ns in small_ns:
        if time.monotonic() > deadline - 25:
            break

        inits = []
        if parent is not None:
            inits.append(resample_sequence(parent, ns))
        inits.append(np.ones(ns, dtype=np.float64) / float(ns))
        inits.append(beta_shape(ns, 0.5, 1.0))
        inits.append(beta_shape(ns, 1.0, 0.5))
        inits.append(beta_shape(ns, 0.35, 0.9))
        try:
            inits.append(normalize_sequence(rng.dirichlet(np.full(ns, 0.6))))
        except Exception:
            pass

        best_small = None
        best_small_s = float("inf")

        for init in inits:
            if time.monotonic() > deadline - 25:
                break
            init = normalize_sequence(init)
            asmp = min(8.0, max(0.5, (deadline - time.monotonic()) * 0.25))
            a_s, s_s = active_lp_phase(init, best_small_s, rng, asmp)
            lt = min(12.0, max(0.5, (deadline - time.monotonic()) * 0.5))
            a_s, s_s = run_lse_lbfgs(a_s, lt)
            if np.isfinite(s_s) and s_s < best_small_s:
                best_small_s = s_s
                best_small = a_s

        if best_small is not None and best_small_s < 2.0:
            up = resample_sequence(best_small, final_n)
            rem = max(0.0, deadline - time.monotonic())
            up, su = active_lp_phase(up, best_score, rng, min(25.0, 0.4 * rem))
            rem = max(0.0, deadline - time.monotonic())
            up, su = run_lse_lbfgs(up, min(35.0, 0.5 * rem))
            if np.isfinite(su) and su < best_score:
                best_score = su
                best_a = up
                print(f"  multi-resolution upsample: {best_score:.6f}", flush=True)

    return best_a, best_score


def final_polish(parent, best_a, best_score, rng, time_budget):
    best_a = normalize_sequence(best_a)
    if time_budget <= 0:
        return best_a, best_score

    best_a = normalize_sequence(best_a)
    s0 = quick_score(best_a)
    if np.isfinite(s0) and s0 < best_score:
        best_score = s0

    deadline = time.monotonic() + max(0.0, time_budget - 20.0)
    n = best_a.size
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    it = 0
    no_improve = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        if time.monotonic() > deadline - 5:
            break

        if it % 3 == 1:
            p = [40, 80, 140][it % 3]
            b, _ = active_lp_better(best_a, p, max_adds=2)
            if b is not None:
                cand, s_c = line_search_mix(best_a, b, best_score)
                if np.isfinite(s_c) and s_c < best_score:
                    best_score = s_c
                    best_a = cand
                    no_improve = 0
                    last_improve = time.monotonic()
                    print(f"  final LP polish: {best_score:.6f}", flush=True)

        elif it % 3 == 2:
            c = fft_conv(best_a)
            cmax = float(np.max(c)) if c.size else 0.0
            if cmax > 0 and np.isfinite(cmax):
                w = peak_weights(c, cmax * 0.7)
                grad = 2.0 * correlate_fft(w, best_a)
                if np.all(np.isfinite(grad)):
                    for step in [0.35, 0.18, 0.09, 0.04, 0.02]:
                        cand = project_simplex(best_a - step * grad)
                        s_new = quick_score(cand)
                        if np.isfinite(s_new) and s_new < best_score:
                            best_score = s_new
                            best_a = cand
                            no_improve = 0
                            last_improve = time.monotonic()
                            print(f"  final gradient polish: {best_score:.6f}", flush=True)
                            break

        else:
            improved = False
            if it % 8 == 0:
                for beta in [0.45, 1.0, -0.45, -1.0, 1.8, -1.8]:
                    cand = normalize_sequence(best_a * np.exp(beta * (t - 0.5)))
                    s_new = quick_score(cand)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = cand
                        improved = True
                        no_improve = 0
                        last_improve = time.monotonic()
                        print(f"  final tilt: {best_score:.6f}", flush=True)
                        break
            if not improved:
                sig = 0.008 + 0.02 * (it % 5) / 5.0
                cand = normalize_sequence(best_a * np.exp(rng.normal(0.0, sig, n)))
                s_new = quick_score(cand)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = cand
                    no_improve = 0
                    last_improve = time.monotonic()
                    print(f"  final noise: {best_score:.6f}", flush=True)

        if not (time.monotonic() - last_improve < max(30.0, time_budget / 8.0)):
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

    if time.monotonic() < deadline + 20 - 35 and best_a.size * 2 <= 9612:
        try:
            high = resample_sequence(best_a, best_a.size * 2)
            rem = max(0.0, deadline + 20 - time.monotonic() - 35)
            if rem > 25:
                high, s_h = active_lp_phase(high, best_score, rng, min(50.0, 0.55 * rem))
                rem = max(0.0, deadline + 20 - time.monotonic() - 35)
                if rem > 25:
                    high, s_h = run_lse_lbfgs(high, min(55.0, 0.60 * rem))
                if np.isfinite(s_h) and s_h < best_score:
                    best_score = s_h
                    best_a = high
                    print(f"  final upsample: {best_score:.6f}", flush=True)
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    total_budget = float(budget_s) if budget_s is not None else 2400.0
    if not np.isfinite(total_budget) or total_budget <= 60.0:
        total_budget = 2400.0
    start = time.monotonic()
    hard_deadline = start + max(60.0, total_budget - 35.0)
    rng = np.random.default_rng(1234567)

    parent = None
    if "height_sequence_1" in globals():
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            raw = np.where(np.isfinite(raw), raw, 0.0)
            s = float(np.sum(raw))
            if raw.size > 0 and s > 1e-300:
                parent = raw / s
        except Exception:
            parent = None

    if parent is not None:
        pl = parent.size
        if pl < 4096:
            main_n = 4806
        elif pl > 9612:
            main_n = 9612
        else:
            main_n = max(pl, 4806)
        main_parent = resample_sequence(parent, main_n)
    else:
        main_n = 4806
        main_parent = None

    main_n = int(max(64, min(9612, main_n)))
    best_a = main_parent if main_parent is not None else np.ones(main_n, dtype=np.float64) / float(main_n)
    best_score = quick_score(best_a)
    if not np.isfinite(best_score):
        best_score = float("inf")

    print(f"Initial main_n={main_n}, quick_score={best_score:.6f}", flush=True)

    if time.monotonic() < hard_deadline - 45:
        inits = generate_initial_candidates(main_parent, main_n, rng)
        for init in inits:
            if time.monotonic() > hard_deadline - 45:
                break
            try:
                sc = quick_score(init)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = normalize_sequence(init)
                    print(f"  initial candidate: {best_score:.6f}", flush=True)
            except Exception:
                pass

    phase_budget = total_budget - 90.0
    phase_budget = max(60.0, phase_budget)

    remaining = min(phase_budget * 0.22, max(30.0, hard_deadline - time.monotonic() - 90.0))
    best_a, best_score = active_lp_phase(best_a, best_score, rng, remaining)

    remaining = min(phase_budget * 0.18, max(20.0, hard_deadline - time.monotonic() - 90.0))
    best_a, best_score = run_lse_lbfgs(best_a, remaining)

    remaining = min(phase_budget * 0.08, max(10.0, hard_deadline - time.monotonic() - 90.0))
    best_a, best_score = mirror_phase(best_a, best_score, remaining)

    remaining = min(phase_budget * 0.07, max(10.0, hard_deadline - time.monotonic() - 90.0))
    best_a, best_score = subgradient_phase(best_a, best_score, rng, remaining)

    remaining = min(phase_budget * 0.18, max(20.0, hard_deadline - time.monotonic() - 90.0))
    best_a, best_score = multi_resolution_phase(main_parent, best_a, best_score, remaining, rng)

    remaining = min(phase_budget * 0.35, max(30.0, hard_deadline - time.monotonic() - 40.0))
    best_a, best_score = final_polish(main_parent, best_a, best_score, rng, remaining)

    candidates = [best_a]
    if main_parent is not None and main_parent.size == best_a.size:
        candidates.append(main_parent)
    candidates.append(np.ones(best_a.size, dtype=np.float64) / float(best_a.size))

    final_best = best_a
    final_direct = direct_score(best_a)
    for cand in candidates:
        if time.monotonic() > hard_deadline:
            break
        try:
            sc = direct_score(cand)
            if np.isfinite(sc) and sc < final_direct:
                final_direct = sc
                final_best = normalize_sequence(cand)
        except Exception:
            pass

    final_best = normalize_sequence(final_best)
    final_best = np.clip(np.asarray(final_best, dtype=np.float64), 0.0, 1000.0)
    final_best = np.where(np.isfinite(final_best), final_best, 0.0)

    try:
        fd = direct_score(final_best)
        print(f"Final returned direct score: {fd:.6f}", flush=True)
    except Exception:
        print("Final direct score could not be computed.", flush=True)

    return [float(x) for x in final_best]
```
