```python
from scipy.optimize import linprog
from scipy import sparse

def solve_lp_direct(a, M, n):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n and 0 <= j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = M * np.ones(2*n - 1)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
    a_new = (1.0 - best_t) * a + best_t * g
    a_new = _normalize(a_new, n)
    return a_new, _max_conv(a_new), True


def _momentum_descent(a, n, deadline, tag="mom", it_limit=500000):
    """Momentum-based gradient descent on max convolution component."""
    best = a.copy()
    best_obj = _obj(_max_conv(a), n)
    momentum = np.zeros(n)
    beta = 0.85
    eta = 0.15
    stall = 0

    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        conv_a = np.convolve(a, a)
        k = int(np.argmax(conv_a))
        M = float(conv_a[k])
        cur_obj = _obj(M, n)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best = a.copy()
            stall = 0
        else:
            stall += 1

        # Gradient of conv(a,a)[k] w.r.t. a
        g = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je + 1] = 2.0 * a[k - np.arange(js, je + 1)]
        g -= g.mean()

        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break

        momentum = beta * momentum + g
        step = eta * momentum / (np.max(np.abs(momentum)) + 1e-12)
        a = np.maximum(a - step, 0.0)
        a = _normalize(a, n)

        if stall > 3000:
            eta = max(eta * 0.7, 1e-6)
            beta = max(beta * 0.95, 0.5)
            stall = 0
            momentum *= 0.5

    return best, best_obj


def _smooth_descent(a, n, deadline, tag="smooth", it_limit=500000):
    """Entropy-smoothed gradient descent."""
    best = a.copy()
    best_obj = _obj(_max_conv(a), n)
    tau = n / 3.0
    eta = 0.06
    stall = 0

    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))
        cur_obj = _obj(M, n)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best = a.copy()
            stall = 0
        else:
            stall += 1

        # Softmax weights
        mx = float(np.max(conv_a))
        p = np.exp((conv_a - mx) / tau)
        p_sum = float(p.sum())
        if p_sum < 1e-300:
            p = np.full_like(conv_a, 1.0 / len(conv_a))
        else:
            p = p / p_sum

        # Gradient: 2 * cross-correlation of p with a (valid mode)
        g = 2.0 * np.correlate(p, a, mode='valid')
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break

        a = np.maximum(a - eta * g / mg, 0.0)
        a = _normalize(a, n)
        tau = max(tau * 0.99998, n * 0.002)

        if stall > 5000:
            eta = max(eta * 0.5, 1e-7)
            stall = 0

    return best, best_obj


def _perturb(a, n, rng, strength=0.3):
    """Random multiplicative perturbation of a block."""
    b = a.copy()
    w = max(2, n // 30)
    s = int(rng.integers(0, max(1, n - w)))
    b[s:s + w] *= np.exp(rng.normal(0.0, strength, size=w))
    b = np.maximum(b, 0.0)
    return _normalize(b, n)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main optimization: LP-guided iterations + gradient refinement + perturbations.
    Returns best non-negative sequence found within budget_s seconds.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0

    # Try to load parent sequence
    n = 3809
    a = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.all(np.isfinite(parent)):
            a = np.clip(parent, 0.0, 1000.0)
    except Exception:
        pass
    if a is None:
        a = np.ones(n, dtype=np.float64)

    a = _normalize(a, n)
    best_a = a.copy()
    best_obj = _obj(_max_conv(a), n)
    print(f"Init: n={n} obj={best_obj:.6f}")

    # --- Phase 1: LP-based optimization (55% of budget) ---
    p1_deadline = t0 + 0.55 * (budget_s - 20.0)
    lp_it = 0
    lp_stall = 0

    while time.monotonic() < p1_deadline:
        lp_it += 1
        result = _lp_step(a, n)
        if result is not None:
            a_new, M_new, _ = result
            obj_new = _obj(M_new, n)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                lp_stall = 0
            else:
                a = a_new
                lp_stall += 1
            if lp_it % 50 == 0:
                print(f"  LP it={lp_it} obj={_obj(_max_conv(a), n):.6f} "
                      f"best={best_obj:.6f} t={time.monotonic()-t0:.0f}s")
        else:
            lp_stall += 1

        # If stuck, perturb and try momentum descent briefly
        if lp_stall > 15:
            a_p = _perturb(a, n, rng, strength=0.2 + 0.1 * min(lp_stall / 10, 1.0))
            sub_dl = min(time.monotonic() + 8.0, p1_deadline)
            a_p, obj_p = _momentum_descent(a_p, n, sub_dl, "lp_pert")
            if obj_p < best_obj:
                best_obj = obj_p
                best_a = a_p.copy()
            a = a_p if obj_p < _obj(_max_conv(a), n) else a
            lp_stall = 0
            print(f"  LP restart after stall at it={lp_it}, obj={best_obj:.6f}")

    print(f"Phase 1 done: obj={best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Phase 2: Smooth + momentum gradient descent (25% of budget) ---
    p2_deadline = t0 + 0.80 * (budget_s - 20.0)
    a = best_a.copy()
    a, obj_s = _smooth_descent(a, n, min(time.monotonic() + 0.12 * (budget_s - 20), p2_deadline), "sm2")
    if obj_s < best_obj:
        best_obj = obj_s
        best_a = a.copy()
    a, obj_m = _momentum_descent(a, n, p2_deadline, "mom2")
    if obj_m < best_obj:
        best_obj = obj_m
        best_a = a.copy()
    print(f"Phase 2 done: obj={best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    # --- Phase 3: Perturbation + refinement loop (20% of budget) ---
    p3_deadline = t0 + 0.95 * (budget_s - 20.0)
    attempt = 0
    while time.monotonic() < p3_deadline:
        attempt += 1
        sig = float(rng.uniform(0.15, 0.50))
        a_p = _perturb(best_a, n, rng, strength=sig)

        # Quick smooth
        sub1 = min(time.monotonic() + 10.0, p3_deadline - 10.0)
        a_q, o_q = _smooth_descent(a_p, n, sub1, f"s{attempt}", it_limit=20000)
        if o_q < best_obj:
            best_obj = o_q
            best_a = a_q.copy()
            print(f"  Perturb+smooth {attempt}: obj={best_obj:.6f}")

        # Quick momentum
        sub2 = min(time.monotonic() + 8.0, p3_deadline - 5.0)
        a_q2, o_q2 = _momentum_descent(a_q, n, sub2, f"m{attempt}", it_limit=20000)
        if o_q2 < best_obj:
            best_obj = o_q2
            best_a = a_q2.copy()
            print(f"  Perturb+momentum {attempt}: obj={best_obj:.6f}")

        # Very short LP from current best
        if time.monotonic() < p3_deadline - 15.0:
            sub3 = min(time.monotonic() + 12.0, p3_deadline)
            a_lp = best_a.copy()
            for _ in range(3):
                if time.monotonic() >= sub3:
                    break
                res = _lp_step(a_lp, n)
                if res is not None:
                    a_lp, _, _ = res
            obj_lp = _obj(_max_conv(a_lp), n)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                print(f"  Perturb+LP {attempt}: obj={best_obj:.6f}")

    # --- Final: one more momentum pass ---
    if time.monotonic() < deadline - 3.0:
        a_f, obj_f = _momentum_descent(best_a, n, deadline - 1.0, "final", it_limit=200000)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f.copy()
    print(f"FINAL: obj={best_obj:.6f} n={n} time={time.monotonic()-t0:.0f}s")

    # Validate and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    if not np.isfinite(sc):
        result = [1.0] * n
    print(f"Evaluated score: {sc:.6f}")
    return result
```
