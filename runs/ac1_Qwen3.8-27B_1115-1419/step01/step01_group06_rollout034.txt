"""
Search for non-negative coefficient sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated gradient descent in log-space with progressive sharpening,
coordinate descent refinement, and multiple initialization strategies.
Key insight: non-symmetric sequences can achieve score < 2 by reducing the
central convolution peak (which equals the cross-correlation with the reverse).

import numpy as np
import time


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    arr = np.asarray(v, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(max(1, arr.size), dtype=np.float64) / max(1, arr.size)
    return arr / s


def softmax_stable(w):
    w = w - np.max(w)
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size, dtype=np.float64) / w.size
    return e / s


def fft_conv(a, n_fft):
    n = len(a)
    A = np.fft.rfft(a, n_fft)
    c = np.fft.irfft(A * A, n_fft)[:2 * n - 1]
    return A, np.maximum(c, 0.0)


def soft_max_value(c, beta):
    cm = float(np.max(c))
    if cm <= 0:
        return 0.0
    x = beta * (c - cm)
    x = np.clip(x, -50.0, 0.0)
    return cm + float(np.log(np.sum(np.exp(x)))) / beta


def soft_max_weights(c, beta):
    cm = float(np.max(c))
    if cm <= 0:
        return np.ones(len(c), dtype=np.float64) / len(c)
    x = beta * (c - cm)
    x = np.clip(x, -50.0, 0.0)
    e = np.exp(x)
    s = e.sum()
    if s <= 0:
        return np.ones(len(c), dtype=np.float64) / len(c)
    return e / s


def exact_convolution(a):
    n = len(a)
    c = np.convolve(a, a)
    return np.maximum(c, 0.0)


def score_normalized(f):
    n = len(f)
    _, c = fft_conv(f, 1)  # dummy, just for structure
    c = exact_convolution(f)
    mx = float(np.max(c))
    s = float(np.sum(f))
    if s <= 0:
        return float('inf')
    return 2.0 * n * mx / (s * s)


def score_fft(f, n_fft):
    n = len(f)
    _, c = fft_conv(f, n_fft)
    mx = float(np.max(c))
    s = float(np.sum(f))
    if s <= 0 or not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * mx / (s * s)


def gradient_smooth_max_fft(f, n_fft, beta):
    """Compute gradient of soft_max_β(conv(f,f)) w.r.t. f using FFT cross-correlation."""
    n = len(f)
    A, c = fft_conv(f, n_fft)
    p = soft_max_weights(c, beta)
    # gradient = 2 * (p cross-correlation f)
    # (p ⋆ f)[j] = ifft(FFT(p) * conj(FFT(f)))
    p_pad = np.zeros(n_fft, dtype=np.float64)
    p_pad[:len(p)] = p
    P = np.fft.rfft(p_pad, n_fft)
    grad_f = 2.0 * np.fft.irfft(P * np.conjugate(A), n_fft)
    grad_f = grad_f[:n]
    return grad_f


def log_space_gradient(grad_f, f):
    """Convert gradient w.r.t. f to gradient w.r.t. log-space w."""
    return f * (grad_f - float(np.dot(f, grad_f)))


def adam_update(w, grad, m, v, t, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
    return w, m, v


def run_gradient_descent(f_init, n_fft, beta_start, beta_end, n_steps, lr, seed=42):
    """Main gradient descent loop in log-space with progressive sharpening."""
    rng = np.random.default_rng(seed)
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    w = np.log(f)
    w -= np.max(w)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    best_f = f.copy()
    best_loss = float('inf')
    
    for step in range(n_steps):
        # Progressive sharpening
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        # Adaptive learning rate (decay)
        lr_eff = lr * (1.0 - 0.5 * frac)
        
        # Compute gradient
        grad_f = gradient_smooth_max_fft(f, n_fft, beta)
        grad_f = np.where(np.isfinite(grad_f), grad_f, 0.0)
        grad_w = log_space_gradient(grad_f, f)
        grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
        
        # Adam step
        w, m, v = adam_update(w, grad_w, m, v, t + 1, lr_eff)
        t += 1
        w = np.where(np.isfinite(w), w, 0.0)
        
        # Convert back
        f_new = softmax_stable(w)
        
        # Evaluate
        _, c = fft_conv(f_new, n_fft)
        loss = soft_max_value(c, beta)
        
        if loss < best_loss:
            best_loss = loss
            best_f = f_new.copy()
        
        # Occasionally check actual max (for monitoring)
        if step % 500 == 0:
            actual_score = 2.0 * n * float(np.max(c))
            # print(f"  step {step}: soft_loss={loss:.6f}, actual_max_conv={np.max(c):.8f}, score~{actual_score:.6f}")
        
        f = f_new
    
    return best_f


def coordinate_descent_refine(f, n_fft, n_iter=100, trial_step=0.01):
    """Try small coordinate perturbations to reduce the max convolution."""
    n = len(f)
    f = f.copy()
    _, c_current = fft_conv(f, n_fft)
    current_max = float(np.max(c_current))
    
    for it in range(n_iter):
        # Find where the max is
        k_max = int(np.argmax(c_current))
        
        # The gradient at the max position: c[k_max] = sum_i f[i]*f[k_max-i]
        # To reduce it, decrease f[i] for i where f[k_max-i] is large, or vice versa
        # Try random coordinate changes
        idx = int(np.random.randint(0, n))
        partner = k_max - idx
        if partner < 0 or partner >= n:
            partner = None
        
        # Try reducing f[idx]
        old_val = f[idx]
        for delta in (-trial_step * old_val, -0.5 * trial_step * old_val, 0.5 * trial_step * old_val):
            f[idx] = max(0.0, old_val + delta)
            s = f.sum()
            if s <= 0:
                f[idx] = old_val
                continue
            f_norm = f / s
            _, c_test = fft_conv(f_norm, n_fft)
            if np.max(c_test) < current_max:
                current_max = float(np.max(c_test))
                c_current = c_test
                break
        f[idx] = old_val
    
    return f

        total_budget = min(float(budget_s), 2390.0)
    except (TypeError, ValueError):
        total_budget = 2390.0
    
    start_time = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Load initial sequence
    init_seq = load_initial_sequence()
    
    # Determine sequence length
    if init_seq is not None:
        n = init_seq.size
        if n > 12000:
            # Downsample
            target = 8192
            arr = init_seq[:target]
            init_seq = arr / arr.sum()
            n = target
    else:
        n = 4096
        init_seq = np.ones(n) / n
    
    n_fft = next_pow2(2 * n - 1)
    
    # Verify initial sequence score
    init_score = evaluate_search_candidate(init_seq, n_fft)
    print(f"Initial sequence: n={n}, n_fft={n_fft}, score={init_score:.6f}")
    
    best_f = init_seq.copy()
    best_score = init_score
    
    # Time budget allocation
    time_so_far = time.monotonic() - start_time
    remaining = total_budget - time_so_far - 5.0  # 5s buffer for finalization
    
    # Strategy 1: Gradient descent from SOTA (or initial)
    print(f"\nStrategy 1: Gradient descent from initial (time remaining: {remaining:.1f}s)")
    t1_start = time.monotonic()
    t1_budget = min(remaining * 0.4, 900.0)
    
    if t1_budget > 10:
        result1, score1 = refine_with_local_search(
            init_seq, n_fft, budget_s=t1_budget, seed=seed
        )
        print(f"  Strategy 1 result: score={score1:.6f}")
        if score1 < best_score:
            best_score = score1
            best_f = result1.copy()
    
    time_so_far = time.monotonic() - start_time
    remaining = total_budget - time_so_far - 5.0
    
    # Strategy 2: Exponential initialization
    print(f"\nStrategy 2: Exponential init (time remaining: {remaining:.1f}s)")
    t2_budget = min(remaining * 0.25, 600.0)
    
    if t2_budget > 10:
        for alpha in [1.0, 2.0, -1.0, -2.0, 3.0, -3.0]:
            if time.monotonic() - start_time > total_budget - 15:
                break
            f_exp = try_exponential_init(n, alpha, rng)
            exp_score = evaluate_search_candidate(f_exp, n_fft)
            print(f"  alpha={alpha}: initial score={exp_score:.6f}")
            if exp_score < best_score + 0.5:  # only optimize if promising
                sub_budget = min(120.0, (total_budget - (time.monotonic() - start_time) - 10))
                if sub_budget > 15:
                    result2, score2 = refine_with_local_search(
                        f_exp, n_fft, budget_s=sub_budget, seed=seed + alpha * 100
                    )
                    print(f"    optimized: score={score2:.6f}")
                    if score2 < best_score:
                        best_score = score2
                        best_f = result2.copy()
    
    time_so_far = time.monotonic() - start_time
    remaining = total_budget - time_so_far - 5.0
    
    # Strategy 3: Two-block initialization
    print(f"\nStrategy 3: Two-block init (time remaining: {remaining:.1f}s)")
    t3_budget = min(remaining * 0.2, 400.0)
    
    if t3_budget > 10:
        for split_frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            for left_alpha in [0.5, 1.5, 2.0, 0.3]:
                if time.monotonic() - start_time > total_budget - 15:
                    break
                f_block = try_two_block_init(n, split_frac, left_alpha, rng)
                block_score = evaluate_search_candidate(f_block, n_fft)
                if block_score < best_score + 0.3:
                    sub_budget = min(80.0, (total_budget - (time.monotonic() - start_time) - 10))
                    if sub_budget > 15:
                        result3, score3 = refine_with_local_search(
                            f_block, n_fft, budget_s=sub_budget, seed=seed + 500
                        )
                        if score3 < best_score:
                            best_score = score3
                            best_f = result3.copy()
                            print(f"  split={split_frac}, alpha={left_alpha}: score={score3:.6f}")
    
    time_so_far = time.monotonic() - start_time
    remaining = total_budget - time_so_far - 5.0
    
    # Strategy 4: Further refinement of best found
    print(f"\nStrategy 4: Further refinement of best (time remaining: {remaining:.1f}s)")
    
    if remaining > 30:
        result4, score4 = refine_with_local_search(
            best_f, n_fft, budget_s=min(remaining - 5, 600.0), seed=seed + 1000
        )
        print(f"  Strategy 4 result: score={score4:.6f}")
        if score4 < best_score:
            best_score = score4
            best_f = result4.copy()
    
    time_so_far = time.monotonic() - start_time
    remaining = total_budget - time_so_far - 3.0
    
    # Strategy 5: Perturbation-based exploration
    print(f"\nStrategy 5: Perturbation exploration (time remaining: {remaining:.1f}s)")
    
    if remaining > 20:
        n_pert_questions = int(min(remaining - 5, 300) / 30)
        for i in range(n_pert_questions):
            if time.monotonic() - start_time > total_budget - 10:
                break
            sig = float(rng.choice([0.05, 0.1, 0.2, 0.3]))
            w_pert = np.log(np.maximum(best_f, 1e-15)) + rng.normal(0, sig, n)
            f_pert = softmax_stable(w_pert)
            pert_score = evaluate_search_candidate(f_pert, n_fft)
            if pert_score < best_score:
                best_score = pert_score
                best_f = f_pert.copy()
            elif pert_score < best_score + 0.05:
                # Close enough to try refinement
                sub_budget = min(30.0, remaining - 15 - i * 3)
                if sub_budget > 10:
                    result5, score5 = refine_with_local_search(
                        f_pert, n_fft, budget_s=sub_budget, seed=seed + 2000 + i
                    )
                    if score5 < best_score:
                        best_score = score5
                        best_f = result5.copy()
                        print(f"  Perturb {i} (sig={sig}): score={score5:.6f}")
    
    # Final: try a few more refinement passes on the best
    time_so_far = time.monotonic() - start_time
    remaining = total_budget - time_so_far - 2.0
    if remaining > 20:
        result_final, score_final = refine_with_local_search(
            best_f, n_fft, budget_s=min(remaining, 200.0), seed=seed + 9999
        )
        if score_final < best_score:
            best_score = score_final
            best_f = result_final.copy()
    
    print(f"\nFinal best score (FFT-based): {best_score:.6f}")
    print(f"Total time used: {time.monotonic() - start_time:.1f}s")
    
    # Finalize
    result = finalize_sequence(best_f)
    
    # Verify with evaluate_sequence if available
    try:
        if 'evaluate_sequence' in dir() or 'evaluate_sequence' in globals():
            eval_score = evaluate_sequence(result)
            print(f"Verified score from evaluate_sequence: {eval_score:.6f}")
    except Exception as e:
        print(f"Could not verify: {e}")
    
    return result
