def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size

        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)

"""
Multi-start LSE-smoothed gradient descent with Nesterov momentum and adaptive
sharpening for low-peak nonnegative self-convolutions.
Minimizes 2*n*max(a*a)/(sum a)^2 via projected subgradient methods on the simplex.

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _make_uniform(n):
    n = max(int(n), 2)
    return np.ones(n, dtype=np.float64) / n


def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_gaussian_bump(n, center, width):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-((x - center) ** 2) / (2.0 * width * width))
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _make_two_peak(n, c1, w1, mix):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    w2 = w1 * 1.3
    c2 = 1.0 - c1
    a = mix * np.exp(-((x - c1) ** 2) / (2.0 * w1 * w1)) + \
        (1.0 - mix) * np.exp(-((x - c2) ** 2) / (2.0 * w2 * w2))
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _make_random_smooth(n, rng, amp=0.4):
    n = max(int(n), 2)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 12 + 2))
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 1e-300:
        f = f / sd
    base = np.ones(n) / n
    a = base * (1.0 + float(amp) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb(a, rng, strength=0.15):
    n = a.size
    noise = np.exp(rng.normal(0.0, strength, size=n))
    a2 = a * noise
    r = rng.random()
    if r < 0.2:
        a2 = 0.6 * a2 + 0.4 * a2[::-1]
    elif r < 0.35:
        block = max(2, n // 16)
        start = int(rng.integers(0, max(1, n - block)))
        a2[start:start+block] *= float(np.exp(rng.normal(0, strength * 2)))
    return _normalize(a2)


def _lse_and_p(b, beta):
    if not b.size:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _gradient_fft(a, p, fft_size, n):
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad


def _project_simplex(v):
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    n = w.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return w / s

        inits.append(_make_random_smooth(n, rng, 0.5))
        inits.append(_make_gaussian_bump(n, 0.5, 0.35))
        inits.append(_make_random_smooth(n, rng, 0.35))
        inits.append(_make_two_peak(n, 0.4, 0.15, 0.5))

        if best_a is not None and best_a.size > 4:
            inits.append(_resample(best_a, n))
        if given is not None and given.size > 4:
            inits.append(_resample(given, n))

        # Deduplicate by fingerprint
        seen_sigs = set()
        unique_inits = []
        for ic in inits:
            sig = tuple(np.round(ic[::max(1, ic.size // 16)], 4))
            if sig not in seen_sigs:
                seen_sigs.add(sig)
                unique_inits.append(ic)

        for idx, init in enumerate(unique_inits[:num_starts]):
            if time.monotonic() > deadline - 10:
                break
            t_end = min(deadline - 2.0, time.monotonic() + sec_per)
            if time.monotonic() >= t_end:
                continue
            res_a, res_score = _refine(init.copy(), t_end, rng)
            elapsed = time.monotonic() - (deadline - budget + 30)
            print(f"n={res_a.size} start={idx}/{num_starts} "
                  f"score={res_score:.8f} best={best_score:.8f} "
                  f"t={elapsed:.1f}s", flush=True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)

    # Phase 2: If given sequence is promising, refine it directly
    if given is not None and 256 <= given.size <= 8192:
        if best_score > 1.51 and time.monotonic() < deadline - 90:
            sec = max(40, int(90 * scale))
            t_end = min(deadline - 40, time.monotonic() + sec)
            res_a, res_score = _refine(given.copy(), t_end, rng)
            print(f"Given direct: n={res_a.size} score={res_score:.8f} "
                  f"best={best_score:.8f}", flush=True)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)

    # Phase 3: Upscale the best sequence to longer lengths
    if time.monotonic() < deadline - 60:
        n_cur = int(best_a.size)
        for target_n in [1024, 2048, 4096, 8192]:
            if target_n <= n_cur:
                continue
            if time.monotonic() > deadline - 40:
                break
            init = _resample(best_a, target_n)
            # Mix with a smooth exponential for diversity
            exp_init = _make_exponential(target_n, 1.0)
            for mix_w in [0.8, 0.6, 0.4]:
                if time.monotonic() > deadline - 20:
                    break
                mixed = _normalize(float(mix_w) * init + float(1 - mix_w) * exp_init)
                t_end = min(deadline - 5.0, time.monotonic() + max(30, int(50 * scale)))
                res_a, res_score = _refine(mixed, t_end, rng)
                print(f"Upscale {n_cur}->{res_a.size} mix={mix_w} "
                      f"score={res_score:.8f} best={best_score:.8f}", flush=True)
                if np.isfinite(res_score) and res_score < best_score:
                    best_a = np.asarray(res_a, dtype=np.float64).copy()
                    best_score = float(res_score)
                    n_cur = int(res_a.size)

    # Phase 4: Final deep refinement
    if time.monotonic() < deadline - 5:
        res_a, res_score = _refine(best_a.copy(), deadline - 2.0, rng, verbose=True)
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)
            print(f"Final refine: score={best_score:.8f}", flush=True)
        else:
            print(f"Final refine: no improvement, best={best_score:.8f}", flush=True)

    # Validate with evaluate_sequence if available
    try:
        out_list = [float(x) for x in best_a]
        eval_score = evaluate_sequence(out_list)
        print(f"Evaluation check: score={eval_score:.8f}", flush=True)
        if not np.isfinite(eval_score):
            # Fallback to safe exponential
            best_a = _make_exponential(512, 1.0)
            out_list = [float(x) for x in best_a]
    except NameError:
        pass
    except Exception as e:
        print(f"Eval check error: {e}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)

    return [float(x) for x in out]
