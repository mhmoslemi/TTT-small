```python
"""
Search for non-negative step-function heights minimizing
    2 * n * max(convolve(a, a)) / sum(a)^2.

The program:
  - starts from the provided previous construction if available,
  - performs length continuation and warm-up at several lengths,
  - generates diverse candidates by interpolation, symmetrization, random Fourier
    shapes, perturbments, and frequency-domain Newton/root targeting,
  - polishes candidates with a smooth surrogate based on the normalized
    convolution coefficients,
  - uses both L-BFGS-B (via scipy, when available) and a custom Adam-like
    optimizer with backtracking in log-softmax space,
  - keeps the best sequence found and returns it as a Python list of floats.
"""

import numpy as np
import time

try:
    from scipy import optimize as _scipy_opt
    _HAS_SCIPY = True
except Exception:
    _scipy_opt = None
    _HAS_SCIPY = False

_L_STATE = None
_L_SCALE = 1.0


def _pair_first(pair):
    return pair[0]


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def _soft(v, n):
    n = int(n)
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    if v.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        k = min(v.size, n)
        if k > 0:
            tmp[:k] = v[:k]
        v = tmp
    if not np.all(np.isfinite(v)):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    v = v - float(np.mean(v))
    v = np.clip(v, -50.0, 50.0)
    e = np.exp(v)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def _v_from_a(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-300)
    v = np.log(a)
    v = v - float(np.mean(v))
    v = np.clip(v, -30.0, 30.0)
    return v


def _sym(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    out = 0.5 * (a + a[::-1])
    return _normalize(out)


def _interp(a, n):
    n = int(max(1, n))
    if a is None:
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if a.size == 0 or s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    a = a / s
    if a.size == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return _normalize(out)


def _perturb(a, sigma, rng):
    n = a.size
    v = _v_from_a(a)
    v = v + rng.normal(0.0, float(sigma), n)
    v = v - float(np.mean(v))
    v = np.clip(v, -30.0, 30.0)
    return _soft(v, n)


def _random_fourier(n, rng, kmax=20, amp_scale=0.35):
    n = int(max(3, n))
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n, dtype=np.float64)
    for k in range(1, int(max(2, kmax)) + 1):
        amp = amp_scale * float(rng.uniform(0.15, 1.0)) / (float(k) ** 0.7)
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        if float(rng.random()) < 0.5:
            a += amp * np.cos(2.0 * np.pi * float(k) * t + ph)
        else:
            a += amp * np.sin(2.0 * np.pi * float(k) * t + ph)
    a = np.maximum(a, 0.0)
    med = float(np.mean(a))
    if med > 0.0:
        a = np.minimum(a, 8.0 * med)
    return _normalize(a)


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    L = 2 * n - 1
    m = _next_pow2(max(16, L))
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:L]
    np.maximum(b, 0.0, out=b)
    bm = float(np.max(b))
    if not np.isfinite(bm) or bm <= 0.0:
        return np.inf
    return float(2.0 * n * bm / (s * s))


def _raw_f_score(a, m, tau):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    L = 2 * n - 1
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return 1e100, np.inf
    if abs(s - 1.0) > 1e-12:
        a = a / s
    F = np.fft.fft(a, m)
    brow = np.fft.ifft(F * F).real[:L]
    if not np.all(np.isfinite(brow)):
        brow = np.zeros(L, dtype=np.float64)
    np.maximum(brow, 0.0, out=brow)
    bm = float(np.max(brow))
    score = 2.0 * n * bm  # sum(a) is 1 above
    bs = float(np.sum(brow))
    if bs <= 0.0 or not np.isfinite(bs):
        b = brow.copy()
    else:
        b = brow / bs
    if tau < 1e-8:
        tau = 1e-8
    z = (float(L) * b - 1.0) / float(tau)
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return 1e100, score
    z = z - zmax
    np.clip(z, -700.0, 700.0, out=z)
    ex = np.exp(z)
    Z = float(np.sum(ex))
    if Z <= 0.0 or not np.isfinite(Z):
        return zmax / (2.0 * float(L)), score
    u = (zmax + float(np.log(Z / float(L)))) / (2.0 * float(L))
    if not np.isfinite(u):
        u = 1e100
    return float(u), float(score)


def _raw_obj_grad(a, m, tau):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    L = 2 * n - 1
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return 1e100, np.zeros(n, dtype=np.float64), np.inf
    if abs(s - 1.0) > 1e-12:
        a = a / s
    F = np.fft.fft(a, m)
    brow = np.fft.ifft(F * F).real[:L]
    if not np.all(np.isfinite(brow)):
        brow = np.zeros(L, dtype=np.float64)
    np.maximum(brow, 0.0, out=brow)
    bm = float(np.max(brow))
    score = 2.0 * n * bm
    bs = float(np.sum(brow))
    if bs <= 0.0 or not np.isfinite(bs):
        b = brow.copy()
        factor = 1.0
    else:
        b = brow / bs
        factor = 1.0 / bs
    if tau < 1e-8:
        tau = 1e-8
    z = (float(L) * b - 1.0) / float(tau)
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return 1e100, np.zeros(n, dtype=np.float64), score
    z = z - zmax
    np.clip(z, -700.0, 700.0, out=z)
    ex = np.exp(z)
    Z = float(np.sum(ex))
    if Z <= 0.0 or not np.isfinite(Z):
        w = np.full(L, 1.0 / float(L), dtype=np.float64)
    else:
        w = ex / Z
    W = np.fft.fft(w, m)
    c = np.fft.ifft(W * np.conj(F)).real[:n]
    if not np.all(np.isfinite(c)):
        c = np.zeros(n, dtype=np.float64)
    c = c * factor
    cm = float(np.dot(c, a))
    g = a * (c - cm)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    u = (zmax + float(np.log(Z / float(L)) if Z > 0 else 0.0)) / (2.0 * float(L))
    if not np.isfinite(u):
        u = 1e100
    return float(u), g, float(score)


def _nbfgs_fun(v):
    if _L_STATE is None:
        return 1e100
    n, m, tau, L = _L_STATE
    if n <= 0:
        return 1e100
    a = _soft(v, n)
    u, sc = _raw_f_score(a, m, tau)
    if not np.isfinite(u):
        return 1e100
    return float(u * _L_SCALE)


def _lbfgs_jac(v):
    if _L_STATE is None:
        return np.zeros(1, dtype=np.float64)
    n, m, tau, L = _L_STATE
    if n <= 0:
        return np.zeros(1, dtype=np.float64)
    a = _soft(v, n)
    u, g, sc = _raw_obj_grad(a, m, tau)
    if g is None or not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    return (g * _L_SCALE).astype(np.float64)


def _lbfgs_polish(a0, deadline, tau):
    global _L_STATE, _L_SCALE
    if not _HAS_SCIPY:
        return _normalize(a0), _score_fft(a0)

    best_a = _normalize(a0)
    best_sc = _score_fft(best_a)
    n = int(best_a.size)
    if n <= 1:
        return best_a, best_sc
    L = 2 * n - 1
    m = _next_pow2(max(16, L))
    v = _v_from_a(best_a)
    no_improve = 0

    while time.monotonic() < deadline - 4.0:
        if not np.all(np.isfinite(v)):
            v = _v_from_a(best_a)
        a = _soft(v, n)
        u0, g0, sc0 = _raw_obj_grad(a, m, tau)
        rms = float(np.sqrt(np.mean(g0 * g0)))
        scale = 1.0
        if rms > 1e-14 and np.isfinite(rms):
            scale = min(1e9, max(1.0, 1.0 / (rms + 1e-14)))
        _L_STATE = (n, m, tau, L)
        _L_SCALE = float(scale)

        if n <= 1024:
            maxfun = 800
            maxiter = 260
        elif n <= 2048:
            maxfun = 600
            maxiter = 200
        elif n <= 4096:
            maxfun = 340
            maxiter = 120
        else:
            maxfun = 200
            maxiter = 80

        t0 = time.monotonic()
        try:
            res = _scipy_opt.minimize(
                _nbfgs_fun,
                v,
                jac=_lbfgs_jac,
                method="L-BFGS-B",
                options={
                    "maxiter": int(maxiter),
                    "maxfun": int(maxfun),
                    "maxls": 20,
                    "ftol": 1e-10,
                    "gtol": 1e-5,
                },
            )
        except Exception:
            break

        v_new = getattr(res, "x", None)
        if v_new is None or not np.all(np.isfinite(v_new)):
            break
        a_new = _soft(v_new, n)
        sc_new = _score_fft(a_new)

        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = float(sc_new)
            best_a = a_new.copy()
            v = np.asarray(v_new, dtype=np.float64).copy()
            no_improve = 0
        else:
            no_improve += 1
            v = np.asarray(v_new, dtype=np.float64).copy()
            if no_improve >= 3:
                v = _v_from_a(best_a)
                # small deterministic jitter
                z = np.linspace(-1.0, 1.0, n, dtype=np.float64)
                v = v + 0.04 * z
                v = v - float(np.mean(v))
                v = np.clip(v, -30.0, 30.0)
                no_improve = 0

        if time.monotonic() - t0 > max(5.0, 0.22 * max(0.0, deadline - time.monotonic())):
            # do not let one huge call consume the whole deadline
            pass
        if no_improve >= 8:
            break

    a_s = _sym(best_a)
    s_s = _score_fft(a_s)
    if np.isfinite(s_s) and s_s < best_sc:
        best_a = a_s
        best_sc = float(s_s)
    return best_a, float(best_sc)


def _adam_refine(a0, deadline, seed):
    if rng_holder is None:
        rng = np.random.default_rng(int(seed))
    else:
        rng = rng_holder

    best_a = _normalize(a0)
    best_sc = _score_fft(best_a)
    n = int(best_a.size)
    if n <= 1:
        return best_a, best_sc
    L = 2 * n - 1
    m = _next_pow2(max(16, L))

    v = _v_from_a(best_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-6
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0
    tau = 0.30
    alpha = 0.10
    start = time.monotonic()
    last_improve = start
    last_kick = start - 100.0

    while time.monotonic() < deadline - 2.0:
        now = time.monotonic()
        if not np.all(np.isfinite(v)):
            v = _v_from_a(best_a)
            continue
        a = _soft(v, n)
        if now - last_improve > 35.0 and tau > 0.03:
            tau = max(0.03, tau * 0.65)

        u, g, sc = _raw_obj_grad(a, m, tau)
        if not np.isfinite(u) or not np.isfinite(sc):
            v = _v_from_a(best_a)
            u, g, sc = _raw_obj_grad(_soft(v, n), m, tau)
            if not np.isfinite(u) or not np.isfinite(sc):
                break

        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = float(sc)
            best_a = a.copy()
            last_improve = now

        if now - last_improve > 70.0 and now - last_kick > 25.0:
            sigma = 0.08 + 0.05 * (1.0 - min(1.0, (now - start) / max(1.0, deadline - start)))
            v = v + rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -30.0, 30.0)
            mu *= 0.35
            nu *= 0.35
            last_kick = now

        g = np.asarray(g, dtype=np.float64)
        rms = float(np.sqrt(np.mean(g * g)))
        if not np.isfinite(rms) or rms < 1e-14:
            v = v + rng.normal(0.0, 0.04, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -30.0, 30.0)
            continue
        gd = g / rms

        t += 1
        mu = beta1 * mu + (1.0 - beta1) * gd
        nu = beta2 * nu + (1.0 - beta2) * (gd * gd)
        b1 = beta1 ** t
        b2 = beta2 ** t
        mhat = mu / max(1e-15, 1.0 - b1)
        vhat = nu / max(1e-15, 1.0 - b2)
        step = mhat / (np.sqrt(vhat) + eps)
        srms = float(np.sqrt(np.mean(step * step)))
        if not np.isfinite(srms) or srms < 1e-14:
            continue
        step = step / srms
        step = step - float(np.mean(step))

        accepted = False
        trial = alpha
        u_curr = u
        v_curr = v
        a_curr = a
        sc_curr = sc
        for _ in range(4):
            v_try = v_curr - trial * step
            v_try = v_try - float(np.mean(v_try))
            np.clip(v_try, -30.0, 30.0, out=v_try)
            a_try = _soft(v_try, n)
            u_try, sc_try = _raw_f_score(a_try, m, tau)
            if not np.isfinite(u_try):
                trial *= 0.5
                continue
            if (u_try < u_curr - 1e-12 and sc_try < sc_curr + 0.02) or sc_try < best_sc - 1e-9:
                v_curr = v_try
                a_curr = a_try
                u_curr = u_try
                sc_curr = sc_try
                if np.isfinite(sc_try) and sc_try < best_sc:
                    best_sc = float(sc_try)
                    best_a = a_try.copy()
                    last_improve = time.monotonic()
                accepted = True
                break
            trial *= 0.5

        if accepted:
            v = v_curr
            a = a_curr
            u = u_curr
            sc = sc_curr
            alpha = min(0.25, alpha * 1.06)
        else:
            alpha = max(0.003, alpha * 0.5)
            if alpha < 0.004:
                v = v + rng.normal(0.0, 0.025, n)
                v = v - float(np.mean(v))
                v = np.clip(v, -30.0, 30.0)
                mu *= 0.6
                nu *= 0.6

        if t % 160 == 0:
            a_s = _sym(a)
            s_s = _score_fft(a_s)
            if np.isfinite(s_s) and s_s < sc:
                v = _v_from_a(a_s)
                a = a_s
                if s_s < best_sc:
                    best_sc = float(s_s)
                    best_a = a_s.copy()
                    last_improve = time.monotonic()

    a_s = _sym(best_a)
    s_s = _score_fft(a_s)
    if np.isfinite(s_s) and s_s < best_sc:
        best_a = a_s
        best_sc = float(s_s)
    return best_a, float(best_sc)

def _improve_candidate(a0, deadline, rng):
    global rng_holder
    if rng is None:
        rng = np.random.default_rng(1234567)
    rng_holder = rng

    best_a = _normalize(a0)
    best_sc = _score_fft(best_a)
    now = time.monotonic()
    rem = max(0.0, deadline - now)
    if rem < 20.0:
        return best_a, float(best_sc)

    if _HAS_SCIPY and rem > 80.0:
        alloc = min(rem * 0.22, 240.0)
        a1, s1 = _lbfgs_polish(best_a, now + alloc, 0.25)
        if np.isfinite(s1) and s1 < best_sc:
            best_sc = float(s1)
            best_a = a1

    now = time.monotonic()
    rem = max(0.0, deadline - now)
    if _HAS_SCIPY and rem > 50.0:
        alloc = min(rem * 0.16, 150.0)
        a2, s2 = _lbfgs_polish(best_a, now + alloc, 0.10)
        if np.isfinite(s2) and s2 < best_sc:
            best_sc = float(s2)
            best_a = a2

    now = time.monotonic()
    rem = max(0.0, deadline - now)
    if rem > 30.0:
        seed = int(rng.integers(0, 2**31 - 1))
        a3, s3 = _adam_refine(best_a, deadline, seed)
        if np.isfinite(s3) and s3 < best_sc:
            best_sc = float(s3)
            best_a = a3

    a_s = _sym(best_a)
    s_s = _score_fft(a_s)
    if np.isfinite(s_s) and s_s < best_sc:
        best_a = a_s
        best_sc = float(s_s)
    return best_a, float(best_sc)


def _make_targets(b, L):
    modes = [
        ("flat", 0.0),
        ("alpha", 0.30),
        ("alpha", 0.55),
        ("alpha", 0.75),
        ("clip", 0.35),
        ("clip", 0.70),
        ("mix", 0.45),
        ("mix", 0.65),
    ]
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    if b.size != L:
        b = np.zeros(L, dtype=np.float64)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(np.sum(b))
    if s <= 0.0 or not np.isfinite(s):
        b = np.ones(L, dtype=np.float64) / float(L)
    else:
        b = b / s
    r = L * b
    rmax = float(np.max(r))
    out = []
    seen = []
    for mode in modes:
        typ = mode[0]
        arg = float(mode[1])
        if typ == "flat":
            t = np.ones(L, dtype=np.float64)
        elif typ == "alpha":
            if arg <= 0.0:
                t = np.ones(L, dtype=np.float64)
            else:
                rpos = np.maximum(r, 0.0)
                t = np.power(rpos, arg)
        elif typ == "clip":
            cap = 1.0 + arg * max(0.0, rmax - 1.0)
            t = np.minimum(r, cap)
        elif typ == "mix":
            t = arg * 1.0 + (1.0 - arg) * r
        else:
            t = np.ones(L, dtype=np.float64)
        t = np.where(np.isfinite(t), t, 0.0)
        t = np.maximum(t, 0.0)
        ts = float(np.sum(t))
        if ts <= 0.0 or not np.isfinite(ts):
            continue
        t = t / ts
        dup = False
        for q in seen:
            d = float(np.sqrt(np.sum((t - q) * (t - q))))
            if d < 0.01:
                dup = True
                break
        if not dup:
            seen.append(t)
            out.append(t)
    return out


def _spectral_candidate(a0, target, n, m, iters, deadline):
    a = _normalize(a0)
    n = int(n)
    target = np.asarray(target, dtype=np.float64).reshape(-1)
    L = 2 * n - 1
    if target.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:target.size] = target[:L]
        target = tmp
    else:
        target = target[:L].astype(np.float64, copy=True)
    target = np.where(np.isfinite(target), target, 0.0)
    target = np.maximum(target, 0.0)
    ts = float(np.sum(target))
    if ts <= 0.0 or not np.isfinite(ts):
        target = np.ones(L, dtype=np.float64) / float(L)
    else:
        target = target / ts

    T = np.fft.fft(target, m)
    F = np.fft.fft(a, m)
    best = a.copy()
    best_sc = _score_fft(a)

    iters = int(max(1, iters))
    for it in range(iters):
        if time.monotonic() >= deadline:
            break
        absF = np.abs(F)
        phase = np.exp(1j * np.angle(F))
        eps = 1e-12 + 1e-9 * absF
        Fsafe = F + eps * phase
        Fnew = 0.5 * (F + T / Fsafe)
        anew = np.fft.ifft(Fnew).real[:n]
        if not np.all(np.isfinite(anew)):
            break
        pos = np.maximum(anew, 0.0)
        ps = float(np.sum(pos))
        if ps <= 1e-6 or not np.isfinite(ps):
            pos = a.copy()
        pos = _normalize(pos)
        mix = 0.65 - 0.35 * (float(it) / float(iters))
        a_new = (1.0 - mix) * a + mix * pos
        a_new = _normalize(a_new)
        sc = _score_fft(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best = a_new.copy()
            a = a_new
            F = np.fft.fft(a, m)
        else:
            if np.isfinite(sc) and sc < best_sc * 1.25:
                a = a_new
                F = np.fft.fft(a, m)
            else:
                a = best.copy()
                F = np.fft.fft(a, m)
    return best, float(best_sc)


def _diverse_select(cands, scores, max_items, div):
    max_items = int(max(1, max_items))
    if len(cands) == 0:
        return []
    pairs = []
    for i in range(len(cands)):
        pairs.append((float(scores[i]), i))
    pairs.sort(key=_pair_first)
    order = [p[1] for p in pairs]
    selected_idx = []
    seen = []
    for idx in order:
        if len(selected_idx) >= max_items:
            break
        a = cands[idx]
        ok = True
        for b in seen:
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < div:
                ok = False
                break
        if ok:
            selected_idx.append(idx)
            seen.append(a)
    for idx in order:
        if len(selected_idx) >= max_items:
            break
        if idx not in selected_idx:
            selected_idx.append(idx)
    return [cands[i] for i in selected_idx]


def _process_stage(n, deadline, best_a, rng):
    n = int(n)
    deadline = float(deadline)
    now = time.monotonic()
    rem = max(0.0, deadline - now)
    if best_a is None:
        a_best = _random_fourier(n, rng, 18)
        s_best = _score_fft(a_best)
    else:
        a_best = _interp(best_a, n)
        s_best = _score_fft(a_best)

    inits = []
    inits.append(a_best)
    inits.append(_sym(a_best))
    inits.append(_perturb(a_best, 0.07, rng))
    inits.append(_perturb(a_best, 0.16, rng))
    inits.append(_random_fourier(n, rng, 22))
    inits.append(_random_fourier(n, rng, 10))

    if rem > 55.0:
        m = _next_pow2(max(16, 2 * n - 1))
        F = np.fft.fft(a_best, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        targets = _make_targets(b, 2 * n - 1)
        spec_deadline = min(deadline - 25.0, now + max(10.0, rem * 0.22))
        max_targets = 4 if rem > 180.0 else 2
        iters = 26 if n <= 2048 else 18
        for j in range(min(max_targets, len(targets))):
            if time.monotonic() >= spec_deadline:
                break
            a_t, s_t = _spectral_candidate(a_best, targets[j], n, m, iters, spec_deadline)
            if np.isfinite(s_t):
                inits.append(a_t)
                if s_t < s_best:
                    s_best = float(s_t)
                    a_best = a_t

    cands = []
    scores = []
    for a in inits:
        a = _normalize(a)
        sc = _score_fft(a)
        if np.isfinite(sc):
            cands.append(a)
            scores.append(sc)

    if not cands:
        a_best = _normalize(a_best)
        s_best = _score_fft(a_best)
        return a_best, float(s_best)

    max_candidates = 4 if rem > 280.0 else 3
    selected = _diverse_select(cands, scores, max_candidates, 0.045)
    if not selected:
        selected = [cands[0]]

    weights = (0.48, 0.30, 0.16, 0.06)
    initial_rem = max(0.0, deadline - time.monotonic())
    for i, a in enumerate(selected):
        now = time.monotonic()
        rem = max(0.0, deadline - now)
        if rem < 25.0:
            break
        w = weights[i] if i < len(weights) else 0.08
        alloc = max(20.0, min(rem * 0.92, initial_rem * w))
        sub_deadline = min(deadline - 2.0, now + alloc)
        if sub_deadline - now < 20.0:
            break
        a_new, s_new = _improve_candidate(a, sub_deadline, rng)
        if np.isfinite(s_new) and s_new < s_best:
            s_best = float(s_new)
            a_best = a_new

    a_s = _sym(a_best)
    s_s = _score_fft(a_s)
    if np.isfinite(s_s) and s_s < s_best:
        a_best = a_s
        s_best = float(s_s)
    return a_best, float(s_best)


def _try_thresholds(a, sc):
    a = _normalize(a)
    sc = float(sc)
    best_a = a
    best_sc = sc
    for th in (1e-10, 1e-9, 1e-8, 1e-7):
        t = best_a.copy()
        t[t < th] = 0.0
        t = _normalize(t)
        s = _score_fft(t)
        if np.isfinite(s) and s < best_sc:
            best_a = t
            best_sc = float(s)
    return best_a, float(best_sc)


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    ]

    for n, frac in stages:
        now = time.monotonic()
        if now + 45.0 >= deadline:
            break
        stage_time = base * float(frac)
        stage_deadline = min(deadline - 8.0, now + stage_time)
        print("Stage n=%d budget=%.1fs current_best=%.6f" % (n, stage_time, best_sc))
        a_new, s_new = _process_stage(n, stage_deadline, best_a, rng)
        if a_new is not None and np.isfinite(s_new) and s_new < best_sc:
            best_sc = float(s_new)
            best_a = a_new
        print("After stage n=%d best=%.6f" % (n, best_sc))

    now = time.monotonic()
    rem = max(0.0, deadline - now)
    if rem > 260.0 and best_a is not None:
        cands = []
        cands.append(best_a)
        cands.append(_sym(best_a))
        cands.append(_perturb(best_a, 0.09, rng))
        cands.append(_perturb(best_a, 0.22, rng))
        cands.append(_random_fourier(len(best_a), rng, 36))
        cs = []
        for a in cands:
            a = _normalize(a)
            s = _score_fft(a)
            if np.isfinite(s):
                cands[-1] = a
        # Recompute cleanly to avoid bug above
        good_c = []
        good_s = []
        for a in cands:
            a = _normalize(a)
            s = _score_fft(a)
            if np.isfinite(s):
                good_c.append(a)
                good_s.append(s)
        if good_c:
            sel = _diverse_select(good_c, good_s, 3, 0.03)
            init_rem = max(0.0, deadline - time.monotonic())
            weights = (0.48, 0.30, 0.22)
            for i, a in enumerate(sel):
                now = time.monotonic()
                if time.monotonic() + 25.0 >= deadline:
                    break
                w = weights[i] if i < len(weights) else 0.2
                alloc = max(25.0, min((deadline - now) * 0.85, init_rem * w))
                sub_deadline = min(deadline - 3.0, now + alloc)
                if sub_deadline - now < 22.0:
                    break
                a_new, s_new = _improve_candidate(a, sub_deadline, rng)
                if np.isfinite(s_new) and s_new < best_sc:
                    best_sc = float(s_new)
                    best_a = a_new

    now = time.monotonic()
    if now + 25.0 < deadline and best_a is not None:
        a_final, s_final = _improve_candidate(best_a, deadline, rng)
        if np.isfinite(s_final) and s_final < best_sc:
            best_sc = float(s_final)
            best_a = a_final

    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
        best_sc = _score_fft(best_a)

    best_a = _normalize(best_a)
    best_sym = _sym(best_a)
    best_sym_sc = _score_fft(best_sym)
    if np.isfinite(best_sym_sc) and best_sym_sc < best_sc:
        best_a = best_sym
        best_sc = float(best_sym_sc)

    best_a, best_sc = _try_thresholds(best_a, best_sc)

    if len(best_a) <= 4096:
        try:
            ex_upper = time.monotonic()
            if ex_upper < float(budget_s) - 15.0:
                exact = evaluate_sequence([float(x) for x in best_a])
                print("Final FFT score=%.6f exact score=%.6f n=%d" %
                      (best_sc, float(exact), len(best_a)))
        except Exception:
            pass

    return [float(x) for x in best_a]
```
