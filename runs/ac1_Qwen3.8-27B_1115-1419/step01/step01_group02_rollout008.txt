import numpy as np
import time
from scipy.fft import fft, ifft

def _fft_conv(a, b=None):
    if b is None:
        b = a
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    A = fft(a, n)
    B = fft(b, n)
    return ifft(A * B).real[:n_out]

def _score(a):
    """Compute the score for a sequence. Assumes a >= 0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = _fft_conv(a)
    max_c = float(np.max(conv))
    return float(2.0 * n * max_c / (s * s))

def _normalize(a, target_sum):
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

                      momentum_beta=0.9, print_interval=500):

    Subgradient optimization with multi-peak balancing.
    """
    n = len(a)
    target_sum = float(np.sum(a))
    momentum = np.zeros(n)
    lr = lr_init
    best = a.copy()
    best_score = _score(a)
    step = 0
    patience = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        conv = _fft_conv(a)
        max_c = float(np.max(conv))
        s = target_sum  # sum is preserved by normalization
        
        # Find top peaks

        threshold = 0.95 * max_c
        peak_indices = np.where(conv >= threshold)[0]
        
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        
        # Cap at n_peaks for efficiency
        if len(peak_indices) > n_peaks:
            # Pick the n_peaks highest
            top_vals = conv[peak_indices]
            order = np.argsort(top_vals)[-n_peaks:]
            peak_indices = peak_indices[order]
        
        # Compute weighted subgradient (uniform weights for all peaks)
        grad = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            # mirror[j] = a[k-j] for valid j
            lo = max(0, k - (n - 1))

            for j in range(lo, hi + 1):

        conv = _fft_conv(a)
        max_c = float(np.max(conv))
        
        # Find peaks above 90% of max
        thresh = 0.90 * max_c
        peak_pos = np.where(conv >= thresh)[0]
        
        # Limit to top 15
        if len(peak_pos) > 15:
            top_vals = conv[peak_pos]
            top_pos = peak_pos[np.argsort(top_vals)[-15:]]
        else:
            top_pos = peak_pos
        
        # Compute combined gradient
        grad = np.zeros(n)
        for k in top_pos:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[j] += a[k-j] for j in [lo, hi]
            grad[lo:hi+1] += a[k - np.arange(lo, hi+1)]
        
        # Center the gradient (preserve sum)
        grad -= np.mean(grad)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad /= gnorm
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        mnorm = np.linalg.norm(momentum)
        if mnorm > 1e-12:
            momentum /= mnorm
        
        # Update
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, target_sum)
        
        # Evaluate
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
