Enhanced search for low-peak nonnegative self-convolutions.
Uses Adam-optimized LSE gradient descent with coordinate descent polishing,
multi-length exploration, and diverse initialization.
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _conv(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _conv_grad(a, p, fft_size, n):
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size).real
    h = conv[n - 1:2 * n - 1]
    h = np.maximum(h, 0.0)
    return 2.0 * h


def _lse_p(b, beta):
    m = float(np.max(b))
    if beta <= 1.0:
        return np.ones(b.size, dtype=np.float64) / b.size, m
    arg = np.maximum(beta * (b - m), -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    val = m + float(np.log(s)) / beta
    return e / s, val


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b = _conv(a, _next_pow2(2 * n - 1))
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_cosine_taper(n):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _make_gaussian(n, width, center=0.5):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-((x - center) / width) ** 2)
    return _normalize(a)


def _make_mixture(n, alpha, gamma):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-alpha * x) + gamma * np.exp(-alpha * (1.0 - x))
    a = a + 0.1
    return _normalize(a)


def _make_smooth_random(n, rng, base=None, amplitude=0.4):
    n = max(int(n), 2)
    if base is None:
        base = _make_uniform(n)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 12 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = base * (1.0 + amplitude * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_uniform(n):
    return np.ones(max(int(n), 1), dtype=np.float64) / max(int(n), 1)


def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s < 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb_block(a, rng, strength=0.2):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)


def _perturb_mix(a, rng):
    if rng.random() < 0.5:
        return _perturb_block(a, rng, 0.15)
    noise = np.exp(rng.normal(0.0, 0.2, size=a.size))
    a2 = a * noise + 0.01 * rng.random(a.size) / a.size
    return _normalize(a2)


def _adam_lse_optimize(a, deadline, rng, beta_start=8.0, beta_end=500.0,
                       lr_start=0.05, patience=2000):
    """
    Adam-optimized LSE gradient descent on the simplex.
    Returns (best_a, best_score).
    """
    a = _normalize(a)
    n = a.size
    if n < 8:
        a = _make_exponential(128, 1.0)
        n = 128

    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n - 2)

    b = _conv(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    beta = beta_start
    lr = lr_start
    it = 0
    fail = 0
    last_improve = 0

    # Adam moments
    m_v = np.zeros(n, dtype=np.float64)
    v_v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    total_iters_planned = 50000
    beta_schedule = np.linspace(np.log(beta_start), np.log(max(beta_end, beta_start)), total_iters_planned)

    while time.monotonic() < deadline:
        if maxb < 1e-300:
            maxb = 1e-300

        # Adaptive beta: increase over time
        frac = min(1.0, it / total_iters_planned)
        target_beta = float(np.exp(beta_schedule[int(frac * (len(beta_schedule) - 1))]))
        # Also scale by 1/maxb to make LSE track the true max
        beta = min(1e6, max(1.0, target_beta / maxb))

        p, smooth_old = _lse_p(b / maxb, beta)

        grad = _conv_grad(a, p, g_fft, n)
        # Center the gradient (removes the component along the all-ones direction,
        # which doesn't change the score under normalization)
        grad_c = grad - float(np.mean(grad))
        grad_norm = float(np.max(np.abs(grad_c)))

        if grad_norm < 1e-300:
            a = _perturb_mix(best_a, rng)
            b = _conv(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            lr = lr_start
            fail = 0
            m_v[:] = 0.0
            v_v[:] = 0.0
            it += 1
            continue

        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad_c / grad_norm
        v_v = beta2 * v_v + (1 - beta2) * (grad_c / grad_norm) ** 2
        m_hat = m_v / (1 - beta1 ** (it + 1))
        v_hat = v_v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        v = a - step
        a_new = _normalize(v)

        b_new = _conv(a_new, b_fft)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * float(n) * maxb_new
        p_new, smooth_new = _lse_p(b_new / max(maxb_new, 1e-300), beta)

        if score_new < best_score - 1e-15:
            best_a = a_new.copy()
            best_score = score_new
            last_improve = it

        if smooth_new < smooth_old - 1e-12:
            lr = min(0.5, lr * 1.05)
            fail = 0
            a = a_new
            b = b_new
            maxb = maxb_new
            score = score_new
        else:
            lr = max(1e-5, lr * 0.7)
            fail += 1
            # Don't accept the step
            a = a  # keep old
            # b and maxb stay the same

        # Schedule adjustments
        if it % 100 == 0 and fail > 50:
            beta_start = min(100.0, beta_start * 1.1)  # effectively speeds up beta

        # Restart conditions
        if (fail > patience or (it - last_improve > 3000 and it > 500)) and time.monotonic() < deadline - 3.0:
            a = _perturb_mix(best_a, rng)
            b = _conv(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            lr = lr_start
            fail = 0
            m_v[:] = 0.0
            v_v[:] = 0.0

        if fail > 5000 and score > 1.05 * best_score + 0.02:
            a = best_a.copy()
            b = _conv(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            lr = lr_start * 0.5
            fail = 0
            m_v[:] = 0.0
            v_v[:] = 0.0

        it += 1

    return best_a, best_score


def _coordinate_polish(a, deadline, max_passes=20):
    """
    Coordinate descent polish: for each coordinate, try local adjustments.
    """
    a = _normalize(a)
    n = a.size
    b_fft = _next_pow2(2 * n - 1)
    b = _conv(a, b_fft)
    maxb = float(np.max(b))
    best_score = 2.0 * float(n) * maxb
    best_a = a.copy()

    for _pass in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random coordinate order
        order = np.arange(n)
        np.random.shuffle(order)
        for idx in range(n):
            if time.monotonic() > deadline:
                break
            i = int(order[idx])
            # Try a few multipliers for a[i]
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                trial = a.copy()
                trial[i] *= mult
                trial = _normalize(trial)
                b_t = _conv(trial, b_fft)
                mb_t = float(np.max(b_t))
                sc_t = 2.0 * float(n) * mb_t
                if sc_t < best_score - 1e-14:
                    a = trial
                    b = b_t
                    maxb = mb_t
                    best_score = sc_t
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_a, best_score


def _block_polish(a, deadline, block_size=8):
    """
    Block coordinate polish: adjust groups of coordinates together.
    """
    a = _normalize(a)
    n = a.size
    b_fft = _next_pow2(2 * n - 1)
    b = _conv(a, b_fft)
    best_score = 2.0 * float(n) * float(np.max(b))
    best_a = a.copy()
    rng = np.random.default_rng(42)

    n_blocks = max(1, n // block_size)
    for _ in range(min(n_blocks, 50)):
        if time.monotonic() > deadline:
            break
        start = int(rng.integers(0, max(1, n - block_size)))
        stop = min(n, start + block_size)
        # Try scaling the block
        for mult in [0.8, 0.9, 1.1, 1.2]:
            trial = a.copy()
            trial[start:stop] *= mult
            trial = _normalize(trial)
            b_t = _conv(trial, b_fft)
            sc_t = 2.0 * float(n) * float(np.max(b_t))
            if sc_t < best_score - 1e-14:
                best_a = trial.copy()
                best_score = sc_t
                a = trial

    return best_a, best_score

                best_a = pa.copy()
                best_score = ps
                print(f"  Final polish: {best_score:.8f}", flush=True)

        # Block polish
        if time.monotonic() < final_deadline - 1:
            ba, bs = _block_polish(best_a, final_deadline - 1)
            if bs < best_score:
                best_a = ba.copy()
                best_score = bs
                print(f"  Final block polish: {best_score:.8f}", flush=True)

    print(f"\nFinal best: score={best_score:.8f}, length={best_a.size}, "
          f"time={time.monotonic()-t0:.1f}s", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)
    return [float(x) for x in out]
