Search for low-peak nonnegative self-convolutions using enhanced
LSE-smoothed gradient descent with momentum, adaptive beta scheduling,
and multi-length exploration.
"""

import numpy as np
import time

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]

def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb

    m = float(np.max(b))
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0:
        e = np.ones(b.size)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val

    """Compute weights concentrated on top-k elements."""
    sorted_idx = np.argsort(b)[::-1]
    top_k = sorted_idx[:k]
    top_vals = b[top_k]
    m = float(np.max(top_vals))
    arg = (top_vals - m) / max(temperature, 1e-10)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0:
        return np.ones(b.size) / b.size
    p = np.zeros(b.size)
    p[top_k] = e / s
    return p

    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)

    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]

def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb

def _lse_and_p(b, beta):
    m = float(np.max(b))
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0:
        e = np.ones(b.size)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val

def _weighted_gradient(a, weights, fft_size, n):
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    h = np.where(h > 0.0, h, 0.0)
    return 2.0 * h

def _project_simplex(v):
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    n = w.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return w / s

def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)

def _make_gaussian(n, center, width):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return _normalize(a)

def _make_symmetric_exp(n, alpha):
    """Symmetric exponential: high in the middle, decaying toward edges."""
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * np.abs(x - 0.5) * 2)
    return _normalize(a)

def _make_bell(n, alpha, beta_param):
    """Bell-shaped: a(x) = exp(-alpha*(x-0.5)^2) + beta_param (uniform floor)."""
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * (x - 0.5) ** 2) + float(beta_param)
    return _normalize(a)

def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

def _make_uniform(n):
    return np.ones(max(int(n), 1), dtype=np.float64) / max(int(n), 1)

    beta = 20.0  # LSE sharpness

        if maxb <= 1e-300:

        # When beta * maxb is large, the LSE is sharp

        grad = _weighted_gradient(a, p, g_fft, n)

        scale = float(np.max(np.abs(grad_c)))

            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.5
            v *= 0.5
            fail = 0
            it += 1
            if verbose:
                print(f"  it={it} perturbed, score={score:.8f}", flush=True)
            continue

        # Normalized gradient
        grad_norm = grad_c / scale
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad_norm
        v_norm = float(np.linalg.norm(v))
        if v_norm > 0:
            v = v / v_norm
        
        # Step
        a_new = _project_simplex(a - alpha * v)
        
        # Evaluate
        b_new = _autocorr(a_new, b_fft)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * float(n) * maxb_new
        
        # Check improvement in smooth objective

        if smooth_new < smooth_old - 1e-15:
            a = a_new
            b = b_new
            maxb = maxb_new
            score = score_new
            alpha = float(min(5.0, alpha * 1.05))
            fail = 0
            last_improve = it
        else:
            alpha *= 0.7
            v *= 0.5
            fail += 1
        
        if score < best_score - 1e-15:
            best_a = a.copy()
            best_score = score
            if verbose:
                print(f"  it={it} NEW BEST score={best_score:.8f} beta={beta:.1f}", flush=True)

        # Adaptive beta increase
        if it % 50 == 0:
            beta = float(min(beta_max, beta * 1.15))
        if fail > 50:
            beta = float(min(beta_max, beta * 1.1))
        if fail > 200:
            beta = float(min(beta_max, beta * 1.5))

        # Restart if stuck
        if (fail > 500 and it > 200) or (it - last_improve > 2000 and it > 500):
            a = _perturb(best_a, rng)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb

            fail = 0

def _perturb(a, rng, strength=0.3):
    """Random perturbation of the sequence."""
    n = a.size
    # Block perturbation
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    a2 = a.copy()
    a2[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    
    # Occasionally add global noise
    if rng.random() < 0.3:
        noise = np.exp(rng.normal(0.0, 0.2, size=n))
        a2 = 0.8 * a2 + 0.2 * noise / noise.sum()
    
    # Occasionally symmetrize
    if rng.random() < 0.2:
        a2 = 0.5 * (a2 + a2[::-1])
    
    return _normalize(a2)

def _generate_inits(n, given, best, rng):

    inits.append(_make_bell(n, 8.0, 0.05))

    # If we have a given sequence, resample
    if given is not None and given.size > 4:
        inits.append(_resample_sequence(given, n))
    
    # Smooth random
    base = _make_exponential(n, 1.0)

    f = np.fft.irfft(F, n)
    f = (f - f.mean()) / (np.std(f) + 1e-10)
    a = base * (1.0 + 0.3 * f)
    a = np.maximum(a, 1e-10)
    inits.append(_normalize(a))
    
    return inits
