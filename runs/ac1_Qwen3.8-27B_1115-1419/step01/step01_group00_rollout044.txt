An alternative is to use `a = exp(w)` parameterization, which automatically satisfies `a >= 0`, and then we just need to handle the sum constraint.

With `a = n * exp(w) / sum(exp(w))` (softmax parameterization):
- `a >= 0` is automatic
- `sum(a) = n` is automatic
- We optimize over `w` (unconstrained!)
- Gradient w.r.t. `w`: `da/dw_j = a_j * (delta_jk - a_k/n)` (Jacobian of softmax scaled by n)

  `= n * g_j - a_j/n * sum_k g_k * a_k` (when sum(a) = n)

- With `a = n * softmax(w)`, the gradient computation involves the Jacobian of softmax, which is `a_j/n * (delta_jk - a_k/n)`.
- This means the effective gradient in w-space is: `g_w_j = n * g_a_j - a_j/n * sum_k g_a_k * a_k`

This is actually exactly what we want! It's the gradient projected onto the tangent space of the constraint `sum(a) = n`.

But there's a subtlety: the softmax parameterization means `a_j > 0` always (never exactly 0). This could be a problem if the optimal solution has some zero entries. However, for our problem, I don't think the optimal has exact zeros (the optimal should be "spread out"), so this should be fine.

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations:
1. Adam optimizer in softmax w-space (handles non-negativity and sum constraints automatically)
2. Nesterov-momentum smoothed descent in a-space as alternative
3. Multiple structured initializations (cosine, bell, U-shape, two-hump)
4. Targeted perturbations based on autocorrelation peak structure
5. Multiple n values explored with adaptive time allocation
6. Careful temperature annealing schedule for the log-sum-exp smoothed max
"""

import numpy as np
import time


def _conv(a):
    """Full self-convolution of a."""
    return np.convolve(a, a)


def _obj(a):
    """Objective 2*max(conv(a,a))/n assuming sum(a)=n."""
    n = len(a)
    c = _conv(a)
    return 2.0 * float(np.max(c)) / n


def _project(a, n):
    """Project onto {x >= 0, sum(x) = n}."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64)
    return a / s * n


def _safe_softmax(w):
    """Numerically stable softmax."""
    mx = float(w.max())
    e = np.exp(w - mx)
    s = float(e.sum())
    if s < 1e-300:
        return np.full_like(w, 1.0 / len(w), dtype=np.float64)
    return e / s


def _smooth_grad(c, a, t):
    """Gradient of smoothed max of c w.r.t. a (via log-sum-exp)."""
    c_shift = c - c.max()
    e = np.exp(t * c_shift)
    s = float(e.sum())
    if s < 1e-300:
        p = np.full_like(c, 1.0 / len(c))
    else:
        p = e / s
    return 2.0 * np.correlate(p, a, mode='valid')


def _hard_grad(a, k, n):
    """Exact gradient of (a*a)[k] w.r.t. a vector."""
    g = np.zeros(n, dtype=np.float64)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        idx = k - np.arange(js, je)
        g[js:je] = 2.0 * a[idx]
    return g


def _adam_w_descent(a0, n, deadline, tag, lr=0.5):
    """
    Adam-optimized smoothed gradient descent in w-space.
    a = n * softmax(w), so constraints are automatic.
    """
    w = np.log(a0 / n + 1e-12)
    w = w - float(w.mean())
    best_a = a0.copy()
    best_o = _obj(a0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    t = max(n / 6.0, 10.0)
    t_min = 3.0 / n
    t_decay = 0.99992
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        a = n * _safe_softmax(w)
        c = _conv(a)
        mc = float(c.max())
        o = 2.0 * mc / n

        if o < best_o - 1e-12:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it % 5000 == 0:
                print(f"    [{tag}] it={it} obj={o:.8f} t={t:.6f}")
        else:
            stall += 1

        g_a = _smooth_grad(c, a, t)
        cross = float(np.dot(g_a, a))
        g_w = n * g_a - (a / n) * cross

        it += 1
        m = beta1 * m + (1.0 - beta1) * g_w
        v = beta2 * v + (1.0 - beta2) * g_w * g_w
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        grad_norm = float(np.max(np.abs(m_hat / (np.sqrt(v_hat) + eps))))
        if grad_norm < 1e-14:
            break
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -5.0, 5.0)
        w = w - step

        t = max(t * t_decay, t_min)

        if stall > 12000:
            lr = max(lr * 0.5, 1e-12)
            stall = 0

    return best_a, best_o


def _momentum_smooth_descent(a0, n, deadline, tag, lr=0.05):
    """
    Nesterov-momentum smoothed gradient descent in a-space with projection.
    """
    a = a0.copy()
    best_a = a.copy()
    best_o = _obj(a)
    vel = np.zeros(n, dtype=np.float64)
    beta = 0.9
    t = max(n / 8.0, 5.0)
    t_min = 2.0 / n
    t_decay = 0.99995
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        c = _conv(a)
        mc = float(c.max())
        o = 2.0 * mc / n

        if o < best_o - 1e-12:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it % 5000 == 0:
                print(f"    [{tag}] it={it} obj={o:.8f} t={t:.6f}")
        else:
            stall += 1

        g = _smooth_grad(c, a, t)
        g = g - float(g.mean())

        vel = beta * vel - lr * g
        a_new = _project(a + vel, n)

        t = max(t * t_decay, t_min)

        if stall > 10000:
            lr = max(lr * 0.5, 1e-10)
            vel *= 0.3
            stall = 0

        a = a_new
        it += 1

    return best_a, best_o


def _hard_descent(a0, n, deadline, tag, lr=0.02):
    """Exact active-set gradient descent with projection."""
    a = a0.copy()
    best_a = a.copy()
    best_o = _obj(a)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv(a)
        k = int(np.argmax(c))
        mc = float(c[k])
        o = 2.0 * mc / n

        if o < best_o - 1e-12:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it % 5000 == 0:
                print(f"    [{tag}] it={it} obj={o:.8f}")
        else:
            stall += 1

        g = _hard_grad(a, k, n)
        g = g - float(g.mean())
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        a = _project(a - lr * g / mg, n)

        if stall > 6000:
            lr = max(lr * 0.5, 1e-12)
            stall = 0
        it += 1

    return best_a, best_o


def _cosine_init(n):
    """Raised cosine initialization."""
    x = np.linspace(-1.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * x))
    return a / a.sum() * n


def _bell_init(n, w):
    """Gaussian bell initialization."""
    x = np.linspace(-1.0, 1.0, n)
    a = np.exp(-(x / w) ** 2)
    return a / a.sum() * n


def _u_init(n, alpha):
    """U-shaped initialization (higher at edges)."""
    x = np.linspace(-1.0, 1.0, n)
    a = 1.0 + alpha * (x ** 2)
    return a / a.sum() * n


def _two_hump_init(n, pos=0.35):
    """Two-hump initialization."""
    x = np.linspace(-1.0, 1.0, n)
    a = np.exp(-((x + pos) / 0.25) ** 2) + np.exp(-((x - pos) / 0.25) ** 2) + 0.05
    return a / a.sum() * n


def _flat_edge_init(n, edge_frac=0.2):
    """Flat in middle, raised at edges."""
    a = np.ones(n)
    ne = max(1, int(n * edge_frac))
    a[:ne] = 1.5
    a[-ne:] = 1.5
    return a / a.sum() * n


def _generate_inits(n, rng):
    """Generate a diverse set of initializations."""
    inits = {}
    inits['cosine'] = _cosine_init(n)
    inits['bell_03'] = _bell_init(n, 0.3)
    inits['bell_05'] = _bell_init(n, 0.5)
    inits['u_03'] = _u_init(n, 0.3)
    inits['u_05'] = _u_init(n, 0.5)
    inits['two_hump'] = _two_hump_init(n)
    inits['flat_edge'] = _flat_edge_init(n)
    inits['uniform'] = np.ones(n, dtype=np.float64)
    # Random smooth init
    r = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 0.01)
    inits['rand_smooth'] = r / r.sum() * n
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses Adam in softmax space, momentum descent, hard active-set,
    targeted perturbations, and explores multiple n values.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0

    best_a = None
    best_o = np.inf

    # Get parent if available
    parent_a = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size >= 500 and np.all(np.isfinite(parent)) and np.all(parent >= 0) and parent.sum() > 0:
            parent_a = parent.copy()
            pa_norm = parent_a / parent_a.sum() * len(parent_a)
            print(f"Parent: n={len(parent_a)}, obj={_obj(pa_norm):.8f}")
    except Exception:
        pass

    # Determine n values to explore
    n_list = []
    if parent_a is not None:
        n_list.append(len(parent_a))
    for nn in [3809, 4096, 3072, 5000, 6144, 2048]:
        if nn not in n_list and 1000 <= nn <= 10000:
            n_list.append(nn)

    print(f"n candidates: {n_list}")
    print(f"Total budget: {budget_s}s, effective deadline in {deadline - t0:.0f}s")

    for ci, n in enumerate(n_list):
        time_left = deadline - time.monotonic()
        if time_left < 90:
            print(f"Time remaining {time_left:.0f}s < 90s, stopping n exploration.")
            break

        # Time allocation decreasing for later configs
        if ci == 0:
            alloc = min(time_left * 0.50, 800.0)
        elif ci == 1:
            alloc = min(time_left * 0.30, 500.0)
        else:
            alloc = min(time_left * 0.15, 300.0)

        print(f"\n{'='*60}")
        print(f"Config {ci}: n={n}, alloc={alloc:.0f}s")
        print(f"{'='*60}")

        # Generate initializations
        inits = _generate_inits(n, rng)
        if parent_a is not None and len(parent_a) == n:
            pa = np.clip(parent_a, 0, 1000).astype(np.float64)
            pa = pa / pa.sum() * n
            inits['parent'] = pa

        # Rank inits
        init_objs = {k: _obj(v) for k, v in inits.items()}
        sorted_keys = sorted(init_objs.keys(), key=lambda x: init_objs[x])
        print(f"  Init ranking: " + ", ".join(f"{k}={init_objs[k]:.4f}" for k in sorted_keys[:5]))

        # --- Phase 1: Best smooth optimization (top 2 inits) ---
        p1_frac = 0.45
        p1_deadline = time.monotonic() + alloc * p1_frac

        current_best_a = inits[sorted_keys[0]].copy()
        current_best_o = init_objs[sorted_keys[0]]

        # Adam w-space on best init
        a_r, o_r = _adam_w_descent(inits[sorted_keys[0]], n, min(time.monotonic() + alloc * 0.25, p1_deadline),
                                    f"adamw_{sorted_keys[0]}_n{n}", lr=0.5)
        print(f"  Adam-w ({sorted_keys[0]}): obj={o_r:.6f}")
        if o_r < current_best_o:
            current_best_a, current_best_o = a_r, o_r
        if o_r < best_o:
            best_o, best_a = o_r, a_r.copy()

        # Momentum smooth on second best init (if time)
        if time.monotonic() < p1_deadline - 20 and len(sorted_keys) > 1:
            p1b_deadline = min(time.monotonic() + alloc * 0.20, p1_deadline)
            a_r2, o_r2 = _momentum_smooth_descent(inits[sorted_keys[1]], n, p1b_deadline,
                                                   f"mom_{sorted_keys[1]}_n{n}", lr=0.05)
            print(f"  Momentum ({sorted_keys[1]}): obj={o_r2:.6f}")
            if o_r2 < current_best_o:
                current_best_a, current_best_o = a_r2, o_r2
            if o_r2 < best_o:
                best_o, best_a = o_r2, a_r2.copy()

        # Adam w-space on second best if time remains
        if time.monotonic() < p1_deadline - 10 and len(sorted_keys) > 1:
            p1c_deadline = min(time.monotonic() + alloc * 0.10, p1_deadline)
            a_r3, o_r3 = _adam_w_descent(inits[sorted_keys[1]], n, p1c_deadline,
                                          f"adamw2_{sorted_keys[1]}_n{n}", lr=0.4)
            print(f"  Adam-w2 ({sorted_keys[1]}): obj={o_r3:.6f}")
            if o_r3 < current_best_o:
                current_best_a, current_best_o = a_r3, o_r3
            if o_r3 < best_o:
                best_o, best_a = o_r3, a_r3.copy()

        # --- Phase 2: Hard active-set refinement ---
        p2_deadline = time.monotonic() + alloc * 0.25
        a_h, o_h = _hard_descent(current_best_a, n, p2_deadline, f"hard_n{n}", lr=0.02)
        print(f"  Hard descent: obj={o_h:.6f}")
        if o_h < current_best_o:
            current_best_a, current_best_o = a_h, o_h
        if o_h < best_o:
            best_o, best_a = o_h, a_h.copy()

        # --- Phase 3: Targeted perturb + rapid refine loop ---
        p3_deadline = time.monotonic() + (alloc * 0.30)
        pert_iter = 0
        while time.monotonic() < p3_deadline:
            pert_iter += 1
            # Determine perturbation parameters (increasing diversity)
            if pert_iter <= 10:
                sigma = float(rng.uniform(0.03, 0.15))
                width_frac = float(rng.choice([0.005, 0.01, 0.02, 0.05]))
            elif pert_iter <= 30:
                sigma = float(rng.uniform(0.08, 0.35))
                width_frac = float(rng.choice([0.01, 0.02, 0.05, 0.10]))
            else:
                sigma = float(rng.uniform(0.15, 0.60))
                width_frac = float(rng.choice([0.02, 0.05, 0.10, 0.20]))

            width = max(3, int(n * width_frac))
            s = int(rng.integers(0, max(1, n - width)))
            a_p = current_best_a.copy()
            a_p[s:s + width] *= np.exp(rng.normal(0.0, sigma, size=width))
            a_p = _project(a_p, n)

            # Quick hard refinement
            sub_dl = min(time.monotonic() + 12.0, p3_deadline)
            a_q, o_q = _hard_descent(a_p, n, sub_dl, f"p{n}_{pert_iter}", lr=0.015)
            if o_q < current_best_o:
                current_best_a, current_best_o = a_q, o_q
                if o_q < best_o:
                    best_o, best_a = o_q, a_q.copy()
                    print(f"    Perturb {pert_iter}: NEW BEST obj={best_o:.6f}")

            # Every 4th iteration: quick Adam smooth restart from current best
            if pert_iter % 4 == 0 and time.monotonic() < p3_deadline - 15:
                sub_dl2 = min(time.monotonic() + 18.0, p3_deadline)
                a_q2, o_q2 = _adam_w_descent(current_best_a, n, sub_dl2,
                                              f"sm{n}_{pert_iter}", lr=0.3)
                if o_q2 < current_best_o:
                    current_best_a, current_best_o = a_q2, o_q2
                    if o_q2 < best_o:
                        best_o, best_a = o_q2, a_q2.copy()
                        print(f"    Smooth reopt {pert_iter}: NEW BEST obj={best_o:.6f}")

        print(f"  Final for n={n}: obj={current_best_o:.6f} | global best: {best_o:.6f}")

    # --- Final polish on global best ---
    if best_a is not None:
        time_left = deadline - time.monotonic()
        if time_left > 20:
            n_f = len(best_a)
            print(f"\n{'='*60}")
            print(f"Final polish: n={n_f}, time={time_left:.0f}s")
            print(f"{'='*60}")
            pol_dl = deadline - 3.0
            a_fin, o_fin = _hard_descent(best_a, n_f, pol_dl, "FINAL", lr=0.01)
            if o_fin < best_o:
                best_o, best_a = o_fin, a_fin
            print(f"  After final polish: obj={o_fin:.8f}")

    # --- Validate and return ---
    print(f"\n{'='*60}")
    print(f"FINAL: obj={best_o:.8f}, n={len(best_a) if best_a is not None else 'N/A'}")
    print(f"{'='*60}")

    if best_a is None:
        result = [1.0] * 3809
        print("WARNING: No result found, returning uniform fallback.")
        return result

    result = [float(x) for x in best_a]
    try:
        sc = float(evaluate_sequence(result))
        print(f"Evaluated score: {sc:.8f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
        sc = float('inf')

    if not np.isfinite(sc):
        print("WARNING: Invalid evaluation, returning uniform fallback.")
        result = [1.0] * 3809

    return result
