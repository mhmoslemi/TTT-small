= sum (1+B*x_i)*(1-B*x_i) where x_i = 2i/(n-1)-1
= sum (1 - B^2*x_i^2)
= n - B^2 * sum x_i^2
≈ n - B^2 * n/3

So c[n-1] ≈ n(1 - B^2/3) which is LESS than n! The ramp reduces the center peak!

c[n-1-k] = sum a[i]*a[i+k] for a ramp.
For a linear ramp a[i] = 1 + B*(2i/(n-1)-1), this is:
sum (1+B*x_i)*(1+B*x_{i+k}) ≈ n(1 + B^2 * correlation coefficient)

The key is that for a ramp, the sequence and its shifted version are positively correlated for small shifts, so c[n-1-k] is close to n for small k. The maximum might shift to a nearby lag.

```python
"""
Searches for non-negative step-function heights minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

Algorithm:
- FFT-based convolution for fast objective evaluation.
- Weighted top-K peak subgradient with emphasis on highest peaks.
- Multiplicative update with simulated annealing acceptance.
- Adaptive learning rate with backtracking on failure.
- Diverse starts: linear ramps, two-block, cosine modulations, smooth random.
- Coarse-to-fine multi-resolution search with upsampled parent.
- Peak-flattening post-processing pass.
"""

import numpy as np
import time
import math


def fast_convolution(a):
    """FFT-based linear convolution a * a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def fast_score(a):
    """Fast FFT-based score."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fast_convolution(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def normalize_seq(a, target_sum):
    """Normalize sequence to target sum, clip to [0, 1000]."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        a.fill(target_sum / a.size)
        return a
    a *= target_sum / s
    a = np.clip(a, 0.0, 1000.0)
    # Re-normalize after clipping
    s = float(a.sum())
    if s < 1e-15:
        a.fill(target_sum / a.size)
    else:
        a *= target_sum / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def weighted_peak_gradient(a, c, K, temperature):
    """
    Compute weighted subgradient of the peak convolution.
    Weights peaks by (c[k] - baseline) with exponential emphasis.
    """
    n = a.size
    L = c.size
    K = max(1, min(K, L))
    
    if K == 1:
        k = int(np.argmax(c))
        g = np.zeros(n, dtype=np.float64)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                g[m] += a[j]
            if 0 <= m - k < n and (m - k) != j:
                g[m] += a[m - k]
        return 2.0 * g
    
    # Get top K indices
    top_idx = np.argpartition(c, -K)[-K:]
    
    # Weight by excess over 90th percentile (emphasize highest peaks)
    threshold = float(np.percentile(c, 75))
    weights = np.maximum(c[top_idx] - threshold, 1e-10)
    # Exponential emphasis
    weights = np.exp(temperature * weights / max(weights.max(), 1e-10))
    weights /= weights.sum()
    
    g = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for m in range(lo, hi):
            j = k - m
            if 0 <= j < n:
                g[m] += w * a[j]
            if 0 <= m - k < n:
                g[m] += w * a[m - k]
    return 2.0 * g


def vectorized_peak_gradient(a, c, K, emphasis=2.0):
    """
    Vectorized weighted top-K peak subgradient.
    Uses emphasis parameter to weight higher peaks more heavily.
    """
    n = a.size
    L = c.size
    K = max(1, min(K, L))
    
    top_idx = np.sort(np.argpartition(c, -K)[-K:])
    
    # Weight: exponential emphasis on peak height relative to median
    med = float(np.median(c))
    excess = np.maximum(c[top_idx] - med, 1e-10)
    scale = max(float(excess.max()), 1e-10)
    weights = np.exp(emphasis * excess / scale)
    weights /= weights.sum()
    
    g = np.zeros(n, dtype=np.float64)
    
    for w, k in zip(weights, top_idx):
        k = int(k)
        # c[k] = sum_{m=max(0,k-n+1)}^{min(n-1,k)} a[m] * a[k-m]
        # d c[k]/d a[m] = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        m_range = np.arange(lo, hi)
        j_idx = k - m_range  # k - m for m in [lo, hi)
        valid_j = (j_idx >= 0) & (j_idx < n)
        g[m_range[valid_j]] += w * a[j_idx[valid_j]]
        
        j_idx2 = m_range - k  # m - k
        valid_j2 = (j_idx2 >= 0) & (j_idx2 < n)
        # Avoid double counting when m-k == k-m, i.e., 2m == k
        g[m_range[valid_j2]] += w * a[j_idx2[valid_j2]]
    
    return 2.0 * g


def smooth_perturb(a, rng, strength=0.3):
    """Smooth multiplicative perturbation for restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n < 3:
        return a
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, 5):
        amp = strength / (0.5 + 0.3 * m)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    a *= np.exp(z)
    return normalize_seq(a, float(n))


def peak_shave(a, c, n_shaves=200, strength=0.1):
    """
    Direct peak shaving: iteratively reduce the largest convolution entry
    by adjusting the elements that contribute most to it.
    """
    a = a.copy()
    n = a.size
    target = float(n)
    
    for _ in range(n_shaves):
        c = fast_convolution(a)
        peak_val = float(np.max(c))
        peak_idx = int(np.argmax(c))
        
        # Find elements contributing most to this peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n, peak_idx + 1)
        m_range = np.arange(lo, hi)
        
        contributions = np.zeros(n)
        j_idx = peak_idx - m_range
        valid = (j_idx >= 0) & (j_idx < n)
        contributions[m_range[valid]] = a[j_idx[valid]]
        
        j_idx2 = m_range - peak_idx
        valid2 = (j_idx2 >= 0) & (j_idx2 < n)
        contributions[m_range[valid2]] += a[j_idx2[valid2]]
        
        # Reduce top contributors, increase bottom ones
        top_n = max(1, n // 10)
        bottom_n = max(1, n // 10)
        
        contrib_sum = contributions.sum()
        if contrib_sum < 1e-15:
            break
        
        # Scale factor to transfer mass
        delta = strength * peak_val / (n * max(contributions[hi-1-lo] if lo < hi else 1.0, 1e-10))
        delta = min(delta, 0.05 * float(a.mean()))
        
        top_mask = contributions > float(np.percentile(contributions[contributions > 0], 90)) if np.any(contributions > 0) else np.zeros(n, dtype=bool)
        bottom_mask = contributions < float(np.percentile(contributions, 10)) if n > 0 else np.zeros(n, dtype=bool)
        
        a[top_mask] -= delta
        a[bottom_mask] += delta
        a = np.maximum(a, 0.0)
        a = normalize_seq(a, target)
    
    return a


def make_ramp(n, slope, curvature=0.0):
    """Linear ramp with optional curvature."""
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (2.0 * t - 1.0) + curvature * ((2.0 * t - 1.0) ** 2 - 1.0/3.0)
    a = np.maximum(a, 0.01)
    return normalize_seq(a, float(n))


def make_two_block(n, frac, ratio):
    """Two-block sequence."""
    n = int(n)
    m = int(round(frac * n))
    m = max(1, min(n - 1, m))
    denom = m * ratio + (n - m)
    B = float(n) / denom
    A = ratio * B
    a = np.empty(n)
    a[:m] = A
    a[m:] = B
    return normalize_seq(a, float(n))


def make_cosine(n, freq, amp, phase=0.0, base=1.0):
    """Cosine-modulated sequence."""
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    a = base + amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    return normalize_seq(a, float(n))


def make_smooth_random(n, rng, n_harmonics=6, max_amp=1.5):
    """Random smooth sequence from low-frequency harmonics."""
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, n_harmonics + 1):
        amp = rng.uniform(-max_amp, max_amp) / (m ** 0.6)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    z -= float(np.max(z))
    a = np.exp(0.5 * z)
    return normalize_seq(a, float(n))


def upsample(seq, n):
    """Linearly interpolate and renormalize."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    n = int(n)
    if m == 0:
        return normalize_seq(np.ones(n), float(n))
    if m == n:
        return normalize_seq(seq, float(n))
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, seq)
    return normalize_seq(out, float(n))


def local_search(a, budget_s, seed=0, label=""):
    """
    Main local search: multiplicative update with weighted peak subgradient,
    simulated annealing, and adaptive step size.
    """
    a = normalize_seq(np.asarray(a, dtype=np.float64), float(len(a)))
    n = int(a.size)
    target = float(n)
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))
    
    c = fast_convolution(a)
    s = float(a.sum())
    score = float(2.0 * n * float(np.max(c)) / (s * s)) if s > 0 else float('inf')
    
    best = a.copy()
    best_score = score
    
    # Hyperparameters
    eta = 0.15  # base learning rate
    temperature = 0.05  # acceptance temperature
    cooling = 0.9997  # geometric cooling
    momentum_beta = 0.4
    noise_level = 0.015
    stagnant = 0
    it = 0
    step_history = []
    
    mom = np.zeros(n, dtype=np.float64)
    
    print(f"  [{label}] Start: score={score:.6f}, n={n}")
    
    while time.monotonic() < deadline:
        it += 1
        
        # Adaptive K: start with fewer peaks, increase over time
        L = 2 * n - 1
        frac = max(0.005, 0.03 / (1.0 + it / 3000.0))
        K = int(L * frac)
        K = max(2, min(K, min(256, max(2, n // 16))))
        
        # Compute gradient
        grad = vectorized_peak_gradient(a, c, K, emphasis=2.0)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Center gradient to preserve sum
        g = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(g)))
        if scale < 1e-14:
            stagnant += 1
            if stagnant > 100:
                break
            continue
        g /= scale
        
        # Momentum
        mom = momentum_beta * mom + (1.0 - momentum_beta) * g
        
        # Multiplicative update
        logf = -eta * mom
        if noise_level > 1e-8:
            logf += rng.normal(0.0, noise_level, n)
        logf = np.clip(logf, -2.5, 2.5)
        
        cand = a * np.exp(logf)
        cand = np.maximum(cand, 0.0)
        cand = normalize_seq(cand, target)
        
        if not np.all(np.isfinite(cand)):
            eta = max(0.01, eta * 0.5)
            mom *= 0.5
            continue
        
        c2 = fast_convolution(cand)
        s2 = float(cand.sum())
        cand_score = float(2.0 * n * float(np.max(c2)) / (s2 * s2)) if s2 > 0 else float('inf')
        
        if np.isfinite(cand_score) and cand_score <= score * (1.0 + temperature):
            # Accept
            a = cand
            c = c2
            score = cand_score
            
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                stagnant = 0
                
                # Increase learning rate on improvement
                eta = min(0.8, eta * 1.05)
                
                # Cool down
                temperature = max(0.001, temperature * cooling)
                noise_level = max(0.001, noise_level * 0.9998)
            else:
                stagnant += 1
                eta = max(0.03, eta * 0.97)
        else:
            # Reject
            eta = max(0.01, eta * 0.6)
            mom *= 0.5
            stagnant += 1
        
        # Periodic restart
        if stagnant > 500:
            perturb_strength = 0.2 + 0.3 * min(1.0, stagnant / 1000.0)
            a = smooth_perturb(best, rng, perturb_strength)
            c = fast_convolution(a)
            s = float(a.sum())
            score = float(2.0 * n * float(np.max(c)) / (s * s)) if s > 0 else float('inf')
            if not np.isfinite(score) or score > best_score * 1.1:
                a = best.copy()
                c = fast_convolution(a)
                score = best_score
            eta = 0.2
            temperature = 0.03
            noise_level = 0.03
            mom = np.zeros(n, dtype=np.float64)
            stagnant = 0
        
        # Log progress
        if it % 5000 == 0:
            print(f"    [{label}] it={it}, score={score:.6f}, best={best_score:.6f}, eta={eta:.4f}, T={temperature:.5f}")
    
    return best, best_score

def generate_starts(n, rng, parent_seq):
    """Generate diverse starting sequences."""
    n = int(n)
    starts = []
    
    def add(a):
        try:
            a = normalize_seq(np.asarray(a, dtype=np.float64), float(n))
            sc = fast_score(a)
            if np.isfinite(sc):
                starts.append((sc, a))
        except Exception:
            pass
    
    # Parent upsampled
    if parent_seq is not None:
        add(upsample(parent_seq, n))
    
    # Uniform baseline
    add(np.ones(n))
    
    # Linear ramps (key insight: asymmetry reduces center peak)
    for slope in (-1.5, -1.0, -0.7, -0.5, -0.3, 0.3, 0.5, 0.7, 1.0, 1.5):
        add(make_ramp(n, slope))
    
    # Ramps with curvature
    for slope in (-1.0, -0.5, 0.5, 1.0):
        for curv in (-0.5, -0.2, 0.2, 0.5):
            add(make_ramp(n, slope, curv))
    
    # Two-block
    for frac in (0.15, 0.25, 0.35, 0.45):
        for ratio in (0.4, 0.7, 1.3, 2.0, 3.5):
            add(make_two_block(n, frac, ratio))
    
    # Cosine modulated
    for freq in (1, 2, 3):
        for amp in (0.3, 0.6, 1.0):
            for phase in (0, np.pi/2, np.pi):
                add(make_cosine(n, freq, amp, phase))
    
    # Smooth random
    for _ in range(15):
        add(make_smooth_random(n, rng))
    
    if not starts:
        starts.append((2.0, normalize_seq(np.ones(n), float(n))))
    
    starts.sort(key=lambda x: x[0])
    return starts

        print(f"\n=== Level n={level_n}, budget={b:.1f}s ===")
        starts = generate_starts(level_n, rng, prev_best if prev_best is not None else parent_seq)
        
        # Pick top starts and refine
        k = max(1, min(3, len(starts), int(b / 40.0) + 1))
        time_each = b / float(k)
        
        for i in range(k):
            if time.monotonic() >= search_deadline - 100.0:
                break
            sc, sq = starts[i]
            time_this = max(5.0, min(time_each, search_deadline - time.monotonic() - 100.0))
            try:
                seq, score = local_search(sq, time_this, seed=int(seed + 1000 + idx * 10 + i), label=f"L{level_n}_{i}")
                if np.isfinite(score) and score < global_best_score:
                    global_best_score = score
                    global_best_seq = seq
                    print(f"  New global best: {score:.6f}")
            except Exception as e:
                print(f"  Error: {e}")
                continue
        
        if global_best_seq is not None:
            prev_best = global_best_seq
        elif parent_seq is not None:
            prev_best = upsample(parent_seq, level_n)
    
    # Final length optimization
    final_remain = max(0.0, search_deadline - time.monotonic())
    print(f"\n=== Final level n={final_n}, remaining={final_remain:.1f}s ===")
    
    if final_remain > 30.0:
        parent_for_final = global_best_seq if global_best_seq is not None else parent_seq
        starts = generate_starts(final_n, rng, parent_for_final)
        
        # Quick evaluation of all starts
        for sc, sq in starts:
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = sq.copy()
        
        # Refine top starts
        k = max(2, min(5, len(starts), int(final_remain / 120.0) + 1))
        time_each = final_remain * 0.7 / float(k)
        
        for i in range(k):
            if time.monotonic() >= search_deadline - 60.0:
                break
            sc, sq = starts[i]
            time_this = max(10.0, min(time_each, search_deadline - time.monotonic() - 60.0))
            if time_this < 10.0:
                break
            try:
                # Alternate between two search methods for diversity
                if i % 2 == 0:
                    seq, score = local_search(sq, time_this, seed=int(seed + 5000 + i), label=f"F{i}")
                else:
                    seq, score = run_local_search_v2(sq, time_this, seed=int(seed + 6000 + i), label=f"F{i}")
                if np.isfinite(score) and score < global_best_score:
                    global_best_score = score
                    global_best_seq = seq
                    print(f"  New global best: {score:.6f}")
            except Exception as e:
                print(f"  Error: {e}")
                continue
    
    # Final long refinement on global best
    final_remain = max(0.0, search_deadline - time.monotonic())
    if final_remain > 30.0 and global_best_seq is not None:
        print(f"\n=== Final refinement, remaining={final_remain:.1f}s ===")
        try:
            seq, score = local_search(global_best_seq, final_remain, seed=int(seed + 9999), label="FINAL")
            if np.isfinite(score) and score < global_best_score:
                global_best_score = score
                global_best_seq = seq
                print(f"  New global best: {score:.6f}")
        except Exception as e:
            print(f"  Error: {e}")
    
    # Peak shaving post-processing
    if global_best_seq is not None and (search_deadline - time.monotonic()) > 10.0:
        print("\n=== Peak shaving ===")
        shaved = peak_shave(global_best_seq, fast_convolution(global_best_seq), 
                           n_shaves=100, strength=0.05)
        shaved_score = fast_score(shaved)
        if np.isfinite(shaved_score) and shaved_score < global_best_score:
            global_best_score = shaved_score
            global_best_seq = shaved
            print(f"  Shaved score: {shaved_score:.6f}")
    
    # Fallback
    if global_best_seq is None:
        global_best_seq = normalize_seq(np.ones(final_n), float(final_n))
        global_best_score = fast_score(global_best_seq)
    
    # Exact validation
    print(f"\n=== Exact validation ===")
    exact_seq = global_best_seq
    exact_score = float(global_best_score)
    
    try:
        ex = float(evaluate_sequence([float(x) for x in global_best_seq]))
        print(f"  Exact score: {ex:.6f} (fast was {exact_score:.6f})")
        if np.isfinite(ex):
            exact_score = ex
            exact_seq = global_best_seq
    except Exception as e:
        print(f"  Validation error: {e}")
    
    # Try a few other candidates too
    if global_best_seq is not None:
        # Try with different normalization
        for target_mult in [0.8, 1.2]:
            alt = normalize_seq(global_best_seq, float(len(global_best_seq)) * target_mult)
            try:
                ex = float(evaluate_sequence([float(x) for x in alt]))
                if np.isfinite(ex) and ex < exact_score:
                    exact_score = ex
                    exact_seq = alt
                    print(f"  Better with mult {target_mult}: {ex:.6f}")
            except Exception:
                pass
    
    # Sanitize and return
    seq = np.asarray(exact_seq, dtype=np.float64)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 1024
    
    final = [float(x) for x in seq]
    print(f"\nFinal answer: length={len(final)}, score={exact_score:.6f}")
    print(f"Total time: {time.monotonic() - t0:.1f}s")
    return final
```
