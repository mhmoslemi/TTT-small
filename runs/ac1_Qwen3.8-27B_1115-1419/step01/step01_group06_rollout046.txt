import time

def fft_convolve(a, N):
    """Linear convolution of a with itself using FFT."""
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2*len(a)-1]

def fft_convolve_ab(a, b, N):
    """Linear convolution of a and b using FFT."""
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)
    return c[:len(a)+len(b)-1]

def softmax(x, beta):
    """Compute exp(beta*x) / sum(exp(beta*x)) with shift for stability."""
    shifted = beta * (x - np.max(x))
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(x, 1.0/len(x))
    return e / s

def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, float).copy()
    n = len(v)
    if n == 0:
        return v
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

Optimize non-negative sequences to minimize peak autocorrelation using

Uses probability simplex parameterization and FFT-based gradient computation.

import time

def next_fast_len(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def fft_self_conv(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2*n-1]
    return np.maximum(c, 0)

def fft_cross_conv(a, b, N):
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)
    out_len = len(a) + len(b) - 1
    return c[:out_len]

def smooth_max_weights(conv, beta):
    """Compute softmax weights for smoothed max."""
    shifted = beta * (conv - np.max(conv))
    shifted = np.clip(shifted, -700, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(conv, 1.0 / len(conv))
    return e / s

def grad_self_conv(a, conv, N):
    """Gradient of smooth_max(conv(a,a)) w.r.t. a."""
    n = len(a)
    w = smooth_max_weights(conv, 1.0)  # weights already normalized
    # grad = 2 * convolve(w, a)[0:n]
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g = np.fft.irfft(W * A, N)
    return 2.0 * g[:n]

    F = np.fft.rfft(f, N)

def project_simplex(v):
    v = np.asarray(v, float).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    return 2.0 * n * mx

    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    mx = np.max(conv)
    return 2.0 * n * mx / (s*s)

    """Create a diverse initial sequence."""
    choice = rng.integers(5)

    elif choice == 1:

        a = np.maximum(a, 1e-10)
        a /= a.sum()
    elif choice == 2:

        a = np.exp(-0.5 * ((np.arange(n) - center1) / sigma)**2) + np.exp(-0.5 * ((np.arange(n) - center2) / sigma)**2)
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    else:

def run_optimize(a, beta_start, beta_end, max_iters, N, rng):
    """Run smoothed gradient descent with Nesterov acceleration."""
    n = len(a)
    a = project_simplex(a.copy())
    momentum = np.zeros(n)
    mu = 0.9  # Nesterov momentum
    eta = 1.0 / (beta_start * 2 * np.max(a**2).__float__() + 1e-10)
    # Actually, let's use a simpler step size
    eta = 0.5 / np.sqrt(n)  # heuristic
    
    best_a = a.copy()
    best_score = score_normalized(a)
    current_beta = beta_start
    
    for it in range(max_iters):
        # Anneal beta
        progress = it / max(max_iters - 1, 1)
        current_beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute convolution and gradient
        conv = fft_self_conv(a, N)
        mx = np.max(conv)
        score = 2.0 * n * mx
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Gradient of smooth max
        w = smooth_max_weights(conv, current_beta)
        W = np.fft.rfft(w, N)
        A = np.fft.rfft(a, N)
        g = np.fft.irfft(W * A, N)[:n]
        grad = 2.0 * g
        
        # Nesterov step
        momentum = mu * momentum + (1 - mu) * grad
        a_new = a - eta * momentum
        a_new = project_simplex(a_new)
        
        # Check improvement
        conv_new = fft_self_conv(a_new, N)
        score_new = 2.0 * n * np.max(conv_new)
        
        if not np.isfinite(score_new) or score_new > score * 1.01:

            a = a  # keep old
            eta *= 0.7
        else:
            a = a_new
            if score_new < score:
                eta = min(eta * 1.05, 1.0)
        
        # Reset momentum if we reverted too much
        if eta < 1e-6:
            eta = 0.5 / np.sqrt(n)
            momentum = np.zeros(n)
    
    return best_a, best_score

"""
Optimize non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2
using L-BFGS-B in log-space with annealed smooth-max objective,
cross-optimization for escaping local minima, and multi-restart strategy.
"""

import numpy as np
import time
from scipy.optimize import minimize

_EPS = 1e-300


def next_fast_len(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def softmax_stable(w):
    w = np.asarray(w, dtype=np.float64)
    w = w - np.max(w)
    e = np.exp(w)
    s = e.sum()
    if s < _EPS:
        return np.full(w.shape, 1.0 / len(w))
    return e / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_self_conv(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2 * n - 1]
    return np.maximum(c, 0.0)


def fft_cross_conv(a, b, N):
    out_len = len(a) + len(b) - 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)
    return c[:out_len]


def make_obj_fn(w_init, beta, N):
    n = len(w_init)

    def fun_and_grad(w):
        a = softmax_stable(w)
        conv = fft_self_conv(a, N)
        shifted = beta * conv
        M = np.max(shifted)
        shifted2 = shifted - M
        shifted2 = np.clip(shifted2, -700.0, 0.0)
        e = np.exp(shifted2)
        s = e.sum()
        if s < _EPS:
            val = 2.0 * n * M / beta
            w_softmax = np.full_like(conv, 1.0 / len(conv))
        else:
            val = 2.0 * n * (M + np.log(s)) / beta
            w_softmax = e / s
        # Gradient in a-space: g_j = 2 * sum_k w_k * a_{k-j}
        W = np.fft.rfft(w_softmax, N)
        A = np.fft.rfft(a, N)
        g_a = np.fft.irfft(W * A, N)[:n] * 2.0
        # Chain rule to w-space
        grad_w = a * (g_a - np.dot(a, g_a))
        if not np.all(np.isfinite(grad_w)):
            grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
        return val, grad_w

    return fun_and_grad


def make_cross_obj_fn(f, beta, N):
    n = len(f)
    F = np.fft.rfft(f, N)

    def fun_and_grad(w):
        b = softmax_stable(w)
        conv_fb = fft_cross_conv(f, b, N)
        conv_fb = np.maximum(conv_fb, 0.0)
        shifted = beta * conv_fb
        M = np.max(shifted)
        shifted2 = shifted - M
        shifted2 = np.clip(shifted2, -700.0, 0.0)
        e = np.exp(shifted2)
        s = e.sum()
        if s < _EPS:
            val = M / beta
            w_softmax = np.full_like(conv_fb, 1.0 / len(conv_fb))
        else:
            val = (M + np.log(s)) / beta
            w_softmax = e / s
        # Gradient w.r.t. b: g_b_j = sum_k w_k * f_{k-j} (no factor 2)
        W = np.fft.rfft(w_softmax, N)
        g_b = np.fft.irfft(W * F, N)[:n]
        grad_w = b * (g_b - np.dot(b, g_b))
        if not np.all(np.isfinite(grad_w)):
            grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
        return val, grad_w

    return fun_and_grad


def load_parent():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 16:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    arr = arr / s
    if arr.size > 10000:
        target = 8192
        blocks = max(1, arr.size // target)
        use = min(arr.size, target * blocks)
        arr = np.mean(arr[:use].reshape(target, blocks), axis=1)
        s = float(arr.sum())
        if s > 0:
            arr = arr / s
    return arr[:10000]


def exact_score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    mx = float(np.max(conv))
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx / (s * s))


def fft_score(a, N):
    n = len(a)
    conv = fft_self_conv(a, N)
    mx = float(np.max(conv))
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * mx


def make_random_init(n, rng):
    choice = rng.integers(6)
    if choice == 0:
        a = np.ones(n) / n
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2.0) ** 2
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    elif choice == 2:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif choice == 3:
        c1 = n // 4
        c2 = 3 * n // 4
        sig = max(n // 15, 3)
        a = np.exp(-0.5 * ((np.arange(n) - c1) / sig) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / sig) ** 2)
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 4.0 * x * (1.0 - x)
        a = np.maximum(a, 1e-10)
        a /= a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 2.0)
    return a


def run_lbfgsb(w, beta, N, max_iter=200, max_time=30.0):
    t0 = time.monotonic()
    fun_and_grad = make_obj_fn(w, beta, N)
    res = minimize(fun_and_grad, w, method='L-BFGS-B',
                   jac=True, options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                                       'ftol': 1e-10, 'gtol': 1e-8})
    elapsed = time.monotonic() - t0
    a = softmax_stable(res.x)
    return a, res, elapsed


def run_cross_optimize(f, w_b, beta, N, max_iter=100):
    t0 = time.monotonic()
    fun_and_grad = make_cross_obj_fn(f, beta, N)
    res = minimize(fun_and_grad, w_b, method='L-BFGS-B',
                   jac=True, options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                                       'ftol': 1e-10, 'gtol': 1e-8})
    elapsed = time.monotonic() - t0
    b = softmax_stable(res.x)
    return b, res, elapsed


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    budget = min(max(0.0, budget), 2390.0)

    t_start = time.monotonic()
    deadline = t_start + budget - 15.0

    rng = np.random.default_rng(seed)

    # Load parent
    parent = load_parent()
    if parent is not None:
        n = parent.size
        if n < 64:
            n = 2048
            parent = np.ones(n) / n
        if n > 10000:
            parent = parent[:10000]
            n = parent.size
            parent = parent / parent.sum()
    else:
        n = 4096
        parent = np.ones(n) / n

    N = next_fast_len(3 * n - 2)

    # Initial score
    best_a = parent.copy()
    best_fft_score = fft_score(best_a, N)
    best_exact = float('inf')
    best_exact_a = best_a.copy()

    # Try exact on parent
    t0 = time.monotonic()
    sc = exact_score(best_a)
    exact_time = time.monotonic() - t0
    if np.isfinite(sc):
        best_exact = sc
        print(f"Parent exact score: {sc:.6f} (n={n}, time={exact_time:.2f}s)")

    # Phase 1: L-BFGS-B from various starting points with increasing beta
    betas_phase1 = [8.0, 15.0, 30.0, 60.0, 120.0, 300.0, 800.0, 2000.0, 5000.0]
    rng_perturb = [0.0, 0.05, 0.1, 0.2]

    phase_budget = budget * 0.7
    per_beta_budget = phase_budget / len(betas_phase1)

    for bi, beta in enumerate(betas_phase1):
        beta_deadline = t_start + (bi + 1) * per_beta_budget
        if time.monotonic() > beta_deadline - 5.0:
            break

        n_starts = 3 if beta < 50 else 4

        for s in range(n_starts):
            if time.monotonic() > beta_deadline:
                break

            # Starting point
            if s == 0:
                w = np.log(best_a + _EPS)
                w -= np.max(w)
            else:
                sig = rng_perturb[min(s - 1, len(rng_perturb) - 1)]
                w = np.log(best_a + _EPS) + rng.normal(0, sig, size=n)
                w -= np.max(w)

            max_iter = 150 if beta < 100 else 100
            a_new, res, elapsed = run_lbfgsb(w, beta, N, max_iter=max_iter,
                                              max_time=per_beta_budget / n_starts)

            if not np.all(np.isfinite(a_new)):
                continue
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

            sc_new = fft_score(a_new, N)
            if np.isfinite(sc_new) and sc_new < best_fft_score - 1e-10:
                best_fft_score = sc_new
                best_a = a_new.copy()
                print(f"  beta={beta:.0f} start={s}: FFT score {sc_new:.6f} (iter={res.nit}, time={elapsed:.1f}s)")

            # Periodic exact check
            if bi % 2 == 0 and s == 0 and time.monotonic() < deadline - 30:
                t0 = time.monotonic()
                sc_ex = exact_score(best_a)
                dt = time.monotonic() - t0
                if np.isfinite(sc_ex) and sc_ex < best_exact:
                    best_exact = sc_ex
                    best_exact_a = best_a.copy()
                    print(f"  Exact score: {best_exact:.6f} (time={dt:.1f}s)")

    # Phase 2: Cross-optimization to escape local minima
    print(f"\nPhase 2: Cross-optimization. Best FFT={best_fft_score:.6f}")
    phase2_deadline = t_start + budget * 0.9

    cross_beta = 100.0
    cross_iter = 0
    while time.monotonic() < phase2_deadline:
        cross_iter += 1

        # Cross-optimize: fix best_a, find b minimizing smooth_max(conv(best_a, b))
        w_b = np.log(best_a + _EPS)
        w_b -= np.max(w_b)
        b, res_c, elapsed_c = run_cross_optimize(best_a, w_b, cross_beta, N, max_iter=80)

        if not np.all(np.isfinite(b)):
            break

        # Try mixing
        best_mix_score = best_fft_score
        best_mix = best_a.copy()
        for t_mix in (0.3, 0.5, 0.7, 1.0):
            mix = (1.0 - t_mix) * best_a + t_mix * b
            mix = np.maximum(mix, 0.0)
            ms = mix.sum()
            if ms < _EPS:
                continue
            mix = mix / ms
            sc_mix = fft_score(mix, N)
            if np.isfinite(sc_mix) and sc_mix < best_mix_score:
                best_mix_score = sc_mix
                best_mix = mix.copy()
                best_t = t_mix

        if best_mix_score < best_fft_score:
            best_a = best_mix.copy()
            best_fft_score = best_mix_score
            print(f"  Cross iter {cross_iter}: improved to {best_fft_score:.6f} (t={best_t})")
        else:
            # Try small perturbation for next round
            if cross_iter % 5 == 0:
                w = np.log(best_a + _EPS) + rng.normal(0, 0.05, size=n)
                a_pert = softmax_stable(w)
                sc_pert = fft_score(a_pert, N)
                if np.isfinite(sc_pert) and sc_pert < best_fft_score:
                    best_a = a_pert
                    best_fft_score = sc_pert

        # Slowly increase cross beta
        cross_beta = min(cross_beta * 1.3, 2000.0)

        # Exact check every few iterations
        if cross_iter % 4 == 0 and time.monotonic() < deadline - 30:
            t0 = time.monotonic()
            sc_ex = exact_score(best_a)
            dt = time.monotonic() - t0
            if np.isfinite(sc_ex) and sc_ex < best_exact:
                best_exact = sc_ex
                best_exact_a = best_a.copy()
                print(f"  Cross exact: {best_exact:.6f} (time={dt:.1f}s)")

    # Phase 3: Final fine-tuning with high beta
    print(f"\nPhase 3: Fine-tuning. Best FFT={best_fft_score:.6f}")
    final_deadline = deadline

    for beta_final in [2000.0, 5000.0, 10000.0]:
        if time.monotonic() > final_deadline - 10:
            break
        w = np.log(best_a + _EPS)
        w -= np.max(w)
        a_new, res, elapsed = run_lbfgsb(w, beta_final, N, max_iter=80, max_time=30.0)
        if np.all(np.isfinite(a_new)):
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            sc_new = fft_score(a_new, N)
            if np.isfinite(sc_new) and sc_new < best_fft_score:
                best_a = a_new
                best_fft_score = sc_new
                print(f"  Fine-tune beta={beta_final:.0f}: {sc_new:.6f}")

    # Final exact evaluation
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        sc_final = exact_score(best_a)
        dt = time.monotonic() - t0
        if np.isfinite(sc_final) and sc_final < best_exact:
            best_exact = sc_final
            best_exact_a = best_a.copy()
        print(f"\nFinal exact score: {best_exact:.6f} (n={n}, check_time={dt:.1f}s)")

    # Also try a few random restarts if time permits
    if time.monotonic() < deadline - 20:
        for _ in range(5):
            if time.monotonic() > deadline - 10:
                break
            a_r = make_random_init(n, rng)
            sc_r = fft_score(a_r, N)
            if np.isfinite(sc_r) and sc_r < best_fft_score:
                best_a = a_r
                best_fft_score = sc_r

    # Use the best exact-score sequence if we have one, else the FFT-best
    result = best_exact_a if np.isfinite(best_exact) else best_a

    # Finalize: scale to sum sqrt(2n)
    result = np.asarray(result, dtype=np.float64).ravel()
    result = np.where(np.isfinite(result), result, 0.0)
    result = np.maximum(result, 0.0)
    nn = result.size
    s = float(result.sum())
    scale = float(np.sqrt(2.0 * nn))
    if s > _EPS:
        result = result * scale / s
    result = np.clip(result, 0.0, 1000.0)
    s2 = float(result.sum())
    if s2 < 0.01:
        result = result * (0.01 / max(s2, _EPS))
        result = np.clip(result, 0.0, 1000.0)

    return [float(x) for x in result]
