"""
Search for non-negative coefficient sequences minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-based gradient descent with adaptive temperature, backtracking line search,
targeted peak reduction, and multi-phase optimization strategy.
"""

import numpy as np
import time

def _fft_score(a, N):
    """Compute score using FFT. a is normalized to sum=1."""
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2*n-1]
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float('inf'), c
    return float(2.0 * n * mx), c

def _exact_score(a):
    """Exact score using numpy.convolve."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return float('inf')
    b = np.convolve(a, a)
    mx = float(np.max(b))
    val = 2.0 * n * mx / (s * s)
    return val if np.isfinite(val) else float('inf')

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def _softmax_weights(vals, beta):
    """Compute softmax weights for lag selection."""
    vals = np.asarray(vals, dtype=np.float64)
    if len(vals) == 0:
        return np.array([1.0])
    v = np.where(np.isfinite(vals), vals, 0.0)
    v = v - np.max(v)
    y = np.clip(beta * v, -700.0, 0.0)
    e = np.exp(y)
    s = float(np.sum(e))
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros(len(vals))
        p[np.argmax(vals)] = 1.0
        return p
    return e / s

def _gradient_for_lags(a, c, p, n, N, phase):
    """Compute gradient of weighted sum of c_k w.r.t. a, using FFT."""
    # p is the weight vector over lags
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # grad_i = 2 * sum_k p_k * a_{k-i} (circular)
    # In FFT: ifft(P * conj(A)) gives sum_k p_k * a_{k-i}
    conv = np.fft.irfft(P * np.conjugate(A), N)
    grad = 2.0 * conv[:n]  # take first n elements
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad

def _projected_adam_step(a, grad, w, m, v, t, lr, mu=0.9):
    """Adam step in log-space with projection to simplex."""
    t += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * (grad * grad)
    mhat = m / (1 - b1**t)
    vhat = v / (1 - b2**t)
    step = mhat / (np.sqrt(vhat) + 1e-8)
    step = np.clip(step, -5.0, 5.0)
    step = np.where(np.isfinite(step), step, 0.0)
    
    w_new = w - lr * step
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= np.max(w_new)
    e = np.exp(w_new)
    s = float(np.sum(e))
    if s < 1e-300 or not np.isfinite(s):
        a_new = np.ones(len(a)) / len(a)
        w_new = np.zeros(len(a))
    else:
        a_new = e / s
    return a_new, w_new, m, v, t

def _line_search(a, grad_w, w, m, v, t, lr, N, phase, max_iter=20):
    """Backtracking line search to ensure score decreases."""
    n = len(a)
    base_score, base_c = _fft_score(a, N)
    
    cur_lr = lr
    for _ in range(max_iter):
        a_new, w_new, m_new, v_new, t_new = _projected_adam_step(a, grad_w, w, m, v, t, cur_lr)
        sc, _ = _fft_score(a_new, N)
        if np.isfinite(sc) and sc < base_score:
            return a_new, w_new, m_new, v_new, t_new, sc, cur_lr

    a_new, w_new, m_new, v_new, t_new = _projected_adam_step(a, grad_w, w, m, v, t, lr)
    sc, _ = _fft_score(a_new, N)

def _structured_perturb(a, n_peaks, c, rng):
    """Perturb a in directions that target peak reduction."""
    n = len(a)
    score, c = _fft_score(a, 1)  # just for c
    # Use the provided c
    top_idx = np.argsort(c)[::-1][:n_peaks]
    # Create a perturbation that reduces overlap at peak lags
    pert = np.zeros(n)
    for k in top_idx:
        # Gradient of c_k w.r.t. a_i is a_{k-i} + a_{i-k}
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                pert[i] -= a[j1]
            if 0 <= j2 < n:
                pert[i] -= a[j2]
    # Subtract mean to preserve sum
    pert -= np.mean(pert)
    # Normalize perturbation
    pn = np.linalg.norm(pert)
    if pn > 1e-10:
        pert = pert / pn
    else:
        pert = rng.normal(0, 1, n)
        pert -= np.mean(pert)
    
    # Apply perturbation in log-space
    w = np.log(np.maximum(a, 1e-300))
    sig = float(rng.choice([0.05, 0.1, 0.2, 0.3]))
    w += sig * pert
    w += rng.normal(0, 0.05, n)
    w -= np.max(w)
    e = np.exp(w)
    s = float(np.sum(e))
    if s < 1e-300:
        return a.copy()
    return e / s

def _run_optimization(a_init, N, budget, seed, verbose=True):
    """Main gradient-based optimization loop."""
    start = time.monotonic()
    deadline = start + budget
    n = len(a_init)
    a = _normalize(a_init)
    score, c = _fft_score(a, N)
    best_a = a.copy()
    best_score = score
    
    w = np.log(np.maximum(a, 1e-300))
    w -= np.max(w)
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_state = 0
    
    beta = 3.0
    lr = 0.1
    stall = 0
    outer = 0
    
    while time.monotonic() < deadline:
        outer += 1
        remaining = deadline - time.monotonic()
        
        # Adaptive beta: increase over time to sharpen focus on peaks
        progress = 1.0 - remaining / budget
        beta = 3.0 + 50.0 * progress * progress
        if progress > 0.8:
            beta = 53.0 + (progress - 0.8) * 200.0
        
        # Compute weights
        p = _softmax_weights(c, beta)
        
        # Compute gradient
        grad = _gradient_for_lags(a, c, p, n, N, None)
        grad_w = a * (grad - np.dot(a, grad))
        
        # Line search step
        a_new, w, m_state, v_state, t_state, sc_new, lr_used = _line_search(
            a, grad_w, w, m_state, v_state, t_state, lr, N
        )
        
        if sc_new < score:
            a = a_new
            score = sc_new
            _, c = _fft_score(a, N)
            stall = 0
            lr = min(0.3, lr * 1.02)
        else:
            stall += 1
            lr = max(0.01, lr * 0.85)
            # Refresh c
            _, c = _fft_score(a, N)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Periodic refresh of exact c
        if outer % 50 == 0:
            score, c = _fft_score(a, N)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # Restart if stalled
        if stall > 30:
            if np.random.default_rng(outer).random() < 0.5:
                a = best_a.copy()
                score = best_score
            else:
                a = _structured_perturb(best_a, 5, c, np.random.default_rng(outer))
                score, c = _fft_score(a, N)
                if not np.isfinite(score):
                    a = best_a.copy()
                    score = best_score
            w = np.log(np.maximum(a, 1e-300))
            w -= np.max(w)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_state = 0
            lr = 0.1
            stall = 0
    
    return best_a, best_score

def _finalize(a):
    """Convert optimized sequence to valid output format."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-300:
        a = np.ones(64)
        n = 64
        s = 64.0
    
    # The evaluator computes 2*n*max_b/sum_a^2 which is scale invariant.
    # We just need sum >= 0.01 and all values in [0, 1000].
    scale = np.sqrt(2.0 * n) / s
    a = a * scale
    a = np.clip(a, 0.0, 1000.0)
    
    s2 = float(np.sum(a))
    if s2 < 0.01:
        a = a * (1.0 / max(s2, 1e-300))
        a = np.clip(a, 0.0, 1000.0)
    
    return [float(x) for x in a]

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point. Runs multi-phase optimization to find sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    
    Strategy:
    1. Load and normalize the provided starting sequence.
    2. Run gradient-based optimization with adaptive temperature and line search.
    3. Do targeted peak reduction to polish the result.
    4. Try alternative initializations if time remains.
    """
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 2400.0
    budget = min(max(budget, 10.0), 2395.0)
    
    start_all = time.monotonic()
    deadline_all = start_all + budget - 5.0  # reserve 5s for finalization
    
    # Load initial sequence
    a_init = None
    hs1 = globals().get('height_sequence_1', None)
    if hs1 is not None:
        try:
            arr = np.asarray(hs1, dtype=np.float64).ravel()
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            if np.sum(arr) > 0.01 and len(arr) >= 16:
                a_init = arr
        except Exception:
            pass
    
    if a_init is None:
        a_init = np.ones(4096, dtype=np.float64)
    
    # Trim to reasonable size
    MAX_N = 10000
    if len(a_init) > MAX_N:
        # Downsample by averaging blocks
        target = 8192
        blocks = len(a_init) // target
        if blocks > 1:
            use = target * blocks
            a_init = np.mean(a_init[:use].reshape(target, blocks), axis=1)
        else:
            a_init = a_init[:target]
    
    n = len(a_init)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    N = max(N, 2 * (2 * n - 1))
    
    if n < 16:
        a_init = np.ones(4096, dtype=np.float64)
        n = 4096
        N = 1
        while N < 3 * n - 2:
            N <<= 1
    
    a_init = _normalize(a_init)
    score0, c0 = _fft_score(a_init, N)
    print(f"Initial: n={n}, N={N}, score={score0:.6f}")
    
    rng = np.random.default_rng(seed)
    best_a = a_init.copy()
    best_score = score0
    
    # Phase 1: Main optimization (70% of budget)
    phase1_budget = (deadline_all - start_all) * 0.65
    a1, s1 = _run_optimization(a_init, N, phase1_budget, seed, verbose=True)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
    print(f"After Phase 1: score={best_score:.6f}")
    
    elapsed = time.monotonic() - start_all
    remaining = deadline_all - time.monotonic()
    
    # Phase 2: Targeted peak reduction (20% of budget)
    if remaining > 30:
        phase2_budget = min(remaining * 0.6, (deadline_all - start_all) * 0.2)
        a2, s2 = _targeted_peak_reduction(best_a, N, n_lags=8, n_steps=80)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
            print(f"After Phase 2 (peak reduction): score={best_score:.6f}")
    
    remaining = deadline_all - time.monotonic()
    
    # Phase 3: Try alternative starting points if time allows (15% of budget)
    if remaining > 60:
        phase3_budget = remaining * 0.5
        
        # Try a few alternative perturbations of the best
        alts = []
        for i in range(3):
            sig = float(rng.choice([0.1, 0.2, 0.3]))
            w_alt = np.log(np.maximum(best_a, 1e-300))
            w_alt += rng.normal(0, sig, n)
            w_alt -= np.max(w_alt)
            e = np.exp(w_alt)
            s_e = float(np.sum(e))
            if s_e > 1e-300:
                alts.append(e / s_e)
        
        for i, alt in enumerate(alts):
            if time.monotonic() >= deadline_all:
                break
            alt_budget = phase3_budget / (len(alts) - i)
            sc_init, _ = _fft_score(alt, N)
            if sc_init < best_score:
                a_alt, s_alt = _run_optimization(alt, N, alt_budget, seed + 100 + i, verbose=False)
                if s_alt < best_score:
                    best_score = s_alt
                    best_a = a_alt.copy()
                    print(f"  Alt {i}: score={best_score:.6f}")
            else:
                # Still try to optimize it
                a_alt, s_alt = _run_optimization(alt, N, alt_budget, seed + 100 + i, verbose=False)
                if s_alt < best_score:
                    best_score = s_alt
                    best_a = a_alt.copy()
                    print(f"  Alt {i}: score={best_score:.6f}")
    
    remaining = deadline_all - time.monotonic()
    
    # Phase 4: Final polishing - targeted peak reduction with more iterations
    if remaining > 20:
        a_final, s_final = _targeted_peak_reduction(best_a, N, n_lags=5, n_steps=40)
        if s_final < best_score:
            best_score = s_final
            best_a = a_final.copy()
            print(f"Final polish: score={best_score:.6f}")
    
    # Exact verification
    if len(best_a) <= 5000:
        exact_sc = _exact_score(best_a)
        print(f"Exact verification: score={exact_sc:.6f} (FFT was {best_score:.6f})")
    
    print(f"Total time: {time.monotonic() - start_all:.1f}s")
    print(f"Best FFT score: {best_score:.6f}")
    
    return _finalize(best_a)
