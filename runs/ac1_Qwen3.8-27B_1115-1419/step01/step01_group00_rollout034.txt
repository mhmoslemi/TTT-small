"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses momentum gradient descent with adaptive smoothing, multi-lag gradients,
and structured perturbations. Improves on previous approach with Nesterov-style
momentum, better step-size adaptation, and multi-lag active set handling.
"""

import numpy as np
import time


def _project(a, total):
    """Project onto {x >= 0, sum(x) = total}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full_like(a, total / len(a))
    return a * (total / s)


def _conv(a):
    """Self-convolution."""
    return np.convolve(a, a)


def _objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    c = _conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def _grad_exact(a, k):
    """Exact gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def _grad_multi(a, indices, weights):
    """Weighted sum of gradients at multiple lags."""
    n = len(a)
    g = np.zeros(n)
    for k, w in zip(indices, weights):
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] += w * 2.0 * a[k - idx]
    return g


def _smooth_grad(a, tau):
    """
    Gradient of tau * log(sum(exp(c/tau))) where c = conv(a,a).
    Returns (smooth_max, gradient).
    """
    c = _conv(a)
    mx = float(c.max())
    ex = np.exp((c - mx) / tau)
    s = float(ex.sum())
    p = ex / s
    n = len(a)
    g = 2.0 * np.correlate(p, a, 'full')[:n]
    smooth_val = mx + tau * np.log(s)
    return smooth_val, g

                v *= 0.4
                stall = 0
