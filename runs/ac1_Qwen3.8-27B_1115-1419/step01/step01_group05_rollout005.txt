"""
Optimized search for low-peak nonnegative self-convolution sequences.
Uses mirror descent with log-sum-exp smoothing, multi-scale resampling,
and peak-directed refinement on the probability simplex.
"""

import numpy as np
import time

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _fft_convolve(a, b, fft_size=None):
    n = max(len(a), len(b))
    if fft_size is None:
        fft_size = _next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:len(a)+len(b)-1]

def _autocorr(a, fft_size=None):
    n = a.size
    if fft_size is None:
        fft_size = _next_pow2(2*n - 1)
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2*n-1]

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(a.size, 1)
    return a / s

def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n < 1:
        return 1e100
    fs = _next_pow2(2*n-1)
    b = _autocorr(a, fs)
    mb = np.max(b)
    if not np.isfinite(mb):
        return 1e100
    return float(2.0 * n * mb)

def _lse(b, beta):
    m = np.max(b)
    if beta <= 1:
        return m
    arg = beta * (b - m)
    arg = np.clip(arg, -700, 700)
    e = np.exp(arg)
    s = e.sum()
    if s <= 0:
        return m
    return m + np.log(s) / beta

def _lse_grad(b, beta, a, fft_size):
    """Gradient of LSE(beta*b) w.r.t. a, where b = conv(a,a)."""
    m = np.max(b)
    if beta <= 1:
        # Just use max
        idx = np.argmax(b)
        p = np.zeros_like(b)
        p[idx] = 1.0
    else:
        arg = beta * (b - m)
        arg = np.clip(arg, -700, 700)
        e = np.exp(arg)
        p = e / e.sum()
    
    # grad_i = 2 * sum_k p_k * a_{k-i}
    # = 2 * (a * rev(p))_{n-1} convolved... 
    # Actually: sum_k p_k * a_{k-i} where a_j = 0 for j outside [0,n-1]
    # This is a cross-correlation of a with p
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size).real
    h = conv[a.size-1 : 2*a.size-1]
    h = np.maximum(h, 0)
    return 2.0 * h

def _mirror_update(a, grad, eta):
    """Multiplicative update on the simplex."""
    g = grad - np.mean(grad)
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 1e-15)
    return _normalize(a_new)

def _euclidean_update(a, grad, eta):
    """Standard projected gradient."""
    g = grad - np.mean(grad)
    v = a - eta * g
    v = np.maximum(v, 0)
    return _normalize(v)

def _make_exponential(n, alpha, two_sided=False):
    n = max(int(n), 2)
    x = np.arange(n, dtype=np.float64)
    if two_sided:
        c = (n-1)/2.0
        a = np.exp(-alpha * np.abs(x - c) / (n-1))
    else:
        a = np.exp(-alpha * x / (n-1))
    return _normalize(a)

def _make_cosine(n, power):
    n = max(int(n), 2)
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * x / (n-1))
    a = np.maximum(a, 0) ** max(power, 0.1)
    a = a + 1e-10
    return _normalize(a)

def _make_gaussian(n, width, center=0.5):
    n = max(int(n), 2)
    x = np.arange(n, dtype=np.float64) / (n-1)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    return _normalize(a)

def _make_flat_with_edges(n):
    n = max(int(n), 2)
    x = np.arange(n, dtype=np.float64) / (n-1)
    a = np.ones(n)

    return _normalize(a)

def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0)
    new_n = max(int(new_n), 2)
    s = old.sum()
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0/new_n, dtype=np.float64)
    x_old = np.linspace(0, 1, old.size)
    x_new = np.linspace(0, 1, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

    n = a.size
    z = rng.normal(size=n)
    F = np.fft.rfft(z)

    if cutoff < len(F):
        F[cutoff:] = 0
    f = np.fft.irfft(F, n)
    f = (f - f.mean()) / (np.std(f) + 1e-15)
    a_new = a * (1 + amplitude * f)
    a_new = np.maximum(a_new, 1e-12)
    return _normalize(a_new)

def _block_perturb(a, rng, strength=0.2):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= np.exp(rng.normal(0, strength))
    if rng.random() < 0.2:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)

def _next_pow2(size):

        return 1
    return 1 << (int(size) - 1).bit_length()

def _autocorr(a, fs):
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)

    n = a.size

    return a / s
