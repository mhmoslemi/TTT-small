This is exactly what the softmax approach approximates. With a very high beta, the softmax concentrates on the max lags, giving approximately the subgradient.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for nonnegative a.
Strategy: FFT-accelerated gradient descent with softmax-smoothed max, 
Nesterov-Adam in log-space, alternating convex subproblem, line search,
adaptive beta, and diverse restarts.

import numpy as np
import time

# --- Top-level helper functions ---

def _norm_nonneg(v):
    arr = np.asarray(v, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(max(arr.size, 1), dtype=np.float64) / max(arr.size, 1)
    return arr / s

def _softmax_w(c, beta):
    mx = c.max()
    if not np.isfinite(mx) or mx <= 0:
        return np.ones(c.size, dtype=np.float64) / c.size
    x = (c - mx) / mx
    x = np.clip(x * beta, -745.0, 0.0)
    e = np.exp(x)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        p = np.zeros(c.size, dtype=np.float64)
        p[np.argmax(c)] = 1.0
        return p
    return e / s

def _log_weights(a):
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    return w

def _seq_from_log(w):
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size, dtype=np.float64) / w.size
    return e / s

def _score_fft(a, N, n):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2*n-1]
    c = np.maximum(c, 0.0)
    mx = c.max()
    if not np.isfinite(mx):
        return float('inf'), c
    return 2.0 * n * mx, c

def _grad_self(a, p, N, n, freq_idx):
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    phase = np.exp(-2j * np.pi * freq_idx * (n-1) / N)
    g = np.fft.irfft(P * phase * np.conjugate(A), N)
    return 2.0 * g[n-1:2*n-1].copy()

def _grad_cross(f, b, p, N, n, freq_idx):
    F = np.fft.rfft(f, N)
    P = np.fft.rfft(p, N)
    phase = np.exp(-2j * np.pi * freq_idx * (n-1) / N)
    g = np.fft.irfft(P * phase * np.conjugate(F), N)
    return g[n-1:2*n-1].copy()

def _adam_nesterov(grad_w, w, m, v, t, lr):
    t += 1
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1 - b1) * grad_w
    v = b2 * v + (1 - b2) * (grad_w * grad_w)
    mh = m / (1 - b1**t)
    vh = v / (1 - b2**t)
    dw = mh / (np.sqrt(vh) + eps)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -5.0, 5.0)
    w_new = w - lr * dw
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= w_new.max()
    return w_new, m, v, t

def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = a.sum()
    if s <= 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mx = b.max()
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * mx / (s * s)

def _load_parent():

        if s > 0.01:

        pass
    return None

def _finalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = a.sum()
    scale = np.sqrt(2.0 * n)
    if s > 1e-10:
        a = a * scale / s
    else:
        a = np.ones(n) * scale / n
    a = np.clip(a, 0.0, 1000.0)
    s2 = a.sum()
    if s2 < 0.01:
        a *= 0.02 / max(s2, 1e-300)
        a = np.clip(a, 0.0, 1000.0)
    out = [float(x) for x in a]
    return out
