```python
import numpy as np
import time
from scipy.optimize import minimize, differential_evolution

def fft_convolve(a):
    """Fast FFT-based convolution a*a"""
    n = len(a)
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:L]
    return np.maximum(c, 0)

def score(a):
    """Compute the score 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s*s)

def soft_score(a, tau):
    """Smooth approximation of score using log-sum-exp"""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    c_norm = c / (s*s)
    c_max = np.max(c_norm)
    # Stable log-sum-exp
    shifted = tau * (c_norm - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    soft_max = c_max + log_sum / tau
    return 2.0 * n * soft_max
```

"""
Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

Strategy:
- Parametric search: optimize low-dimensional families (Chebyshev, cosine, Gaussian)
  using scipy's derivative-free optimizers to find good global basins.
- Soft-max smooth optimization: use log-sum-exp approximation of max for smooth
  gradient-based refinement via L-BFGS-B.
- Coordinate descent: fast O(n) per-step fine-tuning using incremental convolution updates.
- Multi-length: try several sequence lengths around the previous best.
- Start from height_sequence_1 if available.
"""

import numpy as np
import time
import math
from scipy.optimize import minimize


def _fft_conv(a):
    """FFT-based linear convolution a*a, returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Fast score: 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _fft_conv(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _soft_score(a, tau):
    """Smooth approximation using log-sum-exp."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _fft_conv(a)
    c_norm = c / (s * s)
    c_max = float(np.max(c_norm))
    shifted = tau * (c_norm - c_max)
    # Stable logsumexp
    max_shifted = float(np.max(shifted))
    log_sum = max_shifted + float(np.log(np.sum(np.exp(shifted - max_shifted))))
    soft_max = c_max + log_sum / tau
    return float(2.0 * n * soft_max)


def _normalize(a, target_sum):
    """Normalize sequence to have the given sum, keeping non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 10000.0)
    s = float(a.sum())
    if s < 1e-15:
        a.fill(target_sum / max(1, a.size))
        return a
    a *= target_sum / s
    a = np.clip(a, 0.0, 1000.0)
    # Re-normalize after clipping
    s = float(a.sum())
    if s < 0.01:
        a.fill(target_sum / max(1, a.size))
    else:
        a *= target_sum / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def _chebyshev_window(N, beta, order):
    """Dolph-Chebyshev-type window: a[n] = max(0, 1 - T_order(beta*cos(theta_n))/T_order(beta))"""
    n = np.arange(N, dtype=np.float64)
    theta = np.pi * (2.0 * (n + 0.5)) / (N + 1.0)
    x = beta * np.cos(theta)
    # Compute T_order(x) using recurrence for stability
    if order == 0:
        T = np.ones(N)
    elif order == 1:
        T = x.copy()
    else:
        T_prev = np.ones(N)
        T_curr = x.copy()
        for _ in range(2, order + 1):
            T_next = 2.0 * x * T_curr - T_prev
            T_prev = T_curr
            T_curr = T_next
        T = T_curr
    T_beta = math.cosh(order * math.acosh(min(beta, 50.0))) if beta > 1.0 else 1.0
    if abs(T_beta) < 1e-10:
        T_beta = 1.0
    w = 1.0 - T / T_beta
    return np.maximum(w, 0.0)


def _cosine_window(N, coeffs):
    """Cosine series window: a[n] = max(0, 1 + sum c_m * cos(2*pi*m*(n+0.5)/N))"""
    n = np.arange(N, dtype=np.float64)
    t = 2.0 * np.pi * (n + 0.5) / N
    w = np.ones(N, dtype=np.float64)
    for m, c in enumerate(coeffs, 1):
        w += c * np.cos(m * t)
    return np.maximum(w, 0.0)


def _gaussian_window(N, params):
    """Gaussian mixture: a[n] = A0 + sum A_i * exp(-(n-x_i)^2/(2*sigma_i^2))"""
    n = np.arange(N, dtype=np.float64)
    A0 = params[0]
    w = np.full(N, A0, dtype=np.float64)
    idx = 1
    while idx < len(params):
        A = params[idx]
        x = params[idx + 1]
        sig = max(0.1, params[idx + 2])
        w += A * np.exp(-(n - x) ** 2 / (2.0 * sig * sig))
        idx += 3
    return np.maximum(w, 0.0)


def _power_window(N, p, q):
    """Power window: a[n] = max(0, (1 - |2*(n+0.5)/N - 1|^p)^q)"""
    n = np.arange(N, dtype=np.float64)
    t = np.abs(2.0 * (n + 0.5) / N - 1.0)
    w = np.maximum(1.0 - t ** p, 0.0) ** q
    return w


def _param_chebyshev(params, N):
    """Objective for Chebyshev parametric optimization."""
    beta = params[0]
    order = int(round(params[1]))
    order = max(1, min(order, 30))
    beta = max(1.0, min(beta, 20.0))
    w = _chebyshev_window(N, beta, order)
    s = float(w.sum())
    if s < 0.01:
        return 1.0
    w = w / s
    return _score(w)


def _param_cosine(params, N):
    """Objective for cosine parametric optimization."""
    coeffs = params
    w = _cosine_window(N, coeffs)
    s = float(w.sum())
    if s < 0.01:
        return 1.0
    w = w / s
    return _score(w)


def _param_gaussian(params, N):
    """Objective for Gaussian mixture parametric optimization."""
    w = _gaussian_window(N, params)
    s = float(w.sum())
    if s < 0.01:
        return 1.0
    w = w / s
    return _score(w)


def _param_power(params, N):
    """Objective for power window parametric optimization."""
    p = max(0.1, params[0])
    q = max(0.1, params[1])
    w = _power_window(N, p, q)
    s = float(w.sum())
    if s < 0.01:
        return 1.0
    w = w / s
    return _score(w)

def _soft_max_local_search(a, budget, tau_start=5.0, tau_end=500.0):

            break
        phase_budget = min(60.0, (t_end - time.monotonic()) / max(1, n_phases))

        def objective(x):
            # x is the log-parameterization: a = exp(x - max(x)) * n / sum(exp(x - max(x)))
            x = np.asarray(x, dtype=np.float64)
            if not np.all(np.isfinite(x)):
                return 1.0
            xm = x - np.max(x)
            ex = np.exp(xm)

            a_try = np.clip(a_try, 0.0, 1000.0)
            s = float(a_try.sum())
            if s < 0.01:
                return 1.0
            c = _fft_conv(a_try)
            c_norm = c / (s * s)
            c_max = float(np.max(c_norm))
            shifted = tau * (c_norm - c_max)
            ms = float(np.max(shifted))
            log_sum = ms + float(np.log(np.sum(np.exp(shifted - ms))))
            soft_max = c_max + log_sum / tau
            return float(2.0 * n * soft_max)

        # Current a -> x
        a_safe = np.maximum(a, 1e-10)
        x0 = np.log(a_safe)

        try:
            res = minimize(objective, x0, method='L-BFGS-B',
                           options={'maxiter': 200, 'gtol': 1e-6, 'maxfun': 500})
            if res.fun < 1.0:  # valid result

                    best_score = sc
                    a = a_new
                    print(f"    Soft-max (tau={tau:.1f}): score={best_score:.6f}")

            pass

    return best, best_score


def _coordinate_descent(a, budget, step_size=0.05):
    """
    Fast coordinate descent with incremental convolution updates.
    Each step: pick index i, try a[i] +/- delta, update c in O(n), check max.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = _normalize(a, float(n))
    t_end = time.monotonic() + budget

    c = _fft_conv(a)
    L = len(c)
    best_score = float(2.0 * n * float(np.max(c)) / (float(a.sum()) ** 2))
    best = a.copy()
    c_best = c.copy()

    rng = np.random.default_rng(0)
    delta = step_size
    stagnant = 0
    it = 0

    while time.monotonic() < t_end - 2:
        it += 1
        # Adaptively decrease step size
        if it % 5000 == 0:
            delta *= 0.95
        if delta < 1e-6:
            break

        # Pick index - bias toward indices that affect the peak lags
        if it % 100 == 0:
            # Recompute peak to guide search
            peak_k = int(np.argmax(c))
            # Indices that affect c[peak_k]: i in [max(0, peak_k-n+1), min(n-1, peak_k)]
            lo = max(0, peak_k - n + 1)
            hi = min(n - 1, peak_k)
            if hi > lo:
                i = int(rng.integers(lo, hi + 1))
            else:
                i = int(rng.integers(0, n))
        else:
            i = int(rng.integers(0, n))

        # Try increasing a[i]
        old_ai = a[i]
        new_ai = old_ai + delta
        if new_ai < 0 or new_ai > 1000:
            continue

        # Update c: for k = i to min(L-1, i+n-1), c[k] += 2*delta*a[k-i]
        k_lo = i
        k_hi = min(L - 1, i + n - 1)
        # How much does the max change?
        # We compute the new max by checking affected c[k]
        old_max = float(np.max(c))

        # Compute new c values for affected range
        affected = c[k_lo:k_hi + 1].copy()
        idx_in_a = np.arange(k_hi - k_lo + 1) + k_lo - i  # = k - i for k in range
        idx_in_a = np.arange(k_lo - i, k_hi + 1 - i)
        # a[k-i] for k in [k_lo, k_hi]
        vals = a[max(0, idx_in_a.min()):min(n, idx_in_a.max() + 1)]
        # Adjust: we need a[k-i] where k-i ranges from k_lo-i to k_hi-i
        ki_lo = k_lo - i
        ki_hi = k_hi - i
        if ki_lo < 0 or ki_hi >= n:
            # Partial range
            mask = (ki_lo >= 0) & (ki_hi < n)
        else:
            a_vals = a[ki_lo:ki_hi + 1]
            affected_new = affected + 2.0 * delta * a_vals
            # Subtract delta from a[n-1-i] to maintain sum (or just check without)
            # For simplicity, just check if max decreased (we'll renormalize later)
            new_max = float(np.max(affected_new))
            # Also need to check c outside affected range (unchanged)
            if k_lo > 0:
                new_max = max(new_max, float(np.max(c[:k_lo])))
            if k_hi < L - 1:
                new_max = max(new_max, float(np.max(c[k_hi + 1:])))
            s_new = float(a.sum()) + delta  # sum increased by delta
            new_score = 2.0 * n * new_max / (s_new * s_new)

            if new_score < best_score * (1.0 - 1e-8):
                a[i] = new_ai
                c[k_lo:k_hi + 1] = affected_new
                stagnant = 0
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
                    c_best = c.copy()
            else:
                # Try decreasing
                new_ai2 = old_ai - delta
                if new_ai2 < 0:
                    stagnant += 1
                    continue
                affected2 = c[k_lo:k_hi + 1].copy()
                a_vals2 = a[ki_lo:ki_hi + 1]
                affected2_new = affected2 - 2.0 * delta * a_vals2
                new_max2 = float(np.max(affected2_new))
                if k_lo > 0:
                    new_max2 = max(new_max2, float(np.max(c[:k_lo])))
                if k_hi < L - 1:
                    new_max2 = max(new_max2, float(np.max(c[k_hi + 1:])))
                s_new2 = float(a.sum()) - delta
                if s_new2 < 0.01:
                    stagnant += 1
                    continue
                new_score2 = 2.0 * n * new_max2 / (s_new2 * s_new2)
                if new_score2 < best_score * (1.0 - 1e-8):
                    a[i] = new_ai2
                    c[k_lo:k_hi + 1] = affected2_new
                    stagnant = 0
                    if new_score2 < best_score:
                        best_score = new_score2
                        best = a.copy()
                        c_best = c.copy()
                else:
                    stagnant += 1
        # End if-loop

        if stagnant > 50000:
            break

    # Renormalize best
    best = _normalize(best, float(n))
    best_score = _score(best)
    return best, best_score


def _gradient_local_search(a, budget):
    """
    Improved multiplicative gradient search (similar to previous but with
    different hyperparameters and all-peak weighting).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = _normalize(a, float(n))
    t_end = time.monotonic() + budget
    rng = np.random.default_rng(123)

    c = _fft_conv(a)
    s = float(a.sum())
    score = float(2.0 * n * float(np.max(c)) / (s * s))
    best = a.copy()
    best_score = score

    eta = 0.15
    mom = np.zeros(n, dtype=np.float64)
    beta_mom = 0.4
    it = 0
    stagnant = 0

    while time.monotonic() < t_end - 2:
        it += 1
        # Adaptive K: use top fraction of peaks
        L = 2 * n - 1
        frac = max(0.01, 0.05 * math.exp(-it / 5000.0))
        K = max(1, min(L, int(L * frac), 256))

        # Get top K peaks
        if K >= L:
            idx = np.arange(L)
            weights = np.ones(L) / L
        else:
            idx = np.argpartition(c, -K)[-K:]
            c_idx = c[idx]
            c_max = float(np.max(c_idx))
            # Exponential weighting (sharper focus on largest peaks)
            weights = np.exp((c_idx - c_max) * 2.0)
            weights /= float(np.sum(weights))

        # Compute gradient
        d = np.zeros(n, dtype=np.float64)
        for k, w in zip(idx, weights):
            k = int(k)
            start = max(0, k - n + 1)
            stop = min(n, k + 1)
            s_start = k - (stop - 1)
            s_end = k - start + 1
            d[start:stop] += w * a[s_start:s_end][::-1]
        d *= 2.0

        # Center to preserve sum
        d -= float(np.mean(d))
        scale = float(np.max(np.abs(d)))
        if scale < 1e-14:
            stagnant += 1
            if stagnant > 200:
                break
            continue
        d /= scale

        # Momentum
        mom = beta_mom * mom + (1.0 - beta_mom) * d

        # Multiplicative update
        noise_level = 0.01 * math.exp(-it / 3000.0)
        logf = -eta * mom + rng.normal(0.0, noise_level, n)
        logf = np.clip(logf, -2.0, 2.0)
        cand = a * np.exp(logf)
        cand = np.maximum(cand, 0.0)
        cand = _normalize(cand, float(n))

        c2 = _fft_conv(cand)
        s2 = float(cand.sum())
        cand_score = float(2.0 * n * float(np.max(c2)) / (s2 * s2))

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            stagnant = 0

        if cand_score <= score * (1.0 + 0.01):
            a = cand
            c = c2
            score = cand_score
            eta = min(0.5, eta * 1.03)
            stagnant += 1
        else:
            eta = max(0.02, eta * 0.85)
            mom *= 0.5
            stagnant += 1

        if stagnant > 300:
            # Restart with perturbation
            a = best * (1.0 + rng.normal(0.0, 0.1, n))
            a = _normalize(a, float(n))
            c = _fft_conv(a)
            s = float(a.sum())
            score = float(2.0 * n * float(np.max(c)) / (s * s))
            eta = 0.2
            mom = np.zeros(n, dtype=np.float64)
            stagnant = 0

    return best, best_score


def _upsample(seq, n):
    """Linearly interpolate sequence to length n."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    n = int(n)
    if m == 0:
        return np.ones(n) * (n / max(1, n))
    if m == n:
        return seq.copy()
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return _normalize(out, float(n))


def _generate_diverse_starts(n, rng, base_seq=None):
    """Generate diverse starting sequences."""
    starts = []

    # Base sequence (previous best)
    if base_seq is not None:
        s = _normalize(base_seq[:n] if len(base_seq) >= n else _upsample(base_seq, n), float(n))
        sc = _score(s)
        starts.append((sc, s))

    # Uniform
    u = np.ones(n, dtype=np.float64)
    starts.append((_score(u), u))

    # Chebyshev windows with various parameters
    for beta in [1.5, 2.0, 3.0, 5.0, 8.0]:
        for order in [3, 5, 7, 10, 15]:
            w = _chebyshev_window(n, beta, order)
            s = _normalize(w, float(n))
            sc = _score(s)
            starts.append((sc, s))

    # Cosine windows
    for c1 in [-0.5, -0.3, 0.0, 0.3, 0.5]:
        for c2 in [-0.3, -0.1, 0.1, 0.3]:
            w = _cosine_window(n, [c1, c2])
            s = _normalize(w, float(n))
            sc = _score(s)
            starts.append((sc, s))

    # Power windows
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for q in [0.5, 1.0, 1.5, 2.0]:
            w = _power_window(n, p, q)
            s = _normalize(w, float(n))
            sc = _score(s)
            starts.append((sc, s))

    # Random smooth perturbations of base
    if base_seq is not None:
        for _ in range(5):
            t = np.linspace(0, 1, n)
            z = sum(rng.uniform(-0.5, 0.5) * np.cos(2*np.pi*m*t + rng.uniform(0, 2*np.pi))
                    for m in range(1, 5))
            s = _normalize(base_seq[:n] if len(base_seq) >= n else _upsample(base_seq, n), float(n))
            s *= np.exp(z)
            s = _normalize(s, float(n))
            sc = _score(s)
            starts.append((sc, s))

    starts.sort(key=lambda x: x[0])
    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.
    
    1. Parametric search across multiple families and lengths.
    2. Soft-max L-BFGS local refinement.
    3. Coordinate descent fine-tuning.
    4. Gradient-based local search.
    Returns the best sequence found.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    rng = np.random.default_rng(seed)

    if budget < 30:
        # Tiny budget: return a simple Chebyshev window
        N = 1024
        w = _chebyshev_window(N, 2.0, 5)
        w = _normalize(w, float(N))
        return [float(x) for x in w]

    # Reserve time for final validation
    search_deadline = t0 + max(10.0, budget - 30.0)

    print(f"Total budget: {budget:.0f}s, search deadline: {search_deadline - t0:.0f}s")

    # Get starting sequence
    base_seq = None
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if base_arr.size > 0 and np.all(np.isfinite(base_arr)):
            base_seq = np.clip(base_arr, 0.0, 1000.0)
            print(f"Starting from height_sequence_1: len={len(base_seq)}, score={_score(base_seq):.6f}")
    except Exception:
        pass

    # Sequence lengths to try
    if budget < 300:
        lengths = [2048, 4096]
    elif budget < 900:
        lengths = [2048, 4096, 6733]
    else:
        lengths = [2048, 4096, 6733, 10000, 13107]

    global_best_seq = None
    global_best_score = float('inf')

    # Update global best from base_seq
    if base_seq is not None:
        for N in lengths:
            s = _upsample(base_seq, N)
            sc = _score(s)
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = s
                print(f"  Upsampled base to N={N}: score={sc:.6f}")

    # === PHASE 1: Parametric search (40% of budget) ===
    phase1_budget = 0.40 * (search_deadline - t0)
    print(f"\n=== Phase 1: Parametric search ({phase1_budget:.0f}s) ===")
    t_phase1_end = time.monotonic() + phase1_budget

    for N in lengths:
        if time.monotonic() > t_phase1_end - 20:
            break
        remain = t_phase1_end - time.monotonic()
        if remain < 15:
            break
        alloc = min(60.0, remain / max(1, len(lengths)))
        print(f"\n  N={N}, budget={alloc:.0f}s")
        try:
            sc, seq = _param_search(N, alloc, rng)
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = seq
                print(f"  New global best: {sc:.6f} at N={N}")
        except Exception as e:
            print(f"  Error: {e}")

    # === PHASE 2: Soft-max L-BFGS refinement (25% of budget) ===
    phase2_budget = 0.25 * (budget - (time.monotonic() - t0))
    print(f"\n=== Phase 2: Soft-max L-BFGS ({phase2_budget:.0f}s) ===")
    t_phase2_end = time.monotonic() + phase2_budget

    if global_best_seq is not None and phase2_budget > 20:
        # Try at the best length and one nearby
        best_len = len(global_best_seq)
        trial_lengths = [best_len]
        for N in lengths:
            if N != best_len and N not in trial_lengths:
                trial_lengths.append(N)
                if len(trial_lengths) >= 3:
                    break
        trial_lengths = trial_lengths[:3]

        for N in trial_lengths:
            if time.monotonic() > t_phase2_end - 15:
                break
            remain = t_phase2_end - time.monotonic()
            if remain < 20:
                break
            alloc = min(80.0, remain / max(1, len(trial_lengths)))
            s = _upsample(global_best_seq, N)
            print(f"  Refining N={N}, budget={alloc:.0f}s, start_score={_score(s):.6f}")
            try:
                seq, sc = _soft_max_local_search(s, alloc)
                if sc < global_best_score:
                    global_best_score = sc
                    global_best_seq = seq
                    print(f"  New global best: {sc:.6f} at N={N}")
            except Exception as e:
                print(f"  Error: {e}")

    # === PHASE 3: Coordinate descent (25% of budget) ===
    phase3_budget = 0.25 * (budget - (time.monotonic() - t0))
    print(f"\n=== Phase 3: Coordinate descent ({phase3_budget:.0f}s) ===")
    t_phase3_end = time.monotonic() + phase3_budget

    if global_best_seq is not None and phase3_budget > 20:
        best_len = len(global_best_seq)
        trial_lengths = [best_len]
        for N in lengths:
            if N != best_len:
                trial_lengths.append(N)
                if len(trial_lengths) >= 2:
                    break

        for N in trial_lengths:
            if time.monotonic() > t_phase3_end - 10:
                break
            remain = t_phase3_end - time.monotonic()
            if remain < 20:
                break
            alloc = min(200.0, remain / max(1, len(trial_lengths)))
            s = _upsample(global_best_seq, N)
            print(f"  Coord descent N={N}, budget={alloc:.0f}s")
            try:
                seq, sc = _coordinate_descent(s, alloc, step_size=0.03)
                if sc < global_best_score:
                    global_best_score = sc
                    global_best_seq = seq
                    print(f"  New global best: {sc:.6f} at N={N}")
            except Exception as e:
                print(f"  Error: {e}")

    # === PHASE 4: Gradient-based local search (remaining time) ===
    remaining = max(0.0, search_deadline - time.monotonic())
    print(f"\n=== Phase 4: Gradient search ({remaining:.0f}s) ===")

    if global_best_seq is not None and remaining > 20:
        try:
            seq, sc = _gradient_local_search(global_best_seq, remaining * 0.7)
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = seq
                print(f"  New global best: {sc:.6f}")
        except Exception as e:
            print(f"  Error: {e}")

        # Final polish with coordinate descent
        remaining2 = max(0.0, search_deadline - time.monotonic() - 5)
        if remaining2 > 10 and global_best_seq is not None:
            try:
                seq, sc = _coordinate_descent(global_best_seq, remaining2, step_size=0.01)
                if sc < global_best_score:
                    global_best_score = sc
                    global_best_seq = seq
                    print(f"  Final polish: {sc:.6f}")
            except Exception:
                pass

    # === Final: also try fresh starts via diverse generation ===
    remaining3 = max(0.0, search_deadline - time.monotonic() - 10)
    if remaining3 > 30 and global_best_seq is not None:
        print(f"\n=== Phase 5: Fresh starts ({remaining3:.0f}s) ===")
        for N in lengths[:2]:
            if time.monotonic() > search_deadline - 10:
                break
            starts = _generate_diverse_starts(N, rng, global_best_seq)
            for k, (sc, sq) in enumerate(starts[:3]):
                if time.monotonic() > search_deadline - 10:
                    break
                if sc < global_best_score:
                    global_best_score = sc
                    global_best_seq = sq
                # Quick gradient refinement
                alloc = min(30.0, (search_deadline - time.monotonic() - 5) / 3)
                if alloc > 5:
                    try:
                        seq, sc2 = _gradient_local_search(sq, alloc)
                        if sc2 < global_best_score:
                            global_best_score = sc2
                            global_best_seq = seq
                            print(f"  Fresh start N={N}, k={k}: {sc2:.6f}")
                    except Exception:
                        pass

    # === Validation ===
    print(f"\n=== Validation ===")
    if global_best_seq is None:
        global_best_seq = _chebyshev_window(1024, 2.0, 5)
        global_best_seq = _normalize(global_best_seq, 1024.0)

    final_seq = np.asarray(global_best_seq, dtype=np.float64)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(1024, dtype=np.float64)
    else:
        # Normalize to mean 1 (sum = n) for the evaluator
        scale = float(final_seq.size) / s
        if final_seq.max() * scale <= 1000.0:
            final_seq = final_seq * scale
        final_seq = np.clip(final_seq, 0.0, 1000.0)

    final_list = [float(x) for x in final_seq]
    print(f"Final sequence length: {len(final_list)}")
    print(f"Fast score: {_score(final_seq):.6f}")

    # Exact validation
    try:
        exact_sc = evaluate_sequence(final_list)
        print(f"Exact score: {exact_sc:.6f}")
    except Exception as e:
        print(f"Exact eval error: {e}")

    elapsed = time.monotonic() - t0
    print(f"Total time: {elapsed:.1f}s")

    return final_list
