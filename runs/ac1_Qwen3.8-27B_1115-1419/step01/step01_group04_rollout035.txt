  that keeps the autocorrelation below the current peak level.
- Combined with FFT-based gradient descent for fast local refinement.
- Multi-resolution: start at smaller n, upsample and refine.
- Use the provided height_sequence_1 as a warm start.

import numpy as np

def fast_convolution(a):
    """FFT-based linear convolution a * a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def fast_score_array(a):
    """Fast FFT-based analogue of evaluate_sequence()."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = fast_convolution(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(c)) / (s * s))


def normalize_sequence(a, target_sum):
    """Clip, normalize to target sum."""
    a = np.array(a, dtype=np.float64, copy=True)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    for _ in range(5):
        s = float(a.sum())
        if s <= 1e-15:
            a.fill(float(target_sum) / float(a.size))
            return a
        a *= float(target_sum) / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def build_toeplitz_matrix(a):
    """Build the matrix A where (A @ b)[k] = sum_j a[k-j]*b[j] (convolution)."""
    n = len(a)
    L = 2 * n - 1
    # A is L x n, banded
    # Row k, column j: a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(L):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(L, n))
    return A


def solve_lp_direction(a, M, solver_hint=None):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Returns the optimal b (normalized to sum = sum(a)).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    A = build_toeplitz_matrix(a)
    L = 2 * n - 1

def gradient_topK(a, c, K):
    """Subgradient of max convolution coefficient (averaged over top K)."""
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(int(K), L))
    d = np.zeros(n, dtype=np.float64)
    
    if K == 1:
        k = int(np.argmax(c))
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] = a[s_start:s_end][::-1]
        return 2.0 * d
    
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    inv = 1.0 / float(K)
    for k in idx:
        k = int(k)
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] += inv * a[s_start:s_end][::-1]
    
    return 2.0 * d

            stagnant = 0

            mom *= 0.5

def simulated_annealing(a, budget_s, seed=0):
    """Simulated annealing with block perturbations."""
    a = normalize_sequence(np.asarray(a, dtype=np.float64), float(len(a)))
    n = int(a.size)
    target = float(n)
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))
    
    c = fast_convolution(a)
    score = fast_score_array(a)
    best = a.copy()
    best_score = score
    
    T = 0.1  # initial temperature
    cooling = 0.9995
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        # Block perturbation
        block_size = int(rng.integers(max(1, n // 100), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - block_size + 1)))
        
        cand = a.copy()
        # Randomly scale the block
        factor = np.exp(rng.normal(0.0, 0.5, block_size))
        cand[start:start + block_size] *= factor
        cand = np.maximum(cand, 0.0)
        cand = normalize_sequence(cand, target)
        
        c2 = fast_convolution(cand)
        sc2 = fast_score_array(cand)
        
        delta = sc2 - score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            a = cand
            c = c2
            score = sc2
            if score < best_score:
                best = a.copy()
                best_score = score
        
        T *= cooling
    
    return best, best_score


def upsample_sequence(seq, n):
    """Linearly interpolate a sequence to length n."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    n = int(n)
    if m == 0:
        return normalize_sequence(np.ones(n), float(n))
    if m == n:
        return normalize_sequence(seq, float(n))
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_sequence(out, float(n))


def sanitize_for_return(seq):
    """Return a valid Python list of non-negative finite floats."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return [1.0]
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 512
    scale = float(seq.size) / s
    if seq.max() * scale <= 1000.0:
        seq = seq * scale
    seq = np.clip(seq, 0.0, 1000.0)
    if float(seq.sum()) < 0.01:
        return [1.0] * 512
    return [float(x) for x in seq]

"""
Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

Strategy:
- LP-based descent: Solve LP to find optimal direction that keeps
  cross-convolution below current peak, then blend toward it.
- FFT-based gradient descent for fast local refinement between LP steps.
- Warm start from height_sequence_1.
- Multi-phase: LP phase, gradient phase, fine-tuning phase.
"""

import numpy as np
import time
import math
from scipy.sparse import lil_matrix, csr_matrix


def fast_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def fast_score_array(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = fast_convolution(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def normalize_sequence(a, target_sum):
    a = np.array(a, dtype=np.float64, copy=True)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    for _ in range(5):
        s = float(a.sum())
        if s <= 1e-15:
            a.fill(float(target_sum) / float(a.size))
            return a
        a *= float(target_sum) / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ b)[k] = (a*b)[k] (convolution)."""
    n = len(a)
    L = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(L):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(L, n))


def solve_lp_direction(a, M):
    """Solve max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    A = build_conv_matrix(a)
    
    b = cp.Variable(n)
    constraints = [b >= 0, A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(verbose=False)
            except Exception:
                return None
    
    if b.value is None:
        return None
    b_val = np.maximum(b.value, 0.0)
    if b_val.sum() < 1e-10:
        return None
    return b_val


def gradient_topK(a, c, K):
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(int(K), L))
    d = np.zeros(n, dtype=np.float64)
    
    if K == 1:
        k = int(np.argmax(c))
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] = a[s_start:s_end][::-1]
        return 2.0 * d
    
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    inv = 1.0 / float(K)
    for k in idx:
        k = int(k)
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] += inv * a[s_start:s_end][::-1]
    
    return 2.0 * d

"""
Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

Strategy:
- LP-based descent using HiGHS solver: at each step, solve an LP to find
  the optimal direction that keeps cross-convolution (a*b) below the current
  peak level M, then blend toward that direction.
- Fast FFT-based gradient descent with momentum for moves between LP calls.
- Warm start from height_sequence_1.
- Multi-phase: LP-hybrid phase, pure gradient phase, fine-tuning phase.
- Try both original length and upsampled lengths.
"""

import numpy as np
import time
import math
from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog


def fast_convolution(a):
    """FFT-based linear convolution a * a, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def fast_score_array(a):
    """Fast FFT-based analogue of evaluate_sequence()."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = fast_convolution(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(c)) / (s * s))


def normalize_sequence(a, target_sum):
    """Clip to [0,1000] and normalize to target_sum."""
    a = np.array(a, dtype=np.float64, copy=True)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    for _ in range(5):
        s = float(a.sum())
        if s <= 1e-15:
            a.fill(float(target_sum) / float(a.size))
            return a
        a *= float(target_sum) / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def build_conv_matrix(a):
    """Build sparse matrix A (shape 2n-1 x n) where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    L = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(L):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(L, n))


def solve_lp_direction(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Returns optimal b or None on failure.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    # Fallback to CVXPY
    try:
        import cvxpy as cp
        b = cp.Variable(n)
        constraints = [b >= 0, A @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if b.value is not None:
            return np.maximum(b.value, 0.0)
    except Exception:
        pass
    return None


def gradient_topK(a, c, K):
    """Subgradient of max convolution coefficient (averaged over top K peaks)."""
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(int(K), L))
    d = np.zeros(n, dtype=np.float64)

    if K == 1:
        k = int(np.argmax(c))
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] = a[s_start:s_end][::-1]
        return 2.0 * d

    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]

    inv = 1.0 / float(K)
    for k in idx:
        k = int(k)
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] += inv * a[s_start:s_end][::-1]

    return 2.0 * d

def gradient_descent(a, budget_s, seed=0, eta_init=0.12, beta=0.5, K_frac=0.25):

    mom = np.zeros(n, dtype=np.float64)
    noise = 0.015
    stagnant = 0
    it = 0
    last_print = 0

    while time.monotonic() < deadline - 1.0:
        it += 1
        if not np.all(np.isfinite(a)) or a.sum() <= 0:
            a = best.copy()
            c = fast_convolution(a)
            score = best_score
            mom = np.zeros(n)

        K = max(1, min(256, int(n * K_frac)))
        grad = gradient_topK(a, c, K)
        g = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(g)))
        if scale < 1e-14:
            stagnant += 1
            if stagnant > 500:
                break
            continue

        g /= scale
        mom = beta * mom + (1.0 - beta) * g

        logf = -eta * mom
        logf += rng.normal(0.0, noise, n)
        logf = np.clip(logf, -2.5, 2.5)

        cand = a * np.exp(logf)
        cand = np.maximum(cand, 0.0)
        cand = normalize_sequence(cand, target)

        c2 = fast_convolution(cand)
        sc2 = fast_score_array(cand)

        if not np.isfinite(sc2):
            eta = max(0.005, eta * 0.7)
            stagnant += 1
            continue

        if sc2 < best_score:
            best = cand.copy()
            best_score = sc2
            if it - last_print > 500:
                print(f"      [GradDescent] it={it}, score={score:.6f}, best={best_score:.6f}")
                last_print = it

        if sc2 <= score * (1.0 + 0.01):
            a = cand
            c = c2
            score = sc2
            stagnant = 0
            eta = min(0.5, eta * 1.005)
            noise = max(0.002, noise * 0.9998)
        else:
            eta = max(0.005, eta * 0.92)
            mom *= 0.5
            stagnant += 1

        if stagnant > 800:
            a = best.copy() * np.exp(rng.normal(0.0, 0.35, n))
            a = normalize_sequence(a, target)
            c = fast_convolution(a)
            score = fast_score_array(a)
            stagnant = 0
            eta = eta_init
            noise = 0.03
            mom = np.zeros(n)

    return best, best_score


def smooth_sequence(a, sigma_frac=0.01):
    """Apply a small Gaussian smoothing to the sequence."""
    from scipy.ndimage import gaussian_filter1d
    n = len(a)
    sigma = max(1.0, sigma_frac * n)
    a_s = gaussian_filter1d(a, sigma, mode='nearest')
    return normalize_sequence(a_s, float(a.sum()))


def sanitize_for_return(seq):
    """Return a valid Python list of non-negative finite floats."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return [1.0]
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 512
    scale = float(seq.size) / s
    if seq.max() * scale <= 1000.0:
        seq = seq * scale
    seq = np.clip(seq, 0.0, 1000.0)
    if float(seq.sum()) < 0.01:
        return [1.0] * 512
    return [float(x) for x in seq]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.
    
    Phases:
    1. LP-hybrid descent (45% time) - uses LP for informed directions
    2. Gradient descent (25% time) - fast local refinement
    3. Multi-restart gradient (20% time) - escape local minima
    4. Smoothing + final polish (10% time)
    """
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0

    # Get initial sequence from warm start
    try:
        parent = globals().get('height_sequence_1', None)
        if parent is not None and len(parent) > 0:
            a0 = np.array(parent, dtype=np.float64)
        else:
            a0 = np.ones(512)
    except Exception:
        a0 = np.ones(512)

    n = len(a0)
    a0 = normalize_sequence(a0, float(n))
    init_score = fast_score_array(a0)
    print(f"Initial: n={n}, score={init_score:.6f}")

    best_seq = a0.copy()
    best_score = init_score

    # Phase 1: LP-hybrid descent
    p1_budget = 0.45 * max(0.0, deadline - time.monotonic())
    if p1_budget > 30:
        print(f"\n=== Phase 1: LP-hybrid descent ({p1_budget:.0f}s) ===")
        try:
            seq, sc = lp_hybrid_descent(a0, p1_budget, seed=seed)
            if sc < best_score:
                best_score = sc
                best_seq = seq
            print(f"  Phase 1 result: {sc:.6f}")
        except Exception as e:
            print(f"  Phase 1 error: {e}")

    # Phase 2: Pure gradient descent
    p2_budget = 0.25 * max(0.0, deadline - time.monotonic())
    if p2_budget > 20:
        print(f"\n=== Phase 2: Gradient descent ({p2_budget:.0f}s) ===")
        try:
            seq, sc = gradient_descent(best_seq, p2_budget, seed=seed + 100)
            if sc < best_score:
                best_score = sc
                best_seq = seq
            print(f"  Phase 2 result: {sc:.6f}")
        except Exception as e:
            print(f"  Phase 2 error: {e}")

    # Phase 3: Multi-restart gradient
    p3_budget = 0.2 * max(0.0, deadline - time.monotonic())
    if p3_budget > 30:
        print(f"\n=== Phase 3: Multi-restart gradient ({p3_budget:.0f}s) ===")
        n_restarts = max(2, int(p3_budget / 60))
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            b = remaining / (n_restarts - r)
            try:
                # Perturb best and refine
                rng_r = np.random.default_rng(seed + 500 + r * 37)
                perturbed = best_seq.copy() * np.exp(rng_r.normal(0.0, 0.15, len(best_seq)))
                perturbed = normalize_sequence(perturbed, float(len(best_seq)))
                seq, sc = gradient_descent(perturbed, b * 0.8, seed=seed + 500 + r * 13)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq
                    print(f"  Restart {r}: {sc:.6f}")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Phase 4: Smoothing and final polish
    p4_budget = max(0.0, deadline - time.monotonic())
    if p4_budget > 15:
        print(f"\n=== Phase 4: Smoothing + polish ({p4_budget:.0f}s) ===")
        # Try mild smoothing
        for sigma_frac in [0.005, 0.01, 0.02]:
            try:
                smoothed = smooth_sequence(best_seq, sigma_frac)
                sc = fast_score_array(smoothed)
                if sc < best_score:
                    best_score = sc
                    best_seq = smoothed
                    print(f"  Smooth (sigma={sigma_frac}): {sc:.6f}")
                    # Quick refine
                    b = min(30.0, p4_budget * 0.3)
                    if b > 5:
                        seq, sc2 = gradient_descent(best_seq, b, seed=seed + 900)
                        if sc2 < best_score:
                            best_score = sc2
                            best_seq = seq
                            print(f"  After smooth refine: {sc2:.6f}")
            except Exception:
                pass

        # Final short gradient polish
        remaining = deadline - time.monotonic()
        if remaining > 10:
            try:
                seq, sc = gradient_descent(best_seq, remaining, seed=seed + 777,
                                           eta_init=0.05, K_frac=0.3)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq
                    print(f"  Final polish: {sc:.6f}")
            except Exception:
                pass

    # Final exact validation
    final_seq = sanitize_for_return(best_seq)
    try:
        exact = float(evaluate_sequence(final_seq))
        print(f"\nFinal: n={len(final_seq)}, fast_score={best_score:.6f}, exact_score={exact:.6f}")
    except Exception as e:
        print(f"Validation error: {e}")
        exact = float('inf')

    if not np.isfinite(exact):
        # Fallback
        final_seq = [1.0] * 512
    
    print(f"Returning sequence of length {len(final_seq)}")
    return final_seq
