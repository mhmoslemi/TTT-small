"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Adam optimization with soft-max
surrogate, log-reparameterization, and multi-scale search.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b=None):
    if b is None:
        b = a
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_ff(a):
    n = len(a)
    b = _fft_conv(a)
    max_b = float(np.max(b))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _soft_max_weights(b, tau):
    bm = float(np.max(b))
    e = np.exp(tau * (b - bm))
    se = float(np.sum(e))
    p = e / se
    return p


def _grad_wrt_x(a, p):
    """
    Gradient of soft_max(b(a)) w.r.t. x where a = exp(x).
    = a * grad_a where grad_a[j] = 2 * corr[j]
    corr[j] = sum_k p[k]*a[k-j] (cross-correlation)
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real[:n]
    grad_a = 2.0 * corr
    return a * grad_a


def _adam_optimize(a_init, time_budget, name="", lr_init=0.01, tau_init=30.0,
                   tau_rate=1.002, check_every=500, verbose=True):
    """Adam optimization on x = log(a)."""
    n = len(a_init)
    a0 = np.maximum(a_init, 1e-15).astype(np.float64)
    a0 = a0 / a0.sum()
    x = np.log(a0)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    tau = tau_init

    best_a = a0.copy()
    best_score = _score_ff(a0)
    stuck = 0
    deadline = time.monotonic() + time_budget
    step = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() < deadline:
        a = np.exp(x)
        b = _fft_conv(a)
        p = _soft_max_weights(b, tau)
        gx = _grad_wrt_x(a, p)

        # Normalize gradient for stability
        gn = np.linalg.norm(gx)
        if gn > 1e-12:
            gx = gx / gn

        t += 1
        m = 0.9 * m + 0.1 * gx
        v = 0.999 * v + 0.001 * gx * gx
        bc1 = 1 - 0.9 ** t
        bc2 = 1 - 0.999 ** t
        x -= lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)

        step += 1
        tau = min(tau * tau_rate, 10000.0)

        if step % check_every == 0:
            a_chk = np.exp(x)
            sc = _score_ff(a_chk)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_chk.copy()
                stuck = 0
                if verbose:
                    print(f"  {name} [{step}]: {sc:.6f} (tau={tau:.0f})")
            else:
                stuck += 1
            # Decay lr

            if stuck > 40:

                m[:] = 0
                v[:] = 0
                t = 0
                lr = lr_init
                stuck = 0

    return best_a, best_score

"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using Adam optimization with log-sum-exp
soft-max surrogate, log-reparameterization for non-negativity, FFT-based
convolution, and multi-scale multi-init search strategy.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b=None):
    """Fast linear convolution via FFT."""
    if b is None:
        b = a
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_ff(a):
    """Compute score using FFT convolution (fast internal tracking)."""
    n = len(a)
    b = _fft_conv(a)
    max_b = float(np.max(b))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _soft_max_weights(b, tau):
    """Compute probability weights p[k] = exp(tau*b[k]) / sum(...) for soft-max."""
    bm = float(np.max(b))
    e = np.exp(tau * (b - bm))
    se = float(np.sum(e))
    return e / se


def _compute_grad_x(a, p, sm_val, S):
    """
    Compute gradient of score ~ 2n*SM/S^2 w.r.t. x (where a=exp(x)).
    Returns grad_x[i] proportional to a[i]*(corr[i] - SM/S)
    where corr[i] = sum_k p[k]*a[k-i] (cross-correlation via FFT).
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    corr = ifft(P * np.conj(A)).real[:n]
    # Full score gradient (up to positive constant 4n/S^2):
    grad_x = a * (corr - sm_val / S)
    return grad_x


def _adam_optimize(a_init, time_budget, name="", lr_init=0.01, tau_init=30.0,
                   tau_rate=1.003, check_every=400, verbose=True,
                   recenter_every=500):
    """
    Adam optimization on x = log(a) to minimize the score.
    Uses soft-max surrogate with increasing temperature.
    """
    n = len(a_init)
    a0 = np.maximum(a_init.astype(np.float64), 1e-15)
    a0 = a0 / a0.sum()  # normalize to sum=1
    x = np.log(a0)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    tau = tau_init

    best_a = a0.copy()
    best_score = _score_ff(a0)
    stuck = 0
    deadline = time.monotonic() + time_budget
    step = 0
    rng = np.random.default_rng(99999 + n % 10000)

    while time.monotonic() < deadline:
        a = np.exp(x)
        b = _fft_conv(a)
        bm = float(np.max(b))
        e = np.exp(tau * (b - bm))
        se = float(np.sum(e))
        p = e / se
        sm_val = (bm + np.log(se)) / tau
        S = float(np.sum(a))

        gx = _compute_grad_x(a, p, sm_val, S)

        # Normalize gradient for stability
        gn = np.linalg.norm(gx)
        if gn > 1e-15:
            gx = gx / gn
        else:
            gx = np.zeros(n)

        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * gx
        v = 0.999 * v + 0.001 * gx * gx
        bc1 = 1.0 - 0.9 ** t
        bc2 = 1.0 - 0.999 ** t
        x = x - lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)

        step += 1
        tau = min(tau * tau_rate, 20000.0)

        # Re-center x to prevent overflow

            x = x - np.mean(x)

        if step % check_every == 0:

            sc = _score_ff(a_chk)
            if sc < best_score - 1e-12:

                best_a = a_chk.copy()

                if verbose:

                m[:] = 0
                v[:] = 0
                t = 0

                stuck = 0

def _peak_shave(a, n_iter=200):
    """
    Local peak-shaving: iteratively reduce the dominant autocorrelation peak
    by redistributing mass from high-contribution elements.
    """
    n = len(a)
    a = a.copy()
    S = float(np.sum(a))
    best = a.copy()
    best_sc = _score_ff(a)

    for it in range(n_iter):
        b = _fft_conv(a)
        kstar = int(np.argmax(b))
        # Mirror vector: which a[i] contribute to b[kstar]
        # b[kstar] = sum_i a[i]*a[kstar-i] for valid i
        i_lo = max(0, kstar - n + 1)
        i_hi = min(n - 1, kstar)
        if i_hi < i_lo:
            break
        mirror = a[i_lo:i_hi + 1]
        contrib = 2.0 * mirror  # each a[i] contributes 2*a[kstar-i] to grad
        
        # Find the top contributor and redistribute
        idx_local = int(np.argmax(contrib))
        idx_global = i_lo + idx_local
        # Reduce this element, add to elements far from the peak
        reduce_amt = 0.001 * a[idx_global]
        if reduce_amt < 1e-12:
            break
        a[idx_global] -= reduce_amt
        # Distribute to other positions (those not in the peak window)
        other_mask = np.ones(n, dtype=bool)
        other_mask[max(0, kstar - n + 1):min(n, kstar + 1)] = False
        if other_mask.sum() == 0:
            break
        other_idx = np.where(other_mask)[0]
        a[other_idx] += reduce_amt / len(other_idx)
        a = np.maximum(a, 0.0)
        a = a / a.sum() * S

        sc = _score_ff(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        else:
            a = best.copy()

    return best, best_sc
