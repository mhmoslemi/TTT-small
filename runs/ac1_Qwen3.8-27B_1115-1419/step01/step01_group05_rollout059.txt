"""
Search for low-peak nonnegative self-convolutions using LSE-smoothed
gradient descent with Nesterov momentum, diverse skewed-bump
parametric initializations, progressive sharpening, and resampling.

Key insight: optimal sequences are asymmetric (skewed) bumps since
symmetric sequences provably score >= 2.
"""
import numpy as np
import time

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s <= 1e-300:
        return np.ones(max(a.size, 1), dtype=np.float64) / max(a.size, 1)
    return a / s

def _conv(a):
    n = a.size
    sz = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, sz)
    return np.fft.irfft(A * A, sz)[:2 * n - 1]

def _score(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b = _conv(a)
    mb = float(b.max())
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb

def _lse_and_weights(b, beta):
    m = float(b.max())
    if beta <= 1.0:
        p = np.ones(b.size, dtype=np.float64) / b.size
        return m, p
    shifted = beta * (b - m)
    shifted = np.maximum(shifted, -745.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0.0:
        p = np.ones(b.size, dtype=np.float64) / b.size
        return m, p
    p = e / s
    val = m + float(np.log(s)) / beta
    return val, p

def _gradient(a, p, n):
    """Compute 2 * sum_k p[k] * a[k-i] for i=0..n-1 using FFT."""
    sz = _next_pow2(3 * n)
    ar = a[::-1]
    A = np.fft.rfft(ar, sz)
    P = np.fft.rfft(p, sz)
    conv = np.fft.irfft(A * P, sz)
    h = conv[n - 1:2 * n - 1]
    return 2.0 * np.maximum(h, 0.0)

def _soft_project(v):
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / max(v.size, 1)
    return v / s

def _make_skewed(n, mu, sigma, skew):
    x = np.linspace(0.0, 1.0, n)
    w = np.where(x < mu, sigma * (1.0 - skew), sigma * (1.0 + skew))
    w = np.maximum(w, 0.02)
    a = np.exp(-0.5 * ((x - mu) / w) ** 2)
    return _normalize(a)

def _make_exp_power(n, alpha, beta):
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.exp(-alpha * x ** beta)
    return _normalize(a)

def _make_beta_shape(n, p, q):
    x = np.arange(n, dtype=np.float64)
    eps = 0.5
    log_a = p * np.log(x + eps) + q * np.log(n - 1 - x + eps)
    a = np.exp(log_a - log_a.max())
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return a / s

def _make_gamma_shift(n, loc, shape):
    """Gamma-distribution-like shape shifted to position loc."""
    x = np.arange(n, dtype=np.float64)
    t = x - loc
    t = np.maximum(t, 1e-6)
    a = t ** (shape - 1) * np.exp(-t)
    # Zero out the left side (before loc)
    a[x <= loc] = 0.0
    # Add a small right tail
    left_t = np.maximum(loc - x, 1e-6)
    a_left = left_t ** (shape * 0.5) * np.exp(-left_t) * 0.1
    a = a + a_left
    return _normalize(a)

def _make_two_hump(n, mu1, mu2, w1, w2):
    x = np.linspace(0.0, 1.0, n)
    a1 = np.exp(-0.5 * ((x - mu1) / w1) ** 2)
    a2 = np.exp(-0.5 * ((x - mu2) / w2) ** 2)
    a = 0.7 * a1 + 0.3 * a2
    return _normalize(a)

def _generate_inits(n, rng, given=None):
    inits = []
    # Wide range of skewed bumps
    for mu in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
        for sigma in [0.08, 0.12, 0.15, 0.2, 0.25, 0.3]:
            for skew in [-0.4, -0.2, 0.0, 0.2, 0.4, 0.6]:
                inits.append(_make_skewed(n, mu, sigma, skew))
    # Exponential-power
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
        for beta in [0.4, 0.6, 0.8, 1.0, 1.3, 1.7, 2.0, 3.0]:
            inits.append(_make_exp_power(n, alpha, beta))
    # Beta shapes
    for p in [-0.7, -0.5, -0.3, -0.1, 0.0, 0.1, 0.3, 0.5]:
        for q in [-0.7, -0.5, -0.3, -0.1, 0.0, 0.2, 0.5, 1.0, 1.5]:
            inits.append(_make_beta_shape(n, p, q))
    # Gamma-shift
    for loc in [20, 40, 60, 80, 100]:
        for shape in [1.0, 2.0, 3.0, 5.0]:
            if 0 < loc < n:
                inits.append(_make_gamma_shift(n, loc, shape))
    # Two-hump
    for mu1, mu2, w1, w2 in [(0.2, 0.5, 0.1, 0.15), (0.3, 0.6, 0.12, 0.1),
                               (0.15, 0.4, 0.08, 0.2), (0.25, 0.55, 0.1, 0.12)]:
        inits.append(_make_two_hump(n, mu1, mu2, w1, w2))
    # Given sequence
    if given is not None and given.size >= 16:
        x_old = np.linspace(0, 1, given.size)
        x_new = np.linspace(0, 1, n)
        inits.append(_normalize(np.interp(x_new, x_old, given)))
    # Random smooth
    for _ in range(10):
        base = inits[rng.integers(0, len(inits))]
        z = rng.normal(size=n)
        F = np.fft.rfft(z)
        cutoff = max(3, n // 6)
        if cutoff < len(F):
            F[cutoff:] = 0
        f = np.fft.irfft(F, n)
        f = (f - f.mean()) / (f.std() + 1e-10)
        a = base * (1.0 + 0.4 * f)
        inits.append(_normalize(np.maximum(a, 1e-10)))
    return inits

def _refine(a, deadline, rng, beta_init=6.0):
    """LSE gradient descent with Nesterov momentum and adaptive beta."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    b = _conv(a)
    
    v = np.zeros_like(a)
    lr = 0.08
    beta = beta_init
    momentum = 0.85
    fail = 0
    it = 0
    last_improve = 0
    stall_count = 0
    
    while time.monotonic() < deadline:
        if fail > 300:
            # Hard restart
            a = _normalize(best_a + 0.08 * rng.normal(size=n))
            v = np.zeros_like(a)
            lr = 0.06
            beta = max(4.0, beta * 0.25)
            fail = 0
            it += 1
            continue
        
        val_old, p = _lse_and_weights(b, beta)
        grad = _gradient(a, p, n)
        
        # Nesterov look-ahead
        a_look = _soft_project(a - momentum * v)
        b_look = _conv(a_look)
        val_look, p_look = _lse_and_weights(b_look, beta)
        grad_look = _gradient(a_look, p_look, n)
        
        # Update velocity and position
        v = momentum * v + lr * grad_look
        a_new = _soft_project(a_look - v)
        
        b_new = _conv(a_new)
        val_new, _ = _lse_and_weights(b_new, beta)
        score_new = 2.0 * n * float(b_new.max())
        
        if val_new < val_old * (1.0 - 1e-10):
            a = a_new
            b = b_new
            lr = min(1.0, lr * 1.03)
            fail = 0
            if it - last_improve > 200:
                stall_count += 1
            last_improve = it
        else:
            lr *= 0.6
            v = momentum * 0.5 * v
            fail += 1
        
        if score_new < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = score_new
            last_improve = it
        
        # Adaptive beta increase
        if it % 40 == 0 and it > 0:
            beta = min(5000.0, beta * 1.12 + 0.5)
        
        # Moderate restart with diversity
        if (fail > 150 or (it - last_improve > 800 and it > 200)):
            r = rng.random()
            if r < 0.3:
                # Mix with a fresh random smooth shape
                z = rng.normal(size=n)
                F = np.fft.rfft(z)
                cutoff = max(3, n // 10)
                if cutoff < len(F):
                    F[cutoff:] = 0
                f = np.fft.irfft(F, n)
                f = (f - f.mean()) / (f.std() + 1e-10)
                a = _normalize(best_a * (1.0 + 0.3 * f))
            elif r < 0.6:
                # Block perturbation
                a = best_a.copy()
                w = max(4, n // 20)
                s = int(rng.integers(0, n - w))
                a[s:s+w] *= np.exp(rng.normal(0, 0.4, size=w))
                a = _normalize(a)
            else:
                # Smooth noise
                a = _normalize(best_a * np.exp(rng.normal(0, 0.15, size=n)))
            v = np.zeros_like(a)
            lr = 0.05
            beta = max(4.0, beta * 0.3)
            fail = 0
        
        # Reduce momentum when close to convergence
        if fail > 50:
            momentum = max(0.5, momentum - 0.01)
        
        it += 1
    
    return best_a, best_score

def _resample(old, new_n):
    old = _normalize(old)
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0, 1, old.size)
    x_new = np.linspace(0, 1, new_n)
    return _normalize(np.interp(x_new, x_old, old))

def _polish(a, deadline, rng):
    """Intensive polishing with high beta and small steps."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    b = _conv(a)
    lr = 0.02
    beta = 100.0
    it = 0
    
    while time.monotonic() < deadline:
        val_old, p = _lse_and_weights(b, beta)
        grad = _gradient(a, p, n)
        
        # Simple gradient step (no momentum for stability)
        a_new = _soft_project(a - lr * grad)
        b_new = _conv(a_new)
        val_new, _ = _lse_and_weights(b_new, beta)
        score_new = 2.0 * n * float(b_new.max())
        
        if val_new < val_old:
            a = a_new
            b = b_new
            lr = min(0.5, lr * 1.05)
            if score_new < best_score:
                best_a = a_new.copy()
                best_score = score_new
        else:
            lr *= 0.7
            if lr < 1e-6:
                # Perturb and reset
                a = _normalize(best_a + 0.02 * rng.normal(size=n))
                b = _conv(a)
                lr = 0.02
                beta = max(20.0, beta * 0.5)
                continue
        
        if it % 100 == 0:
            beta = min(10000.0, beta * 1.1)
        
        it += 1
    
    return best_a, best_score

def propose_candidate(seed=0, budget_s=2400, **kwargs):
