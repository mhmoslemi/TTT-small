        # df/da_i = 2n/s^2 * grad_conv[i] - 4n*bmax/s^3 * a_i

        mu = b1 * mu + (1 - b1) * grad_v
        nu = b2 * nu + (1 - b2) * (grad_v ** 2)
        mu_hat = mu / (1 - b1**t)
        nu_hat = nu / (1 - b2**t)

        step = mu_hat / (np.sqrt(nu_hat) + eps)
        step = np.clip(step, -1.0, 1.0)
        
        v_y = v_y - lr * step
        v_y = v_y - v_y.mean()

            a_check = a_check / a_check.sum()

                best_a = a_check.copy()
                last_improve = time.monotonic()
        
        # Update Nesterov momentum
        beta_curr = 0.5 * (1 + np.sqrt(1 + 4 * beta**2))
        v = v_y  # This is the Nesterov step

    a_final = a_final / a_final.sum()
    sc = _score(a_final)
    if sc < best_score:
        best_score = sc

def _make_inits(n, rng, prev_a=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)
    
    # Previous solution (interpolated)
    if prev_a is not None:
        pa = np.asarray(prev_a, dtype=np.float64)
        pa = np.maximum(pa, 0)
        ps = pa.sum()
        if ps > 0:
            if pa.size != n:
                x_old = np.linspace(0, 1, pa.size)
                x_new = np.linspace(0, 1, n)
                pa_interp = np.interp(x_new, x_old, pa / ps)
            else:
                pa_interp = pa / ps
            pa_interp = np.maximum(pa_interp, 0)
            pss = pa_interp.sum()
            if pss > 0:
                inits.append(pa_interp / pss)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Arcsine-like
    left = i + 1.0
    right = n - i
    a = 1.0 / np.sqrt(left * right)
    inits.append(_proj_simplex(a))
    
    # Exponentials
    for c in [0.3, 0.5, 1.0, 2.0, 4.0]:
        a = np.exp(-c * t)
        inits.append(_proj_simplex(a))
    
    # Oscillating non-negative (squared cosine)
    for freq in [1, 2, 3, 5, 7, 10, 15]:
        for alpha in [0.3, 0.5, 0.7]:
            a = (1 + alpha * np.cos(2 * np.pi * freq * t)) ** 2
            inits.append(_proj_simplex(a))
    
    # Random Fourier non-negative
    for _ in range(5):
        a = 1.0
        nc = min(20, n // 2)
        for k in range(1, nc + 1):
            amp = rng.uniform(0.1, 0.5) / (k ** 0.5)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + ph)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            inits.append(a / s)
    
    # Tapered (raised cosine)
    for beta in [0.3, 0.5, 1.0, 2.0]:
        a = 0.5 * (1 + np.cos(np.pi * beta * (2 * t - 1)))
        a = np.maximum(a, 1e-10)
        inits.append(_proj_simplex(a))
    
    return inits

import numpy as np
import time


"""
Search for non-negative step-function heights minimizing 2*n*max(convolve(a,a))/sum(a)^2.

Strategy:
  - Spectral initialization targeting flat/tapered auto-convolution.
  - Diverse parametric initializations (arcsine, oscillating, random Fourier).
  - Nesterov-accelerated Adam optimization in log-softmax space.

  - Multiple sequence lengths (2048, 4096, 8192) with time allocation.

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj_simplex(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(a.shape, 1.0 / max(1, a.size), dtype=np.float64)
    return a / s


def _score(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if bmax <= 0 or not np.isfinite(bmax):
        return np.inf
    return 2.0 * n * bmax / (s * s)


def _score_m(a, m):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if bmax <= 0 or not np.isfinite(bmax):
        return np.inf
    return 2.0 * n * bmax / (s * s)


def _grad_conv(a, b, w, m):
    P = np.fft.fft(w, m)
    F = np.fft.fft(a, m)
    c = np.fft.ifft(P * np.conj(F)).real[:len(a)]
    return 2.0 * c


def _spectral_init(n, m, target, a0, iters=50, blend=0.5):
    L = 2 * n - 1
    target = np.asarray(target, dtype=np.float64)
    if target.size < L:
        t = np.zeros(L, dtype=np.float64)
        t[:target.size] = target
        target = t
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    tm = float(target.max())
    if tm > 0 and np.isfinite(tm):
        target = target / tm
    else:
        target = np.ones(L, dtype=np.float64)

    T = np.fft.fft(target, m)

    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    am = float(a.max())
    if am > 0 and np.isfinite(am):
        a = a / am
    else:
        a = np.ones(n, dtype=np.float64) / n

    sqrt_T = np.fft.ifft(np.sqrt(np.real(T))).real[:n]
    sp = np.maximum(sqrt_T, 0.0)
    spm = float(sp.max())
    if spm > 0 and np.isfinite(spm):
        sp = sp / spm
    else:
        sp = a.copy()

    a = 0.5 * a + 0.5 * sp
    a = np.maximum(a, 1e-15)

    T_norm = float(np.max(np.abs(T)))
    if not np.isfinite(T_norm) or T_norm <= 0:
        T_norm = 1.0

    best_a = a.copy()
    best_sc = _score_m(a, m)

    for it in range(iters):
        F = np.fft.fft(a, m)
        F_abs_sq = np.abs(F) ** 2 + 1e-30 * T_norm ** 2
        F_new = 0.5 * (F + T * np.conj(F) / F_abs_sq)
        if not np.all(np.isfinite(F_new)):
            break

        a_new = np.fft.ifft(F_new).real[:n]
        a_pos = np.maximum(a_new, 0.0)
        ps = float(a_pos.sum())
        if ps < 1e-10:
            a_pos = a.copy()

        bld = max(0.2, blend * (1.0 - 0.5 * it / max(1, iters)))
        a = (1.0 - bld) * a + bld * a_pos
        a = np.maximum(a, 1e-15)
        am = float(a.max())
        if am > 0 and np.isfinite(am):
            a = a / am

        sc = _score_m(a, m)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    best_a = _proj_simplex(best_a)
    return best_a, _score_m(best_a, m)


def _optim_nesterov(a_init, max_time, seed, m):
    n = len(a_init)
    rng = np.random.default_rng(seed)

    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-30)
    a = a / float(a.sum())
    v = np.log(a)
    v = v - float(v.mean())

    best_a = a.copy()
    best_score = _score_m(a, m)

    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    start = time.monotonic()
    deadline = start + max_time
    last_improve = start
    check_every = max(3, min(30, n // 50))

    v_momentum = v.copy()
    beta_nest = 0.9

    while time.monotonic() < deadline:
        elapsed = max(1e-9, time.monotonic() - start)
        progress = min(1.0, elapsed / max_time)

        stuck_limit = min(45.0, 0.25 * max_time)
        if time.monotonic() - last_improve > stuck_limit:
            sigma = 0.10 + 0.30 * (1.0 - progress)
            v_momentum += rng.normal(0.0, sigma, n)
            v_momentum = v_momentum - float(v_momentum.mean())
            v_momentum = np.clip(v_momentum, -20.0, 20.0)
            last_improve = time.monotonic()
            continue

        v = 0.5 * (v + v_momentum)
        v_shifted = v - float(v.max())
        v_shifted = np.clip(v_shifted, -30.0, 30.0)
        a = np.exp(v_shifted)
        a = a / float(a.sum())

        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())

        if not np.isfinite(bmax) or bmax <= 1e-300:
            break

        tau_frac = 0.20 * (1.0 - progress) ** 1.5 + 0.02
        if rng.random() < 0.15:
            tau_frac *= rng.uniform(0.3, 2.5)
        tau = max(1e-15, bmax * tau_frac)

        x = (b - bmax) / tau
        x = np.clip(x, -60.0, 0.0)
        w = np.exp(x)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0:
            w = np.full(b.shape, 1.0 / b.size, dtype=np.float64)
        else:
            w = w / ws

        top_k = max(1, int(0.03 * b.size))
        if top_k < b.size:
            idx = np.argpartition(b, -top_k)[-top_k:]
            w[idx] += 0.3
            w = w / float(w.sum())

        c = _grad_conv(a, b, w, m)

        s = float(a.sum())
        grad_v = a * c
        grad_v = grad_v - float(grad_v.mean())

        gnorm = float(np.sqrt(np.mean(grad_v ** 2)))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            sigma = 0.05 + 0.15 * (1.0 - progress)
            v_momentum += rng.normal(0.0, sigma, n)
            v_momentum = v_momentum - float(v_momentum.mean())
            v_momentum = np.clip(v_momentum, -20.0, 20.0)
            continue

        grad_v = grad_v / gnorm
        t += 1
        mu = b1 * mu + (1.0 - b1) * grad_v
        nu = b2 * nu + (1.0 - b2) * (grad_v * grad_v)

        b1_t = b1 ** t
        b2_t = b2 ** t
        mu_hat = mu / max(1e-12, 1.0 - b1_t)
        nu_hat = nu / max(1e-12, 1.0 - b2_t)

        lr = 0.12 * (1.0 - 0.6 * progress)
        step = mu_hat / (np.sqrt(nu_hat) + eps)
        step = np.clip(step, -0.8, 0.8)

        beta_curr = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta_nest * beta_nest))
        v_momentum = v_momentum - beta_curr * lr * step
        v_momentum = v_momentum - float(v_momentum.mean())
        v_momentum = np.clip(v_momentum, -20.0, 20.0)

        if t % check_every == 0:
            a_check = np.exp(v_momentum - float(v_momentum.max()))
            a_check = a_check / float(a_check.sum())
            sc = _score_m(a_check, m)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_a = a_check.copy()
                last_improve = time.monotonic()

    v_final = v_momentum - float(v_momentum.max())
    a_final = np.exp(np.clip(v_final, -30, 30))
    a_final = a_final / float(a_final.sum())
    sc = _score_m(a_final, m)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return _proj_simplex(best_a), float(best_score)


def _make_inits(n, rng, prev_a=None):
    inits = []
    t = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)

    if prev_a is not None:
        pa = np.asarray(prev_a, dtype=np.float64)
        pa = np.maximum(pa, 0)
        ps = float(pa.sum())
        if ps > 0:
            if pa.size != n:
                x_old = np.linspace(0, 1, pa.size)
                x_new = np.linspace(0, 1, n)
                pa_interp = np.interp(x_new, x_old, pa / ps)
            else:
                pa_interp = pa / ps
            pa_interp = np.maximum(pa_interp, 0)
            pss = float(pa_interp.sum())
            if pss > 0:
                inits.append(pa_interp / pss)
                # Symmetrized version
                pa_sym = 0.5 * (pa_interp + pa_interp[::-1])
                pa_sym = np.maximum(pa_sym, 0)
                pss2 = float(pa_sym.sum())
                if pss2 > 0:
                    inits.append(pa_sym / pss2)

    inits.append(np.ones(n) / n)

    left = i + 1.0
    right = n - i
    a = 1.0 / np.sqrt(left * right)
    inits.append(_proj_simplex(a))

    for c in [0.3, 0.5, 1.0, 2.0, 4.0]:
        inits.append(_proj_simplex(np.exp(-c * t)))

    for freq in [1, 2, 3, 5, 7, 10, 15, 20]:
        for alpha in [0.3, 0.5, 0.7]:
            a = (1.0 + alpha * np.cos(2.0 * np.pi * freq * t)) ** 2
            inits.append(_proj_simplex(a))

    for _ in range(4):
        a = 1.0
        nc = min(15, n // 3)
        for k in range(1, nc + 1):
            amp = rng.uniform(0.1, 0.4) / (k ** 0.6)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t + ph)
        a = np.maximum(a, 0)
        s = float(a.sum())
        if s > 0:
            inits.append(a / s)

    for beta in [0.3, 0.5, 1.0, 2.0, 3.0]:
        a = 0.5 * (1.0 + np.cos(np.pi * beta * (2.0 * t - 1.0)))
        a = np.maximum(a, 1e-10)
        inits.append(_proj_simplex(a))

    # Bimodal
    for center_spread in [0.2, 0.3, 0.4]:
        a = np.exp(-((t - 0.25) / center_spread) ** 2) + np.exp(-((t - 0.75) / center_spread) ** 2)
        inits.append(_proj_simplex(a))

    return inits

    """
    seed = int(seed)
    budget_s = float(max(60.0, budget_s))
    start = time.monotonic()
    deadline = start + max(60.0, budget_s - 40.0)

    rng = np.random.default_rng(seed)

    prev_a = None
    try:
        h = height_sequence_1
        if isinstance(h, list):
            arr = np.asarray(h, dtype=np.float64)
        else:
            arr = np.asarray(h, dtype=np.float64)
        arr = arr.ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.maximum(arr, 0)
            s = float(arr.sum())
            if s > 0.01:
                prev_a = arr
    except Exception:
        prev_a = None

    best_a = None
    best_score = np.inf

    if prev_a is not None:
        best_a = prev_a / float(prev_a.sum())
        best_score = _score(best_a)
        print("Previous score: %.8f (n=%d)" % (best_score, len(best_a)))

    stage_defs = [
        (2048, 0.45),
        (4096, 0.35),
        (8192, 0.20),
    ]

    for n, frac in stage_defs:
        now = time.monotonic()
        if now + 30.0 >= deadline:
            break

        stage_time = (deadline - start) * frac
        stage_deadline = min(deadline, now + stage_time)
        m = _next_pow2(2 * n - 1)
        L = 2 * n - 1

        print("\n=== Stage n=%d, budget=%.1fs ===" % (n, stage_time))

        # Spectral initialization
        tL = np.linspace(0, 1, L)
        edge_dist = np.abs(tL - 0.5) * 2.0
        inits = _make_inits(n, rng, best_a)

        spectral_inits = []
        a0 = inits[0] if inits else np.ones(n) / n

        n_spec = 6 if n <= 4096 else 4
        spec_iters = 40 if n <= 4096 else 25
        total_spec_time = min(60.0, stage_time * 0.15)
        per_spec = total_spec_time / n_spec
        spec_deadline = time.monotonic() + total_spec_time

        targets = []
        for ff in [1.0, 0.7, 0.4]:
            for eb in [0.0, 0.5, 1.0]:
                base = 1.0 + eb * (edge_dist ** 1.5)
                bm = float(base.max())
                if bm > 0:
                    base = base / bm
                target = ff * base + (1.0 - ff) * (np.ones(L) / L)
                target = np.maximum(target, 0)
                tm = float(target.max())
                if tm > 0:
                    target = target / tm
                targets.append(target)

        for idx, target in enumerate(targets[:n_spec]):
            if time.monotonic() >= spec_deadline:
                break
            a_spec, sc_spec = _spectral_init(n, m, target, a0, iters=spec_iters, blend=0.5)
            spectral_inits.append(a_spec)
            print("  Spectral %d: score=%.6f" % (idx, sc_spec))

        all_inits = inits + spectral_inits

        # Score and select diverse candidates
        scored = []
        seen = []
        for a in all_inits:
            a = _proj_simplex(a)
            sc = _score_m(a, m)
            if not np.isfinite(sc):
                continue
            if sc >= best_score and best_score < 2.0:
                continue
            diverse = True
            for b in seen:
                d = float(np.sqrt(np.sum((a - b) ** 2)))
                if d < 0.03:
                    diverse = False
                    break
            if diverse:
                scored.append((sc, a))
                seen.append(a)
                if len(seen) >= 8:
                    break

        scored.sort(key=lambda x: x[0])

        n_cands = min(4, len(scored))
        cands = [scored[i][1] for i in range(n_cands)]
        if not cands:
            cands = [np.ones(n) / n]

        for ci in range(len(cands)):
            print("  Candidate %d init score: %.6f" % (ci, _score_m(cands[ci], m)))

        # Time allocation for optimization
        opt_deadline = stage_deadline - 5.0
        n_opt = len(cands)
        
        if n_opt >= 4:
            weights = [0.40, 0.30, 0.20, 0.10]
        elif n_opt == 3:
            weights = [0.50, 0.30, 0.20]
        elif n_opt == 2:
            weights = [0.60, 0.40]
        else:
            weights = [1.0]

        for ci in range(n_opt):
            now = time.monotonic()
            if now + 15.0 >= opt_deadline:
                break
            rem = opt_deadline - now
            alloc = rem * weights[ci]
            alloc = max(20.0, min(alloc, 900.0))

            a_opt, sc_opt = _optim_nesterov(cands[ci], alloc, int(rng.integers(0, 2**31 - 1)), m)

            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt
                print("  ** New best: %.8f (n=%d, cand %d)" % (best_score, n, ci))
            else:
                print("  Cand %d: %.6f (no improvement, best=%.6f)" % (ci, sc_opt, best_score))

        # Extended run on best if time permits
        now = time.monotonic()
        if now + 30.0 < opt_deadline:
            rem = opt_deadline - now
            if best_a is not None and len(best_a) == n:
                print("  Extended run for %.1fs..." % rem)
                a_ext, sc_ext = _optim_nesterov(best_a, rem, int(rng.integers(0, 2**31 - 1)), m)
                if np.isfinite(sc_ext) and sc_ext < best_score:
                    best_score = sc_ext
                    best_a = a_ext
                    print("  ** New best after extended: %.8f" % best_score)
            elif best_a is not None:
                # Interpolate best to current n and optimize
                x_old = np.linspace(0, 1, len(best_a))
                x_new = np.linspace(0, 1, n)
                a_interp = np.interp(x_new, x_old, best_a)
                a_interp = _proj_simplex(a_interp)
                print("  Extended run (interpolated) for %.1fs..." % rem)
                a_ext, sc_ext = _optim_nesterov(a_interp, rem, int(rng.integers(0, 2**31 - 1)), m)
                if np.isfinite(sc_ext) and sc_ext < best_score:
                    best_score = sc_ext
                    best_a = a_ext
                    print("  ** New best after extended: %.8f" % best_score)

    # Try one more length if time permits
    now = time.monotonic()
    if now + 120.0 < deadline and best_a is not None:
        n_cur = len(best_a)
        for n_try in [n_cur * 2, n_cur // 2]:
            if n_try < 512 or n_try > 16384:
                continue
            now = time.monotonic()
            if now + 60.0 >= deadline:
                break
            rem = deadline - now - 10.0
            if rem < 40:
                break
            print("\n=== Trying n=%d ===" % n_try)
            m_try = _next_pow2(2 * n_try - 1)
            x_old = np.linspace(0, 1, n_cur)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_a)
            a_try = _proj_simplex(a_try)
            a_opt, sc_opt = _optim_nesterov(a_try, rem, int(rng.integers(0, 2**31 - 1)), m_try)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt
                print("  ** New best: %.8f (n=%d)" % (best_score, n_try))
            else:
                print("  n=%d: %.6f (no improvement)" % (n_try, sc_opt))

    # Final validation
    if best_a is None:
        best_a = np.ones(2048, dtype=np.float64) / 2048

    best_a = _proj_simplex(best_a)
    fft_score = _score(best_a)

    try:
        if len(best_a) <= 10000:
            exact = evaluate_sequence([float(x) for x in best_a])
            print("\nFinal: FFT=%.8f exact=%.8f n=%d" % (fft_score, exact, len(best_a)))
        else:
            print("\nFinal: FFT=%.8f n=%d (exact skipped)" % (fft_score, len(best_a)))
    except Exception as e:
        print("Validation: %s" % str(e))

    print("Total time: %.1fs" % (time.monotonic() - start))
    return [float(x) for x in best_a]
