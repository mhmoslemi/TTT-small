Strategy: Minimize 2*n*max(auto_conv) / sum^2 for non-negative sequences.
Uses adaptive soft-max gradient descent in log-space, spectral equalization,
and peak-directed corrections. Multiple lengths and initializations explored.

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones_like(a) / a.size
    return a / s


def _conv_and_score(a):
    n = len(a)
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    c = np.fft.ifft(F * F).real[:2 * n - 1]
    c = np.maximum(c, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return c, np.inf
    cmax = c.max()
    sc = float(2.0 * n * cmax / (s * s))
    return c, sc


def _score(a):
    _, sc = _conv_and_score(a)
    return sc


def _log_softmax(v):
    v = v - v.max()
    e = np.exp(np.clip(v, -700, 700))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(v) / v.size
    return e / s


def _interp_to_length(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones(n) / n
    a = a / s
    if len(a) == n:
        return a
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss <= 0:
        out = np.ones(n) / n
    else:
        out = out / ss
    return out


def _make_inits(n, prev_a, rng):

    inits = []
    
    # From previous best
    if prev_a is not None:
        inits.append(_proj(_interp_to_length(prev_a, n)))
    
    # Uniform

    # Cosine bumps at various positions
    t = np.linspace(0, 1, n)
    for center in (0.3, 0.5, 0.7):
        for width in (0.2, 0.4, 0.6):
            a = np.exp(-((t - center) / width) ** 2)
            inits.append(_proj(a))
    
    # Power-law edges
    i = np.arange(n, dtype=float)
    for alpha in (0.5, 1.0, 1.5, 2.0):
        a = (i + 1) ** alpha
        inits.append(_proj(a))
        a = (n - i) ** alpha
        inits.append(_proj(a))
        a = (i + 1) ** 0.5 * (n - i) ** 0.5
        inits.append(_proj(a))
    
    # Fourier-based
    for nharms in (3, 5, 7, 10):
        a = np.ones(n)
        for k in range(1, nharms + 1):
            amp = 0.3 / (k ** 0.5)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + ph)
        inits.append(_proj(a))
    
    # Sine-squared
    for freq in (1, 2, 3, 4):
        a = np.sin(np.pi * t * freq) ** 2 + 0.1
        inits.append(_proj(a))
    
    return inits

    Adam optimization in log-space with soft-max loss.
    v is the log-parameter; p = softmax(v).
    """
    rng = np.random.default_rng(seed)
    n = int(n)
    
    v = v_init.copy().astype(np.float64)
    v = v - v.mean()
    
    m_conv = _next_pow2(2 * n - 1)
    m_grad = _next_pow2(3 * n)
    
    p = _log_softmax(v)
    best_p = p.copy()
    best_sc = _score(p)
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    mu = np.zeros(n)
    nu = np.zeros(n)
    t_step = 0
    
    start = time.monotonic()
    deadline = start + max_time
    last_improve = start
    check_int = max(10, min(100, 2000 // n))
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Adaptive hyperparameters
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_start + (lr_end - lr_start) * progress
        # Add some randomness to tau
        if rng.random() < 0.1:
            tau *= rng.uniform(0.5, 2.0)
        
        # Compute convolution
        F = np.fft.fft(p, m_conv)
        c = np.fft.ifft(F * F).real[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cmax = c.max()
        if not np.isfinite(cmax) or cmax <= 0:
            break
        
        # Soft-max weights with top-K boost
        tau_abs = tau * cmax + 1e-15
        x = c / tau_abs
        x = x - x.max()
        w = np.exp(np.clip(x, -60, 60))
        
        # Top-K boost
        k = max(1, int(top_k_frac * (2 * n - 1)))
        if k < 2 * n - 1:
            idx = np.argpartition(c, -k)[-k:]
            w[idx] *= 3.0
            w[idx : idx[0] + 1]  # sanity
        ws = w.sum()
        if ws <= 0 or not np.isfinite(ws):
            break
        w = w / ws
        
        # Gradient: g_p[i] = 2 * sum_k w[k] * p[k-i] / tau_abs
        # = 2 * (w ⊛ p)[i] / tau_abs  where ⊛ is cross-correlation
        # cross-correlation of w and p = convolve(w, p_reversed)
        p_rev = p[::-1]
        W_f = np.fft.fft(w, m_grad)
        PR_f = np.fft.fft(p_rev, m_grad)
        g_p = np.fft.ifft(W_f * PR_f).real[n - 1 : 2 * n - 1]
        g_p = 2.0 * g_p / tau_abs
        
        # Convert to v-gradient: g_v = p * (g_p - p @ g_p)
        g_mean = np.dot(p, g_p)
        g_v = p * (g_p - g_mean)
        
        # Normalize
        g_norm = np.linalg.norm(g_v)
        if g_norm < 1e-14:
            v += rng.normal(0, 0.1, n)
            v = v - v.mean()
            continue
        g_v = g_v / g_norm
        
        # Adam update
        t_step += 1
        mu = beta1 * mu + (1 - beta1) * g_v
        nu = beta2 * nu + (1 - beta2) * g_v ** 2
        mu_hat = mu / (1 - beta1 ** t_step)
        nu_hat = nu / (1 - beta2 ** t_step)
        
        v = v - lr * mu_hat / (np.sqrt(nu_hat) + eps)
        v = v - v.mean()
        v = np.clip(v, -30, 30)
        
        p = _log_softmax(v)
        
        # Check improvement
        if t_step % check_int == 0:
            sc = _score(p)
            if sc < best_sc - 1e-9:
                best_sc = sc
                best_p = p.copy()
                last_improve = time.monotonic()
        
        # Stuck detection
        if time.monotonic() - last_improve > max(15, 0.25 * max_time):
            sigma = 0.1 + 0.3 * (1 - progress)
            v += rng.normal(0, sigma, n)
            v = v - v.mean()
            v = np.clip(v, -30, 30)
            last_improve = time.monotonic()
    
    p = _log_softmax(v)
    sc = _score(p)
    if sc < best_sc:
        best_sc = sc
        best_p = p
    
    return _proj(best_p), float(best_sc)


def _spectral_equalize(p, n, strength=0.3):
    """Project toward flat power spectrum in frequency domain."""
    m = _next_pow2(2 * n - 1)
    P = np.fft.fft(p, m)
    mag = np.abs(P)
    
    # Target: flat power spectrum (constant |P|)
    target_mag = np.ones_like(mag)
    # Scale to match current energy
    current_energy = np.sum(mag ** 2)
    target_energy = np.sum(target_mag ** 2)
    if target_energy > 0:
        target_mag = target_mag * np.sqrt(current_energy / target_energy)
    
    # Blend
    phase = P / (mag + 1e-30)
    new_mag = mag * (1 - strength) + target_mag * strength
    P_new = phase * new_mag
    
    p_new = np.fft.ifft(P_new).real[:n]
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s <= 0:
        return p.copy()
    return p_new / s


def _run_stage(n, prev_a, max_time, seed):
    """Run optimization for a given length n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max_time
    
    best_a = None
    best_sc = np.inf
    
    if prev_a is not None:
        best_a = _proj(_interp_to_length(prev_a, n))
        best_sc = _score(best_a)
        print(f"  n={n}: initial score = {best_sc:.6f}")
    
    inits = _make_inits(n, prev_a, rng)
    
    # Score and sort initializations
    scored = []
    for a in inits:
        sc = _score(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda x: x[0])
    
    # Take top 4 diverse inits
    top_inits = []
    seen = []
    for sc, a in scored:
        diverse = True
        for b in seen:
            if np.linalg.norm(a - b) < 0.05:
                diverse = False
                break
        if diverse:
            top_inits.append(a
