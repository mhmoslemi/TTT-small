**Structural insight:**

"""
Minimize 2n*max(a*a)/(sum(a))^2 using FISTA with exact subgradient on the
probability simplex. Start from height_sequence_1 and polish with multiple
strategies: exact-subgradient FISTA, smooth-surrogate FISTA with annealing,
and random-restart local search.

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv(a, k):
    """Gradient of (a*a)_k w.r.t. a: result[i] = 2*a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    # Valid range: 0 <= k-i < n => k-n+1 <= i <= k
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        g[lo:hi] = 2.0 * a[k - np.arange(lo, hi, dtype=np.int64)]
    return g


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def normalize_simplex(a):
    """Normalize a to the probability simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fista_exact_subgrad(a0, duration, step=0.4, check_interval=20, verbose=False):
    """FISTA with exact subgradient of max_k (a*a)_k on the simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a_prev = a0.copy().astype(np.float64)
    t = 1.0
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        
        c_y = np.convolve(y, y)
        k_max = int(np.argmax(c_y))
        g = grad_conv(y, k_max)
        
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        if it % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if verbose and it % 10000 == 0:
                print(f"    FISTA it={it}, score={best_score:.8f}, no_improve={no_improve}")
            
            if no_improve > 500:
                # Restart with perturbation
                a = best_a + np.random.randn(n) * 0.001
                a = normalize_simplex(a)
                a_prev = a.copy()
                t = 1.0
                no_improve = 0
                if verbose:
                    print(f"    FISTA restart at it={it}, score={best_score:.8f}")
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

    """FISTA with smooth log-sum-exp surrogate, annealing tau."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a_prev = a0.copy().astype(np.float64)
    t = 1.0
    best_a = a0.copy()
    best_score = fast_score(a0)
    start_time = time.monotonic()
    deadline = time.monotonic() + duration
    it = 0
    
    while time.monotonic() < deadline:
        # Anneal tau
        frac = (time.monotonic() - start_time) / max(duration, 1e-6)
        tau = tau_start + (tau_end - tau_start) * frac
        # Step size inversely proportional to tau
        step_cur = max(0.001, 0.5 / tau * 0.5)  # conservative
        
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        
        c_y = np.convolve(y, y)

        tau_c = tau * c_y
        tau_c -= tau_c.max()
        p = np.exp(tau_c)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(c_y)
            p[int(np.argmax(c_y))] = 1.0
        else:
            p /= p_sum

The standard "restarted FISTA" (Teboulle) works as follows:
- After each step, if $f(a_{t+1}) \leq f(a_t)$, accept the step and continue with momentum.
- If $f(a_{t+1}) > f(a_t)$, reject the step (set $a_{t+1} = a_t$) and reset the momentum ($t = 1$, $a_{prev} = a_t$).

This guarantees monotone decrease of the function value.

"""
Minimize 2n*max(a*a)/(sum(a))^2 using restarted FISTA (exact subgradient)
on the probability simplex, with smooth-surrogate polishing and multiple
step-size trials. Starts from height_sequence_1 and iteratively refines.
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv(a, k):
    """Gradient of (a*a)_k w.r.t. a: result[i] = 2*a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        g[lo:hi] = 2.0 * a[k - np.arange(lo, hi, dtype=np.int64)]
    return g


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 efficiently."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def normalize_simplex(a):
    """Normalize a to the probability simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def fista_restarted(a0, duration, step=0.45, verbose=True):
    """
    Restarted FISTA with exact subgradient on the simplex.
    At each step, computes the subgradient (gradient of the active max),
    takes a projected step, and restarts momentum if the function value increases.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a_prev = a0.copy().astype(np.float64)
    t = 1.0
    best_a = a.copy()
    best_f = np.convolve(a, a).max()
    f_curr = best_f
    deadline = time.monotonic() + duration
    it = 0

    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy

        # Compute gradient at y
        c_y = np.convolve(y, y)
        k_max = int(np.argmax(c_y))
        g = grad_conv(y, k_max)

        # Trial step
        a_trial = y - step * g
        a_trial = project_simplex(a_trial)

        # Evaluate trial (one more convolution)
        c_trial = np.convolve(a_trial, a_trial)
        f_trial = c_trial.max()

        if f_trial <= f_curr:
            a_prev = a
            a = a_trial
            f_curr = f_trial
            if f_curr < best_f:
                best_f = f_curr
                best_a = a.copy()
                if verbose and best_f * 2 * n < 1.503:
                    print(f"    NEW BEST: score={2*n*best_f:.8f} at it={it}")
        else:
            # Reject: reset momentum
            a_prev = a.copy()
            t = 1.0

        it += 1
        if verbose and it % 5000 == 0:
            print(f"    FISTA it={it}, best_score={2*n*best_f:.8f}, "
                  f"remaining={deadline - time.monotonic():.0f}s")

    return best_a, best_f


def fista_smooth_surrogate(a0, duration, tau=500.0, step=0.002, verbose=True):
    """
    FISTA with smooth log-sum-exp surrogate for the max.
    Uses the fact that grad logsumexp(tau*c) = sum p_k * grad c_k
    which can be computed as a cross-correlation via FFT.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a_prev = a0.copy().astype(np.float64)
    t = 1.0
    best_a = a.copy()
    best_f = np.convolve(a, a).max()
    deadline = time.monotonic() + duration
    it = 0

    while time.monotonic() < deadline:
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy

        c_y = np.convolve(y, y)

        # Softmax weights
        tc = tau * c_y
        tc -= tc.max()
        p = np.exp(tc)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(c_y)
            p[int(np.argmax(c_y))] = 1.0
        else:
            p /= p_sum

        # Gradient via cross-correlation: 2 * correlate(p, y)[:n]
        h = np.correlate(p, y, mode='full')[:n]
        g = 2.0 * h

        a_trial = y - step * g
        a_trial = project_simplex(a_trial)

        c_trial = np.convolve(a_trial, a_trial)
        f_trial = c_trial.max()

            a = a_trial

                best_a = a.copy()
        else:

            t = 1.0
