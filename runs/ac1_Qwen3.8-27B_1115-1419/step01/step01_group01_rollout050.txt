def _proj_simplex(v):
    v=np.asarray(v,float)
    if v.size==0: return v
    u=sorted(v, reverse=True)
    cssum=np.cumsum(u)
    rho=np.nonzero(u*(np.arange(1,len(u)+1)) > (cssum-1))[0][0]

```python
"""
Search for non-negative step heights minimizing
    score = 2*n*max(np.convolve(a,a)) / sum(a)^2.

The program combines:
  * interpolation of the supplied previous best construction,
  * cheap deterministic transforms (reversal blends, power warps, index warps),
  * L-BFGS optimization in softmax/log space with annealed log-sum-exp
    surrogates for the maximum convolution coefficient,
  * an Adam-style smooth top-peak optimizer as a finisher,
  * progressive lengths (2048, 4096, 8192) to reduce finite-length effects.
"""

import time
import numpy as np
from scipy.optimize import minimize


class _TimeLimit(Exception):
    pass


_G = {}


def _pair_score(pair):
    return pair[0]


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj_nonneg_sum(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def _conv_self_fft(a, m=None):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if m is None or int(m) < (2 * n - 1):
        m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    c = np.fft.ifft(F * F).real[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.inf
    a = a / s
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    c = np.fft.ifft(F * F).real[:2 * n - 1]
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.inf
    return float(2.0 * n * cmax)


def _cross_corr(a, w, m=None):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if m is None or int(m) < (2 * n - 1):
        m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    G = np.fft.fft(w, m)
    return np.fft.ifft(F * np.conj(G)).real[:n]


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    n = int(z.size)
    if n == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    z = z - float(np.mean(z))
    z = z - float(np.max(z))
    z = np.clip(z, -745.0, 709.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def _z_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    z = np.log(a)
    z = z - float(np.mean(z))
    return z


def _interp(a, n):
    a = np.asarray(a, dtype=np.float64)
    n = int(max(1, n))
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = a / s
    if a.size == n:
        return a.copy()
    xo = np.linspace(0.0, 1.0, a.size)
    xn = np.linspace(0.0, 1.0, n)
    out = np.interp(xn, xo, a)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return out / ss


def _interp_nearest(a, n):
    a = np.asarray(a, dtype=np.float64)
    n = int(max(1, n))
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = a / s
    idx = np.floor((np.arange(n, dtype=np.float64) + 0.5) * float(a.size) / float(n)).astype(int)
    idx = np.clip(idx, 0, a.size - 1)
    out = a[idx]
    ss = float(out.sum())
    if ss <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return out / ss


def _append_init(res, a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0.0 and np.isfinite(s):
        res.append(a / s)


def _param_inits(n, rng):
    n = int(max(3, n))
    res = []
    x = np.linspace(0.0, 1.0, n)
    _append_init(res, np.ones(n, dtype=np.float64))

    # Low-frequency asymmetric envelopes.  The reversal-symmetric case is not
    # useful here; asymmetric shapes are what can beat the symmetric lower bound.
    for e1 in (-0.6, -0.4, -0.2, 0.2, 0.4, 0.6):
        for e2 in (-0.3, 0.0, 0.3):
            a = 1.0 + e1 * np.cos(np.pi * x) + e2 * np.cos(2.0 * np.pi * x)
            _append_init(res, a)

    for c in (0.3, 0.7, 1.5, 3.0, 6.0):
        _append_init(res, np.exp(-c * x))
        _append_init(res, np.exp(-c * (1.0 - x)))
        _append_init(res, np.exp(-c * x - 0.2 * c * (1.0 - x)))
        _append_init(res, np.exp(-0.2 * c * x - c * (1.0 - x)))

    left = x + 1e-6
    right = 1.0 - x + 1e-6
    for p, q in ((0.0, -0.3), (-0.3, 0.0), (0.3, -0.3), (-0.3, 0.3),
                 (0.5, -0.2), (-0.2, 0.5), (0.2, -0.5), (-0.5, 0.2)):
        a = (left ** p) * (right ** q)
        _append_init(res, a)

    kmax = max(1, min(12, n // 2))
    for _ in range(8):
        a = np.ones(n, dtype=np.float64)
        ks = rng.choice(np.arange(1, kmax + 1, dtype=np.int64),
                        size=min(4, kmax), replace=False)
        for k in ks:
            amp = float(rng.uniform(0.1, 0.5))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * k * x + ph)
            a += 0.7 * amp * np.sin(2.0 * np.pi * k * x + 1.3 * ph)
        _append_init(res, a)

    return res


def _quick_transforms(a, n):
    a = np.asarray(a, dtype=np.float64)
    n = int(max(3, a.size))
    out = []

    ar = a[::-1]
    for t in (0.15, 0.35, 0.5, 0.65, 0.85):
        b = (1.0 - t) * a + t * ar
        _append_init(out, b)

    for beta in (0.75, 0.9, 1.1, 1.3):
        b = np.maximum(a, 0.0) ** beta
        _append_init(out, b)

    x = np.linspace(0.0, 1.0, n)
    for p in (0.85, 1.15):
        idx = np.clip(np.floor((x ** p) * float(n - 1)).astype(int), 0, n - 1)
        _append_init(out, a[idx])

    for k, ph in ((3, 0.5), (5, 1.3), (9, 2.1)):
        b = a * (1.0 + 0.18 * np.cos(2.0 * np.pi * k * x + ph))
        _append_init(out, b)

    return out


def _lbfgs_objective(z):
    n = int(_G.get("n", 0))
    tau = float(_G.get("tau", 1.0))
    m = int(_G.get("m", _next_pow2(2 * n - 1) if n else 1))

    z = np.asarray(z, dtype=np.float64)
    if z.size != n or not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
        if z.size != n:
            z = np.zeros(n, dtype=np.float64)

    zz = z - float(np.mean(z))
    zz = zz - float(np.max(zz))
    zz = np.clip(zz, -745.0, 709.0)
    ez = np.exp(zz)
    se = float(ez.sum())
    if se <= 0.0 or not np.isfinite(se):
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
    else:
        a = ez / se

    F = np.fft.fft(a, m)
    c = np.fft.ifft(F * F).real[:2 * n - 1]
    c = np.maximum(c, 0.0)

    y = c * float(n)
    x = y / tau
    xmax = float(np.max(x))
    ex = np.exp(x - xmax)
    sex = float(ex.sum())
    if sex <= 0.0 or not np.isfinite(sex):
        ex = np.ones_like(x)
        sex = float(x.size)

    f = xmax + tau * np.log(sex)
    p = ex / sex

    G = np.fft.fft(p, m)
    grad_c = 2.0 * np.fft.ifft(F * np.conj(G)).real[:n]
    grad_y = float(n) * grad_c
    grad_f = grad_y / tau
    dot = float(np.dot(a, grad_f))
    grad = grad_f - a * dot

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    if not np.isfinite(f):
        f = 1e100

    _G["calls"] = int(_G.get("calls", 0)) + 1
    sc = float(2.0 * n * np.max(c))
    if np.isfinite(sc) and sc < float(_G.get("best_score", np.inf)):
        _G["best_score"] = sc
        _G["best_a"] = a.copy()
        _G["last_z"] = z.copy()
    elif _G["calls"] % 20 == 0:
        _G["last_z"] = z.copy()

    if time.monotonic() > float(_G.get("deadline", 0.0)):
        raise _TimeLimit

    return f, grad


def _optimize_lbfgs(a, n, seconds, seed):
    n = int(max(3, n))
    a = _proj_nonneg_sum(a)
    best_a = a.copy()
    best_sc = _score_fft(a)
    if not np.isfinite(best_sc):
        best_a = np.ones(n, dtype=np.float64) / float(n)
        best_sc = _score_fft(best_a)

    start = time.monotonic()
    end = start + max(1.0, float(seconds))
    z = _z_from_a(best_a)

    # Absolute tau is scaled by current best max-y level, which is score/2.
    # This keeps annealing meaningful when starting from bad candidates.
    tau_bases = (1.0, 0.6, 0.35, 0.20, 0.12, 0.07, 0.04)
    weights = (0.12, 0.18, 0.20, 0.18, 0.15, 0.10, 0.07)

    factor = max(0.05, min(5.0, float(best_sc) * 0.5))
    cum = 0.0
    for i, tb in enumerate(tau_bases):
        cum += float(weights[i])
        tau_deadline = min(end, start + float(seconds) * cum)
        if tau_deadline - time.monotonic() < 2.0:
            break

        _G["n"] = int(n)
        _G["m"] = int(_next_pow2(2 * n - 1))
        _G["tau"] = float(tb) * factor
        _G["deadline"] = float(tau_deadline)
        _G["best_score"] = float(best_sc)
        _G["best_a"] = best_a.copy()
        _G["calls"] = 0
        _G["last_z"] = z.copy()

        try:
            res = minimize(
                _lbfgs_objective,
                z,
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": 3000,
                    "maxls": 30,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                },
            )
            z_new = np.asarray(res.x, dtype=np.float64) if (res is not None and hasattr(res, "x")) else z
        except _TimeLimit:
            z_new = _G.get("last_z", z)
        except Exception:
            z_new = z

        if z_new is not None and z_new.size == n and np.all(np.isfinite(z_new)):
            z_new = z_new - float(np.mean(z_new))
            if float(np.max(np.abs(z_new))) > 40.0:
                z_new = np.clip(z_new, -40.0, 40.0)
            a_new = _softmax_from_z(z_new)
            sc_new = _score_fft(a_new)
            if np.isfinite(sc_new) and sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
            z = z_new

        if float(_G.get("best_score", np.inf)) < best_sc:
            best_sc = float(_G["best_score"])
            ba = _G.get("best_a")
            if ba is not None and np.all(np.isfinite(ba)):
                best_a = np.asarray(ba, dtype=np.float64).copy()
                z = _z_from_a(best_a)

        if time.monotonic() >= end:
            break

    return _proj_nonneg_sum(best_a), float(best_sc)


def _soft(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    v = v - float(np.max(v))
    v = np.clip(v, -745.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / float(max(1, v.size)), dtype=np.float64)
    return e / s


def _score_from_F(a, F, m):
    n = int(a.size)
    L = 2 * n - 1
    b = np.fft.ifft(F * F).real[:L]
    b = np.maximum(b, 0.0)
    S = float(a.sum())
    if S <= 1e-300 or not np.isfinite(S):
        return np.inf
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _optimize_v(a_init, n, max_seconds, seed):
    """
    Adam-style optimizer in log/softmax space using annealed smooth top-peak
    weights.  This is a robust finisher for L-BFGS candidates.
    """
    n = int(max(3, n))
    max_seconds = float(max(1.0, max_seconds))
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    v = _z_from_a(_proj_nonneg_sum(a_init))
    v = v - float(np.mean(v))
    v = np.clip(v, -25.0, 25.0)

    m = _next_pow2(2 * n - 1)
    a = _soft(v)
    best_a = a.copy()
    best_score = _score_fft(a)
    if not np.isfinite(best_score):
        best_a = np.ones(n, dtype=np.float64) / float(n)
        v = _z_from_a(best_a)
        best_score = _score_fft(best_a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0

    start = time.monotonic()
    deadline = start + max_seconds
    last_improve = start
    check_interval = max(2, min(18, 1800 // max(1, n)))

    while time.monotonic() < deadline:
        elapsed = max(1e-9, time.monotonic() - start)
        progress = min(1.0, elapsed / max_seconds)

        stuck_limit = min(90.0, 0.35 * max_seconds)
        if time.monotonic() - last_improve > stuck_limit:
            sigma = 0.10 + 0.35 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            last_improve = time.monotonic()
            continue

        a = _soft(v)
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break

        tau_frac = 0.18 * (1.0 - progress) + 0.03
        if rng.random() < 0.25:
            tau_frac *= float(rng.uniform(0.3, 3.0))
        tau = max(1e-15, bmax * max(tau_frac, 1e-4))

        x = (b - bmax) / tau
        x = np.clip(x, -60.0, 60.0)
        x = x - float(x.max())
        w = np.exp(x)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            break
        w = w / ws

        blen = int(b.size)
        top_k = max(1, int(0.05 * blen))
        if top_k < blen:
            idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            idx = np.arange(blen)
        w[idx] += 0.5
        ws = float(w.sum())
        w = w / ws

        c = 2.0 * _cross_corr(a, w, m)
        np.maximum(c, 0.0, out=c)

        cm = float(np.dot(c, a))
        gv = a * (c - cm)
        grms = float(np.sqrt(np.mean(gv * gv)))
        if not np.isfinite(grms) or grms < 1e-14:
            sigma = 0.05 + 0.20 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            continue

        gv = gv / grms
        t += 1
        mu = beta1 * mu + (1.0 - beta1) * gv
        nu = beta2 * nu + (1.0 - beta2) * (gv * gv)

        b1 = beta1 ** t
        b2 = beta2 ** t
        denom1 = max(1e-12, 1.0 - b1)
        denom2 = max(1e-12, 1.0 - b2)
        mhat = mu / denom1
        vhat = nu / denom2

        lr = 0.10 * (1.0 - 0.65 * progress)
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.5, 0.5)
        v = v - lr * step
        v = v - float(np.mean(v))
        v = np.clip(v, -25.0, 25.0)

        if t % check_interval == 0:
            an = _soft(v)
            Fn = np.fft.fft(an, m)
            sc = _score_from_F(an, Fn, m)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_a = an.copy()
                last_improve = time.monotonic()

    an = _soft(v)
    sc = _score_fft(an)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = an

    return _proj_nonneg_sum(best_a), float(best_score)


def _process_stage(n, stage_deadline, best_a, rng):
    n = int(max(3, n))
    inits = []

    if best_a is not None:
        ba = np.asarray(best_a, dtype=np.float64)
        if ba.size > 0 and np.all(np.isfinite(ba)):
            inits.append(_interp(best_a, n))
            inits.append(_interp_nearest(best_a, n))
            if ba.size == n:
                inits.append(ba / float(max(1.0, ba.sum())) if ba.sum() > 0 else ba)
                inits.extend(_quick_transforms(best_a, n))

    inits.extend(_param_inits(n, rng))

    scored = []
    seen = []
    for a in inits:
        a = _proj_nonneg_sum(a)
        if a.size != n:
            continue
        sc = _score_fft(a)
        if not np.isfinite(sc):
            continue
        diverse = True
        for b in seen:
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < 0.035:
                diverse = False
                break
        if diverse:
            scored.append((sc, a))
            seen.append(a)
            if len(seen) >= 24:
                break

    scored.sort(key=_pair_score)
    top = []
    for _, a in scored:
        top.append(a)
        if len(top) >= 3:
            break
    if not top:
        top.append(np.ones(n, dtype=np.float64) / float(n))

    stage_start = time.monotonic()
    stage_len = max(1.0, float(stage_deadline) - stage_start)

    stage_best_a = best_a.copy() if (best_a is not None and np.all(np.isfinite(best_a))) else top[0].copy()
    stage_best_sc = _score_fft(stage_best_a) if best_a is not None else np.inf
    if not np.isfinite(stage_best_sc):
        stage_best_sc = np.inf

    if len(top) == 1:
        allocs = (1.0,)
    elif len(top) == 2:
        allocs = (0.65, 0.35)
    else:
        allocs = (0.55, 0.28, 0.17)

    for i, a in enumerate(top):
        now = time.monotonic()
        if now >= float(stage_deadline) - 8.0:
            break
        frac = float(allocs[i])
        cand_alloc = max(5.0, stage_len * frac)

        lb_alloc = cand_alloc * 0.55
        v_alloc = cand_alloc * 0.38

        if lb_alloc > 4.0 and time.monotonic() < float(stage_deadline) - 8.0:
            a1, sc1 = _optimize_lbfgs(
                a,
                n,
                max(1.0, lb_alloc - (time.monotonic() - now)),
                int(rng.integers(0, 2**31 - 1)),
            )
            if np.isfinite(sc1) and sc1 < stage_best_sc:
                stage_best_sc = sc1
                stage_best_a = a1.copy()
        else:
            a1 = _proj_nonneg_sum(a)
            sc1 = _score_fft(a1)

        if v_alloc > 4.0 and time.monotonic() < float(stage_deadline) - 8.0:
            a2, sc2 = _optimize_v(
                a1,
                n,
                max(1.0, v_alloc - (time.monotonic() - now)),
                int(rng.integers(0, 2**31 - 1)),
            )
            if np.isfinite(sc2) and sc2 < stage_best_sc:
                stage_best_sc = sc2
                stage_best_a = a2.copy()

    if stage_best_a is not None and time.monotonic() < float(stage_deadline) - 12.0:
        polish_time = max(1.0, float(stage_deadline) - time.monotonic() - 5.0)
        a3, sc3 = _optimize_v(
            stage_best_a,
            n,
            polish_time,
            int(rng.integers(0, 2**31 - 1)),
        )
        if np.isfinite(sc3) and sc3 < stage_best_sc:
            stage_best_sc = sc3
            stage_best_a = a3.copy()

    if stage_best_a is None:
        stage_best_a = np.ones(n, dtype=np.float64) / float(n)
        stage_best_sc = _score_fft(stage_best_a)

    return _proj_nonneg_sum(stage_best_a), float(stage_best_sc)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    seed = int(seed)
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    if not np.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 2400.0
    budget_s = max(120.0, min(2400.0, budget_s))

    total_start = time.monotonic()
    work_budget = max(60.0, budget_s - 25.0)
    global_deadline = total_start + work_budget
    rng = np.random.default_rng(seed)

    # Load parent construction, if available.  Long parents are capped so final
    # exact evaluation by np.convolve remains reasonable.
    parent = None
    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.maximum(arr, 0.0)
            s = float(arr.sum())
            if s > 0.0:
                if arr.size > 8192:
                    arr = _interp(arr, 8192)
                parent = arr / float(arr.sum())
    except Exception:
        parent = None

    best_a = parent
    if best_a is not None:
        best_score = _score_fft(best_a)
    else:
        best_score = np.inf

    print("initial parent score: %.6f" % (best_score if np.isfinite(best_score) else np.inf))

    # Cheap transform scan on the parent length, before staged optimization.
    if best_a is not None and best_a.size <= 8192:
        for a in _quick_transforms(best_a, best_a.size):
            sc = _score_fft(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        print("after quick transforms score: %.6f" % best_score)

    if best_a is None:
        stage_defs = [
            (1024, 0.10),
            (2048, 0.35),
            (4096, 0.30),
            (8192, 0.25),
        ]
    else:
        stage_defs = [
            (2048, 0.38),
            (4096, 0.32),
            (8192, 0.30),
        ]

    cum = 0.0
    for n, frac in stage_defs:
        now = time.monotonic()
        cum += float(frac)
        stage_end = min(global_deadline, total_start + work_budget * cum)
        if now + 35.0 >= stage_end:
            break

        print("stage n=%d deadline_in=%.1fs current_best=%.6f" %
              (n, stage_end - now, best_score if np.isfinite(best_score) else np.inf))

        a_new, sc_new = _process_stage(int(n), float(stage_end), best_a, rng)
        if a_new is not None and np.all(np.isfinite(a_new)) and np.isfinite(sc_new):
            if np.isinf(best_score) or sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()

        print("after stage n=%d best=%.6f" % (n, best_score if np.isfinite(best_score) else np.inf))

        if best_a is not None and best_a.size > 8192:
            best_a = _interp(best_a, 8192)
            best_score = _score_fft(best_a)

    # Final polish on the current global best.
    if best_a is not None and time.monotonic() < global_deadline - 5.0:
        remaining = max(5.0, float(global_deadline) - time.monotonic() - 3.0)
        a_final, sc_final = _optimize_v(
            best_a,
            int(best_a.size),
            remaining,
            int(rng.integers(0, 2**31 - 1)),
        )
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        print("after final polish score: %.6f" % best_score)

    if best_a is None:
        best_a = np.ones(2048, dtype=np.float64) / 2048.0
        best_score = _score_fft(best_a)
    else:
        best_a = _proj_nonneg_sum(best_a)
        if best_a.size > 8192:
            best_a = _interp(best_a, 8192)
            best_score = _score_fft(best_a)
        else:
            sc = _score_fft(best_a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc

    # Optional exact validation for moderate sizes.
    if best_a is not None and best_a.size <= 4096:
        try:
            exact = evaluate_sequence([float(x) for x in best_a])
            if np.isfinite(exact):
                print("exact score: %.6f" % exact)
                best_score = float(exact)
        except Exception:
            pass

    return [float(x) for x in best_a]
```
