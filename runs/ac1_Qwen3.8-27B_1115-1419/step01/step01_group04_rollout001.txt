def fast_convolution(a):
    """FFT-based linear convolution a*a, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c

def fast_score(a):
    """Fast score computation."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size

    s = float(a.sum())
    if s < 0.01:
        return float('inf')

def gradient_topk(a, c, K):
    """Subgradient from top-K convolution peaks."""
    a = np.asarray(a, dtype=np.float64)

"""
Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

Strategy:
- FFT-based convolution for fast evaluation
- Parametric optimization (sum of cosines) for coarse structure
- Full local search with adaptive top-peak subgradient
- Multi-resolution coarse-to-fine
- Simulated annealing with diverse restarts

import numpy as np
import time

from scipy.optimize import minimize as scipy_minimize


def fast_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def fast_score(a):
    n = a.size
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fast_convolution(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def normalize(a, target):
    a = np.array(a, dtype=np.float64, copy=True)
    a[~np.isfinite(a)] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s > 1e-15:
        a *= target / s
    else:
        a.fill(target / max(1, len(a)))
    return np.clip(a, 0.0, 1000.0)


def upsample(seq, n):
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    n = int(n)
    if m == 0:
        return normalize(np.ones(n), float(n))
    if m == n:
        return normalize(seq, float(n))
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize(out, float(n))


def gradient_topk(a, c, K):
    n = a.size
    L = 2 * n - 1
    K = min(K, L)
    d = np.zeros(n, dtype=np.float64)
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    inv = 1.0 / float(K)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        d[lo:hi] += inv * a[k - hi + 1 : k - lo + 1][::-1]
    return 2.0 * d


def parametric_objective(params, n):
    """a_i = 1 + sum_{m=1}^{M} A_m * cos(2*pi*m*i/n + phi_m), clipped to >= 0"""
    M = len(params) // 2
    A = params[:M]
    phi = params[M:]
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for m in range(1, M + 1):
        a += A[m-1] * np.cos(2.0 * np.pi * m * i / n + phi[m-1])
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return 1e6
    a = a / s * n  # normalize to sum = n
    c = fast_convolution(a)
    return float(2.0 * n * c.max() / (n * n))


def parametric_search(n, deadline, M=4):
    """Optimize parametric sequence with M harmonics."""
    rng = np.random.default_rng(123)
    nparams = 2 * M
    best_params = None
    best_score = float('inf')
    
    # Multiple random starts
    for restart in range(30):
        if time.monotonic() >= deadline:
            break
        if restart == 0:
            A = np.array([0.3, 0.15, 0.1, 0.05])[:M]
            phi = np.zeros(M)
        else:
            A = rng.uniform(-0.8, 0.8, M) * np.array([1.0, 0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.04, 0.03])[:M]
            phi = rng.uniform(0, 2*np.pi, M)
        
        x0 = np.concatenate([A, phi])
        try:
            res = scipy_minimize(
                parametric_objective, x0, args=(n,),
                method='Nelder-Mead',
                options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-8}
            )
            sc = res.fun
            if sc < best_score:
                best_score = sc
                best_params = res.x
                print(f"  Parametric restart {restart}: score={sc:.6f}")
        except Exception:
            pass
    
    if best_params is None:
        return None, float('inf')
    
    # Reconstruct
    M2 = len(best_params) // 2
    A = best_params[:M2]

    i = np.arange(n, dtype=np.float64)

    score = fast_score(a)
    return a, score

                    best_score = new_score
