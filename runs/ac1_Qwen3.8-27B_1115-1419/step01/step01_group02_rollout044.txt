"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using smooth L-BFGS-B optimization
with soft-max approximation, multi-start diversity, and adaptive temperature.
Key insight: the gradient of the soft-max smoothed objective can be computed
via FFT in O(n log n), enabling fast optimization for large n.
"""

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def _fft_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n_out = len(a) + len(b) - 1
    N = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, N) * fft(b, N)).real[:n_out]


def _score_fast(a):
    """Fast score using FFT and hard max."""
    n = len(a)
    b = _fft_conv(a, a)
    b = np.maximum(b, 0.0)  # remove FFT numerical artifacts
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _soft_obj_grad(a, T, sum_penalty=1e-5, target_sum=None):
    """
    Returns (objective_value, gradient) for soft-max smoothed score.
    The soft-max approximates the hard max of the convolution.
    """
    n = len(a)
    b = _fft_conv(a, a)
    b = np.maximum(b, 0.0)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return 1e10, np.ones(n)

    # Soft max computation
    max_b = float(np.max(b))
    exp_vals = np.exp(T * (b - max_b))
    sum_exp = float(np.sum(exp_vals))
    soft_max = max_b + np.log(sum_exp) / T
    w = exp_vals / sum_exp

    # Score
    score = 2.0 * n * soft_max / (sum_a * sum_a)

    # Gradient of soft-max w.r.t. a
    # dS/d(a_m) = 2 * sum_j a[j] * w[j+m] for m=0..n-1
    # Computed as 2 * (flip(a) * w)[n-1 : 2n-1]
    flipped_a = a[::-1]
    corr = _fft_conv(flipped_a, w)
    grad_sm = 2.0 * corr[n - 1:2 * n - 1]

    # Full gradient of score
    grad = (2.0 * n / (sum_a * sum_a)) * grad_sm - (4.0 * n * soft_max / (sum_a ** 3))

    # Sum penalty to break scale invariance
    if target_sum is not None and sum_penalty > 0:
        diff = sum_a - target_sum
        score += sum_penalty * diff * diff
        grad += 2.0 * sum_penalty * diff

    return float(score), grad


def _topk_obj_grad(a, k, sum_penalty=1e-5, target_sum=None):
    """
    Returns (objective, gradient) using sum of top-k convolution values.
    Smoother than hard max, rougher than soft-max for large T.
    """
    n = len(a)
    b = _fft_conv(a, a)
    b = np.maximum(b, 0.0)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return 1e10, np.ones(n)

    # Find top-k indices
    if k >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argsort(b)[-k:]

    topk_sum = float(np.sum(b[top_idx]))
    score = 2.0 * n * topk_sum / (sum_a * sum_a)

    # Gradient: 2 * sum_{j in top_k} a_{j-m} for valid j-m
    w = np.zeros(len(b))
    w[top_idx] = 1.0
    flipped_a = a[::-1]
    corr = _fft_conv(flipped_a, w)
    grad_topk = 2.0 * corr[n - 1:2 * n - 1]

    grad = (2.0 * n / (sum_a * sum_a)) * grad_topk - (4.0 * n * topk_sum / (sum_a ** 3))

    if target_sum is not None and sum_penalty > 0:
        diff = sum_a - target_sum
        score += sum_penalty * diff * diff
        grad += 2.0 * sum_penalty * diff

    return float(score), grad


def _run_lbfgsb(a0, obj_type='soft', T=100.0, k=5, target_sum=None,

            return _soft_obj_grad(x, T, target_sum=target_sum)

        def func_grad(x):
            return _topk_obj_grad(x, k, target_sum=target_sum)

        'maxiter': maxiter,
        'ftol': ftol,
        'gtol': gtol,
        'maxls': 50,
        'maxfun': max(2000, maxiter * 10),
    }

    try:
        result = minimize(func_grad, a0.copy(), jac=True, method='L-BFGS-B',
                          bounds=bounds, options=options)
        return result.x, result.fun
    except Exception:
        return a0, np.inf

        result_a, result_val = _run_lbfgsb(a0, obj_type='soft', T=float(T),
                                           target_sum=target_sum, maxiter=500)
        actual_score = float(evaluate_sequence(result_a.tolist()))
        if actual_score < best_score:
            best_score = actual_score
            best_a = result_a.copy()
            print(f"  soft T={T:4d}: {actual_score:.8f} *** IMPROVED ***")
        else:
            print(f"  soft T={T:4d}: {actual_score:.8f}")

    # Top-k objectives
    for k in [1, 2, 3, 5, 10]:
        if time.monotonic() > deadline - 30:
            break
        a0 = best_a.copy()
        result_a, result_val = _run_lbfgsb(a0, obj_type='topk', k=k,
                                           target_sum=target_sum, maxiter=500)
        actual_score = float(evaluate_sequence(result_a.tolist()))
        if actual_score < best_score:
            best_score = actual_score
            best_a = result_a.copy()
            print(f"  topk k={k:3d}: {actual_score:.8f} *** IMPROVED ***")
        else:
            print(f"  topk k={k:3d}: {actual_score:.8f}")

    # Phase 3: Perturb-and-optimize loop
    print(f"\n{'='*60}")
    print(f"Phase 3: Perturb & optimize (T=150), best={best_score:.8f}")
    print(f"{'='*60}")

    T3 = 150.0
    scales = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.005]
    attempt = 0
    no_improve_count = 0
    max_no_improve = 30

    while time.monotonic() < deadline - 30:
        scale = scales[attempt % len(scales)]
        # Different perturbation strategies
        strategy = attempt % 3
        if strategy == 0:
            # Gaussian multiplicative
            a0 = best_a * (1.0 + scale * rng.standard_normal(n))
        elif strategy == 1:
            # Additive with random support
            a0 = best_a.copy()
            mask = rng.random(n) < 0.3
            a0[mask] += scale * target_sum / n * rng.standard_normal(mask.sum())
        else:
            # Coordinate swap perturbation
            a0 = best_a.copy()
            idx = rng.choice(n, size=max(1, n // 10), replace=False)
            idx2 = rng.choice(n, size=len(idx), replace=False)
            swap_amount = a0[idx] * 0.1
            a0[idx] -= swap_amount
            a0[idx2] += swap_amount

        a0 = np.maximum(a0, 0.0)
        a0 = _normalize_to_target(a0, target_sum)

        result_a, result_val = _run_lbfgsb(a0, obj_type='soft', T=T3,
                                           target_sum=target_sum, maxiter=400)
        actual_score = float(evaluate_sequence(result_a.tolist()))
        attempt += 1

        if actual_score < best_score:
            best_score = actual_score
            best_a = result_a.copy()
            no_improve_count = 0
            print(f"  Attempt {attempt:4d} (scale={scale:.3f}, strat={strategy}): "
                  f"{actual_score:.8f} *** IMPROVED ***")
        else:
            no_improve_count += 1
            if attempt % 20 == 0:
                print(f"  Attempt {attempt:4d}: best_attempt={actual_score:.8f}, "
                      f"global_best={best_score:.8f}, no_improve={no_improve_count}")

        # If stuck for too long, try a more aggressive restart
        if no_improve_count >= max_no_improve:
            print(f"  Stuck for {no_improve_count} attempts, trying aggressive restart...")
            # Random restart with structure
            base_profile = 1.0 + 0.5 * rng.random(n)
            a0 = best_a * 0.5 + base_profile * 0.5
            a0 = np.maximum(a0, 0.0)
            a0 = _normalize_to_target(a0, target_sum)
            result_a, _ = _run_lbfgsb(a0, obj_type='soft', T=80,
                                      target_sum=target_sum, maxiter=500)
            actual_score = float(evaluate_sequence(result_a.tolist()))
            if actual_score < best_score:
                best_score = actual_score
                best_a = result_a.copy()
                print(f"  Aggressive restart: {actual_score:.8f} *** IMPROVED ***")
            no_improve_count = 0

    # Phase 4: Final fine-tuning with very high T (close to hard max)
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*60}")
        print(f"Phase 4: Final fine-tuning, best={best_score:.8f}")
        print(f"{'='*60}")

        for T in [300, 500, 800, 1000]:
            if time.monotonic() > deadline - 5:
                break
            result_a, _ = _run_lbfgsb(best_a, obj_type='soft', T=float(T),
                                      target_sum=target_sum, maxiter=300)
            actual_score = float(evaluate_sequence(result_a.tolist()))
            if actual_score < best_score:
                best_score = actual_score
                best_a = result_a.copy()
                print(f"  T={T:4d}: {actual_score:.8f} *** IMPROVED ***")
            else:
                print(f"  T={T:4d}: {actual_score:.8f}")

        # One final top-1 (exact max) via iterative L-BFGS-B
        if time.monotonic() < deadline - 5:
            for k in [1, 2]:
                result_a, _ = _run_lbfgsb(best_a, obj_type='topk', k=k,
                                          target_sum=target_sum, maxiter=400)
                actual_score = float(evaluate_sequence(result_a.tolist()))
                if actual_score < best_score:
                    best_score = actual_score
                    best_a = result_a.copy()
                    print(f"  final topk k={k}: {actual_score:.8f} *** IMPROVED ***")
                else:
                    print(f"  final topk k={k}: {actual_score:.8f}")

    # Final validation and cleanup
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = np.round(best_a, 10)  # clean up floating point noise

    final_score = float(evaluate_sequence(best_a.tolist()))
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {final_score:.8f}")
    print(f"Sequence length: {n}")
    print(f"Avg value: {np.mean(best_a):.6f}, Max: {np.max(best_a):.6f}, Min: {np.min(best_a):.6f}")
    print(f"{'='*60}")

    return [float(x) for x in best_a]
