```python
import cvxpy as cp
import numpy as np

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, non_neg=True)
    # Constraints: (f * g)[k] <= M for all k
    # (f*g)[k] = sum_j f[j] * g[k-j] where 0 <= j < n and 0 <= k-j < n
    constraints = []
    for k in range(2*n - 1):
        # g[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        # and 0 <= j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # (f*g)[k] = sum_{j=j_start}^{j_end} f[j] * g[k-j]
            idx = [k - j for j in range(j_start, j_end + 1)]
            coeffs = [f[j] for j in range(j_start, j_end + 1)]
            constraints.append(sum(c * g[i] for c, i in zip(coeffs, idx)) <= M)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or GLOP, CBC, etc.
    return np.array(g.value)
```

def _top_k_subgradient(a, n, k=5):
    """Compute the average subgradient from the top-k max coefficients."""
    c = _full_conv(a)
    # Get top k indices
    top_k = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for kk in top_k:
        kk = int(kk)
        js = max(0, kk - n + 1)
        je = min(n - 1, kk) + 1
        if js < je:
            ai = kk - np.arange(js, je)
            grad[js:je] += 2.0 * a[ai]
    grad /= k
    return grad, c.max()
