```
def project_simplex(v):
    n = v.size
    if n == 0: return v.copy()
    if n == 1: return np.array([1.0], dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    ar = np.arange(1, n+1, dtype=np.float64)
    cond = u * ar > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0/n, dtype=np.float64)
    out /= s
    return out
```

"""
Hybrid sparse/support-aware search for
    minimize 2*n*max(conv(a,a))/(sum(a))^2,  a_i >= 0.

Strategy:
- Keep the supplied sequence as a dense starting point and polish it with an
  FFT-based smoothed projected gradient method.
- Convert promising dense solutions to low-dimensional weighted supports.
- Construct additional supports with a greedy low-pair-sum heuristic.
- Optimize fixed supports with direct sparse convolutions, a projected-gradient
  subgradient/LSE method, and a KKT-style balance solve on the support matrix.
- Mutate supports (add/remove/swap points) with short local optimizations.
- Return the best sparse or dense sequence found, verified with evaluate_sequence.
"""

import math
import time
import numpy as np

MAX_DIRECT_M = 520
BALANCE_MAX_M = 320


def next_pow2(m):
    r = 1
    while r < m:
        r <<= 1
    return r


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(2, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    a /= s
    return a


def trim_sequence(a, tol=1e-14):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(2, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    nz = np.nonzero(a > tol)[0]
    if nz.size == 0:
        return np.ones(2, dtype=np.float64)
    a = a[int(nz[0]):int(nz[-1]) + 1]
    a[a < 1e-15] = 0.0
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(max(1, a.size), dtype=np.float64)
    a /= s
    return a


def direct_dense_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    ar = np.arange(1, n + 1, dtype=np.float64)
    cond = u * ar > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    out /= s
    return out


def softmax_weights(q, tau):
    q = np.asarray(q, dtype=np.float64)
    L = int(q.size)
    if L == 0:
        return 0.0, np.empty(0, dtype=np.float64)
    M = float(q.max())
    if M <= 0.0:
        return 0.0, np.full(L, 1.0 / L, dtype=np.float64)
    if tau <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(q))] = 1.0
        return M, p
    z = (q - M) / tau
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(q))] = 1.0
        return M, p
    p = ez / s
    F = M + tau * math.log(s)
    if not np.isfinite(F):
        F = M
    return float(F), p.astype(np.float64, copy=False)


def smooth_value(q, tau):
    q = np.asarray(q, dtype=np.float64)
    L = int(q.size)
    if L == 0:
        return 0.0
    M = float(q.max())
    if M <= 0.0:
        return 0.0
    if tau <= 0.0:
        return M
    z = (q - M) / tau
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        return M
    F = M + tau * math.log(s)
    return F if np.isfinite(F) else M


def active_weights(q, gap=0.03, top_k=64, max_active=2048):
    q = np.asarray(q, dtype=np.float64)
    L = int(q.size)
    if L == 0:
        return 0.0, np.empty(0, dtype=np.float64)
    M = float(q.max())
    if M <= 0.0:
        return 0.0, np.full(L, 1.0 / L, dtype=np.float64)
    top_k = max(1, int(top_k))
    max_active = min(max(1, int(max_active)), L)
    if top_k >= L:
        top_idx = np.arange(L)
    else:
        top_idx = np.argpartition(q, -top_k)[-top_k:]
    mask = q >= (M * (1.0 - gap))
    mask[top_idx] = True
    am = int(mask.sum())
    if am <= 0:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(q))] = 1.0
        return M, p
    if am > max_active:
        if max_active >= L:
            keep = np.arange(L)
        else:
            keep = np.argpartition(q, -max_active)[-max_active:]
        mask = np.zeros(L, dtype=bool)
        mask[keep] = True
        am = int(max_active)
    p = np.zeros(L, dtype=np.float64)
    p[mask] = 1.0 / float(am)
    return M, p


def dense_eval(a, N, L):
    fa = np.fft.rfft(a, N)
    q = np.fft.irfft(fa * fa, N)[:L]
    q = np.ascontiguousarray(q)
    np.maximum(q, 0.0, out=q)
    return q, fa


def dense_grad_weights(a, p, fa, N, n):
    p_rev = p[::-1]
    fp = np.fft.rfft(p_rev, N)
    d = np.fft.irfft(fa * fp, N)
    return d[n - 1:2 * n - 1][::-1]


def perturb_dense(a, rng, scale=0.04):
    b = np.asarray(a, dtype=np.float64).copy()
    n = b.size
    if n == 0:
        return b
    mask = rng.random(n) < 0.3
    if mask.any():
        b[mask] *= rng.normal(1.0, 1.0, int(mask.sum())).clip(0.1, 5.0)
    b += rng.normal(0.0, scale / max(1.0, math.sqrt(n)), n)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s > 1e-300:
        b /= s
    else:
        b = a.copy()
    return b


def run_dense_fista(a0, n, duration, rng, beta_start=8.0, beta_end=80.0,
                    exact_mode=False, step=0.22, name="dense"):
    a0 = np.asarray(a0, dtype=np.float64).copy()
    if a0.size != int(n):
        a0 = a0[:int(n)]
        n = int(a0.size)
    if n <= 0:
        return a0, np.inf
    a = normalize_nonnegative(a0)
    N = next_pow2(max(1, 3 * n - 2))
    L = 2 * n - 1

    q0, _ = dense_eval(a, N, L)
    M0 = float(q0.max())
    C0 = 2.0 * n * M0
    best_a = a.copy()
    best_score = C0
    cur_score = C0

    a_prev = a.copy()
    t = 1.0
    base_step = float(step)
    min_step = 1e-5
    max_step = 1.0
    start = time.monotonic()
    deadline = start + max(0.0, float(duration))
    it = 0
    stall = 0
    last_improve = 0

    print(f"[{name}] n={n}, start_score={C0:.8f}, budget={max(0.0, duration):.1f}s", flush=True)

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start
        frac = elapsed / max(1.0, float(duration))
        if frac > 1.0:
            frac = 1.0
        beta = beta_start + (beta_end - beta_start) * frac

        t_old = t
        t = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_old * t_old))
        mom = (t_old - 1.0) / t
        y = a + mom * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = float(y.sum())
        if sy <= 1e-300:
            y = a.copy()
        else:
            y = y / sy

        q_y, fa_y = dense_eval(y, N, L)
        M_y = float(q_y.max())
        C_y = 2.0 * n * M_y
        if C_y < best_score:
            best_score = C_y
            best_a = y.copy()
            last_improve = it

        if exact_mode:
            top_k = max(32, min(2048, n // 24))
            F_y, p = active_weights(q_y, gap=0.03, top_k=top_k, max_active=2048)
            tau = 0.0
        else:
            tau = max(1e-18, M_y / max(1.0, beta))
            F_y, p = softmax_weights(q_y, tau)

        g = dense_grad_weights(y, p, fa_y, N, n)
        g = 2.0 * g

        if exact_mode:
            limit = max(best_score * 1.0005, cur_score * 1.0005)
        else:
            limit = max(best_score * 1.03, cur_score * 1.01, C_y * 1.0005)

        accepted = False
        used_step = 0.0
        a_trial = None
        C_trial = np.inf

        for mult in (1.0, 0.5, 0.25, 0.125):
            bs = base_step * mult
            if bs < min_step:
                break
            z = y - bs * g
            cand = project_simplex(z)
            q_c, _ = dense_eval(cand, N, L)
            M_c = float(q_c.max())
            C_c = 2.0 * n * M_c
            if exact_mode:
                ok = (M_c < M_y - 1e-15) or (C_c < cur_score - 1e-12)
            else:
                F_c = smooth_value(q_c, tau)
                ok = (F_c < F_y - 1e-12) or (C_c < cur_score - 1e-12)
            if ok and C_c <= limit:
                accepted = True
                a_trial = cand
                C_trial = C_c
                used_step = bs
                break

        if accepted:
            a_prev = a.copy()
            a = a_trial
            cur_score = C_trial
            stall = 0
            if used_step < base_step * 0.999:
                base_step = max(min_step, used_step * 1.1)
            elif it % 80 == 0:
                base_step = min(max_step, base_step * 1.04)
        else:
            a_prev = a.copy()
            t = 1.0
            base_step = max(min_step, base_step * 0.75)
            stall += 1
            if stall > 1200:
                a = perturb_dense(a, rng, 0.04)
                a_prev = a.copy()
                t = 1.0
                base_step = 0.22
                stall = 0

        if it % 1000 == 0:
            print(f"[{name}] it={it}, cur={cur_score:.8f}, best={best_score:.8f}, step={base_step:.4f}", flush=True)

    qf, _ = dense_eval(a, N, L)
    Cf = 2.0 * n * float(qf.max())
    if Cf < best_score:
        best_score = Cf
        best_a = a.copy()

    print(f"[{name}] done, best={best_score:.8f}", flush=True)
    return best_a, best_score


def sparse_conv_ps(ps, w, L):
    ww = np.outer(w, w)
    q = np.bincount(ps.ravel(), weights=ww.ravel(), minlength=L)
    if q.size > L:
        q = q[:L]
    q = q.astype(np.float64, copy=False)
    return q


def support_uniform_score(s):
    s = np.asarray(s, dtype=np.int64)
    m = int(s.size)
    if m == 0:
        return np.inf, 0
    s = s - int(s.min())
    n = int(s.max()) + 1
    if n <= 0:
        return np.inf, 0
    ps = s[:, None] + s[None, :]
    counts = np.bincount(ps.ravel(), minlength=2 * n - 1)
    C = float(counts.max())
    score = 2.0 * n * C / (float(m) * float(m))
    return score, n


def greedy_support(S, m, rng, K=48, top_div=0):
    S = int(max(1, S))
    m = int(max(1, min(m, S)))
    if m == 1:
        return np.array([0], dtype=np.int64)
    if m == S:
        return np.arange(S, dtype=np.int64)

    if S <= 2:
        return np.arange(S, dtype=np.int64)

    s_arr = np.array([0, S - 1], dtype=np.int64)
    if m == 2:
        return s_arr

    L = 2 * S - 1
    counts = np.zeros(L, dtype=np.int64)
    for i in range(s_arr.size):
        counts[int(2 * s_arr[i])] += 1
    for i in range(s_arr.size):
        for j in range(i + 1, s_arr.size):
            counts[int(s_arr[i] + s_arr[j])] += 2
    cur_max = int(counts.max())
    used = {int(s_arr[0]), int(s_arr[1])}

    while s_arr.size < m:
