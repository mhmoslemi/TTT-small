   if mode < 0.4:

"""
Iterative FFT-based optimizer for nonnegative sequences minimizing
2*n*max(convolve(a,a))/(sum(a)**2).

The code normalizes sequences to sum n, uses real FFTs for convolution and
gradients, and alternates between:
  - smooth power-mean surrogate gradients,
  - exact quadratic line search along projected gradient directions,
  - multiplicative mirror-descent steps,
  - randomized block/Fourier perturbations to escape symmetric stationary points.

It also briefly warm-starts several sequence lengths before committing most of
the budget to the best candidate.
"""

import numpy as np
import time

_MAX_N = 32768

    ("pow", 4),
    ("top", 32),
    ("pow", 8),
    ("top", 16),
    ("pow", 16),
    ("topw", 32),

    ("top", 64),
    ("pow", 8),
    ("top", 8),
    ("pow", 2),
    ("topw", 16),
)


def _next_pow2(m):
    x = 1
    while x < m:
        x *= 2
    return x


def _normalize_floor(a, n, floor=1e-12):
    """Nonnegative, finite, upper-bounded, normalized to sum n with tiny floor."""
    a = np.asarray(a, dtype=np.float64)
    if a.size != n:
        b = np.zeros(int(n), dtype=np.float64)
        m = min(a.size, int(n))
        if m > 0:
            b[:m] = a.reshape(-1)[:m]
        a = b

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, float(floor))

    s = float(np.sum(a))
    if s < 1e-300:
        a = np.ones(int(n), dtype=np.float64)
    else:
        a = a * (float(n) / s)

    if n > 0 and a.size:
        if float(np.max(a)) > 1000.0:
            for _ in range(10):
                a = np.minimum(a, 1000.0)
                s = float(np.sum(a))
                if s < 1e-300:
                    a = np.full(int(n), 1.0, dtype=np.float64)
                    break
                a = a * (float(n) / s)
                if float(np.max(a)) <= 1000.0 * (1.0 + 1e-12):
                    break

    return a


def _finalize_sequence(a, n):
    """Clean sequence for return: finite, nonnegative, bounded, sum n."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    thresh = 1e-12
    a = a.copy()
    a[a < thresh] = 0.0

    s = float(np.sum(a))
    if s < 1e-12:
        return np.ones(int(n), dtype=np.float64)

    a = a * (float(n) / s)
    a = np.minimum(a, 1000.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.ones(int(n), dtype=np.float64)
    a = a * (float(n) / s)
    return a


def _resample_array(parent, n):
    """Linearly resample a parent sequence to length n and normalize to sum n."""
    m = int(parent.size)
    n = int(n)
    if m == n:
        return _normalize_floor(parent.copy(), n, 1e-12)
    if m == 1:
        return _normalize_floor(np.full(n, float(parent[0])), n, 1e-12)

    xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.interp(t, xp, parent)
    return _normalize_floor(a, n, 1e-12)


def _random_init(n, rng):
    """Random nonnegative smooth-ish initialization on [0,1]."""
    n = int(n)
    a = np.ones(n, dtype=np.float64)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    for m in range(1, 10):
        amp = float(rng.normal(0.0, 0.25 / float(m)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a += amp * np.sin(2.0 * np.pi * float(m) * t + ph)
    a = np.maximum(a, 0.0)
    return _normalize_floor(a, n, 1e-12)


def _initial_for_n(parent, n, rng):
    if parent is None:
        return _random_init(n, rng)
    return _resample_array(parent, n)


def _candidate_ns(parent):
    """A sensible list of lengths to try/use."""
    ns = set()
    if parent is not None:
        p = int(parent.size)
        if 128 <= p <= _MAX_N:
            ns.add(p)
        elif p > _MAX_N:
            ns.add(_MAX_N)
        else:
            ns.add(1024)

        if p <= _MAX_N:
            for f in (2, 4):
                q = int(p * f)
                if 128 <= q <= _MAX_N:
                    ns.add(q)

    for nn in (1024, 2048, 4096, 8192, 12288, 16384, 24576, 32768):
        if 128 <= nn <= _MAX_N:
            ns.add(nn)

    out = sorted(ns)
    if not out:
        out = [4096]
    return out


def _get_parent():
    """Return normalized parent sequence or None."""
    try:
        raw = height_sequence_1  # noqa: F821
    except Exception:
        return None

    try:
        p = np.asarray(raw, dtype=np.float64).reshape(-1)
    except Exception:
        return None

    if p.size == 0:
        return None

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    p = np.minimum(p, 1000.0)
    s = float(np.sum(p))
    if s < 1e-12:
        return None

    p = p * (float(p.size) / s)
    return p


def _conv_with_rfft(a, L, n):
    """
    Return (conv_coefficients, max_coefficient, objective, rfft(a)).
    conv_coefficients has length 2*n-1.
    """
    K = 2 * int(n) - 1
    A = np.fft.rfft(a, L)
    c_full = np.fft.irfft(A * A, L)
    c = c_full[:K]
    mc = float(np.max(c))
    S = float(np.sum(a))
    if S < 1e-12:
        obj = float("inf")
    else:
        obj = float(2.0 * n * mc / (S * S))
    return c, mc, obj, A


def _grad_from_rfft(p, A, L, n):
    """
    Gradient of sum_k p[k] * conv(a,a)[k] with respect to a.
    p should be normalized to sum 1.
    """
    P = np.fft.rfft(p, L)
    g = np.fft.irfft(P * np.conj(A), L)[:n]
    return 2.0 * g


def _build_weights(c, mc, typ, par):
    """
    Build coefficient weights of sum 1 from convolution coefficients c.
    typ in {'pow','top','topw'}.
    """
    K = int(c.size)
    if K <= 0:
        return np.zeros(0)
    if mc <= 1e-300 or not np.isfinite(mc):
        return np.full(K, 1.0 / K, dtype=np.float64)

    if typ == "pow":
        q = float(par)
        if q <= 1.0:
            return np.full(K, 1.0 / K, dtype=np.float64)
        x = c / mc
        np.clip(x, 0.0, 1.0, out=x)
        p = np.power(x, q)
        s = float(np.sum(p))
        if s <= 1e-300:
            p = np.zeros(K, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            p = p / s
        return p

    # top or topw
    B = int(par)
    if B <= 0 or B >= K:
        return np.full(K, 1.0 / K, dtype=np.float64)

    idx = np.argpartition(c, -B)[-B:]
    p = np.zeros(K, dtype=np.float64)

    if typ == "top":
        p[idx] = 1.0 / float(B)
    else:
        vals = c[idx]
        vals = np.maximum(vals, 0.0)
        s = float(np.sum(vals))
        if s <= 1e-300:
            p[idx] = 1.0 / float(B)
        else:
            p[idx] = vals / s

    return p


def _pos_bound(a, d):
    """Largest t in [0, cap] such that a + t*d remains nonnegative."""
    neg = d < 0.0
    if not np.any(neg):
        return 1.0
    ratios = a[neg] / (-d[neg])
    if ratios.size == 0:
        return 1.0
    t = float(np.min(ratios))
    if not np.isfinite(t):
        return 0.0
    return max(0.0, min(t * 0.999, 1.0))


def _quad_select_t(c, lin, quad, mc, t_max):
    """
    On the ray a + t*d, the convolution coefficients are
    c + t*lin + t^2*quad.  Sample t in [0,t_max] and return the best t.
    """
    if t_max <= 1e-12:
        return 0.0, float(mc)

    best_t = 0.0
    best_o = float(mc)
    K = int(c.size)
    ns = 24 if K <= 40000 else 16

    if t_max > 1e-6:
        lo = max(t_max * 1e-8, 1e-12)
        ts1 = np.geomspace(lo, t_max, ns // 2 + 1)
        ts2 = np.linspace(0.0, t_max, ns // 2 + 1)
        ts = np.unique(np.concatenate((ts1, ts2)))
    else:
        ts = np.linspace(1e-12, t_max, ns)

    for tv in ts:
        t = float(tv)
        if t <= 0.0:
            continue
        val = c + t * lin + t * t * quad
        o = float(np.max(val))
        if np.isfinite(o) and o < best_o:
            best_o = o
            best_t = t

    return best_t, best_o


def _perturb(a, n, rng, scale):
    """Randomized perturbation to escape local minima."""
    n = int(n)
    a = np.asarray(a, dtype=np.float64)
    mode = float(rng.random())

    if mode < 0.45:
        b = a.copy()
        widths = (0.005, 0.01, 0.02, 0.05, 0.10, 0.18)
        w = max(2, int(n * float(rng.choice(widths))))
        if w >= n:
            w = max(2, n // 10)
        s = int(rng.integers(0, n - w + 1))
        sig = max(0.05, float(scale) * float(rng.choice((0.5, 1.0, 1.8))))
        b[s:s + w] *= np.exp(rng.normal(0.0, sig, size=w))
        return _normalize_floor(b, n, 1e-12)

    # Smooth Fourier/log perturbation.
    x = np.log(np.maximum(a, 1e-12))
    x -= float(np.mean(x))
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    mmax = int(rng.integers(2, 21))
    base = max(0.05, float(scale)) * float(rng.uniform(0.6, 1.6))
    for m in range(1, mmax + 1):
        amp = base / float(m)
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        if float(rng.random()) < 0.5:
            x += amp * np.cos(2.0 * np.pi * float(m) * t + ph)
        else:
            x += amp * np.sin(2.0 * np.pi * float(m) * t + ph)

    b = np.exp(x)
    return _normalize_floor(b, n, 1e-12)


def run_local(a, n, L, deadline, rng, tag="local"):
    """
    Local optimizer for fixed length n.  Returns (best_a, best_obj).
    """
    n = int(n)
    K = 2 * n - 1
    a = _normalize_floor(a, n, 1e-12)
    c, mc, cur_obj, A = _conv_with_rfft(a, L, n)

    best_a = a.copy()
    best_obj = float(cur_obj)

    strategies = _STRATEGIES
    ns = len(strategies)
    cyc = int(rng.integers(ns))
    good_idx = cyc
    keep = 0
    fails = 0
    pert_level = 0
    it = 0
    start_t = time.monotonic()
    next_print = start_t + 20.0

    while time.monotonic() < deadline:
        it += 1

        if cur_obj < best_obj - 1e-10:
            best_obj = float(cur_obj)
            best_a = a.copy()

        if keep > 0:
            keep -= 1
            idx = good_idx
        else:
            idx = cyc
            cyc = (cyc + 1) % ns
            good_idx = idx

        typ, par = strategies[idx]
        p = _build_weights(c, mc, typ, par)
        g = _grad_from_rfft(p, A, L, n)
        s = g - float(np.mean(g))
        scale_s = float(np.max(np.abs(s)))
        improved = False

        if scale_s > 1e-12 and np.isfinite(scale_s):
            # Additive step with exact quadratic line search along the
            # normalized projected gradient direction.
            d = -s / scale_s
            t_max = _pos_bound(a, d)
            if t_max > 1e-10:
                D = np.fft.rfft(d, L)
                lin_full = 2.0 * np.fft.irfft(A * D, L)
                quad_full = np.fft.irfft(D * D, L)
                t_opt, pred_mc = _quad_select_t(c, lin_full[:K], quad_full[:K], mc, t_max)

                if t_opt > 0.0 and np.isfinite(pred_mc) and pred_mc < mc * (1.0 - 1e-12):
                    cand = a + t_opt * d
                    cand = _normalize_floor(cand, n, 1e-12)
                    c_new, mc_new, obj_new, A_new = _conv_with_rfft(cand, L, n)
                    if np.isfinite(obj_new) and obj_new < cur_obj - 1e-10:
                        a = cand
                        c = c_new
                        mc = mc_new
                        cur_obj = obj_new
                        A = A_new
                        improved = True
                        fails = 0
                        keep = 2

            # If additive step was not useful, try multiplicative mirror steps.
            if not improved:
                multis = (0.25, 0.5, 1.0, 0.1, 0.02)
                for mult in multis:
                    eta = min(10.0, float(mult) / max(scale_s, 1e-12))
                    tmp = -eta * s
                    np.clip(tmp, -8.0, 8.0, out=tmp)
                    cand = a * np.exp(tmp)
                    cand = _normalize_floor(cand, n, 1e-12)
                    c_new, mc_new, obj_new, A_new = _conv_with_rfft(cand, L, n)
                    if np.isfinite(obj_new) and obj_new < cur_obj - 1e-10:
                        a = cand
                        c = c_new
                        mc = mc_new
                        cur_obj = obj_new
                        A = A_new
                        improved = True
                        fails = 0
                        keep = 1
                        break
        else:
            # Flat gradient: try perturbation soon.
            fails += 1

        if not improved:
            if scale_s <= 1e-12:
                fails += 1
            else:
                fails += 1

            if fails >= 8:
                fails = 0
                pert_level = min(15, pert_level + 1)
                scale = 0.15 + 0.10 * float(pert_level)
                b = _perturb(a, n, rng, scale)
                c_new, mc_new, obj_new, A_new = _conv_with_rfft(b, L, n)

                rel = 0.03 + 0.004 * float(pert_level)
                threshold = best_obj * (1.0 + rel) if np.isfinite(best_obj) else 1e100
                if np.isfinite(obj_new) and (obj_new < max(threshold, cur_obj)):
                    a = b
                    c = c_new
                    mc = mc_new
                    cur_obj = obj_new
                    A = A_new
                    pert_level = max(0, pert_level - 2)

        if time.monotonic() > next_print:
            elapsed = time.monotonic() - start_t
            print(
                f"[{tag}] it={it} elapsed={elapsed:.0f}s "
                f"best={best_obj:.6f} cur={cur_obj:.6f} fails={fails} perturb={pert_level}"
            )
            next_print = time.monotonic() + 30.0

    # Check a symmetrized copy; sometimes it is better.
    try:
        a_rev = a[::-1].copy()
        a_sym = _normalize_floor(0.5 * (a + a_rev), n, 1e-12)
        c_sym, mc_sym, obj_sym, _ = _conv_with_rfft(a_sym, L, n)
        if np.isfinite(obj_sym) and obj_sym < best_obj - 1e-10:
            best_obj = float(obj_sym)
            best_a = a_sym.copy()
    except Exception:
        pass

    return best_a, float(best_obj)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search function invoked by the harness.
    Returns a list of nonnegative floats.
    """
    total = float(budget_s)
    start = time.monotonic()

    main_deadline = start + max(60.0, total - 90.0)
    warm_total = min(150.0, max(30.0, 0.05 * total))
    warm_deadline = min(start + warm_total, main_deadline - 30.0)

    rng = np.random.default_rng(seed)
    parent = _get_parent()

    global_n = None
    global_L = None
    global_best_a = None
    global_best_obj = float("inf")

    # ---- Raw candidates, no nontrivial optimization yet. ----
    raw_res = []
    idx = 0
    for n in _candidate_ns(parent):
        if time.monotonic() > warm_deadline - 5.0:
            break
        try:
            a = _initial_for_n(parent, n, rng)
            L = _next_pow2(2 * n - 1)
            _, _, obj, _ = _conv_with_rfft(a, L, n)
        except Exception:
            continue

        if np.isfinite(obj):
            raw_res.append((float(obj), idx, int(n), a, L))
            print(f"raw n={n} obj={obj:.6f}")
        idx += 1

    if not raw_res:
        # Fallback: safe uniform sequence.
        n = 4096
        a = np.ones(n, dtype=np.float64)
        return [float(x) for x in a]

    raw_res.sort()  # sorts by objective, then index, then length.

    init_obj, _, init_n, init_a, init_L = raw_res[0]
    global_best_a = init_a.copy()
    global_n = int(init_n)
    global_L = int(init_L)
    global_best_obj = float(init_obj)

    # ---- Brief warm-up on a few promising lengths. ----
    warm_ns = []
    for item in raw_res[:2]:
        nn = int(item[2])
        if nn not in warm_ns:
            warm_ns.append(nn)

    if parent is not None:
        pn = int(parent.size)
        if 128 <= pn <= _MAX_N and pn not in warm_ns:
            warm_ns.append(pn)

    if len(warm_ns) < 3:
        for item in raw_res:
            nn = int(item[2])
            if nn not in warm_ns:
                warm_ns.append(nn)
                if len(warm_ns) >= 3:
                    break

    if time.monotonic() < warm_deadline - 15.0 and len(warm_ns) > 0:
        warm_each = max(5.0, warm_total / float(len(warm_ns)))
        for n in warm_ns:
            # locate raw initialization for this n
            aa = None
            LL = None
            for item in raw_res:
                if int(item[2]) == int(n):
                    aa = item[3]
                    LL = item[4]
                    break
            if aa is None or LL is None:
                continue

            sub_deadline = time.monotonic() + warm_each
            if sub_deadline > main_deadline - 30.0:
                sub_deadline = main_deadline - 30.0
            if sub_deadline <= time.monotonic():
                break

            try:
                ba, bo = run_local(aa, n, LL, sub_deadline, rng, tag=f"warm{n}")
                if np.isfinite(bo) and bo < global_best_obj - 1e-10:
                    global_best_obj = float(bo)
                    global_best_a = ba
                    global_n = int(n)
                    global_L = int(LL)
            except Exception:
                pass

    # ---- Main optimization. ----
    if time.monotonic() < main_deadline - 20.0 and global_best_a is not None:
        try:
            global_best_a, global_best_obj = run_local(
                global_best_a,
                int(global_n),
                int(global_L),
                main_deadline,
                rng,
                tag="main",
            )
        except Exception:
            pass

    # ---- Final cleanup/validation. ----
    try:
        n_final = int(global_n)
        out = _finalize_sequence(global_best_a, n_final)
        out = np.asarray(out, dtype=np.float64)

        # Local objective check.
        L_final = _next_pow2(2 * n_final - 1)
        _, _, local_obj, _ = _conv_with_rfft(out, L_final, n_final)
        print(f"FINAL local_obj={local_obj:.6f} n={n_final}")

        try:
            score = evaluate_sequence([float(x) for x in out])  # noqa: F821
            if np.isfinite(score):
                print(f"FINAL evaluated={float(score):.6f}")
            else:
                raise ValueError("invalid evaluated")
        except Exception:
            # If evaluation is unavailable or fails, return the cleaned candidate.
            pass

        return [float(x) for x in out]

    except Exception:
        n = int(global_n) if global_n is not None else 4096
        if n < 1:
            n = 4096
        return [1.0 for _ in range(n)]
