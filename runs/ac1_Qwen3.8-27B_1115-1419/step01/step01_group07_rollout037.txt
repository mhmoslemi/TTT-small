def safe_evaluate(a):
    """Evaluate with safety checks."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s * np.sqrt(2 * len(a))
    c = np.convolve(a_norm, a_norm)
    return float(c.max())


def lse_smooth_max(x, tau):
    """Log-sum-exp approximation of max."""
    x_max = x.max()
    if x_max < 1e-30:
        return 0.0
    shifted = (x - x_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.exp(shifted).sum())
    return x_max + tau * log_sum


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def block_lp_update(a, block_start, block_size, target_sum):
    """
    Solve LP to optimize a block of the sequence.

def block_coordinate_descent(a, block_size, max_sweeps=50, time_limit=None):
    """Run block LP coordinate descent."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = safe_evaluate(a)
    t0 = time.monotonic()

    for sweep in range(max_sweeps):
        if time_limit and (time.monotonic() - t0) > time_limit:
            break
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            end = start + block_size
            target_sum = a[start:end].sum()
            if target_sum < 1e-15:
                continue
            a_block_new = block_lp_update(a, start, block_size, target_sum)
            a_copy = a.copy()
            a_copy[start:end] = a_block_new
            score = safe_evaluate(a_copy)
            if score < best_score - 1e-12:
                a = a_copy
                best_score = score
                best_a = a.copy()
                improved = True

        if not improved and sweep > 2:
            break

    return best_a, best_score

        # Transfer mass from one to the other
        delta = (rng.random() - 0.5) * 0.02 * min(a[i], a[j]) * 10
        if delta > 0:
            # Move from i to j
            transfer = min(delta, a[i] * 0.5)
            a[i] -= transfer
            a[j] += transfer
        else:
            # Move from j to i
            transfer = min(-delta, a[j] * 0.5)
            a[j] -= transfer
            a[i] += transfer

        score = safe_evaluate(a)
        eval_count += 1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            accept_count += 1
        elif T > 1e-8:
            # Metropolis acceptance
            if rng.random() < np.exp(-(score - best_score) / T):
                accept_count += 1
            else:
                # Revert
                if delta > 0:
                    a[i] += transfer
                    a[j] -= transfer
                else:
                    a[j] += transfer
                    a[i] -= transfer

        T *= cool

        # Periodic restart from best if stuck
        if eval_count % 50000 == 0 and T > 1e-6:
            if best_score < 1.51:
                a = best_a.copy()
            # Add small noise to escape
            noise = rng.normal(0, 0.002, n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score, eval_count


def lse_lbfgs(a, time_limit, tau_init=None):
    """L-BFGS on LSE-smoothed objective with log parameterization."""
    n = len(a)
    a = a / a.sum()
    x = np.log(a + 1e-30)
    best_a = a.copy()
    best_score = safe_evaluate(a)
    t0 = time.monotonic()

    if tau_init is None:
        tau_init = 0.05 / n

    taus = [tau_init, tau_init * 0.5, tau_init * 0.25, tau_init * 0.1]

    for tau in taus:
        if (time.monotonic() - t0) > time_limit * 0.8:
            break

        def obj(x_):
            a_ = softmax(x_)
            c = np.convolve(a_, a_)
            return lse_smooth_max(c, tau)

        def grad(x_):
            a_ = softmax(x_)
            c = np.convolve(a_, a_)
            c_max = c.max()
            shifted = (c - c_max) / tau
            shifted = np.clip(shifted, -500, 500)
            w = np.exp(shifted)
            w = w / w.sum()
            # Gradient of smoothed max w.r.t. a
            grad_a = 2.0 * np.correlate(w, a_, mode='valid')
            # Chain rule through softmax
            dot = np.dot(a_, grad_a)
            return a_ * (grad_a - dot)

        try:
            result = minimize(obj, x, jac=grad, method='L-BFGS-B',
                              options={'maxiter': 300, 'ftol': 1e-14,
                                       'gtol': 1e-10, 'maxfun': 500})
            x = result.x
            a_new = softmax(x)
            score = safe_evaluate(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        except Exception:
            break

    return best_a, best_score


def subgradient_polish(a, time_limit, n_active=30, step_init=0.05):
    """Subgradient method with top-k peaks and adaptive step."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = safe_evaluate(a)
    step = step_init
    t0 = time.monotonic()

    while (time.monotonic() - t0) < time_limit:
        c = np.convolve(a, a)
        max_c = c.max()
        if max_c < 1e-15:
            break

        # Top-k peaks
        k = min(n_active, len(c))
        top_idx = np.argsort(c)[::-1][:k]

        # Average subgradient over top-k
        s = np.zeros(n)
        for idx in top_idx:
            for i in range(n):
                j = idx - i
                if 0 <= j < n:
                    s[i] += 2.0 * a[j]
        s /= k

        # Polyak step size
        score = safe_evaluate(a)
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        if score > lower_bound + 1e-6:
            s_norm = np.dot(s, s)
            if s_norm > 1e-15:
                polyak_step = (score - lower_bound) / s_norm
                step = max(0.001, min(step, polyak_step))

        # Update
        a_new = a - step * s
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum > 1e-15:
            a_new = a_new / a_new_sum
        else:
            a_new = a.copy()

        score_new = safe_evaluate(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            step = min(step * 1.05, 0.2)
        else:
            step *= 0.95
            if step < 0.0005:
                # Try a larger random step
                a_perturbed = a + np.random.normal(0, 0.01, n)
                a_perturbed = np.maximum(a_perturbed, 0)
                sp = a_perturbed.sum()
                if sp > 0:
                    a_perturbed /= sp
                sp_score = safe_evaluate(a_perturbed)
                if sp_score < best_score:
                    a = a_perturbed
                    best_a = a.copy()
                    best_score = sp_score
                    step = 0.02
                else:
                    step = 0.005

    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """Main search: block LP CD + simulated annealing + L-BFGS + subgradient polish."""
    # Initial sequence
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        a = np.ones(4096, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f"Initial: n={len(a)}, score={best_score:.6f}")

    global_deadline = time.monotonic() + budget_s - 60
    rng = np.random.default_rng(42)

    # Phase 1: L-BFGS warmup (120s)
    print("\n=== Phase 1: L-BFGS warmup ===")
    phase_time = min(120.0, (global_deadline - time.monotonic()) / 4)
    a_new, score = lse_lbfgs(a, phase_time)
    if score < best_score:
        best_score = score
        best_a = a_new.copy()
    print(f"After L-BFGS: {best_score:.6f}")

    # Phase 2: Block LP coordinate descent (600s)
    print("\n=== Phase 2: Block LP coordinate descent ===")
    phase_time = min(600.0, (global_deadline - time.monotonic()) / 3)
    a_start = best_a.copy()

    # Try different block sizes
    for bs in [4, 8]:
        t_start = time.monotonic()
        if time.monotonic() > global_deadline:
            break
        a_blk, score_blk = block_coordinate_descent(
            a_start, bs,
            max_sweeps=200,
            time_limit=(global_deadline - time.monotonic()) / 2
        )
        if score_blk < best_score:
            best_score = score_blk
            best_a = a_blk.copy()
            print(f"  Block size {bs}: {best_score:.6f} (time={time.monotonic()-t_start:.0f}s)")
        a_start = best_a.copy()

    print(f"After block CD: {best_score:.6f}")

    # Phase 3: Simulated annealing (800s)
    print("\n=== Phase 3: Simulated annealing ===")
    remaining = global_deadline - time.monotonic()
    sa_time = min(800.0, remaining * 0.5)
    if sa_time > 30:
        a_sa, score_sa, n_evals = simulated_annealing(
            best_a, sa_time, rng, T0=0.005, cool=0.9998
        )
        if score_sa < best_score:
            best_score = score_sa
            best_a = a_sa.copy()
        print(f"After SA ({n_evals} evals): {best_score:.6f}")

    # Phase 4: Subgradient polish (300s)
    print("\n=== Phase 4: Subgradient polish ===")
    remaining = global_deadline - time.monotonic()
    sg_time = min(300.0, remaining * 0.4)
    if sg_time > 20:
        a_sg, score_sg = subgradient_polish(best_a, sg_time, n_active=40, step_init=0.03)
        if score_sg < best_score:
            best_score = score_sg
            best_a = a_sg.copy()
        print(f"After subgradient: {best_score:.6f}")

    # Phase 5: Try a different length
    print("\n=== Phase 5: Different length attempt ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 120:
        for trial_n in [2048, 6144]:
            if time.monotonic() > global_deadline - 30:
                break
            # Create initial sequence of new length by interpolation
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, trial_n)
            f = interp1d(x_old, best_a, kind='linear', fill_value='extrapolate')
            a_trial = f(x_new)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()

            # Quick L-BFGS
            t_phase = min(80.0, (global_deadline - time.monotonic()) / 2)
            a_t, score_t = lse_lbfgs(a_trial, t_phase)
            if score_t < best_score:
                best_score = score_t
                best_a = a_t.copy()
                print(f"  n={trial_n}: {best_score:.6f}")
            # Quick SA
            t_phase2 = min(40.0, (global_deadline - time.monotonic()) / 2)
            if t_phase2 > 10:
                a_t2, score_t2, _ = simulated_annealing(a_t, t_phase2, rng, T0=0.003)
                if score_t2 < best_score:
                    best_score = score_t2
                    best_a = a_t2.copy()
                    print(f"  n={trial_n} after SA: {best_score:.6f}")

    # Phase 6: Final intensive polish
    print("\n=== Phase 6: Final polish ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        # Alternate between block CD and SA
        for round_idx in range(5):
            if time.monotonic() > global_deadline - 10:
                break
            chunk = min(60.0, (global_deadline - time.monotonic()) - 10)
            if chunk < 15:
                break
            # Block CD
            a_blk, score_blk = block_coordinate_descent(
                best_a, 4, max_sweeps=50, time_limit=chunk * 0.5
            )
            if score_blk < best_score:
                best_score = score_blk
                best_a = a_blk.copy()
            # SA
            a_sa, score_sa, _ = simulated_annealing(
                best_a, chunk * 0.4, rng, T0=0.002, cool=0.9997
            )
            if score_sa < best_score:
                best_score = score_sa
                best_a = a_sa.copy()
            print(f"  Round {round_idx + 1}: {best_score:.6f}")

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_a])
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_a)}, score={final_score:.6f}")
    print(f"{'='*50}")

    return [float(x) for x in best_a]
