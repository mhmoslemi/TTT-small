"""
Search for low-peak nonnegative self-convolution sequences.

This program searches for a nonnegative sequence a_i minimizing
    score(a) = 2 * n * max(a * a) / (sum a)^2
by working with sum-normalized sequences and using a smooth auxiliary
objective based on the min/max range of the full convolution:
    LSE(b) + gamma * LSE(-b)  ~  max(b) - min(b)
when the log-sum-exp temperature is large.  The maximization of min(b)
helps fill under-filled lags and escapes one-sided active-set stationary
points that only minimize max(b).

Optimization is done by mirror descent (exponentiated gradient) on the
logits of a softmax sequence, using FFT-based convolution and gradient
evaluation.  Several prior starts are used: one-sided power-law profiles,
power-law profiles with Fourier perturbations, resamples of the provided
or previously best sequence, and interior-boosted power profiles.
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    """Smallest power of two >= size."""
    size = int(max(1, size))
    return 1 << (size - 1).bit_length()


def _make_uniform(n):
    n = max(int(n), 1)
    return np.full(n, 1.0 / n, dtype=np.float64)


def _normalize(a):
    """Make a nonnegative finite 1D array with sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return _make_uniform(a.size)
    return a / s


def _calc_convolution(a, size, n):
    """Full plain convolution a*a, using FFT and clipping tiny negatives."""
    a = np.asarray(a, dtype=np.float64)
    c = np.fft.rfft(a, size)
    b = np.fft.irfft(c * c, size)
    b = b[:2 * int(n) - 1]
    return np.maximum(b, 0.0)


def _score_of(a):
    """Exact FFT-based score for a sum-normalized nonnegative sequence."""
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    size = _next_pow2(2 * n - 1)
    b = _calc_convolution(a, size, n)
    mb = float(np.max(b)) if b.size else 0.0
    if not np.isfinite(mb):
        return 1e100
    return float(2.0 * n * mb)


def _exact_score_small(a):
    """Direct score for small/moderate sequences, useful for final logging."""
    a = _normalize(a)
    n = a.size
    if n > 4096:
        return _score_of(a)
    b = np.convolve(a, a)
    s = float(a.sum())
    if s <= 1e-300:
        return 1e100
    return float(2.0 * n * float(np.max(b)) / (s * s))


def _sanitize_y(y):
    """Center and gently compress logits to avoid overflow/underflow."""
    y = np.asarray(y, dtype=np.float64).ravel()
    y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
    if y.size == 0:
        return y
    y = y - float(np.mean(y))
    span = float(np.max(y) - np.min(y))
    scale = 30.0
    if span > 2.0 * scale:
        y = scale * np.tanh(y / scale)
    else:
        y = np.clip(y, -40.0, 40.0)
    return y


def _softmax(y):
    """Stable softmax, interpreted as sum-normalized nonnegative sequence."""
    y = _sanitize_y(y)
    if y.size == 0:
        return np.array([1.0], dtype=np.float64)
    m = float(np.max(y))
    if not np.isfinite(m):
        return _make_uniform(y.size)
    ev = np.exp(y - m)
    s = float(ev.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return _make_uniform(y.size)
    return ev / s


def _smooth_noise(rng, n, bandwidth):
    """Lowpass Gaussian noise for smooth random restarts."""
    n = max(int(n), 2)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    bandwidth = max(1, int(bandwidth))
    cut = max(2, min(len(F), n // bandwidth + 2))
    if cut < len(F):
        F[cut:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 1e-300:
        f = f / sd
    else:
        f = np.zeros(n, dtype=np.float64)
    return f


def _perturb_y(y, rng, strength):
    """Perturb logits with smooth low-frequency noise."""
    y = np.asarray(y, dtype=np.float64).copy()
    n = y.size
    if n == 0:
        return y
    strength = float(max(0.0, strength))
    if strength > 0.0 and n > 8:
        f = _smooth_noise(rng, n, max(4, n // 48))
        y = y + strength * f
        if rng.random() < 0.35:
            width = max(1, n // 24)
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            y[start:stop] += strength * 0.5 * rng.normal(size=stop - start)
    return _sanitize_y(y)


def _make_power(n, p=0.5, alpha=0.5):
    """One-sided integrable power-law profile."""
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    alpha = max(1e-3, float(alpha))
    p = max(0.01, min(2.0, float(p)))
    y = -p * np.log(i + alpha)
    y = _sanitize_y(y)
    return _normalize(_softmax(y))


def _make_power_interior(n, interior=0.35, p=0.5, alpha=0.5):
    """Power profile with a smooth interior boost."""
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    alpha = max(1e-3, float(alpha))
    p = max(0.01, min(2.0, float(p)))
    y = -p * np.log(i + alpha)
    y = y + float(interior) * np.sin(np.pi * x)
    y = _sanitize_y(y)
    return _normalize(_softmax(y))


def _make_power_fourier(n, rng, p=0.5, alpha=0.5, amp=0.35, K=10, interior=0.18):
    """Power profile plus small smooth Fourier perturbations."""
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    alpha = max(1e-3, float(alpha))
    p = max(0.01, min(2.0, float(p)))
    y = -p * np.log(i + alpha)
    y = y + float(interior) * np.sin(np.pi * x)

    amp = float(max(0.0, amp))
    for m in range(1, max(1, int(K)) + 1):
        coeff = amp / (0.6 + 0.45 * m)
        if rng.random() < 0.75:
            y = y + rng.normal(0.0, coeff) * np.cos(np.pi * m * x)
        if rng.random() < 0.40:
            y = y + rng.normal(0.0, 0.75 * coeff) * np.sin(np.pi * m * x)

    y = _sanitize_y(y)
    return _normalize(_softmax(y))


def _make_smooth_random(n, rng, base, amp=0.6):
    """Log multiplicative smooth perturbation of a base sequence."""
    base = _normalize(base)
    base_y = np.log(np.maximum(base, 1e-300))
    base_y = _sanitize_y(base_y)
    f = _smooth_noise(rng, n, max(4, n // 64))
    y = base_y + float(amp) * f
    return _normalize(_softmax(y))


def _resample_sequence(old, new_n):
    """Resample a normalized nonnegative sequence to a new length."""
    old = _normalize(old)
    new_n = max(int(new_n), 2)
    m = old.size
    if m == 0:
        return _make_uniform(new_n)
    if m == new_n:
        return old.copy()
    if m == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)

    xo = np.linspace(0.0, 1.0, m, dtype=np.float64)
    xn = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(xn, xo, old)
    return _normalize(new)


def _get_given_sequence():
    """Read the optional global previously found sequence, if present."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0:
        return None
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return arr / s


def _dedup_pairs(pairs):
    """Remove nearly duplicate initial sequences while preserving gamma."""
    out = []
    seen = []
    for seq, gamma in pairs:
        a = np.asarray(seq, dtype=np.float64).ravel()
        if a.size == 0:
            continue
        step = max(1, a.size // 64)
        fp = tuple(np.round(a[::step], 6))
        if fp not in seen:
            seen.append(fp)
            out.append((seq, float(gamma)))
    return out

def _lse_p(x, beta):
    """Return weights exp(beta*x)/sum and log-sum-exp value."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.ones(1), 0.0
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0
    m = float(np.max(x))
    if not np.isfinite(m):
        p = np.full(x.size, 1.0 / x.size, dtype=np.float64)
        return p, 0.0

    if beta < 1e-8:
        p = np.full(x.size, 1.0 / x.size, dtype=np.float64)
        val = m + float(np.log(x.size))
        return p, val

    d = x - m
    d = np.minimum(d, 0.0)
    arg = beta * d
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full(x.size, 1.0 / x.size, dtype=np.float64)
        return p, m
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _lse_value(x, beta):
    """Stable log-sum-exp value."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return 0.0
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0
    m = float(np.max(x))
    if not np.isfinite(m):
        return 0.0
    if beta < 1e-8:
        return m + float(np.log(x.size))

    d = x - m
    d = np.minimum(d, 0.0)
    arg = beta * d
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return m
    return m + float(np.log(s)) / beta


def _refine_mirror(a, deadline, rng, gamma=0.0, beta_start=10.0,
                   beta_max=700.0, eta_start=0.40, max_eta=2.0):
    """
    Mirror descent on logits.

    Objective minimized in the line search:
        J = LSE(b, beta_high) + gamma * LSE(-b, beta_low)
    with b = a*a.  For large beta this approximates max(b) - gamma*min(b),
    encouraging both peak reduction and filling low lags.
    """
    a = _normalize(a)
    n = a.size
    if n < 8:
        a = _make_power(128)
        n = 128

    y = np.log(np.maximum(a, 1e-300))
    y = _sanitize_y(y)
    a = _softmax(y)

    b_size = _next_pow2(2 * n - 1)
    g_size = _next_pow2(3 * n - 2)

    b = _calc_convolution(a, b_size, n)
    mb = float(np.max(b))
    score = float(2.0 * n * mb)

    best_a = a.copy()
    best_score = score
    best_y = y.copy()

    if not np.isfinite(best_score):
        fallback = _make_power(n)
        return fallback, _score_of(fallback)

    use_low = gamma > 1e-7
    eta = float(eta_start)
    beta = float(beta_start)
    fail = 0
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        if not np.isfinite(score) or mb <= 0.0:
            y = best_y.copy()
            a = _softmax(y)
            b = _calc_convolution(a, b_size, n)
            mb = float(np.max(b))
            score = float(2.0 * n * mb)

        beta_h = min(1e8, max(1.0, beta / max(mb, 1e-300)))
        p_h, f_h = _lse_p(b, beta_h)

        if use_low:
            minb = float(np.min(b))
            span = max(mb - minb, mb, 1e-300)
            beta_l = min(1e8, max(1.0, beta / span))
            p_l, f_l = _lse_p(-b, beta_l)
            F = float(f_h + gamma * f_l)
            w = p_h - gamma * p_l
        else:
            beta_l = beta_h
            F = float(f_h)
            w = p_h

        if not np.isfinite(F):
            y = _perturb_y(best_y, rng, 0.35)
            a = _softmax(y)
            b = _calc_convolution(a, b_size, n)
            mb = float(np.max(b))
            score = float(2.0 * n * mb)
            beta = max(6.0, 0.6 * beta)
            eta = float(eta_start)
            fail = 0
            it += 1
            continue

        ar = a[::-1]
        FAr = np.fft.rfft(ar, g_size)
        Fw = np.fft.rfft(w, g_size)
        conv = np.fft.irfft(FAr * Fw, g_size).real
        D = 2.0 * conv[n - 1:2 * n - 1]
        D = np.nan_to_num(D, nan=0.0, posinf=0.0, neginf=0.0)

        Dc = D - float(np.mean(D))
        scale = float(np.max(np.abs(Dc)))
        if not np.isfinite(scale) or scale < 1e-300:
            y = _perturb_y(best_y, rng, 0.40)
            a = _softmax(y)
            b = _calc_convolution(a, b_size, n)
            mb = float(np.max(b))
            score = float(2.0 * n * mb)
            beta = max(6.0, 0.6 * beta)
            eta = float(eta_start)
            fail = 0
            it += 1
            continue

        d = Dc / scale
        F_base = F
        score_base = score
        accepted = False
        trial_eta = eta

        for _ in range(7):
            if time.monotonic() >= deadline:
                break

            ty = y - trial_eta * d
            ty = _sanitize_y(ty)
            ta = _softmax(ty)

            bt = _calc_convolution(ta, b_size, n)
            mb_t = float(np.max(bt))
            score_t = float(2.0 * n * mb_t)

            if use_low:
                minbt = float(np.min(bt))
                spant = max(mb_t - minbt, mb_t, 1e-300)
                beta_lt = min(1e8, max(1.0, beta / spant))
                Ft = float(_lse_value(bt, beta_h) + gamma * _lse_value(-bt, beta_lt))
            else:
                Ft = float(_lse_value(bt, beta_h))

            f_tol = 1e-15 * max(1.0, abs(F_base))
            s_tol = 1e-14
            s_f_tol = 1e-9 * max(1.0, abs(F_base))

            good = False
            if np.isfinite(score_t) and np.isfinite(Ft):
                if Ft < F_base - f_tol:
                    good = True
                elif score_t < score_base - s_tol and Ft <= F_base + s_f_tol:
                    good = True

            if good:
                y = ty
                a = ta
                b = bt
                mb = mb_t
                score = score_t
                F = Ft
                accepted = True
                fail = 0

                if score_t < best_score - 1e-14:
                    best_score = score_t
                    best_a = a.copy()
                    best_y = y.copy()
                    last_improve = it

                if Ft < F_base - 1e-15 * max(1.0, abs(F_base)):
                    eta = min(float(max_eta), eta * 1.10)
                else:
                    eta = min(float(max_eta), eta * 1.02)
                break

            trial_eta = 0.5 * trial_eta
            if trial_eta < 1e-4:
                break

        if accepted:
            if it % 25 == 0:
                beta = min(float(beta_max), beta * 1.055 + 0.15)
        else:
            fail += 1
            eta = max(1e-4, 0.5 * eta)
            if fail > 35:
                if np.isfinite(score) and score > best_score + 0.01:
                    y = best_y.copy()
                else:
                    y = _perturb_y(best_y, rng, 0.45)
                a = _softmax(y)
                b = _calc_convolution(a, b_size, n)
                mb = float(np.max(b))
                score = float(2.0 * n * mb)
                beta = max(8.0, 0.55 * beta)
                eta = float(eta_start)
                fail = 0

        if it - last_improve > 500 and time.monotonic() < deadline - 3.0:
            y = _perturb_y(best_y, rng, 0.55)
            a = _softmax(y)
            b = _calc_convolution(a, b_size, n)
            mb = float(np.max(b))
            score = float(2.0 * n * mb)
            beta = max(8.0, 0.50 * beta)
            eta = float(eta_start)
            fail = 0

        it += 1

    return _normalize(best_a), float(best_score)


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Search budget-bounded and return the best nonnegative sequence found.
    """
    try:
        seed_int = int(seed) % (2 ** 32)
    except Exception:
        seed_int = 0
    rng = np.random.default_rng(seed_int)

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0

    start_time = time.monotonic()
    # Leave time for final normalization/return, but never run longer than budget.
    deadline = start_time + max(5.0, budget - 35.0)

    if budget < 25.0:
        seq = _make_power(256)
        return [float(x) for x in seq]

    given = _get_given_sequence()

    # Robust starting point: a sufficiently long one-sided power profile.
    best_a = _make_power(1024)
    best_score = _score_of(best_a)
    if given is not None:
        g_score = _score_of(given)
        if g_score < best_score:
            best_a = given.copy()
            best_score = g_score

    print(
        f"initial best len={best_a.size} score={best_score:.8f} t={time.monotonic() - start_time:.1f}s",
        flush=True,
    )

    # Choose maximum problem length depending on budget.
    if budget < 60.0:
        max_n = 128
    elif budget < 180.0:
        max_n = 512
    elif budget < 450.0:
        max_n = 1024
    elif budget < 900.0:
        max_n = 2048
    elif budget < 1500.0:
        max_n = 4096
    elif budget < 2100.0:
        max_n = 8192
    else:
        max_n = 16384

    raw_configs = [
        # length, number of starts, per-start seconds at full 2400s budget
        (128, 1, 8),
        (256, 2, 12),
        (512, 2, 25),
        (1024, 3, 50),
        (2048, 3, 100),
        (4096, 3, 180),
        (8192, 3, 240),
    ]
    if budget >= 2000.0:
        raw_configs.append((16384, 1, 250))

    scale = min(1.0, max(0.15, budget / 2400.0))
    configs = []
    for n, num, tps in raw_configs:
        if n > max_n:
            break
        sec = max(3, int(float(tps) * scale))
        configs.append((int(n), int(num), int(sec)))

    if not configs:
        configs = [(max(128, min(512, max_n)), 1, max(10, int(budget - 45)))]

    for n, num, total_sec in configs:
        if time.monotonic() > deadline - 20.0:
            break

        pairs = _generate_pairs(n, given, rng, best_a.copy(), num)
        if not pairs:
            continue

        per_sec = max(3, total_sec // max(1, len(pairs)))
        for idx, (init, gamma) in enumerate(pairs):
            if time.monotonic() > deadline - 15.0:
                break

            t_end = min(deadline - 5.0, time.monotonic() + float(per_sec))
            if t_end <= time.monotonic():
                continue

            res_a, res_score = _refine_mirror(
                init.copy(),
                t_end,
                rng,
                gamma=gamma,
                beta_start=8.0,
                beta_max=650.0,
                eta_start=0.40,
                max_eta=2.0,
            )

            print(
                f"n={res_a.size} gamma={gamma:.2f} start={idx}/{len(pairs)} "
                f"score={res_score:.8f} best={best_score:.8f} "
                f"t={time.monotonic() - start_time:.1f}s",
                flush=True,
            )

            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)

    # If largest length was not reached by configs, but there is significant time left,
    # upscale the current best once/twice.
    if time.monotonic() < deadline - 180.0 and best_a.size < max_n:
        n_cur = int(best_a.size)
        for _ in range(3):
            if n_cur >= max_n or time.monotonic() >= deadline - 150.0:
                break
            n2 = min(max_n, max(512, 2 * n_cur))
            if n2 <= n_cur:
                break

            init = _resample_sequence(best_a, n2)
            init = _normalize(0.86 * init + 0.14 * _make_power(n2))

            t_end = min(deadline - 90.0, time.monotonic() + max(45.0, 0.18 * (deadline - start_time)))
            res_a, res_score = _refine_mirror(
                init.copy(),
                t_end,
                rng,
                gamma=0.22,
                beta_start=12.0,
                beta_max=700.0,
                eta_start=0.40,
                max_eta=2.0,
            )
            print(
                f"upscale {n_cur}->{res_a.size} score={res_score:.8f} "
                f"best={best_score:.8f} t={time.monotonic() - start_time:.1f}s",
                flush=True,
            )

            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                n_cur = int(res_a.size)
            else:
                break

    # Final polishing: pure peak minimization from the current best shape.
    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        t_end = min(deadline - 3.0, time.monotonic() + min(0.45 * remaining, 300.0))
        res_a, res_score = _refine_mirror(
            best_a.copy(),
            t_end,
            rng,
            gamma=0.0,
            beta_start=40.0,
            beta_max=1000.0,
            eta_start=0.30,
            max_eta=1.4,
        )
        print(
            f"final pure score={res_score:.8f} best={best_score:.8f} "
            f"t={time.monotonic() - start_time:.1f}s",
            flush=True,
        )
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)

    # Secondary final polish with mild range term.
    remaining = deadline - time.monotonic()
    if remaining > 25.0:
        t_end = min(deadline - 2.0, time.monotonic() + min(0.35 * remaining, 220.0))
        res_a, res_score = _refine_mirror(
            best_a.copy(),
            t_end,
            rng,
            gamma=0.12,
            beta_start=30.0,
            beta_max=900.0,
            eta_start=0.30,
            max_eta=1.3,
        )
        print(
            f"final range score={res_score:.8f} best={best_score:.8f} "
            f"t={time.monotonic() - start_time:.1f}s",
            flush=True,
        )
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)

    # If the directly provided sequence is short enough and currently looks competitive,
    # include it in a final direct comparison.
    if given is not None and given.size <= 4096:
        g_exact = _exact_score_small(given)
        b_exact = _exact_score_small(best_a)
        print(
            f"direct scores: given={g_exact:.8f} best={b_exact:.8f}",
            flush=True,
        )
        if np.isfinite(g_exact) and g_exact < b_exact:
            best_a = given.copy()
            best_score = float(g_exact)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if out.size == 0:
        out = _make_uniform(256)
    if not np.isfinite(s) or s <= 1e-12:
        out = _make_uniform(max(1, out.size))
    else:
        out = out / s

    # Safety: avoid extremely large final evaluation in the direct evaluator.
    if out.size > 20000:
        out = _resample_sequence(out, 16384)

    # Final finite, normalized, nonnegative list.
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-12:
        out = _make_uniform(max(1, out.size))
    else:
        out = out / s

    print(
        f"FINAL len={out.size} internal_score={best_score:.8f} "
        f"direct_score={_exact_score_small(out):.8f} total_time={time.monotonic() - start_time:.1f}s",
        flush=True,
    )

    return [float(x) for x in out]
