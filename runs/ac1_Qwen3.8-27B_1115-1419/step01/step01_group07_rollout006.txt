    fft_size = 1
    while fft_size < size:

    A = np.fft.rfft(a, fft_size)

    e = np.exp(xm)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def conv_full(a):
    """Full convolution a*a."""
    return np.convolve(a, a)

def score(a):
    """Direct score computation."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = conv_full(a)
    max_c = c.max()
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def lse_obj_and_grad(x, tau):
    """LSE-smoothed objective and gradient."""
    a = softmax(x)
    n = len(a)
    c = conv_full(a)
    c_max = c.max()
    cs = c - c_max
    w = np.exp(cs / tau)
    w_sum = w.sum()
    w /= w_sum
    obj = tau * np.log(w_sum) + c_max
    # Gradient
    corr = np.correlate(w, a, mode='full')
    grad_a = 2.0 * corr[:n]
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x

def subgradient_step(a, n_active, step, rng):
    """Subgradient step with multiple active constraints."""
    n = len(a)
    c = conv_full(a)
    c_max = c.max()
    if c_max < 1e-15:
        return a
    # Get top n_active peaks
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    # Also add some random sub-peaks for diversity
    if rng is not None and len(c) > n_active:
        extra = rng.choice(len(c), size=min(n_active//4, len(c)-n_active), replace=False)
        active = np.concatenate([active, extra])
        active = np.sort(active)
    
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(active)
    
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

        p = max(2, n // 40)
        pattern = np.abs(np.sin(np.pi * np.linspace(0, 1, p)))
        x = np.tile(pattern, n // p + 1)[:n]
        x = np.maximum(x, 0.05)
    elif variant == 11 and best_a is not None:
        resized = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        x = resized
    else:
        x = np.ones(n)
    x = np.maximum(x, 1e-10)
    x = x / x.sum()
    return x


def propose_candidate(budget_s=2400, **kwargs):
    global _g_tau, _g_n

    start_time = time.monotonic()
    deadline = start_time + budget_s - 45

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        a0 = np.ones(256, dtype=np.float64)

    n0 = len(a0)
    a0 = np.maximum(a0, 1e-10)
    a0 = a0 / a0.sum()

    best_score = _quick_score(a0)
    best_a = a0.copy()
    rng = np.random.default_rng(777)

    print(f"Start: n={n0}, score={best_score:.6f}")

    # Multi-length targets
    target_lengths = sorted(set([
        n0, max(128, n0 // 2), min(12000, n0 * 2),
        400, 800, 1500, 3000, 5000, 7000, 10000
    ]))
    target_lengths = [l for l in target_lengths if 64 <= l <= 12000]

    # Generate diverse inits
    all_inits = []
    for n in target_lengths:
        for v in range(12):
            init = _generate_init(n, v, rng, best_a if v == 11 else None)
            all_inits.append(init)

    # Score all inits, keep top
    scored = [(_quick_score(a), a) for a in all_inits]
    scored.sort(key=lambda t: t[0])
    top_inits = scored[:10]
    for s, a in top_inits[:5]:
        print(f"  Init: n={len(a)}, score={s:.6f}")
    if top_inits[0][0] < best_score:
        best_score = top_inits[0][0]
        best_a = top_inits[0][1].copy()

    # Phase 2: L-BFGS with LSE smoothing
    print(f"\nPhase 2: L-BFGS. Best={best_score:.6f}")
    num_runs = 7
    for ri in range(num_runs):
        if time.monotonic() > deadline - 400:
            break
        if ri < len(top_inits):
            init = top_inits[ri][1].copy()
        else:
            init = best_a.copy()

        n_init = len(init)
        _g_n = n_init
        x = np.log(init + 1e-30)
        tau_schedule = [10.0 / n_init, 5.0 / n_init, 2.5 / n_init, 1.0 / n_init, 0.5 / n_init]

        print(f"  Run {ri+1}/{num_runs}: n={n_init}")
        for tau in tau_schedule:
            if time.monotonic() > deadline - 400:
                break
            _g_tau = tau
            t0 = time.monotonic()
            try:
                result = minimize(
                    _lse_objective, x, jac=_lse_gradient,
                    method='L-BFGS-B',
                    options={'maxiter': 250, 'ftol': 1e-13, 'gtol': 1e-11,
                             'maxfun': 600, 'maxls': 40}
                )
                x = result.x
            except Exception:
                break
            a_curr = _softmax(x)
            s_curr = _quick_score(a_curr)
            elapsed = time.monotonic() - t0
            if np.isfinite(s_curr) and s_curr < best_score:
                best_score = s_curr
                best_a = a_curr.copy()
                print(f"    tau={tau:.5f}: {s_curr:.6f} ({elapsed:.1f}s) *NEW*")
            else:
                print(f"    tau={tau:.5f}: {s_curr:.6f} ({elapsed:.1f}s)")

    # Phase 3: Subgradient polishing
    print(f"\nPhase 3: Subgradient. Best={best_score:.6f}")
    a = best_a.copy()
    n = len(a)
    step = 0.06
    best_in_phase = best_score
    stall = 0

    while time.monotonic() < deadline - 120:
        for _ in range(80):
            if time.monotonic() > deadline - 120:
                break
            a = _ave_peak_subgradient(a, n_active=40, step=step, rng=rng)
        s = _quick_score(a)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a.copy()
            stall = 0
        else:
            a = best_a.copy()
            step *= 0.85
            stall += 1
        step *= 0.997
        if step < 0.001:
            step = 0.02
        if stall > 60:
            # Escape local min: random perturbation
            pert = a + rng.normal(0, 0.02, n) * (a + 1e-6)
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            s_p = _quick_score(pert)
            if s_p < best_score:
                best_score = s_p
                best_a = pert.copy()
                a = pert.copy()
                print(f"  Escape: {best_score:.6f} *NEW*")
            stall = 0
            step = 0.04

        if (int)(time.monotonic() - start_time) % 30 < 1 and ri % 5 == 4:
            print(f"  SG: {best_score:.6f} step={step:.4f}")

    # Phase 4: Ultra-fine polish
    print(f"\nPhase 4: Fine polish. Best={best_score:.6f}")
    a = best_a.copy()
    step = 0.004
    for i in range(500):
        if time.monotonic() > deadline:
            break
        a = _subgradient_step(a, n_active=25, step=step, rng=rng)
        if i % 40 == 39:
            s = _quick_score(a)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_a = a.copy()
                print(f"  Fine {i}: {best_score:.6f} *NEW*")
        step *= 0.998

    # Final: one more LSE polish on best
    if time.monotonic() < deadline - 30:
        n_fin = len(best_a)
        _g_n = n_fin
        _g_tau = 0.3 / n_fin
        x_fin = np.log(best_a + 1e-30)
        try:
            result = minimize(
                _lse_objective, x_fin, jac=_lse_gradient,
                method='L-BFGS-B',
                options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-12,
                         'maxfun': 200, 'maxls': 30}
            )
            a_fin = _softmax(result.x)
            s_fin = _quick_score(a_fin)
            if np.isfinite(s_fin) and s_fin < best_score:
                best_score = s_fin
                best_a = a_fin.copy()
                print(f"  Final LSE: {best_score:.6f} *NEW*")
        except Exception:
            pass

    print(f"\nDONE: n={len(best_a)}, score={best_score:.6f}")
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    return [float(v) for v in best_a]
