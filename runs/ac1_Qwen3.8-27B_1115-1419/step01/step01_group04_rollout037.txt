def normalize(a, target_sum):

    s = a.sum()

def _greedy_polish(a, budget_s):
    """Direct peak reduction: repeatedly reduce the element contributing most to the peak."""
    a = _normalize(a.copy(), float(a.size))
    n = a.size
    target = float(n)
    t_start = time.monotonic()
    deadline = t_start + budget_s

    c = _fft_conv(a)
    score = _fast_score(a)
    best_a = a.copy()
    best_score = score
    stagnant = 0

    while time.monotonic() < deadline:
        k_star = int(np.argmax(c))
        max_val = float(c[k_star])

        ms = max(0, k_star - n + 1)
        me = min(n, k_star + 1)
        # products[i] = a[ms+i] * a[k_star - (ms+i)]
        idx1 = np.arange(ms, me)
        idx2 = k_star - idx1
        idx2 = idx2[(idx2 >= 0) & (idx2 < n)]
        idx1 = idx1[(idx1 >= 0) & (idx1 < n)]
        valid = (idx2 >= 0) & (idx2 < n)
        idx1 = idx1[valid]
        idx2 = idx2[valid]

        if len(idx1) == 0:
            break

        products = a[idx1] * a[idx2]
        top_k = min(10, len(products))
        top_local = np.argpartition(products, -top_k)[-top_k:]

        # Reduce the larger element in each top pair
        for li in top_local:
            i1 = int(idx1[li])
            i2 = int(idx2[li])
            if a[i1] >= a[i2]:
                a[i1] *= 0.92
            else:
                a[i2] *= 0.92

        a = _normalize(a, target)
        c = _fft_conv(a)
        new_score = _fast_score(a)

        if new_score < best_score:
            best_a = a.copy()
            best_score = new_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                break

    return best_a, best_score

    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    if m == 0:
        return np.ones(n, dtype=np.float64)
    if m == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, seq)
    return _normalize(out, float(n))


def _sanitize(seq):
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return [0.5] * 1024
    # Scale so sum >= 0.01
    if s < 1.0:
        seq *= 1.0 / s
    return [float(x) for x in seq]
