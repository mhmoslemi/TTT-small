import time
from scipy.optimize import linprog

def fast_conv(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def compute_score(a):
    """Compute 2n*max(conv)/sum^2."""
    n = float(len(a))
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * float(c.max()) / (s * s))

def normalize(a, target_sum):
    """Normalize non-negative array to target sum."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def lp_solve(f, M, verbose=False):
    """
    Solve: max sum(b) s.t. (f*b)[k] <= M for all k in [0, 2n-2], b >= 0.
    Returns the optimal b or None if infeasible.
    """
    n = len(f)
    if n <= 1:
        return np.array([M / max(f[0], 1e-15) if f[0] > 0 else M])
    
    # Build constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # For efficiency, use structured construction
    
    # The constraint (f*b)[k] = sum_j f[k-j] * b[j] <= M
    # where f[k-j] = 0 if k-j < 0 or k-j >= n
    
    # Build A as (2n-1) x n
    # Column j has f[0:n] shifted by j positions (only valid entries)
    
    # Actually, let's think of it differently:
    # A[k, j] = f[k-j] for max(0, k-n+1) <= j <= min(k, n-1)
    
    # This is a Toeplitz-like matrix. Let me build it efficiently.
    
    # For row k: A[k, :] = [f[k], f[k-1], ..., f[k-n+1]] with 0-padding
    # where f[i] is defined for i in [0, n-1]
    
    A = np.zeros((2*n-1, n), dtype=np.float64)
    
    # For each row k (0 to 2n-2):
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    # j ranges from max(0, k-n+1) to min(k, n-1)
    
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        if j_start > j_end:
            continue
        # A[k, j_start:j_end+1] = f[k-j_end : k-j_start+1] (reversed)
        f_slice = f[k-j_end : k-j_start+1][::-1]
        A[k, j_start:j_end+1] = f_slice
    
    b_ub = M * np.ones(2*n-1)
    c_obj = -np.ones(n)  # minimize -sum(b)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        if verbose:
            print(f"LP failed: {result.message}")
        return None

def gradient_step(a, c, K=16, eta=0.3, rng=None):
    """Multiplicative gradient step in log-space targeting top-K peaks."""
    n = len(a)
    L = len(c)
    K = min(K, L)
    
    # Get top-K indices
    if K == L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    # Subgradient for active peaks
    grad = np.zeros(n)
    inv_K = 1.0 / K
    for k in idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        # d(b[k])/d(a[j]) = a[k-j]
        # grad[j] += a[k-j] / K
        a_slice = a[k-j_end : k-j_start+1][::-1]
        grad[j_start:j_end+1] += inv_K * a_slice
    
    grad *= 2.0  # because b = a*a, dB/dA = 2A*(other)
    
    # Center the gradient (to preserve sum approximately)
    grad -= grad.mean()
    
    # Normalize gradient
    scale = np.max(np.abs(grad))
    if scale < 1e-15:
        return a
    
    grad /= scale
    
    # Multiplicative update in log-space
    if rng is not None:
        noise = rng.normal(0, 0.01, n)
    else:
        noise = 0
    
    log_mult = np.clip(-eta * grad + noise, -2, 2)
    new_a = a * np.exp(log_mult)
    new_a = np.maximum(new_a, 0)
    new_a = normalize(new_a, a.sum())
    
    return new_a

def run_lp_iteration(f, t=0.3, verbose=False):
    """One LP-based iteration."""
    n = len(f)
    c = fast_conv(f)
    M = float(c.max())
    s = float(f.sum())
    
    b = lp_solve(f, M, verbose=verbose)
    if b is None:
        return f, M, s
    
    # Rescale b to have same sum as f
    b_sum = float(b.sum())
    if b_sum > 1e-10:
        b = b * (s / b_sum)
    else:
        return f, M, s
    
    # Clip
    b = np.clip(b, 0, 1000)
    b = normalize(b, s)
    
    # Convex combination
    new_f = (1 - t) * f + t * b
    new_f = np.clip(new_f, 0, 1000)
    new_f = normalize(new_f, s)
    
    return new_f, M, s

def polish(a, time_budget, rng):
    """Gradient-based polishing."""
    n = len(a)
    t0 = time.monotonic()
    deadline = t0 + time_budget

    s = float(a.sum())

        c = fast_conv(a)
        score = float(2.0 * n * float(c.max()) / s**2)
        
        if score < best_score:

            stagnant = 0
            eta = min(0.8, eta * 1.05)
        else:

            eta = max(0.02, eta * 0.95)

        a = gradient_step(a, c, K=K, eta=eta, rng=rng)
        s = float(a.sum())
    
    return best_a, best_score

"""
LP + gradient optimization for minimizing 2n*max(conv(a))/(sum(a))^2.
Combines linear programming iterations (paper's method) with FFT-based
multiplicative gradient polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _fast_conv(a):
    n = len(a)
    L = 2*n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = float(len(a))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = _fast_conv(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))

def _normalize(a, target=None):
    if target is None:
        target = float(np.sum(a))
    a = np.clip(a.astype(np.float64), 0.0, None)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full(len(a), target / len(a))
    return a * (target / s)

def _lp_solve(f, M):
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js > je:
            continue
        A[k, js:je+1] = f[k - je : k - js + 1][::-1]
    b_ub = np.full(2*n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True,
                      'time_limit': 120})
        if res.success:
            return res.x
    except:
        pass
    return None
