def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a):

    s = a.sum()

        return np.inf

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_topk(c, a, top_k):
    """Compute weighted gradient from top-k active constraints."""
    n = len(a)
    if top_k >= c.size:
        idx = np.arange(c.size)
    else:

    count = len(idx)
    for mk in idx:
        mk = int(mk)
        lo = max(0, mk - n + 1)
        hi = min(n, mk + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * a[mk - np.arange(lo, hi)]
    g /= count
    return g
