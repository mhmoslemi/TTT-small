"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with annealed log-sum-exp smooth max approximation
2. Gerchberg-Saxton-like spectral iterations
3. Multi-start with diverse initializations
4. Progressive length increase
"""

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p

def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - float(np.max(v))
    e = np.exp(np.clip(v, -700, 700))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(v.shape, dtype=np.float64) / v.size
    return e / s

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if bmax <= 0 or not np.isfinite(bmax):
        return np.inf
    return float(2.0 * n * bmax / (s * s))

def _obj_and_grad(v, t, m, n):
    """Compute smooth objective and gradient w.r.t. v (log-space params)."""
    a = _softmax(v)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    
    bmax = float(b.max())
    x = (b - bmax) / t
    x = np.clip(x, -50.0, 50.0)
    exp_x = np.exp(x)
    log_sum_exp = bmax + t * float(np.log(exp_x.sum()))
    
    w = exp_x / float(exp_x.sum())
    
    # Gradient of f_t w.r.t. a: 2 * cross_correlation(w, a)
    W = np.fft.fft(w, m)
    cross_corr = np.fft.ifft(W * np.conj(F)).real[:n]
    grad_a = 2.0 * cross_corr
    
    # Chain rule through softmax: grad_v = a * (grad_a - dot(grad_a, a))
    ga_dot_a = float(np.dot(grad_a, a))
    grad_v = a * (grad_a - ga_dot_a)
    
    obj = 2.0 * n * log_sum_exp
    grad_v = 2.0 * n * grad_v
    
    return obj, grad_v

def _lbfgs_fun(v, t, m, n):
    return _obj_and_grad(v, t, m, n)

def _lbfgs_optimize(v_init, n, t_values, max_iter_per_t, max_time, verbose=True):
    """Run L-BFGS with annealed temperature schedule."""
    v = v_init.copy()
    m = _next_pow2(2 * n - 1)
    
    best_a = _softmax(v)
    best_score = _score(best_a)
    
    start = time.monotonic()
    
    for ti, t in enumerate(t_values):
        if time.monotonic() - start > max_time:
            break
        
        if verbose:
            print("    L-BFGS t=%.4f (iter %d/%d)" % (t, ti, len(t_values)))
        
        try:
            res = minimize(_lbfgs_fun, v, args=(t, m, n),
                          method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter_per_t,
                                   'ftol': 1e-12, 'gtol': 1e-10,
                                   'maxls': 40,
                                   'maxfun': max_iter_per_t * 3})
            v = res.x
        except Exception as e:
            if verbose:
                print("    L-BFGS error at t=%.4f: %s" % (t, str(e)))
            continue
        
        a = _softmax(v)
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            if verbose:
                print("    t=%.4f score=%.6f (improved!)" % (t, best_score))
        elif verbose:
            print("    t=%.4f score=%.6f (current best=%.6f)" % (t, sc, best_score))
    
    return best_a, float(best_score)

def _gs_iterations(a_init, n_iter, m, verbose=False):
    """Gerchberg-Saxton-like iterations for flat convolution spectrum."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0:
        return np.ones(a.size) / a.size
    a = a / s
    
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    
    # Compute target |F| for flat linear convolution
    t_flat = np.zeros(m)
    t_flat[:2*n-1] = 1.0 / (2 * n - 1)
    T_flat = np.abs(np.fft.fft(t_flat))
    
    for it in range(n_iter):
        F = np.fft.fft(a, m)
        absF = np.abs(F)
        
        # Blend current |F| with target
        alpha = 0.3 + 0.4 * float(it) / max(1, n_iter - 1)  # increase alpha over time
        absF_new = (1 - alpha) * absF + alpha * T_flat
        absF_new[0] = absF[0]  # preserve DC
        
        # Reconstruct with new magnitudes and old phases
        phase = F / np.maximum(absF, 1e-300)
        F_new = absF_new * phase
        a_new = np.fft.ifft(F_new).real[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new <= 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        
        if verbose and (it + 1) % 100 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            print("    GS iter %d score=%.6f" % (it+1, sc))
    
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a

    m = _next_pow2(2 * n - 1)
    n_params = len(v)
    
    m_adam = np.zeros(n_params)
    v_adam = np.zeros(n_params)
    t_adam = 0

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for ti, t in enumerate(t_values):

def _get_initializations(n, rng, prev_best=None):
