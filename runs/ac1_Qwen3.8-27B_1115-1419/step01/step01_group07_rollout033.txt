def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def fast_convolve(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    N = 2 * n - 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def lp_smoothed_objective(x, p):
    """Lp norm of a*a where a = softmax(x)."""
    a = softmax(x)
    c = fast_convolve(a)
    return np.sum(c ** p) ** (1.0 / p)

def generate_spectral_init(n, rng, target_flatness=0.3):
    """Generate initialization using spectral design."""
    # We want |A(f)|^2 to be as flat as possible
    # A(0) = 1 (DC component, fixed by sum=1)
    # For other frequencies, choose magnitude and random phase
    
    N = 2 * n - 1
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0  # DC
    
    # Choose magnitudes for non-DC frequencies
    # Flat target: |A_k|^2 = constant for all k
    # But we need A to correspond to a real non-negative sequence
    # Just use random phases with moderate magnitude
    for k in range(1, N):
        # Decay magnitude to help with non-negativity
        decay = np.exp(-target_flatness * k / N)
        mag = decay * rng.uniform(0.5, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = mag * np.exp(1j * phase)
        # For real signal: A[N-k] = conj(A[k])
        A[N - k] = np.conj(A[k])
    
    # Also handle the Nyquist if N is even
    if N % 2 == 0:
        A[N // 2] = np.exp(1j * rng.uniform(0, np.pi)) * np.exp(-target_flatness * 0.5)
    
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

            right = np.minimum(left + 1, len(p) - 1)
            frac = indices - left
            p_resampled = p[left] * (1 - frac) + p[right] * frac
            p_resampled = np.maximum(p_resampled, 1e-10)
            p_resampled /= p_resampled.sum()
            inits.append(p_resampled)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine-modulated (various frequencies)
    for freq in [1, 2, 3, 5, 8]:
        x = np.ones(n)
        for i in range(n):
            x[i] = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * i / n)
        x = np.maximum(x, 0.01)
        inits.append(x / x.sum())
    
    # Asymmetric linear ramp
    x = np.linspace(2.0, 0.2, n)
    inits.append(x / x.sum())
    
    # Reverse ramp
    x = np.linspace(0.2, 2.0, n)
    inits.append(x / x.sum())
    
    # Exponential decay
    for rate in [1.0, 2.0, 4.0]:
        x = np.exp(-rate * np.linspace(0, 1, n))
        inits.append(x / x.sum())
    
    # Reverse exponential
    for rate in [1.0, 2.0]:
        x = np.exp(-rate * (1 - np.linspace(0, 1, n)))
        inits.append(x / x.sum())
    
    # Two peaks (asymmetric)
    for pos in [0.25, 0.4, 0.6]:
        x = np.zeros(n)
        x[int(pos * n)] = 1.0
        # Spread
        sigma = n * 0.1
        for i in range(n):
            x[i] = 1.0 + np.exp(-((i - pos * n) ** 2) / (2 * sigma ** 2))
        x = np.maximum(x, 0.01)
        inits.append(x / x.sum())
    
    # Spectral initialization
    for _ in range(3):
        inits.append(generate_spectral_init(n, rng, target_flatness=rng.uniform(0.2, 0.8)))
    
    # Random Dirichlet with various concentrations
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    return inits


def run_optimization(a_init, n, time_budget, rng):
    """Run the full optimization pipeline on a single initialization."""
    best_score = float('inf')
    best_a = a_init.copy()
    score = evaluate_sequence(a_init.tolist())
    if np.isfinite(score):
        best_score = score
    
    x = np.log(np.maximum(a_init, 1e-30))
    deadline = time.monotonic() + time_budget
    
    # Phase 1: Lp smoothing (p=4, then p=8)
    for p in [4, 8]:
        if time.monotonic() > deadline - 10:
            break
        
        def obj(x_):
            return lp_smoothed_objective(x_, p)
        
        def grad(x_):
            return lp_smoothed_gradient(x_, p)
        
        try:
            result = minimize(
                obj, x, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': 500, 'maxls': 30}
            )
            x = result.x
        except Exception:
            pass
        
        a = softmax(x)
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    Lp={p}: {best_score:.6f}")
    
    # Phase 2: LSE smoothing with decreasing tau
    for tau_factor in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if time.monotonic() > deadline - 10:
            break
        tau = tau_factor / n
        
        def obj_lse(x_):
            return lse_objective(x_, tau)
        
        def grad_lse(x_):
            return lse_gradient(x_, tau)
        
        try:
            result = minimize(
                obj_lse, x, jac=grad_lse,
                method='L-BFGS-B',
                options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxfun': 600, 'maxls': 40}
            )
            x = result.x
        except Exception:
            pass
        
        a = softmax(x)
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    LSE tau={tau_factor/n:.6f}: {best_score:.6f}")
    
    # Phase 3: Frank-Wolfe / subgradient polishing
    a = best_a.copy()
    fw_deadline = min(time.monotonic() + time_budget * 0.3, deadline - 5)
    step = 0.1
    stuck_count = 0
    
    while time.monotonic() < fw_deadline:
        a_new, _, _ = frank_wolfe_step(a, n_active=80, step=step)
        score = evaluate_sequence(a_new.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            step *= 0.9
            if stuck_count > 50:
                # Try a perturbation
                pert = a + rng.normal(0, 0.003, n)
                pert = np.maximum(pert, 0)
                pert /= pert.sum()
                score_p = evaluate_sequence(pert.tolist())
                if score_p < best_score:
                    best_score = score_p
                    best_a = pert.copy()
                    a = pert.copy()
                    step = 0.05
                    stuck_count = 0
                else:
                    break
    
    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search function: finds best sequence minimizing the evaluation.
    Uses multi-phase optimization with diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30  # Leave 30s buffer
    
    rng = np.random.default_rng(2024)
    
    # Get parent sequence if available
    parent = None
    if 'height_sequence_1' in globals():
        parent = height_sequence_1
    
    # Choose sequence lengths to try
    lengths = [3000, 5000, 7000, 10000, 12000]
    
    best_score = float('inf')
    best_a = None
    best_n = None
    
    # Phase 1: Global search across lengths and initializations
    print("=== Phase 1: Global Search ===")
    phase1_time = min(500, budget_s * 0.2)
    time_per_length = phase1_time / len(lengths)
    
    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        
        print(f"\nLength n={n}:")
        inits = generate_diverse_inits(n, parent, rng)
        
        # Limit initializations to save time
        max_inits = max(3, min(8, int(time_per_length / 30)))
        inits = inits[:max_inits]
        
        for idx, init in enumerate(inits):
            if time.monotonic() > deadline - 600:
                break
            
            time_per_init = min(40.0, (deadline - 600 - time.monotonic()) / max(1, len(inits) - idx))
            if time_per_init < 5:
                break
            
            a, score = run_optimization(init, n, time_per_init, rng)
            
            if score < best_score:
                best_score = score
                best_a = a
                best_n = n
                print(f"  New best: n={n}, score={best_score:.6f}")
    
    print(f"\nAfter Phase 1: best_score={best_score:.6f}, best_n={best_n}")
    
    # Phase 2: Deep optimization of the best candidate
    print("\n=== Phase 2: Deep Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a = best_a.copy()
        n = best_n
        
        # Extended L-BFGS with very fine tuning
        x = np.log(np.maximum(a, 1e-30))
        
        for tau_factor in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            if time.monotonic() > deadline - 200:
                break
            tau = tau_factor / n
            
            def obj_lse2(x_):
                return lse_objective(x_, tau)
            
            def grad_lse2(x_):
                return lse_gradient(x_, tau)
            
            try:
                result = minimize(
                    obj_lse2, x, jac=grad_lse2,
                    method='L-BFGS-B',
                    options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12,
                             'maxfun': 1000, 'maxls': 50}
                )
                x = result.x
            except Exception:
                pass
            
            a_new = softmax(x)
            score = evaluate_sequence(a_new.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()
                print(f"  LSE tau={tau_factor/n:.6f}: {best_score:.6f}")
        
        # Frank-Wolfe polishing with varying active set sizes
        a = best_a.copy()
        n = best_n
        fw_time = min(300, max(60, (deadline - time.monotonic()) * 0.4))
        fw_deadline = time.monotonic() + fw_time
        
        for n_active in [20, 50, 100, 200, 500]:
            if time.monotonic() > fw_deadline:
                break
            
            step = 0.08
            stuck = 0
            while time.monotonic() < fw_deadline and stuck < 100:
                a_new, _, _ = frank_wolfe_step(a, n_active=n_active, step=step)
                score = evaluate_sequence(a_new.tolist())
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new.copy()
                    stuck = 0
                else:
                    stuck += 1
                    step *= 0.92
                    if step < 0.002:
                        break
            
            if time.monotonic() > fw_deadline:
                break
            print(f"  FW n_active={n_active}: best={best_score:.6f}")
        
        # Subgradient polishing
        a = best_a.copy()
        sg_deadline = min(time.monotonic() + 100, deadline - 50)
        step = 0.03
        while time.monotonic() < sg_deadline:
            c = fast_convolve(a)
            max_c = c.max()
            threshold = max_c * 0.98
            active_idx = np.where(c >= threshold)[0]
            if len(active_idx) > 100:
                sorted_local = np.argsort(c[active_idx])[::-1][:100]
                active_idx = active_idx[sorted_local]
            
            g = np.zeros(len(a))
            for k in active_idx:
                lo = max(0, k - len(a) + 1)
                hi = min(len(a) - 1, k)
                g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1] + a[lo:hi + 1]
            g /= len(active_idx)
            
            a_new = a - step * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0.01:
                a_new /= s
            else:
                continue
            
            score = evaluate_sequence(a_new.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                step *= 0.97
            if step < 0.0005:
                break
        
        print(f"  After FW+SG: {best_score:.6f}")
    
    # Phase 3: Try different lengths from best candidate
    print("\n=== Phase 3: Length Exploration ===")
    if time.monotonic() < deadline - 100:
        # Try to find if a different length works better
        for n_try in [best_n - 1000, best_n + 1000, best_n + 3000]:
            if n_try < 1000 or time.monotonic() > deadline - 50:
                continue
            
            # Resample best to new length
            p = best_a
            indices = np.linspace(0, len(p) - 1, n_try)
            left = np.floor(indices).astype(int)
            right = np.minimum(left + 1, len(p) - 1)
            frac = indices - left
            p_new = p[left] * (1 - frac) + p[right] * frac
            p_new = np.maximum(p_new, 1e-10)
            p_new /= p_new.sum()
            
            time_alloc = min(60.0, max(20.0, (deadline - time.monotonic()) - 30))
            a, score = run_optimization(p_new, n_try, time_alloc, rng)
            
            if score < best_score:
                best_score = score
                best_a = a
                best_n = n_try
                print(f"  Length {n_try}: new best {best_score:.6f}")
    
    # Phase 4: Final fine-tuning
    print("\n=== Phase 4: Final Polish ===")
    if time.monotonic() < deadline - 10:
        a = best_a.copy()
        n = best_n
        step = 0.01
        polish_iters = 0
        while time.monotonic() < deadline - 5:
            polish_iters += 1
            # Small subgradient steps
            c = fast_convolve(a)
            max_c = c.max()
            # Use tighter active set
            threshold = max_c * 0.995
            active_idx = np.where(c >= threshold)[0]
            if len(active_idx) > 30:
                sorted_local = np.argsort(c[active_idx])[::-1][:30]
                active_idx = active_idx[sorted_local]
            
            g = np.zeros(n)
            for k in active_idx:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1] + a[lo:hi + 1]
            g /= len(active_idx)
            
            a_new = a - step * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0.01:
                a_new /= s
            
            score = evaluate_sequence(a_new.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                step *= 0.95
                if step < 0.0003:
                    # Occasional perturbation
                    if polish_iters % 100 == 0:
                        pert = a + rng.normal(0, 0.001, n)
                        pert = np.maximum(pert, 0)
                        pert /= pert.sum()
                        score_p = evaluate_sequence(pert.tolist())
                        if score_p < best_score:
                            best_score = score_p
                            best_a = pert.copy()
                            a = pert.copy()
                            step = 0.005
                        else:
                            step = 0.005  # Reset for diversity
        
        print(f"  After final polish: {best_score:.6f}")
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== Done in {elapsed:.0f}s ===")
    print(f"Final: n={best_n}, score={best_score:.6f}")
    
    return [float(val) for val in best_a]
