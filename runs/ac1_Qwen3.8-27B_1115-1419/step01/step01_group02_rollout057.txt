def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    N = 1 << (n_out - 1).bit_length()
    return np.fft.ifft(np.fft.fft(a, N) * np.fft.fft(b, N)).real[:n_out]

    """Compute score assuming p is a probability vector (sum=1). Score = 2n*max(convolve(p,p))."""
    c = _fft_conv(p, p)
    return 2.0 * n * c.max()

def _compute_grad_smooth(p, tau, n, N2):
    """
    Compute gradient of smoothed objective tau*log(sum exp(c_k/tau)) where c=convolve(p,p).
    Returns (grad, smooth_max_value, c).

    P = np.fft.fft(p, N2)
    c = np.fft.ifft(P * P).real[:2 * n - 1]

    grad = 2.0 * h

    while time.monotonic() < deadline:
        grad, smooth_max, c, w = _compute_grad_smooth(p, tau, n, N2)
        
        # Adaptive step size: scale by gradient magnitude
        grad_scale = np.max(np.abs(grad))
        if grad_scale > 1e-15:
            eta_eff = eta * 0.5 / grad_scale
        else:
            eta_eff = eta
        
        p = _multiplicative_update(p, grad, eta_eff)
        
        tau = max(tau * tau_decay, min_tau)
        eta = max(eta * eta_decay, 1e-10)
        step += 1
        
        if step % eval_interval == 0:
            score = 2.0 * n * c.max()
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck >= patience:
                # Perturb and reset
                perturbation = np.random.standard_normal(n) * 0.05
                p = best_p + perturbation * best_p
                p = np.maximum(p, 1e-10)
                p /= p.sum()
                tau = max(tau * 5, min_tau * 10)
                stuck = 0
    
    # Final score
    c = _fft_conv(p, p)
    final_score = 2.0 * n * c.max()
    if final_score < best_score:
        best_score = final_score
        best_p = p.copy()
    
    return best_p, best_score

def _subgradient_refine(p, n, deadline, N2=None):
    """Final refinement using exact subgradient (non-smooth) with top-k peaks."""
    if N2 is None:
        N2 = 1 << (3 * n - 2).bit_length()
    
    p = p.copy()
    best_p = p.copy()
    c = _fft_conv(p, p)
    best_score = 2.0 * n * c.max()
    
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.02
    top_k = 3
    
    while time.monotonic() < deadline:
        c = _fft_conv(p, p)
        
        # Top-k peaks
        if top_k >= len(c):
            top_k = len(c)
        top_idx = np.argsort(c)[-top_k:]
        top_vals = c[top_idx]
        weights = np.exp(top_vals / max(tau_small, 1e-8)) 
        tau_small = 1e-4
        weights = np.exp(top_vals / tau_small)
        weights /= weights.sum()
        
        # Weighted mirror
        mirror = np.zeros(n)
        for wt, k in zip(weights, top_idx):
            k = int(k)
            # Mirror at position k: p[j] contributes to c_k when j + (k-j) = k
            # dc_k/dp_m = 2*p_{k-m}, so mirror contribution at m is 2*p_{k-m}
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                mirror[lo:hi+1] += wt * 2.0 * p[hi:lo-1:-1] if hi > lo else wt * 2.0 * p[k- lo:k-lo+1]
        
        # Simpler: just use the gradient formula
        # grad_m = 2 * sum_k w_k * p_{k-m}
        # = 2 * cross-correlation
        grad = np.zeros(n)
        for wt, k in zip(weights, top_idx):
            k = int(k)
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += wt * 2.0 * p[j]
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        p = p - lr * momentum
        p = np.maximum(p, 0)
        p /= p.sum()
        
        lr *= 0.9999
        
        if c.max() * 2 * n < best_score:
            best_score = 2.0 * n * c.max()
            best_p = p.copy()
    
    return best_p, best_score

def _subgradient_refine_fast(p, n, deadline, N2=None, top_k=5):
    """Faster subgradient refinement using FFT for the mirror computation."""
    if N2 is None:
        N2 = 1 << (3 * n - 2).bit_length()
    
    p = p.copy()
    best_p = p.copy()
    c = _fft_conv(p, p)
    best_score = 2.0 * n * c.max()
    
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.02
    stuck = 0
    
    while time.monotonic() < deadline:
        c = _fft_conv(p, p)
        current_score = 2.0 * n * c.max()
        
        # Softmax weights on top peaks
        tau_sm = current_score / (2 * n) * 0.1  # relative to peak
        tau_sm = max(tau_sm, 1e-6)
        c_scaled = c / tau_sm
        c_max = c_scaled.max()
        exp_c = np.exp(c_scaled - c_max)
        w = exp_c / exp_c.sum()
        
        # Gradient via cross-correlation: grad_m = 2 * sum_j p_j * w_{j+m}
        P = np.fft.fft(p, N2)
        W = np.fft.fft(w, N2)
        h = np.fft.ifft(W * np.conj(P)).real[:n]
        grad = 2.0 * h
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        p = p - lr * momentum
        p = np.maximum(p, 1e-15)
        p /= p.sum()
        lr = max(lr * 0.9998, 1e-6)
        
        if current_score < best_score:
            best_score = current_score
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 100:
            p = best_p + 0.02 * np.random.standard_normal(n)
            p = np.maximum(p, 1e-10)
            p /= p.sum()
            momentum = np.zeros(n)
            lr = 0.02
            stuck = 0
    
    return best_p, best_score
