**Convex-Concave Procedure (CCP):**

The objective is F(f) = max_k (f*f)[k], which is jointly non-convex in (first f, second f) but convex in the second argument. The CCP replaces the second f with the current iterate:

Given f^{(t)}, solve: min_g max_k (f^{(t)}*g)[k] s.t. g >= 0, sum(g) = 1

```python
import numpy as np
from scipy.sparse import csc_matrix
from scipy.optimize import linprog
import time

def next_fast_len(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def fft_conv_max(a, N=None):
    """Compute max of self-convolution using FFT."""
    n = len(a)
    if N is None:
        N = next_fast_len(2*n - 1)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2*n-1]
    return float(np.max(c))

def fft_conv(a, N=None):
    """Compute full self-convolution using FFT."""
    n = len(a)
    if N is None:
        N = next_fast_len(2*n - 1)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2*n-1]

def build_lp_data(f, n):
    """Build the data, indices, indptr for the LP constraint matrix (CSC)."""
    m = 2 * n - 1
    total_nz = n * n + m
    
    data = np.empty(total_nz, dtype=np.float64)
    indices = np.empty(total_nz, dtype=np.int32)
    indptr = np.zeros(n + 2, dtype=np.int32)
    
    for j in range(n):
        start = j * n
        data[start:start+n] = f
        indices[start:start+n] = np.arange(j, j+n, dtype=np.int32)
        indptr[j+1] = (j+1) * n
    
    start = n * n
    data[start:start+m] = -1.0
    indices[start:start+m] = np.arange(m, dtype=np.int32)
    indptr[n+1] = n * n + m
    
    return data, indices, indptr

def solve_inner_lp(f, n, data, indices, indptr):
    """Solve the inner LP given precomputed structure and current f."""
    m = 2 * n - 1
    # Update data: g-columns get f values
    for j in range(n):
```
