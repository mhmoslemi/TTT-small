def adam_update(grad, m, v, t, beta1, beta2, eps=1e-8):
    t += 1
    m = m * beta1 + (1.0 - beta1) * grad
    v = v * beta2 + (1.0 - beta2) * (grad * grad)
    mhat = m / (1.0 - beta1 ** t)
    vhat = v / (1.0 - beta2 ** t)
    update = mhat / (np.sqrt(vhat) + eps)
    update = np.where(np.isfinite(update), update, 0.0)
    return update, m, v, t

    grad = 2.0 * conv[n - 1:2 * n - 1]

                f_try[other_top] = max(0, f_try[other_top] - reduction * (f[other_top] / f[j_top]) if f[j_top] > 0 else 0)

            non_peak_js = [j for j in range(n) if (pk - j) < 0 or (pk - j) >= n or j not in set(js)]
            if non_peak_js:
                targets = rng.choice(non_peak_js, size=min(10, len(non_peak_js)), replace=False)
                per_target = (f_try[j_top] - f_try[j_top])  # This is getting complicated
                # Simpler: just add back the mass uniformly to non-peak positions
                mass_to_redistribute = f[j_top] - f_try[j_top] + (f[other_top] - f_try[other_top]) if j_top != other_top else f[j_top] - f_try[j_top]
                if mass_to_redistribute > 0:
                    per = mass_to_redistribute / len(targets)
                    for t_j in targets:
                        f_try[t_j] += per
            
            total = f_try.sum()
            if total > 0 and np.isfinite(total):
                f_try = f_try / total
                sc = fft_score_only(f_try, N)
                if np.isfinite(sc) and sc < best_local_score:
                    best_local_score = sc
                    best_local = f_try.copy()
        
        if best_local is not None:
            f = best_local
            best_score = best_local_score
        
        # Also try: increase a random non-peak position and decrease a peak contributor
        if rng.random() < 0.5:
            f_try = f.copy()
            j_dec = j_top
            j_inc = int(rng.integers(0, n))
            # Make sure j_inc doesn't contribute to the peak
            while (0 <= pk - j_inc < n) and j_inc != j_dec:
                j_inc = int(rng.integers(0, n))
            transfer = 0.05 * f[j_dec]
            f_try[j_dec] = max(0, f[j_dec] - transfer)
            f_try[j_inc] += transfer
            total = f_try.sum()
            if total > 0 and np.isfinite(total):
                f_try = f_try / total
                sc = fft_score_only(f_try, N)
                if np.isfinite(sc) and sc < best_local_score:
                    f = f_try
                    best_score = best_local_score
    
    return f, best_score


def load_initial_sequence():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None

        arr = np.asarray(obj, dtype=np.float64).ravel()

    arr = np.where(np.isfinite(arr), arr, 0.0)

def resize_sequence(arr, target_n):
    """Resize a sequence to a target length using interpolation or averaging."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = arr.size
    if n == target_n:
        return arr.copy()
    if n > target_n:
        # Average blocks
        blocks = n // target_n
        use = target_n * blocks
        arr = np.mean(arr[:use].reshape(target_n, blocks), axis=1)
        if arr.size != target_n:
            arr = arr[:target_n]
    else:
        # Interpolate
        old_idx = np.linspace(0, n - 1, n)
        new_idx = np.linspace(0, n - 1, target_n)
        arr = np.interp(new_idx, old_idx, arr)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s > 0 and np.isfinite(s):
        arr = arr / s
    else:
        arr = np.ones(target_n, dtype=np.float64) / target_n
    return arr


def run_optimization(f, n, seed=42, time_limit=60.0, verbose=False):
    """Run the main optimization for a given initial sequence and length."""
    start = time.monotonic()
    deadline = start + time_limit
    
    f = normalize_nonneg(f)
    n = f.size
    
    N = next_fast_len(2 * n)
    phase = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / float(N))
    
    # Initial score
    c = fft_convolve(f, N)
    best_score = float(2.0 * n * np.max(c)) if c.size > 0 else INF
    
    if not np.isfinite(best_score):
        f = np.ones(n, dtype=np.float64) / n
        c = fft_convolve(f, N)
        best_score = float(2.0 * n * np.max(c))
    
    best_f = f.copy()
    
    # Log-space state
    w = log_weights_from_sequence(f)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0
    
    # Cross-subproblem state
    b = f.copy()
    w_b = log_weights_from_sequence(b)
    m_b = np.zeros(n, dtype=np.float64)
    v_b = np.zeros(n, dtype=np.float64)
    t_b_adam = 0
    
    beta = 3.0
    lr = 0.08
    stall = 0
    outer = 0
    rng = np.random.default_rng(seed)
    
    # Annealing temperature
    T_anneal = 1.0
    
    while time.monotonic() < deadline:
        outer += 1
        
        # Safety check
        if not np.all(np.isfinite(f)) or not np.all(np.isfinite(b)):
            f = best_f.copy()
            b = f.copy()
            w = log_weights_from_sequence(f)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t_adam = 0
            w_b = log_weights_from_sequence(b)
            m_b = np.zeros(n, dtype=np.float64)
            v_b = np.zeros(n, dtype=np.float64)
            t_b_adam = 0
            continue
        
        # --- Cross-subproblem step (optimize b given f) ---
        F = np.fft.rfft(f, N)
        FR = phase * np.conjugate(F)
        
        n_inner = 3
        if outer % 7 == 0:
            n_inner = 5
        
        for _ in range(n_inner):
            grad_b, h = compute_cross_gradient(f, b, beta, N, phase)
            # Project gradient to maintain sum constraint (in log-space)
            grad_w_b = b * (grad_b - float(np.dot(b, grad_b)))
            grad_w_b = np.where(np.isfinite(grad_w_b), grad_w_b, 0.0)
            
            update, m_b, v_b, t_b_adam = adam_update(grad_w_b, m_b, v_b, t_b_adam, 0.9, 0.999)
            update = np.clip(update, -5.0, 5.0)
            w_b = w_b - lr * update
            w_b = np.where(np.isfinite(w_b), w_b, 0.0)
            w_b -= float(np.max(w_b))
            b = sequence_from_log_weights(w_b)
        
        # Try convex combinations of f and b
        prev_score = best_score
        best_cand = None
        best_cand_score = best_score
        
        for tval in (0.3, 0.5, 0.7, 1.0):
            cand = (1.0 - tval) * f + tval * b
            cs = float(np.sum(cand))
            if not np.isfinite(cs) or cs <= 0.0:
                continue
            cand = cand / cs
            sc = fft_score_only(cand, N)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand = cand
                best_t = tval
        
        if best_cand is not None:
            f = best_cand.copy()
            
            # Annealing: accept worse with some probability
            if best_cand_score >= best_score and rng.random() < 0.05 * T_anneal:
                pass  # Still accept (annealing)
            
            if best_cand_score < best_score:
                best_score = best_cand_score
                best_f = f.copy()
            
            if best_cand_score < prev_score:
                stall = 0
                lr = min(0.25, lr * 1.03)
            else:
                stall += 1
                lr = max(0.01, lr * 0.95)
            
            # Reset log state
            w = log_weights_from_sequence(f)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t_adam = 0
        else:
            stall += 1
            lr = max(0.01, lr * 0.97)
        
        # --- Self-gradient step (periodic) ---
        if (outer % 10 == 0 or stall > 8) and time.monotonic() < deadline:
            c = fft_convolve(f, N)
            sc_now = float(2.0 * n * np.max(c)) if c.size > 0 else INF
            
            if np.isfinite(sc_now) and sc_now < best_score:
                best_score = sc_now
                best_f = f.copy()
            
            # Try self-gradient step with multiple step sizes
            grad = compute_self_gradient(f, c, beta, N, phase)
            grad_w = f * (grad - float(np.dot(f, grad)))
            grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
            
            best_self = None
            best_self_score = best_score
            
            for mult in (0.5, 1.0, 2.0, 0.25):
                update, m_try, v_try, t_try = adam_update(grad_w, m.copy(), v.copy(), t_adam, 0.9, 0.999)
                update = np.clip(update, -5.0, 5.0)
                w_try = w - lr * mult * update
                w_try = np.where(np.isfinite(w_try), w_try, 0.0)
                w_try -= float(np.max(w_try))
                cand = sequence_from_log_weights(w_try)
                sc = fft_score_only(cand, N)
                if np.isfinite(sc) and sc < best_self_score:
                    best_self_score = sc
                    best_self = (cand, w_try, m_try, v_try, t_try)
                    best_mult = mult
            
            if best_self is not None:
                f, w, m, v, t_adam = best_self
                if best_self_score < best_score:
                    best_score = best_self_score
                    best_f = f.copy()
                stall = 0
                lr = min(0.25, lr * 1.03)
            else:
                lr = max(0.01, lr * 0.95)
        
        # --- Restart logic ---
        if stall > 30:
            # Reset to best and add noise
            if rng.random() < 0.4:
                f = best_f.copy()
            else:
                sig = float(rng.choice((0.1, 0.2, 0.3, 0.5)))
                w_try = log_weights_from_sequence(best_f) + rng.normal(0.0, sig, size=n)
                f = sequence_from_log_weights(w_try)
            
            b = f.copy()
            w = log_weights_from_sequence(f)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t_adam = 0
            w_b = log_weights_from_sequence(b)
            m_b = np.zeros(n, dtype=np.float64)
            v_b = np.zeros(n, dtype=np.float64)
            t_b_adam = 0
            stall = 0
            lr = 0.08
            T_anneal = max(0.1, T_anneal * 0.8)
        
        # Sharpen beta
        if outer % 40 == 0:
            beta = min(200.0, beta * 1.06 + 0.3)
        
        # Decay annealing
        T_anneal = max(0.0, T_anneal - 0.002)
    
    return best_f, best_score
