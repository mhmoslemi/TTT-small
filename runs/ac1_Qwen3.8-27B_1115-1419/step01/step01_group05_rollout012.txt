"""
Search for low-peak non-negative self-convolutions.

The objective is
    score(a) = 2*n*max_k (a*a)_k / (sum a)^2,      a_i >= 0,
which is scale invariant.  During search sequences are normalized to sum 1.

The program uses:
  * many structured and randomized starts (exponentials, polynomials,
    cosines, Gaussians, low-pass perturbations, and the supplied sequence);
  * FFT-based exact scores and gradients of log-sum-exp / active-peak
    smoothed surrogates;
  * projected/multiplicative gradient steps with adaptive beta, top-peak
    weights, and restarts;
  * occasional scipy L-BFGS refinements on log-softmax variables;
  * occasional LP best-response moves when scipy is available;
  * multi-scale resampling :-small lengths to discover shapes, larger
    lengths to reduce discretization error.
"""

import numpy as np
import time
import math
import operator

try:
    from scipy.optimize import linprog, minimize
    _HAS_SCIPY = True
except Exception:
    linprog = None
    minimize = None
    _HAS_SCIPY = False


_SC = {
    "n": 0,
    "beta": 1.0,
    "bfft": 0,
    "gfft": 0,
    "deadline": 0.0,
    "last_z": None,
}


class _TimeLimit(Exception):
    pass


def _next_pow2(size):
    size = int(max(1, size))
    if size <= 1:
        return 1
    return 1 << (size - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _lse_value(b, beta):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return 0.0
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    if beta <= 2.0:
        return float(np.mean(b))
    m = float(np.max(b))
    if not np.isfinite(m):
        return 1e100
    arg = float(beta) * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return m
    val = m + float(np.log(s)) / float(beta)
    return val if np.isfinite(val) else m


def _weights_lse(b, beta):
    b = np.asarray(b, dtype=np.float64).ravel()
    m = b.size
    if m == 0:
        return np.ones(1, dtype=np.float64)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    b = np.maximum(b, 0.0)
    if beta <= 2.0:
        return np.ones(m, dtype=np.float64) / float(m)
    mval = float(np.max(b))
    if mval <= 1e-300:
        return np.ones(m, dtype=np.float64) / float(m)
    arg = float(beta) * (b - mval)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    p = e / s
    return p


def _make_weights(b, beta, topk_frac, topk_mix):
    b = np.asarray(b, dtype=np.float64).ravel()
    m = b.size
    if m == 0:
        return np.ones(1, dtype=np.float64)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    b = np.maximum(b, 0.0)
    p = _weights_lse(b, beta)
    topk_mix = float(min(1.0, max(0.0, topk_mix)))
    if topk_mix > 0.0:
        K = int(float(topk_frac) * float(m))
        K = min(max(1, K), m)
        if K < m:
            idx = np.argpartition(b, -K)[-K:]
        else:
            idx = np.nonzero(b > 0.0)[0]
            if idx.size == 0:
                idx = np.arange(m)
        q = np.zeros(m, dtype=np.float64)
        q[idx] = 1.0
        qs = float(q.sum())
        if qs > 0.0:
            q = q / qs
        else:
            q = np.ones(m, dtype=np.float64) / float(m)
        p = (1.0 - topk_mix) * p + topk_mix * q
    p = (1.0 - 1e-9) * p + (1e-9 / float(m))
    p = np.maximum(p, 0.0)
    ps = float(p.sum())
    if ps <= 0.0:
        return np.ones(m, dtype=np.float64) / float(m)
    return p / ps


def _weighted_gradient(a, p, fft_size, n):
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    h = np.where(h > 0.0, h, 0.0)
    return 2.0 * h


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    bfft = _next_pow2(2 * n - 1)
    b = _autocorr(a, bfft)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.nan_to_num(z, nan=0.0, posinf=700.0, neginf=-700.0)
    m = float(np.max(z))
    w = np.exp(z - m)
    s = float(w.sum())
    if s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / max(1, z.size)
    return w / s


def _bfgs_fun(z):
    if time.monotonic() > _SC["deadline"]:
        raise _TimeLimit
    a = _softmax(z)
    b = _autocorr(a, int(_SC["bfft"]))
    return _lse_value(b, float(_SC["beta"]))


def _bfgs_jac(z):
    if time.monotonic() > _SC["deadline"]:
        raise _TimeLimit
    a = _softmax(z)
    n = a.size
    b = _autocorr(a, int(_SC["bfft"]))
    p = _weights_lse(b, float(_SC["beta"]))
    g = _weighted_gradient(a, p, int(_SC["gfft"]), n)
    fdot = float(np.dot(p, b))
    mu = 2.0 * fdot
    dz = a * (g - mu)
    return np.nan_to_num(dz, nan=0.0, posinf=0.0, neginf=0.0)


def _bfgs_step(z, beta, deadline):
    if not _HAS_SCIPY:
        return z
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n < 2:
        return z
    bfft = _next_pow2(2 * n - 1)
    gfft = _next_pow2(3 * n - 2)
    _SC["n"] = n
    _SC["beta"] = float(beta)
    _SC["bfft"] = bfft
    _SC["gfft"] = gfft
    _SC["deadline"] = float(max(deadline, time.monotonic()))
    _SC["last_z"] = z.copy()

    maxiter = 400 if n <= 96 else (200 if n <= 256 else 80)
    options = dict(maxiter=maxiter, maxls=12, ftol=1e-12, gtol=1e-9, disp=False)
    try:
        res = minimize(_bfgs_fun, z, jac=_bfgs_jac, method="L-BFGS-B", options=options)
        x = getattr(res, "x", None)
        if x is not None and np.all(np.isfinite(x)):
            return np.asarray(x, dtype=np.float64)
    except _TimeLimit:
        pass
    except Exception:
        pass
    return _SC["last_z"]


def _bfgs_refine(a, deadline, rng):
    if not _HAS_SCIPY:
        return a
    a = _normalize(a)
    n = a.size
    if n < 16 or n > 512:
        return a
    if time.monotonic() > deadline - 3.0:
        return a

    best = a.copy()
    best_score = _score_of(a)
    z = np.log(np.maximum(a, 1e-300))

    stages = (2.0, 8.0, 30.0, 120.0)
    for stag in stages:
        if time.monotonic() > deadline - 2.0:
            break
        bfft = _next_pow2(2 * n - 1)
        b = _autocorr(a, bfft)
        mb = float(np.max(b))
        if mb <= 1e-300:
            break
        beta = stag / max(mb, 1e-300)
        local_end = min(deadline - 1.0, time.monotonic() + 8.0)
        z = _bfgs_step(z, beta, local_end)
        a2 = _softmax(z)
        s2 = _score_of(a2)
        if np.isfinite(s2) and s2 < best_score:
            best = a2
            best_score = s2
            a = a2
        else:
            if rng.random() < 0.5:
                z = z + rng.normal(0.0, 0.2, size=z.size)
    return best

    return np.ones(n, dtype=np.float64) / float(n)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    v = np.exp(-float(alpha) * x) + float(floor)

    if reverse:
        x = 1.0 - x + 1e-8
    else:
        x = x + 1e-8
    v = np.power(x, float(p)) + 1e-4
    return _normalize(v)

def _make_two_gauss(n, rng, k=2):

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    return _normalize(v)


def _make_cosine_random(n, rng, harmonics=3, amplitude=0.5):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    v = np.ones(n, dtype=np.float64)
    H = max(1, int(harmonics))
    amp = float(amplitude)
    for h in range(1, H + 1):
        c = amp * float(rng.uniform(-1.0, 1.0)) / float(h)
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        v = v + c * np.cos(2.0 * np.pi * float(h) * x + ph)
    v = v + 0.65
    return _normalize(v)


def _make_smooth_random(n, rng, base=None, amplitude=0.5):
    n = max(int(n), 2)
    if base is None or base.size != n:
        base = _make_uniform(n)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 8 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _random_base(n, rng):
    r = int(rng.integers(0, 6))
    if r == 0:
        return _make_exponential(n, float(rng.uniform(0.5, 2.2)))
    if r == 1:
        return _make_power(n, float(rng.uniform(1.0, 4.0)), bool(rng.random() < 0.5))
    if r == 2:
        return _make_mixture(n, float(rng.uniform(0.5, 2.0)), float(rng.uniform(0.2, 2.0)))
    if r == 3:
        return _make_two_gauss(n, rng, int(rng.integers(2, 5)))
    if r == 4:
        return _make_cosine_random(n, rng, int(rng.integers(2, 6)), float(rng.uniform(0.3, 0.9)))
    return _make_smooth_random(n, rng, _make_uniform(n), 0.8)


def _mix_with_base(a, rng, t_max=0.4):
    n = a.size
    b = _random_base(n, rng)
    t = float(rng.uniform(0.10, float(t_max)))
    return _normalize((1.0 - t) * a + t * b)


def _block_perturb(a, rng, strength=0.25):
    a = _normalize(a)
    n = a.size
    width = max(2, int(n * 0.05))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c = a.copy()
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.25:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)


def _logit_perturb(a, rng, strength=0.5):
    a = _normalize(a)
    n = a.size
    z = np.log(np.maximum(a, 1e-300))
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 8 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    z = z + float(strength) * f
    z = z - float(np.mean(z))
    w = np.exp(np.clip(z, -700.0, 700.0))
    return _normalize(w)


def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(2, int(new_n))
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, old))


def _mix_candidates(a, b, t):
    if a.size != b.size:
        b = _resample_sequence(b, a.size)
    return _normalize((1.0 - float(t)) * a + float(t) * b)


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    s = float(arr.sum())
    if s <= 1e-300:
        return None
    return arr / s


def _lp_response(a):
    if not _HAS_SCIPY:
        return None
    a = _normalize(a)
    n = a.size
    if n < 32 or n > 512:
        return None

    bfft = _next_pow2(2 * n - 1)
    b = _autocorr(a, bfft)
    M = float(np.max(b))
    if M <= 1e-300:
        M = 1.0

    af = a + 1e-10
    af = af / float(af.sum())
    ar = af[::-1]
    A = np.empty((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        lo = n - 1 - (k - s)
        hi = n - 1 - (k - e)
        A[k, s:e + 1] = ar[lo:hi + 1]

    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"maxiter": 5000, "disp": False},
        )
    except Exception:
        return None

    if getattr(res, "status", -1) != 0:
        return None
    y = np.asarray(res.x, dtype=np.float64) if getattr(res, "x", None) is not None else None
    if y is None or y.size != n:
        return None
    y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
    y = np.maximum(y, 0.0)
    ys = float(y.sum())
    if ys <= 1e-300:
        return None
    return y / ys


def _lp_update(a, deadline, tries=None):
    if tries is None:
        tries = (0.2, 0.4, 0.7, 1.0)
    if not _HAS_SCIPY:
        return a
    a = _normalize(a)
    n = a.size
    if n < 32 or n > 512:
        return a
    if time.monotonic() > deadline - 3.0:
        return a

    y = _lp_response(a)
    if y is None:
        return a

    best = a.copy()
    best_score = _score_of(a)
    for t in tries:
        if time.monotonic() > deadline - 1.0:
            break
        t = float(t)
        cand = _normalize((1.0 - t) * a + t * y)
        s = _score_of(cand)
        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s
    return best


def _spectral_candidate(a, n_iter=25, clip_frac=0.12, deadline=None):
    a = _normalize(a)
    n = a.size
    if n < 16:
        return a
    N = _next_pow2(2 * n - 1)
    m0 = 2 * n - 1
    cur = a.copy()
    n_iter = max(1, int(n_iter))

    for _ in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(cur, N)
        b = np.fft.irfft(A * A, N)[:m0]
        maxb = float(np.max(b))
        if maxb <= 1e-300:
            break
        level = max(1e-15, maxb * (1.0 - float(clip_frac)))
        t = np.zeros(m0, dtype=np.float64)
        np.minimum(b, level, out=t)
        sumt = float(t.sum())
        if sumt < 1.0:
            t += (1.0 - sumt) / float(m0)
        elif sumt > 1.0:
            t *= 1.0 / sumt
        t = np.maximum(t, 0.0)

        target_full = np.zeros(N, dtype=np.float64)
        target_full[:m0] = t
        C = np.fft.rfft(target_full, N)
        B2 = np.maximum(C.real, 0.0)
        B = np.sqrt(B2)

        mag = np.abs(A)
        scale = np.zeros_like(B)
        np.divide(B, mag, out=scale, where=mag > 1e-300)
        A_new = A * scale

        a2 = np.fft.irfft(A_new, N)[:n].real
        a2 = np.maximum(a2, 0.0)
        s = float(a2.sum())
        if s <= 1e-300:
            break
        cur = a2 / s

    return _normalize(cur)


def _refine_sequence(a, deadline, rng, allow_lp=False, allow_spectral=True):
    a = _normalize(a)
    n = a.size
    if n < 16:
        a = _make_exponential(128, 1.0)
        n = 128

    bfft = _next_pow2(2 * n - 1)
    gfft = _next_pow2(3 * n - 2)

    b = _autocorr(a, bfft)
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb

    best = a.copy()
    best_score = score

    beta_s = 6.0
    alpha = 0.35
    topk_frac = 0.006
    topk_mix = 0.0

    it = 0
    fail = 0
    last_best = 0

    while time.monotonic() < deadline:
        st = time.monotonic()
        if st > deadline:
            break

        if not np.isfinite(maxb) or maxb <= 1e-300:
            a = best.copy()
            b = _autocorr(a, bfft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            fail = 0
            continue

        beta = beta_s / maxb
        smooth_old = _lse_value(b, beta)
        p = _make_weights(b, beta, topk_frac, topk_mix)
        grad = _weighted_gradient(a, p, gfft, n)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

        fdot = float(np.dot(p, b))
        mean_grad = float(np.mean(grad))
        h_box = grad - mean_grad
        h_log = grad - 2.0 * fdot

        scale_box = float(np.max(np.abs(h_box)))
        scale_log = float(np.max(np.abs(h_log)))
        if (scale_box <= 1e-300 or not np.isfinite(scale_box)) and (scale_log <= 1e-300 or not np.isfinite(scale_log)):
            a = _block_perturb(best, rng, 0.35)
            b = _autocorr(a, bfft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            fail = 0
            continue

        if (it % 2 == 1) and scale_log > 1e-300 and scale_log < 1e50:
            eta = alpha / max(scale_log, 1e-300)
            logf = -eta * h_log
            logf = np.clip(logf, -50.0, 50.0)
            cand = _normalize(a * np.exp(logf))
        else:
            cand = _normalize(np.maximum(a - alpha * h_box, 0.0))

        b2 = _autocorr(cand, bfft)
        b2 = np.nan_to_num(b2, nan=0.0, posinf=0.0, neginf=0.0)
        max2 = float(np.max(b2))
        score2 = 2.0 * float(n) * max2
        smooth2 = _lse_value(b2, beta)

        if score2 < best_score - 1e-15:
            best = cand.copy()
            best_score = score2
            last_best = it

        accept = False
        if (score2 < score - 1e-14) or (smooth2 < smooth_old - 1e-13):
            accept = True
        elif rng.random() < 0.012 and np.isfinite(score2) and score2 < best_score * 1.08:
            accept = True

        if accept:
            a = cand
            b = b2
            maxb = max2
            score = score2
            fail = 0
            if smooth2 < smooth_old:
                alpha = min(2.0, alpha * 1.1)
            else:
                alpha = max(0.05, alpha * 0.85)
        else:
            alpha = max(0.02, alpha * 0.55)
            fail += 1

        if fail > 80:
            alpha = max(0.05, alpha * 1.25)

        if it % 100 == 0 and it > 0:
            beta_s = min(520.0, beta_s * 1.08)
            topk_frac = min(0.045, topk_frac + 0.0006)
            topk_mix = min(0.65, topk_mix + 0.03)

        if fail > 260 or (it - last_best > 1100 and it > 350):
            a = best.copy()
            if st < deadline - 12.0:
                if allow_lp and n <= 384:
                    a_lp = _lp_update(a, min(deadline, st + 10.0), tries=(0.2, 0.4, 0.7))
                    s_lp = _score_of(a_lp)
                    if np.isfinite(s_lp) and s_lp < best_score:
                        best = a_lp
                        best_score = s_lp
                    if np.isfinite(s_lp) and s_lp <= score + 1e-12:
                        a = _normalize(0.5 * best + 0.5 * a_lp)
                    else:
                        a = _mix_with_base(a, rng, 0.25)
                else:
                    if allow_spectral and n <= 2048 and st < deadline - 20.0:
                        spec = _spectral_candidate(a, 18, 0.12, min(deadline, st + 15.0))
                        s_spec = _score_of(spec)
                        if np.isfinite(s_spec) and s_spec < best_score:
                            best = spec.copy()
                            best_score = s_spec
                            a = spec.copy()
                        else:
                            if rng.random() < 0.35:
                                a = _logit_perturb(a, rng, 0.45)
                            else:
                                a = _block_perturb(a, rng, 0.35)
                    else:
                        if rng.random() < 0.35:
                            a = _logit_perturb(a, rng, 0.45)
                        else:
                            a = _block_perturb(a, rng, 0.35)
            else:
                a = _block_perturb(a, rng, 0.35)

            b = _autocorr(a, bfft)
            b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            fail = 0
            alpha = 0.30
            beta_s = max(6.0, beta_s * 0.5)
            last_best = it

        it += 1
        if it > 300000:
            break

    if best_score >= score:
        best = a.copy()
        best_score = score
    return best, best_score


def _generate_inits(n, given, best, rng):
    n = max(32, int(n))
    inits = []

    base_exp = _make_exponential(n, 1.0)
    inits.append(base_exp)

    if best is not None:
        inits.append(_resample_sequence(best, n))
    if given is not None:
        inits.append(_resample_sequence(given, n))

    inits.append(_make_exponential(n, 0.65))
    inits.append(_make_exponential(n, 1.7))
    inits.append(_make_power(n, 2.0, False))
    inits.append(_make_power(n, 3.0, True))
    inits.append(_make_power(n, 1.4, True))

    inits.append(_make_mixture(n, 1.0, 0.6))
    inits.append(_make_mixture(n, 0.7, 1.5))
    inits.append(_make_mixture(n, 1.4, 0.4))

    inits.append(_make_cosine_params(n, 0.35, 0.0, 0.0, 0.0))
    inits.append(_make_cosine_params(n, 0.50, math.pi, 0.0, 0.0))
    inits.append(_make_cosine_params(n, 0.65, math.pi / 2.0, 0.30, math.pi))
    inits.append(_make_cosine_params(n, 0.45, math.pi / 3.0, 0.45, 2.0 * math.pi / 3.0))
    inits.append(_make_cosine_params(n, 0.55, 5.0 * math.pi / 6.0, 0.35, 1.0))

    inits.append(_make_two_gauss(n, rng, 2))
    inits.append(_make_two_gauss(n, rng, 3))
    inits.append(_make_cosine_random(n, rng, 4, 0.55))
    inits.append(_make_cosine_random(n, rng, 6, 0.85))
    inits.append(_make_smooth_random(n, rng, base_exp, 0.55))
    inits.append(_make_smooth_random(n, rng, _make_uniform(n), 0.85))
    inits.append(_random_base(n, rng))
    inits.append(_random_base(n, rng))

    out = []
    seen = set()
    step = max(1, n // 20)
    for v in inits:
        arr = np.asarray(v, dtype=np.float64).ravel()
        if arr.size != n or not np.all(np.isfinite(arr)):
            continue
        sig = tuple(float(x) for x in np.round(arr[::step], 4))
        if sig in seen:
            continue
        seen.add(sig)
        out.append(_normalize(arr))

    if not out:
        out = [base_exp]
    return out


def _search_length(n, num, sec, deadline, rng, best_a, given):
    n = max(32, int(n))
    num = max(1, int(num))
    sec = max(3.0, float(sec))

    best_stage_a = _resample_sequence(best_a, n)
    best_stage_score = _score_of(best_stage_a)

    inits = _generate_inits(n, given, best_a, rng)
    scored = []
    for init in inits:
        sc = _score_of(init)
        if np.isfinite(sc):
            scored.append((sc, init))

    if not scored:
        return best_stage_a, best_stage_score

    scored.sort(key=operator.itemgetter(0))

    selected = []
    half = max(1, num // 2)
    selected.extend([x[1] for x in scored[:half]])

    rest = scored[half:]
    if len(selected) < num and len(rest) > 0:
        idxs = list(range(len(rest)))
        rng.shuffle(idxs)
        for idx in idxs[: max(0, num - len(selected))]:
            selected.append(rest[idx][1])

    bfgs_limit = 2 if n <= 256 else (1 if n <= 512 else 0)
    bfgs_used = 0

    for i, init in enumerate(selected[:num]):
        if time.monotonic() > deadline - 5.0:
            break
        t_end = min(deadline - 2.0, time.monotonic() + sec)
        if time.monotonic() >= t_end:
            continue

        a = _normalize(init)
        sc = _score_of(a)
        if sc < best_stage_score:
            best_stage_a = a.copy()
            best_stage_score = sc

        if (
            _HAS_SCIPY
            and n <= 512
            and bfgs_used < bfgs_limit
            and t_end > time.monotonic() + 12.0
        ):
            local_deadline = min(t_end, time.monotonic() + 24.0)
            a_b = _bfgs_refine(a.copy(), local_deadline, rng)
            sc_b = _score_of(a_b)
            if np.isfinite(sc_b) and sc_b < sc:
                a = a_b
                sc = sc_b
                if sc < best_stage_score:
                    best_stage_a = a.copy()
                    best_stage_score = sc
                bfgs_used += 1
                print(
                    "  bfgs n={} score={:.8f} best={:.8f}".format(n, sc, best_stage_score),
                    flush=True,
                )

        if (
            _HAS_SCIPY
            and n <= 256
            and t_end > time.monotonic() + 8.0
            and rng.random() < 0.75
        ):
            local_deadline = min(t_end, time.monotonic() + 8.0)
            a_lp = _lp_update(a.copy(), local_deadline)
            sc_lp = _score_of(a_lp)
            if np.isfinite(sc_lp) and sc_lp < sc:
                a = a_lp
                sc = sc_lp
                if sc < best_stage_score:
                    best_stage_a = a.copy()
                    best_stage_score = sc
                print(
                    "  lp n={} score={:.8f} best={:.8f}".format(n, sc, best_stage_score),
                    flush=True,
                )

        res_a, res_score = _refine_sequence(
            a.copy(),
            t_end,
            rng,
            allow_lp=(n <= 384),
            allow_spectral=(n <= 2048),
        )
        if np.isfinite(res_score) and res_score < best_stage_score:
            best_stage_a = np.asarray(res_a, dtype=np.float64).copy()
            best_stage_score = float(res_score)
            print(
                "n={} init{}/{} refined score={:.8f} best={:.8f}".format(
                    n, i + 1, num, res_score, best_stage_score
                ),
                flush=True,
            )
        else:
            print(
                "n={} init{}/{} no improve best={:.8f}".format(n, i + 1, num, best_stage_score),
                flush=True,
            )

    return best_stage_a, best_stage_score


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    fallback = _make_exponential(1024, 1.0)
    try:
        start = time.monotonic()
        try:
            budget = float(budget_s)
        except Exception:
            budget = 2400.0
        if not np.isfinite(budget) or budget <= 0:
            budget = 2400.0
        budget = max(10.0, budget)

        reserve = 35.0 if budget > 180.0 else min(6.0, 0.2 * budget)
        deadline = start + max(5.0, budget - reserve)

        rng = np.random.default_rng(int(seed) if seed is not None else 0)
        given = _get_given_sequence()

        best_a = fallback.copy()
        best_score = _score_of(best_a)
        print("initial fallback len={} score={:.10f}".format(best_a.size, best_score), flush=True)

        if given is not None:
            g = _normalize(given)
            g_score = _score_of(g)
            if np.isfinite(g_score) and g_score < best_score:
                best_a = g.copy()
                best_score = g_score
                print("given sequence len={} score={:.10f}".format(g.size, g_score), flush=True)

        configs = [
            (128, 4, 32),
            (256, 6, 48),
            (512, 6, 68),
            (1024, 5, 92),
            (2048, 4, 116),
            (4096, 3, 140),
        ]
        base_total = float(sum(num * sec for _, num, sec in configs))
        available = max(30.0, budget - 70.0)
        scale = max(0.15, min(1.0, available / max(1.0, base_total)))

        for n_cfg, num_cfg, sec_cfg in configs:
            sec = sec_cfg * scale
            if time.monotonic() > deadline - max(5.0, sec) - 4.0:
                continue
            cand_a, cand_score = _search_length(
                int(n_cfg),
                int(num_cfg),
                sec,
                deadline,
                rng,
                best_a,
                given,
            )
            if np.isfinite(cand_score) and cand_score < best_score:
                best_a = np.asarray(cand_a, dtype=np.float64).copy()
                best_score = float(cand_score)
                print(
                    "stage n={} best overall={:.10f}".format(best_a.size, best_score),
                    flush=True,
                )

        if time.monotonic() < deadline - 80.0:
            cur_n = int(best_a.size)
            next_n = min(4096, max(512, 2 * cur_n))
            upscaled = False
            while next_n <= 4096 and next_n > cur_n and time.monotonic() < deadline - 70.0:
                if cur_n >= next_n:
                    break
                init = _resample_sequence(best_a, next_n)
                init = _normalize(0.86 * init + 0.14 * _make_exponential(next_n, 1.0))
                t_end = min(
                    deadline - 30.0,
                    time.monotonic() + max(45.0, min(150.0, 0.25 * (deadline - time.monotonic()))),
                )
                res_a, res_score = _refine_sequence(
                    init, t_end, rng, allow_lp=(next_n <= 384), allow_spectral=(next_n <= 2048)
                )
                if np.isfinite(res_score) and res_score < best_score:
                    best_a = np.asarray(res_a, dtype=np.float64).copy()
                    best_score = float(res_score)
                    cur_n = int(best_a.size)
                    upscaled = True
                    next_n = min(4096, max(cur_n + 1, 2 * cur_n))
                    print(
                        "upscale to n={} score={:.10f}".format(cur_n, best_score),
                        flush=True,
                    )
                else:
                    break

        if time.monotonic() < deadline - 6.0:
            res_a, res_score = _refine_sequence(
                best_a.copy(),
                deadline - 3.0,
                rng,
                allow_lp=(int(best_a.size) <= 384),
                allow_spectral=True,
            )
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                print("final refine score={:.10f}".format(best_score), flush=True)
            else:
                print("final refine no improve best={:.10f}".format(best_score), flush=True)

        out = _normalize(best_a)
        out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
        out = np.maximum(out, 0.0)
        out = _normalize(out)
        out_list = [float(x) for x in out]

        eval_fn = globals().get("evaluate_sequence", None)
        if callable(eval_fn):
            try:
                exact = float(eval_fn(out_list))
                print("returned len={} internal={:.10f} exact={}".format(len(out_list), best_score, exact), flush=True)
            except Exception as exc:
                print("evaluation check failed: {}".format(exc), flush=True)
        else:
            ps = _score_of(out)
            print("returned len={} score={:.10f}".format(len(out_list), ps), flush=True)

        return out_list

    except Exception as exc:
        print("fatal error: {}".format(exc), flush=True)
        fallback = _make_exponential(1024, 1.0)
        return [float(x) for x in fallback]
