"""
Search for minimum-peak nonnegative self-convolution sequences.
Combines top-K weighted gradient descent, block simulated annealing,
equioscillation enforcement, and multi-length exploration.
Target: score < 1.5027.

import numpy as np
import time

def _conv(a, b):
    """Linear convolution via FFT."""
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.array([0.0])
    size = 1 << (n1 + n2 - 2).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n1 + n2 - 1]

def _acs(a):
    """Autocorrelation a*a (linear)."""
    return _conv(a, a)

def _sc(a):
    """Score = 2*n*max(a*a) for sum-normalized nonnegative a."""
    n = len(a)
    if n == 0:
        return 1e100
    b = _acs(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * n * mb

def _norm(a):
    """Normalize to nonnegative, sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.full(a.size, 1.0 / max(a.size, 1), dtype=np.float64)
    return a / s

def _resamp(old, new_n):
    """Resample sequence to new length via linear interpolation."""
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    s = float(old.sum())
    if s < 1e-300 or old.size == 0:
        return np.ones(max(new_n, 2), dtype=np.float64) / max(new_n, 2)
    old = old / s
    if old.size == new_n:
        return old.copy()
    x1 = np.linspace(0.0, 1.0, old.size)
    x2 = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x2, x1, old)
    return _norm(new)

def _grad_weighted(a, w):
    """
    Gradient of sum_k w[k]*(a*a)[k] w.r.t. a_j.
    = 2 * sum_k w[k] * a[k-j] for 0<=k-j<n
    Computed as 2 * conv(a_reversed, w)[n-1 : 2n-1]
    """
    n = len(a)
    ar = a[::-1]
    c = _conv(ar, w)
    return 2.0 * c[n - 1:n - 1 + n]

def _topk_weights(b, beta, K):
    """
    Create weights concentrated on top-K peaks with soft falloff.
    """
    n = len(b)
    mb = float(np.max(b))
    if mb < 1e-300:
        return np.ones(n) / n
    # Soft: exp(beta * (b - mb)/mb), then boost top K
    arg = beta * (b - mb) / mb
    arg = np.maximum(arg, -50.0)
    w = np.exp(arg)
    # Boost top K
    top_idx = np.argsort(b)[-K:]
    w[top_idx] *= 3.0
    s = float(w.sum())
    if s < 1e-300:
        return np.ones(n) / n
    return w / s

def _gradient_optimize(a, deadline, rng, max_iter=20000):
    """
    Gradient-based optimization with top-K weighting and adaptive step.
    Returns (best_a, best_score).
    """
    a = _norm(a.copy())
    n = len(a)
    b = _acs(a)
    best_a = a.copy()
    best_s = _sc(a)

    beta = 12.0
    K = 3
    step = 0.08
    it = 0
    stall = 0
    last_beta_up = 0
    last_K_up = 0

    while time.monotonic() < deadline and it < max_iter:
        mb = float(np.max(b))
        if mb < 1e-300:
            break

        g = _grad_weighted(a, w)
        g_c = g - float(np.mean(g))

        if float(np.max(np.abs(g_c))) < 1e-15:

            a = _norm(best_a + 0.05 * rng.standard_normal(n))
            b = _acs(a)
            step = 0.08
            stall = 0
            it += 1
            continue

        # Try the step
        a_new = np.maximum(0.0, a - step * g_c)
        sn = float(a_new.sum())
        if sn < 1e-300:
            step *= 0.5
            it += 1
            continue
        a_new /= sn
        b_new = _acs(a_new)
        s_new = _sc(a_new)

        if s_new < best_s - 1e-15:
            best_a = a_new.copy()
            best_s = float(s_new)
            stall = 0
            step = min(0.8, step * 1.03)
            a = a_new
            b = b_new

                stall = 0
            else:
                a = a_new

def _block_anneal(a, deadline, rng, T_init=0.3, T_min=0.01, n_iterations=3000):
    """
    Block-level simulated annealing.
    Perturbs random blocks and accepts based on Metropolis criterion.
    """
    a = _norm(a.copy())
    n = len(a)
    best_a = a.copy()
    b = _acs(a)
    best_s = _sc(a)
    T = T_init

    block_sizes = [4, 8, 12, 16, 24, 32]
    it = 0
    while time.monotonic() < deadline and it < n_iterations:
        # Pick random block
        bw = int(rng.choice(block_sizes, p=[0.2, 0.25, 0.2, 0.15, 0.1, 0.1]))
        bw = min(bw, max(4, n // 4))
        start = int(rng.integers(0, max(1, n - bw + 1)))
        stop = min(n, start + bw)

        a_new = a.copy()
        # Perturb block: multiplicative noise
        noise = np.exp(rng.standard_normal(stop - start) * 0.3)
        a_new[start:stop] *= noise
        # Also try sometimes to shift mass
        if rng.random() < 0.2:
            shift_start = int(rng.integers(0, max(1, n - bw + 1)))
            shift_stop = min(n, shift_start + bw)
            a_new[shift_start:shift_stop] *= np.exp(rng.standard_normal(shift_stop - shift_start) * 0.2)

        a_new = _norm(a_new)
        b_new = _acs(a_new)
        s_new = _sc(a_new)

        delta = s_new - _sc(a)
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            a = a_new
            b = b_new
            if s_new < best_s:
                best_a = a_new.copy()
                best_s = float(s_new)

        # Cool
        T *= 0.997
        T = max(T, T_min)
        it += 1

    return best_a, best_s

def _equioscillate(a, deadline, rng, K=8):
    """
    Try to make top-K autocorrelation values equal by solving a small
    least-squares system (linearized).
    """
    a = _norm(a.copy())
    n = len(a)
    best_a = a.copy()
    b = _acs(a)
    best_s = _sc(a)

    for iteration in range(50):
        if time.monotonic() > deadline:
            break
        b = _acs(a)
        mb = float(np.max(b))
        if mb < 1e-300:
            break
        top_idx = np.argsort(b)[-K:]
        top_vals = b[top_idx]
        target = float(np.mean(top_vals))
        # Want to reduce max of top_vals toward target
        # Gradient-based: push down those above target
        w = np.zeros(2 * n - 1)
        for idx in top_idx:
            w[idx] = max(0.0, b[idx] - target) + 0.1
        w /= max(1e-300, w.sum())

        g = _grad_weighted(a, w)
        g_c = g - float(np.mean(g))
        step = 0.02
        a_new = _norm(a - step * g_c)
        b_new = _acs(a_new)
        s_new = _sc(a_new)
        if s_new < best_s:
            best_a = a_new.copy()
            best_s = float(s_new)
        a = a_new

    return best_a, best_s

        _make_cosine_window(n0, 1),
        _make_smooth_random(n0, rng, 1.0, 0.3),
        _make_smooth_random(n0, rng, 1.5, 0.4),
    ]
    if given is not None and given.size >= 50:
        inits_0.append(_resamp(given, n0))

    for idx, init in enumerate(inits_0):
        s = _sc(init)
        if s < best_score:
            best_a = init.copy()
            best_score = float(s)
    print(f"Phase 0: best={best_score:.8f} (len={best_a.size})", flush=True)

    # --- Phase 1: Gradient optimization from best at n=256 ---
    if time.monotonic() < deadline - 120:
        t_end = min(deadline - 120, time.monotonic() + 200)
        a1, s1 = _gradient_optimize(best_a.copy(), t_end, rng)
        print(f"Phase 1 (grad opt n={n0}): score={s1:.8f}", flush=True)
        if s1 < best_score:
            best_a = a1
            best_score = float(s1)

        # Also try from a couple other inits
        for idx in [0, 3, 7]:
            if time.monotonic() > deadline - 120:
                break
            t_end2 = min(deadline - 120, time.monotonic() + 60)
            a2, s2 = _gradient_optimize(inits_0[idx].copy(), t_end2, rng)
            if s2 < best_score:
                best_a = a2
                best_score = float(s2)
                print(f"  Phase 1 alt start {idx}: score={s2:.8f}", flush=True)

    # --- Phase 2: Block annealing on best ---
    if time.monotonic() < deadline - 100:
        t_end = min(deadline - 100, time.monotonic() + 120)
        a2, s2 = _block_anneal(best_a.copy(), t_end, rng, T_init=0.2)
        print(f"Phase 2 (anneal): score={s2:.8f}", flush=True)
        if s2 < best_score:
            best_a = a2
            best_score = float(s2)

    # --- Phase 3: Equioscillation enforcement ---
    if time.monotonic() < deadline - 80:
        t_end = min(deadline - 80, time.monotonic() + 60)
        a3, s3 = _equioscillate(best_a.copy(), t_end, rng, K=8)
        print(f"Phase 3 (equioscillate K=8): score={s3:.8f}", flush=True)
        if s3 < best_score:
            best_a = a3
            best_score = float(s3)
        a3b, s3b = _equioscillate(best_a.copy(), min(deadline - 60, time.monotonic() + 40), rng, K=12)
        print(f"Phase 3b (equioscillate K=12): score={s3b:.8f}", flush=True)
        if s3b < best_score:
            best_a = a3b
            best_score = float(s3b)

    # --- Phase 4: Try different lengths ---
    lengths = [224, 240, 288, 320, 384, 448, 512]
    for n_try in lengths:
        if time.monotonic() > deadline - 80:
            break
        init = _resamp(best_a, n_try)
        # Mix with a fresh exponential to avoid degenerate resampling
        init = _norm(0.8 * init + 0.2 * _make_exponential(n_try, 1.2))

        t_end = min(deadline - 80, time.monotonic() + 80)
        a4, s4 = _gradient_optimize(init.copy(), t_end, rng, max_iter=8000)
        if s4 < best_score:
            best_a = a4
            best_score = float(s4)
            print(f"Phase 4 (len={n_try}): score={s4:.8f} NEW BEST", flush=True)
        else:
            print(f"Phase 4 (len={n_try}): score={s4:.8f} (best={best_score:.8f})", flush=True)

    # --- Phase 5: More gradient optimization on best (longer run) ---
    if time.monotonic() < deadline - 30:
        t_end = min(deadline - 20, time.monotonic() + 180)
        a5, s5 = _gradient_optimize(best_a.copy(), t_end, rng, max_iter=50000)
        print(f"Phase 5 (long grad): score={s5:.8f}", flush=True)
        if s5 < best_score:
            best_a = a5
            best_score = float(s5)

    # --- Phase 6: Final anneal + equioscillate ---
    if time.monotonic() < deadline - 10:
        t_end = min(deadline - 10, time.monotonic() + 40)
        a6, s6 = _block_anneal(best_a.copy(), t_end, rng, T_init=0.08)
        if s6 < best_score:
            best_a = a6
            best_score = float(s6)
        if time.monotonic() < deadline - 5:
            a6b, s6b = _equioscillate(best_a.copy(), deadline - 3, rng, K=10)
            if s6b < best_score:
                best_a = a6b
                best_score = float(s6b)

    # --- Phase 7: Final gradient polish ---
    if time.monotonic() < deadline - 3:
        a7, s7 = _gradient_optimize(best_a.copy(), deadline - 1, rng, max_iter=20000)
        if s7 < best_score:
            best_a = a7
            best_score = float(s7)
            print(f"Phase 7 (final polish): score={s7:.8f}", flush=True)

    final_a = _norm(best_a)
    final_s = _sc(final_a)
    print(f"FINAL: len={final_a.size}, score={final_s:.8f}", flush=True)

    out = np.asarray(final_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 0:
        out = out / s
    return [float(x) for x in out]
