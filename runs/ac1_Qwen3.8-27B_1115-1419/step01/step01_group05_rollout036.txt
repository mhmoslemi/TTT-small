    return a / s

def _lse_loss(a, beta, z, zmap_size):
    """Compute LSE loss and its gradient w.r.t. a, then w.r.t. z."""
    n = a.size
    # Compute convolution
    conv = np.fft.irfft(np.fft.rfft(a, 2*zmap_size)**2, 2*zmap_size)[:2*n-1]
    # LSE
    m = conv.max()
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    lse = m + np.log(e.sum()) / beta
    # Gradient w.r.t. conv: softmax
    w = e / e.sum()
    # Gradient w.r.t. a: 2 * convolve(w_reversed_flipped, a) ... 
    # Actually: dL/da_i = 2 * sum_k w_k * a_{k-i} (for valid k-i)
    # This is 2 * (a * w_flipped)[i] where w_flipped = w[::-1]
    # Using FFT:
    w_rev = w[::-1]
    grad_a = 2.0 * np.fft.irfft(np.fft.rfft(a, 2*zmap_size) * np.fft.rfft(w_rev, 2*zmap_size), 2*zmap_size)[:n]
    # Gradient w.r.t. z:
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return lse, grad_z

def _nesterov_step(z, grad, alpha, mu, z_prev=None, zmap_size=256):
    """Nesterov momentum step."""
    if z_prev is None:
        return z - alpha * grad, z
    y = z + mu * (z - z_prev)
    # For simplicity, use the gradient at current point (not true Nesterov)
    new_z = y - alpha * grad
    return new_z, z

"""
Search for low-peak nonnegative self-convolutions using LSE-smoothed
projected gradient descent in log-parameter space (softmax reparameterization).

Key improvements over previous approach:
- Softmax parameterization eliminates projection steps
- Nesterov momentum for faster convergence
- Adaptive beta schedule with annealing restarts
- Multi-start population with diverse initializations
- Targeted perturbation based on peak-contributing elements
- Multi-length exploration with upscaling
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(max(a.size, 1)) / max(a.size, 1)
    return a / s


def _make_a_from_z(z):
    z = z - z.max()
    z = np.clip(z, -700, 700)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_z_from_a(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-30)
    return np.log(a) - np.log(a.sum())


def _convolve_fft(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2 * n - 1]


def _score_from_conv(conv, n):
    mb = float(np.max(conv))
    return 2.0 * n * mb


def _lse_and_grad_z(a, z, beta, fft_size):
    """Compute LSE loss and gradient w.r.t. z."""
    n = a.size
    conv = _convolve_fft(a, fft_size)
    m = float(np.max(conv))
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0:
        w = np.ones_like(conv) / len(conv)
        lse = m
    else:
        w = e / s
        lse = m + float(np.log(s)) / beta
    # grad w.r.t. a: dL/da_i = 2 * sum_k w_k * a_{k-i}
    # = 2 * convolve(w[::-1], a)[n-1 : 2n-1] ... actually it's correlation
    # d(conv_k)/da_i = a_{k-i} + a_{k-i} (factor of 2 from product rule)
    # dL/da_i = 2 * sum_k w_k * a_{k-i} for valid indices
    # This equals 2 * (a * w_flipped)_i in the linear convolution sense
    w_flip = w[::-1]
    grad_a = 2.0 * np.fft.irfft(
        np.fft.rfft(a, fft_size) * np.fft.rfft(w_flip, fft_size), fft_size
    )[:n]
    blend = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - blend)
    return lse, grad_z, m


def _make_exponential(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_symmetric_exp(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    xc = x - 0.5
    a = np.exp(-alpha * xc * xc)
    return _normalize(a)


def _make_clipped_sin(n, freq, phase):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return _normalize(a)


def _make_smooth_random(n, rng, base_freq=4, seed_offset=0):
    z = rng.normal(0, 1, size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // max(1, n / base_freq))
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - f.mean()
    sd = f.std()
    if sd > 0:
        f = f / sd
    a = np.exp(0.5 * f)
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.maximum(old, 0.0)
    s = old.sum()
    if s <= 0:
        return np.ones(new_n) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb_z(z, rng, smoothness=0.4):
    """Add smooth random perturbation to z."""
    n = z.size
    noise = rng.normal(0, 1, size=n)
    F = np.fft.rfft(noise)
    cutoff = max(2, n // 8)
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - f.mean()
    sd = f.std()
    if sd > 0:
        f = f / sd
    return z + smoothness * f


def _refine_z(z, deadline, rng, start_beta=8.0, max_beta=5000.0):
    """
    Optimize z (log-parameters) to minimize LSE of convolution.
    Uses Nesterov momentum with adaptive step size.
    Returns (best_z, best_score, best_a).
    """
    n = z.size
    a = _make_a_from_z(z)
    fft_size = _next_pow2(2 * n)

    # Initial evaluation
    conv = _convolve_fft(a, fft_size)
    maxb = float(np.max(conv))
    score = _score_from_conv(conv, n)
    best_z = z.copy()
    best_score = score
    best_a = a.copy()

    beta = start_beta
    alpha = 0.5
    mu = 0.85  # momentum
    v = np.zeros_like(z)  # velocity for momentum
    fail = 0
    it = 0
    last_improve_it = 0
    stagnation = 0

    while time.monotonic() < deadline:
        a = _make_a_from_z(z)
        loss, grad_z, maxb = _lse_and_grad_z(a, z, beta, fft_size)
        score = _score_from_conv(_convolve_fft(a, fft_size), n)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            last_improve_it = it

        # Nesterov momentum step
        v = mu * v + (1 - mu) * grad_z
        z_new = z - alpha * v

        # Line search (simplified: accept if smooth loss decreased)
        a_new = _make_a_from_z(z_new)
        conv_new = _convolve_fft(a_new, fft_size)
        m_new = float(np.max(conv_new))
        shifted_new = beta * (conv_new - m_new)
        shifted_new = np.clip(shifted_new, -700.0, 700.0)
        loss_new = m_new + float(np.log(np.exp(shifted_new).sum())) / beta

        if loss_new < loss - 1e-14:
            z = z_new
            alpha = min(2.0, alpha * 1.1)
            fail = 0
        else:
            alpha *= 0.5
            fail += 1
            if alpha < 0.001:
                # Restart from best with perturbation
                z = _perturb_z(best_z, rng, 0.3)
                v = np.zeros_like(z)
                alpha = 0.5
                beta = max(start_beta, beta * 0.5)
                fail = 0

        # Increase beta periodically
        if it % 50 == 0:
            beta = min(max_beta, beta * 1.15)

        # Annealing restart when stuck
        stagnation = it - last_improve_it
        if stagnation > 800 and fail > 100:
            z = _perturb_z(best_z, rng, 0.6)
            v = np.zeros_like(z)
            alpha = 0.8
            beta = max(start_beta * 0.5, beta * 0.3)
            fail = 0
            stagnation = 0

        if it % 2000 == 0:
            print(f"  it={it} beta={beta:.1f} score={score:.6f} best={best_score:.6f} "
                  f"fail={fail} stag={stagnation}", flush=True)

        it += 1

    return best_z, best_score, best_a


def _coordinate_descent(a, deadline, rng, iterations=20000):
    """Simple coordinate descent: try small changes to individual elements."""
    n = a.size
    a = _normalize(a.copy())
    fft_size = _next_pow2(2 * n)
    conv = _convolve_fft(a, fft_size)
    score = _score_from_conv(conv, n)
    best_a = a.copy()
    best_score = score

    for _ in range(iterations):
        if time.monotonic() > deadline:
            break
        i = int(rng.integers(0, n))
        orig = a[i]
        # Try multiplicative perturbation
        factor = np.exp(rng.normal(0, 0.3))
        a[i] = max(0.0, orig * factor)
        s = a.sum()
        if s < 1e-30:
            a = best_a.copy()
            continue
        a = a / s
        conv = _convolve_fft(a, fft_size)
        new_score = _score_from_conv(conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        else:
            a = best_a.copy()

    return best_a, best_score
