"""
Minimize 2n*max(convolve(a,a))/(sum(a))^2 using LP-based active-set direction finding,
simulated annealing restarts, and diverse structured initializations.
"""
import numpy as np
import time
import sys
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    if n_out < 2:
        return np.array([a[0] * b[0]])
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score(a):
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _norm(a, s):
    cur = float(np.sum(a))
    if cur < 1e-15:
        return np.full_like(a, s / len(a))
    return a * (s / cur)


def _subgrad_step(a, lr, momentum, beta, k_peak):
    n = len(a)
    b = _fft_conv(a, a)
    sa = float(np.sum(a))
    if k_peak >= n:
        k_peak = n
    idx = np.argsort(b)[-k_peak:]
    vals = b[idx]
    vmin = vals[0] if len(vals) > 0 else 0.0
    wts = (vals - vmin + 1e-12)
    wts /= wts.sum()
    mir = np.zeros(n)
    for wt, mi in zip(wts, idx):
        mi = int(mi)
        lo = max(0, mi - n + 1)
        hi = min(n - 1, mi)
        if hi >= lo:
            mir[lo:hi + 1] += wt * a[mi - np.arange(lo, hi + 1)]
    wpeak = float(np.dot(wts, vals))
    grad = mir * sa - wpeak
    momentum = beta * momentum + (1.0 - beta) * grad
    a_new = np.maximum(a - lr * momentum, 0.0)
    a_new = _norm(a_new, sa)
    return a_new, momentum, float(np.max(b))


def _run_subgrad(a, deadline, lr_i=0.06, decay=0.9998, beta=0.9, kpk=5, nskip=150):

    stuck = 0
    st = 0

                a = _norm(a, np.sum(best_a))
                mom[:] = 0.0
                lr = lr_i * 0.4
                stuck = 0
    return best_a, best_s

    subgradient descent, simulated annealing restarts, and diverse initializations.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    n = 5582
    try:
        _p = np.asarray(height_sequence_1, dtype=np.float64)
        if _p.ndim == 1 and _p.size > 0 and np.isfinite(_p).all():
            n = int(_p.size)
    except Exception:
        pass

    target = float(np.sqrt(2.0 * n))
    best = np.ones(n, dtype=np.float64)
    best_sc = np.inf

    # Evaluate parent
    try:
        _p = np.asarray(height_sequence_1, dtype=np.float64)
        if _p.ndim == 1 and _p.size == n and np.isfinite(_p).all():
            pa = _norm(np.clip(_p, 0.0, 1e6), target)
            s0 = float(evaluate_sequence(pa.tolist()))
            if np.isfinite(s0) and s0 < best_sc:
                best_sc, best = s0, pa.astype(np.float64).copy()
                print(f"[parent] score={best_sc:.6f} n={n}")
    except Exception:
        pass

    # Build initializations
    inits = []
    if np.isfinite(best_sc) and best_sc < 10.0:
        inits.append(("par", best.copy()))
    k = np.arange(n, dtype=np.float64)
    inits.append(("uni", np.ones(n, dtype=np.float64)))
    inits.append(("gam", np.random.standard_normal(n) * 0.5 + 2.0))
    for cval in [1.0, 2.0, 3.0, 4.0, 6.0]:
        r = 1.0 + cval / n
        inits.append((f"exp{cval:g}", np.power(r, k)))
    for cval in [1.5, 2.5, 3.5]:
        r = 1.0 + cval / n
        inits.append((f"rev{cval:g}", np.power(r, n - 1 - k)))
    for f in [1, 3, 7, 15]:
        inits.append((f"cos{f}", 1.5 + 0.8 * np.cos(2.0 * np.pi * f * k / n)))
    inits.append(("ramp", 1.0 + 1.5 * k / n))
    inits.append(("iramp", 2.5 - 1.5 * k / n))
    inits.append(("bimod", np.concatenate([np.full(n // 2, 2.0), np.full(n - n // 2, 0.3)])))
    inits.append(("tri", 2.0 - 4.0 * np.abs(k / n - 0.5)))

    # Phase 1: diverse subgradient exploration
    p1b = min(50.0, (deadline - time.monotonic()) * 0.18)
    print(f"\n=== Phase 1: {len(inits)} inits, {p1b:.0f}s each ===")
    cands = []
    for nm, init in inits:
        if time.monotonic() > deadline - 120.0:
            break
        ia = _norm(np.maximum(np.asarray(init, dtype=np.float64), 0.0), target)
        lb, ls = _run_subgrad(ia, time.monotonic() + p1b, lr_i=0.07, decay=0.9997,
                              beta=0.9, kpk=5, nskip=120)
        if ls < best_sc:
            try:
                av = float(evaluate_sequence(lb.tolist()))
                if np.isfinite(av) and av < best_sc:
                    best_sc, best = av, lb.astype(np.float64).copy()
            except Exception:
                pass
            print(f"  [{nm}] FFT={ls:.6f} -> updated best={best_sc:.6f}")
        cands.append((ls, lb))
        print(f"  [{nm}] FFT={ls:.6f}")
    cands.sort(key=lambda x: x[0])
    print(f"Top-1: {cands[0][0]:.6f}, Top-2: {cands[1][0]:.6f}, Top-3: {cands[2][0]:.6f}")

    # Phase 2: LP polish top candidates
    p2b = min(500.0, (deadline - time.monotonic()) * 0.55)
    print(f"\n=== Phase 2: LP polish, {p2b:.0f}s total ===")
    topk = min(4, len(cands))
    blk = p2b / topk
    for ci in range(topk):
        if time.monotonic() > deadline - 80.0:
            break
        ca, cs = cands[ci]
        lb, ls = _polish_lp_init(ca, blk)
        if ls < best_sc:
            try:
                av = float(evaluate_sequence(lb.tolist()))
                if np.isfinite(av) and av < best_sc:
                    best_sc, best = av, lb.astype(np.float64).copy()
                    print(f"  [polish-{ci}] FFT={ls:.6f} actual={av:.6f} -> updated")
                else:
                    print(f"  [polish-{ci}] FFT={ls:.6f} actual={av:.6f}")
            except Exception:
                print(f"  [polish-{ci}] FFT={ls:.6f} updated (unverified)")
        cands[ci] = (ls, lb)
    cands.sort(key=lambda x: x[0])

    # Phase 3: extended LP on best
    p3b = min(900.0, (deadline - time.monotonic()) * 0.60)
    print(f"\n=== Phase 3: extended LP polish, {p3b:.0f}s ===")
    blk3 = p3b / 4.0
    for pass_i in range(4):
        if time.monotonic() > deadline - 60.0:
            break
        da = np.maximum(best + 0.02 * rng.standard_normal(n), 0.0)
        da = _norm(da, target)
        lb, ls = _polish_lp(da, blk3)
        if ls < best_sc:
            try:
                av = float(evaluate_sequence(lb.tolist()))
                if np.isfinite(av) and av < best_sc:
                    best_sc, best = av, lb.astype(np.float64).copy()
                    print(f"  [ext-{pass_i}] FFT={ls:.6f} actual={av:.6f} -> updated")
                else:
                    print(f"  [ext-{pass_i}] FFT={ls:.6f} actual={av:.6f}")
            except Exception:
                print(f"  [ext-{pass_i}] FFT={ls:.6f} updated (unverified)")
    # Phase 4: final fine-tuning
    if time.monotonic() < deadline - 30.0:
        print(f"\n=== Phase 4: fine-tuning ===")
        fb = deadline - time.monotonic()
        da = best.copy()
        lb, ls = _run_subgrad(da, deadline - 5.0, lr_i=0.015, decay=0.99995,
                              beta=0.95, kpk=3, nskip=80)
        if ls < best_sc:
            try:
                av = float(evaluate_sequence(lb.tolist()))
                if np.isfinite(av) and av < best_sc:
                    best_sc, best = av, lb.astype(np.float64).copy()
                    print(f"  [fine] FFT={ls:.6f} actual={av:.6f} -> updated")
                else:
                    print(f"  [fine] FFT={ls:.6f} actual={av:.6f}")
            except Exception:
                print(f"  [fine] FFT={ls:.6f} updated (unverified)")

    # Final verification
    final = float(evaluate_sequence(best.tolist()))
    print(f"\nFinal verified score: {final:.6f}")
    print(f"Tracked best score:   {best_sc:.6f}")
    if final < best_sc:
        best_sc = final
    return [float(x) for x in best]
