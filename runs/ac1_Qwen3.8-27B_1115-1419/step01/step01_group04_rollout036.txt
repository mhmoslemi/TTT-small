"""
Hybrid subgradient + simulated annealing optimizer for minimizing
2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.

Strategy:
- Start from parent sequence (height_sequence_1) at its native length.
- Use FFT-based convolution for fast evaluation.
- Phase 1: Explore with adaptive subgradient + SA acceptance, varying K.
- Phase 2: Refine with smaller steps, focusing on top peaks.
- Phase 3: Polish with coordinate descent and very fine steps.
- Periodic smooth perturbations to escape local minima.
"""

import numpy as np
import time


def _normalize(a, target_sum):
    """Normalize array to have the given sum, keeping non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) * (target_sum / len(a))
    a *= target_sum / s
    return np.clip(a, 0.0, 1000.0)


def _fft_conv(a):
    """Fast FFT-based linear self-convolution, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(a, c):
    """Compute objective from precomputed convolution."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * float(c.max()) / (s * s))


def _score(a):
    """Compute objective directly."""
    c = _fft_conv(a)
    return _score_from_conv(a, c)


def _subgrad_topK(a, c, K):
    """Subgradient of average of top-K convolution values w.r.t. a."""
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(int(K), L))
    grad = np.zeros(n, dtype=np.float64)

    if K == 1:
        k_idx = int(np.argmax(c))
        start = max(0, k_idx - (n - 1))
        end = min(n, k_idx + 1)
        j_start = k_idx - (end - 1)
        j_end = k_idx - start + 1
        grad[start:end] = a[j_start:j_end][::-1]
        return 2.0 * grad
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        for k_idx in top_idx:
            k_idx = int(k_idx)
            start = max(0, k_idx - (n - 1))
            end = min(n, k_idx + 1)
            j_start = k_idx - (end - 1)
            j_end = k_idx - start + 1
            grad[start:end] += a[j_start:j_end][::-1]
        return 2.0 * grad / float(K)


def _smooth_perturb(a, rng, strength):
    """Apply a smooth multiplicative perturbation."""
    n = len(a)
    if n < 3:
        return a.copy()
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n)
    for m in range(1, 5):
        amp = strength / (0.5 + 0.5 * m)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    a_new = a * np.exp(z)
    # Also add local noise
    block = max(2, n // 16)
    for _ in range(3):
        start = int(rng.integers(0, max(1, n - block)))
        a_new[start:start+block] *= np.exp(rng.normal(0, strength * 0.5, block))
    return _normalize(a_new, float(a.sum()))


def _coordinate_descent_step(a, c, n, rng, num_steps=100):
    """Try random coordinate perturbations and accept improvements."""
    best = a.copy()
    best_score = _score_from_conv(a, c)
    for _ in range(num_steps):
        i = int(rng.integers(0, n))
        old_val = a[i]
        # Try a few multiplicative factors
        for factor in [0.7, 0.85, 1.0 / 0.85, 1.3]:
            a[i] = max(0.0, old_val * factor)
            s = a.sum()
            if s < 0.01:
                a[i] = old_val
                continue
            # Quick score: only recompute affected convolution terms
            # For speed, just compute full score occasionally
            c_new = _fft_conv(a)
            sc = _score_from_conv(a, c_new)
            if sc < best_score:
                best = a.copy()
                best_score = sc
                c = c_new
            else:
                break
        a[i] = old_val
    return best, best_score, c
