```python
"""
Searches for a non-negative step function minimizing
  2*n*max(convolve(a,a)) / (sum(a))^2.

This implementation uses a multi-resolution continuous search.  The core
driver is a natural-gradient / multiplicative projected update in log-space,
with FFT-based convolution and an FFT-based convolution-correlation gradient.
Instead of a hard active set, the gradient uses a smooth power-weighted max
surrogate w_k proportional to c_k^p, with p annealed upward while optimizing.
The method is warm-started from the supplied construction when available and
also uses diverse analytic and random starts.  Long sequences are refined
with a final global phase followed by a sharper polishing phase.
"""

import numpy as np
import time


def next_power_of_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _score_key(pair):
    return pair[0]


def normalize_sum(x, target_sum):
    """Clip to valid range and rescale to the requested sum, if possible."""
    a = np.asarray(x, dtype=np.float64)
    if a.size == 0:
        return np.zeros(0, dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    n = int(a.size)
    target = float(target_sum)
    if target <= 0.0:
        target = float(n)

    for _ in range(6):
        s = float(a.sum())
        if s <= 1e-300:
            a = np.full(n, target / max(1, n), dtype=np.float64)
            break
        a = a * (target / s)
        a = np.clip(a, 0.0, 1000.0)
        s2 = float(a.sum())
        if abs(s2 - target) <= 1e-12 * max(1.0, target):
            break

    s = float(a.sum())
    if s <= 1e-300:
        a.fill(1.0)
    else:
        a = a * (target / s)
        a = np.clip(a, 0.0, 1000.0)
    return a


def fast_convolution_square(a, N=None):
    """FFT linear self-convolution of a, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    if N is None:
        N = next_power_of_two(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def fast_score_array(a):
    """Fast floatFFT score equivalent to evaluate_sequence for valid input."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")

    c = fast_convolution_square(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(c.max()) / (s * s))


def score_from_c(a, c):
    """Score from an already computed self-convolution."""
    n = int(np.asarray(a).size)
    s = float(np.sum(np.asarray(a)))
    if s < 0.01 or c.size == 0:
        return float("inf")
    return float(2.0 * n * float(c.max()) / (s * s))


def grad_from_c_power(a, c, p, N, L, w_buf):
    """
    Gradient for the smooth surrogate based on weights proportional to c_k^p.

    For active weights w_k summing to 1,
        d(c_w)/d(a_i) = 2 * sum_k w_k a_{k-i}.
    The sum is computed as a zero-padded correlation using FFT.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    s = np.zeros(n, dtype=np.float64)

    maxc = float(np.max(c)) if c.size else 0.0
    if maxc <= 1e-300 or not np.isfinite(maxc):
        return 2.0 * s

    cm = c / maxc
    cm = np.where(np.isfinite(cm), cm, 0.0)
    p = float(p)

    if p <= 1.0:
        w = cm * 1.0
    else:
        logw = p * np.log(cm + 1e-300)
        if logw.size:
            logw -= float(np.max(logw))
        w = np.exp(logw)

    w = np.where(np.isfinite(w), w, 0.0)
    ws = float(w.sum())
    if ws <= 1e-300:
        w = np.full(L, 1.0 / max(1, L), dtype=np.float64)
    else:
        w = w / ws

    if w_buf.size != N:
        w_buf = np.zeros(N, dtype=np.float64)
    w_buf[:L] = w
    if L < N:
        w_buf[L:] = 0.0

    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_buf, N)
    s = np.fft.irfft(fa * np.conjugate(fw), N)[:n]
    s = np.where(np.isfinite(s), s, 0.0)
    s = np.maximum(s, 0.0)
    return 2.0 * s


def upsample_sequence(seq, n):
    """Linearly interpolate sequence to length n and normalize it."""
    n = int(n)
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    m = int(seq.size)
    if m == 0:
        return normalize_sum(np.ones(n, dtype=np.float64), n)
    if m == n:
        return normalize_sum(seq, n)

    xold = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    xnew = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    out = np.interp(xnew, xold, seq)
    return normalize_sum(out, n)


def smooth_perturbed_sequence(a, rng, strength):
    """A smooth random multiplicative perturbation, with a mild anti-palindromic part."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    if n == 0:
        return a
    if n > 2:
        t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        z = np.zeros(n, dtype=np.float64)
        for m in range(1, 5):
            amp = float(strength) / (1.0 + 0.7 * float(m))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)
        a *= np.exp(z)

    if n > 4:
        za = rng.normal(0.0, float(strength), n)
        za = za - za[::-1]
        za_max = float(np.max(np.abs(za))) if za.size else 0.0
        if za_max > 1e-300:
            a *= np.exp(0.5 * float(strength) * np.tanh(za / za_max))

    return normalize_sum(a, n)


def make_uniform(n):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    return normalize_sum(np.ones(n, dtype=np.float64), n)


def make_anti_sine(n, alpha):
    """Positive sequence with anti-palindromic shape, useful for reducing center peak."""
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = 1.0 + float(alpha) * np.sin(np.pi * (2.0 * t - 1.0))
    a = np.maximum(a, 1e-8)
    return normalize_sum(a, n)


def make_random_anti(n, rng, beta):
    """Smooth anti-palindromic random pattern."""
    n = int(n)
    if n <= 2:
        return np.ones(max(1, n), dtype=np.float64)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    phase = np.pi * (2.0 * t - 1.0)
    q = np.zeros(n, dtype=np.float64)
    for m in range(1, 7):
        amp = rng.uniform(-0.7, 0.7) / (0.6 + 0.55 * float(m))
        q += amp * np.sin((2 * m - 1) * phase)
    q = q - q[::-1]
    q_max = float(np.max(np.abs(q))) if q.size else 0.0
    if q_max > 1e-300:
        q /= q_max
    a = 1.0 + float(beta) * np.tanh(q)
    a = np.maximum(a, 1e-8)
    return normalize_sum(a, n)


def make_random_sym(n, rng, beta):
    """Smooth low-frequency positive random pattern."""
    n = int(n)
    if n <= 2:
        return np.ones(max(1, n), dtype=np.float64)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, 7):
        amp = rng.uniform(-0.8, 0.8) / (0.6 + 0.6 * float(m))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)
    z -= float(np.mean(z))
    z_max = float(np.max(np.abs(z))) if z.size else 0.0
    if z_max > 1e-300:
        z /= z_max
    a = 1.0 + float(beta) * np.tanh(z)
    a = np.maximum(a, 1e-8)
    return normalize_sum(a, n)


def make_two_block(n, frac, ratio):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    m = int(round(float(frac) * float(n)))
    m = max(1, min(n - 1, m))
    denom = float(m) * float(ratio) + float(n - m)
    if denom <= 0.0:
        B = 1.0
    else:
        B = float(n) / denom
    A = B * float(ratio)
    a = np.empty(n, dtype=np.float64)
    a[:m] = A
    a[m:] = B
    a = np.maximum(a, 1e-8)
    return normalize_sum(a, n)


def make_block_sequence(n, rng, num_blocks):
    n = int(n)
    if n <= 2:
        return np.ones(max(1, n), dtype=np.float64)

    num_blocks = int(num_blocks)
    num_blocks = max(2, min(num_blocks, max(2, n // 2)))
    edges = np.linspace(0.0, float(n), num_blocks + 1).astype(int)
    edges[0] = 0
    edges[-1] = n
    lengths = np.diff(edges)
    if np.any(lengths <= 0):
        lengths = np.ones(num_blocks, dtype=int)
        lengths[-1] = n - int(lengths.sum()) + int(lengths[-1])
        if lengths[-1] <= 0:
            lengths = np.repeat(int(n / num_blocks), num_blocks)
            lengths[-1] += n - int(lengths.sum())

    vals = rng.uniform(0.2, 2.5, num_blocks)
    if rng.random() < 0.35:
        half = max(1, num_blocks // 2)
        vals[:half] *= rng.uniform(0.35, 0.95)
        vals[half:] *= rng.uniform(0.75, 1.4)

    a = np.repeat(vals, lengths)
    a = np.maximum(a, 1e-8)
    return normalize_sum(a, n)


def add_start(out, seq, target_n):
    """Evaluate a start candidate and append it if finite."""
    try:
        seq = np.asarray(seq, dtype=np.float64).reshape(-1)
        if seq.size != target_n:
            seq = upsample_sequence(seq, target_n)
        seq = normalize_sum(seq, target_n)
        sc = fast_score_array(seq)
        if np.isfinite(sc):
            out.append((float(sc), np.asarray(seq, dtype=np.float64).copy()))
    except Exception:
        pass


def make_starts(n, parent, rng, count=40):
    """Create a diverse pool of starting sequences for length n."""
    n = int(n)
    out = []

    parent_seq = None
    if parent is not None:
        try:
            parent_seq = upsample_sequence(parent, n)
        except Exception:
            parent_seq = None

    if parent_seq is not None:
        add_start(out, parent_seq, n)
        for strength in (0.12, 0.30, 0.55):
            add_start(out, smooth_perturbed_sequence(parent_seq, rng, strength), n)

    add_start(out, make_uniform(n), n)

    for alpha in (0.25, 0.45, 0.65, 0.80, 0.92, 0.98):
        add_start(out, make_anti_sine(n, alpha), n)

    for _ in range(6):
        beta = rng.uniform(0.35, 0.95)
        add_start(out, make_random_anti(n, rng, beta), n)

    for _ in range(6):
        beta = rng.uniform(0.35, 0.90)
        add_start(out, make_random_sym(n, rng, beta), n)

    for _ in range(6):
        max_blocks = max(5, min(64, n // 2 + 1))
        nb = int(rng.integers(4, max_blocks + 1))
        add_start(out, make_block_sequence(n, rng, nb), n)

    for frac in (0.20, 0.30, 0.40, 0.50):
        for ratio in (0.45, 0.75, 1.35, 2.20):
            add_start(out, make_two_block(n, frac, ratio), n)

    out.sort(key=_score_key)
    return out


def local_search(a, budget_s, seed=0, p_start=2.0, p_end=8.0,
                 step_init=0.25, momentum=0.40, noise=0.01, stagnation_limit=1000):
    """
    Log-space natural-gradient multiplicative local search.

    Maintains a nonnegative sequence with sum = n.  Direction is centered
    natural direction a_i * grad_i - mean(a_i * grad_i).  The power p of
    the smooth max surrogate is annealed during the phase.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    if n == 0:
        return a, float("inf")

    a = normalize_sum(a, n)
    budget = max(0.0, float(budget_s))

    c = fast_convolution_square(a)
    score = score_from_c(a, c)

    if budget < 1.0:
        return a, score

    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    deadline = t0 + budget

    best = a.copy()
    best_score = score

    step = float(step_init)
    step_min = 0.02
    step_max = 1.0
    beta = float(momentum)
    vel = np.zeros(n, dtype=np.float64)
    no_imp = 0
    accept_tol = 0.025
    phase_start = t0

    L = 2 * n - 1
    N = next_power_of_two(3 * n - 2)
    w_buf = np.zeros(N, dtype=np.float64)

    floor = 1e-12

    while time.monotonic() < deadline - 1e-3:
        if not np.all(np.isfinite(a)) or float(a.sum()) <= 0.0:
            a = best.copy()
            c = fast_convolution_square(a)
            score = score_from_c(a, c)
            vel = np.zeros(n, dtype=np.float64)

        phase_elapsed = min(budget, max(0.0, time.monotonic() - phase_start))
        prog = float(phase_elapsed) / max(1e-9, float(budget))
        p = float(p_start) + (float(p_end) - float(p_start)) * (prog ** 0.8)

        grad = grad_from_c_power(a, c, p, N, L, w_buf)
        if not np.all(np.isfinite(grad)):
            break

        glog = a * grad
        gm = float(np.mean(glog))
        d = glog - gm
        scale = float(np.max(np.abs(d))) if n > 0 else 0.0

        if scale < 1e-14:
            if no_imp > stagnation_limit:
                break
            vel = rng.normal(0.0, 0.06, n)
        else:
            d = d / scale
            vel = beta * vel - step * d

        if noise > 0.0 and rng.random() < 0.02:
            vel = vel + float(noise) * rng.normal(0.0, 1.0, n)

        vmax = float(np.max(np.abs(vel))) if n > 0 else 0.0
        if vmax > 2.0:
            vel = vel * (2.0 / vmax)

        cand = a * np.exp(vel)
        cand = np.maximum(cand, floor)
        cand = normalize_sum(cand, n)

        if not np.all(np.isfinite(cand)) or float(cand.sum()) < 0.01:
            cand = best.copy()
            vel = np.zeros(n, dtype=np.float64)
            step = max(step_min, step * 0.7)
            no_imp += 1
        else:
            c2 = fast_convolution_square(cand)
            sc2 = score_from_c(cand, c2)
            accepted = False

            if np.isfinite(sc2) and sc2 <= score * (1.0 + accept_tol) + 1e-14:
                a = cand
                c = c2
                score = sc2
                accepted = True

                if sc2 < best_score:
                    best = cand.copy()
                    best_score = sc2
                    no_imp = 0
                    step = min(step_max, step * 1.07)
                    beta = min(0.65, beta + 0.004)
                    accept_tol = max(1e-5, accept_tol * 0.999)
                else:
                    if sc2 < score_from_c(a, c) * 1.0000001:
                        step = min(step_max, step * 1.01)
                        accept_tol = max(1e-5, accept_tol * 0.9995)
                    else:
                        step = max(step_min, step * 0.98)
                    vel = vel * 0.85
                    no_imp += 1
            else:
                step = max(step_min, step * 0.7)
                vel = vel * 0.5
                no_imp += 1

        if no_imp > stagnation_limit:
            strength = 0.15 + 0.40 * float(rng.random())
            a = smooth_perturbed_sequence(best, rng, strength)
            c = fast_convolution_square(a)
            score = score_from_c(a, c)
            vel = np.zeros(n, dtype=np.float64)
            step = float(step_init)
            beta = float(momentum)
            accept_tol = 0.03
            no_imp = 0
            phase_start = time.monotonic()
            if not np.isfinite(score) or score > best_score * 1.25:
                a = best.copy()
                c = fast_convolution_square(a)
                score = score_from_c(a, c)

    return best, best_score


def run_level(n, budget_s, prev, seed=0, p_end=6.0, step_init=0.25):
    """Optimize several starts at one sequence length."""
    n = int(n)
    budget = max(0.0, float(budget_s))
    t0 = time.monotonic()
    deadline = t0 + budget

    if budget < 12.0:
        if prev is not None and np.asarray(prev).size > 0:
            a = upsample_sequence(prev, n)
        else:
            a = normalize_sum(np.ones(n, dtype=np.float64), n)
        return a, fast_score_array(a)

    rng = np.random.default_rng(int(seed))
    count = min(64, max(28, int(n // 64) + 20))
    starts = make_starts(n, prev, rng, count)

    if not starts:
        a = normalize_sum(np.ones(n, dtype=np.float64), n)
        sc = fast_score_array(a)
        starts = [(sc, a)]

    idxs = [0]
    if len(starts) > 8:
        idxs.append(len(starts) // 4)
    if len(starts) > 20:
        idxs.append(len(starts) // 2)

    if budget < 120.0:
        idxs = idxs[:1]
    elif budget < 320.0:
        idxs = idxs[:2]
    else:
        idxs = idxs[:3]

    best_seq = starts[idxs[0]][1].copy()
    best_score = float(starts[idxs[0]][0])

    fracs = [0.60, 0.30, 0.10]
    seen = set()

    for i, idx in enumerate(idxs):
        now = time.monotonic()
        rem = deadline - now
        if rem < 18.0:
            break
        if idx in seen or idx < 0 or idx >= len(starts):
            continue
        seen.add(idx)

        alloc = min(budget * fracs[min(i, len(fracs) - 1)], rem - 5.0)
        if alloc < 10.0:
            break

        sc, sq = starts[idx]
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = sq.copy()

        try:
            seq, sc2 = local_search(
                sq,
                alloc,
                seed=int(seed * 100 + idx),
                p_start=2.0,
                p_end=p_end,
                step_init=step_init,
                momentum=0.40,
                noise=0.01,
                stagnation_limit=800,
            )
        except Exception:
            continue

        if seq is not None and np.isfinite(sc2) and sc2 < best_score:
            best_score = float(sc2)
            best_seq = np.asarray(seq, dtype=np.float64).copy()

    return best_seq, best_score


def add_candidate_pool(pool, seq, score):
    """Maintain a small sorted pool of candidate sequences."""
    if seq is None:
        return
    try:
        sc = float(score)
    except Exception:
        return
    if not np.isfinite(sc):
        return

    arr = np.asarray(seq, dtype=np.float64).reshape(-1).copy()
    if arr.size == 0:
        return

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return

    arr = normalize_sum(arr, arr.size)
    sc = fast_score_array(arr)
    if not np.isfinite(sc):
        return

    pool.append((float(sc), arr))
    pool.sort(key=_score_key)
    if len(pool) > 8:
        pool.pop()


def get_previous_sequence():
    """Return the global height_sequence_1 if present, otherwise None."""
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).reshape(-1)
        if arr.size == 0:
            return None
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s < 0.01:
            return None
        return normalize_sum(arr, arr.size)
    except Exception:
        return None


def sanitize_return(seq):
    """Convert final sequence into a valid Python list of nonnegative finite floats."""
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    if seq.size == 0:
        return [1.0] * 1024

    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    n = int(seq.size)

    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 1024

    target = float(n)
    scale = target / s
    if seq.max() * scale <= 1000.0:
        seq = seq * scale
    else:
        seq = np.clip(seq, 0.0, 1000.0)
        s2 = float(seq.sum())
        if s2 < 0.01:
            return [1.0] * 1024

    seq = np.clip(seq, 0.0, 1000.0)
    if float(seq.sum()) < 0.01:
        return [1.0] * 1024

    return [float(x) for x in seq]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.

    Uses coarse-to-fine multi-resolution local search, warm-started from the
    previous construction when available.  Long phases and a final sharpening
    phase are run at a larger length.  A few top candidates are exactly
    validated before returning.
    """
    t0 = time.monotonic()
    budget = min(2400.0, max(0.0, float(budget_s)))

    prev = get_previous_sequence()
    if prev is None:
        prev = make_anti_sine(512, 0.82)

    if budget < 20.0:
        return sanitize_return(prev)

    reserve = 130.0
    if budget < reserve + 30.0:
        reserve = min(30.0, 0.2 * budget)
    search_deadline = t0 + max(10.0, budget - reserve)

    global_best = normalize_sum(prev, prev.size)
    global_score = fast_score_array(global_best)
    if not np.isfinite(global_score):
        global_best = make_uniform(1024)
        global_score = fast_score_array(global_best)

    print("initial fast score:", global_score, flush=True)

    pool = []
    add_candidate_pool(pool, global_best, global_score)
    rng = np.random.default_rng(int(seed))

    rem = search_deadline - time.monotonic()
    if rem <= 30.0:
        return sanitize_return(global_best)

    if rem >= 1500.0:
        final_n = 16384
        levels = [1024, 2048, 4096, 8192]
    elif rem >= 750.0:
        final_n = 8192
        levels = [1024, 2048, 4096]
    elif rem >= 350.0:
        final_n = 4096
        levels = [1024, 2048]
    elif rem >= 180.0:
        final_n = 2048
        levels = [1024]
    else:
        final_n = 1024
        levels = []

    # Coarse levels.
    if levels:
        coarse_budget = min(900.0, 0.35 * rem)
        if coarse_budget > 60.0:
            weights = [1.0, 1.8, 2.8, 4.0][:len(levels)]
            total_w = float(sum(weights))
            scale_budget = coarse_budget / max(1e-9, total_w)
            parent = global_best

            for idx, lv in enumerate(levels):
                now = time.monotonic()
                if now >= search_deadline - 80.0:
                    break
                b = scale_budget * float(weights[idx])
                if b < 15.0:
                    continue
                try:
                    seq, sc = run_level(
                        lv,
                        b,
                        parent,
                        seed=int(seed + 1000 + idx),
                        p_end=6.0,
                        step_init=0.25,
                    )
                except Exception:
                    seq, sc = None, float("inf")

                if seq is not None and np.isfinite(sc):
                    parent = seq
                    if float(sc) < global_score:
                        global_score = float(sc)
                        global_best = np.asarray(seq, dtype=np.float64).copy()
                        add_candidate_pool(pool, global_best, global_score)
                        print("coarse n=", lv, "score=", global_score, flush=True)

    # Final length start pool.
    rem = search_deadline - time.monotonic()
    if rem <= 20.0:
        return sanitize_return(global_best)

    count = 56 if final_n > 8192 else 40
    starts = make_starts(final_n, global_best, rng, count)
    if not starts:
        starts = [(fast_score_array(make_uniform(final_n)), make_uniform(final_n))]

    for sc, sq in starts[:8]:
        if np.isfinite(sc) and float(sc) < global_score:
            global_score = float(sc)
            global_best = np.asarray(sq, dtype=np.float64).copy()
            add_candidate_pool(pool, global_best, global_score)

    # Quick diverse refinement at final length.
    rem = search_deadline - time.monotonic()
    if rem > 180.0:
        quick_budget = min(1000.0, 0.35 * rem)
        if quick_budget > 40.0:
            idxs = [0]
            if len(starts) > 6:
                idxs.append(1)
            if len(starts) > 12:
                idxs.append(len(starts) // 4)
            if len(starts) > 24:
                idxs.append(len(starts) // 2)
            if len(starts) > 40:
                idxs.append(3 * len(starts) // 4)

            max_runs = 1
            if quick_budget > 250.0:
                max_runs = 2
            if quick_budget > 600.0:
                max_runs = 3
            idxs = idxs[:max_runs]

            fracs = [0.58, 0.30, 0.12]
            seen = set()

            for i, idx in enumerate(idxs):
                now = time.monotonic()
                rem_local = search_deadline - now
                if rem_local < 60.0:
                    break
                if idx in seen or idx < 0 or idx >= len(starts):
                    continue
                seen.add(idx)

                alloc = min(quick_budget * fracs[min(i, 2)], rem_local - 45.0)
                if alloc < 20.0:
                    break

                sc, sq = starts[idx]
                try:
                    seq, sc2 = local_search(
                        sq,
                        alloc,
                        seed=int(seed + 7000 + idx),
                        p_start=2.0,
                        p_end=8.0,
                        step_init=0.25,
                        momentum=0.40,
                        noise=0.01,
                        stagnation_limit=800,
                    )
                except Exception:
                    continue

                if seq is not None and np.isfinite(sc2) and float(sc2) < global_score:
                    global_score = float(sc2)
                    global_best = np.asarray(seq, dtype=np.float64).copy()
                    add_candidate_pool(pool, global_best, global_score)
                    print("final quick idx=", idx, "score=", global_score, flush=True)

    # Long global refinement.
    rem = search_deadline - time.monotonic()
    if rem > 100.0:
        long_budget = min(0.55 * rem, rem - 130.0)
        if long_budget > 40.0:
            try:
                seq, sc = local_search(
                    global_best,
                    long_budget,
                    seed=int(seed + 12345),
                    p_start=3.0,
                    p_end=12.0,
                    step_init=0.22,
                    momentum=0.45,
                    noise=0.005,
                    stagnation_limit=1800,
                )
            except Exception:
                seq, sc = None, float("inf")

            if seq is not None and np.isfinite(sc) and float(sc) < global_score:
                global_score = float(sc)
                global_best = np.asarray(seq, dtype=np.float64).copy()
                add_candidate_pool(pool, global_best, global_score)
                print("final long score=", global_score, flush=True)

    # Sharper polishing phase.
    rem = search_deadline - time.monotonic()
    if rem > 70.0:
        polish_budget = min(0.60 * rem, rem - 30.0)
        if polish_budget > 30.0:
            try:
                seq, sc = local_search(
                    global_best,
                    polish_budget,
                    seed=int(seed + 99991),
                    p_start=6.0,
                    p_end=22.0,
                    step_init=0.12,
                    momentum=0.35,
                    noise=0.002,
                    stagnation_limit=3500,
                )
            except Exception:
                seq, sc = None, float("inf")

            if seq is not None and np.isfinite(sc) and float(sc) < global_score:
                global_score = float(sc)
                global_best = np.asarray(seq, dtype=np.float64).copy()
                add_candidate_pool(pool, global_best, global_score)
                print("final polish score=", global_score, flush=True)

    add_candidate_pool(pool, global_best, global_score)
    pool.sort(key=_score_key)

    final_best = global_best.copy() if global_best is not None else make_uniform(final_n)
    final_score = global_score

    # Exact validation of one or two best candidates, if time and length allow.
    for i in range(min(2, len(pool))):
        if time.monotonic() > t0 + budget - 30.0:
            break
        seq = pool[i][1]
        if seq.size == 0 or seq.size > 20000:
            continue
        try:
            ex = float(evaluate_sequence([float(x) for x in seq]))
            if np.isfinite(ex) and ex < final_score:
                final_score = ex
                final_best = seq
                print("exact validated score:", ex, "length:", seq.size, flush=True)
        except Exception:
            pass

    print("returned fast/exact best score:", final_score, "length:", int(final_best.size), flush=True)
    return sanitize_return(final_best)
```
