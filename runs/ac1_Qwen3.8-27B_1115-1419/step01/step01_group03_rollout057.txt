d(p0*x^2 + p1*2xy + p2*y^2)/dy = 2*p1*x + 2*p2*y = 2*(p1*x + p2*y) ✓

So the formula is simply: `grad = 2 * correlate(p, a, 'valid')`

This is clean and fast.

L(a) = (1/T) * log(sum_k exp(T * conv[k]))
dL/da = (1/T) * (1/sum_k exp(T*conv[k])) * sum_k exp(T*conv[k]) * d(conv[k])/da
= sum_k p[k] * d(conv[k])/da
where p[k] = exp(T*conv[k]) / sum_k exp(T*conv[k])

This is beautiful and simple.

def project_simplex(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()

            elif stuck == 250:
                # Perturbation restart
                a = best_a.copy()
                noise = rng.normal(0.0, 0.02 / np.sqrt(n), n)
                a = a + noise
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a = a / sa
                else:
                    a = best_a.copy()
                m[:] = 0.0
                v[:] = 0.0
                lr = lr_init
                stuck = 0
                print(f"    it={it}, T={T:.1f}, reset, score={best_score:.8f}")

            if it % 2000 == 0:
                elapsed_s = duration - (deadline - time.monotonic())
                print(f"    it={it}, T={T:.2f}, lr={lr:.5f}, score={best_score:.8f}, "
                      f"el={elapsed_s:.0f}s")

    # Final
    final_sc = fast_score_simplex(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()

    return best_a, best_score


def run_subgradient_polish(a0, n, duration, rng, n_peaks=5):
    """
    Local polishing using subgradient from top-n_peaks convolution coefficients.
    Uses a small step size projected gradient method.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv0 = fft_convolve_full(a)
    best_score = 2.0 * n * np.max(conv0)

    deadline = time.monotonic() + duration
    it = 0
    step = 0.05 / n
    stuck = 0

    while time.monotonic() < deadline:
        conv = fft_convolve_full(a)
        # Find top n_peaks
        if n_peaks >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]

        # Gradient: for each peak k, grad[j] += a[k-j] (valid)
        # Using correlate: grad = sum over top peaks of correlate(delta_k, a)
        # Which is: grad[j] = sum_{k in top} a[k-j] for valid k-j
        # Equivalently: for each j, grad[j] = sum_{k in top} a[k-j] = sum_{m} a[m] * [j+m in top]
        grad = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                idx = k - np.arange(lo, hi)
                grad[lo:hi] += a[idx]
            if k % 2 == 0:
                j = k // 2
                if 0 <= j < n:
                    grad[j] += a[j]  # diagonal term (already counted if k-j=j)

        # Actually the correct gradient is 2*a[k-j] for off-diagonal, a[j] for diagonal
        # Let me redo: grad[j] = 2*sum_{k in top, 0<=k-j<n} a[k-j] - sum_{k in top: 2j=k} a[j]
        # But for simplicity with top-peaks, just use:
        # grad[j] = 2 * sum_{k in top} a[k-j] * [0<=k-j<n]  (approximate)
        # The -a[j]*[2j in top] correction is small

        grad /= float(len(top_idx))

        a_new = a - step * grad
        a = project_simplex(a_new)

        it += 1
        if it % 20 == 0:
            sc = fast_score_simplex(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck == 30:
                step *= 0.7
            if stuck == 150:
                a = best_a.copy()
                step = 0.05 / n
                stuck = 0

    final_sc = fast_score_simplex(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()

    return best_a, best_score


def make_zadoff_chu_init(n, rng, clip_factor=1.0):
    """Create initialization from magnitude of Zadoff-Chu-like sequence."""
    # Use a chirp: a[k] = exp(i*pi*(k-k0)^2/(n-k0))
    k0 = n // 2
    k = np.arange(n, dtype=np.float64)
    phase = np.pi * (k - k0) ** 2 / (n + 1)
    complex_seq = np.exp(1j * phase)
    # Take magnitude (which is 1 for all), so instead use real part + offset
    real_part = np.abs(complex_seq)  # all ones, not useful
    # Better: use a random phase sequence
    phases = rng.uniform(0, 2 * np.pi, n)
    complex_seq = np.exp(1j * phases)
    # Take magnitude of windows (this gives a "texture")
    # Instead, use: a[k] = |sum_{m=k}^{k+w} exp(i*phi_m)| for some window
    w = max(2, n // 20)
    a = np.zeros(n)
    for i in range(n):
        end = min(i + w, n)
        a[i] = np.abs(np.sum(complex_seq[i:end])) / w
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_periodic_structure(n, rng):
    """Create a periodic structure with random phase."""
    period = rng.integers(max(3, n // 200), max(4, n // 50))
    period = min(period, n // 2)
    base = rng.uniform(0.3, 1.0, period)
    a = np.tile(base, n // period + 1)[:n]
    # Add small random perturbation
    a += rng.uniform(0, 0.1, n) * base.max()
    a = np.maximum(a, 0.0)
    return a / a.sum()


def propose_candidate(budget_s=2400):
    """
    Main search function. Uses Adam with log-sum-exp, FFT-based convolution,
    scipy.correlate for gradient, progressive temperature, simplex projection,
    multiple diverse initializations, and multi-phase optimization.
    """
    rng = np.random.default_rng(12345)
    total_deadline = time.monotonic() + float(budget_s) - 25.0

    # Get initial sequence
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.full(n, 1.0, dtype=np.float64)

    if n < 2:
        n = 4806
        arr = np.full(n, 1.0, dtype=np.float64)

    print(f"=== n = {n} ===")

    # Build diverse initializations
    candidates = []
    names = []

    # 1. Given sequence
    S = arr.sum()
    if S > 1e-10:
        candidates.append(arr.astype(np.float64) / S)
        names.append("given")

    # 2. Uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)
    names.append("uniform")

    # 3. Gaussian
    x = np.linspace(-3, 3, n)
    g = np.exp(-0.5 * x ** 2)
    candidates.append(g / g.sum())
    names.append("gaussian")

    # 4. Raised cosine
    x = np.linspace(-1, 1, n)
    rc = 0.5 * (1.0 + np.cos(np.pi * x))
    rc = np.maximum(rc, 1e-4)
    candidates.append(rc / rc.sum())
    names.append("raised_cosine")

    # 5. Lorentzian
    x = np.arange(n, dtype=np.float64) - n / 2.0
    lor = 1.0 / (1.0 + (x / (n * 0.08)) ** 2)
    candidates.append(lor / lor.sum())
    names.append("lorentzian")

    # 6. Two bumps
    b1 = np.zeros(n, dtype=np.float64)
    for i in range(n):
        b1[i] = max(0.0, 1.0 - abs(i - n * 0.25) / (n * 0.15)) + \
                 max(0.0, 1.0 - abs(i - n * 0.75) / (n * 0.15))
    if b1.sum() > 0:
        candidates.append(b1 / b1.sum())
        names.append("two_bumps")

    # 7. Periodic structure
    candidates.append(make_periodic_structure(n, rng))
    names.append("periodic")

    # 8. Zadoff-Chu magnitude
    candidates.append(make_zadoff_chu_init(n, rng))
    names.append("zch_magnitude")

    # 9. Random softmax (high temp)
    r = rng.standard_normal(n) * 0.5
    sr = np.exp(r - r.max())
    candidates.append(sr / sr.sum())
    names.append("random_soft")

    # 10. Block structure
    nblocks = rng.integers(5, 30)
    block_size = n // nblocks
    a_blk = np.zeros(n, dtype=np.float64)
    for b in range(nblocks):
        a_blk[b * block_size:(b + 1) * block_size] = rng.uniform(0.5, 1.0)
    candidates.append(a_blk / a_blk.sum())
    names.append("blocks")

    print("Initial scores (on simplex, score = 2n*max_conv):")
    best_cand = None
    best_cand_score = np.inf
    cand_scores = []
    for c, nm in zip(candidates, names):
        sc = fast_score_simplex(c)
        cand_scores.append(sc)
        print(f"  {nm:15s}: {sc:.8f}")
        if sc < best_cand_score:
            best_cand_score = sc
            best_cand = c.copy()

    print(f"Best initial: {best_cand_score:.8f}")

    best_overall = best_cand.copy()
    best_score_overall = best_cand_score

    # === Phase 1: Adam-LSE from best candidate (exploration, low T) ===
    time_left = total_deadline - time.monotonic()
    p1_time = min(time_left * 0.35, 800.0)
    if p1_time > 50:
        print(f"\n--- Phase 1: Adam-LSE explore (T: 0.5->100), {p1_time:.0f}s ---")
        a1, s1 = run_adam_lse(best_cand, n, p1_time, rng,
                              T_start=0.5, T_end=100.0, lr_init=0.03)
        if s1 < best_score_overall:
            best_score_overall = s1
            best_overall = a1.copy()
        print(f"Phase 1 best: {min(s1, best_cand_score):.8f}")

    # === Phase 2: Adam-LSE from 2nd/3rd best candidates ===
    time_left = total_deadline - time.monotonic()
    if time_left > 100:
        # Try the second best and a structurally different one
        sorted_idx = np.argsort(cand_scores)
        tried = set()
        for rank in sorted_idx[:4]:
            if candidates[rank] is best_cand or id(candidates[rank]) in tried:
                continue
            tried.add(id(candidates[rank]))
            time_left = total_deadline - time.monotonic()
            if time_left < 80:
                break
            p2_time = min(time_left * 0.25, 400.0)
            if p2_time < 50:
                break
            nm = names[rank]
            print(f"\n--- Phase 2: Adam-LSE from '{nm}', T: 2->200, {p2_time:.0f}s ---")
            a2, s2 = run_adam_lse(candidates[rank], n, p2_time, rng,
                                  T_start=2.0, T_end=200.0, lr_init=0.02)
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = a2.copy()
                print(f"  Improved to: {s2:.8f}")
            else:
                print(f"  No improvement (was {s2:.8f})")

    # === Phase 3: Fine polishing (high T, small steps) ===
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        p3_time = min(time_left * 0.25, 500.0)
        if p3_time > 40:
            print(f"\n--- Phase 3: Polish (T: 50->2000), {p3_time:.0f}s ---")
            a3, s3 = run_adam_lse(best_overall, n, p3_time, rng,
                                  T_start=50.0, T_end=2000.0, lr_init=0.01)
            if s3 < best_score_overall:
                best_score_overall = s3
                best_overall = a3.copy()
            print(f"Phase 3 best: {s3:.8f}")

    # === Phase 4: Subgradient top-k polish ===
    time_left = total_deadline - time.monotonic()
    if time_left > 50:
        p4_time = min(time_left * 0.3, 300.0)
        if p4_time > 30:
            print(f"\n--- Phase 4: Top-5 subgradient polish, {p4_time:.0f}s ---")
            a4, s4 = run_subgradient_polish(best_overall, n, p4_time, rng, n_peaks=5)
            if s4 < best_score_overall:
                best_score_overall = s4
                best_overall = a4.copy()
            print(f"Phase 4 best: {s4:.8f}")

    # === Phase 5: More restarts if time ===
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        print(f"\n--- Phase 5: Extra Adam-LSE runs, {time_left-10:.0f}s remaining ---")
        for rk in range(10):
            time_left = total_deadline - time.monotonic()
            if time_left < 40:
                break
            # Create perturbed start
            scale = (0.03 + 0.02 * rk) / np.sqrt(n)
            a_start = best_overall + rng.normal(0, scale, n)
            np.maximum(a_start, 0, out=a_start)
            sa = a_start.sum()
            if sa > 1e-12:
                a_start = a_start / sa
            else:
                a_start = best_overall.copy()

            t_budget = min(time_left - 10, 250.0)
            T_start = 10.0 + 20.0 * rk
            T_end = 500.0 + 500.0 * rk
            a_r, s_r = run_adam_lse(a_start, n, t_budget, rng,
                                    T_start=T_start, T_end=T_end, lr_init=0.015)
            if s_r < best_score_overall:
                best_score_overall = s_r
                best_overall = a_r.copy()
                print(f"  Restart {rk}: improved to {s_r:.8f}")
            else:
                print(f"  Restart {rk}: {s_r:.8f}")

    # === Final subgradient polish on best ===
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        a_f, s_f = run_subgradient_polish(best_overall, n, time_left - 10, rng, n_peaks=10)
        if s_f < best_score_overall:
            best_score_overall = s_f
            best_overall = a_f.copy()

    # Final top-1 subgradient
    time_left = total_deadline - time.monotonic()
    if time_left > 20:
        a_f2, s_f2 = run_subgradient_polish(best_overall, n, time_left - 5, rng, n_peaks=1)
        if s_f2 < best_score_overall:
            best_score_overall = s_f2
            best_overall = a_f2.copy()

    # === Final verification ===
    final_list = [float(max(0.0, x)) for x in best_overall]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.8f}")

    # Fallback check
    if not np.isfinite(final_score):
        print("Score is inf, falling back to best candidate")
        final_list = [float(max(0.0, x)) for x in best_cand]
        final_score = float(evaluate_sequence(final_list))

    # Ensure sum >= 0.01
    fsum = sum(final_list)
    if fsum < 0.01:
        scale = 0.1 / max(fsum, 1e-15)
        final_list = [x * scale for x in final_list]
        final_score = float(evaluate_sequence(final_list))

    print(f"=== FINAL SCORE: {final_score:.8f} (n={n}) ===")
    return final_list
