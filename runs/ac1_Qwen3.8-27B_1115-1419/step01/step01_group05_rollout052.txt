"""
Search for low-peak nonnegative self-convolutions using multi-method optimization.
Combines LSE gradient descent, multiplicative (mirror) descent, and block coordinate
descent with diverse initializations and adaptive length scaling.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s <= 1e-300:
        return np.ones(max(a.size, 1), dtype=np.float64) / max(a.size, 1)
    return a / s

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]

def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb

def _make_uniform(n):
    return np.ones(max(int(n), 1), dtype=np.float64) / max(int(n), 1)

def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)

def _make_raised_cosine(n, width=1.0):
    n = max(int(n), 2)
    x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    a = np.cos(np.pi * x / (2.0 * width)) ** 2
    a = np.clip(a, 0.0, None)
    return _normalize(a)

def _make_gaussian(n, sigma=0.3):
    n = max(int(n), 2)
    x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    a = np.exp(-x**2 / (2.0 * sigma**2))
    return _normalize(a)

def _make_bipolar_smooth(n, rng):
    """Create a smooth sequence using Fourier components with random phases."""
    n = max(int(n), 2)
    a = np.ones(n, dtype=np.float64)
    num_terms = max(3, n // 50)
    for k in range(1, num_terms + 1):
        amp = 0.3 / k
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n, dtype=np.float64)
        a += amp * np.cos(2 * np.pi * k * x / n + phase)
    a = np.maximum(a, 1e-10)
    return _normalize(a)

def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

def _lse_and_grad(a, beta, fft_size, g_fft):
    """Compute LSE value and its gradient w.r.t. a."""
    n = a.size
    b = _autocorr(a, fft_size)
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
        val = m
    else:
        arg = beta * (b - m)
        arg = np.maximum(arg, -745.0)
        e = np.exp(arg)
        s = float(e.sum())
        if s <= 0:
            p = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
            val = m
        else:
            p = e / s
            val = m + float(np.log(s)) / beta
    # Gradient: 2 * (a * p)_i where (a*p) is cross-correlation
    # (a*a)_k = sum_i a_i a_{k-i}
    # d/d a_j of sum_k p_k (a*a)_k = 2 * sum_k p_k a_{k-j}
    # This is 2 * (p * a reversed) _j, i.e., 2 * (p convolve a_reversed) shifted
    ar = a[::-1]
    A = np.fft.rfft(ar, g_fft)
    P = np.fft.rfft(p, g_fft)
    conv = np.fft.irfft(A * P, g_fft).real
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return val, grad, b, m

def _multiplicative_update(a, grad, step):
    """Mirror descent step on simplex: a_i * exp(-step * grad_i)"""
    log_a = np.log(np.maximum(a, 1e-300))
    new_a = log_a - step * grad
    new_a = np.exp(new_a)
    return _normalize(new_a)

def _block_optimize(a, block_size, deadline, rng):
    """Block coordinate descent: optimize one block at a time."""
    n = a.size
    best_a = a.copy()
    best_score = _score_of(best_a)
    fft_size = _next_pow2(2 * n - 1)
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Pick a random block
        nb = max(1, n // block_size)
        bidx = int(rng.integers(0, nb))
        start = bidx * (n // nb)
        stop = min(n, (bidx + 1) * (n // nb) + 1)
        
        # Current convolution
        b_full = _autocorr(best_a, fft_size)
        
        # Try perturbations in this block
        improved = False
        for scale in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            candidate = best_a.copy()
            candidate[start:stop] *= scale
            candidate = _normalize(candidate)
            s = _score_of(candidate)
            if s < best_score - 1e-12:
                best_a = candidate
                best_score = s
                improved = True
                break
        
        if not improved:
            # Try random perturbation in the block
            for _ in range(3):
                candidate = best_a.copy()
                candidate[start:stop] *= np.exp(rng.normal(0, 0.3, size=stop - start))
                candidate = _normalize(candidate)
                s = _score_of(candidate)
                if s < best_score - 1e-12:
                    best_a = candidate
                    best_score = s
                    break
        
        if it % 500 == 0:
            print(f"  block_opt: it={it} score={best_score:.8f}", flush=True)
    
    return best_a, best_score

def _refine_lse(a, deadline, rng, beta_start=10.0):
    """LSE gradient descent with adaptive step and restarts."""
    a = _normalize(a)
    n = a.size
    if n < 32:
        return a, _score_of(a)
    
    fft_size = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n - 2)
    
    best_a = a.copy()
    best_score = _score_of(best_a)
    alpha = 0.1
    beta = beta_start
    fail = 0
    last_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        val, grad, b, maxb = _lse_and_grad(a, beta, fft_size, g_fft)
        
        # Center the gradient
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))
        
        if scale <= 1e-300:
            # Perturb and restart
            a = a * np.exp(rng.normal(0, 0.3, size=n))
            a = _normalize(a)
            alpha = 0.1
            fail = 0
            it += 1
            continue

        step = alpha / max(scale, 1e-10)
        v = a - step * grad_c
        a = _normalize(np.maximum(v, 0.0))
        
        score = _score_of(a)
        val_new, _, _, _ = _lse_and_grad(a, beta, fft_size, g_fft)
        
        if val_new < val - 1e-14:
            alpha = min(2.0, alpha * 1.1)
            fail = 0
            last_improve = it
        else:
            alpha *= 0.5
            fail += 1
        
        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score

        if it % 50 == 0:
            beta = min(200.0, beta + 2.0)
