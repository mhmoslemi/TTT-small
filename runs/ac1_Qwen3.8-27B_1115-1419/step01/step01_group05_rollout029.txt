def _next_pow2(size):

def _normalize(a):

def _autocorr(a, fft_size):

    n = a.size
    A = np.fft.rfft(a, fft_size)

def _project_simplex(v, target_sum=1.0):
    """True Euclidean projection onto {x >= 0, sum x = target_sum}."""
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, target_sum)])
    
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = (cssv - target_sum) / np.arange(1, n+1)
    # Find the largest rho such that u[rho] > rho_candidates[rho]
    # This is the standard algorithm
    cond = u > rho_candidates
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = rho_candidates[rho]
    w = np.maximum(v - theta, 0.0)
    # Normalize to ensure exact sum (numerical correction)
    s = w.sum()
    if s > 0:
        w = w * (target_sum / s)
    else:
        w = np.full(n, target_sum / n)
    return w

    x = np.arange(n, dtype=np.float64)

    else:  # symmetric (V-shaped dip)

    x = np.arange(1, n+1, dtype=np.float64)

def _resample(old, new_n):
    """Resample a normalized sequence to a new length."""
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.maximum(old, 0.0)
    s = old.sum()
    if s < 1e-300 or old.size == 0:
        return np.ones(new_n) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

        p, lse_old = _lse_and_p(b, beta)
        
        # Compute gradient
        grad = _weighted_conv(a, p, g_fft, n)
        
        # Apply momentum
        v_momentum = momentum * v_momentum + (1 - momentum) * grad
        if momentum == 0:
            v_momentum = grad
        
        # Gradient descent step
        v = a - alpha * v_momentum
        a_new = _project_simplex(v, 1.0)
        
        # Evaluate
        b_new = _autocorr(a_new, b_fft)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * n * maxb_new
        lse_new = _lse_value(b_new, beta)
        
        # Line search / step size adaptation
        if lse_new < lse_old - 1e-15:
            # Accepted
            a = a_new
            b = b_new
            maxb = maxb_new
            score = score_new
            alpha = min(1.0, alpha * 1.05)
            fail_count = 0
            momentum = min(0.9, momentum + 0.01)
            if score < best_score - 1e-12:
                best_a = a.copy()
                best_score = score
                last_improve = it
        else:
            # Rejected
            alpha *= 0.5
            fail_count += 1
            momentum = max(0.0, momentum - 0.02)
        
        # Adaptive beta: increase as we converge
        if it % 50 == 0:
            # Target: beta should be high enough that LSE ≈ max
            # Current "effective" beta = 1 / (std of b at the top)
            top_vals = np.sort(b)[::-1][:max(1, n//4)]
            spread = top_vals[0] - top_vals[-1]
            if spread > 1e-15:
                beta_target = 2.0 / spread
            else:
                beta_target = 1e6
            beta = min(1e6, max(beta, 0.8 * beta_target))
        
        # Restart if stuck
        if fail_count > 200 or (it - last_improve > 500 and it > 100):
            # Try a perturbation of the best
            if rng.random() < 0.5:
                a = _normalize(best_a * np.exp(rng.normal(0, 0.1, n)))
            else:
                a = _normalize(best_a + 0.05 * rng.normal(size=n))
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.1
            beta = max(5.0, 10.0 * maxb)
            momentum = 0.0
            v_momentum = np.zeros(n)
            fail_count = 0
        
        it += 1
    
    return best_a, best_score

def _lse_value(b, beta):
    m = float(np.max(b))
    if beta < 1.0:
        return m
    arg = beta * (b - m)
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def _peak_attack(a, b, max_iter=100, deadline=None):
    """Directly try to reduce the current peak by targeted redistribution."""
    n = a.size
    b_fft = _next_pow2(2*n - 1)
    best_a = a.copy()
    maxb = float(np.max(b))
    best_score = 2.0 * n * maxb
    
    k_star = int(np.argmax(b))  # current peak position
    
    # The peak b[k_star] = sum_j a[j] * a[k_star - j]
    # To reduce it, reduce the products a[j]*a[k_star-j]
    # Strategy: reduce a[j] for j where a[j]*a[k_star-j] is large,
    # and add mass to positions that don't contribute to b[k_star]
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        k_star = int(np.argmax(b))
        # Contributions to b[k_star]: a[j] * a[k_star - j] for valid j
        js = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                js.append((j, k2, a[j] * a[k2]))
        
        if not js:
            break
        
        # Sort by contribution
        js.sort(key=lambda x: -x[2])
        
        # Reduce the top contributors
        total_contrib = sum(x[2] for x in js)
        if total_contrib < 1e-15:
            break
        
        # Move mass from top contributor to "safe" position
        top_j, top_k2, top_val = js[0]
        if top_val < 1e-15:
            break
        
        # Find a "safe" position: one where adding mass doesn't increase any b[k] much
        # Simple heuristic: position 0 or n-1 (edges) are usually safest
        # Better: find position i that minimizes sum_k p[k] * a[k-i] (doesn't help much)
        
        # Simple approach: reduce a at the peak-contributing position, add to the minimum
        reduction = 0.1 * a[top_j]
        if reduction < 1e-15:
            break
        
        # Add to the position with smallest current value (least likely to create new peaks)
        min_idx = int(np.argmin(a))
        
        a_new = a.copy()
        a_new[top_j] -= reduction
        a_new[min_idx] += reduction
        a_new = np.maximum(a_new, 0)
        a_new = _normalize(a_new)
        
        b_new = _autocorr(a_new, b_fft)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * n * maxb_new
        
        if score_new < best_score:
            best_a = a_new.copy()
            best_score = score_new
            a = a_new
            b = b_new
            maxb = maxb_new
        else:
            # Try smaller reduction
            reduction = 0.01 * a[top_j]
            if reduction < 1e-15:
                break
            a_new = a.copy()
            a_new[top_j] -= reduction
            a_new[min_idx] += reduction
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new)
            b_new = _autocorr(a_new, b_fft)
            maxb_new = float(np.max(b_new))
            score_new = 2.0 * n * maxb_new
            if score_new < best_score:
                best_a = a_new.copy()
                best_score = score_new
                a = a_new
                b = b_new
                maxb = maxb_new
            else:
                break
    
    return best_a, best_score

def _weighted_conv(a, p, fft_size, n):
    """Compute gradient: g[i] = 2 * sum_j a[j] * p[i+j] for i=0..n-1."""
    ar = a[::-1]
    result = np.fft.irfft(np.fft.rfft(ar, fft_size) * np.fft.rfft(p, fft_size), fft_size)
    return 2.0 * result[n-1:2*n-1]
