def _score_and_grad(v, tau, m):
    """Compute smooth-max score and gradient w.r.t. v in log-space."""
    n = len(v)
    a = np.exp(v - np.max(v))
    L = 2*n - 1
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:L]
    b = np.maximum(b, 1e-300)  # avoid log of negative (shouldn't happen for a>=0)
    
    S = np.sum(a)
    if S <= 1e-300:
        return np.inf, np.zeros(n)
    
    # Soft-max of b
    tb = tau * b
    tb = tb - np.max(tb)
    w = np.exp(tb)
    w_sum = np.sum(w)
    w = w / w_sum
    soft_max = (np.log(w_sum) + np.max(tau*b)) / tau  # log-sum-exp / tau
    
    # Gradient of soft_max w.r.t. a
    W = np.fft.fft(w, m)
    A = np.fft.fft(a, m)
    c = np.fft.ifft(np.conj(W) * A).real[:n]
    
    dsoft_max_da = 2.0 * c
    
    # Gradient of score w.r.t. a
    dscore_da = 2.0 * n * (S * dsoft_max_da - soft_max) / (S * S * S)
    
    # Gradient w.r.t. v: dscore/dv_j = dscore/da_j * a_j
    dscore_dv = dscore_da * a
    
    score = 2.0 * n * soft_max / (S * S)

def _np2(x):

    p = 1
    while p < x:

def _score_gradient(v, tau, m):

    F = np.fft.rfft(a, m)

    b = np.maximum(b, 0.0)
    S = a.sum()
    if S < 1e-30:
        return np.inf
    return 2.0 * n * b.max() / (S * S)
