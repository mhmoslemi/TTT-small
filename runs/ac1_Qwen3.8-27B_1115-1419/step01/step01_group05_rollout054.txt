"""
Multi-strategy search for low-peak nonnegative self-convolutions.

Combines LP-based direction finding, LSE-surrogate gradient descent,
and coordinate descent to minimize 2*n*max(a*a)/(sum(a))^2.
The LP approach solves: min t s.t. (a*g)(k)<=t, sum(g)=1, g>=0,
giving a principled direction that reduces the convolution peak.
"""

import numpy as np
import time
from scipy.optimize import linprog

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def _conv(a):
    n = len(a)
    if n < 64:
        return np.convolve(a, a, mode='full')
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _score(a):
    a = _norm(a)
    n = len(a)
    b = _conv(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = a
    return C


def _lp_direction(a, verbose=False):
    """
    Solve: min t subject to (a*g)(k) <= t for all k, sum(g)=1, g>=0.
    Returns (g*, t*) or (None, None) on failure.
    """
    n = len(a)
    if n > 4096:
        return None, None

    C = _build_conv_matrix(a)  # (2n-1) x n

    # Variables: [g_0,...,g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t

    # Inequality: C*g - t <= 0  =>  [C, -1] @ x <= 0
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n + 1), dtype=np.float64)
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_cons)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 30})

    if res.status != 0 or not hasattr(res, 'x'):
        return None, None

    g = res.x[:n]
    t = res.x[n]
    g = np.maximum(g, 0.0)
    sg = g.sum()
    if sg <= 0:
        return None, None
    g = g / sg
    return g, float(t)


def _lp_max_sum_direction(a, M):
    """
    Solve: max sum(g) subject to (a*g)(k) <= M, g >= 0.
    Returns (g*, sum_g) or (None, None).
    """
    n = len(a)
    if n > 4096:
        return None, None

    C = _build_conv_matrix(a)

    c = -np.ones(n)  # minimize -sum(g)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 30})

    if res.status != 0 or not hasattr(res, 'x'):
        return None, None

    g = np.maximum(res.x, 0.0)
    sg = float(g.sum())
    if sg <= 0:
        return None, None
    return g, sg


def _lse_and_p(b, beta):
    m = float(np.max(b))
    if beta <= 1.0:
        return np.full(b.size, 1.0 / b.size), m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = e.sum()
    if s <= 0:
        e = np.ones(b.size)
        s = b.size
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _weighted_grad(a, weights, n):
    ar = a[::-1]
    size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(ar, size)
    W = np.fft.rfft(weights, size)
    conv = np.fft.irfft(A * W, size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * np.maximum(h, 0.0)


def _refine_lse(a, deadline, rng, beta_start=8.0):
    """LSE-surrogate projected gradient descent."""
    a = _norm(a.copy())
    n = a.size
    b_fft = 1 << (2 * n - 1).bit_length()
    g_fft = 1 << (3 * n - 2).bit_length()

    b = _conv(a)
    mb = float(np.max(b))
    score = 2.0 * n * mb
    best_a = a.copy()
    best_score = score

    alpha = 0.15
    beta_t = beta_start
    it = 0
    fail = 0

    while time.monotonic() < deadline:
        mb_safe = max(mb, 1e-300)
        beta = min(1e5, max(1.0, beta_t / mb_safe))
        p, smooth_old = _lse_and_p(b, beta)
        grad = _weighted_grad(a, p, n)
        grad_c = grad - grad.mean()
        scale = np.max(np.abs(grad_c))
        if scale <= 1e-300:
            a = _norm(a * np.exp(rng.normal(0, 0.15, n)))
            b = _conv(a)
            mb = float(np.max(b))
            score = 2.0 * n * mb
            fail = 0
            it += 1
            continue

        v = a - alpha * grad_c / scale * mb
        a = _norm(v)
        b = _conv(a)
        mb = float(np.max(b))
        score = 2.0 * n * mb
        smooth_new = -float(np.log(p @ b / p.sum())) / beta  # approximate

        # Better: recompute smooth
        _, smooth_new = _lse_and_p(b, beta)

        if smooth_new < smooth_old - 1e-14:
            alpha = min(1.0, alpha * 1.1)
            fail = 0
        else:
            alpha *= 0.5
            fail += 1

        if score < best_score - 1e-15:
            best_a = a.copy()
            best_score = score

        if it % 80 == 0:
            beta_t = min(500.0, beta_t + 3.0)
        if fail > 200:
            a = _norm(best_a * np.exp(rng.normal(0, 0.2, n)))
            b = _conv(a)
            mb = float(np.max(b))
            score = 2.0 * n * mb
            alpha = 0.15
            fail = 0
            beta_t = max(8.0, beta_t * 0.5)

        it += 1

    return best_a, best_score


def _coord_descent(a, deadline, rng):
    """Coordinate descent: adjust individual elements to reduce max peak."""
    a = _norm(a.copy())
    n = len(a)
    b = _conv(a)
    score = 2.0 * n * float(np.max(b))
    best_a = a.copy()
    best_score = score

    it = 0
    while time.monotonic() < deadline:
        # Find top peak
        k_max = int(np.argmax(b))
        peak = float(b[k_max])

        # Find which a[j] contribute most to this peak
        # b[k_max] = sum_j a[j] * a[k_max - j]
        contribs = np.zeros(n)
        for j in range(n):
            k2 = k_max - j
            if 0 <= k2 < n:
                contribs[j] = a[j] * a[k2]

        # Decrease the largest contributors, increase others
        top_idx = np.argsort(contribs)[-min(8, n):]
        bottom_idx = np.argsort(contribs)[:min(8, n)]

        frac = 0.02 + 0.03 * rng.random()
        for idx in top_idx:
            a[idx] *= (1.0 - frac)
        for idx in bottom_idx:
            a[idx] *= (1.0 + frac)

        a = _norm(a)
        b = _conv(a)
        score = 2.0 * n * float(np.max(b))

        if score < best_score - 1e-15:
            best_a = a.copy()
            best_score = score
        else:
            a = best_a.copy()
            # Random coordinate perturbation
            for _ in range(min(16, n)):
                idx = int(rng.integers(0, n))
                a[idx] *= np.exp(rng.normal(0, 0.1))
            a = _norm(a)
            b = _conv(a)
            score = 2.0 * n * float(np.max(b))
            if score < best_score:
                best_a = a.copy()
                best_score = score

        it += 1
        if it % 50 == 0:
            # Occasionally try a bigger perturbation
            if rng.random() < 0.1:
                mask = rng.random(n) < 0.2
                a[mask] *= np.exp(rng.normal(0, 0.3, mask.sum()))
                a = _norm(a)

    return best_a, best_score


def _make_exponential(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _norm(a)


def _make_cosine(n, mode=1):
    x = np.linspace(0.0, 1.0, n)
    a = np.maximum(0.3 + 0.7 * np.cos(mode * np.pi * (2 * x - 1)), 0.01)
    return _norm(a)


def _make_mixed(n, rng):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-1.0 * x) + 0.5 * np.exp(-0.5 * (1 - x)) + 0.3 * np.abs(np.sin(2 * np.pi * 3 * x))
    a = a + 0.1
    return _norm(a)


def _make_random_smooth(n, rng, base=None, amp=0.4):
    if base is None:
        base = _norm(np.ones(n))
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 16 + 2)
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - f.mean()
    sd = f.std()
    if sd > 0:
        f = f / sd
    a = base * (1.0 + amp * f)
    a = np.maximum(a, 1e-10)
    return _norm(a)


def _resample(a, new_n):
    a = _norm(a)
    if a.size == new_n:
        return a.copy()
    if a.size < 2:
        return _norm(np.ones(new_n))
    x_old = np.linspace(0, 1, a.size)
    x_new = np.linspace(0, 1, new_n)
    new = np.interp(x_new, x_old, a)
    return _norm(new)

def _lp_optimize(a, deadline, rng):

    a = _norm(a.copy())

    b = _conv(a)

    best_a = a.copy()
    best_score = score
    t_step = 0.15
    it = 0

    while time.monotonic() < deadline:
        M = float(np.max(b))

        # Solve min-t LP
        g, t_opt = _lp_direction(a)
        if g is not None:
            # Update
            a_new = (1 - t_step) * a + t_step * g
            a_new = _norm(a_new)
            b_new = _conv(a_new)
            s_new = 2.0 * n * float(np.max(b_new))

            if s_new < score - 1e-15:
                a = a_new
                b = b_new
                score = s_new
                t_step = min(0.5, t_step * 1.05)
                if score < best_score - 1e-15:
                    best_a = a.copy()
                    best_score = score
            else:

                if t_step < 0.01:

                        b_cand = _conv(a_candidate)
                        s_cand = 2.0 * n * float(np.max(b_cand))
                        if s_cand < best_score:
                            a = (1 - 0.1) * a + 0.1 * a_candidate
                            a = _norm(a)
                            b = _conv(a)
                            score = 2.0 * n * float(np.max(b))
                            if score < best_score:
                                best_a = a.copy()
                                best_score = score
                    t_step = 0.15
                    # Inject randomness
                    a = _norm(a * np.exp(rng.normal(0, 0.05, n)))
                    b = _conv(a)
                    score = 2.0 * n * float(np.max(b))
        else:
            # LP failed, perturb and continue
            a = _norm(a * np.exp(rng.normal(0, 0.1, n)))
            b = _conv(a)
            score = 2.0 * n * float(np.max(b))
            if score < best_score:
                best_a = a.copy()
                best_score = score
            t_step = 0.15

        it += 1
        if it % 200 == 0:
            print(f"    LP iter {it}: score={best_score:.8f} t_step={t_step:.4f}", flush=True)

    return best_a, best_score
