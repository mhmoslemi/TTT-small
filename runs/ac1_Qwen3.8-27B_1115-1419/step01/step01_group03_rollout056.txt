    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def soft_max_gradient(a, c, T):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*c_k)) w.r.t. a.
    This equals sum_k w_k * grad(c_k) where w_k = softmax(T*c).
    """
    n = len(a)
    # Compute softmax weights
    Tc = T * c
    Tc -= Tc.max()  # numerical stability
    w = np.exp(Tc)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform weights
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    
    # Gradient: g[j] = 2 * sum_m a[m] * w[m+j] = 2 * convolve(a, w, 'valid')[j]
    g = 2.0 * np.convolve(a, w, mode='valid')
    return g


def run_soft_max_adam(a0, n, duration, rng, T_start=10.0, T_end=500.0, 
                      lr_start=0.05, lr_end=0.001, beta1=0.9, beta2=0.999, 
                      tol=1e-8):
    """
    Run soft-max Adam optimization on the simplex.
    """
    a = a0.copy()
    a = project_simplex(a)
    
    m = np.zeros(n, dtype=np.float64)  # First moment
    v = np.zeros(n, dtype=np.float64)  # Second moment
    step_count = 0
    
    best_a = a.copy()
    best_score = fast_score(a)
    
    deadline = time.monotonic() + duration
    
    last_improve = 0
    best_score_at_improve = best_score
    T = T_start
    
    while time.monotonic() < deadline:
        step_count += 1
        
        # Compute current convolution and gradient
        c = np.convolve(a, a)
        g = soft_max_gradient(a, c, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** step_count)
        v_hat = v / (1 - beta2 ** step_count)
        
        # Learning rate schedule (linear decay)
        progress = min(1.0, step_count / max(1, int(duration * 50)))  # assume ~50 iters/sec
        lr = lr_start * (1 - progress) + lr_end * progress
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)
        a = a_new
        
        # Periodic score check (every 20 steps)
        if step_count % 20 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = step_count
                best_score_at_improve = best_score
            
            # Anneal T
            elapsed_frac = min(1.0, (time.monotonic() - (deadline - duration)) / duration)
            T = T_start * (T_end / T_start) ** elapsed_frac
            
            # Check for stagnation
            if step_count - last_improve > 2000:
                # Perturb and restart
                print(f"    Stagnation detected at step {step_count}, score={best_score:.7f}, T={T:.1f}")
                a = best_a + rng.normal(0.0, 0.02 / np.sqrt(n), n)
                a = project_simplex(a)
                m[:] = 0
                v[:] = 0
                step_count = 0
                last_improve = 0
                T = max(10.0, T * 0.5)  # Reset T to be less peaked
                lr = lr_start
                print(f"    Restarted with T={T:.1f}, lr={lr:.4f}")
    
    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

    k = np.fft.fftfreq(n)
    # Random phase
    phase = rng.uniform(0, 2*np.pi, n)
    # Magnitude: roughly 1, with some random variation
    mag = 1.0 + 0.1 * rng.normal(size=n)

    A = mag * np.exp(1j * phase)
    A[0] = abs(A[0])  # Make DC positive

    a_sum = a.sum()
    if a_sum < 1e-10:

    return a / a_sum

"""
Minimize 2n*max(conv(a))/(sum(a))^2 using a multi-phase approach:
1. Soft-max smoothed objective with Adam optimizer and simplex projection
2. Adaptive temperature annealing for progressive peak focusing
3. Multiple diverse initializations including Fourier-domain construction
4. Active-set refinement and high-precision polishing

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1} in O(n log n)."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def soft_max_weights(c, T):
    """Compute softmax(T * c) with numerical stability."""
    Tc = T * c
    Tc = Tc - Tc.max()
    w = np.exp(Tc)
    s = w.sum()
    if s < 1e-300:
        # All effectively zero after shift; return uniform
        w = np.ones_like(c)
        s = len(c)
    w /= s
    return w


def compute_gradient(a, w):
    """
    Compute gradient of sum_k w_k * c_k w.r.t. a.
    c_k = sum_i a_i * a_{k-i}, so d(c_k)/d(a_j) = 2*a_{k-j} (if 0<=k-j<n).
    Result: g[j] = 2 * sum_k w_k * a_{k-j} (for valid k-j)
            = 2 * sum_m a_m * w_{m+j}
            = 2 * np.correlate(w, a, 'valid')[j]  (when len(w) >= len(a))
    Wait, let me use np.convolve(a, w, 'valid') which gives result[k] = sum_m a[m]*w[k+m].
    So g = 2 * np.convolve(a, w, 'valid')
    """
    return 2.0 * np.convolve(a, w, mode='valid')


def run_adam_soft_max(a0, n, duration, rng, T_start=5.0, T_end=500.0,
                      lr_start=0.1, lr_end=0.0005,
                      beta1=0.9, beta2=0.999, tol=1e-9,
                      stagnation_limit=3000, verbose=True):
    """
    Soft-max Adam optimization on the simplex.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    step = 0
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    start_time = time.monotonic()
    deadline = start_time + duration
    last_improve_step = 0
    T = T_start
    
    while time.monotonic() < deadline:
        step += 1
        
        # Compute convolution and soft-max gradient
        c = np.convolve(a, a)
        w = soft_max_weights(c, T)
        g = compute_gradient(a, w)
        
        # Normalize gradient for scale invariance
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        # Learning rate: cosine schedule
        progress = min(1.0, step / max(1, int(duration * 100)))
        lr = 0.5 * (lr_start + lr_end) + 0.5 * (lr_start - lr_end) * np.cos(np.pi * progress)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a_new)
        
        # Periodic evaluation
        if step % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                last_improve_step = step
            
            # Anneal T based on elapsed time fraction
            elapsed = time.monotonic() - start_time
            frac = min(1.0, elapsed / duration)
            T = T_start * (T_end / T_start) ** (frac * 1.5)  # Slightly superlinear annealing
            
            # Stagnation detection
            if step - last_improve_step > stagnation_limit:
                if verbose:
                    print(f"    [Adam] Stagnation at step {step}, score={best_sc:.7f}, T={T:.1f}")
                # Perturb best and restart
                a = best_a + rng.normal(0.0, 0.03 / np.sqrt(n), n)
                a = project_simplex(a)
                m[:] = 0.0
                v[:] = 0.0
                step = 0
                last_improve_step = 0
                T = max(5.0, T * 0.3)
                
                if verbose and (step == 1 or step % 5000 == 0):
                    print(f"    [Adam] Restarted, T={T:.1f}")
    
    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc


def fourier_init(n, rng):
    """Initialize using a flat-ish random spectrum."""
    # Random complex spectrum
    phase = rng.uniform(0, 2 * np.pi, n)
    mag = 1.0 + 0.2 * rng.standard_normal(n)
    mag = np.maximum(mag, 0.1)
    A = mag * np.exp(1j * phase)
    A[n // 2] = float(abs(A[n // 2]))  # Nyquist real
    A[0] = float(abs(A[0]))  # DC positive
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    """Create structured initializations."""
    inits = []
    
    # Multi-frequency modulation
    for freq in [1, 2, 3, 5, 8]:
        x = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
        x = np.maximum(x, 0.01)
        x /= x.sum()
        inits.append(x)
    
    # Block structure
    for n_blocks in [3, 5, 7]:
        block_size = n // n_blocks
        x = np.zeros(n)
        for b in range(n_blocks):
            x[b*block_size:(b+1)*block_size] = rng.exponential(1.0)
        x = np.maximum(x, 0)
        x /= x.sum()
        inits.append(x)
    
    return inits


def polish_top_peaks(a, n, duration, num_peaks=5):
    """
    Polish by focusing on the top peaks using exact subgradient.
    This is essentially FISTA with the gradient from the top-k peaks.
    """
    a = a.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    deadline = time.monotonic() + duration
    step = 0
    lr = 0.05
    stuck = 0
    
    while time.monotonic() < deadline:
        step += 1
        c = np.convolve(a, a)
        
        # Get top peaks
        if num_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -num_peaks)[-num_peaks:]
        
        # Average gradient of top peaks
        g = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                idx = k - np.arange(lo, hi, dtype=np.int64)
                g[lo:hi] += 2.0 * a[idx]
        g /= len(top_idx)
        
        # Normalize
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g /= gn
        
        # Gradient step + project
        a_new = a - lr * g
        a = project_simplex(a_new)
        
        if step % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck == 50:
                lr *= 0.8
            if stuck > 200:
                a = best_a + np.random.default_rng().normal(0, 0.01/np.sqrt(n), n)
                a = project_simplex(a)
                lr = 0.05
                stuck = 0
    
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc

"""
Minimize 2n*max(conv(a))/(sum(a))^2 using multi-phase soft-max Adam optimization
on the probability simplex with adaptive temperature annealing, Fourier-domain
initialization, and top-peak polishing for final refinement.
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1} in O(n log n)."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def soft_max_weights(c, T):
    """Compute softmax(T * c) with numerical stability."""
    Tc = T * c
    Tc = Tc - Tc.max()
    w = np.exp(Tc)
    s = w.sum()
    if s < 1e-300:
        w = np.ones_like(c)
        s = w.size
    w /= s
    return w


def compute_gradient(a, w):
    """
    Compute gradient of sum_k w_k * c_k w.r.t. a.
    Uses identity: g[j] = 2 * sum_m a[m] * w[m+j] = 2 * convolve(a, w, 'valid')[j]
    where len(w) = 2*len(a) - 1.
    """
    return 2.0 * np.convolve(a, w, mode='valid')


def run_adam_soft_max(a0, n, duration, rng, T_start=5.0, T_end=300.0,
                      lr_start=0.05, lr_end=0.001,
                      beta1=0.9, beta2=0.999,
                      stagnation_limit=2000, eval_interval=50,
                      normalize_grad=True, verbose=True):
    """
    Soft-max Adam optimization on the simplex with adaptive annealing.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    step = 0

    best_a = a.copy()
    best_sc = compute_score(a)
    start_time = time.monotonic()
    deadline = start_time + duration
    last_improve_step = 0
    T = T_start
    restart_count = 0

    while time.monotonic() < deadline:
        step += 1

        c = np.convolve(a, a)
        w = soft_max_weights(c, T)
        g = compute_gradient(a, w)

        if normalize_grad:
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                g = g / g_norm
            else:
                continue

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / duration)
        lr = 0.5 * (lr_start + lr_end) + 0.5 * (lr_start - lr_end) * np.cos(np.pi * frac)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a_new)

        if step % eval_interval == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                last_improve_step = step

            # Anneal temperature (superlinear for faster convergence to sharp regime)
            T = T_start * (T_end / T_start) ** (frac ** 0.7)

            # Stagnation detection and restart
            if step - last_improve_step > stagnation_limit:
                restart_count += 1
                if verbose:
                    print(f"    [Adam] Restart #{restart_count} at step {step}, "
                          f"score={best_sc:.7f}, T={T:.1f}")
                a = best_a + rng.normal(0.0, 0.03 / np.sqrt(n), n)
                a = project_simplex(a)
                m[:] = 0.0
                v[:] = 0.0
                step = 0
                last_improve_step = 0
                T = max(T_start, T * 0.4)

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def polish_top_peaks(a0, n, duration, num_peaks=5, lr_start=0.03, verbose=True):
    """
    Polish using exact subgradient from top-k peaks with adaptive step size.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    deadline = time.monotonic() + duration
    step = 0
    lr = lr_start
    stuck = 0
    rng_local = np.random.default_rng(999)

    # FISTA momentum
    a_prev = a.copy()
    t_mom = 1.0

    while time.monotonic() < deadline:
        step += 1
        c = np.convolve(a, a)

        top_k = min(num_peaks, len(c))
        top_idx = np.argpartition(c, -top_k)[-top_k:]

        g = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                idx = k - np.arange(lo, hi, dtype=np.int64)
                g[lo:hi] += 2.0 * a[idx]
        g /= float(top_k)

        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g /= gn

        # FISTA momentum
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_mom
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy > 1e-15 and abs(sy - 1.0) > 0.05:
            y /= sy

        a_prev = a
        a = project_simplex(y - lr * g)

        if step % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck == 30:
                lr *= 0.85
            if stuck > 150:
                a = best_a + rng_local.normal(0, 0.01 / np.sqrt(n), n)
                a = project_simplex(a)
                a_prev = a.copy()
                t_mom = 1.0
                lr = lr_start
                stuck = 0

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def fourier_init(n, rng):
    """Initialize using a flat-ish random spectrum in Fourier domain."""
    phase = rng.uniform(0, 2 * np.pi, n)
    mag = 1.0 + 0.15 * rng.standard_normal(n)
    mag = np.maximum(mag, 0.05)
    A = mag * np.exp(1j * phase)
    A[0] = float(abs(A[0]))
    if n % 2 == 0:
        A[n // 2] = float(abs(A[n // 2]))
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def generate_diverse_inits(n, rng):
    """Generate diverse initial candidates."""
    inits = []
    names = []

    # Uniform
    inits.append(np.ones(n) / n)
    names.append("uniform")

    # Fourier inits
    for seed in range(4):
        r = np.random.default_rng(5000 + seed * 7)
        inits.append(fourier_init(n, r))
        names.append(f"fourier_{seed}")

    # Multi-frequency modulation
    for freq in [1, 2, 3, 5]:
        x = 1.0 + 0.25 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        x = np.maximum(x, 0.01)
        x /= x.sum()
        inits.append(x)
        names.append(f"mod_f{freq}")

    # Block structure
    for nb in [4, 7]:
        bs = n // nb
        x = np.zeros(n)
        for b in range(nb):
            x[b * bs:(b + 1) * bs] = rng.exponential(1.0)
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s > 1e-10:
            x /= s
            inits.append(x)
            names.append(f"blocks_{nb}")

    # Raised cosine
    x = 0.5 * (1.0 + np.cos(2 * np.pi * np.arange(n) / n))
    x = np.maximum(x, 0.01)
    x /= x.sum()
    inits.append(x)
    names.append("raised_cos")

    return inits, names
