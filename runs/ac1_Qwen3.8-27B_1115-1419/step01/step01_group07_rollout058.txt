    # This is 2 * cross-correlation of w and a, at offset i.
    # In numpy: 2 * np.correlate(w, a, mode='valid')[i] if w is longer than a.
    # w has length 2n-1, a has length n. 'valid' mode gives length (2n-1) - n + 1 = n.
    # np.correlate(w, a, 'valid')[i] = sum_j w[i+j]*a[j] (I need to verify the convention).
    
    # Actually, let me just compute it directly.
    # grad_a[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    
    n = len(a)
    grad_a = np.zeros(n)
    # For each i, grad_a[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1 where i+j < 2n-1
    # This is a valid cross-correlation
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # Verify: np.correlate(w, a, 'valid')[i] = sum_j w[i+j] * a[j]?
    # From numpy docs: cross-correlation c[k] = sum_i a[i+k] * conj(v[i])
    # So np.correlate(w, a, 'full')[k] = sum_i w[i+k] * a[i]
    # For 'valid' mode (w longer than a): result has length len(w) - len(a) + 1
    # result[i] = sum_j w[i+j] * a[j] for j = 0, ..., len(a)-1
    # Yes, that's what we want!
    
    # Chain rule through softmax: a = softmax(x), so da[i]/dx[m] = a[i]*(delta_{im} - a[m])
    # d(lse)/dx[m] = sum_i (d(lse)/da[i]) * (da[i]/dx[m])
    # = sum_i grad_a[i] * a[i]*(delta_{im} - a[m])
    # = grad_a[m]*a[m] - a[m]*sum_i grad_a[i]*a[i]
    # = a[m] * (grad_a[m] - a @ grad_a)
    
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x * 2 * n  # Scale by 2n from the objective


def subgradient_step(a, n_active, step_size):
    """One exact subgradient step on the probability simplex."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    if max_c < 1e-15:
        return a
    
    # Find active set
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    
    # Compute subgradient: average of gradients of active constraints
    # d(c[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[i] += 2*a[k-i] for i in [lo, hi]
        indices = np.arange(lo, hi + 1)
        grad[indices] += 2.0 * a[k - indices]
    grad /= len(active)
    
    # Step and project
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def sum_sq_objective(x):
    """Sum of squares of convolution, as a smooth objective."""
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    a = a / np.sum(a)
    c = np.convolve(a, a)
    return np.sum(c ** 2)


def sum_sq_gradient(x):
    """Gradient of sum of squares of convolution w.r.t. x."""
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    a = a / np.sum(a)
    n = len(a)
    c = np.convolve(a, a)
    # grad of sum(c^2) wrt a[i] = 2 * sum_k c[k] * 2*a[k-i] * 1[0<=k-i<n]
    # = 4 * sum_j c[i+j] * a[j]
    # = 4 * np.correlate(c, a, 'valid')[i]
    grad_a = 4.0 * np.correlate(c, a, mode='valid')
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x
