"""
Optimize step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterative frequency-domain flat autocorrelation design (Papoulis-style Wiener iteration),
projected subgradient polishing on exact max, log-parameterized L-BFGS refinement,
and multi-restart with diverse initializations.
Key insight: minimize peak-to-average ratio of autocorrelation via frequency-domain flattening.
"""

import numpy as np
import time
from scipy.optimize import minimize


def flat_top_iteration(a, N_fft, epsilon=1e-12, damping=0.7):
    """One step of frequency-domain flat autocorrelation design with damping."""
    n = len(a)
    A = np.fft.rfft(a, N_fft)
    psd = np.abs(A) ** 2
    target = np.mean(psd)
    corr = np.sqrt(target / (psd + epsilon))
    A_new = A * corr
    a_flat = np.fft.irfft(A_new, N_fft)
    a_flat = a_flat[:n]
    a_flat = np.maximum(a_flat, 0.0)
    s = a_flat.sum()
    if s < 1e-15:
        return a.copy()
    a_flat = a_flat / s
    # Damped update to avoid oscillation
    a_new = (1.0 - damping) * a + damping * a_flat
    a_new = np.maximum(a_new, 0.0)
    s2 = a_new.sum()
    if s2 < 1e-15:
        return a.copy()
    return a_new / s2


def subgradient_step(a, n_active=100, step_size=0.01):
    """Subgradient step for the true max of conv(a,a) using active set."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = conv.max()
    if max_c < 1e-15:
        return a
    sorted_idx = np.argsort(conv)[::-1]
    active = sorted_idx[:min(n_active, len(conv))]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    # Project gradient to tangent space of simplex (remove component along 1)
    g = g - np.mean(g)
    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def evaluate_score(a):
    """Quick score evaluation without function call overhead."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def log_param_objective(x, tau, n):
    """Smoothed objective using log parameterization (softmax)."""
    a = softmax(x)
    conv = np.convolve(a, a)
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    lse = c_max + tau * np.log(np.exp(shifted).sum())
    return 2.0 * n * lse


def log_param_gradient(x, tau, n):
    """Gradient of smoothed objective w.r.t. log-parameterization."""
    a = softmax(x)
    conv = np.convolve(a, a)
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    w_norm = w / w_sum
    g_a = np.zeros(n)
    for k in range(len(conv)):
        if w_norm[k] < 1e-16:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g_a[lo:hi + 1] += w_norm[k] * 2.0 * a[k - hi:k - lo + 1][::-1]
    dot = np.dot(a, g_a)
    grad_x = a * (g_a - dot)
    grad_x *= 2.0 * n
    return grad_x

        a = a.copy()

    print("\n=== Phase 2: L-BFGS refinement ===")
