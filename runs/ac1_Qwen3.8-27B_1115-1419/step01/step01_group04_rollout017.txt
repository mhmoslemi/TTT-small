```python
import numpy as np
import time
from scipy.optimize import linprog

def lp_refine(a, M, max_iter=10):
    n = len(a)
    S = a.sum()
    
    for it in range(max_iter):
        # Build LP: max sum(g) s.t. (f*g)_k <= M, g >= 0
        # (f*g)_k = sum_j g_j * f_{k-j}
        # For k = 0..2n-2, f_{k-j} is 0 if k-j < 0 or k-j >= n
        
        c = -np.ones(n)  # minimize -sum(g)
        
        A_ub = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A_ub[k, j] = a[idx]
        
        b_ub = np.full(2*n-1, M)
        
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        Sg = g.sum()
        if Sg < 1e-10:
            break
        
        # Rescale g to match the normalization of a
        g = g * S / Sg
        
        # Update
        t = 0.5  # step size
        a = (1-t)*a + t*g
        
        # Recompute M
        b = np.convolve(a, a)
        M = b.max()
    
    return a
```

"""
Searches for non-negative step-function heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Algorithm:
1. Generate diverse initial sequences at the target length.
2. Refine using multiplicative gradient descent (subgradient of max-convolution).
3. Periodically perform LP-based refinement for larger jumps.
4. Use random perturbations to escape local minima.
5. Validate the best candidate with the exact evaluator.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog

def fast_conv(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(c, 0.0)

def score_arr(a):
    """Compute the score for a numpy array."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a, target_sum=None):
    """Clip to [0, 1000] and normalize."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    if target_sum is None:
        target_sum = a.size
    s = a.sum()
    if s < 1e-15:
        a = np.full(a.size, target_sum / a.size)
    else:
        a = a * target_sum / s
    return np.clip(a, 0.0, 1000.0)

def subgrad_peak(a, c, K=1):
    """Subgradient of the max of the convolution, averaged over top K peaks."""
    n = a.size
    L = 2 * n - 1
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    d = np.zeros(n)
    inv_K = 1.0 / K
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            d[j] += inv_K * 2.0 * a[k - j]
    return d

def lp_refine_step(a, t=0.5):
    """One step of LP-based refinement."""
    n = a.size
    c = fast_conv(a)
    M = c.max()
    S = a.sum()

    c = np.fft.irfft(fa*fa, N)[:L]

    s = a.sum()
    if s < 0.01: return float('inf')
    c = fast_conv(a)
    return float(2*n*c.max()/s**2)

    a = np.clip(a, 0, 1000.0)
    if S is None: S = a.size
    s = a.sum()
    if s < 1e-15:

    a *= S/s
    return np.clip(a, 0, 1000.0)

        idx = np.argpartition(c, -K)[-K:]

        seg = a[k-hi:k-lo+1][::-1]

$\frac{\partial S}{\partial a_j} = \frac{2n}{S_a^2} \cdot 2 a_{k^*-j} \cdot \mathbf{1}_{valid} - \frac{4n b_{k^*}}{S_a^3}$

The second term is a constant (doesn't depend on $j$), so it doesn't affect the direction of descent (only the step size). So I can ignore it and just use $2 a_{k^*-j}$ as the "uncentered" subgradient.

For the centered subgradient (to preserve the sum constraint): $d_j = 2 a_{k^*-j} - \bar{d}$ where $\bar{d} = \frac{1}{n} \sum_j 2 a_{k^*-j}$.

This is what the previous code does. Good.

For the multiplicative update: $a_j \leftarrow a_j \exp(-\eta d_j)$. This automatically preserves non-negativity. The sum changes, so we renormalize.

The choice of $\eta$ is important. Too large: oscillation. Too small: slow convergence. I'll use an adaptive $\eta$ that increases on improvement and decreases on stagnation.

Let me also consider using a "line search" to find the best $\eta$ at each step. Given the direction $d$, find $\eta^* = \arg\min_\eta S(a \cdot \exp(-\eta d))$. This is a 1D optimization, which can be done with golden section search or bisection.

But the function $S(a \cdot \exp(-\eta d))$ is not unimodal in general, so bisection might not work. Golden section search is more robust but requires multiple evaluations.

For simplicity, I'll use a fixed $\eta$ with adaptive adjustment.

"""
Multi-phase search minimizing 2n*max(conv(a,a))/sum(a)^2 for non-negative a.
Combines: FFT subgradient descent, LP-based refinement (HiGHS), adaptive annealing,
diverse analytical + random initialization, and exact final validation.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def fast_conv(a):
    n = a.size
    if n == 0:
        return np.empty(0)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def calc_score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def do_normalize(a, target=None):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return a
    if target is None:
        target = float(a.size)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target / a.size)
    a = a * target / s
    return np.clip(a, 0.0, 1000.0)


def compute_subgrad(a, c, K):
    n = a.size
    L = 2 * n - 1
    K = max(1, min(K, L))
    if K < L:
        idx = np.argpartition(c, -K)[-K:]
    else:
        idx = np.arange(L)
    d = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        d[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    d /= float(K)
    return d


def upsample_seq(seq, n):
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    n = int(n)
    if m == 0:
        return do_normalize(np.ones(n), float(n))
    if m == n:
        return do_normalize(seq, float(n))
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    return do_normalize(np.interp(xnew, xold, seq), float(n))


def smooth_perturb(a, rng, strength=0.3):
    a = a.copy()
    n = a.size
    if n > 2:
        t = np.linspace(0.0, 1.0, n)
        z = np.zeros(n)
        for m in range(1, 5):
            z += (strength / (0.5 + 0.5 * m)) * np.cos(2 * np.pi * m * t + rng.uniform(0, 2 * np.pi))
        a *= np.exp(z)
    if n > 8:
        w = max(1, n // 16)
        st = int(rng.integers(0, n - w))
        a[st:st + w] *= np.exp(rng.normal(0, strength, w))
    return do_normalize(a, float(n))


def lp_refine(a, t=0.4, relax=1.005):
    n = a.size
    c = fast_conv(a)
    M = c.max() * relax
    S = a.sum()
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    b_ub = np.full(2 * n - 1, M)
    bnds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bnds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return a
    if res is None or res.status != 0 or res.x is None:
        return a
    g = np.maximum(res.x, 0.0)
    Sg = g.sum()
    if Sg < 1e-15:
        return a
    g = g * S / Sg
    return do_normalize((1.0 - t) * a + t * g, S)


def gen_starts(n, rng, prev_best):
    starts = []
    starts.append(do_normalize(np.ones(n), float(n)))
    # Two-block grid
    for mf in [0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
        m = max(1, int(mf * n))
        for ratio in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
            a = np.empty(n)
            a[:m] = ratio
            a[m:] = 1.0
            starts.append(do_normalize(a, float(n)))
            a2 = a.copy()
            a2[:m] = 1.0
            a2[m:] = ratio
            starts.append(do_normalize(a2, float(n)))
    # Global shapes
    t = np.linspace(0.0, 1.0, n)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        starts.append(do_normalize(1.0 + alpha * np.cos(np.pi * (2 * t - 1)), float(n)))
        starts.append(do_normalize(1.0 + alpha * (2 * t - 1) ** 2, float(n)))
        starts.append(do_normalize(np.sin(np.pi * t) ** 0.5 + 0.1, float(n)))
        starts.append(do_normalize(np.sin(np.pi * t) ** 2 + 0.1, float(n)))
    # Random smooth
    for _ in range(15):
        z = np.zeros(n)
        for m in range(1, 7):
            z += rng.uniform(-1, 1) / (m + 0.5) ** 0.8 * np.cos(2 * np.pi * m * t + rng.uniform(0, 2 * np.pi))
        starts.append(do_normalize(np.exp(z - z.max()), float(n)))
    # Random blocks
    for _ in range(8):
        nb = int(rng.integers(3, 13))
        edges = np.sort(rng.choice(range(1, n), nb - 1, replace=False))
        edges = np.concatenate([[0], edges, [n]])
        vals = rng.uniform(0.1, 3.0, nb)
        if rng.random() < 0.3:
            vals[int(rng.integers(0, nb))] = 0.0
        a = np.repeat(vals, np.diff(edges).astype(int))
        starts.append(do_normalize(a, float(n)))
    # Chirp-like
    if n > 4:
        tt = np.arange(n, dtype=np.float64)
        starts.append(do_normalize(np.abs(np.cos(np.pi * tt * tt / n)) + 0.05, float(n)))
        starts.append(do_normalize(np.abs(np.sin(np.pi * tt * tt / n)) + 0.05, float(n)))
    # Previous best
    if prev_best is not None:
        starts.append(upsample_seq(prev_best, n))
        starts.append(upsample_seq(prev_best[::-1], n))
    # height_sequence_1
    try:
        hs1 = np.asarray(global_this.get("height_sequence_1", []), dtype=np.float64)
        if hs1.size > 0:
            starts.append(upsample_seq(hs1, n))
            starts.append(upsample_seq(hs1[::-1], n))
            smoothed = np.convolve(hs1, np.ones(5) / 5, mode='same')
            starts.append(upsample_seq(smoothed, n))
    except Exception:
        pass
    # Deduplicate and score
    seen = set()
    scored = []
    for a in starts:
        key = tuple(np.round(a, 3))
        if key in seen:
            continue
        seen.add(key)
        sc = calc_score(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda x: x[0])
    return scored


def quick_grad_search(a0, steps, rng, n, initial_eta=0.25):
    a = a0.copy()
    best_a = a.copy()
    best_sc = calc_score(a)
    eta = initial_eta
    noise = 0.04
    mom = np.zeros(n)
    beta = 0.3
    stagnant = 0
    for _ in range(steps):
        c = fast_conv(a)
        S_a = a.sum()
        K = max(1, min(32, n // 32))
        d = compute_subgrad(a, c, K)
        d -= d.mean()
        sc = d.max() - d.min()
        if sc < 1e-14:
            stagnant += 10
            continue
        d /= sc
        mom = beta * mom + (1 - beta) * d
        if noise > 0.003:
            mom += noise * rng.standard_normal(n)
        cand = a * np.exp(-eta * mom)
        cand = do_normalize(cand, S_a)
        cc = fast_conv(cand)
        sc2 = float(2 * n * cc.max() / (S_a * S_a))
        cur_sc = float(2 * n * c.max() / (S_a * S_a))
        if sc2 < best_sc:
            best_a = cand.copy()
            best_sc = sc2
            stagnant = 0
        if sc2 <= cur_sc * 1.03:
            a = cand
            if sc2 < cur_sc:
                eta = min(0.8, eta * 1.06)
            else:
                eta = max(0.05, eta * 0.97)
            noise = max(0.003, noise * 0.998)
        else:
            eta = max(0.05, eta * 0.75)
            mom *= 0.5
            stagnant += 1
        if stagnant > 150:
            a = smooth_perturb(best_a, rng, 0.25)
            eta = 0.3
            noise = 0.06
            mom = np.zeros(n)
            stagnant = 0
    return best_a, best_sc
