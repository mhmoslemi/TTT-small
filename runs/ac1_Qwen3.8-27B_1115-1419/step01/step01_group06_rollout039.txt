        p <<= 1

    w -= w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0:
        return np.ones_like(w) / len(w)
    return e / s

    w = np.log(a)
    w -= w.max()
    return w

def _fft_conv(a, N):

    c = np.fft.irfft(A * A, N)
    n = len(a)
    c = c[:2*n-1]
    return np.maximum(c, 0.0)

    c = _fft_conv(a, N)

    s = a.sum()
    if s < 1e-10:

"""
Hybrid optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: (1) FFT-based smoothed-Linf descent with adaptive parameters,
      (2) exact coordinate descent with incremental convolution updates,
      (3) multi-start with parametric initializations.

import numpy as np
import time


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _softmax(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size == 0:
        return np.ones(1)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size) / w.size
    return e / s


def _log_weights(a):
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    return w


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    n = len(a)
    c = c[:2 * n - 1]
    return np.maximum(c, 0.0)


def _soft_weights(c, beta):
    mx = c.max()
    if mx <= 0 or not np.isfinite(mx):
        return np.ones_like(c) / len(c)
    x = (c - mx) / mx
    x = np.clip(x * beta, -745, 0)
    e = np.exp(x)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[c.argmax()] = 1.0
        return p
    return e / s


def _adam_step(grad, w, m, v, t, lr):
    b1, b2 = 0.9, 0.999
    t += 1
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    mhat = m / (1 - b1 ** t)
    vhat = v / (1 - b2 ** t)
    step = mhat / (np.sqrt(vhat) + 1e-12)
    step = np.clip(step, -10, 10)
    w_new = w - lr * step
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= w_new.max()
    return _softmax(w_new), w_new, m, v, t


def _load_start():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 8:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return arr / s


def _make_parametric_starts(n, rng):
    starts = []
    # Cosine modulation
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = (1 + 0.4 * np.cos(t)) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    starts.append(a)
    # Dual frequency
    a = (1 + 0.3 * np.cos(2 * t) + 0.2 * np.cos(4 * t)) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    starts.append(a)
    # Gaussian-like
    x = np.arange(n) / n - 0.5
    a = np.exp(-20 * x * x)
    a = a / a.sum()
    starts.append(a)
    # Bimodal
    a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x + 0.25) ** 2)
    a = a / a.sum()
    starts.append(a)
    # Random smooth
    rough = rng.random(n)
    kernel = np.exp(-0.5 * (np.linspace(-6, 6, min(101, n))) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(rough, kernel, mode='same')
    a = np.maximum(a, 0)
    a = a / a.sum()
    starts.append(a)
    return starts


def _fft_optimize(a, budget_s, rng, n_override=None):
    """FFT-based smoothed-Linf optimization. Returns best (a, score)."""
    start = time.monotonic()
    n = len(a)
    N = _next_pow2(3 * n - 2)
    phase = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / N)

    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    A = np.fft.rfft(a, N)
    c = _fft_conv(a, N)
    score = 2.0 * n * c.max()

    best_a = a.copy()
    best_score = score

    w = _log_weights(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = 4.0
    lr = 0.06
    stall = 0
    iters = 0

    FR = phase * np.conjugate(A)

    while time.monotonic() - start < budget_s:
        iters += 1

        p = _soft_weights(c, beta)
        P = np.fft.rfft(p, N)
        conv_grad = np.fft.irfft(P * FR, N)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        grad = np.where(np.isfinite(grad), grad, 0.0)

        grad_w = a * (grad - a @ grad)
        a, w, m, v, t = _adam_step(grad_w, w, m, v, t, lr)

        A = np.fft.rfft(a, N)
        c = _fft_conv(a, N)
        score = 2.0 * n * c.max()

        if not np.isfinite(score):
            a = best_a.copy()
            score = best_score
            A = np.fft.rfft(a, N)
            c = _fft_conv(a, N)
            w = _log_weights(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            FR = phase * np.conjugate(A)
            continue

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
            lr = min(0.18, lr * 1.015)
        else:
            stall += 1
            lr = max(0.008, lr * 0.98)

        FR = phase * np.conjugate(A)

        if iters % 40 == 0:
            beta = min(300.0, beta * 1.04 + 0.2)

        if stall > 60:
            r = rng.random()
            if r < 0.4:
                a = best_a.copy()
                score = best_score
            elif r < 0.7:
                sig = rng.choice([0.03, 0.08, 0.15, 0.3])
                w_try = _log_weights(best_a) + rng.normal(0, sig, n)
                a = _softmax(w_try)
                A = np.fft.rfft(a, N)
                c = _fft_conv(a, N)
                score = 2.0 * n * c.max()
            else:
                # Block permutation
                a2 = a.copy()
                block = max(8, n // 32)
                nblocks = n // block
                if nblocks > 1:
                    perm = rng.permutation(nblocks)
                    a2[:nblocks * block] = a[:nblocks * block].reshape(nblocks, block)[perm].ravel()
                a = a2
                A = np.fft.rfft(a, N)
                c = _fft_conv(a, N)
                score = 2.0 * n * c.max()

            w = _log_weights(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            FR = phase * np.conjugate(A)
            stall = 0
            lr = 0.06
            beta = max(4.0, beta * 0.5)

    return best_a, best_score


def _exact_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def _coord_descent(a, budget_s, rng):
    """Exact coordinate descent with incremental convolution updates."""
    start = time.monotonic()
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    s = a.sum()
    if s <= 0:
        return a
    a = a / s

    c = np.convolve(a, a)
    cur_max = c.max()
    cur_obj = 2.0 * n * cur_max

    multipliers = np.array([0.2, 0.4, 0.6, 0.75, 0.85, 0.92,
                            1.08, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0])

    passes = 0
    no_improve_rounds = 0

    while time.monotonic() - start < budget_s:
        passes += 1
        order = rng.permutation(n)
        pass_improved = 0

        for i in order:
            if time.monotonic() - start > budget_s:
                break

            a_i = a[i]
            if a_i < 1e-14:
                # Try setting to a small value
                target = 0.01 / n  # small target
                delta = target - a_i
                if delta <= 1e-15:
                    continue
            else:
                deltas = a_i * (multipliers - 1.0)
                # Also try setting to 0
                deltas = np.append(deltas, -a_i * 0.999)
                # Filter out deltas that would make a_i negative
                deltas = deltas[deltas > -a_i + 1e-15]
                deltas = deltas[np.abs(deltas) > 1e-15 * max(1, a_i)]
                if len(deltas) == 0:
                    continue

            best_delta = None
            best_obj = cur_obj

            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            affected_len = k_hi - k_lo + 1
            idx = np.arange(k_lo, k_hi + 1)

            base = a[idx - i]  # a[k-i] for k in affected range
            has_2i = (2 * i >= k_lo) and (2 * i <= k_hi)
            pos_2i = 2 * i - k_lo if has_2i else -1

            # Vectorized: compute new max for all deltas at once
            affected_base = c[idx]  # shape: (affected_len,)

            # For each delta d: affected_new = affected_base + 2*d*base + d^2*(k==2i)
            # Shape: (num_deltas, affected_len)
            d2 = deltas * base[None, :]  # (D, L)
            affected_new = affected_base[None, :] + 2.0 * d2  # (D, L)
            if has_2i:
                affected_new[:, pos_2i] += deltas ** 2

            max_affected = affected_new.max(axis=1)  # (D,)

            # Unaffected max
            if k_lo > 0:
                if k_hi < 2 * n - 2:
                    max_unaffected = max(c[:k_lo].max(), c[k_hi + 1:].max())
                else:
                    max_unaffected = c[:k_lo].max()
            elif k_hi < 2 * n - 2:
                max_unaffected = c[k_hi + 1:].max()
            else:
                max_unaffected = 0.0

            max_temp = np.maximum(max_affected, max_unaffected)
            new_sums = 1.0 + deltas
            valid = new_sums > 1e-10
            if not valid.all():
                max_temp = max_temp[valid]
                deltas = deltas[valid]
                new_sums = new_sums[valid]

            obj_new = 2.0 * n * max_temp / (new_sums * new_sums)
            best_idx = obj_new.argmin()

            if obj_new[best_idx] < best_obj - 1e-13:
                best_obj = obj_new[best_idx]
                best_delta = deltas[best_idx]

            if best_delta is not None:
                # Apply
                a[i] += best_delta
                new_sum = a.sum()
                a = a / new_sum

                # Update c incrementally
                # c_temp = c + delta_pattern (before normalization)
                # c_new = c_temp / new_sum^2  (but new_sum should be ~1+best_delta)
                # Actually new_sum = 1 + best_delta (since sum was 1 before)
                # Let's just recompute the affected part
                c[idx] += 2.0 * best_delta * base  # base was a[k-i] before change
                if has_2i:
                    c[pos_2i + k_lo] += best_delta ** 2
                # Normalize c
                c = c / (new_sum * new_sum)

                cur_max = c.max()
                cur_obj = 2.0 * n * cur_max
                pass_improved += 1

        # Periodic exact recompute to fix drift
        if passes % 20 == 0:
            c = np.convolve(a, a)
            cur_max = c.max()
            cur_obj = 2.0 * n * cur_max

        if pass_improved == 0:
            no_improve_rounds += 1
            if no_improve_rounds >= 8:
                break
        else:
            no_improve_rounds = 0

        print(f"  CD pass {passes}: improved={pass_improved}, obj={cur_obj:.6f}", flush=True)

    return a
