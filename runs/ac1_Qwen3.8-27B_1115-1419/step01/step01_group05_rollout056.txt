    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s

def _full_conv(a):
    """Full self-convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = len(a)
    b = _full_conv(a)
    return 2.0 * n * float(b.max())

def _grad_fft(a, w):
    """
    Gradient of sum_k w_k * (a*a)_k w.r.t. a.
    Uses cross-correlation via FFT: grad_i = 2 * sum_j a_j * w_{i+j}
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.rfft(a, N)
    W_rev = w[::-1]
    W = np.fft.rfft(W_rev, N)
    conv = np.fft.irfft(A * W, N)
    C = conv[2 * n - 2 : n - 1 : -1]
    return 2.0 * C

def _grad_direct(a, w):
    """Direct O(n^2) gradient computation."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        max_j = min(n - 1, 2 * n - 2 - i)
        if max_j >= 0:
            grad[i] = 2.0 * np.dot(a[:max_j + 1], w[i:i + max_j + 1])
    return grad

def _lse_smooth(b, beta):
    """Log-sum-exp approximation of max(b)."""
    m = float(b.max())
    if m < 1e-300:
        return m
    x = np.clip(beta * (b - m), -700.0, 0.0)
    return m + float(np.log(np.sum(np.exp(x)))) / beta

def _lse_weights(b, beta):
    """Softmax weights proportional to exp(beta * b)."""
    m = float(b.max())
    x = np.clip(beta * (b - m), -700.0, 0.0)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return e / s

def _exp_weight(b):
    """Exponential weighting based on how close each element is to max."""
    m = float(b.max())
    if m < 1e-300:
        return np.ones(len(b)) / len(b)
    # Weight by how "active" each constraint is
    ratio = b / m
    return ratio ** 20.0 / (ratio ** 20.0).sum()

def _get_initial(n):
    """Get initial sequence from global or create default."""
    seq = globals().get("height_sequence_1", None)
    if seq is not None:
        try:
            arr = np.asarray(seq, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, None)
            if arr.size == n and arr.sum() > 1e-30:
                return arr / arr.sum()
            if arr.size > 4:
                x_old = np.linspace(0, 1, arr.size)
                x_new = np.linspace(0, 1, n)
                new = np.interp(x_new, x_old, arr)
                return _normalize(new)
        except Exception:
            pass
    # Default: slightly modulated uniform
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
    return _normalize(a)

def _perturb(a, rng, scale=0.1):
    """Multiplicative perturbation."""
    n = len(a)
    noise = np.exp(rng.normal(0, scale, size=n))
    return _normalize(a * noise)

def _block_perturb(a, rng, strength=0.3):
    """Perturb a random block."""
    n = len(a)
    c = a.copy()
    width = max(4, n // 16)
    start = int(rng.integers(0, n - width))
    c[start:start + width] *= np.exp(rng.normal(0, strength, size=width))
    return _normalize(c)

def _lse_optimize(a, deadline, rng, time_budget=60.0):
    """
    LSE-smoothed projected gradient descent.
    Returns (best_a, best_score).
    """
    n = len(a)
    t_start = time.monotonic()
    t_end = min(deadline, t_start + time_budget)
    
    best_a = a.copy()
    best_score = _score(a)
    b = _full_conv(a)
    
    lr = 0.05
    beta = 10.0
    fail_count = 0
    it = 0
    
    while time.monotonic() < t_end:
        smooth_val = _lse_smooth(b, beta)
        weights = _lse_weights(b, beta)
        grad = _grad_fft(a, weights)
        
        # Center the gradient (remove mean to stay on simplex better)
        grad_centered = grad - grad.mean()
        grad_norm = np.abs(grad_centered).max()
        if grad_norm < 1e-15:
            break
        
        # Adaptive step
        step = lr * (b.max() / max(grad_norm, 1e-15))
        v = a - step * grad_centered
        a_new = _normalize(v)
        
        b_new = _full_conv(a_new)
        smooth_new = _lse_smooth(b_new, beta)
        score_new = 2.0 * n * float(b_new.max())
        
        if smooth_new < smooth_val - 1e-12:
            a = a_new
            b = b_new
            lr = min(lr * 1.1, 0.5)
            fail_count = 0
            if score_new < best_score:
                best_a = a.copy()
                best_score = score_new
        else:
            lr *= 0.6
            fail_count += 1
            if fail_count > 50:
                # Restart with perturbation
                a = _perturb(best_a, rng, 0.15)
                b = _full_conv(a)
                lr = 0.05
                beta = max(5.0, beta * 0.5)
                fail_count = 0
        
        # Increase beta periodically
        if it % 80 == 0:
            beta = min(beta * 1.3, 200.0)
        
        it += 1
    
    # Final check
    final_score = _score(a)
    if final_score < best_score:
        best_a = a.copy()
        best_score = final_score
    
    return best_a, best_score

def _build_conv_matrix(f):
    """
    Build the convolution matrix F where (F @ g)_k = (f*g)_k.
    F has shape (2n-1, n) with F[k,i] = f[k-i] if 0<=k-i<n else 0.
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_step(f, solver="CLARABEL"):
    """
    Solve LP: minimize M s.t. (f*g)_k <= M, g >= 0, sum(g) = 1.
    Returns optimal g or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(f)
    F = _build_conv_matrix(f)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [
        cp.sum(g) == 1,
        F @ g <= M
    ]
    
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None and np.isfinite(g.value).all():
            return np.maximum(g.value, 0.0)
    except Exception:
        pass
    
    # Try alternative solver
    try:
        prob.solve(solver="ECOS")
        if g.value is not None and np.isfinite(g.value).all():
            return np.maximum(g.value, 0.0)
    except Exception:
        pass
    
    return None

def _lp_optimize(a, deadline, rng, time_budget=60.0):
    """
    LP-based optimization: iteratively find best g under current peak, blend.
    """
    t_start = time.monotonic()
    t_end = min(deadline, t_start + time_budget)
    
    best_a = a.copy()
    best_score = _score(a)
    n = len(a)
    
    lp_count = 0
    while time.monotonic() < t_end:
        g = _lp_step(a)
        lp_count += 1
        if g is None:
            print("LP solver failed, switching to gradient", flush=True)
            break
        
        g = _normalize(g)
        g_score = _score(g)
        
        # Try different blend factors
        improved = False
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            blend = _normalize((1 - t) * a + t * g)
            s = _score(blend)
            if s < best_score:
                best_a = blend.copy()
                best_score = s
                a = blend
                improved = True
                break
            if s < _score(a):
                a = blend
        
        if not improved:
            # Try symmetric LP
            g_sym = _normalize((g + g[::-1]) / 2.0)
            for t in [0.3, 0.5, 0.7]:
                blend = _normalize((1 - t) * a + t * g_sym)
                s = _score(blend)
                if s < best_score:
                    best_a = blend.copy()
                    best_score = s
                    a = blend
                    improved = True
                    break
        
        if not improved and lp_count > 3:
            break
    
    print(f"LP optimization: {lp_count} LPs, best_score={best_score:.8f}", flush=True)
    return best_a, best_score

def _coordinate_descent(a, deadline, rng, time_budget=30.0):
    """
    Coordinate descent: try adjusting each coordinate.
    """
    t_start = time.monotonic()
    t_end = min(deadline, t_start + time_budget)
    
    best_a = a.copy()
    best_score = _score(a)
    n = len(a)
    
    while time.monotonic() < t_end:
        improved_any = False
        # Random order of coordinates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > t_end:
                break
            old_val = a[i]
            if old_val < 1e-10:
                # Try adding a small value
                for mult in [0.01, 0.005]:
                    a[i] = mult * a.mean()
                    s = _score(a)
                    if s < best_score:
                        best_a = a.copy()
                        best_score = s
                        improved_any = True
                        break
                else:
                    a[i] = old_val
                continue
            
            # Try several multiplicative adjustments
            for mult in [0.5, 0.7, 1.3, 1.5, 0.3, 1.7]:
                a[i] = old_val * mult
                a = _normalize(a)
                s = _score(a)
                if s < best_score:
                    best_a = a.copy()
                    best_score = s
                    improved_any = True
                    old_val = a[i]  # update for next trial
                else:
                    a[i] = old_val
                    a = _normalize(a)
        
        if not improved_any:
            break
    
    print(f"Coordinate descent: best_score={best_score:.8f}", flush=True)
    return best_a, best_score

def _pair_swap(a, deadline, rng, time_budget=20.0):
    """
    Try swapping mass between pairs of coordinates.
    """
    t_start = time.monotonic()
    t_end = min(deadline, t_start + time_budget)
    
    best_a = a.copy()
    best_score = _score(a)
    n = len(a)
    
    while time.monotonic() < t_end:
        improved = False
        for _ in range(100):
            if time.monotonic() > t_end:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            vi = a[i]
            vj = a[j]
            if vi < 1e-10 and vj < 1e-10:
                continue
            
            # Try transferring mass from i to j or vice versa
            for frac in [0.1, 0.25, 0.5, 0.75]:
                a[i] = vi * (1 - frac) + vj * frac * 0.5
                a[j] = vj * (1 - frac) + vi * frac * 0.5
                a = _normalize(a)
                s = _score(a)
                if s < best_score:
                    best_a = a.copy()
                    best_score = s
                    improved = True
                    vi = a[i]
                    vj = a[j]
        
        if not improved:
            break
    
    print(f"Pair swap: best_score={best_score:.8f}", flush=True)
    return best_a, best_score

def _multiple_beta_optimize(a, deadline, rng, time_budget=60.0):
    """
    Try multiple beta values for LSE to escape local minima.
    """
    t_start = time.monotonic()
    t_end = min(deadline, t_start + time_budget)
    
    best_a = a.copy()
    best_score = _score(a)
    n = len(a)
    
    betas = [5.0, 10.0, 20.0, 50.0, 100.0]
    time_per = time_budget / len(betas)
    
    for beta0 in betas:
        if time.monotonic() >= t_end:
            break
        
        a_curr = _normalize(a)
        b = _full_conv(a_curr)
        lr = 0.03
        it = 0
        no_improve = 0
        
        beta = beta0
        while time.monotonic() < min(t_end, time.monotonic() + time_per):
            smooth_val = _lse_smooth(b, beta)
            weights = _lse_weights(b, beta)
            grad = _grad_fft(a_curr, weights)
            grad_centered = grad - grad.mean()
            grad_norm = np.abs(grad_centered).max()
            if grad_norm < 1e-15:
                break
            
            step = lr
            v = a_curr - step * grad_centered / max(grad_norm, 1e-15) * b.max()
            a_new = _normalize(v)
            
            b_new = _full_conv(a_new)
            smooth_new = _lse_smooth(b_new, beta)
            score_new = 2.0 * n * float(b_new.max())
            
            if smooth_new < smooth_val - 1e-13:
                a_curr = a_new
                b = b_new
                lr = min(lr * 1.05, 0.3)
                no_improve = 0
                if score_new < best_score:
                    best_a = a_curr.copy()
                    best_score = score_new
            else:
                lr *= 0.7
                no_improve += 1
                if no_improve > 30:
                    break
            
            if it % 50 == 0:
                beta *= 1.2
            it += 1
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for the best non-negative sequence of length 256
    minimizing 2*n*max(a*a)/(sum a)^2.
    """
    n = 256
    total_budget = float(budget_s) if np.isfinite(budget_s) and budget_s > 0 else 2400.0
    deadline = time.monotonic() + total_budget - 40.0
    rng = np.random.default_rng(seed)
    
    print(f"Starting search with budget={total_budget:.0f}s, deadline in {deadline - time.monotonic():.0f}s", flush=True)
    
    # Initialize
    best_a = _get_initial(n)
    best_score = _score(best_a)
    print(f"Initial score: {best_score:.8f} (len={n})", flush=True)
    
    # Also try a few alternative initializations
    inits = [
        _get_initial(n),
        _normalize(np.exp(-1.5 * np.linspace(0, 1, n))),
        _normalize(np.exp(-1.0 * np.abs(np.linspace(-1, 1, n)))),
        _normalize(1.0 + 0.5 * np.cos(4 * np.pi * np.linspace(0, 1, n))),
        _normalize(1.0 + 0.3 * np.sin(2 * np.pi * np.linspace(0, 1, n))),
        _normalize(1.0 + 0.4 * np.cos(2 * np.pi * 3 * np.linspace(0, 1, n))),
    ]
    
    for idx, init in enumerate(inits):
        s = _score(_normalize(init))
        if s < best_score:
            best_a = _normalize(init).copy()
            best_score = s
            print(f"Better init {idx}: score={s:.8f}", flush=True)
    
    # Main optimization loop with alternating methods
    phase = 0
    stagnation = 0
    
    while time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        if phase == 0:
            # LSE gradient descent
            t_budget = min(120, remaining * 0.25)
            a_new, s_new = _lse_optimize(best_a, deadline, rng, time_budget=t_budget)
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                stagnation = 0
                print(f"Phase 0 (LSE): score={best_score:.8f}", flush=True)
            else:
                stagnation += 1
            phase = 1
        
        elif phase == 1:
            # LP-based optimization
            t_budget = min(90, remaining * 0.15)
            a_new, s_new = _lp_optimize(best_a, deadline, rng, time_budget=t_budget)
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                stagnation = 0
                print(f"Phase 1 (LP): score={best_score:.8f}", flush=True)
            else:
                stagnation += 1
            phase = 2
        
        elif phase == 2:
            # Multi-beta LSE
            t_budget = min(90, remaining * 0.15)
            a_new, s_new = _multiple_beta_optimize(best_a, deadline, rng, time_budget=t_budget)
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                stagnation = 0
                print(f"Phase 2 (Multi-beta): score={best_score:.8f}", flush=True)
            else:
                stagnation += 1
            phase = 3
        
        elif phase == 3:
            # Coordinate descent
            t_budget = min(60, remaining * 0.1)
            a_new, s_new = _coordinate_descent(best_a, deadline, rng, time_budget=t_budget)
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                stagnation = 0
                print(f"Phase 3 (Coord): score={best_score:.8f}", flush=True)
            else:
                stagnation += 1
            phase = 4
        
        elif phase == 4:
            # Pair swap
            t_budget = min(40, remaining * 0.08)
            a_new, s_new = _pair_swap(best_a, deadline, rng, time_budget=t_budget)
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                stagnation = 0
                print(f"Phase 4 (Swap): score={best_score:.8f}", flush=True)
            else:
                stagnation += 1
            phase = 0
        
        # If stuck, do a restart with perturbation
        if stagnation >= 5:
            print(f"Stagnation detected (score={best_score:.8f}), restarting with perturbation", flush=True)
            a_pert = _perturb(best_a, rng, 0.2)
            # Quick LSE from perturbed start
            a_new, s_new = _lse_optimize(a_pert, deadline, rng, time_budget=min(60, remaining * 0.1))
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                print(f"After restart: score={best_score:.8f}", flush=True)
            stagnation = 0
            phase = 0
    
    # Final polish with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"Final polish with {remaining:.1f}s remaining", flush=True)
        a_new, s_new = _lse_optimize(best_a, deadline - 5, rng, time_budget=remaining * 0.5)
        if s_new < best_score:
            best_a = a_new
            best_score = s_new
        
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_new, s_new = _coordinate_descent(best_a, deadline - 5, rng, time_budget=remaining)
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
    
    # Final symmetric refinement: try symmetrizing
    a_sym = _normalize((best_a + best_a[::-1]) / 2.0)
    s_sym = _score(a_sym)
    if s_sym < best_score:
        best_a = a_sym
        best_score = s_sym
        print(f"Symmetrization improved: score={best_score:.8f}", flush=True)
    
    print(f"Final best score: {best_score:.8f} (length={len(best_a)})", flush=True)
    
    # Ensure valid output
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s < 0.01:
        out = np.ones(n) / n
    else:
        out = out / s
    
    return [float(x) for x in out]
