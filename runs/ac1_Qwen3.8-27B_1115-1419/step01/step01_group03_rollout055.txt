    while p < m:

    return p

def project_simplex(v):
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0/n)
    r = np.where(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
