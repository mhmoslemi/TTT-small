"""
Multi-phase optimization of step function to minimize 2n*max(autocorr)/sum^2.
Uses LP-based descent (primary), subgradient refinement (secondary),
and diverse initializations for global exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n_padded = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_padded) * fft(b, n_padded)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum=None):
    """Normalize array to target sum (or keep current sum if None)."""
    s = float(np.sum(a))
    if target_sum is None:
        target_sum = s
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _build_constraint_matrix(a, peak_indices):
    """Build dense constraint matrix for LP.
    A[i, j] = a[k_i - j] if 0 <= k_i - j < len(a)
    """
    n = len(a)
    conv_len = 2 * n - 1
    A = np.zeros((len(peak_indices), n))
    for i, k in enumerate(peak_indices):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_end > j_start:
            # A[i, j] = a[k-j] for j in [j_start, j_end)
            # When j = j_start: a[k - j_start]
            # When j = j_end-1: a[k - (j_end-1)]
            A[i, j_start:j_end] = a[k_int - (j_end - 1): k_int - j_start + 1][::-1]
    return A


def _lp_direction(a, top_k=200, timeout_s=30):
    """Solve LP to find best ascent direction for sum(g) subject to conv(a,g) <= M.
    Returns (g, M) or (None, M) on failure.
    """
    n = len(a)
    conv = _fft_conv(a, a)
    M = float(np.max(conv))
    
    if M < 1e-15:
        return None, M
    
    # Get top-k peak positions
    peak_idx = np.argsort(conv)[-top_k:]
    
    # Build constraint matrix
    A = _build_constraint_matrix(a, peak_idx)
    
    # Solve LP: maximize sum(g) subject to A@g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0, None)] * n
    
    start = time.monotonic()
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout_s})
    
    if time.monotonic() - start > timeout_s + 5:
        return None, M
    
    if res.status == 0:  # Optimal
        return res.x, M
    elif res.status == 3:  # Infeasible (shouldn't happen)
        return None, M
    else:
        # For status 1 (max iter) or 4 (timeout), use whatever we have
        if res.x is not None and res.x is not None:
            # Check if it's feasible
            if np.all(A @ res.x <= b_ub + 1e-6) and np.all(res.x >= -1e-9):
                return np.maximum(res.x, 0), M
        return None, M


def _lp_step(a, top_k=200, t=0.3, timeout_s=30):

    n = len(a)
    a_sum = float(np.sum(a))

    g_sum = float(np.sum(g))

def _subgrad_step(a, lr, top_k=10):

    n = len(a)
    conv = _fft_conv(a, a)
    sum_a = float(np.sum(a))

    a_sum = float(np.sum(a))
    best = a.copy()
    best_score = _score_fast(a)
    t = t_init
    stuck = 0
    iteration = 0
    
    while time.monotonic() < deadline - 10:

        a = _lp_step(a, top_k=top_k, t=t, timeout_s=timeout_s)

                a = _subgrad_step(a, lr=0.01, top_k=5)

            best = a.copy()
            stuck = 0

            stuck += 1

            a = best + 0.01 * np.random.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = _normalize(a, a_sum)
            stuck = 0
            t = t_init * 0.5
        
        if iteration % 5 == 0:

"""
Multi-phase optimization minimizing 2n*max(self-conv)/(sum)^2 for non-negative sequences.
Primary method: LP-based descent (solve max sum(g) s.t. conv(a,g)<=M, g>=0).
Secondary: subgradient refinement and diverse restarts.
Exploits scale invariance (fix sum(a)=sqrt(2n), minimize max(conv)).
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n_padded = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_padded) * fft(b, n_padded)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize array to target sum."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones_like(a) * (target_sum / len(a))
    return a * (target_sum / s)


def _build_A(a, peak_indices):
    """Build constraint matrix: A[i,j] = a[k_i - j] if valid."""
    n = len(a)
    n_pk = len(peak_indices)
    A = np.zeros((n_pk, n))
    for i in range(n_pk):
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        if je > js:
            A[i, js:je] = a[k - (je - 1): k - js + 1][::-1]
    return A


def _solve_lp(a, top_k, timeout_s=60):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M for top-k peaks, g>=0.
    Returns (g, M, conv) or (None, M, conv) on failure."""
    n = len(a)
    conv = _fft_conv(a, a)
    M = float(np.max(conv))
    if M < 1e-15:
        return None, M, conv

    peak_idx = np.argsort(conv)[-top_k:]
    A = _build_A(a, peak_idx)
    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0.0, None)] * n

    start = time.monotonic()
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout_s, 'presolve': True})
    elapsed = time.monotonic() - start

    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        return g, M, conv
    elif res.x is not None:
        g = np.maximum(res.x, 0.0)
        if np.all(A @ g <= b_ub * 1.001) and float(np.sum(g)) > 1e-10:
            return g, M, conv
    return None, M, conv


def _lp_descent_step(a, top_k=200, t=0.3, timeout_s=60):
    """One LP-based descent step with blending."""
    n = len(a)
    a_sum = float(np.sum(a))
    g, M, conv = _solve_lp(a, top_k, timeout_s=timeout_s)

    if g is None:
        return a, M

    g_sum = float(np.sum(g))
    if g_sum < 1e-15:
        return a, M

    g = g * (a_sum / g_sum)
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, a_sum)
    return a_new, M


def _subgrad_step(a, lr, top_k=10):
    """Subgradient descent step for max(conv) with fixed sum."""
    n = len(a)
    conv = _fft_conv(a, a)
    sum_a = float(np.sum(a))
    peak_idx = np.argsort(conv)[-top_k:]

    grad = np.zeros(n)
    for k in peak_idx:
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n, ki + 1)
        if je > js:
            grad[js:je] += a[ki - (je - 1): ki - js + 1][::-1]
    grad /= len(peak_idx)

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, sum_a)
    return a_new


def _run_lp_loop(a, deadline, top_k=200, t_init=0.35, t_decay=0.9992,
                 subgrad_every=4, timeout_s=50, perturb_thresh=12,
                 eval_every=3):
    """Main LP descent loop with periodic subgradient and restarts."""
    n = len(a)
    a_sum = float(np.sum(a))
    best = a.copy()
    best_score = _score_fast(a)
    t = t_init
    stuck = 0
    it = 0

    while time.monotonic() < deadline - 5:
        t = max(t * t_decay, 0.02)
        a, _ = _lp_descent_step(a, top_k=top_k, t=t, timeout_s=timeout_s)
        it += 1

            for _ in range(15):
                if time.monotonic() > deadline - 5:
                    break
                a = _subgrad_step(a, lr=0.008, top_k=7)

        if it % eval_every == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stuck = 0
            else:
                stuck += 1

            if it % (eval_every * 10) == 0:
                print(f"    it={it}, score={best_score:.7f}, t={t:.5f}, stuck={stuck}")

                a = np.maximum(a, 0.0)
