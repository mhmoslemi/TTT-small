Minimize 2n*max(a*a)/(sum(a))^2 using FISTA with adaptive top-k peak weighting.
Uses FFT for fast convolution, multiple initializations, and an annealing
schedule on the peak-weighting sharpness parameter.

import numpy as np
import time


def convolve_fast(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho_arr > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = convolve_fast(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def compute_gradient(a, c, top_k, sharpness):
    """
    Compute the gradient direction using top-k peaks with exponential weighting.
    
    Parameters:
    - a: current point (probability vector)
    - c: convolution at current point
    - top_k: number of top peaks to use
    - sharpness: controls how much weight the top peaks get relative to lower ones
                 (higher = more concentrated on the very top)
    """
    n = len(a)
    m = min(top_k, len(c))
    top_idx = np.argpartition(c, -m)[-m:]
    
    # Exponential weighting based on how close each peak is to the max
    c_max = c.max()
    c_top = c[top_idx]
    if sharpness > 0:
        weights = np.exp(sharpness * (c_top - c_max))
    else:
        weights = np.ones(m)
    weights /= weights.sum()
    
    # Compute gradient: g_i = sum_l weights[l] * a[top_idx[l] - i]
    g = np.zeros(n, dtype=np.float64)
    for l in range(m):
        k = int(top_idx[l])
        w = weights[l]
        if w < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            idx = k - np.arange(lo, hi, dtype=np.int64)
            g[lo:hi] += w * a[idx]
    
    return g

    a = a0.copy()
    a_prev = a0.copy()

    c = convolve_fast(a)

    while time.monotonic() < deadline:

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))

        if sy < 1e-15:

                best_a = a.copy()
                stuck = 0
