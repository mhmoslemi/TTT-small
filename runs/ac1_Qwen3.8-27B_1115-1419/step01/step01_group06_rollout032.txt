import time

def next_power_of_2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def fft_self_conv(a, N):
    """Compute linear self-convolution a*a via FFT."""
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2*n-1]
    return np.maximum(c, 0.0)

def objective(a, N):
    """Compute 2*n*max(conv) / sum(a)^2."""
    n = len(a)
    c = fft_self_conv(a, N)
    s = a.sum()
    if s < 1e-300:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)

def normalized_objective(a, N):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    c = fft_self_conv(a, N)
    return 2.0 * n * c.max()

def softmax_weights(values, beta):
    """Compute softmax weights with temperature 1/beta."""
    v = np.asarray(values, dtype=np.float64)
    if v.size == 0:
        return np.array([1.0])
    v = np.where(np.isfinite(v), v, -1e300)
    v_max = v.max()
    if not np.isfinite(v_max) or v_max <= 0:
        idx = v.argmax()
        w = np.zeros(v.size)
        w[idx] = 1.0
        return w
    x = beta * (v - v_max) / v_max  # Scale relative to max
    x = np.clip(x, -745.0, 0.0)

    s = e.sum()
    if s < 1e-300:
        idx = v.argmax()
        w = np.zeros(v.size)
        w[idx] = 1.0
        return w
    return e / s

        FR = phase * np.conjugate(A)  # Frequency response for gradient
        c = fft_self_conv(a, N)
        score = c.max() * 2.0 * n

            best_score = score
            best_a = a.copy()
            stall = 0

        P = np.fft.rfft(w, N)
