from scipy.fft import fft, ifft

def _fft_conv(a, b):

    n = 1 << (n_out - 1).bit_length()

    """
    Solve LP to find best perturbation direction.
    
    We want to find δa such that:
    - sum(δa) = 0
    - |δa_i| <= eps for all i
    - a_i + δa_i >= 0
    - The predicted max peak is minimized
    
    The predicted c_k after perturbation: c_k + 2 * M_k · δa
    where M_k[i] = a_{k-i} if valid, 0 otherwise.

    subject to: 2 * M_k · δa - t <= -c_k  for k in top_k
                sum δa = 0
                -eps <= δa_i <= eps
                δa_i >= -a_i  (non-negativity)
    """
    n = len(a)
    K = len(top_k_indices)
    
    # Variables: [δa_0, ..., δa_{n-1}, t]
    # Total: n + 1 variables
    
    # Objective: minimize t -> c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each peak k: 2 * M_k · δa - t <= -c_k
    # i.e., 2*M_k[0], 2*M_k[1], ..., 2*M_k[n-1], -1 | x | <= -c_k
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for idx, k in enumerate(top_k_indices):
        # M_k[i] = a[k-i] if 0 <= k-i < n
        m = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:

        A_ub[idx, :n] = 2.0 * m
        A_ub[idx, n] = -1.0
        b_ub[idx] = -c[k]

    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Actually, we want δa_i >= -a_i (for non-negativity) and |δa_i| <= eps

    lower = np.maximum(-eps, -a)
    upper = np.full(n, eps)
    bounds = list(zip(lower, upper)) + [(-np.inf, np.inf)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
