def self_conv_and_score(a, N, n=None):

def make_fourier_basis(n, d, rng):
    d = max(1, min(d, max(1, n//2)))
    B = np.zeros((n, d), dtype=np.float64)
    for j in range(d):
        freq = j + 1
        phase = rng.uniform(0.0, 2.0*np.pi)
        idx = np.arange(n)
        B[:, j] = np.cos(2.0*np.pi*freq*idx/float(n) + phase)
    B -= np.mean(B, axis=0, keepdims=True)
    try:
        Q,R = np.linalg.qr(B)
        if Q.shape[1] >= d:
            B = Q[:, :d]
        else:
            B = Q
    except Exception:
        norms = np.linalg.norm(B, axis=0)
        B = np.divide(B, norms, out=np.zeros_like(B), where=norms>1e-12)
    return B

def lowdim_objective(x, B, N, beta, n):

"""
Hybrid optimizer for nonnegative step-function coefficient sequences.

The objective is scale-invariant, so internally sequences are normalized to
sum 1 and the score is 2*n*max(np.convolve(a,a)).  The search combines:
  * correct FFT-based gradients for the squared-polynomial peak,
  * alternating convex subproblem updates min_b max(a*b),
  * adaptive learning-rate line search in log-softmax space,
  * low-dimensional Fourier/BFGS polishing for diverse starts,
  * occasional resizing to larger lengths,
  * exact validation through the provided evaluate_sequence oracle when available.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize as scipy_minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

EPS = 1e-300
_INF = float("inf")

_ORACLE_BEST_SCORE = _INF
_ORACLE_BEST_SEQ = None
_ORACLE_TIME = 0.0
_ORACLE_CALLS = 0


def next_fast_len(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    """Finite nonnegative vector normalized to sum 1."""
    arr = np.asarray(v, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def softmax_stable(w):
    """Stable softmax over a 1D array."""
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size == 0:
        return np.ones(1, dtype=np.float64)
    w = np.where(np.isfinite(w), w, -1e300)
    w = w - float(np.max(w))
    w = np.clip(w, -745.0, 0.0)
    e = np.exp(w)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    return e / s


def log_weights(a):
    """Finite log weights for a nonnegative vector."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, EPS)
    w = np.log(a)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= float(np.max(w))
    return w


def make_weights(vals, beta):
    """Smoothly peaked weights concentrated near the maximum of vals."""
    vals = np.asarray(vals, dtype=np.float64).ravel()
    L = vals.size
    if L == 0:
        return np.ones(1, dtype=np.float64)
    mx = float(np.max(vals))
    if not np.isfinite(mx) or mx <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    x = (vals - mx) / mx
    x = np.where(np.isfinite(x), x, -1e300)
    if beta <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    y = np.exp(np.clip(beta * x, -745.0, 0.0))
    sy = float(np.sum(y))
    if not np.isfinite(sy) or sy <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(vals))] = 1.0
        return p
    p = y / sy
    p = p + 1e-15
    p = p / float(np.sum(p))
    return p


def self_conv_and_score(a, N, n=None):
    """Exact linear self-convolution by FFT and scale-invariant score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if n is None:
        n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64), _INF
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(2 * n - 1, dtype=np.float64), _INF
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2 * n - 1]
    if c.size:
        np.maximum(c, 0.0, out=c)
        mx = float(np.max(c))
    else:
        mx = 0.0
    if not np.isfinite(mx):
        return c, _INF
    val = float(2.0 * n * mx / (s * s))
    return c, (val if np.isfinite(val) else _INF)


def fft_score_only(a, N, n=None):
    """Convenience wrapper returning only FFT score."""
    _, sc = self_conv_and_score(a, N, n)
    return sc


def gradient_corr(a, q, N, n, factor=1.0):
    """
    For c_k = sum_i a_i q_{i+k}, gradient wrt a is factor * that sum.
    This is computed by A * conj(Q) / inverse FFT without shift.
    """
    A = np.fft.rfft(a, N)
    Q = np.fft.rfft(q, N)
    g = np.fft.irfft(A * np.conjugate(Q), N)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    return factor * g


def log_gradient_from_euclidean(g, a):
    """Convert Euclidean gradient wrt a to gradient wrt log-softmax parameters."""
    return a * (g - float(np.dot(a, g)))


def adam_log_step(w, grad_w, m, v, t, lr):
    """One Adam step in log-softmax space."""
    gnorm = float(np.dot(grad_w, grad_w))
    if not np.isfinite(gnorm) or gnorm <= 1e-30:
        return softmax_stable(w), w, m, v, t
    t += 1
    beta1 = 0.9
    beta2 = 0.999
    m = beta1 * m + (1.0 - beta1) * grad_w
    v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
    mhat = m / (1.0 - beta1 ** t)
    vhat = v / (1.0 - beta2 ** t)
    dw = -lr * mhat / (np.sqrt(vhat) + 1e-8)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -25.0, 25.0)
    w = w + dw
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - float(np.max(w))
    a = softmax_stable(w)
    return a, w, m, v, t


def simple_log_candidate(w, grad_w, lr):
    """One multiplicative / log-softmax step."""
    w_try = w - lr * grad_w
    w_try = np.where(np.isfinite(w_try), w_try, 0.0)
    w_try = w_try - float(np.max(w_try))
    w_try = np.clip(w_try, -700.0, 0.0)
    a_try = softmax_stable(w_try)
    return a_try, w_try


def add_top(top, a, score, cap=12):
    """Maintain a small pool of best candidates."""
    if not np.isfinite(score):
        return top
    a = np.asarray(a, dtype=np.float64).copy()
    score = float(score)
    if len(top) < cap:
        top.append((score, a))
        return top
    worst_idx = 0
    worst_sc = float(top[0][0])
    for i in range(1, len(top)):
        sc = float(top[i][0])
        if sc > worst_sc:
            worst_sc = sc
            worst_idx = i
    if score < worst_sc:
        top[worst_idx] = (score, a)
    return top


def get_top_min(top):
    """Return (sequence, score) with smallest score in top list."""
    if not top:
        return None, _INF
    best_a = None
    best_sc = _INF
    for sc, a in top:
        sc = float(sc)
        if sc < best_sc:
            best_sc = sc
            best_a = a
    return best_a, best_sc


def resize_sequence(a, m):
    """Resample a nonnegative sequence to length m using linear interpolation."""
    a = normalize_nonneg(a)
    n = a.size
    m = int(max(1, min(m, 30000)))
    if m == n:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=np.float64) / float(m)
    x_old = np.linspace(0.0, 1.0, n, endpoint=True)
    x_new = np.linspace(0.0, 1.0, m, endpoint=True)
    y = np.interp(x_new, x_old, a)
    return normalize_nonneg(y)

def finalize_sequence(a):
    """Return a valid Python list with finite nonnegative floats, sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(a.size, dtype=np.float64) / float(a.size)
    else:
        a = a / s
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(np.sum(a))
    if s2 < 0.01:
        a = a * (1.0 / max(s2, EPS))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]

    if _ORACLE_TIME > 320.0 or _ORACLE_CALLS > 130:

        _ORACLE_BEST_SCORE = float(sc)

def lowdim_objective(x, B, N, beta, n):
    """Smoothed self-convolution objective for a linear Fourier-parametrized sequence."""
    x = np.asarray(x, dtype=np.float64).ravel()
    w = B @ x
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - float(np.max(w))
    w = np.clip(w, -700.0, 0.0)
    e = np.exp(w)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        return 1e100, np.zeros_like(x)
    a = e / se

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    if c.size == 0:
        return 0.0, np.zeros_like(x)
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx <= 0.0:
        return 0.0, np.zeros_like(x)

    if beta <= 0.0:
        f = mx
        q = np.zeros(c.size, dtype=np.float64)
        q[int(np.argmax(c))] = 1.0
    else:
        y = np.exp(np.clip(beta * (c - mx), -745.0, 0.0))
        sy = float(np.sum(y))
        if not np.isfinite(sy) or sy <= 0.0:
            return mx, np.zeros_like(x)
        q = y / sy
        f = mx + math.log(sy) / beta

    Q = np.fft.rfft(q, N)
    ga = 2.0 * np.fft.irfft(A * np.conjugate(Q), N)[:n]
    ga = np.where(np.isfinite(ga), ga, 0.0)
    gw = log_gradient_from_euclidean(ga, a)
    grad = B.T @ gw
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(f), grad


def make_fourier_basis(n, d, rng):
    """Orthonormal random Fourier basis with zero mean."""
    d = int(max(1, min(d, max(1, n // 2))))
    B = np.zeros((n, d), dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    base = 2.0 * np.pi * idx / float(n)
    for j in range(d):
        freq = float(j + 1)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        B[:, j] = np.cos(freq * base + phase)
    B -= np.mean(B, axis=0, keepdims=True)
    try:
        Q, R = np.linalg.qr(B)
        if Q.shape[1] >= d:
            B = Q[:, :d]
        else:
            B = Q
    except Exception:
        norms = np.linalg.norm(B, axis=0)
        B = np.divide(B, norms, out=np.zeros_like(B), where=norms[:, None] > 1e-12)
    return B


def lowdim_search(a0, deadline, seed=0, max_seconds=80.0):
    """Run a few BFGS passes in low-dimensional Fourier models."""
    start = time.monotonic()
    a = normalize_nonneg(a0)
    n = a.size
    N = next_fast_len(3 * n - 2)
    best = a.copy()
    best_score = fft_score_only(a, N, n)

    if not HAVE_SCIPY:
        return best, best_score

    rng = np.random.default_rng(seed)
    d = int(min(72, max(16, n // 70)))
    loga = log_weights(a)
    center = loga - float(np.mean(loga))

    betas = (30.0, 80.0)
    bases_done = 0
    max_bases = 5

    while bases_done < max_bases:
        if time.monotonic() > start + max_seconds or time.monotonic() > deadline - 90.0:
            break
        B = make_fourier_basis(n, d, rng)
        if bases_done == 0:
            x0 = B.T @ center
        else:
            x0 = rng.normal(0.0, 0.9, size=d)
        x0 = np.where(np.isfinite(x0), x0, 0.0)
        x0 = np.clip(x0, -6.0, 6.0)

        for beta in betas:
            if time.monotonic() > start + max_seconds or time.monotonic() > deadline - 90.0:
                break
            try:
                res = scipy_minimize(
                    lowdim_objective,
                    x0,
                    args=(B, N, beta, n),
                    jac=True,
                    method="L-BFGS-B",
                    bounds=[(-8.0, 8.0)] * d,
                    options={"maxiter": 110, "gtol": 1e-7, "maxls": 20},
                )
                x1 = res.x
            except Exception:
                x1 = x0
            if x1 is None or len(x1) == 0:
                x1 = x0
            w1 = B @ x1
            a1 = softmax_stable(w1)
            sc1 = fft_score_only(a1, N, n)
            if np.isfinite(sc1) and sc1 < best_score:
                best = a1
                best_score = sc1
        bases_done += 1

    return best, best_score


def full_optimize(a0, deadline, seed=0):
    """
    Main high-dimensional optimizer.

    Alternates:
      - optimize b to reduce max(a*b) for fixed a (convex subproblem),
      - mix b into a,
      - line-search log-softmax self-convolution gradient step.
    """
    start = time.monotonic()
    a = normalize_nonneg(a0)
    n = a.size
    if n < 64:
        a = np.ones(2048, dtype=np.float64) / 2048.0
        n = a.size
    if n > 20000:
        a = resize_sequence(a, 16384)
        n = a.size

    N = next_fast_len(3 * n - 2)
    _, score = self_conv_and_score(a, N, n)

    best = a.copy()
    best_score = score
    top = [(score, a.copy())]
    w = log_weights(a)
    rng = np.random.default_rng(seed)

    if score > 1.95:
        for _ in range(24):
            sig = float(rng.choice((0.05, 0.10, 0.20, 0.50)))
            wt = log_weights(a) + rng.normal(0.0, sig, size=n)
            at = softmax_stable(wt)
            sc = fft_score_only(at, N, n)
            if np.isfinite(sc) and sc < score:
                a = at
                score = sc
                w = log_weights(a)
        best = a.copy()
        best_score = score
        top = [(score, a.copy())]

    b = a.copy()
    wb = log_weights(b)
    mb = np.zeros(n, dtype=np.float64)
    vb = np.zeros(n, dtype=np.float64)
    tb = 0

    lr_self = 0.14
    lr_cross = 0.11
    beta = 10.0
    beta_cross = 12.0
    stall = 0
    outer = 0

    while time.monotonic() < deadline - 12.0:
        outer += 1

        if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.isfinite(score)):
            a = best.copy()
            score = best_score
            w = log_weights(a)
            b = a.copy()
            wb = log_weights(b)
            mb = np.zeros(n, dtype=np.float64)
            vb = np.zeros(n, dtype=np.float64)
            tb = 0
            stall = 0
            continue

        # --- Convex subproblem: vary b to reduce max(a*b). ---
        A = np.fft.rfft(a, N)
        inner = 3 + (outer % 8)
        if stall > 30:
            inner = 6
        for _ in range(inner):
            B = np.fft.rfft(b, N)
            h = np.fft.irfft(A * B, N)[:2 * n - 1]
            np.maximum(h, 0.0, out=h)
            q = make_weights(h, beta_cross)
            Q = np.fft.rfft(q, N)
            g = np.fft.irfft(A * np.conjugate(Q), N)[:n]
            g = np.where(np.isfinite(g), g, 0.0)
            gw = log_gradient_from_euclidean(g, b)
            b, wb, mb, vb, tb = adam_log_step(wb, gw, mb, vb, tb, lr_cross)

        # --- Update a from the optimized b. ---
        best_mix = None
        best_mix_score = score
        for tval in (1.0, 0.65, 0.35, 0.15):
            cand = (1.0 - tval) * a + tval * b
            cand = normalize_nonneg(cand)
            sc = fft_score_only(cand, N, n)
            if np.isfinite(sc) and sc < best_mix_score - 1e-12:
                best_mix_score = sc
                best_mix = cand

        if best_mix is not None:
            a = best_mix
            score = best_mix_score
            w = log_weights(a)
            b = a.copy()
            wb = log_weights(b)
            mb = np.zeros(n, dtype=np.float64)
            vb = np.zeros(n, dtype=np.float64)
            tb = 0
            stall = 0
            lr_self = min(0.9, lr_self * 1.10)
            lr_cross = min(0.35, lr_cross * 1.05)
            if score < best_score:
                best_score = score
                best = a.copy()
                top = add_top(top, best, score)
        else:
            stall += 1
            lr_cross = max(0.02, lr_cross * 0.95)
            if stall % 7 == 0:
                b = a.copy()
                wb = log_weights(b)
                mb = np.zeros(n, dtype=np.float64)
                vb = np.zeros(n, dtype=np.float64)
                tb = 0

        # --- Direct smoothed-max self descent. ---
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        if c.size:
            sc_curr = float(2.0 * n * np.max(c))
        else:
            sc_curr = _INF
        if np.isfinite(sc_curr) and sc_curr < score:
            score = sc_curr

        q = make_weights(c, beta)
        Q = np.fft.rfft(q, N)
        g = 2.0 * np.fft.irfft(A * np.conjugate(Q), N)[:n]
        g = np.where(np.isfinite(g), g, 0.0)
        gw = log_gradient_from_euclidean(g, a)

        best_try = None
        best_try_score = score
        best_mult = 1.0
        best_w_try = w
        for mult in (2.0, 1.0, 0.5):
            lr = lr_self * mult
            at, wt = simple_log_candidate(w, gw, lr)
            sc = fft_score_only(at, N, n)
            if np.isfinite(sc) and sc < best_try_score - 1e-12:
                best_try_score = sc
                best_try = at
                best_w_try = wt
                best_mult = mult

        if best_try is not None:
            a = best_try
            w = best_w_try
            score = best_try_score
            stall = 0
            if best_mult > 1.0:
                lr_self = min(1.2, lr_self * 1.15)
            elif best_mult < 0.75:
                lr_self = max(0.02, lr_self * 0.85)
            else:
                lr_self = min(1.2, lr_self * 1.05)
            if score < best_score:
                best_score = score
                best = a.copy()
                top = add_top(top, best, score)
        else:
            stall += 1
            lr_self = max(0.02, lr_self * 0.85)

        # --- Sharpen peak weights over time. ---
        if outer % 20 == 0:
            beta = min(180.0, beta + 2.0 + 0.05 * beta)
            beta_cross = min(300.0, beta_cross + 3.0 + 0.05 * beta_cross)

        # --- Restart / exploration. ---
        if stall > 35:
            r = rng.random()
            if r < 0.4:
                a = best.copy()
                score = best_score
            elif r < 0.8:
                a_min, sc_min = get_top_min(top)
                if a_min is not None:
                    a = a_min.copy()
                    score = float(sc_min)
                else:
                    a = best.copy()
                    score = best_score
            else:
                sig = float(rng.choice((0.05, 0.10, 0.20, 0.40)))
                wt = log_weights(best) + rng.normal(0.0, sig, size=n)
                at = softmax_stable(wt)
                sc = fft_score_only(at, N, n)
                if np.isfinite(sc) and sc < best_score + 0.08:
                    a = at
                    score = sc
                else:
                    a = best.copy()
                    score = best_score
            w = log_weights(a)
            b = a.copy()
            wb = log_weights(b)
            mb = np.zeros(n, dtype=np.float64)
            vb = np.zeros(n, dtype=np.float64)
            tb = 0
            beta = 25.0
            beta_cross = 30.0
            lr_self = 0.14
            lr_cross = 0.11
            stall = 0

    return best, float(best_score) if np.isfinite(best_score) else 2.0


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point required by the harness.

    Returns the best finite nonnegative coefficient sequence found before
    the 2400s deadline.
    """
    global _ORACLE_BEST_SCORE, _ORACLE_BEST_SEQ, _ORACLE_TIME, _ORACLE_CALLS

    _ORACLE_BEST_SCORE = _INF
    _ORACLE_BEST_SEQ = None
    _ORACLE_TIME = 0.0
    _ORACLE_CALLS = 0

    total_start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    cutoff = total_start + min(max(budget, 60.0), 2390.0)

    parent = load_parent()
    if parent is None:
        parent = np.ones(4096, dtype=np.float64) / 4096.0
    parent = normalize_nonneg(parent)
    if parent.size > 20000:
        parent = resize_sequence(parent, 16384)

    Np = next_fast_len(3 * parent.size - 2)
    parent_score = fft_score_only(parent, Np, parent.size)

    # Low-dimensional Fourier polish can find useful diverse starts early.
    lowdim_best = parent.copy()
    lowdim_score = parent_score
    if HAVE_SCIPY:
        lowdim_budget = min(80.0, 0.12 * max(1.0, cutoff - total_start))
        if lowdim_budget > 25.0:
            a_low, sc_low = lowdim_search(parent, total_start + lowdim_budget, seed + 7, lowdim_budget - 5.0)
            Nlow = next_fast_len(3 * a_low.size - 2)
            if np.isfinite(sc_low) and sc_low < lowdim_score:
                lowdim_best = a_low
                lowdim_score = sc_low

    if lowdim_score < parent_score:
        main_start = lowdim_best
    else:
        main_start = parent

    # Main high-dimensional optimization.
    deadline_main = total_start + 0.64 * max(1.0, cutoff - total_start)
    a_main, sc_main = full_optimize(main_start, deadline_main, seed)
    validate_oracle(a_main, cutoff, allow_large=False)

    a_final = np.asarray(a_main, dtype=np.float64).copy()
    sc_final = float(sc_main)

    # Try a larger construction if enough time remains.
    time_left = cutoff - time.monotonic()
    if time_left > 260.0:
        n_main = int(a_final.size)
        target = 0
        if n_main < 4096:
            target = 8192
        elif n_main < 8192:
            target = min(2 * n_main, 16384)
        elif n_main < 14000 and time_left > 520.0:
            target = 16384

        if target > n_main + 300:
            a_up = resize_sequence(a_final, target)
            Nup = next_fast_len(3 * target - 2)
            sc_up = fft_score_only(a_up, Nup, target)
            # Only spend time on upsized starts that look plausible.
            if np.isfinite(sc_up) and sc_up < sc_final + 0.25:
                deadline_up = cutoff - 40.0
                a_up2, sc_up2 = full_optimize(a_up, deadline_up, seed + 23)
                if np.isfinite(sc_up2) and sc_up2 < sc_final:
                    a_final = a_up2
                    sc_final = sc_up2
                validate_oracle(a_final, cutoff, allow_large=False)

    # Final exact oracle validation for the most promising candidates.
    candidates = (a_final, a_main, lowdim_best, parent)
    seen = set()
    for cand in candidates:
        if time.monotonic() > cutoff - 10.0:
            break
        cand = np.asarray(cand, dtype=np.float64).ravel()
        key = (cand.size, round(float(np.sum(cand)), 12))
        if key in seen:
            continue
        seen.add(key)
        validate_oracle(cand, cutoff, allow_large=True)
        if time.monotonic() > cutoff - 10.0:
            break

    # Prefer a sequence actually validated by the oracle when available.
    if _ORACLE_BEST_SEQ is not None and np.isfinite(_ORACLE_BEST_SCORE):
        return finalize_sequence(_ORACLE_BEST_SEQ)

    return finalize_sequence(a_final)
