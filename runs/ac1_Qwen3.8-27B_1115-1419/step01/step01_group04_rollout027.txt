"""

import numpy as np
import time

def normalize_a(a, target_sum):
    """Normalize a to have the given sum, keeping values in [0, 1000]."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    for _ in range(5):
        s = a.sum()
        if s < 1e-15:
            a.fill(target_sum / len(a))
            break
        a *= target_sum / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def fft_convolution(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def score(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_convolution(a)
    return 2.0 * n * c.max() / (s * s)


def gradient_topK(a, c, K):
    """Subgradient of max(c) using top K peaks."""
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(K, L))
    
    if K == 1:
        k = int(np.argmax(c))
        start = max(0, k - n + 1)
        stop = min(n, k + 1)
        g = np.zeros(n)
        g[start:stop] = a[k - stop + 1 : k - start + 1][::-1]
        return g
    
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    g = np.zeros(n)
    weights = c[idx] / c[idx].sum()  # Weight by magnitude
    for k, w in zip(idx, weights):
        k = int(k)
        start = max(0, k - n + 1)
        stop = min(n, k + 1)
        g[start:stop] += w * a[k - stop + 1 : k - start + 1][::-1]
    return g


def local_search_v2(a, budget_s, seed=0):

    Improved local search using:
    - Log-domain multiplicative updates
    - Adaptive top-K with magnitude weighting
    - Cosine annealing for step size
    - Backtracking acceptance
    """
    a = normalize_a(a.copy(), float(len(a)))
    n = len(a)
    rng = np.random.default_rng(seed)
    
    t0 = time.monotonic()
    c = fft_convolution(a)
    sc = 2.0 * n * c.max() / (a.sum() ** 2)
    best = a.copy()
    best_score = sc
    
    # Parameters
    eta = 0.4  # Initial step size
    eta_min = 0.01
    beta = 0.5  # Momentum
    noise_scale = 0.05
    accept_margin = 0.02
    
    mom = np.zeros(n)
    stagnant = 0
    it = 0
    total_iters = 0
    
    while time.monotonic() - t0 < budget_s:
        it += 1
        total_iters += 1
        
        # Adaptive K
        progress = min(1.0, it / 5000.0)
        K_frac = 0.15 * (1 - progress) + 0.02 * progress
        K = max(1, min(int((2*n-1) * K_frac), 256, n // 2))
        
        # Compute gradient
        g = gradient_topK(a, c, K)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Momentum
        mom = beta * mom + (1 - beta) * g
        
        # Cosine annealing for step size
        cos_phase = 0.5 * (1 + np.cos(np.pi * progress))
        current_eta = eta * cos_phase + eta_min
        
        # Log-domain update with noise
        logf = -current_eta * mom
        if noise_scale > 0.001:
            logf += rng.normal(0, noise_scale * cos_phase, n)
        logf = np.clip(logf, -2.0, 2.0)

        cand = normalize_a(cand, float(n))
        
        c2 = fft_convolution(cand)
        cand_score = 2.0 * n * c2.max() / (cand.sum() ** 2)

            eta *= 0.8
            continue
        
        # Acceptance criterion
        if cand_score <= sc * (1 + accept_margin):
            a = cand
            c = c2
            sc = cand_score
            mom *= 0.9
            
            if cand_score < best_score:
                best = a.copy()
                best_score = cand_score
                stagnant = 0
                # Decrease noise slowly
                noise_scale *= 0.999
                accept_margin = max(0.001, accept_margin * 0.999)
            else:
                stagnant += 1
        else:
            stagnant += 1
            eta = max(eta_min, eta * 0.95)
            mom *= 0.5
        
        # Restart if stagnant
        if stagnant > 500:
            # Perturb from best
            pert = best * np.exp(rng.normal(0, 0.2, n))
            pert = normalize_a(pert, float(n))
            c_p = fft_convolution(pert)
            sc_p = 2.0 * n * c_p.max() / (pert.sum() ** 2)

                a = pert
                c = c_p
                sc = sc_p
            else:
                a = best.copy()
                c = fft_convolution(a)
                sc = best_score
            mom = np.zeros(n)
            stagnant = 0
            eta = 0.3
            accept_margin = 0.05
            noise_scale = 0.08

def upsample(seq, n):
    """Resample sequence to length n."""
    seq = np.asarray(seq, dtype=np.float64)
    m = len(seq)
    if m == n:
        return normalize_a(seq, float(n))
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, seq)
    return normalize_a(out, float(n))


def make_structured_start(n, rng):
    """Create a structured starting sequence."""
    choice = rng.integers(0, 5)
    
    if choice == 0:
        # Two-block
        m = int(n * rng.uniform(0.2, 0.5))
        ratio = rng.uniform(0.5, 3.0)
        a = np.empty(n)
        a[:m] = ratio
        a[m:] = 1.0
    elif choice == 1:
        # Cosine taper
        t = np.linspace(0, np.pi, n)
        a = 1.0 + 0.5 * np.cos(t)
    elif choice == 2:
        # Random smooth
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            freq = rng.integers(1, 5)
            amp = rng.uniform(-0.5, 0.5)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        # Block structure
        nblocks = rng.integers(4, 12)
        edges = np.sort(rng.integers(1, n, nblocks - 1))
        edges = np.concatenate([[0], edges, [n]])
        vals = rng.uniform(0.3, 2.0, nblocks)
        a = np.repeat(vals, np.diff(edges))
        a = a[:n]
    else:
        # Exponential decay + noise
        t = np.linspace(0, 1, n)
        a = np.exp(-3 * t) + 0.3 + rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.05)
    
    return normalize_a(a, float(n))

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative step functions.
Uses log-space subgradient with Armijo line search, adaptive peak weighting,

def _normalize(a, target_sum):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    for _ in range(5):
        s = a.sum()
        if s < 1e-15:
            a.fill(target_sum / max(1, len(a)))
            return a
        a *= target_sum / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def _fft_conv(a):
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_weighted(a, c, K=16):
    """Subgradient weighted by peak magnitude (softmax-like)."""
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(K, L))
    
    if K == 1:
        k = int(np.argmax(c))
        start = max(0, k - n + 1)
        stop = min(n, k + 1)
        g = np.zeros(n)
        g[start:stop] = a[k - stop + 1:k - start + 1][::-1]
        return g
    
    if K >= L:
        idx = np.arange(L)
        vals = c.copy()
    else:
        idx = np.argpartition(c, -K)[-K:]
        idx = idx[np.argsort(c[idx])[::-1]]
        vals = c[idx].copy()
    
    # Exponential weighting (softer than uniform)
    max_val = vals.max()
    if max_val > 0:
        weights = np.exp((vals - max_val) / (max_val * 0.1 + 1e-10))
        weights /= weights.sum()
    else:
        weights = np.ones(K) / K
    
    g = np.zeros(n)
    for k, w in zip(idx, weights):
        k = int(k)
        start = max(0, k - n + 1)
        stop = min(n, k + 1)
        g[start:stop] += w * a[k - stop + 1:k - start + 1][::-1]
    return g

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        g_centered = g - g.mean()

                a = cand
                c = c2
                sc = cand_sc

def _upsample(seq, n):
    seq = np.asarray(seq, dtype=np.float64)
    m = len(seq)
    if m == n:
        return _normalize(seq.copy(), float(n))
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, seq)
    return _normalize(out, float(n))


def _structured_start(n, rng):

    t = np.linspace(0, 1, n)

        edges = np.sort(rng.choice(np.arange(1, n), nb-1, replace=False))

        a = rng.uniform(0.1, 2.0, n)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative step functions.
Uses log-space subgradient with exponential peak weighting, Nesterov momentum,
cosine-annealed step size, and diverse mutation-based restarts.
Multi-track parallel search with periodic best-sharing.
"""

import numpy as np
import time
import math


def _normalize(a, target_sum):
    """Normalize array to target sum, clip to [0, 1000]."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    for _ in range(5):
        s = a.sum()
        if s < 1e-15:
            a.fill(target_sum / max(1, len(a)))
            return a
        a *= target_sum / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def _fft_conv(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_exp_weighted(a, c, K, temperature=0.1):
    """
    Subgradient of max(c) using exponential weighting over top-K peaks.
    Weight is proportional to exp(c_k / temperature_range).
    """
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(K, L))

    if K == 1:
        k = int(np.argmax(c))
        start = max(0, k - n + 1)
        stop = min(n, k + 1)
        g = np.zeros(n)
        g[start:stop] = a[k - stop + 1:k - start + 1][::-1]
        return g

    if K >= L:
        idx = np.arange(L)
        vals = c
    else:
        idx = np.argpartition(c, -K)[-K:]
        vals = c[idx]

    # Exponential weighting: peaks that are closer to max get more weight
    max_val = vals.max()
    min_val = vals.min()
    if max_val > min_val + 1e-15:
        scaled = (vals - min_val) / (max_val - min_val)
        weights = np.exp(scaled / temperature)
        weights /= weights.sum()
    else:
        weights = np.ones(K) / K

    g = np.zeros(n)
    for k, w in zip(idx, weights):
        k = int(k)
        if w < 1e-10:
            continue
        start = max(0, k - n + 1)
        stop = min(n, k + 1)
        g[start:stop] += w * a[k - stop + 1:k - start + 1][::-1]
    return g


def _local_search(a, budget_s, seed=0, verbose=False):
    """
    Log-space subgradient with Nesterov momentum, adaptive K,
    cosine-annealed step, and diverse restarts.
    """
    a = _normalize(a.copy(), float(len(a)))
    n = len(a)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    c = _fft_conv(a)
    sc = float(2.0 * n * c.max() / (a.sum() ** 2))
    best = a.copy()
    best_score = sc

    # Nesterov parameters
    beta = 0.7  # momentum
    eta = 0.6   # base step size
    eta_min = 0.02
    noise_init = 0.04
    accept_tol = 0.015
    temp = 0.15  # temperature for exponential weighting

    mom = np.zeros(n)
    x_prev = a.copy()
    stagnant = 0
    it = 0
    total_iters = 0
    restart_count = 0

    while time.monotonic() - t0 < budget_s:
        it += 1
        total_iters += 1
        progress = min(1.0, it / 15000.0)

        # Cosine annealing for step size
        cos_val = 0.5 * (1.0 + math.cos(math.pi * progress))
        current_eta = eta * cos_val + eta_min

        # Temperature annealing
        current_temp = temp * (1.0 - 0.5 * progress)

        # Adaptive K
        K = max(2, int(48 * (1.0 - progress * 0.7)))
        K = min(K, max(2, n // 4))

        # Nesterov extrapolation
        y = a + beta * (a - x_prev)
        y = np.maximum(y, 0.0)
        y = _normalize(y, float(n))

        # Gradient at y
        c_y = _fft_conv(y)
        g = _grad_exp_weighted(y, c_y, K, current_temp)

        # Center to preserve sum approximately
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            a = best.copy()
            break

        # Nesterov step
        x_prev = a.copy()
        noise_scale = noise_init * (1.0 - progress * 0.8)
        logf = -current_eta * (g / g_norm)
        if noise_scale > 0.001:
            logf += rng.normal(0, noise_scale, n)
        logf = np.clip(logf, -2.5, 2.5)

        cand = y * np.exp(logf)
        cand = np.maximum(cand, 0.0)
        cand = _normalize(cand, float(n))

        # Adaptively try smaller steps if needed
        accepted = False
        for shrink in range(4):
            c2 = _fft_conv(cand)
            cand_sc = float(2.0 * n * c2.max() / (cand.sum() ** 2))

            if not np.isfinite(cand_sc):
                break

            if cand_sc <= sc * (1.0 + accept_tol):
                a = cand
                c = c2
                sc = cand_sc
                accepted = True
                break
            else:
                # Try smaller step
                cand = a * np.exp(-0.5 * current_eta * (g / g_norm))
                cand = np.maximum(cand, 0.0)
                cand = _normalize(cand, float(n))

        if not accepted:
            a = best.copy()
            c = _fft_conv(a)
            sc = best_score
            stagnant += 1
            mom *= 0.5
            continue

        # Update best
        if sc < best_score:
            best = a.copy()
            best_score = sc
            stagnant = 0
            # Slowly tighten
            accept_tol = max(0.001, accept_tol * 0.9995)
            noise_init = max(0.002, noise_init * 0.9998)
        else:
            stagnant += 1

        # Diverse restart
        if stagnant > 200:
            restart_count += 1
            r = rng.random()

            if r < 0.35:
                # Smooth Gaussian perturbation
                t = np.linspace(0, 1, n)
                z = np.zeros(n)
                for m in range(1, 4):
                    amp = 0.4 / m
                    ph = rng.uniform(0, 2 * np.pi)
                    z += amp * np.cos(2 * np.pi * m * t + ph)
                pert = best * np.exp(z)
            elif r < 0.60:
                # Block mutations at peak locations
                pert = best.copy()
                # Find where autocorrelation has high peaks
                c_b = _fft_conv(pert)
                top_peaks = np.argpartition(c_b, -min(8, len(c_b)))[-min(8, len(c_b)):]
                n_mut = int(rng.integers(1, 4))
                for _ in range(n_mut):
                    # Pick a peak location, mutate nearby region
                    peak_k = int(top_peaks[rng.integers(0, len(top_peaks))])
                    center = peak_k // 2
                    width = int(rng.integers(max(2, n // 200), n // 15))
                    start = max(0, center - width // 2)
                    end = min(n, start + width)
                    pert[start:end] *= np.exp(rng.normal(0, 0.6))
            elif r < 0.80:
                # Region rescaling
                pert = best.copy()
                nsplit = int(rng.integers(2, 5))
                splits = np.sort(rng.integers(1, n, max(1, nsplit - 1)))
                splits = np.concatenate([[0], splits, [n]])
                for i in range(len(splits) - 1):
                    pert[splits[i]:splits[i + 1]] *= rng.uniform(0.4, 1.8)
            else:
                # Copy a random portion from a new structured sequence
                pert = best.copy()
                new_struct = _structured_start(n, rng)
                frac = rng.uniform(0.1, 0.4)
                start = int(rng.integers(0, max(1, n - int(frac * n))))
                end = min(n, start + int(frac * n))
                pert[start:end] = new_struct[start:end]

            pert = _normalize(np.maximum(pert, 0.0), float(n))
            c_p = _fft_conv(pert)
            sc_p = float(2.0 * n * c_p.max() / (pert.sum() ** 2))

            if np.isfinite(sc_p) and sc_p < best_score * 1.15:
                a = pert
                c = c_p
                sc = sc_p
                x_prev = a.copy()
            else:
                a = best.copy()
                c = _fft_conv(a)
                sc = best_score
                x_prev = a.copy()

            stagnant = 0
            # Reset some params
            accept_tol = 0.03
            noise_init = max(noise_init, 0.04)
            beta = min(0.8, beta + 0.05)

        # Occasionally bump momentum
        if stagnant > 50 and stagnant % 50 == 0:
            beta = min(0.85, beta + 0.02)

        # Print progress
        if verbose and it % 2000 == 0:
            print(f"  it={it}, score={sc:.6f}, best={best_score:.6f}, K={K}, eta={current_eta:.4f}")

    return best, best_score


def _upsample(seq, n):
    """Resample sequence to target length."""
    seq = np.asarray(seq, dtype=np.float64)
    m = len(seq)
    if m == n:
        return _normalize(seq.copy(), float(n))
    if m == 0:
        return _normalize(np.ones(n), float(n))
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, seq)
    return _normalize(out, float(n))


def _structured_start(n, rng):
    """Generate a diverse structured initial sequence."""
    t = np.linspace(0, 1, n)
    choice = int(rng.integers(0, 8))

    if choice == 0:
        # Two-level with random split
        m = max(1, int(n * rng.uniform(0.15, 0.60)))
        ratio = rng.uniform(0.3, 5.0)
        a = np.empty(n)
        a[:m] = ratio
        a[m:] = 1.0
    elif choice == 1:
        # Cosine window
        alpha = rng.uniform(0.3, 2.5)
        a = 1.0 + alpha * np.cos(np.pi * (2 * t - 1))
        a = np.maximum(a, 0.05)
    elif choice == 2:
        # Multi-sine
        a = np.ones(n)
        ncomp = int(rng.integers(2, 7))
        for m in range(1, ncomp + 1):
            amp = rng.uniform(-0.7, 0.7) / (0.5 + 0.3 * m)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.sin(2 * np.pi * m * t + ph)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        # Piecewise constant blocks
        nb = int(rng.integers(3, 15))
        if nb >= n:
            nb = max(2, n // 2)
        edges = sorted(rng.choice(np.arange(1, n), min(nb - 1, n - 1), replace=False).tolist())
        edges = [0] + edges + [n]
        vals = rng.uniform(0.15, 2.5, len(edges) - 1)
        a = np.repeat(vals, np.diff(edges))[:n]
    elif choice == 4:
        # Exponential decay with floor
        rate = rng.uniform(0.5, 6.0)
        floor = rng.uniform(0.05, 0.4)
        a = floor + (1.0 - floor) * np.exp(-rate * t)
    elif choice == 5:
        # Quadratic
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.5)
        a = 1.0 + rng.uniform(-1, 1) * ((t - center) / width) ** 2
        a = np.maximum(a, 0.05)
    elif choice == 6:
        # Smoothed random
        a = rng.uniform(0.2, 2.0, n)
        # Simple moving average smoothing
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
    else:
        # Alternating high-low
        period = int(rng.integers(2, max(3, n // 10)))
        a = np.tile([1.0, rng.uniform(0.2, 0.8)], n // 2 + 1)[:n]
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.05)

    return _normalize(a, float(n))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-track subgradient search with diverse initializations.
    Uses parent sequence as warm start if available.
    Returns list of non-negative floats minimizing the objective.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))

    if budget < 15:
        return [1.0] * 512

    # Choose final length
    if budget < 200:
        n_final = 1024
    elif budget < 600:
        n_final = 2048
    elif budget < 1500:
        n_final = 4096
    else:
        n_final = 6733

    # Reserve time for final validation and potential extra polish
    search_deadline = t0 + budget - 45.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = globals().get('height_sequence_1', None)
    if parent is not None and len(parent) > 0:
        try:
            parent = np.asarray(parent, dtype=np.float64)
            parent = np.where(np.isfinite(parent), parent, 0.0)
            parent = np.clip(parent, 0, 1000)
            if parent.sum() > 0.01:
                parent = _normalize(parent, float(len(parent)))
                sc = _score(parent)
                if np.isfinite(sc):
                    best_seq = parent.copy()
                    best_score = sc
                    print(f"Parent loaded: score={sc:.6f}, n={len(parent)}")
        except Exception as e:
            print(f"Parent load error: {e}")
            parent = None

    # === Phase 1: Small-scale warm start ===
    small_levels = [128, 256, 512, 1024, 2048]
    small_levels = [x for x in small_levels if x < n_final]

    if small_levels and (search_deadline - time.monotonic()) > 250:
        small_budget = min(200.0, 0.10 * (search_deadline - time.monotonic()))
        weights = np.array([1, 1.5, 2, 3, 4][:len(small_levels)], dtype=float)
        weights /= weights.sum()

        for idx, sn in enumerate(small_levels):
            b = small_budget * weights[idx]
            if b < 8 or time.monotonic() > search_deadline - 120:
                break

            # Build starts for this scale
            starts = []
            if best_seq is not None:
                starts.append(_upsample(best_seq, sn))
            for _ in range(5):
                starts.append(_structured_start(sn, rng))

            # Run local search on top 3 starts
            scored_starts = []
            for s in starts[:6]:
                sc_s = _score(s)
                if np.isfinite(sc_s):
                    scored_starts.append((sc_s, s))
            scored_starts.sort(key=lambda x: x[0])

            n_try = min(3, len(scored_starts))
            b_each = b / max(1, n_try)

            for i in range(n_try):
                if time.monotonic() > search_deadline - 80:
                    break
                seq, sc = _local_search(scored_starts[i][1], b_each,
                                        seed=int(seed + 5000 * idx + i))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq
                    print(f"Small n={sn}, start {i}: {sc:.6f}")

    # === Phase 2: Multi-track at final scale ===
    # Build diverse start pool
    final_starts = []
    if best_seq is not None:
        final_starts.append(_upsample(best_seq, n_final))

    # Multiple structured starts
    for i in range(12):
        final_starts.append(_structured_start(n_final, rng, ))

    # Uniform
    final_starts.append(np.ones(n_final, dtype=np.float64))

    # Smooth ramp
    t = np.linspace(0, 1, n_final)
    final_starts.append(_normalize(1.0 + 0.5 * t, float(n_final)))
    final_starts.append(_normalize(1.0 - 0.5 * t, float(n_final)))

    # Score and rank
    scored_final = []
    for s in final_starts:
        sc = _score(s)
        if np.isfinite(sc):
            scored_final.append((sc, s))
    scored_final.sort(key=lambda x: x[0])

    if scored_final:
        print(f"Best final-scale start: {scored_final[0][0]:.6f}, n={n_final}")
        if scored_final[0][0] < best_score:
            best_score = scored_final[0][0]
            best_seq = scored_final[0][1].copy()

    # Run multi-track refinement
    remaining = search_deadline - time.monotonic()
    n_tracks = 3
    top_n = min(n_tracks, len(scored_final))

    if top_n > 0 and remaining > 30:
        track_budget = remaining * 0.7 / top_n

        # Track results
        track_results = []
        for i in range(top_n):
            if time.monotonic() > search_deadline - 30:
                break
            b = track_budget
            sc0, s0 = scored_final[i]
            seq, sc = _local_search(s0, b, seed=int(seed * 10 + i * 7))
            track_results.append((sc, seq))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"Track {i}: {sc:.6f}")

        # Pick best track and continue
        if track_results:
            track_results.sort(key=lambda x: x[0])
            best_track_sc, best_track_seq = track_results[0]
            if best_track_sc < best_score:
                best_score = best_track_sc
                best_seq = best_track_seq

    # === Phase 3: Extended polish on global best ===
    polish_time = search_deadline - time.monotonic()
    if polish_time > 20 and best_seq is not None:
        # Run polish with fresh parameters
        seq, sc = _local_search(best_seq.copy(), polish_time,
                                seed=int(seed * 100 + 42), verbose=True)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
            print(f"Polish result: {sc:.6f}")

    # === Phase 4: Quick final refinement if time remains ===
    if time.monotonic() < search_deadline and best_seq is not None:
        extra_t = search_deadline - time.monotonic()
        if extra_t > 10:
            seq, sc = _local_search(best_seq.copy(), extra_t,
                                    seed=int(seed * 1000 + 99))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"Extra polish: {sc:.6f}")

    # Prepare final output
    if best_seq is None:
        best_seq = np.ones(512, dtype=np.float64)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with exact evaluator
    try:
        exact = evaluate_sequence(result)
        print(f"Final exact score: {exact:.6f}, n={len(result)}")
    except Exception as e:
        print(f"Exact eval error: {e}")
        exact = float('inf')

    return result
