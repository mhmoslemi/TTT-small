from scipy.fft import fft, ifft

def convolve_fft(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    C = ifft(A * A)
    return C[:size].real

def soft_max(x, t=0.1):
    """Smooth approximation of max."""
    # Use log-sum-exp for numerical stability
    x_max = x.max()
    if x_max > 100:
        return x_max  # Avoid overflow
    return t * np.log(np.sum(np.exp((x - x_max) / t))) + x_max

def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""

    A = fft(a, fft_size)
    C = ifft(A * A)
    return C[:size].real

        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)

with adaptive temperature and Nesterov momentum.

import numpy as np

def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho_idx = np.where(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)

def full_convolve(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    C = ifft(A * A)
    return C[:size].real

def compute_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = full_convolve(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))

def sqrt_dirichlet(n):
    """Generate the 'square root of Dirichlet kernel' sequence: C(2k,k)/4^k."""
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return a

def gradient_soft_max(a, t):
    """
    Compute gradient of soft_max(conv(a,a); t) wrt a.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    conv = full_convolve(a)
    
    # Soft max
    c = conv - conv.max()
    e = np.exp(c / t)
    p = e / e.sum()
    sm = t * np.log(e.sum()) + conv.max()
    
    # Gradient: grad_i = sum_j p[i+j] * a[j] for j=0,...,n-1
    # Use FFT: grad = convolve(p, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = (2*n - 1) + n - 1  # = 3n - 2
    L = 1 << (conv_len - 1).bit_length()
    P = fft(p, L)
    AR = fft(a_rev, L)
    C = ifft(P * AR)
    grad = C[n-1 : 2*n-1].real
    
    return sm, grad

    Optimize the sequence using projected gradient with Nesterov momentum
    and adaptive soft-max temperature.
    """
    n = len(a0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_prev = a.copy()
    t_mom = 1.0  # Nesterov momentum parameter
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Initial temperature: large enough for smooth optimization
    t = 1.0 / np.sqrt(n)
    t_min = 0.005 / np.sqrt(n)
    lr = t  # Learning rate proportional to temperature
    
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    
    if verbose:
        print(f"    Init: score={best_score:.8f}, n={n}, t={t:.6f}")
    
    while time.monotonic() < deadline:
        # Nesterov momentum
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        alpha = (t_old - 1.0) / t_mom
        y = a + alpha * (a - a_prev)
        # Project y onto simplex for stability
        y = project_simplex(y)
        
        # Compute gradient at y
        sm, grad = gradient_soft_max(y, t)
        
        # Gradient step + project
        a_new = y - lr * grad
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation

            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stuck = 0

                stuck += 1

                lr = t
            elif stuck == 200:

                lr = t
                # Reset momentum

                t_mom = 1.0

                if sa > 1e-10:

                a_prev = a.copy()
