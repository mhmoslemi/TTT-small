    w = w - w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(w) / len(w)
    return e / s

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2*len(a)-1]

def smoothed_max(c, beta):
    """Smoothed maximum of c, and the weight vector p."""
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    s = e.sum()
    val = (m + np.log(s)) / beta
    p = e / s
    return val, p

def obj_and_grad(w, beta, N):
    """Objective and gradient in log-space."""
    a = softmax(w)
    n = len(a)
    c = compute_conv(a, N)
    c = np.maximum(c, 0.0)

    p_pad[:len(p)] = p

    a_pad[:n] = a

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _softmax(w):
    w = np.asarray(w, dtype=np.float64)
    w = w - w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size, dtype=np.float64) / w.size
    return e / s

def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[: 2 * len(a) - 1]

def _smooth_max_and_weights(c, beta):
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    s = e.sum()
    val = (m + np.log(s)) / beta
    p = e / s
    return val, p

def _obj_grad(w, beta, N):

    n = a.size
    c = _fft_conv(a, N)
    c = np.maximum(c, 0.0)
    sm, p = _smooth_max_and_weights(c, beta)
    val = 2.0 * n * sm

    p_pad = np.zeros(N, dtype=np.float64)

    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[: n] = a
    conv_pa = np.fft.irfft(np.fft.rfft(p_pad, N) * np.fft.rfft(a_pad, N))
    grad_a = 2.0 * conv_pa[n - 1 : 2 * n - 1]
    
    dot = np.dot(a, grad_a)
    grad_w = 2.0 * n * a * (grad_a - dot)
    grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
    grad_w = np.clip(grad_w, -1e4, 1e4)
