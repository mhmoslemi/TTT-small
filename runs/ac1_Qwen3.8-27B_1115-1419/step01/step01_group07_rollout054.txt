    n = len(v)

    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_objective_and_grad(x, tau, n):
    """Compute LSE-smoothed objective and gradient w.r.t. log-parameter x."""
    a = softmax_fn(x)
    c = compute_conv(a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    shifted = np.minimum(shifted, 50.0)
    vals = np.exp(shifted)
    w = vals / vals.sum()
    
    obj = c_max + tau * np.log(vals.sum())
    
    # Gradient w.r.t. a
    g_a = np.zeros(n)
    # Only process non-negligible weights
    active = w > 1e-12
    for k in np.where(active)[0]:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Chain rule through softmax
    dot = np.dot(a, g_a)
    grad_x = a * (g_a - dot)
    return obj, grad_x


def run_lbfgs_polish(a, max_time=60, verbose=False):
    """L-BFGS refinement on log-parameterization with decreasing tau."""
    n = len(a)
    x = np.log(np.maximum(a, 1e-30))
    best_score = objective_value(softmax_fn(x))
    best_a = softmax_fn(x).copy()
    deadline = time.monotonic() + max_time
    
    for tau in [0.05 / n, 0.02 / n, 0.01 / n, 0.005 / n]:
        if time.monotonic() > deadline:
            break
        
        def obj_fn(x_):
            o, _ = lse_objective_and_grad(x_, tau, n)
            return o
        
        def grad_fn(x_):
            _, g = lse_objective_and_grad(x_, tau, n)
            return g
        
        try:
            result = minimize(obj_fn, x, jac=grad_fn, method='L-BFGS-B',
                              options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-11,
                                       'maxls': 30})
            x = result.x
        except Exception:
            break
        
        a_new = softmax_fn(x)
        score = objective_value(a_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            if verbose:
                print(f"      L-BFGS (tau={tau:.6f}): {best_score:.6f}")
    
    return best_a, best_score


def run_coordinate_descent(a, max_iters=5000, n_active=30, verbose=False):
    """Cyclic coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a_curr = a.copy()
    best_score = objective_value(a_curr)
    best_a = a_curr.copy()
    rng = np.random.default_rng(99)
    
    for it in range(max_iters):
        # Pick a random coordinate
        i = rng.integers(0, n)
        
        # Binary search for optimal a[i] given others fixed
        # The objective as a function of a[i] is max_k (c0[k] + 2*a[i]*a[k-i] + a[i]^2*1[k=2i])
        # where c0 is the convolution with a[i] = 0
        a_zero = a_curr.copy()
        old_val = a_zero[i]
        a_zero[i] = 0.0
        c0 = compute_conv(a_zero)
        rest_sum = a_zero.sum()
        
        # For each peak k, c[k] = c0[k] + 2*a[i]*a[k-i] (if 0<=k-i<n) [+ a[i]^2 if k=2i]
        # We want to minimize max_k c[k]
        # This is a 1D convex problem in a[i] (piecewise quadratic)
        # Use ternary search
        
        def f_1d(val):
            a_test = a_zero.copy()
            a_test[i] = val
            a_test_sum = a_test.sum()
            if a_test_sum < 0.01:
                return float('inf')
            c_test = compute_conv(a_test)
            return 2.0 * n * c_test.max() / (a_test_sum * a_test_sum)
        
        # Ternary search on [0, 5*mean]
        lo = 0.0
        hi = 5.0 * (1.0 / n)
        
        # First check if current is good
        f_curr = f_1d(old_val)
        
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = f_1d(m1)
            f2 = f_1d(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        
        optimal = (lo + hi) / 2.0
        f_opt = f_1d(optimal)
        
        if f_opt < f_curr:
            a_curr[i] = optimal
            if f_opt < best_score:
                best_score = f_opt
                best_a = a_curr.copy()
                if verbose and it % 500 == 0:
                    print(f"      Coord it={it}: {best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """Main search function. Combines FISTA, multiplicative updates, L-BFGS,
    and coordinate descent with multi-restart strategy."""
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30
    rng = np.random.default_rng(2024)
    
    # Get initial sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a0 = np.array(height_sequence_1, dtype=np.float64)
        print(f"Loaded parent sequence, length={len(a0)}")
    else:
        a0 = np.ones(4806, dtype=np.float64)
        print("No parent available, using uniform init length=4806")
    
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 0.01:
        a0 = np.ones(n)
    a0 = a0 / a0.sum()
    
    best_score = objective_value(a0)
    best_a = a0.copy()
    print(f"Initial score: {best_score:.6f}, n={n}, norm2={np.linalg.norm(a0):.4f}")
    
    # === Phase 1: Quick FISTA to get a good baseline ===
    print("\n=== Phase 1: FISTA (quick) ===")
    t_phase = time.monotonic()
    a = a0.copy()
    a, score = run_fista(a, max_iters=2000, n_active=20, step_scale=0.8)
    if score < best_score:
        best_score = score
        best_a = a.copy()
        print(f"  FISTA1: {best_score:.6f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: Multiplicative updates ===
    print("\n=== Phase 2: Multiplicative weight updates ===")
    t_phase = time.monotonic()
    a = best_a.copy()
    for eta in [0.05, 0.03, 0.02, 0.01, 0.005]:
        if time.monotonic() > deadline - 600:
            break
        a, score = run_multiplicative(a, max_iters=1500, n_active=30, eta=eta)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  Mult (eta={eta}): {best_score:.6f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: L-BFGS polishing ===
    print("\n=== Phase 3: L-BFGS polishing ===")
    t_phase = time.monotonic()
    a = best_a.copy()
    while time.monotonic() < deadline - 400:
        a, score = run_lbfgs_polish(a, max_time=90, verbose=True)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  L-BFGS: {best_score:.6f} (total {time.monotonic()-start_time:.0f}s)")
        else:
            # Try perturbation before giving up
            a_pert = best_a + rng.normal(0, 0.02, n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_p = objective_value(a_pert)
            if score_p < best_score:
                best_score = score_p
                best_a = a_pert.copy()
                a = a_pert.copy()
                print(f"  Perturbation helped: {best_score:.6f}")
            else:
                break
    
    # === Phase 4: Extended FISTA with larger active set ===
    print("\n=== Phase 4: FISTA (extended, larger active set) ===")
    t_phase = time.monotonic()
    a = best_a.copy()
    for n_act in [40, 60, 80, 100]:
        if time.monotonic() > deadline - 200:
            break
        a, score = run_fista(a, max_iters=3000, n_active=n_act, step_scale=0.5)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  FISTA (K={n_act}): {best_score:.6f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: Coordinate descent on a subset ===
    print("\n=== Phase 5: Coordinate descent ===")
    if time.monotonic() < deadline - 150:
        t_phase = time.monotonic()
        a = best_a.copy()
        a, score = run_coordinate_descent(a, max_iters=2000, n_active=30, verbose=True)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  CoordDesc: {best_score:.6f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 6: Final refinement loop ===
    print("\n=== Phase 6: Final refinement ===")
    t_phase = time.monotonic()
    a = best_a.copy()
    improve_count = 0
    while time.monotonic() < deadline - 10:
        # Alternate between FISTA and multiplicative
        if improve_count % 2 == 0:
            a, score = run_fista(a, max_iters=1500, n_active=50, step_scale=0.4)
        else:
            a, score = run_multiplicative(a, max_iters=1000, n_active=50, eta=0.01)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            improve_count += 1
            print(f"  Final ref it={improve_count}: {best_score:.6f} ({time.monotonic()-t_phase:.0f}s)")
        else:
            # Small perturbation to escape
            scale = 0.005 + 0.01 * np.exp(-improve_count * 0.1)
            a_pert = best_a + rng.normal(0, scale, n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_p = objective_value(a_pert)
            if score_p < best_score:
                best_score = score_p
                best_a = a_pert.copy()
                a = a_pert.copy()
                improve_count += 1
                print(f"  Perturbation helped: {best_score:.6f}")
            else:
                # Try L-BFGS as last resort
                if time.monotonic() < deadline - 60:
                    a, score = run_lbfgs_polish(best_a, max_time=45, verbose=True)
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                        improve_count += 1
                        print(f"  L-BFGS rescue: {best_score:.6f}")
                    else:
                        break
    
    # Final verification
    final_local = objective_value(best_a)
    print(f"\n=== SUMMARY ===")
    print(f"Local score: {final_local:.6f}")
    print(f"Best score: {best_score:.6f}")
    print(f"Sequence length: {n}")
    print(f"Sum: {best_a.sum():.6f}")
    print(f"Max element: {best_a.max():.6f}")
    print(f"Elapsed: {time.monotonic() - start_time:.0f}s / {budget_s}s")
    
    # Verify with the actual evaluator
    try:
        eval_score = evaluate_sequence([float(x) for x in best_a])
        print(f"Evaluator score: {eval_score:.6f}")
        if eval_score < best_score and np.isfinite(eval_score):
            # Update if evaluator gives better (shouldn't happen but just in case)
            pass
    except Exception as e:
        print(f"Evaluator error: {e}")
    
    return [float(x) for x in best_a]
