import numpy as np
from scipy.fft import fft, ifft
import time

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth energy objective (sum of c[k]^2) with projected gradient + Nesterov momentum
2. Subgradient polishing of the exact max
3. Adaptive learning rate with restarts
4. FFT-based fast convolution for O(n log n) gradient computation
"""


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    """Fast score using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize to target sum preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _project_sum_nonneg(a, S):
    """Project onto {x: sum(x)=S, x>=0} via bisection water-filling."""
    n = len(a)
    lo = 0.0
    hi = float(np.max(a)) + S / n + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.maximum(a - mid, 0.0)
        s = np.sum(proj)
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - (lo + hi) * 0.5, 0.0)


def _grad_energy(a):
    """
    Compute gradient of sum_k c[k]^2 where c = convolve(a, a).
    Result has same length as a.
    grad[i] = 4 * sum_k c[k] * a[k-i] (valid indices)
    = 4 * convolve(c, a_rev)[i + n - 1]
    """
    n = len(a)
    c = _fft_conv(a, a)  # length 2n-1
    a_rev = a[::-1]
    # convolve(c, a_rev)[i+n-1] for i=0..n-1
    conv = _fft_conv(c, a_rev)  # length (2n-1) + n - 1 = 3n-2
    return 4.0 * conv[n - 1:2 * n - 1]


def _subgrad_max(a):
    """
    Compute subgradient of max(c[k]) where c = convolve(a,a).
    Uses all peaks within 2% of max for smoother gradient.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    c = _fft_conv(a, a)
    max_c = float(np.max(c))
    threshold = 0.98 * max_c
    peak_mask = c >= threshold
    # Weight peaks by how much they exceed threshold
    weights = np.where(peak_mask, c - threshold + 1e-10, 0.0)
    total_w = np.sum(weights)
    if total_w < 1e-15:
        # If only one peak, use it
        k_max = int(np.argmax(c))
        a_rev = a[::-1]
        conv = _fft_conv(np.eye(1)[None, 0] * c, a_rev)  # just c convolved
        # Actually for single peak: grad[i] = 2*a[k_max - i] (valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        return grad
    weights = weights / total_w
    # grad[i] = 2 * sum_k weights[k] * a[k-i]
    # = 2 * convolve(weights, a_rev)[i+n-1]
    a_rev = a[::-1]
    conv = _fft_conv(weights, a_rev)
    return 2.0 * conv[n - 1:2 * n - 1]


def _run_smooth_opt(a, deadline, lr_init=0.005, momentum=0.9, eval_interval=100):
    """
    Projected gradient with Nesterov momentum on sum(c^2) objective.
    Returns best sequence and score found.
    """
    n = len(a)
    S = float(np.sum(a))
    best = a.copy()
    best_score = _score_fast(a)
    
    v = np.zeros(n)  # momentum velocity
    y = a.copy()  # Nesterov extrapolation point
    lr = lr_init
    stuck = 0
    step = 0
    
    while time.monotonic() < deadline:
        # Nesterov: evaluate gradient at extrapolated point
        y_new = y + momentum * (y - a)
        y = np.maximum(y_new, 0.0)
        y = _normalize(y, S)
        
        grad = _grad_energy(y)
        
        # Gradient step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_sum_nonneg(a_new, S)
        
        # Update momentum
        # (Nesterov: v is implicit in the y update above)
        a = a_new
        step += 1
        
        if step % eval_interval == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best = a.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 30:
                    # Restart with perturbation
                    a = best + 0.03 * S * np.random.standard_normal(n) / np.sqrt(n)
                    a = np.maximum(a, 0.0)
                    a = _project_sum_nonneg(a, S)
                    y = a.copy()
                    lr = lr_init
                    stuck = 0
            
            # Adaptive lr
            if stuck % 10 == 0 and stuck > 0:
                lr = max(lr * 0.8, 1e-7)
    
    return best, best_score


def _polish_subgrad(a, deadline, lr_init=0.02, eval_interval=50):
    """
    Subgradient polishing on the exact max objective.
    """
    n = len(a)
    S = float(np.sum(a))
    best = a.copy()
    best_score = _score_fast(a)
    v = np.zeros(n)
    beta = 0.85
    lr = lr_init
    stuck = 0
    step = 0
    
    while time.monotonic() < deadline:
        grad = _subgrad_max(a)
        v = beta * v + (1.0 - beta) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_sum_nonneg(a_new, S)
        a = a_new
        lr = max(lr * 0.99995, 1e-7)
        step += 1
        
        if step % eval_interval == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best = a.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 40:
                    a = best + 0.02 * S * np.random.standard_normal(n) / np.sqrt(n)
                    a = np.maximum(a, 0.0)
                    a = _project_sum_nonneg(a, S)
                    v = np.zeros(n)
                    lr = lr_init * 0.5
                    stuck = 0
    
    return best, best_score


def _try_temperature_opt(a, deadline, T_init=50.0, T_decay=0.999, lr=0.003):
    """
    Soft-max (log-sum-exp) based optimization with temperature annealing.
    Minimize (1/T)*log(sum(exp(T*c[k]/C))) which interpolates between mean and max.
    """
    n = len(a)
    S = float(np.sum(a))
    C = S * S
    best = a.copy()
    best_score = _score_fast(a)
    T = T_init
    v = np.zeros(n)
    beta = 0.9
    stuck = 0
    step = 0
    
    while time.monotonic() < deadline:
        c = _fft_conv(a, a)
        # Soft gradient: grad[i] = (1/C) * sum_k (c[k]/C) * exp(T*c[k]/C) * 2*a[k-i] / sum(exp(...))
        # Simplified: weight[k] = c[k] * exp(T*(c[k]-max_c)/C)
        max_c = float(np.max(c))
        weights = c * np.exp(T * (c - max_c) / C)
        total_w = np.sum(weights)
        if total_w < 1e-30:
            weights = np.zeros_like(c)
            weights[int(np.argmax(c))] = 1.0
        else:
            weights = weights / total_w
        
        a_rev = a[::-1]
        conv = _fft_conv(weights, a_rev)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        v = beta * v + (1.0 - beta) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_sum_nonneg(a_new, S)
        a = a_new
        T *= T_decay
        step += 1
        
        if step % 100 == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best = a.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 30:
                    a = best + 0.02 * S * np.random.standard_normal(n) / np.sqrt(n)
                    a = np.maximum(a, 0.0)
                    a = _project_sum_nonneg(a, S)
                    v = np.zeros(n)
                    stuck = 0
                    T = T_init
    
    return best, best_score
