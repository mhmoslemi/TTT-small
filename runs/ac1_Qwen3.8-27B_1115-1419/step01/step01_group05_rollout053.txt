```python
def project_simplex(x):
    n = len(x)
    if n == 0:
        return x
    # Find lambda
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n+1)
    cond = u - (cssv - 1) / r > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond))
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(x - theta, 0)
```

```python
"""
Search for low-peak nonnegative self-convolutions using Adam optimization
with LSE surrogate, spectral initialization, and L-BFGS-B polishing.
"""
import numpy as np
import time

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s

def _project_simplex(x):
    """Exact projection onto {x >= 0, sum(x) = 1}."""
    n = x.size
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / r > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(x - theta, 0.0)

def _convolve(a, b, L=None):
    if L is None:
        L = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:len(a) + len(b) - 1]

def _autocorr(a):
    L = _next_pow2(2 * len(a) - 1)
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * len(a) - 1]

def _lse(b, beta):
    m = b.max()
    eb = np.exp(beta * (b - m))
    s = eb.sum()
    if s <= 0:
        return m
    return m + np.log(s) / beta

def _lse_grad(b, a, beta):
    """Gradient of (1/beta)*log(sum exp(beta*b_k)) w.r.t. a."""
    m = b.max()
    eb = np.exp(beta * (b - m))
    s = eb.sum()
    if s <= 0:
        p = np.ones(len(b)) / len(b)
    else:
        p = eb / s
    # grad_j = (2/beta) * sum_k p_k * a_{k-j}
    n = len(a)
    L = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, L)
    fp = np.fft.rfft(p[::-1], L)  # reversed p
    conv = np.fft.irfft(fa * fp, L)
    c = conv[n-1:2*n-1][::-1]
    return (2.0 / beta) * c

def _score(a):
    a = _normalize(a)
    n = a.size
    b = _autocorr(a)
    return 2.0 * n * b.max()

def _spectral_init(n, rng, iters=200):
    """Iterative spectral method to design flat-spectrum non-negative sequence."""
    a = np.ones(n) / n
    for it in range(iters):
        A = np.fft.fft(a)
        B = np.abs(A)**2
        # Target: flat B (except DC)
        B_target = B.copy()
        B_target[1:] = B[1:].mean() if n > 1 else 0
        # Adjust magnitude, keep phase
        new_mag = np.sqrt(np.maximum(B_target, 0))
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        new_A[0] = np.abs(A[0])  # DC is real positive
        a = np.real(np.fft.ifft(new_A))
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            break
    return a

def _adam_optimize(a, deadline, beta_start=10.0, beta_end=200.0):
    """Adam optimization with LSE surrogate on the simplex."""
    a = _normalize(a.copy())
    n = a.size
    L_conv = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    
    lr = 0.05 / np.sqrt(n) * 10  # Scale with n
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta_lse = beta_start
    
    best_a = a.copy()
    best_score = _score(a)
    best_lse = np.inf
    
    stall = 0

        # Compute b = a * a
        fa = np.fft.rfft(a, L_conv)
        b = np.fft.irfft(fa * fa, L_conv)[:2*n-1]

        if s <= 0:

        a_new = _project_simplex(a_new)
