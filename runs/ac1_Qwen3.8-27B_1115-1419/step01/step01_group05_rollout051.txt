    s = a.sum()
    if s < 1e-30:
        return np.ones(a.shape[0]) / a.shape[0]

    return np.convolve(a, a, mode='full')

    a = _normalize(a)
    b = _full_conv(a)

    if beta < 2.0:

        if s < 1e-300:

            p[int(np.argmax(b))] = 1.0
            val = m
        else:
            p = exps / s
            val = m + np.log(s) / beta
    
    # Compute gradient: grad[i] = 2 * sum_k p[k] * a[k-i]

        if i < 2*n-1:

            grad[i] = 2.0 * np.dot(p[idx], a[idx - i])

    n = len(a)
    idx = np.argsort(b)[-k:]
    p = np.zeros(len(b))
    p[idx] = 1.0 / k

"""
Minimize peak self-convolution of non-negative sequences of length 256.
Combines LSE-smoothed gradient descent with momentum, top-k peak targeting,
adaptive step sizes, and local search refinement.
"""

import numpy as np
import time

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(a.shape[0], dtype=np.float64) / a.shape[0]
    return a / s

def _conv(a):
    return np.convolve(a, a, mode='full')

def _score_direct(a):
    b = _conv(a)
    return b.max()

def _lse_val(b, beta):
    m = b.max()
    if beta < 2.0:
        return m
    shifted = beta * (b - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def _lse_weights(b, beta):
    m = b.max()
    if beta < 2.0:
        p = np.zeros(len(b))
        p[int(np.argmax(b))] = 1.0
        return p
    shifted = beta * (b - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        p = np.zeros(len(b))
        p[int(np.argmax(b))] = 1.0
        return p
    return e / s

def _grad_from_weights(a, p):
    """Compute grad[i] = 2 * sum_k p[k] * a[k-i] for valid indices."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for i in range(n):
        lo = i
        hi = min(2 * n - 1, i + n)
        if lo < hi:
            grad[i] = 2.0 * np.dot(p[lo:hi], a[:hi - lo])
    return grad

def _project_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    n = v.shape[0]
    if s < 1e-30:
        return np.ones(n, dtype=np.float64) / n
    return v / s

def _make_exponential(n, alpha):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * x / n)
    return _normalize(a)

def _make_raised_cosine(n):
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    return _normalize(a)

def _make_gaussian(n, sigma_frac):
    center = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64) - center
    a = np.exp(-0.5 * (x / (sigma_frac * n)) ** 2)
    return _normalize(a)

def _make_mixture(n, alpha_mix=0.5, gamma=1.0):
    a1 = np.exp(-gamma * np.arange(n, dtype=np.float64) / n)
    a2 = np.exp(-gamma * (n - 1 - np.arange(n, dtype=np.float64)) / n)
    a = alpha_mix * a1 + (1 - alpha_mix) * a2
    return _normalize(a)

def _make_smooth_random(n, rng, base=None, amplitude=0.3):
    if base is None:
        base = np.ones(n) / n
    z = rng.normal(size=n)
    # Low-pass filter
    F = np.fft.rfft(z)
    cutoff = max(2, n // 8)
    F[cutoff:] = 0
    f = np.fft.irfft(F, n)
    f = (f - f.mean()) / (f.std() + 1e-10)
    a = base * (1.0 + amplitude * f)
    a = np.maximum(a, 1e-10)
    return _normalize(a)

def _resample(a, new_n):
    a = _normalize(a)
    if a.shape[0] == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, a.shape[0])
    x_new = np.linspace(0, 1, new_n)
    return _normalize(np.interp(x_new, x_old, a))

def _lse_optimize(a, deadline, rng, beta_init=30.0, momentum=0.9):
    """LSE-smoothed gradient descent with momentum and backtracking."""
    a = _normalize(a)
    n = a.shape[0]
    
    best_a = a.copy()
    best_score = _score_direct(a)
    b = _conv(a)
    
    alpha = 0.1  # step size
    beta = beta_init
    vel = np.zeros(n, dtype=np.float64)  # momentum velocity
    
    it = 0
    stall = 0
    max_iter = 50000
    
    while time.monotonic() < deadline and it < max_iter:
        b = _conv(a)
        cur_score = b.max()
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute LSE gradient
        p = _lse_weights(b, beta)
        grad = _grad_from_weights(a, p)
        
        # Center the gradient (projection onto simplex tangent space)
        grad_c = grad - grad.mean()
        gnorm = np.linalg.norm(grad_c)
        
        if gnorm < 1e-15:
            # No progress possible, restart
            a = _perturb(a, rng, 0.2)
            vel = np.zeros(n, dtype=np.float64)
            alpha = 0.1
            it += 1
            continue
        
        # Normalize gradient for stable step size
        grad_n = grad_c / gnorm
        
        # Momentum update
        vel = momentum * vel + (1 - momentum) * grad_n
        direction = -vel  # descent direction
        
        # Backtracking line search
        a_trial = _project_simplex(a + alpha * direction)
        b_trial = _conv(a_trial)
        smooth_trial = _lse_val(b_trial, beta)
        smooth_old = _lse_val(b, beta)
        
        step_ok = False
        for _ in range(10):
            if smooth_trial < smooth_old:
                step_ok = True
                break
            alpha *= 0.5
            a_trial = _project_simplex(a + alpha * direction)
            b_trial = _conv(a_trial)
            smooth_trial = _lse_val(b_trial, beta)

            alpha = min(alpha * 1.05, 1.0)

def _perturb(a, rng, strength=0.15):
    noise = np.exp(rng.normal(0, strength, a.shape[0]))
    return _normalize(a * noise)

def _local_search(a, deadline, rng):
    """Local search with peak-directed moves."""
    a = _normalize(a)
    n = a.shape[0]
    best_a = a.copy()
    best_b = _conv(a)
    best_peak = best_b.max()
    
    it = 0
    max_it = 200000
    
    while time.monotonic() < deadline and it < max_it:
        # Strategy 1: Peak-directed move
        if rng.random() < 0.6:
            b = _conv(a)
            k_peak = int(np.argmax(b))
            # Find the j that contributes most to b[k_peak]
            j_lo = max(0, k_peak - n + 1)
            j_hi = min(n - 1, k_peak)
            if j_lo >= j_hi:
                a = _perturb(a, rng, 0.05)
                it += 1
                continue
            contribs = np.array([a[j] * a[k_peak - j] for j in range(j_lo, j_hi + 1)])
            j_bad = j_lo + int(np.argmax(contribs))
            # Move mass from the "bad" index to a random other index
            j_good = int(rng.integers(0, n))
            delta = a[j_bad] * 0.1
            if j_good != j_bad and a[j_bad] > delta:
                a_new = a.copy()
                a_new[j_bad] -= delta
                a_new[j_good] += delta
                a_new = _normalize(a_new)
                b_new = _conv(a_new)
                if b_new.max() < best_peak:
                    a = a_new
                    best_a = a.copy()
                    best_b = b_new
                    best_peak = b_new.max()
        # Strategy 2: Random pair swap
        elif rng.random() < 0.8:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = min(a[i], a[j]) * 0.05
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _normalize(a_new)
                b_new = _conv(a_new)
                if b_new.max() < best_peak:
                    a = a_new
                    best_a = a.copy()
                    best_b = b_new
                    best_peak = b_new.max()
        # Strategy 3: Small multiplicative perturbation
        else:
            a = _perturb(a, rng, 0.03)
            b_check = _conv(a)
            if b_check.max() < best_peak:
                best_a = a.copy()
                best_b = b_check
                best_peak = b_check.max()
            else:
                a = best_a.copy()
        
        it += 1
    
    return best_a, best_peak

"""
Minimize peak self-convolution of non-negative sequences of length 256.
Uses LSE-smoothed projected gradient descent with vectorized gradients,
backtracking line search, momentum, top-k peak targeting, and local search
refinement with peak-directed moves.

import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(a.shape[0], dtype=np.float64) / max(a.shape[0], 1)
    return a / s


def _conv(a):
    return np.convolve(a, a, mode='full')


def _score_direct(a):
    b = _conv(a)
    return float(b.max())


def _lse_val(b, beta):
    m = float(b.max())
    if beta < 2.0:
        return m
    shifted = beta * (b - m)
    shifted = np.maximum(shifted, -700.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        return m
    return m + float(np.log(s)) / beta


def _lse_weights(b, beta):
    m = float(b.max())
    p = np.zeros(len(b), dtype=np.float64)
    if beta < 2.0:
        p[int(np.argmax(b))] = 1.0
        return p
    shifted = beta * (b - m)
    shifted = np.maximum(shifted, -700.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        p[int(np.argmax(b))] = 1.0
        return p
    return e / s


def _topk_weights(b, k):
    p = np.zeros(len(b), dtype=np.float64)
    idx = np.argsort(b)[-k:]
    p[idx] = 1.0 / k
    return p

    """Vectorized gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (valid indices)."""
    n = len(a)
    conv_full = np.convolve(p, a[::-1], mode='full')
    return 2.0 * conv_full[n - 1:2 * n - 1]

    s = v.sum()
    n = v.shape[0]
    if s < 1e-30:
        return np.ones(n, dtype=np.float64) / n
    return v / s

    return _normalize(a)

    a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / float(n)) ** 2

    center = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64) - center

    """Raised cosine to the power beta_w."""
    x = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / float(n)) ** 2
    a = x ** beta_w
    return _normalize(a)


def _make_rectified_smooth(n, rng, bandwidth=0.3):
    """Generate band-limited noise, rectify, normalize."""
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, int(n * bandwidth))
    if cutoff < len(F):
        F[cutoff:] = 0
    f = np.fft.irfft(F, n)
    a = np.maximum(f, 0.0)
    if a.sum() < 1e-30:
        a = np.ones(n) / n
    return _normalize(a)

    x = np.arange(n, dtype=np.float64)

def _make_smooth_random(n, rng, base=None, amplitude=0.3):

    a = base * (1.0 + amplitude * f)
    a = np.maximum(a, 1e-12)

def _resample(a, new_n):
    a = _normalize(a)
    if a.shape[0] == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, a.shape[0])
    x_new = np.linspace(0, 1, new_n)
    return _normalize(np.interp(x_new, x_old, a))

    noise = np.exp(rng.normal(0, strength, a.shape[0]))

    return _normalize(a_new)
