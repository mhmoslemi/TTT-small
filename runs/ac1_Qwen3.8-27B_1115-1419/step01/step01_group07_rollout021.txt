    w[top_idx] = 1.0 / n_active
    # Gradient: 2 * correlate(w, a, 'valid')
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    a_new = a - step * grad_a
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def run_lbfgs(x, n, max_iters_per_tau=150, time_limit=120.0):
    """Run L-BFGS with decreasing tau schedule. Returns (best_a, best_score, x)."""
    deadline = time.monotonic() + time_limit
    a = softmax_func(x)
    best_a = a.copy()
    best_score = compute_score(best_a)
    
    # Tau schedule scaled with n
    base_tau = 1.0 / (4.0 * np.sqrt(n))
    taus = [base_tau * 4.0, base_tau * 2.0, base_tau, base_tau * 0.5, base_tau * 0.25]
    
    for tau in taus:
        if time.monotonic() > deadline:
            break
        if tau < 1e-10:
            break
        
        global _current_tau
        _current_tau = tau
        
        try:
            result = minimize(
                _obj_wrapper, x, jac=_grad_wrapper,
                method='L-BFGS-B',
                options={'maxiter': max_iters_per_tau, 'ftol': 1e-13,
                         'gtol': 1e-11, 'maxfun': max_iters_per_tau * 2,
                         'maxls': 25}
            )
            x = result.x
        except Exception:
            break
        
        a = softmax_func(x)
        score = compute_score(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score, x


def polish_subgradient(a, deadline, initial_step=0.015, max_active=60):
    """Adaptive subgradient polishing."""
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    step = initial_step
    consecutive_no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() < deadline - 3:
        a = subgradient_step(a, n_active=min(max_active, 2 * n - 1), step=step)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        # Adaptive step
        step *= 0.998
        if step < 0.0008:
            step = 0.004
            # Try perturbation to escape local min
            if consecutive_no_improve > 200:
                pert = rng.normal(0, 0.004, n)
                a_p = a + pert
                a_p = np.maximum(a_p, 0)
                s = a_p.sum()
                if s > 1e-10:
                    a_p /= s
                    sp = compute_score(a_p)
                    if sp < best_score:
                        best_score = sp
                        best_a = a_p.copy()
                        a = a_p.copy()
                        consecutive_no_improve = 0
                        print(f"    Perturbation improved to {best_score:.6f}")
    
    return best_a, best_score


def gen_ramp(n, power=1.0):
    """Power-law ramp: a[i] proportional to (i+1)^power."""
    x = np.arange(1, n + 1, dtype=np.float64) ** power
    return x / x.sum()


def gen_exp_ramp(n, rate=3.0):
    """Exponential ramp."""
    x = np.exp(np.linspace(0.0, rate, n))
    return x / x.sum()


def gen_quadratic_ramp(n):
    """Quadratic ramp."""
    x = np.arange(1, n + 1, dtype=np.float64) ** 2
    return x / x.sum()


def gen_cosine_modulated_ramp(n, freq=1, amp=0.3):
    """Ramp with cosine modulation."""
    base = np.arange(1, n + 1, dtype=np.float64)
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
    x = np.maximum(base * mod, 0.01)
    return x / x.sum()


def gen_parent_like(n):
    """Generate a sequence similar to the parent solution (asymmetric, peaked left)."""
    # Create an asymmetric shape: heavy on left, tapering on right
    t = np.linspace(0, 1, n)
    x = (1.0 - t) ** 0.7 + 0.1 * t
    x = np.maximum(x, 0.01)
    return x / x.sum()


def upsample(a, new_n):
    """Upsample a sequence to a larger length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s > 1e-10:
        new_a /= s
    return new_a


def propose_candidate(budget_s=2400, **kwargs):
    """Main search: find best sequence minimizing the evaluation score."""
    global _current_tau
    
    deadline = time.monotonic() + budget_s - 45
    rng = np.random.default_rng(2024)
    
    best_score = np.inf
    best_a = None
    
    # Get parent if available
    parent = None
    if 'height_sequence_1' in globals():
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 1e-15)
        parent = parent / parent.sum()
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
        if parent_score < best_score:
            best_score = parent_score
            best_a = parent.copy()
    
    # Define lengths to try
    lengths = [4000, 5000, 6000, 8000]
    
    # Phase 1: Coarse optimization at small n to find good shape
    print("\n=== Phase 1: Coarse optimization (n=500) ===")
    coarse_n = 500
    coarse_deadline = min(time.monotonic() + 60, deadline - 600)
    
    coarse_best_a = gen_ramp(coarse_n, power=1.0)
    coarse_best_score = compute_score(coarse_best_a)
    print(f"  Ramp init: score={coarse_best_score:.6f}")
    
    for init_func in [lambda: gen_ramp(coarse_n, 1.0),
                      lambda: gen_ramp(coarse_n, 0.7),
                      lambda: gen_ramp(coarse_n, 1.5),
                      lambda: gen_exp_ramp(coarse_n, 3.0),
                      lambda: gen_exp_ramp(coarse_n, 5.0)]:
        if time.monotonic() > coarse_deadline:
            break
        a = init_func()
        x = np.log(a + 1e-15)
        a_opt, score, x = run_lbfgs(x, coarse_n, max_iters_per_tau=80, time_limit=10)
        if score < coarse_best_score:
            coarse_best_score = score
            coarse_best_a = a_opt.copy()
            print(f"  Coarse improved: {score:.6f}")
    
    print(f"  Coarse best: {coarse_best_score:.6f}")
    
    # Phase 2: Multi-length, multi-init optimization
    print("\n=== Phase 2: Multi-length optimization ===")
    
    # Generate initializations for each length
    all_inits = []
    for n in lengths:
        inits_for_n = []
        # Upsampled coarse solution
        inits_for_n.append(upsample(coarse_best_a, n))
        # Pure ramp
        inits_for_n.append(gen_ramp(n, 1.0))
        # Slightly sub-linear ramp
        inits_for_n.append(gen_ramp(n, 0.8))
        # Slightly super-linear ramp
        inits_for_n.append(gen_ramp(n, 1.3))
        # Exponential
        inits_for_n.append(gen_exp_ramp(n, 4.0))
        # Cosine-modulated ramp
        inits_for_n.append(gen_cosine_modulated_ramp(n, 1, 0.2))
        # Parent upsampled
        if parent is not None:
            inits_for_n.append(upsample(parent, n))
        # Random perturbation of ramp
        a_rand = gen_ramp(n, 1.0) + rng.normal(0, 0.02, n)
        a_rand = np.maximum(a_rand, 0)
        inits_for_n.append(a_rand / a_rand.sum())
        
        for a_init in inits_for_n:
            all_inits.append((n, a_init))
    
    # Sort by length (shorter first for quick wins)
    all_inits.sort(key=lambda x: x[0])
    
    # Time budget per init
    time_per_init = min(80.0, (deadline - time.monotonic() - 300.0) / max(1, len(all_inits)))
    
    for i, (n, a_init) in enumerate(all_inits):
        if time.monotonic() > deadline - 300:
            break
        
        t0 = time.monotonic()
        x = np.log(a_init + 1e-15)
        
        # Run L-BFGS
        a_opt, score, x = run_lbfgs(x, n, max_iters_per_tau=120, 
                                     time_limit=min(time_per_init, 100.0))
        
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Init {i+1}/{len(all_inits)} (n={n}): new best {best_score:.6f}")
        
        # Quick subgradient polish
        polish_deadline = min(time.monotonic() + 20, deadline - 200)
        a_opt, score = polish_subgradient(a_opt, polish_deadline, 
                                           initial_step=0.01, max_active=40)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  After polish (n={n}): {best_score:.6f}")
        
        elapsed = time.monotonic() - t0
        if (i + 1) % 5 == 0:
            print(f"  Progress: {i+1}/{len(all_inits)}, best={best_score:.6f}, "
                  f"elapsed={time.monotonic()-t0+sum(0 for _ in []):.0f}s total")
    
    # Phase 3: Extended polishing on best found
    print(f"\n=== Phase 3: Extended polishing ===")
    print(f"  Best before polish: {best_score:.6f}, n={len(best_a)}")
    
    a = best_a.copy()
    n = len(a)
    
    # Multiple polishing rounds with increasing active set
    for round_num, (n_active, step0) in enumerate([(30, 0.01), (50, 0.008), (80, 0.005), (120, 0.003)]):
        if time.monotonic() > deadline - 10:
            break
        a, score = polish_subgradient(a, min(time.monotonic() + 120, deadline - 10),
                                       initial_step=step0, max_active=n_active)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  Polish round {round_num+1} (active={n_active}): {best_score:.6f}")
    
    # Phase 4: Final fine polishing
    print(f"\n=== Phase 4: Final polishing ===")
    a = best_a.copy()
    a, score = polish_subgradient(a, deadline - 5, initial_step=0.003, max_active=100)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    # Verify with evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nFinal: n={len(best_a)}, compute_score={best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}")
    
    if not np.isfinite(final_score):
        # Fallback to parent or uniform
        if parent is not None:
            return [float(v) for v in parent]
        return [1.0 / 100] * 100
    
    return [float(v) for v in best_a]
