import numpy as np
import time

def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient(a, n_peaks=10):
    """Subgradient of max(conv(a,a)) w.r.t. a using top peaks."""
    n = len(a)
    c = np.convolve(a, a)
    idx = np.argpartition(c, -n_peaks)[-n_peaks:]
    idx = idx[np.argsort(c[idx])[::-1]]  # Sort by value descending
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(idx)
    return g

    """Compute the objective (without the 2n/S^2 factor, just max_b)."""
    n = len(a)
    c = np.convolve(a, a)
    return c.max()

Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using projected subgradient on the simplex,
L-BFGS with LSE smoothing, and peak-shaving heuristics.
Multi-length search with diverse restarts.

import numpy as np
import time
from scipy.optimize import minimize


def simplex_project(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, 1+n) > (cssv - 1)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_subgradient(a, n_peaks=20):
    """Compute subgradient of max(conv(a,a)) w.r.t. a at top peaks."""
    n = len(a)
    c = np.convolve(a, a)
    # Get top n_peaks indices
    if n_peaks >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        idx = idx[np.argsort(c[idx])[::-1]]
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(idx)
    return g, c


def objective(a):
    """2*n*max_b / S^2 where S = sum(a)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def run_subgradient(a0, time_limit=60, n_peaks=20, initial_step=0.05):
    """Projected subgradient with adaptive step."""
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_obj = objective(best_a)
    step = initial_step
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_limit:
        g, c = get_subgradient(a, n_peaks)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            break
        g = g / norm_g  # Normalize to unit step
        
        # Try a few step sizes
        best_local_a = a.copy()
        best_local_obj = best_obj if objective(a) >= best_obj else objective(a)
        
        a_new = simplex_project(a - step * g)
        obj_new = objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.05, 0.5)
            stall = 0
        elif obj_new < objective(a):
            a = a_new
            step = min(step * 1.01, 0.5)
            stall += 1
        else:
            step *= 0.9

                best_obj = obj_check
                best_a = a.copy()
            stall += 2

            stall = 0

def peak_shave(a, n_it=1000, time_limit=30):
    """Direct peak shaving: reduce elements contributing to peaks."""
    a = simplex_project(a.copy())
    best_a = a.copy()
    best_obj = objective(best_a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    for it in range(n_it):
        if time.monotonic() - start > time_limit:
            break
        n = len(a)
        c = np.convolve(a, a)
        k_peak = np.argmax(c)
        # Find which elements contribute to the peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        contrib = np.zeros(n)
        contrib[lo:hi+1] = a[k_peak-hi:k_peak-lo+1][::-1]
        
        # Reduce top contributors, add to least contributors
        top_idx = np.where(contrib > 0)[0]
        if len(top_idx) == 0:
            break
        
        # Reduce the largest contributors
        red_amount = 0.001
        j = top_idx[np.argmax(contrib[top_idx])]
        if a[j] <= red_amount:
            red_amount = a[j] * 0.1
        if red_amount < 1e-10:
            break
        
        a[j] -= red_amount
        # Add to a position that contributes least to the peak
        zero_contrib = np.where(contrib == 0)[0]
        if len(zero_contrib) > 0:
            # Add to random position with zero contribution
            j2 = zero_contrib[rng.integers(len(zero_contrib))]
        else:
            # Add to position with minimum contribution
            j2 = top_idx[np.argmin(contrib[top_idx])]

        a = simplex_project(a)
        
        obj = objective(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def run_lbfgs_smooth(a0, time_limit=120):
    """L-BFGS on log-softmax parameterization with LSE smoothing."""
    n = len(a0)
    x = np.log(np.maximum(a0, 1e-15))
    best_a = a0.copy()
    best_obj = objective(best_a)
    start = time.monotonic()
    
    def lse_smooth(x_, tau):
        x_ = x_ - x_.max()
        e = np.exp(x_)
        a_ = e / e.sum()
        c_ = np.convolve(a_, a_)
        m = c_.max()
        return tau * np.log(np.sum(np.exp((c_ - m) / tau))) + m
    
    def lse_grad(x_, tau):
        x_ = x_ - x_.max()
        e = np.exp(x_)
        a_ = e / e.sum()
        c_ = np.convolve(a_, a_)
        m = c_.max()
        w = np.exp((c_ - m) / tau)
        w /= w.sum()
        # grad of smoothed max w.r.t. a_: 2 * correlate(w, a_, 'valid')
        grad_a = 2.0 * np.correlate(w, a_, mode='valid')
        # Chain rule for softmax
        dot = np.dot(a_, grad_a)
        return a_ * (grad_a - dot)
    
    taus = [0.1/n, 0.05/n, 0.02/n, 0.01/n]
    
    for tau in taus:
        if time.monotonic() - start > time_limit:
            break
        
        result = minimize(
            lambda x_: lse_smooth(x_, tau),
            x,
            jac=lambda x_: lse_grad(x_, tau),
            method='L-BFGS-B',
            options={'maxiter': 300, 'ftol': 1e-15, 'maxfun': 500}
        )
        x = result.x
        a_ = np.exp(x - x.max())
        a_ /= a_.sum()
        obj_ = objective(a_)
        if obj_ < best_obj:
            best_obj = obj_
            best_a = a_.copy()
    
    return best_a, best_obj

import numpy as np
import time
from scipy.optimize import minimize


def simplex_project(v):
    """Project v onto {a >= 0, sum(a) = 1} via water-filling."""
    n = len(v)
    d = np.sort(v)[::-1]
    cssv = np.cumsum(d)
    cond = d * np.arange(1, 1 + n) > (cssv - 1)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def weighted_subgradient(a, sigma_peaks=5.0, n_peaks=30):
    """Compute Gaussian-weighted subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    # Gaussian weights around the max
    weights = np.exp(-0.5 * ((c - c_max) / (sigma_peaks * c_max + 1e-15))**2)
    # Focus on top peaks
    threshold = c_max - 2.0 * sigma_peaks * c_max
    weights[c < threshold] = 0.0
    if weights.sum() < 1e-15:
        # Fallback: just the argmax
        weights = np.zeros_like(c)
        weights[np.argmax(c)] = 1.0
    else:
        # Keep top n_peaks among weighted
        top_idx = np.argsort(c)[::-1][:n_peaks]
        mask = np.zeros_like(c)
        mask[top_idx] = 1.0
        weights *= mask
        weights /= weights.sum()
    
    g = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        g[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g, c


def compute_objective(a):
    """Compute 2*n*max_b / S^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def run_subgradient(a0, time_limit=120, n_peaks=30, sigma=5.0, step0=0.08):
    """Projected subgradient with backtracking and periodic diversification."""
    a = simplex_project(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_objective(best_a)
    step = step0
    start = time.monotonic()
    stall_count = 0
    diversify_every = 2000
    it = 0
    
    while time.monotonic() - start < time_limit:
        it += 1
        g, c = weighted_subgradient(a, sigma_peaks=sigma, n_peaks=n_peaks)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Backtracking line search
        s = step
        improved = False
        for trial in range(5):
            a_trial = simplex_project(a - s * g)
            obj_trial = compute_objective(a_trial)
            if obj_trial < best_obj - 1e-12:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                step = min(step * 1.03, 0.4)
                stall_count = 0
                break
            elif obj_trial < compute_objective(a):
                a = a_trial
                step = min(step * 1.01, 0.4)
                s *= 0.7
            else:
                s *= 0.5
        if not improved:
            step *= 0.92
            stall_count += 1
        
        # Periodic diversification
        if stall_count > 200 or it % diversify_every == 0:
            noise_scale = 0.005 + 0.02 * min(stall_count / 500.0, 1.0)
            a = simplex_project(a + noise_scale * np.random.randn(len(a)))
            obj_a = compute_objective(a)
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()
            step = step0 * np.exp(-np.random.rand() * 0.5)
            stall_count = 0
        
        if step < 0.002:
            step = step0 * 0.1  # Reset
    
    return best_a, best_obj


def run_lbfgs_smooth(a0, time_limit=120):
    """L-BFGS on log-softmax parameterization with decreasing-temperature LSE."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x = np.log(a0)
    best_a = a0.copy()
    best_obj = compute_objective(best_a)
    start = time.monotonic()
    
    def lse_obj(x_):
        x_ = x_ - x_.max()
        e = np.exp(x_)
        a_ = e / e.sum()
        c_ = np.convolve(a_, a_)
        m = c_.max()
        tau = 0.05 / n
        shifted = (c_ - m) / tau
        shifted = np.clip(shifted, -500, 500)
        return tau * (m / tau + np.log(np.sum(np.exp(shifted))))
    
    def lse_obj_val(x_, tau):
        x_ = x_ - x_.max()
        e = np.exp(x_)
        a_ = e / e.sum()
        c_ = np.convolve(a_, a_)
        m = c_.max()
        shifted = (c_ - m) / tau
        shifted = np.clip(shifted, -500, 500)
        return tau * np.log(np.sum(np.exp(shifted))) + m
    
    def lse_grad(x_, tau):
        x_ = x_ - x_.max()
        e = np.exp(x_)
        a_ = e / e.sum()
        c_ = np.convolve(a_, a_)
        m = c_.max()
        shifted = (c_ - m) / tau
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w /= w.sum()
        grad_a = 2.0 * np.correlate(w, a_, mode='valid')
        dot = np.dot(a_, grad_a)
        return a_ * (grad_a - dot)
    
    for tau in [0.1 / n, 0.05 / n, 0.02 / n, 0.01 / n, 0.005 / n]:
        if time.monotonic() - start > time_limit:
            break
        try:
            result = minimize(
                lambda x_: lse_obj_val(x_, tau),
                x,
                jac=lambda x_: lse_grad(x_, tau),
                method='L-BFGS-B',
                options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': 600, 'maxls': 40}
            )
            x = result.x
        except Exception:
            break
        a_ = np.exp(x - x.max())
        a_ /= a_.sum()
        obj_ = compute_objective(a_)
        if obj_ < best_obj:
            best_obj = obj_
            best_a = a_.copy()
    
    return best_a, best_obj


def peak_shave(a0, time_limit=30, step=0.002):
    """Direct peak shaving: reduce top contributors to peak, add to non-contributors."""
    a = simplex_project(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_obj = compute_objective(best_a)
    start = time.monotonic()
    rng = np.random.default_rng(77)
    
    for it in range(200000):
        if time.monotonic() - start > time_limit:
            break
        n = len(a)
        c = np.convolve(a, a)
        # Top 3 peaks
        top3 = np.argsort(c)[::-1][:3]
        for k_peak in top3:
            lo = max(0, k_peak - n + 1)
            hi = min(n - 1, k_peak)
            contrib = np.zeros(n)
            contrib[lo:hi + 1] = a[k_peak - hi:k_peak - lo + 1][::-1]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            # Reduce top contributor
            top_pos = np.where(contrib > 0)[0]
            if len(top_pos) == 0:
                continue
            j_red = top_pos[np.argmax(contrib[top_pos])]
            amt = min(step * a.mean() * 10, a[j_red] * 0.3)
            if amt < 1e-12:
                continue
            # Add to position with least contribution to ALL peaks
            zero_mask = np.ones(n, dtype=bool)
            for kp in top3:
                lo2 = max(0, kp - n + 1)
                hi2 = min(n - 1, kp)
                zero_mask[lo2:hi2 + 1] = False
            candidates = np.where(zero_mask)[0]
            if len(candidates) == 0:
                candidates = top_pos[np.argmin(contrib[top_pos])]
                candidates = np.array([candidates])
            j_add = candidates[rng.integers(len(candidates))]
            a[j_red] -= amt
            a[j_add] += amt
            a = simplex_project(a)
        
        if it % 500 == 0:
            obj = compute_objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    final_obj = compute_objective(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    return best_a, best_obj


def make_diverse_initializations(n, rng, base_seq=None):

    inits.append(x / x.sum())
    
    # Cosine (1st harmonic)

    x = np.ones(n)

    s = best_sequence.sum()
    if s < 0.01:
        best_sequence = np.ones(1024) / 1024
        s = 1.0
    best_sequence = np.minimum(best_sequence, 1000.0)
    
    final_score = evaluate_sequence(best_sequence.tolist())
    print(f"\n{'='*50}")
    print(f"FINAL: score={final_score:.6f}, n={len(best_sequence)}")
    print(f"{'='*50}")
    
    return [float(v) for v in best_sequence]
