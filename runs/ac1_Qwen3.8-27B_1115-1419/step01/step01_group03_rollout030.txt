"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using projected subgradient method
with exact subgradients, FFT-based convolution, and adaptive step sizes.
Uses multiple initializations and a multi-phase approach for robust convergence.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def convolve_full(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    fa = fft(a, fft_size)
    c = ifft(fa * fa).real[:size]
    return c


def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    if np.all(v <= 0):
        return np.ones(n, dtype=np.float64) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(c, a, eps_frac=1e-4):

    Compute the subgradient of max(c) where c = conv(a,a).
    Uses the formula: dc_k/da_j = 2*a[k-j] if 0 <= k-j < n, else 0.
    The subgradient is the average over active peaks.
    
    Efficient computation: g[j] = 2 * sum_{k in S} a[k-j] * I(0 <= k-j < n)
    This is a cross-correlation between a and the indicator of S.
    """
    n = len(a)
    c_max = c.max()
    threshold = c_max - eps_frac * c_max
    S = np.where(c >= threshold)[0]
    
    if len(S) == 0:
        S = np.where(c == c_max)[0]
    
    # g[j] = 2 * sum_{k in S} a[k-j] * I(0 <= k-j < n)
    # = 2 * sum_{d=0}^{n-1} a[d] * I(j+d in S)
    # This is the cross-correlation of a and 1_S at lag j.
    
    # Use FFT: cross-correlation can be computed as ifft(conj(fft(a)) * fft(1_S))
    # But we need to be careful about the definition.
    # corr[a, b][j] = sum_i a[i] * b[i+j]
    # We want: g[j]/2 = sum_d a[d] * I_S[j+d] = corr[a, 1_S][j]
    
    # Using FFT: corr[a, b] = ifft(conj(fft(a, N)) * fft(b, N)) for appropriate N
    # But we need to handle the linear (not circular) correlation.
    
    # Actually, let's just compute it directly since S might be small.
    # For each k in S, g += 2 * a shifted by k (reversed).
    # g[j] += 2 * a[k-j] if 0 <= k-j < n
    
    g = np.zeros(n, dtype=np.float64)
    for k in S:
        # g[j] += 2*a[k-j] for j such that 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            # g[lo:hi] += 2 * a[k-lo : k-hi+1][::-1]
            # j = lo, lo+1, ..., hi-1
            # k-j = k-lo, k-lo-1, ..., k-(hi-1) = k-hi+1
            # So a[k-j] = a[k-lo], a[k-lo-1], ..., a[k-hi+1]
            # That's a[k-hi+1 : k-lo+1][::-1]
            g[lo:hi] += 2.0 * a[k-(hi-1): k-lo+1][::-1]
    
    g /= float(len(S))
    return g


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = convolve_full(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def run_projected_subgrad(a0, n, duration, rng, eps_frac=1e-4):
    """
    Run projected subgradient method on the simplex.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    a = a0.copy()
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    
    # Initialize step size
    eta = 0.5
    t = 1
    it = 0
    stuck = 0
    
    # Also track running average (for convergence of subgradient method)
    a_avg = a0.copy()
    
    while time.monotonic() < deadline:
        # Compute convolution
        c = convolve_full(a)
        c_max = c.max()
        score = 2.0 * n * c_max  # since sum(a) = 1
        
        # Track best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute subgradient
        g = compute_subgrad(c, a, eps_frac)
        
        # Step size schedule
        # Use eta_t = base_eta / sqrt(t)
        base_eta = 0.3
        eta_t = base_eta / np.sqrt(t)
        
        # Gradient step + projection
        a_new = project_simplex(a - eta_t * g)
        a = a_new
        
        # Update running average
        a_avg = (1.0 - 1.0/t) * a_avg + (1.0/t) * a
        
        t += 1
        it += 1
        
        # Check running average score periodically
        if it % 200 == 0:
            sa = a_avg.sum()
            if sa > 1e-15:
                a_avg_norm = a_avg / sa
                sc_avg = fast_score(a_avg_norm)
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = a_avg_norm.copy()
                    stuck = 0
        
        if it % 1000 == 0:
            sc_avg = fast_score(a_avg / max(a_avg.sum(), 1e-15))
            print(f"  PSG it={it}, current_score={score:.7f}, avg_score={sc_avg:.7f}, best={best_score:.7f}, stuck={stuck}")
            
            # Adaptive: if stuck for too long, restart
            if stuck > 2000:
                print(f"  PSG restart at it={it}")
                a = best_a + rng.normal(0, 0.01/np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-15:
                    a /= sa
                a_avg = a.copy()
                t = 1
                stuck = 0
    
    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def run_fista_exact(a0, n, duration, rng, eps_frac=1e-4):
    """
    Run FISTA-like accelerated projected subgradient with exact subgradients.
    """
    a = a0.copy()
    a_prev = a0.copy()
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    
    t_k = 1.0
    base_step = 0.3
    it = 0
    stuck = 0
    
    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_k
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy  # Renormalize to simplex
        
        # Compute subgradient at y
        c = convolve_full(y)
        g = compute_subgrad(c, y, eps_frac)
        
        # Step
        step = base_step / np.sqrt(t_k)
        a_new = project_simplex(y - step * g)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic check
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck > 500:
                # Reset momentum
                a_prev = best_a.copy()
                t_k = 1.0
                stuck = 0
                
            if stuck > 2000:
                # Full restart
                a = best_a + rng.normal(0, 0.02/np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-15:
                    a /= sa
                a_prev = a.copy()
                t_k = 1.0
                stuck = 0
                print(f"  FISTA restart at it={it}, score={best_score:.7f}")
        
        if it % 2000 == 0:
            sc = fast_score(a)
            print(f"  FISTA it={it}, score={sc:.7f}, best={best_score:.7f}")
    
    # Final
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

Minimize 2n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
Phase 1: Fast projected subgradient with top-peaks subgradient.

import numpy as np
import time
from scipy.fft import fft, ifft


def convolve_full(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = fft(a, fft_size)
    return ifft(fa * fa).real[:size]


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_from_peaks(c, a, n_peaks='all', eps_abs=0.0, eps_frac=0.0):
    """
    Compute subgradient of max(c) at a, where c = conv(a,a).
    
    If n_peaks == 'all': use all k with c[k] >= c_max - eps.
    If n_peaks is int: use top n_peaks.
    
    Returns gradient vector g of length n.
    """
    n = len(a)
    c_max = c.max()
    
    if n_peaks == 'all':
        threshold = c_max - max(eps_abs, eps_frac * c_max)
        active = np.where(c >= threshold)[0]
    else:
        active = np.argpartition(c, -n_peaks)[-n_peaks:]
    
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += a[k - (hi - 1): k - lo + 1][::-1]
    
    g *= 2.0 / len(active)
    return g
