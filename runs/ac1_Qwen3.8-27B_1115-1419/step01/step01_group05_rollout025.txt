    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]

def project_simplex(v):

    n = v.size

        return np.array([1.0])

    rho = np.arange(1, n + 1)

        tau = v - 1.0 / n
        return np.maximum(tau, 0.0)

Multi-method search for low-peak nonnegative self-convolutions.
Uses LSE gradient descent with Nesterov momentum, exact simplex projection,
peak shaving, and diverse multi-start with length scaling.

import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _project_simplex(v):
    """Exact Euclidean projection onto {a >= 0, sum(a) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho > (sv - 1.0))
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (sv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def _autocorr(a):
    n = a.size
    fs = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:2 * n - 1]

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return 1e10
    a = a / s
    n = a.size
    b = _autocorr(a)

    m = float(b.max())
    if beta <= 1.5:
        return m
    arg = beta * (b - m)
    arg = np.clip(arg, -700.0, 0.0)
