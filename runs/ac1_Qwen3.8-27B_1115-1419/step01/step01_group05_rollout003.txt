def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(a.size) / a.size
    return a / s

def _conv(a, b):
    """Full convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

    n = len(a)

"""
Search for low-peak nonnegative self-convolutions using LSE gradient descent
with periodic LP-based active-set updates.

The objective is score(a) = 2*n*max(a*a) / (sum(a))^2, which is scale-invariant.
We normalize to sum=1 and minimize 2*n*max(a*a).

Key techniques:
- Log-sum-exp smoothing of the max for differentiability
- FFT-based convolution and gradient computation
- Periodic LP update (solve max sum(b) s.t. (a*b) <= M, b>=0)
- Multiple parametric and structured initializations
- Adaptive step size, beta annealing, and restarts
"""

import numpy as np
import time
from scipy.optimize import linprog

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _normalize(a):
    """Clip to nonnegative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _fft_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _score(a):
    """Compute score = 2*n*max(a*a) assuming sum(a)=1."""
    n = len(a)
    b = _fft_conv(a, a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _lse_value(b, beta):
    """Stable LSE: (1/beta)*log(sum(exp(beta*b)))."""
    m = float(np.max(b))
    if beta < 2.0:
        return m
    e = np.exp(beta * (b - m))
    s = float(e.sum())
    if s < 1e-300:
        return m
    return m + float(np.log(s)) / beta


def _lse_and_p(b, beta):
    """Return (p_k proportional to exp(beta*b_k), lse_value)."""
    m = float(np.max(b))
    if beta < 2.0:
        p = np.ones(len(b), dtype=np.float64) / len(b)
        return p, m
    e = np.exp(beta * (b - m))
    s = float(e.sum())
    if s < 1e-300:
        e = np.ones(len(b), dtype=np.float64)
        s = float(len(b))
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _lse_gradient(a, p):
    """
    Gradient of sum_k p_k * (a*a)_k w.r.t. a.
    = 2 * cross_correlate(p, a) restricted to [0,n).
    """
    n = len(a)
    a_ext = np.zeros(2 * n - 1, dtype=np.float64)
    a_ext[:n] = a
    size = 1 << (2 * (2 * n - 1)).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a_ext, size)
    xcorr = np.fft.irfft(P * np.conj(A), size)[:2 * n - 1]
    return 2.0 * xcorr[:n]


def _lp_update(a, M, time_limit=20.0):
    """
    Given a (sum=1) and M = max(a*a), solve LP:
      maximize sum(b) s.t. b>=0, (a*b)_k <= M for all k.
    Returns normalized b if sum(b) > 1, else a.
    """
    n = len(a)
    # Build A: A[k,j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    c = -np.ones(n, dtype=np.float64)
    rhs = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            b_sol = np.maximum(res.x, 0.0)
            s = float(b_sol.sum())
            if s > 1.0 + 1e-8:
                return b_sol / s
            elif s > 1e-300:
                return b_sol / s
    except Exception:
        pass
    return a.copy()


def _make_exponential(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_symmetric_bow(n, strength):
    """High at ends, low in middle - bow-tie shape."""
    x = np.linspace(0.0, 1.0, n)
    mid = 0.5
    a = np.exp(-strength * (x - mid) ** 2 / (0.25 * (1 - 2 * mid + mid**2)))
    # Add end peaks
    a += 0.5 * (np.exp(-20 * x**2) + np.exp(-20 * (1 - x)**2))
    return _normalize(a)


def _make_two_bump(n, center_frac, width):
    x = np.linspace(0.0, 1.0, n)
    c1 = center_frac
    c2 = 1.0 - center_frac
    a = np.exp(-((x - c1) ** 2) / (2 * width ** 2))
    a += np.exp(-((x - c2) ** 2) / (2 * width ** 2))
    a += 0.01 * np.ones(n)
    return _normalize(a)


def _make_cubic(n):
    """a_i proportional to (i+1)*(n-1-i)*(something) for end-peaked shape."""
    x = np.linspace(0.0, 1.0, n)
    a = x * (1 - x)
    a = a ** 0.5  # soften
    # Actually for end-peaked: a = 1 - (2x-1)^2
    a = 1.0 - (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.01)
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.maximum(old, 0.0)
    s = old.sum()
    if s < 1e-300 or old.size == 0:
        return _make_exponential(new_n, 1.0)
    old = old / s
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old, right=0.0, left=0.0)
    return _normalize(new)


def _perturb(a, rng, sigma=0.25):
    noise = np.exp(rng.normal(0.0, sigma, size=a.size))
    return _normalize(a * noise)


def _block_perturb(a, rng, nblocks=3, strength=0.4):
    n = len(a)
    c = a.copy()
    block_size = max(2, n // (nblocks * 2))
    for _ in range(nblocks):
        start = rng.integers(0, max(1, n - block_size))
        end = min(n, start + block_size)
        c[start:end] *= np.exp(rng.normal(0, strength, size=end - start))
    return _normalize(c)


def _refine_sequence(a, deadline, rng, verbose=True):

    Main refinement: LSE gradient descent with periodic LP updates.
    Returns (best_a, best_score).
    """
    a = _normalize(a)
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    beta = 15.0
    alpha = 0.05
    fail = 0
    it = 0
    last_improve_it = 0
    lp_interval = 80

    while time.monotonic() < deadline:
        # Compute LSE surrogate gradient
        b = _fft_conv(a, a)
        maxb = float(np.max(b))
        p, smooth_old = _lse_and_p(b, beta)
        grad = _lse_gradient(a, p)

        # Center the gradient for simplex (remove mean component)
        grad_c = grad - float(np.mean(grad))
        gscale = float(np.max(np.abs(grad_c)))

        if gscale < 1e-300:
            a = _perturb(best_a, rng, 0.3)
            fail = 0
            it += 1
            continue

        # Mirror descent step in log space
        if alpha < 0.001:
            alpha = 0.05
        step = alpha / max(gscale, 1e-300)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - step * grad_c
        log_a_new -= float(np.max(log_a_new))
        a_new = np.exp(log_a_new)
        a_new = _normalize(a_new)

        score_new = _score(a_new)
        b_new = _fft_conv(a_new, a_new)
        smooth_new = _lse_value(b_new, beta)

            a = a_new
            alpha = min(2.0, alpha * 1.08)
            fail = 0
        else:
            alpha *= 0.5
            fail += 1

            best_score = score_new

                fail = 0

            alpha = 0.05
            fail = 0

        try:
            arr = np.asarray(gseq, dtype=np.float64).ravel()
            arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
            arr = np.maximum(arr, 0.0)
            if arr.size > 4 and arr.sum() > 1e-6:
                given = _normalize(arr)
                print(f"Loaded given sequence, len={given.size}, score={_score(given):.8f}", flush=True)
        except Exception:
            pass

    # Generate diverse initializations
    inits = []

    # From given sequence
    if given is not None:
        inits.append(_resample(given, n))
        inits.append(_perturb(_resample(given, n), rng, 0.1))

    # Exponential family
    for alpha in [0.8, 1.2, 2.0, 3.0]:
        inits.append(_make_exponential(n, alpha))

    # Symmetric bow-tie
    inits.append(_make_symmetric_bow(n, 3.0))
    inits.append(_make_symmetric_bow(n, 6.0))

    # Two-bump
    inits.append(_make_two_bump(n, 0.25, 0.1))
    inits.append(_make_two_bump(n, 0.3, 0.08))

    # Cubic end-peaked
    inits.append(_make_cubic(n))

    # Uniform (baseline score=2, but useful as fallback)
    inits.append(_normalize(np.ones(n)))

    # Smooth random
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(3, n // 10)
    F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = (f - f.min()) / (f.max() - f.min() + 1e-300)
    inits.append(_normalize(0.5 * _make_exponential(n, 1.0) + 0.5 * f))

    # Evaluate all initializations
    best_a = None
    best_score = 1e100
    for i, init in enumerate(inits):
        s = _score(init)
        if s < best_score:
            best_score = s
            best_a = init.copy()
    print(f"Best init: score={best_score:.10f}", flush=True)

    # Phase 1: Short optimization of top-5 inits
    init_scores = [( _score(init), init) for init in inits]
    init_scores.sort(key=lambda x: x[0])
    top_inits = [ic for _, ic in init_scores[:5]]

    phase1_deadline = time.monotonic() + max(30, budget * 0.15)
    per_init_time = (phase1_deadline - time.monotonic()) / len(top_inits)

    for i, init in enumerate(top_inits):
        if time.monotonic() > phase1_deadline - 5:
            break
        t_end = min(phase1_deadline, time.monotonic() + per_init_time)
        if time.monotonic() >= t_end:
            continue
        res_a, res_score = _refine_sequence(init, t_end, rng, verbose=False)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()
        print(f"  Phase1 init{i}: score={res_score:.10f} best={best_score:.10f}", flush=True)

    # Phase 2: Deep optimization of the best
    phase2_deadline = time.monotonic() + max(60, budget * 0.45)
    res_a, res_score = _refine_sequence(best_a.copy(), phase2_deadline, rng, verbose=True)
    if res_score < best_score:
        best_score = res_score
        best_a = res_a.copy()
    print(f"Phase2 done: best={best_score:.10f}", flush=True)

    # Phase 3: Try symmetric refinement
    if time.monotonic() < total_deadline - 60:
        t_end = min(total_deadline - 30, time.monotonic() + max(30, budget * 0.15))
        res_a, res_score = _refine_symmetric(best_a.copy(), t_end, rng)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()
            print(f"Symmetric improve: {best_score:.10f}", flush=True)
        else:
            print(f"Symmetric no improve: {res_score:.10f} vs {best_score:.10f}", flush=True)

    # Phase 4: Try different lengths (resample and optimize)
    if time.monotonic() < total_deadline - 40:
        for n2 in [128, 512, 1024]:
            if time.monotonic() > total_deadline - 30:
                break
            if n2 == n:
                continue
            init_r = _resample(best_a, n2)
            t_end = min(total_deadline - 20, time.monotonic() + max(20, budget * 0.08))
            res_a, res_score = _refine_sequence(init_r, t_end, rng, verbose=False)
            eff_score = _score(res_a)  # rescore at its own length
            # Compare: score is scale-invariant in n for smooth shapes,
            # but the actual evaluate_sequence uses the actual n
            if eff_score < best_score:
                best_score = eff_score
                best_a = res_a.copy()
                print(f"  n={n2} improve: {best_score:.10f}", flush=True)
            else:
                print(f"  n={n2} no improve: {eff_score:.10f}", flush=True)

    # Phase 5: Final deep polish
    if time.monotonic() < total_deadline - 10:
        res_a, res_score = _refine_sequence(best_a.copy(), total_deadline - 3.0, rng, verbose=True)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()
        print(f"Final: best_score={best_score:.10f}", flush=True)

    # Validate with evaluate_sequence
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)

    # Safety check
    try:
        eval_score = evaluate_sequence([float(x) for x in out])
        print(f"Final evaluate_sequence score: {eval_score:.10f}, len={len(out)}", flush=True)
    except Exception as e:
        print(f"evaluate_sequence failed: {e}", flush=True)

    return [float(x) for x in out]
