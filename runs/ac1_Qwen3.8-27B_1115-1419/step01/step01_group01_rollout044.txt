"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative a.
Uses LP-based active-set descent combined with smooth-max gradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p

def _score(a):
    """Compute the score using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 1e-300:
        return np.inf
    m = _next_pow2(2*n-1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F*F).real[:2*n-1]
    b = np.maximum(b, 0)
    bmax = b.max()
    if not np.isfinite(bmax):
        return np.inf
    return 2*n*bmax/(S*S)

def _score_normalized(p):
    """Score for a probability vector (sum=1). Returns 2*n*max(p*p)."""
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    m = _next_pow2(2*n-1)
    F = np.fft.fft(p, m)
    b = np.fft.ifft(F*F).real[:2*n-1]
    b = np.maximum(b, 0)
    return 2*n*b.max()

def _convolve_self(a, m):
    """Compute self-convolution of a using FFT."""
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F*F).real[:2*len(a)-1]
    return np.maximum(b, 0)

def _convolve_cross(a, b, m):
    """Compute cross-convolution of a and b using FFT."""
    Fa = np.fft.fft(a, m)
    Fb = np.fft.fft(b, m)
    c = np.fft.ifft(Fa*Fb).real[:len(a)+len(b)-1]
    return np.maximum(c, 0)

def _normalize(p):
    """Normalize to sum=1, non-negative."""
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(p))/len(p)
    return p/s

def _lp_descent_step(p, M, m):
    """
    Solve LP: max sum(g) s.t. (p*g)_k <= M for all k, g >= 0.
    Returns the LP solution g (normalized to sum=1).
    """
    n = len(p)
    L = 2*n - 1
    
    # Constraint matrix: A_k = (p*g)_k for k=0,...,L-1
    # (p*g)_k = sum_{i: 0<=i<n, 0<=k-i<n} p_i * g_{k-i}
    # So A[k, j] = p_{k-j} if 0 <= k-j < n, else 0
    # i.e., A[k, j] = p[k-j] if 0 <= k-j < n
    
    # Build constraint matrix
    A = np.zeros((L, n))
    for k in range(L):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = p[idx]
    
    # Maximize sum(g) s.t. A @ g <= M, g >= 0
    # linprog minimizes, so negate: minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-300:
        return None
    return g/s  # Normalize to sum=1

def _smooth_max_gradient(p, m, tau):
    """
    Compute the gradient of the smooth-max objective in log-space.
    Returns (score_estimate, gradient in x-space).
    """
    n = len(p)
    L = 2*n - 1
    self_conv = _convolve_self(p, m)
    bmax = self_conv.max()
    
    if bmax < 1e-300:
        return np.inf, np.zeros(n)
    
    # Smooth weights
    x = (self_conv - bmax) / tau
    x = np.clip(x, -50, 0)
    w = np.exp(x)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(L)
        w[np.argmax(self_conv)] = 1
    else:
        w = w/ws
    
    # Smooth max value
    smax = bmax + tau * np.log(ws)
    
    # Gradient of smooth-max w.r.t. x (log-space)
    # (∇c_k)_m = 2*p_m*p_{k-m} (if 0<=k-m<n)
    # ∇(smooth_max) = sum_k w_k * ∇c_k
    
    grad = np.zeros(n)
    # For each active lag k, add w_k * 2*p_m*p_{k-m}
    # This is 2 * (w * p) convolved appropriately... 
    # Actually, let's compute it as: grad_m = 2 * sum_k w_k * p_m * p_{k-m}
    # = 2 * p_m * sum_k w_k * p_{k-m}
    # = 2 * p_m * (w * p)_{m}  (cross-correlation of w and p, evaluated at m)
    
    # (w * p)_m = sum_k w_k * p_{k-m} (with p_j = 0 outside [0,n-1])
    # This is the cross-correlation of w (length L) and p (length n)
    # = (w reversed) * p, evaluated at appropriate position
    
    # Actually: sum_k w_k * p_{k-m} = (w * p)_{m} where the convolution is
    # sum_k w_k * p_{k-m}
    # Let's compute this via FFT: it's the cross-correlation.
    
    # w has length L = 2n-1, p has length n
    # sum_k w_k * p_{k-m}: for m in [0, n-1]
    # = cross-correlation of w and p at lag m
    
    wc = _cross_corr(w, p, m)  # length n + L - 1 = 3n - 2, we take [0:n]
    grad = 2 * p * wc[:n]

def _next_pow2(x):
    x = int(max(1, x))

def _score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-300:

    F = np.fft.fft(a, m)

    b = np.maximum(b, 0.0)
    bmax = float(b.max())

    a = np.ones(n)
    for k in range(1, kc + 1):
        a += (0.4 / k**0.9) * np.sin(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
    res.append(_normalize(a))
    # Random non-negative
    res.append(_normalize(rng.random(n) ** 2))
    res.append(_normalize(np.abs(rng.normal(0, 1, n)) + 0.1))
    return res


def _lp_step(p, m, deadline):
    """Solve LP: max sum(g) s.t. (p*g)_k <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(p)
    L = 2 * n - 1
    self_c = _self_conv(p, m)
    M = float(self_c.max())
    if M < 1e-300:
        return None
    # Build constraint matrix: A[k,j] = p[k-j] if 0<=k-j<n
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        # A[k, j] = p[k-j] for k in [j, j+n-1]
        lo = j
        hi = min(j + n, L)
        if lo < hi:
            A[lo:hi, j] = p[max(0, lo - j):hi - j]
    c = -np.ones(n)
    b_ub = np.full(L, M)
    bounds = [(0, None)] * n
    method = 'highs'
    opts = {'time_limit': max(1.0, deadline - time.monotonic())}
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method, options=opts)
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-300:
        return None
    return g / s


def _lp_step_active(p, m, deadline, frac=0.15):
    """LP using only active constraints (nears max). More efficient for large n."""
    n = len(p)
    L = 2 * n - 1
    self_c = _self_conv(p, m)
    M = float(self_c.max())
    if M < 1e-300:
        return None
    # Select active lags: those with self_c > (1-frac)*M
    threshold = (1.0 - frac) * M
    active = np.where(self_c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(self_c))])
    # Build constraint matrix for active lags only
    A = np.zeros((len(active), n), dtype=np.float64)
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        # A[i, j] = p[k-j] for j in [lo, hi)
        if lo < hi:
            A[i, lo:hi] = p[k - lo:k - lo + (hi - lo)][::-1]
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(1.0, deadline - time.monotonic())})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-300:
        return None
    return g / s


def _mirror_descent(p, m, max_seconds, seed, tau_init=0.1, tau_min=0.01,
                    top_frac=5.0, alpha_init=0.8, alpha_min=0.05,
                    stuck_threshold=60.0, use_subgrad=False):
    """Mirror descent (multiplicative weight) in log-space with smooth-max gradient."""
    rng = np.random.default_rng(seed)
    n = len(p)
    p = _normalize(p.copy())
    x = np.log(p + 1e-300)
    x = x - float(x.mean())

    best_p = p.copy()
    best_score = _score_fft(p)
    if not np.isfinite(best_score):
        best_p = np.ones(n) / n
        x = np.log(best_p)
        best_score = 2.0

    start = time.monotonic()
    deadline = start + max_seconds
    last_improve = start
    t_iter = 0
    check_interval = max(1, min(20, 600 // n))

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_seconds)

        # Stuck detection
        if time.monotonic() - last_improve > stuck_threshold:
            sigma = 0.15 + 0.30 * (1.0 - progress)
            x += rng.normal(0, sigma, n)
            x = x - float(x.mean())
            x = np.clip(x, -30, 30)
            last_improve = time.monotonic()
            p = _normalize(np.exp(x))
            continue

        # Compute self-convolution
        c = _self_conv(p, m)
        bmax = float(c.max())
        if bmax < 1e-300:
            break

        # Compute smooth weights on top lags
        tau = max(tau_min, tau_init * (1.0 - 0.9 * progress) + tau_min * progress)
        if rng.random() < 0.2:
            tau *= rng.uniform(0.4, 2.0)

        # Top-k selection for efficiency
        k_sel = max(1, int(top_frac * n))
        if k_sel < 2 * n - 1:
            # Weight only top k_sel lags
            top_idx = np.argpartition(c, -min(k_sel, len(c)))[:min(k_sel, len(c))]
            w = np.zeros(2 * n - 1)
            w[top_idx] = np.exp((c[top_idx] - bmax) / tau)
        else:
            w = np.exp(np.clip((c - bmax) / tau, -50, 0))

        ws = float(w.sum())
        if ws < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws

        # Compute gradient: grad_m = 2 * p_m * sum_k w_k * p_{k-m}
        # sum_k w_k * p_{k-m} is cross-correlation of w (len 2n-1) and p (len n)
        M_cross = _next_pow2((2 * n - 1) + n - 1)
        c_conv = np.fft.ifft(np.conj(np.fft.fft(w, M_cross)) * np.fft.fft(p, M_cross)).real
        # r_m = sum_k w_k * p_{k-m} = c_conv[(-m) % M_cross]
        r = c_conv[np.mod(-np.arange(n), M_cross)]
        grad = 2.0 * p * r

        # Normalize gradient
        grms = float(np.sqrt(np.mean(grad * grad)))
        if not np.isfinite(grms) or grms < 1e-14:
            x += rng.normal(0, 0.1, n)
            x = x - float(x.mean())
            x = np.clip(x, -30, 30)
            p = _normalize(np.exp(x))
            continue
        grad = grad / grms

        # Step size
        alpha = max(alpha_min, alpha_init * (1.0 - 0.7 * progress))
        if rng.random() < 0.1:
            alpha *= rng.uniform(0.5, 2.0)

        # Multiplicative weight update (mirror descent)
        x = x - alpha * grad
        x = x - float(x.mean())
        x = np.clip(x, -30, 30)
        p = _normalize(np.exp(x))
        t_iter += 1

        # Periodic score check
        if t_iter % check_interval == 0:
            sc = _score_fft(p)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_p = p.copy()
                last_improve = time.monotonic()

    sc = _score_fft(p)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p
    return best_p, float(best_score)


def _lp_phase(p, m, max_seconds, deadline):
    """Run LP-based iterations."""
    n = len(p)
    p = _normalize(p.copy())
    best_p = p.copy()
    best_score = _score_fft(p)
    if not np.isfinite(best_score):
        best_p = np.ones(n) / n
        best_score = 2.0

    step_sizes = [0.15, 0.10, 0.07, 0.05, 0.03, 0.02]
    ss_idx = 0
    last_improve = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        # Try LP step
        g = _lp_step_active(p, m, deadline, frac=0.12)
        if g is None:
            g = _lp_step(p, m, deadline)
        if g is None:
            break

        # Line search: try different step sizes
        improved = False
        for t in ([0.3, 0.2, 0.1, 0.05][:max(1, 3 - ss_idx)]):
            p_new = _normalize((1.0 - t) * p + t * g)
            sc = _score_fft(p_new)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                p = p_new
                best_score = sc
                best_p = p_new.copy()
                last_improve = time.monotonic()
                improved = True
                break

        if not improved:
            # Try smaller step or perturbation
            p_new = _normalize(p + 0.01 * (g - p))
            sc = _score_fft(p_new)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                p = p_new
                best_score = sc
                best_p = p_new.copy()
                last_improve = time.monotonic()
            else:
                # Stuck - try perturbation
                p = _normalize(p + 0.05 * np.random.default_rng(it).standard_normal(n))
                last_improve = time.monotonic()

        it += 1
        if it % 20 == 0:
            print("      LP iter %d, score=%.6f" % (it, best_score))

    return best_p, float(best_score)


def _spectral_init(n, rng):
    """Try to initialize by making |FFT| as flat as possible."""
    m = _next_pow2(2 * n - 1)
    # Target: flat magnitude spectrum
    # a_k >= 0, a_k = 0 for k >= n
    # In frequency: F_m = sum_{k=0}^{n-1} a_k exp(-2pi i m k / m)
    # We want |F_m| to be constant for all m
    # This is the "flat spectrum" or "CAZAC-like" condition
    # For non-negative a, this is hard. Let's try iterative water-filling.

    p = np.ones(n) / n
    for it in range(50):
        F = np.fft.fft(p, m)
        mag = np.abs(F)
        mag_max = mag.max()
        if mag_max < 1e-10:
            break
        # Try to reduce the peaks in the magnitude spectrum
        # Gradient of max(|F|) w.r.t. a: related to the phase of F at the peak
        peak_idx = np.argmax(mag)
        # F_peak = sum p_k exp(-2pi i peak_idx k / m)
        # d|F_peak|/dp_k = Re(exp(2pi i peak_idx k / m) * conj(F_peak) / |F_peak|)
        #                = Re(exp(2pi i peak_idx k / m) * conj(F_peak)) / |F_peak|
        phase = np.exp(2.0j * np.pi * peak_idx * np.arange(n) / m)
        grad = np.real(phase * np.conj(F[peak_idx])) / mag_max
        # Descent: p <- p - eta * grad, then project to simplex
        eta = 0.1
        p_new = p - eta * grad
        p_new = _normalize(p_new)
        if np.allclose(p_new, p, atol=1e-8):
            p = p_new
            break
        p = p_new
    return p


def _adam_optimize(p_init, n, m, max_seconds, seed):
    """Adam optimization in log-space (similar to previous but with improvements)."""
    rng = np.random.default_rng(seed)
    x = np.log(np.maximum(p_init, 1e-300))
    x = x - float(x.mean())
    x = np.clip(x, -20, 20)

    p = _normalize(np.exp(x))
    best_p = p.copy()
    best_score = _score_fft(p)
    if not np.isfinite(best_score):
        best_p = np.ones(n) / n
        x = np.log(best_p)
        best_score = 2.0

    b1, b2, eps = 0.9, 0.999, 1e-8
    mu = np.zeros(n)
    nu = np.zeros(n)
    t = 0
    start = time.monotonic()
    deadline = start + max_seconds
    last_improve = start
    check_interval = max(2, min(30, 900 // n))

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_seconds)

        if time.monotonic() - last_improve > min(90.0, 0.3 * max_seconds):
            sigma = 0.10 + 0.30 * (1.0 - progress)
            x += rng.normal(0, sigma, n)
            x = x - float(x.mean())
            x = np.clip(x, -25, 25)
            p = _normalize(np.exp(x))
            last_improve = time.monotonic()
            continue

        c = _self_conv(p, m)
        bmax = float(c.max())
        if bmax < 1e-300:
            break

        # Adaptive temperature
        tau = max(0.02, 0.15 * (1.0 - 0.85 * progress) + 0.02)
        if rng.random() < 0.2:
            tau *= rng.uniform(0.3, 3.0)

        # Smooth weights
        xw = (c - bmax) / tau
        xw = np.clip(xw, -50, 0)
        w = np.exp(xw)
        ws = float(w.sum())
        if ws < 1e-300:
            break
        w = w / ws

        # Gradient via cross-correlation
        M_cross = _next_pow2((2 * n - 1) + n - 1)
        c_conv = np.fft.ifft(np.conj(np.fft.fft(w, M_cross)) * np.fft.fft(p, M_cross)).real
        r = c_conv[np.mod(-np.arange(n), M_cross)]
        grad = 2.0 * p * r

        # Also add a "spread" term to encourage flatness
        smax = bmax + tau * np.log(ws)
        target = smax / (2 * n - 1)  # Average level
        spread_w = np.clip((c - target) / (tau * 2), -20, 20)
        spread_w = np.exp(spread_w)
        sws = float(spread_w.sum())
        if sws > 1e-300:
            spread_w = spread_w / sws
            c_conv2 = np.fft.ifft(np.conj(np.fft.fft(spread_w, M_cross)) * np.fft.fft(p, M_cross)).real
            r2 = c_conv2[np.mod(-np.arange(n), M_cross)]
            grad += 0.3 * 2.0 * p * r2

        grms = float(np.sqrt(np.mean(grad * grad)))
        if not np.isfinite(grms) or grms < 1e-14:
            x += rng.normal(0, 0.05, n)
            x = x - float(x.mean())
            x = np.clip(x, -25, 25)
            p = _normalize(np.exp(x))
            continue
        grad = grad / grms

        t += 1
        mu = b1 * mu + (1 - b1) * grad
        nu = b2 * nu + (1 - b2) * (grad * grad)
        mhat = mu / (1 - b1**t)
        vhat = nu / (1 - b2**t)
        lr = 0.12 * (1.0 - 0.6 * progress)
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.5, 0.5)
        x = x - lr * step
        x = x - float(x.mean())
        x = np.clip(x, -25, 25)
        p = _normalize(np.exp(x))

        if t % check_interval == 0:
            sc = _score_fft(p)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_p = p.copy()
                last_improve = time.monotonic()

    sc = _score_fft(p)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p
    return best_p, float(best_score)


def _process_length(n, deadline, prev_a, rng):
    """Process a single length: try multiple methods and keep the best."""
    n = int(max(3, n))
    m = _next_pow2(2 * n - 1)
    now = time.monotonic()
    if now >= deadline:
        return None, np.inf

    best_a = None
    best_score = np.inf

    # Build initial candidates
    inits = []
    if prev_a is not None:
        inits.append(_interp_to(prev_a, n))
    inits.append(_spectral_init(n, rng))
    inits.extend(_param_inits(n, rng))

    # Score and deduplicate
    scored = []
    seen = []
    for a in inits:
        a = _normalize(a)
        sc = _score_fft(a)
        if not np.isfinite(sc):
            continue
        div = True
        for b in seen:
            if float(np.sqrt(np.sum((a - b)**2))) < 0.04:
                div = False
                break
        if div:
            scored.append((sc, a))
            seen.append(a)
            if len(seen) >= 12:
                break

    scored.sort(key=lambda x: x[0])
    if scored:
        best_score = scored[0][0]
        best_a = scored[0][1].copy()

    cands = [a for _, a in scored[:4]]
    if not cands:
        cands = [np.ones(n) / n]

    # Phase 1: LP (for n <= 2048)
    if n <= 2048:
        now = time.monotonic()
        rem = deadline - now
        if rem > 30:
            lp_time = min(rem * 0.35, 300.0)
            lp_deadline = now + lp_time
            print("    LP phase n=%d budget=%.0fs" % (n, lp_time))
            for a in cands[:2]:
                now = time.monotonic()
                if now >= lp_deadline:
                    break
                a_new, sc_new = _lp_phase(a, m, lp_deadline - now, lp_deadline)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new

    # Phase 2: Adam optimization
    now = time.monotonic()
    rem = max(0.0, deadline - now - 10)
    if rem > 20:
        print("    Adam phase n=%d budget=%.0fs" % (n, rem))
        # Distribute time among candidates
        n_cands = min(3, len(cands))
        per_cand = rem / n_cands
        for i in range(n_cands):
            now = time.monotonic()
            if now >= deadline - 10:
                break
            a_start = cands[i] if i < len(cands) else best_a
            if i > 0 and best_a is not None:
                a_start = _normalize(0.5 * a_start + 0.5 * best_a)
            seed = int(rng.integers(0, 2**31 - 1))
            a_new, sc_new = _adam_optimize(a_start, n, m, max(10.0, rem - (time.monotonic() - now)), seed)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new
                print("      Adam cand %d improved: %.6f" % (i, best_score))

    # Phase 3: Mirror descent polishing
    now = time.monotonic()
    rem = max(0.0, deadline - now - 5)
    if rem > 15 and best_a is not None:
        print("    Mirror descent phase n=%d budget=%.0fs" % (n, rem))
        seed = int(rng.integers(0, 2**31 - 1))
        a_new, sc_new = _mirror_descent(best_a, m, rem, seed,
                                        tau_init=0.05, tau_min=0.01,
                                        alpha_init=0.5, alpha_min=0.03)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new
            print("      Mirror descent improved: %.6f" % best_score)

    # Phase 4: Final LP polish if time allows
    if n <= 2048 and best_a is not None:
        now = time.monotonic()
        rem = max(0.0, deadline - now - 3)
        if rem > 20:
            print("    Final LP polish n=%d budget=%.0fs" % (n, rem))
            a_new, sc_new = _lp_phase(best_a, m, rem, now + rem)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new

    if best_a is not None:
        best_a = _normalize(best_a)
    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(convolve(a,a))/sum(a)^2.
    Uses LP-based active-set descent, Adam in log-space, mirror descent,
    and spectral initialization across multiple sequence lengths.
    """
    seed = int(seed)
    budget_s = float(max(60.0, budget_s))
    start = time.monotonic()
    global_deadline = start + max(60.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    # Load previous best
    prev = None
    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0 and np.isfinite(arr).all():
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 0.0:
                prev = arr / float(arr.sum())
    except Exception:
        pass

    best_a = prev
    best_score = _score_fft(prev) if prev is not None else np.inf
    print("Initial score: %.6f (n=%d)" % (best_score, prev.size if prev is not None else 0))

    # Try multiple lengths with increasing budget
    # Allocate time proportionally, giving more to intermediate sizes
    base = max(60.0, budget_s - 30.0)
    stage_defs = [
        (1024, 0.10),
        (2048, 0.30),
        (3072, 0.15),
        (4096, 0.25),
        (6144, 0.12),
        (8192, 0.08),
    ]

    for n, frac in stage_defs:
        now = time.monotonic()
        if now + 30.0 >= global_deadline:
            break
        stage_time = base * frac
        stage_deadline = min(global_deadline - 5.0, now + stage_time)
        print("Stage n=%d budget=%.0fs current_best=%.6f" % (n, stage_time, best_score))
        a_new, sc_new = _process_length(int(n), stage_deadline, best_a, rng)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_a = a_new
            best_score = sc_new
        print("After n=%d best=%.6f" % (n, best_score))

    # Final: try the best length found one more time with remaining budget
    now = time.monotonic()
    rem = max(0.0, global_deadline - now - 5)
    if rem > 30 and best_a is not None:
        n_final = len(best_a)
        m = _next_pow2(2 * n_final - 1)
        print("Final refinement n=%d budget=%.0fs" % (n_final, rem))
        seed = int(rng.integers(0, 2**31 - 1))
        if n_final <= 2048:
            a_new, sc_new = _lp_phase(best_a, m, rem, now + rem)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new
            now = time.monotonic()
            rem2 = max(0.0, global_deadline - now - 5)
            if rem2 > 10:
                a_new, sc_new = _mirror_descent(best_a, m, rem2, seed,
                                                tau_init=0.03, tau_min=0.005,
                                                alpha_init=0.3, alpha_min=0.02)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new
        else:
            a_new, sc_new = _mirror_descent(best_a, m, rem, seed,
                                            tau_init=0.05, tau_min=0.01,
                                            alpha_init=0.5, alpha_min=0.03)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new

    if best_a is None:
        best_a = np.ones(512, dtype=np.float64)

    best_a = _normalize(best_a)
    final_score = _score_fft(best_a)
    print("Final FFT score=%.6f n=%d" % (final_score, len(best_a)))

    # Validate with exact evaluator if possible
    try:
        if len(best_a) <= 6000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Exact score=%.6f" % exact)
            if not np.isfinite(exact):
                best_a = np.ones(512, dtype=np.float64)
                final_score = 2.0
    except Exception as e:
        print("Validation: %s" % e)

    return [float(x) for x in best_a]
