import numpy as np

def project_simplex_fast(v):
    """Fast projection onto probability simplex via clip-and-normalize (KL projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        n = len(v)
        return np.full(n, 1.0 / n, dtype=np.float64)
    return v / s


def project_simplex_exact(v):
    """Exact Euclidean projection onto the standard simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def fft_grad_active(a, s):
    """
    Compute grad[i] = 2 * sum_k s[k] * a[k-i] for i=0..n-1.
    Here s is the active set indicator (length 2n-1), a has length n.
    Uses FFT cross-correlation.
    """
    n = len(a)
    # Need N >= (2n-1) + n - 1 = 3n-2 for linear cross-correlation
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    Fs = np.fft.rfft(s, N)
    Fa = np.fft.rfft(a, N)
    R = np.fft.irfft(Fs * np.conj(Fa), N)
    return 2.0 * R[:n]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    C = fft_conv_self(a)
    return 2.0 * n * C.max()

    plateau[center + width - edge:center + width] = np.linspace(1, 0, edge)
    if plateau.sum() > 0:
        candidates.append(plateau / plateau.sum())
        names.append("plateau")

    # --- Evaluate all candidates ---
    print("\n=== Evaluating initializations ===")
    best_cand
    = None
    best_cand_score = np.inf
    scores = []
    for c, nm in zip(candidates, names):
        sc = compute_score(c)
        scores.append((sc, nm))
        print(f"  {nm:20s}: {sc:.8f}")
        if sc < best_cand_score:
            best_cand_score = sc
            best_cand = c.copy()

    scores.sort()
    print(f"\nBest initial: {scores[0][1]} with score {scores[0][0]:.8f}")

    # --- Phase 1: Main FISTA from best candidate ---
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        print(f"\n=== Phase 1: FISTA from '{scores[0][1]}' ({time_left:.0f}s) ===")
        best_a, best_score = run_fista_fft(best_cand, time_left - 300.0, rng,
                                            initial_step=0.4, use_exact_proj=False)
        print(f"Phase 1 result: {best_score:.8f}")
    else:
        best_a = best_cand.copy()
        best_score = best_cand_score

    # --- Phase 2: Try second-best candidate if time allows ---
    time_left = total_deadline - time.monotonic()
    if time_left > 120 and scores[1][0] < best_score + 0.05:
        # The second best might lead to a different basin (though convex, numerical issues)
        second_cand = None
        for c, nm in zip(candidates, names):
            if nm == scores[1][1]:
                second_cand = c
                break
        if second_cand is not None:
            print(f"\n=== Phase 2: FISTA from '{scores[1][1]}' ({time_left - 60:.0f}s) ===")
            alt_a, alt_score = run_fista_fft(second_cand, time_left - 60.0, rng,
                                              initial_step=0.35, use_exact_proj=False)
            if alt_score < best_score:
                best_a = alt_a
                best_score = alt_score
                print(f"Phase 2 improved: {best_score:.8f}")
            else:
                print(f"Phase 2 no improvement: {alt_score:.8f} vs {best_score:.8f}")

    # --- Phase 3: Polishing ---
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        print(f"\n=== Phase 3: Polishing ({time_left:.0f}s) ===")
        best_a, best_score = run_polish(best_a, time_left - 30.0, rng)
        print(f"Polish result: {best_score:.8f}")

    # --- Phase 4: Final local search with random perturbations ---
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        print(f"\n=== Phase 4: Local search ({time_left:.0f}s) ===")
        for trial in range(20):
            if time.monotonic() > total_deadline - 30:
                break
            # Random perturbation
            scale = 0.01 / np.sqrt(n) * (1.0 - trial / 20.0)
            a_pert = best_a + rng.normal(0.0, scale, n)
            a_pert = project_simplex_fast(a_pert)
            sc = compute_score(a_pert)
            if sc < best_score:
                best_score = sc
                best_a = a_pert.copy()
                print(f"  Perturb {trial}: improved to {best_score:.8f}")

        # Run short FISTA from best perturb
        time_left = total_deadline - time.monotonic()
        if time_left > 30:
            best_a, best_score = run_fista_fft(best_a, time_left - 20.0, rng,
                                                initial_step=0.2, use_exact_proj=False)
            print(f"  Post-perturb FISTA: {best_score:.8f}")

    # --- Final verification ---
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.8f}")

    # Fallback if something went wrong
    if not np.isfinite(final_score) or final_score > best_cand_score + 0.1:
        print(f"Falling back to best candidate: {best_cand_score:.8f}")
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_cand]
        final_score = best_cand_score

    print(f"=== FINAL SCORE: {final_score:.8f} (n={n}) ===")
    return final_list
