"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses mirror descent with adaptive step size, multi-resolution search,
diverse initializations, and periodic annealing kicks to escape local minima.
"""

import numpy as np
import time
import math


def normalize_sum1(a):
    """Normalize non-negative array to have sum exactly 1."""
    a = np.array(a, dtype=np.float64, copy=True)
    a = np.maximum(a, 0.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()
    if s < 1e-30:
        a[:] = 1.0 / len(a)
    else:
        a /= s
    return a


def fft_conv(a):
    """Linear convolution a * a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.empty(0)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(c, 0.0)


def score_array(a):
    """Fast score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def conv_subgrad(a, c, K=1):
    """
    Subgradient of the max of top-K convolution coefficients w.r.t. a.
    For active index k: grad[j] = 2*a[k-j] if 0<=k-j<n else 0.
    """
    n = len(a)
    L = len(c)
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    inv = 1.0 / len(idx)
    grad = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            src = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += inv * a[src]
    return 2.0 * grad


def mirror_search(a, time_budget, seed=0, verbose=False):

    Minimizes max_k (a*a)_k.

    t0 = time.monotonic()

    c = fft_conv(a)
    best_score = c.max()
    best_a = a.copy()
    cur_score = best_score
    
    eta = 0.15
    patience = 0
    kick_cd = 0
    it = 0
    
    while time.monotonic() < deadline:

        kick_cd = max(0, kick_cd - 1)

        K = 3 if it < 500 else 1
        grad = conv_subgrad(a, c, K)
        
        # Update via mirror descent
        log_step = -eta * grad
        # Clip to avoid underflow/overflow
        log_step = np.clip(log_step, -5.0, 5.0)
        a_new = a * np.exp(log_step)
        a_new = normalize_sum1(a_new)
        
        c_new = fft_conv(a_new)
        new_score = c_new.max()

            best_a = a_new.copy()
            a = a_new
            c = c_new

    final_score = 2.0 * n * best_score  # since sum=1, score = 2n*max_c
    return best_a, final_score


def upsample(a, new_n):
    """Interpolate sequence to new length, renormalize to sum=1."""
    m = len(a)
    new_n = int(new_n)
    if m == 0:
        return np.ones(new_n) / new_n
    if m == new_n:
        return normalize_sum1(a)
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, new_n)
    out = np.interp(x_new, x_old, a)
    return normalize_sum1(out)


def gen_harmonic(n, rng, max_freq=8, max_amp=1.5):
    """Random low-frequency cosine sum, exponentiated."""
    t = np.linspace(0, 1, n)
    z = np.zeros(n)
    for m in range(1, max_freq + 1):
        amp = rng.uniform(-max_amp, max_amp) / (m ** 0.6)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * m * t + ph)
    z -= z.max()
    a = np.exp(z)
    return normalize_sum1(a)


def gen_geometric(n, ratio, rng):
    """Geometric (exponential decay) sequence."""
    a = ratio ** np.arange(n)
    if rng.random() < 0.5:
        a = a[::-1]
    return normalize_sum1(a)


def gen_ramp(n, rng, curvature=0.0):
    """Linear or quadratic ramp."""
    t = np.linspace(0, 1, n)
    a = 1.0 + curvature * (2 * t - 1) ** 2 + 1.5 * (2 * t - 1)
    a = np.maximum(a, 0.01)
    if rng.random() < 0.5:
        a = a[::-1]
    return normalize_sum1(a)


def gen_two_block(n, frac, ratio, rng):
    """Two-block sequence."""
    m = int(round(frac * n))
    m = max(1, min(n - 1, m))
    a = np.empty(n)
    a[:m] = ratio
    a[m:] = 1.0
    if rng.random() < 0.5:
        a = a[::-1]
    return normalize_sum1(a)

    seq = np.asarray(best_seq, dtype=np.float64)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.maximum(seq, 0.0)
    seq = np.minimum(seq, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return [1.0] * 1024
    
    return [float(x) for x in seq]
