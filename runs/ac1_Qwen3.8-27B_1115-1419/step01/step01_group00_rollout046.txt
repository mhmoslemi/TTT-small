= sum_{k=m}^{min(m+n-1, 2n-2)} w[k] * a[k-m]

This is exactly what we want! (The gradient at position m is 2 times this sum.)

In FFT: circular cross-correlation is ifft(fft(W) * conj(fft(A))).

And g = 2 * g_full (after subtracting the mean to maintain the sum constraint).

    """Full self-convolution of a using FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:

    fa = fft(a, N)

def fft_corr(w, a):

    n = len(a)
    N = 1
    while N < 2*n - 1:

import numpy as np
from numpy.fft import fft, ifft
import time

def _fft_conv(a):
    n = len(a)
    N = 1 << (2*n - 2).bit_length()  # next power of 2 >= 2n-1
    fa = fft(a, N)
    return ifft(fa * fa).real[:2*n-1]

def _fft_grad(c_weights, a):
    """Gradient: g[m] = 2 * sum_k c_weights[k] * a[k-m] for valid indices."""
    n = len(a)
    N = 1 << (2*n - 2).bit_length()
    fa = fft(a, N)
    fw = fft(c_weights, N)
    return ifft(fw * np.conj(fa)).real[:n] * 2.0

def _project(a, n):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n, dtype=np.float64)
    return a * (n / s)

def _obj(a):
    n = len(a)
    c = _fft_conv(a)
    return 2.0 * c.max() / n  # assuming sum(a) = n

def _soft_run(a, n, deadline, tag, tau0=None, eta0=0.5):
    best = a.copy()
    best_o = _obj(best)
    if tau0 is None:
        tau = n * 0.1
    else:
        tau = tau0
    eta = eta0
    it = 0
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        mc = c.max()
        o = 2.0 * mc / n
        if o < best_o - 1e-12:
            best_o = o
            best = a.copy()
        
        # Soft-max weights (log-sum-exp trick)
        cm = c / tau
        cm -= cm.max()
        w = np.exp(cm)
        w /= w.sum()
        
        # Gradient
        g = _fft_grad(w, a)
        g -= g.mean()
        
        gn = np.abs(g).max()
        if gn < 1e-14:
            break
        g /= gn
        
        a = _project(a - eta * g, n)
        
        # Schedule
        tau = max(tau * 0.99995, n * 0.002)
        if it % 5000 == 4999:
            eta = max(eta * 0.95, 0.01)
        it += 1
        if it % 10000 == 0:

    return best, best_o

def _hard_run(a, n, deadline, tag, K=1):
    best = a.copy()
    best_o = _obj(best)
    eta = 0.1
    it = 0

        mc = c.max()

            best_o = o
            best = a.copy()

            k = int(np.argmax(c))
            g = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            g[js:je] = 2.0 * a[k - np.arange(js, je)]
        else:

            g = np.zeros(n)
            for k in idx:

                je = min(n, k + 1)
                g[js:je] += 2.0 * a[k - np.arange(js, je)]
        
        g /= K  # normalize
        g -= g.mean()
        gn = np.abs(g).max()
        if gn < 1e-14:

        g /= gn
        
        a = _project(a - eta * g, n)

10-100x more iterations than direct methods.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def _next_pow2(m):
    return 1 << (m - 1).bit_length() if m > 1 else 2


def _fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = fft(a, N)
    return ifft(fa * fa).real[:2 * n - 1]


def _fft_corr(w, a):
    """g[m] = 2 * sum_k w[k]*a[k-m] for m=0..n-1 (gradient of weighted conv max)."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = fft(a, N)
    fw = fft(w, N)
    return ifft(fw * np.conj(fa)).real[:n] * 2.0


def _project(a, n):
    """Project to {x >= 0, sum(x) = n}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n, dtype=np.float64)
    return a * (n / s)


def _obj_val(c, n):
    """Objective from convolution values (assumes sum(a)=n)."""
    return 2.0 * c.max() / n


def _softmax_weights(c, tau):
    """Stable softmax of c/tau."""
    x = c / tau
    x -= x.max()
    w = np.exp(x)
    s = w.sum()
    if s < 1e-300:
        w = np.ones_like(c) / len(c)
    else:
        w /= s
    return w


def _soft_run(a, n, deadline, tag, tau_start=None, eta_start=0.4):
    """Phase 1: Soft gradient descent with annealing."""
    best = a.copy()
    best_o = _obj_val(_fft_conv(best), n)
    tau = tau_start if tau_start else n * 0.08
    tau_min = n * 0.003
    eta = eta_start
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        o = _obj_val(c, n)
        if o < best_o - 1e-12:
            best_o = o
            best = a.copy()
            stall = 0
        else:
            stall += 1
        
        w = _softmax_weights(c, tau)
        g = _fft_corr(w, a)
        g -= g.mean()
        gn = np.abs(g).max()
        if gn < 1e-14:
            break
        g /= gn
        
        a = _project(a - eta * g, n)
        
        # Annealing schedule
        if tau > tau_min:
            tau = max(tau * 0.99998, tau_min)
        if stall > 3000:
            eta = max(eta * 0.7, 0.01)
            stall = 0
        
        it += 1
        if it % 20000 == 0:
            print(f"  [{tag}] it={it} obj={o:.6f} tau={tau:.1f} eta={eta:.3f}")
    
    return best, best_o


def _hard_run(a, n, deadline, tag, K=3, eta_start=0.08):
    """Phase 2: Hard top-K gradient descent."""
    best = a.copy()
    best_o = _obj_val(_fft_conv(best), n)
    eta = eta_start
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        o = _obj_val(c, n)
        if o < best_o - 1e-12:
            best_o = o
            best = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Top-K gradient
        if K == 1:
            k = int(np.argmax(c))
            g = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            idx = np.arange(js, je)
            g[idx] = 2.0 * a[k - idx]
        else:
            topk = np.argpartition(c, -K)[-K:]
            # Sort topk by value (descending)
            topk = topk[np.argsort(c[topk])[::-1]]
            g = np.zeros(n)
            for k in topk:
                js = max(0, k - n + 1)
                je = min(n, k + 1)
                idx = np.arange(js, je)
                g[idx] += 2.0 * a[k - idx]
            g /= K
        
        g -= g.mean()
        gn = np.abs(g).max()
        if gn < 1e-14:
            break
        g /= gn
        
        a = _project(a - eta * g, n)
        
        if stall > 5000:
            eta = max(eta * 0.7, 0.003)
            stall = 0
        if it % 20000 == 0:
            eta = max(eta * 0.98, 0.003)
        
        it += 1
        if it % 20000 == 0:
            print(f"  [{tag}] it={it} obj={o:.6f} K={K} eta={eta:.4f}")
    
    return best, best_o


def _lp_step(a, n, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, ||a * g||_inf <= M.
    Returns the normalized direction g (sum = n).
    """
    from scipy.optimize import linprog
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    
    # For efficiency, build as sparse
    # Row k has non-zeros at positions j where 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # Clamped to [0, n-1]
    
    # Use scipy sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        for j in range(js, je):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: min -sum(g) s.t. A @ g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        g = res.x
        s = g.sum()
        if s > 1e-10:
            return g / s * n
    return None


def _lp_run(a, n, deadline, tag):
    """Phase 3: LP-based active set method."""
    best = a.copy()
    c = _fft_conv(a)
    best_o = _obj_val(c, n)
    t = 0.1  # blending parameter
    
    while time.monotonic() < deadline - 30:
        c = _fft_conv(a)
        M = c.max()
        
        g = _lp_step(a, n, M)
        if g is None:
            break
        
        # Blend
        a_new = (1 - t) * a + t * g
        a_new = _project(a_new, n)
        
        c_new = _fft_conv(a_new)
        o_new = _obj_val(c_new, n)
        
        if o_new < best_o - 1e-12:
            best_o = o_new
            best = a_new.copy()
            a = a_new
        else:
            t = max(t * 0.5, 0.01)
            a = (1 - t) * a + t * g
            a = _project(a, n)
        
        print(f"  [{tag}] LP step: obj={o_new:.6f} t={t:.3f}")
    
    return best, best_o


def _perturb(a, n, rng, sigma=0.1, width_frac=0.05):
    """Random block perturbation."""
    b = a.copy()
    w = max(2, int(n * width_frac))
    s = int(rng.integers(0, n - w + 1))
    b[s:s+w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _project(b, n)

FFT-accelerated multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. FFT-based convolution and gradient computation (O(n log n) vs O(n^2))

import numpy as np

def _next_pow2(m):
    """Next power of 2 >= m."""
    if m <= 1:

def _fft_conv(a):
    """Full self-convolution of a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    N = _next_pow2(2 * n - 1)

def _fft_corr(w, a):

    Compute g[m] = 2 * sum_k w[k] * a[k-m] for m=0..n-1.
    This is the gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    """
    n = len(a)

    a = np.maximum(a, 0.0)

    eta = eta_start
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        o = _obj_from_conv(c, n)
        if o < best_o - 1e-13:
            best_o = o
            best = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Top-K gradient
        if K <= len(c):
            topk = np.argpartition(c, -K)[-K:]
        else:
            topk = np.arange(len(c))
        
        g = np.zeros(n, dtype=np.float64)
        for k in topk:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            if js < je:
                idx = np.arange(js, je)
                g[idx] = g[idx] + 2.0 * a[k - idx]
        g /= max(K, 1)
        g -= g.mean()
        
        gn = np.abs(g).max()
        if gn < 1e-14:
            break
        g = g / gn
        
        a = _project(a - eta * g, n)
        
        if stall > 8000:
            eta = max(eta * 0.7, 0.002)
            stall = 0
        if it % 50000 == 0 and it > 0:
            eta = max(eta * 0.97, 0.002)
        
        it += 1
        if it % 50000 == 0:
            print(f"  [{tag}] it={it} obj={o:.6f} K={K} eta={eta:.5f}")
    
    return best, best_o


def _multi_refine(a, n, deadline, tag):
    """
    Multi-scale refinement: alternate between broad and focused gradients.
    Uses a cyclical K schedule to avoid getting stuck.
    """
    best = a.copy()
    best_o = _obj_from_conv(_fft_conv(best), n)
    it = 0
    cycle = 0
    eta = 0.2
    
    k_schedule = [n // 10, n // 20, n // 50, n // 100, max(5, n // 200), 3, 1]
    
    for k_val in k_schedule * 10:
        if time.monotonic() >= deadline:
            break
        sub_end = min(time.monotonic() + 30.0, deadline)
        
        if k_val >= n // 5:
            # Use soft gradient
            a, o = _adaptsk_run(a, n, sub_end, f"{tag}_soft", tau_start=n * 0.03, eta_start=eta)
        else:
            # Use hard gradient
            a, o = _hard_focus_run(a, n, sub_end, f"{tag}_K{k_val}", K=min(k_val, 20), eta_start=max(eta * 0.3, 0.02))
        
        if o < best_o - 1e-13:
            best_o = o
            best = a.copy()
            print(f"  [{tag}] refined: obj={best_o:.6f} (k={k_val})")
        
        eta = max(eta * 0.9, 0.02)
        cycle += 1
    
    return best, best_o


def _perturb_multi(a, n, rng, mode="block"):
    """Generate a perturbed version of a."""
    b = a.copy()
    if mode == "block":
        w = max(2, int(n * rng.choice([0.005, 0.01, 0.02, 0.05, 0.1, 0.2])))
        s = int(rng.integers(0, n - w + 1))
        sigma = float(rng.uniform(0.03, 0.5))
        b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    elif mode == "global":
        sigma = float(rng.uniform(0.02, 0.15))
        b *= np.exp(rng.normal(0.0, sigma, size=n))
    elif mode == "edge":
        w = max(2, int(n * 0.1))
        sigma = float(rng.uniform(0.05, 0.4))
        b[:w] *= np.exp(rng.normal(0.0, sigma, size=w))
        b[-w:] *= np.exp(rng.normal(0.0, sigma, size=w))
    else:
        # Random sparse perturbation
        frac = float(rng.uniform(0.01, 0.1))
        k = max(1, int(n * frac))
        idx = rng.choice(n, size=k, replace=False)
        b[idx] *= np.exp(rng.normal(0.0, 0.2, size=k))
    
    return _project(b, n)


def _try_n(a, n, deadline, tag, parent_init=None):
    """
    Full optimization pipeline for a given n.
    Returns (best_sequence, best_objective).
    """
    t0 = time.monotonic()
    
    # Initialize
    if parent_init is not None and len(parent_init) == n:
        a = _project(parent_init.copy(), n)
    else:
        # Try structured initializations
        inits = []
        inits.append(np.ones(n, dtype=np.float64))
        # Raised cosine
        x = np.arange(n, dtype=np.float64) / n
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * x))
        inits[-1] = np.maximum(inits[-1], 0.01)
        # Gaussian
        mu = n / 2.0
        sigma_g = n / 4.0
        inits.append(np.exp(-0.5 * ((np.arange(n) - mu) / sigma_g) ** 2))
        # Triangular
        inits.append(np.minimum(x, 1 - x) + 0.05)
        # Random
        inits.append(np.random.default_rng(123).uniform(0.3, 1.5, n))
        
        best_init = None
        best_init_o = float('inf')
        for init in inits:
            a_try = _project(init.astype(np.float64), n)
            o_try = _obj_from_conv(_fft_conv(a_try), n)
            if o_try < best_init_o:
                best_init_o = o_try
                best_init = a_try
        a = best_init
    
    o_init = _obj_from_conv(_fft_conv(a), n)
    elapsed = time.monotonic() - t0
    print(f"  [{tag}] n={n} init_obj={o_init:.6f} (t={elapsed:.1f}s)")
    
    if time.monotonic() >= deadline:
        return a, o_init
    
    # Phase 1: Adaptive soft→hard (45% of available time)
    avail = deadline - time.monotonic()
    p1_end = time.monotonic() + 0.40 * avail
    a, o1 = _adaptsk_run(a, n, p1_end, f"{tag}_soft", tau_start=n * 0.05, eta_start=0.35)
    print(f"  [{tag}] after soft: obj={o1:.6f} (t={time.monotonic()-t0:.0f}s)")
    
    if time.monotonic() >= deadline:
        return a, o1
    
    # Phase 2: Hard focus K=3 (20%)
    avail = deadline - time.monotonic()
    p2_end = time.monotonic() + 0.20 * avail
    a, o2 = _hard_focus_run(a, n, p2_end, f"{tag}_hard3", K=3, eta_start=0.06)
    print(f"  [{tag}] after hard3: obj={o2:.6f} (t={time.monotonic()-t0:.0f}s)")
    
    if time.monotonic() >= deadline:
        return a, o2
    
    # Phase 3: Hard focus K=1 (15%)
    avail = deadline - time.monotonic()
    p3_end = time.monotonic() + 0.15 * avail
    a, o3 = _hard_focus_run(a, n, p3_end, f"{tag}_hard1", K=1, eta_start=0.04)
    print(f"  [{tag}] after hard1: obj={o3:.6f} (t={time.monotonic()-t0:.0f}s)")
    
    # Phase 4: Multi-refine (10%)
    if time.monotonic() < deadline - 20:
        avail = deadline - time.monotonic()
        p4_end = time.monotonic() + 0.10 * avail
        a, o4 = _multi_refine(a, n, p4_end, f"{tag}_multi")
        print(f"  [{tag}] after multi: obj={o4:.6f} (t={time.monotonic()-t0:.0f}s)")
    
    # Phase 5: Perturbation + refinement (remaining)
    rng = np.random.default_rng(42 + n)
    while time.monotonic() < deadline - 15:
        mode = rng.choice(["block", "global", "edge", "sparse"])
        a_p = _perturb_multi(a, n, rng, mode)
        
        # Quick refinement
        sub_end = time.monotonic() + min(10.0, max(3.0, (deadline - time.monotonic() - 5) * 0.3))
        a_q, o_q = _adaptsk_run(a_p, n, sub_end, f"{tag}_pert", tau_start=n * 0.01, eta_start=0.2)
        
        sub_end2 = time.monotonic() + min(5.0, max(2.0, (deadline - time.monotonic() - 3) * 0.3))
        a_q2, o_q2 = _hard_focus_run(a_q, n, sub_end2, f"{tag}_pert_h", K=1, eta_start=0.03)
        
        if o_q2 < o3:
            o3 = o_q2
            a = a_q2
            print(f"  [{tag}] perturbed({mode}): obj={o3:.6f}")
    
    return a, o3


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase Fourier-accelerated optimizer.
    Explores multiple sequence lengths with intelligent time allocation.
    Returns the best non-negative sequence found within budget_s seconds.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 8.0
    rng = np.random.default_rng(seed)
    
    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 100 and np.all(np.isfinite(p)):
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    # Determine n values to try
    # Prioritize: parent's n, then nearby values, then larger
    n_candidates = []
    if parent is not None:
        pn = parent.size
        n_candidates.append(pn)
        for delta in [200, 500, 1000, -200, -500, 1500, -1000]:
            cand = pn + delta
            if 2000 <= cand <= 8000 and cand not in n_candidates:
                n_candidates.append(cand)
    else:
        n_candidates = [4000, 3809, 4500, 5000, 3500, 5500]
    
    # Time allocation: give 55% to first (best) n, 20% to second, 10% to third, rest to others
    total_budget = deadline - t_start
    allocations = []
    remaining = total_budget - 10
    for i in range(len(n_candidates)):
        if i == 0:
            alloc = remaining * 0.52
        elif i == 1:
            alloc = remaining * 0.18
        elif i == 2:
            alloc = remaining * 0.12
        else:
            alloc = remaining * 0.04
        allocations.append(alloc)
    
    best_overall = None
    best_overall_o = float('inf')
    
    for i, n in enumerate(n_candidates):
        elapsed = time.monotonic() - t_start
        if elapsed > total_budget - 30:
            break
        
        # How much time we can actually give this n
        actual_alloc = min(allocations[i], total_budget - elapsed - 10)
        if actual_alloc < 20:
            break
        
        n_deadline = time.monotonic() + actual_alloc
        print(f"\n{'='*60}")
        print(f"  Exploring n={n} with budget={actual_alloc:.0f}s (elapsed={elapsed:.0f}s)")
        print(f"{'='*60}")
        
        a, o = _try_n(None, n, n_deadline, f"n{n}", parent_init=parent)
        
        if o < best_overall_o:
            best_overall_o = o
            best_overall = a.copy()
            print(f"\n  *** NEW BEST: obj={best_overall_o:.6f} at n={n} ***\n")
    
    # Final validation
    if best_overall is not None:
        result = [float(x) for x in best_overall]
        # Ensure all values are in [0, 1000]
        result = [max(0.0, min(1000.0, x)) for x in result]
        try:
            sc = float(evaluate_sequence(result))
        except Exception:
            sc = float('inf')
        if not np.isfinite(sc):
            # Fallback to uniform
            n = len(best_overall)
            result = [1.0] * n
            sc = 2.0
        print(f"\n{'='*60}")
        print(f"  FINAL RESULT: internal_obj={best_overall_o:.6f}  evaluated={sc:.6f}  n={len(result)}")
        print(f"  Target: 1.5027  |  Previous: 1.517603")
        print(f"  {'SUCCESS' if sc < 1.5027 else 'NEEDS IMPROVEMENT'}")
        print(f"{'='*60}")
        return result
    else:
        # Absolute fallback
        n = 3809
        a = _project(np.ones(n), n)
        result = [float(x) for x in a]
        print(f"FALLBACK: uniform n={n}")
        return result
