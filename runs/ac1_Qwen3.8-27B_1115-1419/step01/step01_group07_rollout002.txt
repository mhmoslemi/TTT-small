def lse_max_conv(a, tau):
    """LSE-smoothed max of a*a."""
    c = np.convolve(a, a)
    c_max = c.max()
    if tau < 1e-30:
        return c_max
    c_shifted = c - c_max
    vals = np.exp(c_shifted / tau)
    return float(tau * np.log(vals.sum()) + c_max)

def lse_max_conv_grad(c, c_max, tau):
    """Gradient of LSE-smoothed max w.r.t. convolution entries."""
    if tau < 1e-30:
        w = np.zeros_like(c)
        w[c == c_max] = 1.0 / np.sum(c == c_max)
        return w
    c_shifted = c - c_max
    w = np.exp(c_shifted / tau)
    w /= w.sum()
    return w

def convolve_grad_from_weights(w, a):
    """
    Compute the gradient of the LSE-smoothed max of a*a w.r.t. a.
    w is the weight for each lag k of the convolution.
    Result: grad[i] = sum_k w[k] * d(a*a)_k / da[i]
    """
    n = len(a)
    m = len(w)  # 2n-1
    grad = np.zeros(n)
    for k in range(m):
        if w[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += w[k] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad

def convolve_grad_sparsity(w, a):
    """Faster version using only non-zero weights."""
    n = len(a)
    m = len(w)
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:

        hi = min(n - 1, k)

    """

def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u) - 1.0
    r = np.arange(1, n + 1)
    cond = u - cssv / r > 0
    k = int(np.where(cond)[0][-1])
    theta = cssv[k] / (k + 1)
    return np.maximum(a - theta, 0.0)

def lbfgs_optimize(a, tau, max_iter=200, time_limit=60):
    """
    L-BFGS optimization of LSE-smoothed max of a*a.
    Works in log-space: a = softmax(log_a).
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    # Parameterize as log
    x = np.log(np.maximum(a, 1e-30))
    x -= x.mean()  # center for numerical stability
    
    best_a = a.copy()
    best_score = lse_max_conv(a, tau)
    
    start_time = time.monotonic()
    
    def obj(x_):
        x_shifted = x_ - x_.max()
        e = np.exp(x_shifted)
        a_ = e / e.sum()
        return lse_max_conv(a_, tau)
    
    def grad(x_):
        x_shifted = x_ - x_.max()
        e = np.exp(x_shifted)
        a_ = e / e.sum()
        c = np.convolve(a_, a_)
        c_max = c.max()
        c_shifted = c - c_max
        w = np.exp(c_shifted / tau)
        w /= w.sum()
        grad_a = convolve_grad_sparsity(w, a_)
        # Chain rule through softmax
        dot = np.dot(a_, grad_a)
        grad_x = a_ * (grad_a - dot)
        return grad_x
    
    try:
        result = minimize(
            obj, x, jac=grad,
            method='L-BFGS-B',

            best_a = a_new

    x_old = np.linspace(0, 1, n)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:

    return a_new

"""

Uses multi-length search with FFT-accelerated subgradient methods,

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    C = A * A
    c = np.fft.irfft(C, fft_size)[:size]
    np.maximum(c, 0, out=c)  # Remove tiny negative values from FFT
    return c

def compute_subgrad(a, active):
    """Compute subgradient direction from active set indices."""
    n = len(a)
    K = len(active)
    if K == 0:
        return np.zeros(n)
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        length = hi - lo + 1
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= K
    return g

def subgrad_iter(a, top_k=30, threshold=0.92):
    """One subgradient iteration: compute active set and direction."""
    n = len(a)
    c = fft_convolve(a)
    max_c = c.max()
    if max_c < 1e-15:
        return a, 0.0
    # Use combined criterion: top_k or threshold
    idx = np.argpartition(c, -top_k)[-top_k:]
    c_idx = c[idx]
    mask = c_idx >= threshold * max_c
    if np.any(mask):
        active = idx[mask]
    else:
        active = idx[:5]
    g = compute_subgrad(a, active)
    return g, max_c

def subgradient_loop(a, time_limit=60, top_k=30, threshold=0.92, step_init=0.005, rng=None):
    """Iterative subgradient loop with adaptive step and momentum."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_score = evaluate_sequence(a.tolist())

    no_improve = 0

    momentum = 0.0

    it = 0
    eval_count = 0

    while time.monotonic() - start < time_limit:
        g, max_c = subgrad_iter(a, top_k=top_k, threshold=threshold)

        # Nesterov-like momentum
        v = momentum * (a - py_prev) - step * g
        py_prev = a.copy()
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss > 1e-15:

            a_new = a.copy()

            score = evaluate_sequence(a_new.tolist())
            eval_count += 1
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                no_improve = 0

def lse_objective_and_grad(x, tau):
    """
    LSE-smoothed max of a*a where a = softmax(x).
    Returns (objective, gradient w.r.t. x).
    """
    x_max = x.max()
    e = np.exp(x - x_max)
    s = e.sum()
    if s < 1e-300:
        a = np.ones(len(x)) / len(x)
    else:
        a = e / s

    c = fft_convolve(a)
    c_max = c.max()
    if tau < 1e-30:
        return c_max, np.zeros(len(x))

    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return c_max, np.zeros(len(x))
    w = w / w_sum

    obj = c_max + tau * np.log(w_sum)

    # Gradient of obj w.r.t. a: sum_k w[k] * 2*a[k-i] for valid k-i
    grad_a = np.zeros(len(x))
    nz = np.where(w > 1e-12)[0]
    for k in nz:
        lo = max(0, k - len(x))
        hi = min(len(x) - 1, k)
        grad_a[lo:hi + 1] += 2.0 * w[k] * a[k - hi:k - lo + 1][::-1]

    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x

def lbfgs_polish(a, tau_list=None, time_limit=60, max_iter_per_tau=150):
    """L-BFGS polishing with decreasing tau."""
    n = len(a)
    a = a.copy() / a.sum()

    if tau_list is None:
        tau_list = [0.05 / n, 0.02 / n, 0.01 / n, 0.005 / n, 0.002 / n]

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())

    x = np.log(np.maximum(a, 1e-30))
    start = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - start > time_limit - 5:
            break

        def obj_fn(x_):
            return lse_objective_and_grad(x_, tau)[0]

        def grad_fn(x_):
            return lse_objective_and_grad(x_, tau)[1]

        try:
            result = minimize(
                obj_fn, x, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_tau, 'ftol': 1e-13,
                         'gtol': 1e-11, 'maxfun': max_iter_per_tau * 4}
            )
            x = result.x
        except Exception:
            break

        x_max = x.max()
        e = np.exp(x - x_max)
        a_new = e / e.sum()
        score = evaluate_sequence(a_new.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a

def resample_to_length(a, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

def generate_diverse_inits(n, rng, base_seq=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    i = np.arange(n, dtype=float)

        inits.append(resample_to_length(base_seq, n))

    # Flat with noise

    x = np.linspace(1.6, 0.4, n)
    inits.append(x / x.sum())

    x = np.linspace(0.4, 1.6, n)

    for sigma_frac in [0.2, 0.3, 0.4]:

    x = np.exp(-((i - mid * 0.3)) ** 2 / (2 * (n * 0.1) ** 2)) + \

    x = np.where(rng.random(n) > 0.3, 1.0, 0.15)

    decay = 0.003
    x = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))

            print(f"  Ultra-fine: {best_score:.6f}")

    # Try one more creative restart: take best, add tiny structured perturbation
    time_left = deadline - time.monotonic()
    if time_left > 40:
        for pert_type in ['sine', 'noise']:
            if time.monotonic() > deadline - 10:
                break
            a2 = best_a.copy()
            if pert_type == 'sine':
                i = np.arange(n)
                a2 = a2 * (1 + 0.01 * np.sin(2 * np.pi * 7 * i / n))
            else:
                a2 = a2 + rng.normal(0, 0.003, n)
            a2 = np.maximum(a2, 0)
            a2 /= a2.sum()
            a2 = subgradient_loop(a2, time_limit=min(25, int(time_left * 0.4)),
                                  top_k=20, threshold=0.98, step_init=0.001, rng=rng)
            a2 = lbfgs_polish(a2, tau_list=[0.002 / n],
                              time_limit=min(15, int(time_left * 0.2)),
                              max_iter_per_tau=100)
            sc = evaluate_sequence(a2.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print(f"  Creative restart ({pert_type}): {best_score:.6f}")

    # Ensure validity
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    final_check = evaluate_sequence(best_a.tolist())
    if not np.isfinite(final_check):
        # Fallback to base
        best_a = np.maximum(base, 0.0)
        best_a = np.minimum(best_a, 1000.0)
        final_check = evaluate_sequence(best_a.tolist())

    print(f"\n=== FINAL: n={len(best_a)}, score={final_check:.6f}, "
          f"elapsed={time.monotonic()-t0:.0f}s ===")

    return [float(val) for val in best_a]
