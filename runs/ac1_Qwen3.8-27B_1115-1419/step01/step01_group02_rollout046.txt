"""
Minimize 2n*max(autocorr)/(sum)^2 using L-BFGS-B with log-sum-exp surrogate.
Uses FFT-based gradient computation, tau annealing, multiple sequence lengths,
and diverse initializations including cosine-modulated and structured patterns.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    N = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, N) * fft(b, N)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target):
    """Normalize array to target sum preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / s)


def _make_objective(n):
    """Create objective+gradient function for given n."""
    fft_size = 1 << (3 * n).bit_length()
    conv_size = 1 << (2 * n - 1).bit_length()

    def obj_grad(x, tau):
        a = x
        sum_a = float(np.sum(a))
        if sum_a < 0.01:
            return 1e10, np.zeros_like(x)

        # Compute convolution
        Nc = conv_size
        fa = fft(a, Nc)
        b = ifft(fa * fa).real[:2 * n - 1]

        # Log-sum-exp of b/tau
        b_max = float(np.max(b))
        exp_b = np.exp((b - b_max) / tau)
        sum_exp = float(np.sum(exp_b))
        if sum_exp < 1e-300:
            return 1e10, np.zeros_like(x)
        log_sum_exp = b_max / tau + np.log(sum_exp)
        g = tau * log_sum_exp

        # Objective: 2n * g / sum_a^2
        value = 2.0 * n * g / (sum_a * sum_a)

        # Gradient computation
        w = exp_b / sum_exp
        A = fft(a, fft_size)
        W_flip = fft(np.flip(w), fft_size)
        result = ifft(A * W_flip).real
        dg = 2.0 * result[n - 1:2 * n - 1][::-1]

        grad = (2.0 * n / (sum_a * sum_a)) * (dg - 2.0 * g / sum_a)

        return value, grad

    return obj_grad


def _optimize_from_start(a0, deadline, max_anneal=7, max_iters=200):
    """Run L-BFGS-B with tau annealing from a starting point."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    sum_a0 = float(np.sum(a0))
    if sum_a0 < 0.01:
        return a0, _score_fast(a0)

    a0 = _normalize(a0, np.sqrt(2.0 * n))
    best_a = a0.copy()
    best_score = _score_fast(a0)

    obj = _make_objective(n)
    bounds = [(0.0, 1000.0)] * n

    # Compute initial tau
    b0 = _fft_conv(a0, a0)
    max_b0 = float(np.max(b0))
    tau_start = max_b0 / 3.0

    current = a0.copy()
    for i in range(max_anneal):
        if time.monotonic() > deadline:
            break
        tau = tau_start * (0.35 ** i)
        tau = max(tau, 0.001 * max_b0)

        result = minimize(
            obj, current, args=(tau,),
            method='L-BFGS-B',
            bounds=bounds,
            jac=True,
            options={'maxiter': max_iters, 'gtol': 1e-12, 'maxfun': max_iters * 5}
        )

        current = np.maximum(result.x, 0.0)
        sc = _score_fast(current)
        if sc < best_score:
            best_score = sc
            best_a = current.copy()

    return best_a, best_score


def _try_length(n, deadline, rng, parent=None):
    """Try multiple initializations for a given length n."""
    best_a = None
    best_score = np.inf

    k = np.arange(n)
    inits = []

    # From parent (resample)
    if parent is not None and len(parent) > 0:
        if n == len(parent):
            inits.append(("parent", parent.copy()))
        elif n > len(parent):
            # Cubic upsample
            x_old = np.arange(len(parent))
            x_new = np.linspace(0, len(parent) - 1, n)
            from scipy.interpolate import PchipInterpolator
            try:
                f = PchipInterpolator(x_old, parent)
                inits.append(("parent_up", np.maximum(f(x_new), 0.0)))
            except Exception:
                from scipy.interpolate import interp1d
                f = interp1d(x_old, parent, kind='linear', fill_value='extrapolate')
                inits.append(("parent_up", np.maximum(f(x_new), 0.0)))
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(("parent_dn", parent[idx].copy()))

    # Diverse structured inits
    inits.append(("uniform", np.ones(n)))
    inits.append(("random", rng.random(n) + 0.1))

    # Cosine modulations at various frequencies
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append((f"cos{freq}", 1.0 + 0.35 * np.cos(2 * np.pi * freq * k / n)))

    # Dual cosine
    inits.append(("dual_cos", 1.0 + 0.25 * np.cos(2 * np.pi * 3 * k / n) + 0.2 * np.cos(2 * np.pi * 7 * k / n)))

    # Periodic patterns
    inits.append(("period3", np.where(k % 3 == 0, 1.8, 0.4) + 0.5))
    inits.append(("period5", np.where(k % 5 < 2, 1.5, 0.5)))

    # Ramp and V-shape
    inits.append(("ramp", 0.5 + 0.8 * k / n))
    inits.append(("vshape", 0.5 + 0.8 * np.abs(k / n - 0.5) * 2))

    # Gaussian bump
    sigma = n / 5.0
    inits.append(("gauss", 0.3 + np.exp(-0.5 * ((k - n / 2) / sigma) ** 2)))

    # Sine
    inits.append(("sine3", 1.0 + 0.4 * np.sin(6 * np.pi * k / n)))

    # Two-level with random assignment
    r = rng.random(n)
    inits.append(("two_level", np.where(r > 0.5, 1.4, 0.6)))

    # Water-filled: high in middle, low at edges
    inits.append(("bell", 0.3 + 0.7 * np.sin(np.pi * k / n) ** 2))

    # Time budget per init
    remaining = deadline - time.monotonic()
    per_init = max(remaining / len(inits), 10.0)

    for name, init in inits:
        if time.monotonic() > deadline - 10:
            break
        init = np.maximum(init, 0.0)
        s = float(np.sum(init))
        if s < 0.01:
            continue
        init = _normalize(init, np.sqrt(2.0 * n))

        local_best, local_score = _optimize_from_start(
            init, time.monotonic() + per_init, max_anneal=5, max_iters=150
        )

        if local_score < best_score:
            best_score = local_score
            best_a = local_best.copy()
            if local_score < 1.52:
                print(f"    [{name}] n={n}: {local_score:.6f}")

    return best_a, best_score


def _polish(a, deadline, n_steps=500):
    """Polish using direct subgradient with adaptive step size."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_fast(a)
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.9

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = _fft_conv(a, a)
        sum_a = float(np.sum(a))
        peak_idx = int(np.argmax(b))
        peak_val = float(b[peak_idx])

        # Subgradient at the peak
        # db_k/da_j = 2*a_{k-j} if 0<=k-j<n
        j_range = np.arange(n)
        k_minus_j = peak_idx - j_range
        mask = (k_minus_j >= 0) & (k_minus_j < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * a[k_minus_j[mask]]

        # Full gradient of score wrt a
        # d(score)/da_j = 2n/s^2 * (dg/da_j - 2g/s)
        g = peak_val
        grad = (2.0 * n / (sum_a * sum_a)) * (grad - 2.0 * g / sum_a)

        momentum = beta * momentum + (1 - beta) * grad
        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _normalize(a, sum_a)

        lr *= 0.999

        if step % 50 == 0:
            sc = _score_fast(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Minimize 2n*max(autocorr)/(sum)^2 using L-BFGS-B with log-sum-exp surrogate,
    FFT-based gradients, tau annealing, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Get parent
    parent = None
    parent_n = 5582
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
            parent_n = parent.size
    except (NameError, Exception):
        parent = np.ones(parent_n)

    # Initial best
    best = np.maximum(parent, 0.0)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score:.6f} (n={parent_n})")

    # Phase 1: Optimize at parent length (highest priority)
    t_start = time.monotonic()
    phase1_deadline = t_start + (total_deadline - t_start) * 0.35
    print(f"\nPhase 1: n={parent_n}, budget={phase1_deadline - t_start:.0f}s")

    a_best, s_best = _try_length(parent_n, phase1_deadline, rng, parent)
    if a_best is not None and s_best < best_score:
        actual = float(evaluate_sequence(a_best.tolist()))
        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best = a_best.copy()
            print(f"  Phase 1 improved to: {best_score:.6f}")

    # Polish the best
    if time.monotonic() < phase1_deadline:
        a_pol, s_pol = _polish(best, phase1_deadline - 5)
        if s_pol < best_score:
            actual = float(evaluate_sequence(a_pol.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = a_pol.copy()
                print(f"  After polish: {best_score:.6f}")

    # Phase 2: Try longer sequences
    phase2_deadline = total_deadline - 60
    remaining = phase2_deadline - time.monotonic()
    print(f"\nPhase 2: Longer sequences, budget={remaining:.0f}s")

    longer_lengths = []
    base = parent_n
    for mult in [1.5, 2.0, 3.0, 5.0]:
        L = int(base * mult)
        if L > 100000:
            L = 100000
        if L not in longer_lengths and L > parent_n:
            longer_lengths.append(L)

    # Also try some specific lengths
    for L in [10000, 20000, 40000]:
        if L not in longer_lengths and L > parent_n and L <= 100000:
            longer_lengths.append(L)

    longer_lengths = sorted(set(longer_lengths))
    n_lengths = len(longer_lengths)
    time_per = max(remaining / max(n_lengths, 1), 30.0)

    for L in longer_lengths:
        if time.monotonic() > phase2_deadline - 20:
            break
        L_deadline = min(time.monotonic() + time_per, phase2_deadline - 20)
        print(f"  Trying n={L}...")

        a_cand, s_cand = _try_length(L, L_deadline, rng, best)
        if a_cand is not None and s_cand < best_score:
            actual = float(evaluate_sequence(a_cand.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = a_cand.copy()
                print(f"  n={L}: improved to {best_score:.6f}")
            else:
                print(f"  n={L}: FFT {s_cand:.6f}, actual {actual:.6f}")

    # Phase 3: Final refinement
    if time.monotonic() < total_deadline - 15:
        print(f"\nPhase 3: Final refinement, best={best_score:.6f}")
        n_best = len(best)
        refine_deadline = total_deadline - 5

        # Multiple L-BFGS-B passes with very small tau
        a_curr = best.copy()
        for tau_frac in [0.05, 0.02, 0.01, 0.005]:
            if time.monotonic() > refine_deadline:
                break
            b_curr = _fft_conv(a_curr, a_curr)
            tau = float(np.max(b_curr)) * tau_frac
            obj = _make_objective(n_best)
            result = minimize(
                obj, a_curr, args=(max(tau, 0.0001),),
                method='L-BFGS-B',
                bounds=[(0.0, 1000.0)] * n_best,
                jac=True,
                options={'maxiter': 100, 'gtol': 1e-14}
            )
            a_curr = np.maximum(result.x, 0.0)
            sc = _score_fast(a_curr)
            if sc < best_score:
                actual = float(evaluate_sequence(a_curr.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best = a_curr.copy()
                    print(f"  tau={tau_frac}: {best_score:.6f}")

        # Final subgradient polish
        if time.monotonic() < refine_deadline - 5:
            a_pol, s_pol = _polish(best, refine_deadline, n_steps=300)
            if s_pol < best_score:
                actual = float(evaluate_sequence(a_pol.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best = a_pol.copy()
                    print(f"  Final polish: {best_score:.6f}")

    # Ensure validity
    best = np.maximum(best, 0.0)
    best = np.minimum(best, 1000.0)

    final_score = float(evaluate_sequence([float(x) for x in best]))
    print(f"\nFinal score: {final_score:.6f}")

    if final_score < best_score:
        best_score = final_score

    return [float(x) for x in best]
