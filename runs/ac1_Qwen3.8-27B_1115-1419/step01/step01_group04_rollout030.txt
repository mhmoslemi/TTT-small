"""
Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2 using:
- Logsumexp smoothing with L-BFGS-B for quasi-Newton optimization
- Coarse-to-fine multi-resolution search (256 -> 4096)
- Diverse initializations (U-shapes, random, parent upsample, harmonic)
- Adaptive temperature schedule (large tau -> small tau)
- Final active-set polish step
"""

import numpy as np
import time
import math
from scipy.optimize import minimize


def fast_conv(a):
    """FFT-based linear convolution a * a, truncated to length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def normalize_seq(a, target=None):
    """Clip to [0, inf], normalize to target sum (default n)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    n = a.size
    if target is None:
        target = float(n)
    s = float(a.sum())
    if s < 1e-15:
        a.fill(target / max(1, n))
        return a
    a *= target / s
    return a


def exact_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fast_conv(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * float(c.max()) / (s * s))


def lse_obj_grad(params, tau):
    """
    Logsumexp-smoothed objective and gradient for scipy optimizer.
    params: raw parameter vector (will be clipped and normalized internally)
    tau: smoothing parameter (smaller = sharper, closer to true max)
    Returns (objective, gradient).
    """
    n = params.size
    p = np.maximum(params, 0.0)
    s = float(p.sum())
    if s < 1e-15:
        return 1e10, np.zeros(n)
    a = p * (float(n) / s)  # normalized: sum(a) = n

    c = fast_conv(a)
    cm = float(c.max())
    shifted = (c - cm) / tau
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    obj = 2.0 * (cm + tau * math.log(Z)) / float(n)

    w = exp_shifted / Z

    # Gradient computation:
    # d(obj)/da[j] = 4/n * sum_k w[k] * a[k-j] (k-j valid)
    # = 4/n * cross_correlation(w, a)[j]
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    cross = np.fft.irfft(np.fft.rfft(w[::-1], N) * np.fft.rfft(a, N), N)[:n]
    grad_a = 4.0 * cross / float(n)

    # Since objective is scale-invariant, gradient w.r.t. raw params
    # is proportional to grad_a; L-BFGS-B handles the scale via line search.
    # Project to remove component along a (maintain normalization consistency).
    dot_a = float(np.dot(grad_a, a))
    aa = float(np.dot(a, a))
    if aa > 1e-15:
        grad = grad_a - (dot_a / aa) * a
    else:
        grad = grad_a

    return obj, grad


def upsample_seq(seq, n):
    """Linearly interpolate a sequence to length n, then normalize."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    n = int(n)
    if m == 0:
        return normalize_seq(np.ones(n), float(n))
    if m == n:
        return normalize_seq(seq, float(n))
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_seq(out, float(n))


def make_u_shape(n, sharpness=2.0):
    """U-shaped sequence: high at edges, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + sharpness * (2.0 * t - 1.0) ** 2
    return normalize_seq(np.maximum(a, 0.01), float(n))


def make_m_shape(n, sharpness=2.0):
    """M-shaped sequence: high at quarter and three-quarter points."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + sharpness * (np.cos(2.0 * np.pi * t) ** 2)
    return normalize_seq(np.maximum(a, 0.01), float(n))


def make_harmonic(n, rng, num_harmonics=8, max_amp=2.0):
    """Random harmonic shape, exponentiated to stay positive."""
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, num_harmonics + 1):
        amp = rng.uniform(-max_amp, max_amp) / (m ** 0.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    z -= z.max()
    a = np.exp(z)
    return normalize_seq(a, float(n))


def make_random_blocks(n, rng, num_blocks=16):
    """Random piecewise-constant sequence."""
    num_blocks = max(2, min(num_blocks, n // 2))
    edges = np.sort(rng.choice(np.arange(1, n), size=num_blocks - 1, replace=False))
    boundaries = np.concatenate([[0], edges, [n]])
    lengths = np.diff(boundaries)
    vals = rng.uniform(0.0, 3.0, num_blocks)
    a = np.repeat(vals, lengths)
    return normalize_seq(np.maximum(a, 0.0), float(n))


def make_two_pulse(n, rng):
    """Two Gaussian pulses at random positions."""
    t = np.linspace(0.0, 1.0, n)
    c1 = rng.uniform(0.1, 0.4)
    c2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.02, 0.1)
    w2 = rng.uniform(0.02, 0.1)
    a1 = rng.uniform(0.5, 2.0)
    a2 = rng.uniform(0.5, 2.0)
    a = a1 * np.exp(-((t - c1) / w1) ** 2) + a2 * np.exp(-((t - c2) / w2) ** 2)
    a += 0.01
    return normalize_seq(a, float(n))


def generate_starts(n, parent, rng, count=20):

    # Constant

    # U-shapes with various sharpness
    for sh in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]:
        starts.append(make_u_shape(n, sh))

    # M-shapes
    for sh in [1.0, 2.0, 3.0]:
        starts.append(make_m_shape(n, sh))

    # Cosine-based
    t = np.linspace(0.0, 1.0, n)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        starts.append(normalize_seq(np.maximum(1.0 + alpha * np.cos(np.pi * (2.0 * t - 1.0)), 0.01), float(n)))
        starts.append(normalize_seq(np.maximum(1.0 - alpha * np.cos(np.pi * t), 0.01), float(n)))

    # Two-pulse
    for _ in range(5):
        starts.append(make_two_pulse(n, rng))

    # Random blocks
    for nb in [4, 8, 16, 32]:
        starts.append(make_random_blocks(n, rng, nb))

    # Harmonic
    for _ in range(5):
        starts.append(make_harmonic(n, rng))

    # Pure random
    for _ in range(3):
        starts.append(normalize_seq(np.abs(rng.standard_normal(n)) + 0.01, float(n)))

    # Parent upsample
    if parent is not None and parent.size > 1:
        starts.append(upsample_seq(parent, n))
        # Perturbed parent
        p = upsample_seq(parent, n)
        p *= np.exp(rng.normal(0, 0.3, n))
        starts.append(normalize_seq(p, float(n)))

    # Score and sort
    scored = []
    for s in starts:
        sc = exact_score(s)
        if np.isfinite(sc):
            scored.append((sc, s))
    scored.sort(key=lambda x: x[0])
    return scored


def run_lbfqs(a, n, tau_schedule, max_iter=300, deadline=None):
    """
    Run L-BFGS-B with a schedule of decreasing tau values.
    Returns the best (sequence, score) found.
    """
    best_a = a.copy()
    best_obj = exact_score(a)

    for tau in tau_schedule:
        if deadline is not None and time.monotonic() > deadline:
            break
        try:
            result = minimize(
                lse_obj_grad,
                best_a,
                args=(tau,),
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 30}
            )
            if result.x is not None:
                a_new = normalize_seq(result.x, float(n))
                sc = exact_score(a_new)
                if np.isfinite(sc) and sc < best_obj:
                    best_obj = sc
                    best_a = a_new
        except Exception:
            continue

    # Final direct evaluation
    sc_final = exact_score(best_a)
    if np.isfinite(sc_final) and sc_final < best_obj:
        best_obj = sc_final

    return best_a, best_obj


def polish_active_set(a, n, max_iter=200, deadline=None):
    """
    Polish using active-set gradient steps on the exact (unsmoothed) max.
    Targets the top-K peaks of the convolution.
    """
    best_a = a.copy()
    best_score = exact_score(a)
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        a = best_a.copy()
        c = fast_conv(a)
        cmax = c.max()
        if cmax < 1e-15:
            break

        # Active set: peaks within 5% of max
        threshold = cmax * 0.95
        active = np.where(c >= threshold)[0]
        K = len(active)
        if K == 0:
            break

        # Subgradient: average of gradients at active peaks
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            start = max(0, k - (n - 1))
            stop = min(n, k + 1)
            s_start = k - (stop - 1)
            s_end = k - start + 1
            grad[start:stop] += a[s_start:s_end][::-1]
        grad *= 2.0 / K

        # Center the gradient (to preserve sum)
        grad -= grad.mean()

        # Step size via backtracking line search
        step = 0.5
        for _ in range(20):
            candidate = a - step * grad
            np.maximum(candidate, 0.0, out=candidate)
            candidate = normalize_seq(candidate, float(n))
            sc = exact_score(candidate)
            if sc < best_score * (1.0 - 1e-8):
                best_a = candidate
                best_score = sc
                break
            step *= 0.5

    return best_a, best_score


def sanitize_output(seq):
    """Convert to a valid Python list of non-negative finite floats."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return [1.0] * 512
    seq = np.where(np.isfinite(seq), seq, 0.0)
    np.clip(seq, 0.0, 1000.0, out=seq)
    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 512
    # Normalize to sum = n for clean output
    scale = float(seq.size) / s
    seq = seq * scale
    np.clip(seq, 0.0, 1000.0, out=seq)
    if float(seq.sum()) < 0.01:
        return [1.0] * 512
    return [float(x) for x in seq]

    # Each level: generate starts, run L-BFGS with tau schedule, keep best
    levels = [
        (256, [5.0, 2.0, 1.0, 0.5], 3, 200),
        (512, [3.0, 1.5, 0.7, 0.3, 0.1], 4, 250),
        (1024, [2.0, 1.0, 0.5, 0.2, 0.1, 0.05], 3, 300),
        (2048, [1.5, 0.7, 0.3, 0.1, 0.05, 0.02], 3, 350),
        (4096, [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01], 2, 400),
    ]

    rng = np.random.default_rng(42)

    for level_idx, (n, tau_sched, num_starts, max_iter) in enumerate(levels):
        if time.monotonic() > search_deadline - 60.0:
            break

        # Determine how much time to allocate to this level
        levels_remaining = len(levels) - level_idx
        time_available = max(10.0, search_deadline - time.monotonic() - 60.0)
        weight = (level_idx + 1) ** 1.5  # more time for larger levels
        total_weight = sum((i + 1) ** 1.5 for i in range(level_idx, len(levels)))
        level_time = time_available * weight / total_weight
        level_time = max(5.0, min(level_time, time_available))
        level_deadline = time.monotonic() + level_time

        # Generate starts
        num_gen = max(8, num_starts + 10)
        starts = generate_starts(n, parent if level_idx == 0 else global_best_seq, rng, count=num_gen)

        if not starts:
            starts = [(2.0, normalize_seq(np.ones(n), float(n)))]

        # Update parent for next level
        if starts[0][1].size > 0:
            if global_best_seq is None or starts[0][0] < global_best_score:
                global_best_score = starts[0][0]
                global_best_seq = starts[0][1].copy()

        # Run L-BFGS on top starts
        time_per_start = level_time / float(num_starts)
        for i in range(min(num_starts, len(starts))):
            if time.monotonic() > level_deadline - 2.0:
                break
            sc, a = starts[i]
            # Predict if this start could beat global best
            if sc > global_best_score * 1.5 and i > 0:
                continue

            time_this = max(2.0, min(time_per_start, level_deadline - time.monotonic() - 1.0))
            max_iter_this = min(max_iter, max(20, int(time_this * 100)))

            try:
                a_opt, sc_opt = run_lbfqs(a, n, tau_sched, max_iter=max_iter_this,
                                           deadline=level_deadline)
            except Exception:
                a_opt, sc_opt = a, sc

            if sc_opt < global_best_score:
                global_best_score = sc_opt
                global_best_seq = a_opt.copy()
                print(f"  Level n={n}, start {i+1}/{num_starts}: score = {sc_opt:.6f} (best so far)")

        # After level, polish the best from this level
        if global_best_seq is not None and global_best_seq.size == n:
            polish_deadline = min(level_deadline, time.monotonic() + max(2.0, level_time * 0.1))
            try:
                a_pol, sc_pol = polish_active_set(global_best_seq, n, max_iter=100,
                                                   deadline=polish_deadline)
                if sc_pol < global_best_score:
                    global_best_score = sc_pol
                    global_best_seq = a_pol.copy()
                    print(f"  Level n={n}, after polish: score = {sc_pol:.6f}")
            except Exception:
                pass

    # If we have time, do a final polish on the global best
    if global_best_seq is not None and time.monotonic() < search_deadline:
        n = global_best_seq.size
        polish_time = max(5.0, search_deadline - time.monotonic() - 10.0)
        try:
            a_pol, sc_pol = polish_active_set(global_best_seq, n, max_iter=500,
                                               deadline=search_deadline - 5.0)
            if sc_pol < global_best_score:
                global_best_score = sc_pol
                global_best_seq = a_pol.copy()
                print(f"  Final polish: score = {sc_pol:.6f}")
        except Exception:
            pass

        # One more L-BFGS pass with very small tau if time permits
        if time.monotonic() < search_deadline - 10.0:
            try:
                a_final, sc_final = run_lbfqs(
                    global_best_seq, n,
                    tau_schedule=[0.05, 0.02, 0.01, 0.005],
                    max_iter=200,
                    deadline=search_deadline - 5.0
                )
                if sc_final < global_best_score:
                    global_best_score = sc_final
                    global_best_seq = a_final.copy()
                    print(f"  Final L-BFGS: score = {sc_final:.6f}")
            except Exception:
                pass

    # Fallback
    if global_best_seq is None:
        global_best_seq = normalize_seq(np.ones(512), 512.0)
        global_best_score = exact_score(global_best_seq)

    print(f"\nFinal best score: {global_best_score:.6f} at n={global_best_seq.size}")

    # Validate with exact evaluator
    try:
        val = evaluate_sequence(sanitize_output(global_best_seq))
        if np.isfinite(val):
            print(f"Validated score: {val:.6f}")
        else:
            print(f"Validation failed, fallback")
            global_best_seq = normalize_seq(np.ones(512), 512.0)
    except Exception as e:
        print(f"Validation error: {e}")

    return sanitize_output(global_best_seq)
