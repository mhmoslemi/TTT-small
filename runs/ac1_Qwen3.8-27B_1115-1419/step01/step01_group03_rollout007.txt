def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - 1.0) / rho
    if not cond.any():
        return np.full(n, 1.0/n)
    rho_max = np.where(cond)[0][-1]
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)

def compute_grad_active(a, active_lags):
    """Compute the averaged gradient of max(c_k) over active lags."""

import time


def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()

    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def fft_gradient(a, active_lags, weights=None):
    """
    Compute the averaged gradient of max_k c_k over active lags using FFT.
    grad(c_k)[m] = 2*a[k-m] for 0<=k-m<n.
    Returns gradient array of length n.
    """
    n = len(a)
    if weights is None:
        weights = np.ones(len(active_lags)) / len(active_lags)
    
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    
    phase_sum = np.zeros(N // 2 + 1, dtype=np.complex128)
    for k, w in zip(active_lags, weights):
        phase = np.exp(-2j * np.pi * k * np.arange(N // 2 + 1) / N)
        phase_sum += w * phase
    
    grad_pad = np.fft.irfft(A * phase_sum, n=N)
    grad = 2.0 * grad_pad[:n]
    return grad


def run_fista(a0, duration, rng, step_init=0.5, top_k=5, top_k_max=80,
              restart_interval=300, momentum_decay=0.98):
    """
    FISTA to minimize max convolution coefficient on the simplex.
    Uses corrected gradient (factor of 2), FFT-based multi-lag accumulation,
    adaptive step size, growing active set, and periodic restarts.
    """
    n = len(a0)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    a = a0.copy()
    a_prev = a0.copy()
    t_k = 1.0
    step = step_init
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    current_top_k = top_k
    last_improve = it
    print_freq = max(200, int(duration / 50))
    
    while time.monotonic() < deadline:
        # Nesterov momentum
        t_old = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_k
        # Damp momentum to improve stability for non-convex problems
        momentum *= momentum_decay
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        
        # Compute convolution at y
        c_y = np.convolve(y, y)
        
        # Find active lags (top-k plus lags near max)
        c_max = c_y.max()
        # Threshold: include lags within a fraction of max
        threshold = c_max * (1.0 - 0.05 * (current_top_k / top_k_max + 0.5))
        active_mask = c_y >= threshold
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) > current_top_k:
            # Keep only top-k
            top_idx = np.argpartition(c_y, -current_top_k)[-current_top_k:]
            active_lags = np.sort(top_idx)
        
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(c_y)])
        
        # Compute gradient using FFT
        weights = np.ones(len(active_lags)) / len(active_lags)
        g = fft_gradient(y, active_lags, weights)
        
        # Gradient step + project onto simplex
        a_new = project_simplex(y - step * g)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic score check
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
                last_improve = it
            else:
                stuck += 1
            
            # Adaptive step size
            if stuck == 30:
                step *= 0.92
            elif stuck == 100:
                step *= 0.85
                # Reduce momentum by resetting
                a_prev = a.copy()
                t_k = 1.0
            elif stuck > restart_interval:
                # Restart with perturbation
                sigma = max(0.002, 0.01 * (1.0 - it / max(1, duration * 10)))
                a = best_a + rng.normal(0.0, sigma / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_k = 1.0
                step = step_init
                stuck = 0
                # Grow active set on restart
                if current_top_k < top_k_max:
                    current_top_k = min(current_top_k + 3, top_k_max)
                last_improve = it
        
        # Gradually increase active set
        if it % 3000 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 2, top_k_max)
        
        # Log progress
        if it % print_freq == 0:
            el = time.monotonic()
            print(f"    FISTA it={it}, score={best_score:.7f}, step={step:.4f}, top_k={current_top_k}, stuck={stuck}, rem={max(0, deadline-el):.0f}s")
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def random_local_search(a, duration, rng):
    """
    Gradient-free local search: random perturbations with acceptance.
    Useful for escaping shallow local minima in the non-convex landscape.
    """
    n = len(a)
    a_best = a.copy()
    score_best = fast_score(a)
    deadline = time.monotonic() + duration
    T = 0.01  # initial "temperature"
    T_min = 0.0001
    
    while time.monotonic() < deadline:
        # Generate perturbation
        sigma = T
        idx = rng.integers(0, n, size=max(5, n // 100))
        a_trial = a_best.copy()
        a_trial[idx] += rng.normal(0, sigma / np.sqrt(n), len(idx))
        np.maximum(a_trial, 0, out=a_trial)
        st = a_trial.sum()
        if st > 1e-12:
            a_trial /= st
        
        sc = fast_score(a_trial)
        if sc < score_best:
            a_best = a_trial
            score_best = sc
            T = max(T * 0.95, T_min)
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-min(100.0, (sc - score_best) / max(T, T_min))):
                a_best = a_trial
            T = max(T * 0.99, T_min)
    
    return a_best, score_best


def multiscale_warmup(a, rng):
    """
    Multi-scale warmup: downsample, optimize, upsample.
    Helps find a better basin before fine optimization.
    """
    n = len(a)
    factor = 4
    if n >= 64:
        n_coarse = n // factor
        # Downsample by averaging
        a_coarse = np.convolve(a, np.ones(factor) / factor, mode='valid')[:n_coarse]
        sc = a_coarse.sum()
        if sc > 1e-12:
            a_coarse /= sc
        
        # Quick FISTA on coarse
        a_c, sc_c = run_fista(a_coarse, 5, rng, step_init=0.5, top_k=3, top_k_max=10, restart_interval=20)
        
        # Upsample
        a_up = np.repeat(a_c, factor)[:n]
        np.maximum(a_up, 0, out=a_up)
        su = a_up.sum()
        if su > 1e-12:
            a_up /= su
        return a_up
    return a.copy()


def propose_candidate(budget_s=2400):
    """
    Main search function. Uses FISTA with corrected gradients, FFT-based
    multi-lag accumulation, adaptive step sizes, growing active sets,
    multi-scale warmup, and simulated-annealing perturbations.
    """
    rng = np.random.default_rng(2024)
    
    # --- Get/initialize sequence ---
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.ones(n, dtype=np.float64)
    
    if n < 10:
        n = 4806
        arr = np.ones(n, dtype=np.float64)
    
    # Normalize to simplex
    S = arr.sum()
    if S > 1e-10:
        a = arr / S
    else:
        a = np.ones(n, dtype=np.float64) / n
    
    total_deadline = time.monotonic() + float(budget_s) - 25.0
    
    # Evaluate initial
    init_score = fast_score(a)
    print(f"n={n}, initial score={init_score:.8f}")
    
    # Also try a few diverse candidates for comparison
    candidates = [a]
    cand_names = ["given"]
    
    # Exponential
    exp_c = np.exp(-np.linspace(0, 3, n))
    exp_c /= exp_c.sum()
    candidates.append(exp_c)
    cand_names.append("exp")
    
    # Two-peak
    two = np.zeros(n)
    w = max(3, n // 30)
    two[n//4 - w:n//4 + w] = 1.0
    two[3*n//4 - w:3*n//4 + w] = 1.0
    two /= two.sum()
    candidates.append(two)
    cand_names.append("two_peak")
    
    # Anti-symmetric (high at ends, low in middle)
    x = np.linspace(0, 1, n)
    anti = (1 - 2 * np.abs(x - 0.5))
    anti = np.maximum(anti, 0)
    anti += 0.1  # avoid all zeros
    anti /= anti.sum()
    candidates.append(anti)
    cand_names.append("anti_sym")
    
    # Random
    r = rng.exponential(1.0, n)
    r /= r.sum()
    candidates.append(r)
    cand_names.append("random")
    
    best_a = a.copy()
    best_score = init_score
    for i, (c, nm) in enumerate(zip(candidates, cand_names)):
        sc = fast_score(c)
        if sc < best_score:
            best_score = sc
            best_a = c.copy()
        print(f"  {nm}: {sc:.8f}")
    
    print(f"Best initial: {best_score:.8f}")
    
    # --- Multi-scale warmup (quick) ---
    if n >= 64:
        a_warm = multiscale_warmup(best_a, rng)
        sc_warm = fast_score(a_warm)
        if sc_warm < best_score:
            best_a = a_warm
            best_score = sc_warm
            print(f"Multi-scale improved: {best_score:.8f}")
    
    # --- Phase 1: Main FISTA (60% of remaining time) ---
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        t1 = time_left * 0.55
        print(f"Phase 1: FISTA ({t1:.0f}s)")
        a1, s1 = run_fista(best_a, t1, rng, step_init=0.5, top_k=5, top_k_max=60, restart_interval=200)
        if s1 < best_score:
            best_a, best_score = a1, s1
        print(f"Phase 1 result: {best_score:.8f}")
    
    # --- Phase 2: FISTA from perturbation (20%) ---
    time_left = total_deadline - time.monotonic()
    if time_left > 50:
        t2 = time_left * 0.35
        # Perturb best solution
        a_pert = best_a + rng.normal(0, 0.005 / np.sqrt(n), n)
        np.maximum(a_pert, 0, out=a_pert)
        a_pert /= a_pert.sum()
        print(f"Phase 2: FISTA from perturbation ({t2:.0f}s)")
        a2, s2 = run_fista(a_pert, t2, rng, step_init=0.5, top_k=8, top_k_max=80, restart_interval=150)
        if s2 < best_score:
            best_a, best_score = a2, s2
            print(f"Phase 2 improved: {best_score:.8f}")
        else:
            print(f"Phase 2: no improvement ({s2:.8f})")
    
    # --- Phase 3: Polishing with small steps and large active set (10%) ---
    time_left = total_deadline - time.monotonic()
    if time_left > 20:
        t3 = time_left * 0.5
        print(f"Phase 3: Polishing ({t3:.0f}s)")
        a3, s3 = run_fista(best_a, t3, rng, step_init=0.15, top_k=30, top_k_max=150, restart_interval=100)
        if s3 < best_score:
            best_a, best_score = a3, s3
            print(f"Phase 3 improved: {best_score:.8f}")
        else:
            print(f"Phase 3: no improvement ({s3:.8f})")
    
    # --- Phase 4: Random local search (remaining time) ---
    time_left = total_deadline - time.monotonic()
    if time_left > 15:
        t4 = time_left * 0.7
        print(f"Phase 4: Random local search ({t4:.0f}s)")
        a4, s4 = random_local_search(best_a, t4, rng)
        if s4 < best_score:
            best_a, best_score = a4, s4
            print(f"Phase 4 improved: {best_score:.8f}")
    
    # --- Final verification ---
    final_list = [float(x) for x in best_a]
    # Ensure all values are valid
    final_list = [max(0.0, min(1000.0, x)) for x in final_list]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.8f}")
    
    # Fallback if something went wrong
    if not np.isfinite(final_score) or final_score > init_score:
        print(f"Falling back to initial: {init_score:.8f}")
        final_list = [float(max(0.0, x)) for x in a]
        final_score = init_score
    
    print(f"=== FINAL: {final_score:.8f} (n={n}) ===")
    return final_list
