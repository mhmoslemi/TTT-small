```python
import numpy as np
import time

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1)) / max(n, 1)
    return a / s

def _project_simplex_true(v):
    """True Euclidean projection onto {x: sum x = 1, x >= 0}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    n = v.size
    if n == 0:
        return np.array([1.0])
    if n == 1:
        return np.array([1.0])
    # Algorithm from Goldstein & Tetral (2006)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    condition = mu - (cv - 1) / rho > 0
    rho_star = np.max(rho[condition])
    theta = (cv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)

def _autocorr(a, fft_size=None):
    n = a.size
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_size)
```

```python
Search for low-peak nonnegative self-convolutions using improved LSE optimization
with true Euclidean simplex projection, Nesterov momentum, cosine annealing,
and diverse multi-scale search.

Key improvements over previous approach:
- True Euclidean projection onto the simplex (water-filling)
- Nesterov momentum for faster convergence
- Adaptive step size with backtracking line search
- Cosine annealing schedule for LSE temperature (allows escape from local minima)
- Direct max subgradient refinement after LSE converges
- More diverse initializations including the provided sequence
- Multi-scale search with resampling

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _proj_simplex(v):
    """True Euclidean projection onto {x: x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    n = v.size
    if n == 0:
        return np.array([1.0])
    if n == 1:
        return np.array([1.0])
    # Water-filling algorithm (Goldstein & Tetral, 2006)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = mu - (cv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho_star = int(np.max(rho[cond]))
    theta = (cv[rho_star - 1] - 1.0) / rho_star
    result = np.maximum(v - theta, 0.0)
    s = float(result.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return result / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _compute_grad(a, weights, fft_size, n):
    """
    Gradient of sum_k weights_k * (a*a)_k w.r.t. a.
    (a*a)_k = sum_i a_i * a_{k-i}
    d(a*a)_k / da_j = a_{k-j} + a_{j-k} (out-of-range terms are 0)
    Total grad_j = sum_k weights_k * (a_{k-j} + a_{j-k})
    By symmetry of the convolution, this equals 2 * (w_rev * a)_j (appropriately sliced).
    """
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _lse_and_p(b, beta):
    """Return (p, lse_value) where p_k ~ exp(beta * b_k)."""
    if b.size == 0:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    fft_size = _next_pow2(2 * n - 1)
    b = _autocorr(a, fft_size)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _make_exp(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_mixture(n, alpha, gamma):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x) + float(gamma) * np.exp(-float(alpha) * (1.0 - x))
    a = a + 0.15
    return _normalize(a)


def _make_cosine_mod(n, freq, amp):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 1.0 + float(amp) * np.cos(2.0 * np.pi * float(freq) * x)
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _make_smooth_random(n, rng, base=None, amplitude=0.4):
    n = max(int(n), 2)
    if base is None:
        base = np.ones(n, dtype=np.float64) / n
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 16 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample_seq(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    s = float(old.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _block_perturb(a, rng, strength=0.2):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)


def _perturb_mix(a, rng):
    if rng.random() < 0.5:
        return _block_perturb(a, rng, 0.2)
    noise = np.exp(rng.normal(0.0, 0.25, size=a.size))
    a2 = a * noise + 0.01 * np.ones(a.size, dtype=np.float64) / a.size
    return _normalize(a2)


def _optimize_lse(a, deadline, rng, beta_start=8.0, beta_end=800.0, total_iters_est=5000):
    """
    LSE-based optimization with Nesterov momentum, true simplex projection,
    and cosine annealing for beta.
    """
    a = _normalize(a).copy()
    n = a.size
    if n < 8:
        a = _make_exp(128, 1.0)
        n = a.size

    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n)

    b = _autocorr(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    # Nesterov momentum
    momentum = 0.9
    v = np.zeros(n, dtype=np.float64)

    alpha = 0.3  # step size
    fail_count = 0
    last_improve = 0
    it = 0
    total_time = None

    while time.monotonic() < deadline:
        # Cosine annealing for beta
        progress = min(1.0, float(it) / float(total_iters_est))
        # Double cosine: goes up and comes back down periodically
        cycle = 4.0
        t = (progress * cycle) % 1.0
        beta_scale = 0.5 - 0.5 * np.cos(2.0 * np.pi * t)
        # Also increase the base beta over time
        base_beta = beta_start + (beta_end - beta_start) * progress
        beta = base_beta * (0.3 + 0.7 * beta_scale)
        beta = max(1.0, min(1e6, beta))

        p, smooth_old = _lse_and_p(b, beta)
        grad = _compute_grad(a, p, g_fft, n)

        # Center the gradient (remove mean for invariance to adding constant)
        grad_centered = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_centered)))

        if scale <= 1e-300:
            a = _perturb_mix(best_a, rng)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.3
            fail_count = 0
            v = np.zeros(n, dtype=np.float64)
            it += 1
            continue

        # Nesterov: take step in direction of v + gradient
        v = momentum * v - alpha * grad_centered
        a_new = _proj_simplex(a + v)

        # Evaluate new point
        b_new = _autocorr(a_new, b_fft)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * float(n) * maxb_new

        # Compute smooth value at new point
        _, smooth_new = _lse_and_p(b_new, beta)

        if smooth_new < smooth_old - 1e-15:
            a = a_new
            b = b_new
            maxb = maxb_new
            score = score_new
            # Increase step size
            alpha = min(2.0, alpha * 1.08)
            fail_count = 0
            last_improve = it
        else:
            # Backtrack
            alpha *= 0.5
            fail_count += 1
            # Don't accept the new point
            # Dampen momentum
            v *= 0.5

        if alpha < 0.003:
            alpha = 0.3
            v = np.zeros(n, dtype=np.float64)
            fail_count = 0

        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score

        # Restart if stuck
        if (fail_count > 500 or (it - last_improve > 1200 and it > 200)) and time.monotonic() < deadline - 5.0:
            a = _perturb_mix(best_a, rng)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.3
            fail_count = 0
            v = np.zeros(n, dtype=np.float64)
            beta = beta_start * 0.8  # reset beta lower for exploration

        it += 1

    return best_a, best_score


def _optimize_direct_max(a, deadline, rng, max_iters=50000):
    """
    Direct max subgradient optimization with line search.
    Used as a refinement step after LSE optimization.
    """
    a = _normalize(a).copy()
    n = a.size
    b_fft = _next_pow2(2 * n - 1)

    b = _autocorr(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    alpha = 1.0 / float(np.max(np.abs(a)) + 1e-10) * 0.01
    fail = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        b = _autocorr(a, b_fft)
        k_star = int(np.argmax(b))

        # Subgradient of (a*a)_{k*} w.r.t. a
        # (a*a)_{k*} = sum_i a_i * a_{k*-i}
        # d/da_j = a_{k*-j} + a_{j-k*} (out of range = 0)
        grad = np.zeros(n, dtype=np.float64)
        # a_{k*-j} is valid when 0 <= k*-j < n, i.e., k*-n+1 <= j <= k*
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n - 1, k_star)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += a[k_star - np.arange(lo1, hi1 + 1)]
        # a_{j-k*} is valid when 0 <= j-k* < n, i.e., k* <= j <= k*+n-1
        lo2 = max(0, k_star)
        hi2 = min(n - 1, k_star + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += a[np.arange(lo2, hi2 + 1) - k_star]

        # Take step
        grad_centered = grad - float(np.mean(grad))
        a_new = _proj_simplex(a - alpha * grad_centered)

        b_new = _autocorr(a_new, b_fft)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * float(n) * maxb_new

        if score_new < score * (1.0 - 1e-10):
            a = a_new
            score = score_new
            maxb = maxb_new
            alpha = min(alpha * 1.1, 1.0)
            fail = 0
            if score < best_score:
                best_a = a.copy()
                best_score = score
        else:
            alpha *= 0.7
            fail += 1

        if alpha < 1e-12 or fail > 200:
            break

        it += 1

    return best_a, best_score


def _get_given_sequence():

def _generate_inits(n, given, rng, best_a):
    n = max(int(n), 2)
    inits = []

    # Exponential with various rates
    inits.append(_make_exp(n, 1.0))
    inits.append(_make_exp(n, 0.7))
    inits.append(_make_exp(n, 1.5))

    # Cosine modulations
    inits.append(_make_cosine_mod(n, 2, 0.5))
    inits.append(_make_cosine_mod(n, 3, 0.4))
    inits.append(_make_cosine_mod(n, 5, 0.3))

    # Mixture (symmetric-ish)
    inits.append(_make_mixture(n, 1.0, 0.5))
    inits.append(_make_mixture(n, 0.8, 0.3))

    # Smooth random
    inits.append(_make_smooth_random(n, rng, _make_exp(n, 1.0), 0.5))
    inits.append(_make_smooth_random(n, rng, _make_exp(n, 0.7), 0.4))

    # Resample best
    if best_a is not None and best_a.size > 4:
        inits.append(_resample_seq(best_a, n))

    # Resample given
    if given is not None and given.size > 4:
        inits.append(_resample_seq(given, n))

    # Deduplicate
    seen = set()
    out = []
    for ic in inits:
        sig = tuple(ic[::max(1, ic.size // 16)].round(4))
        if sig not in seen:
            seen.add(sig)
            out.append(ic)
    return out

    if given is not None:
        given_score = _score_of(given)
        print(f"Given: len={given.size}, score={given_score:.8f}", flush=True)
        if given_score < best_score:
            best_a = given.copy()
            best_score = given_score
            print(f"  -> Adopted given as best", flush=True)

    # Phase 1: Quick exploration at small lengths
    print("\n=== Phase 1: Quick exploration ===", flush=True)
    phase1_deadline = t0 + max(30, budget * 0.08)

    for n in [128, 256, 512]:
        if time.monotonic() > phase1_deadline - 10:
            break
        inits = _generate_inits(n, given, rng, best_a)
        for init in inits[:4]:
            if time.monotonic() > phase1_deadline - 5:
                break
            t_end = min(phase1_deadline, time.monotonic() + max(5, (phase1_deadline - time.monotonic()) / 4 - 2))
            if time.monotonic() >= t_end:
                continue
            res_a, res_score = _optimize_lse(init.copy(), t_end, rng,
                                             beta_start=6.0, beta_end=200.0,
                                             total_iters_est=500)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = res_a.copy()
                best_score = float(res_score)
                print(f"  n={res_a.size} score={res_score:.8f} (improved!)", flush=True)

    print(f"Phase 1 best: score={best_score:.8f}, len={best_a.size}", flush=True)

    # Phase 2: Intensive optimization at the best length
    print("\n=== Phase 2: Intensive optimization ===", flush=True)
    phase2_deadline = t0 + max(60, budget * 0.75)
    n_best = best_a.size

    # Try a few lengths around the best
    candidate_lengths = [n_best]
    for dl in [n_best // 2, n_best * 2, n_best - 128, n_best + 128]:
        if 64 <= dl <= 16384 and dl not in candidate_lengths:
            candidate_lengths.append(dl)
    candidate_lengths.sort()

    for n in candidate_lengths:
        if time.monotonic() > phase2_deadline - 30:
            break
        # Resample best to this length
        init = _resample_seq(best_a, n)
        # Add slight perturbation
        init = _make_smooth_random(n, rng, init, 0.15)

        sec_left = max(20, phase2_deadline - time.monotonic() - 20)
        sec_alloc = min(sec_left, max(30, budget * 0.15))
        t_end = min(phase2_deadline - 5, time.monotonic() + sec_alloc)

        res_a, res_score = _optimize_lse(init.copy(), t_end, rng,
                                         beta_start=10.0, beta_end=1000.0,
                                         total_iters_est=3000)

        if np.isfinite(res_score) and res_score < best_score:
            best_a = res_a.copy()
            best_score = float(res_score)
            n_best = res_a.size
            print(f"  n={res_a.size} score={res_score:.8f} (improved!)", flush=True)

        # Direct max refinement
        if time.monotonic() < phase2_deadline - 15:
            t_end2 = min(phase2_deadline - 5, time.monotonic() + max(10, (phase2_deadline - time.monotonic()) * 0.2))
            res_a2, res_score2 = _optimize_direct_max(res_a.copy(), t_end2, rng)
            if np.isfinite(res_score2) and res_score2 < best_score:
                best_a = res_a2.copy()
                best_score = float(res_score2)
                n_best = res_a2.size
                print(f"  direct_max n={res_a.size} score={res_score2:.8f} (improved!)", flush=True)

    print(f"Phase 2 best: score={best_score:.8f}, len={best_a.size}", flush=True)

    # Phase 3: Try to upscale for better scores
    print("\n=== Phase 3: Upscale attempts ===", flush=True)
    phase3_deadline = t0 + max(60, budget * 0.90)

    if best_score > 1.51:
        n_cur = max(n_best, 512)
        while n_cur < 8192 and time.monotonic() < phase3_deadline - 30:
            n2 = min(8192, n_cur * 2)
            if n2 <= n_cur:
                break
            init = _resample_seq(best_a, n2)
            # Blend with a fresh exponential
            init = _normalize(0.9 * init + 0.1 * _make_exp(n2, 1.0))
            t_end = min(phase3_deadline - 5, time.monotonic() + max(30, budget * 0.08))
            res_a, res_score = _optimize_lse(init.copy(), t_end, rng,
                                             beta_start=12.0, beta_end=600.0,
                                             total_iters_est=2000)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = res_a.copy()
                best_score = float(res_score)
                n_cur = res_a.size
                print(f"  Upscale to {n_cur}: score={res_score:.8f} (improved!)", flush=True)
            else:
                print(f"  Upscale to {n2}: score={res_score:.8f} (no improvement)", flush=True)
                break

    # Phase 4: Final deep refinement
    print("\n=== Phase 4: Final refinement ===", flush=True)
    if time.monotonic() < deadline - 10:
        # LSE refinement
        t_end = min(deadline - 20, time.monotonic() + max(20, (deadline - time.monotonic()) * 0.6))
        res_a, res_score = _optimize_lse(best_a.copy(), t_end, rng,
                                         beta_start=50.0, beta_end=2000.0,
                                         total_iters_est=5000)
        if np.isfinite(res_score) and res_score < best_score:
            best_a = res_a.copy()
            best_score = float(res_score)
            print(f"  Final LSE: score={res_score:.8f}", flush=True)
        else:
            print(f"  Final LSE: no improvement, best={best_score:.8f}", flush=True)

        # Direct max refinement
        if time.monotonic() < deadline - 5:
            t_end2 = min(deadline - 2, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.3))
            res_a2, res_score2 = _optimize_direct_max(best_a.copy(), t_end2, rng)
            if np.isfinite(res_score2) and res_score2 < best_score:
                best_a = res_a2.copy()
                best_score = float(res_score2)
                print(f"  Final direct_max: score={res_score2:.8f}", flush=True)
            else:
                print(f"  Final direct_max: no improvement, best={best_score:.8f}", flush=True)

    # Final: try a few small perturbations of the best
    if time.monotonic() < deadline - 3:
        for _ in range(5):
            if time.monotonic() > deadline - 3:
                break
            pert = _block_perturb(best_a, rng, 0.05)
            s = _score_of(pert)
            if s < best_score:
                best_a = pert
                best_score = s
                print(f"  Perturbation improved: {best_score:.8f}", flush=True)

    elapsed = time.monotonic() - t0
    print(f"\nFinal: len={best_a.size}, score={best_score:.8f}, time={elapsed:.1f}s", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s > 0:
        out = out / s
    # Ensure sum is large enough for the evaluator
    if s < 0.01:
        out = out * 1.0

    return [float(x) for x in out]
```
