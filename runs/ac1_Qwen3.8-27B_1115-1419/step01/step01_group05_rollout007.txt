    s = float(a.sum())

    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)

def _conv_ab(a, b, fft_size):

    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)


def _lse_p(b, beta):
    """Probabilities p_k proportional to exp(beta*b_k) and LSE value."""
    m = float(np.max(b))
    if beta < 1.0:
        return np.full(b.size, 1.0 / b.size), m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s < 1e-300:
        e = np.ones(b.size)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _topk_p(b, k, temperature):
    """Weight top-k lags with soft assignment."""
    n_b = b.size
    m = float(np.max(b))
    if temperature < 1e-10:
        p = np.zeros(n_b)
        idx = np.argmax(b)
        p[idx] = 1.0
        return p, m
    arg = (b - m) / temperature
    arg = np.maximum(arg, -745.0)

    sorted_idx = np.argsort(b)[::-1]

def _project_simplex(v):
    """Euclidean projection onto {a >= 0, sum a = 1}."""
    n = v.size
    if n == 0:
        return np.array([1.0])
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

        s = float(np.sum(np.maximum(v - t, 0.0)))
        if s > 1.0:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    a = np.maximum(v - t, 0.0)
    s = float(a.sum())
    if s < 1e-300:

    return a / s

def _make_smooth_random(n, rng, amplitude=0.5):
    n = max(int(n), 2)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(3, min(len(F), n // 8 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0:
        f = f / sd
    base = np.ones(n)
    a = base * (1.0 + amplitude * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n) / new_n
    s = float(old.sum())

        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb(a, rng, strength=0.2):

    n = a.size

    k = max(2, n // 32)
    kernel = np.ones(k) / k
    noise = np.convolve(noise, kernel, mode='same')
    new = a * noise
    new = np.maximum(new, 1e-12)
    return _normalize(new)

    c = np.maximum(c, 1e-12)

    Returns (best_a, best_score).

    best_score = score

    beta = 20.0
    momentum = 0.0
    if use_momentum:
        momentum = 0.85

    it = 0
    fail = 0
    last_improve = 0
    best_smooth = float(np.max(b))

            p = 0.6 * p + 0.4 * p_top
            p = p / float(p.sum())

        # Center the gradient

        g_norm = float(np.max(np.abs(g)))
        if g_norm < 1e-300:

            score = 2.0 * n * maxb
            alpha = 0.01
            fail = 0
            v = np.zeros(n)
            it += 1
            continue

        step = v / max(1e-12, g_norm) * alpha

        smooth_try = float(np.max(b_try))

        if smooth_try < smooth_val - 1e-15:
            a = a_try
            b = b_try
            maxb = maxb_try
            score = score_try
            smooth_val = smooth_try
            alpha = min(0.1, alpha * 1.1)
            fail = 0
            last_improve = it
        else:

            fail += 1

            score = 2.0 * n * maxb
            alpha = 0.01
            fail = 0

            score = 2.0 * n * maxb
            beta = 15.0
            alpha = 0.008
            fail = 0

def _coord_descent_polish(a, t_end, rng, n_passes=3):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed.
    """
    a = _normalize(a.copy())
    n = a.size
    if n < 4:
        return a, _score(a)

    fft_s = _next_pow2(2 * n - 1)
    b = _conv_nn(a, fft_s)
    best_score = float(np.max(b)) * 2.0 * n
    best_a = a.copy()

    for _pass in range(n_passes):
        if time.monotonic() > t_end:
            break
        for i in range(0, n, max(1, n // 200)):
            if time.monotonic() > t_end - 1.0:
                break
            # Fix all except coordinate i, optimize a[i]
            # b_k = sum_j a_j * a_{k-j}
            # When varying a_i: b_k changes by a_i * (a_{k-i} + a_{k-i}) = 2*a_i*a_{k-i}
            # Actually b_k = old_b_k - 2*old_a_i*contrib + 2*new_a_i*contrib where
            # contrib_k = a_{k-i} (for valid k-i)
            # So b_k(new) = b_k(old) + 2*(new_a_i - old_a_i) * a_{k-i}
            
            old_val = a[i]
            contrib = np.zeros(2 * n - 1)
            # a_{k-i} is nonzero when 0 <= k-i < n, i.e., i <= k <= i+n-1

            # b_k new = b_k + 2*(new - old) * contrib_k

            # subject to sum a = 1 (adjust: if we change a[i], we need to renormalize)

            # This is a 1D problem: minimize max_k [b_k + 2*(new-old)*contrib_k]
            # over new >= 0, with sum constraint handled approximately.
            
            # Binary search: the optimal new makes the top of (b + 2*(new-old)*contrib) as flat as possible
            # Simple approach: try a few candidate values
            candidates = [0.0, old_val * 0.5, old_val, old_val * 1.5, old_val * 2.0, 4.0 * old_val]
            # Also try values that would equalize the top 2-3 lags
            top_idx = np.argsort(b)[::-1][:5]
            for ti in top_idx[:3]:
                if contrib[ti] > 0 and old_val > 0:
                    # b[ti] + 2*(new-old)*contrib[ti] = b[ti+1] (try to equalize)
                    target = b[top_idx[ti if ti < len(top_idx) - 1 else 0] + 1] if ti + 1 < len(b) else b[ti]
                    cand = (target - b[ti]) / (2.0 * contrib[ti]) + old_val
                    if cand > 0:
                        candidates.append(cand)
            
            best_new = old_val
            best_max = float(np.max(b))
            for cv in candidates:
                cv = max(0.0, cv)
                new_b = b + 2.0 * (cv - old_val) * contrib
                new_max = float(np.max(new_b))
                if new_max < best_max:
                    best_max = new_max
                    best_new = cv
            
            new_val = best_new
            a[i] = new_val
            # Quick renormalize
            s = float(a.sum())
            if s > 1e-300:
                a = a / s
            
            # Update b
            b = _conv_nn(a, fft_s)
            cur_max = float(np.max(b))
            if cur_max * 2.0 * n < best_score:
                best_score = cur_max * 2.0 * n
                best_a = a.copy()

    return best_a, best_score


def _generate_inits(n, rng, best_a=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    # Various parametric families
    inits.append(_make_exponential(n, 1.0))
    inits.append(_make_exponential(n, 2.0))
    inits.append(_make_exponential(n, 0.5))
    inits.append(_make_symmetric_gauss(n, 0.25))
    inits.append(_make_symmetric_gauss(n, 0.35))
    inits.append(_make_gaussian(n, 0.3, 0.2))
    inits.append(_make_gaussian(n, 0.7, 0.2))
    inits.append(_make_cosine_window(n))
    inits.append(_make_u_shape(n, 2.0))
    inits.append(_make_u_shape(n, 0.5))
    inits.append(_make_triangle(n))
    inits.append(_make_smooth_random(n, rng, 0.5))
    inits.append(_make_smooth_random(n, rng, 0.8))
    
    if best_a is not None and best_a.size > 4:
        inits.append(_resample(best_a, n))
        inits.append(_normalize(0.7 * _resample(best_a, n) + 0.3 * _make_exponential(n, 1.0)))
    
    return inits


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry. Searches for the best nonnegative sequence minimizing
    2*n*max(convolve(a,a))/(sum a)^2.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + max(budget, 60.0) - 25.0
    rng = np.random.default_rng(seed)

    # Try to get the provided sequence
    given = None
    for name in ('height_sequence_1', 'height_sequence_2', 'height_sequence_3'):
        if name in globals():
            try:
                arr = np.asarray(globals()[name], dtype=np.float64).ravel()
                arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
                arr = np.maximum(arr, 0.0)
                if arr.size >= 8 and float(arr.sum()) > 1e-6:
                    given = arr / float(arr.sum())
                    break
            except Exception:
                pass

    # Initial best
    best_a = _make_exponential(512, 1.5)
    best_score = _score(best_a)
    print(f"Initial: n={best_a.size}, score={best_score:.8f}", flush=True)

    if given is not None:
        g_score = _score(given)
        print(f"Given sequence: n={given.size}, score={g_score:.8f}", flush=True)
        if g_score < best_score:
            best_a = given.copy()
            best_score = g_score

    # Multi-stage search
    # Stage 1: Small n - find good shapes quickly
    stage1_configs = [
        (128, 4, 20),
        (256, 5, 30),
    ]

    for n, num_starts, sec_per in stage1_configs:
        if time.monotonic() > deadline - 30:
            break
        inits = _generate_inits(n, rng, best_a)
        for idx in range(min(num_starts, len(inits))):
            if time.monotonic() > deadline - 25:
                break
            t_end = min(deadline - 20, time.monotonic() + sec_per)
            res_a, res_sc = _refine(inits[idx], t_end, rng, n_topk=3)
            print(f"  n={res_a.size} init={idx} score={res_sc:.8f} best={best_score:.8f} t={time.monotonic()-(deadline-budget+25):.0f}s", flush=True)
            if res_sc < best_score:
                best_a = res_a.copy()
                best_score = res_sc

    # Stage 2: Medium n
    stage2_configs = [
        (512, 4, 50),
        (1024, 3, 70),
    ]

    for n, num_starts, sec_per in stage2_configs:
        if time.monotonic() > deadline - 60:
            break
        inits = _generate_inits(n, rng, best_a)
        for idx in range(min(num_starts, len(inits))):
            if time.monotonic() > deadline - 50:
                break
            t_end = min(deadline - 40, time.monotonic() + sec_per)
            res_a, res_sc = _refine(inits[idx], t_end, rng, n_topk=5)
            print(f"  n={res_a.size} init={idx} score={res_sc:.8f} best={best_score:.8f} t={time.monotonic()-(deadline-budget+25):.0f}s", flush=True)
            if res_sc < best_score:
                best_a = res_a.copy()
                best_score = res_sc

    # Stage 3: Large n - refine the best shape
    if time.monotonic() < deadline - 120:
        n_large = 2048
        init = _resample(best_a, n_large)
        # Blend with a fresh exponential to avoid degenerate resampling
        init = _normalize(0.8 * init + 0.2 * _make_exponential(n_large, 1.0))
        t_end = min(deadline - 60, time.monotonic() + 120)
        res_a, res_sc = _refine(init, t_end, rng, n_topk=7)
        print(f"  n={res_a.size} large score={res_sc:.8f} best={best_score:.8f}", flush=True)
        if res_sc < best_score:
            best_a = res_a.copy()
            best_score = res_sc

    # Stage 4: Even larger if time permits
    if time.monotonic() < deadline - 100:
        n_xl = 4096
        init = _resample(best_a, n_xl)
        init = _normalize(0.85 * init + 0.15 * _make_exponential(n_xl, 1.0))
        t_end = min(deadline - 40, time.monotonic() + 100)
        res_a, res_sc = _refine(init, t_end, rng, n_topk=7)
        print(f"  n={res_a.size} xl score={res_sc:.8f} best={best_score:.8f}", flush=True)
        if res_sc < best_score:
            best_a = res_a.copy()
            best_score = res_sc

    # Stage 5: Coordinate descent polish on best
    if time.monotonic() < deadline - 20:
        n_polish = min(best_a.size, 1024)
        a_polish = _resample(best_a, n_polish) if best_a.size != n_polish else best_a.copy()
        t_end = min(deadline - 10, time.monotonic() + 40)
        res_a, res_sc = _coord_descent_polish(a_polish, t_end, rng, n_passes=5)
        print(f"  polish n={res_a.size} score={res_sc:.8f} best={best_score:.8f}", flush=True)
        if res_sc < best_score:
            best_a = res_a.copy()
            best_score = res_sc

    # Stage 6: Final deep refinement
    if time.monotonic() < deadline - 5:
        t_end = deadline - 2.0
        res_a, res_sc = _refine(best_a.copy(), t_end, rng, n_topk=5)
        if res_sc < best_score:
            best_a = res_a.copy()
            best_score = res_sc
            print(f"  final refine score={res_sc:.8f}", flush=True)
        else:
            print(f"  final refine no improvement, best={best_score:.8f}", flush=True)

    # If still above target, try one more creative approach
    if best_score > 1.51 and time.monotonic() < deadline - 30:
        # Try a "dual exponential" shape: high at one end, with a secondary bump
        n_try = 512
        x = np.arange(n_try, dtype=np.float64) / n_try
        a_tw = np.exp(-3.0 * x) + 0.3 * np.exp(-3.0 * (1.0 - x)) + 0.1
        a_tw = _normalize(a_tw)
        t_end = min(deadline - 5, time.monotonic() + 40)
        res_a, res_sc = _refine(a_tw, t_end, rng, n_topk=5)
        print(f"  dual-exp score={res_sc:.8f} best={best_score:.8f}", flush=True)
        if res_sc < best_score:
            best_a = res_a.copy()
            best_score = res_sc

    # Final coordinate descent on the very best
    if time.monotonic() < deadline - 3:
        n_final = min(best_a.size, 512)
        a_f = _resample(best_a, n_final) if best_a.size != n_final else best_a.copy()
        t_end = deadline - 1.0
        res_a, res_sc = _coord_descent_polish(a_f, t_end, rng, n_passes=3)
        if res_sc < best_score:
            best_a = res_a
            best_score = res_sc

    # Prepare output
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = out / float(out.sum())

    final_score = _score(out)
    print(f"FINAL: n={out.size}, score={final_score:.8f}", flush=True)

    return [float(x) for x in out]
