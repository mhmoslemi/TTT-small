    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def _normalize(a, target=1.0):
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / s)

def _adam_init(n):
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    return m, v, t

def _adam_step(theta, m, v, t, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8):

    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    return theta, m, v, t

Actually, let me try a completely different approach: **use the fact that the optimal solution should have the property that the top peaks are equal**.

Algorithm:
1. Start with a.
2. Find the peak position k*.
3. Compute the "shadow" or "mirror" of a at k*: m[i] = a[k*-i].
4. The "excess" at k* is b[k*] = sum(a[i]*a[k*-i]) = dot(a, m).
5. To reduce b[k*], we want to decrease dot(a, m). The best direction (while maintaining sum(a)=1) is to move mass from positions where m is large to positions where m is small.
6. Specifically, the optimal update is: a_new[i] = max(0, a[i] - lr * (m[i] - mean(m))) then renormalize.

This is exactly the subgradient step! But let me think about the optimal learning rate.

If we take a step in the direction -m (the subgradient), the new value at k* is:
b_new[k*] = dot(a - lr*m + O(lr^2), a - lr*m) ≈ dot(a, a) - lr * dot(m, a) - lr * dot(a, m) + ...
= b[k*] - 2*lr*dot(a, m) + ...
= b[k*] - 2*lr*b[k*] + ...
= b[k*] * (1 - 2*lr) + ...

So for small lr, the peak decreases linearly. But other peaks might increase.

The optimal lr depends on the "curvature" of the objective, which is hard to compute exactly.

A practical approach: use a backtracking line search. Start with lr = lr_max, and if the score increases, halve lr.

def _subgrad_line_search(a, lr_init, max_iter=20):

    n = len(a)
    s = np.sum(a)
    best_a = a.copy()
    best_score = _score(a)
    
    for iteration in range(max_iter):
        lr = lr_init / (2 ** iteration)
        if lr < 1e-10:
            break

        k_star = int(np.argmax(b))
        m = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                m[i] = a[j]

        grad = m * s - b[k_star]  # *2 but we absorb the constant
        # Project to maintain sum: subtract mean
        grad = grad - np.mean(grad)

        a_new = _normalize(a_new, s)
        
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            lr_init = lr * 1.5  # can take a bigger step
        else:
            lr_init = lr
    
    return best_a, best_score, lr_init

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score(a):
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _normalize(a, s_target=1.0):
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, s_target / len(a))
    return a * (s_target / s)


def _mirror(a, k, n):
    m = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if hi >= lo:
        m[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return m


def _grad_smooth(a, n, tau):
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    shifted = b - max_b
    shifted = np.clip(shifted, -100.0, 0.0)
    es = np.exp(tau * shifted)
    w = es / np.sum(es)
    soft = max_b + np.log(np.sum(es)) / tau
    r = a[::-1]
    conv = _fft_conv(w, r)
    ga = conv[n - 1:n - 1 + n]
    return 2.0 * n * ga, 2.0 * n * soft


def _run_adam_softmax(a, n, deadline, lr_init=0.02, tau_start=2.0, tau_end=200.0):
    a = _normalize(np.maximum(a, 1e-10), 1.0)
    theta = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0
    best_patience_sc = best_sc
    b1, b2, eps = 0.9, 0.999, 1e-8
    min_time_per_step = 1.0 / 5000.0

    while time.monotonic() < deadline:
        progress = min(1.0, t / 30000.0)
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)
        cur_lr = lr_init * (1.0 - 0.6 * progress)

        a = np.exp(theta - np.max(theta))
        a /= a.sum()

        grad, _ = _grad_smooth(a, n, tau)
        gt = a * (grad - np.dot(grad, a))

        t += 1
        m = b1 * m + (1 - b1) * gt
        v = b2 * v + (1 - b2) * gt * gt
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

def _subgrad_polish(a, deadline, max_rounds=500, top_k=3):
    a = _normalize(np.maximum(a, 0), 1.0)
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.05
    no_improve = 0

    for _ in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b = _fft_conv(a, a)
        if top_k >= len(b):
            top_k = len(b)
        top_idx = np.argsort(b)[-top_k:]
        top_vals = b[top_idx]

        weights = top_vals - top_vals[-1] + 1e-10
        weights = weights / np.sum(weights)

        m = np.zeros(n)
        for w, k in zip(weights, top_idx):
            m += w * _mirror(a, int(k), n)

        weighted_peak = float(np.dot(weights, top_vals))
        grad = m - weighted_peak
        grad = grad - grad.mean()

        improved = False
        for trial in range(12):
            step = lr / (2 ** trial)
            if step < 1e-12:
                break
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)
            a_new = _normalize(a_new, 1.0)
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
                lr = min(step * 1.5, 0.5)
                improved = True
                no_improve = 0
                break
        if not improved:
            lr *= 0.5
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_sc
