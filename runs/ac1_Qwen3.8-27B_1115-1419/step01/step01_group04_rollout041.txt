def project_onto_simplex(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(a)
    # Water filling
    a_sorted = np.sort(a)[::-1]
    cssv = np.cumsum(a_sorted)
    rho = np.arange(1, n+1)
    cond = a_sorted - (cssv - target_sum) / rho > 0
    rho_star = np.max(rho[cond])
    tau = (cssv[rho_star-1] - target_sum) / rho_star
    return np.maximum(a - tau, 0)

def fast_convolution(a):
    """FFT-based linear convolution a*a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.maximum(c, 0.0)
    return c
