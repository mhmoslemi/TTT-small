    """True Euclidean projection onto {a : a>=0, sum(a)=1}."""
    n = v.size
    if n == 0:

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1] if np.any(u * np.arange(1,n+1) > (cssv-1)) else 0

    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]

    a = _normalize(a)
    n = a.size
    b = _autocorr(a, _next_pow2(2 * n - 1))
    mb = float(np.max(b))
    return 2.0 * float(n) * mb

def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.maximum(np.nan_to_num(old), 0.0)
    new_n = max(int(new_n), 2)
    s = old.sum()
    if s <= 0 or old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    xo = np.linspace(0.0, 1.0, old.size)
    xn = np.linspace(0.0, 1.0, new_n)
    return _normalize(np.interp(xn, xo, old))

def _smooth_perturb(a, rng, amp=0.3):
    n = a.size
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 16)
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n)
    f -= f.mean()
    sd = f.std()
    if sd > 0:
        f /= sd
    a2 = a * (1.0 + amp * f)
    return _normalize(np.maximum(a2, 1e-12))

def _refine_lse(a, deadline, rng, beta0=20.0):
    """LSE projected gradient descent."""
    a = _normalize(a)
    n = a.size
    fft_s = _next_pow2(2 * n - 1)
    fft_g = _next_pow2(3 * n)
    
    b = _autocorr(a, fft_s)
    maxb = float(np.max(b))
    score = 2.0 * n * maxb
    best_a = a.copy()
    best_score = score
    
    beta = beta0
    alpha = 0.4
    fail = 0
    it = 0
    last_impr = 0
    smooth_old = maxb  # initial
    
    while time.monotonic() < deadline:
        m = float(np.max(b))
        arg = beta * (b - m)
        arg = np.clip(arg, -50.0, 50.0)
        e = np.exp(arg)
        S = float(e.sum())
        if S < 1e-300:
            p = np.ones(b.size) / b.size
            smooth_val = m
        else:
            p = e / S
            smooth_val = m + float(np.log(S)) / beta
        
        # Gradient
        ar = a[::-1]
        A = np.fft.rfft(ar, fft_g)
        W = np.fft.rfft(p, fft_g)
        conv = np.fft.irfft(A * W, fft_g).real
        grad = 2.0 * conv[n - 1: 2 * n - 1]
        
        gc = grad - grad.mean()
        scale = float(np.max(np.abs(gc)))
        
        if scale < 1e-25:
            a = _smooth_perturb(best_a, rng, 0.2)
            b = _autocorr(a, fft_s)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.4
            fail = 0
            smooth_old = maxb
            it += 1
            continue
        
        a_new = a - alpha * gc
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        b = _autocorr(a_new, fft_s)
        m_new = float(np.max(b))
        arg2 = beta * (b - m_new)
        arg2 = np.clip(arg2, -50.0, 50.0)
        e2 = np.exp(arg2)
        S2 = float(e2.sum())
        smooth_new = m_new + (float(np.log(S2)) / beta if S2 > 1e-300 else 0.0)
        
        a = a_new
        maxb = m_new
        score = 2.0 * n * maxb
        
        if smooth_new < smooth_old - 1e-14:
            alpha = min(3.0, alpha * 1.12)
            fail = 0
            last_impr = it
        else:
            alpha *= 0.5
            fail += 1
        
        smooth_old = smooth_new
        
        if score < best_score - 1e-15:
            best_a = a.copy()
            best_score = score
        
        if it % 30 == 0:
            beta = min(800.0, beta * 1.08)
        
        if fail > 200:
            a = _smooth_perturb(best_a, rng, 0.15)
            b = _autocorr(a, fft_s)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.4
            fail = 0
            beta = max(beta0, beta * 0.6)
        
        if fail > 800 and time.monotonic() < deadline - 5:
            a = _smooth_perturb(best_a, rng, 0.4)
            b = _autocorr(a, fft_s)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.5
            fail = 0
            beta = beta0
        
        it += 1
    
    return best_a, best_score

def _scipy_polish(a, deadline):
    """Polish with scipy L-BFGS-B using softmax parameterization."""
    a = _normalize(a)
    n = a.size
    fft_s = _next_pow2(2 * n - 1)
    beta = 80.0
    
    def obj_grad(x):
        xm = np.max(x)
        ex = np.exp(x - xm)
        a_s = ex / ex.sum()
        
        b = _autocorr(a_s, fft_s)
        m = float(np.max(b))
        arg = beta * (b - m)
        arg = np.clip(arg, -60.0, 60.0)
        e = np.exp(arg)
        S = float(e.sum())
        if S < 1e-300:
            return 2.0 * n * m, np.zeros(n)
        p = e / S
        obj = 2.0 * n * (m + float(np.log(S)) / beta)
        
        m_b = float(np.dot(p, b))
        # s_j = sum_k p[k] * a_s[k-j]
        s = np.zeros(n)
        for j in range(n):
            ks = j
            ke = min(b.size, j + n)
            if ks < ke:
                s[j] = float(np.dot(p[ks:ke], a_s[ks - j:ke - j]))
        
        g = 2.0 * a_s * (s - m_b)
        return obj, g
    
    x0 = np.log(np.maximum(a, 1e-30))
    
    try:
        res = minimize(obj_grad, x0, jac=True, method='L-BFGS-B',
                       options={'maxiter': 300, 'gtol': 1e-10, 'maxfun': 500})
        xm = np.max(res.x)
        a_new = np.exp(res.x - xm)
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss > 1e-300:
            a_new = a_new / ss
        else:
            a_new = a.copy()
    except Exception:
        a_new = a.copy()
    
    return a_new

def _get_initials(n, rng, given, best):
    """Generate diverse initializations."""
    inits = []
    inits.append(_make_hann(n))
    inits.append(_make_blackman(n))
    inits.append(_make_exp(n, 1.0))
    inits.append(_make_exp(n, 0.7))
    inits.append(_make_exp(n, 1.5))
    inits.append(_make_cosine_pow(n, 2.0))
    inits.append(_make_cosine_pow(n, 3.0))
    inits.append(_make_mixed(n, 1.0, 0.6))
    
    if given is not None and given.size > 4:
        inits.append(_resample(given, n))
    if best is not None and best.size > 4:
        inits.append(_resample(best, n))
    
    # Add smooth random perturbations
    inits.append(_smooth_perturb(_make_hann(n), rng, 0.5))
    inits.append(_smooth_perturb(_make_exp(n, 1.0), rng, 0.5))
    inits.append(_smooth_perturb(_make_blackman(n), rng, 0.4))
    
    # Deduplicate
    seen = set()
    out = []
    for ic in inits:
        key = ic[::max(1, ic.size // 16)].round(4).tobytes()
        if key not in seen:
            seen.add(key)
            out.append(ic)
    return out

def propose_candidate(seed=0, budget_s=2400, **kwargs):

    """
    budget = max(float(budget_s), 60.0)
    t0 = time.monotonic()
    deadline = t0 + budget - 30.0
    rng = np.random.default_rng(seed)

        if seq is not None:

            given = np.maximum(np.nan_to_num(given), 0.0)
            s = given.sum()
            if s > 1e-300:
                given = given / s
            else:
                given = None
    except Exception:
        given = None
    
    # Initial fallback
    best_a = _make_hann(256)
    best_score = _score_of(best_a)
    print(f"Initial: n={best_a.size} score={best_score:.8f}", flush=True)
    
    # If given sequence is good, use it
    if given is not None:
        gs = _score_of(given)
        print(f"Given: n={given.size} score={gs:.8f}", flush=True)
        if gs < best_score:
            best_a = given.copy()
            best_score = gs
    
    # Staged search over different lengths
    lengths = [128, 192, 256, 320, 384, 512]
    
    # Allocate time: more time for promising lengths
    # Rough allocation: 128: 40s, 192: 50s, 256: 120s, 320: 80s, 384: 80s, 512: 100s
    time_alloc = {128: 40, 192: 50, 256: 130, 320: 80, 384: 80, 512: 100}
    
    for n in lengths:
        if time.monotonic() > deadline - 40:
            break
        
        alloc = time_alloc.get(n, 60)
        if time.monotonic() + alloc * 0.3 > deadline:
            alloc = max(20, int((deadline - time.monotonic()) * 0.2))
        
        inits = _get_initials(n, rng, given, best_a)
        
        # For each init, run LSE refinement
        n_starts = min(len(inits), max(2, alloc // 25))
        
        for idx in range(n_starts):
            if time.monotonic() > deadline - 35:
                break
            t_start = time.monotonic()
            t_end = min(deadline - 30, t_start + alloc / max(n_starts, 1))
            
            if t_end <= t_start:
                continue
            
            res_a, res_s = _refine_lse(inits[idx].copy(), t_end, rng, beta0=20.0)
            
            # Also try with different beta
            if time.monotonic() < deadline - 30 and res_s < best_score:
                t_end2 = min(deadline - 25, time.monotonic() + 15)
                res_a2, res_s2 = _refine_lse(res_a.copy(), t_end2, rng, beta0=60.0)
                if res_s2 < res_s:
                    res_a, res_s = res_a2, res_s2
            
            print(f"n={res_a.size} init={idx}/{n_starts} score={res_s:.8f} best={best_score:.8f} t={time.monotonic()-t0:.0f}s", flush=True)
            
            if res_s < best_score:
                best_a = res_a.copy()
                best_score = res_s
    
    # Scipy polishing phase
    if time.monotonic() < deadline - 10:
        print(f"Scipy polish: n={best_a.size} score={best_score:.8f} t={time.monotonic()-t0:.0f}s", flush=True)
        polished = _scipy_polish(best_a.copy(), deadline - 5)
        ps = _score_of(polished)
        print(f"After scipy: score={ps:.8f} best={best_score:.8f}", flush=True)
        if ps < best_score:
            best_a = polished
            best_score = ps
    
    # Final LSE refinement
    if time.monotonic() < deadline - 3:
        fa, fs = _refine_lse(best_a.copy(), deadline - 2, rng, beta0=80.0)
        if fs < best_score:
            best_a = fa
            best_score = fs
            print(f"Final LSE: score={fs:.8f}", flush=True)
        else:
            print(f"Final LSE: no improvement, best={best_score:.8f}", flush=True)
    
    # Final output
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)
    
    final_score = _score_of(out)
    print(f"FINAL: n={out.size} score={final_score:.8f}", flush=True)
    
    return [float(x) for x in out]
