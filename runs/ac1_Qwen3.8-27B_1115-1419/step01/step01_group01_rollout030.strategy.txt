def objective_function(a: List[float]) -> float:
    n = len(a)
    s = np.float64(sum(a))
    # Protect against invalid sequences
    if s < 0.01: return np.inf  # Lower sum bound
    conv = np.convolve(a, a, mode="full")
    b_max = np.float64(max(conv))
    # Ensure positive max
    if b_max <= 0.0: return np.inf  # Undefined max
    return float(2*n*b_max/(s*s))

def gradient(a: List[float],index_peak: bool) -> List[float]:
    """Compute gradient based on convolution maximum and window information."""
    conv = np.convolve(a, a, mode="full")
    peak_index = int(np.argmax(conv))
    avg_coef = np.float64(a[len(a)//2])  # Typical value
    # Derivative respect to peak term
    grad = np.zeros(len(a))
    for i in range(max(1,peak_index-5),min(peak_index+5,len(conv))):
        idx = i - len(a) + 1
        if 0 <= idx < len(a):
            grad[idx] = -conv[i]/(s*s)  # Simplified gradient component
    # Additional spectral terms not evaluated directly
    grad[peak_index-len(a)+1] += 0.3/s**2 * (1.0+0.2*np.log(avg_coef))
    return np.array(grad)
