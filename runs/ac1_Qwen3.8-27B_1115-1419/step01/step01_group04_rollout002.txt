Search for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2 using Adam optimization on log-space,
peak-lag targeting, genetic crossover, and simulated annealing.
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa*fa, N)[:L]
    return np.maximum(c, 0)

def compute_score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = fast_conv(a)
    return float(2.0 * n * np.max(c) / (s*s))

def normalize(a, target_sum=None):
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones_like(a)
    if target_sum is None:
        target_sum = float(len(a))
    a = a * (target_sum / s)
    return np.clip(a, 0, 1000.0)

def peak_subgrad(a, c, K=5):
    """Subgradient of max(c) w.r.t. a, averaged over top K peaks."""
    n = len(a)
    L = 2*n - 1
    if K >= L:
        peaks = np.arange(L)
    else:
        peaks = np.argpartition(c, -K)[-K:]
    
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        # dc_k/da_m = 2*a_{k-m} if 0<=k-m<n else 0
        valid_m = np.arange(max(0, k-n+1), min(n, k+1))
        if len(valid_m) > 0:
            indices = k - valid_m
            grad[valid_m] += 2.0 * a[indices]
    return grad / K

def adam_log_update(a, c, params, K=5, lr=0.05):
    """Adam update on log(a) with peak subgradient."""
    n = len(a)
    grad = peak_subgrad(a, c, K)
    
    # Center gradient to preserve sum
    grad = grad - np.sum(grad) * a / np.sum(a)  # weighted centering
    
    # Actually, for log-space: d/d(log a_i) f = a_i * df/da_i
    log_grad = a * grad
    log_grad -= np.sum(log_grad)  # to preserve sum (approximately)
    
    # Adam update
    params['m'] = 0.9 * params['m'] + 0.1 * log_grad
    params['v'] = 0.99 * params['v'] + 0.01 * log_grad**2
    params['t'] += 1
    
    m_hat = params['m'] / (1 - 0.9**params['t'])
    v_hat = params['v'] / (1 - 0.99**params['t'])
    
    new_log = np.log(np.maximum(a, 1e-10)) - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    new_a = np.exp(np.clip(new_log, -10, 10))
    new_a = normalize(new_a, float(n))
    return new_a

Search for non-negative step-function heights minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

Strategy:
- Use FFT-based convolution for fast objective evaluation.
- Adam optimizer on log(a) with top-K peak subgradient.

import numpy as np
import time


def fast_conv(a):
    """FFT-based linear self-convolution, length 2n-1."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def compute_score(a):
    """Fast score: 2*n*max(c)/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def normalize(a, target=None):
    """Clip to [0,1000] and normalize to target sum (default: len(a))."""
    a = np.array(a, dtype=np.float64, copy=True)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if target is None:
        target = float(len(a))
    s = float(np.sum(a))
    if s < 1e-15:
        a.fill(target / max(1, len(a)))
        return np.clip(a, 0.0, 1000.0)
    a *= target / s
    return np.clip(a, 0.0, 1000.0)


def peak_subgrad(a, c, K=20):
    """
    Averaged subgradient of c over its top-K values.
    dc_k/da_j = 2*a_{k-j} if 0<=k-j<n, else 0.
    Returns gradient in a-space (shape n).
    """
    n = len(a)
    L = 2 * n - 1
    if K >= L:
        peaks = np.arange(L)
    else:
        peaks = np.argpartition(c, -K)[-K:]
    peaks = list(peaks)

    grad = np.zeros(n, dtype=np.float64)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo >= hi:
            continue
        j_indices = np.arange(lo, hi)
        mirror = k - j_indices
        # All mirror values are valid by construction (0<=mirror<n)
        grad[j_indices] += 2.0 * a[mirror]
    return grad / float(len(peaks))


def adam_step(a, grad_a, state, lr):
    """
    Adam update in log-space to minimize peak while preserving sum.
    grad_a: subgradient in a-space.
    state: dict with 'm', 'v', 't' arrays/scalars.
    Returns new a (normalized).
    """
    n = len(a)
    # Convert to log-space gradient: d f / d log(a_j) = a_j * d f / d a_j
    lg = a * grad_a
    # Project to preserve sum: subtract weighted mean
    w = a / (np.sum(a) + 1e-15)
    lg -= np.sum(lg * w) / np.sum(w)

    # Adam
    beta1, beta2 = 0.9, 0.999
    state['t'] += 1
    t = state['t']
    state['m'] = beta1 * state['m'] + (1 - beta1) * lg
    state['v'] = beta2 * state['v'] + (1 - beta2) * (lg * lg)
    m_hat = state['m'] / (1 - beta1 ** t)
    v_hat = state['v'] / (1 - beta2 ** t)
    log_a = np.log(np.maximum(a, 1e-12)) - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    new_a = np.exp(np.clip(log_a, -20, 20))
    new_a = normalize(new_a, float(n))
    return new_a


def targeted_peak_reduction(a, c, top_frac=0.05, strength=0.15):
    """
    Directly reduce elements that contribute most to the top peak.
    For peak lag k*, contribution of a_j is a_j * a_{k* - j}.
    """
    n = len(a)
    k_star = int(np.argmax(c))
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    indices = np.arange(lo, hi)
    contributions = a[indices] * a[k_star - indices]
    max_contrib = float(np.max(contributions))
    if max_contrib < 1e-15:
        return a

    # Identify top-contributing elements
    threshold = max_contrib * (1.0 - top_frac)
    mask = contributions > threshold
    if not np.any(mask):
        return a

    # Reduce top contributors
    a = a.copy()
    a[indices[mask]] *= (1.0 - strength)
    # Slightly boost low contributors to maintain structure
    low_mask = contributions < max_contrib * 0.3
    if np.any(low_mask):
        a[indices[low_mask]] *= (1.0 + strength * 0.5)

    return normalize(a, float(n))

    # Ramp + oscillation
    for alpha in (-1.0, -0.5, 0.5, 1.0):
        starts.append(1.0 + alpha * (2 * t - 1) + 0.3 * np.sin(4 * np.pi * t))

    # Parent upsampled
    if parent is not None and len(parent) > 1:
        p = np.array(parent, dtype=np.float64)
        x1 = np.linspace(0, 1, len(p))
        x2 = np.linspace(0, 1, n)
        p_resized = np.interp(x2, x1, p)
        starts.append(p_resized)
        # Also add slightly perturbed parent
        starts.append(p_resized * np.exp(rng.normal(0, 0.05, n)))

    # Normalize and filter
    valid = []
    for a in starts:
        a = normalize(np.asarray(a, dtype=np.float64), float(n))
        if np.all(np.isfinite(a)) and np.sum(a) > 0.01 and a.size == n:
            valid.append(a)

    if not valid:
        valid.append(np.ones(n))
    return valid


def local_search(a, budget, seed=0, K_init=30, K_min=3):
    """
    Adam-based local search with adaptive peak focus and restarts.
    Returns (best_sequence, best_score).
    """
    n = len(a)
    a = normalize(a, float(n))
    rng = np.random.default_rng(seed)

    c = fast_conv(a)
    score = compute_score(a)

    best_a = a.copy()
    best_score = score

    state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
    lr = 0.08
    stagnant = 0
    t_start = time.monotonic()
    deadline = t_start + budget
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # Adaptive K: start broad, narrow over time
        progress = min(1.0, it / 8000.0)
        K = max(K_min, int(K_init * (1.0 - 0.7 * progress)))

        # Compute gradient
        grad = peak_subgrad(a, c, K=K)

        # Adam step
        new_a = adam_step(a, grad, state, lr)

        # Evaluate
        new_c = fast_conv(new_a)
        new_score = compute_score(new_a)

        if not np.isfinite(new_score):
            lr *= 0.7
            state['m'] *= 0.5
            state['v'] *= 0.5
            stagnant += 1
            if stagnant > 200:
                # Hard restart
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = normalize(a, float(n))
                c = fast_conv(a)
                score = compute_score(a)
                state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
                lr = 0.06
                stagnant = 0
            continue

        if new_score < best_score:
            best_a = new_a.copy()
            best_score = new_score
            stagnant = 0
            lr = min(0.12, lr * 1.01)
        elif new_score <= score * 1.005:
            a = new_a
            c = new_c
            score = new_score
            stagnant += 1
            lr *= 0.98
        else:
            lr *= 0.85
            state['m'] *= 0.7
            stagnant += 1

        # Occasional targeted peak reduction
        if stagnant % 200 == 100 and stagnant < 2000:
            tp_a = targeted_peak_reduction(a, c, top_frac=0.1, strength=0.08)
            tp_c = fast_conv(tp_a)
            tp_score = compute_score(tp_a)
            if tp_score < score:
                a = tp_a
                c = tp_c
                score = tp_score
                if score < best_score:
                    best_a = a.copy()
                    best_score = score
                    stagnant = 0

        # Simulated annealing restart
        if stagnant > 800:
            temp = 0.05 + 0.03 * np.log1p(stagnant / 400)
            a = best_a * np.exp(rng.normal(0, temp, n))
            a = normalize(a, float(n))
            c = fast_conv(a)
            score = compute_score(a)
            if score > best_score * 1.05:
                a = best_a.copy()
                c = fast_conv(a)
                score = best_score
            state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
            lr = 0.06
            stagnant = 0

        if stagnant > 5000:
            break

    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search entry point.
    """
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    rng = np.random.default_rng(12345)

    # Get parent
    parent_seq = None
    try:
        parent_seq = globals().get('height_sequence_1', None)
        if parent_seq is not None:
            parent_seq = np.array(parent_seq, dtype=np.float64)
            if not np.all(np.isfinite(parent_seq)):
                parent_seq = None
    except Exception:
        parent_seq = None

    if budget < 30:
        return [1.0] * 1024

    best_seq = None
    best_score = float('inf')

    # --- Phase 1: Quick exploration at n=1024 (up to 120s) ---
    n1 = 1024
    phase1_end = t0 + min(120.0, budget * 0.08)
    print(f"[Phase 1] n={n1}, budget={phase1_end - t0:.0f}s")

    starts = generate_starts(n1, rng, parent_seq)
    scored = []
    for a in starts:
        sc = compute_score(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda x: x[0])
    print(f"[Phase 1] {len(scored)} starts, best={scored[0][0]:.6f}")

    top1 = scored[:min(4, len(scored))]
    for i, (sc, a) in enumerate(top1):
        elapsed = time.monotonic() - t0
        remain = phase1_end - t0 - elapsed
        if remain < 3:
            break
        alloc = max(2.0, remain / (len(top1) - i))
        seq, score = local_search(a, alloc, seed=100 + i, K_init=20, K_min=3)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq
            print(f"[Phase 1] start {i}: {score:.6f}")

    # --- Phase 2: Refine at n=2048 (main phase) ---
    n2 = 2048
    phase2_end = t0 + min(budget - 90.0, t0 + budget * 0.82) - t0 + t0
    phase2_end = t0 + (budget * 0.82)
    print(f"[Phase 2] n={n2}, deadline={phase2_end - t0:.0f}s")

    # Build parent for phase 2
    parent2 = None
    if best_seq is not None:
        x1 = np.linspace(0, 1, len(best_seq))
        x2 = np.linspace(0, 1, n2)
        parent2 = np.interp(x2, x1, best_seq)
    elif parent_seq is not None:
        x1 = np.linspace(0, 1, len(parent_seq))
        x2 = np.linspace(0, 1, n2)
        parent2 = np.interp(x2, x1, parent_seq)

    starts2 = generate_starts(n2, rng, parent2)
    scored2 = []
    for a in starts2:
        sc = compute_score(a)
        if np.isfinite(sc):
            scored2.append((sc, a))
    scored2.sort(key=lambda x: x[0])
    print(f"[Phase 2] {len(scored2)} starts, best={scored2[0][0]:.6f}")

    # Run local search on top candidates
    top2 = scored2[:min(6, len(scored2))]
    for i, (sc, a) in enumerate(top2):
        elapsed = time.monotonic() - t0
        remain = phase2_end - t0 - elapsed
        if remain < 15:
            break
        alloc = max(10.0, remain / (len(top2) - i))
        print(f"[Phase 2] start {i} (score={sc:.6f}), alloc={alloc:.0f}s")
        seq, score = local_search(a, alloc, seed=1000 + i * 7, K_init=40, K_min=4)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq
            print(f"[Phase 2] New best: n={n2}, score={score:.6f}")

        # Crossover with current best
        if i > 0 and best_seq is not None:
            elapsed2 = time.monotonic() - t0
            remain2 = phase2_end - t0 - elapsed2
            if remain2 > 30:
                child = crossover_seqs(a, best_seq, rng, noise=0.02)
                alloc2 = max(10.0, remain2 * 0.4)
                seq2, score2 = local_search(child, alloc2, seed=5000 + i * 13, K_init=30, K_min=3)
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_seq = seq2
                    print(f"[Phase 2] Crossover {i} improved: {score2:.6f}")

    # --- Phase 3: Final polish ---
    phase3_end = t0 + (budget - 30.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        if remaining > 20:
            print(f"[Phase 3] Final polish, {remaining:.0f}s")
            seq, score = local_search(best_seq, remaining, seed=99999, K_init=20, K_min=2)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq
                print(f"[Phase 3] Final: {score:.6f}")

    # --- Validation with exact evaluator ---
    if best_seq is not None:
        seq_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            exact_sc = evaluate_sequence(seq_list)
            print(f"[Validate] Exact score: {exact_sc:.6f}")
            if np.isfinite(exact_sc):
                return seq_list
        except Exception as e:
            print(f"[Validate] Error: {e}")

    # Fallback
    print("[Fallback] Using uniform sequence")
    return [1.0] * 1024
