    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _convolve(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _project_simplex(v):
    """Exact Euclidean projection onto {x : sum x = 1, x >= 0}."""
    n = v.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return w  # Already sums to 1 by construction


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _convolve(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _lse_and_p(b, beta):
    """Return (p_k proportional to exp(beta*b_k), lse_value)."""
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.ones(b.size, dtype=np.float64) / b.size
        return p, m
    arg = beta * (b - m)
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s < 1e-300:
        p = np.zeros(b.size, dtype=np.float64)
        p[np.argmax(b)] = 1.0
        return p, m
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _compute_grad(a, p, fft_size):
    """Gradient of sum_k p_k*(a*a)_k w.r.t. a. Uses FFT."""
    n = a.size
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def _make_exponential(n, alpha):
    n = max(int(n), 4)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_bell(n, center, width):
    n = max(int(n), 4)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - float(center)) / float(width)) ** 2)
    return _normalize(a)


def _make_asymmetric(n, left_alpha, right_alpha):
    """Two-sided exponential: peaks where left/right balance is asymmetric."""
    n = max(int(n), 4)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-float(left_alpha) * x - float(right_alpha) * (1.0 - x))
    return _normalize(a)


def _make_smooth_random(n, rng, base=None, amplitude=0.4):
    n = max(int(n), 4)
    if base is None:
        base = _make_exponential(n, 1.0)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 8 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 4)
    if old.size == 0:
        return _make_exponential(new_n, 1.0)
    s = float(old.sum())
    if s < 1e-300:
        return _make_exponential(new_n, 1.0)
    old = old / s
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb(a, rng, mode=0, strength=0.2):
    """Diverse perturbation strategies."""
    n = a.size
    if mode == 0:
        # Block perturbation
        c = a.copy()
        width = max(2, n // 16)
        start = int(rng.integers(0, max(1, n - width)))
        c[start:start + width] *= np.exp(rng.normal(0.0, float(strength)))
    elif mode == 1:
        # Global noise
        noise = np.exp(rng.normal(0.0, float(strength), size=n))
        c = a * noise
    elif mode == 2:
        # Mirror blend
        c = a * rng.random() + a[::-1] * rng.random()
        c = c + 0.05 * rng.random(n)
    elif mode == 3:
        # Large random kick
        mask = rng.random(n) < 0.3
        c = a.copy()
        c[mask] *= np.exp(rng.normal(0.0, 0.5, size=int(mask.sum())))
    else:
        c = a * np.exp(rng.normal(0.0, float(strength), size=n))
    return _normalize(c)


def _refine(a, t_end, rng):
    """
    LSE-projected gradient descent with backtracking on true score.
    Returns (best_a, best_score).
    """
    a = _normalize(a.copy())
    n = a.size
    if n < 8:
        a = _make_exponential(64, 1.0)
        n = a.size

    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n - 2)

    b = _convolve(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    alpha = 0.1
    beta_target = 8.0
    it = 0
    stale = 0

    while time.monotonic() < t_end:
        maxb = max(maxb, 1e-300)
        beta = float(min(1e6, max(1.0, beta_target / maxb)))

        p, smooth_val = _lse_and_p(b, beta)
        grad = _compute_grad(a, p, g_fft)

        # Subtract mean to project onto sum=0 subspace (tangent to simplex)
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale < 1e-300:
            # Stuck: restart
            a = _perturb(best_a, rng, mode=int(rng.integers(0, 4)), strength=0.25)
            b = _convolve(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.1
            beta_target = max(6.0, beta_target * 0.5)
            it += 1
            continue

        # Try step with backtracking
        step_taken = False
        for _bt in range(20):
            v = a - alpha * grad_c / scale
            new_a = _project_simplex(v)
            new_b = _convolve(new_a, b_fft)
            new_maxb = float(np.max(new_b))
            new_score = 2.0 * float(n) * new_maxb

            if new_score < score - 1e-15:
                a = new_a
                b = new_b
                maxb = new_maxb
                score = new_score
                step_taken = True
                break
            alpha *= 0.7

        if step_taken:
            if score < best_score - 1e-14:
                best_a = a.copy()
                best_score = score
                stale = 0
            else:
                stale += 1
            alpha = float(min(2.0, alpha * 1.15))
        else:
            alpha *= 0.5
            stale += 1

        # Adapt beta
        if it % 50 == 0:
            beta_target = float(min(500.0, beta_target + 2.0))

        # Periodic restarts
        if (stale > 300 or (it > 500 and stale > 150)) and time.monotonic() < t_end - 0.5:
            mode = int(rng.integers(0, 4))
            a = _perturb(best_a, rng, mode=mode, strength=0.3)
            b = _convolve(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.1
            beta_target = max(6.0, beta_target * 0.4)
            stale = 0

        # If we've drifted too far, reset
        if score > 1.5 * best_score + 0.05:
            a = best_a.copy()
            b = _convolve(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            beta_target = 8.0
            alpha = 0.1
            stale = 0

        it += 1

    return best_a, best_score


def _refine_subgrad(a, t_end, rng):
    """
    Subgradient method: directly chase the active peak.
    At each step, compute the argmax of a*a and take a step in the
    negative direction of that peak's gradient.
    """
    a = _normalize(a.copy())
    n = a.size
    if n < 8:
        a = _make_exponential(64, 1.0)
        n = a.size

    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n - 2)

    b = _convolve(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    alpha = 0.05
    stale = 0

    while time.monotonic() < t_end:
        # Find active peak(s): use top few peaks with soft weighting
        k = 2 * n - 1
        # Use a modest softmax over top peaks
        sorted_idx = np.argsort(b)[::-1]
        top_m = min(8, k)
        top_idx = sorted_idx[:top_m]
        top_vals = b[top_idx]
        # Soft weights: exp(-gap / temp)
        temp = max(1e-10, (top_vals[0] - top_vals[-1]) + 1e-10) * 0.1
        w = np.exp(-(top_vals[0] - top_vals) / temp)
        w = w / w.sum()

        # Build sparse weight vector
        p = np.zeros(k, dtype=np.float64)
        p[top_idx] = w

        # Compute gradient
        grad = _compute_grad(a, p, g_fft)
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale < 1e-300:
            a = _perturb(best_a, rng, mode=int(rng.integers(0, 4)), strength=0.2)
            b = _convolve(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            continue

        # Backtracking
        step_taken = False
        for _bt in range(15):
            v = a - alpha * grad_c / scale
            new_a = _project_simplex(v)
            new_b = _convolve(new_a, b_fft)
            new_score = 2.0 * float(n) * float(np.max(new_b))
            if new_score < score - 1e-15:
                a = new_a
                b = new_b
                maxb = float(np.max(new_b))
                score = new_score
                step_taken = True
                break
            alpha *= 0.7

        if step_taken:
            if score < best_score - 1e-14:
                best_a = a.copy()
                best_score = score
                stale = 0
            else:
                stale += 1
            alpha = float(min(1.0, alpha * 1.1))
        else:
            alpha *= 0.5
            stale += 1

        if stale > 200 and time.monotonic() < t_end - 0.5:
            a = _perturb(best_a, rng, mode=int(rng.integers(0, 4)), strength=0.3)
            b = _convolve(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.05
            stale = 0

    return best_a, best_score


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0:
        return None
    s = float(arr.sum())
    if s < 1e-300:
        return None
    return arr / s


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point. Runs for at most budget_s seconds and returns the best
    nonnegative coefficient list found.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + max(budget, 60.0) - 20.0
    rng = np.random.default_rng(seed)

    given = _get_given_sequence()

    # Initial best: use given sequence if available, else exponential
    if given is not None and given.size >= 16:
        best_a = given.copy()
    else:
        best_a = _make_exponential(256, 1.0)
    best_score = _score_of(best_a)
    print(f"Initial best: len={best_a.size} score={best_score:.8f}", flush=True)

    scale = float(max(0.1, min(1.0, budget / 2400.0)))

    # === Phase 1: Refine at the given/best length ===
    n0 = best_a.size
    if time.monotonic() < deadline - 60:
        t_end = min(deadline - 40, time.monotonic() + 300 * scale)
        res_a, res_score = _refine(best_a.copy(), t_end, rng)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"Phase1a: len={best_a.size} score={best_score:.8f}", flush=True)

        # Subgradient refinement
        if time.monotonic() < deadline - 40:
            t_end = min(deadline - 20, time.monotonic() + 150 * scale)
            res_a, res_score = _refine_subgrad(best_a.copy(), t_end, rng)
            if res_score < best_score:
                best_a = res_a.copy()
                best_score = res_score
                print(f"Phase1b (subgrad): len={best_a.size} score={best_score:.8f}", flush=True)

    # === Phase 2: Try different initializations at same length ===
    n0 = best_a.size
    inits = []
    # Asymmetric shapes
    inits.append(_make_asymmetric(n0, 2.0, 0.5))
    inits.append(_make_asymmetric(n0, 0.8, 1.5))
    inits.append(_make_exponential(n0, 0.7))
    inits.append(_make_exponential(n0, 1.5))
    inits.append(_make_bell(n0, 0.3, 0.25))
    inits.append(_make_bell(n0, 0.7, 0.25))
    inits.append(_make_smooth_random(n0, rng, _make_asymmetric(n0, 1.5, 0.8), 0.5))
    inits.append(_make_smooth_random(n0, rng, best_a, 0.3))

    # Remove duplicates (rough check)
    seen_sigs = set()
    unique_inits = []
    for init in inits:
        sig = tuple(init[::max(1, init.size // 16)].round(4))
        if sig not in seen_sigs:
            seen_sigs.add(sig)
            unique_inits.append(init)

    per_init_time = 40 * scale
    for idx, init in enumerate(unique_inits):
        if time.monotonic() > deadline - 30:
            break
        t_end = min(deadline - 20, time.monotonic() + per_init_time)
        if time.monotonic() >= t_end:
            continue
        res_a, res_score = _refine(init, t_end, rng)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"Phase2 init={idx}: len={best_a.size} score={best_score:.8f}", flush=True)

    # === Phase 3: Try other lengths ===
    other_lengths = [n0 // 2, n0 * 2, n0 * 3, n0 // 4]
    other_lengths = [l for l in other_lengths if 64 <= l <= 8192 and l != n0]
    # Deduplicate
    other_lengths = list(dict.fromkeys(other_lengths))[:4]

    for nl in other_lengths:
        if time.monotonic() > deadline - 80:
            break
        # Resample best to new length
        init = _resample(best_a, nl)
        # Also try a fresh asymmetric init at this length
        init2 = _make_asymmetric(nl, 1.5, 0.7)

        for idx, ini in enumerate([init, init2]):
            if time.monotonic() > deadline - 30:
                break
            t_end = min(deadline - 20, time.monotonic() + 60 * scale)
            if time.monotonic() >= t_end:
                continue
            res_a, res_score = _refine(ini, t_end, rng)
            if res_score < best_score:
                best_a = res_a.copy()
                best_score = res_score
                print(f"Phase3 len={nl} init={idx}: score={best_score:.8f}", flush=True)

    # === Phase 4: Final deep refinement of the best ===
    if time.monotonic() < deadline - 5:
        t_end = deadline - 2.0
        res_a, res_score = _refine(best_a.copy(), t_end, rng)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"Phase4 final: score={best_score:.8f}", flush=True)
        else:
            print(f"Phase4 final: no improvement, best={best_score:.8f}", flush=True)

    # === Phase 5: One more subgradient pass if time allows ===
    if time.monotonic() < deadline - 5 and best_score > 1.505:
        t_end = deadline - 1.0
        res_a, res_score = _refine_subgrad(best_a.copy(), t_end, rng)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"Phase5 subgrad: score={best_score:.8f}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)
    print(f"FINAL: len={out.size} score={best_score:.8f}", flush=True)
    return [float(x) for x in out]
