import numpy as np
import time

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p

def _softmax_stable(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - v.max()
    v = np.clip(v, -700, 700)
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(v) / v.size
    return e / s

def _score(a):
    """Compute the objective score for sequence a (assumes sum(a) > 0, a >= 0)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-300:
        return np.inf
    m = _next_pow2(2*n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2*n-1]
    b = np.maximum(b, 0)
    bmax = b.max()
    return 2.0 * n * bmax / (s * s)

def _compute_conv(a, m):
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2*len(a)-1]
    b = np.maximum(b, 0)
    return F, b

def _compute_grad(q, a, m_corr):
    """
    Compute gradient of sum_k q[k] * conv(a,a)[k] w.r.t. a.
    = 2 * cross_correlation(q, a)
    q has length 2n-1, a has length n.
    """
    n = len(a)
    q_ext = np.zeros(m_corr)
    q_ext[:len(q)] = q
    a_ext = np.zeros(m_corr)
    a_ext[:n] = a
    # cross-correlation: sum_k q[k] * a[k-m]
    # = ifft(FFT(q) * conj(FFT(a)))
    Q = np.fft.fft(q_ext)
    A = np.fft.fft(a_ext)
    corr = np.fft.ifft(Q * np.conj(A)).real[:n]
    return 2.0 * corr

def _adam_update(w, grad, t, m_hat, v_hat, b1=0.9, b2=0.999, eps=1e-8, lr=0.1):
    m_hat = b1 * m_hat + (1-b1) * grad
    v_hat = b2 * v_hat + (1-b2) * grad * grad
    bc1 = 1 - b1**t
    bc2 = 1 - b2**t
    m_hat_c = m_hat / bc1
    v_hat_c = v_hat / bc2
    w = w - lr * m_hat_c / (np.sqrt(v_hat_c) + eps)
    return w, m_hat, v_hat

    """
    Optimize in log-space w. a = softmax(w).
    Objective: 2n * max(conv(a,a)) (since sum(a)=1 for softmax)
    """
    rng = np.random.default_rng(seed)
    n = int(n)
    m_conv = _next_pow2(2*n - 1)
    m_corr = _next_pow2(3*n - 2)
    
    w = np.array(w_init, dtype=np.float64).copy()
    w = w - w.mean()  # center
    w = np.clip(w, -30, 30)
    
    a = _softmax_stable(w)
    best_a = a.copy()
    best_score = _score(a)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    last_improve = time.monotonic()
    start = time.monotonic()
    
    score_at = max(5, 2000 // n)  # score every this many iterations
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive temperature
        T = T_start + (T_end - T_start) * (progress ** 0.7)
        
        # Compute convolution
        F, b = _compute_conv(a, m_conv)
        bmax = b.max()
        if bmax < 1e-300:
            break
        
        # Compute soft-max weights
        x = (b - bmax) / max(1e-15, T)  # actually T should be in the exponent
        # Actually: q[k] = exp(T * (b[k] - bmax)) / sum(exp(T*(b - bmax)))
        # This is softmax(T * b) but shifted for numerical stability
        Tx = T * (b - bmax)
        Tx = np.clip(Tx, -700, 0)  # b[k] - bmax <= 0, so Tx <= 0
        q = np.exp(Tx)
        qs = q.sum()
        if qs < 1e-300:
            q = np.zeros(len(b))
            q[np.argmax(b)] = 1.0
        else:
            q /= qs
        
        # Compute gradient
        grad_a = _compute_grad(q, a, m_corr)
        
        # Convert to w-space: grad_w = (I - aa^T) grad_a = grad_a - a*(a @ grad_a)
        a_dot_grad = np.dot(a, grad_a)
        grad_w = grad_a - a * a_dot_grad
        # Normalize
        grms = np.sqrt(np.mean(grad_w**2))
        if grms < 1e-14:
            # Stuck, perturb
            w += rng.normal(0, 0.1, n)
            w -= w.mean()
            w = np.clip(w, -30, 30)
            a = _softmax_stable(w)
            last_improve = time.monotonic()
            continue
        grad_w /= grms
        
        # Adam update
        lr = 0.15 * (1.0 - 0.6 * progress)
        t += 1
        w, m_hat, v_hat = _adam_update(w, grad_w, t, m_hat, v_hat, lr=lr)
        w = w - w.mean()
        w = np.clip(w, -30, 30)
        a = _softmax_stable(w)
        
        # Periodic scoring
        if t % score_at == 0:
            sc = 2.0 * n * bmax  # since sum(a)=1, score = 2n*bmax
            # Actually need to recompute bmax from current a
            F, b = _compute_conv(a, m_conv)
            bmax = b.max()
            sc = 2.0 * n * bmax
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                last_improve = time.monotonic()
        
        # Stuck detection
        if time.monotonic() - last_improve > min(60, 0.2 * time_budget):
            sigma = 0.05 + 0.3 * (1 - progress)
            w += rng.normal(0, sigma, n)
            w -= w.mean()
            w = np.clip(w, -30, 30)
            a = _softmax_stable(w)
            last_improve = time.monotonic()
    
    # Final score
    sc = _score(best_a)
    if sc < best_score:
        best_score = sc
    
    return best_a, best_score

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequence a.
Uses smooth-max gradient descent in log-space with adaptive temperature,
spectral initialization, and diverse restarts.
"""
import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _softmax_stable(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - v.max()
    v = np.clip(v, -700.0, 700.0)
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(v) / v.size
    return e / s


def _score_direct(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bm = float(b.max())
    return 2.0 * n * bm / (s * s)


def _conv_fft(a, m):
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * len(a) - 1]
    b = np.maximum(b, 0.0)
    return F, b


def _grad_conv(q, a, m_corr):
    """
    Gradient of sum_k q[k]*conv(a,a)[k] w.r.t. a.
    = 2 * cross_correlation(q, a)
    q has length 2n-1, a has length n.
    """
    n = len(a)
    q_ext = np.zeros(m_corr)
    q_ext[:len(q)] = q
    a_ext = np.zeros(m_corr)
    a_ext[:n] = a
    Q = np.fft.fft(q_ext)
    A = np.fft.fft(a_ext)
    corr = np.fft.ifft(Q * np.conj(A)).real[:n]
    return 2.0 * corr


def _make_inits(n, rng, prev_a=None):
    """Generate diverse non-negative initial sequences of length n, sum=1."""
    inits = []
    t = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # From previous solution (interpolated)
    if prev_a is not None and len(prev_a) > 0:
        pa = np.maximum(prev_a, 0.0)
        ps = pa.sum()
        if ps > 0:
            pa = pa / ps
            if len(pa) != n:
                x_old = np.linspace(0, 1, len(pa))
                x_new = np.linspace(0, 1, n)
                pa = np.interp(x_new, x_old, pa)
                pa = np.maximum(pa, 0)
                ps2 = pa.sum()
                if ps2 > 0:
                    pa /= ps2
            inits.append(pa.copy())

    # Cosine bell
    a = np.cos(np.pi * (t - 0.5)) ** 2
    a = a / a.sum()
    inits.append(a)

    # Gaussian (various widths)
    for sigma in (0.15, 0.25, 0.35, 0.5):
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            inits.append(a / s)

    # Power-law edges
    left = i + 1.0
    right = n - i
    for p in (0.5, 1.0, 1.5, 2.0):
        a = (left / n) ** p + (right / n) ** p
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            inits.append(a / s)

    # Random smooth
    kc = min(20, n)
    a = 1.0
    for k in range(1, kc + 1):
        amp = 0.3 / (k ** 0.8)
        ph = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + ph)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    inits.append(a)

    a = 1.0
    for k in range(1, kc + 1):
        amp = 0.25 / (k ** 0.9)
        ph = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * k * t + ph)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    inits.append(a)

    return inits


def _spectral_init(n, target_flatness, rng, prev_a=None):
    """Spectral method: find a whose convolution approximates a target shape."""
    L = 2 * n - 1
    m = _next_pow2(max(L, 4 * n))
    tL = np.linspace(0, 1, L)
    edge = np.abs(tL - 0.5) * 2.0

    if prev_a is not None and len(prev_a) == n:
        pa = np.maximum(prev_a, 0)
        s = pa.sum()
        if s > 0:
            pa /= s
            F = np.fft.fft(pa, m)
            b = np.fft.ifft(F * F).real[:L]
            b = np.maximum(b, 0)
            bm = b.max()
            if bm > 0:
                b_norm = b / bm
    else:
        b_norm = np.ones(L)

    base = 1.0 + 0.5 * (edge ** 1.5)
    bm = base.max()
    base /= bm

    target = target_flatness * base + (1 - target_flatness) * b_norm
    target = np.maximum(target, 0)
    tm = target.max()
    if tm > 0:
        target /= tm

    T = np.fft.fft(target, m)
    # sqrt in frequency
    sqrt_T = np.fft.ifft(np.sqrt(np.maximum(T, 0))).real[:n]
    a = np.maximum(sqrt_T, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a

    m_corr = _next_pow2(3 * n - 2)

    w = np.array(w_init, dtype=np.float64).copy()
    w = w - w.mean()
    w = np.clip(w, -25.0, 25.0)
    a = _softmax_stable(w)

    best_a = a.copy()
    best_score = 2.0 * n * float(_conv_fft(a, m_conv)[1].max())

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t_count = 0
    start_time = time.monotonic()
    last_improve = start_time
    score_interval = max(3, 3000 // n)
    stuck_thresh = min(45.0, 0.15 * time_budget)

    while time.monotonic() - start_time < time_budget:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / max(1e-9, time_budget))

        # Adaptive temperature: slow rise then fast rise
        T = T_start + (T_end - T_start) * (progress ** 1.5)

        # Compute convolution
        F, b = _conv_fft(a, m_conv)
        bmax = float(b.max())
        if bmax < 1e-300:
            break

        # Soft-max weights on convolution lags
        Tx = T * (b - bmax)
        Tx = np.clip(Tx, -700.0, 0.0)
        q = np.exp(Tx)
        qs = q.sum()
        if qs < 1e-300:
            q = np.zeros(len(b))
            q[int(np.argmax(b))] = 1.0
        else:
            q /= qs

        # Also add uniform component for global flatness (more at low progress)
        uniform_weight = 0.3 * (1 - progress)
        if uniform_weight > 0.01:
            q = (1 - uniform_weight) * q + uniform_weight * (np.ones(len(b)) / len(b))
            qs = q.sum()
            if qs > 0:
                q /= qs

        # Gradient: 2 * cross_correlation(q, a)
        grad_a = _grad_conv(q, a, m_corr)

        # Project to log-space tangent: grad_w = grad_a - a*(a·grad_a)
        a_dot_g = np.dot(a, grad_a)
        grad_w = grad_a - a * a_dot_g

        # Normalize
        grms = float(np.sqrt(np.mean(grad_w ** 2)))
        if grms < 1e-15:
            # Stuck - perturb
            w += rng.normal(0, 0.15 * (1 - progress), n)
            w -= w.mean()
            w = np.clip(w, -25, 25)
            a = _softmax_stable(w)
            last_improve = time.monotonic()
            continue
        grad_w /= grms

        # Adam step
        lr = 0.12 * (1.0 - 0.7 * progress)
        t_count += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m_hat = b1 * m_hat + (1 - b1) * grad_w
        v_hat = b2 * v_hat + (1 - b2) * (grad_w * grad_w)
        bc1 = 1 - b1 ** t_count
        bc2 = 1 - b2 ** t_count
        step = (m_hat / bc1) / (np.sqrt(v_hat / bc2) + eps)
        step = np.clip(step, -1.0, 1.0)
        w = w - lr * step
        w = w - w.mean()
        w = np.clip(w, -25, 25)
        a = _softmax_stable(w)

        # Score check
        if t_count % score_interval == 0:
            Fn, bn = _conv_fft(a, m_conv)
            sc = 2.0 * n * float(bn.max())
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                last_improve = time.monotonic()

        # Stuck detection
        if time.monotonic() - last_improve > stuck_thresh:
            sigma = 0.08 + 0.4 * (1 - progress)
            w += rng.normal(0, sigma, n)
            w -= w.mean()
            w = np.clip(w, -25, 25)
            a = _softmax_stable(w)
            last_improve = time.monotonic()
            # Reset Adam state
            m_hat *= 0.5
            v_hat *= 0.5

    # Final eval
    sc = 2.0 * n * float(_conv_fft(best_a, m_conv)[1].max())
    return best_a, sc


def _optimize_direct(a_init, n, time_budget, seed, T_start=1.0, T_end=60.0):
    """
    Optimize directly in a-space with projection to a>=0, sum=1.
    """
    rng = np.random.default_rng(seed)
    m_conv = _next_pow2(2 * n - 1)
    m_corr = _next_pow2(3 * n - 2)

    a = np.maximum(np.array(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(_conv_fft(a, m_conv)[1].max())

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t_count = 0
    start_time = time.monotonic()
    last_improve = start_time
    score_interval = max(3, 3000 // n)
    stuck_thresh = min(40.0, 0.12 * time_budget)

    while time.monotonic() - start_time < time_budget:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / max(1e-9, time_budget))
        T = T_start + (T_end - T_start) * (progress ** 1.3)

        F, b = _conv_fft(a, m_conv)
        bmax = float(b.max())
        if bmax < 1e-300:
            break

        # Soft-max weights
        Tx = T * (b - bmax)
        Tx = np.clip(Tx, -700.0, 0.0)
        q = np.exp(Tx)
        qs = q.sum()
        if qs < 1e-300:
            q = np.zeros(len(b))
            q[int(np.argmax(b))] = 1.0
        else:
            q /= qs

        # Uniform blend
        uw = 0.25 * (1 - progress)
        if uw > 0.01:
            q = (1 - uw) * q + uw * (np.ones(len(b)) / len(b))
            qs = q.sum()
            if qs > 0:
                q /= qs

        # Gradient
        grad_a = _grad_conv(q, a, m_corr)

        # Subtract mean to handle sum constraint (projection onto tangent)
        a_dot_g = np.dot(a, grad_a)
        grad_a = grad_a - a_dot_g  # direction that preserves sum approximately

        grms = float(np.sqrt(np.mean(grad_a ** 2)))
        if grms < 1e-15:
            a += rng.normal(0, 0.05, n)
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            last_improve = time.monotonic()
            continue
        grad_a /= grms

        lr = 0.08 * (1.0 - 0.6 * progress)
        t_count += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m_hat = b1 * m_hat + (1 - b1) * grad_a
        v_hat = b2 * v_hat + (1 - b2) * (grad_a * grad_a)
        bc1 = 1 - b1 ** t_count
        bc2 = 1 - b2 ** t_count
        step = (m_hat / bc1) / (np.sqrt(v_hat / bc2) + eps)
        step = np.clip(step, -1.0, 1.0)

        a = a - lr * step
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a /= s

        if t_count % score_interval == 0:
            Fn, bn = _conv_fft(a, m_conv)
            sc = 2.0 * n * float(bn.max())
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                last_improve = time.monotonic()

        if time.monotonic() - last_improve > stuck_thresh:
            sigma = 0.03 + 0.15 * (1 - progress)
            a = a + rng.normal(0, sigma, n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            last_improve = time.monotonic()
            m_hat *= 0.5
            v_hat *= 0.5

    sc = 2.0 * n * float(_conv_fft(best_a, m_conv)[1].max())
    return best_a, sc


def _polish(a_init, n, time_budget, seed):
    """High-T, low-lr polish to squeeze out last bits of improvement."""
    m_conv = _next_pow2(2 * n - 1)
    m_corr = _next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)

    a = np.maximum(np.array(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_score = 2.0 * n * float(_conv_fft(a, m_conv)[1].max())

    w = np.log(np.maximum(a, 1e-30))
    w = w - w.mean()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t_count = 0
    start_time = time.monotonic()

    while time.monotonic() - start_time < time_budget:
        a = _softmax_stable(w)
        F, b = _conv_fft(a, m_conv)
        bmax = float(b.max())
        if bmax < 1e-300:
            break

        # Very sharp soft-max (almost hard max)
        T = 40.0 + 20.0 * np.random.random()
        Tx = T * (b - bmax)
        Tx = np.clip(Tx, -700.0, 0.0)
        q = np.exp(Tx)
        qs = q.sum()
        if qs < 1e-300:
            q = np.zeros(len(b))
            q[int(np.argmax(b))] = 1.0
        else:
            q /= qs

        grad_a = _grad_conv(q, a, m_corr)
        a_dot_g = np.dot(a, grad_a)
        grad_w = grad_a - a * a_dot_g
        grms = float(np.sqrt(np.mean(grad_w ** 2)))
        if grms < 1e-15:
            w += rng.normal(0, 0.02, n)
            w -= w.mean()
            continue
        grad_w /= grms

        lr = 0.03
        t_count += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m_hat = b1 * m_hat + (1 - b1) * grad_w
        v_hat = b2 * v_hat + (1 - b2) * (grad_w * grad_w)
        bc1 = 1 - b1 ** t_count
        bc2 = 1 - b2 ** t_count
        step = (m_hat / bc1) / (np.sqrt(v_hat / bc2) + eps)
        step = np.clip(step, -0.3, 0.3)
        w = w - lr * step
        w = w - w.mean()
        w = np.clip(w, -25, 25)

        if t_count % 10 == 0:
            an = _softmax_stable(w)
            sc = 2.0 * n * float(_conv_fft(an, m_conv)[1].max())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = an.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses smooth-max gradient in log-space and direct space, with spectral
    initialization and diverse restarts.
    """
    seed = int(seed)
    budget_s = float(max(60.0, min(2400.0, budget_s)))
    start_global = time.monotonic()
    deadline = start_global + budget_s - 30.0  # 30s safety margin
    rng = np.random.default_rng(seed)

    # Load previous best if available
    prev_a = None
    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0:
                prev_a = arr / s
    except Exception:
        prev_a = None

    best_a = None
    best_score = np.inf

    if prev_a is not None:
        best_a = prev_a.copy()
        best_score = _score_direct(prev_a)
    else:
        best_a = np.ones(2048) / 2048
        best_score = 2.0

    print("Starting. Previous best score: %.6f" % best_score)

    # Define stages: (n fraction_of_budget)
    stages = [
        (2048, 0.35),
        (4096, 0.30),
        (1024, 0.15),
        (8192, 0.12),
        (2048, 0.08),  # second pass on 2048
    ]

    for stage_idx, (n, frac) in enumerate(stages):
        now = time.monotonic()
        if now > deadline:
            break
        stage_budget = frac * (budget_s - 30.0)
        stage_deadline = min(deadline, now + stage_budget)
        print("\n=== Stage %d: n=%d, budget=%.0fs ===" % (stage_idx + 1, n, stage_budget))

        # Generate initializations
        inits = _make_inits(n, rng, prev_a if prev_a is not None else best_a)

        # Add spectral inits
        for tf in (0.7, 0.4, 0.2):
            try:
                si = _spectral_init(n, tf, rng, best_a)
                inits.append(si)
            except Exception:
                pass

        # Score and sort inits
        scored_inits = []
        for ai in inits:
            ai = np.maximum(ai, 0.0)
            s = ai.sum()
            if s > 0:
                ai = ai / s
            sc = 2.0 * n * float(_conv_fft(ai, _next_pow2(2 * n - 1))[1].max())
            if np.isfinite(sc):
                scored_inits.append((sc, ai))

        scored_inits.sort(key=lambda x: x[0])
        # Take top 4 diverse
        chosen = []
        for sc, ai in scored_inits:
            diverse = True
            for (_, bj) in chosen:
                d = np.sqrt(np.sum((ai - bj) ** 2))
                if d < 0.03:
                    diverse = False
                    break
            if diverse:
                chosen.append((sc, ai))
            if len(chosen) >= 4:
                break
        if not chosen:
            chosen.append((2.0, np.ones(n) / n))

        print("  Chosen inits: %d (best init score: %.6f)" % (len(chosen), chosen[0][0]))

        # Optimize each
        n_cands = len(chosen)
        per_cand_time = max(5.0, (stage_budget - 20.0) / max(1, n_cands))

        for ci, (sc_init, ai) in enumerate(chosen):
            now = time.monotonic()
            if now > stage_deadline - 10:
                break
            alloc = min(per_cand_time, max(5.0, stage_deadline - now - 5.0))
            cseed = int(rng.integers(0, 2**31))

            # Alternate between log-space and direct
            if ci % 2 == 0:
                w = np.log(np.maximum(ai, 1e-20))
                w = w - w.mean()
                a_opt, sc_opt = _optimize_logspace(w, n, alloc, cseed)
            else:
                a_opt, sc_opt = _optimize_direct(ai, n, alloc, cseed)

            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt
                print("  n=%d cand%d: improved to %.6f" % (n, ci, best_score))
            else:
                print("  n=%d cand%d: score=%.6f (best=%.6f)" % (n, ci, sc_opt, best_score))

        # Polish the best
        now = time.monotonic()
        if now < stage_deadline - 5 and best_a is not None and len(best_a) == n:
            polish_budget = min(30.0, stage_deadline - now - 2.0)
            if polish_budget > 5:
                a_pol, sc_pol = _polish(best_a, n, polish_budget, int(rng.integers(0, 2**31)))
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_a = a_pol
                    print("  n=%d polished: %.6f" % (n, best_score))

        print("  Stage %d done. Best: %.6f" % (stage_idx + 1, best_score))

    # Final: resize best to a reasonable length and verify
    if best_a is None:
        best_a = np.ones(2048) / 2048
    best_a = np.maximum(best_a, 0.0)
    bs = best_a.sum()
    if bs > 0:
        best_a = best_a / bs

    # Verify with evaluate_sequence
    final_score = best_score
    try:
        if len(best_a) <= 20000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("\nFinal: n=%d, FFT score=%.6f, exact score=%.6f" %
                  (len(best_a), best_score, exact))
            if not np.isfinite(exact):
                best_a = np.ones(2048) / 2048
                final_score = 2.0
            else:
                final_score = exact
        else:
            print("\nFinal: n=%d, score=%.6f (exact check skipped)" % (len(best_a), best_score))
    except Exception as e:
        print("Validation error: %s" % e)
        final_score = best_score

    print("Returned sequence length: %d, score: %.6f" % (len(best_a), final_score))
    return [float(x) for x in best_a]
