"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a via adaptive
smoothed-inf gradient descent in log-probability space with multiple
sequence lengths, structured restarts, and periodic coordinate refinement.
Uses FFT for O(n log n) convolution throughout.
"""

import numpy as np
import time


def next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def stable_softmax(w):
    w = np.asarray(w, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - np.max(w)
    e = np.exp(np.clip(w, -745.0, 709.0))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    return e / s


def log_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-300)
    return np.log(a)


def fft_conv_self(a, N):
    n = a.size
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[:2 * n - 1]
    b = np.maximum(b, 0.0)
    return b


def smoothed_inf_and_grad(a, b, N, beta):
    n = a.size
    b_max = float(b.max())
    x = beta * (b - b_max)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros(b.size, dtype=np.float64)
        p[int(b.argmax())] = 1.0
        smax = b_max
    else:
        p = e / s
        smax = b_max + np.log(s) / beta

    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    grad_a = 2.0 * np.fft.irfft(P * np.conjugate(A), N)[:n]
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)
    return smax, grad_a


def fft_score(a, N):
    b = fft_conv_self(a, N)
    mx = float(b.max())
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * float(a.size) * mx


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return float('inf')
    b = np.convolve(a, a)
    mx = float(b.max())
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx / (s * s))


def load_parent():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 16:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return arr / s


def run_single(n_target, seed, budget_s, N_override=None):
    t0 = time.monotonic()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    N = N_override if N_override else next_pow2(2 * n_target)
    n = n_target

    # Initialize
    parent = load_parent()
    if parent is not None and abs(parent.size - n) < n // 4:
        # Resample to target length
        if parent.size == n:
            a = parent.copy()
        else:
            idx = np.linspace(0, parent.size - 1, n)
            a = np.interp(idx, np.arange(parent.size), parent)
        s = float(a.sum())
        if s > 0 and np.isfinite(s):
            a = a / s
        else:
            a = np.ones(n, dtype=np.float64) / n
    else:
        # Structured initialization: raised cosine
        x = np.arange(n, dtype=np.float64) / (n - 1)
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = a ** 1.5
        s = float(a.sum())
        if s > 0 and np.isfinite(s):
            a = a / s
        else:
            a = np.ones(n, dtype=np.float64) / n

    w = log_from_a(a)

    # Best tracking
    b = fft_conv_self(a, N)
    best_score = float(2.0 * n * b.max())
    best_a = a.copy()
    best_w = w.copy()

    top = [(best_score, best_a.copy())]

    # Optimizer state
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    t_adam = 0
    beta = 5.0
    lr = 0.04
    stall = 0
    outer = 0
    best_improve_iter = 0

    # Adam params
    a1, a2 = 0.9, 0.999
    eps = 1e-8

    while time.monotonic() < deadline:
        outer += 1

        # Safety check
        if not np.all(np.isfinite(w)) or not np.all(np.isfinite(a)):
            w = best_w.copy()
            a = stable_softmax(w)
            m_adam = np.zeros(n, dtype=np.float64)
            v_adam = np.zeros(n, dtype=np.float64)
            t_adam = 0
            stall = 0
            continue

        # Compute convolution
        b = fft_conv_self(a, N)
        score = float(2.0 * n * b.max())

        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_w = w.copy()
            best_improve_iter = outer
            stall = 0
            lr = min(0.12, lr * 1.005)
            if len(top) < 12:
                top.append((score, best_a.copy()))
            else:
                worst_i = int(np.argmax([t[0] for t in top]))
                if score < top[worst_i][0]:
                    top[worst_i] = (score, best_a.copy())
        else:
            stall += 1
            if stall % 50 == 0:
                lr = max(0.008, lr * 0.96)

        if outer % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  n={n} iter={outer} t={elapsed:.0f}s score={score:.6f} best={best_score:.6f} beta={beta:.1f} lr={lr:.4f} stall={stall}")

        # Gradient computation
        smax, grad_a = smoothed_inf_and_grad(a, b, N, beta)

        # Log-space gradient
        mean_grad = float(np.dot(a, grad_a))
        grad_w = a * (grad_a - mean_grad)
        grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)

        # Adam update
        t_adam += 1
        m_adam = a1 * m_adam + (1.0 - a1) * grad_w
        v_adam = a2 * v_adam + (1.0 - a2) * grad_w * grad_w
        m_hat = m_adam / (1.0 - a1 ** t_adam)
        v_hat = v_adam / (1.0 - a2 ** t_adam)
        dw = m_hat / (np.sqrt(v_hat) + eps)
        dw = np.clip(dw, -5.0, 5.0)
        dw = np.where(np.isfinite(dw), dw, 0.0)

        w = w - lr * dw
        w = np.where(np.isfinite(w), w, 0.0)
        a = stable_softmax(w)

        # Periodic: try a "sharpen" step - focus more on the max
        if outer % 30 == 0 and stall > 20 and time.monotonic() < deadline - 5:
            # Identify hot spots
            b_hot = b.max()
            thresh = 0.98 * b_hot
            hot_idx = np.where(b >= thresh)[0]

            # Try to redistribute mass at hot coordinates
            # For each hot k, b[k] = sum_i a[i]*a[k-i]
            # Reduce a[i] where a[i]*a[k-i] is large, add to a[j] where it's small
            a_try = a.copy()
            for k in hot_idx[:10]:
                # Find the top contributors to b[k]
                valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
                if len(valid_i) == 0:
                    continue
                contrib = a_try[valid_i] * a_try[k - valid_i]
                # Move mass from top to bottom
                if len(valid_i) > 2:
                    top_i = valid_i[np.argmax(contrib)]
                    bot_i = valid_i[np.argmin(contrib)]
                    delta = 0.001 * a_try[top_i]
                    a_try[top_i] -= delta
                    a_try[bot_i] += delta
                    a_try = np.maximum(a_try, 0.0)
                    s_try = float(a_try.sum())
                    if s_try > 0:
                        a_try = a_try / s_try

            sc_try = fft_score(a_try, N)
            if np.isfinite(sc_try) and sc_try < score:
                a = a_try
                w = log_from_a(a)
                m_adam = np.zeros(n, dtype=np.float64)
                v_adam = np.zeros(n, dtype=np.float64)
                t_adam = 0

        # Adaptive beta
        if outer % 100 == 0:
            if stall < 10:
                beta = min(300.0, beta * 1.08 + 0.2)
            elif stall < 50:
                beta = min(300.0, beta * 1.03 + 0.05)
            else:
                beta = max(3.0, beta * 0.95)

        # Restart logic
        if stall > 80:
            r = rng.random()
            if r < 0.25:
                # Small perturbation from best
                sigma = 0.05
                w = best_w.copy() + rng.normal(0.0, sigma, n)
            elif r < 0.50:
                # Medium perturbation from best
                sigma = 0.15
                w = best_w.copy() + rng.normal(0.0, sigma, n)
            elif r < 0.70:
                # From top candidate
                sc_t, a_t = top[int(rng.integers(len(top)))]
                w = log_from_a(a_t) + rng.normal(0.0, 0.1, n)
            elif r < 0.85:
                # Structured perturbation (Gaussian window)
                x = np.arange(n, dtype=np.float64)
                center = rng.uniform(0.2 * n, 0.8 * n)
                width = rng.uniform(0.1 * n, 0.4 * n)
                pert = np.exp(-0.5 * ((x - center) / width) ** 2)
                w = best_w.copy() + 0.3 * pert
            else:
                # Large exploration
                sigma = 0.5
                w = best_w.copy() + rng.normal(0.0, sigma, n)

            a = stable_softmax(w)
            m_adam = np.zeros(n, dtype=np.float64)
            v_adam = np.zeros(n, dtype=np.float64)
            t_adam = 0
            stall = 0
            beta = 5.0
            lr = 0.04

        # Very periodic: try concave/transformation perturbation
        if stall > 150 and outer % 100 == 0:
            # Power transformation
            p_val = float(rng.choice([0.5, 0.7, 1.3, 1.5, 2.0]))
            a_t = np.power(np.maximum(best_a, 1e-300), p_val)
            s_t = float(a_t.sum())
            if s_t > 0 and np.isfinite(s_t):
                a_t = a_t / s_t
                sc_t = fft_score(a_t, N)
                if np.isfinite(sc_t) and sc_t < best_score:
                    best_score = sc_t
                    best_a = a_t.copy()
                    best_w = log_from_a(a_t)
                    a = a_t.copy()
                    w = best_w.copy()
                    m_adam = np.zeros(n, dtype=np.float64)
                    v_adam = np.zeros(n, dtype=np.float64)
                    t_adam = 0
                    stall = 0

    return best_a, best_score
