def _lp_iteration(a, n, deadline, verbose=False):
    """One LP iteration with line search. Returns (a_new, obj_new, improved)."""
    a_lp = _solve_tangent(a, n)
    if a_lp is None:
        return a, _obj_n(a, n), False
    
    a_lp = _proj(a_lp, n)
    best_a = a.copy()
    best_o = _obj_n(a, n)
    
    for tau in [1.0, 0.8, 0.6, 0.5, 0.3, 0.2, 0.1, 0.05]:
        a_try = (1.0 - tau) * a + tau * a_lp
        a_try = _proj(a_try, n)
        o = _obj_n(a_try, n)
        if o < best_o - 1e-12:
            best_o = o
            best_a = a_try.copy()
        if o >= best_o:
            break  # No point trying smaller tau if this tau already worse
    
    return best_a, best_o, (best_o < _obj_n(a, n) - 1e-12)

def _grad_step(a, n, eta):
    c = _conv_full(a)
    k = int(np.argmax(c))
    g = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    idx = np.arange(i_lo, i_hi + 1)
    g[idx] = 2.0 * a[k - idx]
    g -= g.mean()
    gn = float(np.linalg.norm(g))
    if gn < 1e-15:
        return a.copy()
    a_new = a - eta * g / gn
    return _proj(a_new, n)

Tangent-plane LP optimizer for min 2n*max(conv(a,a))/sum(a)^2.
Iteratively solves an LP on tangent planes to the convolution,
with line search, multi-start perturbation, and gradient fallback.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import time

def _conv(a):
    return np.convolve(a, a)

def _proj(a, S):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)

def _obj(a, n):
    return 2.0 * float(np.max(_conv(a))) / n

def _lp_tangent(a0, n):
    K = 2 * n - 1
    c0 = _conv(a0)
    
    # Build sparse A_ub (K x (n+1))
    # a-part: n columns, each with n nonzeros
    row_a = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel().astype(np.int32)
    col_a = np.repeat(np.arange(n, dtype=np.int32), n)
    dat_a = np.tile(2.0 * a0, n)
    
    # t-part
    row_t = np.arange(K, dtype=np.int32)
    col_t = np.full(K, n, dtype=np.int32)
    dat_t = np.full(K, -1.0)
    
    A_ub = csc_matrix((np.concatenate([dat_a, dat_t]),
                       (np.concatenate([row_a, row_t]),
                        np.concatenate([col_a, col_t]))),
                      shape=(K, n + 1))
    
    c_obj = np.zeros(n + 1); c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    b_eq = np.array([float(n)])
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=c0, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    if not res.success:
        return None
    return np.maximum(res.x[:n], 0.0)

def _grad_step(a, n, eta=0.05):
    c = _conv(a)
    k = int(np.argmax(c))
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    g -= g.mean()
    gn = float(np.linalg.norm(g))
    if gn < 1e-15:
        return a.copy()
    return _proj(a - eta * g / gn, n)

def _smooth_grad_step(a, n, tau, eta):
    c = _conv(a)
    mx = c.max()
    p = np.exp((c - mx) / tau)
    p /= p.sum()
    g = 2.0 * np.correlate(p, a, mode='valid')
    g -= g.mean()
    gn = float(np.linalg.norm(g))
    if gn < 1e-15:
        return a.copy()
    return _proj(a - eta * g / gn, n)

def _line_search(a, a_lp, n):
    best_a = a.copy()
    best_o = _obj(a, n)
    for tau in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]:
        a_t = _proj((1 - tau) * a + tau * a_lp, n)
        o = _obj(a_t, n)
        if o < best_o - 1e-12:
            best_o = o
            best_a = a_t.copy()
    return best_a, best_o
