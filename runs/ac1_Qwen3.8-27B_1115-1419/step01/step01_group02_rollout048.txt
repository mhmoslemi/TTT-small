Optimization of step function heights to minimize 2n*max(autocorr)/sum^2.

import numpy as np

from scipy.fft import fft, ifft
from scipy.optimize import minimize


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score(a):
    """Compute the score 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = _fft_conv(a, a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def _normalize(a):
    """Normalize to sum=1."""
    S = float(np.sum(a))
    if S < 1e-15:
        return np.ones_like(a) / len(a)
    return a / S


def _soft_max_and_p(c, T):
    """Compute soft-max value and probability vector from c array."""
    c_max = float(np.max(c))
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    L = c_max + float(np.log(sum_exp)) / T
    p = exp_s / sum_exp
    return L, p


def _cross_corr(a, p):
    """Compute (a⋆p)[m] = sum_j a[j]*p[j+m] for m=0..len(a)-1."""
    n = len(a)
    p_rev = p[::-1]
    conv_ap = _fft_conv(a, p_rev)
    # (a⋆p)[m] = conv(a, p_rev)[2n-2-m] for m=0..n-1
    return conv_ap[n-1:2*n-1][::-1]


def _grad_soft_max(a, T):
    """Gradient of soft_max(conv(a,a)) w.r.t. a (assuming sum(a)=1)."""
    n = len(a)
    c = _fft_conv(a, a)
    L, p = _soft_max_and_p(c, T)
    cross = _cross_corr(a, p)
    grad = 2.0 * cross
    return L, grad


def _run_mirror_descent(a, deadline, T_init=1.0, T_growth=1.3, eta_init=1.0,
                        eta_decay=0.998, eval_interval=200, max_T=5000.0):
    """Run mirror descent with soft-max annealing. Returns best a and score."""
    n = len(a)
    a = _normalize(a.copy())
    T = T_init
    eta = eta_init
    best = a.copy()
    best_score = _score(a)
    step = 0
    patience = 0
    
    while time.monotonic() < deadline:
        c = _fft_conv(a, a)
        L, p = _soft_max_and_p(c, T)
        cross = _cross_corr(a, p)
        grad = 2.0 * cross
        
        # Multiplicative update
        log_update = -eta * grad
        # Clip to avoid overflow/underflow
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        S = float(np.sum(a))
        if S < 1e-300:
            a = np.ones(n) / n
        else:
            a = a / S
        
        step += 1
        eta *= eta_decay
        
        if step % eval_interval == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                patience = 0
                if best_score < 1.55:
                    print(f"  MD: step={step}, T={T:.1f}, score={sc:.6f}")
            else:
                patience += 1
            
            # Anneal T
            if patience >= 5 and T < max_T:
                T *= T_growth
                eta = min(eta_init, eta * 1.5)
                patience = 0
            
            # Restart if stuck too long
            if patience > 20:
                a = _normalize(best + 0.005 * np.random.standard_normal(n))
                T = max(1.0, T * 0.5)
                eta = eta_init * 0.5
                patience = 0
    
    return best, best_score
