import numpy as np

def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def compute_lse_gradient(a, eps):
    """
    Compute gradient of (1/eps)*log(sum_k exp(eps*c[k])) where c = conv(a,a).
    Returns (gradient, c) for efficiency.
    """
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    
    # Softmax weights: p[k] = exp(eps*c[k]) / sum(exp(eps*c[j]))
    ec = eps * c
    ec_max = ec.max()
    exp_c = np.exp(ec - ec_max)
    p = exp_c / exp_c.sum()
    
    # Gradient: g[j] = 2 * sum_k p[k] * a[k-j] (for valid k,j)
    # = 2 * convolve(a[::-1], p)[n-1:2*n-1]
    g = 2.0 * np.convolve(a[::-1], p)[n-1:2*n-1]
    
    return g, c

def compute_flatness_gradient(a):
    """
    Compute gradient of sum_k (c[k] - mu)^2 where c=conv(a,a), mu=1/(2n-1).
    Returns (gradient, c).
    """
    n = len(a)
    c = np.convolve(a, a)
    mu = 1.0 / (2 * n - 1)
    d = c - mu
    # grad[j] = 4 * sum_k d[k] * a[k-j] = 4 * convolve(a[::-1], d)[n-1:2*n-1]
    g = 4.0 * np.convolve(a[::-1], d)[n-1:2*n-1]
    return g, c

        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
