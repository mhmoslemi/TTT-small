```python
"""
Search for non-negative step-function heights minimizing
    2 * n * max(convolve(a, a)) / (sum(a))^2

The optimizer combines:
  - symmetric probability-simplex parametrization (pair-mass softmax),
  - smoothed log-sum-exp / near-active-set subgradients from FFT convolution,
  - adaptive multiplicative (natural-gradient) steps with line search,
  - low-dimensional tangent LP directions around the current point,
  - safe convex mixing of old and LP-directed points,
  - multiple initializations and final active-polish passes.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog as _linprog
except Exception:
    _linprog = None


# -----------------------------------------------------------------------
# Basic helpers
# -----------------------------------------------------------------------

def _pow2_ge(x):
    x = int(max(1, x))
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def sym_normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    a = 0.5 * (a + a[::-1])
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def downsample_sequence(arr, target):
    arr = np.asarray(arr, dtype=np.float64)
    m = len(arr)
    if m <= target:
        return arr.copy()
    x_old = np.arange(m, dtype=np.float64)
    x_new = np.linspace(0.0, float(m - 1), int(target), dtype=np.float64)
    return np.interp(x_new, x_old, arr, left=0.0, right=0.0).astype(np.float64)


def softmax_pair(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    bad = ~np.isfinite(z)
    if bad.any():
        z[bad] = 0.0
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    e /= s
    return e


def expand_pair(b, n):
    b = np.asarray(b, dtype=np.float64)
    p = b.size
    a = np.empty(n, dtype=np.float64)
    if n % 2 == 0:
        a[:p] = 0.5 * b
        a[n - 1:p - 1:-1] = 0.5 * b
    else:
        a[:p - 1] = 0.5 * b[:p - 1]
        a[n - 1:p - 1:-1] = 0.5 * b[:p - 1]
        if p >= 1:
            a[p - 1] = b[p - 1]
    return a


def pair_mass(a, n):
    a = np.asarray(a, dtype=np.float64)
    p = (n + 1) // 2
    b = np.empty(p, dtype=np.float64)
    if n % 2 == 0:
        b[:] = 0.5 * (a[:p] + a[n - 1:p - 1:-1])
    else:
        b[:p - 1] = 0.5 * (a[:p - 1] + a[n - 1:p - 1:-1])
        if p >= 1:
            b[p - 1] = a[p - 1]
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        b[:] = 1.0 / float(p)
    else:
        b /= s
    return b


def pair_gradient(g, n):
    g = np.asarray(g, dtype=np.float64)
    p = (n + 1) // 2
    gb = np.empty(p, dtype=np.float64)
    if n % 2 == 0:
        gb[:] = 0.5 * (g[:p] + g[n - 1:p - 1:-1])
    else:
        gb[:p - 1] = 0.5 * (g[:p - 1] + g[n - 1:p - 1:-1])
        if p >= 1:
            gb[p - 1] = g[p - 1]
    return gb


# -----------------------------------------------------------------------
# Convolution and scores
# -----------------------------------------------------------------------

def convolve_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    if n <= 80:
        return np.asarray(np.convolve(a, a), dtype=np.float64)
    total = 2 * n - 1
    L = _pow2_ge(total)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:total]
    c = np.maximum(c, 0.0)
    return c.astype(np.float64, copy=False)


def convolve_fft_ab(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    la, lb = a.size, b.size
    if la == 0 or lb == 0:
        return np.array([], dtype=np.float64)
    total = la + lb - 1
    if total <= 120:
        return np.asarray(np.convolve(a, b), dtype=np.float64)
    L = _pow2_ge(total)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)[:total]
    c = np.maximum(c, 0.0)
    return c.astype(np.float64, copy=False)


def score_from_a(a, n=None):
    a = np.asarray(a, dtype=np.float64)
    if n is None:
        n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.inf
    c = convolve_fft(a)
    m = float(c.max()) if c.size else 0.0
    return float(2.0 * n * m / (s * s))


def score_from_z(z, n):
    b = softmax_pair(z)
    a = expand_pair(b, n)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.inf, a
    c = convolve_fft(a)
    m = float(c.max()) if c.size else 0.0
    return float(2.0 * n * m / (s * s)), a

    try:

    except Exception:
        try:
            arr = np.asarray(lst, dtype=np.float64)
            return score_from_a(arr, len(arr))
        except Exception:
            return np.inf


# -----------------------------------------------------------------------
# Gradient weights
# -----------------------------------------------------------------------

def smooth_weights(c, tau):
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return c.copy()
    cmax = float(c.max())
    tau = max(1e-300, float(tau))
    z = (c - cmax) / tau
    bad = ~np.isfinite(z)
    if bad.any():
        z[bad] = -700.0
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    e /= s
    return e.astype(np.float64, copy=False)


def active_weights(c, frac, top_k):
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return c.copy()
    cmax = float(c.max())
    if cmax <= 0:
        w = np.full(m, 1.0 / float(m), dtype=np.float64)
        return w
    thr = cmax - max(1e-300, float(frac) * cmax)
    mask = c >= thr
    cnt = int(mask.sum())
    if cnt < int(top_k):
        k = min(m, int(top_k))
        if k <= 0:
            k = 1
        idx = np.argpartition(c, -k)[-k:]
        mask = np.zeros(m, dtype=bool)
        mask[idx] = True
    if not mask.any():
        mask = np.zeros(m, dtype=bool)
        mask[int(np.argmax(c))] = True
    w = np.zeros(m, dtype=np.float64)
    w[mask] = 1.0
    w /= float(w.sum())
    return w


def make_weights(c, tau_frac, active_final):
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return c.copy()
    if active_final:
        aw = active_weights(c, 1e-4, 10)
        uw = np.full(m, 1.0 / float(m), dtype=np.float64)
        return 0.25 * uw + 0.75 * aw
    cmax = float(c.max()) if c.size else 0.0
    tau = max(1e-300, float(tau_frac) * max(cmax, 1e-300))
    return smooth_weights(c, tau)


def compute_gradient_from_a_weights(a, weights):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(weights, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _pow2_ge(m + n - 1)
    fa = np.fft.rfft(a, L)
    q = w[::-1]
    fq = np.fft.rfft(q, L)
    conv = np.fft.irfft(fa * fq, L)[:m + n - 1]
    h = conv[m - n:m][::-1].copy()
    bad = ~np.isfinite(h)
    if bad.any():
        h[bad] = 0.0
    return h.astype(np.float64, copy=False)


# -----------------------------------------------------------------------
# Bases for candidates / LP directions
# -----------------------------------------------------------------------

def make_cosine_basis(n, K):
    if n <= 1:
        return np.ones((max(n, 1), max(K, 1)), dtype=np.float64)
    u = 2.0 * np.arange(n, dtype=np.float64) / float(n - 1) - 1.0
    B = np.empty((n, K), dtype=np.float64)
    B[:, 0] = 1.0
    for r in range(1, K):
        B[:, r] = np.cos(math.pi * float(r) * u)
    # Normalize columns to about RMS 1.
    for r in range(K):
        col = B[:, r]
        nm = float(np.linalg.norm(col))
        if nm > 1e-14:
            col *= float(np.sqrt(float(n))) / nm
    return B


def make_random_smooth_basis(n, K, rng):
    if n <= 1:
        B = np.ones((max(n, 1), max(K, 1)), dtype=np.float64)
        return B
    B = np.empty((n, K), dtype=np.float64)
    B[:, 0] = 1.0
    if K > 1:
        freq = np.fft.rfftfreq(n)
        cutoff = min(freq.size, max(3, int(n // 40)))
        mask = np.ones(freq.size, dtype=np.float64)
        if cutoff > 0:
            mask[cutoff:] = 0.0
        for r in range(1, K):
            v = rng.normal(size=n)
            vf = np.fft.rfft(v)
            vf[1:] *= mask
            smooth = np.fft.irfft(vf, n)
            smooth = 0.5 * (smooth + smooth[::-1])
            smooth -= float(smooth.mean())
            maxabs = float(np.max(np.abs(smooth))) if smooth.size else 0.0
            if maxabs < 1e-12:
                smooth = rng.normal(size=n)
                smooth = 0.5 * (smooth + smooth[::-1])
                smooth -= float(smooth.mean())
                maxabs = float(np.max(np.abs(smooth))) if smooth.size else 1.0
            if maxabs > 1e-300:
                smooth /= maxabs
            nm = float(np.linalg.norm(smooth))
            if nm > 1e-14:
                smooth *= float(np.sqrt(float(n))) / nm
            B[:, r] = smooth
    return B


# -----------------------------------------------------------------------
# Low-dimensional LP direction (tangent LP around current f)
# -----------------------------------------------------------------------

def solve_lp_direction(f, basis, M):
    if _linprog is None:
        return None
    f = np.asarray(f, dtype=np.float64)
    basis = np.asarray(basis, dtype=np.float64)
    n = f.size
    m = 2 * n - 1
    K = basis.shape[1]
    if K <= 0 or n <= 0:
        return None

    cfa = convolve_fft(f)
    if cfa.size < m:
        tmp = np.zeros(m, dtype=np.float64)
        tmp[:cfa.size] = cfa
        cfa = tmp
    slack = 1e-9 * max(float(M), 1e-12)
    b_conv = max(0.0, float(M)) - cfa + slack
    b_conv = np.maximum(b_conv, 0.0)

    L = _pow2_ge(m)
    ff = np.fft.rfft(f, L)
    Co = np.empty((m, K), dtype=np.float64)
    for r in range(K):
        fb = np.fft.rfft(basis[:, r], L)
        co = np.fft.irfft(ff * fb, L)[:m]
        Co[:, r] = np.maximum(co, 0.0)

    A = np.vstack((Co, -basis))
    b = np.concatenate((b_conv, f))
    cobj = -basis.sum(axis=0)
    bounds = [(-10.0, 10.0) for _ in range(K)]

    res = None
    last_exc = None
    try:
        res = _linprog(
            cobj,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": 20.0, "presolve": True}
        )
    except Exception as e1:
        last_exc = e1
        try:
            res = _linprog(cobj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        except Exception as e2:
            last_exc = e2
            try:
                res = _linprog(cobj, A_ub=A, b_ub=b, bounds=bounds)
            except Exception as e3:
                last_exc = e3
                res = None

    if res is None:
        return None
    if not getattr(res, "success", False):
        return None
    x = np.asarray(res.x, dtype=np.float64)
    g = f + basis @ x
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    g /= s
    g = sym_normalize(g)
    return g


def mix_two(a, b, ca, cb, cab, n):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    ca = np.asarray(ca, dtype=np.float64)
    cb = np.asarray(cb, dtype=np.float64)
    cab = np.asarray(cab, dtype=np.float64)
    mlen = min(ca.size, cb.size, cab.size)
    if mlen <= 0:
        return sym_normalize(a), np.inf

    ca = ca[:mlen]
    cb = cb[:mlen]
    cab = cab[:mlen]

    def vals_for_ts(ts):
        ts = np.asarray(ts, dtype=np.float64)
        vv = np.empty(ts.size, dtype=np.float64)
        for i in range(ts.size):
            t = float(ts[i])
            one = 1.0 - t
            c = (one * one) * ca + (2.0 * one * t) * cab + (t * t) * cb
            vv[i] = 2.0 * float(n) * float(c.max())
        return vv

    ts = np.linspace(0.0, 1.0, 221, dtype=np.float64)
    vals = vals_for_ts(ts)
    idx = int(np.argmin(vals))
    best_t = float(ts[idx])
    best_val = float(vals[idx])

    if 0.0 < best_t < 1.0:
        dt = 1.0 / 221.0
        lo = max(0.0, best_t - 2.0 * dt)
        hi = min(1.0, best_t + 2.0 * dt)
        ts2 = np.linspace(lo, hi, 221, dtype=np.float64)
        vals2 = vals_for_ts(ts2)
        idx2 = int(np.argmin(vals2))
        best_t = float(ts2[idx2])
        best_val = float(vals2[idx2])

    best_a = (1.0 - best_t) * a + best_t * b
    best_a = sym_normalize(best_a)
    exact = score_from_a(best_a, n)
    if not np.isfinite(exact):
        exact = best_val
    return best_a, float(exact)


# -----------------------------------------------------------------------
# Main local optimizer: multiplicative/natural-gradient on symmetric probability simplex
# -----------------------------------------------------------------------

def optimize_multiplicative(a0, duration, rng, active_final=False, name=""):
    a0 = np.asarray(a0, dtype=np.float64)
    n = a0.size
    if n <= 1:
        return sym_normalize(a0), np.inf
    if duration <= 0.0:
        return sym_normalize(a0), score_from_a(a0, n)

    a = sym_normalize(a0)
    # tiny positivity floor
    a = 0.99995 * a + (0.00005 / float(n)) * np.ones(n, dtype=np.float64)
    a = sym_normalize(a)

    b = pair_mass(a, n)
    b = np.clip(b, 1e-15, None)
    b /= float(b.sum())
    z = np.log(b).astype(np.float64)
    z -= float(np.mean(z))

    best_a = a.copy()
    best_z = z.copy()
    best_score = score_from_a(best_a, n)

    base_lr = 0.05 if not active_final else 0.018
    max_lr = 0.35 if not active_final else 0.06
    min_lr = 1e-4
    tau_frac = 0.08 if not active_final else 0.0005
    min_tau = 0.0003 if not active_final else 0.00002
    alphas = (0.5, 1.0, 2.0) if not active_final else (0.25, 0.5, 1.0)

    no_improve = 0
    stall = 0
    deadline = time.monotonic() + float(duration)
    last_print = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = softmax_pair(z)
        a = expand_pair(b, n)

        c = convolve_fft(a)
        cm = float(c.max()) if c.size else 0.0
        cur = 2.0 * float(n) * cm

        if not np.isfinite(cur):
            z = best_z.copy()
            base_lr = 0.03
            continue

        if cur > best_score * 1.05 + 1e-9:
            z = best_z.copy()
            base_lr = min(base_lr, 0.05)
            no_improve += 15
            continue

        if cur < best_score:
            best_score = float(cur)
            best_a = a.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if (it % 120 == 0) or (time.monotonic() - last_print > 12.0):
            last_print = time.monotonic()
            print(f"    {name} it={it} score={best_score:.8f} lr={base_lr:.4f} tau={tau_frac:.6g} noimp={no_improve}")

        if no_improve > 25 and tau_frac > min_tau:
            tau_frac = max(min_tau, tau_frac * 0.995)

        if no_improve > 180:
            scale = 0.04 + 0.018 * float(stall)
            z = z + scale * rng.normal(size=z.size)
            z = np.where(np.isfinite(z), z, 0.0)
            z -= float(np.mean(z))
            base_lr = min(max_lr, base_lr * 1.35)
            tau_frac = max(min_tau, tau_frac * 1.4)
            no_improve = 0
            stall += 1
            print(f"    {name} multiplicative restart {stall}, score={best_score:.8f}")
            if stall > (4 if active_final else 8):
                break

        weights = make_weights(c, tau_frac, active_final)
        g = compute_gradient_from_a_weights(a, weights)
        gb = pair_gradient(g, n)
        base_dot = float(np.dot(b, gb))
        d = gb - base_dot
        bad = ~np.isfinite(d)
        if bad.any():
            d[bad] = 0.0
        nrm = float(np.sqrt(np.dot(b, d * d)))
        if not np.isfinite(nrm) or nrm < 1e-18:
            z = z + 0.02 * rng.normal(size=z.size)
            z = np.where(np.isfinite(z), z, 0.0)
            z -= float(np.mean(z))
            continue
        d = d / nrm

        best_trial_score = cur
        best_trial_z = z
        best_trial_a = a

        for alpha in alphas:
            step = float(base_lr) * float(alpha)
            zt = z - step * d
            zt = np.where(np.isfinite(zt), zt, 0.0)
            zt -= float(np.mean(zt))
            st, at = score_from_z(zt, n)
            if np.isfinite(st) and st < best_trial_score:
                best_trial_score = float(st)
                best_trial_z = zt
                best_trial_a = at

        if best_trial_score < cur - 1e-12:
            z = best_trial_z
            a = best_trial_a
            base_lr = min(max_lr, base_lr * 1.06)
        else:
            base_lr = max(min_lr, base_lr * 0.82)

    final_score = score_from_z(best_z, n)[0]
    final_a = expand_pair(softmax_pair(best_z), n)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = float(final_score)
        best_a = final_a

    return sym_normalize(best_a), float(best_score)


# -----------------------------------------------------------------------
# Low-dimensional LP-search phase
# -----------------------------------------------------------------------

def lowdim_lp_search(a0, duration, rng, cos_basis=None):
    a0 = sym_normalize(a0)
    n = a0.size
    if n <= 1 or duration <= 0.0:
        return a0, score_from_a(a0, n)
    if _linprog is None:
        return a0, np.inf

    best_a = a0.copy()
    best_score = score_from_a(best_a, n)
    deadline = time.monotonic() + float(duration)
    K = 32
    if cos_basis is None:
        cos_basis = make_cosine_basis(n, K)
    no_improve = 0
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        rem = deadline - time.monotonic()
        if rem <= 1.0:
            break

        M = best_score / (2.0 * float(n))
        if not np.isfinite(M) or M <= 0.0:
            M = 1.0 / float(n)

        if it % 3 == 1:
            basis = cos_basis
        else:
            basis = make_random_smooth_basis(n, K, rng)

        g = solve_lp_direction(best_a, basis, M)
        if g is None:
            no_improve += 1
            continue

        if np.linalg.norm(g - best_a) < 1e-9:
            no_improve += 1
            continue

        c_best = convolve_fft(best_a)
        c_g = convolve_fft(g)
        c_bg = convolve_fft_ab(best_a, g)

        cand_a, cand_score = mix_two(best_a, g, c_best, c_g, c_bg, n)
        sg = score_from_a(g, n)

        improved = False
        if np.isfinite(cand_score) and cand_score < best_score:
            best_a = cand_a
            best_score = float(cand_score)
            improved = True
        elif np.isfinite(sg) and sg < best_score:
            best_a = sym_normalize(g)
            best_score = float(sg)
            improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 10:
            noise = np.exp(rng.normal(0.0, 0.06, n))
            best_a = sym_normalize(best_a * noise)
            best_score = score_from_a(best_a, n)
            no_improve = 0

        if (it % 10 == 0) or (time.monotonic() - last_print > 12.0):
            last_print = time.monotonic()
            print(f"    lowdim-LP it={it} score={best_score:.8f} noimp={no_improve}")

        if no_improve > 80 and it > 50:
            # diversify by starting LP from a noisy copy
            noise = np.exp(rng.normal(0.0, 0.12, n))
            pert = sym_normalize(best_a * noise)
            sc = score_from_a(pert, n)
            if np.isfinite(sc) and sc < best_score * 1.05:
                best_a = pert
                best_score = float(sc)
            no_improve = 30

    return sym_normalize(best_a), float(best_score)


# -----------------------------------------------------------------------
# Candidate generation
# -----------------------------------------------------------------------

def generate_candidates(n, given, rng, cos_basis):
    pool = []
    idx = 0

    def _try_add(name, a):
        nonlocal idx
        a = np.asarray(a, dtype=np.float64)
        if a.size != n:
            return
        a = sym_normalize(a)
        s = score_from_a(a, n)
        if np.isfinite(s):
            pool.append((float(s), idx, name, a.copy()))
            idx += 1

    if given is not None:
        try:
            _try_add("given", given)
        except Exception:
            pass

    _try_add("uniform", np.ones(n, dtype=np.float64) / float(n))

    if n > 1:
        u = 2.0 * np.arange(n, dtype=np.float64) / float(n - 1) - 1.0
        c1 = np.cos(math.pi * u)
        edge_low = 1.0 - c1     # high at edges, low at center
        center_low = 1.0 + c1   # low at edges, high at center
        edge_low = np.maximum(edge_low, 1e-12)
        center_low = np.maximum(center_low, 1e-12)

        for p in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 4.0):
            _try_add(f"edge_pow_{p}", np.power(edge_low, float(p)))
            _try_add(f"center_pow_{p}", np.power(center_low, float(p)))

        for alpha in (-0.85, -0.6, -0.35, -0.12, 0.12, 0.35, 0.6, 0.85):
            _try_add(f"cos_alpha_{alpha}", np.exp(alpha * c1))
            _try_add(f"edge_alpha_{alpha}", np.exp(alpha * edge_low - float(np.mean(np.exp(alpha * edge_low)))))

    if cos_basis is not None and cos_basis.size:
        K = min(32, cos_basis.shape[1])
        Kb = cos_basis[:, :K]
        ramp_sigmas = (3.0, 5.0, 8.0, 12.0)
        for sigma in ramp_sigmas:
            rrr = np.arange(K, dtype=np.float64)
            ramp = np.exp(-0.5 * (rrr / float(sigma)) ** 2)
            for _ in range(12):
                coeff = rng.normal(size=K) * ramp
                v = Kb @ coeff
                v = np.asarray(v, dtype=np.float64)
                v -= float(v.mean())
                sd = float(v.std())
                if sd > 1e-14:
                    v *= 0.65 / sd
                a = np.exp(v)
                _try_add(f"cos_ramp_{sigma}", a)

    # A few fully random smooth candidates.
    for _ in range(8):
        rb = make_random_smooth_basis(n, 12, rng)
        coeff = rng.normal(size=12)
        v = rb @ coeff
        v -= float(v.mean())
        sd = float(v.std())
        if sd > 1e-14:
            v *= 0.65 / sd
        _try_add("rand_smooth", np.exp(v))

    # Sort by score, then unique idx. The tuples start with numeric score.
    try:
        pool.sort(key=lambda z: z[0])
    except Exception:
        pool.sort()

    # De-duplicate arrays approximately by score/name is not important; return top objects.
    out = []
    seen = set()
    for sc, i_id, nm, a in pool:
        key = f"{nm}_{i_id}"
        if key in seen:
            continue
        seen.add(key)
        out.append(a)
        if len(out) >= 8:
            break
    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out


# -----------------------------------------------------------------------
# Main entry point
# -----------------------------------------------------------------------

def propose_candidate(budget_s=2400, **kwargs):
    """
    Proposed sequence finder. Returns a Python list of non-negative floats.
    """
    start_time = time.monotonic()
    total_budget = float(budget_s)
    stop_time = start_time + max(60.0, total_budget) - 8.0
    rng = np.random.default_rng(20250923)

    # Determine n and optional starting sequence.
    raw = kwargs.get("height_sequence_1", None)
    if raw is None:
        try:
            raw = globals().get("height_sequence_1", None)
        except Exception:
            raw = None

    n = 4806
    given = None
    if raw is not None:
        try:
            arr = np.asarray(raw, dtype=np.float64)
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            if arr.size > 0:
                n = int(arr.size)
                given = arr
        except Exception:
            n = 4806
            given = None

    # Keep size in a practical range for np.convolve evaluation and FFT search.
    if n < 512:
        n = 2048
        given = None
    if n > 5000:
        if given is not None:
            given = downsample_sequence(given, min(n, 5000))
        n = max(512, min(int(n), 5000))

    print(f"=== Search starting: n={n}, budget={total_budget:.1f}s ===")

    K = 32
    cos_basis = make_cosine_basis(n, K)

    # Initial candidate pool.
    cands = []
    try:
        cands = generate_candidates(n, given, rng, cos_basis)
    except Exception as e:
        print("Candidate generation failed:", e)
        cands = [np.ones(n, dtype=np.float64) / float(n)]

    if not cands:
        cands = [np.ones(n, dtype=np.float64) / float(n)]

    best_a = sym_normalize(cands[0])
    best_score = score_from_a(best_a, n)
    finalists = [best_a.copy(), cands[0].copy()]
    print(f"Best initial fast score = {best_score:.8f}")

    # Phase 1: local natural-gradient refinement from best initializer.
    remain = stop_time - time.monotonic()
    t1 = max(15.0, min(700.0, remain - 1300.0))
    if t1 > 20.0:
        try:
            a1, s1 = optimize_multiplicative(best_a, t1, rng, active_final=False, name="phase1")
            if np.isfinite(s1) and s1 < best_score:
                best_a = a1
                best_score = float(s1)
                finalists.append(best_a.copy())
            elif np.isfinite(s1):
                finalists.append(a1.copy())
            print(f"Phase1 done: score={best_score:.8f}")
        except Exception as e:
            print("Phase1 failed:", e)

    # Phase 2: low-dimensional tangent LP exploration.
    remain = stop_time - time.monotonic()
    if _linprog is not None and remain > 220.0:
        t2 = max(40.0, min(1050.0, remain - 720.0))
        if t2 > 35.0:
            try:
                a2, s2 = lowdim_lp_search(best_a, t2, rng, cos_basis=cos_basis)
                if np.isfinite(s2) and s2 < best_score:
                    best_a = a2
                    best_score = float(s2)
                    finalists.append(best_a.copy())
                elif np.isfinite(s2):
                    finalists.append(a2.copy())
                print(f"Phase2 done: score={best_score:.8f}")
            except Exception as e:
                print("Phase2 failed:", e)
    else:
        print("Skipping lowdim-LP phase.")

    # If possible, also try a second local start to avoid the starting-basin trap.
    remain = stop_time - time.monotonic()
    if remain > 480.0 and best_score > 1.5029 and len(cands) > 1:
        t_alt = max(60.0, min(350.0, remain - 360.0))
        if t_alt > 40.0:
            alt = sym_normalize(cands[1])
            try:
                a_alt, s_alt = optimize_multiplicative(alt, t_alt, rng, active_final=False, name="alt-start")
                if np.isfinite(s_alt) and s_alt < best_score:
                    best_a = a_alt
                    best_score = float(s_alt)
                    finalists.append(best_a.copy())
                elif np.isfinite(s_alt):
                    finalists.append(a_alt.copy())
                print(f"Alt-start done: score={best_score:.8f}")
            except Exception as e:
                print("Alt-start failed:", e)

    # Phase 3: full local refinement after LP/exploration.
    remain = stop_time - time.monotonic()
    t3 = max(15.0, min(900.0, remain - 260.0))
    if t3 > 20.0:
        try:
            a3, s3 = optimize_multiplicative(best_a, t3, rng, active_final=False, name="phase3")
            if np.isfinite(s3) and s3 < best_score:
                best_a = a3
                best_score = float(s3)
            finalists.append(a3.copy())
            print(f"Phase3 done: score={best_score:.8f}")
        except Exception as e:
            print("Phase3 failed:", e)

    # Phase 4: fine active-set polishing.
    remain = stop_time - time.monotonic()
    t4 = max(10.0, remain - 8.0)
    if t4 > 5.0:
        try:
            a4, s4 = optimize_multiplicative(best_a, t4, rng, active_final=True, name="final-polish")
            if np.isfinite(s4) and s4 < best_score:
                best_a = a4
                best_score = float(s4)
            finalists.append(a4.copy())
            print(f"Final polish done: score={best_score:.8f}")
        except Exception as e:
            print("Final polish failed:", e)

    # Final verification using the actual evaluator if available.
    # Prefer the lowest actually-evaluated finite score, with fast score as tiebreak.
    best_actual = np.inf
    best_final_list = [float(x) for x in best_a]
    tried = set()

    for cand in finalists:
        try:
            lst = [float(float(x)) for x in np.asarray(cand, dtype=np.float64).ravel()]
        except Exception:
            continue
        if len(lst) != n:
            continue
        # Hash by a few entries plus length to avoid duplicate slow evaluates.
        key = (len(lst), round(float(lst[0]), 12), round(float(lst[-1]), 12),
               round(float(np.mean(np.asarray(lst, dtype=np.float64))), 15),
               round(float(np.linalg.norm(np.asarray(lst, dtype=np.float64) - best_a)), 10))
        if key in tried:
            continue
        # Cap number of expensive evaluations.
        if len(tried) > 24:
            break
        tried.add(key)
        val = evaluate_list_safe(lst)
        if np.isfinite(val) and val < best_actual:
            best_actual = float(val)
            best_final_list = lst

    # One last direct check of the current best.
    try:
        lst = [float(float(x)) for x in best_a]
        val = evaluate_list_safe(lst)
        if np.isfinite(val) and val < best_actual:
            best_actual = float(val)
            best_final_list = lst
    except Exception:
        pass

    # Ultimate safe fallback: best known non-empty candidate.
    if (not best_final_list) or (not all(np.isfinite(float(x)) for x in best_final_list)):
        best_final_list = [float(x) for x in np.ones(n, dtype=np.float64) / float(n)]
        best_actual = evaluate_list_safe(best_final_list)

    # Ensure non-negative clean floats.
    clean = []
    for x in best_final_list:
        try:
            y = float(x)
        except Exception:
            y = 0.0
        if not np.isfinite(y) or y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        clean.append(y)
    s = sum(clean)
    if s <= 0.0:
        clean = [1.0 / float(len(clean))] * len(clean)
        s = 1.0
    if s < 0.01:
        clean = [x * (0.0105 / s) for x in clean]

    print(f"=== FINAL returned score={best_actual:.8f}, n={len(clean)} ===")
    return clean
```
