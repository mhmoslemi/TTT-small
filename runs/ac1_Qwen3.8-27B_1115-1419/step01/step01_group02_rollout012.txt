```python
"""
Searches for a non-negative sequence minimizing
    2 * n * max(np.convolve(a, a)) / (sum(a))**2.

The optimizer combines:
  - FFT-based fast self-convolution scoring,
  - logarithmic mirror descent on exp(y) variables with log-sum-exp /
    active-peak subgradients and exact line search,
  - a spectral fitting heuristic that pushes the self-convolution toward
    flatter coefficient vectors,
  - L-BFGS-B refinement on a smooth log-sum-exp surrogate,
  - a low-dim polynomial template search for diverse basins,
  - an optional longer-length exploration,
  - a final direct-convolution polish for numerical tightening.
"""

import time
import math

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
from scipy.optimize import minimize

_FFT_CACHE = {}


def _fft_len(n, kind):
    """Cached FFT length needed for self-convolution or gradient convolution."""
    n = int(n)
    key = (n, kind)
    val = _FFT_CACHE.get(key)
    if val is None:
        if kind == 0:
            out = 2 * n - 1
        else:
            out = 3 * n - 2
        val = int(next_fast_len(max(1, out)))
        _FFT_CACHE[key] = val
    return val


def _load_parent():
    """Load the externally provided parent sequence, if present."""
    try:
        seq = height_sequence_1
    except NameError:
        return None
    except Exception:
        return None

    try:
        a = np.asarray(seq, dtype=np.float64)
    except Exception:
        return None

    if a.ndim != 1 or a.size == 0:
        return None

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if float(np.sum(a)) <= 0.0:
        return None

    # Keep search tractable even if the parent is unexpectedly huge.
    if a.size > 12000:
        m = 12000
        blocks = a.size // m
        if blocks > 0:
            a = a[:blocks * m].reshape(m, blocks).mean(axis=1)
        else:
            a = a[:m]

    return a


def _target_sum(n):
    """Normalization used internally: sum(a)^2 = 2n."""
    return math.sqrt(2.0 * float(n))


def _normalize_to_sum(a, s_target):
    """Clip to non-negative finite values and scale to the target sum."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return np.full_like(a, s_target / max(1, len(a)), dtype=np.float64)

    return a * (s_target / s)


def _conv_self(a):
    """Fast full self-convolution using FFT. Small negatives are clipped."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    nfft = _fft_len(n, 0)
    A = rfft(a, nfft)
    b = irfft(A * A, nfft)[:need]
    return np.maximum(b.real, 0.0)


def _fast_evaluate(a):
    """Fast internal evaluation matching evaluate_sequence for valid finite inputs."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf

    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf

    b = _conv_self(a)
    m = float(np.max(b))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _fast_score_fixed(a):
    """Assuming a is already normalized to sum sqrt(2n), score is max(selfconv)."""
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    b = _conv_self(a)
    m = float(np.max(b))
    if not np.isfinite(m):
        return np.inf
    return float(m)


def _a_from_y(y, s_target):
    """Convert log-parameter y to a non-negative vector with sum s_target."""
    y = np.asarray(y, dtype=np.float64)
    z = y - float(np.max(y))
    ea = np.exp(z)
    s = float(np.sum(ea))
    if not np.isfinite(s) or s <= 1e-15:
        return np.full_like(y, s_target / max(1, y.size), dtype=np.float64)
    return (s_target / s) * ea


def _softmax_weights(b, tau):
    """Stable normalized weights proportional to exp(b / tau)."""
    b = np.asarray(b, dtype=np.float64)
    b = np.maximum(b, 0.0)
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return np.full_like(b, 1.0 / max(1, b.size), dtype=np.float64)

    if tau <= 1e-12:
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return w

    e = np.exp((b - m) / tau)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return w

    w = e / s
    return w


def _top_weights(b, k):
    """Weights on the top-k coefficients, proportional to coefficient values."""
    b = np.asarray(b, dtype=np.float64)
    b = np.maximum(b, 0.0)
    n = b.size
    k = min(int(k), n)

    if k <= 0:
        return np.full_like(b, 1.0 / max(1, n), dtype=np.float64)
    if k >= n:
        s = float(np.sum(b))
        if s <= 0.0:
            return np.full_like(b, 1.0 / n, dtype=np.float64)
        return b / s

    idx = np.argpartition(b, -k)[-k:]
    w = np.zeros_like(b, dtype=np.float64)
    vals = b[idx]
    s = float(np.sum(vals))
    if s <= 0.0:
        w[idx] = 1.0 / k
    else:
        w[idx] = vals / s
    return w


def _grad_weighted(a, w):
    """
    Gradient of sum_k w_k * c_k(a), where c_k are full self-conv coefficients.

    For one peak k, d c_k / d a_t = 2 a_{k-t} when valid.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    ng = _fft_len(n, 1)

    Af = rfft(a, ng)
    Wf = rfft(w[::-1], ng)
    c = irfft(Af * Wf, ng)
    c = c.real[:3 * n - 2]

    grad = c[n - 1:2 * n - 1][::-1].copy()
    grad *= 2.0
    return np.maximum(grad, 0.0)


def _direct_conv(a):
    """Direct exact full convolution, used for final polishing."""
    a = np.asarray(a, dtype=np.float64)
    b = np.convolve(a, a, mode="full")
    return np.maximum(b, 0.0)


def _disturb(A, s_target, rng, strength=0.05):
    """Perturb A around its best/baseline shape, keeping it valid."""
    A = np.asarray(A, dtype=np.float64)
    if strength > 0:
        A = A + strength * float(np.max(A)) * rng.standard_normal(A.size)
    A = np.maximum(A, 0.0)
    if float(np.sum(A)) <= 1e-15:
        A = np.full_like(A, s_target / max(1, A.size), dtype=np.float64)
    return _normalize_to_sum(A, s_target)


def _optimize_mirror(A, deadline, rng, tag="mirror"):
    """
    Main local search.

    Works in log variables y, where a = S * softmax(y). The direction is a
    mirror-descent subgradient / log-sum-exp gradient of the self-convolution
    peak. Line search uses exact FFT score.
    """
    A = np.asarray(A, dtype=np.float64)
    n = A.size
    if n < 3:
        return A, _fast_evaluate(A)

    S = _target_sum(n)
    A = _normalize_to_sum(A, S)

    A = np.maximum(A, 1e-300)
    y = np.log(A)
    y -= float(np.mean(y))

    best = A.copy()
    best_score = _fast_score_fixed(best)
    printed = best_score

    start = time.monotonic()
    total_dur = max(1.0, deadline - start)

    # (tau, initial mirror step, fraction of total time)
    schedule = [
        (1.0,    0.90, 0.03),
        (0.5,    0.80, 0.05),
        (0.25,   0.70, 0.08),
        (0.12,   0.65, 0.10),
        (0.06,   0.60, 0.14),
        (0.025,  0.55, 0.20),
        (0.008,  0.50, 0.20),
        (0.003,  0.45, 0.15),
        (0.0012, 0.40, 0.05),
    ]

    cum = 0.0
    for tau, alpha0, frac in schedule:
        if time.monotonic() > deadline:
            break

        cum += frac
        stage_deadline = min(deadline, start + cum * total_dur)
        alpha = alpha0
        beta = 0.72 if tau >= 0.1 else 0.85
        v = np.zeros(n, dtype=np.float64)
        no_best_iters = 0
        fails = 0

        b = _conv_self(A)

        while time.monotonic() < stage_deadline:
            cur = float(np.max(b))
            if not np.isfinite(cur):
                break

            if cur < best_score - 1e-12:
                best_score = cur
                best = A.copy()
                no_best_iters = 0
                if best_score < printed - 1e-6:
                    print(f"[{tag}] best: {best_score:.8f}")
                    printed = best_score

            if no_best_iters > 3000 and alpha < 0.25:
                A = _disturb(best, S, rng, strength=0.04 + 0.06 * (best_score > 1.55))
                y = np.log(np.maximum(A, 1e-300))
                y -= float(np.mean(y))
                v = np.zeros(n, dtype=np.float64)
                alpha = alpha0
                b = _conv_self(A)
                no_best_iters = 0
                fails = 0
                continue

            w = _softmax_weights(b, tau)
            g = _grad_weighted(A, w)
            v = beta * v + (1.0 - beta) * g
            mean_part = float(np.dot(A, v)) / S
            gy = A * (v - mean_part)
            maxg = float(np.max(np.abs(gy)))
            if not np.isfinite(maxg) or maxg <= 1e-22:
                break

            d = gy / maxg
            accepted = False

            for a_try in (alpha, 0.5 * alpha, 0.25 * alpha):
                yn = y - a_try * d
                An = _a_from_y(yn, S)
                bn = _conv_self(An)
                sc = float(np.max(bn))

                if np.isfinite(sc) and sc < cur - 1e-11:
                    y = yn
                    y -= float(np.mean(y))
                    A = An
                    b = bn
                    cur = sc
                    accepted = True
                    break

            if accepted:
                if cur < best_score - 1e-12:
                    best_score = cur
                    best = A.copy()
                    no_best_iters = 0
                    if best_score < printed - 1e-6:
                        print(f"[{tag}] best: {best_score:.8f}")
                        printed = best_score
                    alpha = min(1.8, alpha * 1.07)
                else:
                    no_best_iters += 1
                    alpha = min(1.5, alpha * 1.01)
                    fails = 0
            else:
                alpha = max(0.02, alpha * 0.5)
                fails += 1
                no_best_iters += 5

                if fails > 40 or alpha <= 0.02:
                    A = _disturb(best, S, rng, strength=0.035)
                    y = np.log(np.maximum(A, 1e-300))
                    y -= float(np.mean(y))
                    v = np.zeros(n, dtype=np.float64)
                    alpha = alpha0
                    b = _conv_self(A)
                    fails = 0

    return best, _fast_score_fixed(best)


def _optimize_spectral(A, deadline, tag="spectral"):
    """
    Spectral heuristic: iteratively multiply A_hat by sqrt(target / current)
    so that the self-convolution moves toward a flatter coefficient vector.
    """
    A = np.asarray(A, dtype=np.float64)
    n = A.size
    if n < 3:
        return A, _fast_evaluate(A)

    S = _target_sum(n)
    A = _normalize_to_sum(A, S)
    L = _fft_len(n, 0)

    best = A.copy()
    best_score = _fast_score_fixed(best)
    printed = best_score

    reduction = 0.45
    fails = 0

    while time.monotonic() < deadline:
        AH = fft(A, L)
        mag2 = np.abs(AH) ** 2
        b = ifft(mag2, L).real[:2 * n - 1]
        b = np.maximum(b, 0.0)

        cur = float(np.max(b))
        bsum = float(np.sum(b))
        if not np.isfinite(cur) or not np.isfinite(bsum) or bsum <= 0:
            break

        if cur < best_score - 1e-12:
            best_score = cur
            best = A.copy()
            if best_score < printed - 1e-6:
                print(f"[{tag}] best: {best_score:.8f}")
                printed = best_score

        mean_b = bsum / max(1.0, float(b.size))
        M = mean_b + (1.0 - reduction) * (cur - mean_b)
        if M <= mean_b + 1e-12:
            M = cur * 0.999

        target = b.copy()
        target[target > M] = M
        target_sum = float(np.sum(target))
        if target_sum <= 1e-15:
            break

        scale = bsum / target_sum
        if not np.isfinite(scale) or scale > 1.35:
            reduction *= 0.5
            fails += 1
            if fails > 20:
                break
            continue

        target *= scale
        BT = fft(target, L)

        ratio = np.ones_like(BT, dtype=np.complex128)
        mask = mag2 > 1e-30
        ratio[mask] = BT[mask] / mag2[mask]

        AH2 = AH * np.sqrt(ratio)
        A2 = ifft(AH2, L).real[:n]
        A2 = np.where(np.isfinite(A2), A2, 0.0)
        A2 = np.maximum(A2, 0.0)

        if float(np.sum(A2)) <= 1e-15:
            reduction *= 0.5
            fails += 1
            continue

        A2 = _normalize_to_sum(A2, S)
        sc2 = _fast_score_fixed(A2)

        if np.isfinite(sc2) and sc2 < cur - 1e-12:
            A = A2
            b = _conv_self(A)
            reduction = min(1.0, reduction * 1.1)
            fails = 0
        elif np.isfinite(sc2) and sc2 < cur + 0.002:
            A = A2
            b = _conv_self(A)
            reduction *= 0.85
            fails = 0
        else:
            reduction *= 0.5
            fails += 1
            if fails > 30 or reduction < 0.03:
                break

    return best, _fast_score_fixed(best)


def _lbfgs_objective(y, args):
    n, S, tau = args
    A = _a_from_y(y, S)
    b = _conv_self(A)
    m = float(np.max(b))
    if not np.isfinite(m):
        return 1e100
    e = np.exp((b - m) / tau)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return 1e100
    return float(m + tau * math.log(s))


def _lbfgs_jacobian(y, args):
    n, S, tau = args
    A = _a_from_y(y, S)
    b = _conv_self(A)
    if tau <= 1e-12:
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = _softmax_weights(b, tau)

    gA = _grad_weighted(A, w)
    mean_part = float(np.dot(A, gA)) / S
    gy = A * (gA - mean_part)
    return gy / max(tau, 1e-12)


def _optimize_lbfgs(A, deadline, tag="lbfgs"):
    """L-BFGS-B refinement on a smooth log-sum-exp surrogate in log-variables."""
    A = np.asarray(A, dtype=np.float64)
    n = A.size
    if n < 3:
        return A, _fast_evaluate(A)

    S = _target_sum(n)
    A = _normalize_to_sum(A, S)

    best = A.copy()
    best_score = _fast_evaluate(best)

    y = np.log(np.maximum(A, 1e-300))
    y -= float(np.mean(y))

    stages = [
        (0.25, 100),
        (0.08, 140),
        (0.025, 160),
        (0.008, 180),
    ]

    for tau, maxfun in stages:
        if time.monotonic() > deadline:
            break

        args = (n, S, tau)
        try:
            res = minimize(
                _lbfgs_objective,
                y,
                jac=_lbfgs_jacobian,
                args=args,
                method="L-BFGS-B",
                options={
                    "maxiter": max(30, maxfun // 4),
                    "maxfun": int(maxfun),
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxcor": 12,
                    "disp": False,
                },
            )
        except Exception:
            res = None

        if res is not None and np.all(np.isfinite(res.x)):
            z = np.asarray(res.x, dtype=np.float64)
            z -= float(np.mean(z))
            y = z
            An = _a_from_y(z, S)
            sc = _fast_evaluate(An)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best = An
                print(f"[{tag}] best: {best_score:.8f}")

    return best, _fast_evaluate(best)


def _poly_objective(p, ctx):
    bases, n = ctx
    x = np.zeros(n, dtype=np.float64)
    for c, base in zip(p, bases):
        x += c * base
    x -= float(np.mean(x))
    x = np.clip(x, -12.0, 12.0)
    a = np.exp(x)
    a = np.maximum(a, 1e-12)
    a = _normalize_to_sum(a, _target_sum(n))
    return _fast_evaluate(a)


def _poly_p_to_a(p, ctx):
    bases, n = ctx
    x = np.zeros(n, dtype=np.float64)
    for c, base in zip(p, bases):
        x += c * base
    x -= float(np.mean(x))
    x = np.clip(x, -12.0, 12.0)
    a = np.exp(x)
    a = np.maximum(a, 1e-12)
    return _normalize_to_sum(a, _target_sum(n))


def _optimize_polynomial(n, deadline, rng, tag="poly"):
    """
    Low-dimensional global-ish search over smooth non-negative templates.
    """
    n = int(n)
    if n < 16:
        return None, np.inf

    u = np.linspace(0.0, 1.0, n, dtype=np.float64)
    bases_list = [
        u,
        1.0 - u,
        u * u,
        (1.0 - u) * (1.0 - u),
        u * (1.0 - u),
        np.sin(2.0 * math.pi * u),
        np.cos(2.0 * math.pi * u),
        u * u * (1.0 - u),
    ]
    bases = tuple(bases_list)
    k = len(bases)
    ctx = (bases, n)
    bounds = [(-4.0, 4.0)] * k

    best_a = None
    best_score = np.inf

    num_starts = 3
    if deadline - time.monotonic() > 1200:
        num_starts = 4

    for st in range(num_starts):
        if time.monotonic() > deadline:
            break

        if st == 0:
            p0 = np.zeros(k, dtype=np.float64)
            p0[0] = 0.35
            p0[1] = -0.35
            p0[2] = 0.8
            p0[3] = 0.8
        else:
            p0 = rng.uniform(-1.0, 1.0, size=k, dtype=np.float64)

        try:
            res = minimize(
                _poly_objective,
                p0,
                args=ctx,
                method="Powell",
                bounds=bounds,
                options={
                    "maxiter": 160,
                    "maxfev": 500,
                    "tol": 1e-7,
                    "disp": False,
                },
            )
        except Exception:
            res = None

        if res is not None and np.all(np.isfinite(res.x)):
            sc = float(_fast_evaluate(_poly_p_to_a(res.x, ctx)))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = _poly_p_to_a(res.x, ctx)
                print(f"[{tag}] poly candidate: {best_score:.8f}")
        elif time.monotonic() > deadline:
            break

    if best_a is None:
        return None, np.inf
    return best_a, _fast_evaluate(best_a)


def _upsample(a, m):
    """Linear upsample to length m, preserving shape."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = int(m)
    if m <= 0:
        return a.copy()
    if m == n:
        return a.copy()
    x = np.linspace(0.0, float(n - 1), m, dtype=np.float64)
    return np.interp(x, np.arange(n, dtype=np.float64), a)


def _optimize_direct_polish(A, deadline, rng, tag="direct"):
    """
    Final very-local polishing using direct convolution for exact scores.
    Uses top-active-set subgradients and mirror line search.
    """
    A = np.asarray(A, dtype=np.float64)
    n = A.size
    if n < 3 or time.monotonic() > deadline:
        return A, _fast_evaluate(A)

    S = _target_sum(n)
    A = _normalize_to_sum(A, S)

    best = A.copy()
    best_score = _direct_conv(A)
    best_score = float(2.0 * n * np.max(best_score) / (float(np.sum(A)) ** 2))
    printed = best_score

    A_safe = np.maximum(A, 1e-300)
    y = np.log(A_safe)
    y -= float(np.mean(y))
    v = np.zeros(n, dtype=np.float64)
    alpha = 0.28
    b = _direct_conv(A)
    fails = 0
    no_best = 0

    while time.monotonic() < deadline:
        cur = float(2.0 * n * np.max(b) / (float(np.sum(A)) ** 2))
        if not np.isfinite(cur):
            break

        if cur < best_score - 1e-12:
            best_score = cur
            best = A.copy()
            no_best = 0
            if best_score < printed - 1e-7:
                print(f"[{tag}] best: {best_score:.8f}")
                printed = best_score

        if no_best > 120 and alpha < 0.12:
            A = _disturb(best, S, rng, strength=0.025)
            y = np.log(np.maximum(A, 1e-300))
            y -= float(np.mean(y))
            v = np.zeros(n, dtype=np.float64)
            alpha = 0.28
            b = _direct_conv(A)
            no_best = 0
            fails = 0
            continue

        w = _top_weights(b, 24)
        g = _grad_weighted(A, w)
        v = 0.65 * v + 0.35 * g
        mean_part = float(np.dot(A, v)) / S
        gy = A * (v - mean_part)
        maxg = float(np.max(np.abs(gy)))
        if not np.isfinite(maxg) or maxg <= 1e-22:
            break

        d = gy / maxg
        accepted = False

        for a_try in (alpha, 0.5 * alpha, 0.25 * alpha):
            yn = y - a_try * d
            An = _a_from_y(yn, S)
            bn = _direct_conv(An)
            sc = float(2.0 * n * np.max(bn) / (float(np.sum(An)) ** 2))

            if np.isfinite(sc) and sc < cur - 1e-12:
                y = yn
                y -= float(np.mean(y))
                A = An
                b = bn
                cur = sc
                accepted = True
                break

        if accepted:
            if cur < best_score - 1e-12:
                best_score = cur
                best = A.copy()
                no_best = 0
                if best_score < printed - 1e-7:
                    print(f"[{tag}] best: {best_score:.8f}")
                    printed = best_score
                alpha = min(0.8, alpha * 1.08)
            else:
                no_best += 1
                alpha = min(0.6, alpha * 1.01)
                fails = 0
        else:
            alpha = max(0.02, alpha * 0.5)
            fails += 1
            no_best += 4
            if fails > 25 or alpha <= 0.02:
                A = _disturb(best, S, rng, strength=0.025)
                y = np.log(np.maximum(A, 1e-300))
                y -= float(np.mean(y))
                v = np.zeros(n, dtype=np.float64)
                alpha = 0.28
                b = _direct_conv(A)
                fails = 0

    return best, _fast_evaluate(best)


def _actual_score(a):
    """Call the provided evaluator safely."""
    try:
        vals = [float(x) for x in a]
        sc = evaluate_sequence(vals)
        sc = float(sc)
        if np.isfinite(sc):
            return sc
    except Exception:
        pass
    return np.inf


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Entry point used by the judge.

    Search algorithm:
      1) start from height_sequence_1 if available,
      2) apply FFT mirror-descent log-sum-exp optimization,
      3) apply spectral flattening,
      4) polish with L-BFGS-B on a smooth surrogate,
      5) attempt low-dimensional polynomial templates for diversity,
      6) optionally explore a longer length by up-sampling,
      7) final direct-convolution micro-polish,
      8) return the best valid non-negative list.
    """
    rng = np.random.default_rng(int(seed))
    start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0

    hard_deadline = start + max(60.0, budget - 35.0)

    parent = _load_parent()
    if parent is None:
        init = np.ones(5582, dtype=np.float64)
    else:
        init = parent

    n0 = int(init.size)
    S0 = _target_sum(n0)
    A = _normalize_to_sum(init, S0)
    best = A.copy()
    best_score = _fast_evaluate(best)
    print(f"Initial internal score: {best_score:.8f} (n={n0})")

    # Quick deterministic variants on the same length.
    rev = _normalize_to_sum(0.5 * (A + A[::-1]), S0)
    sc = _fast_evaluate(rev)
    if sc < best_score:
        best = rev.copy()
        best_score = sc
        print(f"Reversed-average improved: {best_score:.8f}")

    for mix in (0.8, 0.6):
        noise = np.maximum(A * (1.0 + 0.08 * rng.standard_normal(n0)), 0.0)
        cand = _normalize_to_sum(mix * best + (1.0 - mix) * noise, S0)
        sc = _fast_evaluate(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            print(f"Initial noise mix improved: {best_score:.8f}")

    # Primary long local search from the best current sequence.
    if hard_deadline - time.monotonic() > 300:
        dl = time.monotonic() + 0.55 * (hard_deadline - time.monotonic())
        A2, sc2 = _optimize_mirror(best, dl, rng, tag="mirror-main")
        if sc2 < best_score:
            best, best_score = A2, sc2

    # Spectral flattening.
    rem = hard_deadline - time.monotonic()
    if rem > 240:
        dl = time.monotonic() + 0.28 * rem
        A2, sc2 = _optimize_spectral(best, dl, tag="spectral")
        if sc2 < best_score:
            best, best_score = A2, sc2

    # L-BFGS surrogate refinement.
    rem = hard_deadline - time.monotonic()
    if rem > 360:
        dl = time.monotonic() + 0.30 * rem
        A2, sc2 = _optimize_lbfgs(best, dl, tag="lbfgs")
        if sc2 < best_score:
            best, best_score = A2, sc2

    # Low-dimensional global templates on current best length.
    rem = hard_deadline - time.monotonic()
    if rem > 420:
        nm = len(best)
        dl = time.monotonic() + 0.32 * rem
        A2, sc2 = _optimize_polynomial(nm, dl, rng, tag="poly")
        if A2 is not None and sc2 < best_score:
            # Give the template a short local polish before accepting.
            dlpol = time.monotonic() + min(300.0, 0.35 * max(0.0, hard_deadline - time.monotonic()))
            A2, sc2 = _optimize_mirror(A2, dlpol, rng, tag="poly-polish")
        if A2 is not None and sc2 < best_score:
            best, best_score = A2, sc2

    # Try one or two longer lengths, if enough time remains.
    rem = hard_deadline - time.monotonic()
    n_best = len(best)
    if rem > 1200 and n_best <= 7000:
        for factor in (1.35, 1.75):
            n2 = int(min(12000, max(n_best + 1, round(n_best * factor))))
            if n2 <= n_best:
                continue
            if time.monotonic() > hard_deadline - 420:
                break

            A2 = _upsample(best, n2)
            A2 = _normalize_to_sum(A2, _target_sum(n2))
            sc2 = _fast_evaluate(A2)

            # Only spend time if up-sampling did not destroy the sequence too badly.
            if sc2 > best_score * 1.08 + 0.02:
                print(f"Length {n2}: upsample score {sc2:.8f} too high; skip")
                continue

            dlf = time.monotonic() + min(420.0, 0.45 * (hard_deadline - time.monotonic()))
            A2, sc2 = _optimize_mirror(A2, dlf, rng, tag=f"len{n2}")
            if sc2 < best_score:
                best, best_score = A2, sc2
                print(f"Longer length {n2} improved: {best_score:.8f}")
                n_best = n2

    # Optional final mirror pass if time remains.
    rem = hard_deadline - time.monotonic()
    if rem > 420:
        dl = time.monotonic() + 0.55 * rem
        A2, sc2 = _optimize_mirror(best, dl, rng, tag="mirror-final")
        if sc2 < best_score:
            best, best_score = A2, sc2

    # Direct convolution micro-polish for final numerical tightening.
    rem = hard_deadline - time.monotonic()
    if rem > 220 and len(best) <= 12000:
        dl = time.monotonic() + min(260.0, 0.85 * rem)
        A2, sc2 = _optimize_direct_polish(best, dl, rng, tag="direct-polish")
        if sc2 < best_score:
            best, best_score = A2, sc2

    # Final safe validation: use the actual evaluator on a few short candidates.
    candidates = []
    candidates.append(("best", best))

    if parent is not None and parent.size <= 8000 and parent.size != len(best):
        candidates.append(("parent", _normalize_to_sum(parent, _target_sum(parent.size))))

    if len(best) <= 8000:
        candidates.append(("uniform", np.full(len(best), 1.0, dtype=np.float64)))

    chosen = best
    chosen_score = np.inf

    for label, cand in candidates:
        if time.monotonic() > hard_deadline:
            break
        if cand is None:
            continue
        cand = np.asarray(cand, dtype=np.float64)
        if cand.size == 0:
            continue
        sc = _actual_score(cand)
        if np.isfinite(sc) and sc < chosen_score:
            chosen_score = sc
            chosen = cand

    if not np.isfinite(chosen_score) or chosen is None:
        chosen = np.ones(max(1, n0), dtype=np.float64)
        chosen_score = _actual_score(chosen)

    # Ensure every returned value is a finite Python float in [0, 1000].
    out = np.asarray(chosen, dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(np.sum(out))
    if s < 0.01:
        out = np.ones(max(1, out.size), dtype=np.float64)
    else:
        out = out * (0.5 * math.sqrt(2.0 * out.size) / s)
        out = np.clip(out, 0.0, 1000.0)

    final = [float(x) for x in out]
    print(f"Final internal/tracked score: {best_score:.8f}")
    print(f"Final evaluated subset score: {chosen_score:.8f}")
    print(f"Final length: {len(final)}")
    return final
```
