def _conv(a):

def _score(a):

    n = len(a)
    S = float(np.sum(a))

    """max of conv(a,a)."""
    return float(np.max(_conv(a)))


def _proj(a, S):
    """Project to {x>=0, sum=S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def _peak_grad(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1) + 1
    if lo < hi:
        idx = np.arange(lo, hi)
        g[idx] = 2.0 * a[k - idx]
    return g

    c = _conv(a)

    return np.fft.irfft(spec, n)

        return a, False

        mc_new = _mc(a_new)

        lr *= 0.5
    return a, False
