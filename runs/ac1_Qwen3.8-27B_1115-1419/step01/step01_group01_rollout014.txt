  - FFT evaluation of the self-convolution.
  - Softmax parameterization to enforce non-negativity and unit sum.
  - Smooth max / top-lag subgradients with exact convolution gradients.
  - Adam-style nonconvex optimization.
  - Custom L-BFGS polish on annealed smooth-max surrogates.
  - Several constructive starts, including one-sided half-power shapes,
    exponentials, cosine perturbations, and the supplied previous sequence.
  - Progressive lengths and final roof-polishing.
"""

import numpy as np
import time
import math

# Global best sequence tracker.
RUN = {"best_a": None, "best_score": float("inf")}


def _pair_score(item):
    return item[0]


def _update_run(p, score):
    if p is None:
        return
    score = float(score)
    if not np.isfinite(score):
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 1e-300:
        return
    p = p / s
    if score < RUN["best_score"]:
        RUN["best_score"] = score
        RUN["best_a"] = p.copy()


def _fft_len(n):
    n = int(max(1, n))
    need = 2 * n - 1
    m = 1
    while m < need:
        m <<= 1
    return m


def _project_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        n = 512
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def _score_projected(a):
    a = _project_seq(a)
    n = int(a.size)
    if n == 0:
        return float("inf"), a
    m = _fft_len(n)
    F = np.fft.rfft(a, m)
    b = np.fft.irfft(F * F, m)[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return float("inf"), a
    return float(2.0 * n * bmax), a


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    vm = float(np.max(v))
    if not np.isfinite(vm):
        return np.full(v.size, 1.0 / float(max(1, v.size)), dtype=np.float64)
    x = np.clip(v - vm, -745.0, 700.0)
    e = np.exp(x)
    s = float(np.sum(e))
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(v.size, 1.0 / float(max(1, v.size)), dtype=np.float64)
    return e / s


def _score_v(v, n, m=None):
    n = int(max(1, n))
    if m is None:
        m = _fft_len(n)
    p = _softmax(v)
    F = np.fft.rfft(p, m)
    b = np.fft.irfft(F * F, m)[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return float("inf"), p, b, F
    return float(2.0 * n * bmax), p, b, F


def _seq_to_v(a):
    a = _project_seq(a)
    v = np.log(np.maximum(a, 1e-300))
    v = v - float(np.mean(v))
    v = np.clip(v, -40.0, 40.0)
    return v


def _interp_to(n, a):
    n = int(max(1, n))
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if a.size == n:
        return _project_seq(a)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(np.where(np.isfinite(out), out, 0.0), 0.0)
    return _project_seq(out)


def _load_prev():
    try:
        cand = height_sequence_1
    except Exception:
        return None
    try:
        if isinstance(cand, list):
            arr = np.asarray(cand, dtype=np.float64)
        else:
            arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size == 0:
            return None
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(np.sum(arr))
        if s <= 0.0:
            return None
        return arr / s
    except Exception:
        return None


def _fallback_seq():
    n = 2048
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-0.5)
    return _project_seq(a)


def _make_inits(n, rng, extra=None):
    n = int(max(8, n))
    res = []
    if extra is not None:
        for e in extra:
            try:
                res.append(np.asarray(e, dtype=np.float64).ravel())
            except Exception:
                pass

    i = np.arange(n, dtype=np.float64)
    t = i / float(max(1, n - 1))

    # One-sided and two-sided half-power shapes.
    for alpha in (0.0, 0.5, 1.0, 2.0):
        res.append((i + 1.0 + alpha) ** (-0.5))
        res.append((n - i + 1.0 + alpha) ** (-0.5))

    for beta in (0.35, 0.45, 0.6, 0.75):
        res.append((i + 1.0) ** (-beta))
        res.append((np.maximum(n - i, 1.0)) ** (-beta))

    # Exponential envelopes.
    for c in (0.5, 1.0, 2.0, 4.0):
        res.append(np.exp(-c * t))
        res.append(np.exp(-c * (1.0 - t)))
        res.append(np.exp(-c * (t - 0.5) ** 2 * 2.0))

    # Endpoint-pair shapes.
    left = (i + 1.0) / float(n)
    right = (n - i) / float(n)
    left = np.maximum(left, 1e-8)
    right = np.maximum(right, 1e-8)
    for p, q in ((0.25, 0.25), (0.5, 0.25), (0.25, 0.5),
                 (1.0, 0.25), (0.25, 1.0), (0.25, 0.05)):
        res.append(left ** p * right ** q + 1e-9)

    # Low-frequency positive cosine perturbations.
    for rep in range(4):
        a = np.ones(n, dtype=np.float64)
        kc = min(12, n)
        for k in range(1, kc + 1):
            amp = 0.55 / (float(k) ** 0.85)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t + ph)
        res.append(a)

    # Small random nonnegative perturbations of the best half-power shape.
    hp = (i + 1.0) ** (-0.5)
    for rep in range(3):
        a = hp * (1.0 + 0.25 * np.sin(2.0 * np.pi * t + rng.uniform(0.0, 6.28)))
        a += 1e-6 * rng.random(n)
        res.append(a)

    return res


def _select_inits(inits, max_count=6):
    scored = []
    seen = []
    for a in inits:
        a = _project_seq(a)
        if a.size == 0:
            continue
        sc, a = _score_projected(a)
        if not np.isfinite(sc):
            continue
        diverse = True
        for b in seen:
            if b.size != a.size:
                continue
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < 0.035:
                diverse = False
                break
        if diverse:
            scored.append((sc, a))
            seen.append(a)

    scored.sort(key=_pair_score)
    out = []
    for sc, a in scored:
        out.append(a)
        if len(out) >= max_count:
            break
    if not out:
        out.append(_fallback_seq())
    return out


def _obj_grad_params(v, n, m, beta, vmax, pen):
    """
    Smooth objective for p = softmax(v), with lazy log-sum-exp max
    and exact gradient for the convolution-square coefficients.
    """
    n = int(max(1, n))
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size != n:
        if v.size == 0:
            v = np.zeros(n, dtype=np.float64)
        else:
            v = np.interp(np.linspace(0.0, 1.0, n),
                          np.linspace(0.0, 1.0, v.size),
                          v).astype(np.float64)

    vc = v - float(np.mean(v))

    penf = 0.0
    gpen = np.zeros(n, dtype=np.float64)
    if pen > 0.0:
        over = vc - np.clip(vc, -vmax, vmax)
        if np.any(over != 0.0):
            penf = 0.5 * float(pen) * float(np.dot(over, over))
            gpen = float(pen) * over

    p = _softmax(vc)
    F = np.fft.rfft(p, m)
    b = np.fft.irfft(F * F, m)[:2 * n - 1]
    b = np.maximum(b, 0.0)

    bmax = float(np.max(b))
    if not np.isfinite(bmax):
        return 1e100, np.zeros(n, dtype=np.float64)

    score = 2.0 * float(n) * bmax
    _update_run(p, score)

    x = beta * (b - bmax)
    x = np.clip(x, -745.0, 700.0)
    mx = float(np.max(x))
    e = np.exp(x - mx)
    ws = float(np.sum(e))
    if ws <= 1e-300:
        w = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)
        lse = 0.0
    else:
        w = e / ws
        lse = mx + math.log(ws)

    f = score / 2.0 + float(lse) / float(beta) + penf

    Fw = np.fft.rfft(w, m)
    g = 2.0 * float(n) * np.fft.irfft(Fw * np.conj(F), m).real[:n]
    if not np.all(np.isfinite(g)):
        return 1e100, np.zeros(n, dtype=np.float64)

    gdot = float(np.dot(p, g))
    gv = p * (g - gdot)
    gv -= float(np.mean(gv))
    gv += gpen

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(n, dtype=np.float64)

    if not np.isfinite(f):
        f = 1e100 + penf
    return float(f), gv


def _adam_optimize(v0, n, seconds, seed, beta0=8.0, beta1_final=80.0,
                   active=False, K=12):
    n = int(max(8, n))
    seconds = float(max(0.0, seconds))
    if seconds <= 0.5:
        return _seq_to_v(np.ones(n, dtype=np.float64)), float("inf")

    v = np.asarray(v0, dtype=np.float64).ravel()
    if v.size != n:
        v = np.interp(np.linspace(0.0, 1.0, n),
                      np.linspace(0.0, 1.0, max(1, v.size)),
                      v).astype(np.float64)
    v = v - float(np.mean(v))
    v = np.clip(v, -45.0, 45.0)

    m = _fft_len(n)
    L = 2 * n - 1
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)

    sc, p, b, F = _score_v(v, n, m)
    if not np.isfinite(sc):
        v = np.zeros(n, dtype=np.float64)
        sc, p, b, F = _score_v(v, n, m)
    _update_run(p, sc)

    local_v = v.copy()
    local_sc = sc
    start = time.monotonic()
    deadline = start + seconds

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0
    last_improve = start
    stuck_limit = min(25.0, max(3.0, 0.18 * seconds))

    print("    ADAM n=%d start=%.6f budget=%.2fs active=%s K=%d"
          % (n, local_sc, seconds, int(active), int(K)))

    while time.monotonic() < deadline:
        now = time.monotonic()
        progress = min(1.0, max(0.0, (now - start) / seconds))

        if now - last_improve > stuck_limit:
            sigma = 0.04 + 0.22 * (1.0 - progress)
            v = local_v + rng.normal(0.0, sigma, size=n)
            v = v - float(np.mean(v))
            v = np.clip(v, -45.0, 45.0)
            last_improve = now

        p = _softmax(v)
        F = np.fft.rfft(p, m)
        b = np.fft.irfft(F * F, m)[:L]
        b = np.maximum(b, 0.0)
        s = float(n) * b
        smax = float(np.max(s))
        sc = 2.0 * smax
        if not np.isfinite(sc):
            break

        if sc < local_sc - 1e-10:
            local_sc = sc
            local_v = v.copy()
            last_improve = now
        if sc < RUN["best_score"]:
            _update_run(p, sc)

        if active:
            Kuse = int(min(max(1, K), L))
            if Kuse >= L:
                w = np.ones(L, dtype=np.float64) / float(L)
            else:
                idx = np.argpartition(s, -Kuse)
                idx = idx[-Kuse:]
                w = np.zeros(L, dtype=np.float64)
                w[idx] = 1.0 / float(Kuse)
        else:
            beta = float(beta0) + (float(beta1_final) - float(beta0)) * progress
            beta = max(1e-3, beta)
            x = beta * (s - smax)
            x = np.clip(x, -745.0, 700.0)
            mx = float(np.max(x))
            e = np.exp(x - mx)
            ws = float(np.sum(e))
            if ws <= 1e-300:
                w = np.ones(L, dtype=np.float64) / float(L)
            else:
                w = e / ws

        Fw = np.fft.rfft(w, m)
        g = 2.0 * float(n) * np.fft.irfft(Fw * np.conj(F), m).real[:n]

        if not np.all(np.isfinite(g)):
            v = local_v + rng.normal(0.0, 0.05, size=n)
            v = v - float(np.mean(v))
            v = np.clip(v, -45.0, 45.0)
            continue

        gdot = float(np.dot(p, g))
        gv = p * (g - gdot)
        gv -= float(np.mean(gv))
        grms = float(np.sqrt(np.mean(gv * gv)))

        if grms <= 1e-15 or not np.isfinite(grms):
            sigma = 0.03 + 0.12 * (1.0 - progress)
            v = local_v + rng.normal(0.0, sigma, size=n)
            v = v - float(np.mean(v))
            v = np.clip(v, -45.0, 45.0)
            continue

        t += 1
        mu = beta1 * mu + (1.0 - beta1) * gv
        nu = beta2 * nu + (1.0 - beta2) * (gv * gv)
        b1 = beta1 ** t
        b2 = beta2 ** t
        mhat = mu / max(1e-12, 1.0 - b1)
        vhat = nu / max(1e-12, 1.0 - b2)

        lr = 0.12 * (1.0 - 0.65 * progress) + 0.005
        step = mhat / (np.sqrt(vhat) + eps)
        np.clip(step, -0.65, 0.65, out=step)

        v = v - lr * step
        v = v - float(np.mean(v))
        v = np.clip(v, -45.0, 45.0)

        if t % 4000 == 0:
            print("      Adam n=%d iter=%d local=%.6f global=%.6f beta=%.1f"
                  % (n, t, local_sc, RUN["best_score"],
                     beta if not active else -1.0))

    sc, p, b, F = _score_v(local_v, n, m)
    if np.isfinite(sc):
        _update_run(p, sc)
    print("    ADAM done n=%d local=%.6f global=%.6f"
          % (n, local_sc, RUN["best_score"]))
    return local_v, local_sc


def _lbfgs(v0, n, beta, seconds, hist=20, vmax=35.0, pen=0.25):
    """
    Custom L-BFGS on the smooth-max surrogate.
    """
    n = int(max(8, n))
    seconds = float(max(0.0, seconds))
    if seconds <= 0.5:
        v = _seq_to_v(np.ones(n, dtype=np.float64))
        return v, float("inf")

    m = _fft_len(n)
    v = np.asarray(v0, dtype=np.float64).ravel()
    if v.size != n:
        v = np.interp(np.linspace(0.0, 1.0, n),
                      np.linspace(0.0, 1.0, max(1, v.size)),
                      v).astype(np.float64)
    v = v - float(np.mean(v))
    v = np.clip(v, -vmax, vmax)

    f, g = _obj_grad_params(v, n, m, float(beta), float(vmax), float(pen))
    if not np.isfinite(f):
        v = np.zeros(n, dtype=np.float64)
        f, g = _obj_grad_params(v, n, m, float(beta), float(vmax), float(pen))

    S = []
    Y = []
    RHO = []
    start = time.monotonic()
    deadline = start + seconds
    last_alpha = 1.0
    it = 0

    sc, p, b, F = _score_v(v, n, m)
    if np.isfinite(sc):
        _update_run(p, sc)

    print("    LBFGS n=%d beta=%.1f budget=%.2fs start=%.6f"
          % (n, beta, seconds, RUN["best_score"]))

    while time.monotonic() < deadline:
        if not np.isfinite(f) or not np.all(np.isfinite(g)):
            break

        gnorm2 = float(np.dot(g, g))
        if gnorm2 <= 1e-16 or not np.isfinite(gnorm2):
            break

        d = -g.copy()
        if S:
            q = g.copy()
            alphas = []
            for i in range(len(S) - 1, -1, -1):
                si = S[i]
                yi = Y[i]
                ai = float(np.dot(si, q)) * RHO[i]
                alphas.append(ai)
                q = q - ai * yi
            vvar = q.copy()
            for i in range(len(S)):
                si = S[i]
                ai = alphas[i]
                bi = float(np.dot(si, vvar)) * RHO[i]
                vvar = vvar + (bi - ai) * si
            d = vvar

        gd = float(np.dot(g, d))
        if not np.isfinite(gd) or gd >= 0.0:
            d = -g.copy()
            gd = -gnorm2
        else:
            if -gd < 1e-8 * max(1.0, gnorm2):
                d = -g.copy()
                gd = -gnorm2
            else:
                scale = gnorm2 / (-gd)
                if not np.isfinite(scale) or scale <= 0.0:
                    d = -g.copy()
                    gd = -gnorm2
                else:
                    scale = min(max(scale, 1e-6), 1e3)
                    d = d * scale
                    gd = -gnorm2

        if gd >= 0.0:
            break

        alpha = last_alpha
        if not np.isfinite(alpha) or alpha <= 0.0:
            alpha = 1.0

        dnorm2 = float(np.dot(d, d))
        if not np.isfinite(dnorm2) or dnorm2 <= 0.0:
            break
        dnorm = math.sqrt(dnorm2)
        if dnorm > 1.0:
            alpha = min(alpha, 1.5 / dnorm)

        accepted = False
        vnew = None
        fnew = None
        gnew = None

        for _ls in range(45):
            vtrial = v + alpha * d
            vtrial = vtrial - float(np.mean(vtrial))
            vtrial = np.clip(vtrial, -vmax - 10.0, vmax + 10.0)
            ft, gt = _obj_grad_params(vtrial, n, m, float(beta), float(vmax), float(pen))
            if np.isfinite(ft) and ft <= f + 1e-4 * alpha * gd + 1e-12:
                accepted = True
                vnew = vtrial
                fnew = ft
                gnew = gt
                break
            alpha *= 0.5
            if alpha < 1e-13:
                break

        if not accepted:
            # Fallback steepest unit step.
            gnorm = math.sqrt(gnorm2)
            if gnorm <= 0.0 or not np.isfinite(gnorm):
                break
            d = -g / gnorm
            gd = -gnorm
            alpha = 0.1
            for _ls in range(20):
                vtrial = v + alpha * d
                vtrial = vtrial - float(np.mean(vtrial))
                vtrial = np.clip(vtrial, -vmax - 10.0, vmax + 10.0)
                ft, gt = _obj_grad_params(vtrial, n, m, float(beta), float(vmax), float(pen))
                if np.isfinite(ft) and ft <= f + 1e-4 * alpha * gd + 1e-12:
                    accepted = True
                    vnew = vtrial
                    fnew = ft
                    gnew = gt
                    break
                alpha *= 0.5
                if alpha < 1e-13:
                    break
            if not accepted:
                break

        sstep = vnew - v
        ystep = gnew - g
        curv = float(np.dot(sstep, ystep))
        if curv > 1e-12 and np.isfinite(curv):
            S.append(sstep)
            Y.append(ystep)
            RHO.append(1.0 / curv)
            if len(S) > hist:
                S.pop(0)
                Y.pop(0)
                RHO.pop(0)

        v = vnew
        f = fnew
        g = gnew
        last_alpha = min(1.0, alpha * 1.08)
        it += 1

        if it % 250 == 0:
            scnow, _, _, _ = _score_v(v, n, m)
            print("      LBFGS n=%d beta=%.1f iter=%d score=%.6f"
                  % (n, beta, it, scnow if np.isfinite(scnow) else RUN["best_score"]))

    sc, p, b, F = _score_v(v, n, m)
    if np.isfinite(sc):
        _update_run(p, sc)
    print("    LBFGS done n=%d beta=%.1f global=%.6f" % (n, beta, RUN["best_score"]))
    return v, sc


def _optimize_pool(selected, n, budget_s, seed_base):
    if not selected:
        return
    budget_s = float(max(0.0, budget_s))
    if budget_s <= 8.0:
        return

    n = int(max(8, n))
    weights = [0.50, 0.25, 0.15, 0.06, 0.04, 0.04]
    weights = weights[:len(selected)]
    if not weights:
        return
    total_w = float(sum(weights))
    if total_w <= 0.0:
        return

    print("  Optimizing pool n=%d with %d starts, budget=%.1fs"
          % (n, len(selected), budget_s))

    for idx, a in enumerate(selected):
        if budget_s <= 0.0:
            break
        a = _project_seq(a)
        sc0, a = _score_projected(a)
        if np.isfinite(sc0):
            _update_run(a, sc0)

        w = weights[idx] / total_w
        alloc = budget_s * w
        if alloc < 10.0:
            continue

        if np.isfinite(sc0) and sc0 > RUN["best_score"] + 0.30 and idx > 0:
            alloc *= 0.45

        adam_s = 0.58 * alloc
        lb_s = 0.42 * alloc

        v = _seq_to_v(a)
        v, _local = _adam_optimize(
            v, n, adam_s,
            int(seed_base) * 1000000 + idx * 917 + 17,
            beta0=8.0,
            beta1_final=70.0,
            active=False,
            K=16
        )

        if RUN["best_a"] is not None and RUN["best_a"].size == n:
            v = _seq_to_v(RUN["best_a"])
        if not np.isfinite(RUN["best_score"]) and RUN["best_a"] is None:
            v = _seq_to_v(a)

        lb_betas = (20.0, 55.0, 110.0)
        if lb_s > 5.0:
            per = lb_s / float(len(lb_betas))
            for b in lb_betas:
                if per < 5.0:
                    continue
                v, _ = _lbfgs(v, n, float(b), per, hist=20, vmax=35.0, pen=0.25)
                if RUN["best_a"] is not None and RUN["best_a"].size == n:
                    v = _seq_to_v(RUN["best_a"])

        # Very short active top-peak Adam at the end of a candidate polish.
        if alloc > 45.0 and RUN["best_a"] is not None and RUN["best_a"].size == n:
            v = _seq_to_v(RUN["best_a"])
            extra = min(12.0, 0.06 * budget_s)
            _adam_optimize(v, n, extra,
                           int(seed_base) * 1000000 + idx * 503 + 88,
                           beta0=30.0, beta1_final=120.0,
                           active=True, K=8)

        budget_s -= alloc

    print("  Pool done n=%d global=%.6f" % (n, RUN["best_score"]))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point invoked by the judge.
    Returns a Python list of nonnegative floats.
    """
    global RUN
    RUN = {"best_a": None, "best_score": float("inf")}

    budget_s = float(max(120.0, budget_s))
    seed = int(seed)
    start_all = time.monotonic()
    base_budget = max(60.0, budget_s - 90.0)
    deadline = start_all + base_budget

    rng = np.random.default_rng(seed + 101)

    # Load and normalize the provided previous solution, if present.
    prev = _load_prev()
    prev_for_fallback = None
    if prev is not None:
        if prev.size > 16384:
            interp_prev = _interp_to(8192, prev)
        else:
            interp_prev = _project_seq(prev)
        sc_prev, interp_prev = _score_projected(interp_prev)
        _update_run(interp_prev, sc_prev)
        prev_for_fallback = interp_prev.copy()
        print("Provided previous: n=%d FFT score=%.6f" % (interp_prev.size, sc_prev))
    else:
        fb = _fallback_seq()
        sc_fb, fb = _score_projected(fb)
        _update_run(fb, sc_fb)
        prev_for_fallback = fb.copy()
        print("no previous solution available; fallback score=%.6f n=%d"
              % (sc_fb, fb.size))

    if RUN["best_a"] is None:
        p = _fallback_seq()
        sc, p = _score_projected(p)
        _update_run(p, sc)

    # Choose main search length from current best, but keep it practical.
    cur = RUN["best_a"]
    if cur.size < 256:
        n_main = 2048
    elif cur.size > 16384:
        n_main = 8192
        cur = _interp_to(n_main, cur)
        sc, cur = _score_projected(cur)
        _update_run(cur, sc)
    else:
        n_main = int(cur.size)

    # Stage 1: main length.
    extra_main = []
    if RUN["best_a"] is not None:
        extra_main.append(_interp_to(n_main, RUN["best_a"]))
        extra_main.append(_interp_to(n_main, RUN["best_a"][::-1].copy()))

    inits_main = _make_inits(n_main, rng, extra_main)
    selected_main = _select_inits(inits_main, max_count=7)

    stage1_budget = min(0.46 * base_budget, max(0.0, deadline - time.monotonic()))
    if stage1_budget > 30.0:
        _optimize_pool(selected_main, n_main, stage1_budget, seed_base=123457)

    # Stage 2: try a larger length if enough time remains.
    now = time.monotonic()
    if now < deadline - 120.0 and RUN["best_a"] is not None:
        cur_n = int(RUN["best_a"].size)
        n_long = min(8192, max(2 * cur_n, 2048))
        if n_long > cur_n:
            extra_long = [
                _interp_to(n_long, RUN["best_a"]),
                _interp_to(n_long, RUN["best_a"][::-1].copy()),
            ]
            inits_long = _make_inits(n_long, rng, extra_long)
            selected_long = _select_inits(inits_long, max_count=5)
            stage2_budget = min(0.38 * base_budget, max(0.0, deadline - time.monotonic()))
            if stage2_budget > 60.0:
                _optimize_pool(selected_long, n_long, stage2_budget, seed_base=987654321)

    # Stage 3: roof-polishing on the best current sequence.
    for pass_i in range(4):
        now = time.monotonic()
        if now + 45.0 >= deadline:
            break
        if RUN["best_a"] is None:
            break

        best_p = _project_seq(RUN["best_a"])
        n = int(best_p.size)
        if n > 16384:
            best_p = _interp_to(8192, best_p)
            sc, best_p = _score_projected(best_p)
            _update_run(best_p, sc)
            n = int(best_p.size)

        rem = max(0.0, deadline - time.monotonic())
        if rem <= 30.0:
            break

        v = _seq_to_v(best_p)

        # Fast active-lag Adam.
        adam_rem = min(110.0, 0.30 * rem)
        _adam_optimize(v, n, adam_rem,
                       seed=789 + pass_i,
                       beta0=20.0, beta1_final=140.0,
                       active=True, K=8)

        rem = max(0.0, deadline - time.monotonic())
        if rem <= 25.0:
            break

        if RUN["best_a"] is not None:
            v = _seq_to_v(_project_seq(RUN["best_a"]))

        # L-BFGS polishes with increasingly sharp max surrogates.
        l1 = min(55.0, 0.35 * rem)
        if l1 > 15.0:
            v, _ = _lbfgs(v, n, 55.0, l1, hist=20, vmax=35.0, pen=0.20)

        rem = max(0.0, deadline - time.monotonic())
        if rem <= 20.0:
            break
        if RUN["best_a"] is not None:
            v = _seq_to_v(_project_seq(RUN["best_a"]))

        l2 = min(55.0, 0.30 * rem)
        if l2 > 10.0:
            v, _ = _lbfgs(v, n, 130.0, l2, hist=20, vmax=35.0, pen=0.20)

    # Finalize.
    out = RUN["best_a"]
    if out is None:
        out = prev_for_fallback if prev_for_fallback is not None else _fallback_seq()
    out = _project_seq(out)

    if out.size > 10000:
        out = _interp_to(8192, out)
        sc, out = _score_projected(out)
    else:
        sc, out = _score_projected(out)

    _update_run(out, sc)
    out = _project_seq(out)

    out_list = [float(x) for x in out]

    # Validate with exact evaluator when available and/large enough.
    try:
        exact = evaluate_sequence(out_list)
        print("Final FFT score=%.6f  exact score=%.6f  n=%d"
              % (RUN["best_score"], float(exact), len(out_list)))
        if not np.isfinite(exact):
            if prev_for_fallback is not None:
                out_list = [float(x) for x in _project_seq(prev_for_fallback)]
            else:
                out_list = [float(x) for x in _project_seq(_fallback_seq())]
    except Exception:
        print("Exact validation skipped or unavailable; FFT best=%.6f n=%d"
              % (RUN["best_score"], len(out_list)))

    # Safety: force finite nonnegative floats.
    cleaned = []
    total = 0.0
    for x in out_list:
        try:
            y = float(x)
        except Exception:
            y = 0.0
        if math.isnan(y) or math.isinf(y):
            y = 0.0
        if y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        cleaned.append(y)
        total += y
    if total <= 0.0:
        fallback = _project_seq(_fallback_seq())
        cleaned = [float(x) for x in fallback]
    else:
        cleaned = [x / total for x in cleaned]

    return cleaned
