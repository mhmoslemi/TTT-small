import numpy as np
import time

def _fft_conv(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fc = np.fft.irfft(fa * fa, size)
    return fc[:2*n-1]

def _obj_val(a, n):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)

def _project_nonneg_sum(a, S):
    """Project a to non-negative with sum S."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.full_like(a, S / len(a))
    return a * (S / s)

def _smooth_grad(a, k_list, c, tau):
    """Weighted gradient using top-K convolution coefficients with softmax weights."""
    # Use softmax over the top K coefficients
    top_vals = c[k_list]
    mx = np.max(top_vals)
    w = np.exp((top_vals - mx) / tau)
    w = w / np.sum(w)
    
    n = len(a)
    g = np.zeros(n)
    for idx, ki in enumerate(k_list):
        # Gradient of c[ki] w.r.t. a[i] is 2*a[ki-i] if 0 <= ki-i < n
        js = max(0, ki - n + 1)
        je = min(n, ki + 1)
        if js < je:
            ai = ki - np.arange(js, je)
            g[js:je] += w[idx] * 2.0 * a[ai]
    return g

def _hard_grad_at(a, k, n):
    """Exact gradient of c[k] w.r.t. a."""
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n, k + 1)
    if js < je:
        ai = k - np.arange(js, je)
        g[js:je] = 2.0 * a[ai]
    return g

def _optimize_smooth(a, n, deadline, K=50, tau_init=None, eta=0.05):
    """Smooth gradient descent using top-K coefficients."""
    if tau_init is None:
        tau_init = n / 10.0
    tau = tau_init
    S = float(np.sum(a))
    best = a.copy()
    best_obj = _obj_val(a, n)
    
    it = 0
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        # Find top K indices
        if K < len(c):
            k_list = np.argsort(c)[-K:][::-1]
        else:
            k_list = np.arange(len(c))
        
        g = _smooth_grad(a, k_list, c, tau)
        g -= np.mean(g)  # Maintain sum
        mg = np.max(np.abs(g))
        if mg < 1e-14:
            break
        a = _project_nonneg_sum(a - eta * g / mg, S)
        
        it += 1
        if it % 1000 == 0:
            obj = _obj_val(a, n)
            if obj < best_obj:
                best = a.copy()
                best_obj = obj
                print(f"    smooth it={it} obj={obj:.6f} tau={tau:.1f}")
            tau = max(tau * 0.99995, n * 0.001)
    
    # Final check
    obj = _obj_val(a, n)
    if obj < best_obj:
        best = a.copy()
        best_obj = obj
    return best, best_obj

def _optimize_hard(a, n, deadline, K=5):
    """Hard active-set descent on top K coefficients."""
    S = float(np.sum(a))
    best = a.copy()
    best_obj = _obj_val(a, n)
    eta = 0.03
    
    it = 0
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        if K < len(c):
            k_list = np.argsort(c)[-K:][::-1]
        else:
            k_list = np.arange(len(c))
        
        # Combined gradient
        g = np.zeros(n)
        for ki in k_list:
            g += _hard_grad_at(a, ki, n)
        g /= len(k_list)
        g -= np.mean(g)
        mg = np.max(np.abs(g))
        if mg < 1e-14:
            break
        a = _project_nonneg_sum(a - eta * g / mg, S)

                best_obj = obj
                print(f"    hard it={it} obj={obj:.6f}")

                eta = max(eta * 0.95, 1e-6)
    
    obj = _obj_val(a, n)
    if obj < best_obj:
        best = a.copy()
        best_obj = obj

def _perturb(a, n, rng, method='block'):
    """Perturb the sequence using various methods."""
    a_p = a.copy()
    S = float(np.sum(a))

        s1 = int(rng.integers(0, n - 2*w + 1))
        s2 = int(rng.integers(s1 + w, n - w + 1))
        a_p[s1:s1+w], a_p[s2:s2+w] = a_p[s2:s2+w].copy(), a_p[s1:s1+w].copy()
    elif method == 'mirror_partial':
        # Reverse a random sub-block
        w = max(2, int(n * rng.uniform(0.1, 0.5)))
        s = int(rng.integers(0, n - w + 1))
        a_p[s:s+w] = a_p[s:s+w][::-1]
    elif method == 'asymmetry_boost':
        # Boost asymmetry: multiply first half by (1+eps), second by (1-eps)
        eps = float(rng.uniform(0.05, 0.3))
        mid = n // 2
        a_p[:mid] *= (1 + eps)
        a_p[mid:] *= (1 - eps)
    elif method == 'random_restart':
        a_p = np.abs(rng.normal(1.0, 0.5, size=n))

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
1. FFT-based convolution (4x faster) enabling far more iterations
2. Multiple structured initializations that break symmetry
3. Top-K active set optimization (not just single argmax)
4. Structured perturbations targeting asymmetry
5. Adaptive learning rate with restarts
"""

import numpy as np
import time


def _fft_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fc = np.fft.irfft(fa * fa, size)
    return fc[:2 * n - 1]


def _project(a, S):
    """Project onto non-negative simplex with sum S."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def _obj(a, n):
    """Objective: 2*n*max(conv(a,a))/sum(a)^2."""
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def _grad_at(a, k, n):
    """Gradient of conv(a,a)[k] w.r.t a."""
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n, k + 1)
    if js < je:
        g[js:je] = 2.0 * a[k - np.arange(js, je)]
    return g


def _smooth_phase(a, n, deadline, K=30, tau0=None, eta0=0.05):
    """Entropy-smoothed gradient descent on top-K convolution entries."""
    if tau0 is None:
        tau0 = n / 15.0
    tau = tau0
    eta = eta0
    S = float(a.sum())
    best = a.copy()
    best_o = _obj(a, n)
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        idx = np.argpartition(c, -K)[-K:]
        vals = c[idx]
        mx = vals.max()
        w = np.exp((vals - mx) / tau)
        w /= w.sum()
        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += w[j] * _grad_at(a, ki, n)
        g -= g.mean()
        mg = float(np.abs(g).max())
        if mg < 1e-14:

        a = _project(a - eta * g / mg, S)
        it += 1

            if o < best_o - 1e-10:
                best_o = o
                best = a.copy()
                stall = 0
            else:
                stall += 1
            if stall > 15:
                eta = max(eta * 0.6, 1e-8)
                tau = max(tau * 0.99, n * 0.0005)
                stall = 0
            tau = max(tau * 0.99997, n * 0.0005)
