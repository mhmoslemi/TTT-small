"""
Search for low-peak nonnegative self-convolutions.
Uses a combination of LSE gradient descent, direct peak reduction,
and coordinate descent polishing to minimize 2*n*max(a*a)/(sum(a))^2.
"""

import numpy as np
import time
import math

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_gaussian(n, sigma):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return _normalize(a)


def _make_powerlaw(n, alpha):
    n = max(int(n), 2)
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-float(alpha))
    return _normalize(a)


def _make_sine(n, freq, phase=0.0):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 1.0 + float(freq) * np.sin(2.0 * np.pi * float(phase) * x)
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _make_cosine_sq(n, k=1, p=1.0):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.cos(np.pi * float(k) * x) ** (2.0 * float(p))
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _make_smooth_random(n, rng, amplitude=0.4):
    n = max(int(n), 2)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 16 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = 1.0 + float(amplitude) * f
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    s = float(old.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb(a, rng, strength=0.15):
    """Perturb a sequence multiplicatively."""
    n = a.size
    noise = np.exp(rng.normal(0.0, strength, size=n))
    a2 = a * noise
    a2 = a2 + 1e-10 * np.ones(n)
    return _normalize(a2)


def _block_perturb(a, rng, strength=0.2):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)


def _peak_reduction(a, c, top_k=3, strength=0.3):
    """
    Direct peak reduction: identify the top-k peaks in the convolution,
    and reduce the mass at positions contributing most to those peaks.
    """
    n = a.size
    c = np.asarray(c, dtype=np.float64).ravel()
    # Find top-k peak positions
    peak_positions = np.argsort(c)[-top_k:][::-1]
    peak_value = c[peak_positions[0]]
    
    # Compute sensitivity: for each position i, how much does it contribute to the peaks?
    sensitivity = np.zeros(n, dtype=np.float64)
    for k in peak_positions:
        # c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # dc[k]/da[i] = a[k-i] (if k-i is valid) + a[k-i] (the other term if i != k-i)
        # Actually dc[k]/da[i] = a[k-i] for valid k-i, doubled if i == k-i
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    sensitivity[i] += a[i]  # derivative of a[i]^2 is 2*a[i], but we use a[i] as weight
                else:
                    sensitivity[i] += a[j]
    
    # Normalize sensitivity
    s_max = float(np.max(sensitivity))
    if s_max > 1e-300:
        sensitivity = sensitivity / s_max
    
    # Reduce mass at high-sensitivity positions, add to low-sensitivity
    mask_high = sensitivity > 0.7
    mask_low = sensitivity < 0.3
    
    if mask_high.any() and mask_low.any():
        total_high = float(a[mask_high].sum())
        total_low = float(a[mask_low].sum())
        if total_high > 0 and total_low > 0:
            # Move some mass from high to low
            transfer = float(strength) * total_high
            n_high = int(mask_high.sum())
            n_low = int(mask_low.sum())
            reduction = transfer / n_high
            increase = transfer / n_low
            a[mask_high] = np.maximum(a[mask_high] - reduction, 0.0)
            a[mask_low] += increase
        return _normalize(a)
    
    return _normalize(a)


def _coordinate_descent_polish(a, n_iters=100, deadline=None):
    """Polish the sequence by trying small shifts of mass between pairs of positions."""
    n = a.size
    best_a = a.copy()
    best_score = _score_of(best_a)
    b_fft = _next_pow2(2 * n - 1)
    c = _autocorr(best_a, b_fft)
    best_peak = float(np.max(c))
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        # Try moving mass from position i to position j
        # Only try a subset of pairs for speed
        if n > 512:
            indices = np.random.choice(n, size=min(n, 200), replace=False)
        else:
            indices = np.arange(n)
        
        for i in indices:
            if a[i] < 1e-10:
                continue
            # Try moving a small amount from i to nearby positions
            delta = a[i] * 0.05
            for j in range(max(0, i - 5), min(n, i + 6)):
                if i == j:
                    continue
                old_i = a[i]
                old_j = a[j]
                a[i] = max(0.0, old_i - delta)
                a[j] = old_j + delta
                s = float(a.sum())
                if s < 1e-10:
                    a[i] = old_i
                    a[j] = old_j
                    continue
                a = a / s
                c_new = _autocorr(a, b_fft)
                peak = float(np.max(c_new))
                if peak < best_peak - 1e-14:
                    best_peak = peak
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = old_i
                    a[j] = old_j
                    a = a / s
        
        if not improved and it > 10:
            break
    
    return best_a, best_score  # Note: best_score might be stale, recompute


def _lse_gradient_step(a, c, beta, lr):
    """
    One step of LSE gradient descent.
    a: current sequence (sum=1)
    c: self-convolution of a
    beta: LSE temperature
    lr: learning rate
    Returns updated a.
    """
    n = a.size
    c = np.asarray(c, dtype=np.float64).ravel()
    m = float(np.max(c))
    if beta <= 1.0:
        weights = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        arg = beta * (c - m)
        arg = np.maximum(arg, -745.0)
        e = np.exp(arg)
        s = float(e.sum())
        if s <= 0:
            weights = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            weights = e / s
    
    # Gradient of sum_k weights[k] * c[k] w.r.t. a[i]
    # c[k] = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} a[i] * a[k-i]
    # dc[k]/da[i] = a[k-i] (if k-i valid and i != k-i) or 2*a[i] (if i == k-i)
    # So grad[i] = sum_k weights[k] * dc[k]/da[i]
    #           = sum_k weights[k] * a[k-i] (for valid k-i, with factor 2 if k=2i)
    
    # This is equivalent to: grad = 2 * (reversed_a * weights_reversed) convolved appropriately
    # Let's compute it directly via FFT
    
    # grad[i] = sum_{k} weights[k] * a[k-i] where k ranges over [0, 2n-2] and k-i is in [0, n-1]
    # Let's define: for each k, the contribution to grad[i] is weights[k] * a[k-i]
    # This is a cross-correlation of weights with a (reversed).
    
    # Actually: grad[i] = sum_{k} weights[k] * a[k-i] = (weights * a_reversed)[i + (n-1)]
    # where * denotes cross-correlation
    
    # More precisely:
    # grad[i] = sum_{j=0}^{n-1} a[j] * sum_{k: k-i=j} weights[k]
    #         = sum_{j=0}^{n-1} a[j] * weights[i+j]
    
    # So grad[i] = sum_j a[j] * weights[i+j]
    # This is a "sliding dot product" of a with weights starting at position i.
    
    # We can compute this via FFT:
    # Define w_ext of length 3n-2: w_ext[k] = weights[k] for k in [0, 2n-2], 0 otherwise.
    # Then grad[i] = sum_j a[j] * w_ext[i+j] = (a_rev * w_ext)[(n-1) + i] where * is convolution
    # Actually: let's define b = a reversed, length n.
    # (b * w_ext)[k] = sum_j b[j] * w_ext[k-j] = sum_j a[n-1-j] * w_ext[k-j]
    # We want grad[i] = sum_j a[j] * w_ext[i+j]
    # Let j' = n-1-j: grad[i] = sum_{j'} a[n-1-j'] * w_ext[i + n-1-j']
    # = (b * w_ext)[i + n-1] where b[j'] = a[n-1-j']
    
    # So: compute conv = irfft(fft(b, 3n-2) * fft(w_ext, 3n-2))
    # grad = conv[n-1 : 2n-1]
    
    b = a[::-1]
    w_ext = np.zeros(3 * n - 2, dtype=np.float64)
    w_ext[:2 * n - 1] = weights
    fft_size = _next_pow2(3 * n - 2)
    B = np.fft.rfft(b, fft_size)
    W = np.fft.rfft(w_ext, fft_size)
    conv = np.fft.irfft(B * W, fft_size).real
    grad = conv[n - 1:2 * n - 1]
    
    # Apply gradient step
    a_new = a - lr * grad
    # Center the gradient (subtract mean) for stability
    # Actually, let's just clip and normalize
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-300:
        return a.copy()
    return a_new / s

Search for low-peak nonnegative self-convolutions using LSE gradient descent

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:

    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb
