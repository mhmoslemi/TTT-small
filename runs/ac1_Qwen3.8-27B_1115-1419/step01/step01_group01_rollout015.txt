"""
Search for non-negative sequences minimizing
    2*n*max(np.convolve(a,a)) / (sum(a)^2)

Approach:
  - Use the previously known good sequence as a primary seed if present.
  - Build asymmetric non-negative initial shapes (power-law envelopes,
    shifted Gaussians, sine shapes, random spectra) to avoid symmetric
    local minima.
  - Refine sequences with two complementary optimizers:
      * a log-softmax Adam-style optimizer
      * a simpler simplex projected-gradient line-search optimizer
  - The gradient uses FFT-based evaluation of weighted self-convolution.
  - Try progressive lengths (2048, 4096, 8192) and keep the best sequence.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _fft_len(n):
    n = int(n)
    if n <= 0:
        return 2
    return _next_pow2(2 * n - 1)


def _proj_fast(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300 or not np.isfinite(s):
        v.fill(1.0 / float(v.size))
    else:
        v /= s
    return v


def _proj_exact(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return v.copy()
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)
    n = v.size
    sv = np.sort(v)[::-1]
    idx = np.arange(1, n + 1, dtype=np.float64)
    ssv = np.cumsum(sv)
    cond = sv > (ssv - 1.0) / idx
    if not np.any(cond):
        k = n - 1
    else:
        k = int(np.nonzero(cond)[0][-1])
    theta = (ssv[k] - 1.0) / (k + 1.0)
    x = np.maximum(v - theta, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        x.fill(1.0 / float(n))
    else:
        x /= s
    return x


def _conv_self(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.zeros(1, dtype=np.float64), np.ones(1, dtype=np.complex128)
    if m < 2 * n - 1:
        m = _fft_len(n)
    F = np.fft.rfft(a, m)
    b = np.fft.irfft(F * F, m)[:2 * n - 1]
    np.maximum(b, 0.0, out=b)
    return b, F


def _score_fft(a, m=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-300:
        return np.inf
    if m is None or m < 2 * n - 1:
        m = _fft_len(n)
    b, _ = _conv_self(a, m)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _interp_to(a, n):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a / s
    if a.size == n:
        return _proj_fast(a)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return _proj_fast(out)


def _soft(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return v.copy()
    v = v - float(np.max(v))
    v = np.clip(v, -745.0, 0.0)

    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / float(v.size), dtype=np.float64)

    return w


def _perturb(a, rng, sigma):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a.copy()
    noise = rng.normal(0.0, float(sigma), size=a.size)
    v = a * np.exp(noise)
    return _proj_fast(v)


def _base_shape_power(n, p, q):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    eps = 1e-6
    a = np.power(t + eps, float(p)) * np.power(1.0 - t + eps, float(q))
    return _proj_fast(a)


def _base_shape_gauss(n, c, sig):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * np.square((t - float(c)) / float(sig)))
    return _proj_fast(a)


def _base_shape_sine(n, p, beta):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    a = np.power(np.sin(np.pi * t) + 1e-6, float(p)) * np.exp(float(beta) * t)
    return _proj_fast(np.maximum(a, 0.0))


def _base_shape_random(n, rng, kc=12):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n, dtype=np.float64)
    kc = int(max(1, kc))
    for k in range(1, kc + 1):
        amp = rng.uniform(0.05, 0.30) / max(1.0, float(k) ** 0.7)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t + ph)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
    else:
        a /= s
    return _proj_fast(a)


def _generate_candidates(base, n, rng):
    n = int(n)
    cands = []
    limit = 70

    if base is not None and len(base) > 0:
        b0 = _interp_to(base, n)
        cands.append(b0)
        cands.append(b0[::-1].copy())

        i = np.arange(n, dtype=np.float64)
        for gam in (-1.5, -0.75, -0.35, 0.35, 0.75, 1.5):
            if len(cands) < limit:
                f = np.power((i + 1.0) / float(n), gam)
                cands.append(_proj_fast(b0 * f))
            if len(cands) < limit:
                f = np.power((float(n) - i) / float(n), gam)
                cands.append(_proj_fast(b0 * f))

        for sig in (0.08, 0.18, 0.35):
            if len(cands) < limit:
                cands.append(_perturb(b0, rng, sig))

    pw = (0.1, 0.3, 0.6, 1.0, 2.0, 3.5)
    for p in pw:
        for q in pw:
            if len(cands) >= limit:
                break
            cands.append(_base_shape_power(n, p, q))
        if len(cands) >= limit:
            break

    for c in (0.30, 0.40, 0.50, 0.60, 0.70):
        for sig in (0.15, 0.25, 0.35):
            if len(cands) >= limit:
                break
            cands.append(_base_shape_gauss(n, c, sig))
        if len(cands) >= limit:
            break

    for p in (0.5, 1.0, 2.0):
        for beta in (-1.5, -0.5, 0.5, 1.5):
            if len(cands) >= limit:
                break
            cands.append(_base_shape_sine(n, p, beta))
        if len(cands) >= limit:
            break

    for _ in range(6):
        if len(cands) >= limit:
            break
        cands.append(_base_shape_random(n, rng, 12))

    return cands


def _score_of_pair(pair):
    return pair[0]


def _select_diverse(cands, n, max_k):
    n = int(n)
    scored = []
    for a in cands:
        try:
            a = _proj_fast(a)
            sc = _score_fft(a)
            if np.isfinite(sc):
                scored.append((float(sc), a))
        except Exception:
            pass

    if not scored:
        return []

    scored.sort(key=_score_of_pair)
    threshold = max(0.02, 0.75 / float(np.sqrt(max(2, n))))
    selected = []
    for sc, a in scored:
        if not selected:
            selected.append((sc, a))
            continue
        ok = True
        for _, b in selected:
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < threshold:
                ok = False
                break
        if ok:
            selected.append((sc, a))
            if len(selected) >= max_k:
                break
    return selected


def optimize_adam(a0, n, max_seconds, seed, goal_score):
    n = int(max(3, n))
    max_seconds = float(max_seconds)
    a = _interp_to(a0, n)
    m = _fft_len(n)

    best_a = a.copy()
    best_score = _score_fft(a, m)
    if not np.isfinite(best_score):
        a = np.ones(n, dtype=np.float64) / float(n)
        best_a = a.copy()
        best_score = 2.0

    if max_seconds <= 5.0:
        return _proj_fast(best_a), float(best_score)

    theta = np.log(np.maximum(a, 1e-300))
    theta -= float(np.mean(theta))
    theta = np.clip(theta, -30.0, 30.0)

    rng = np.random.default_rng(int(seed))
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    start = time.monotonic()
    deadline = start + max_seconds
    last_improve = start
    stuck_s = min(20.0, max(5.0, 0.25 * max_seconds))

    while time.monotonic() < deadline:
        a = _soft(theta)
        b, F = _conv_self(a, m)
        bmax = float(b.max()) if b.size else 0.0
        if not np.isfinite(bmax) or bmax <= 0.0:
            break
        score = 2.0 * n * bmax
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > stuck_s:
            sigma = 0.22
            theta = theta + rng.normal(0.0, sigma, size=n)
            theta -= float(np.mean(theta))
            theta = np.clip(theta, -35.0, 35.0)
            last_improve = time.monotonic()
            continue

        progress = float(np.clip((time.monotonic() - start) / max_seconds, 0.0, 1.0))
        w = _make_weights(b, n, progress, float(goal_score), 1.0)
        Fw = np.fft.rfft(w, m)
        c = np.fft.irfft(F * np.conj(Fw), m)[:n]
        np.maximum(c, 0.0, out=c)

        cm = float(np.dot(c, a))
        gv = a * (c - cm)
        grms = float(np.sqrt(np.mean(gv * gv)))
        if not np.isfinite(grms) or grms < 1e-14:
            theta = theta + rng.normal(0.0, 0.10, size=n)
            theta -= float(np.mean(theta))
            theta = np.clip(theta, -35.0, 35.0)
            continue

        gv = gv / grms
        t += 1
        mu = beta1 * mu + (1.0 - beta1) * gv
        nu = beta2 * nu + (1.0 - beta2) * (gv * gv)

        mhat = mu / max(1e-12, 1.0 - beta1 ** t)
        vhat = nu / max(1e-12, 1.0 - beta2 ** t)
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.5, 0.5)

        lr = 0.25 * (1.0 - 0.60 * progress)
        theta = theta - lr * step
        theta -= float(np.mean(theta))
        theta = np.clip(theta, -35.0, 35.0)

    a_final = _soft(theta)
    s_final = _score_fft(a_final, m)
    if np.isfinite(s_final) and s_final < best_score:
        best_score = float(s_final)
        best_a = a_final.copy()

    return _proj_fast(best_a), float(best_score)


def optimize_projected(a0, n, max_seconds, seed, goal_score):
    n = int(max(3, n))
    max_seconds = float(max_seconds)
    a = _interp_to(a0, n)
    m = _fft_len(n)

    best_a = a.copy()
    best_score = _score_fft(a, m)
    if not np.isfinite(best_score):
        a = np.ones(n, dtype=np.float64) / float(n)
        best_a = a.copy()
        best_score = 2.0

    if max_seconds <= 5.0:
        return _proj_fast(best_a), float(best_score)

    rng = np.random.default_rng(int(seed))
    start = time.monotonic()
    deadline = start + max_seconds
    last_improve = start
    stuck_s = min(10.0, max(4.0, 0.12 * max_seconds))
    last_lam = 0.20
    mom = np.zeros(n, dtype=np.float64)
    proj = _proj_exact if n <= 2048 else _proj_fast

    while time.monotonic() < deadline:
        b, F = _conv_self(a, m)
        bmax = float(b.max()) if b.size else 0.0
        S = float(a.sum())
        if not np.isfinite(bmax) or bmax <= 0.0 or S <= 1e-300:
            break
        score_cur = 2.0 * n * bmax / (S * S)
        if np.isfinite(score_cur) and score_cur < best_score:
            best_score = float(score_cur)
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > stuck_s:
            p = _perturb(a, rng, 0.16)
            sp = _score_fft(p, m)
            if np.isfinite(sp):
                if sp < score_cur or sp < 1.04 * best_score:
                    a = p
                    score_cur = sp
                    if sp < best_score:
                        best_score = float(sp)
                        best_a = a.copy()
            mom.fill(0.0)
            last_lam = 0.20
            last_improve = time.monotonic()
            continue

        progress = float(np.clip((time.monotonic() - start) / max_seconds, 0.0, 1.0))
        w = _make_weights(b, n, progress, float(goal_score), S)
        Fw = np.fft.rfft(w, m)
        c = np.fft.irfft(F * np.conj(Fw), m)[:n]
        np.maximum(c, 0.0, out=c)

        g = c - float(np.mean(c))
        mom = 0.70 * mom + 0.30 * g
        scale = max(1e-15, float(np.max(np.abs(mom))))
        d = -mom / scale
        d -= float(np.mean(d))
        d = np.clip(d, -4.0, 4.0)
        d -= float(np.mean(d))

        base_lam = 0.30 * (1.0 - 0.55 * progress)
        lams = [max(0.005, last_lam),
                base_lam,
                max(0.004, 0.5 * base_lam),
                max(0.003, 0.20 * base_lam),
                0.005]

        best_sc = np.inf
        best_trial = a
        chosen_lam = 0.005

        for lam in lams:
            lam = float(lam)
            if lam <= 0.0:
                continue
            trial = a + lam * d
            trial = proj(trial)
            sc = _score_fft(trial, m)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_trial = trial
                chosen_lam = lam

        if np.isfinite(best_sc) and best_sc < score_cur - 1e-12:
            a = best_trial
            last_lam = chosen_lam
            if best_sc < best_score - 1e-12:
                best_score = float(best_sc)
                best_a = a.copy()
                last_improve = time.monotonic()
        else:
            if np.isfinite(best_sc) and best_sc < 1.003 * score_cur:
                a = best_trial
                last_lam = chosen_lam
            else:
                mom = 0.5 * mom
                last_lam = max(0.005, 0.7 * chosen_lam)

    s_final = _score_fft(a, m)
    if np.isfinite(s_final) and s_final < best_score:
        best_score = float(s_final)
        best_a = a.copy()

    return _proj_fast(best_a), float(best_score)


def _optimize_two(a0, n, max_seconds, seed, goal_score):
    n = int(max(3, n))
    max_seconds = float(max_seconds)
    if max_seconds <= 1.0:
        a = _interp_to(a0, n)
        return _proj_fast(a), _score_fft(a)

    start = time.monotonic()
    deadline = start + max_seconds
    a = _interp_to(a0, n)
    best_a = a.copy()
    best_score = _score_fft(a)
    if not np.isfinite(best_score):
        a = np.ones(n, dtype=np.float64) / float(n)
        best_a = a.copy()
        best_score = _score_fft(a)

    try:
        if max_seconds > 45.0:
            t_adam = min(max_seconds * 0.35, 120.0)
            rem = deadline - time.monotonic()
            if rem > 5.0:
                a1, s1 = optimize_adam(best_a, n, min(t_adam, rem), int(seed), goal_score)
                if np.isfinite(s1) and s1 < best_score:
                    best_score = float(s1)
                    best_a = _proj_fast(a1)

            rem = deadline - time.monotonic()
            if rem > 10.0:
                a2, s2 = optimize_projected(best_a, n, rem, int(seed) + 101, goal_score)
                if np.isfinite(s2) and s2 < best_score:
                    best_score = float(s2)
                    best_a = _proj_fast(a2)
        else:
            rem = deadline - time.monotonic()
            a2, s2 = optimize_projected(best_a, n, rem, int(seed), goal_score)
            if np.isfinite(s2) and s2 < best_score:
                best_score = float(s2)
                best_a = _proj_fast(a2)
    except Exception as ee:
        print("optimizer error caught: %s" % str(ee))

    return _proj_fast(best_a), float(best_score)


def _process_stage(n, stage_deadline, best_a, best_score, rng, goal_score):
    n = int(n)
    now = time.monotonic()
    if now >= stage_deadline or best_a is None:
        return best_a, best_score

    cands = _generate_candidates(best_a, n, rng)
    selected = _select_diverse(cands, n, 5)
    if not selected:
        a = _interp_to(best_a, n)
        s = _score_fft(a)
        if not np.isfinite(s):
            return best_a, best_score
        selected = [(float(s), a)]

    k = len(selected)
    if k == 1:
        weights = [1.0]
    elif k == 2:
        weights = [0.60, 0.40]
    elif k == 3:
        weights = [0.50, 0.30, 0.20]
    elif k == 4:
        weights = [0.45, 0.30, 0.18, 0.07]
    else:
        weights = [0.40, 0.25, 0.18, 0.12, 0.05]

    total_w = float(np.sum(np.asarray(weights[:k], dtype=np.float64)))
    if total_w <= 0.0:
        total_w = 1.0

    start_stage = time.monotonic()
    total_span = max(0.0, stage_deadline - start_stage)
    if total_span < 5.0:
        return best_a, best_score

    seed = int(rng.integers(0, 2**31 - 1))
    for i in range(k):
        now = time.monotonic()
        if now + 8.0 >= stage_deadline:
            break
        alloc = max(8.0, min(stage_deadline - now,
                             total_span * (float(weights[i]) / total_w)))
        a_cand = selected[i][1]
        a_opt, s_opt = _optimize_two(a_cand, n, alloc, seed + i * 17, goal_score)
        if np.isfinite(s_opt) and s_opt < best_score:
            best_score = float(s_opt)
            best_a = _proj_fast(a_opt)
            print("  n=%d candidate %d score=%.6f" % (n, i + 1, best_score))

    now = time.monotonic()
    if now + 10.0 < stage_deadline:
        rem = min(stage_deadline - now, max(10.0, total_span * 0.2))
        a_opt, s_opt = _optimize_two(best_a, n, rem, seed + 9000, goal_score)
        if np.isfinite(s_opt) and s_opt < best_score:
            best_score = float(s_opt)
            best_a = _proj_fast(a_opt)
            print("  n=%d final stage polish score=%.6f" % (n, best_score))

    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Returns a non-negative Python list of heights.
    """
    start = time.monotonic()
    budget = float(max(120.0, float(budget_s)))
    global_deadline = start + max(60.0, budget - 35.0)
    rng = np.random.default_rng(int(seed))
    goal_score = 1.485

    best_a = None
    best_score = np.inf

    try:
        prev = globals().get("height_sequence_1", None)
        if prev is not None:
            arr = np.asarray(prev, dtype=np.float64).ravel()
            if arr.ndim != 1:
                arr = arr.ravel()
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            s = float(arr.sum())
            if arr.size > 0 and s > 0.0:
                best_a = arr / s
                best_score = _score_fft(best_a)
                print("loaded previous sequence: n=%d fft_score=%.6f" %
                      (len(best_a), best_score))
    except Exception as ee:
        print("could not load previous sequence: %s" % str(ee))

    if best_a is None or not np.isfinite(best_score):
        best_a = _base_shape_power(1024, 0.6, 1.2)
        best_score = _score_fft(best_a)
        print("no previous sequence; using simple power-law start. score=%.6f" %
              best_score)

    n_best = int(len(best_a))
    if n_best <= 1024:
        specs = [(1024, 0.18), (2048, 0.30), (4096, 0.30), (8192, 0.15)]
    elif n_best <= 2048:
        specs = [(2048, 0.42), (4096, 0.33), (8192, 0.18)]
    elif n_best <= 4096:
        specs = [(4096, 0.45), (8192, 0.35)]
    else:
        specs = [(8192, 0.55), (4096, 0.25)]

    total_frac = 0.0
    for _, f in specs:
        total_frac += float(f)
    if total_frac <= 0.0:
        total_frac = 1.0

    main_total = max(120.0, (budget - 60.0) * 0.92)
    main_deadline = start + main_total
    main_start = start
    cum_frac = 0.0

    for idx, (n, f) in enumerate(specs):
        now = time.monotonic()
        if now + 30.0 >= main_deadline:
            break
        cum_frac += float(f)
        stage_deadline = min(main_deadline - 5.0,
                             main_start + main_total * (cum_frac / total_frac))
        if stage_deadline <= now + 10.0:
            continue
        print("stage n=%d budget=%.1fs current_best=%.6f" %
              (n, stage_deadline - now, best_score))
        best_a, best_score = _process_stage(int(n), stage_deadline,
                                            best_a, best_score,
                                            rng, goal_score)
        print("after stage n=%d best=%.6f" % (n, best_score))

    now = time.monotonic()
    if now + 15.0 < global_deadline:
        rem = global_deadline - 15.0 - now
        if rem > 5.0:
            print("final polish at n=%d budget=%.1fs" % (len(best_a), rem))
            a_opt, s_opt = _optimize_two(best_a, len(best_a), rem,
                                         int(seed) + 234567, goal_score)
            if np.isfinite(s_opt) and s_opt < best_score:
                best_score = float(s_opt)
                best_a = _proj_fast(a_opt)
            print("after final polish best=%.6f" % best_score)

    best_a = _proj_fast(np.asarray(best_a, dtype=np.float64).ravel())
    best_a = np.clip(best_a, 0.0, 1000.0)
    if float(best_a.sum()) > 0.0:
        best_a = best_a / float(best_a.sum())
    else:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    exact_score = np.inf
    try:
        eval_func = globals().get("evaluate_sequence", None)
        if callable(eval_func) and len(best_a) <= 12288:
            exact_score = float(eval_func([float(x) for x in best_a]))
            print("FFT score=%.6f exact score=%.6f n=%d" %
                  (best_score, exact_score, len(best_a)))
            if not np.isfinite(exact_score):
                raise ValueError("invalid exact score")
    except Exception as ee:
        print("exact validation skipped or failed: %s" % str(ee))
        fallback = globals().get("height_sequence_1", None)
        if fallback is not None:
            try:
                arr = np.asarray(fallback, dtype=np.float64).ravel()
                arr = np.where(np.isfinite(arr), arr, 0.0)
                arr = np.maximum(arr, 0.0)
                s = float(arr.sum())
                if arr.size > 0 and s > 0.0:
                    best_a = arr / s
                    exact_score = np.inf
            except Exception:
                pass
        if best_a is None or float(np.sum(best_a)) <= 0.0:
            best_a = np.ones(1024, dtype=np.float64) / 1024.0

    best_a = np.clip(best_a, 0.0, 1000.0)
    if float(best_a.sum()) > 0.0:
        best_a = best_a / float(best_a.sum())
    else:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0

    return [float(x) for x in best_a]
