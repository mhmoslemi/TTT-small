import numpy as np
from scipy.optimize import linprog
import time

def fast_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0. 0, out=c)
    return c

def score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolution(a)
    return 2.0 * n * c.max() / (s * s)

def build_A(f):
    """Build the convolution constraint matrix for the LP."""
    n = len(f)
    L = 2 * n - 1
    # A[k, i] = f[i] if 0 <= k-i < n
    A = np.zeros((L, n))
    for i in range(n):
        if f[i] > 0:
            A[i:i+n, i] = f[i]  # rows i to i+n-1, column i
    return A

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:

    b = res.x
    s_b = b.sum()
    s_f = f.sum()
    if s_b < 1e-10:

"""
Multi-resolution LP-based search for non-negative step-function heights
minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

Uses alternating LP projection: at each step, solve a top-K linear program
to find the best non-negative direction with bounded cross-correlation,
then mix with the current sequence. Multi-resolution coarse-to-fine search
with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolution(a):
    """FFT-based linear convolution a * a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def fast_score(a):
    """Fast FFT-based score computation."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fast_convolution(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def normalize_seq(a, target=None):
    """Normalize sequence to non-negative, finite, with target sum."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        a.fill(1.0)
        s = float(a.sum())
    if target is not None:
        a *= target / s
        a = np.clip(a, 0.0, 1000.0)
    return a


def lp_step(f, K=30, upper_bound=None):
    """
    One LP-based step: find direction b maximizing sum(b)
    subject to (f*b)[k] <= M for top-K k values, b >= 0.
    Returns the normalized direction g (same sum as f).
    """
    n = len(f)
    c = fast_convolution(f)
    M = float(c.max())
    if M < 1e-15:
        return f.copy()
    
    L = 2 * n - 1
    K = min(K, L)
    
    # Get top-K constraint indices
    if K < L:
        top_k = np.argpartition(c, -K)[-K:]
    else:
        top_k = np.arange(L)
    
    # Build constraint matrix: A[idx, i] = f[i] if i in [max(0,k-n+1), min(n-1,k)]
    A = np.zeros((K, n), dtype=np.float64)
    for idx, k in enumerate(top_k):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        A[idx, i_start:i_end + 1] = f[i_start:i_end + 1]
    
    # LP: max sum(b) s.t. A@b <= M, 0 <= b <= ub
    if upper_bound is None:
        upper_bound = float(f.sum()) * 2.0 + 1.0
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, upper_bound)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return f.copy()
        b = res.x
    except Exception:
        return f.copy()
    
    s_b = float(b.sum())
    if s_b < 1e-15:
        return f.copy()
    
    # Normalize to match sum of f
    s_f = float(f.sum())
    g = b * (s_f / s_b)
    g = np.maximum(g, 0.0)
    return g
