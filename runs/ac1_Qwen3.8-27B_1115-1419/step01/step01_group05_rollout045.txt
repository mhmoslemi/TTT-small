def _next_pow2(size):

    if size <= 1:

    """Clip to nonnegative, normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()

    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:2 * n - 1]

def _score(a):
    """Exact score: 2*n*max(a*a) with sum(a)=1."""
    a = _normalize(a)
    n = a.size
    b = _autocorr(a)
    mb = float(np.max(b))
    return 2.0 * n * mb if np.isfinite(mb) else 1e100

def _lse_w(b, beta):
    """Return (LSE value, softmax weights) of b at temperature beta."""
    m = float(np.max(b))
    if beta <= 1.0:
        return m, np.ones(b.size, dtype=np.float64) / b.size
    arg = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s < 1e-300:
        return m, np.ones(b.size, dtype=np.float64) / b.size
    w = e / s
    return m + float(np.log(s)) / beta, w

def _grad_a(a, w, fs):
    """Gradient of LSE w.r.t. a: 2 * cross_corr(w, a)."""
    n = a.size
    ar = a[::-1]
    conv = np.fft.irfft(
        np.fft.rfft(w, fs) * np.fft.rfft(ar, fs), fs
    ).real
    return 2.0 * conv[n - 1:2 * n - 1]

def _proj_simplex(v):
    """Euclidean projection onto {x>=0, sum=1}."""
    n = v.size
    if n <= 1:
        return np.ones(n, dtype=np.float64) / max(n, 1)
    v = np.asarray(v, dtype=np.float64)
    vs = np.sort(v)[::-1]
    css = np.cumsum(vs)
    idx = np.arange(1, n + 1, dtype=np.float64)
    ok = vs - (css - 1.0) / idx > 0
    ii = np.where(ok)[0]
    if ii.size == 0:
        return np.ones(n, dtype=np.float64) / n
    k = ii[-1]
    theta = (css[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)

def _make_exp(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-alpha * x))

def _make_bell(n, c=0.5, w=0.25):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-0.5 * ((x - c) / w) ** 2))

def _make_sym(n, alpha=1.0):
    """Symmetric exponential (high in middle)."""
    x = np.linspace(-1.0, 1.0, n)
    return _normalize(np.exp(-alpha * x ** 2))

def _make_smooth_rand(n, rng, base=None, amp=0.3):
    if base is None:
        base = np.ones(n, dtype=np.float64) / n
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cut = max(2, n // 12 + 2)
    if cut < len(F):
        F[cut:] = 0.0
    f = np.fft.irfft(F, n)
    f -= f.mean()
    sd = f.std()
    if sd > 1e-15:
        f /= sd
    a = base * (1.0 + amp * f)
    return _normalize(np.maximum(a, 1e-12))

def _resample(old, new_n):
    old = _normalize(old)
    if old.size == new_n:
        return old.copy()
    xo = np.linspace(0.0, 1.0, old.size)
    xn = np.linspace(0.0, 1.0, new_n)
    return _normalize(np.interp(xn, xo, old))

def _fft_refine(a, deadline, rng, beta0=8.0, verbose=True):
    """
    LSE gradient descent with true simplex projection + backtracking.
    """
    a = _normalize(a)
    n = a.size
    fs_b = _next_pow2(2 * n - 1)
    fs_g = _next_pow2(3 * n - 2)
    
    best_a = a.copy()
    best_score = _score(a)
    beta = beta0
    alpha = 0.2
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        b = _autocorr(a, fs_b)
        sc = 2.0 * n * float(np.max(b))
        
        if sc < best_score - 1e-13:
            best_a, best_score = a.copy(), sc
            stall = 0
        else:
            stall += 1
        
        lse_val, w = _lse_w(b, beta)
        g = _grad_a(a, w, fs_g)
        gc = g - g.mean()
        gn = float(np.dot(gc, gc))
        
        if gn < 1e-28:
            a = _make_smooth_rand(n, rng, best_a, 0.35)
            beta = max(5.0, beta * 0.5)
            alpha = 0.2
            stall = 0
            it += 1
            continue
        
        # Backtracking line search (max 15 steps)
        step_found = False
        a_try = a
        for _ in range(15):
            a_try = _proj_simplex(a - alpha * gc)
            b_try = _autocorr(a_try, fs_b)
            lv, _ = _lse_w(b_try, beta)
            if lv <= lse_val - 1e-4 * alpha * gn:
                step_found = True
                break
            alpha *= 0.5
        
        if not step_found:
            if stall > 50:
                a = _make_smooth_rand(n, rng, best_a, 0.4)
                beta = max(5.0, beta * 0.5)
                alpha = 0.2
                stall = 0
            else:
                a = a_try
        else:
            a = a_try
        
        # Adaptive beta
        if it % 80 == 0:
            beta = min(1000.0, beta * 1.15)
        
        # Reset alpha floor
        if alpha < 0.002:
            alpha = 0.15
        
        # Major restart
        if stall > 800 and time.monotonic() < deadline - 3:
            a = _make_smooth_rand(n, rng, best_a, 0.5)
            beta = max(8.0, beta0)
            alpha = 0.25
            stall = 0
        
        it += 1
        
        if verbose and it % 5000 == 0:
            print(f"  fft_refine: it={it} beta={beta:.1f} score={best_score:.8f}", flush=True)
    
    return best_a, best_score

Search for minimum-peak nonnegative self-convolutions.

Key improvements over previous approach:
1. TRUE Euclidean projection to the probability simplex (sort-based) instead of
   soft clip+normalize, giving better descent directions.
2. Backtracking line search for adaptive step sizes.

import numpy as np
import time
from scipy.optimize import minimize as _scipy_min

def _next_pow2(size):
    """Smallest power of 2 >= size."""
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s

def _autocorr(a, fs=None):
    """Full self-convolution via zero-padded FFT."""
    n = a.size
    if fs is None:
        fs = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:2 * n - 1]

def _score(a):
    """Exact score: 2*n*max(a*a) assuming sum(a)=1."""
    a = _normalize(a)
    n = a.size
    b = _autocorr(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb

def _lse_w(b, beta):
    """Return (LSE value, softmax weights) of b at temperature beta."""
    m = float(np.max(b))
    if beta <= 1.0:
        return m, np.ones(b.size, dtype=np.float64) / b.size
    arg = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s < 1e-300:
        return m, np.ones(b.size, dtype=np.float64) / b.size
    w = e / s
    return m + float(np.log(s)) / beta, w

def _grad_a(a, w, fs):
    """
    Gradient of LSE(b,beta) w.r.t. a: g_i = 2 * sum_j w_{j+i} * a_j.
    Computed as 2 * cross_corr(w, a) via FFT.
    """
    n = a.size
    ar = a[::-1]
    conv = np.fft.irfft(
        np.fft.rfft(w, fs) * np.fft.rfft(ar, fs), fs
    ).real
    return 2.0 * conv[n - 1:2 * n - 1]

def _proj_simplex(v):
    """
    True Euclidean projection onto {x >= 0, sum(x) = 1}.
    Uses the Athaydes-Moreira algorithm (sort-based, O(n log n)).
    """
    n = v.size
    if n <= 1:
        return np.ones(n, dtype=np.float64) / max(n, 1)
    v = np.asarray(v, dtype=np.float64).copy()
    vs = np.sort(v)[::-1]
    css = np.cumsum(vs)
    idx = np.arange(1, n + 1, dtype=np.float64)
    ok = vs - (css - 1.0) / idx > 0
    ii = np.where(ok)[0]
    if ii.size == 0:
        return np.ones(n, dtype=np.float64) / n
    k = ii[-1]
    theta = (css[k] - 1.0) / (k + 1.0)
    u = np.maximum(v - theta, 0.0)
    s = u.sum()
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return u / s

def _make_exp(n, alpha=1.0):
    """Exponentially decaying sequence."""
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-alpha * x))

def _make_bell(n, c=0.5, w=0.25):
    """Gaussian bell centered at c with width w."""
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-0.5 * ((x - c) / w) ** 2))

def _make_sym(n, alpha=1.0):
    """Symmetric Gaussian (peak at center)."""
    x = np.linspace(-1.0, 1.0, n)
    return _normalize(np.exp(-alpha * x * x))

def _make_arcsine(n):
    """Arcsine distribution (mass near edges)."""
    x = np.linspace(0.0, 1.0, n)
    x = np.clip(x, 1e-6, 1.0 - 1e-6)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    return _normalize(a)

def _make_uniform(n):
    """Uniform sequence."""
    return np.ones(n, dtype=np.float64) / n

def _make_smooth_rand(n, rng, base=None, amp=0.3):
    """Base modulated by smooth (bandlimited) noise."""
    if base is None:
        base = _make_uniform(n)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cut = max(2, n // 12 + 2)
    if cut < len(F):
        F[cut:] = 0.0
    f = np.fft.irfft(F, n)
    f -= f.mean()
    sd = f.std()
    if sd > 1e-15:
        f /= sd
    a = base * (1.0 + amp * f)
    return _normalize(np.maximum(a, 1e-12))

def _resample(old, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    old = _normalize(old)
    if old.size == new_n:
        return old.copy()
    xo = np.linspace(0.0, 1.0, old.size)
    xn = np.linspace(0.0, 1.0, new_n)
    return _normalize(np.interp(xn, xo, old))

def _peak_shave(a, rng, strength=0.15):
    """
    Informed perturbation: reduce elements that contribute most to the peak
    of the self-convolution, then redistribute.
    """
    n = a.size
    b = _autocorr(a, _next_pow2(2 * n - 1))
    a_new = a.copy()
    # Identify top peak positions
    top_k = np.argsort(b)[::-1][:max(3, n // 8)]
    for k in top_k[:5]:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # Contributions: a[lo:hi] * a[k-lo : k-hi+n]
        idx_first = np.arange(lo, hi)
        idx_second = k - idx_first
        valid = (idx_second >= 0) & (idx_second < n)
        if not valid.any():
            continue
        contrib = a[idx_first[valid]] * a[idx_second[valid]]
        cmax = contrib.max()
        if cmax > 1e-30:
            a_new[idx_first[valid]] *= (1.0 - strength * contrib / cmax)
    # Add small random component to avoid symmetry
    a_new += 1e-6 * rng.random(n) * a.max()
    return _normalize(a_new)

def _fft_refine(a, deadline, rng, beta0=8.0, verbose=False):
    """
    LSE gradient descent with TRUE simplex projection and backtracking line search.
    This is the core optimization loop.
    """
    a = _normalize(a)
    n = a.size
    fs_b = _next_pow2(2 * n - 1)
    fs_g = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = _score(a)
    beta = beta0
    alpha = 0.2
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        b = _autocorr(a, fs_b)
        sc = 2.0 * n * float(np.max(b))

        if sc < best_score - 1e-13:
            best_a, best_score = a.copy(), sc
            stall = 0
        else:
            stall += 1

        lse_val, w = _lse_w(b, beta)
        g = _grad_a(a, w, fs_g)
        gc = g - g.mean()
        gn = float(np.dot(gc, gc))

        if gn < 1e-28:
            # Gradient vanishing: informed restart
            a = _peak_shave(best_a, rng, 0.2)
            beta = max(5.0, beta * 0.5)
            alpha = 0.2
            stall = 0
            it += 1
            continue

        # Backtracking line search (Armijo condition)
        step_ok = False
        a_new = a
        a_test = a
        for _ls in range(18):
            a_test = _proj_simplex(a - alpha * gc)
            b_test = _autocorr(a_test, fs_b)
            lv, _ = _lse_w(b_test, beta)
            if lv <= lse_val - 1e-4 * alpha * gn:
                step_ok = True
                a_new = a_test
                break
            alpha *= 0.5
        if not step_ok:
            if stall > 80:
                # Informed restart: peak shave + random
                if rng.random() < 0.5:
                    a = _peak_shave(best_a, rng, 0.25)
                else:
                    a = _make_smooth_rand(n, rng, best_a, 0.4)
                beta = max(5.0, beta0)
                alpha = 0.2
                stall = 0
            else:
                a_new = a_test
        a = a_new

        # Adaptive beta: increase to sharpen LSE focus
        if it % 60 == 0:
            beta = min(2000.0, beta * 1.12)

        # Alpha floor
        if alpha < 0.002:
            alpha = 0.15

        # Major restart when badly stuck
        if stall > 1200 and time.monotonic() < deadline - 3.0:
            r = rng.random()
            if r < 0.3:
                a = _peak_shave(best_a, rng, 0.3)
            elif r < 0.6:
                a = _make_smooth_rand(n, rng, best_a, 0.5)
            elif r < 0.8:
                a = _make_exp(n, float(rng.uniform(0.3, 3.0)))
            else:
                a = _make_bell(n, float(rng.uniform(0.3, 0.7)), 0.2)
            beta = beta0
            alpha = 0.25
            stall = 0

        it += 1
        if verbose and it % 3000 == 0:
            print(f"    fft_ref n={n} it={it} beta={beta:.0f} best={best_score:.8f}", flush=True)

    return best_a, best_score

def _lbfgs_refine(a, timeout, beta=25.0, maxiter=300):
    """
    Fine-tune using scipy L-BFGS-B on softmax parameterization.
    x -> a = softmax(x), objective = LSE(a*a, beta).
    """
    a = _normalize(a)
    n = a.size
    fs_b = _next_pow2(2 * n - 1)
    fs_g = _next_pow2(3 * n - 2)
    t_start = time.monotonic()
    t_end = t_start + min(timeout, 90.0)

    x0 = np.log(np.maximum(a, 1e-12))

    def obj_grad(x):
        if time.monotonic() > t_end:
            return float('inf'), np.zeros_like(x)
        am = x - x.max()
        e = np.exp(am)
        a_s = e / e.sum()
        b = _autocorr(a_s, fs_b)
        m = float(np.max(b))
        arg = np.clip(beta * (b - m), -745.0, 0.0)
        ev = np.exp(arg)
        s = float(ev.sum())
        if s < 1e-300:
            w = np.ones(b.size) / b.size
            val = m
        else:
            w = ev / s
            val = m + float(np.log(s)) / beta
        ga = _grad_a(a_s, w, fs_g)
        c = float(np.dot(ga, a_s))
        gx = a_s * (ga - c)
        return val, gx

    try:
        res = _scipy_min(
            obj_grad, x0, jac=True, method='L-BFGS-B',
            options={'maxiter': maxiter, 'gtol': 1e-10, 'maxfun': maxiter * 3}
        )
    except Exception:
        return a, _score(a)

    a_opt = _normalize(np.exp(res.x - res.x.max()))
    sc = _score(a_opt)
    return a_opt, sc

def _get_given():
    """Retrieve the provided starting sequence."""
    seq = globals().get('height_sequence_1', None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or arr.sum() < 1e-300:
        return None
    return arr / arr.sum()
