"""
Search for nonnegative sequences minimizing peak self-convolution score.
Combines MPA (error reduction /Gerchberg-Saxton style) with LSE gradient descent,
multi-length exploration, and peak equalization heuristics.
Score = 2*n*max(convolve(a,a)) / (sum(a))^2, scale-invariant.
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _convolve_fft(a, N):
    """Linear convolution a*a via zero-padded FFT."""
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * len(a) - 1]


def _score(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    N = _next_pow2(2 * n - 1)
    c = _convolve_fft(a, N)
    mb = float(np.max(c))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * n * mb


def _autocorr_full(a, N):
    """Return full circular convolution (length N) of zero-padded a with itself."""
    A = np.fft.fft(a, N)
    C = A * A
    c = np.fft.ifft(C).real
    return c


def _mpa_step(a, alpha, N, num_peaks=5):
    """
    One MPA (Minimum Peak Amplitude) step.
    Modifies the convolution to reduce peaks, then inverts to get new a.
    """
    n = a.size
    if N is None:
        N = _next_pow2(2 * n - 1)

    # Compute current convolution
    A = np.fft.fft(a, N)
    c = np.fft.ifft(A * A).real

    # Work on the linear convolution part
    c_lin = c[:2 * n - 1].copy()
    c_max = float(np.max(c_lin))
    c_mean = float(np.mean(c_lin))

    if c_max <= 1e-300:
        return a.copy()

    # Find top peaks and target level
    sorted_idx = np.argsort(c_lin)[::-1]
    k = min(num_peaks, len(c_lin))
    top_peaks = c_lin[sorted_idx[:k]]
    target_level = float(np.mean(top_peaks)) * (1.0 - alpha * 0.5)

    # Reduce values above target
    excess = np.maximum(0.0, c_lin - target_level)
    c_lin_new = c_lin - alpha * excess

    # Rebuild full circular convolution
    c_new = np.zeros(N, dtype=np.float64)
    c_new[:2 * n - 1] = c_lin_new

    # Compute new A via square root
    C_new = np.fft.fft(c_new)
    A_new = np.sqrt(C_new.astype(np.complex128))

    # Get new a
    a_new = np.fft.ifft(A_new)[:n].real
    a_new = np.maximum(a_new, 0.0)
    return _normalize(a_new)


def _lse_and_p(c, beta):
    """Return softmax weights and log-sum-exp value."""
    if c.size == 0:
        return np.ones(1), 0.0
    m = float(np.max(c))
    if beta <= 1.0:
        p = np.full(c.size, 1.0 / c.size, dtype=np.float64)
        return p, m
    arg = beta * (c - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(c.size, dtype=np.float64)
        s = float(c.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _weighted_gradient(a, weights, N, n):
    """
    Gradient of sum_k w_k * c_k w.r.t. a_i.
    c = convolve(a, a), dc_k/da_i = a[k-i] + a[i-k] (symmetric).
    Result = 2 * convolve(a, weights)[n-1 : 2n-1]
    """
    ar = a[::-1]
    A = np.fft.rfft(ar, N)
    W = np.fft.rfft(weights, N)
    conv = np.fft.irfft(A * W, N).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    """Clip to nonnegative and normalize to sum 1."""
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    n = w.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return w / s


def _make_exponential(n, alpha):
    x = np.linspace(0.0, 1.0, max(int(n), 2))
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_cosine_window(n):
    x = np.linspace(0.0, np.pi, max(int(n), 2))
    a = np.abs(np.cos(x / 2.0))
    return _normalize(a)


def _make_two_bumps(n, ratio=0.6):
    """Sequence with two unequal segments."""
    n = max(int(n), 4)
    split = int(n * ratio)
    a = np.zeros(n, dtype=np.float64)
    a[:split] = 1.0 / split
    a[split:] = 1.0 / (n - split)
    return _normalize(a)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.maximum(np.nan_to_num(old, nan=0.0), 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n) / new_n
    s = old.sum()
    if s <= 1e-300:
        return np.ones(new_n) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb(a, rng, strength=0.15):
    """Random multiplicative perturbation."""
    noise = np.exp(rng.normal(0.0, strength, size=a.size))
    a2 = a * noise
    return _normalize(a2)


def _block_perturb(a, rng, strength=0.25):
    n = a.size
    c = a.copy()
    width = max(2, n // 16)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)


def _optimize_lse(a, deadline, rng, max_iters=5000):
    """LSE gradient descent on the simplex."""
    a = _normalize(a)
    n = a.size
    N_conv = _next_pow2(2 * n - 1)
    N_grad = _next_pow2(2 * n - 1)

    c = _convolve_fft(a, N_conv)
    maxb = float(np.max(c))
    score = 2.0 * n * maxb
    best_a = a.copy()
    best_score = score

    alpha = 0.1
    beta = 8.0
    it = 0
    fail = 0
    last_improve = 0

    # Adam-like momentum
    m Adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.85, 0.95, 1e-8
    lr_base = 0.3

    while time.monotonic() < deadline and it < max_iters:
        c = _convolve_fft(a, N_conv)
        maxb = float(np.max(c))
        if maxb <= 1e-300:
            maxb = 1e-300

        # Adaptive beta
        beta = max(2.0, min(200.0, 16.0 / maxb))

        p, smooth_old = _lse_and_p(c, beta)
        grad = _weighted_gradient(a, p, N_grad, n)
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale <= 1e-300:
            a = _perturb(best_a, rng, 0.12)
            it += 1
            continue

        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad_c
        v_adam = beta2 * v_adam + (1 - beta2) * grad_c ** 2
        m_hat = m_adam / (1 - beta1 ** (it + 1))
        v_hat = v_adam / (1 - beta2 ** (it + 1))
        update = m_hat / (np.sqrt(v_hat) + eps)
        update_norm = float(np.max(np.abs(update)))
        if update_norm > 0:
            update = update / update_norm

        a_new = a - lr_base * alpha * update
        a_new = _project_simplex(a_new)

        c_new = _convolve_fft(a_new, N_conv)
        maxb_new = float(np.max(c_new))
        score_new = 2.0 * n * maxb_new
        smooth_new_val = _lse_and_p(c_new, beta)[1]

        if smooth_new_val < smooth_old - 1e-14:
            alpha = min(1.0, alpha * 1.1)
            fail = 0
            last_improve = it
            a = a_new
        else:
            alpha *= 0.7
            fail += 1

        if score_new < best_score - 1e-15:
            best_a = a_new.copy()
            best_score = score_new

        # Restart if stuck
        if (fail > 200 or (it - last_improve > 500 and it > 100)) and time.monotonic() < deadline - 2.0:
            a = _block_perturb(best_a, rng, 0.2)
            alpha = 0.15
            fail = 0
            m_adam = np.zeros(n, dtype=np.float64)
            v_adam = np.zeros(n, dtype=np.float64)

        it += 1

    return best_a, best_score


def _optimize_mpa(a, deadline, rng, max_iters=2000):
    """MPA (error reduction) optimization."""
    a = _normalize(a)
    n = a.size
    N = _next_pow2(2 * n - 1)

    c = _convolve_fft(a, N)
    score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    best_score = score

    alpha = 0.3
    it = 0
    fail = 0

    while time.monotonic() < deadline and it < max_iters:
        a_new = _mpa_step(a, alpha, N, num_peaks=5)
        c_new = _convolve_fft(a_new, N)
        score_new = 2.0 * n * float(np.max(c_new))

        if score_new < best_score - 1e-15:
            best_a = a_new.copy()
            best_score = score_new
            a = a_new
            alpha = min(0.8, alpha * 1.05)
            fail = 0
        else:
            # Try smaller alpha
            alpha = max(0.02, alpha * 0.7)
            fail += 1
            if fail > 50:
                # Perturb and restart
                a = _perturb(best_a, rng, 0.1)
                alpha = 0.3
                fail = 0

        it += 1

    return best_a, best_score


def _optimize_combined(a, deadline, rng):
    """Alternate between LSE and MPA."""
    a = _normalize(a)
    n = a.size
    c = _convolve_fft(a, _next_pow2(2 * n - 1))
    score = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    best_score = score

    phase = 0
    segment_time = min(30.0, max(5.0, (deadline - time.monotonic()) / 6.0))

    while time.monotonic() < deadline - 2.0:
        seg_end = min(deadline - 1.0, time.monotonic() + segment_time)

        if phase % 3 == 0:
            res_a, res_score = _optimize_lse(a, seg_end, rng, max_iters=3000)
        elif phase % 3 == 1:
            res_a, res_score = _optimize_mpa(a, seg_end, rng, max_iters=1000)
        else:
            # Mix: LSE then small MPA polish
            half = time.monotonic() + (seg_end - time.monotonic()) * 0.7
            res_a, _ = _optimize_lse(a, half, rng, max_iters=1000)
            res_a, res_score = _optimize_mpa(res_a, seg_end, rng, max_iters=300)

        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)
            a = best_a.copy()
        else:
            a = best_a.copy()

        phase += 1
        # Gradually increase segment time
        segment_time = min(60.0, segment_time * 1.15)

    return best_a, best_score


def _get_initial_sequence():
    """Try to get the provided sequence."""
    seq = globals().get("height_sequence_1", None)
    if seq is not None:
        try:
            arr = np.asarray(seq, dtype=np.float64).ravel()
            arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if arr.size > 4 and s > 1e-10:
                return arr / s
        except Exception:
            pass
    return None


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for the best nonnegative coefficient sequence
    minimizing 2*n*max(convolve(a,a))/(sum(a))^2 within the time budget.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + max(budget, 60.0) - 20.0
    rng = np.random.default_rng(seed)

    print(f"Budget: {budget:.0f}s, deadline in {max(budget,60)-20:.0f}s", flush=True)

    # Get initial sequence
    given = _get_initial_sequence()
    if given is not None:
        best_a = given.copy()
        best_score = _score(best_a)
        print(f"Given sequence: n={best_a.size}, score={best_score:.8f}", flush=True)
    else:
        best_a = _make_exponential(512, 1.0)
        best_score = _score(best_a)
        print(f"No given sequence, using exponential: n={best_a.size}, score={best_score:.8f}", flush=True)

    # Phase 1: Try different initializations at small lengths
    small_lengths = [128, 256, 512]
    for n_test in small_lengths:
        if time.monotonic() > deadline - 60:
            break
        inits = []
        inits.append(_make_exponential(n_test, 1.2))
        inits.append(_make_cosine_window(n_test))
        inits.append(_make_two_bumps(n_test, 0.6))
        inits.append(_make_two_bumps(n_test, 0.55))
        inits.append(_make_exponential(n_test, 0.8))

        if given is not None:
            inits.append(_resample(given, n_test))

        t_per = max(5, min(15, (deadline - time.monotonic()) / (len(inits) * 3)))
        for init in inits:
            if time.monotonic() > deadline - 30:
                break
            t_end = time.monotonic() + t_per
            res_a, res_score = _optimize_lse(init.copy(), t_end, rng, max_iters=2000)
            if res_score < best_score:
                best_a = res_a.copy()
                best_score = res_score
                print(f"  Small n={n_test}: improved to {best_score:.8f}", flush=True)

    print(f"After Phase 1: best_score={best_score:.8f} (n={best_a.size})", flush=True)

    # Phase 2: Multi-length combined optimization
    # Try increasing lengths, always starting from current best
    lengths_to_try = [512, 1024, 2048, 4096, 8192]

    for n_target in lengths_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Determine time allocation for this length
        if n_target <= 1024:
            time_alloc = min(remaining * 0.35, 200)
        elif n_target <= 2048:
            time_alloc = min(remaining * 0.3, 250)
        elif n_target <= 4096:
            time_alloc = min(remaining * 0.25, 300)
        else:
            time_alloc = min(remaining * 0.2, 350)

        time_alloc = max(15, time_alloc)
        t_end = time.monotonic() + time_alloc

        # Create initial for this length
        init = _resample(best_a, n_target)
        # Mix with exponential to avoid degenerate resampling
        exp_base = _make_exponential(n_target, 1.0)
        init = _normalize(0.8 * init + 0.2 * exp_base)

        # Run combined optimization
        res_a, res_score = _optimize_combined(init, t_end, rng)

        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"  n={n_target}: improved to {best_score:.8f}", flush=True)
        else:
            print(f"  n={n_target}: no improvement (got {res_score:.8f}, best {best_score:.8f})", flush=True)

    # Phase 3: Try MPA specifically on the best sequence
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_cur = best_a.size
        # Also try resampling to a few other lengths
        for n_try in [int(n_cur * 1.5), int(n_cur * 2), max(512, int(n_cur * 0.75))]:
            if time.monotonic() > deadline - 20:
                break
            n_try = min(16384, max(256, n_try))
            if n_try == n_cur:
                continue
            init = _resample(best_a, n_try)
            t_end = min(deadline - 15, time.monotonic() + min(60, remaining * 0.2))
            res_a, res_score = _optimize_mpa(init, t_end, rng)
            if res_score < best_score:
                best_a = res_a.copy()
                best_score = res_score
                print(f"  MPA n={n_try}: improved to {best_score:.8f}", flush=True)

    # Phase 4: Final LSE refinement of the best
    remaining = deadline - time.monotonic()
    if remaining > 10:
        t_end = deadline - 2.0
        res_a, res_score = _optimize_lse(best_a.copy(), t_end, rng, max_iters=10000)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"Final LSE: improved to {best_score:.8f}", flush=True)
        else:
            print(f"Final LSE: best remains {best_score:.8f}", flush=True)

    # Output
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)

    final_score = _score(out)
    print(f"Final result: n={out.size}, score={final_score:.8f}", flush=True)

    return [float(x) for x in out]
