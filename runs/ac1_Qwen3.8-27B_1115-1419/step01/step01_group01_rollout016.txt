"""
Search for nonnegative step-function heights minimizing

    2 * n * max(convolve(a, a)) / sum(a)**2.

This implementation uses:
  - FFT-based scoring,
  - low-dimensional parametric band-limited density initialization,
  - direct smooth-max Adam optimization in log/softmax space,
  - spectral square-root seeds from flattened/self-shrunk current coefficient
    profiles,
  - staged length extension (4096, then 8192) with final polishing.
"""

import time
import math
import numpy as np

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj_nonneg_sum(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a.fill(1.0 / float(a.size))
    else:
        a /= s
    return a


def _soft(v):
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.clip(v, -25.0, 25.0)
    v -= float(np.max(v))
    e = np.exp(v)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(v.shape, dtype=np.float64) / max(1, v.size)
    return e / s


def _logplus(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 1e-300, out=a)
    return np.log(a)


def _conv_b(a, m):
    n = int(a.size)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.where(np.isfinite(b), b, 0.0)
    np.maximum(b, 0.0, out=b)
    return b, F


def _score_norm_m(a, m):
    a = _proj_nonneg_sum(a)
    n = int(a.size)
    if n == 0:
        return float("inf")
    b, _ = _conv_b(a, m)
    bm = float(b.max()) if b.size else float("inf")
    if not np.isfinite(bm) or bm <= 0.0:
        return float("inf")
    return float(2.0 * n * bm)


def _score_fft(a):
    a = _proj_nonneg_sum(a)
    n = int(a.size)
    if n == 0:
        return float("inf")
    return _score_norm_m(a, _next_pow2(2 * n - 1))


def _interp(a, n):
    a = _proj_nonneg_sum(a)
    n = int(max(1, n))
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return _proj_nonneg_sum(out)


def _pair_score(pair):
    return pair[0]


def _param_sequence(x, basis):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = int(basis[0].size)
    v = np.zeros(n, dtype=np.float64)
    half = len(basis) // 2
    for j in range(min(half, x.size // 2)):
        v += x[2 * j] * basis[2 * j]
        v += x[2 * j + 1] * basis[2 * j + 1]
    # Slightly bound the log-density to keep exp stable and avoid saturation.
    v = 2.5 * np.tanh(v / 2.5)
    v -= float(np.mean(v))
    a = np.exp(np.clip(v, -25.0, 25.0))
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _param_objective(x, basis, m):
    a = _param_sequence(x, basis)
    sc = _score_norm_m(a, m)
    return sc


def _score_arrays(arrays, cap, thresh):
    raw = []
    for a in arrays:
        if a is None:
            continue
        an = _proj_nonneg_sum(a)
        if an.size == 0:
            continue
        sc = _score_fft(an)
        if np.isfinite(sc):
            raw.append((sc, an))
    if not raw:
        return []
    raw.sort(key=_pair_score)

    kept = []
    seen = []
    used = set()

    for idx, (sc, a) in enumerate(raw):
        diverse = True
        for b in seen:
            d = float(np.sqrt(np.sum((a - b) * (a - b))))
            if d < thresh:
                diverse = False
                break
        if diverse:
            kept.append((sc, a))
            seen.append(a)
            used.add(idx)
            if len(kept) >= cap:
                return kept

    for idx, (sc, a) in enumerate(raw):
        if idx not in used:
            kept.append((sc, a))
            used.add(idx)
            if len(kept) >= cap:
                break
    return kept


def _param_search(n, wall, rng):
    if time.monotonic() >= wall - 20:
        return []

    n = int(max(256, min(1024, n)))
    freqs = [1, 2, 3, 4, 5, 6, 8, 10, 12]
    idx = np.arange(n, dtype=np.float64)
    basis = []
    for q in freqs:
        ang = 2.0 * math.pi * q * idx / float(n)
        basis.append(np.cos(ang))
        basis.append(np.sin(ang))

    m = _next_pow2(2 * n - 1)
    p = 2 * len(freqs)
    res = []

    starts = [np.zeros(p, dtype=np.float64)]
    for _ in range(4):
        starts.append(rng.normal(0.0, 0.85, size=p).astype(np.float64))

    # Strong tilted-sine starts, which are useful for reducing the middle
    # convolution coefficient.
    x0 = np.zeros(p, dtype=np.float64)
    x0[1] = 0.95
    x0[3] = 0.35
    x0[5] = -0.45
    starts.append(x0)

    x0 = np.zeros(p, dtype=np.float64)
    x0[3] = 0.80
    x0[7] = 0.30
    x0[9] = -0.25
    starts.append(x0)

    for x in starts:
        a = _param_sequence(x, basis)
        sc = _score_norm_m(a, m)
        if np.isfinite(sc):
            res.append((sc, a))

    if _scipy_minimize is not None:
        for st in starts[:4]:
            if time.monotonic() >= wall - 15:
                break
            try:
                opt = _scipy_minimize(
                    _param_objective,
                    st,
                    args=(basis, m),
                    method="Nelder-Mead",
                    options={"maxfev": 450, "xatol": 0.05, "fatol": 1e-7},
                )
                a = _param_sequence(opt.x, basis)
                sc = _score_norm_m(a, m)
                if np.isfinite(sc):
                    res.append((sc, a))
            except Exception:
                pass
    else:
        for _ in range(500):
            if time.monotonic() >= wall - 5:
                break
            x = rng.normal(0.0, 1.0, size=p).astype(np.float64)
            a = _param_sequence(x, basis)
            sc = _score_norm_m(a, m)
            if np.isfinite(sc):
                res.append((sc, a))

    if not res:
        return []

    res.sort(key=_pair_score)
    out = []
    seen = []
    for sc, a in res:
        diverse = True
        for b in seen:
            if float(np.sqrt(np.sum((a - b) * (a - b)))) < 0.05:
                diverse = False
                break
        if diverse:
            out.append(a)
            seen.append(a)
            if len(out) >= 5:
                break
    return out


def _random_seeds(n, rng, max_count=20):
    out = []
    n = int(max(16, n))
    t = np.arange(n, dtype=np.float64) / float(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    j = n + 1.0 - i

    # Tilted sine starts: often move away from the symmetric stationary point.
    for alpha in (0.40, 0.65, 0.85, 1.00):
        a = 1.0 + alpha * np.sin(2.0 * math.pi * t)
        out.append(np.maximum(a, 1e-8))

    for _ in range(8):
        a = np.ones(n, dtype=np.float64)
        q1 = int(rng.integers(1, 9))
        a += rng.uniform(-1.35, 1.35) * np.sin(
            2.0 * math.pi * q1 * t + rng.uniform(0.0, 2.0 * math.pi)
        )
        if rng.random() < 0.75:
            q2 = int(rng.integers(1, 9))
            a += rng.uniform(-0.95, 0.95) * np.sin(
                2.0 * math.pi * q2 * t + rng.uniform(0.0, 2.0 * math.pi)
            )
        if rng.random() < 0.50:
            q3 = int(rng.integers(1, 9))
            a += rng.uniform(-0.55, 0.55) * np.cos(
                2.0 * math.pi * q3 * t + rng.uniform(0.0, 2.0 * math.pi)
            )
        out.append(np.maximum(a, 1e-8))

    # Smooth band-limited log shapes.
    for _ in range(7):
        v = np.zeros(n, dtype=np.float64)
        for q in range(1, 8):
            amp = rng.uniform(-1.25, 1.25) / (float(q) ** 0.65)
            ph = rng.uniform(0.0, 2.0 * math.pi)
            v += amp * np.sin(2.0 * math.pi * q * t + ph)
        out.append(np.exp(np.clip(v, -3.5, 3.5)))

    # End-weighted beta-like shapes.
    for _ in range(5):
        p = rng.uniform(-0.55, 1.30)
        q = rng.uniform(-0.55, 1.30)
        out.append(np.power(np.maximum(i, 1e-9), p) * np.power(np.maximum(j, 1e-9), q))

    # Exponentially skewed shapes.
    for _ in range(4):
        lam = rng.uniform(0.20, 3.00)
        mu = rng.uniform(0.20, 3.00)
        w = rng.uniform(0.00, 2.00)
        out.append(np.exp(-lam * t) + w * np.exp(-mu * (1.0 - t)))

    if len(out) > max_count:
        out = out[:max_count]
    return out


def _spectral_sqrt(target, a_init, n, m, iters=30, blend=0.55):
    n = int(max(16, n))
    L = 2 * n - 1
    target = np.asarray(target, dtype=np.float64).ravel()
    z = np.zeros(m, dtype=np.float64)
    pad = min(L, target.size, m)
    if pad > 0:
        z[:pad] = target[:pad]
    z = np.where(np.isfinite(z), z, 0.0)
    np.maximum(z, 1e-12, out=z)
    zm = float(z.max())
    if zm <= 0.0 or not np.isfinite(zm):
        z.fill(1.0)
    else:
        z /= zm

    T = np.fft.fft(z)

    best_a = None
    best_sc = float("inf")

    # Direct principal square root candidate.
    sqrtT = np.sqrt(T)
    cand = np.fft.ifft(sqrtT).real[:n]
    cand = np.where(np.isfinite(cand), cand, 0.0)
    np.maximum(cand, 0.0, out=cand)
    cand = _proj_nonneg_sum(cand)
    sc = _score_norm_m(cand, m)
    if np.isfinite(sc):
        best_a = cand.copy()
        best_sc = sc

    a = _proj_nonneg_sum(a_init)
    if best_a is not None:
        a2 = 0.50 * a + 0.50 * best_a
        a2 = _proj_nonneg_sum(a2)
        sc2 = _score_norm_m(a2, m)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
        a = a2

    F = np.fft.fft(a, m)

    for it in range(max(1, int(iters))):
        if not np.all(np.isfinite(F)):
            break

        scale = float(np.max(np.abs(F)))
        if not np.isfinite(scale) or scale <= 0.0:
            scale = 1.0
        eps = 1e-10 * scale
        denom = F.copy()
        small = np.abs(denom) < eps
        if np.any(small):
            denom[small] = eps + 0.0j

        Fnew = 0.5 * (F + T / denom)
        anew = np.fft.ifft(Fnew).real[:n]
        if not np.all(np.isfinite(anew)):
            break

        pos = np.maximum(anew, 0.0)
        ps = float(pos.sum())
        if not np.isfinite(ps) or ps < 1e-8 * max(1.0, float(a.sum())):
            pos = a.copy()

        if it % 10 == 0 and best_a is not None:
            pos = 0.75 * pos + 0.25 * best_a

        a_new = (1.0 - blend) * a + blend * pos
        a_new = _proj_nonneg_sum(a_new)

        if (it + 1) % 5 == 0:
            sc = _score_norm_m(a_new, m)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()

        a = a_new
        F = np.fft.fft(a, m)

    sc = _score_norm_m(a, m)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    if best_a is None:
        best_a = _proj_nonneg_sum(a)
    return best_a, float(best_sc)


def _spectral_seeds(a, n, m, rng, count=2, iters=None):
    a = _proj_nonneg_sum(a)
    n = int(a.size)
    b, _ = _conv_b(a, m)
    bm = float(b.max())
    if b.size == 0 or bm <= 0.0 or not np.isfinite(bm):
        return []

    r = b / bm
    L = int(b.size)
    tt = np.arange(L, dtype=np.float64) / float(max(1, L - 1))
    edge = np.abs(tt - 0.5) * 2.0

    if iters is None:
        iters = 22 if n >= 8192 else 32

    out = []
    for _ in range(int(max(1, count))):
        eta = float(rng.uniform(0.05, 0.80))
        shape = (1.0 - eta) * r + eta

        mode = rng.random()
        amp = float(rng.uniform(-0.65, 0.85))
        if mode < 0.50:
            # Raise edges / lower middle.
            shape = shape * (1.0 - amp * (1.0 - edge))
        else:
            # Low-frequency modulation in coefficient index.
            q = int(rng.integers(1, 5))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            shape = shape + amp * np.sin(2.0 * math.pi * q * tt + ph)

        shape = np.where(np.isfinite(shape), shape, 0.0)
        np.maximum(shape, 1e-12, out=shape)
        smax = float(shape.max())
        if smax <= 0.0 or not np.isfinite(smax):
            shape = np.ones_like(shape)
        else:
            shape /= smax

        try:
            a_new, sc = _spectral_sqrt(shape, a, n, m, iters=iters)
            if np.isfinite(sc):
                out.append(a_new)
        except Exception:
            pass

    return out


def _optimize_log(a0, seconds, rng):
    a = _proj_nonneg_sum(a0)
    n = int(a.size)
    if n <= 0:
        return a, float("inf")
    m = _next_pow2(2 * n - 1)
    seconds = float(max(5.0, seconds))
    start = time.monotonic()
    deadline = start + seconds

    v = _logplus(a)
    v = np.where(np.isfinite(v), v, 0.0)
    v -= float(np.mean(v))
    v = np.clip(v, -25.0, 25.0)
    a = _soft(v)

    b, F = _conv_b(a, m)
    best_bmax = float(b.max()) if b.size else float("inf")
    if not np.isfinite(best_bmax) or best_bmax <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
        v = _logplus(a)
        b, F = _conv_b(a, m)
        best_bmax = float(b.max())
    best_a = a.copy()
    last_improve = start

    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tau_frac = 0.18

    while time.monotonic() < deadline:
        now = time.monotonic()
        progress = min(1.0, max(0.0, (now - start) / max(1e-9, seconds)))
        lr = 0.045 + 0.105 * (1.0 - progress)
        target_tau = 0.035 + 0.120 * (1.0 - progress)
        tau_frac = 0.88 * tau_frac + 0.12 * target_tau

        # If stuck long enough, perturb log variables and reset Adam state.
        stuck_limit = max(45.0, 0.12 * seconds)
        if now - last_improve > stuck_limit:
            sigma = 0.20 + 0.55 * (1.0 - progress)
            v = v + rng.normal(0.0, sigma, n)
            v = np.where(np.isfinite(v), v, 0.0)
            v -= float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            mu.fill(0.0)
            nu.fill(0.0)
            t = 0
            last_improve = now
            continue

        a = _soft(v)
        b, F = _conv_b(a, m)
        bmax = float(b.max()) if b.size else float("inf")
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break

        if bmax < best_bmax - 1e-14:
            best_bmax = bmax
            best_a = a.copy()
            last_improve = now

        # If badly outside the current best, pull the iterates back.
        if bmax > best_bmax * 1.25:
            v2 = _logplus(best_a)
            v = 0.55 * v + 0.45 * v2
            v = np.where(np.isfinite(v), v, 0.0)
            v -= float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            mu.fill(0.0)
            nu.fill(0.0)
            t = 0
            last_improve = now
            continue

        tau_noise = float(rng.lognormal(0.0, 0.25))
        tau = max(1e-15, bmax * max(0.012, tau_frac) * tau_noise)

        x = (b - bmax) / tau
        x = np.clip(x, -60.0, 0.0)
        w = np.exp(x)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.full(b.size, 1.0 / max(1, b.size), dtype=np.float64)
        else:
            w /= ws

        soft_maxw = float(w.max())

        # Add a small floor and extra weight on near-active lags. This avoids
        # the exact-max saddle and helpfully broadens the active set.
        w += 1e-6
        active = b > (bmax - 2.0 * tau)
        cnt = int(np.count_nonzero(active))
        if cnt > 0:
            w[active] += 0.25
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.full(b.size, 1.0 / max(1, b.size), dtype=np.float64)
        else:
            w /= ws

        # Adapt smoothing based on how concentrated the smooth max was.
        if soft_maxw < 0.07:
            tau_frac = max(0.015, 0.85 * tau_frac)
        elif soft_maxw > 0.82:
            tau_frac = min(0.45, 1.15 * tau_frac)

        P = np.fft.fft(w, m)
        c = 2.0 * np.fft.ifft(np.conj(F) * P).real[:n]
        if not np.all(np.isfinite(c)):
            v = v + rng.normal(0.0, 0.15 + 0.4 * (1.0 - progress), n)
            v = np.where(np.isfinite(v), v, 0.0)
            v -= float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            mu.fill(0.0)
            nu.fill(0.0)
            t = 0
            continue

        cm = float(np.dot(a, c))
        gv = a * (c - cm)
        if not np.all(np.isfinite(gv)):
            continue
        grms = float(np.sqrt(np.mean(gv * gv)))
        if grms <= 1e-14 or not np.isfinite(grms):
            v = v + rng.normal(0.0, 0.15 + 0.4 * (1.0 - progress), n)
            v = np.where(np.isfinite(v), v, 0.0)
            v -= float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            mu.fill(0.0)
            nu.fill(0.0)
            t = 0
            continue

        gv /= grms
        t += 1
        mu = beta1 * mu + (1.0 - beta1) * gv
        nu = beta2 * nu + (1.0 - beta2) * (gv * gv)

        b1 = beta1 ** t
        b2 = beta2 ** t
        denom1 = 1.0 - b1 if b1 < 1.0 - 1e-16 else 1.0
        denom2 = 1.0 - b2 if b2 < 1.0 - 1e-16 else 1.0
        mhat = mu / denom1
        vhat = nu / denom2
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.where(np.isfinite(step), step, 0.0)
        step = np.clip(step, -0.5, 0.5)

        v = v - lr * step
        v = np.where(np.isfinite(v), v, 0.0)
        v -= float(np.mean(v))
        v = np.clip(v, -25.0, 25.0)

    a_final = _soft(v)
    b, _ = _conv_b(a_final, m)
    bm = float(b.max()) if b.size else float("inf")
    if np.isfinite(bm) and bm < best_bmax:
        best_bmax = bm
        best_a = a_final.copy()

    best_a = _proj_nonneg_sum(best_a)
    sc = _score_norm_m(best_a, m)
    if not np.isfinite(sc):
        sc = float(2.0 * n * best_bmax)
    return best_a, float(sc)


def _stage(n, wall, best_a_in, best_score_in, param_shapes, rng):
    n = int(max(128, n))
    wall = float(wall)
    start = time.monotonic()
    if wall <= start + 30:
        return best_a_in, best_score_in

    arrays = []
    if best_a_in is not None:
        arrays.append(_interp(best_a_in, n))
    arrays.append(np.ones(n, dtype=np.float64) / float(n))

    if param_shapes:
        for a in param_shapes[:4]:
            arrays.append(_interp(a, n))

    arrays.extend(_random_seeds(n, rng, max_count=18))

    scored = _score_arrays(arrays, cap=12, thresh=0.035)
    m = _next_pow2(2 * n - 1)

    # Spectral seeds from the best direct candidates.
    if time.monotonic() < wall - 60 and len(scored) > 0:
        extra = []
        limit = 3 if n <= 4096 else 2
        for pair in scored[:limit]:
            try:
                extra.extend(_spectral_seeds(pair[1], n, m, rng, count=2, iters=None))
            except Exception:
                pass
        if extra:
            all_arrays = []
            for pair in scored:
                all_arrays.append(pair[1])
            all_arrays.extend(extra)
            scored2 = _score_arrays(all_arrays, cap=14, thresh=0.035)
            if scored2:
                scored = scored2

    if not scored:
        scored = [(best_score_in, _interp(best_a_in, n) if best_a_in is not None
                   else (np.ones(n, dtype=np.float64) / float(n)))]

    # Run local optimization on several promising candidates.
    n_cand = 5 if n <= 4096 else 4
    if time.monotonic() < wall - 900:
        n_cand = min(len(scored), n_cand + 1)
    candidates = scored[: max(1, n_cand)]

    for idx, (sc0, a0) in enumerate(candidates):
        now = time.monotonic()
        remaining_cands = max(1, len(candidates) - idx)
        limit = wall - now - 12.0
        if limit < 25.0:
            break
        alloc = (wall - now) / float(remaining_cands)
        alloc = min(alloc * 1.25, limit)
        alloc = max(25.0, alloc)

        try:
            a_opt, sc_opt = _optimize_log(a0, alloc, rng)
            if np.isfinite(sc_opt) and sc_opt < best_score_in:
                best_score_in = float(sc_opt)
                best_a_in = _proj_nonneg_sum(a_opt)
                print("      stage n=%d cand %d score=%.6f" % (n, idx, best_score_in))
        except Exception:
            pass

    # Final polish on the current best at this length.
    now = time.monotonic()
    if best_a_in is not None and wall - now > 110:
        if len(best_a_in) == n:
            a_best = best_a_in.copy()
        else:
            a_best = _interp(best_a_in, n)
        sc_best = _score_fft(a_best)
        if np.isfinite(sc_best) and sc_best < best_score_in:
            best_score_in = float(sc_best)
            best_a_in = _proj_nonneg_sum(a_best)

        alloc = max(60.0, wall - now - 35.0)
        try:
            a_opt, sc_opt = _optimize_log(a_best, alloc, rng)
            if np.isfinite(sc_opt) and sc_opt < best_score_in:
                best_score_in = float(sc_opt)
                best_a_in = _proj_nonneg_sum(a_opt)
                print("      stage n=%d polish score=%.6f" % (n, best_score_in))
        except Exception:
            pass

    return best_a_in, float(best_score_in)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    seed = int(seed)
    budget = float(max(60.0, float(budget_s)))
    start = time.monotonic()
    deadline = start + max(60.0, budget - 45.0)
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float("inf")

    # Use the available previous construction if present.
    try:
        seq = height_sequence_1
        arr = np.asarray(seq, dtype=np.float64).ravel()
        if arr.size > 0:
            arr = np.where(np.isfinite(arr), arr, 0.0)
            np.maximum(arr, 0.0, out=arr)
            s = float(arr.sum())
            if s > 1e-12:
                best_a = arr / s
                best_score = _score_fft(best_a)
    except Exception:
        best_a = None

    if best_a is None or not np.isfinite(best_score):
        best_a = np.ones(512, dtype=np.float64) / 512.0
        best_score = _score_fft(best_a)

    print("Seed length=%d score=%.6f" % (len(best_a), best_score))

    # Band-limited parametric search: gives strong starts away from the
    # symmetric stationary point.
    param_shapes = []
    try:
        if time.monotonic() < deadline - 240:
            p_wall = min(deadline, time.monotonic() + 210.0)
            param_shapes = _param_search(1024, p_wall, rng)
            for a in param_shapes:
                sc = _score_fft(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_a = _proj_nonneg_sum(a)
            print("After parametric search best=%.6f" % best_score)
    except Exception as e:
        print("Parametric search failed:", e)

    try:
        # Stage 1: 4096.
        if time.monotonic() < deadline - 420:
            wall4 = min(deadline, time.monotonic() + max(300.0, (budget - 300.0) * 0.22))
            best_a, best_score = _stage(4096, wall4, best_a, best_score, param_shapes, rng)
            print("After 4096 best=%.6f len=%d" % (best_score, len(best_a)))
    except Exception as e:
        print("Stage 4096 failed:", e)

    try:
        # Stage 2: 8192.
        if time.monotonic() < deadline - 260:
            wall8 = min(deadline, time.monotonic() + max(650.0, (budget - 420.0) * 0.55))
            best_a, best_score = _stage(8192, wall8, best_a, best_score, param_shapes, rng)
            print("After 8192 best=%.6f len=%d" % (best_score, len(best_a)))
    except Exception as e:
        print("Stage 8192 failed:", e)

    # If we are still at lower length, test an 8192 interpolation and polish
    # only if it is competitive.
    try:
        if best_a is not None and len(best_a) < 8192 and time.monotonic() < deadline - 80:
            a8192 = _interp(best_a, 8192)
            sc8192 = _score_fft(a8192)
            if np.isfinite(sc8192) and sc8192 < best_score:
                best_score = float(sc8192)
                best_a = a8192

            remaining = max(0.0, deadline - time.monotonic() - 35.0)
            if remaining > 90.0 and np.isfinite(sc8192) and sc8192 < min(best_score, 1.60) + 1e-9:
                a_opt, sc_opt = _optimize_log(a8192, remaining, rng)
                if np.isfinite(sc_opt) and sc_opt < best_score:
                    best_score = float(sc_opt)
                    best_a = _proj_nonneg_sum(a_opt)
            print("Final short-check best=%.6f len=%d" % (best_score, len(best_a)))
    except Exception as e:
        print("Final check failed:", e)

    if best_a is None:
        best_a = np.ones(512, dtype=np.float64) / 512.0

    best_a = _proj_nonneg_sum(best_a)

    try:
        if len(best_a) <= 3000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            if np.isfinite(exact):
                print("Exact validation score=%.6f len=%d" % (exact, len(best_a)))
    except Exception:
        pass

    return [float(x) for x in best_a]
