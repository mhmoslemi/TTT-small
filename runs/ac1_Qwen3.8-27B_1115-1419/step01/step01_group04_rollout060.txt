def chebyshev_t(m, t):
    t = np.asarray(t, dtype=np.float64)
    if m == 0: return np.ones_like(t)
    if m == 1: return t.copy()
    t0 = np.ones_like(t)
    t1 = t.copy()
    for _ in range(2, m+1):
       tm = 2.0 * t * t1 - t0
       t0, t1 = t1, tm
    return t1

def make_basis(n):

       return np.zeros((8,n), dtype=np.float64)

```python
"""
Searches for non-negative step-function heights minimizing

    2 * n * max(convolve(a, a)) / (sum(a))^2.

The program combines several complementary strategies:
- Cheap FFT-based scoring and FFT-based dense/sparse subgradient steps.
- Low-dimensional Chebyshev parameterization with derivative-free global search.
- Multi-resolution coarse-to-fine search, starting from the supplied
  height_sequence_1 when available.
- Optional LP-based alternating refinement for moderate lengths.
- Final exact validation using evaluate_sequence().
"""

import time
import math

import numpy as np

try:
    import scipy.optimize as scipy_opt
    HAVE_SCIPY_OPT = True
except Exception:
    HAVE_SCIPY_OPT = False


def pair_score(pair):
    return float(pair[0])


def normalize_to_sum1(seq):
    a = np.asarray(seq, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(1024, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    a = a / s
    a = np.maximum(a, 1e-300)
    a = a / float(a.sum())
    return a


def upsample_sequence(seq, n):
    seq = np.asarray(seq, dtype=np.float64).ravel()
    n = int(n)
    if seq.size == 0:
        out = np.ones(n, dtype=np.float64)
        return out / float(n)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.size == n:
        return normalize_to_sum1(seq)
    xold = np.linspace(0.0, 1.0, seq.size)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_to_sum1(out)


def convolve_fft_raw(a, N):
    n = int(a.size)
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.maximum(c, 0.0)
    return c, fa


def convolve_fft(a, N=None):
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    if N is None:
        N = 1 << (L - 1).bit_length()
    c, _ = convolve_fft_raw(a, N)
    return c


def fast_score_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = int(a.size)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = convolve_fft(a)
    if c.size == 0:
        return float("inf")
    mc = float(np.max(c))
    val = 2.0 * float(n) * mc / (s * s)
    if not np.isfinite(val):
        return float("inf")
    return float(val)


def normalize_loga_array(loga):
    loga = np.asarray(loga, dtype=np.float64).ravel()
    if not np.all(np.isfinite(loga)):
        loga = np.where(np.isfinite(loga), loga, -700.0)
    m = float(loga.max())
    a = np.exp(loga - m)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones_like(loga, dtype=np.float64)
        s = float(a.size)
    a = a / s
    a = np.maximum(a, 1e-300)
    a = a / float(a.sum())
    loga2 = np.log(a)
    return a, loga2


def active_gradient(a, d, K, tau):
    n = int(a.size)
    L = int(d.size)
    K = max(1, min(int(K), L))
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(d, -K)[-K:]
    if tau <= 1e-8:
        tau = 1e-8
    maxd = float(np.max(d[idx]))
    logw = (d[idx] - maxd) / tau
    logw = np.clip(logw, -50.0, 0.0)
    w = np.exp(logw)
    ws = float(w.sum())
    if ws <= 1e-300:
        w = np.full(K, 1.0 / max(1, K), dtype=np.float64)
    else:
        w = w / ws
    g = np.zeros(n, dtype=np.float64)
    thresh = float(np.max(w)) * 1e-12
    for ii in range(K):
        wk = float(w[ii])
        if wk <= thresh:
            continue
        k = int(idx[ii])
        st = max(0, k - (n - 1))
        en = min(n, k + 1)
        src_start = k - (en - 1)
        src_end = k - st + 1
        g[st:en] += wk * a[src_start:src_end][::-1]
    return 2.0 * g


def gradient_dense(a, d, tau, N, A):
    n = int(a.size)
    L = int(d.size)
    if tau <= 1e-8:
        tau = 1e-8
    maxd = float(np.max(d)) if L else 0.0
    if maxd <= 0.0:
        return np.zeros(n, dtype=np.float64)
    logw = (d - maxd) / tau
    logw = np.clip(logw, -50.0, 0.0)
    w = np.exp(logw)
    ws = float(w.sum())
    if ws <= 1e-300:
        w = np.full(L, 1.0 / max(1, L), dtype=np.float64)
    else:
        w = w / ws
    wN = np.zeros(N, dtype=np.float64)
    wN[:L] = w
    W = np.fft.rfft(wN)
    grad_full = np.fft.irfft(W * np.conjugate(A), N)
    g = 2.0 * grad_full[:n].real
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    return g


def local_search(a_init, budget_s, seed=0, label=""):
    a = normalize_to_sum1(a_init)
    n = int(a.size)
    base_score = fast_score_array(a)
    if n == 0 or budget_s <= 0.5:
        return a, base_score

    N = 1 << ((2 * n - 2).bit_length())
    t0 = time.monotonic()
    deadline = t0 + max(0.5, float(budget_s))

    loga = np.log(np.maximum(a, 1e-300))
    c, A = convolve_fft_raw(a, N)
    d = n * c
    s0 = float(a.sum())
    peak0 = float(np.max(d)) if d.size else 0.0
    best_score = float(2.0 * peak0 / (s0 * s0)) if s0 > 0 else float("inf")
    best = a.copy()
    current_score = best_score

    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)

    eta = 0.18
    noise = 0.025
    tau = 0.12
    Kfrac = 0.0025
    Kmin = 8
    Kmax = 48 if n > 2048 else 96
    beta = 0.35

    mom = np.zeros(n, dtype=np.float64)
    it = 0
    last_best = 0

    while time.monotonic() < deadline:
        it += 1

        if not np.all(np.isfinite(a)) or float(a.sum()) <= 0.0:
            a = best.copy()
            loga = np.log(np.maximum(a, 1e-300))
            c, A = convolve_fft_raw(a, N)
            d = n * c
            s = float(a.sum())
            peak = float(np.max(d)) if d.size else 0.0
            current_score = float(2.0 * peak / (s * s)) if s > 0 else best_score
            mom = np.zeros(n, dtype=np.float64)

        L = 2 * n - 1
        K = int(L * Kfrac)
        K = max(Kmin, min(Kmax, K))
        if it - last_best > 300:
            K = min(2 * K, 2 * Kmax)

        dense_now = (n <= 1024) or (it % 25 == 0) or (it - last_best > 180 and it % 5 == 0)

        if dense_now:
            g = gradient_dense(a, d, tau, N, A)
        else:
            g = active_gradient(a, d, K, tau)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        center = float(np.dot(a, g))
        g = g - center
        scale = float(np.max(np.abs(g))) if g.size else 0.0

        if scale < 1e-15:
            loga_new = loga + rng.normal(0.0, 0.05, n)
            a_new, loga_new = normalize_loga_array(loga_new)
        else:
            g = g / scale
            mom = beta * mom + (1.0 - beta) * g
            direction = mom
            dscale = float(np.max(np.abs(direction))) if direction.size else 0.0
            if dscale < 1e-15 or not np.isfinite(dscale):
                direction = g
                dscale = 1.0
            direction = direction / dscale
            loga_new = loga - eta * direction
            if noise > 1e-8:
                loga_new = loga_new + rng.normal(0.0, noise, n)
            a_new, loga_new = normalize_loga_array(loga_new)

        c_new, A_new = convolve_fft_raw(a_new, N)
        d_new = n * c_new
        s_new = float(a_new.sum())
        peak_new = float(np.max(d_new)) if d_new.size else 0.0
        score_new = float(2.0 * peak_new / (s_new * s_new)) if s_new > 0 else float("inf")

        if not np.isfinite(score_new):
            eta = max(0.02, eta * 0.8)
            mom *= 0.5
            continue

        if score_new < best_score:
            best = a_new.copy()
            best_score = score_new
            last_best = it

        accept = score_new <= current_score * (1.0 + 0.02) + 1e-12
        if accept:
            old = current_score
            a = a_new
            loga = loga_new
            c = c_new
            A = A_new
            d = d_new
            current_score = score_new
            mom *= 0.85

            if score_new < old * (1.0 - 1e-9):
                eta = min(0.60, eta * 1.07)
                tau = max(0.020, tau * 0.998)
                noise = max(0.0, noise * 0.996)
                Kfrac = max(0.001, Kfrac * 0.998)
            else:
                eta = max(0.03, eta * 0.98)
        else:
            eta = max(0.02, eta * 0.8)
            mom *= 0.5

        if current_score > best_score * 1.30 + 1e-10:
            a = best.copy()
            loga = np.log(np.maximum(a, 1e-300))
            c, A = convolve_fft_raw(a, N)
            d = n * c
            s = float(a.sum())
            peak = float(np.max(d)) if d.size else 0.0
            current_score = float(2.0 * peak / (s * s)) if s > 0 else best_score
            mom = np.zeros(n, dtype=np.float64)
            eta = 0.20
            noise = 0.04
            tau = 0.15
            Kfrac = 0.005
            last_best = it

        if it - last_best > 400:
            loga_p = np.log(np.maximum(best, 1e-300))
            amp = 0.10 + 0.25 * float(rng.random())
            z = np.zeros(n, dtype=np.float64)
            for m in range(1, 5):
                z += (amp / (0.5 + 0.5 * float(m))) * np.cos(
                    2.0 * math.pi * float(m) * x + rng.uniform(0.0, 2.0 * math.pi)
                )
            if rng.random() < 0.4:
                z += rng.normal(0.0, 0.15) * (2.0 * x - 1.0)
            loga_p += z
            a_p, loga_p = normalize_loga_array(loga_p)
            c_p, A_p = convolve_fft_raw(a_p, N)
            d_p = n * c_p
            s_p = float(a_p.sum())
            peak_p = float(np.max(d_p)) if d_p.size else 0.0
            score_p = float(2.0 * peak_p / (s_p * s_p)) if s_p > 0 else float("inf")

            if np.isfinite(score_p) and score_p < best_score * 1.20:
                a = a_p
                loga = loga_p
                c = c_p
                A = A_p
                d = d_p
                current_score = score_p
            else:
                a = best.copy()
                loga = np.log(np.maximum(a, 1e-300))
                c, A = convolve_fft_raw(a, N)
                d = n * c
                s = float(a.sum())
                peak = float(np.max(d)) if d.size else 0.0
                current_score = float(2.0 * peak / (s * s)) if s > 0 else best_score

            mom = np.zeros(n, dtype=np.float64)
            eta = 0.25
            noise = 0.04
            tau = 0.15
            Kfrac = 0.006
            last_best = it

        if it % 200 == 0:
            print(f"local {label} it={it} best={best_score:.6f} cur={current_score:.6f}", flush=True)

    return best, best_score


def chebyshev_t(m, t):
    t = np.asarray(t, dtype=np.float64)
    if m == 0:
        return np.ones_like(t)
    if m == 1:
        return t.copy()
    t0 = np.ones_like(t)
    t1 = t.copy()
    for _ in range(2, m + 1):
        tm = 2.0 * t * t1 - t0
        t0, t1 = t1, tm
    return t1


def make_basis(n):
    n = int(n)
    if n <= 1:
        return np.zeros((8, max(1, n)), dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    cols = []
    for m in (1, 3, 5, 7, 2, 4, 6, 8):
        arr = chebyshev_t(m, t)
        arr = arr - float(np.mean(arr))
        std = float(np.std(arr))
        if std <= 1e-14:
            arr = np.zeros_like(arr)
        else:
            arr = arr / std
            arr *= 1.0 / math.sqrt(float(m) + 1.0)
        cols.append(arr)
    return np.vstack(cols).astype(np.float64)


def param_to_seq(p, n, basis):
    n = int(n)
    p = np.asarray(p, dtype=np.float64).ravel()
    z = basis.dot(p)
    z = z - float(np.mean(z))
    z = np.clip(z, -8.0, 8.0)
    a = np.exp(z)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        out = np.ones(n, dtype=np.float64)
        return out / float(n)
    a = a / s
    return a


def param_objective(p, n, basis):
    try:
        a = param_to_seq(p, n, basis)
        return fast_score_array(a)
    except Exception:
        return 1e6


def param_search(n, budget_s, seed):
    if budget_s <= 10.0:
        return None, float("inf")

    n = int(n)
    basis = make_basis(n)
    m = int(basis.shape[0])

    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s))
    rng = np.random.default_rng(seed)

    candidates = []
    p0 = np.zeros(m, dtype=np.float64)
    candidates.append((param_objective(p0, n, basis), p0.copy()))

    for _ in range(50):
        scale = float(rng.uniform(0.30, 2.20))
        p = rng.normal(0.0, scale, m)
        sc = param_objective(p, n, basis)
        candidates.append((sc, p.copy()))
        if time.monotonic() > t0 + 0.25 * budget_s:
            break

    if HAVE_SCIPY_OPT and budget_s > 150.0 and time.monotonic() < deadline - 25.0:
        bounds = [(-3.0, 3.0)] * m
        maxiter = 30 if n <= 1024 else 15
        popsize = 8
        try:
            res = scipy_opt.differential_evolution(
                param_objective,
                bounds,
                args=(n, basis),
                maxiter=maxiter,
                popsize=popsize,
                seed=seed,
                tol=1e-6,
                init="latinhypercube",
                polish=False,
            )
            if res is not None and hasattr(res, "x") and res.x is not None:
                pde = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), -4.0, 4.0)
                candidates.append((param_objective(pde, n, basis), pde))
        except Exception:
            pass

    candidates.sort(key=pair_score)
    best_seq = None
    best_score = float("inf")

    top_k = min(5, len(candidates))
    for i in range(top_k):
        if time.monotonic() >= deadline - 5.0:
            break
        sc, p = candidates[i]
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = param_to_seq(p, n, basis)

        if HAVE_SCIPY_OPT:
            maxiter = 200 if n <= 1024 else 100
            try:
                res = scipy_opt.minimize(
                    param_objective,
                    np.asarray(p, dtype=np.float64).ravel(),
                    args=(n, basis),
                    method="Nelder-Mead",
                    options={"maxiter": maxiter, "xatol": 1e-7, "fatol": 1e-9},
                )
                p2 = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), -4.0, 4.0)
                sc2 = param_objective(p2, n, basis)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = param_to_seq(p2, n, basis)
            except Exception:
                pass

    if best_seq is not None:
        best_seq = normalize_to_sum1(best_seq)
        best_score = fast_score_array(best_seq)

    return best_seq, best_score


def smooth_random_sequence(n, rng):
    n = int(n)
    if n < 2:
        out = np.ones(n, dtype=np.float64)
        return out / max(1, n)
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, 6):
        amp = float(rng.uniform(-1.0, 1.0)) / (0.5 + float(m))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        z += amp * np.cos(2.0 * math.pi * float(m) * t + ph)
    if rng.random() < 0.35:
        z += float(rng.uniform(-0.6, 0.6)) * (2.0 * t - 1.0)
    z = z - float(np.mean(z))
    a = np.exp(z)
    return normalize_to_sum1(a)


def add_start(starts, seq, n):
    try:
        a = normalize_to_sum1(seq)
        if a.size != int(n):
            a = upsample_sequence(a, int(n))
        sc = fast_score_array(a)
        if np.isfinite(sc):
            starts.append((sc, a.astype(np.float64)))
    except Exception:
        pass


def make_conv_matrix(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = int(f.size)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def lp_alternate(a_init, budget_s, seed):
    a = normalize_to_sum1(a_init)
    n = int(a.size)
    best_score = fast_score_array(a)
    if not HAVE_SCIPY_OPT or n > 1024 or budget_s <= 60.0:
        return a, best_score

    t0 = time.monotonic()
    deadline = t0 + max(60.0, float(budget_s))

    for _ in range(4):
        if time.monotonic() > deadline - 60.0:
            break

        try:
            f_eps = a + 1e-8
            maxf = float(f_eps.max())
            if maxf <= 1e-300:
                break
            f = f_eps / maxf

            c = convolve_fft(a)
            M = float(np.max(c)) if c.size else 0.0
            if M <= 1e-300:
                break

            A = make_conv_matrix(f)
            b = np.full(2 * n - 1, M / maxf, dtype=np.float64)
            ub = max(10.0, 10.0 * float(n) * (M / maxf))

            res = scipy_opt.linprog(
                -np.ones(n, dtype=np.float64),
                A_ub=A,
                b_ub=b,
                bounds=(0.0, ub),
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        if g.size != n:
            break
        s = float(g.sum())
        if not np.isfinite(s) or s <= 1e-15:
            break
        g = g / s

        cur_score = fast_score_array(a)
        improved = False
        for lam in (0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.45, 0.65, 0.85):
            cand = (1.0 - float(lam)) * a + float(lam) * g
            cand = normalize_to_sum1(cand)
            sc = fast_score_array(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                a = cand.copy()
                improved = True

        if improved:
            print(f"LP improve: {best_score:.6f}", flush=True)
            if cur_score - best_score < 1e-7:
                break
        else:
            break

    final_score = fast_score_array(a)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    return a, best_score


def optimize_level(n, budget_s, parent, seed):
    n = int(n)
    if budget_s <= 5.0:
        seq = upsample_sequence(parent, n) if parent is not None else None
        if seq is None:
            seq = np.ones(n, dtype=np.float64) / float(n)
        return seq, fast_score_array(seq)

    level_deadline = time.monotonic() + max(5.0, float(budget_s))
    starts = []

    if parent is not None:
        add_start(starts, upsample_sequence(parent, n), n)
    add_start(starts, np.ones(n, dtype=np.float64), n)

    t = np.linspace(0.0, 1.0, n)
    for alpha in (0.30, 0.60, 0.90):
        for sgn in (1.0, -1.0):
            add_start(starts, 1.0 + sgn * alpha * (2.0 * t - 1.0), n)
        for beta in (-0.5, 0.5):
            base = 1.0 + alpha * (2.0 * t - 1.0) + beta * (((2.0 * t - 1.0) ** 2) - 1.0 / 3.0)
            add_start(starts, base, n)

    rng = np.random.default_rng(seed)
    param_seq = None
    param_budget = min(0.35 * float(budget_s), 220.0)
    if param_budget > 25.0 and time.monotonic() < level_deadline - 30.0:
        try:
            pseq, pscore = param_search(n, param_budget, seed + 1)
            if pseq is not None and np.isfinite(pscore):
                param_seq = pseq
                add_start(starts, pseq, n)
        except Exception:
            pass

    for _ in range(30):
        if time.monotonic() > level_deadline - 60.0:
            break
        add_start(starts, smooth_random_sequence(n, rng), n)

    if not starts:
        add_start(starts, np.ones(n, dtype=np.float64), n)

    starts.sort(key=pair_score)
    best_seq = starts[0][1].copy()
    best_score = starts[0][0]

    lp_budget = 0.0
    if n <= 1024 and HAVE_SCIPY_OPT and float(budget_s) > 150.0:
        lp_budget = min(300.0, 0.28 * float(budget_s))

    local_budget = max(0.0, (level_deadline - time.monotonic()) - lp_budget)
    if not np.isfinite(local_budget) or local_budget <= 0.0:
        local_budget = 0.0

    max_local_starts = 5 if n > 2048 else 7
    k = max(1, min(len(starts), max_local_starts, int(local_budget / 45.0) + 1))

    minsc = starts[0][0]
    weights = []
    for i in range(k):
        sc = starts[i][0]
        weights.append(1.0 / max(1e-6, (float(sc) - float(minsc)) + 0.05))
    total_w = float(sum(weights))
    time_rem = float(local_budget)

    for i in range(k):
        time_to_lp = level_deadline - time.monotonic() - lp_budget
        if time_to_lp <= 10.0 or time_rem <= 0.0:
            break
        wi = float(weights[i]) / total_w if total_w > 0 else 1.0 / float(k)
        t_i = max(8.0, min(float(time_rem) * wi, time_to_lp - 5.0, float(local_budget)))
        t_start = time.monotonic()
        try:
            seq, sc = local_search(starts[i][1], t_i, int(seed + 1000 + i), label=f"n{n}_s{i}")
        except Exception:
            seq, sc = starts[i][1], starts[i][0]
        elapsed = time.monotonic() - t_start
        time_rem = max(0.0, time_rem - elapsed)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq

    if param_seq is not None and time.monotonic() < level_deadline - lp_budget - 20.0:
        try:
            seq, sc = local_search(param_seq, 60.0, int(seed + 999), label=f"n{n}_param")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
        except Exception:
            pass

    if lp_budget > 0.0 and time.monotonic() < level_deadline - 45.0:
        try:
            seq, sc = lp_alternate(best_seq, min(lp_budget, level_deadline - time.monotonic()), seed + 777)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
        except Exception:
            pass

    return best_seq, best_score


def add_candidate_pool(pool, seq, score):
    if seq is None or score is None:
        return
    try:
        sc = float(score)
    except Exception:
        return
    if not np.isfinite(sc):
        return
    a = np.asarray(seq, dtype=np.float64).ravel()
    if a.size == 0:
        return
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return
    a = a / s
    pool.append((sc, a.astype(np.float64)))
    pool.sort(key=pair_score)
    if len(pool) > 24:
        pool.pop()


def sanitize_for_return(seq):
    a = np.asarray(seq, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(1024, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(1024, dtype=np.float64)
        s = float(a.size)
    if s > 0.0:
        a = a / s
    a = np.maximum(a, 1e-12)
    a = a / float(a.sum())
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0 or float(a.sum()) < 0.01:
        a = np.ones(1024, dtype=np.float64)
        a = a / float(a.size)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0

    if budget < 25.0:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            seq = normalize_to_sum1(h)
        else:
            seq = np.ones(1024, dtype=np.float64) / 1024.0
        return sanitize_for_return(seq)

    search_deadline = t0 + max(10.0, budget - 60.0)

    parent = None
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            parent = normalize_to_sum1(h)
            psc = fast_score_array(parent)
            print(f"parent fast score = {psc:.6f}, len = {parent.size}", flush=True)
        except Exception:
            parent = None

    pool = []
    global_best_seq = None
    global_best_score = float("inf")

    if parent is not None:
        psc = fast_score_array(parent)
        global_best_seq = parent
        global_best_score = psc
        add_candidate_pool(pool, parent, psc)

    if budget < 350.0:
        final_n = 1024
    elif budget < 1200.0:
        final_n = 2048
    else:
        final_n = 4096

    if final_n <= 512:
        levels = [max(64, int(final_n))]
    else:
        base = [512]
        lv = 1024
        while lv < final_n:
            base.append(lv)
            lv *= 2
        base.append(final_n)
        levels = []
        for x in base:
            x = int(x)
            if not levels or x != levels[-1]:
                levels.append(x)

    if len(levels) == 1:
        weights = [1.0]
    elif len(levels) == 2:
        weights = [0.45, 0.55]
    elif len(levels) == 3:
        weights = [0.25, 0.30, 0.45]
    else:
        weights = [0.17, 0.22, 0.24, 0.37]

    search_time = max(10.0, min(search_deadline - t0, budget - 60.0))
    wsum = float(sum(weights))

    for idx, nlev in enumerate(levels):
        if time.monotonic() >= search_deadline - 20.0:
            break
        level_alloc = search_time * (float(weights[idx]) / wsum)
        level_budget = min(level_alloc, max(5.0, search_deadline - time.monotonic() - 10.0))
        if level_budget < 15.0:
            seq = upsample_sequence(global_best_seq, nlev) if global_best_seq is not None else np.ones(nlev, dtype=np.float64) / float(nlev)
            sc = fast_score_array(seq)
            add_candidate_pool(pool, seq, sc)
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best_seq = seq
            continue

        try:
            seq, sc = optimize_level(nlev, level_budget, global_best_seq, int(seed + 1000 * (idx + 1)))
        except Exception:
            seq = upsample_sequence(global_best_seq, nlev) if global_best_seq is not None else np.ones(nlev, dtype=np.float64) / float(nlev)
            sc = fast_score_array(seq)

        add_candidate_pool(pool, seq, sc)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best_seq = seq
            print(f"level {nlev} new global best fast = {sc:.6f}", flush=True)

    extra_time = max(0.0, search_deadline - time.monotonic())
    if extra_time > 90.0 and global_best_seq is not None:
        n_global = int(global_best_seq.size)
        target_n = final_n if final_n > n_global else n_global
        try:
            if target_n != n_global:
                seq0 = upsample_sequence(global_best_seq, target_n)
                sc0 = fast_score_array(seq0)
                add_candidate_pool(pool, seq0, sc0)
                if np.isfinite(sc0) and sc0 < global_best_score:
                    global_best_score = sc0
                    global_best_seq = seq0
            else:
                seq0 = global_best_seq.copy()
                sc0 = global_best_score
            seq, sc = local_search(seq0, max(5.0, extra_time - 20.0), int(seed + 90909), label="extra_final")
            add_candidate_pool(pool, seq, sc)
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best_seq = seq
        except Exception:
            pass

    pool.sort(key=pair_score)

    exact_deadline = t0 + max(10.0, budget - 1.0)
    exact_best_seq = None
    exact_best_score = float("inf")
    evaluated = 0
    max_evaluated = 8

    if global_best_seq is not None:
        add_candidate_pool(pool, global_best_seq, global_best_score)
        pool.sort(key=pair_score)

    for _, seq in pool[:24]:
        if time.monotonic() >= exact_deadline:
            break
        if evaluated >= max_evaluated:
            break
        nseq = int(seq.size)
        if nseq > 4096 and evaluated >= 4:
            break
        fast_sc = pair_score(0, seq) if False else 0.0
        # Use pool ordering; score is tuple first, not accessible here.
        try:
            lst = sanitize_for_return(seq)
            ex = float(evaluate_sequence(lst))
        except Exception:
            ex = float("inf")
        evaluated += 1
        print(f"exact eval len={nseq} score={ex:.6f}", flush=True)
        if np.isfinite(ex) and ex < exact_best_score:
            exact_best_score = ex
            exact_best_seq = lst

    if exact_best_seq is None:
        if pool:
            exact_best_seq = sanitize_for_return(pool[0][1])
        elif global_best_seq is not None:
            exact_best_seq = sanitize_for_return(global_best_seq)
        else:
            exact_best_seq = sanitize_for_return(np.ones(1024, dtype=np.float64) / 1024.0)

    return exact_best_seq
```
