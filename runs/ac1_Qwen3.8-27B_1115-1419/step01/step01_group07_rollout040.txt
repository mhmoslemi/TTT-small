import time

def fft_convolve(a, b):
    """Compute linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # Next power of 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def lse_max(x, tau):
    """Log-sum-exp approximation of max."""
    x_max = x.max()
    if tau < 1e-10:
        return x_max
    return x_max + tau * np.log(np.exp((x - x_max) / tau).sum())

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:

    conv = fft_convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

import numpy as np
import time

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def _compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = _fft_convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def _lse_value(a, tau):
    conv = _fft_convolve(a, a)
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    return c_max + tau * np.log(np.exp(shifted).sum())

def _lse_gradient(a, tau):
    n = len(a)
    conv = _fft_convolve(a, a)
    c_max = conv.max()
    w = np.exp((conv - c_max) / tau)
    w /= w.sum()
    # Cross-correlation: g[j] = 2 * sum_k w[k] * a[k-j]
    # Using FFT: g = 2 * ifft(fft(w_pad) * conj(fft(a_pad)))
    corr_len = len(conv) + n - 1
    size = 1 << (corr_len - 1).bit_length()
    w_pad = np.zeros(size)
    w_pad[:len(conv)] = w
    a_pad = np.zeros(size)
    a_pad[:n] = a
    F_w = np.fft.rfft(w_pad, size)
    F_a = np.fft.rfft(a_pad, size)
    corr = np.fft.irfft(F_w * np.conj(F_a), size)[:n]
    return 2.0 * corr

def _project(v, target=1.0):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.full(len(v), target / len(v))
    return v * (target / s)

def _subgradient(a, n_top=20):
    n = len(a)
    conv = _fft_convolve(a, a)
    if n_top >= len(conv):
        n_top = len(conv)
    idx = np.argpartition(conv, -n_top)[-n_top:]
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= n_top
    return g

def _lse_descent(a, tau, max_iter=1000, time_budget=None):
    """LSE-smoothed projected gradient descent."""
    n = len(a)
    a = a.copy()
    best_val = _lse_value(a, tau)
    best_a = a.copy()
    step = 1.0 / n  # Initial step
    start = time.monotonic()
    
    for it in range(max_iter):
        if time_budget and time.monotonic() - start > time_budget:
            break
        grad = _lse_gradient(a, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Normalize gradient for stable step
        grad_normed = grad / (gnorm + 1e-15)
        a_new = _project(a - step * grad_normed, 1.0)
        val_new = _lse_value(a_new, tau)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.1, 1.0)
        else:
            step *= 0.7
            if step < 1e-8:
                break
    return best_a, best_val
