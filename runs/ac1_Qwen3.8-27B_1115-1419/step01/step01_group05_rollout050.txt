**New approach: Use the structure of the optimal solution.**

For the optimal sequence, by the KKT conditions, there should be a "balance" where the gradient of the objective is a linear combination of the constraint gradients.

The objective is f(a) = max_k (a*a)_k. At the optimum, the subgradient of f is in the normal cone of the feasible set {a: a >= 0, sum(a) = 1}.

If the max is achieved at a unique point k*, then the gradient of (a*a)_{k*} w.r.t. a is:
∂(a*a)_{k*}/∂a_i = a_{k*-i} (for valid indices) + a_{k*-i'} (for the symmetric term)

More precisely: (a*a)_k = sum_i a_i * a_{k-i}, so ∂/∂a_i = a_{k-i} (if k-i is valid).

So the gradient is a "reversed and shifted" copy of a.

At the optimum: a_{k*-i} = λ + μ_i for all i where a_i > 0, where λ is the Lagrange multiplier for the sum constraint and μ_i >= 0 is the multiplier for a_i >= 0.

This means: for the optimal sequence, a_{k*-i} is "almost constant" (= λ) over the support of a.

If the support is all of [0, n-1], then a_{k*-i} = λ for all i, which means a is constant = uniform. But we know uniform isn't optimal (score = 2).

So the support must be a proper subset, OR the max is achieved at multiple points (in which case we get a weighted average of the shifted reversed sequences).

If the max is achieved at multiple points k1, k2, ..., km, then:
sum_j w_j * a_{kj - i} = λ for all i in the support, with sum w_j = 1, w_j > 0.

This is a system of linear equations that determines the structure of a!

**This is the key insight for a better algorithm:**

1. Identify the "active set" $S = \{k : (a*a)_k = M\}$ where $M = \max(a*a)$.
2. The KKT condition says: for $i$ in the support of $a$:
   $\sum_{k \in S} w_k \cdot a_{k-i} = \lambda$ (where $w_k$ are weights summing to 1).
3. For $i$ NOT in the support: $\sum_{k \in S} w_k \cdot a_{k-i} \leq \lambda$.

This gives us a system to solve! If we fix the active set S and the support, we can solve for a.

**Algorithm:**
1. Start with a good initial guess (e.g., from the previous solution).
2. Compute the active set S.
3. Solve the linear system: find a >= 0, sum(a) = 1, such that for i in support: sum_{k in S} w_k * a_{k-i} = λ.
4. Update S based on the new a.
5. Iterate.

```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp(a, M):
    """Solve max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    # Objective: max sum(b) => min -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M, b >= 0
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x, res.fun  # res.fun = -sum(b)
    return None, None
```

"""
LP-based Frank-Wolfe optimization for minimizing max self-convolution peak.

Core algorithm: iteratively solve the LP
    max sum(b)  s.t.  (a*b)_k <= M for all k,  b >= 0
where M = max(a*a), then update a <- (1-t)*a + t*b with line search.
This gives the exact best linearized direction at each step (Frank-Wolfe method).

Complemented by spectral flattening and LSE gradient polish for escaping
local minima and final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _score(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def _build_conv_matrix(a):
    n = a.size
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            v = a[k - j]
            if v > 0:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_direction(a, M):
    n = a.size
    if n < 2:
        return np.ones(1)
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_optimize(a, deadline, rng, verbose=True, max_iters=5000):
    n = a.size
    a = _normalize(a)
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    best_score = 2.0 * n * best_M
    it = 0
    stuck = 0

    while it < max_iters and time.monotonic() < deadline:
        M = float(np.convolve(a, a).max())

        b = _lp_direction(a, M)
        if b is None:
            break
        s = float(b.sum())
        if s < 1e-12:
            break
        b = b / s

        # Line search
        best_t = 0.0
        best_new_M = M
        best_a_new = None
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            sn = float(a_new.sum())
            if sn < 1e-12:
                continue
            a_new = a_new / sn
            new_M = float(np.convolve(a_new, a_new).max())
            if new_M < best_new_M:
                best_new_M = new_M
                best_t = t
                best_a_new = a_new

        if best_t == 0.0:
            stuck += 1
            if stuck > 20:
                # Perturb to escape
                a = _normalize(a + 0.05 * rng.standard_normal(n))
                stuck = 0
            it += 1
            continue

        a = best_a_new
        stuck = 0

        if best_new_M < best_M - 1e-14:
            best_M = best_new_M
            best_a = a.copy()
            best_score = 2.0 * n * best_M
            if verbose and it % 50 == 0:
                print(f"      LP iter {it}: score={best_score:.8f}", flush=True)
        else:
            # No improvement, check convergence
            if best_new_M >= best_M * (1.0 - 1e-12):
                # Try smaller step or perturb
                if it % 100 == 0 and it > 0:
                    a = _normalize(best_a + 0.02 * rng.standard_normal(n))

        if best_M < 1.0 / n * 1.01:
            break

        it += 1

    return best_a, best_score


def _lse_optimize(a, deadline, rng):
    """LSE gradient descent as complement/polish."""
    n = a.size
    if n < 4:
        return a, _score(a)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a)

    def _autocorr(x, fft_size):
        X = np.fft.rfft(x, fft_size)
        return np.fft.irfft(X * X, fft_size)[:2 * n - 1]

    fft_size = 1 << (2 * n - 1).bit_length()
    g_fft = 1 << (3 * n - 2).bit_length()

    beta = 20.0
    alpha_lr = 0.1
    it = 0
    fail = 0

    while time.monotonic() < deadline and it < 5000:
        b = _autocorr(a, fft_size)
        maxb = float(b.max())
        if maxb < 1e-30:
            maxb = 1e-30
        beta_cur = min(1e6, max(1.0, beta / maxb))

        # LSE gradient
        m = float(b.max())
        arg = beta_cur * (b - m)
        arg = np.maximum(arg, -745.0)
        p = np.exp(arg)
        ps = float(p.sum())
        if ps < 1e-30:
            p = np.ones_like(b)
            ps = float(b.size)
        p = p / ps

        # Gradient of LSE(a*a) w.r.t. a
        ar = a[::-1]
        A_f = np.fft.rfft(ar, g_fft)
        W_f = np.fft.rfft(p, g_fft)
        conv = np.fft.irfft(A_f * W_f, g_fft).real
        grad = 2.0 * conv[n - 1:2 * n - 1]
        grad = np.maximum(grad, 0.0)
        grad_c = grad - float(grad.mean())
        scale = float(np.max(np.abs(grad_c)))
        if scale < 1e-30:
            break

        v = a - alpha_lr * grad_c / scale
        a = _normalize(v)

        new_score = _score(a)
        if new_score < best_score:
            best_a = a.copy()
            best_score = new_score
            alpha_lr = min(2.0, alpha_lr * 1.1)
            fail = 0
        else:
            alpha_lr *= 0.7
            fail += 1
            if fail > 100:
                a = _normalize(best_a + 0.03 * rng.standard_normal(n))
                fail = 0
                beta = max(15.0, beta * 0.5)

        if it % 200 == 0:
            beta = min(500.0, beta + 2.0)

        it += 1

    return best_a, best_score


def _spectral_flatten(a, n_iters=50, rng=None):
    """Iterative spectral flattening heuristic."""
    n = a.size
    a = _normalize(a)
    best_a = a.copy()
    best_s = _score(a)

    for it in range(n_iters):
        A = np.fft.fft(a)
        power = np.abs(A) ** 2
        # Target: reduce peaks in power spectrum
        avg_p = float(power.mean())
        # Create target with reduced peaks
        target = np.minimum(power, 1.5 * avg_p)
        # Reconstruct with new amplitude, keep phase
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        new_A = np.sqrt(target) * phase
        a_new = np.real(np.fft.ifft(new_A))
        a_new = np.maximum(a_new, 0.0)
        sn = float(a_new.sum())
        if sn < 1e-12:
            break
        a_new = a_new / sn
        s_new = _score(a_new)
        if s_new < best_s:
            best_a = a_new.copy()
            best_s = s_new
        a = a_new

    return best_a, best_s


def _make_geometric(n, alpha):
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    return _normalize(a)


def _make_two_sided(n, alpha):
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-alpha * np.abs(i - c))
    return _normalize(a)


def _make_cosine(n):
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / n) ** 2 + 0.05
    return _normalize(a)


def _make_raised_cosine(n):
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / n)) + 0.1
    return _normalize(a)


def _make_blackman(n):
    n = max(int(n), 4)
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_smooth_random(n, rng, base=None, amp=0.3):
    n = max(int(n), 2)
    if base is None:
        base = np.ones(n) / n
    z = rng.standard_normal(n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 10 + 2)
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n)
    f = f - f.mean()
    sd = f.std()
    if sd > 0:
        f = f / sd
    a = base * (1.0 + amp * f)
    a = np.maximum(a, 1e-10)
    return _normalize(a)


def _resample(a, n_new):
    n_new = max(int(n_new), 2)
    a = _normalize(a)
    if a.size == n_new:
        return a.copy()
    if a.size == 1:
        return np.full(n_new, 1.0 / n_new)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n_new)
    new = np.interp(x_new, x_old, a)
    return _normalize(new)


def _get_given():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 1e-300:
        return None
    return arr / s


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0
    deadline = time.monotonic() + budget - 20.0
    rng = np.random.default_rng(seed)

    given = _get_given()

    # Initial best: use given if available, else geometric
    if given is not None and given.size >= 16:
        best_a = _normalize(given.copy())
    else:
        best_a = _make_geometric(256, 1.0)
    best_score = _score(best_a)
    print(f"Initial: len={best_a.size}, score={best_score:.8f}", flush=True)

    # Generate diverse initializations
    n_base = best_a.size
    inits = []

    # Given sequence (if available)
    if given is not None and given.size >= 16:
        inits.append(("given", _normalize(given.copy())))

    # Geometric variants
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append((f"geo_{alpha}", _make_geometric(n_base, alpha)))

    # Two-sided (symmetric bump)
    for alpha in [1.0, 2.0, 4.0]:
        inits.append((f"2sided_{alpha}", _make_two_sided(n_base, alpha)))

    # Cosine family
    inits.append(("cosine", _make_cosine(n_base)))
    inits.append(("raised_cos", _make_raised_cosine(n_base)))
    inits.append(("blackman", _make_blackman(n_base)))

    # Smooth random perturbations of best
    for _ in range(4):
        inits.append(("rand_smooth", _make_smooth_random(n_base, rng, best_a, 0.3)))

    # Uniform
    inits.append(("uniform", np.ones(n_base) / n_base))

    print(f"Generated {len(inits)} initializations for n={n_base}", flush=True)

    # Phase 1: LP-optimize each init (time-bounded)
    time_per_init = max(10.0, (budget * 0.4) / len(inits))
    used_time = 0.0

    for name, init in inits:
        if time.monotonic() > deadline - 60:
            break
        t_end = min(deadline - 30, time.monotonic() + time_per_init)
        if time.monotonic() >= t_end:
            continue
        init = _normalize(init)
        if init.size != n_base:
            init = _resample(init, n_base)
        t0 = time.monotonic()
        res_a, res_s = _lp_optimize(init.copy(), t_end, rng, verbose=False, max_iters=3000)
        dt = time.monotonic() - t0
        used_time += dt
        print(f"  [{name}] n={res_a.size} score={res_s:.8f} time={dt:.1f}s", flush=True)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"    *** New best: {best_score:.8f}", flush=True)

    # Phase 2: Resample to larger lengths and optimize
    lengths_to_try = []
    cur_n = best_a.size
    for L in [512, 1024, 2048, 4096]:
        if L > cur_n and time.monotonic() < deadline - 120:
            lengths_to_try.append(L)

    for L in lengths_to_try:
        if time.monotonic() > deadline - 90:
            break
        init_big = _resample(best_a, L)
        # Mix with a fresh geometric to avoid degeneracy
        geo = _make_geometric(L, 1.5)
        init_big = _normalize(0.8 * init_big + 0.2 * geo)

        # Use LP for smaller L, LSE for larger
        if L <= 1024:
            t_end = min(deadline - 60, time.monotonic() + budget * 0.15)
            res_a, res_s = _lp_optimize(init_big.copy(), t_end, rng, verbose=False, max_iters=2000)
        else:
            t_end = min(deadline - 60, time.monotonic() + budget * 0.12)
            res_a, res_s = _lse_optimize(init_big.copy(), t_end, rng)

        print(f"  [upscale to {L}] score={res_s:.8f}", flush=True)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"    *** New best: {best_score:.8f}", flush=True)

    # Phase 3: Spectral flattening on best
    if time.monotonic() < deadline - 60:
        t_end = min(deadline - 30, time.monotonic() + 120)
        res_a, res_s = _spectral_flatten(best_a.copy(), n_iters=200, rng=rng)
        print(f"  [spectral] score={res_s:.8f}", flush=True)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"    *** New best: {best_score:.8f}", flush=True)

    # Phase 4: Final LP polish on best
    if time.monotonic() < deadline - 10:
        n_best = best_a.size
        if n_best <= 1024:
            t_end = min(deadline - 5, time.monotonic() + (deadline - time.monotonic()) * 0.4)
            res_a, res_s = _lp_optimize(best_a.copy(), t_end, rng, verbose=False, max_iters=5000)
        else:
            t_end = min(deadline - 5, time.monotonic() + (deadline - time.monotonic()) * 0.4)
            res_a, res_s = _lse_optimize(best_a.copy(), t_end, rng)
        print(f"  [final polish] score={res_s:.8f}", flush=True)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"    *** New best: {best_score:.8f}", flush=True)

    # Phase 5: Very last LSE refinement
    if time.monotonic() < deadline - 3:
        t_end = deadline - 2
        res_a, res_s = _lse_optimize(best_a.copy(), t_end, rng)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s

    print(f"Final: len={best_a.size}, score={best_score:.8f}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)
    return [float(x) for x in out]
