"""
Optimize step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key insight: asymmetric power-law shapes (a[i] = (i/n)^p) give much better
scores than symmetric shapes. The optimal p balances the center peak and
end peak of the convolution. Uses LSE-smoothed L-BFGS with FFT-based
convolution for fast optimization.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a):
    """Compute full convolution using FFT for O(n log n) speed."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    return c


def lse_max(conv, tau):
    """Log-sum-exp approximation of max."""
    c_max = conv.max()
    if c_max == 0:
        return 0.0
    shifted = conv - c_max
    return c_max + tau * np.log(np.exp(shifted / tau).sum())


def smooth_objective_and_grad(a, tau):
    """Compute LSE-smoothed objective and its gradient w.r.t. a."""
    conv = fft_convolve(a)
    # LSE approximation of max
    c_max = conv.max()
    shifted = conv - c_max
    exp_vals = np.exp(shifted / tau)
    lse = c_max + tau * np.log(exp_vals.sum())
    # Weights for gradient
    w = exp_vals / exp_vals.sum()
    # Gradient: 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return lse, grad

def exact_subgradient_step(a, n_active, step):
    """One exact subgradient step using true max."""
    conv = fft_convolve(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return a
    
    # Find active set
    sorted_idx = np.argsort(conv)[::-1]
    active = sorted_idx[:min(n_active, len(conv))]
    
    # Compute subgradient at active points
    grad = np.zeros(len(a))
    for k in active:
        # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j <= n-1
        lo = max(0, k - len(a) + 1)
        hi = min(len(a) - 1, k)
        # a[k-j] for j in [lo, hi] is a[hi:lo-1:-1]
