def project_simplex(v, out=None):
    n=v.size
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.arange(1,n+1,dtype=float)
    cond = u*rho > cssv - 1.0
    if not cond.any(): return np.full(n,1/n)
    r=np.nonzero(cond)[0][-1]
    theta=(cssv[r]-1.0)/(r+1.0)
    w=np.maximum(v-theta,0.0)
    # due fp, sum may not exactly 1; adjust if sum
    s=w.sum()
    if abs(s-1)>1e-12: w += (1-s)/n? but may break nonneg. Better w/=s? If due numeric, divide.
    return w

"""
Search for low-scoring nonnegative step functions.

Strategy:
- Work on the normalized simplex (sum = 1, values >= 0).
- Use FFT-based convolution and FFT-based gradients for weighted peak reductions.
- Apply a restart-capable accelerated projected gradient search with adaptive
  peak weights (softmax over convolution peaks), multiple phases from broad
  peak weighting to sharp peak weighting, and Fourier/Gaussian kicks to escape
  local minima.
- Seed the search from the provided sequence, its reversals/kicks, and many
  deterministic random Fourier-style sequences, then polish the best seeds.
"""

import numpy as np
import time
import math


SEARCH_SCHEDULE = [
    (0.20, 0.70, 0.22, True),
    (0.25, 1.50, 0.28, True),
    (0.25, 4.00, 0.28, True),
    (0.15, 12.00, 0.24, False),
    (0.15, 35.00, 0.20, False),
]

FINAL_SCHEDULE = [
    (0.25, 1.00, 0.25, True),
    (0.30, 3.00, 0.25, True),
    (0.45, 10.00, 0.22, False),
]

SEED_WEIGHTS = [0.34, 0.25, 0.18, 0.13, 0.07, 0.03]


def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def normalize_simplex(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([1.0], dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-15:
        return np.full(arr.shape, 1.0 / float(arr.size), dtype=np.float64)
    return arr / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(v.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    r = int(np.nonzero(cond)[0][-1])
    theta = (cssv[r] - 1.0) / float(r + 1)
    w = v - theta
    np.maximum(w, 0.0, out=w)
    s = float(w.sum())
    if s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if abs(s - 1.0) > 1e-12:
        w /= s
    return w


def compute_q_max(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return 0.0
    L = 2 * n - 1
    if N is None:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    q = np.fft.irfft(A * A, N)[:L]
    if q.size == 0:
        return 0.0
    return float(max(0.0, q.max()))


def get_parent_sequence():
    val = globals().get("height_sequence_1", None)
    if val is None:
        return None
    try:
        arr = np.asarray(val, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    max_n = 20000
    if arr.size > max_n:
        idx = np.linspace(0, arr.size - 1, max_n, dtype=np.int64)
        arr = arr[idx]
    if arr.size < 2:
        return None
    return arr


def score_key(item):
    return item[0]


def l1_ratio(p, q):
    n = max(1, int(p.size))
    return float(np.abs(np.asarray(p) - np.asarray(q)).sum()) / float(n)


def select_top_seeds(cands, K):
    if not cands:
        return []
    cands = sorted(cands, key=score_key)
    selected = []
    for score, p in cands:
        duplicate = False
        for _, q in selected:
            if l1_ratio(p, q) < 1e-5:
                duplicate = True
                break
        if not duplicate:
            selected.append((float(score), p))
            if len(selected) >= K:
                break
    if len(selected) < K:
        selected = cands[:K]
    return selected


def add_seed_to_list(cands, p, n):
    if p is None:
        return
    p = normalize_simplex(p)
    if p.size != n:
        return
    qmax = compute_q_max(p, None)
    if math.isfinite(qmax):
        cands.append((float(qmax), p))


def make_fourier_seed(n, rng, kind):
    if n <= 1:
        return np.ones(n, dtype=np.float64) / float(max(1, n))

    i = np.arange(n, dtype=np.float64)
    scale = 2.0 * math.pi / float(n)
    v = np.zeros(n, dtype=np.float64)
    max_low = max(2, min(80, n // 2))

    if kind == 0:
        m = int(1 + rng.integers(0, max(1, min(16, n // 2))))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        if rng.random() < 0.5:
            v[:] = np.cos(scale * m * i + ph)
        else:
            v[:] = np.sin(scale * m * i + ph)

    elif kind == 1:
        R = int(1 + rng.integers(0, 8))
        for _ in range(R):
            m = int(1 + int((rng.random() ** 2) * (max_low - 1)))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            amp = float(rng.uniform(0.3, 1.0))
            if rng.random() < 0.5:
                v += amp * np.cos(scale * m * i + ph)
            else:
                v += amp * np.sin(scale * m * i + ph)

    elif kind == 2:
        raw = np.zeros(n, dtype=np.float64)
        R = int(1 + rng.integers(0, 8))
        for _ in range(R):
            m = int(1 + int((rng.random() ** 2) * (max_low - 1)))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            amp = float(rng.uniform(0.3, 1.0))
            if rng.random() < 0.5:
                raw += amp * np.cos(scale * m * i + ph)
            else:
                raw += amp * np.sin(scale * m * i + ph)
        v = 0.5 * (raw - raw[::-1])

    else:
        m1 = int(1 + rng.integers(0, max(1, min(24, n // 2))))
        m2 = int(1 + rng.integers(0, max(1, min(24, n // 2))))
        ph1 = float(rng.uniform(0.0, 2.0 * math.pi))
        ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
        v = np.sin(scale * m1 * i + ph1) * np.cos(scale * m2 * i + ph2)

    v = v - float(v.mean())
    maxabs = float(np.max(np.abs(v))) if v.size else 0.0
    if maxabs > 1e-12:
        v *= 0.95 / maxabs
    else:
        v[:] = 0.0

    p = (1.0 + v) / float(n)
    return project_simplex(p)


def kick_sequence(base, n, rng, amp):
    if amp <= 1e-14:
        return np.asarray(base, dtype=np.float64).copy()

    v = np.asarray(base, dtype=np.float64).ravel() * float(n) - 1.0
    v = np.where(np.isfinite(v), v, 0.0)

    if n > 8 and rng.random() < 0.35:
        i = np.arange(n, dtype=np.float64)
        scale = 2.0 * math.pi / float(n)
        m = int(1 + rng.integers(0, max(1, min(64, n // 2))))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        if rng.random() < 0.5:
            v += amp * np.cos(scale * m * i + ph)
        else:
            v += amp * np.sin(scale * m * i + ph)
    else:
        v += rng.normal(0.0, amp, n)

    v -= float(v.mean())
    np.clip(v, -0.999, 20.0, out=v)
    p = (1.0 + v) / float(n)
    return project_simplex(p)


def fill_weights(q, w, m, gamma):
    L = int(w.size)
    if L <= 0:
        return
    if gamma <= 0.0 or not math.isfinite(gamma):
        gamma = 1.0

    w[:] = q
    if m <= 1e-300 or not math.isfinite(m):
        w.fill(1.0 / float(L))
        return

    w -= m
    w *= gamma / m
    np.exp(w, out=w)
    s = float(w.sum())

    if not math.isfinite(s) or s <= 1e-300:
        k = min(32, L)
        w.fill(0.0)
        if k >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(q, -k)[-k:]
        if idx.size:
            w[idx] = 1.0 / float(idx.size)
        else:
            w.fill(1.0 / float(L))
    else:
        w *= 1.0 / s


def run_search(a0, seconds, rng, schedule, anchor_p, anchor_q):
    a0 = normalize_simplex(a0)
    n = int(a0.size)

    if seconds <= 2.0 or n <= 0:
        return anchor_p, anchor_q, a0, compute_q_max(a0, None), a0

    work = float(max(1.0, seconds - 2.0))
    start_time = time.monotonic()
    deadline = start_time + work

    L = 2 * n - 1
    N = next_pow2(max(1, 3 * n - 2))
    freqs = np.arange(N // 2 + 1, dtype=np.float64)
    phase = np.exp((-2j * math.pi) * freqs * float(n - 1) / float(N))

    total_frac = 0.0
    for frac, _, _, _ in schedule:
        total_frac += float(frac)
    if total_frac <= 0.0 or not math.isfinite(total_frac):
        total_frac = 1.0

    phase_times = []
    gammas = []
    steps = []
    moms = []
    acc = 0.0
    for frac, g, st, mom in schedule:
        acc += float(frac)
        frac_frac = min(1.0, acc / total_frac)
        phase_times.append(start_time + frac_frac * work)
        gammas.append(float(g))
        steps.append(float(st))
        moms.append(bool(mom))

    phase_idx = 0
    gamma = gammas[0]
    step = steps[0]
    use_mom = moms[0]

    p = a0.copy()
    p_prev = a0.copy()
    t = 1.0

    qmax_init = compute_q_max(p, N)
    local_best_p = p.copy()
    local_best_q = qmax_init

    global_best_p = np.asarray(anchor_p, dtype=np.float64).copy()
    global_best_q = float(anchor_q)

    if math.isfinite(local_best_q) and local_best_q < global_best_q:
        global_best_p = local_best_p.copy()
        global_best_q = local_best_q

    plateau = 0
    last_ref = local_best_q
    it = 0
    check_it = 0
    w = np.empty(L, dtype=np.float64)

    while time.monotonic() < deadline:
        if phase_idx + 1 < len(phase_times) and time.monotonic() >= phase_times[phase_idx + 1]:
            phase_idx += 1
            gamma = gammas[phase_idx]
            step = steps[phase_idx]
            if moms[phase_idx] != use_mom:
                use_mom = moms[phase_idx]
                t = 1.0
                p_prev = p.copy()
            last_ref = local_best_q
            plateau = 0

        if use_mom:
            t_old = t
            t = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_old * t_old))
            mom = (t_old - 1.0) / t
            y = p + mom * (p - p_prev)
            np.maximum(y, 0.0, out=y)
            sy = float(y.sum())
            if sy <= 1e-15:
                y = p.copy()
            else:
                y /= sy
        else:
            y = p

        A = np.fft.rfft(y, N)
        A2 = A * A
        q = np.fft.irfft(A2, N)[:L]
        np.maximum(q, 0.0, out=q)
        mm = float(q.max()) if q.size else 0.0

        if not math.isfinite(mm) or mm <= 0.0:
            break

        fill_weights(q, w, mm, gamma)

        B = phase * np.conjugate(A)
        W = np.fft.rfft(w, N)
        h = np.fft.irfft(B * W, N)[n - 1:2 * n - 1]
        direction = (2.0 * step) * h
        trial = project_simplex(y - direction)

        p_prev = p
        p = trial
        it += 1

        if mm < local_best_q:
            local_best_p = y.copy()
            local_best_q = mm
        if mm < global_best_q:
            global_best_p = y.copy()
            global_best_q = mm

        if it - check_it >= 50:
            check_it = it
            qmax_p = compute_q_max(p, N)
            if qmax_p < local_best_q:
                local_best_p = p.copy()
                local_best_q = qmax_p
            if qmax_p < global_best_q:
                global_best_p = p.copy()
                global_best_q = qmax_p
            if qmax_p < last_ref - 1e-13:
                last_ref = qmax_p
                plateau = 0

        if it % 20 == 0:
            if mm < last_ref - 1e-13:
                last_ref = mm
                plateau = 0
                step = min(0.8, step * 1.02)
            else:
                plateau += 20
                if plateau > 400:
                    step = max(0.02, step * 0.9)
                    plateau = 0
                elif plateau > 1200:
                    amp = 0.4 + 0.8 * float(rng.random())
                    p = kick_sequence(local_best_p, n, rng, amp)
                    p_prev = p.copy()
                    t = 1.0
                    step = steps[phase_idx]
                    plateau = 0
                    last_ref = local_best_q
                elif plateau > 3000:
                    amp = 1.0 + 2.0 * float(rng.random())
                    p = kick_sequence(local_best_p, n, rng, amp)
                    p_prev = p.copy()
                    t = 1.0
                    step = steps[phase_idx]
                    plateau = 0
                    last_ref = local_best_q

    qmax_p = compute_q_max(p, N)
    if qmax_p < local_best_q:
        local_best_p = p.copy()
        local_best_q = qmax_p
    if qmax_p < global_best_q:
        global_best_p = p.copy()
        global_best_q = qmax_p

    return global_best_p, global_best_q, local_best_p, local_best_q, p


def seed_scan(n, rng, parent, time_limit):
    cands = []

    if parent is None:
        parent = np.ones(n, dtype=np.float64) / float(n)

    add_seed_to_list(cands, parent, n)
    add_seed_to_list(cands, parent[::-1].copy(), n)

    for amp in (0.2, 0.5, 1.0, 1.8, 3.0):
        add_seed_to_list(cands, kick_sequence(parent, n, rng, amp), n)

    deadline = time.monotonic() + float(max(0.0, time_limit))
    count = 0
    while time.monotonic() < deadline:
        kind = count % 4
        p = make_fourier_seed(n, rng, kind)
        add_seed_to_list(cands, p, n)
        count += 1
        if count > 50000:
            break

    return select_top_seeds(cands, 6)

        )
        print(
            f"seed {idx}: qmax={2.0 * float(n) * anchor_q:.10f} "
            f"score={2.0 * float(n) * anchor_q:.10f} t={time.monotonic() - start_time:.1f}"
        )

    if time.monotonic() < deadline - 15.0:
        final_allow = max(5.0, deadline - time.monotonic() - 15.0)
        anchor_p, anchor_q, local_p, local_q, cur_p = run_search(
            anchor_p.copy(), final_allow, rng, FINAL_SCHEDULE, anchor_p, anchor_q
        )
        print(f"final polish: score={2.0 * float(n) * anchor_q:.10f}")

    best_list = [float(max(0.0, x)) for x in anchor_p]
    for idx, val in enumerate(best_list):
        if not math.isfinite(val):
            best_list[idx] = 0.0

    try:
        best_eval = float(evaluate_sequence(best_list))
    except Exception:
        best_eval = 2.0 * float(n) * anchor_q

    if not math.isfinite(best_eval):
        best_eval = 2.0 * float(n) * anchor_q

    if math.isfinite(best_eval) and best_eval < parent_eval:
        return best_list

    if parent_exists:
        return parent_list

    return best_list
