"""
Optimized search for low-peak nonnegative self-convolutions using Adam optimizer
on a log-sum-exp surrogate with systematic annealing and multi-restart strategy.
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        n = a.size
        return np.ones(max(n, 1)) / max(n, 1)
    return a / s

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2*n-1]

def _gradient_lse(a, b, p, fft_size):
    """Compute gradient of LSE(beta*b) w.r.t. a, where p = softmax(beta*(b-max(b)))."""
    n = a.size
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size).real
    h = conv[n-1:2*n-1]
    return 2.0 * h

def _softmax_weights(b, beta):
    """Compute p_j proportional to exp(beta * b_j)."""
    m = float(np.max(b))
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones(b.size) / b.size
    return e / s

def _score(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    fft_size = _next_pow2(2*n-1)
    b = _autocorr(a, fft_size)
    return 2.0 * float(n) * float(np.max(b))

def _make_exponential(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)

def _make_bilateral(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x) + np.exp(-alpha * (1.0 - x))
    return _normalize(a)

def _make_unimodal(n, center, width):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-((x - center) / width) ** 2)
    return _normalize(a)

def _resample(old, new_n):
    old = _normalize(old)
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

class AdamOptimizer:
    def __init__(self, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
        self.m = None
        self.v = None
    
    def reset(self, shape):
        self.t = 0
        self.m = np.zeros(shape)
        self.v = np.zeros(shape)
    
    def step(self, x, grad):
        self.t += 1
        if self.m is None or self.m.size != x.size:
            self.reset(x.size)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad * grad
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def _project_simplex(v):
    """Project to probability simplex (nonnegative, sum=1)."""
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.ones(w.size) / w.size
    return w / s

    if rng is None:
        rng = np.random.default_rng()
    
    a = _normalize(a)
    n = a.size
    fft_b = _next_pow2(2*n - 1)
    fft_g = _next_pow2(3*n - 2)  # for gradient computation

    b = _autocorr(a, fft_b)
    maxb = float(np.max(b))
    score = 2.0 * n * maxb
    best_a = a.copy()

    adam = AdamOptimizer(lr=lr, beta1=0.8, beta2=0.999, eps=1e-8)
    
    # Beta schedule: geometric annealing
    beta = beta_start
    no_improve = 0
    max_no_improve = 500
    it = 0
    
    while time.monotonic() < deadline - 5.0:

        # Compute gradient
        grad = _gradient_lse(a, b, p, fft_g)
        
        # Center the gradient (removes the component along the all-ones direction)
        grad_centered = grad - float(np.mean(grad))
        
        # Adam step
        v = adam.step(a.copy(), grad_centered)
        
        # Project to simplex
        a_new = _project_simplex(v)

        b_new = _autocorr(a_new, fft_b)

        score_new = 2.0 * n * maxb_new

            best_a = a_new.copy()
            best_score = score_new
            no_improve = 0
            a = a_new
            b = b_new
            maxb = maxb_new
        else:
            no_improve += 1
            a = a_new  # Still accept the step (Adam handles this)
            b = b_new
            maxb = maxb_new
        
        # Adaptive beta increase

            # Increase beta based on how peaked b is
            ratio = maxb / float(np.mean(b))
            beta = min(beta_end, beta * (1.0 + 0.02 * ratio))

            # Perturb from best
            noise = rng.normal(0, 0.1, size=n)

            b = _autocorr(a, fft_b)

            score = 2.0 * n * maxb
            adam = AdamOptimizer(lr=lr * 0.5, beta1=0.8, beta2=0.999, eps=1e-8)
            no_improve = 0

        it += 1
    
    # Final check
    final_score = _score(best_a)

Optimized search for low-peak nonnegative self-convolutions.
Uses log-softmax parameterization with Adam optimizer on LSE surrogate,
multi-scale resampling, and L-BFGS-B polishing.
"""

import numpy as np
import time
import math

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        n = a.size
        return np.ones(max(n, 1)) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _softmax(x):
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones(x.size) / x.size
    return e / s


def _lse_weights(b, beta):
    m = float(np.max(b))
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones(b.size) / b.size
    return e / s


def _gradient_wrt_a(a, p, fft_size):
    """Gradient of sum_k p_k * (a*a)_k w.r.t. a."""
    n = a.size
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _gradient_wrt_x(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. x where a=softmax(x)."""
    s = float(np.dot(grad_a, a))
    return grad_a - a * s


def _score(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    fft_size = _next_pow2(2 * n - 1)
    b = _autocorr(a, fft_size)
    return 2.0 * float(n) * float(np.max(b))


def _make_exponential(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_bilateral(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x) + np.exp(-alpha * (1.0 - x))
    return _normalize(a)


def _make_gaussian(n, center, width):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-((x - center) / width) ** 2)
    return _normalize(a)


def _make_cosine(n, k=1):
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * k * x)
    a = np.maximum(a, 0.01)
    return _normalize(a)


def _resample(old, new_n):
    old = _normalize(old)
    old_n = old.size
    if old_n == new_n:
        return old.copy()
    if old_n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

        a_best = _softmax(best_x)
        return a_best, _score(a_best)

    return a0, _score(a0)


def _block_local_search(a, deadline, rng):
    """
    Try local improvements by optimizing small blocks of coordinates.
    """
    n = a.size
    a = _normalize(a)
    best = a.copy()
    best_s = _score(a)
    fft_size = _next_pow2(2 * n - 1)

    block_size = max(4, n // 32)
    attempts = 0
    max_attempts = max(50, int((deadline - time.monotonic()) * 10))

    while time.monotonic() < deadline - 5.0 and attempts < max_attempts:
        attempts += 1
        c = best.copy()
        start = int(rng.integers(0, max(1, n - block_size)))
        stop = min(n, start + block_size)

        # Randomly perturb the block
        c[start:stop] *= np.exp(rng.normal(0, 0.8, size=stop - start))
        c = _normalize(c)

        # Try to improve by coordinate descent within the block
        for _ in range(20):
            improved = False
            for i in range(start, stop):
                old_val = c[i]
                # Try a few multipliers
                for mult in [0.5, 0.7, 0.85, 1.1, 1.3, 1.5, 0.0, 2.0]:
                    c2 = c.copy()
                    c2[i] = old_val * mult
                    c2 = _normalize(c2)
                    b2 = _autocorr(c2, fft_size)
                    s2 = 2.0 * n * float(np.max(b2))
                    if s2 < best_s - 1e-12:
                        best = c2.copy()
                        best_s = s2
                        c = c2.copy()
                        improved = True
                        break
            if not improved:
                break

        # Also try setting some coordinates to 0
        if rng.random() < 0.3:
            c = best.copy()
            for _ in range(rng.integers(1, 4)):
                idx = int(rng.integers(0, n))
                c[idx] = 0.0
            c = _normalize(c)
            b_c = _autocorr(c, fft_size)
            s_c = 2.0 * n * float(np.max(b_c))
            if s_c < best_s - 1e-12:
                best = c.copy()
                best_s = s_c

    return best, best_s


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point. Uses log-softmax parameterization with Adam and L-BFGS-B
    to minimize the peak self-convolution score.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + budget - 25.0
    rng = np.random.default_rng(seed)

    # Load provided sequence
    given = None
    if 'height_sequence_1' in globals():
        try:
            seq = globals()['height_sequence_1']
            arr = np.asarray(seq, dtype=np.float64).ravel()
            arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
            arr = np.maximum(arr, 0.0)
            if arr.size > 4 and arr.sum() > 1e-10:
                given = arr / arr.sum()
        except Exception:
            given = None

    # Generate initial candidates
    candidates = []
    if given is not None:
        candidates.append(given.copy())

    for n in [128, 256, 512]:
        candidates.append(_make_exponential(n, 1.0))
        candidates.append(_make_bilateral(n, 1.5))
        candidates.append(_make_gaussian(n, 0.5, 0.25))
        candidates.append(_make_cosine(n, 1))
        candidates.append(_make_cosine(n, 2))

    # Evaluate all
    best_a = None
    best_score = float('inf')
    for i, cand in enumerate(candidates):
        s = _score(cand)
        if s < best_score:
            best_score = s
            best_a = cand.copy()

    print(f"Best initial: n={best_a.size}, score={best_score:.8f}", flush=True)

    # Phase 1: Quick Adam optimization at native resolution
    if best_a.size <= 512:
        t_end1 = min(deadline - 60.0, time.monotonic() + min(120.0, budget * 0.1))
        res_a, res_s = _adam_smooth(best_a.copy(), t_end1,
                                     beta_start=5.0, beta_end=500.0,
                                     lr=0.05, rng=rng)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"Phase 1 improved: n={best_a.size}, score={best_score:.8f}", flush=True)

    # Phase 2: Multi-scale search
    target_lengths = [256, 512, 1024, 2048]

    for n_target in target_lengths:
        if time.monotonic() > deadline - 80.0:
            break

        # Multiple starting points for this length
        starts = []
        starts.append(_resample(best_a, n_target))

        # Mix with parametric
        if n_target == 256:
            starts.append(_normalize(0.6 * _resample(best_a, n_target) +
                                     0.4 * _make_exponential(n_target, 1.0)))
            starts.append(_normalize(0.7 * _resample(best_a, n_target) +
                                     0.3 * _make_bilateral(n_target, 1.0)))

        # Random perturbation of best
        a_pert = _resample(best_a, n_target)
        a_pert = _normalize(a_pert * np.exp(rng.normal(0, 0.3, size=n_target)))
        starts.append(a_pert)

        best_for_n = best_a.copy()
        best_score_for_n = best_score

        for s_idx, start in enumerate(starts):
            if time.monotonic() > deadline - 40.0:
                break

            # Allocate time
            remaining = deadline - time.monotonic()
            per_start = min(remaining * 0.3, 200.0)
            if per_start < 30:
                break
            t_end = time.monotonic() + per_start

            # Adam optimization
            res_a, res_s = _adam_smooth(start.copy(), t_end,
                                         beta_start=3.0, beta_end=3000.0,
                                         lr=0.03, rng=rng)

            if res_s < best_score_for_n:
                best_score_for_n = res_s
                best_for_n = res_a.copy()

        if best_score_for_n < best_score:
            best_a = best_for_n.copy()
            best_score = best_score_for_n
            print(f"Phase 2 n={n_target}: improved to {best_score:.8f}", flush=True)
        else:
            print(f"Phase 2 n={n_target}: best={best_score_for_n:.8f} (no improvement over {best_score:.8f})",
                  flush=True)

    # Phase 3: L-BFGS-B polishing at the best length
    if time.monotonic() < deadline - 60.0:
        t_end3 = min(deadline - 10.0, time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.5))
        res_a, res_s = _lbfgs_polish(best_a.copy(), t_end3, beta=500.0, rng=rng)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"Phase 3 (L-BFGS): improved to {best_score:.8f}", flush=True)
        else:
            print(f"Phase 3 (L-BFGS): no improvement (got {res_s:.8f})", flush=True)

    # Phase 4: Block local search
    if time.monotonic() < deadline - 20.0:
        t_end4 = deadline - 10.0
        res_a, res_s = _block_local_search(best_a.copy(), t_end4, rng)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"Phase 4 (block): improved to {best_score:.8f}", flush=True)

    # Final: one more Adam pass with high beta
    if time.monotonic() < deadline - 5.0:
        res_a, res_s = _adam_smooth(best_a.copy(), deadline - 2.0,
                                     beta_start=50.0, beta_end=20000.0,
                                     lr=0.005, rng=rng)
        if res_s < best_score:
            best_a = res_a.copy()
            best_score = res_s
            print(f"Final Adam: improved to {best_score:.8f}", flush=True)

    # Final verification
    final_score = _score(best_a)
    if final_score < best_score:
        best_score = final_score

    print(f"FINAL: n={best_a.size}, score={best_score:.8f}", flush=True)

    out = _normalize(best_a)
    return [float(x) for x in out]
