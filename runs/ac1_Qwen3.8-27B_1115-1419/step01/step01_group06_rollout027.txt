"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses multi-start initialization, FFT-based smoothed-max gradient descent in
log-space with Adam optimizer, adaptive beta annealing, coordinate refinement,
and periodic restarts. Key improvements over prior approach: better parametric
initialization, peak-equalization steps, multi-scale optimization, and more
aggressive exploration-exploitation balance.
"""

import numpy as np
import time


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_autocorr(a, N):
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)
    return b[: 2 * len(a) - 1]


def _fft_score(a, N):
    n = len(a)
    b = _fft_autocorr(a, N)
    b = np.maximum(b, 0.0)
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * mx


def _make_weights(b, beta):
    b = np.maximum(b, 0.0)
    mx = float(np.max(b))
    if mx <= 1e-300:
        return np.ones(len(b)) / len(b)
    y = beta * (b / mx - 1.0)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(len(b))
        p[int(np.argmax(b))] = 1.0
        return p
    return e / s


def _grad_from_weights(a, w, N):
    n = len(a)
    W = np.fft.rfft(w, N)
    AR = np.fft.rfft(a[::-1], N)
    C = np.fft.irfft(W * AR, N)
    return 2.0 * C[n - 1: 2 * n - 1]


def _log_w(a):
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w -= float(np.max(w))
    return w


def _softmax(w):
    w = np.asarray(w, dtype=np.float64)
    w = w - float(np.max(w))
    e = np.exp(w)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size) / w.size
    return e / s


def _adam_update(grad, m, v, t, lr):
    t += 1
    m = 0.9 * m + 0.1 * grad
    v = 0.999 * v + 0.001 * grad * grad
    m_h = m / (1.0 - 0.9 ** t)
    v_h = v / (1.0 - 0.999 ** t)
    step = lr * m_h / (np.sqrt(v_h) + 1e-10)
    return np.clip(step, -5.0, 5.0), m, v, t


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0 or not np.isfinite(s):
        return np.ones(a.size) / a.size
    return a / s


def _raised_cosine(n):
    j = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (j + 0.5) / n) ** 2


def _gaussian_seq(n, sigma_frac=0.4):
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((j - center) / sigma) ** 2)


def _slepian_like(n, band_frac=0.3):
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (j - center) / n
    b = band_frac / 2.0
    # Approximate prolate spheroidal via sinc window
    t = x / b
    sinc_val = np.sinc(t)
    return np.maximum(sinc_val, 0.0) ** 2


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 0.01:
        a = a * (0.01 / s) if s > 0 else a
        s = float(np.sum(a))
    n = len(a)
    b = np.convolve(a, a)
    mx = float(np.max(b))
    return 2.0 * n * mx / (s * s)


def _finalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        a = np.ones(min(n, 4096), dtype=np.float64)
        n = a.size
        s = float(a.sum())
    scale = np.sqrt(2.0 * n)
    a = a * scale / s
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.02 / max(s2, 1e-30))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def _optimize_from_start(a0, N, deadline, seed=42, n_opt=None):
    """Core optimization loop from a starting sequence."""
    rng = np.random.default_rng(seed)
    f = _normalize(a0.astype(np.float64).copy())
    n = f.size
    if n_opt is not None and n_opt != n:
        # Resample to target length
        old_idx = np.linspace(0, n - 1, n_opt)
        new_f = np.interp(np.arange(n_opt), old_idx, f)
        f = _normalize(new_f)
        n = f.size
        N = _next_pow2(4 * n)
    
    N = _next_pow2(4 * n)
    
    w = _log_w(f)
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    best_f = f.copy()
    best_score = _fft_score(f, N)
    curr_score = best_score
    
    lr = 0.06
    beta = 3.0
    stall = 0
    no_improve = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        
        # Compute autocorrelation and score
        b = _fft_autocorr(f, N)
        b = np.maximum(b, 0.0)
        curr_score = float(2.0 * n * np.max(b))
        
        if not np.isfinite(curr_score):
            f = best_f.copy()
            w = _log_w(f)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            curr_score = best_score
            continue
        
        # Update best
        if curr_score < best_score:
            best_score = curr_score
            best_f = f.copy()
            no_improve = 0
            stall = 0
        else:
            no_improve += 1
        
        # Compute weights and gradient
        weights = _make_weights(b, beta)
        grad_a = _grad_from_weights(f, weights, N)
        
        # Project to log-space (maintain sum=1 constraint)
        mean_a_grad = float(np.dot(f, grad_a))
        grad_w = f * (grad_a - mean_a_grad)
        grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
        
        # Adam step
        step, m_state, v_state, t_adam = _adam_update(grad_w, m_state, v_state, t_adam, lr)
        w = w - step
        w = np.where(np.isfinite(w), w, 0.0)
        w -= float(np.max(w))
        f = _softmax(w)
        
        # Adaptive learning rate
        new_score = _fft_score(f, N)
        if np.isfinite(new_score) and new_score < curr_score:
            lr = min(0.15, lr * 1.02)
        elif np.isfinite(new_score):
            # Step was bad, revert and reduce lr
            f = _softmax(w + step)
            w = w + step
            lr = max(0.01, lr * 0.95)
            stall += 1
        
        # Anneal beta (sharpen peak detection over time)
        progress = min(1.0, iteration / 2000.0)
        beta = 3.0 + 80.0 * (progress ** 1.5)
        
        # Periodic coordinate refinement
        if iteration % 50 == 0 and stall > 5:
            b_new = _fft_autocorr(f, N)
            b_new = np.maximum(b_new, 0.0)
            # Find peak positions and try to reduce them
            peak_thresh = float(np.max(b_new)) * 0.9
            peak_mask = b_new >= peak_thresh
            peak_idx = np.where(peak_mask)[0]
            
            if len(peak_idx) > 0:
                # Reduce mass at positions contributing to peaks
                reduction = np.zeros(n)
                for k in peak_idx[:20]:
                    for j in range(n):
                        k_j = k - j
                        if 0 <= k_j < n:
                            reduction[j] += f[k_j]
                # Scale reduction by how much it contributes
                red_scale = 0.05 * float(np.max(reduction)) / max(float(np.max(f)), 1e-30)
                reduction *= min(red_scale, 0.3)
                
                w_refine = _log_w(f)
                w_refine = w_refine - reduction / (f + 1e-30) * 0.1
                w_refine = np.where(np.isfinite(w_refine), w_refine, 0.0)
                f_refine = _softmax(w_refine)
                sc_refine = _fft_score(f_refine, N)
                
                if np.isfinite(sc_refine) and sc_refine < best_score:
                    f = f_refine
                    w = _log_w(f)
                    best_score = sc_refine
                    best_f = f.copy()
                    stall = 0
                    no_improve = 0
        
        # Restart logic
        if stall > 40 or no_improve > 200:
            r = rng.random()
            if r < 0.4:
                # Revert to best with small perturbation
                sig = float(rng.choice([0.02, 0.05, 0.1, 0.2]))
                w_try = _log_w(best_f) + rng.normal(0, sig, n)
                f = _softmax(w_try)
            elif r < 0.7:
                # Larger perturbation
                sig = float(rng.choice([0.2, 0.4, 0.6]))
                w_try = _log_w(best_f) + rng.normal(0, sig, n)
                f = _softmax(w_try)
            else:
                # Try reversed sequence
                f = _normalize(f[::-1].copy())
            
            w = _log_w(f)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            lr = 0.06
            beta = max(3.0, beta * 0.5)
            stall = 0
            no_improve = 0
        
        # Occasional beta boost for peak targeting
        if iteration % 100 == 0:
            beta = min(150.0, beta * 1.1)
    
    return best_f, best_score


def _try_multi_scale(a_best_full, N_full, deadline, seed=42):
    """Try multi-scale: downsample, optimize, upsample back."""
    n = len(a_best_full)
    
    # Downsample to 1/4 size
    small_n = max(256, n // 4)
    if small_n % 2 != 0:
        small_n -= 1
    small_n = min(small_n, n)
    
    if small_n >= 256 and small_n < n:
        # Average blocks
        blocks = n // small_n
        if blocks > 1:
            use = small_n * blocks
            small = np.mean(a_best_full[:use].reshape(small_n, blocks), axis=1)
            small = _normalize(small)
            
            N_small = _next_pow2(4 * small_n)
            deadline_small = time.monotonic() + max(60, (deadline - time.monotonic()) * 0.3)
            
            small_opt, small_score = _optimize_from_start(
                small, N_small, deadline_small, seed=seed + 100, n_opt=small_n
            )
            
            # Upsample back
            upsampled = np.interp(
                np.linspace(0, small_n - 1, n),
                np.arange(small_n),
                small_opt
            )
            upsampled = _normalize(upsampled)
            
            return upsampled, small_score
    
    return a_best_full, None
