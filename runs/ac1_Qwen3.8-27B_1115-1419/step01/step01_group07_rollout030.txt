def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, step_size, n_active=100):

    c = np.convolve(a, a, mode='full')
    c_max = c.max()

    active = np.where(c >= c_max - 1e-10)[0]

        active = sorted_idx

def project_simplex(v):
    """Project vector v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            step = min(step * 1.02, 1.0 / n * 20)
        else:
            step *= 0.85
            v = v * 0.5

        a = a_new
        count += 1

    return best_a, 2.0 * n * best_score / (best_a.sum() ** 2)


def smac_lbfgs_polish(a, time_limit):
    """Polish using SMAC + L-BFGS."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_score = 2.0 * n * np.convolve(a, a).max()
    best_a = a.copy()
    deadline = time.monotonic() + time_limit

    taus = [50.0 / n, 20.0 / n, 10.0 / n, 5.0 / n, 2.0 / n]

    for tau in taus:
        if time.monotonic() > deadline:
            break

        current_a = best_a.copy()

        def obj(x):
            a_ = np.maximum(x, 1e-15)
            a_ = a_ / a_.sum()
            c = np.convolve(a_, a_)
            c_max = c.max()
            c_shifted = c - c_max
            log_val = np.log(np.exp(c_shifted / tau).sum())
            return tau * log_val + c_max

        def grad(x):
            a_ = np.maximum(x, 1e-15)
            a_ = a_ / a_.sum()
            c = np.convolve(a_, a_)
            c_max = c.max()
            c_shifted = c - c_max
            w = np.exp(c_shifted / tau)
            w = w / w.sum()
            grad_a = 2.0 * np.correlate(w, a_, mode='valid')[:n]
            return grad_a

        try:
            result = minimize(
                obj, current_a, jac=grad,
                method='L-BFGS-B',
                bounds=[(1e-15, None)] * n,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': 500}
            )
            a_trial = np.maximum(result.x, 1e-15)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * np.convolve(a_trial, a_trial).max()
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                if verbose:
                    print(f"    SMAC tau={tau:.1f}: score={score:.6f}")
        except Exception:
            pass

    return best_a, 2.0 * n * best_score / (best_a.sum() ** 2)


def make_initializations(n, rng, seed_seq=None):
    """Generate diverse initializations of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Resample from seed sequence if provided
    if seed_seq is not None and len(seed_seq) > 0:
        seed = np.array(seed_seq, dtype=np.float64)
        seed = np.maximum(seed, 0)
        if seed.sum() > 0:
            seed = seed / seed.sum()
            # Interpolate to length n
            old_indices = np.linspace(0, 1, len(seed))
            new_indices = np.linspace(0, 1, n)
            resampled = np.interp(new_indices, old_indices, seed)
            resampled = np.maximum(resampled, 0)
            if resampled.sum() > 0:
                inits.append(resampled / resampled.sum())

    # Sine-modulated (various frequencies)
    for freq in [1, 2, 3, 5]:
        for alpha in [0.3, 0.5, 0.7]:
            x = 1.0 + alpha * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
            x = np.maximum(x, 1e-6)
            inits.append(x / x.sum())

    # Cosine-modulated
    for freq in [1, 2, 4]:
        x = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        x = np.maximum(x, 1e-6)
        inits.append(x / x.sum())

    # Asymmetric patterns
    # Linear ramp
    x = np.linspace(1.0, 0.1, n)
    inits.append(x / x.sum())

    # Exponential decay
    x = np.exp(-3.0 * np.arange(n) / n)
    inits.append(x / x.sum())

    # Bowed in (high at edges)
    mid = n / 2.0
    x = 0.3 + 0.7 * np.abs(np.arange(n) - mid) / mid
    inits.append(x / x.sum())

    # Bowed out (high in middle)
    x = 1.0 - 0.7 * np.abs(np.arange(n) - mid) / mid
    x = np.maximum(x, 0.05)
    inits.append(x / x.sum())

    # Random (Dirichlet with low concentration - sparse-like)
    for alpha in [0.3, 1.0, 3.0]:
        x = rng.dirichlet(np.ones(n) * alpha)
        inits.append(x)

    # Random with structure (low-frequency noise)
    for _ in range(3):
        base = np.ones(n) / n
        noise = np.zeros(n)
        for k in range(1, 6):
            noise += rng.normal(0, 1.0 / k) * np.cos(2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        x = base + 0.3 * noise
        x = np.maximum(x, 1e-8)
        inits.append(x / x.sum())

    return inits


def verbose_print(*args):
    """Print helper."""
    print(*args)


def propose_candidate(budget_s=2400, **kwargs):
    """Main search function: finds best sequence minimizing the evaluation."""
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30  # Leave 30s buffer
    rng = np.random.default_rng(42)

    # Get seed sequence
    seed_seq = None
    if 'height_sequence_1' in globals():
        seed_seq = np.array(height_sequence_1, dtype=np.float64)
        seed_seq = np.maximum(seed_seq, 0)
        if seed_seq.sum() > 0:
            seed_seq = seed_seq / seed_seq.sum()
        else:
            seed_seq = None

    best_score = float('inf')
    best_a = None

    # Try multiple sequence lengths
    lengths = [256, 512, 1024, 2048, 4096]

    # Allocate time: explore each length with a few initializations
    # Phase 1: Global exploration
    # Phase 2: Local refinement on best
    phase1_time = 0.5 * (budget_s - 30)
    phase2_time = 0.3 * (budget_s - 30)
    phase3_time = 0.2 * (budget_s - 30)

    verbose_print(f"=== Phase 1: Exploration ({phase1_time:.0f}s) ===")
    verbose_print(f"Seed sequence available: {seed_seq is not None}")

    results = []  # (score, a, n)

    time_per_length = phase1_time / len(lengths)

    for n in lengths:
        if time.monotonic() > start_time + phase1_time:
            break

        verbose_print(f"\n  n = {n}: exploring with {time_per_length:.0f}s budget")
        n_deadline = min(time.monotonic() + time_per_length, start_time + phase1_time)

        inits = make_initializations(n, rng, seed_seq)

        # Time per init
        time_per_init = (n_deadline - time.monotonic()) / len(inits)
        if time_per_init < 5:
            time_per_init = 5

        length_best_score = float('inf')
        length_best_a = None

        for idx, init in enumerate(inits):
            if time.monotonic() > n_deadline:
                break

            # Run coordinate descent
            a_res, score = run_coordinate_descent(init, time_per_init)

            if score < length_best_score:
                length_best_score = score
                length_best_a = a_res.copy()

            if score < best_score:
                best_score = score
                best_a = a_res.copy()

        if length_best_a is not None:
            results.append((length_best_score, length_best_a, n))
            verbose_print(f"  n={n} best: {length_best_score:.6f}")

    verbose_print(f"\nPhase 1 complete. Best: {best_score:.6f}")

    # Phase 2: Refine the best result
    verbose_print(f"\n=== Phase 2: Refinement ({phase2_time:.0f}s) ===")
    if best_a is not None:
        n = len(best_a)
        verbose_print(f"  Refining n={n}, current score={best_score:.6f}")

        # Try SMAC + L-BFGS on the best
        time_left = max(0, min(phase2_time, deadline - time.monotonic()))
        if time_left > 30:
            a_res, score = smac_lbfgs_polish(best_a, time_left * 0.5)
            if score < best_score:
                best_score = score
                best_a = a_res.copy()
                verbose_print(f"  After SMAC polish: {best_score:.6f}")

        # Subgradient polish
        time_left = max(0, deadline - time.monotonic())
        if time_left > 30:
            a_res, score = subgradient_polish(best_a, min(time_left * 0.5, 300))
            if score < best_score:
                best_score = score
                best_a = a_res.copy()
                verbose_print(f"  After subgradient polish: {best_score:.6f}")

        # More coordinate descent
        time_left = max(0, deadline - time.monotonic() - 60)
        if time_left > 30:
            a_res, score = run_coordinate_descent(best_a, time_left)
            if score < best_score:
                best_score = score
                best_a = a_res.copy()
                verbose_print(f"  After more CD: {best_score:.6f}")

    # Phase 3: Try different approaches on best
    verbose_print(f"\n=== Phase 3: Final polishing ({phase3_time:.0f}s) ===")
    if best_a is not None:
        n = len(best_a)
        time_left = max(0, deadline - time.monotonic())

        # Alternating CD and subgradient
        while time.monotonic() < deadline - 10:
            seg_time = min(30, deadline - time.monotonic() - 10)
            if seg_time < 5:
                break

            # Coordinate descent
            a_res, score = run_coordinate_descent(best_a, seg_time * 0.6)
            if score < best_score:
                best_score = score
                best_a = a_res.copy()

            # Subgradient
            seg_time = min(30, deadline - time.monotonic() - 10)
            if seg_time < 5:
                break
            a_res, score = subgradient_polish(best_a, seg_time * 0.4)
            if score < best_score:
                best_score = score
                best_a = a_res.copy()

        # Verify with evaluate_sequence
        final_score = evaluate_sequence([float(x) for x in best_a])
        verbose_print(f"\nFinal verified score: {final_score:.6f}")
        verbose_print(f"Sequence length: {len(best_a)}")
        verbose_print(f"Total time used: {time.monotonic() - start_time:.1f}s")

    if best_a is None:
        # Fallback: uniform sequence
        best_a = np.ones(256) / 256
        verbose_print("Fallback to uniform sequence!")

    return [float(val) for val in best_a]
