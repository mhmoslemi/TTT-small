def next_pow2(size):
    if size <= 1: return 1
    return 1 << (size - 1).bit_length()

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0)
    s = a.sum()
    if s < 1e-300: return np.ones(len(a)) / len(a)
    return a / s

def autocorr_fft(a, fft_size):
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2*len(a)-1]

def lse_and_p(b, beta):
    m = np.max(b)
    if beta <= 1:
        return np.ones(len(b))/len(b), m
    arg = np.maximum(beta*(b-m), -745)
    e = np.exp(arg)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(b))/len(b), m
    p = e/s
    val = m + np.log(s)/beta
    return p, val

def weighted_grad(a, weights, fft_size):
    """Gradient of sum_k w_k * (a*a)_k w.r.t. a"""
    n = len(a)
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    g = conv[n-1:2*n-1]
    return 2.0 * np.maximum(g, 0)

def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    n = a.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n,1))/max(n,1)
    return a/s

def _acorr(a, fs):
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A*A, fs)[:2*a.size-1]

def _lse_p(b, beta):
    m = b.max()
    if beta <= 1.0:
        return np.full(b.size, 1.0/b.size), float(m)
    arg = np.clip(beta*(b-m), -745, 709)
    e = np.exp(arg)
    s = e.sum()
    if s < 1e-300:
        return np.full(b.size, 1.0/b.size), float(m)
    return e/s, float(m + np.log(s)/beta)

def _wgrad(a, w, fs):
    n = a.size
    ar = a[::-1]
    A = np.fft.rfft(ar, fs)
    W = np.fft.rfft(w, fs)
    c = np.fft.irfft(A*W, fs).real
    return 2.0*c[n-1:2*n-1]

def _score(a):
    a = _norm(a)
    n = a.size
    if n < 1: return 1e100
    b = _acorr(a, _npow2(2*n-1))
    mb = b.max()
    return 2.0*n*mb if np.isfinite(mb) else 1e100

def _expseq(n, al):
    x = np.linspace(0,1,n)
    return _norm(np.exp(-al*x))

def _mixseq(n, al, g):
    x = np.linspace(0,1,n)
    return _norm(np.exp(-al*x)+g*np.exp(-al*(1-x))+0.08)

def _smrand(n, rng, base, amp):
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cut = max(2, n//20+2)
    if cut < len(F): F[cut:] = 0
    f = np.fft.irfft(F, n)
    f = (f-f.mean())/max(f.std(),1e-10)
    return _norm(base*(1+amp*f))

def _rsmp(a, nn):
    a = _norm(a)
    if a.size == nn: return a.copy()
    xo = np.linspace(0,1,a.size)
    xn = np.linspace(0,1,nn)
    return _norm(np.interp(xn, xo, a))

def _fpert(a, rng, st):
    n = a.size
    fs = _npow2(n)
    A = np.fft.rfft(a, fs)
    lo = max(1, n//64)
    hi = len(A) - max(1, n//64)
    P = A.copy()
    P[lo:hi] += st*(rng.normal(size=hi-lo) + 1j*rng.normal(size=hi-lo))
    a2 = np.fft.irfft(P, fs)[:n]
    return _norm(a2)

def _bpert(a, rng, st):
    n = a.size
    c = a.copy()
    w = max(2, n//20)
    s = int(rng.integers(0, max(1, n-w)))
    c[s:s+w] *= np.exp(rng.normal(0, st))
    if rng.random() < 0.2:
        c = 0.5*(c+c[::-1])
    return _norm(c)

def _pert(a, rng):
    r = rng.random()
    if r < 0.3: return _fpert(a, rng, 0.08)
    elif r < 0.6: return _bpert(a, rng, 0.2)
    else:
        noise = np.exp(rng.normal(0, 0.15, size=a.size))
        return _norm(a*noise)

def _refine(a, dl, rng):
    a = _norm(a)
    n = a.size
    bfs = _npow2(2*n-1)
    gfs = _npow2(3*n-1)
    b = _acorr(a, bfs)
    mb = b.max()
    sc = 2*n*mb
    ba = a.copy()
    bs = sc
    beta = 80.0
    al = 0.12
    it = 0
    fail = 0
    li = 0
    best_p = None
    
    while time.monotonic() < dl:
        if mb < 1e-300: mb = 1e-300
        bt = min(1e7, max(1.0, beta/max(mb, 1e-10)))
        p, lo = _lse_p(b, bt)
        g = _wgrad(a, p, gfs)
        gc = g - g.mean()
        sc_ = gc.max() - gc.min()
        
        if sc_ < 1e-15:
            if time.monotonic() < dl - 1:
                a = _pert(ba, rng)
                b = _acorr(a, bfs)
                mb = b.max()
                sc = 2*n*mb
                al = 0.12
                fail = 0
            it += 1
            continue
        
        # Adaptive step size
        step = al * 0.5 / (1 + sc_*0.1)
        v = a - step*gc
        a = _norm(v)
        
        b = _acorr(a, bfs)
        mb = b.max()
        sc = 2*n*mb
        ln = _lse_p(b, bt)[1]
        
        if ln < lo - 1e-13:
            al = min(3.0, al*1.18)
            fail = 0
            li = it
        else:
            al *= 0.45
            fail += 1
        
        if sc < bs - 1e-14:
            ba = a.copy()
            bs = sc
            li = it
        
        # Increase beta
        if it % 80 == 0:
            beta *= 1.6
        
        # Restart
        if (fail > 400 or (it-li > 800 and it > 200)) and time.monotonic() < dl-3:
            a = _pert(ba, rng)
            b = _acorr(a, bfs)
            mb = b.max()
            sc = 2*n*mb
            al = 0.1
            fail = 0
            beta = max(40, beta*0.4)
        
        # Reset if too far
        if fail > 1000 and sc > 1.05*bs + 0.02:
            a = ba.copy()
            b = _acorr(a, bfs)
            mb = b.max()
            sc = 2*n*mb
            beta = 50.0
            al = 0.1
            fail = 0
        
        it += 1
    return ba, bs

def _coord_desc(a, dl, rng):
    """Simple coordinate descent: reduce coordinates that contribute to peak."""
    a = _norm(a)
    n = a.size
    bfs = _npow2(2*n-1)
    gfs = _npow2(3*n-1)
    b = _acorr(a, bfs)
    mb = b.max()
    sc = 2*n*mb
    ba = a.copy()
    bs = sc
    
    sweep = 0
    while time.monotonic() < dl:
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > dl: break
            # Reduce a[i] slightly and see if score improves
            old = a[i]
            for frac in [0.5, 0.2, 0.05, -0.1]:
                a[i] = max(0, old*(1+frac))
                if a.sum() < 1e-10:
                    a[i] = old
                    continue
                a = _norm(a)
                b2 = _acorr(a, bfs)
                mb2 = b2.max()
                sc2 = 2*n*mb2
                if sc2 < sc - 1e-12:
                    b = b2
                    mb = mb2
                    sc = sc2
                    if sc < bs:
                        ba = a.copy()
                        bs = sc
                    break
                a[i] = 0  # reset, will re-normalize
            a = _norm(a) if a.sum() > 1e-10 else ba.copy()
            b = _acorr(a, bfs)
            mb = b.max()
            sc = 2*n*mb
        sweep += 1
    return ba, bs
