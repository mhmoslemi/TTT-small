import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = (u - (cssv - 1.0) / rho) > 0
    if not cond.any():
        return np.full(n, 1.0/n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv_and_grad(y, max_active=200):
    """Compute convolution, max, and subgradient at y."""
    n = len(y)
    conv = np.convolve(y, y)
    M = conv.max()
    
    # Find active peaks
    if max_active >= len(conv):
        active = np.arange(len(conv))
    else:
        # Get peaks within tolerance of max
        tol = 1e-8 * M
        active_mask = conv >= M - tol
        active = np.where(active_mask)[0]
        if len(active) > max_active:
            # Use top max_active
            active = np.argsort(conv)[-max_active:]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            idx = k - np.arange(lo, hi)
            g[lo:hi] += y[idx]
    g /= len(active)
    
    return conv, M, g

    n = len(a)
    s = a.sum()
    if s < 0.01:

    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

    a_prev = a0.copy()
    t_k = 1.0

    deadline = time.monotonic() + duration
    it = 0

    while time.monotonic() < deadline:

        # Gradient step + project

        a_prev = a

            if obj < best_obj - 1e-14:

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - (cssv - 1.0) / rho) > 0
    if not cond.any():
        return np.full(n, 1.0 / n)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute subgradient of max(conv) at y using active peaks."""
    n = len(y)
    M = conv.max()
    # Active peaks: within relative tolerance
    tol = max(1e-12, 1e-6 * M)
    active = np.where(conv >= M - tol)[0]

        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += y[k - np.arange(lo, hi)]
    g /= len(active)
    return g, M, len(active)

        # This equals a^T D_k a where D_k[i,j] = 1 if i+j=k, else 0
        # But D_k is not symmetric in general... wait, D_k[i,j] = 1 iff i+j=k
        # D_k[j,i] = 1 iff j+i=k, same condition. So D_k IS symmetric.
        # And a^T D_k a = sum_{i,j: i+j=k} a[i]*a[j] = (a*a)[k]. Good.
        
        # Build D_k as a sparse matrix
        # D_k has entries at positions (i, k-i) for valid i
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        rows = list(range(lo, hi + 1))
        cols = [k - r for r in rows]
        vals = [1.0] * len(rows)
        
        from scipy.sparse import csr_matrix
        D_k = csr_matrix((vals, (rows, cols)), shape=(n, n))
        # Make symmetric (it should already be, but let's ensure)
        # Actually, D_k is already symmetric since if (i, k-i) is in D_k, so is (k-i, i)
        # Let me verify: D_k[i, k-i] = 1 and D_k[k-i, i] = 1. Yes, symmetric.
        
        terms.append(a.T @ D_k @ a)
    
    # Minimize the maximum
    t = cp.Variable()
    constraints += [term <= t for term in terms]
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, max_iters=10000)
    
    if a.value is not None:
        return np.maximum(a.value, 0)
    return a_init

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1, dtype=np.float64)
    cond = (u - (cssv - 1.0)/rho) > 0
    if not cond.any():
        return np.full(n, 1.0/n)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2*n - 1

    fa = np.fft.rfft(a, ft)
    return np.fft.irfft(fa*fa, ft)[:size]
