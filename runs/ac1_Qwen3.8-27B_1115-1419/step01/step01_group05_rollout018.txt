```python
"""
Search for low-peak nonnegative self-convolutions.

The objective is
    score(a) = 2 n max_k (a*a)_k / (sum a)^2 ,          a_i >= 0.
It is scale invariant, so all sequences are normalized to sum 1 during search.

Algorithm:
  * Use several cheap parametric/random starts, including the previous best
    global sequence when available.
  * Optimize a smooth log-sum-exp surrogate of max(a*a) by line-search
    gradient descent in log-space (softmax parameterization).
  * Compute gradients with FFT-based convolution/correlation.
  * Periodically use the LP best-response from the old state-of-the-art
    method (max sum g subject to a*g <= M) as a global/escape move.
  * Periodically resample to a larger length, optimize there, then resample
    back to the output length.  The output is always at the original length
    when the supplied global sequence has a reasonable length.
"""

import math
import time

import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _normalize(a):
    """Return a nonnegative array proportional to a and with sum 1."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        arr = np.ones(1, dtype=np.float64)

    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)

    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.where(arr > 0.0, arr, 0.0)

    s = float(arr.sum())
    if not math.isfinite(s) or s <= 1e-300:
        n = max(1, int(arr.size))
        return np.ones(n, dtype=np.float64) / float(n)

    return arr / s


def _clean_global_sequence(x):
    """Clean a global sequence.  Return None if unusable."""
    if x is None:
        return None
    arr = _normalize(x)
    if arr.size < 4:
        return None
    return arr


def _make_uniform(n):
    n = max(2, int(n))
    return np.ones(n, dtype=np.float64) / float(n)


def _make_exponential(n, alpha, floor=0.0):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x) + float(floor)
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _make_mixture(n, alpha, gamma, floor=0.0):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x) + float(gamma) * np.exp(-float(alpha) * (1.0 - x)) + float(floor)
    a = np.maximum(a, 1e-15)
    return _normalize(a)


def _resample_sequence(old, new_n):
    """Interpolatory resampling on normalized nonnegative sequences."""
    old = _normalize(old)
    new_n = max(2, int(new_n))

    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _conv_fft(u, v):
    """Linear convolution by zero-padded FFT."""
    u = np.asarray(u, dtype=np.float64).ravel()
    v = np.asarray(v, dtype=np.float64).ravel()
    if u.size == 0 or v.size == 0:
        return np.empty(0, dtype=np.float64)
    L = 1 << (u.size + v.size - 2).bit_length()
    U = np.fft.rfft(u, L)
    V = np.fft.rfft(v, L)
    c = np.fft.irfft(U * V, L).real
    return c[:u.size + v.size - 1]


def _score_of(a, use_fft=True):
    """Fast approximate score; matches evaluator up to FFT roundoff."""
    a = _normalize(a)
    n = int(a.size)
    if n == 0:
        return float("inf")

    if not use_fft:
        b = np.convolve(a, a)
    else:
        b = _conv_fft(a, a)

    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    b = np.where(b > 0.0, b, 0.0)

    if b.size == 0:
        return float("inf")

    mb = float(np.max(b))
    if not math.isfinite(mb):
        return float("inf")

    return 2.0 * float(n) * mb


def _smooth_perturb(a, rng, amp=0.25):
    """Add low-frequency smooth multiplicative noise."""
    a = _normalize(a)
    n = a.size
    if n < 2:
        return a

    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(F.size, n // 6 + 2))
    if cutoff < F.size:
        F = F.copy()
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n).real

    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 1e-12:
        f = f / sd

    c = a * (1.0 + float(amp) * f)
    c = np.maximum(c, 1e-12)
    return _normalize(c)


def _block_perturb(a, rng, width_frac=0.12, strength=0.25):
    """Multiply a random block by a random positive factor."""
    a = _normalize(a)
    n = a.size
    c = a.copy()

    w = max(2, int(n * max(0.01, float(width_frac))))
    w = min(w, n)
    stop = max(1, n - w + 1)
    s = int(rng.integers(0, stop))
    stop_min = min(n, s + w)

    c[s:stop_min] *= np.exp(rng.normal(0.0, float(strength), stop_min - s))

    if rng.random() < 0.30:
        c = 0.5 * (c + c[::-1])

    return _normalize(c)


def _candidate_shapes(n, rng):
    """Deterministic and random smooth starts."""
    out = []
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    alphas = (0.3, 0.5, 0.7, 1.0, 1.4, 2.0, 3.0, 5.0)
    for al in alphas:
        out.append(_make_exponential(n, al))

    mixes = (
        (1.0, 0.3, 0.05),
        (1.0, 0.5, 0.10),
        (1.0, 0.7, 0.15),
        (1.0, 1.0, 0.20),
        (0.7, 0.7, 0.10),
        (1.5, 0.5, 0.25),
        (2.5, 0.5, 0.50),
    )
    for al, ga, fl in mixes:
        out.append(_make_mixture(n, al, ga, fl))

    # Polygons, including non-symmetric monotane variants.
    for beta in (-0.6, -0.3, 0.0, 0.3, 0.6):
        for gamma in (-0.3, 0.0, 0.3):
            shape = 1.0 + beta * x + gamma * x * x
            if float(np.min(shape)) > 1e-4:
                out.append(_normalize(shape))

    # Small trigonometric perturbations; useful for breaking plateaus.
    for m in (1, 2, 3):
        for amp in (0.2, 0.5, 0.75):
            for ph in (0.0, 0.5, 1.0, 1.5):
                shape = 1.0 + amp * np.cos(2.0 * math.pi * m * x + ph)
                if float(np.min(shape)) > 1e-4:
                    out.append(_normalize(shape))

    # Power windows, including tapered one-sided windows.
    powers = (0.5, 0.7, 1.3, 2.0, 3.0)
    for p in powers:
        out.append(_normalize(np.maximum((1.0 - x) ** p, 1e-12)))
        out.append(_normalize(np.maximum(x ** p, 1e-12)))
        for fl in (0.05, 0.2):
            out.append(_normalize(np.maximum((1.0 - x) ** p + fl, 1e-12)))
            out.append(_normalize(np.maximum(x ** p + fl, 1e-12)))

    base1 = _make_exponential(n, 1.0)
    base2 = _make_uniform(n)
    for base in (base1, base2):
        out.append(_smooth_perturb(base, rng, 0.30))
        out.append(_smooth_perturb(base, rng, 0.60))

    out.append(_block_perturb(base1, rng, 0.12, 0.20))
    out.append(_block_perturb(base2, rng, 0.10, 0.20))

    # A few Dirichlet starts; these can provide ragged local structure.
    for conc in (0.3, 1.0, 5.0):
        try:
            d = rng.dirichlet(np.ones(n, dtype=np.float64) * conc)
            out.append(_normalize(d))
        except Exception:
            pass

    # Symmetrize a few early starts.
    limited = out[:25]
    for arr in limited:
        s = 0.5 * (arr + arr[::-1])
        out.append(_normalize(s))

    return out


def _softmax(theta):
    """Stable softmax with simple recovery."""
    theta = np.asarray(theta, dtype=np.float64).ravel()
    if theta.size == 0:
        return np.ones(1, dtype=np.float64)

    if not np.all(np.isfinite(theta)):
        n = theta.size
        return _make_uniform(n)

    m = float(theta.max())
    e = np.exp(theta - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return _make_uniform(theta.size)

    return e / s


def _seq_to_theta(a):
    """Represent a positive simplex sequence as log-coefficients."""
    a = _normalize(a)
    n = a.size
    if n == 0:
        return np.zeros(1, dtype=np.float64)

    floor = 1e-12 / float(n)
    a = np.maximum(a, floor)
    a = a / float(a.sum())
    theta = np.log(a)
    theta = theta - float(theta.mean())
    return theta


def _lse_stats(b, temp):
    """
    Return value, probability weights p, and max of LSE(b/temp) style smooth
    maximum.  Here val = max(b) + temp*log(sum(exp((b-max(b))/temp))).
    """
    b = np.asarray(b, dtype=np.float64).ravel()
    if b.size == 0:
        return 0.0, np.empty(0, dtype=np.float64), 0.0

    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    b = np.where(b > 0.0, b, 0.0)

    m = float(np.max(b))
    if not math.isfinite(m) or m <= 1e-300:
        p = np.zeros(b.size, dtype=np.float64)
        p[0] = 1.0
        return 0.0, p, 0.0

    if temp <= 1e-300:
        p = np.zeros(b.size, dtype=np.float64)
        idx = int(np.argmax(b))
        p[idx] = 1.0
        return m, p, m

    z = (b - m) / temp
    z = np.where(z < -745.0, -745.0, z)
    e = np.exp(z)
    s = float(e.sum())

    if not math.isfinite(s) or s <= 1e-300:
        p = np.zeros(b.size, dtype=np.float64)
        idx = int(np.argmax(b))
        p[idx] = 1.0
        return m, p, m

    p = e / s
    val = m + temp * math.log(s)
    return val, p, m


def _grad_from_weights(a, p):
    """
    For weights p on convolution lags, compute
        v_i = sum_k p_k a_{k-i}
    which is one half of the gradient of sum_k p_k (a*a)_k.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    lp = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if lp == 0:
        return np.zeros(n, dtype=np.float64)

    q = p[::-1]
    L = 1 << (n + lp - 2).bit_length()
    A = np.fft.rfft(a, L)
    Q = np.fft.rfft(q, L)
    c = np.fft.irfft(A * Q, L).real

    S = 2 * n - 2
    start = S - n + 1
    end = S + 1

    if start < 0:
        start = 0
    if end > c.size:
        end = c.size
    if start >= end:
        return np.zeros(n, dtype=np.float64)

    v = c[start:end][::-1]
    if v.size > n:
        v = v[:n]
    elif v.size < n:
        tmp = np.zeros(n, dtype=np.float64)
        tmp[:v.size] = v
        v = tmp

    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    v = np.where(v > 0.0, v, 0.0)
    return v


def _lp_response(a, max_len=1024):
    """
    Solve the old SOTA LP best-response:
        maximize sum(g)
        subject to (a*g)_k <= M,   g >= 0,
    where M = max(a*a).  Return g normalized to sum 1, or None.
    """
    if linprog is None:
        return None

    a = _normalize(a)
    n = int(a.size)
    if n < 16 or n > int(max_len):
        return None

    b = _conv_fft(a, a)
    M = float(np.max(b))
    if not math.isfinite(M) or M <= 1e-300:
        return None

    # Constraint matrix A has A[k,j] = a_{k-j} when valid.
    A = np.empty((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        # For j in [lo, hi], coefficient is a[k-j].
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    A = A / M
    b_ub = np.ones(2 * n - 1, dtype=np.float64)
    cvec = -np.ones(n, dtype=np.float64)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 10.0))

    t0 = time.monotonic()
    try:
        res = linprog(cvec, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(cvec, A_ub=A, b_ub=b_ub, bounds=bounds)
        except Exception:
            return None

    if not bool(getattr(res, "success", False)):
        return None

    # Optional safety: if this was extremely slow, do not use it.
    if time.monotonic() - t0 > 30.0:
        return None

    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g = np.where(g > 0.0, g, 0.0)
    s = float(g.sum())
    if not math.isfinite(s) or s <= 1e-15:
        return None

    return g / s


def _optimize_smooth(a, deadline, rng, tag=""):
    """
    Smooth log-sum-exp gradient descent in log-space, with LP best-response
    moves and restarts.  Returns (best_sequence, best_score).
    """
    a = _normalize(a)
    n = int(a.size)
    if n < 2:
        return a, _score_of(a, use_fft=True)

    best_a = a.copy()
    b = _conv_fft(a, a)
    best_score = 2.0 * float(n) * float(np.max(b))

    theta = _seq_to_theta(a)
    soft = 6.0
    step = 0.12
    fail = 0
    it = 0
    last_lp = time.monotonic() - 2.0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        maxb = float(np.max(b))
        if not math.isfinite(maxb) or maxb <= 1e-300:
            maxb = 1e-300

        temp = maxb / float(soft)
        val, p, mb = _lse_stats(b, temp)
        if mb <= 0 and maxb > 0:
            mb = maxb
        maxb = mb

        v = _grad_from_weights(a, p)
        mu = float(np.dot(p, b))
        g = a * (v - mu)
        g = g - float(np.mean(g))

        rms = float(np.sqrt(np.mean(g * g)))
        if (not math.isfinite(rms)) or rms < 1e-14:
            a = _smooth_perturb(best_a, rng, 0.20)
            theta = _seq_to_theta(a)
            b = _conv_fft(a, a)
            soft = max(5.0, 0.7 * soft)
            step = 0.12
            fail = 0
            it += 1
            continue

        direction = g / rms
        val_old = val
        accepted = False
        improved_val = False
        s = step
        trials = 0

        while s > 1e-7 and trials < 25 and time.monotonic() < deadline - 0.1:
            theta_trial = theta - s * direction
            a_trial = _softmax(theta_trial)

            # Guard against numeric collapse.
            if float(a_trial.sum()) <= 1e-300:
                s *= 0.5
                trials += 1
                continue

            b_trial = _conv_fft(a_trial, a_trial)
            maxb_trial = float(np.max(b_trial))
            if not math.isfinite(maxb_trial):
                s *= 0.5
                trials += 1
                continue

            val_trial, _, mb_trial = _lse_stats(b_trial, temp)
            score_trial = 2.0 * float(n) * mb_trial
            trials += 1

            tol_val = 1e-13 * max(1.0, abs(val))
            accept = False
            if val_trial < val - tol_val:
                accept = True
                improved_val = val_trial < val_old
            elif score_trial < best_score - 1e-12:
                # Allow exact improvement even if surrogate is slightly worse.
                if val_trial < val + max(1e-10, 0.1 * abs(val)):
                    accept = True
                    improved_val = False

            if accept:
                theta = theta_trial
                a = a_trial
                b = b_trial
                maxb = mb_trial
                val = val_trial
                accepted = True

                if score_trial < best_score:
                    best_a = a_trial.copy()
                    best_score = score_trial
                break

            s *= 0.5

        if accepted:
            if improved_val:
                step = min(1.60, step * 1.12)
            else:
                step = min(1.60, step * 1.03)
            fail = 0
        else:
            step = max(1e-5, step * 0.55)
            fail += 1

        it += 1

        if it % 50 == 0:
            theta = theta - float(theta.mean())

        # Sharpen smoothly, but not too fast.
        if it % 80 == 0:
            soft = min(30000.0, soft * 1.07)

        if fail > 80 and it % 15 == 0:
            soft = max(5.0, soft * 0.80)

        score_now = 2.0 * float(n) * float(np.max(b))

        # Restart if stuck.
        stuck = (
            fail > 450
            or (it % 400 == 0 and score_now > best_score * 1.02 + 1e-12)
        )
        if stuck and time.monotonic() < deadline - 5.0:
            amp = 0.18
            if fail > 1200:
                amp = 0.34
            a = _smooth_perturb(best_a, rng, amp)
            theta = _seq_to_theta(a)
            b = _conv_fft(a, a)
            soft = max(5.0, 0.5 * soft)
            step = 0.12
            fail = 0

        # Periodic LP best-response move.
        if (
            linprog is not None
            and it > 80
            and time.monotonic() - last_lp > 2.0
            and time.monotonic() < deadline - 5.0
        ):
            last_lp = time.monotonic()

            score_now_lp = 2.0 * float(n) * float(np.max(b))
            lp_src = a
            if score_now_lp > best_score * 1.05:
                lp_src = best_a

            g = _lp_response(lp_src, max_len=1024)
            if g is not None:
                maxb_b = float(np.max(b))
                score_b = 2.0 * float(n) * float(maxb_b)

                for tt in (0.10, 0.20, 0.35, 0.05):
                    mix = _normalize((1.0 - tt) * lp_src + tt * g)
                    mix_score = _score_of(mix, use_fft=True)
                    if not math.isfinite(mix_score):
                        continue

                    improve_best = mix_score < best_score - 1e-13
                    improve_current = mix_score < score_b - 1e-14

                    if improve_best or improve_current:
                        a = mix
                        theta = _seq_to_theta(a)
                        b = _conv_fft(a, a)
                        step = 0.10
                        soft = max(5.0, 0.7 * soft)
                        fail = 0

                        if improve_best:
                            best_a = a.copy()
                            best_score = mix_score
                        break

        if it % 1000 == 0 and time.monotonic() - last_print > 45.0:
            last_print = time.monotonic()
            print(
                f"optimize[{tag}] n={n} it={it} soft={soft:.1f} "
                f"score={score_now:.8f} best={best_score:.8f}",
                flush=True,
            )

    best_a = _normalize(best_a)
    # Recompute with a slightly exact score (still FFT for speed).
    best_score = _score_of(best_a, use_fft=True)
    return best_a, float(best_score)


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main search entry point.  Runs until near the supplied time budget and
    returns a nonnegative list with the best score found.
    """
    t0 = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0

    if not math.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = min(budget, 2400.0)

    if budget < 20.0:
        budget = 20.0

    reserve = 25.0 if budget > 300.0 else 5.0
    total_deadline = t0 + max(1.0, budget - reserve)

    rng = np.random.default_rng(seed)

    original_n = 256
    given = None

    try:
        gseq = globals().get("height_sequence_1", None)
    except Exception:
        gseq = None

    if gseq is not None:
        arr = _clean_global_sequence(gseq)
        if arr is not None:
            original_n = int(arr.size)
            given = arr

    n_out = original_n
    if n_out < 16:
        n_out = 256
    if n_out > 4096:
        # Preserve a computational upper bound.  The final output is still
        # resampled to the original length if one was provided.
        n_out = 2048
        if given is not None:
            given = _resample_sequence(given, n_out)

    # Deterministic/random start material at output length.
    shape_list = _candidate_shapes(n_out, rng)

    scored = []
    seen = []
    for arr in shape_list:
        arr = _resample_sequence(arr, n_out)
        sc = _score_of(arr, use_fft=True)
        if not math.isfinite(sc):
            continue
        # Rough duplicate rejection by coarse fingerprint.
        fp = tuple(arr[::max(1, arr.size // 64)].astype(np.int64))
        if fp in seen:
            continue
        seen.append(fp)
        scored.append((sc, arr))

    vals = np.empty(len(scored), dtype=np.float64)
    for i, item in enumerate(scored):
        vals[i] = item[0]
    if vals.size:
        order = np.argsort(vals)
        scored_ordered = []
        for idx in order:
            scored_ordered.append(scored[int(idx)])
        scored = scored_ordered
    else:
        scored = []

    if given is not None:
        given = _resample_sequence(given, n_out)
        best_a = given.copy()
        best_score = _score_of(given, use_fft=True)
    else:
        best_a = None
        best_score = float("inf")

    start_list = []
    if given is not None:
        start_list.append(given)

    for sc, arr in scored:
        if len(start_list) >= 4:
            break
        start_list.append(arr)

    if best_a is None or best_score == float("inf"):
        if start_list:
            best_a = start_list[0].copy()
            best_score = _score_of(best_a, use_fft=True)
        else:
            best_a = _make_exponential(n_out, 1.0)
            best_score = _score_of(best_a, use_fft=True)
            start_list.append(best_a)

    if len(start_list) == 0:
        best_a = _make_exponential(n_out, 1.0)
        best_score = _score_of(best_a, use_fft=True)
        start_list.append(best_a)

    print(
        f"search start n={n_out} initial_best={best_score:.8f} "
        f"candidates={len(scored)}",
        flush=True,
    )

    # Phase A: try several starts at the output length.
    if budget > 60.0:
        phaseA_deadline = min(total_deadline, t0 + max(30.0, 0.50 * budget))
    else:
        phaseA_deadline = total_deadline - 1.0

    if phaseA_deadline > time.monotonic() + 5.0:
        for i, st in enumerate(start_list):
            remaining_count = len(start_list) - i
            if remaining_count <= 0:
                break
            slice_time = (phaseA_deadline - time.monotonic() - 5.0) / float(remaining_count)
            if slice_time < 3.0:
                break
            end = min(phaseA_deadline - 0.5, time.monotonic() + slice_time)
            if end <= time.monotonic() + 1.0:
                break

            res_a, res_score = _optimize_smooth(st.copy(), end, rng, tag=f"start{i}")
            if math.isfinite(res_score) and res_score < best_score:
                best_a = _resample_sequence(res_a, n_out)
                best_score = float(res_score)
                print(
                    f"phaseA start{i} n={n_out} score={best_score:.8f} "
                    f"t={time.monotonic()-t0:.1f}s",
                    flush=True,
                )

    # Plan remaining time: deep refine at output length, optional upscale
    # detour, then final polish.
    now = time.monotonic()
    rem = max(0.0, total_deadline - now)

    upsize_budget = 0.0
    if n_out <= 1024 and rem > 500.0:
        upsize_budget = min(360.0, 0.25 * rem)

    deep_budget = max(0.0, rem - upsize_budget - 240.0)
    deep_budget = min(deep_budget, 1200.0)

    if deep_budget > 20.0:
        end = min(total_deadline - upsize_budget - 240.0, time.monotonic() + deep_budget)
        if end > time.monotonic() + 5.0:
            res_a, res_score = _optimize_smooth(best_a.copy(), end, rng, tag="deep")
            if math.isfinite(res_score) and res_score < best_score:
                best_a = _resample_sequence(res_a, n_out)
                best_score = float(res_score)
                print(
                    f"deep n={n_out} score={best_score:.8f} "
                    f"t={time.monotonic()-t0:.1f}s",
                    flush=True,
                )

    # Upscale detour: optimize a longer interpolation, then resample back.
    # This often finds a better low-frequency shape and then returns it to
    # the output length.
    if upsize_budget > 60.0 and n_out <= 1024:
        now = time.monotonic()
        avail = max(0.0, total_deadline - now - 120.0)
        if avail > 30.0:
            n_up = int(min(2048, max(512, 2 * n_out)))
            if n_up > n_out:
                init_up = _resample_sequence(best_a, n_up)
                exp_mix = _make_exponential(n_up, 1.0)
                init_up = _normalize(0.80 * init_up + 0.20 * exp_mix)
                init_up = _smooth_perturb(init_up, rng, 0.15)

                end = min(total_deadline - 30.0, time.monotonic() + min(upsize_budget, avail))
                if end > time.monotonic() + 5.0:
                    res_up, res_score_up = _optimize_smooth(init_up, end, rng, tag="up")
                    back = _resample_sequence(res_up, n_out)
                    back_score = _score_of(back, use_fft=True)
                    if math.isfinite(back_score) and back_score < best_score:
                        best_a = _resample_sequence(back, n_out)
                        best_score = float(back_score)
                        print(
                            f"upsize {n_out}->{n_up} back n={n_out} "
                            f"score={best_score:.8f} t={time.monotonic()-t0:.1f}s",
                            flush=True,
                        )

    # Final polish.
    now = time.monotonic()
    if total_deadline - now > 10.0:
        end = total_deadline - 1.0
        res_a, res_score = _optimize_smooth(best_a.copy(), end, rng, tag="final")
        if math.isfinite(res_score) and res_score < best_score:
            best_a = _resample_sequence(res_a, n_out)
            best_score = float(res_score)
            print(f"final n={n_out} score={best_score:.8f}", flush=True)
        else:
            print(f"final no improvement n={n_out} best={best_score:.8f}", flush=True)

    # Final exact-ish recomputation at the chosen length.
    best_a = _resample_sequence(best_a, n_out)
    best_a = _normalize(best_a)
    if best_a.size == 0:
        best_a = _make_exponential(n_out, 1.0)

    exact_score = _score_of(best_a, use_fft=False)
    if math.isfinite(exact_score):
        best_score = float(exact_score)

    out_list = [float(x) for x in best_a]

    # If the official evaluator is present, validate the final list.
    ev_val = float("inf")
    try:
        ev_val = float(evaluate_sequence(out_list))
    except Exception:
        ev_val = float(_score_of(best_a, use_fft=False))

    if not math.isfinite(ev_val) or ev_val > 10.0:
        # Robust fallback to a simple non-pathological sequence.
        best_a = _make_exponential(n_out, 1.0)
        out_list = [float(x) for x in best_a]
        try:
            ev_val = float(evaluate_sequence(out_list))
        except Exception:
            ev_val = float(_score_of(best_a, use_fft=False))

    print(
        f"final_return len={len(out_list)} eval={ev_val:.10f} "
        f"internal={_score_of(best_a, use_fft=False):.10f} "
        f"time={time.monotonic()-t0:.1f}s",
        flush=True,
    )

    return out_list
```
