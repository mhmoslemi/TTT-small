```python
def lp_direction(f, M):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # For each lag k, sum_i f[i] * g[k-i] <= M

    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve()
```

```python
def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize array to target sum."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _compute_grad_lse(a, temp):
    """
    Compute gradient of LSE-smoothed max of conv(a,a) w.r.t. a.
    Returns (gradient, lse_value, softmax_weights).
    """
    n = len(a)
    c = _fft_conv(a, a)  # length 2n-1
    c_max = float(np.max(c))
    exp_c = np.exp((c - c_max) / temp)
    softmax = exp_c / np.sum(exp_c)
    lse_val = c_max + temp * np.log(np.sum(exp_c))
    
    # Compute gradient via FFT
    # grad[j] = 2 * sum_{m=0}^{n-1} softmax[j+m] * a[m]
    # = 2 * conv(v, s_rev)[2n-2-j] where v = a padded to 2n-1, s_rev = softmax[::-1]
    v = np.zeros(2 * n - 1)
    v[:n] = a
    s_rev = softmax[::-1]
    N = 1 << (len(v) + len(s_rev) - 1).bit_length()
    conv_result = ifft(fft(v, N) * fft(s_rev, N)).real
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    
    return grad, lse_val, softmax, c


def _adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8, target_sum=None):
    """One Adam step with projection and normalization."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    if target_sum is not None:
        a_new = _normalize(a_new, target_sum)
    return a_new, m, v
```

```python
def _sparse_lp_direction(f, M, K=100):
    """
    Solve sparse LP: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for top-K lags.
    Returns the optimal g.
    """
    import cvxpy as cp
    from scipy.sparse import csr_matrix
    
    n = len(f)
    c = _fft_conv(f, f)
    
    # Get top-K lag indices
    top_lags = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix for these lags
    # For lag k: conv(f,g)[k] = sum_i f[i] * g[k-i] (valid indices)
    # This is a row of the convolution matrix: T[k,:] @ g
    # T[k, j] = f[k-j] if 0 <= k-j < n
    
    rows_data = []
    cols_data = []
    vals_data = []
    
    for idx, k in enumerate(top_lags):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows_data.append(idx)
                cols_data.append(j)
                vals_data.append(f[i])
    
    T = csr_matrix((vals_data, (rows_data, cols_data)), shape=(K, n))
    
    # Solve LP
    g = cp.Variable(n)
    constraints = [T @ g <= M * np.ones(K), g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except:
        pass
    
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except:
        pass
    
    return None
```

```python
"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Sparse LP-based direction finding (top-K convolution constraints)
2. Adam optimizer with log-sum-exp smoothed objective
3. Temperature annealing and diverse restarts
4. Local peak-reduction heuristics
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    """Fast score computation using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize array to target sum, preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _compute_grad_lse(a, temp):
    """
    Compute gradient of LSE-smoothed max of conv(a,a) w.r.t. a.
    Returns (gradient, conv_values).
    """
    n = len(a)
    c = _fft_conv(a, a)
    c_max = float(np.max(c))
    exp_c = np.exp((c - c_max) / temp)
    softmax = exp_c / np.sum(exp_c)

    # grad[j] = 2 * sum_{m=0}^{n-1} softmax[j+m] * a[m]
    # = 2 * conv(v, s_rev)[2n-2-j]
    v = np.zeros(2 * n - 1)
    v[:n] = a
    s_rev = softmax[::-1]
    N = 1 << (len(v) + len(s_rev) - 1).bit_length()
    conv_result = ifft(fft(v, N) * fft(s_rev, N)).real
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]

    return grad, c


def _run_adam_lse(a, deadline, target_sum, temp_init=0.5, temp_min=0.01,
                  lr=0.003, eval_interval=300, rng=None):
    """Run Adam with LSE smoothing and temperature annealing."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best = a.copy()
    best_score = _score_fast(a)
    stuck = 0
    temp = temp_init
    # Estimate total steps
    total_steps_est = 200000
    temp_factor = (temp_min / temp_init) ** (1.0 / max(10, total_steps_est / 10))

    while time.monotonic() < deadline:
        grad, c = _compute_grad_lse(a, temp)
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = np.maximum(a, 0.0)
        a = _normalize(a, target_sum)
        lr = max(lr * 0.999995, 1e-7)
        temp = max(temp * temp_factor, temp_min)

        if t % eval_interval == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-9:
                best_score = sc
                best = a.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 25:
                a = best + 0.02 * rng.standard_normal(n) * target_sum / np.sqrt(n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, target_sum)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stuck = 0
                temp = max(temp * 10, temp_min)

    return best, best_score


def _sparse_lp_direction(f, M, K=80):
    """
    Solve sparse LP: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for top-K lags.
    Returns optimal g or None.
    """
    n = len(f)
    c = _fft_conv(f, f)
    top_lags = np.argsort(c)[::-1][:K]

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_lags):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            rows.append(idx)
            cols.append(j)
            vals.append(f[i])

    if len(rows) == 0:
        return None
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n))
    b_ub = M * np.ones(K)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _run_lp_optimization(a, deadline, target_sum, K=80, max_iters=500, rng=None):
    """
    LP-based alternating optimization.
    At each step: find direction g via sparse LP, then step towards g.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    best = a.copy()
    best_score = _score_fast(a)
    current = a.copy()
    stuck = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        c = _fft_conv(current, current)
        M = float(np.max(c))

        g = _sparse_lp_direction(current, M, K=K)
        if g is None or np.sum(g) < 1e-10:
            break

        g = _normalize(g, target_sum)

        # Try different step sizes
        improved = False
        for t_frac in [0.5, 0.3, 0.2, 0.1, 0.05]:
            candidate = (1 - t_frac) * current + t_frac * g
            candidate = np.maximum(candidate, 0.0)
            candidate = _normalize(candidate, target_sum)
            sc = _score_fast(candidate)
            if sc < _score_fast(current) - 1e-10:
                current = candidate
                improved = True
                break

        if improved:
            sc = _score_fast(current)
            if sc < best_score - 1e-9:
                best_score = sc
                best = current.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
            # Perturb and retry
            if stuck > 10:
                current = best + 0.03 * rng.standard_normal(n) * target_sum / np.sqrt(n)
                current = np.maximum(current, 0.0)
                current = _normalize(current, target_sum)
                stuck = 0

        it += 1

    return best, best_score


def _local_peak_reduction(a, target_sum, n_attempts=200, rng=None):
    """Try to reduce the peak by adjusting individual coordinates."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    best = a.copy()
    best_score = _score_fast(a)

    for _ in range(n_attempts):
        c = _fft_conv(best, best)
        peak_idx = int(np.argmax(c))

        # Find contributing pairs
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)
        pairs = []
        for i in range(i_lo, i_hi + 1):
            j = peak_idx - i
            pairs.append((best[i] * best[j], i, j))
        pairs.sort(reverse=True)

        improved = False
        for contrib, i, j in pairs[:5]:
            if contrib < 1e-10:
                break
            # Try reducing the larger of a[i], a[j]
            if best[i] >= best[j]:
                reduce_idx = i
                other_idx = j
            else:
                reduce_idx = j
                other_idx = i

            for frac in [0.01, 0.02, 0.05]:
                reduction = frac * best[reduce_idx]
                candidate = best.copy()
                candidate[reduce_idx] -= reduction
                # Redistribute to a random position
                k = rng.integers(n)
                candidate[k] += reduction
                candidate = np.maximum(candidate, 0.0)
                candidate = _normalize(candidate, target_sum)
                sc = _score_fast(candidate)
                if sc < best_score - 1e-10:
                    best = candidate
                    best_score = sc
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best, best_score
```
