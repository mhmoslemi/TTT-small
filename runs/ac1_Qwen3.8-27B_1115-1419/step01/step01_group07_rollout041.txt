```python
def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
```

```python
"""
Optimize step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key insight: conv(a,a)[n-1] = sum(a_i * a_{n-1-i}), which for asymmetric
sequences can be much less than sum(a_i^2), allowing objective < 2.

Uses FFT-accelerated log-sum-exp smoothed optimization with L-BFGS,
followed by exact subgradient polishing. Multiple diverse asymmetric
initializations to avoid local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve_self(a):
    """Compute conv(a, a) using FFT for O(n log n) complexity."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    # Due to numerical errors, clip tiny negatives
    result[: 2*n-1] = np.maximum(result[: 2*n-1], 0.0)
    return result[: 2*n-1]

def softamp(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def lse_smoothed_max(c, tau):
    """Compute LSE-smoothed max of c with temperature tau."""
    c_max = c.max()
    vals = np.exp((c - c_max) / tau)
    return tau * np.log(vals.sum()) + c_max

def lse_weights(c, tau):
    """Compute weights for LSE-smoothed max."""
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    return w

def smooth_obj(x, tau):
    """LSE-smoothed objective using softmax parameterization."""
    a = softamp(x)
    c = fft_convolve_self(a)
    return lse_smoothed_max(c, tau)

def smooth_grad(x, tau):
    """Gradient of LSE-smoothed objective w.r.t. log-parameterization."""
    a = softamp(x)
    c = fft_convolve_self(a)
    w = lse_weights(c, tau)
    # grad_a[i] = a[i] + sum_j w[i+j] * a[j]
    cross = np.correlate(w, a, mode='valid')
    grad_a = a + cross
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x

def exact_objective(a):
    """Exact objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * max_c / (sum_a * sum_a)

def subgradient_step(a, n_active=80, step=0.01):
    """One projected subgradient step on the exact objective."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-15:
        return a
    
    # Find top peaks
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d(conv(a,a)[k])/da[i] = a[k-i] + a[i] for i in [lo, hi]
        # For i in [lo, hi]: a[k-i] is valid
        g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        g[lo:hi+1] += a[lo:hi+1]
    
    g /= len(active)
    
    # Projected gradient step (onto probability simplex)
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def run_optimization(a_init, n, max_time=120, verbose=True):
    """Run L-BFGS + subgradient polishing from initial point a_init."""
    best_a = a_init.copy()
    best_score = exact_objective(a_init)
    x = np.log(np.maximum(a_init, 1e-30))
    deadline = time.monotonic() + max_time
    
    # LSE smoothing with decreasing temperature
    base_tau = 1.0 / n
    taus = [base_tau * 5, base_tau * 2, base_tau * 1, base_tau * 0.5, base_tau * 0.2]
    
    for tau_mult in taus:
        tau = base_tau * tau_mult
        if time.monotonic() > deadline - 30:
            break
        
        def obj(x_):
            return smooth_obj(x_, tau)
        
        def grad(x_):
            return smooth_grad(x_, tau)
        
        try:
            result = minimize(
                obj, x, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': 500, 'maxls': 40}
            )
            x = result.x
        except Exception as e:
            if verbose:
                print(f"  L-BFGS error: {e}")
            break
        
        a = softamp(x)
        score = exact_objective(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(f"  L-BFGS (tau={tau:.1e}): {best_score:.6f}")
    
    # Subgradient polishing
    a = best_a.copy()
    step = 0.05
    polish_deadline = deadline - 5
    while time.monotonic() < polish_deadline:
        for _ in range(50):
            if time.monotonic() > polish_deadline:
                break
            a = subgradient_step(a, n_active=100, step=step)
        score = exact_objective(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
        step *= 0.98
        if step < 0.001:
            step = 0.01  # Periodic reset for exploration
    
    return best_a, best_score
```

```python
"""
Optimize step function to minimize 2*n*max(conv(a,a))/sum(a)^2.

Key mathematical insight: conv(a,a)[n-1] = sum(a_i * a_{n-1-i}) (the "palindromic sum"),
which for asymmetric sequences can be well below sum(a_i^2), allowing objective < 2.

Strategy:
1. Generate diverse asymmetric initializations (ramps, decays, off-center Gaussians, etc.)
2. FFT-accelerated L-BFGS with log-sum-exp smoothed max of convolution
3. Exact subgradient polishing with adaptive active set
4. Periodic mini-restarts to escape local minima
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Compute conv(a, a) using FFT for O(n log n) complexity."""
    n = len(a)
    result_len = 2 * n - 1
    size = 1
    while size < result_len:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:result_len]


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_smoothed_max(c, tau):
    """Compute LSE-smoothed max of c with temperature tau."""
    c_max = c.max()
    vals = np.exp((c - c_max) / tau)
    return tau * np.log(vals.sum()) + c_max


def lse_weights(c, tau):
    """Compute probability weights for LSE-smoothed max."""
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    return w


def smooth_objective(x, tau):
    """LSE-smoothed objective using softmax parameterization."""
    a = softmax_fn(x)
    c = fft_convolve_self(a)
    return lse_smoothed_max(c, tau)


def smooth_gradient(x, tau):
    """Gradient of LSE-smoothed objective w.r.t. log-parameterization x."""
    a = softmax_fn(x)
    c = fft_convolve_self(a)
    w = lse_weights(c, tau)
    # dL/da[i] = a[i] + sum_j w[i+j] * a[j]
    cross = np.correlate(w, a, mode='valid')
    grad_a = a + cross
    # Chain rule through softmax: dL/dx = a * (grad_a - a·grad_a)
    dot = np.dot(a, grad_a)
    return a * (grad_a - dot)


def exact_obj(a):
    """Exact objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * max_c / (sum_a * sum_a)


def subgradient_step(a, n_active=100, step=0.02):
    """One projected subgradient step on exact objective (probability simplex)."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    if max_c < 1e-15:
        return a

    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]

    # Compute subgradient: g[i] = (1/|A|) * sum_{k in A} (a[k-i] + a[i])
    cross_vals = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        cross_vals[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]

    g = (cross_vals + a) / len(active)

    # Projected gradient step onto probability simplex
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def run_lbfgs_phase(a_init, max_time=90, verbose=True):
    """Run L-BFGS with decreasing tau schedule. Returns (best_a, best_score)."""
    best_a = a_init.copy()
    best_score = exact_obj(a_init)
    n = len(a_init)
    x = np.log(np.maximum(a_init, 1e-30))
    deadline = time.monotonic() + max_time

    base_tau = 1.0 / n
    tau_multipliers = [10.0, 5.0, 2.5, 1.0, 0.5, 0.25, 0.1]

    for mult in tau_multipliers:
        if time.monotonic() > deadline - 15:
            break
        tau = base_tau * mult

        def obj(x_):
            return smooth_objective(x_, tau)

        def grad(x_):
            return smooth_gradient(x_, tau)

        try:
            result = minimize(
                obj, x, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': 600, 'maxls': 40}
            )
            x_new = result.x
            a_new = softmax_fn(x_new)
            score = exact_obj(a_new)
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                x = x_new
            if verbose:
                print(f"    tau={tau:.2e} (x{mult:.0f}): score={score:.6f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at tau={tau:.2e}: {e}")
            break

    return best_a, best_score


def run_subgradient_phase(a_init, max_time=60, verbose=False):
    """Adaptive subgradient polishing. Returns (best_a, best_score)."""
    best_a = a_init.copy()
    best_score = exact_obj(a_init)
    a = a_init.copy()
    step = 0.04
    n_active = 80
    deadline = time.monotonic() + max_time
    stagnation = 0

    while time.monotonic() < deadline:
        # Adaptive n_active: increase when stuck
        if stagnation > 50:
            n_active = min(200, n_active + 20)
            step = min(0.08, step * 1.5)
            stagnation = 0

        for _ in range(30):
            if time.monotonic() > deadline:
                break
            a = subgradient_step(a, n_active=n_active, step=step)

        score = exact_obj(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        step *= 0.99
        if step < 0.001:
            step = 0.01

    return best_a, best_score


def generate_inits(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []

    # Strongly asymmetric: linear ramp (heavy start)
    x = np.linspace(1.0, 0.05, n)
    inits.append(x / x.sum())

    # Reverse ramp
    x = np.linspace(0.05, 1.0, n)
    inits.append(x / x.sum())

    # Exponential decay (fast)
    x = np.exp(-np.linspace(0, 5, n))
    inits.append(x / x.sum())

    # Exponential decay (slow)
    x = np.exp(-np.linspace(0, 2, n))
    inits.append(x / x.sum())

    # Two-block: heavy first third, medium middle, light last
    x = np.ones(n)
    x[:n // 3] = 2.5
    x[n // 3:2 * n // 3] = 1.0
    x[2 * n // 3:] = 0.2
    inits.append(x / x.sum())

    # Three-block: peak at 1/4, dip in middle, rise at 3/4
    x = np.ones(n)
    x[:n // 4] = 2.0
    x[n // 4:n // 2] = 0.5
    x[n // 2:3 * n // 4] = 1.5
    x[3 * n // 4:] = 0.7
    inits.append(x / x.sum())

    # Off-center Gaussians (various positions and widths)
    for center_frac, width in [(0.2, 0.1), (0.3, 0.15), (0.15, 0.08),
                                (0.8, 0.1), (0.7, 0.12), (0.5, 0.2)]:
        center = int(n * center_frac)
        x = np.exp(-0.5 * ((np.arange(n) - center) / (n * width)) ** 2)
        x = np.maximum(x, 1e-6)
        inits.append(x / x.sum())

    # Cosine-modulated with phase shifts (breaks symmetry)
    for freq in [2, 3, 5]:
        for phase in [0.5, 1.0, 2.0]:
            x = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
            x = np.maximum(x, 0.01)
            inits.append(x / x.sum())

    # Random Dirichlet (various concentrations)
    for alpha in [0.2, 0.5, 1.0, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))

    # Random smooth (Gaussian process-like)
    for _ in range(2):
        freqs = rng.uniform(0, 3, 5)
        phases = rng.uniform(0, 2 * np.pi, 5)
        amps = rng.uniform(0.5, 1.5, 5)
        x = np.ones(n)
        for f, p, a in zip(freqs, phases, amps):
            x += a * np.cos(2 * np.pi * f * np.arange(n) / n + p)
        x = np.maximum(x, 0.01)
        inits.append(x / x.sum())

    # Periodic asymmetric pattern (short period)
    for period in [3, 4, 7]:
        pattern = np.array([2.0, 0.5, 1.0, 0.3, 1.5, 0.8, 0.4])[:period]
        x = np.tile(pattern, n // period + 1)[:n]
        inits.append(x / x.sum())

    # Sparse peaks
    for _ in range(2):
        x = np.full(n, 0.05)
        n_peaks = max(3, n // 200)
        positions = rng.choice(n, size=n_peaks, replace=False)
        x[positions] = rng.uniform(5, 30, n_peaks)
        inits.append(x / x.sum())

    return inits
```
