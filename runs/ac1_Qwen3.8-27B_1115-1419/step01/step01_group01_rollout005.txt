import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def _np2(x):
    p = 1
    while p < x:
        p <<= 1
    return p

def _gauss_sequence(params, n, M):
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for j in range(M):
        c = max(0.0, params[3*j])
        mu = float(params[3*j+1])
        sig = max(0.5, float(params[3*j+2]))
        x = (t - mu) / sig
        a += c * np.exp(-0.5 * x * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def _score_raw(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return 1e10, None
    a_n = a / s
    m = _np2(2*n - 1)
    F = np.fft.fft(a_n, m)
    b = np.fft.ifft(F*F).real[:2*n-1]
    b = np.maximum(b, 0.0)
    return 2.0 * n * b.max(), b

def _gauss_obj(params, n, M, m, tau):
    a = _gauss_sequence(params, n, M)
    if a is None:
        return 1e10
    m2 = m
    F = np.fft.fft(a, m2)
    b = np.fft.ifft(F*F).real[:2*n-1]
    b = np.maximum(b, 0.0)
    bm = b.max()
    if bm < 1e-30:
        return 1e10

    tb = np.clip(tb, -60.0, 60.0)
    sm = bm + np.log(np.sum(np.exp(tb)) + 1e-30) / tau
    sc = 2.0 * n * sm
    return float(sc) if np.isfinite(sc) else 1e10
