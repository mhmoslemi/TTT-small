"""
Search for non-negative step-function heights minimizing:
    2 * n * max(convolve(a,a)) / sum(a)^2

The algorithm combines:
- FFT-based fast scoring and gradient evaluation
- L-BFGS-B on a smooth log-sum-exponential surrogate, using log-softmax
  parameterization to enforce non-negativity and normalization
- Multiplicative / mirror-descent steps that flatten weighted peak mirrors
- A primal-dual multiplicative algorithm adapting peak weights dynamically
- Random local reordering, smoothing, trimming, and length resizing
"""

import time
import numpy as np

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None

# Global state used by the top-level scipy callback (no closures/lambdas)
_STATE_T = 1.0
_STATE_N = 0
_STATE_REG = 1e-12
_STATE_TRACK = False
_STATE_FAST_BEST_SCORE = float("inf")
_STATE_FAST_BEST_X = None


def _next_pow2(n):
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _self_conv(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    nfft = _next_pow2(need)
    fa = rfft(x, nfft)
    c = irfft(fa * fa, nfft)
    return c[:need]


def _hankel_mul(x, w):
    """h_i = sum_j x_j w_{i+j}, with w non-negative and len(w)=2*len(x)-1."""
    n = int(x.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    need = n + m - 1
    nfft = _next_pow2(need)
    fa = rfft(x[::-1], nfft)
    fb = rfft(w, nfft)
    c = irfft(fa * fb, nfft)
    return c[n - 1:n - 1 + n]


def _normalize_one(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x.copy()
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(x.shape, 1.0 / x.size, dtype=np.float64)
    return x / s


def _log_normalize(logx):
    logx = np.asarray(logx, dtype=np.float64)
    n = logx.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(logx)):
        return np.full(n, 1.0 / n, dtype=np.float64)
    mx = float(np.max(logx))
    ex = np.exp(logx - mx)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return ex / s


def _fast_score(x):
    x = _normalize_one(x)
    n = x.size
    if n == 0:
        return float("inf")
    q = _self_conv(x)
    if q.size == 0:
        return float("inf")
    mq = float(np.max(q))
    if not np.isfinite(mq) or mq < 0.0:
        mq = 0.0
    return 2.0 * n * mq


def _clean_sequence(seq):
    if seq is None:
        return None
    arr = np.asarray(seq, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    if not np.all(np.isfinite(arr)):
        return None
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return arr / s


def _update_best(seq, best_seq, best_score, score_hint=None):
    arr = _clean_sequence(seq)
    if arr is None:
        return best_seq, best_score
    if score_hint is None:
        sc = _fast_score(arr)
    else:
        sc = float(score_hint)
        if not np.isfinite(sc):
            sc = _fast_score(arr)
    if np.isfinite(sc) and sc < best_score:
        return arr.copy(), sc
    return best_seq, best_score


def _load_parent():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(np.sum(arr))
            if s > 0.0:
                return arr / s
    except Exception:
        pass
    return None


def _resize_sequence(x, m):
    x = _normalize_one(x)
    n = x.size
    m = int(m)
    if m <= 0:
        return x.copy()
    if m == n:
        return x.copy()
    if n == 1:
        return np.full(m, 1.0 / m, dtype=np.float64)
    old = np.linspace(0.0, 1.0, n)
    new = np.linspace(0.0, 1.0, m)
    y = np.interp(new, old, x)
    return _normalize_one(y)


def _smooth_sequence(x, width):
    x = _normalize_one(x)
    if width <= 0 or x.size < 5:
        return x
    width = float(max(0.25, width))
    radius = int(max(1, np.ceil(3.0 * width)))
    k = np.arange(-radius, radius + 1, dtype=np.float64)
    ker = np.exp(-0.5 * (k / width) ** 2)
    ker /= float(np.sum(ker))
    y = np.convolve(x, ker, mode="same")
    y = np.maximum(y, 0.0)
    return _normalize_one(y)


def _trim_sequence(x, max_steps=50):
    x = _normalize_one(x)
    if x.size < 3:
        return x
    candidate = x.copy()
    best = x.copy()
    cur_sc = _fast_score(x)
    best_sc = cur_sc
    for _ in range(int(max_steps)):
        improved = False
        if candidate.size > 16:
            nxt = _normalize_one(candidate[1:])
            sc = _fast_score(nxt)
            if np.isfinite(sc) and sc < cur_sc - 1e-12:
                candidate = nxt
                cur_sc = sc
                improved = True
                if sc < best_sc:
                    best = nxt.copy()
                    best_sc = sc
        if candidate.size > 16:
            nxt = _normalize_one(candidate[:-1])
            sc = _fast_score(nxt)
            if np.isfinite(sc) and sc < cur_sc - 1e-12:
                candidate = nxt
                cur_sc = sc
                improved = True
                if sc < best_sc:
                    best = nxt.copy()
                    best_sc = sc
        if not improved:
            break
    return best


def _random_local_reorders(x, rng, trials=30):
    x = _normalize_one(x)
    n = x.size
    if n < 4 or trials <= 0:
        return x, _fast_score(x)
    best = x.copy()
    best_sc = _fast_score(x)
    trials = int(trials)
    for _ in range(trials):
        y = x.copy()
        mode = int(rng.integers(0, 2))
        if mode == 0:
            sh = int(rng.integers(1, n))
            y = np.roll(y, sh)
        else:
            a = int(rng.integers(0, n - 2))
            length = int(rng.integers(2, n - a))
            y[a:a + length] = y[a + length - 1:a - 1:-1]
        cand = _normalize_one(y)
        sc = _fast_score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best = cand.copy()
            best_sc = sc
    return best, best_sc


def _perturb_logx(x, rng, sigma):
    x = np.maximum(np.asarray(x, dtype=np.float64), 1e-300)
    logx = np.log(x)
    if logx.size == 0:
        return x
    noise = rng.normal(0.0, float(sigma), size=logx.size)
    noise -= float(np.mean(noise))
    logx += noise
    return _log_normalize(logx)


def _x_to_u(x):
    x = _normalize_one(x)
    if x.size <= 1:
        return np.array([], dtype=np.float64)
    logx = np.log(np.maximum(x, 1e-300))
    logx -= float(np.mean(logx))
    return logx[:-1].copy()


def _u_to_x(u, n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size < n - 1:
        tmp = np.zeros(n - 1, dtype=np.float64)
        tmp[:u.size] = u
        u = tmp
    elif u.size > n - 1:
        u = u[:n - 1]
    y = np.empty(n, dtype=np.float64)
    y[:n - 1] = u
    y[n - 1] = -float(np.sum(u))
    return _log_normalize(y)


def _softmax_q(q, tau):
    q = np.asarray(q, dtype=np.float64)
    if q.size == 0:
        return np.array([], dtype=np.float64)
    qmax = float(np.max(q))
    tau = float(max(1e-9, tau))
    c = np.exp((q - qmax) / tau)
    s = float(np.sum(c))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(q)
        w[int(np.argmax(q))] = 1.0
        return w
    return c / s


def _obj_core(u, T, n, reg):
    global _STATE_FAST_BEST_SCORE, _STATE_FAST_BEST_X

    n = int(n)
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)

    expected = max(n - 1, 0)
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size != expected:
        tmp = np.zeros(expected, dtype=np.float64)
        m = min(u.size, expected)
        if m > 0:
            tmp[:m] = u[:m]
        u = tmp

    y = np.empty(n, dtype=np.float64)
    if n == 1:
        y[0] = 0.0
    else:
        if not np.all(np.isfinite(u)):
            return 1e100, np.zeros(max(n - 1, 0), dtype=np.float64)
        y[:n - 1] = u
        y[n - 1] = -float(np.sum(u))

    if not np.all(np.isfinite(y)):
        return 1e100, np.zeros(max(n - 1, 0), dtype=np.float64)

    mx = float(np.max(y))
    exp_y = np.exp(y - mx)
    sx = float(np.sum(exp_y))
    if not np.isfinite(sx) or sx <= 0.0:
        return 1e100, np.zeros(max(n - 1, 0), dtype=np.float64)

    x = exp_y / sx
    q = _self_conv(x)
    if q.size == 0:
        qmax = 0.0
    else:
        qmax = float(np.max(q))
    if not np.isfinite(qmax) or qmax < 0.0:
        qmax = 0.0

    f = qmax
    gy = np.zeros(n, dtype=np.float64)

    if T > 0.0:
        scaled = (q - qmax) / T
        c = np.exp(np.maximum(scaled, -745.0))
        wsum = float(np.sum(c))
        if np.isfinite(wsum) and wsum > 0.0:
            w = c / wsum
        else:
            w = np.zeros_like(q)
            if q.size > 0:
                w[int(np.argmax(q))] = 1.0
        f += float(T) * float(np.log(max(wsum, 1e-300)))

        h = _hankel_mul(x, w)
        h = np.asarray(h, dtype=np.float64)
        if h.size != n:
            h = np.zeros(n, dtype=np.float64)
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)

        dFdx = (2.0 / float(T)) * h
        inner = float(np.dot(dFdx, x))
        gy = dFdx * x - x * inner

    if n > 1:
        gu = gy[:n - 1] - gy[n - 1]
        if reg > 0.0:
            gu = gu + reg * u
            f += 0.5 * float(reg) * float(np.dot(u, u))
    else:
        gu = np.array([], dtype=np.float64)

    if not np.all(np.isfinite(gu)):
        gu = np.nan_to_num(gu, nan=0.0, posinf=0.0, neginf=0.0)

    if _STATE_TRACK:
        sc = 2.0 * float(n) * qmax
        if np.isfinite(sc) and sc < _STATE_FAST_BEST_SCORE:
            _STATE_FAST_BEST_SCORE = sc
            _STATE_FAST_BEST_X = x.copy()

    return float(f), gu


def _scipy_obj(v):
    return _obj_core(v, _STATE_T, _STATE_N, _STATE_REG)


def _lbfgs_one(x, T, maxfun, maxiter=None):
    global _STATE_T, _STATE_N, _STATE_REG, _STATE_TRACK
    global _STATE_FAST_BEST_SCORE, _STATE_FAST_BEST_X

    x = _normalize_one(x)
    n = x.size
    maxfun = int(maxfun)
    if n < 2 or maxfun < 3 or scipy_minimize is None:
        return x

    init_sc = _fast_score(x)
    _STATE_T = float(max(1e-12, T))
    _STATE_N = int(n)
    _STATE_REG = 1e-12
    _STATE_TRACK = True
    _STATE_FAST_BEST_SCORE = init_sc
    _STATE_FAST_BEST_X = x.copy()

    u0 = _x_to_u(x)
    if maxiter is None:
        maxiter = max(3, maxfun)

    options = {
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "ftol": 1e-15,
        "gtol": 1e-10,
        "maxls": 12,
    }

    res = None
    try:
        res = scipy_minimize(_scipy_obj, u0, method="L-BFGS-B", jac=True,
                             options=options, tol=1e-10)
    except Exception:
        res = None

    _STATE_TRACK = False

    best_x = _STATE_FAST_BEST_X if _STATE_FAST_BEST_X is not None else x.copy()
    best_sc = _STATE_FAST_BEST_SCORE

    if res is not None and hasattr(res, "x"):
        try:
            xx = _u_to_x(res.x, n)
            sc = _fast_score(xx)
            if np.isfinite(sc) and sc < best_sc:
                best_x = xx.copy()
                best_sc = sc
        except Exception:
            pass

    return _normalize_one(best_x)


def _run_lbfgs_schedule(x, time_limit, seed, maxfun=None, levels=8,
                        max_stag=70.0, factor_start=10.0, factor_end=0.02,
                        T_min=1e-7):
    x = _normalize_one(x)
    n = x.size
    if n < 2:
        return x, _fast_score(x)

    best = x.copy()
    best_sc = _fast_score(x)
    rng = np.random.default_rng(seed)
    step = 0
    last_improve = time.monotonic()

    while time.monotonic() < float(time_limit) - 3.0:
        cycle = step // max(1, int(levels))
        decay = 0.78 ** min(cycle, 3)
        fs = max(float(factor_end), float(factor_start) * decay)
        fe = max(0.005, float(factor_end) * decay)
        if fs < fe:
            fs, fe = fe, fs

        frac = (step % max(1, int(levels))) / max(1, int(levels) - 1)
        factor = float(np.exp(np.log(fs) + frac * (np.log(fe) - np.log(fs))))

        r = best_sc / (2.0 * max(best.size, 1))
        if not np.isfinite(r) or r <= 0.0:
            r = 1e-4
        T = float(max(T_min, min(1e-2, r * factor)))

        mode = step % 6
        if mode == 0:
            start = _perturb_logx(best, rng, 0.18 + 0.06 * min(cycle, 3))
        elif mode == 1:
            start = _perturb_logx(best, rng, 0.07)
        elif mode == 2:
            start = best.copy()
        elif mode == 3:
            start = _smooth_sequence(best, 1.0)
        elif mode == 4:
            start = _trim_sequence(best, 10)
        else:
            start = _perturb_logx(best, rng, 0.12)

        scale = int(700000 / max(best.size, 1))
        scale = max(15, min(180, scale))
        if maxfun is None:
            mr = scale
        else:
            mr = min(scale, int(maxfun))

        rem = float(time_limit) - time.monotonic()
        if rem < 3.0:
            break
        if rem < 10.0:
            mr = min(mr, 20)

        cand = _lbfgs_one(start, T, mr)
        sc = _fast_score(cand)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best = cand.copy()
            best_sc = sc
            last_improve = time.monotonic()

        step += 1

        if time.monotonic() - last_improve > float(max_stag):
            big = _perturb_logx(best, rng, 0.35)
            scb = _fast_score(big)
            if scb < best_sc:
                best = big.copy()
                best_sc = scb
                last_improve = time.monotonic()
            else:
                break

    return best.copy(), float(best_sc)


def _mirror_refine(x, p, alpha, time_limit, rng=None, max_stag=60.0):
    x = _normalize_one(x)
    best = x.copy()
    best_sc = _fast_score(x)
    current = x.copy()
    cur_sc = best_sc

    if rng is None:
        rng = np.random.default_rng(555)

    alpha = float(alpha)
    init_alpha = alpha
    alpha_cur = alpha
    last_improve = time.monotonic()
    no_improve = 0

    while time.monotonic() < float(time_limit) - 2.0:
        q = _self_conv(current)
        if q.size == 0:
            break
        qmax = float(np.max(q))
        if not np.isfinite(qmax) or qmax <= 0.0:
            break

        w = np.power(np.maximum(q, 0.0), float(p))
        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= 1e-300:
            w = np.full_like(q, 1.0 / max(q.size, 1), dtype=np.float64)
        else:
            w = w / ws

        h = _hankel_mul(current, w)
        h = np.asarray(h, dtype=np.float64)
        if h.size != current.size:
            h = np.zeros(current.size, dtype=np.float64)
        h = np.nan_to_num(h, nan=0.0, posinf=1e6, neginf=0.0)

        target = float(np.median(h))
        if not np.isfinite(target) or target <= 0.0:
            target = 1.0 / max(current.size, 1)

        h_safe = np.clip(h, target * 1e-6, target * 1e6)
        logx = np.log(np.clip(current, 1e-300, None))
        log_ratio = np.log(target / h_safe)
        log_ratio = np.clip(log_ratio, -8.0, 8.0)
        logx_new = logx + float(alpha_cur) * log_ratio

        cand = _log_normalize(logx_new)
        sc_cand = _fast_score(cand)

        if np.isfinite(sc_cand) and sc_cand < best_sc - 1e-12:
            current = cand
            cur_sc = sc_cand
            best = cand.copy()
            best_sc = sc_cand
            last_improve = time.monotonic()
            no_improve = 0
        elif np.isfinite(sc_cand) and sc_cand < cur_sc - 1e-12:
            current = cand
            cur_sc = sc_cand
            no_improve = 0
        else:
            no_improve += 1
            alpha_cur *= 0.85
            if alpha_cur < 1e-3:
                alpha_cur = init_alpha * 0.25

            if no_improve > 25 or time.monotonic() - last_improve > float(max_stag):
                current = _perturb_logx(best, rng, 0.05 + 0.05 * min(no_improve / 10.0, 3.0))
                cur_sc = _fast_score(current)
                alpha_cur = init_alpha
                no_improve = 0
                last_improve = time.monotonic()

    return best.copy(), float(best_sc)


def _mwa_refine(x, time_limit, seed, alpha=0.25, tau_frac=0.25, max_stag=60.0):
    x = _normalize_one(x)
    n = x.size
    if n < 2:
        return x, _fast_score(x)

    best = x.copy()
    best_sc = _fast_score(x)
    current = x.copy()
    cur_sc = best_sc
    rng = np.random.default_rng(seed)

    alpha = float(alpha)
    alpha_cur = alpha
    init_alpha = alpha
    last_improve = time.monotonic()
    no_improve = 0

    q = _self_conv(current)
    r = cur_sc / (2.0 * n)
    tau = max(1e-8, float(tau_frac) * r)
    u = _softmax_q(q, tau)

    while time.monotonic() < float(time_limit) - 2.0:
        h = _hankel_mul(current, u)
        h = np.asarray(h, dtype=np.float64)
        if h.size != n:
            h = np.zeros(n, dtype=np.float64)
        h = np.nan_to_num(h, nan=0.0, posinf=1e6, neginf=0.0)

        hmax = float(np.max(h)) if h.size > 0 else 1.0
        if not np.isfinite(hmax) or hmax <= 0.0:
            hnorm = np.zeros_like(h)
        else:
            hnorm = h / hmax

        logx = np.log(np.clip(current, 1e-300, None))
        logx_new = logx - float(alpha_cur) * hnorm
        cand = _log_normalize(logx_new)

        q_new = _self_conv(cand)
        if q_new.size == 0:
            sc = _fast_score(cand)
        else:
            sc = 2.0 * n * float(np.max(q_new))

        if np.isfinite(sc) and sc < cur_sc + 1e-8:
            current = cand
            cur_sc = sc
            q = q_new
            if sc < best_sc:
                best = cand.copy()
                best_sc = sc
                last_improve = time.monotonic()
                no_improve = 0
        else:
            no_improve += 1
            alpha_cur *= 0.85
            if no_improve > 20 or time.monotonic() - last_improve > float(max_stag):
                current = _perturb_logx(best, rng, 0.10)
                cur_sc = _fast_score(current)
                q = _self_conv(current)
                alpha_cur = init_alpha
                no_improve = 0
                last_improve = time.monotonic()

        r = cur_sc / (2.0 * max(current.size, 1))
        if not np.isfinite(r) or r <= 0.0:
            r = 1e-4
        tau = max(1e-8, float(tau_frac) * r)
        u = _softmax_q(q, tau)

    return best.copy(), float(best_sc)


def propose_candidate(seed=42, budget_s=2400.0):
    """Top-level 2400-second search for the returned height sequence."""
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0
    total_deadline = start + max(90.0, budget - 75.0)

    rng = np.random.default_rng(12345 + int(seed))

    best_seq = None
    best_score = float("inf")

    parent = _load_parent()
    if parent is not None:
        best_seq, best_score = _update_best(parent, best_seq, best_score)

    if best_seq is None:
        best_seq = _normalize_one(np.ones(5582, dtype=np.float64))
        best_score = _fast_score(best_seq)

    base_n = int(best_seq.size)
    print(f"initial fast score: {best_score:.10f}, n={base_n}")

    # Cheap deterministic improvements.
    cand = _trim_sequence(best_seq, 30)
    best_seq, best_score = _update_best(cand, best_seq, best_score)

    for w in (1.0, 2.0):
        cand = _smooth_sequence(best_seq, w)
        best_seq, best_score = _update_best(cand, best_seq, best_score)

    cand, sc = _random_local_reorders(best_seq, rng, 60)
    best_seq, best_score = _update_best(cand, best_seq, best_score, sc)

    # Warm optimization on the current length.
    now = time.monotonic()
    if total_deadline - now > 70.0:
        lim = min(now + 45.0, total_deadline - 35.0)
        best_seq, best_score = _run_lbfgs_schedule(best_seq, lim, seed=11,
                                                    maxfun=90, max_stag=30)
        print(f"after warm L-BFGS: {best_score:.10f}")

    now = time.monotonic()
    if total_deadline - now > 50.0:
        lim = min(now + 35.0, total_deadline - 35.0)
        best_seq, best_score = _mirror_refine(best_seq, 6, 0.14, lim, rng,
                                               max_stag=25)
        print(f"after warm mirror: {best_score:.10f}")

    # Explore nearby lengths from the current best shape.
    now = time.monotonic()
    if total_deadline - now > 150.0:
        cap = max(10000, base_n)
        mults = [0.75, 0.9, 1.05, 1.10, 1.25, 1.40, 1.60, 2.00]
        tried = set([base_n])
        for mult in mults:
            now = time.monotonic()
            if total_deadline - now < 75.0:
                break
            m = int(round(base_n * float(mult)))
            m = int(max(512, min(cap, m)))
            if m in tried:
                continue
            if abs(m - base_n) < max(10, int(0.01 * base_n)):
                continue
            tried.add(m)

            x = _resize_sequence(best_seq, m)
            x = _smooth_sequence(x, 1.0)
            x = _trim_sequence(x, 20)

            x, sc = _random_local_reorders(x, rng, 8)
            best_seq, best_score = _update_best(x, best_seq, best_score, sc)

            now = time.monotonic()
            if total_deadline - now > 45.0:
                x, sc = _mirror_refine(x, 4, 0.16,
                                       min(now + 15.0, total_deadline - 25.0),
                                       rng, max_stag=12)
                best_seq, best_score = _update_best(x, best_seq, best_score, sc)

            now = time.monotonic()
            if total_deadline - now > 45.0:
                x, sc = _run_lbfgs_schedule(x,
                                            min(now + 25.0, total_deadline - 25.0),
                                            seed=2000 + m,
                                            maxfun=70, max_stag=15)
                best_seq, best_score = _update_best(x, best_seq, best_score, sc)
                print(f"length m={m}: fast best {best_score:.10f}")

    # Main alternating local-search loop.
    block = 0
    last_score = best_score
    last_improve = time.monotonic()

    while time.monotonic() < total_deadline - 70.0:
        now = time.monotonic()
        if best_score < last_score - 1e-8:
            last_score = best_score
            last_improve = now

        sl = min(120.0, total_deadline - now)
        if sl < 25.0:
            break

        t_end = now + sl
        r_seed = 10000 + 17 * block
        bmod = block % 4

        if bmod == 0:
            cand, sc = _run_lbfgs_schedule(best_seq, t_end, seed=r_seed,
                                           maxfun=140, max_stag=70)
        elif bmod == 1:
            cand, sc = _mwa_refine(best_seq, t_end, seed=r_seed, alpha=0.25,
                                   tau_frac=0.25, max_stag=55)
        elif bmod == 2:
            cand, sc = _mirror_refine(best_seq, 8, 0.10, t_end, rng,
                                      max_stag=55)
        else:
            cand, sc = _mirror_refine(best_seq, 16, 0.05, t_end, rng,
                                      max_stag=55)

        best_seq, best_score = _update_best(cand, best_seq, best_score, sc)

        cand2, sc2 = _random_local_reorders(cand, rng, 20)
        best_seq, best_score = _update_best(cand2, best_seq, best_score, sc2)

        cand3 = _trim_sequence(cand, 30)
        best_seq, best_score = _update_best(cand3, best_seq, best_score)

        cand4 = _smooth_sequence(cand, 1.0)
        best_seq, best_score = _update_best(cand4, best_seq, best_score)

        print(f"block {block}: n={best_seq.size}, fast={best_score:.10f}")
        block += 1

        if time.monotonic() - last_improve > 240.0:
            # Large perturbation + random reordering + dedicated L-BFGS.
            x = _perturb_logx(best_seq, rng, 0.40)
            x, sc = _random_local_reorders(x, rng, 30)
            x, sc = _run_lbfgs_schedule(
                x,
                min(time.monotonic() + 90.0, total_deadline - 35.0),
                seed=r_seed + 999,
                maxfun=85,
                max_stag=40,
            )
            best_seq, best_score = _update_best(x, best_seq, best_score, sc)
            last_improve = time.monotonic()

    # Final polish.
    now = time.monotonic()
    if total_deadline - now > 70.0:
        best_seq, best_score = _mirror_refine(
            best_seq, 20, 0.035,
            min(now + 35.0, total_deadline - 45.0), rng, max_stag=25)

    now = time.monotonic()
    if total_deadline - now > 55.0:
        best_seq, best_score = _mwa_refine(
            best_seq,
            min(now + 30.0, total_deadline - 45.0),
            seed=777, alpha=0.12, tau_frac=0.15, max_stag=20)

    now = time.monotonic()
    if total_deadline - now > 55.0:
        best_seq, best_score = _run_lbfgs_schedule(
            best_seq,
            min(now + 40.0, total_deadline - 40.0),
            seed=888, maxfun=110, max_stag=25)

    cand, sc = _random_local_reorders(best_seq, rng, 60)
    best_seq, best_score = _update_best(cand, best_seq, best_score, sc)

    cand = _trim_sequence(best_seq, 150)
    best_seq, best_score = _update_best(cand, best_seq, best_score)

    cand = _smooth_sequence(best_seq, 1.0)
    best_seq, best_score = _update_best(cand, best_seq, best_score)

    print(f"final fast score: {best_score:.10f}, n={int(best_seq.size)}")

    # Safely produce a valid Python list.
    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    if best_seq.size == 0:
        best_seq = np.ones(1024, dtype=np.float64)

    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    s = float(np.sum(best_seq))
    if not np.isfinite(s) or s <= 1e-300:
        best_seq = np.ones(1024, dtype=np.float64)
    else:
        best_seq = best_seq / s

    out = [float(v) for v in best_seq]

    try:
        exact = evaluate_sequence(out)
        print(f"final exact score: {exact}")
        if not np.isfinite(exact):
            fallback = [1.0] * 1024
            try:
                if np.isfinite(evaluate_sequence(fallback)):
                    out = fallback
            except Exception:
                pass
    except Exception as e:
        print("final exact evaluation skipped:", e)

    return out
