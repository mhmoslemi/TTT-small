Minimize 2n*max(conv(a,a))/(sum(a))^2 using smoothed FISTA (log-sum-exp
approximation of the max) with FFT-based gradient computation, multiple
diverse initializations, and adaptive temperature annealing.
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho_val = np.where(cond)[0][-1] + 1
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    conv = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def smoothed_conv_gradient(a, alpha, fft_size=None):
    """
    Compute the gradient of (1/alpha)*log(sum_k exp(alpha*conv(a,a)[k])).
    Uses FFT for efficiency.
    """
    n = len(a)
    conv = np.convolve(a, a)
    # Numerical stability
    conv_max = conv.max()
    shifted = alpha * (conv - conv_max)
    
    # Compute softmax weights
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    wsum = exp_shifted.sum()
    if wsum < 1e-30:
        # Fallback: uniform weights
        weights = np.ones(len(conv)) / len(conv)
    else:
        weights = exp_shifted / wsum
    
    # Gradient via FFT: g[m] = 2 * sum_k weights[k] * a[k-m]
    # = 2 * correlate(weights, a)[m]
    # Using FFT: irfft(FFT(weights) * conj(FFT(a)))
    if fft_size is None:
        fft_size = 1
        while fft_size < 3 * n:
            fft_size *= 2
    
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    g = 2.0 * corr[:n]
    
    return g, conv_max

    s = a.sum()
    if s < 1e-15:

    else:
        a /= s

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    
    # Precompute the full convolution size
    # conv has length 2n-1
    
    while time.monotonic() < deadline:
        it += 1

        frac = min(1.0, it / max(1, int(duration * 10)))  # approximately
        # Actually, let's base it on iterations
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / 5000.0)
        
        # FISTA extrapolation
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy

        g, _ = smoothed_conv_gradient(y, alpha, fft_size)
        
        # Gradient step + project
        a_new = project_simplex(y - step * g)
        a_prev = a
        a = a_new
        
        # Periodic evaluation (every 50 iterations to save time)
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck == 100:
                step *= 0.7
            elif stuck >= 300:
                # Restart

                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-10:
                    a /= sa

                a_prev = a.copy()

            print(f"  FISTA it={it}, score={best_score:.8f}, alpha={alpha:.1f}, step={step:.4f}, time_left={remaining:.0f}s")
    
    # Final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def run_exact_fista(a0, duration, rng, step=0.5, top_k_frac=0.05):
    """
    Run FISTA with exact (or near-exact) subgradient from top-k peaks.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # FISTA extrapolation
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        
        # Compute convolution
        conv = np.convolve(y, y)
        conv_max = conv.max()
        
        # Find peaks within a threshold of the max
        threshold = conv_max - 0.01 * conv_max
        peak_mask = conv >= threshold
        peak_count = peak_mask.sum()
        
        # Use top-k if too many peaks
        min_top_k = max(3, int(0.01 * len(conv)))
        max_top_k = max(10, int(top_k_frac * len(conv)))
        if peak_count > max_top_k:
            # Use top max_top_k
            top_idx = np.argpartition(conv, -max_top_k)[-max_top_k:]
        elif peak_count < min_top_k:
            # Use at least min_top_k
            top_idx = np.argpartition(conv, -min_top_k)[-min_top_k:]
        else:
            top_idx = np.where(peak_mask)[0]
        
        # Compute gradient as average over selected peaks
        # grad conv[k] w.r.t. a[m] = 2 * a[k-m] if 0 <= k-m < n
        # g[m] = 2 * (1/|S|) * sum_{k in S} a[k-m]
        # This is 2/|S| * sum_{k in S} a[k-m]
        # = 2/|S| * (selective correlation of indicator_S with a)
        
        # For efficiency, use the FFT approach with the peak indicator
        S = len(top_idx)
        indicator = np.zeros(len(conv))
        indicator[top_idx] = 1.0
        indicator /= S
        
        # g[m] = 2 * sum_k indicator[k] * a[k-m]
        fft_size = 1
        while fft_size < 3 * n:
            fft_size *= 2
        W = np.fft.rfft(indicator, fft_size)
        A = np.fft.rfft(y, fft_size)
        corr = np.fft.irfft(W * np.conj(A), fft_size)
        g = 2.0 * corr[:n]
        
        # Gradient step + project
        a_new = project_simplex(y - step * g)
        a_prev = a
        a = a_new
        
        # Periodic evaluation
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck == 100:
                step *= 0.7
            elif stuck >= 300:
                a = best_a + rng.normal(0, 0.005 / np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-10:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t = 1.0
                step = 0.5
                stuck = 0
        
        if it % 2000 == 0:
            remaining = deadline - time.monotonic()
            print(f"  Exact FISTA it={it}, score={best_score:.8f}, step={step:.4f}, peaks={peak_count}, time_left={remaining:.0f}s")
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score
