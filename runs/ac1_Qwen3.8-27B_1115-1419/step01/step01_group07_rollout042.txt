def solve_lp_for_g(f, n):
    """Solve LP: minimize t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1"""
    g = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [g >= 0, cp.sum(g) == 1]
    
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i, k-i
    # For k = 0: f[0]*g[0]
    # For k = 1: f[0]*g[1] + f[1]*g[0]
    # ...
    # For k = n-1: f[0]*g[n-1] + f[1]*g[n-2] + ... + f[n-1]*g[0]
    # ...
    # For k = 2n-2: f[n-1]*g[n-1]
    
    # We can write this as: for each k, sum_{i} f[i] * g[k-i] <= t
    # where the inner sum is over valid i (0 <= i < n and 0 <= k-i < n)
    
    # More efficiently: (f*g)[k] = (f * g)[k] which is the k-th element of the convolution
    # We can construct this as matrix-vector product: C[k,:] @ g <= t
    # where C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Build the constraint matrix
    # Actually, let's just build the convolution constraints directly
    for k in range(2*n - 1):
        # (f*g)[k] = sum_{j} f[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # (f*g)[k] = sum_{j=lo}^{hi} f[k-j] * g[j]
        coeff = np.zeros(n)

            coeff[j] = f[k - j]
        constraints.append(coeff @ g <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC)
    
    return g.value, t.value

    g = cp.Variable(n)
    t = cp.Variable()

        g >= 0,
        cp.sum(g) == 1,
        C @ g <= t
    ]

    try:
        prob.solve(solver=cp.CBC, verbose=False)

        return f.copy(), f @ f  # fallback
    if g.value is None:
        return f.copy(), f @ f
    g_val = np.maximum(g.value, 0)
    s = g_val.sum()
    if s < 1e-10:

"""
Optimize step function sequence using LP-based active set iteration with
subgradient polishing. Key innovation: each step solves an LP to find the
optimal direction on the simplex that minimizes the cross-convolution with
the current sequence, providing exact (unsmoothed) gradient information and
enabling faster convergence than LSE-smoothed methods.
"""

import numpy as np
import time
import cvxpy as cp


def build_conv_matrix_rows(f, n, row_indices):
    """Build selected rows of the convolution matrix C where C@g = conv(f,g)."""
    rows = len(row_indices)
    C = np.zeros((rows, n))
    for idx, k in enumerate(row_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # C[idx, j] = f[k-j] for j in [lo, hi]
        C[idx, lo:hi + 1] = f[k - hi:k - lo + 1]
    return C


def solve_lp_direction(f, n, active_rows, C_active):
    """Solve LP: minimize t s.t. C_active@g <= t, g>=0, sum(g)=1."""
    g_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        g_var >= 0,
        cp.sum(g_var) == 1,
        C_active @ g_var <= t_var
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False)
    except Exception:
        return f.copy(), None
    if g_var.value is None:
        return f.copy(), None
    g_val = np.maximum(g_var.value, 0.0)
    s = g_val.sum()
    if s < 1e-10:
        return f.copy(), None
    g_val = g_val / s
    return g_val, t_var.value


def subgradient_step(a, n_active=50, step=0.01):
    """One exact subgradient step on the max-convolution objective."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    if max_c < 1e-15:
        return a
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_c / (sum_a ** 2))


def lp_iteration(a, n, n_active=200):
    """One LP-based iteration: find best direction and blend."""
    c = np.convolve(a, a)
    M = np.max(c)
    # Get active set (top peaks)
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    # Build constraint matrix for active rows
    C_active = build_conv_matrix_rows(a, n, active)
    # Solve LP
    g, t_val = solve_lp_direction(a, n, active, C_active)
    if t_val is None:
        return a, 0.0
    return g, t_val

        s = fast_eval(a_new)
        if s < best_s:

            best_a = a_new.copy()
    return best_a, best_s
