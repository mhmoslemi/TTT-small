    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones_like(a) / a.size
    return a / s

    """Full linear convolution using FFT."""
    n_a = len(a)
    n_b = len(b)
    n_fft = 1 << (n_a + n_b - 2).bit_length()
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    result = np.fft.irfft(A * B, n_fft)

    """
    n = len(a)

Search for low-peak nonnegative self-convolutions using LP-based direction finding
and gradient-based refinement.

The score is 2*n*max(a*a)/(sum(a))^2, minimized over nonnegative sequences a.
Approach:
1. LP-based directional updates (find best direction keeping convolution bounded)
2. Log-sum-exp gradient descent for fine-tuning
3. Multi-length search with resampling
"""

import numpy as np
from scipy.optimize import linprog
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _full_conv(a, b):
    n_a, n_b = len(a), len(b)
    n_fft = 1 << (n_a + n_b - 2).bit_length()
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    return np.fft.irfft(A * B, n_fft)[:n_a + n_b - 1]


def _score(a):
    a = _normalize(a)
    n = len(a)
    b = _full_conv(a, a)
    return 2.0 * n * float(np.max(b))


def _conv_matrix(a):
    """Build the convolution matrix C where C[k,j] = a[k-j] if valid."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    # C[k, j] = a[k-j] if 0 <= k-j < n
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            C[k, j] = a[k - j]
    return C


def _lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    C = _conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        return np.maximum(0, res.x)
    return np.ones(n) / n

"""
Search for low-peak nonnegative self-convolutions using LP-based directional
optimization and gradient descent with log-sum-exp relaxation.

The objective is score(a) = 2*n*max(a*a) / (sum(a))^2, minimized over
nonnegative sequences. The approach alternates between:
1. LP-based directional steps (find best direction keeping convolution bounded)
2. Gradient descent on a smooth LSE surrogate
3. Peak-targeted perturbations
Multi-length search with resampling finds the optimal shape.
"""

import numpy as np
from scipy.optimize import linprog
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _norm(a):
    """Normalize to sum=1, nonnegative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _conv(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n + m - 1]


def _self_conv(a):
    """Full self-convolution a*a."""
    n = len(a)
    sz = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, sz)
    return np.fft.irfft(A * A, sz)[:2 * n - 1]


def _eval(a):
    """Score: 2*n*max(a*a)."""
    a = _norm(a)
    b = _self_conv(a)
    return 2.0 * len(a) * float(np.max(b))


def _build_C(a):
    """Build convolution matrix C[k,j] = a[k-j] if valid."""
    n = len(a)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)

"""
Search for low-peak nonnegative self-convolutions using LP-based directional
optimization combined with gradient descent on a log-sum-exp surrogate.

The objective is score(a) = 2*n*max(a*a) / (sum(a))^2 over nonnegative sequences.
Key innovations over the previous approach:
1. LP-based directional steps: find the best direction that keeps the cross-
   convolution bounded, giving structured improvements.
2. Efficient FFT-based gradient computation for the LSE surrogate.
3. Multi-length exploration with intelligent resampling.
4. Peak-targeted perturbations for escaping local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _norm(a):
    """Normalize nonnegative array to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _conv(a, b):
    """Full linear convolution via zero-padded FFT."""
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n + m - 1]


def _self_conv(a):
    """Full self-convolution a*a via FFT."""
    n = len(a)
    sz = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, sz)
    return np.fft.irfft(A * A, sz)[:2 * n - 1]


def _eval(a):
    """Compute score = 2*n*max(a*a) for normalized a."""
    a = _norm(a)
    b = _self_conv(a)
    return 2.0 * len(a) * float(np.max(b))


def _build_C(a):
    """Build convolution constraint matrix C[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        C[k, jlo:jhi + 1] = a[k - jhi:k - jlo + 1][::-1]
    return C


def _lp_direction(a, M, time_limit=20.0):
    """
    Solve LP: maximize sum(b) subject to (a*b)_k <= M for all k, b >= 0.
    Returns optimal b (unnormalized) or uniform fallback.
    """
    n = len(a)
    C = _build_C(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return np.ones(n, dtype=np.float64) / n


def _lp_refine(a, deadline, rng, verbose=True):
    """
    LP-based iterative refinement.
    Alternates between LP direction finding and convex combination updates.
    Returns (best_a, best_score).
    """
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_score = _eval(a)
    t = 0.25
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        b_conv = _self_conv(a)
        M = float(np.max(b_conv))
        if M <= 1e-300:
            break

        b_dir = _lp_direction(a, M, time_limit=min(20.0, max(2.0, deadline - time.monotonic() - 5)))
        s = float(b_dir.sum())
        if s <= 1e-300:
            break
        b_dir = b_dir / s

        # Try current step size
        a_new = _norm((1.0 - t) * a + t * b_dir)
        new_score = _eval(a_new)

        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            t = min(0.45, t * 1.08)
        else:
            # Try smaller step
            a_new2 = _norm((1.0 - t * 0.4) * a + t * 0.4 * b_dir)
            new_score2 = _eval(a_new2)
            if new_score2 < best_score - 1e-14:
                best_score = new_score2
                best_a = a_new2.copy()
                a = a_new2
                stall = 0
                t = max(0.03, t * 0.6)
            else:
                t = max(0.03, t * 0.75)
                stall += 1

        # Perturbation to escape local minima
        if stall > 25 and time.monotonic() < deadline - 10:
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = _norm(best_a * noise)
            t = 0.2
            stall = 0

        it += 1
        if verbose and it % 50 == 0:
            print(f"  LP it={it} score={best_score:.8f} t={t:.4f}", flush=True)

    return best_a, best_score


def _gradient_refine(a, deadline, rng, verbose=True):
    """
    Gradient descent on LSE surrogate with FFT-based gradient.
    Returns (best_a, best_score).
    """
    a = _norm(a)
    n = len(a)
    best_a = a.copy()
    best_score = _eval(a)
    alpha = 0.15
    beta = 15.0
    fail = 0
    last_improve = 0
    it = 0

    while time.monotonic() < deadline:
        b_conv = _self_conv(a)
        maxb = float(np.max(b_conv))
        score = 2.0 * n * maxb

        if maxb <= 1e-300:
            maxb = 1e-300

        beta_eff = min(500.0, max(5.0, 40.0 / maxb))

        # LSE weights
        arg = beta_eff * (b_conv - maxb)
        arg = np.maximum(arg, -700.0)
        w = np.exp(arg)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.ones_like(b_conv)
            w_sum = float(len(b_conv))
        p = w / w_sum

        # Gradient: grad[i] = 2 * sum_j a[j] * p[j+i]
        # Computed via FFT cross-correlation
        p_rev = p[::-1]
        conv_ap = _conv(a, p_rev)
        # grad[i]/2 = conv_ap[2n-2-i] for i=0..n-1
        # conv_ap has length n + (2n-1) - 1 = 3n-2
        # We need conv_ap[2n-2], conv_ap[2n-3], ..., conv_ap[n-1]
        grad = 2.0 * conv_ap[n - 1:2 * n - 1][::-1]

        # Center gradient for simplex-friendly projection
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale <= 1e-300:
            # Stuck, perturb
            a = _norm(best_a * np.exp(rng.normal(0.0, 0.2, size=n)))
            alpha = 0.1
            fail = 0
            it += 1
            continue

        a_new = _norm(a - alpha * grad_c / scale)
        new_score = _eval(a_new)

        # Check smooth objective improvement
        b_new = _self_conv(a_new)
        mb_new = float(np.max(b_new))
        if mb_new <= 1e-300:
            mb_new = 1e-300
        arg_new = beta_eff * (b_new - mb_new)
        arg_new = np.maximum(arg_new, -700.0)
        lse_new = mb_new + float(np.log(max(float(np.exp(arg_new).sum()), 1e-300))) / beta_eff
        lse_old = maxb + float(np.log(w_sum)) / beta_eff

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            last_improve = it

        if lse_new < lse_old - 1e-12:
            a = a_new
            alpha = min(1.5, alpha * 1.1)
            fail = 0
        else:
            alpha *= 0.5
            fail += 1

        # Periodic beta increase
        if it % 80 == 0:
            beta = min(500.0, beta + 3.0)

        # Escaping local minima
        if (fail > 80 or (it - last_improve > 500 and it > 100)) and time.monotonic() < deadline - 5:
            a = _norm(best_a * np.exp(rng.normal(0.0, 0.15, size=n)))
            alpha = 0.12
            fail = 0
            beta = max(10.0, beta * 0.5)

        it += 1
        if verbose and it % 100 == 0:
            print(f"  GD it={it} score={best_score:.8f} alpha={alpha:.4f} beta={beta:.1f}", flush=True)

    return best_a, best_score


def _resample(old, new_n):
    """Resample sequence to new length using linear interpolation."""
    old = _norm(old)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _norm(new)


def _make_exp(n, alpha=1.0):
    """Exponential decay sequence."""
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-alpha * x)
    return _norm(a)


def _make_gauss(n, sigma=0.3):
    """Gaussian centered sequence."""
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return _norm(a)


def _make_ramp(n, beta=1.0):
    """Power-law decay (generalized exponential)."""
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (1.0 - x) ** beta + 0.01
    return _norm(a)


def _smooth_perturb(a, rng, strength=0.15):
    """Apply smooth (band-limited) multiplicative perturbation."""
    n = len(a)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 8)
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = (f - float(np.mean(f))) / (float(np.std(f)) + 1e-300)
    a2 = a * np.exp(strength * f)
    return _norm(a2)


def _peak_chase(a, rng, top_k=5):
    """
    Peak-targeted perturbation: identify peaks in a*a and reduce
    contributions from the largest elements near those peaks.
    """
    n = len(a)
    b = _self_conv(a)
    peak_indices = np.argsort(b[-1:top_k * 3])[-top_k:]
    
    a2 = a.copy()
    for k in peak_indices:
        # Contributions to b[k] come from pairs (i, k-i)
        # Find the i that maximizes a[i]*a[k-i]
        valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(valid_i) == 0:
            continue
        vals = a[valid_i] * a[k - valid_i]
        j_max = valid_i[np.argmax(vals)]
        # Reduce the larger of the pair
        if a[j_max] >= a[k - j_max]:
            a2[j_max] *= 0.95
        else:
            a2[k - j_max] *= 0.95
    
    # Add small smooth noise to avoid degeneracy
    a2 *= np.exp(rng.normal(0, 0.02, size=n))
    return _norm(a2)

    phase = 0

            if res_score < best_score:

                best_a = res_a.copy()
                a = res_a
            else:
                a = best_a.copy()
            if verbose:

        elif phase % 3 == 1:

            if time.monotonic() > deadline - 45:
                break
            
            t_end = min(deadline - 30, time.monotonic() + time_per_start)
            if t_end <= time.monotonic() + 10:
                continue
            
            print(f"\nLength={n_target}, start={idx+1}/{len(starts)}, time_alloc={time_per_start:.0f}s", flush=True)
            
            # Combined refinement
            t_end = min(deadline - 60, time.monotonic() + time_per_start)
            res_a, res_score = _combined_refine(a_s, t_end, rng, use_lp=(n_target <= 2048), verbose=True)
            
            if res_score < best_score:
                best_score = res_score
                best_a = res_a.copy()
                print(f"  NEW BEST: {best_score:.8f} (n={n_target})", flush=True)
            else:
                print(f"  No improvement (got {res_score:.8f}, best {best_score:.8f})", flush=True)

    # Final deep refinement on the best found
    if time.monotonic() < deadline - 30:
        print(f"\nFinal refinement (n={best_a.size}, score={best_score:.8f})", flush=True)
        t_end = deadline - 15.0
        res_a, res_score = _combined_refine(best_a.copy(), t_end, rng, use_lp=(best_a.size <= 2048), verbose=True)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()
            print(f"  Final score: {best_score:.8f}", flush=True)
        else:
            print(f"  No final improvement, keeping {best_score:.8f}", flush=True)

    # Additional: try gradient-only refinement in the remaining time
    if time.monotonic() < deadline - 10:
        t_end = deadline - 5.0
        res_a, res_score = _gradient_refine(best_a.copy(), t_end, rng, verbose=False)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()
            print(f"  Post-GD score: {best_score:.8f}", flush=True)

    # Validate and prepare output
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 0.01:
        out = np.ones_like(out) / len(out) * 1.0
        s = float(out.sum())
    out = out / s  # normalize (score is scale-invariant, but ensure sum > 0.01)
    
    # Make sure sum is well above 0.01 for the evaluator
    if float(out.sum()) < 1.0:
        out = out * (1.0 / max(float(out.sum()), 1e-10))
    
    # Cap values at 1000 as the evaluator does
    out = np.minimum(out, 1000.0)
    
    final_score = _eval(out)
    print(f"\nFinal answer: n={len(out)}, eval_score={final_score:.8f}", flush=True)
    
    return [float(x) for x in out]
