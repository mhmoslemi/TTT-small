    if s > 0:

    return a_proj

    score, grad = _score_and_grad(a, top_k)

def _fft_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    n_fft = 1 << (2 * n - 2).bit_length()
    A = fft(a, n_fft)
    return ifft(A * A).real[:2 * n - 1]


def _score_fast(a):
    """Compute score (2*n*max_b/sum_a^2) using FFT."""
    n = len(a)
    b = _fft_conv(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * b.max() / (s * s))


def _grad_at_peaks(a, top_k=5):
    """
    Compute subgradient of max_b (= max of conv(a,a)) w.r.t. a.
    Uses top-k peak weighting for smoothness.
    Returns (weighted_peak_value, subgradient_vector).
    """
    n = len(a)
    b = _fft_conv(a)
    n_b = 2 * n - 1
    k_eff = min(top_k, n_b)
    
    top_idx = np.argpartition(b, -k_eff)[-k_eff:]
    vals = b[top_idx]
    base = float(np.median(b))
    wts = np.maximum(vals - base, 1e-12)
    wts /= wts.sum()
    
    subgrad = np.zeros(n)
    for w, k in zip(wts, top_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if hi >= lo:
            subgrad[lo:hi + 1] += w * 2.0 * a[ki - np.arange(lo, hi + 1)]
    
    weighted_peak = float(np.dot(wts, vals))
    return weighted_peak, subgrad


def _objective(x, top_k=5):

    a = np.maximum(x, 0.0)
    s = float(a.sum())

    n = len(a)
    b = _fft_conv(a)
    max_b = float(b.max())

def _objective_grad(x, top_k=5):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = _fft_conv(a)
    max_b = float(b.max())
    score = 2.0 * n * max_b / (s * s)
    
    # Subgradient of max_b
    k_eff = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(b, -k_eff)[-k_eff:]
    vals = b[top_idx]
    base = float(np.median(b))
    wts = np.maximum(vals - base, 1e-12)
    wts /= wts.sum()
    
    subgrad = np.zeros(n)
    for w, k in zip(wts, top_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if hi >= lo:
            subgrad[lo:hi + 1] += w * 2.0 * a[ki - np.arange(lo, hi + 1)]
    
    wmax = float(np.dot(wts, vals))
    # d(score)/da[m] = 2n*(subgrad[m]*s - 2*wmax*1) / s^3
    grad = 2.0 * n * (subgrad * s - 2.0 * wmax) / (s ** 3)
    
    # For x where x[m] <= 0, gradient should be 0 (handled by bounds)
    # But since we clamp with np.maximum, the effective gradient for the
    # optimizer is just grad (L-BFGS-B with bounds handles the rest)
    
    return score, grad


def _project_simple(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _run_lbfqsb(a0, deadline, top_k=5, max_iter=500):
    """Run L-BFGS-B optimization from a0 until deadline."""
    n = len(a0)

        return _objective_grad(x, top_k=top_k)

    x_opt = result.x
    x_opt = np.maximum(x_opt, 0.0)

Minimizes 2*n*max(autocorrelation)/(sum)^2 using a multi-phase strategy:
1. L-BFGS-B with top-k subgradient from parent and diverse starts
2. Adaptive projected gradient descent with momentum
3. Coarse-to-fine optimization at multiple resolutions
4. Final L-BFGS polish with sharp subgradients
Uses FFT for O(n log n) convolution. All operations vectorized.
"""
import numpy as np

from scipy.optimize import minimize
from scipy.fft import fft, ifft


def _fft_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    n_fft = 1 << (2 * n - 2).bit_length()
    A = fft(a, n_fft)
    return ifft(A * A).real[:2 * n - 1]


def _score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    b = _fft_conv(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * b.max() / (s * s))


def _score_grad(x, top_k=5):
    """Returns (score, gradient) using top-k peak subgradient."""
    n = len(x)
    s = float(x.sum())
    if s < 0.01:
        return 1e10, np.zeros(n)
    b = _fft_conv(x)
    max_b = float(b.max())
    score = 2.0 * n * max_b / (s * s)

    k_eff = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(b, -k_eff)[-k_eff:]
    vals = b[top_idx]
    base = float(np.percentile(b, 50))
    wts = np.maximum(vals - base, 1e-12)
    wts /= wts.sum()

    sg = np.zeros(n)
    for w, k in zip(wts, top_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if hi >= lo:
            sg[lo:hi + 1] += w * 2.0 * x[ki - np.arange(lo, hi + 1)]

    wmax = float(np.dot(wts, vals))
    grad = 2.0 * n * (sg * s - 2.0 * wmax) / (s ** 3)
    return score, grad


def _normalize(a, target=1.0):
    s = float(a.sum())
    if s < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / s)

    best = a.copy()

        _, g = _score_grad(a, top_k=top_k)

        a = a - lr * momentum
        a = np.maximum(a, 0.0)

            momentum = np.zeros(n)

    """L-BFGS-B with wall-clock time limit."""
