    a = np.zeros(n, dtype=np.float64)

    a = a / (4.0 ** j)

    if s > 0:

    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:len(a)*2-1]
    return np.maximum(b, 0.0)

    n = len(a)
    m = _next_pow2(2*n - 1)
    b = _conv_fft(a, m)
    S = a.sum()
    if S < 1e-300:
        return np.inf
    return float(2.0 * n * b.max() / (S*S))

    t = state['t'] + 1
    state['t'] = t
    state['m'] = 0.9 * state['m'] + 0.1 * grad
    state['v'] = 0.999 * state['v'] + 0.001 * grad * grad

    v_hat = state['v'] / (1 - 0.999**t)
    v = v - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    return v

        tau_frac = 0.15 * (1.0 - progress) + 0.02
        tau = tau_frac * best_score / (2.0 * n)  # Scale by current score
        
        # Top-k fraction schedule
        top_k_frac = max(0.03, 0.20 * (1.0 - progress) + 0.03)
        
        # Compute a, b
        v_max = v.max()
        e = np.exp(v - v_max)
        a = e / e.sum()
        
        b = _conv_fft(a, m)
        bmax = b.max()
        if bmax < 1e-300:
            break
        
        # Soft-max weights
        x = b / (tau_frac * bmax + 1e-30)
        x = x - x.max()
        w = np.exp(x)
        ws = w.sum()
        if ws < 1e-300 or not np.isfinite(ws):

        w /= ws

        k = max(1, int(top_k_frac * len(b)))

            w[idx] *= 3.0

        P_w = np.fft.fft(w, m)

        if not np.isfinite(grms) or grms < 1e-15:
            # Add noise
            v += rng.normal(0, 0.1, n)
            v = v - v.mean()
            continue
        grad_v /= grms

        lr = 0.3 * (1.0 - 0.7 * progress)

        v = np.clip(v, -30, 30)
        v = v - v.mean()
        
        it_count += 1

                best_a = a.copy()
                last_improve = time.monotonic()

            v = np.clip(v, -30, 30)
            last_improve = time.monotonic()

    best_score = _score(a_final)

Uses Adam optimization in log-space with:
- Correct gradient: 4n*a*(cross_corr(w,a) - dot(w,b))
- Adaptive soft-max temperature for peak focusing
- Top-k boost for additional peak concentration
- Cosine learning rate annealing
- Stuck detection with random perturbations

import numpy as np
import time
import sys
from math import lgamma

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p

def _binomial_seq(n):
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    for j in range(1, n):
        a[j] = a[j-1] * (2.0*j - 1.0) / (2.0*j)
    s = a.sum()
    if s > 0:
        a /= s
    return a

def _conv_fft(a, m):
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2*len(a)-1]
    return np.maximum(b, 0.0)

def _score(a):
    n = a.size
    m = _next_pow2(2*n - 1)
    b = _conv_fft(a, m)
    S = a.sum()
    if S < 1e-300:
        return np.inf
    return float(2.0 * n * b.max() / (S*S))

    Adam optimization in log-space to minimize 2n*max(conv(a,a))
    (with sum(a)=1 parameterization).

    v = np.clip(v, -30, 30)

    best_score = float(2.0 * n * best_bmax)  # since sum(a)=1
    
    last_improve = start_time
    it_count = 0

        if elapsed < 1e-6:

        progress = min(1.0, elapsed / max(time_budget, 1.0))

        v_max = v.max()
        e = np.exp(v - v_max)
        a = e / e.sum()

        if bmax < 1e-300 or not np.isfinite(bmax):
