"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using accelerated gradient (FISTA) with
FFT-based convolution, soft-max subgradient, multiple diverse restarts, and
periodic perturbation to escape local minima. The problem is non-convex, so
we use a multi-start strategy with diverse initializations.

import numpy as np
import time


def project_simplex(v):
    """Project v onto {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    fc = fa * fa
    c = np.fft.irfft(fc, fft_size)[:size]
    # Clean up tiny negatives from FFT
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def subgradient_softmax(a, c, k_idx, temperature):
    """
    Compute a soft subgradient: weighted average of gradients of top-k
    convolution coefficients, weighted by softmax of their values.
    """
    n = len(a)
    # Get top-k indices
    if k_idx >= len(c):
        k_idx = len(c)
    top_vals = np.partition(c, -k_idx)[-k_idx:]
    top_sorted = np.sort(top_vals)[::-1]
    
    # Softmax weights
    shifted = top_sorted - top_sorted[0]
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    
    # Get actual indices
    # This is expensive if k_idx is large; for small k_idx it's fine
    if k_idx <= 50:
        mask = c >= top_vals[0] - 1e-15  # all that are at max (approximately)
        # Actually, let's just use the top-k_idx by value
        top_indices = np.argpartition(c, -k_idx)[-k_idx:]
        top_indices = top_indices[np.argsort(c[top_indices])[::-1]]
    else:
        top_indices = np.argsort(c)[::-1][:k_idx]
    
    g = np.zeros(n)
    for w, k in zip(weights, top_indices):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        if j_lo < j_hi:
            js = np.arange(j_lo, j_hi)
            kjs = k - js
            g[js] += 2.0 * w * a[kjs]
    return g


def run_fista_fista(a0, n, duration, rng, top_k=5, temp=0.01):
    """
    FISTA with FFT convolution and soft subgradient.
    Returns (best_a, best_score).
    """
    a = a0.copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    a_prev = a.copy()
    t = 1.0
    L = 2.0  # Lipschitz constant
    step = 1.0 / L
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + duration
    it = 0
    no_improve = 0
    sc_same_count = 0
    
    print(f"    FISTA start: score={best_score:.8f}")
    
    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute convolution
        c = fast_convolve(y)
        
        # Soft subgradient
        g = subgradient_softmax(y, c, top_k, temp)
        
        # Gradient step
        a_new = y - step * g
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                sc_same_count = 0
            else:
                no_improve += 1
            
            # Adaptive: if stuck, perturb
            if no_improve > 200:
                # Small random perturbation
                noise = rng.normal(0, 0.005 / np.sqrt(n), n)
                a = best_a + noise
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t = 1.0
                no_improve = 0
                # Increase temperature (smoother gradient)
                temp = min(1.0, temp * 1.1)
                print(f"    FISTA perturbation at it={it}, score={best_score:.8f}, temp={temp:.4f}")
            
            # Gradually increase top_k
            if it % 10000 == 0 and top_k < 20:
                top_k += 2
                temp *= 0.9  # Make it sharper as we converge
            
            # Occasionally decrease step size
            if no_improve > 100:
                step *= 0.95
                if step < 0.1:
                    step = 1.0 / L
        
        if it % 50000 == 0:
            print(f"    FISTA it={it}, score={best_score:.8f}, top_k={top_k}, temp={temp:.4f}")
    
    # Final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    
    # 1. Uniform
    inits.append(("uniform", np.ones(n) / n))
    
    # 2. Symmetric raised cosine
    x = np.linspace(-1, 1, n)
    rc = 0.5 * (1 + np.cos(np.pi * x))
    inits.append(("raised_cosine", rc / rc.sum()))
    
    # 3. Asymmetric: linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(("ramp", ramp / ramp.sum()))
    
    # 4. Two-block (asymmetric)
    tb = np.ones(n)
    tb[n//3:] = 0.5
    inits.append(("two_block", tb / tb.sum()))
    
    # 5. Gaussian bump (off-center)
    centers = [n // 4, n // 2, 3 * n // 4]
    for ci, center in enumerate(centers):
        sigma = n // 8
        g = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        inits.append((f"gauss_{ci}", g / g.sum()))
    
    # 6. Random exponential
    rx = rng.exponential(1.0, n)
    inits.append(("exp_rand", rx / rx.sum()))
    
    # 7. Sparse random
    sr = rng.exponential(1.0, n)
    mask = rng.random(n) > 0.7
    sr[~mask] = 0
    if sr.sum() > 1e-15:
        inits.append(("sparse_rand", sr / sr.sum()))
    
    # 8. Cube wave (alternating blocks)
    cw = np.zeros(n)
    block = max(1, n // 20)
    for i in range(0, n, 2 * block):
        cw[i:min(i + block, n)] = 1.0
    if cw.sum() > 1e-15:
        inits.append(("cube_wave", cw / cw.sum()))
    
    # 9. Sine-based
    freqs = [1, 2, 3]
    for f in freqs:
        s = np.sin(np.pi * f * np.arange(n) / n) ** 2
        if s.sum() > 1e-15:
            inits.append((f"sine_{f}", s / s.sum()))
    
    # 10. Power law
    pl = np.arange(1, n + 1) ** 0.5
    inits.append(("power_law", pl / pl.sum()))
    
    # 11. Triangle (asymmetric)
    tri = np.zeros(n)
    peak = n // 3
    for i in range(n):
        tri[i] = 1.0 - abs(i - peak) / max(peak, n - peak)
    tri = np.maximum(tri, 0)
    if tri.sum() > 1e-15:
        inits.append(("asym_triangle", tri / tri.sum()))
    
    # 12. Hyperbolic decay
    hd = 1.0 / (1.0 + np.arange(n).astype(np.float64))
    inits.append(("hyperbolic", hd / hd.sum()))
    
    return inits
