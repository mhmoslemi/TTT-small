```python
"""
LP-based optimization for minimum peak self-convolution of nonnegative sequences.

Strategy:
- Uses the state-of-the-art LP direction-finding method: at each step, solve an LP
  to find the best direction g that keeps cross-convolution (f*g) bounded by current peak M.
- Combines with bisection line search, multiple sequence lengths, and LSE gradient polish.
"""

import numpy as np
import time
from scipy.optimize import linprog

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _convolve_full(a, fft_size=None):
    n = a.size
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2 * n - 1]


def _convolve_cross(a, b, fft_size=None):
    n = a.size
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:2 * n - 1]


def _score(a):
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return 1e100
    b = _convolve_full(a)
    return 2.0 * n * b.max() / (s * s)


def _normalize(a):
    a = np.maximum(a, 0.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(a.size) / a.size
    return a / s


def _build_lp_matrix(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_direction_lp(f, M, n):
    A = _build_lp_matrix(f, n)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 5})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _line_search(a, g, M, n, max_iter=40):
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = 0.5 * (lo + hi)
        a_new = (1.0 - mid) * a + mid * g
        b_new = _convolve_full(a_new)
        if b_new.max() <= M * (1.0 + 1e-12):
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    return lo


def _make_exponential(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_symmetric_exp(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x) + 0.5 * np.exp(-alpha * (1.0 - x))
    return _normalize(a)


def _make_gaussian_mix(n, rng):
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.1
    for _ in range(rng.integers(1, 5)):
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.3)
        a += np.exp(-((x - center) / width) ** 2)
    return _normalize(a)


def _resample_to_len(a, new_n):
    a = _normalize(a)
    if a.size == new_n:
        return a.copy()
    if a.size == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return _normalize(np.interp(x_new, x_old, a))


def _perturb(a, rng, strength=0.15):
    noise = np.exp(rng.normal(0, strength, size=a.size))
    a2 = a * noise
    return _normalize(a2)


def _lse_beta_value_grad(b, beta):
    """Compute LSE(b)/beta and its gradient w.r.t. a (via weights p)."""
    m = b.max()
    if beta <= 1.0:
        p = np.ones(b.size) / b.size
        val = m
    else:
        arg = np.maximum(beta * (b - m), -745.0)
        e = np.exp(arg)
        s = e.sum()
        if s < 1e-300:
            p = np.ones(b.size) / b.size
            val = m
        else:
            p = e / s
            val = m + np.log(s) / beta
    return val, p


def _lse_gradient_step(a, p, n):
    """Compute gradient of LSE(@(a*a), beta) w.r.t. a using weights p."""
    # grad_i = 2 * sum_k p_k * a_{k-i} (with appropriate index handling)
    # This is 2 * cross_correlation(p, a) truncated
    fft_size = _next_pow2(3 * n)
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _lse_polish(a, deadline, rng, max_iter=200000):
    """Polish using LSE gradient descent with adaptive beta."""
    a = _normalize(a.copy())
    n = a.size
    if n < 8:
        return a, _score(a)

    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n)

    b = _convolve_full(a, b_fft)
    M = b.max()
    score = 2.0 * n * M
    best_a = a.copy()
    best_score = score

    beta = 50.0
    alpha = 0.5 / n
    it = 0
    stall = 0

    while time.monotonic() < deadline and it < max_iter:
        val_old, p = _lse_beta_value_grad(b, beta)
        grad = _lse_gradient_step(a, p, n)

        # Center the gradient to stay on simplex (approximately)
        grad_centered = grad - grad.sum() / n
        grad_norm = np.sqrt((grad_centered ** 2).sum())
        if grad_norm < 1e-30:
            break

        # Step
        a_new = a - alpha * grad_centered
        a_new = _normalize(a_new)

        b_new = _convolve_full(a_new, b_fft)
        M_new = b_new.max()
        val_new, _ = _lse_beta_value_grad(b_new, beta)

        if val_new < val_old - 1e-15:
            a = a_new
            b = b_new
            M = M_new
            alpha = min(2.0 / n, alpha * 1.05)
            stall = 0
        else:
            alpha *= 0.7
            stall += 1

        s_curr = 2.0 * n * M
        if s_curr < best_score - 1e-12:
            best_score = s_curr
            best_a = a.copy()

        if it % 200 == 0:
            beta = min(500.0, beta * 1.3)

            a = _perturb(best_a, rng, 0.1)
            b = _convolve_full(a, b_fft)
            M = b.max()
            alpha = 0.5 / n
            beta = max(30.0, beta * 0.5)
            stall = 0

        it += 1

    return best_a, best_score


def _lp_optimize(a, deadline, rng, stagnation_limit=500):

    a = _normalize(a.copy())
    n = a.size
    if n < 8:
        n = 128
        a = _make_exponential(n)

    b = _convolve_full(a)
    M = b.max()
    score = 2.0 * n * M
    best_a = a.copy()
    best_score = score
    it = 0
    stall = 0
    last_print = 0

    while time.monotonic() < deadline:
        # Solve LP for direction
        g0 = _solve_direction_lp(a, M, n)
        if g0 is None or g0.sum() < 1e-10:
            stall += 1
            if stall > 50:
                break
            a = _perturb(best_a, rng, 0.1)
            b = _convolve_full(a)
            M = b.max()
            continue

        s_g0 = g0.sum()
        if s_g0 > best_score / (2.0 * n) + 1e-10:
            # There's room to improve
            pass

        g = g0 / s_g0

        # Line search
        t = _line_search(a, g, M, n)

        if t < 1e-8:
            stall += 1
            if stall > stagnation_limit:
                # Try larger perturbation
                a = _perturb(best_a, rng, 0.25)
                b = _convolve_full(a)
                M = b.max()
                stall = 0
            elif stall > 20:
                a = _perturb(best_a, rng, 0.05)
                b = _convolve_full(a)
                M = b.max()
            continue

        # Update
        a = (1.0 - t) * a + t * g
        a = _normalize(a)
        b = _convolve_full(a)
        M_new = b.max()
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        M = M_new

        if it % 100 == 0 and time.monotonic() - last_print > 10:
```

LP-based optimization for minimum peak self-convolution of nonnegative sequences.

Strategy:
- Uses the state-of-the-art LP direction-finding method: at each step, solve an LP
  to find the best direction g that keeps cross-convolution (f*g) bounded by current peak M.
- Combines with bisection line search, multiple sequence lengths, and LSE gradient polish.

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _convolve_full(a, fft_size=None):
    n = a.size
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:2 * n - 1]


def _score(a):
    n = a.size

    b = _convolve_full(a)
    return 2.0 * n * b.max() / (s * s)


def _normalize(a):
    a = np.maximum(a, 0.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()

    for j in range(n):

    return A

                      options={"presolve": True, "time_limit": 10})
