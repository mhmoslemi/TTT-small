"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Subgradient method with active peak tracking on the simplex
- Simulated annealing with periodic perturbations
- Block coordinate refinement for fine-tuning
- Multiple restarts with diverse initializations
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def grad_conv_k(a, k):
    """Gradient of the k-th convolution coefficient w.r.t. a."""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        idx = k - np.arange(lo, hi, dtype=np.int64)
        g[lo:hi] = a[idx]
    # For the diagonal term (when k is even and k//2 is in range)
    if k % 2 == 0:
        j = k // 2
        if 0 <= j < n:
            g[j] += a[j]
    return g


def subgradient_step(a, top_k=5, step_size=0.5, rng=None):
    """Take one subgradient step on the max convolution coefficient."""
    n = len(a)
    c = np.convolve(a, a)
    
    if top_k >= c.size:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    # Weight by proximity to max (soft selection)
    weights = c[top_idx]
    w_max = weights.max()
    w_min = weights.min()
    if w_max - w_min > 1e-15:
        weights = (weights - w_min) / (w_max - w_min) + 0.1
    else:
        weights = np.ones_like(weights)
    weights /= weights.sum()
    
    g = np.zeros(n, dtype=np.float64)
    for mk, w in zip(top_idx, weights):
        g += w * grad_conv_k(a, int(mk))
    
    a_new = a - step_size * g
    a_new = project_simplex(a_new)
    return a_new, c.max(), top_idx

        for _ in range(anneal_interval):
            if time.monotonic() >= deadline:

                best_score = sc_pert
                best_a = a_pert.copy()

            elif sc_pert2 < best_score:
                a = a_pert2
                best_score = sc_pert2
                best_a = a_pert2.copy()
                print(f"    Anneal (from best) improved to {best_score:.7f}")

    # Final check

def block_coordinate_refine(a0, duration, rng, block_size=200):
    """
    Refine using block coordinate descent.
    Randomly select blocks of coordinates and optimize within them.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    
    it = 0
    while time.monotonic() < deadline:
        # Select random block
        start = rng.integers(0, max(1, n - block_size))
        end = start + block_size
        block = a[start:end].copy()
        
        # Optimize the block: we want to minimize max(conv(a))
        # while keeping sum(a) = 1 and a >= 0
        # Fix all other coordinates, optimize the block
        # The convolution coefficients depend on the block in a specific way
        
        # Simple approach: do a few gradient steps within the block
        for _ in range(5):
            c = np.convolve(a, a)
            top_idx = np.argpartition(c, -3)[-3:]
            
            g_block = np.zeros(block_size, dtype=np.float64)
            for mk in top_idx:
                g_full = grad_conv_k(a, int(mk))
                g_block += g_full[start:end]
            g_block /= len(top_idx)
            
            # Gradient step on the block

    if sc < best_score:

"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using corrected subgradient method
with simulated annealing, population-based evolution, and block refinement.
Key improvement: corrected gradient computation (factor of 2).
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def correct_grad_conv_k(a, k):
    """
    Correct gradient of the k-th full-convolution coefficient w.r.t. a.
    g[j] = 2 * a[k-j] for j in [max(0,k-n+1), min(n-1,k)], else 0.
    """
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    j_min = max(0, k - n + 1)
    j_max = min(n, k + 1)
    if j_min < j_max:
        g[j_min:j_max] = 2.0 * a[k - np.arange(j_min, j_max, dtype=np.int64)]
    return g


def compute_subgradient(a, top_k=5, soft_weight=True):
    """Compute a weighted subgradient of max_k conv_k(a)."""
    n = len(a)
    c = np.convolve(a, a)
    cmax = c.max()
    
    if top_k >= c.size:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    if soft_weight:
        # Weight by proximity to max
        c_top = c[top_idx]
        w_min = c_top.min()
        w_range = cmax - w_min
        if w_range > 1e-15:
            weights = (c_top - w_min) / w_range + 0.1
        else:
            weights = np.ones(len(top_idx))
    else:
        weights = np.ones(len(top_idx))
    
    weights = weights / weights.sum()
    
    g = np.zeros(n, dtype=np.float64)
    for mk, w in zip(top_idx, weights):
        g += w * correct_grad_conv_k(a, int(mk))
    
    return g, cmax


def run_optimization(a0, duration, rng, step_size=0.15, top_k=5,

    Run subgradient optimization with annealing.

    """
    n = len(a0)
    a = a0.copy()
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    step = step_size
    anneal = initial_anneal
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Compute subgradient and take step
        g, cmax = compute_subgradient(a, top_k=top_k, soft_weight=True)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            a = a - step * g / g_norm * np.max(np.abs(g))
            # Actually, let's use a simpler step:
            a = a - step * g
        a = project_simplex(a)
        it += 1
        
        # Periodic evaluation
        if it % 200 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Annealing
        if it % anneal_every == 0 and it > 0:
            if no_improve > 5:

                noise = rng.normal(0, anneal / np.sqrt(n), n)

                else:

                if no_improve > 15:

    Try multiple random restarts from a0 and keep the best.

    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration_per * n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() >= deadline:
            break
        
        # Create perturbed start

        a_start = a0 + noise
        a_start = project_simplex(a_start)
        
        sc_start = fast_score(a_start)
        if sc_start < best_score:
            best_a = a_start.copy()
            best_score = sc_start
        
        # Optimize from this start
        a_r, s_r = run_optimization(
            a_start, duration_per, rng,
            step_size=step_size, top_k=top_k,
            anneal_every=200, initial_anneal=0.005,
            anneal_decay=0.97, min_step=0.01
        )
        
        if s_r < best_score:
