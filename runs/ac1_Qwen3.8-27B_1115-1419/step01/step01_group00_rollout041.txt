FFT-accelerated mirror descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative updates (mirror descent) with adaptive annealing,
multiple diverse candidates, and backtracking line search.
Key speedup: FFT-based convolution (O(n log n) vs O(n^2)).

import numpy as np
import time


def _fft_conv(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _soft_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    c = _fft_conv(a)
    p = np.exp((c - c.max()) / tau)
    p /= p.sum()
    # grad[i] = 2 * sum_j p[j] * a[(j-i) mod ...]
    # Actually, we need: sum_j p[j] * a[j-i] for 0 <= j-i < n
    # This is the cross-correlation of p and a.
    # Using FFT: IFFT(conj(FFT(p)) * FFT(a))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(fp) * fa, size)[:n]
    return 2.0 * corr, c


def _hard_grad(a, k):
    """Exact gradient of conv(a,a)_k w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    idx = np.arange(n)
    valid = (k - idx >= 0) & (k - idx < n)
    grad[valid] = 2.0 * a[k - idx[valid]]
    return grad


def _mirror_update(a, grad, eta):
    """Mirror descent: a <- a * exp(-eta*(grad - mean(grad))), preserve sum."""
    centered = grad - grad.mean()
    # Clip to avoid overflow
    centered = np.clip(centered, -10.0, 10.0)
    a_new = a * np.exp(-eta * centered)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    target = a.sum()
    if s < 1e-30:
        return a.copy()
    return a_new * (target / s)


def _additive_update(a, grad, eta, n):
    """Additive gradient descent with projection to simplex {x>=0, sum(x)=target}."""
    centered = grad - grad.mean()
    a_new = a - eta * centered
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    target = a.sum()
    if s < 1e-30:
        return a.copy()
    return a_new * (target / s)

    s = a.sum()
    if s < 1e-10:

    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s * s)

    s = a_new.sum()
    if s < 1e-30:

    return a_new * (a.sum() / s)

"""
FFT-accelerated mirror descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative (mirror) updates with adaptive temperature annealing,
multiple diverse candidates, backtracking, and periodic perturbation.
Key speedup: FFT-based self-convolution gives O(n log n) per iteration instead of O(n^2).
"""

import numpy as np
import time


def _fft_conv(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _compute_objective(a, n):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * float(c.max()) / (s * s)


def _mirror_step(a, grad, eta):
    """Mirror descent update: a_new[i] = a[i]*exp(-eta*g[i]) normalized to same sum."""
    scaled = -eta * grad
    scaled = np.clip(scaled, -50.0, 50.0)
    e = np.exp(scaled)
    aw = a * e
    Z = aw.sum()
    if Z < 1e-30:
        return a.copy()
    a_new = aw / Z * a.sum()
    return np.maximum(a_new, 0.0)


def _additive_step(a, grad, eta):
    """Additive gradient descent with projection to {x>=0} and sum preservation."""
    centered = grad - grad.mean()
    a_new = a - eta * centered
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new * (a.sum() / s)


def _smoothed_gradient(a, c, tau, n, fft_size):
    """Compute softmax-weighted gradient of the convolution peak."""
    p = np.exp((c - c.max()) / tau)
    p /= p.sum()
    fp = np.fft.rfft(p, fft_size)
    fa = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(fp) * fa, fft_size)[:n]
    return 2.0 * corr


def _hard_gradient(a, k, n):
    """Exact gradient of conv(a,a)_k w.r.t. a."""
    grad = np.zeros(n, dtype=np.float64)
    idx = np.arange(n)
    valid = (k - idx >= 0) & (k - idx < n)
    grad[valid] = 2.0 * a[k - idx[valid]]
    return grad


def _perturb(a, rng, n_blocks, fracs, sigma_lo, sigma_hi):
    """Apply random multiplicative block perturbations."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        f = float(rng.choice(fracs))
        w = max(2, int(n * f))
        s = int(rng.integers(0, max(1, n - w)))
        sig = float(rng.uniform(sigma_lo, sigma_hi))
        a_new[s:s + w] *= np.exp(rng.normal(0.0, sig, size=w))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new * (a.sum() / s)


def _optimize(candidate, n, deadline, tau_start, tau_end, eta_start, tag, rng,
              use_smooth=True, it_cap=50000000):
    """
    Optimize a single candidate until deadline.
    Returns (best_a, best_obj).
    """
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    a = candidate.copy()
    tau = tau_start
    eta = eta_start
    best = a.copy()
    best_o = _compute_objective(a, n)
    stall = 0
    it = 0
    tau_range = tau_start - tau_end
    total_iters_est = 200000  # rough estimate for annealing schedule

    while time.monotonic() < deadline:
        c = _fft_conv(a)
        o_cur = 2.0 * n * float(c.max()) / (a.sum() ** 2)

        if o_cur < best_o - 1e-10:
            best_o = o_cur
            best = a.copy()
            stall = 0
        else:
            stall += 1

        if use_smooth:
            grad = _smoothed_gradient(a, c, tau, n, fft_size)
        else:
            k = int(np.argmax(c))
            grad = _hard_gradient(a, k, n)

        # Mirror descent step
        a_new = _mirror_step(a, grad, eta)
        c_new = _fft_conv(a_new)
