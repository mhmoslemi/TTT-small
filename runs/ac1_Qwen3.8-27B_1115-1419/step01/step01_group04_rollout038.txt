Wait, there's a bug in the c update. When I update a[j] from `base` to `best_local`, the delta is `best_local - base`. The update to c is:
- For k ≠ 2j: c_k += 2 * delta * a_{k-j}
- For k = 2j: c_k += best_local^2 - base^2

But I'm using `a[idx[mask]]` which is the CURRENT value of a at those indices. Since I'm doing coordinate descent (changing one at a time), the other a_i values are fixed, so this is correct.

However, I need to be careful: after I update `a[j] = best_local`, the `a` array has the new value. But in the c update, I'm using `a[idx[mask]]` where idx = ks - j. Since j is the coordinate I'm changing, and idx ranges over [0, n-1], one of these could be j itself (when k - j = j, i.e., k = 2j). But I handle k=2j separately, so the mask excludes it. For other values of idx, they're not j (since idx = k-j and k ≠ 2j means idx ≠ j). So `a[idx[mask]]` gives the correct (unchanged) values. ✓

Let me also think about the order of coordinates. A fixed order (0, 1, 2, ..., n-1) might be fine, but a random order might help avoid systematic biases. I'll use a random order each pass.

2. Renormalize.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using iterative coordinate descent
with perturbative restarts.

Key ideas:
- FFT-based convolution for fast objective evaluation
- Coordinate descent with analytical 1D sub-optimization
- Iterated local search with structured perturbations
- Multiple sequence lengths explored

import numpy as np
import time


def fast_conv(a):
    """FFT-based linear convolution a*a."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def obj_value(a):
    """Compute the objective value."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def normalize(a, target_sum=None):
    """Normalize a to have target_sum (default: len(a))."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    if target_sum is None:
        target_sum = float(n)
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S > 1e-15:
        a *= target_sum / S
    else:
        a.fill(target_sum / n)
    return a


def coord_descent(a, n, max_passes=500, seed=0, verbose=False):

    Coordinate descent to minimize max(c_k) / S^2.
    Works with unnormalized a (objective is scale-invariant).
    """
    rng = np.random.default_rng(seed)
    c = fast_conv(a)
    S = float(a.sum())
    best_score = 2.0 * n * float(c.max()) / (S * S)
    best_a = a.copy()
    no_improve = 0
    
    for p in range(max_passes):
        # Random order for this pass
        order = np.random.default_rng(seed + p).permutation(n)
        
        pass_improved = False
        for j in order:
            base = a[j]
            if base < 1e-12:
                candidates = [0.0, S * 1e-6, S / n * 0.1, S / n * 0.5]
            else:
                candidates = [0.0, base * 0.1, base * 0.3, base * 0.7, base * 1.3, base * 2.0]
            
            best_local = base
            best_local_max = float(c.max())
            best_local_S = S
            
            for aj in candidates:
                aj = max(0.0, aj)
                delta = aj - base
                if abs(delta) < 1e-14:
                    continue
                
                # Compute new max c
                # Affected range: k in [j, j+n-1]
                k_min = j
                k_max = j + n - 1
                if k_max >= 2 * n - 1:
                    k_max = 2 * n - 2
                
                # Max of un-affected part
                if k_min > 0:
                    um = c[:k_min].max()
                else:
                    um = 0.0
                if k_max < 2 * n - 2:
                    um2 = c[k_max+1:].max()
                    if um2 > um:
                        um = um2
                
                # Affected c values
                ks = np.arange(k_min, k_max + 1)
                idx = ks - j  # 0, 1, ..., n-1 (or part of it)
                # c_k' = c_k + 2*delta*a[idx] (for k != 2j)
                # c_{2j}' = c_{2j} + aj^2 - base^2
                
                if 2 * j >= k_min and 2 * j <= k_max:
                    mask = ks != (2 * j)
                    aff_vals = c[ks[mask]] + 2 * delta * a[idx[mask]]
                    aff_2j = c[2 * j] + aj * aj - base * base
                    aff_max = max(aff_vals.max(), aff_2j)
                else:
                    aff_vals = c[ks] + 2 * delta * a[idx]
                    aff_max = aff_vals.max()

                if S_new < 1e-12:
                    continue

                    # But since 2*n is constant, comparing new_max_c/S_new^2 is equivalent
                    best_local_max = sc
                    best_local = aj
                    best_local_S = S_new

                delta = best_local - a[j]
                a[j] = best_local
                
                k_min = j
                k_max = j + n - 1
                if k_max >= 2 * n - 1:
                    k_max = 2 * n - 2
                ks = np.arange(k_min, k_max + 1)
                idx = ks - j
                
                if 2 * j >= k_min and 2 * j <= k_max:

                    c[ks] += 2 * delta * a[idx]

                pass_improved = True

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0

            no_improve += 1

            print(f"  CoordDesc pass {p}: best={best_score:.10f}, current={current_score:.10f}")

    return best_a, best_score


def perturb(a, rng, strength=0.3):
    """Apply a multiplicative perturbation to escape local minima."""
    a = a.copy()
    n = len(a)
    
    # Strategy: multiply a random block by a random factor
    num_blocks = rng.integers(1, 4)
    for _ in range(num_blocks):
        start = rng.integers(0, n)
        length = rng.integers(max(1, n // (2 * num_blocks + 1)), n // (num_blocks + 1) + 1)
        end = min(n, start + length)
        factor = np.exp(rng.normal(0, strength))
        a[start:end] *= factor
    
    # Also try a smooth global perturbation
    t = np.linspace(0, 1, n)
    z = np.zeros(n)
    for m in range(1, 4):
        amp = strength / (m + 1)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * m * t + ph)
    a *= np.exp(z)
    
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n)
    return a


def local_search_ils(a, budget_s, seed=0, verbose=False):
    """
    Iterated Local Search using coordinate descent.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s
    
    best_a = a.copy()
    best_score = obj_value(a)
    
    if verbose:
        print(f"  ILS start: score={best_score:.10f}")

def generate_starts(n, rng, parent_seq=None):

    for freq in (1, 2, 3, 5, 8):
