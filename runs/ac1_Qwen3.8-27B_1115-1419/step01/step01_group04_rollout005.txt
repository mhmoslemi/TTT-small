            a = w * best + (1 - w) * np.exp(rng.normal(0, 0.1, n))
            a = _norm(a, float(n))
            if symmetric:
                a = _symmetrize(a)
                a = _norm(a, float(n))
            c = _fft_conv(a)
            s = a.sum()
            sc = float(2.0 * n * c.max() / (s * s)) if s > 1e-12 else float('inf')
            eta = 0.15
            noise_scale = 0.03
            mom = np.zeros(n)
            stagn = 0

    return best, best_sc


def _peak_shave(a, K_peaks=3, n_shave=10, alpha=0.3):
    """
    Heuristic: directly reduce elements contributing most to top peaks,
    redistribute to least-contributing elements.
    """
    n = a.size
    c = _fft_conv(a)
    top_k = np.argpartition(c, -K_peaks)[-K_peaks:]

    grad = np.zeros(n)
    for k in top_k:
        k = int(k)
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        grad[m_lo:m_hi + 1] += a[k - m_hi:k - m_lo + 1][::-1]

    # Reduce top n_shave grad elements, increase bottom n_shave
    n_shave = min(n_shave, n // 4)
    top_idx = np.argpartition(grad, -n_shave)[-n_shave:]
    bot_idx = np.argpartition(grad, n_shave - 1)[:n_shave] if n_shave > 0 else np.array([], dtype=int)

    cand = a.copy()
    mass_removed = 0.0
    for i in top_idx:
        red = min(alpha * a[i], 0.5 * a[i])
        cand[i] -= red
        mass_removed += red
    if len(bot_idx) > 0 and mass_removed > 0:
        per_bot = mass_removed / len(bot_idx)
        for i in bot_idx:
            cand[i] += per_bot

    return _norm(cand, float(n))


def _try_scipy_polish(a, budget, seed):
    """Use scipy smooth max optimization for final polish."""
    try:
        from scipy.optimize import minimize
    except ImportError:
        return a.copy(), _score_fast(a)

    n = a.size
    a0 = _norm(a.copy(), float(n))

    def objective(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-10:
            return 1e10
        c = _fft_conv(ax)
        T = 0.05
        # Smooth max via log-sum-exp
        cm = c - c.max()
        smax = c.max() + T * np.log(np.sum(np.exp(cm / T)))
        return 2.0 * n * smax / (s * s)

    def jac(x):
        ax = np.maximum(x, 1e-15)
        s = ax.sum()
        c = _fft_conv(ax)
        T = 0.05
        cm = c - c.max()
        weights = np.exp(cm / T)
        weights /= weights.sum()
        # gradient of smooth max
        smax_grad = weights  # shape (2n-1,)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            w = smax_grad[k]
            if w < 1e-10:
                continue
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            g[m_lo:m_hi + 1] += 2.0 * w * ax[k - m_hi:k - m_lo + 1][::-1]
        # Full score gradient: 2n * smax_grad * max_c / s^2 - 2 * 2n*smax / s^2
        smax_val = c.max() + T * np.log(np.sum(np.exp(cm / T)))
        score_grad = (2.0 * n / (s * s)) * g - (4.0 * n * smax_val / (s * s))
        return score_grad

    t0 = time.monotonic()
    best_x = a0.copy()
    best_f = objective(a0)

    for attempt in range(3):
        if time.monotonic() - t0 > budget:
            break
        try:
            res = minimize(objective, best_x, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 50, 'ftol': 1e-10})
            if res.fun < best_f:
                best_x = res.x
                best_f = res.fun
        except Exception:
            break

    best_x = _norm(best_x, float(n))
    return best_x, _score_fast(best_x)


def _upsample(seq, n):
    """Linearly interpolate to length n and normalize."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    if m == 0:
        return np.ones(n)
    if m == n:
        return _norm(seq, float(n))
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    return _norm(np.interp(x_new, x_old, seq), float(n))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-length subgradient optimization with diverse starts,
    symmetry, peak shaving, and scipy polish. Returns best sequence found.
    """
    t_global = time.monotonic()
    budget = max(0.0, float(budget_s))
    print(f"Total budget: {budget:.1f}s")

    # Get parent sequence
    parent_seq = None
    try:
        if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
            parent_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if parent_seq.size > 0:
                print(f"Parent sequence: n={parent_seq.size}, score={_score_fast(parent_seq):.6f}")
    except Exception:
        parent_seq = None

    if parent_seq is None:
        try:
            h = kwargs.get('height_sequence_1', None)
            if h is not None:
                parent_seq = np.asarray(h, dtype=np.float64)
        except Exception:
            pass

    # Time-critical deadline (leave 30s for final eval)
    search_deadline = t_global + max(10.0, budget - 30.0)

    global_best = None
    global_best_sc = float('inf')

    def update_global(seq, sc):
        nonlocal global_best, global_best_sc
        if seq is not None and np.isfinite(sc) and sc < global_best_sc:
            global_best = seq.copy()
            global_best_sc = float(sc)
            print(f"  New global best: score={sc:.6f}, n={seq.size}")

    # Determined lengths to try
    if budget < 120:
        lengths = [512]
    elif budget < 400:
        lengths = [256, 512]
    else:
        lengths = [256, 512, 1024]

    # If parent is 512, make sure we try that
    if parent_seq is not None and parent_seq.size not in lengths:
        if budget > 200:
            lengths.append(parent_seq.size)
            lengths.sort()
        else:
            lengths = [parent_seq.size]

    # Allocate time across lengths (weighted toward larger)
    remain = max(0.0, search_deadline - time.monotonic())
    weights = [1.0, 2.0, 2.5, 3.0, 3.5]
    w_sum = sum(weights[:len(lengths)])
    time_each = [w / w_sum * remain * 0.7 for w in weights[:len(lengths)]]
    # Reserve 30% for final polish
    polish_budget = remain * 0.3

    print(f"Lengths to try: {lengths}")
    print(f"Time allocation: {[f'{t:.0f}s' for t in time_each]}, polish: {polish_budget:.0f}s")

    for li, n in enumerate(lengths):
        if time.monotonic() >= search_deadline - 5:
            break
        tl = min(time_each[li], max(5.0, search_deadline - time.monotonic() - 5.0))
        if tl < 10:
            break

        print(f"\n=== Length n={n}, budget={tl:.1f}s ===")
        rng = np.random.default_rng(int(seed + li * 1000))

        # Generate starts
        starts = _make_starts(n, parent_seq, rng)
        print(f"  Generated {len(starts)} starts, best={starts[0][0]:.6f}")

        # Try top starts with local search
        k_starts = min(len(starts), max(2, int(tl / 40.0) + 1, 3))
        time_each_start = tl / k_starts

        for si in range(k_starts):
            if time.monotonic() >= search_deadline - 5:
                break
            sc0, a0 = starts[si]
            t_rem = max(5.0, min(time_each_start, search_deadline - time.monotonic() - 5.0))
            if t_rem < 5:
                break

            # Try both symmetric and non-symmetric
            for sym in (False, True):
                if time.monotonic() >= search_deadline - 5:
                    break
                a_init = a0.copy()
                if sym:
                    a_init = _norm(_symmetrize(a_init), float(n))

                try:
                    seq, sc = _local_search(a_init, t_rem * 0.9, seed=int(seed + li * 1000 + si * 2 + (1 if sym else 0)), symmetric=sym)
                    update_global(seq, sc)
                except Exception as ex:
                    print(f"    Error: {ex}")

            # Peak shaving on best found so far
            if global_best is not None and global_best.size == n:
                for ps_iter in range(3):
                    if time.monotonic() >= search_deadline - 5:
                        break
                    shaved = _peak_shave(global_best, K_peaks=3, n_shave=max(5, n // 50), alpha=0.2)
                    sc_shave = _score_fast(shaved)
                    if sc_shave < global_best_sc:
                        update_global(shaved, sc_shave)

        print(f"  Best after n={n}: {global_best_sc:.6f}")

    # === Final polish phase ===
    if global_best is not None:
        polish_time = max(10.0, search_deadline - time.monotonic())
        print(f"\n=== Polish phase: {polish_time:.1f}s, current best={global_best_sc:.6f} ===")

        if polish_time > 20:
            # More aggressive local search on the best
            try:
                seq, sc = _local_search(global_best, polish_time * 0.5, seed=int(seed + 99999), symmetric=False)
                update_global(seq, sc)
            except Exception:
                pass

            # Symmetric polish
            if time.monotonic() < search_deadline - 10:
                try:
                    a_sym = _norm(_symmetrize(global_best), float(global_best.size))
                    seq, sc = _local_search(a_sym, polish_time * 0.2, seed=int(seed + 88888), symmetric=True)
                    update_global(seq, sc)
                except Exception:
                    pass

            # Scipy polish
            if time.monotonic() < search_deadline - 10 and global_best.size <= 2048:
                try:
                    seq, sc = _try_scipy_polish(global_best, polish_time * 0.2, seed=int(seed + 77777))
                    update_global(seq, sc)
                except Exception as ex:
                    print(f"  scipy failed: {ex}")

            # Additional peak shaving rounds
            if time.monotonic() < search_deadline - 10:
                for _ in range(10):
                    if time.monotonic() >= search_deadline - 5:
                        break
                    for alpha in (0.1, 0.2, 0.4, 0.6):
                        shaved = _peak_shave(global_best, K_peaks=5, n_shave=max(5, global_best.size // 40), alpha=alpha)
                        sc = _score_fast(shaved)
                        if sc < global_best_sc:
                            update_global(shaved, sc)

    # Fallback
    if global_best is None:
        global_best = np.ones(512)
        global_best_sc = _score_fast(global_best)
        print(f"  Fallback to uniform: score={global_best_sc:.6f}")

    # === Exact validation ===
    print(f"\n=== Final validation ===")
    final_list = [float(x) for x in global_best]
    for x in final_list:
        if x < 0:
            final_list[final_list.index(x)] = 0.0

    try:
        exact_sc = float(evaluate_sequence(final_list))
        print(f"  Exact score: {exact_sc:.6f} (n={len(final_list)})")
    except Exception as ex:
        print(f"  Exact eval failed: {ex}")
        exact_sc = float('inf')

    # Also check a few alternatives
    alternatives = []
    if parent_seq is not None and parent_seq.size > 0:
        alternatives.append(parent_seq)
    alternatives.append(np.ones(512))
    alternatives.append(np.ones(1024) * (0.5 + 0.5 * np.cos(np.linspace(0, 2 * np.pi, 1024))))

    best_exact = exact_sc
    best_final = final_list

    for alt in alternatives:
        try:
            alt_list = [float(max(0, x)) for x in alt]
            es = float(evaluate_sequence(alt_list))
            if np.isfinite(es) and es < best_exact:
                best_exact = es
                best_final = alt_list
                print(f"  Better alternative: {es:.6f}")
        except Exception:
            pass

    print(f"\nFinal score: {best_exact:.6f}")
    return best_final
