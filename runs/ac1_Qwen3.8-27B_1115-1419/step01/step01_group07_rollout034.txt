import numpy as np
import time

def compute_convolution(a):
    """Compute the full self-convolution of a."""
    return np.convolve(a, a)

def compute_subgradient(a, K=50):
    """Compute the subgradient of max(a*a) using top K peaks."""
    n = len(a)
    c = np.convolve(a, a)
    # Find top K peaks
    if K >= len(c):
        active = np.arange(len(c))
    else:
        # Use argpartition for efficiency
        top_k_indices = np.argpartition(c, -K)[-K:]
        active = top_k_indices
    g = np.zeros(n)
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        g[lo:hi+1] += a[k-hi : k-lo+1][::-1]
    g /= len(active)
    return g, c.max()

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def objective(a):

    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 1e-10:

    return 2.0 * n * c.max() / S**2
