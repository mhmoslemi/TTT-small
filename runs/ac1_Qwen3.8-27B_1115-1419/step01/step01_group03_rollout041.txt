Minimize 2n*max(conv(a,a))/(sum(a))^2 using LP-based iterative method.
Each iteration solves a linear program to find the "most spread out" 
sequence bounded by the current peak, then updates via convex combination.
Combines LP iteration with FISTA polishing for final convergence.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix(n, a):
    """Build sparse matrix A (2n-1 x n) where (A@b)_k = sum_j a[j]*b[k-j] (full conv)."""
    rows = np.zeros(n*(n+1)//2, dtype=np.int32)
    cols = np.zeros(n*(n+1)//2, dtype=np.int32)
    vals = np.zeros(n*(n+1)//2, dtype=np.float64)
    idx = 0
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n, k+1)
        length = hi - lo
        rows[idx:idx+length] = k
        cols[idx:idx+length] = np.arange(lo, hi)
        vals[idx:idx+length] = a[lo:hi]
        idx += length
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0*n*c.max()/(s*s)


def lp_iteration(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def grad_max_conv(a, k):
    """Gradient of (a*a)_k w.r.t. a. Result: 2*a_{k-i} for valid i."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k-n+1)
    hi = min(n, k+1)
    if lo < hi:
        # g[lo:hi] = 2*a[hi-1-lo+1 : hi-1-lo+1 ... ]
        # g[i] = 2*a[k-i] for i in [lo, hi)
        idx = k - np.arange(lo, hi)
        g[lo:hi] = 2.0 * a[idx]
    return g


def fista_polish(a, duration, rng, top_k=5, step=None):
    """FISTA to polish the solution."""
    n = len(a)
    a = a.copy()
    a_prev = a.copy()
    t_mom = 1.0
    if step is None:
        step = 0.5 / (2.0 * np.sqrt(n))  # Conservative step
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    
    while time.monotonic() < deadline:
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t_old*t_old))
        momentum = (t_old - 1.0) / t_mom
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        c_y = np.convolve(y, y)
        if top_k < c_y.size:
            top_idx = np.argpartition(c_y, -top_k)[-top_k:]
        else:
            top_idx = np.arange(c_y.size)
        
        g = np.zeros(n)
        for mk in top_idx:
            g += grad_max_conv(y, int(mk))
        g /= len(top_idx)
        
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1

            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:

                a_prev = best_a.copy()
                t_mom = 1.0
                stuck = 0

    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

    while N < 2*n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa*fa, N)[:2*n-1]

def sc(a):
    """Score: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0*n*conv2(a).max()/(s*s)

def solve_lp(a, M):
    """Solve max sum(b) s.t. A(a)@b <= M, b >= 0."""
    n = len(a)
    A = build_A(n, a)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0.0, None)]*n, method='highs')
    return res

def psimplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    c = u*np.arange(1, n+1) > (cs - 1.0)
    if not c.any():
        return np.full(n, 1.0/n)
    r = np.where(c)[0][-1]
    th = (cs[r]-1.0)/(r+1)
    return np.maximum(v-th, 0.0)

def gk(a, k):
    """Gradient of (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k-n+1)
    hi = min(n, k+1)
    if lo < hi:
        g[lo:hi] = 2.0*a[k-np.arange(lo, hi)]
    return g

def fista(a0, dur, rng, top_k=5, step=0.3):
    """FISTA to minimize score on simplex."""
    n = len(a0)
    a = a0.copy()
    ap = a0.copy()
    tm = 1.0
    ba = a0.copy()
    bs = sc(a0)
    dl = time.monotonic() + dur
    it = 0
    stk = 0
    while time.monotonic() < dl:
        to = tm
        tm = 0.5*(1.0+np.sqrt(1.0+4.0*to*to))
        y = a + (to-1.0)/tm*(a-ap)
        np.maximum(y, 0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        cy = conv2(y)
        if top_k < cy.size:
            ti = np.argpartition(cy, -top_k)[-top_k:]
        else:
            ti = np.arange(cy.size)
        g = np.zeros(n)
        for mk in ti:
            g += gk(y, int(mk))
        g /= len(ti)
        an = psimplex(y - step*g)
        ap = a
        a = an
        it += 1
        if it % 20 == 0:
            s2 = sc(a)
            if s2 < bs - 1e-12:
                bs = s2
                ba = a.copy()
                stk = 0
            else:
                stk += 1
            if stk > 50:
                step = max(0.01, step*0.8)
                ap = ba.copy()
                tm = 1.0
                stk = 0
                if step <= 0.01:
                    # Restart with perturbation
                    step = 0.3
                    a = ba + rng.normal(0, 0.01/np.sqrt(n), n)
                    np.maximum(a, 0, out=a)
                    a /= a.sum()
                    ap = a.copy()
                    tm = 1.0
    s2 = sc(a)
    if s2 < bs:
        bs = s2
        ba = a.copy()
    return ba, bs
