Minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequence a.

Strategy:
  - Multi-start with diverse initializations (spectral sqrt, Fourier, perturbed previous best)

import numpy as np

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p << 1
    return p


def _proj_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _score_fft(a, m=None):
    """Compute score using FFT-based convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-300:
        return np.inf
    if m is None:
        m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _convolve_self(a, m):
    """Compute self-convolution using FFT."""
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * a.size - 1]
    return np.maximum(b, 0.0)


def _gradient_soft(a, b, m, tau):
    """
    Compute gradient of soft-max(b) w.r.t. a.
    Returns (gradient, bmax, w) where w are the soft-max weights.
    """
    n = a.size
    bmax = float(b.max())
    if bmax <= 1e-300:
        return np.zeros(n), 0.0, np.ones(2 * n - 1) / (2 * n - 1)
    
    x = (b - bmax) / tau
    x = np.clip(x, -50.0, 50.0)
    x = x - float(x.max())
    w = np.exp(x)
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        return np.zeros(n), bmax, np.ones(2 * n - 1) / (2 * n - 1)
    w = w / ws
    
    # Gradient: d(softmax(b))/da_j = sum_k w_k * (db_k/da_j)
    # db_k/da_j = a_{j+k} + a_{j-k} (linear convolution, with boundaries)
    # Using FFT: if P = fft(w), then c = ifft(P * conj(F))_real[:n] gives the gradient
    F = np.fft.fft(a, m)
    P = np.fft.fft(w, m)
    c = np.fft.ifft(P * np.conj(F)).real[:n]
    
    return c, bmax, w

def _spectral_init(n, target='flat', seed=0):

def _fourier_init(n, n_harmonics=8, seed=0):
    """Generate a non-negative sequence as a sum of cosines."""
    rng = np.random.default_rng(seed)
    idx = np.arange(n) / n
    
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(-0.3, 0.3) / np.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * idx + ph)
    
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def _interp_to_length(a, n):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = float(a.sum())
    if a.size == 0 or s <= 0:
        return np.ones(n) / n
    if a.size == n:
        return a / s
    x_old = np.linspace(0, 1, a.size)
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0)
    ss = float(out.sum())
    if ss <= 0:
        return np.ones(n) / n
    return out / ss

            if float(arr.sum()) > 0:
                prev = arr / float(arr.sum())
    except Exception:
        prev = None
    
    best_a = prev
    if best_a is not None:
        best_score = _score_fft(best_a)
    else:
        best_a = np.ones(512) / 512
        best_score = 2.0
    
    print("Starting: n=%d, score=%.6f" % (len(best_a), best_score))
    
    # Progressive length search
    # Allocate time based on expected benefit
    # Larger n gives more "degrees of freedom" to reduce the peak
    stages = [
        (1024, 0.04),   # Quick check
        (2048, 0.20),   # Main length (matches previous)
        (4096, 0.30),   # Extended
        (8192, 0.35),   # Large
        (16384, 0.11),  # Very large (if time permits)
    ]
    
    total_base = max(60.0, budget_s - 25.0)
    
    for n, frac in stages:
        now = time.monotonic()
        if now + 20 >= global_deadline:
            break
        
        # Interpolate previous best to this length
        if best_a is not None:
            a_interp = _interp_to_length(best_a, n)
        else:
            a_interp = np.ones(n) / n
        
        sc_interp = _score_fft(a_interp)
        print("Stage n=%d: interpolated score=%.6f" % (n, sc_interp))
        
        # Time budget for this stage
        stage_time = min(total_base * frac, global_deadline - now - 10)
        if stage_time < 20:
            break
        stage_deadline = now + stage_time
        
        # Update global best if interpolation already improved
        if sc_interp < best_score:
            best_score = sc_interp
            best_a = a_interp
        
        a_new, sc_new = _process_length(n, stage_deadline, a_interp, rng, stage_time)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new
            print("  Improved to %.6f at n=%d" % (best_score, n))
        else:
            print("  No improvement at n=%d (best=%.6f)" % (n, best_score))
    
    # Final fine-tuning on the best sequence
    now = time.monotonic()
    remaining = max(0, global_deadline - now - 5)
    if remaining > 30 and best_a is not None:
        print("Final polish: n=%d, remaining=%.1fs" % (len(best_a), remaining))
        a_final, sc_final = _hill_climb(best_a, len(best_a), remaining, seed=rng.integers(0, 2**31))
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_a = a_final
            print("  Final polish score: %.6f" % best_score)
        
        # One more gradient pass
        remaining2 = max(0, global_deadline - time.monotonic() - 5)
        if remaining2 > 20:
            a_grad, sc_grad = _optimize_adaptive(best_a, len(best_a), remaining2,
                                                  seed=rng.integers(0, 2**31))
            if np.isfinite(sc_grad) and sc_grad < best_score:
                best_score = sc_grad
                best_a = a_grad
                print("  Final grad score: %.6f" % best_score)
    
    # Ensure valid output
    best_a = np.asarray(best_a, dtype=np.float64)
    best_a = np.maximum(best_a, 0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if s <= 0:
        best_a = np.ones(512) / 512
    else:
        best_a = best_a / s  # Normalize (score is scale-invariant, but keep values reasonable)
    
    # Validate with exact evaluation if length is manageable
    try:
        if len(best_a) <= 10000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Final: FFT=%.6f, exact=%.6f, n=%d" % (best_score, exact, len(best_a)))
            if not np.isfinite(exact):
                best_a = np.ones(512) / 512
        else:
            print("Final: FFT=%.6f, n=%d (exact skipped)" % (best_score, len(best_a)))
    except Exception as e:
        print("Validation error: %s" % str(e))
    
    elapsed = time.monotonic() - start
    print("Done in %.1fs. Best score: %.6f, length: %d" % (elapsed, best_score, len(best_a)))
    
    return [float(x) for x in best_a]
