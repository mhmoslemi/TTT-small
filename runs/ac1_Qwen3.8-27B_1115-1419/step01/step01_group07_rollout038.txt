"""
Optimize step function sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a multi-phase approach: LSE-smoothed L-BFGS for initial convergence,
Adam with exact subgradient for refinement, multiple lengths and inits.
Key: non-symmetric sequences beat the Cauchy-Schwarz bound of 2.
"""

import numpy as np
import time
from scipy.optimize import minimize


def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a)


def lse_value_and_weights(c, tau):
    """Compute LSE approximation of max(c) and softmax weights."""
    c_max = c.max()
    shifted = (c - c_max) / tau
    e = np.exp(shifted)
    e_sum = e.sum()
    if e_sum < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return c_max, w
    w = e / e_sum
    lse_val = c_max + tau * np.log(e_sum)
    return lse_val, w


def smooth_obj_grad(a, tau):
    """Compute LSE-smoothed objective and its gradient w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    sum_a = a.sum()
    if sum_a < 1e-15:
        return 1e10, np.zeros(n)
    lse_val, w = lse_value_and_weights(c, tau)
    # Gradient of LSE w.r.t. a[i] = sum_k w[k] * a[k-i] for valid k-i
    # This is np.correlate(w, a, mode='valid')
    grad_lse = np.correlate(w, a, mode='valid')
    obj = 2.0 * n * lse_val / (sum_a ** 2)
    grad = 2.0 * n / (sum_a ** 2) * (grad_lse - lse_val / sum_a)
    return obj, grad


def exact_subgradient(a, n_top=10):
    """Exact subgradient of max(conv(a,a)) w.r.t. a, averaged over top peaks."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    # Get top n_top convolution lags
    sorted_idx = np.argsort(c)[::-1][:n_top]
    w = np.zeros_like(c)
    w[sorted_idx] = 1.0 / len(sorted_idx)
    grad = np.correlate(w, a, mode='valid')
    return grad


def project_simplex(v):
    """Project v onto the probability simplex (non-negative, sum=1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return v / s


def adam_step(a, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step with simplex projection. Returns (new_a, m, v)."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = project_simplex(a_new)
    return a_new, m, v


def make_inits(n, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    
    # Use parent if available
    if 'height_sequence_1' in globals():
        parent = np.array(height_sequence_1, dtype=np.float64)
        if len(parent) == n:
            inits.append(parent / parent.sum())
        else:
            # Resample to length n
            indices = np.linspace(0, len(parent) - 1, n)
            i0 = indices.astype(int)
            i1 = np.minimum(i0 + 1, len(parent) - 1)
            frac = indices - i0
            resampled = (1 - frac) * parent[i0] + frac * parent[i1]
            resampled = np.maximum(resampled, 0)
            s = resampled.sum()
            if s > 1e-10:
                inits.append(resampled / s)
    
    # Linear ramp (asymmetric!)
    for alpha in [0.5, 1.0, 2.0]:
        x = 1.0 + alpha * np.arange(n) / n
        inits.append(x / x.sum())
    
    # Exponential ramp
    for rate in [1.0, 2.0, 3.0]:
        x = np.exp(rate * np.arange(n) / n)
        inits.append(x / x.sum())
    
    # Two-block asymmetric
    x = np.ones(n)
    x[:n // 3] = 2.0
    x[n // 3:2 * n // 3] = 1.0
    x[2 * n // 3:] = 0.3
    inits.append(x / x.sum())
    
    # Cosine with phase (asymmetric)
    for freq in [1, 2, 3]:
        for phase in [0.3, 0.7, 1.5]:
            x = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
            x = np.maximum(x, 0.01)
            inits.append(x / x.sum())
    
    # Quadratic (parabola shifted to break symmetry)
    t = np.arange(n) / n
    x = (t - 0.3) ** 2 + 0.1
    inits.append(x / x.sum())
    
    # Random Dirichlet
    for concentration in [0.3, 1.0, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * concentration))
    
    # Sparse random
    x = np.where(rng.random(n) > 0.5, 1.0, 0.1)
    inits.append(x / x.sum())
    
    # Near-uniform with small asymmetric perturbation
    x = np.ones(n) + 0.1 * np.arange(n) / n
    inits.append(x / x.sum())
    
    return inits


def run_lbfgs_phase(a0, tau_schedule, max_iter=300, ftol=1e-13):
    """Run L-BFGS on the LSE-smoothed objective with a tau schedule."""
    n = len(a0)
    best_a = a0.copy()
    best_score = np.inf
    x_log = np.log(np.maximum(a0, 1e-30))
    
    for tau in tau_schedule:
        tau_abs = tau * n  # Scale tau with n for consistency
        
        def objective_x(x):
            a = np.exp(x - x.max())
            a /= a.sum()
            obj, _ = smooth_obj_grad(a, tau_abs)
            return obj
        
        def gradient_x(x):
            a = np.exp(x - x.max())
            a /= a.sum()
            _, grad_a = smooth_obj_grad(a, tau_abs)
            # Chain rule: d/dx_j log_softmax = grad_a[j] - sum(a*grad_a)
            dot = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot)
            return grad_x
        
        try:
            result = minimize(
                objective_x, x_log, jac=gradient_x,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10,
                         'maxfun': max_iter * 3}
            )
            x_log = result.x
        except Exception:
            pass
        
        a = np.exp(x_log - x_log.max())

def run_adam_phase(a0, max_time, lr_init=0.05, lr_min=0.0005, n_top=20,
                   restart_interval=2000, restart_perturb=0.02):
    """Run Adam with exact subgradient for a given time budget."""
    n = len(a0)
    a = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start_time = time.monotonic()
    stagnant_count = 0
    
    while time.monotonic() - start_time < max_time:
        t += 1
        # Adaptive learning rate: decay over time
        progress = min(1.0, (time.monotonic() - start_time) / max_time)
        lr = lr_init * (lr_min / lr_init) ** progress
        
        # Compute subgradient
        c = np.convolve(a, a)
        c_max = c.max()
        sorted_idx = np.argsort(c)[::-1][:n_top]
        w = np.zeros_like(c)
        w[sorted_idx] = 1.0 / len(sorted_idx)
        grad = np.correlate(w, a, mode='valid')
        
        sum_a = a.sum()
        # Full objective gradient (accounting for normalization)
        grad = (2.0 * n / (sum_a ** 2)) * (grad - c_max / sum_a)
        
        a, m, v = adam_step(a, grad, m, v, t, lr=lr)
        
        # Evaluate periodically
        if t % 50 == 0 or t <= 5:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                stagnant_count = 0
            else:
                stagnant_count += 1
        
        # Periodic restart to escape local minima
        if stagnant_count > restart_interval // 50:
            a = best_a.copy()
            perturbation = np.random.default_rng(t).normal(0, restart_perturb, n)
            a = a + perturbation
            a = project_simplex(a)
            m = np.zeros(n)
            v = np.zeros(n)
            stagnant_count = 0
            restart_perturb *= 0.8
    
    return best_a, best_score


def run_hybrid(a0, max_time):
    """Hybrid: LSE-LBFGS then Adam."""
    n = len(a0)
    
    # Phase A: LSE-LBFGS (short)
    lbfgs_time = min(120, max_time * 0.3)
    tau_schedule = [5.0/n, 2.0/n, 1.0/n, 0.5/n, 0.2/n, 0.1/n]
    
    if time.monotonic() < time.monotonic() + lbfgs_time:  # always true, just for clarity

        a = a0.copy()
        score = evaluate_sequence(a.tolist())
    
    # Phase B: Adam refinement
    adam_time = max(30, max_time - lbfgs_time - 10)
    a, score = run_adam_phase(a, adam_time)
    
    return a, score
