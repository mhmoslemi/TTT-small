```python
"""
LP-based iterative optimization for low-peak nonnegative self-convolutions.

Uses the linear programming approach from the SOTA description:
- Maintain a nonnegative sequence f normalized to sum 1
- At each iteration, solve an LP to find a direction g that doesn't
  increase the peak autocorrelation
- Update f via convex combination with g
"""

import numpy as np
import time
from scipy.optimize import linprog

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s

def _autocorr_full(a):
    n = a.size
    b = np.convolve(a, a, mode='full')
    return b

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-300:
        return float('inf')
    b = np.convolve(a, a, mode='full')
    return 2.0 * n * float(np.max(b)) / (s * s)

def _build_constraint_matrix(f):
    """Build the (2n-1) x n matrix A where A[k,i] = f[k-i] if valid."""
    n = f.size
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    return A

def _solve_lp(f, M):
    """Solve the LP: maximize sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = f.size
    A = _build_constraint_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    return res.x

def _lp_iteration(f, t=0.3, max_iters=200, deadline=None):
    """Run LP-based iterations starting from f."""
    f = _normalize(f)
    n = f.size
    best_f = f.copy()
    best_score = _score(f)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = np.convolve(f, f, mode='full')
        M = float(np.max(b))
        
        if M <= 1e-300:
            break
        
        g = _solve_lp(f, M)
        if g is None:
            break
        
        g_sum = float(g.sum())
        if g_sum <= 1e-300:
            break
        
        g = g / g_sum  # normalize to sum 1
        
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = _normalize(f_new)
        
        new_score = _score(f_new)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
        
        # Check convergence
        b_new = np.convolve(f, f, mode='full')
        M_new = float(np.max(b_new))
        if abs(M_new - M) < 1e-10 * M:
            break
    
    return best_f, best_score
```

```python
    return float(2.0 * n * np.max(b) / (s * s))
```

```python
    s = float(a.sum())

    return A


def _solve_lp(f, M):

    A = _build_A(f)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2*n-1, M, dtype=np.float64)

    return res.x.astype(np.float64)


def _lp_refine(a, deadline, max_iters=1000, verbose=True, label=""):
    a = _normalize(a.astype(np.float64).copy())
    n = a.size
    best_a = a.copy()
    best_score = _score_of(a)

    t_pool = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]

    it = 0
    imp_count = 0
    stagnant = 0

    while it < max_iters and time.monotonic() < deadline:
        b = np.convolve(a, a, mode='full')
        M = float(np.max(b))
        if M < 1e-18:

        g = _solve_lp(a, M)
        if g is None:
            break
        gs = float(g.sum())
        if gs < 1e-18:
            break
        g = g / gs

        best_t = 0.0
        best_sc = best_score
        for t in t_pool:
            a2 = (1.0 - t) * a + t * g
            a2 = np.maximum(a2, 0.0)
            s2 = a2.sum()
            if s2 < 1e-18:
                continue
            a2 = a2 / s2
            sc = _score_of(a2)
            if sc < best_sc:
                best_sc = sc
                best_t = t

            stagnant += 1
            if stagnant > 8:

            a = _normalize(a + 0.005 * rng.normal(size=n))
            it += 1
            continue

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = _normalize(a)

                      flush=True)

        it += 1

def _coord_polish(a, n_steps=2000, verbose=False):
    a = _normalize(a.astype(np.float64).copy())
    n = a.size
    best_score = _score_of(a)
    best_a = a.copy()
    rng = np.random.default_rng(42)

    for step in range(n_steps):
        i = int(rng.integers(n))
        for mult in [0.85, 0.9, 1.05, 1.1]:
            a2 = a.copy()

            a2 = _normalize(a2)
            sc = _score_of(a2)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a2.copy()
                a = a2
                break
    return best_a, best_score


def _make_exp(n, alpha=1.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_gauss(n, width=0.25, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    return _normalize(a)


def _make_cos(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2.0) ** 2
    return _normalize(a)


def _make_two_sided(n, alpha=1.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * x) + 0.5 * np.exp(-alpha * (1 - x))
    return _normalize(a)


def _resample_seq(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = _normalize(old)
    new_n = max(int(new_n), 2)
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0, 1, old.size)
    x_new = np.linspace(0, 1, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)
```

LP-based iterative minimization of peak nonnegative self-convolution.

The score for a nonnegative sequence a (length n) is
    score = 2n * max(a*a) / (sum(a))^2
which is scale-invariant.  We normalize to sum(a)=1 and minimize 2n*max(a*a).

Main algorithm:
  1. Generate several parametric and perturbed initial sequences.
  2. LP iteration: given f, solve max sum(g) s.t. (f*g)_k <= max(f*f), g>=0.
     Take a convex-combination step with line search.
  3. Coordinate-descent polish.
  4. Upscale to longer sequences and repeat.
