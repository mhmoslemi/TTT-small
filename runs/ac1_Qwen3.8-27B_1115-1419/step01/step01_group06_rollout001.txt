```python
"""
Hybrid search combining FFT-based gradient smoothing with exact
coordinate-wise minimization of max self-convolution. The coordinate
phase solves a 1-D convex subproblem at each step, guaranteeing per-
coordinate optimality and typically converging faster than pure
gradient methods. Multiple restarts and length sweeps provide robustness.
"""

import numpy as np
import time

MAX_N = 8192

def fft_conv_max(a):
    n = len(a)
    N = 1
    while N < 2*n:
        N <<= 1
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A*A, N)[:2*n-1]
    return max(np.max(c), 0.0)

def obj_val(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * fft_conv_max(a) / (s*s)

def exact_obj(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * max(np.max(c), 0.0) / (s*s)

def load_parent():
    obj = globals().get('height_sequence_1', None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size < 8 or arr.sum() < 1e-15:
        return None
    return arr

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - (n-1)/2.0
    sig = sigma_frac * n
    g = np.exp(-x*x/(2*sig*sig))
    return g / g.sum()

def make_exponential(n, decay=0.5):
    x = np.exp(-decay * np.abs(np.arange(n) - (n-1)/2.0) / n)
    return x / x.sum()

def fft_gradient_step(a, c, beta, lr, N=None, phase=None):
    n = len(a)
    if N is None:
        N = 1
        while N < 2*n:
            N <<= 1
        phase = np.exp(-2j*np.pi*np.arange(N//2+1)*(n-1)/float(N))
    if phase is None:
        phase = np.exp(-2j*np.pi*np.arange(N//2+1)*(n-1)/float(N))
    
    # Smoothed max of c
    L = np.max(c)
    x = (c - L) / max(L, 1e-300)
    x = np.clip(beta * x, -700, 0)
    p = np.exp(x)
    p /= p.sum()
    
    # Gradient: grad_i = sum_k p[k] * (a[k-i] + a[i-k])
    # Using FFT: grad = irfft(fft(p) * conj(phase * fft(a)))
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    FR = phase * np.conjugate(A)
    conv = np.fft.irfft(P * FR, N)
    grad = 2.0 * conv[n-1:2*n-1]
    # Center the gradient (remove mean to maintain sum)
    grad -= grad.mean() * n  # hmm, this isn't quite right
    
    # Actually, for scale invariance, we can just subtract the mean
    grad = grad - np.mean(grad)
    a_new = a * np.exp(lr * grad / max(np.max(np.abs(grad)), 1e-15))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new, grad

def coordinate_descent(a, c, n_passes, seed=42, verbose=False):
    """
    Coordinate descent on a to minimize max(c) where c = conv(a,a).
    Each step: for coordinate i, find d minimizing max(c + 2d*a + d^2*I(k=2i))
    subject to a[i] + d >= 0.
    """
    n = len(a)
    c_len = 2*n - 1
    rng = np.random.default_rng(seed)
    
    for p in range(n_passes):
        order = rng.permutation(n)
        total_improve = 0.0
        
        for i in order:
            ai = a[i]
            if ai < 1e-300:
                # If a[i] is essentially 0, the only valid d is d >= 0
                lo = 0.0
            else:
                lo = -ai
            
            hi = ai + 2.0/n if ai > 0 else 1.0/n*2
            
            # Precompute: the segment of c that gets modified is c[i:i+n]
            # and it gets 2*d*a added
            # c[2i] additionally gets d^2
            
            # Ternary search for d in [lo, hi]
            # f(d) = max(c + 2d*a_shift + d^2*I)
            # where a_shift[k] = a[k-i] for valid k
            
            # For efficiency, note that:
            # vals = c + 2*d*a_shift
            # vals[2i] += d^2
            # f = max(vals)
            
            # Since a_shift = a (for k in [i, i+n)), and 0 elsewhere:
            # vals[k] = c[k] + 2*d*a[k-i] for k in [i, i+n)
            # vals[k] = c[k] for k outside [i, i+n)
            
            # Precompute the baseline max outside the affected region
            # and the affected values
            
            base_max = 0.0
            # k from 0 to i-1 and i+n to 2n-2
            if i > 0:
                base_max = max(base_max, np.max(c[:i]))
            if i + n < c_len:
                base_max = max(base_max, np.max(c[i+n:]))
            
            affected = c[i:i+n].copy()  # length n
            is_2i = (2*i < c_len)  # whether k=2i is in the affected range
            idx_2i = 2*i - i  # index within affected where k=2i → index = 2i - i = i
            # Wait: affected[j] corresponds to c[i+j], so k = i+j
            # k = 2i → j = i. So idx_2i = i (within affected)
            
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                # f(m1) = max(max(base_max, max(affected + 2*m1*a + m1^2*I(j==i))))
                v1 = affected + 2.0 * m1 * a
                if is_2i and 0 <= i < n:
                    v1[i] += m1 * m1
                f1 = max(base_max, np.max(v1))
                
                v2 = affected + 2.0 * m2 * a
                if is_2i and 0 <= i < n:
                    v2[i] += m2 * m2
                f2 = max(base_max, np.max(v2))
                
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1
            
            d = (lo + hi) / 2.0
            if abs(d) < 1e-15:
                continue
            
            # Update
            old_max = np.max(c)
            c[i:i+n] += 2.0 * d * a  # use OLD a (before updating a[i])
            if is_2i:
                c[2*i] += d * d
            a[i] += d
            if a[i] < 0:
                a[i] = 0
                # Revert c
                c[i:i+n] -= 2.0 * d * a
                if is_2i:
                    c[2*i] -= d * d
            
            new_max = np.max(c)
            total_improve += max(0, old_max - new_max)
        
        if verbose and p % 5 == 0:
            print(f"  Coord pass {p}: improvement={total_improve:.6f}, max(c)={np.max(c):.8f}")
    
    return a, c

def run_search(budget=2400.0):
    t0 = time.monotonic()
    deadline = t0 + budget
    
    parent = load_parent()
    
    best_score = float('inf')
    best_a = None
    
    # Candidate initializations
    inits = []
    if parent is not None and parent.size >= 32:
        p = parent.copy()
        if p.size > MAX_N:
            # Downsample
            target = MAX_N
            blocks = p.size // target
            if blocks > 1:
                use = target * blocks
                p = np.mean(p[:use].reshape(target, blocks), axis=1)
            else:
                p = p[:target]
        p = np.maximum(p, 0)
        s = p.sum()
        if s > 0:
            p /= s
        inits.append(p)
    
    for n in [2048, 4096, 8192]:
        inits.append(make_gaussian(n, 0.3))
        inits.append(make_gaussian(n, 0.5))
        inits.append(make_exponential(n, 0.5))
        inits.append(make_exponential(n, 1.0))
    
    # Add a uniform + noise init
    n_mid = 4096
    rng0 = np.random.default_rng(0)
    noisy = np.ones(n_mid) + 0.1 * rng0.standard_normal(n_mid)
    noisy = np.maximum(noisy, 0)
    noisy /= noisy.sum()
    inits.append(noisy)
    
    # Evaluate all inits and pick top few
    scored_inits = []
    for a in inits:
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        N = 1
        while N < 2*len(a):
            N <<= 1
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A*A, N)[:2*len(a)-1]
        mx = max(np.max(c), 0)
        sc = 2.0 * len(a) * mx
        scored_inits.append((sc, a))
    
    scored_inits.sort(key=lambda x: x[0])
    print(f"Initialized {len(scored_inits)} candidates, best={scored_inits[0][0]:.6f}")
    
    # Process top candidates
    top_k = min(5, len(scored_inits))
    
    for init_idx in range(top_k):
        if time.monotonic() > deadline - 60:
            break
        
        a = scored_inits[init_idx][1].copy()
        n = len(a)
        
        if n < 32 or n > MAX_N:
            continue
        
        print(f"\nProcessing init {init_idx} (n={n}, score={scored_inits[init_idx][0]:.6f})")
        
        # Phase 1: FFT gradient smoothing (short)
        N = 1
        while N < 2*n:
            N <<= 1
        phase = np.exp(-2j*np.pi*np.arange(N//2+1)*(n-1)/float(N))
        
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A*A, N)[:2*n-1]
        c = np.maximum(c, 0)
        cur_score = 2.0 * n * np.max(c)
        
        beta = 5.0
        lr = 0.1
        stall = 0
        max_grad_iters = 200
        iter_time_budget = min(120.0, (deadline - time.monotonic()) * 0.15)
        
        for it in range(max_grad_iters):
            if time.monotonic() - t0 + iter_time_budget > deadline - 120:
                break
            if not np.isfinite(a).all() or a.sum() < 1e-15:
                break
            
            L = np.max(c)
            if L < 1e-300:
                break
            x = (c - L) / L
            x = np.clip(beta * x, -700, 0)
            p_sm = np.exp(x)
            p_sm /= p_sm.sum()
            
            A = np.fft.rfft(a, N)
            P = np.fft.rfft(p_sm, N)
            FR = phase * np.conjugate(A)
            conv = np.fft.irfft(P * FR, N)
            grad = 2.0 * conv[n-1:2*n-1]
            grad = grad - np.mean(grad)
            
            # Adaptive step
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-15:
                break
            step = lr * grad / gnorm
            a_new = a * np.exp(np.clip(step, -5, 5))
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            
            # Evaluate
            A2 = np.fft.rfft(a_new, N)
            c2 = np.fft.irfft(A2*A2, N)[:2*n-1]
            c2 = np.maximum(c2, 0)
            new_score = 2.0 * n * np.max(c2)
            
            if new_score < cur_score * (1 - 1e-10):
                a = a_new
                c = c2
                cur_score = new_score
                stall = 0
                lr = min(0.5, lr * 1.02)
            else:
                a = a_new  # accept anyway (non-greedy)
                c = c2
                cur_score = 2.0 * n * np.max(c)
                stall += 1
                lr = max(0.01, lr * 0.9)
            
            if stall > 30:
                # Re-compute c exactly from a
                A_real = np.fft.rfft(a, N)
                c = np.fft.irfft(A_real*A_real, N)[:2*n-1]
                c = np.maximum(c, 0)
                cur_score = 2.0 * n * np.max(c)
                stall = 0
        
        print(f"  After gradient phase: score={cur_score:.6f}")
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        # Phase 2: Coordinate descent (the main improvement)
        # Recompute c exactly
        A_real = np.fft.rfft(a, N)
        c = np.fft.irfft(A_real*A_real, N)[:2*n-1]
        c = np.maximum(c, 0)
        
        # How many coordinate passes can we afford?
        time_remaining = deadline - time.monotonic() - 30
        time_per_pass_est = 2.0 * n * 50 * 1e-6  # rough estimate
        n_passes = max(1, min(100, int(time_remaining / max(time_per_pass_est, 0.1))))
        n_passes = min(n_passes, 60)  # cap
        print(f"  Coord descent: {n_passes} passes (est time {n_passes * time_per_pass_est:.1f}s)")
        
        # Do coordinate descent with multiple passes
        a_cd = a.copy()
        c_cd = c.copy()
        
        cd_start = time.monotonic()
        passes_done = 0
        last_improve = 0.0
        
        for p in range(n_passes):
            if time.monotonic() > deadline - 45:
                break
            prev_max = np.max(c_cd)
            a_cd, c_cd = coordinate_descent(a_cd, c_cd, 1, seed=init_idx*1000+p, verbose=False)
            new_max = np.max(c_cd)
            this_improve = prev_max - new_max
            passes_done += 1
            
            score_cd = 2.0 * n * new_max
            if score_cd < best_score:
                best_score = score_cd
                best_a = a_cd.copy()
            
            if p % 10 == 0 or this_improve < 1e-10:
                elapsed = time.monotonic() - cd_start
                print(f"    CD pass {p}: max(c)={new_max:.8f}, score={score_cd:.6f}, "
                      f"improve={this_improve:.2e}, elapsed={elapsed:.0f}s")
            
            if this_improve < 1e-12:
                break
        
        print(f"  After CD ({passes_done} passes): best_score={best_score:.6f}")
        
        # Phase 3: Short FFT gradient refinement after CD
        if time.monotonic() < deadline - 30:
            a_ref = a_cd.copy()
            A_real = np.fft.rfft(a_ref, N)
            c_ref = np.fft.irfft(A_real*A_real, N)[:2*n-1]
            c_ref = np.maximum(c_ref, 0)
            score_ref = 2.0 * n * np.max(c_ref)
            
            beta2 = 20.0
            lr2 = 0.05
            for it in range(100):
                if time.monotonic() > deadline - 30:
                    break
                L = np.max(c_ref)
                if L < 1e-300:
                    break
                x = (c_ref - L) / L
                x = np.clip(beta2 * x, -700, 0)
                p_sm = np.exp(x)
                p_sm /= p_sm.sum()
                
                A2 = np.fft.rfft(a_ref, N)
                P = np.fft.rfft(p_sm, N)
                FR = phase * np.conjugate(A2)
                conv = np.fft.irfft(P * FR, N)
                grad = 2.0 * conv[n-1:2*n-1] - np.mean(2.0 * conv[n-1:2*n-1])
                
                gnorm = np.max(np.abs(grad))
                if gnorm < 1e-15:
                    break
                step = lr2 * grad / gnorm
                a_new = a_ref * np.exp(np.clip(step, -3, 3))
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    break
                a_new /= s
                
                A3 = np.fft.rfft(a_new, N)
                c3 = np.fft.irfft(A3*A3, N)[:2*n-1]
                c3 = np.maximum(c3, 0)
                new_sc = 2.0 * n * np.max(c3)
                
                if new_sc < score_ref:
                    a_ref = a_new
                    c_ref = c3
                    score_ref = new_sc
                    if score_ref < best_score:
                        best_score = score_ref
                        best_a = a_ref.copy()
            
            print(f"  After refinement: score={score_ref:.6f}")
    
    if best_a is None:
        best_a = np.ones(4096) / 4096.0
    
    return best_a
