def run_coord_polish(f0, n, deadline, verbose=True):
    """Cycle coordinate ascent: adjust one coordinate + renormalize."""
    f, best_f = f0.copy(), f0.copy()
    best_score = score_fast(f0)
    rng = np.random.default_rng(7)
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() < deadline:
        improved = 0
        for i in rng.permutation(n):
            if time.monotonic() - t0 > 20:
                break
            for md in [1.02, 0.98, 1.05, 0.95]:
                fc = f.copy()
                old = fc[i]
                fc[i] = old * md
                if fc[i] < 0:
                    fc[i] = 0

                if abs(delta) < 1e-15:
                    continue
                others = np.diff(np.r_[np.r_[0, np.nonzero(fc[:i])[0]], i +

    c3[:] = 0; c3[:n // 3] = 1; c3[2 * n // 3:] = 1
    if c3.sum() > 0:
        c3 /= c3.sum(); cands.append(c3); names.append("three_block")
    xx = np.arange(n, dtype=np.float64)
    lo[:] = np.maximum(0, 1 - (2 * xx / (n - 1) - 1) ** 2); lo /= lo.sum()
    cands.append(lo); names.append("triangle_low")
    sp[:] = rng.exponential(1, n)
    sp += 0.1 * np.sin(2 * np.pi * 3 * xx / n)
    np.maximum(sp, 0, out=sp); sp /= sp.sum(); cands.append(sp); names.append("mod_exp")
    return cands, names


def propose_candidate(budget=2400, **kwargs):
    """
    Search for the best non-negative coefficient sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2 using LP direction finding + FISTA +
    coordinate-descent polishing, with multiple starting points.
    Returns the best list of non-negative floats found.
    """
    rng = np.random.default_rng(2024)
    deadline = time.monotonic() + float(budget) - 25.0

    # ---- Build initial sequence ----
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.maximum(arr, 0.0)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.ones(n, dtype=np.float64)
    if n < 10:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    # ---- Candidates ----
    cands, names = make_candidates(arr, n, rng)
    best_c, best_init_sc = None, np.inf
    for c, nm in zip(cands, names):
        sc = score_fast(c)
        if finite := np.isfinite(sc):
            print(f"  cand '{nm}': {sc:.8f}")
            if sc < best_init_sc:
                best_init_sc, best_c = sc, c.copy()
    print(f"Best init: {best_init_sc:.8f}")

    # ---- Pre-compute LP matrix structure ----
    rows, cols = build_matrix_structure(n)
    print(f"LP structure: {len(rows)} nnz")

    # ---- Phase 1: LP from best candidate ----
    tl = deadline - time.monotonic()
    if tl > 60:
        print(f"Phase 1: LP ({tl:.0f}s)")
        p1_f, p1_sc = run_lp_phase(best_c, n, rows, cols,
                                   deadline - 300, verbose=True)
    else:
        p1_f, p1_sc = best_c.copy(), best_init_sc

    # ---- Phase 2: FISTA from two diverse starts ----
    tl = deadline - time.monotonic()
    if tl > 60 and best_init_sc > 1.503:
        print(f"Phase 2a: FISTA from LP result ({tl / 2 - 10:.0f}s)")
        p2a_f, p2a_sc = run_fista_phase(p1_f, n,
                                         time.monotonic() + max(30, tl / 2 - 20),
                                         top_k_init=5, verbose=True)
        tl = deadline - time.monotonic()
        if tl > 40:
            print(f"Phase 2b: FISTA from best init ({tl - 20:.0f}s)")
            p2b_f, p2b_sc = run_fista_phase(best_c, n,
                                             deadline - 50, top_k_init=7,
                                             verbose=True)
            best_f = p2a_f if p2a_sc <= p2b_sc else p2b_f
            best_sc = min(p2a_sc, p2b_sc)
        else:
            best_f, best_sc = p2a_f, p2a_sc
    else:
        best_f, best_sc = p1_f, p1_sc

    # ---- Phase 3: Short LP from FISTA result ----
    tl = deadline - time.monotonic()
    if tl > 80 and best_sc > 1.503:
        print(f"Phase 3: LP from FISTA ({tl - 100:.0f}s)")
        p3_f, p3_sc = run_lp_phase(best_f, n, rows, cols,
                                   deadline - 120, verbose=True)
        if p3_sc < best_sc:
            best_f, best_sc = p3_f, p3_sc
            tl = deadline - time.monotonic()
            if tl > 40:
                p3b_f, p3b_sc = run_fista_phase(p3_f, n, deadline - 60,
                                                 top_k_init=8, verbose=True)
                if p3b_sc < best_sc:
                    best_f, best_sc = p3b_f, p3b_sc

    # ---- Phase 4: Coordinate-descent polish ----
    tl = deadline - time.monotonic()
    if tl > 30:
        print(f"Phase 4: Coord polish ({tl - 5:.0f}s)")
        best_f, best_sc = run_coord_polish(best_f, n, deadline - 5, verbose=True)

    # ---- Final verify ----
    fl = [float(max(0, v)) for v in best_f]
    fs = evaluate_sequence(fl)
    print(f"Final eval: {fs:.8f}  n={n}")
    if not np.isfinite(fs) and np.isfinite(best_init_sc):
        fl = [float(max(0, v)) for v in best_c]
    return fl
