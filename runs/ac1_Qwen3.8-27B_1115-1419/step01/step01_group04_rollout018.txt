def normalize_to_sum1(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    # if max too large, scale to avoid overflow
    mx = float(x.max())
    if mx > 1e100:
        x = x / mx
    s = float(x.sum())
    if s <= 0.0:
        return np.full(x.size, 1.0 / float(x.size), dtype=np.float64)
    x = x / s
    return x
```

"""
Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

This version uses:
- FFT-based convolution and a workspace-based local search.
- Weighted stochastic subgradients from near-maximum convolution peaks.
- Adaptive peak-width, step size, and active-set width.
- Multi-length, multi-start portfolio, with periodic refinement from the
  previously best known sequence and from smooth/generated starts.
"""

import numpy as np
import time
import math


def _first(pair):
    return pair[0]


def normalize_to_sum1(x):
    """Clip to non-negative finite values and normalize sum to 1."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return np.ones(1, dtype=np.float64)

    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)

    mx = float(x.max())
    if mx > 1e100:
        x = x / mx

    s = float(x.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.full(x.size, 1.0 / float(x.size), dtype=np.float64)

    x = x / s
    return x


def convolution_into(a, spec, out, N):
    """
    Compute the full linear convolution a*a into out[:2*n-1].

    a has length n. N must be at least 2*n-1. spec is a complex
    half-spectrum buffer, out is a real time-domain buffer of length N.
    """
    np.fft.rfft(a, N, out=spec)
    spec *= spec
    if N % 2 == 0:
        spec[N // 2] = float(spec[N // 2].real)
    np.fft.irfft(spec, N, out=out)
    np.maximum(out, 0.0, out=out)


def fast_score_array(a):
    """FFT-based score, assuming scale invariance and normalizing to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")

    a = normalize_to_sum1(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()

    spec = np.empty(N // 2 + 1, dtype=np.complex128)
    out = np.empty(N, dtype=np.float64)
    convolution_into(a, spec, out, N)

    m = float(np.max(out[:L]))
    if not math.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)

    """Call the official evaluator if available."""

        seq = np.asarray(a, dtype=np.float64).tolist()
        return float(evaluate_sequence(seq))
    except Exception:
        return float("inf")


def choose_weighted_active(c, rng, Kloop, Tcap, width):
    """
    Choose a small weighted active set among near-maximum convolution peaks.

    c is the convolution vector. The return is (active_indices, weights).
    """
    L = int(c.size)
    Kloop = max(1, min(int(Kloop), L))

    if Kloop >= L:
        idx = np.arange(L, dtype=np.int64)
        w = np.ones(L, dtype=np.float64)
        return idx, w

    T = max(Kloop, min(int(Tcap), L))

    if T >= L:
        topT = np.arange(L, dtype=np.int64)
    else:
        topT = np.argpartition(c, -T)[-T:]

    vals = np.asarray(c[topT], dtype=np.float64)
    M = float(vals.max())

    gap = max(1e-15 * M, float(width) * M)
    base = M - gap
    w = np.maximum(vals - base, 0.0)
    wsum = float(w.sum())

    if wsum <= 1e-30:
        if T > Kloop:
            order = np.argsort(vals)[-Kloop:]
            topK = topT[order]
        else:
            topK = topT
        return topK, np.ones(topK.size, dtype=np.float64)

    if T > Kloop:
        probs = w / wsum
        probs = probs + 1e-300
        probs = probs / float(probs.sum())
        chosen = rng.choice(T, size=Kloop, replace=False, p=probs)
        active = topT[chosen]
        weights = w[chosen].astype(np.float64, copy=True)
    else:
        active = topT.copy()
        weights = w.astype(np.float64, copy=True)

    i_max = int(np.argmax(c))
    if len(active) == 0 or not np.any(active == i_max):
        if len(active) < Kloop:
            active = np.concatenate([active.astype(np.int64, copy=False),
                                     np.array([i_max], dtype=np.int64)])
            weights = np.concatenate([weights,
                                      np.array([max(1e-3, float(c[i_max] - base)]),
                                               dtype=np.float64)])
        else:
            pos = int(np.argmin(weights))
            active[pos] = i_max
            weights[pos] = max(1e-3, float(c[i_max] - base))

    weights = np.where(weights > 1e-30, weights, 1e-3)
    return active, weights


def topk_gradient_weighted(p, active, weights):
    """
    Weighted average subgradient contribution from the selected peaks.

    For peak k, the derivative-like vector is p[k-i] over valid i.
    A constant factor is irrelevant after normalization.
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    d = np.zeros(n, dtype=np.float64)
    wsum = float(np.sum(weights))
    if wsum <= 0.0:
        return d

    inv = 1.0 / wsum

    for k, w in zip(active, weights):
        ww = float(w) * inv
        if ww <= 0.0:
            continue

        k = int(k)
        if k <= n - 1:
            d[:k + 1] += ww * p[:k + 1][::-1]
        else:
            s = k - n + 1
            d[s:] += ww * p[s:n][::-1]

    return d


def smooth_perturb(p, rng, strength):
    """
    Smooth multiplicative perturbation used to escape stagnation.
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    out = p.copy()

    if n > 2:
        t = np.linspace(0.0, 1.0, n)
        z = np.zeros(n, dtype=np.float64)

        freqs = (1, 2, 3, 4)
        for m in freqs:
            amp = strength / (0.5 + 0.5 * float(m))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            if rng.random() < 0.5:
                z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)
            else:
                z += amp * np.sin(2.0 * np.pi * float(m) * t + ph)

        # Preserve/encourage asymmetry with a mild trend.
        z += 0.45 * strength * (float(rng.random()) * 2.0 - 1.0) * (2.0 * t - 1.0)
        out *= np.exp(z)

    if n > 8 and strength > 0.05:
        wmin = max(1, n // 120)
        wmax = max(wmin + 1, n // 20)
        width = int(rng.integers(wmin, wmax + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        out[start:start + width] *= np.exp(
            rng.normal(0.0, 0.5 * strength, width)
        )

    return normalize_to_sum1(out)


def upsample_sequence(seq, n):
    """Linearly resample a sequence to length n and normalize to sum 1."""
    seq = np.asarray(seq, dtype=np.float64)
    m = int(seq.size)
    n = int(n)

    if m == 0:
        return np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
    if m == n:
        return normalize_to_sum1(seq)

    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_to_sum1(out)


def make_uniform(n):
    n = int(n)
    if n <= 0:
        return np.array([1.0])
    return np.ones(n, dtype=np.float64) / float(n)


def make_linear(n, slope):
    n = int(n)
    if n <= 1:
        return make_uniform(n)
    t = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(slope) * (2.0 * t - 1.0)
    p = np.maximum(p, 1e-3)
    return normalize_to_sum1(p)


def make_power(n, a_left, a_right, w_left, w_right):
    """Two-sided endpoint power shape."""
    n = int(n)
    if n <= 1:
        return make_uniform(n)

    x = np.linspace(0.0, 1.0, n)
    eps = 1.0 / (n + 1.0)

    a_left = max(-0.55, min(0.55, float(a_left)))
    a_right = max(-0.55, min(0.55, float(a_right)))

    left = (x + eps) ** a_left
    right = (1.0 - x + eps) ** a_right
    p = float(w_left) * left + float(w_right) * right + 0.05
    return normalize_to_sum1(p)


def make_fourier(n, rng):
    """Random low-frequency positive shape."""
    n = int(n)
    if n <= 1:
        return make_uniform(n)

    t = np.linspace(0.0, 1.0, n)
    z = 0.7 * (float(rng.random()) * 2.0 - 1.0) * (2.0 * t - 1.0)

    max_freq = 9
    for m in range(1, max_freq + 1):
        amp = float(rng.uniform(-1.6, 1.6)) / (0.6 + 0.35 * float(m))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        if rng.random() < 0.5:
            z += amp * np.cos(2.0 * np.pi * float(m) * t + ph)
        else:
            z += amp * np.sin(2.0 * np.pi * float(m) * t + ph)

    z = z - float(np.max(z))
    p = np.exp(0.8 * z) + 1e-6
    return normalize_to_sum1(p)


def make_blocks(n, rng):
    """Random piecewise-constant sequence."""
    n = int(n)
    if n <= 1:
        return make_uniform(n)

    nb = int(rng.integers(4, min(36, n // 4) + 1))
    edges = np.round(np.linspace(0.0, n, nb + 1)).astype(int)
    edges = np.unique(edges)
    if edges.size < 3 or edges[0] != 0 or edges[-1] != n:
        edges = np.linspace(0.0, n, nb + 1).astype(int)
        edges[0] = 0
        edges[-1] = n
        edges = np.unique(edges)

    lengths = np.diff(edges)
    if lengths.size == 0 or np.any(lengths <= 0):
        return make_uniform(n)

    vals = rng.exponential(1.0, lengths.size)
    if lengths.size > 2 and rng.random() < 0.25:
        vals[int(rng.integers(0, lengths.size))] = 0.0
    vals += 1e-4

    p = np.repeat(vals, lengths)
    return normalize_to_sum1(p)


def make_two_lobe(n, rng):
    """Two asymmetric Gaussian lobes plus floor."""
    n = int(n)
    if n <= 1:
        return make_uniform(n)

    t = np.linspace(0.0, 1.0, n)
    c1 = float(rng.uniform(0.05, 0.42))
    c2 = float(rng.uniform(0.58, 0.95))
    s1 = float(rng.uniform(0.05, 0.28))
    s2 = float(rng.uniform(0.05, 0.28))
    w1 = float(rng.exponential(1.0) + 0.1)
    w2 = float(rng.exponential(1.0) + 0.1)

    p = 0.04
    p += w1 * np.exp(-0.5 * ((t - c1) / s1) ** 2)
    p += w2 * np.exp(-0.5 * ((t - c2) / s2) ** 2)

    return normalize_to_sum1(p)


def add_start_template(starts, p):
    """Normalize, score, and append to starts if finite."""
    p = normalize_to_sum1(p)
    sc = fast_score_array(p)
    if math.isfinite(sc):
        starts.append((float(sc), p))


def generate_starts(n, rng, global_best, random_count=30):
    """Generate diverse scored starts for a given length."""
    n = int(n)
    if n < 2:
        n = 2

    add_start_template = local_add_start_placeholder = None
    # Avoid accidental closure; use top-level helper.
    starts = []

    start_add = add_start_template
    start_add(starts, make_uniform(n))

    if global_best is not None:
        start_add(starts, upsample_sequence(global_best, n))

    slopes = (-1.8, -1.2, -0.7, -0.35, 0.35, 0.7, 1.2, 1.8)
    for sl in slopes:
        start_add(starts, make_linear(n, sl))

    # A modest deterministic grid of power-law starts.
    alphas = (-0.50, -0.25, 0.0, 0.25, 0.50)
    weights = (1.0, 1.8, 0.5, 3.0, 0.3)
    for al in alphas:
        for ar in alphas:
            for wl in (1.0, 2.0, 0.5):
                wr = max(0.1, 3.0 / wl)
                if al == ar and abs(wl - wr) <= 1e-12:
                    # Skip the most symmetric case.
                    continue
                start_add(starts, make_power(n, al, ar, wl, wr))

    for _ in range(max(0, int(random_count))):
        r = rng.random()
        if r < 0.30:
            p = make_fourier(n, rng)
        elif r < 0.55:
            p = make_blocks(n, rng)
        elif r < 0.80:
            p = make_two_lobe(n, rng)
        else:
            p = make_power(
                n,
                float(rng.uniform(-0.5, 0.5)),
                float(rng.uniform(-0.5, 0.5)),
                float(rng.exponential(1.0) + 0.1),
                float(rng.exponential(1.0) + 0.1),
            )
        start_add(starts, p)

    # Add a reversed copy of the current best if it exists; reversal is an
    # isometry of the objective, but it can lead local search elsewhere.
    if global_best is not None:
        rev = global_best[::-1]
        start_add(starts, upsample_sequence(rev, n))

    starts.sort(key=_first)
    return starts


def local_add_start_placeholder():
    return None


def add_candidate(pool, p, sc):
    """Add a candidate to the portfolio, keeping it sorted and bounded."""
    if p is None:
        return
    try:
        sc = float(sc)
    except Exception:
        return
    if not math.isfinite(sc):
        return

    p = np.asarray(p, dtype=np.float64).copy()
    pool.append((sc, p))
    pool.sort(key=_first)

    if len(pool) > 24:
        pool.pop()


def local_search(p0, budget_s, seed=0, label=""):
    """
    Local search using weighted stochastic subgradients.

    The sequence is kept nonnegative and normalized to sum 1.
    """
    p = normalize_to_sum1(p0)
    n = int(p.size)
    if n <= 1:
        return p, fast_score_array(p)

    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()

    spec = np.empty(N // 2 + 1, dtype=np.complex128)
    buf_cur = np.empty(N, dtype=np.float64)
    buf_tmp = np.empty(N, dtype=np.float64)

    convolution_into(p, spec, buf_cur, N)
    init_score = float(2.0 * n * np.max(buf_cur[:L]))

    best = p.copy()
    best_score = init_score

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))

    lr = 0.18
    min_lr = 0.02
    max_lr = 0.95

    beta = 0.70
    tol = 0.025
    min_tol = 0.0012

    mom = np.zeros(n, dtype=np.float64)
    floor = max(1e-30, 1e-14 / float(n))

    base_K = max(12, min(64, n // 80))
    Kloop = base_K
    max_K = min(128, max(base_K + 24, base_K * 2))
    Tcap = min(384, max(192, base_K * 4))

    width = 0.08
    min_width = 0.004
    max_width = 0.25

    stagnant = 0
    shakes = 0
    it = 0

    print(f"[{label}] local_search start n={n} score={init_score:.10f} budget={float(budget_s):.1f}s")

    while time.monotonic() < deadline - 0.05:
        convolution_into(p, spec, buf_cur, N)
        curc = buf_cur[:L]
        M = float(np.max(curc))
        if not math.isfinite(M):
            break

        score = float(2.0 * n * M)

        if score < best_score:
            best = p.copy()
            best_score = score
            stagnant = 0
            lr = min(max_lr, lr * 1.035)
            width = max(min_width, width * 0.992)
            if Kloop > base_K:
                Kloop = max(base_K, Kloop - 1)

        active, weights = choose_weighted_active(curc, rng, Kloop, Tcap, width)
        grad = topk_gradient_weighted(p, active, weights)

        mean_g = float(np.mean(grad))
        g = grad
        g -= mean_g

        var = float(g.dot(g)) / float(n)
        if not math.isfinite(var) or var <= 1e-24:
            stagnant += 1
            if stagnant > 120:
                break

            g = rng.normal(0.0, 1.0, n)
            g -= float(np.mean(g))
            var = float(g.dot(g)) / float(n)
            if not math.isfinite(var) or var <= 1e-30:
                continue

        inv_scale = 1.0 / math.sqrt(var)
        g *= inv_scale

        mom *= beta
        mom += (1.0 - beta) * g

        mom_norm = math.sqrt(float(mom.dot(mom)) / float(n))
        if not math.isfinite(mom_norm) or mom_norm <= 1e-16:
            stagnant += 1
            if stagnant > 120:
                break
            continue

        e = -lr * mom / mom_norm
        np.clip(e, -1.2, 1.2, out=e)

        cand = p * np.exp(e)
        np.maximum(cand, floor, out=cand)
        s = float(cand.sum())
        if s <= 0.0 or not math.isfinite(s):
            break
        cand /= s

        convolution_into(cand, spec, buf_tmp, N)
        cand_score = float(2.0 * n * np.max(buf_tmp[:L]))

        accept = False
        improved = False

        if math.isfinite(cand_score):
            if cand_score <= score * (1.0 + tol) + 1e-15:
                accept = True
                improved = cand_score < score * (1.0 - 1e-12)
            else:
                e2 = e * 0.5
                np.clip(e2, -0.6, 0.6, out=e2)
                cand2 = p * np.exp(e2)
                np.maximum(cand2, floor, out=cand2)
                s2 = float(cand2.sum())
                if s2 > 0.0 and math.isfinite(s2):
                    cand2 /= s2
                    convolution_into(cand2, spec, buf_tmp, N)
                    cand_score2 = float(2.0 * n * np.max(buf_tmp[:L]))
                    if math.isfinite(cand_score2) and cand_score2 <= score * (1.0 + 0.5 * tol) + 1e-15:
                        cand = cand2
                        cand_score = cand_score2
                        accept = True
                        improved = cand_score < score

        if accept:
            p = cand
            score = cand_score
            if improved:
                lr = min(max_lr, lr * 1.04)
                tol = max(min_tol, tol * 0.994)
                stagnant = 0
                mom *= 0.85
            else:
                lr = max(min_lr, lr * 0.98)
                stagnant += 1
        else:
            lr = max(min_lr, lr * 0.62)
            mom *= 0.7
            stagnant += 1

        if stagnant > 320:
            strength = 0.25 + 0.35 * min(1.0, float(stagnant) / 1200.0)
            pert = smooth_perturb(best, rng, strength)
            p = normalize_to_sum1(0.5 * best + 0.5 * pert)
            mom = np.zeros(n, dtype=np.float64)
            lr = 0.28
            width = min(max_width, width * 1.6)
            Kloop = min(max_K, Kloop + 8)
            stagnant = 60
            shakes += 1
            if shakes > 30:
                break

        it += 1
        if it % 5000 == 0 and (time.monotonic() - t0) > 15.0:
            print(f"[{label}] iter={it} current={score:.10f} best={best_score:.10f}")

    return best, best_score


def get_previous_sequence():
    """Load the previously known sequence if present, capped at 8192."""
    try:
        obj = globals().get("height_sequence_1", None)
    except Exception:
        obj = None

    if obj is None:
        return None

    try:
        arr = np.asarray(obj, dtype=np.float64)
    except Exception:
        return None

    if arr.size == 0:
        return None

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)

    if arr.size > 8192:
        arr = upsample_sequence(arr, 8192)

    return normalize_to_sum1(arr)


def sanitize_for_return(p):
    """Return a valid Python list of non-negative finite floats."""
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return [1.0]

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)

    s = float(p.sum())
    if s < 0.01 or not math.isfinite(s):
        return [1.0]

    p = p / s
    p = np.clip(p, 0.0, 1000.0)

    if float(p.sum()) < 0.01:
        return [1.0]

    return [float(x) for x in p]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point.

    Runs a multi-length portfolio of starts and local searches, then
    exact-validates the top candidates before returning the best sequence.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))

    prev = get_previous_sequence()

    if budget < 30.0:
        if prev is not None:
            return sanitize_for_return(prev)
        return sanitize_for_return(make_uniform(1024))

    rng_seed = int(seed)
    rng = np.random.default_rng(rng_seed)

    pool = []

    if prev is not None:
        add_candidate(pool, prev, fast_score_array(prev))
        add_candidate(pool, prev[::-1], fast_score_array(prev[::-1]))

    add_candidate(pool, make_uniform(1024), 2.0)

    # Keep a safety reserve for exact validation and final cleanup.
    reserve = 45.0 if budget > 250.0 else min(10.0, 0.1 * budget)
    search_total = max(60.0, budget - reserve)
    search_deadline = t0 + search_total

    if search_total > 1600.0:
        known_alloc = min(330.0, 0.11 * search_total)
        med_alloc = min(480.0, 0.16 * search_total)
        final_alloc = min(1380.0, 0.47 * search_total)
        long_target = 8192
    elif search_total > 800.0:
        known_alloc = min(220.0, 0.12 * search_total)
        med_alloc = min(300.0, 0.18 * search_total)
        final_alloc = min(850.0, 0.55 * search_total)
        long_target = 4096
    elif search_total > 300.0:
        known_alloc = min(120.0, 0.18 * search_total)
        med_alloc = 0.0
        final_alloc = max(0.0, 0.72 * search_total)
        long_target = 2048
    else:
        known_alloc = max(0.0, 0.25 * search_total)
        med_alloc = 0.0
        final_alloc = max(0.0, search_total - 5.0)
        long_target = 1024

    # Stage 1: refine the previously known sequence at its native length.
    if prev is not None:
        b = min(known_alloc, search_deadline - time.monotonic() - 10.0)
        if b > 15.0:
            if b > 90.0:
                b1 = b * 0.55
                b2 = b * 0.45
                if time.monotonic() < search_deadline - 10.0:
                    seq, sc = local_search(prev, b1, rng_seed + 1, "known")
                    add_candidate(pool, seq, sc)
                if time.monotonic() < search_deadline - 20.0:
                    seq, sc = local_search(prev[::-1], b2, rng_seed + 2, "known_rev")
                    add_candidate(pool, seq, sc)
            else:
                seq, sc = local_search(prev, b, rng_seed + 1, "known")
                add_candidate(pool, seq, sc)

    pool.sort(key=_first)
    if pool:
        print(f"[portfolio] after known stage best={pool[0][0]:.10f} len={pool[0][1].size} t={time.monotonic()-t0:.1f}s")

    # Stage 2: intermediate length.
    b_med = min(med_alloc, search_deadline - time.monotonic() - 20.0)
    if b_med > 50.0 and med_alloc > 0.0:
        n_med = 2048 if long_target >= 4096 else 1024
        best_p = pool[0][1] if pool else make_uniform(n_med)

        starts = generate_starts(n_med, rng, best_p, random_count=18)
        if prev is not None:
            add_start_template(starts, upsample_sequence(prev, n_med))
        if starts:
            starts.sort(key=_first)
            k = min(3, len(starts), max(1, int(b_med / 130.0) + 1))
            ws = (0.55, 0.30, 0.15)
            for i in range(k):
                if time.monotonic() >= search_deadline - 20.0:
                    break
                b = b_med * ws[i]
                if b < 12.0:
                    continue
                sc, sq = starts[i]
                seq, sc2 = local_search(
                    sq,
                    min(b, search_deadline - time.monotonic() - 10.0),
                    rng_seed + 100 + i,
                    f"med_{i}"
                )
                add_candidate(pool, seq, sc2)

    pool.sort(key=_first)
    if pool:
        print(f"[portfolio] after med stage best={pool[0][0]:.10f} len={pool[0][1].size} t={time.monotonic()-t0:.1f}s")

    # Stage 3: main final length.
    b_fin = min(final_alloc, search_deadline - time.monotonic() - 30.0)
    if b_fin > 50.0:
        n_final = 4096 if long_target >= 4096 else long_target
        if n_final < 512:
            n_final = 512

        best_p = pool[0][1] if pool else make_uniform(n_final)
        starts = generate_starts(n_final, rng, best_p, random_count=40)
        if prev is not None:
            add_start_template(starts, upsample_sequence(prev, n_final))

        if starts:
            starts.sort(key=_first)
            # After generating, always include the current best upsampled.
            add_start_template(starts, upsample_sequence(pool[0][1] if pool else best_p, n_final))
            starts.sort(key=_first)

            k = min(4, len(starts), max(1, int(b_fin / 230.0) + 1))
            ws = (0.45, 0.25, 0.18, 0.12)
            for i in range(k):
                if time.monotonic() >= search_deadline - 30.0:
                    break
                b = b_fin * ws[i]
                if b < 15.0:
                    continue
                sc, sq = starts[i]
                seq, sc2 = local_search(
                    sq,
                    min(b, search_deadline - time.monotonic() - 10.0),
                    rng_seed + 200 + i,
                    f"final_{i}"
                )
                add_candidate(pool, seq, sc2)

    pool.sort(key=_first)
    if pool:
        print(f"[portfolio] after final stage best={pool[0][0]:.10f} len={pool[0][1].size} t={time.monotonic()-t0:.1f}s")

    # Stage 4: long refinement on the best found so far, optionally upsampling
    # to a larger length when the best is shorter.
    remaining = search_deadline - time.monotonic()
    if remaining > 80.0 and pool:
        best_sc, best_p = pool[0]
        n_best = int(best_p.size)

        # If the best is small, spend a short prefix improving it directly.
        if n_best <= 1024 and remaining > 300.0:
            b_small = min(remaining * 0.30, 280.0)
            if b_small > 30.0:
                seq, sc = local_search(
                    best_p,
                    min(b_small, search_deadline - time.monotonic() - 10.0),
                    rng_seed + 777,
                    "long_small"
                )
                add_candidate(pool, seq, sc)
                pool.sort(key=_first)
                remaining = search_deadline - time.monotonic()

        if remaining > 80.0:
            # Choose a longer target if we have reason to think it can help.
            if n_best < long_target and long_target >= 2048:
                n_long = min(8192, max(n_best * 2, long_target))
            else:
                n_long = max(n_best, long_target)

            n_long = min(8192, max(512, n_long))

            if n_best == n_long:
                start_long = pool[0][1]
            else:
                start_long = upsample_sequence(pool[0][1], n_long)

            b_long = max(0.0, remaining - 5.0)
            if b_long > 30.0:
                seq, sc = local_search(
                    start_long,
                    min(b_long, search_deadline - time.monotonic() - 5.0),
                    rng_seed + 999,
                    "long"
                )
                add_candidate(pool, seq, sc)

    pool.sort(key=_first)

    if pool:
        print(f"[portfolio] before exact best_fast={pool[0][0]:.10f} len={pool[0][1].size} t={time.monotonic()-t0:.1f}s")

    return_deadline = t0 + max(10.0, budget - 2.0)

    exact_best_p = None
    exact_best_score = float("inf")

    max_exact_evals = 4
    if pool and pool[0][1].size > 4096:
        max_exact_evals = 2

    top_exact = min(max_exact_evals, len(pool))
    for i in range(top_exact):
        if time.monotonic() > return_deadline - 25.0:
            break

        sc, p = pool[i]
        if p.size == 0 or p.size > 8192:
            continue

        ex = exact_score_array(p)
        if math.isfinite(ex) and ex < exact_best_score:
            exact_best_score = ex
            exact_best_p = p
            print(f"[exact] candidate {i} len={p.size} exact={ex:.10f}")

        if time.monotonic() > return_deadline - 10.0:
            break

    if exact_best_p is None:
        if pool:
            exact_best_p = pool[0][1]
        else:
            exact_best_p = make_uniform(1024)

    print(f"[final] returning len={np.asarray(exact_best_p).size} fast={pool[0][0] if pool else 'na'} t={time.monotonic()-t0:.1f}s")
    return sanitize_for_return(exact_best_p)
