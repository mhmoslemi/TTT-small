    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = max_c + np.log(np.sum(exp_shifted)) / 1.0  # (1/t)*log(sum(exp(t*c)))
    obj = max_c + np.log(np.sum(exp_shifted)) / t
    
    # Softmax weights
    w = exp_shifted / np.sum(exp_shifted)
    
    # Gradient computation
    R = cross_correl(a, w)  # R[m] = sum_j a[j] * w[j+m]
    
    # Correction term: w[2m] * a[m] for m = 0, ..., n-1
    # w has length 2n-1, indices 0 to 2n-2. 2m ranges from 0 to 2n-2. OK.
    correction = w[2 * np.arange(n)] * a
    grad = 2.0 * R - correction
    
    return obj, grad


def project(a, target_sum):
    """Project onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    a = a * (target_sum / s)
    # Re-clip and re-normalize (one more pass)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    a = a * (target_sum / s)
    return a


def adam_optimize(a0, budget_s, target_sum, seed=42, t_start=50.0, t_end=500.0):
    """
    Adam optimization of the smoothed max-autocorrelation.
    
    a0: initial sequence
    budget_s: time budget in seconds
    target_sum: target sum for normalization
    t_start, t_end: initial and final temperature for log-sum-exp
    """
    a = project(a0.astype(np.float64).copy()), 
    n = len(a)
    t0 = time.monotonic()
    deadline = t0 + budget_s
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    
    best_a = a.copy()
    
    # Initial evaluation
    c = fft_conv(a)
    best_score = 2.0 * n * np.max(c) / (target_sum ** 2)
    current_score = best_score
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Adaptive temperature
        frac = min(1.0, it / max(1, int(budget_s * 20)))  # ramp up over time
        t = t_start + (t_end - t_start) * frac
        
        obj, grad = smooth_objective_and_grad(a, t)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        a_new = project(a_new, target_sum)
        
        # Evaluate
        c_new = fft_conv(a_new)
        score = 2.0 * n * np.max(c_new) / (target_sum ** 2)
        
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        # Accept with some probability (simulated annealing style)
        if score <= current_score or (np.random.random() < np.exp(-(score - current_score) / (0.01 + 0.1 / (1 + it/100)))):
            a = a_new
            current_score = score
            lr = min(2.0, lr * 1.002)  # slowly increase lr

                break

            m = np.zeros(n)
            v = np.zeros(n)
            lr = 0.3
            no_improve = 0
            current_score = best_score
            it = 0

def fft_conv(a):
    """FFT-based linear convolution a*a, returns length 2n-1."""
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(c, 0.0)


def cross_correl_first_n(a, w):
    """
    Compute R[m] = sum_j a[j] * w[j+m] for m = 0,...,n-1.
    a has length n, w has length 2n-1.
    Uses FFT for efficiency.
    """
    n = len(a)
    Lw = len(w)
    needed = n + Lw - 1
    N = 1 << (needed - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(fw * np.conj(fa), N)[:n]
    return result

    eps = 1e-8
    lr = lr_init

    best_a = a.copy()
    best_score = exact_score(a)
    current_score = best_score
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # Adaptive temperature: ramp from t_start to t_end
        progress = min(1.0, it / max(100, max_iter_per_restart))
        t = t_start + (t_end - t_start) * progress

        obj, grad = smooth_obj_grad(a, t)

        if not np.all(np.isfinite(grad)):
            break

        # Gradient in log-space: dL/db = a * dL/da
        log_grad = a * grad

        # Adam update in log-space
        m = beta1 * m + (1.0 - beta1) * log_grad
        v = beta2 * v + (1.0 - beta2) * log_grad ** 2
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Update in log-space
        log_step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_step = np.clip(log_step, -2.0, 2.0)

        a_new = a * np.exp(-log_step)
        a_new = project_sum(a_new, target_sum)

        # Evaluate
        score = exact_score(a_new)

        if not np.isfinite(score):
            lr = max(0.01, lr * 0.8)
            continue

        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # Accept/reject (simulated annealing)
        delta = score - current_score
        temp = 0.005 + 0.05 / (1.0 + it / 200.0)
        if delta <= 0 or rng.random() < math.exp(-delta / temp):
            a = a_new
            current_score = score
            if delta < -1e-10:
                lr = min(1.5, lr * 1.005)
            else:
                lr = max(0.05, lr * 0.997)
        else:
            lr = max(0.02, lr * 0.9)

        # Restart if stuck
        if no_improve > 800:
            if time.monotonic() > deadline - 15:
                break
            # Structured perturbation
            pert = best_a * np.exp(rng.normal(0, 0.15 + 0.1 * no_improve / 500.0, n))
            a = project_sum(pert, target_sum)
            current_score = exact_score(a)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0
            it = 0

    return best_a, best_score


def subgradient_polish(a0, budget_s, target_sum, seed=0):
    """
    Direct subgradient method targeting the top autocorrelation peaks.
    Uses top-K averaging for a smoother signal.
    """
    a = project_sum(a0.astype(np.float64).copy(), target_sum)
    n = len(a)
    t0 = time.monotonic()
    deadline = t0 + budget_s

    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_score = exact_score(a)
    eta = 0.3
    no_improve = 0

    K = max(3, n // 50)

    while time.monotonic() < deadline:
        c = fft_conv(a)
        L = 2 * n - 1

        # Top-K peaks
        if K >= L:
            top_idx = np.arange(L)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        # Compute subgradient as average over top-K
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for m in range(i_min, i_max + 1):
                j = k - m
                if m == j:
                    grad[m] += a[j]
                else:
                    grad[m] += a[j]
                    grad[j] += a[m]
        grad /= float(K)

        # Center to preserve sum
        grad -= np.mean(grad)

        scale = np.max(np.abs(grad))
        if scale < 1e-15:
            break
        grad /= scale

        a_new = a - eta * grad
        a_new = project_sum(a_new, target_sum)
        score = exact_score(a_new)

        if not np.isfinite(score):
            eta *= 0.7
            continue

        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        if score <= exact_score(a) * 1.001:
            a = a_new
            if score < best_score:
                eta = min(1.0, eta * 1.03)
        else:
            eta = max(0.01, eta * 0.8)

        if no_improve > 500:
            if time.monotonic() > deadline - 10:
                break
            a = project_sum(best_a * np.exp(rng.normal(0, 0.2, n)), target_sum)
            eta = 0.3
            no_improve = 0

    return best_a, best_score


def generate_diverse_starts(n, seed=0):
    """Generate a diverse pool of starting sequences."""
    rng = np.random.default_rng(seed)
    starts = []
    t = np.linspace(0.0, 1.0, n)

    # Basic shapes
    starts.append(np.ones(n))
    starts.append(np.sin(np.pi * t) ** 2)
    starts.append(0.5 - 0.5 * np.cos(2 * np.pi * t))
    starts.append(np.exp(-((t - 0.5) ** 2) / 0.05))
    starts.append(np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02) + 0.05)
    starts.append(np.minimum(t * 5, (1 - t) * 5, 1) + 0.1)

    # Polynomial shapes
    for p in [2, 3, 4]:
        starts.append(((t - 0.5) ** 2 * 4 + 0.1) ** p)
        starts.append(1.0 - p * (t - 0.5) ** 2)

    # Low-frequency random
    for _ in range(15):
        z = np.zeros(n)
        for k in range(1, 6):
            amp = rng.uniform(-1, 1) / k
            phase = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.exp(z - np.max(z)) + 0.05
        starts.append(a)

    # Block sequences
    for _ in range(5):
        nblocks = int(rng.integers(3, 8))
        edges = np.sort(rng.integers(1, n - 1, nblocks - 1))
        edges = np.concatenate([[0], edges, [n]])
        a = np.ones(n)
        for i in range(nblocks):
            a[edges[i]:edges[i + 1]] = rng.uniform(0.1, 2.0)
        starts.append(a)

    # Tapered with random internal structure
    for _ in range(5):
        base = np.sin(np.pi * t) ** 2
        internal = 1.0 + 0.3 * np.sin(2 * np.pi * rng.integers(2, 8) * t)
        starts.append(base * internal)

    # From parent if available
    try:
        parent = height_sequence_1
        if parent is not None and len(parent) > 0:
            parent = np.array(parent, dtype=np.float64)
            if len(parent) == n:
                starts.append(parent.copy())
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                starts.append(np.interp(x_new, x_old, parent))
    except (NameError, AttributeError):
        pass

    # Normalize
    result = []
    for s in starts:
        s = np.maximum(s, 0.0).astype(np.float64)
        ss = np.sum(s)
        if ss > 1e-10:
            s = s * n / ss
        else:
            s = np.ones(n)
        result.append(s)

    return result


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.
    Multi-phase: explore -> optimize -> polish -> validate.
    """
    t0 = time.monotonic()
    total_deadline = t0 + budget_s - 40.0

    global _best_seq, _best_score
    _best_seq = None
    _best_score = float('inf')

    def update_best(seq, score):
        global _best_seq, _best_score
        if seq is not None and np.isfinite(score) and score < _best_score:
            _best_seq = seq.copy()
            _best_score = float(score)

    # === Phase 1: Multi-length exploration ===
    lengths = [512, 1024, 2048]
    n_alloc = len(budget_s) if isinstance(budget_s, list) else 1

    # Try each length with a quick Adam run
    exploration_budget = min(600.0, (total_deadline - t0) * 0.25)
    per_length = exploration_budget / len(lengths)

    for li, n in enumerate(lengths):
        if time.monotonic() > total_deadline - 100:
            break
        starts = generate_diverse_starts(n, seed + li * 100)

        # Quick score all starts
        scored = []
        for s in starts:
            sc = exact_score(s)
            if np.isfinite(sc):
                scored.append((sc, s))
        scored.sort(key=lambda x: x[0])

        update_best(scored[0][1], scored[0][0])

        # Quick Adam on top 2 starts
        time_this = min(per_length * 0.7, total_deadline - time.monotonic() - 60)
        if time_this > 10:
            for i in range(min(2, len(scored))):
                if time.monotonic() > total_deadline - 60:
                    break
                sc, s = scored[i]
                if sc > _best_score * 1.05:
                    break
                t_this = min(time_this / (2 - i), total_deadline - time.monotonic() - 30)
                if t_this < 5:
                    break
                try:
                    seq, score = adam_logspace(s, t_this, float(n),
                                               seed=seed + li * 100 + i,
                                               t_start=20.0, t_end=200.0,
                                               lr_init=0.4)
                    update_best(seq, score)
                except Exception:
                    pass

        print(f"Phase 1: n={n}, best score so far = {_best_score:.6f}")

    # === Phase 2: Intensive optimization at best length ===
    if _best_seq is not None and time.monotonic() < total_deadline - 60:
        n_best = len(_best_seq)
        target = float(n_best)
        remaining = total_deadline - time.monotonic()

        # Extended Adam with higher temperature range
        adam_budget = min(remaining * 0.5, 900.0)
        if adam_budget > 30:
            try:
                seq, score = adam_logspace(_best_seq, adam_budget, target,
                                           seed=seed + 500,
                                           t_start=50.0, t_end=800.0,
                                           lr_init=0.5)
                update_best(seq, score)
            except Exception as e:
                print(f"Phase 2a error: {e}")

            # Second Adam run with different params
            if time.monotonic() < total_deadline - 60:
                adam_budget2 = min(remaining * 0.2, 400.0)
                if adam_budget2 > 20:
                    try:
                        seq, score = adam_logspace(_best_seq, adam_budget2, target,
                                                   seed=seed + 600,
                                                   t_start=100.0, t_end=2000.0,
                                                   lr_init=0.2)
                        update_best(seq, score)
                    except Exception as e:
                        print(f"Phase 2b error: {e}")

        print(f"Phase 2: best score = {_best_score:.6f}")

    # === Phase 3: Subgradient polish ===
    if _best_seq is not None and time.monotonic() < total_deadline - 30:
        n_best = len(_best_seq)
        target = float(n_best)
        sg_budget = min(200.0, total_deadline - time.monotonic() - 20)
        if sg_budget > 15:
            try:
                seq, score = subgradient_polish(_best_seq, sg_budget, target,
                                                seed=seed + 700)
                update_best(seq, score)
            except Exception as e:
                print(f"Phase 3 error: {e}")

        print(f"Phase 3: best score = {_best_score:.6f}")

    # === Phase 4: Try upsampling to larger length ===
    if _best_seq is not None and time.monotonic() < total_deadline - 100:
        n_cur = len(_best_seq)
        n_large = n_cur * 2
        if n_large <= 8192:
            x_old = np.linspace(0, 1, n_cur)
            x_new = np.linspace(0, 1, n_large)
            up = np.interp(x_new, x_old, _best_seq)
            up = project_sum(up, float(n_large))

            up_budget = min(300.0, total_deadline - time.monotonic() - 30)
            if up_budget > 20:
                try:
                    seq, score = adam_logspace(up, up_budget, float(n_large),
                                               seed=seed + 800,
                                               t_start=30.0, t_end=300.0,
                                               lr_init=0.4)
                    update_best(seq, score)
                except Exception as e:
                    print(f"Phase 4 error: {e}")

            print(f"Phase 4 (n={n_large}): best score = {_best_score:.6f}")

    # === Phase 5: Final fine-tuning ===
    if _best_seq is not None and time.monotonic() < total_deadline - 10:
        n_best = len(_best_seq)
        target = float(n_best)
        final_budget = total_deadline - time.monotonic() - 10
        if final_budget > 10:
            try:
                seq, score = adam_logspace(_best_seq, final_budget, target,
                                           seed=seed + 900,
                                           t_start=200.0, t_end=3000.0,
                                           lr_init=0.1)
                update_best(seq, score)
            except Exception as e:
                print(f"Phase 5 error: {e}")

        print(f"Phase 5: best score = {_best_score:.6f}")

    # === Final validation and return ===
    if _best_seq is None:
        fallback = np.ones(1024)
        return [1.0] * 1024

    # Normalize to mean 1
    result = _best_seq.astype(np.float64).copy()
    result = np.maximum(result, 0.0)
    result = np.minimum(result, 1000.0)
    s = np.sum(result)
    if s < 0.01:
        return [1.0] * 1024
    result = result * (len(result) / s)
    result = np.clip(result, 0.0, 1000.0)

    # Exact validation
    try:
        exact = evaluate_sequence(result.tolist())
        print(f"Final exact evaluation: {exact:.6f}")
        if not np.isfinite(exact):
            result = np.ones(1024)
    except Exception:
        pass

    return [float(x) for x in result]
