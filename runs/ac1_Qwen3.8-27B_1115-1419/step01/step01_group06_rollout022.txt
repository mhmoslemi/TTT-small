Hybrid FFT-based smoothed-Linf optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Uses:
- Smoothed-Linf gradient in log-probability space with Adam optimizer
- Adaptive beta (smoothing) schedule
- Adaptive learning rate with stall detection
- Periodic restarts with structured perturbations
- Multi-scale exploration (different perturbation sizes)

import numpy as np
import time


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(v) / len(v)
    return e / s


def _log_softmax(v):
    """Log of v, centered for numerical stability."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 1e-300)
    return np.log(v)


def _compute_score(a, N):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 using FFT."""
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2*n-1]
    c = np.maximum(c, 0)
    s = a.sum()
    if s < 1e-300:
        return float('inf')
    return float(2.0 * n * c.max() / (s * s))


def _exact_score(a):
    """Compute exact score using np.convolve."""
    n = len(a)
    b = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * b.max() / (s * s))

    """
    Compute the gradient of the smoothed-Linf objective.
    
    grad_j = 2 * sum_{l=0}^{n-1} a[l] * w[j+l]

    g_full = np.fft.irfft(Ar * W, N)
