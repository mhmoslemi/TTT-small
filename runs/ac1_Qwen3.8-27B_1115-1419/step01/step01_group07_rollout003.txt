Optimize step function sequence to minimize 2*n*max(a*a)/(sum(a))^2.
Uses Nesterov-accelerated subgradient with adaptive step sizes,
periodic L-BFGS polishing on LSE-smoothed objective, and peak-shaving heuristics.

import numpy as np

from scipy.optimize import minimize

def compute_score(a):
    """Compute the score directly (for internal use)."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def subgradient_of_max_conv(a):
    """Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns gradient vector and the active set indices."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros(n), 0
    
    # Active set: all indices where c[k] >= max_c - epsilon
    epsilon = max_c * 1e-10 + 1e-15
    active = np.where(c >= max_c - epsilon)[0]
    
    g = np.zeros(n)
    for k in active:
        # d/d a[i] of conv(a,a)[k] = a[k-i] for valid i
        # conv(a,a)[k] = sum_i a[i] * a[k-i] (where 0<=i<n, 0<=k-i<n)
        # So d/da[i] = a[k-i] for 0<=k-i<n, i.e., k-n+1 <= i <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # a[k-i] for i in [lo, hi] gives a[k-hi] to a[k-lo]
        g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    
    g /= len(active)  # Average over active set (subgradient of max)
    return g, len(active)

def project_simplex(a):
    """Project onto {x: x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def nesterov_subgradient(a, max_iter=10000, initial_step=0.01, time_limit=60):
    """Nesterov-accelerated subgradient method for minimizing max(conv(a,a))."""
    n = len(a)
    a = project_simplex(a.copy())
    
    x = a.copy()  # Current iterate
    y = a.copy()  # Nesterov extrapolation point
    t = 1.0
    step0 = initial_step
    best_a = a.copy()
    best_cmax = np.convolve(a, a).max()
    
    start = time.monotonic()
    for iteration in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        # Evaluate at y
        c_y = np.convolve(y, y)
        max_y = c_y.max()
        
        if max_y < best_cmax:
            best_cmax = max_y
            best_a = y.copy()
        
        # Subgradient at y
        g, n_active = subgradient_of_max_conv(y)
        
        # Step size (Polyak-like or diminishing)
        step = step0 / (1.0 + 0.01 * iteration)
        
        # Update x
        x_new = y - step * g
        x_new = project_simplex(x_new)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + (t - 1.0) / t_new * (x_new - x)
        y_new = project_simplex(y_new)
        
        x = x_new
        y = y_new
        t = t_new
    
    return best_a, best_cmax

def lse_smoothed_max_conv(a, tau):
    """LSE-smoothed max of conv(a,a)."""
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    vals = np.exp(shifted - shifted.max())  # Extra stability
    result = tau * np.log(vals.sum()) + c_max
    return result

def lse_gradient(x, tau):
    """Gradient of LSE-smoothed max of conv(softmax(x), softmax(x)) w.r.t. x."""
    a = np.exp(x - x.max())
    a /= a.sum()
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    
    # d/d a[i] of smoothed_max = sum_k w[k] * d conv(a,a)[k]/da[i]
    grad_a = np.zeros(len(a))
    n = len(a)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_a[lo:hi+1] += w[k] * a[k-hi:k-lo+1][::-1]
    
    # Chain rule: dx = a * (grad_a - a.dot(grad_a))
    dot = a.dot(grad_a)
    return a * (grad_a - dot), a

def lbfgs_smooth_opt(x, n, tau_schedule, max_time=120):
    """L-BFGS on LSE-smoothed objective with decreasing tau."""
    a_curr = np.exp(x - x.max())
    a_curr /= a_curr.sum()
    best_a = a_curr.copy()
    best_score = compute_score(a_curr)

            return lse_smoothed_max_conv(a_t, tau)

            g, _ = lse_gradient(x_, tau)
            return g

                obj, x, jac=grad,

        a_curr /= a_curr.sum()
        score = compute_score(a_curr)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_curr.copy()
    
    return best_a, best_score, x

def peak_shave(a, n_peaks=10, strength=0.01):
    """Directly reduce values that contribute most to peaks in convolution."""
    n = len(a)
    c = np.convolve(a, a)
    
    # Find the top peaks
    sorted_idx = np.argsort(c)[::-1][:n_peaks]
    
    a_new = a.copy()
    for k in sorted_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # The contribution to c[k] from a[i] is a[i]*a[k-i]
        # Reduce a[i] proportionally to its contribution
        for i in range(lo, hi+1):
            j = k - i
            if 0 <= j < n:

                a_new[i] -= strength * contrib / a[i] if a[i] > 1e-15 else 0
                a_new[j] -= strength * contrib / a[j] if a[j] > 1e-15 else 0
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

"""
Optimize step function sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated subgradient with adaptive peak focusing,
L-BFGS on LSE-smoothed objective with aggressive annealing,
peak-shaving heuristics, and multi-phase time allocation.
Key improvements over previous: efficient O(n) gradient via correlate,
Nesterov momentum, adaptive single-vs-multi-peak strategy, and
stricter convergence control.
"""

import numpy as np
import time
from scipy.optimize import minimize


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def convolve_and_grad(a, w):
    """Compute weighted sum of convolution contributions.
    w has length 2n-1 (weights for each conv index).
    Returns grad_a of length n: grad_a[i] = sum_j w[i+j]*a[j]
    This equals correlate(w, a, 'valid').
    """
    # grad_a[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is cross-correlation: (w * a)[i] where * is correlation
    grad_a = np.correlate(w, a, mode='valid')
    return grad_a


def subgradient_single_peak(a, k_peak):
    """Subgradient from a single peak at index k_peak."""
    n = len(a)
    lo = max(0, k_peak - n + 1)
    hi = min(n - 1, k_peak)
    g = np.zeros(n)
    # d conv(a,a)[k]/da[i] = a[k-i] for i in [lo, hi]
    # a[k-hi] ... a[k-lo] reversed
    g[lo:hi+1] = a[k_peak - hi:k_peak - lo + 1][::-1]
    return g


def subgradient_multi_peak(a, active_indices):
    """Averaged subgradient over multiple active peak indices."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += a[k - hi:k - lo + 1][::-1]
    g /= len(active_indices)
    return g


def softmax_vec(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_obj_and_grad(x, tau):
    """LSE-smoothed max of conv(softmax(x), softmax(x)) and its gradient."""
    a = softmax_vec(x)
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    obj = tau * np.log(w_sum) + c_max
    w /= w_sum
    # grad w.r.t. a: correlate(w, a, 'valid')
    grad_a = np.correlate(w, a, mode='valid')
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x


def nesterov_subgradient(a_init, max_iter=20000, initial_step=0.01,
                         time_limit=60, focus='auto'):
    """Nesterov-accelerated subgradient for min max(conv(a,a)) s.t. a>=0, sum=1.
    focus: 'single' (only worst peak), 'multi' (all near-max), or 'auto'.
    """
    n = len(a_init)
    x = a_init.copy()
    y = a_init.copy()
    t = 1.0
    best_a = a_init.copy()
    best_cmax = np.convolve(a_init, a_init).max()
    no_improve_count = 0

    start = time.monotonic()
    step = initial_step

    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break

        c_y = np.convolve(y, y)
        max_y = c_y.max()

        if max_y < best_cmax - 1e-12:
            best_cmax = max_y
            best_a = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 500:
                # Reset momentum, try fresh step
                step = initial_step * 0.5
                no_improve_count = 0

        # Determine active peaks
        gap = max_y - np.sort(c_y)[-2] if len(c_y) > 1 else 0
        if focus == 'single' or (focus == 'auto' and gap > max_y * 0.01):
            k_peak = np.argmax(c_y)
            g = subgradient_single_peak(y, k_peak)
        else:
            threshold = max_y - max(max_y * 1e-6, 1e-10)
            active = np.where(c_y >= threshold)[0]
            if len(active) == 1:
                g = subgradient_single_peak(y, active[0])
            else:
                g = subgradient_multi_peak(y, active)

        # Adaptive step size
        step = initial_step / (1.0 + 0.005 * it)
        # Occasional larger steps to escape plateaus
        if it > 0 and it % 500 == 0:
            step *= 3.0

        # Nesterov update
        x_new = y - step * g
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-30:
            x_new = y.copy()
        else:
            x_new /= s

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + (t - 1.0) / t_new * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        s2 = y_new.sum()
        if s2 < 1e-30:
            y_new = x_new.copy()
        else:
            y_new /= s2

        x = x_new
        y = y_new
        t = t_new

    return best_a, best_cmax


def lbfgs_phase(x, tau_list, max_time=120):
    """Run L-BFGS with a schedule of decreasing tau values."""
    n = len(x)
    best_a = softmax_vec(x).copy()
    best_score = compute_score(best_a)
    deadline = time.monotonic() + max_time

    for tau in tau_list:
        if time.monotonic() > deadline:
            break

        def obj(x_):
            v, _ = lse_obj_and_grad(x_, tau)
            return v

        def grad(x_):
            _, g = lse_obj_and_grad(x_, tau)
            return g

        try:
            result = minimize(
                obj, x, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-13,
                         'maxfun': 600, 'maxls': 40}
            )
            x = result.x
        except Exception:
            break

        a_curr = softmax_vec(x)
        score = compute_score(a_curr)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_curr.copy()

    return best_a, best_score, x


def peak_shave(a, n_peaks=8, strength=0.003):
    """Reduce values contributing most to the top peaks of convolution."""
    n = len(a)
    c = np.convolve(a, a)
    top = np.argsort(c)[::-1][:n_peaks]

    a_new = a.copy()
    for k in top:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                # Reduce a[i] and a[j] proportionally
                if a_new[i] > 1e-15:
                    a_new[i] *= (1.0 - strength)
                if j != i and a_new[j] > 1e-15:
                    a_new[j] *= (1.0 - strength * 0.5)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def coordinate_descent_polish(a, n_steps=100, time_limit=30):
    """Try small perturbations of individual elements to reduce max conv."""
    n = len(a)
    best_a = a.copy()
    best_cmax = np.convolve(a, a).max()
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() - start > time_limit:
            break
        # Pick a random element to perturb
        i = rng.integers(0, n)
        for delta in [0.5, -0.5, 0.1, -0.1]:
            a_try = best_a.copy()
            a_try[i] = max(0, a_try[i] * (1.0 + delta * 0.01))
            s = a_try.sum()
            if s > 1e-15:
                a_try /= s
            cmax = np.convolve(a_try, a_try).max()
            if cmax < best_cmax:
                best_cmax = cmax
                best_a = a_try.copy()

    return best_a
