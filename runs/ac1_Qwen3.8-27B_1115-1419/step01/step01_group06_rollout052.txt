import numpy as np
import time

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / ind > 0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    i = np.where(cond)[0][-1]
    tau = (cssv[i] - 1.0) / (i + 1)
    return np.maximum(v - tau, 0.0)

def compute_conv(a, N):
    """Compute linear self-convolution via FFT."""
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2 * len(a) - 1]
    c = np.maximum(c, 0.0)  # numerical safety
    return c

def grad_smooth_max(a, c, beta, N):
    """
    Compute gradient of smoothed max of conv(a,a) w.r.t. a.
    Returns grad of (1/beta) * log(sum exp(beta * c_k))
    which is sum_k p_k * grad_c_k where p = softmax(beta*c).
    """
    n = len(a)
    # Softmax weights
    c_max = c.max()
    if c_max <= 0:
        return np.zeros(n)
    logits = beta * (c - c_max) / c_max  # normalized for stability
    logits = np.clip(logits, -700, 0)
    exp_logits = np.exp(logits)

```python
import numpy as np
import time

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / ind > 0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    i = np.where(cond)[0][-1]
    tau = (cssv[i] - 1.0) / (i + 1)
    return np.maximum(v - tau, 0.0)

def compute_conv(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2 * n - 1]
    return np.maximum(c, 0.0)

def smooth_gradient(a, c, beta, N2):
    """Compute gradient of smoothed max of c = conv(a,a)."""
    n = len(a)
    c_max = c.max()
    if c_max <= 0:
        return np.zeros(n)
    # Softmax weights for the "max" approximation
    logits = beta * (c - c_max) / c_max
    logits = np.clip(logits, -700.0, 0.0)
    exp_l = np.exp(logits)
    s = exp_l.sum()
    if s < 1e-300:
        p = np.zeros(len(c))
        p[c.argmax()] = 1.0
    else:
        p = exp_l / s
    
    # Gradient = 2 * full_conv(p, reversed_a)[n-1 : 2n-1]
    rev_a = a[::-1]
    P = np.fft.rfft(p, N2)
    R = np.fft.rfft(rev_a, N2)
    conv_pr = np.fft.irfft(P * R, N2)
    grad = 2.0 * conv_pr[n-1:2*n-1]
    return grad
```
