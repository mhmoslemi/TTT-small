Minimize 2n*max(conv(a))/(sum(a))^2 using improved FISTA + LP correction.
Key improvements:
- All near-active constraints for subgradient (not just top-k)
- Periodic LP correction for global direction
- Multiple n values explored
- Better momentum management with strong steps

import numpy as np
import time
import math
from scipy.optimize import linprog


def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = u * arange > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def compute_subgradient(a, c, eps_frac, max_active=2000):
    """Compute subgradient using all near-active constraints (capped at max_active)."""
    n = len(a)
    max_c = c.max()
    threshold = max_c * (1.0 - eps_frac)
    active = np.where(c >= threshold)[0]
    
    # If too many active, use the top max_active
    if len(active) > max_active:
        # Sort by value and take top max_active
        active_vals = c[active]
        top_idx = np.argsort(active_vals)[::-1][:max_active]
        active = active[top_idx]
    
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            idx = k - np.arange(lo, hi, dtype=np.int64)
            g[lo:hi] += a[idx]
        if k % 2 == 0:
            j = k // 2
            if 0 <= j < n:
                g[j] += a[j]
    
    g /= float(len(active))
    return g


def solve_lp_correction(f, M):
    """
    Solve the LP: maximize sum(b) subject to b >= 0 and convolve(f, b) <= M.
    Returns the normalized solution b/sum(b) or None if infeasible.
    """
    n = len(f)
    
    # Build constraint matrix A such that A @ b <= M * ones(2n-1)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # This is a banded matrix
    
    # For efficiency, build A explicitly (it's a band matrix)
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # f[j] contributes to A[k, j] for k in [j, j+n-1]
        A[j:j+n, j] = f[j]
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: 0 <= b_j (no upper bound needed, but let's add a reasonable one)
    bounds = [(0, None)] * n
    
    # Solve with HiGHS
    try:
        result = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                        bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            b = result.x
            b = np.maximum(b, 0)
            sb = b.sum()
            if sb > 1e-15:
                return b / sb
    except Exception:
        pass
    
    return None
