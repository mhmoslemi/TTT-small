"""
Hybrid smooth-∞ descent with adaptive schedule for minimizing
2n * max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time

def next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def softmax_stable(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, -1e30)
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(x) / len(x)
    return e / s

def log_normalize(a):
    """Convert non-negative sequence to log-weights (centered)."""
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w -= np.max(w)
    return w

def exp_normalize(w):
    """Convert log-weights back to normalized sequence."""
    w = np.where(np.isfinite(w), w, -1e30)
    w = w - np.max(w)
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(w) / len(w)
    return e / s

def conv_fft(a, N):
    """Full linear self-convolution via FFT."""
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    n = len(a)
    return c[:2*n-1]

def score_from_conv(c, n, s=1.0):
    """Compute 2n*max(c)/s^2."""
    mx = c.max()
    return 2.0 * n * mx / (s * s)

def grad_smooth_max(c, p, a, N):
    """
    Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a.
    c: self-convolution (length 2n-1)
    p: normalized softmax weights on c (length 2n-1)
    a: sequence (length n)
    N: FFT size >= 3n-2
    """
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # conv(a, p) and cross-correlation
    conv = np.fft.irfft(A * P, N)
    xcorr = np.fft.irfft(np.conjugate(A) * P, N)
    n = len(a)
    grad = (conv + xcorr)[:n]
    return grad

def exact_score(a):
    """Exact score using np.convolve."""
    n = len(a)
    s = a.sum()
    if s <= 0:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def adam_step(w, grad_w, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update on log-weights."""
    t += 1
    m = b1 * m + (1 - b1) * grad_w
    v = b2 * v + (1 - b2) * (grad_w ** 2)
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= np.max(w_new)
    return w_new, m, v, t

def optimize_sequence(a_init, beta_start, beta_end, n_iters, N, n_grad_N, 
                       lr_start, lr_end, seed=42, time_budget=2400.0):
    """
    Main optimization loop for a single sequence.
    Returns the best sequence and its score found.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = a_init.copy()
    w = log_normalize(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    c = conv_fft(a, N)
    score = score_from_conv(c, n)
    best_a = a.copy()
    best_score = score
    
    start_time = time.monotonic()
    stall = 0
    
    for it in range(n_iters):
        # Check time
        if time.monotonic() - start_time > time_budget:
            break
        
        # Progress [0, 1]
        prog = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * prog
        lr = lr_start + (lr_end - lr_start) * prog
        
        # Compute softmax weights on convolution
        p = softmax_stable(beta * c)
        p = p / p.sum()  # ensure normalized
        
        # Compute gradient
        grad_a = grad_smooth_max(c, p, a, n_grad_N)
        
        # Convert to log-weight gradient
        grad_w = a * (grad_a - np.dot(a, grad_a))
        grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
        
        # Adam step
        w, m, v, t = adam_step(w, grad_w, m, v, t, lr)
        a_new = exp_normalize(w)
        
        # Evaluate
        c_new = conv_fft(a_new, N)
        score_new = score_from_conv(c_new, n)
        
        if np.isfinite(score_new) and score_new < score:
            a = a_new
            c = c_new
            score = score_new
            stall = 0
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            stall += 1
            # Backtrack: try smaller step
            if stall > 5:
                # Reset to best and perturb
                a = best_a.copy()
                w = log_normalize(a)
                c = conv_fft(a, N)
                score = best_score
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
                
                # Small random perturbation
                w += rng.normal(0, 0.05, n)
                w -= w.max()
                a = exp_normalize(w)

"""
Smooth-∞ descent for minimizing 2n*max(convolve(a,a))/(sum(a))^2
over non-negative sequences. Uses FFT-based gradient of the softmax
approximation to max, with adaptive beta schedule, Adam optimizer,
and periodic exploration to escape local minima.
"""

import numpy as np
import time


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _softmax_stable(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, -1e30)
    x = x - np.max(x)
    e = np.exp(x)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(x) / max(1, len(x))
    return e / s


def _log_w(a):
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w -= np.max(w)
    return w


def _from_log_w(w):
    w = np.where(np.isfinite(w), w, -1e30)
    w = w - np.max(w)
    e = np.exp(w)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(w) / len(w)
    return e / s


def _self_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    n = len(a)
    c = c[:2 * n - 1]
    if c.size:
        c = np.maximum(c, 0.0)
    return c


def _grad_smooth(c, p, a, N_grad):
    A = np.fft.rfft(a, N_grad)
    P = np.fft.rfft(p, N_grad)
    conv = np.fft.irfft(A * P, N_grad)
    xcorr = np.fft.irfft(np.conjugate(A) * P, N_grad)
    n = len(a)
    g = (conv + xcorr)[:n]
    return np.where(np.isfinite(g), g, 0.0)


def _exact_score(a):
    n = len(a)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-10:
        return float('inf')
    c = np.convolve(a, a)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx / (s * s))


def _adam_step(w, gw, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    t += 1
    m = b1 * m + (1.0 - b1) * gw
    v = b2 * v + (1.0 - b2) * (gw * gw)
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= np.max(w_new)
    return w_new, m, v, t


def _optimize_one(a_init, budget, seed):

    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = float(a.sum())
    if s <= 0:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    N_conv = _next_pow2(2 * n - 1)
    N_grad = _next_pow2(3 * n - 2)

    w = _log_w(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = _self_conv(a, N_conv)
    cur_score = float(2.0 * n * np.max(c))
    best_a = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    stall = 0
    stall_threshold = 30

    beta = 5.0
    lr = 0.03
    best_score_this_run = best_score
    it = 0

        elapsed = time.monotonic() - start_time
        if elapsed > budget:
            break

        it += 1

        # Adaptive beta: start moderate, increase over time
        beta = 5.0 + 195.0 * progress

        gw = a * (grad_a - np.dot(a, grad_a))

                v = np.zeros(n)
                t = 0

                    cur_score = best_score

    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point. Runs smooth-inf descent to minimize
    2n * max(convolve(a,a)) / (sum(a))^2.
    """
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    total_budget = min(max(total_budget, 10.0), 2395.0)

    start_time = time.monotonic()
    print(f"Total budget: {total_budget:.1f}s")

    # Load initial sequence
    a_init = _load_initial()
    if a_init is not None:
        n = len(a_init)
        print(f"Loaded initial sequence of length {n}")
        init_score = _exact_score(a_init)
        print(f"Initial exact score: {init_score:.6f}")
    else:
        n = 4096
        a_init = _make_gaussian_init(n)
        print(f"Using Gaussian init of length {n}")

    # Ensure n is reasonable
    if n < 256:
        n_target = 4096
        a_init = _make_gaussian_init(n_target)
        n = n_target

    # Phase 1: Main optimization (70% of budget)
    phase1_budget = 0.70 * total_budget
    print(f"Phase 1: main optimization, budget={phase1_budget:.1f}s")

    a_best, score_best = _optimize_one(a_init, phase1_budget, seed)
    exact_best = _exact_score(a_best)
    print(f"Phase 1 done. FFT score={score_best:.6f}, exact score={exact_best:.6f}")

    # Phase 2: Try different initializations (15% of budget)
    phase2_budget = 0.15 * total_budget
    print(f"Phase 2: alternative inits, budget={phase2_budget:.1f}s")

    inits_to_try = []
    inits_to_try.append(("cosine", _make_cosine_init(n)))
    inits_to_try.append(("narrow_gauss", _make_gaussian_init(n) * 0 + np.exp(-0.5 * ((np.arange(n) - (n-1)/2) / (n/10.0))**2)))

    # Also try a perturbed version of the best
    rng = np.random.default_rng(seed + 77)
    a_pert_init = a_best + rng.normal(0, 0.1, n)
    a_pert_init = np.maximum(a_pert_init, 0)
    a_pert_init = a_pert_init / a_pert_init.sum()
    inits_to_try.append(("perturbed_best", a_pert_init))

    per_init_budget = phase2_budget / len(inits_to_try)
    for name, ai in inits_to_try:
        if time.monotonic() - start_time > total_budget - 10.0:
            break
        a_try, sc_try = _optimize_one(ai, per_init_budget, seed + len(name))
        exact_try = _exact_score(a_try)
        print(f"  {name}: FFT={sc_try:.6f}, exact={exact_try:.6f}")
        if exact_try < exact_best:
            exact_best = exact_try
            a_best = a_try.copy()

    # Phase 3: Fine polish on the best (10% of budget)
    phase3_budget = 0.10 * total_budget
    print(f"Phase 3: fine polish, budget={phase3_budget:.1f}s")
    if phase3_budget > 5.0:
        a_polished, sc_polished = _optimize_one(a_best, phase3_budget, seed + 999)
        exact_polished = _exact_score(a_polished)
        print(f"  Polished: FFT={sc_polished:.6f}, exact={exact_polished:.6f}")
        if exact_polished < exact_best:
            exact_best = exact_polished
            a_best = a_polished.copy()

    # Phase 4: Try structural variants (remaining time)
    remaining = total_budget - (time.monotonic() - start_time)
    print(f"Phase 4: variants, remaining={remaining:.1f}s")
    if remaining > 2.0:
        a_variant, sc_variant = _try_variants(a_best, remaining, seed)
        print(f"  Best variant: exact={sc_variant:.6f}")
        if sc_variant < exact_best:
            exact_best = sc_variant
            a_best = a_variant.copy()

    # Final: one more optimization pass if time allows
    remaining = total_budget - (time.monotonic() - start_time)
    print(f"Final pass, remaining={remaining:.1f}s")
    if remaining > 10.0:
        a_final, sc_final = _optimize_one(a_best, remaining, seed + 4242)
        exact_final = _exact_score(a_final)
        print(f"  Final: FFT={sc_final:.6f}, exact={exact_final:.6f}")
        if exact_final < exact_best:
            exact_best = exact_final
            a_best = a_final.copy()

    # Output final result
    print(f"\nFinal best exact score: {exact_best:.6f}")
    print(f"Sequence length: {len(a_best)}")

    # Scale to satisfy evaluator constraints
    a_out = np.asarray(a_best, dtype=np.float64).ravel()
    a_out = np.where(np.isfinite(a_out), a_out, 0.0)
    a_out = np.maximum(a_out, 0.0)
    n_out = len(a_out)
    s_out = float(a_out.sum())
    if s_out > 0 and np.isfinite(s_out):
        # Scale so sum = sqrt(2n) (this makes the score = 2*max_c, which is what we minimized / sum^2)
        # Actually the score is 2*n*max_c/s^2, and we normalized sum=1, so score = 2*n*max_c.
        # The evaluator will compute 2*n*max_c/s^2 on whatever we return.
        # Since the score is scale-invariant, we can return any scaling.
        # Let's return it normalized to sum=1 for cleanliness, but make sure sum >= 0.01.
        a_out = a_out / s_out
        # Ensure sum >= 0.01
        if a_out.sum() < 0.01:
            a_out = a_out * (0.01 / a_out.sum())
        # Cap at 1000
        a_out = np.minimum(a_out, 1000.0)
    else:
        a_out = np.ones(n_out) * 1.0 / n_out

    return [float(x) for x in a_out]
