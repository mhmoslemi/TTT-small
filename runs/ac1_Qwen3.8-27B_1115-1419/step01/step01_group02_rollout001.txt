from scipy.fft import fft, ifft

def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    n_fft = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_fft) * fft(b, n_fft)).real[:n_out]

def _score_fast(a):
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _normalize(a, target_sum):
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _softmax_mirror(a, b, temperature):
    """Compute softmax-weighted mirror subgradient."""
    n = len(a)
    # Softmax weights (numerically stable)
    b_max = float(np.max(b))
    weights = np.exp((b - b_max) / temperature)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        weights = np.zeros(len(b))
        weights[np.argmax(b)] = 1.0
        return _hard_mirror(a, b, top_k=1)
    weights /= w_sum
    
    mirror = np.zeros(n)
    # For efficiency, only process lags with significant weight
    threshold = 1e-6 / len(b)
    active = np.where(weights > threshold)[0]
    
    for k in active:
        w = weights[k]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            idx = k - np.arange(i_lo, i_hi + 1)
            mirror[i_lo:i_hi+1] += w * a[idx]
    
    return mirror

def _hard_mirror(a, b, top_k=5):
    """Hard top-k mirror."""
    n = len(a)
    top_indices = np.argsort(b)[-top_k:]
    mirror = np.zeros(n)
    for k in top_indices:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            idx = k - np.arange(i_lo, i_hi + 1)
            mirror[i_lo:i_hi+1] += a[idx]
    return mirror / top_k

def _subgrad_step(a, lr, temperature):
    n = len(a)
    b = _fft_conv(a, a)
    S = float(np.sum(a))
    
    mirror = _softmax_mirror(a, b, temperature)
    peak = float(np.max(b))
    
    # Subgradient of max_b: mirror
    # Subgradient of score = 2n*max_b/S^2:
    # d(score)/da = 2n * (mirror/S^2 - 2*max_b/S^3) = 2n/S * (mirror/S - 2*max_b/S^2)
    # = 2n/S * (mirror/S - 2*max_b/S^2)
    # For descent: a_new = a - lr * grad
    # = a - lr * 2n/S * (mirror/S - 2*max_b/S^2)
    # = a - lr * 2n/S^2 * (mirror - 2*max_b/S)
    # = a - lr * 2n/S^2 * (mirror - 2*max_b/S)
    
    # Actually, let's just use: a_new = a - lr * (mirror - peak_mean)
    # where peak_mean is chosen to maintain the sum
    mean_mirror = float(np.sum(mirror))
    # To maintain sum: subtract the mean of the adjustment
    adjustment = mirror - 2.0 * peak / S * float(np.ones(n))
    # Actually, let's use the proper subgradient:
    # We want to reduce max_b while maintaining S.
    # Direction: -mirror (reduces peak) + uniform (maintains sum)
    # a_new = a - lr * mirror + lr * (sum(mirror)/n) * ones = a - lr * (mirror - mean(mirror)*ones)
    
    adjustment = mirror - np.mean(mirror)
    a_new = a - lr * adjustment
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, S)  # Renormalize to same sum
    
    return a_new
