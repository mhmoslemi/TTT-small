"""
Search for low-peak self-convolutions of nonnegative step sequences.

Objective (scale invariant):
    score(a) = 2 * n * max_k (a * a)_k / (sum a)^2

The search combines:
  * parametric and spectrum-flat starts,
  * log-sum-exp / active-peak multiplicative gradient descent in the log-simplex,
  * spectral-factor projection onto modified convolution spectra,
  * occasional LP "best response" direction for cross-convolution flatness,
  * multi-purpose restarts, crossovers, and length expansion.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_array, hstack, eye

except Exception:

def next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def full_convolution(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    m = 2 * n - 1
    if fft_size is None:
        fft_size = next_pow2(m)
    x = a / float(np.sum(a)) if float(np.sum(a)) > 1e-300 else a
    A = np.fft.rfft(x, fft_size)
    c = np.fft.irfft(A * A, fft_size).real[:m]
    c = np.maximum(c, 0.0)
    return c


def score_normalized(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return 1e100
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if s <= 1e-300:
        return 1e100
    if fft_size is None:
        fft_size = next_pow2(2 * n - 1)
    c = full_convolution(a / s, fft_size)
    mb = float(np.max(c))
    if not np.isfinite(mb):
        return 1e100
    return float(2.0 * n * mb)


def score_array(a, fft_size=None):
    return score_normalized(a, fft_size)


def softmax_stable(theta):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    n = theta.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    m = float(np.max(theta))
    z = np.maximum(theta - m, -745.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def init_opt_state(a):
    theta = np.log(np.maximum(np.asarray(a, dtype=np.float64), 1e-300))
    theta -= float(theta.mean())
    v = np.zeros_like(theta)
    base_k = 60.0
    alpha = max(8.0, 0.35 * a.size)

def blend_arith(a, b, w):

def resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    s = float(old.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, old[0], dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    out = np.interp(x_new, x_old, old)
    return normalize_sequence(out)


def lse_weights(c, k):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = c.size
    if m == 0:
        return np.zeros(0, dtype=np.float64)
    maxc = float(np.max(c))
    if maxc <= 1e-300:
        return np.full(m, 1.0 / m, dtype=np.float64)
    beta = min(1.0e6, float(k) / maxc)
    if beta <= 1.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    arg = beta * (c - maxc)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        p = np.zeros(m, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def weighted_gradient(a, p, fft_size, n=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    if n is None:
        n = a.size
    m = 2 * n - 1
    if p.size != m:
        p = np.resize(p, m)
    ps = float(p.sum())
    if ps <= 1e-300:
        p = np.full(m, 1.0 / m, dtype=np.float64)
    else:
        p = p / ps
    q = p[::-1]
    if fft_size is None:
        fft_size = next_pow2(3 * n - 2)
    A = np.fft.rfft(a, fft_size)
    Q = np.fft.rfft(q, fft_size)
    conv = np.fft.irfft(A * Q, fft_size).real
    seg = conv[n - 1:2 * n - 1]
    return 2.0 * seg[::-1]


def project_box_sum(c, t):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = c.size
    if m == 0:
        return np.zeros(0, dtype=np.float64)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s <= 1e-300:
        return np.full(m, 1.0 / m, dtype=np.float64)
    c = c / s
    t = float(t)
    avg = 1.0 / m
    if t < avg:
        t = avg
    if float(np.max(c)) <= t * (1.0 + 1e-12):
        return c.copy()

    hi = max(float(t - float(np.min(c))), 1e-12)
    # Ensure hi is high enough.
    for _ in range(10):
        shi = float(np.sum(np.clip(c + hi, 0.0, t)))
        if shi >= 1.0:
            break
        hi *= 2.0

    lo = 0.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        smid = float(np.sum(np.clip(c + mid, 0.0, t)))
        if smid < 1.0:
            lo = mid
        else:
            hi = mid

    d = np.clip(c + lo, 0.0, t)
    sd = float(d.sum())
    if sd <= 1e-300:
        d = np.clip(c + hi, 0.0, t)
        sd = float(d.sum())
    if sd > 0.0:
        d /= sd
    else:
        d = np.full(m, avg, dtype=np.float64)
    return d


def spectral_factor_from_current(a, d, fft_size):
    a = np.asarray(a, dtype=np.float64).ravel()
    d = np.asarray(d, dtype=np.float64).ravel()
    n = a.size
    if n < 2 or d.size != 2 * n - 1:
        return None
    if fft_size is None:
        fft_size = next_pow2(2 * n - 1)

    A = np.fft.rfft(a, fft_size)
    Df = np.fft.rfft(d, fft_size)

    angA = np.angle(A)
    angD = np.angle(Df)
    twoA = 2.0 * angA
    k = np.round((twoA - angD) / (2.0 * np.pi))
    angD_adj = angD + 2.0 * np.pi * k
    ang_new = 0.5 * angD_adj
    mag_new = np.sqrt(np.maximum(np.abs(Df), 0.0))
    newA = mag_new * np.exp(1j * ang_new)
    newA[0] = float(np.real(newA[0]))
    if newA.size > 1:
        newA[-1] = float(np.real(newA[-1]))

    a_full = np.fft.irfft(newA, fft_size).real

    best = None
    best_quality = -1e300
    for lag in range(-3, 4):
        cand = np.roll(a_full, -lag)[:n]
        pos = np.maximum(cand, 0.0)
        neg = np.maximum(-cand, 0.0)
        pos_sum = float(pos.sum())
        neg_sum = float(neg.sum())
        if pos_sum <= 1e-12:
            continue
        quality = pos_sum - 0.25 * neg_sum
        if quality > best_quality:
            best_quality = quality
            best = pos
    if best is None or best_quality <= 1e-10:
        return None
    bs = float(best.sum())
    if bs <= 1e-300:
        return None
    return best / bs


def spectral_flat_start(n, target_ratio, rng=None):
    if n < 4:
        return None
    m = 2 * n - 1
    a = np.ones(n, dtype=np.float64) / float(n)
    fft = next_pow2(m)
    c = full_convolution(a, fft)
    s = float(c.sum())
    if s <= 0.0:
        return None
    c = c / s
    t = max(1.001 / m, float(target_ratio) / (2.0 * n))
    d = project_box_sum(c, t)
    flat = np.full(m, 1.0 / m, dtype=np.float64)
    d = 0.98 * d + 0.02 * flat
    ds = float(d.sum())
    if ds <= 0.0:
        return None
    d = d / ds
    return spectral_factor_from_current(a, d, fft)


def lp_best_response(f):
    if not _HAS_SCIPY:
        return None
    f = normalize_sequence(f)
    n = f.size
    if n < 4 or n > 768:
        return None
    m = 2 * n - 1

    col_idx = np.repeat(idx, n)
    vals = np.repeat(np.asarray(f, dtype=np.float64), n)

    try:
        A = coo_array((vals, (row_idx, col_idx)), shape=(m, n)).tocsr()
        I = eye(m, format="csc")

        res = linprog(
            c_obj,

def make_inits(n, rng, given, pool, num):
    n = max(int(n), 2)
    num = max(int(num), 1)
    cands = []

    for item in pool[:3]:
        cands.append(resample_sequence(item[1], n))
    if given is not None:
        cands.append(resample_sequence(given, n))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    eps = 1e-8
    xx = x + eps
    yy = 1.0 - x + eps

    cands.append(np.ones(n, dtype=np.float64))

    for alpha in (0.5, 1.0, 2.0, 3.0, 5.0):
        cands.append(np.exp(-alpha * x))
        cands.append(np.exp(-alpha * x) * yy)
        cands.append(np.exp(-alpha * x) * yy * yy)
        cands.append(np.exp(-alpha * yy))
        cands.append(np.exp(-alpha * yy) * xx)

    for p in (-0.30, -0.15, 0.0, 0.20, 0.50, 0.80):
        for q in (0.0, 0.5, 1.0):
            cands.append(xx ** p * yy ** q)

    cands.append(np.sin(np.pi * x) ** 2)
    cands.append(np.sin(np.pi * x) ** 4)
    cands.append(1.0 - 0.65 * np.cos(2.0 * np.pi * x))

    base = np.ones(n, dtype=np.float64)
    if pool:
        base = resample_sequence(pool[0][1], n)
    cands.append(base * (1.0 + 0.30 * np.fft.irfft(np.fft.rfft(rng.normal(size=n))).real))
    cands.append(base * (1.0 + 0.55 * np.fft.irfft(np.fft.rfft(rng.normal(size=n))).real))
    if given is not None:
        gb = resample_sequence(given, n)
        cands.append(gb * (1.0 + 0.35 * np.fft.irfft(np.fft.rfft(rng.normal(size=n))).real))

    for ratio in (1.18, 1.32, 1.45, 1.60):
        s = spectral_flat_start(n, ratio, rng)
        if s is not None:
            cands.append(s)

    scored = []
    for c in cands:
        c = normalize_sequence(c)
        sc = score_array(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=_scored_key)
    if len(scored) == 0:
        return [np.ones(n, dtype=np.float64) / float(n)]
    return [item[1] for item in scored[:num]]


def spectral_refine(a, end_time, rng, target_min=1.12):
    a = normalize_sequence(a)
    n = a.size
    if n < 8:
        return a, score_array(a)

    b_fft = next_pow2(2 * n - 1)
    m = 2 * n - 1
    avg = 1.0 / m

    best_a = a.copy()
    best_score = score_normalized(a, b_fft)
    current = a.copy()
    score = best_score

    descent_factor = 0.985
    target_score = max(target_min, min(best_score * 0.99, 1.65))
    fail = 0

    while time.monotonic() < end_time:
        c = full_convolution(current, b_fft)
        cs = c.copy()
        ss = float(cs.sum())
        if ss > 0.0:
            cs = cs / ss
        maxc = float(np.max(c))
        score = float(2.0 * n * maxc)
        if not np.isfinite(score):
            current = best_a.copy()
            continue

        if score < best_score - 1e-13:
            best_a = current.copy()
            best_score = score
            fail = 0
        else:
            fail += 1

        if fail > 5:
            desired = max(target_min, min(score * 0.998, best_score * 0.999))
        else:
            desired = max(target_min, min(target_score * descent_factor, score * 0.999))
        if desired > score * 0.999:
            desired = score * 0.995

        t = desired / (2.0 * n)
        if t < avg * 1.0005:
            t = avg * 1.0005

        d = project_box_sum(cs, t)
        flat = np.full(m, avg, dtype=np.float64)
        d = 0.985 * d + 0.015 * flat
        ds = float(d.sum())
        if ds > 0.0:
            d = d / ds

        a_new = spectral_factor_from_current(current, d, b_fft)
        improved = False

        if a_new is not None:
            for w in (0.35, 0.70, 1.00):
                cand = blend_arith(current, a_new, w)
                sc = score_normalized(cand, b_fft)
                if sc < score - 1e-14:
                    current = cand
                    score = sc
                    improved = True
                    break
                cand2 = blend_log(current, a_new, w)
                sc2 = score_normalized(cand2, b_fft)
                if sc2 < score - 1e-14:
                    current = cand2
                    score = sc2
                    improved = True
                    break

        if improved:
            target_score = max(target_min, min(score * descent_factor, best_score * 0.995))
            fail = 0
        else:
            if target_score < score:
                target_score = min(score * 0.999, max(target_score * 1.02, target_min))
            if fail > 18 and time.monotonic() < end_time - 5.0:
                sc = 0.25 if n < 2048 else 0.15
                current = log_perturb(best_a, rng, sc)
                fail = 0
                target_score = max(target_min, best_score * 0.99)

        if fail > 60:
            break

    return best_a, best_score


def gradient_refine(a, end_time, rng, use_lp=True, use_spec=True):
    a = normalize_sequence(a)
    n = a.size
    if n < 8:
        return a, score_array(a)

    b_fft = next_pow2(2 * n - 1)
    g_fft = next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = score_normalized(a, b_fft)
    current = a.copy()
    score = best_score

    theta, v, base_k, alpha = init_opt_state(current)
    alpha_max = max(50.0, 3.0 * n)
    alpha_min = 0.05
    momentum = 0.55
    it = 0
    fail = 0
    no_best = 0
    lp_calls = 0
    last_mutation = 0
    target_min = 1.12

    while time.monotonic() < end_time:
        c = full_convolution(current, b_fft)
        maxc = float(np.max(c))
        score = float(2.0 * n * maxc)
        if not np.isfinite(score):
            current = best_a.copy()
            score = best_score
            theta, v, base_k, alpha = init_opt_state(current)
            fail = 0
            no_best = 0
            it += 1
            continue

        if score < best_score - 1e-13:
            best_a = current.copy()
            best_score = score
            no_best = 0
        else:
            no_best += 1

        if no_best > 1200 and it > 300:
            sc = 0.30 if n < 1024 else 0.20
            current = log_perturb(best_a, rng, sc)
            score = score_normalized(current, b_fft)
            theta, v, base_k, alpha = init_opt_state(current)
            fail = 0
            no_best = 0
            it += 1
            continue

        if it % 100 == 50:
            sym = normalize_sequence(0.5 * (current + current[::-1]))
            sc = score_normalized(sym, b_fft)
            if sc < best_score - 1e-13:
                best_a = sym
                best_score = sc
                current = sym
                score = sc
                theta, v, base_k, alpha = init_opt_state(current)
                no_best = 0
            elif sc < score - 1e-13:
                current = sym
                score = sc
                theta, v, base_k, alpha = init_opt_state(current)

        p = lse_weights(c, base_k)
        old_obj = float(np.dot(p, c))
        grad = weighted_gradient(current, p, g_fft, n)
        wmean = float(np.dot(current, grad))
        grad_c = grad - wmean
        gvmax = float(np.max(np.abs(grad_c)))

        if gvmax <= 1e-18:
            fail += 1
            if fail > 10:
                current = log_perturb(current, rng, 0.12)
                theta, v, base_k, alpha = init_opt_state(current)
                fail = 0
            it += 1
            continue

        v = momentum * v + (1.0 - momentum) * grad_c
        vmax = float(np.max(np.abs(v)))
        if vmax > 1e-30:
            step = alpha * vmax
            if step > 1.5:
                v = v * (1.5 / step)

        accepted = False
        for _ in range(5):
            theta_try = theta - alpha * v
            a_try = softmax_stable(theta_try)
            c_try = full_convolution(a_try, b_fft)
            max_try = float(np.max(c_try))
            exact_try = float(2.0 * n * max_try)
            obj_try = float(np.dot(p, c_try))
            if np.isfinite(exact_try) and np.isfinite(obj_try):
                tol = 1e-14 * max(1.0, abs(old_obj))
                if obj_try < old_obj - tol or (exact_try < score - 1e-12 and obj_try <= old_obj + 10.0 * tol):
                    current = a_try
                    theta = theta_try
                    score = exact_try
                    accepted = True
                    break
            alpha = max(alpha_min, alpha * 0.5)

        if accepted:
            alpha = min(alpha_max, alpha * 1.08)
            v *= 0.5
            fail = 0
        else:
            fail += 1
            alpha = max(alpha_min, alpha * 0.3)
            v *= 0.5
            if fail > 20:
                current = log_perturb(current, rng, 0.12)
                theta, v, base_k, alpha = init_opt_state(current)
                fail = 0

        if no_best > 300:
            base_k = min(5000.0, base_k * 1.05 + 10.0)
        elif no_best < 50:
            base_k = max(35.0, base_k * 0.97)

        if use_spec and it - last_mutation > 220 and time.monotonic() < end_time - 25:
            c_norm = c.copy()
            ssum = float(c_norm.sum())
            if ssum > 0.0:
                c_norm = c_norm / ssum
            desired = max(target_min, min(score * 0.99, best_score * 0.998))
            t = desired / (2.0 * n)
            if t < (1.0 / (2 * n - 1)) * 1.0005:
                t = (1.0 / (2 * n - 1)) * 1.0005
            d = project_box_sum(c_norm, t)
            a_new = spectral_factor_from_current(current, d, b_fft)
            if a_new is not None:
                for w in (0.35, 0.70, 1.00):
                    cand = blend_arith(current, a_new, w)
                    sc = score_normalized(cand, b_fft)
                    if sc < score - 1e-13:
                        current = cand
                        score = sc
                        theta, v, base_k, alpha = init_opt_state(current)
                        fail = 0
                        no_best = 0
                        break
            last_mutation = it

        if (
            use_lp
            and n <= 768
            and lp_calls < 2
            and it - last_mutation > 260
            and time.monotonic() < end_time - 50
        ):
            g = lp_best_response(current)
            lp_calls += 1
            last_mutation = it
            if g is not None:
                for w in (0.25, 0.50, 0.75, 1.00):
                    cand = blend_arith(current, g, w)
                    sc = score_normalized(cand, b_fft)
                    if sc < score - 1e-13:
                        current = cand
                        score = sc
                        theta, v, base_k, alpha = init_opt_state(current)
                        fail = 0
                        no_best = 0
                        break
                    cand2 = blend_log(current, g, w)
                    sc2 = score_normalized(cand2, b_fft)
                    if sc2 < score - 1e-13:
                        current = cand2
                        score = sc2
                        theta, v, base_k, alpha = init_opt_state(current)
                        fail = 0
                        no_best = 0
                        break
            it += 1
            continue

        it += 1

    return best_a, best_score


def combined_refine(a, end_time, rng, use_spec=True, use_lp=True):
    a = normalize_sequence(a)
    sc = score_array(a)
    if not np.isfinite(sc):
        return a, sc
    t_start = time.monotonic()
    total = max(0.0, float(end_time) - t_start)
    best_a = a
    best_score = sc

    if use_spec and total > 12.0:
        if total > 60.0:
            spec_time = 0.45 * total
        elif total > 20.0:
            spec_time = 0.40 * total
        else:
            spec_time = 0.30 * total
        spec_end = min(float(end_time) - 4.0, t_start + max(5.0, spec_time))
        if time.monotonic() < spec_end:
            sa, ss = spectral_refine(a, spec_end, rng)
            if ss < best_score:
                best_a = sa
                best_score = ss
                a = sa

    if time.monotonic() < float(end_time) - 1.0:
        ga, gs = gradient_refine(a, float(end_time), rng, use_lp=use_lp, use_spec=use_spec)
        if gs < best_score:
            best_a = ga
            best_score = gs

    return best_a, best_score


def add_to_pool(pool, a, max_size=24):
    a = normalize_sequence(a)
    sc = score_array(a)
    if not np.isfinite(sc):
        return
    pool.append((sc, a.copy(), int(a.size)))
    pool.sort(key=_pool_key)
    while len(pool) > max_size:
        pool.pop()


def get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0:
        return None
    s = float(arr.sum())
    if s <= 1e-300:
        return None
    url = arr / s
    if arr.size > 20000:
        url = resample_sequence(url, 4096)
    return url


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0

    t0 = time.monotonic()
    hard_leeway = max(10.0, 0.015 * budget)
    deadline = t0 + max(60.0, budget) - hard_leeway
    final_reserve = max(120.0, 0.10 * budget)

    rng = np.random.default_rng(seed)

    given = get_given_sequence()

    pool = []
    best_a = np.ones(256, dtype=np.float64) / 256.0
    best_score = score_array(best_a)

    if given is not None:
        gs = score_array(given)
        print(f"given length={given.size} local_score={gs:.8f}", flush=True)
        if np.isfinite(gs) and gs < best_score:
            best_a = given.copy()
            best_score = gs
    add_to_pool(pool, best_a)

    # Quick useful starts.
    for n in (256, 512):
        for init in make_inits(n, rng, given, pool, 2):
            add_to_pool(pool, init)
    best_score = float(pool[0][0])
    best_a = pool[0][1].copy()

    plan = [
        (64, 2, 15, True, False),
        (128, 3, 25, True, False),
        (256, 4, 45, True, False),
        (512, 3, 70, True, True),
        (1024, 3, 110, True, True),
        (2048, 2, 150, True, False),
        (4096, 2, 180, True, False),
        (8192, 1, 180, True, False),
    ]

    scale = max(0.10, min(1.5, budget / 2400.0))

    for n, num, sec, use_spec, use_lp in plan:
        sec = max(8, int(sec * scale))
        if time.monotonic() > deadline - final_reserve - sec:
            continue

        num_use = max(1, min(num, int(num * scale) + 1))
        inits = make_inits(int(n), rng, given, pool, num_use)

        for idx, init in enumerate(inits):
            if time.monotonic() > deadline - final_reserve - sec:
                break
            t_end = min(deadline - final_reserve, time.monotonic() + sec)
            if time.monotonic() >= t_end:
                continue
            res_a, res_score = combined_refine(init, t_end, rng, use_spec=use_spec, use_lp=use_lp)
            if np.isfinite(res_score):
                print(
                    f"n={res_a.size} start={idx}/{num_use} score={res_score:.8f} "
                    f"best={best_score:.8f} t={time.monotonic() - t0:.1f}s",
                    flush=True,
                )
            if np.isfinite(res_score) and res_score < best_score - 1e-15:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
            add_to_pool(pool, res_a)

    # Crossover phase from the strongest pool members.
    if len(pool) >= 2 and time.monotonic() < deadline - max(120.0, 0.07 * budget):
        cross_n = int(best_a.size)
        if cross_n < 512:
            cross_n = 512
        if cross_n > 8192:
            cross_n = 8192

        pairs = []
        if len(pool) >= 2:
            pairs.append((0, 1))
        if len(pool) >= 3:
            pairs.append((0, 2))
        if len(pool) >= 4:
            pairs.append((1, 3))

        for i, j in pairs[:2]:
            if time.monotonic() > deadline - max(60.0, 0.05 * budget):
                break
            a1 = resample_sequence(pool[i][1], cross_n)
            a2 = resample_sequence(pool[j][1], cross_n)
            candidates = [blend_arith(a1, a2, 0.5), blend_log(a1, a2, 0.5)]
            for cand in candidates:
                if time.monotonic() > deadline - max(45.0, 0.04 * budget):
                    break
                sec = max(25, int(70 * scale))
                t_end = min(deadline - max(30.0, 0.03 * budget), time.monotonic() + sec)
                if time.monotonic() >= t_end:
                    continue
                res_a, res_score = combined_refine(cand, t_end, rng, use_spec=True, use_lp=(cross_n <= 768))
                if np.isfinite(res_score) and res_score < best_score - 1e-15:
                    best_a = np.asarray(res_a, dtype=np.float64).copy()
                    best_score = float(res_score)
                    print(f"crossover best={best_score:.8f} len={best_a.size}", flush=True)
                add_to_pool(pool, res_a)

    # Final deep refinement.
    if time.monotonic() < deadline - 5.0:
        t_end = min(deadline - 1.0, time.monotonic() + max(60.0, deadline - time.monotonic()))
        res_a, res_score = combined_refine(
            best_a,
            t_end,
            rng,
            use_spec=True,
            use_lp=(best_a.size <= 768),
        )
        if np.isfinite(res_score) and res_score < best_score - 1e-15:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)
            print(f"final refine best={best_score:.8f} len={best_a.size}", flush=True)
        else:
            print(f"final no improvement best={best_score:.8f} len={best_a.size}", flush=True)

    out = normalize_sequence(best_a)
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        out = np.ones(256, dtype=np.float64) / 256.0

    ans = []
    for v in out:
        ans.append(float(v))

    # Optional validation print if the evaluator is present.
    try:
        ev = evaluate_sequence(ans)  # noqa: F821
        print(f"evaluator score={ev:.8f}", flush=True)
    except Exception:
        pass

    return ans
