"""
Mirror descent with weighted multi-peak subgradient for minimizing
2n*max(autocorr)/(sum)^2 over non-negative sequences.
Uses multiplicative (exponential) updates that naturally preserve
non-negativity, with adaptive step size and stuck detection.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """Fast linear convolution via FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score(a):
    """Compute 2n*max(a*a)/(sum a)^2 using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _norm_to(a, target):
    """Scale a to have sum = target, preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full(len(a), target / len(a))
    return a * (target / s)


def _subgrad_peaks(a, thresh=0.95, cap=100):
    """
    Compute a subgradient of max(a*a) using weighted mirrors at peak positions.
    Returns (gradient vector, autocorrelation array, max value).
    """
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    
    # Find peaks above threshold
    idx = np.where(b >= thresh * max_b)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(b)])
    if len(idx) > cap:
        # Keep top 'cap' peaks
        vals = b[idx]
        cutoff = np.partition(vals, len(vals) - cap)[len(vals) - cap]
        idx = idx[vals >= cutoff]
    
    # Build weighted subgradient
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        w = float(b[k])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            g[lo:hi+1] += w * a[k - np.arange(lo, hi + 1)]
    
    total = float(np.sum(b[idx]))
    if total > 0:
        g /= total
    return g, b, max_b


def _mirror_update(a, g, eta):
    """
    Multiplicative (mirror descent) update.
    a_new_j prop to a_j * exp(-eta * g_j)
    Then renormalize to preserve sum.
    """
    # Shift gradient to avoid overflow
    g_shift = g - np.max(g)
    # Work in log space for stability
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * g_shift
    log_a_new = np.clip(log_a_new, -700, 700)
    a_new = np.exp(log_a_new)
    a_new = np.maximum(a_new, 0.0)
    
    # Renormalize
    s_old = float(np.sum(a))
    s_new = float(np.sum(a_new))
    if s_new < 1e-15:
        return a.copy()
    return a_new * (s_old / s_new)


def _grad_update(a, g, eta):
    """
    Additive (gradient descent) update with projection.
    """
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    s_old = float(np.sum(a))
    a_new = _norm_to(a_new, s_old)
    return a_new


def _optimize_mirror(a, deadline, eta0=0.05, thresh=0.95, cap=100,
                     eval_every=15, stuck_patience=60):
    """
    Main optimization loop using mirror descent.
    Returns (best_sequence, best_score).
    """
    best = a.copy()
    best_sc = _score(a)
    eta = eta0
    stuck = 0
    step = 0
    last_eta_reduce = 0
    
    while time.monotonic() < deadline:
        g, b, _ = _subgrad_peaks(a, thresh, cap)
        a = _mirror_update(a, g, eta)
        step += 1
        
        if step % eval_every == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = a.copy()
                stuck = 0
                # Slightly increase eta on progress
                if step - last_eta_reduce > eval_every * 3:
                    eta = min(eta * 1.02, 0.5)
            else:
                stuck += 1
                if stuck == 1:
                    last_eta_reduce = step
                if stuck % 5 == 0:
                    eta = max(eta * 0.9, 0.005)
            
            if stuck >= stuck_patience:
                # Restart with perturbation
                noise = 0.03 * np.random.standard_normal(len(a))
                a = best * (1.0 + noise)
                a = np.maximum(a, 0.0)
                a = _norm_to(a, np.sum(best))
                eta = max(eta0 * 0.5, eta * 2.0)
                stuck = 0
    
    return best, best_sc
