    # Find the largest rho such that 1 + u[rho]/rho > 0

Multi-strategy optimization for low-peak nonnegative self-convolutions.

def _next_pow2(size):

def _conv_fft(a, fft_size):
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    n = len(a)
    return b[:2 * n - 1]


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return np.array([])
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)

    idx = np.arange(1, n + 1, dtype=np.float64)

        return m, p

    val = m + float(np.log(s)) / beta
    return val, p

        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if arr.size == 0 or s < 1e-300:
        return None
    return arr / s


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Search for low-peak nonnegative self-convolution sequences.
    Combines FISTA with LSE smoothing, POCS, and coordinate descent.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + max(budget, 60.0) - 30.0
    rng = np.random.default_rng(seed)
    given = _get_given()

    # Initial best
    best_a = None
    best_score = 1e100

    # Try multiple lengths with diverse strategies
    lengths = [128, 256, 512, 1024, 2048]

    # Adjust lengths based on budget
    if budget < 600:
        lengths = [256, 512]
    elif budget < 1200:
        lengths = [256, 512, 1024]

    # Phase 1: Quick evaluation of parametric families
    print("Phase 1: Parametric families", flush=True)
    t0 = time.monotonic()

    for n in lengths:
        inits = [
            _make_uniform(n),
            _make_exponential(n, 0.8),
            _make_exponential(n, 1.2),
            _make_gaussian(n, 0.2),
            _make_gaussian(n, 0.35),
            _make_raised_cosine(n),
            _make_triangular(n),
            _make_two_bump(n, 0.3),
            _make_fourier(n, 4),
        ]
        for init in inits:
            sc = _score_of(init)
            if sc < best_score:
                best_score = sc
                best_a = init.copy()
                print(f"  n={n} score={sc:.6f}", flush=True)

    if given is not None:
        for n in lengths:
            g = _resample(given, n)
            sc = _score_of(g)
            if sc < best_score:
                best_score = sc
                best_a = g.copy()
                print(f"  given resampled n={n} score={sc:.6f}", flush=True)

    print(f"Phase 1 done: best={best_score:.6f} t={time.monotonic()-t0:.1f}s", flush=True)

    # Phase 2: FISTA optimization from best starts
    print("Phase 2: FISTA optimization", flush=True)

    # Select top initializations for deeper optimization
    n_best = best_a.size
    start_seqs = []

    # Best found so far
    start_seqs.append(best_a.copy())

    # Resample best to different lengths
    for n in lengths:
        start_seqs.append(_resample(best_a, n))

    # Given sequence resampled
    if given is not None:
        for n in [256, 512, 1024]:
            start_seqs.append(_resample(given, n))

    # Add diverse parametric starts at best length
    start_seqs.append(_make_gaussian(n_best, 0.25))
    start_seqs.append(_make_raised_cosine(n_best))
    start_seqs.append(_make_fourier(n_best, 5))

    # Deduplicate by size and rough signature
    seen = set()
    unique_starts = []
    for s in start_seqs:
        key = (s.size, round(float(s[:16].sum()), 6))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)

    # Allocate time per start
    n_starts = min(len(unique_starts), 8)
    time_per_start = max(10.0, (deadline - time.monotonic() - 120.0) / n_starts)

    for idx in range(n_starts):
        if time.monotonic() > deadline - 60:
            break
        init = unique_starts[idx].copy()
        t_end = min(deadline - 30, time.monotonic() + time_per_start)
        if time.monotonic() >= t_end:
            continue

        res_a, res_score = _fista_optimize(init, t_end, rng)
        print(f"  FISTA n={res_a.size} idx={idx} score={res_score:.6f} best={best_score:.6f}", flush=True)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()

    # Phase 3: POCS from best
    print("Phase 3: POCS optimization", flush=True)
    if time.monotonic() < deadline - 60:
        t_pocs = min(deadline - 30, time.monotonic() + 120.0)
        res_a, res_score = _pocs_optimize(best_a.copy(), t_pocs, rng)
        print(f"  POCS n={res_a.size} score={res_score:.6f} best={best_score:.6f}", flush=True)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()

        # POCS from a different perspective: start from flat spectrum
        if time.monotonic() < deadline - 60:
            n_pocs = best_a.size
            flat_start = _make_raised_cosine(n_pocs)
            res_a, res_score = _pocs_optimize(flat_start, min(deadline - 30, time.monotonic() + 60.0), rng)
            if res_score < best_score:
                best_score = res_score
                best_a = res_a.copy()
                print(f"  POCS2 n={res_a.size} score={res_score:.6f}", flush=True)

    # Phase 4: Try larger versions of best
    print("Phase 4: Upscaling", flush=True)
    if time.monotonic() < deadline - 120:
        n_cur = best_a.size
        for target_n in [512, 1024, 2048]:
            if target_n <= n_cur:
                continue
            if time.monotonic() > deadline - 60:
                break
            init = _resample(best_a, target_n)
            # Mix with parametric to avoid degenerate resampling
            init = _project_simplex(0.8 * init + 0.2 * _make_gaussian(target_n, 0.3))
            t_end = min(deadline - 30, time.monotonic() + 100.0)
            res_a, res_score = _fista_optimize(init, t_end, rng)
            print(f"  Upscale {n_cur}->{res_a.size} score={res_score:.6f} best={best_score:.6f}", flush=True)
            if res_score < best_score:
                best_score = res_score
                best_a = res_a.copy()
                n_cur = res_a.size

    # Phase 5: More FISTA runs with different hyperparameters
    print("Phase 5: Additional FISTA runs", flush=True)
    if time.monotonic() < deadline - 30:
        # Run from best with different beta schedules
        for beta_s, beta_e in [(12, 300), (20, 500), (5, 200)]:
            if time.monotonic() > deadline - 15:
                break
            t_end = min(deadline - 5, time.monotonic() + 60.0)
            if time.monotonic() >= t_end:
                break
            init = best_a.copy()
            # Small perturbation
            init = _project_simplex(init + rng.normal(0, 0.05 / np.sqrt(init.size), init.size))
            res_a, res_score = _fista_optimize(init, t_end, rng, beta_start=beta_s, beta_end=beta_e)
            if res_score < best_score:
                best_score = res_score
                best_a = res_a.copy()
                print(f"  FISTA bs={beta_s} be={beta_e} score={res_score:.6f}", flush=True)

    # Phase 6: Coordinate descent polish
    print("Phase 6: Coordinate descent polish", flush=True)
    if time.monotonic() < deadline - 5:
        t_cd = deadline - 3.0
        res_a, res_score = _coord_descent(best_a.copy(), t_cd, rng)
        print(f"  CD n={res_a.size} score={res_score:.6f} best={best_score:.6f}", flush=True)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()

    # Final round of FISTA with remaining time
    if time.monotonic() < deadline - 5:
        t_final = deadline - 2.0
        res_a, res_score = _fista_optimize(best_a.copy(), t_final, rng, beta_start=20.0, beta_end=600.0)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()
            print(f"  Final FISTA score={res_score:.6f}", flush=True)

    print(f"\nFinal: n={best_a.size} score={best_score:.8f}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 1e-300:
        out = out / s
    else:
        out = _make_uniform(256)

    return [float(x) for x in out]
