```
def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = a.astype(np.float64, copy=True)
    a[~np.isfinite(a)] = 0.0
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / s
```

```
def _alpha_shape(n, gamma=0.5):
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** (-gamma)
    return _norm(a)
def _exp_shape(n, rate=1.5):
    x = np.arange(n, dtype=np.float64) / max(1, n-1)
    a = np.exp(-rate * x)
    return _norm(a)
def _pow_shape(n, p=1.0):
    x = np.arange(n, dtype=np.float64) / max(1, n)
    a = (1.0 - x + 1e-4) ** p
    return _norm(a)
def _random_smooth_shape(n, rng):
    w = max(20, int(n // 80))
    kernel = np.exp(-np.arange(w) / max(1.0, w / 6.0))
    kernel /= kernel.sum()
    b = rng.normal(0.0, 1.0, size=n)
    b = np.convolve(b, kernel, mode='same')
    a = np.abs(b) + 0.25 + 0.1 * rng.random(n)
    return _norm(a)
def _interp_shape(src, n):

    if src.size == 0:

    xs = np.linspace(0.0,1.0,src.size)
    xt = np.linspace(0.0,1.0,n)
    y = np.interp(xt, xs, np.maximum(src,0.0))
```

```
def _nextfft(n):
    m=2*n-1
    N=1
    while N < m: N <<=1
    return N
def _fft_conv(a, N, m, fa=None):
    if fa is None:
        fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N, n=N)
    c = c[:m]
    # small negatives
    np.maximum(c, 0.0, out=c)
    return c, fa
def _grad_from_p(p, N, fa, n):
    ph = np.fft.rfft(p, N)
    g = 2.0 * np.fft.irfft(ph * np.conj(fa), N, n=N)
    return g[:n].copy()
```

```
def _make_weights(c, delta, blend):
    cmax = float(np.max(c))
    L = c.size
    if cmax <= 0.0:
        p = np.zeros(L, dtype=np.float64); p[0]=1.0; return p
    tau = cmax * float(delta)
    idx = int(np.argmax(c))
    if tau <= 1e-300:
        p = np.zeros(L, dtype=np.float64); p[idx]=1.0; return p
    e = np.exp((c - cmax) / tau)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(L, dtype=np.float64); p[idx]=1.0; return p
    p = e / s
    if blend > 0.0:
        b = float(blend)
        if b > 1.0: b = 1.0
        if b > 1e-8:
            p *= (1.0 - b)
            p[idx] += b
            # renormalize to 1 (floating)
            ss = float(p.sum())
            if ss > 0.0: p /= ss
    return p
```

```
def _safe_mult_update(a, mom, eta):

        return a.copy()
```

```python
"""
Fast FFT-based projected multiplicative subgradient search for the sequence
minimizing 2*n*max(np.convolve(a,a))/sum(a)^2 with a_i >= 0.

The code uses:
- normalization sum(a)=1, so the local objective is 2*n*max(conv)
- power-of-two FFT convolution and FFT-based gradients
- entropy/softmax weights over active convolution lags, blended with one hot
  active lag
- line search with small annealed hill-climbing tolerance
- momentum, restarts by multiplicative block noise, transfers, and alpha
  re-blending
- several starting shapes and a shorter larger-length refinement
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _norm(a):
    """Normalize to non-negative values with sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = a.astype(np.float64, copy=True)
    a[~np.isfinite(a)] = 0.0
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / s


def _alpha_shape(n, gamma=0.5):
    """Slowly decaying power-law shape."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** (-float(gamma))
    return _norm(a)


def _exp_shape(n, rate=1.5):
    """Exponential decay shape."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.exp(-float(rate) * x)
    return _norm(a)


def _pow_shape(n, p=1.0):
    """Polynomial decay toward the end."""
    x = np.arange(n, dtype=np.float64) / max(1, n)
    a = (1.0 - x + 1e-4) ** float(p)
    return _norm(a)


def _random_smooth_shape(n, rng):
    """Smooth random initialization."""
    w = max(20, int(n // 80))
    kernel = np.exp(-np.arange(w) / max(1.0, w / 6.0))
    kernel /= kernel.sum()
    b = rng.normal(0.0, 1.0, size=n)
    b = np.convolve(b, kernel, mode="same")
    a = np.abs(b) + 0.25 + 0.1 * rng.random(n)
    return _norm(a)


def _interp_shape(src, n):
    """Linearly resample a sequence to length n and normalize."""
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return _alpha_shape(n, 0.5)
    if src.size == n:
        return _norm(np.maximum(src, 0.0))
    xs = np.linspace(0.0, 1.0, src.size)
    xt = np.linspace(0.0, 1.0, n)
    y = np.interp(xt, xs, np.maximum(src, 0.0))
    return _norm(y)


def _nextfft(n):
    """Power-of-two FFT length at least 2*n-1."""
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N


def _fft_conv(a, N, m, fa=None):
    """Full self-convolution by FFT, length m=2n-1, plus reuse FFT(a)."""
    if fa is None:
        fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N, n=N)
    c = c[:m]
    np.maximum(c, 0.0, out=c)
    return c, fa


def _grad_from_p(p, N, fa, n):
    """
    Gradient of sum_k p_k conv(a,a)[k] w.r.t. a.
    Returns 2 * correlation(p, a) on first n coordinates.
    """
    ph = np.fft.rfft(p, N)
    g = 2.0 * np.fft.irfft(ph * np.conj(fa), N, n=N)
    return g[:n].copy()


def _make_weights(c, delta, blend):
    """
    Softmax weights over convolution coefficients.
    delta controls temperature as delta*cmax.
    blend adds weight to a single max coefficient.
    """
    L = c.size
    cmax = float(np.max(c))
    if cmax <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[0] = 1.0
        return p
    tau = cmax * float(delta)
    idx = int(np.argmax(c))
    if tau <= 1e-300:
        p = np.zeros(L, dtype=np.float64)
        p[idx] = 1.0
        return p
    e = np.exp((c - cmax) / tau)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[idx] = 1.0
        return p
    p = e / s
    if blend > 0.0:
        b = float(blend)
        if b > 1.0:
            b = 1.0
        if b > 1e-8:
            p *= (1.0 - b)
            p[idx] += b
            ss = float(p.sum())
            if ss > 0.0:
                p /= ss
    return p


def _safe_mult_update(a, mom, eta):
    """Multiplicative update and renormalization."""
    if eta <= 0.0:
        return a.copy()
    try:
        a2 = a * np.exp(-eta * mom)
    except Exception:
        return None
    if not np.all(np.isfinite(a2)):
        return None
    a2 = np.maximum(a2, 1e-300)
    s = float(a2.sum())
    if not np.isfinite(s) or s <= 1e-200:
        return None
    return a2 / s


def _transfer_perturb(a, rng, eps):
    """Move a small amount of mass from one coordinate to another."""
    b = np.asarray(a, dtype=np.float64).copy()
    n = len(b)
    if n == 0:
        return b
    i = 0
    for _ in range(30):
        i = int(rng.integers(0, n))
        if b[i] > 1e-12:
            break
    amt = float(eps) * float(b[i])
    j = int(rng.integers(0, n))
    if i == j:
        return _norm(b)
    b[i] -= amt
    b[j] += amt
    b = np.maximum(b, 0.0)
    return _norm(b)


def _perturb(a, rng, strength):
    """Random mutation used for restarts."""
    n = len(a)
    r = float(rng.random())
    if r < 0.2:
        return _transfer_perturb(a, rng, min(0.25, max(0.02, 0.5 * strength)))
    if r < 0.4:
        alpha = _alpha_shape(n, 0.5)
        mix = min(0.5, max(0.05, 0.7 * strength))
        b = (1.0 - mix) * a + mix * alpha
        return _norm(b)
    b = np.maximum(np.asarray(a, dtype=np.float64), 1e-300).copy()
    widths = (0.002, 0.005, 0.01, 0.02, 0.05, 0.10)
    wf = float(rng.choice(widths))
    w = max(2, int(n * wf))
    if w >= n:
        w = max(2, n // 10)
    if n > w:
        start = int(rng.integers(0, n - w + 1))
    else:
        start = 0
    b[start:start + w] *= np.exp(rng.normal(0.0, float(strength), size=w))
    return _norm(b)


def _kind_params(kind):
    """Parameters for each search task."""
    if kind == "parentA":
        return (0.15, 0.002, 0.30, 0.80, 0.35, 0.88, 0.004, 25.0)
    if kind == "big":
        return (0.12, 0.002, 0.30, 0.75, 0.32, 0.88, 0.005, 30.0)
    if kind == "alpha":
        return (0.55, 0.005, 0.20, 0.60, 0.45, 0.85, 0.02, 30.0)
    if kind == "exp":
        return (0.55, 0.005, 0.20, 0.60, 0.45, 0.85, 0.02, 30.0)
    if kind == "pow":
        return (0.60, 0.005, 0.20, 0.60, 0.45, 0.85, 0.02, 30.0)
    if kind == "rand":
        return (0.80, 0.010, 0.10, 0.45, 0.50, 0.85, 0.03, 35.0)
    return (0.35, 0.002, 0.25, 0.70, 0.35, 0.87, 0.01, 25.0)


def _optimize_sequence(
    a,
    N,
    n,
    m,
    deadline,
    tag,
    rng,
    delta_start=0.25,
    delta_end=0.001,
    blend_start=0.25,
    blend_end=0.75,
    eta0=0.35,
    beta=0.86,
    temper_base=0.01,
    stall_time=25.0,
    verbose_every=5000,
):
    """
    Optimize a single sequence.
    Returns (best_sequence, best_objective), with objective = 2*n*max(conv)
    when sum(sequence)=1.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size != n:
        a = _interp_shape(a, n)
    a = np.maximum(a, 1e-12)
    a = a / float(a.sum())

    c, fa = _fft_conv(a, N, m)
    obj = 2.0 * n * float(np.max(c))

    best_a = a.copy()
    best_obj = obj

    mom = np.zeros(n, dtype=np.float64)
    eta = float(eta0)

    t_start = time.monotonic()
    duration = max(1e-6, float(deadline) - t_start)
    last_improve = t_start
    wide_until = 0.0
    fails = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            last_improve = now

        frac = min(1.0, (now - t_start) / duration)
        delta = delta_start + (delta_end - delta_start) * frac
        blend = blend_start + (blend_end - blend_start) * frac

        if now < wide_until:
            delta = min(0.25, max(delta, 0.12))
            blend = min(blend, 0.45)

        p = _make_weights(c, delta, blend)
        g = _grad_from_p(p, N, fa, n)
        if not np.all(np.isfinite(g)):
            break

        # Project to the subspace preserving sum to first order.
        h = g - float(np.dot(a, g))
        habs = float(np.max(np.abs(h)))
        if not np.isfinite(habs) or habs < 1e-15:
            break

        hs = h / habs
        mom = beta * mom + (1.0 - beta) * hs
        mom -= float(np.dot(a, mom))
        momax = float(np.max(np.abs(mom)))
        if not np.isfinite(momax) or momax < 1e-14:
            break
        mom = mom / momax

        old_obj = obj
        step = eta
        hot = temper_base * max(0.0, 1.0 - 1.5 * frac)
        limit = old_obj * (1.0 + hot)
        if best_obj < np.inf:
            cap = best_obj * 1.004
            if limit > cap:
                limit = cap

        accept = False
        best_trial_obj = np.inf
        best_a2 = None
        best_c2 = None
        best_fa2 = None

        for trial in range(5):
            a2 = _safe_mult_update(a, mom, step)
            if a2 is None:
                step *= 0.5
                continue
            c2, fa2 = _fft_conv(a2, N, m)
            o2 = 2.0 * n * float(np.max(c2))
            if o2 < best_trial_obj:
                best_trial_obj = o2
                best_a2 = a2
                best_c2 = c2
                best_fa2 = fa2
            if (o2 <= old_obj + 1e-10) or (hot > 0.0 and o2 <= limit):
                accept = True
                break
            step *= 0.5

        if accept:
            a = best_a2
            c = best_c2
            fa = best_fa2
            obj = best_trial_obj
            fails = 0
            if obj < old_obj - 1e-10:
                eta = min(eta * 1.05, 1.2)
            else:
                eta = max(eta * 0.95, 1e-4)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                last_improve = time.monotonic()
        else:
            fails += 1
            eta = max(eta * 0.6, 1e-5)

            # Safety: if some trial was actually lower, take it.
            if best_trial_obj < old_obj - 1e-10 and best_a2 is not None:
                a = best_a2
                c = best_c2
                fa = best_fa2
                obj = best_trial_obj
                fails = 0
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    last_improve = time.monotonic()
            elif fails >= 8 or (time.monotonic() - last_improve) > stall_time:
                elapsed_stall = time.monotonic() - last_improve
                strength = min(
                    0.8,
                    0.07 + 0.004 * float(max(0.0, elapsed_stall / 10.0)),
                )
                if best_obj < obj / 1.01:
                    a0 = best_a.copy()
                else:
                    a0 = a
                a = _perturb(a0, rng, strength)
                c, fa = _fft_conv(a, N, m)
                obj = 2.0 * n * float(np.max(c))
                mom = np.zeros(n, dtype=np.float64)
                eta = float(eta0)
                fails = 0
                last_improve = time.monotonic()
                wide_until = time.monotonic() + max(5.0, 0.05 * duration)

        if it % max(1, int(verbose_every)) == 0:
            print(
                f"[{tag}] it={it} curr={obj:.6f} best={best_obj:.6f} "
                f"delta={delta:.4f} blend={blend:.2f} eta={eta:.4f}",
                flush=True,
            )

    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    return best_a, best_obj


def _resolution_upgrade(a, obj, hard_deadline, rng):
    """
    Try to improve by interpolating the current best to a larger length,
    especially up to 8191, and doing a short refinement.
    """
    n = len(a)
    if n <= 0:
        return a, obj

    n2 = min(8191, max(n + 1000, 2 * n))
    if n2 <= n or n2 > 8191:
        return a, obj

    init = _interp_shape(a, n2)
    N2 = _nextfft(n2)
    m2 = 2 * n2 - 1
    c2, _ = _fft_conv(init, N2, m2)
    obj2 = 2.0 * n2 * float(np.max(c2))

    # If interpolation badly hurts the objective, do not spend time here.
    if obj2 > obj + 0.005:
        return a, obj

    now = time.monotonic()
    sub_deadline = min(float(hard_deadline), now + 120.0)
    if sub_deadline > now + 30.0:
        init, obj2 = _optimize_sequence(
            init,
            N2,
            n2,
            m2,
            sub_deadline,
            "upsample",
            rng,
            delta_start=0.08,
            delta_end=0.002,
            blend_start=0.40,
            blend_end=0.80,
            eta0=0.30,
            beta=0.88,
            temper_base=0.003,
            stall_time=20.0,
            verbose_every=2000,
        )

    if obj2 < obj:
        return init, obj2
    return a, obj


def propose_candidate():
    """Search for a low-objective non-negative sequence within 2400 seconds."""
    t0 = time.monotonic()
    total_budget = 2400.0
    deadline = t0 + total_budget - 10.0
    rng = np.random.default_rng(123456789)

    # Parent sequence from previous state of the art, if available.
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None:
        if parent.ndim != 1 or parent.size == 0:
            parent = None
        else:
            parent = parent.astype(np.float64, copy=True)
            parent[~np.isfinite(parent)] = 0.0
            parent = np.maximum(parent, 0.0)
            if parent.size > 20000:
                xs = np.linspace(0.0, 1.0, parent.size)
                xt = np.linspace(0.0, 1.0, 3809)
                parent = np.interp(xt, xs, parent)

    if parent is not None and parent.size >= 50:
        n0 = int(parent.size)
        a0 = _norm(parent)
        use_parent = True
    else:
        n0 = 3809
        a0 = _alpha_shape(n0, 0.5)
        use_parent = False

    N0 = _nextfft(n0)
    m0 = 2 * n0 - 1
    c0, _ = _fft_conv(a0, N0, m0)
    obj0 = 2.0 * n0 * float(np.max(c0))

    global_a = a0.copy()
    global_obj = obj0
    print(f"init n={n0} obj={global_obj:.6f}", flush=True)

    tasks = []
    if use_parent:
        tasks.append((n0, a0, "parentA", 0.40))
        tasks.append((n0, _alpha_shape(n0, 0.5), "alpha", 0.14))
        tasks.append((n0, _exp_shape(n0, 1.5), "exp", 0.04))
        tasks.append((n0, _pow_shape(n0, 1.0), "pow", 0.04))
        tasks.append((n0, _random_smooth_shape(n0, rng), "rand", 0.04))
    else:
        tasks.append((n0, a0, "alpha", 0.35))
        tasks.append((n0, _alpha_shape(n0, 0.45), "alpha", 0.10))
        tasks.append((n0, _exp_shape(n0, 2.0), "exp", 0.10))
        tasks.append((n0, _random_smooth_shape(n0, rng), "rand", 0.10))

    n_big = 4096
    if n_big > n0:
        if use_parent:
            big_init = _interp_shape(parent, n_big)
        else:
            big_init = _alpha_shape(n_big, 0.5)
        tasks.append((n_big, big_init, "big", 0.28 if use_parent else 0.40))

    total_w = sum(float(t[3]) for t in tasks)
    reserve_final = 160.0
    budget_left = max(0.0, deadline - time.monotonic() - reserve_final)
    weight_left = float(total_w)

    for n_t, init, kind, w in tasks:
        now = time.monotonic()
        if budget_left < 15.0 or weight_left <= 1e-12:
            break
        if now > deadline - reserve_final:
            break

        slice_time = budget_left * (float(w) / weight_left)
        max_slice = deadline - now - reserve_final
        if slice_time > max_slice:
            slice_time = max_slice
        if slice_time < 15.0:
            weight_left -= float(w)
            continue

        if len(init) != n_t:
            init = _interp_shape(init, n_t)
        init = _norm(np.maximum(init, 0.0))

        sub_deadline = time.monotonic() + slice_time
        Nt = _nextfft(n_t)
        mt = 2 * n_t - 1
        params = _kind_params(kind)

        try:
            a_best, o_best = _optimize_sequence(
                init,
                Nt,
                n_t,
                mt,
                sub_deadline,
                kind,
                rng,
                delta_start=params[0],
                delta_end=params[1],
                blend_start=params[2],
                blend_end=params[3],
                eta0=params[4],
                beta=params[5],
                temper_base=params[6],
                stall_time=params[7],
            )
        except Exception as e:
            print(f"optimizer error {kind}: {e}", flush=True)
            a_best = init
            o_best = 2.0 * n_t * float(np.max(_fft_conv(init, Nt, mt)[0]))

        if o_best < global_obj - 1e-12:
            global_a = a_best
            global_obj = o_best
            print(f"new global after {kind}: n={n_t} obj={global_obj:.6f}", flush=True)

        weight_left -= float(w)
        budget_left = max(0.0, deadline - time.monotonic() - reserve_final)

    # Optional larger-length upgrade before final polishing.
    if time.monotonic() < deadline - 240.0:
        hard_deadline = min(deadline - 150.0, time.monotonic() + 120.0)
        if hard_deadline > time.monotonic() + 40.0:
            global_a, global_obj = _resolution_upgrade(
                global_a,
                global_obj,
                hard_deadline,
                rng,
            )
            print(f"after upsample: n={len(global_a)} obj={global_obj:.6f}", flush=True)

    # Final hard polishing stage.
    if time.monotonic() < deadline - 5.0:
        n_f = len(global_a)
        Nf = _nextfft(n_f)
        mf = 2 * n_f - 1
        global_a, global_obj = _optimize_sequence(
            global_a,
            Nf,
            n_f,
            mf,
            deadline - 3.0,
            "final",
            rng,
            delta_start=0.02,
            delta_end=0.0,
            blend_start=0.70,
            blend_end=1.0,
            eta0=0.22,
            beta=0.90,
            temper_base=0.0,
            stall_time=15.0,
            verbose_every=2000,
        )

    # Build final output.  Sum=1 is safe and avoids clipping.
    result = []
    for x in np.asarray(global_a, dtype=np.float64).ravel():
        y = float(x)
        if not np.isfinite(y) or y < 1e-300:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        result.append(y)

    # If something went wrong, fall back to a simple valid sequence.
    try:
        ev = evaluate_sequence(result)
    except Exception:
        ev = float("inf")

    if not np.isfinite(ev):
        fallback = _alpha_shape(3809, 0.5)
        result = [float(max(0.0, x)) for x in fallback]
        try:
            ev = evaluate_sequence(result)
        except Exception:
            ev = float("inf")

    print(
        f"FINAL computed={global_obj:.6f} evaluated={ev:.6f} n={len(result)}",
        flush=True,
    )
    return result
```
