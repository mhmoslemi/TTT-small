**LP Approach:**

At each iteration:
1. Given current `a` (normalized to `sum(a) = 1`), compute `b = a*a` and `M = max(b)`.
2. Solve LP: maximize `sum(g)` subject to `g ≥ 0` and `(a*g)[k] ≤ M` for all `k`.
3. Rescale `g` to have `sum(g) = 1`.
4. Blend: `a_new = (1-t)*a + t*g`.

```python
"""
Step-function height sequence optimizer using LP-based direction finding
and soft-gradient polishing.

Main innovation: At each iteration, solve a linear program to find the
"best" direction g that maximizes sum(g) subject to a*g <= M (current
peak autocorrelation). Blend the current solution with g using an
adaptive step size. This is fundamentally different from pure gradient
descent and can take larger, more informed steps.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import scipy.linalg

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s <= 0:
        return np.ones(a.shape, dtype=np.float64) / a.size
    return a / s

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    return float(2.0 * n * bmax / (s * s))

def _conv(a, m=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if m is None:
        m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    return np.maximum(b, 0.0)

def _build_conv_matrix_sparse(a):
    """Build the (2n-1) x n convolution matrix T where (T@g)[k] = (a*g)[k]."""
    n = len(a)
    nrow = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(nrow):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        js = np.arange(j_min, j_max + 1)
        cnt = len(js)
        rows[idx:idx+cnt] = k
        cols[idx:idx+cnt] = js
        vals[idx:idx+cnt] = a[k - js]
        idx += cnt
    # Trim if needed (should be exactly n^2)
    return csc_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(nrow, n))

def _lp_direction(a, M, use_sparse=True):
    """Solve LP: max sum(g) s.t. T@g <= M*1, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    
    if use_sparse:
        A_ub = _build_conv_matrix_sparse(a)
    else:
        # Dense Toeplitz
        col = np.zeros(2 * n - 1)
        col[:n] = a
        row = np.zeros(n)
        row[0] = a[0]
        A_ub = scipy.linalg.toeplitz(col, row)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = res.x
            g = np.maximum(g, 0.0)
            if g.sum() > 0:
                return g
    except Exception:
        pass
    return None

def _lp_iteration(a, max_lp_time=200):
    """Perform LP-based iterations on a. Returns (best_a, best_score)."""
    a = _normalize(a)
    n = a.size
    b = _conv(a)
    M = float(b.max())
    best_a = a.copy()
    best_score = 2.0 * n * M  # since sum(a)=1, score = 2n*max(b)
    
    start = time.monotonic()

            b = _conv(a)

    out = np.maximum(out, 0)

    if ss <= 0:

    return out / ss
```

```python
Minimizes 2*n*max(autocorrelation)/sum(a)^2 for non-negative step functions.

Primary method: LP-based direction finding - at each step, solve a linear program
to find direction g maximizing sum(g) subject to a*g <= M (current peak autocorrelation).
This gives informed, potentially large steps that differ fundamentally from gradient descent.

Secondary methods: Soft-gradient polishing in log-space and mirror descent (exponentiated

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):

    return a / s

        return np.inf
    s = float(a.sum())

    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if not np.isfinite(bmax):
        return np.inf
    return float(2.0 * n * bmax / (s * s))

def _conv(a, m=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if m is None:
        m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    return np.maximum(b, 0.0)


def _build_conv_matrix_sparse(a):
    n = len(a)
    nrow = 2 * n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(nrow):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        cnt = j_max - j_min + 1
        js = np.arange(j_min, j_max + 1, dtype=np.int32)
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = js
        vals[idx:idx + cnt] = a[k - js]
        idx += cnt
    return csc_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(nrow, n))


def _lp_direction(a, M, n):
    c = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    A_ub = _build_conv_matrix_sparse(a)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - float(v.max())
    e = np.exp(np.clip(v, -700.0, 700.0))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(v.shape, dtype=np.float64) / max(1, v.size)
    return e / s


def _interp_to_length(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s <= 0 or a.size == 0:
        return np.ones(n, dtype=np.float64) / n
    a = a / s
    if a.size == n:
        return a
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 0:
        return np.ones(n, dtype=np.float64) / n
    return out / ss


def _make_inits(n, rng, prev_a=None):
    inits = []
    t = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    if prev_a is not None:
        inits.append(_normalize(_interp_to_length(prev_a, n)))

    inits.append(_normalize(np.ones(n)))

    for c in (0.5, 1.0, 2.0, 4.0, 6.0):
        inits.append(_normalize(np.exp(-c * t)))

    left = i + 1.0
    right = n - i
    inits.append(_normalize(1.0 / np.sqrt(left * right)))

    for p in (0.3, 0.5, 0.8, 1.2, 2.0):
        inits.append(_normalize(left ** p))

    for p in (0.3, 0.5, 0.8, 1.2, 2.0):
        inits.append(_normalize(right ** p))

    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t) + 0.2 * np.cos(4.0 * np.pi * t)
    inits.append(_normalize(np.maximum(a, 0)))

    kc = min(10, n)
    a = 1.0
    for k in range(1, kc + 1):
        amp = 0.25 / (float(k) ** 0.8)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t + ph)
    inits.append(_normalize(np.maximum(a, 0)))

    inits.append(_normalize(np.maximum(1.0 - np.abs(2.0 * t - 1.0) ** 2, 0)))
    inits.append(_normalize(np.maximum(1.0 - np.abs(2.0 * t - 1.0) ** 4, 0)))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative step function heights minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Uses LP-based direction finding (solving max sum(g) s.t. a*g <= M, g>=0)
    as the primary optimization engine, combined with soft-gradient Adam
    polishing and exponentiated gradient (mirror descent) for fine-tuning.
    Progressive length increase from 512 to 8192.
    """
    seed = int(seed)
    budget_s = float(max(60.0, budget_s))
    start = time.monotonic()
    deadline = start + budget_s - 40.0  # 40s safety buffer

    # Get previous best if available
    prev_a = None
    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.maximum(arr, 0)
            if float(arr.sum()) > 0:
                prev_a = arr / float(arr.sum())
    except Exception:
        prev_a = None

    best_a = prev_a
    best_score = _score(prev_a) if prev_a is not None else 2.0
    print("Initial best score: %.6f (n=%d)" % (best_score, len(prev_a) if prev_a is not None else 0))

    stages = [
        (512, 0.05),
        (1024, 0.10),
        (2048, 0.30),
        (4096, 0.38),
        (8192, 0.17),
    ]

    rng_main = np.random.default_rng(seed)

    for stage_idx, (n, frac) in enumerate(stages):
        now = time.monotonic()
        rem = deadline - now
        if rem < 60.0:
            break

        stage_time = min(rem * frac, rem - 30.0)
        stage_deadline = now + stage_time
        print("\n=== Stage n=%d, budget=%.1fs, current_best=%.6f ===" % (n, stage_time, best_score))

        # Build initializations
        inits = _make_inits(n, rng_main, best_a)

        # Score and sort initializations
        scored_inits = []
        seen = []
        for a in inits:
            a = _normalize(a)
            sc = _score(a)
            if not np.isfinite(sc):
                continue
            diverse = True
            for b in seen:
                d = float(np.sqrt(np.sum((a - b) ** 2)))
                if d < 0.03:
                    diverse = False
                    break
            if diverse:
                scored_inits.append((sc, a))
                seen.append(a)
                if len(seen) >= 12:
                    break

        scored_inits.sort(key=lambda x: x[0])

        # Pick top candidates
        cands = [a for sc, a in scored_inits[:4]]
        if not cands:
            cands = [np.ones(n) / n]

        best_stage_a = cands[0]
        best_stage_score = _score(cands[0])
        if best_stage_score < best_score:
            best_score = best_stage_score

        # Time allocation: LP gets 50%, gradient 25%, mirror 15%, extra LP 10%
        lp1_time = stage_time * 0.45
        grad_time = stage_time * 0.25
        mirror_time = stage_time * 0.10
        lp2_time = stage_time * 0.10
        lp3_time = max(0.0, stage_time - lp1_time - grad_time - mirror_time - lp2_time - 10.0)

        # Phase 1: LP on best init
        if lp1_time > 5.0:
            now = time.monotonic()
            alloc = min(lp1_time, stage_deadline - now - 10.0)
            if alloc > 5.0:
                a_lp, sc_lp, nlp = _lp_iteration(cands[0], max_time=alloc)
                if sc_lp < best_stage_score:
                    best_stage_score = sc_lp
                    best_stage_a = a_lp
                print("  LP1: score=%.6f (%d iters)" % (best_stage_score, nlp))

        # Phase 2: LP on second candidate if time allows
        now = time.monotonic()
        rem_stage = stage_deadline - now
        if len(cands) > 1 and rem_stage > 30.0:
            alloc2 = min(lp1_time * 0.5, rem_stage * 0.3)
            if alloc2 > 5.0:
                a_lp2, sc_lp2, nlp2 = _lp_iteration(cands[1], max_time=alloc2)
                if sc_lp2 < best_stage_score:
                    best_stage_score = sc_lp2
                    best_stage_a = a_lp2
                print("  LP2 (cand2): score=%.6f (%d iters)" % (best_stage_score, nlp2))

        # Phase 3: Gradient polish
        now = time.monotonic()
        rem_stage = stage_deadline - now
        alloc = min(grad_time, rem_stage * 0.4)
        if alloc > 5.0:
            a_g, sc_g = _gradient_polish(best_stage_a, alloc, seed=seed + n)
            if sc_g < best_stage_score:
                best_stage_score = sc_g
                best_stage_a = a_g
            print("  Gradient: score=%.6f" % best_stage_score)

        # Phase 4: Mirror descent
        now = time.monotonic()
        rem_stage = stage_deadline - now
        alloc = min(mirror_time, rem_stage * 0.3)
        if alloc > 5.0:
            a_m, sc_m = _mirror_descent(best_stage_a, alloc, seed=seed + n + 1)
            if sc_m < best_stage_score:
                best_stage_score = sc_m
                best_stage_a = a_m
            print("  Mirror: score=%.6f" % best_stage_score)

        # Phase 5: More LP polishing
        now = time.monotonic()
        rem_stage = stage_deadline - now
        total_lp_rem = lp2_time + lp3_time
        alloc = min(total_lp_rem, rem_stage)
        if alloc > 5.0:
            a_lp3, sc_lp3, nlp3 = _lp_iteration(best_stage_a, max_time=alloc)
            if sc_lp3 < best_stage_score:
                best_stage_score = sc_lp3
                best_stage_a = a_lp3
            print("  LP3: score=%.6f (%d iters)" % (best_stage_score, nlp3))

        # Phase 6: Final gradient polish
        now = time.monotonic()
        rem_stage = max(0.0, stage_deadline - now)
        if rem_stage > 10.0:
            a_g2, sc_g2 = _gradient_polish(best_stage_a, rem_stage, seed=seed + n + 100)
            if sc_g2 < best_stage_score:
                best_stage_score = sc_g2
                best_stage_a = a_g2
            print("  Final grad: score=%.6f" % best_stage_score)

        if best_stage_score < best_score - 1e-9:
            best_score = best_stage_score
            best_a = best_stage_a
            print("  >>> New global best: %.6f at n=%d" % (best_score, n))
        else:
            print("  Stage best: %.6f (no improvement over global %.6f)" % (best_stage_score, best_score))

    # Final: try a last LP iteration on the best found
    now = time.monotonic()
    rem = deadline - now
    if rem > 20.0 and best_a is not None:
        a_final, sc_final, nfinal = _lp_iteration(best_a, max_time=rem - 10.0)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final
        print("\nFinal LP: %.6f (%d iters)" % (best_score, nfinal))

    # Final gradient polish
    now = time.monotonic()
    rem = deadline - now
    if rem > 10.0 and best_a is not None:
        a_final2, sc_final2 = _gradient_polish(best_a, rem - 5.0, seed=seed + 999)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_a = a_final2
        print("Final gradient: %.6f" % best_score)

    best_a = _normalize(best_a)
    print("\n=== FINAL: score=%.6f, n=%d ===" % (best_score, len(best_a)))

    # Validate with exact evaluator
    try:
        if len(best_a) <= 10000:
            exact = evaluate_sequence([float(x) for x in best_a])
            print("Exact evaluation: %.6f" % exact)
            if not np.isfinite(exact):
                print("Invalid! Falling back to flat.")
                best_a = np.ones(512, dtype=np.float64)
    except Exception as e:
        print("Validation error: %s" % str(e))

    return [float(x) for x in best_a]
```
