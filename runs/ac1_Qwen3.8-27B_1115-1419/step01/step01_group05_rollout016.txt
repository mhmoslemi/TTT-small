```python
"""
Search for low-peak nonnegative self-convolutions.

The objective is
    score(a) = 2 n max_k (a*a)_k / (sum a)^2 ,          a_i >= 0.
It is scale invariant, so search uses sum(a) = 1.

Strategy:
  * Build a broad pool of analytic/randomized starts, including the provided
    previous sequence, inverse-root profiles, exponentials, Gaussian mixtures,
    and formal square-root-of-flat-convolution initializations.
  * Polish starts with an adaptive log-sum-exp smoothed projected-gradient
    method on the probability simplex, using FFT-based gradients.
  * Intersperse random/log-domain escapes, symmetrization trials, and top-peak
    exact gradient polish.
  * Add a small-n local mass-transfer squeeze at the end.
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _safe_array(a):
    """Convert input to a finite 1D nonnegative float array."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return np.zeros(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    return arr


def _normalize(a):
    """Normalize to sum 1 and keep values nonnegative."""
    a = _safe_array(a)
    n = a.size
    if n == 0:
        return np.array([1.0])
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s


def _next_pow2(m):
    """Smallest power of two >= m."""
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _convolve(a, fft_len=None):
    """Linear full convolution a*a via zero-padded FFT."""
    a = _safe_array(a)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, fft_len)
    b = np.fft.irfft(A * A, fft_len)
    b = b[:2 * n - 1]
    b = np.maximum(b, 0.0)
    return b


def _score_from_b(n, b):
    """Score from an FFT convolution of a sum-1 sequence."""
    if b.size == 0:
        return float("inf")
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return float("inf")
    return 2.0 * float(n) * mb


def _score(a):
    """Exact scale-invariant score for arbitrary nonnegative array."""
    a = _safe_array(a)
    n = a.size
    s = float(a.sum())
    if n == 0 or s <= 1e-300:
        return float("inf")
    b = _convolve(a / s)
    return _score_from_b(n, b)


def _score_norm(a, conv_fft=None):
    """Exact score for a sum-1 sequence, with optional cached FFT length."""
    a = _safe_array(a)
    n = a.size
    s = float(a.sum())
    if n == 0 or s <= 1e-300:
        return float("inf")
    b = _convolve(a / s, conv_fft)
    return _score_from_b(n, b)


def _lse_value(b, beta):
    """Stable log-sum-exp(b)/beta, with beta=1 handled."""
    m = b.size
    if m == 0:
        return 0.0
    max_b = float(np.max(b))
    if beta <= 1.0:
        return max_b + float(np.log(float(m))) / max(float(beta), 1e-300)
    arg = beta * (b - max_b)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        return max_b
    return max_b + float(np.log(s)) / float(beta)


def _softmax_weights(b, beta):
    """Return weights p proportional to exp(beta*b), value, and effective size."""
    m = b.size
    if m == 0:
        return np.zeros(0), 0.0, 0.0
    max_b = float(np.max(b))
    if beta <= 1.0:
        return np.full(m, 1.0 / float(m), dtype=np.float64), max_b, float(m)
    arg = beta * (b - max_b)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 1e-300:
        p = np.full(m, 1.0 / float(m), dtype=np.float64)
        return p, max_b, float(m)
    p = e / s
    ss = float(np.sum(p * p))
    neff = float(m) if ss <= 1e-300 else 1.0 / ss
    val = max_b + float(np.log(s)) / float(beta)
    return p, val, neff


def _weighted_gradient(a, weights, fft_len):
    """
    Gradient of sum_k weights_k (a*a)_k with respect to a_i.
    For real nonnegative a and weights, it is 2 * sum_j a_j weights_{i+j}.
    """
    a = _safe_array(a)
    weights = _safe_array(weights)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_len)
    W = np.fft.rfft(weights, fft_len)
    conv = np.fft.irfft(A * W, fft_len)
    h = conv[n - 1:2 * n - 1].real
    h = np.maximum(h, 0.0)
    return 2.0 * h


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([1.0])
    v = np.where(np.isfinite(v), v, 0.0)
    x = np.sort(v)[::-1]
    cssv = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = x * idx - (cssv - 1.0) > 0.0
    if not np.any(cond):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    w = v - theta
    w = np.maximum(w, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return w / s


def _project(v, exact=True):
    """Projection variant used by the optimizer."""
    if exact:
        return _project_simplex(v)
    v = _safe_array(v)
    n = v.size
    if n == 0:
        return np.array([1.0])
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return w / s


def _make_uniform(n):
    n = max(int(n), 1)
    return np.full(n, 1.0 / float(n), dtype=np.float64)


def _make_power(n, alpha, offset=1.0):
    """a_i proportional to (i+offset)^(-alpha)."""
    n = max(int(n), 2)
    idx = np.arange(n, dtype=np.float64) + float(offset)
    if float(alpha) <= 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = np.power(idx, -float(alpha))
    return _normalize(a)


def _make_exp_r(n, r):
    """Geometric decay sequence."""
    n = max(int(n), 2)
    r = float(np.clip(r, 0.5, 0.999999))
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r))
    return _normalize(a)


def _make_mixture_power(n, alpha, gamma, center, width):
    """Left inverse-root/power plus one Gaussian bump."""
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64) + 1.0
    if float(alpha) > 0.0:
        base = np.power(i, -float(alpha))
    else:
        base = np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n)
    z = (x - float(center)) / max(float(width), 1e-4)
    bump = np.exp(-z * z)
    a = base + float(gamma) * bump
    return _normalize(a)


def _make_two_bump_power(n, alpha, g1, c1, w1, g2, c2, w2):
    """Left inverse-root/power plus up to two Gaussian bumps."""
    n = max(int(n), 2)
    i = np.arange(n, dtype=np.float64) + 1.0
    if float(alpha) > 0.0:
        base = np.power(i, -float(alpha))
    else:
        base = np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n)
    z1 = (x - float(c1)) / max(float(w1), 1e-4)
    a = base + float(g1) * np.exp(-z1 * z1)
    if float(g2) > 0.0:
        z2 = (x - float(c2)) / max(float(w2), 1e-4)
        a = a + float(g2) * np.exp(-z2 * z2)
    return _normalize(a)


def _make_sqrt_flat(n, score_target):
    """
    Formal square-root of a flat first-half convolution profile.
    Useful as a nonnegative-ish heuristic initialize.
    """
    n = max(int(n), 8)
    m = float(score_target) / (2.0 * float(n))
    if not np.isfinite(m) or m <= 0.0:
        return _make_power(n, 0.5)
    a = np.zeros(n, dtype=np.float64)
    a0 = float(np.sqrt(m))
    a[0] = a0
    for k in range(1, n):
        if k > 1:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        else:
            s = 0.0
        val = (m - s) / (2.0 * a0)
        if not np.isfinite(val) or val < 0.0:
            val = 0.0
        a[k] = val
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return _make_power(n, 0.5)
    return a / s


def _make_smooth_random(base, rng, amplitude=0.4):
    """Low-pass multiplicative perturbation around base."""
    base = _safe_array(base)
    n = base.size
    if n == 0:
        return np.array([1.0])
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), max(4, n // 16 + 2)))
    if cutoff < len(F):
        F[cutoff:] = 0.0
    f = np.fft.irfft(F, n).real
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample(a, n):
    """Linearly resample shape to length n and normalize."""
    a = _safe_array(a)
    n = max(int(n), 2)
    if a.size == 0:
        return _make_uniform(n)
    s = float(a.sum())
    if s <= 1e-300:
        return _make_uniform(n)
    a = a / s
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    return _normalize(b)


def _symmetrize(a):
    """Averaging with its reversal preserves nonnegativity and may lower peaks."""
    a = _safe_array(a)
    return _normalize(0.5 * (a + a[::-1]))


def _perturb(a, rng, strength=0.3):
    """Random low-pass/block/symmetry escape move."""
    a = _safe_array(a)
    n = a.size
    if n == 0:
        return np.array([1.0])
    r = float(rng.random())
    if r < 0.45:
        loga = np.log(np.maximum(a, 1e-30))
        noise = rng.normal(0.0, float(strength), size=n)
        F = np.fft.rfft(noise)
        cutoff = max(2, min(len(F), max(4, n // 8 + 2)))
        if cutoff < len(F):
            F[cutoff:] = 0.0
        noise = np.fft.irfft(F, n).real
        b = np.exp(loga + noise)
        return _normalize(b)
    elif r < 0.75:
        c = a.copy()
        width = max(2, n // 20)
        hi = max(1, n - width)
        start = int(rng.integers(0, hi))
        stop = min(n, start + width)
        c[start:stop] *= float(np.exp(rng.normal(0.0, float(strength))))
        if rng.random() < 0.25:
            c = 0.5 * (c + c[::-1])
        return _normalize(c)
    else:
        return _symmetrize(a)


def _add_candidate(lst, item):
    lst.append(item)


def _sort_candidates(scored):
    """Sort (score, sequence) pairs by score without lambdas."""
    if not scored:
        return []
    scores = np.empty(len(scored), dtype=np.float64)
    for i, item in enumerate(scored):
        scores[i] = item[0]
    order = np.argsort(scores)
    out = []
    for idx in order:
        out.append(scored[int(idx)][1])
    return out


def _load_given():
    """Read height_sequence_1 if available."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0:
        return None
    s = float(arr.sum())
    if s <= 1e-300:
        return None
    return arr / s


def _build_pools(lengths, given, current_best, rng, deadline):
    """Evaluate a deterministic+random pool of starts for each length."""
    pools = {}
    for n in lengths:
        n = int(n)
        if n < 16:
            continue
        if time.monotonic() > deadline - 75.0:
            break

        cands = []

        if current_best is not None:
            _add_candidate(cands, _resample(current_best, n))
            _add_candidate(cands, _make_smooth_random(_resample(current_best, n), rng, 0.30))
            _add_candidate(cands, _symmetrize(_resample(current_best, n)))

        if given is not None:
            _add_candidate(cands, _resample(given, n))
            _add_candidate(cands, _make_smooth_random(_resample(given, n), rng, 0.25))
            _add_candidate(cands, _symmetrize(_resample(given, n)))

        _add_candidate(cands, _make_uniform(n))

        for alpha in (0.25, 0.35, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80):
            _add_candidate(cands, _make_power(n, alpha))

        for r in (0.70, 0.80, 0.90, 0.95, 0.97, 0.99, 0.999):
            _add_candidate(cands, _make_exp_r(n, r))

        # Mixture shapes.
        if n <= 2048:
            for alpha in (0.50, 0.55, 0.60):
                for gamma in (0.25, 0.6, 1.2, 2.5):
                    for center in (0.30, 0.55, 0.80):
                        for width in (0.07, 0.15, 0.30):
                            _add_candidate(cands, _make_mixture_power(n, alpha, gamma, center, width))

        # More expressive two-bump variants.
        if n <= 4096:
            _add_candidate(cands, _make_two_bump_power(n, 0.50, 1.0, 0.65, 0.25, 0.0, 0.0, 0.0))
            _add_candidate(cands, _make_two_bump_power(n, 0.55, 0.8, 0.35, 0.20, 0.8, 0.85, 0.20))
            _add_candidate(cands, _make_two_bump_power(n, 0.60, 0.6, 0.50, 0.25, 0.6, 0.90, 0.15))
            _add_candidate(cands, _make_two_bump_power(n, 0.45, 1.5, 0.75, 0.18, 0.0, 0.0, 0.0))
            _add_candidate(cands, _make_two_bump_power(n, 0.70, 0.5, 0.45, 0.18, 1.2, 0.85, 0.12))

        rnd_count = 20 if n <= 1024 else 8
        for _ in range(rnd_count):
            alpha = float(rng.uniform(0.35, 0.75))
            g1 = float(rng.uniform(0.0, 3.0))
            c1 = float(rng.uniform(0.10, 0.90))
            w1 = float(rng.uniform(0.05, 0.40))
            g2 = float(rng.uniform(0.0, 1.5))
            c2 = float(rng.uniform(0.30, 1.00))
            w2 = float(rng.uniform(0.05, 0.40))
            _add_candidate(cands, _make_two_bump_power(n, alpha, g1, c1, w1, g2, c2, w2))

        if n <= 1024:
            for st in (1.35, 1.40, 1.45, 1.50, 1.55, 1.65, 1.80):
                _add_candidate(cands, _make_sqrt_flat(n, st))

        for _ in range(6):
            base = _make_power(n, float(rng.uniform(0.40, 0.60)))
            _add_candidate(cands, _make_smooth_random(base, rng, float(rng.uniform(0.25, 0.80))))

        if len(cands) > 320:
            cands = cands[:320]

        scored = []
        for c in cands:
            if time.monotonic() > deadline - 60.0:
                break
            sc = _score(c)
            if np.isfinite(sc):
                scored.append((sc, c))

        sorted_cands = _sort_candidates(scored)
        pools[n] = sorted_cands[:14]
        if sorted_cands:
            print(f"pool n={n} best_startup={float(np.max(_convolve(sorted_cands[0]) * 0.0 + sorted_cands[0].sum())):.6f} score={sorted_cands[0][0]:.8f}", flush=True)
        else:
            print(f"pool n={n} no valid candidates", flush=True)

    return pools


def _get_starts(n, pools, given, best_a, rng):
    """Get initial starts for a length, respecting pool and current best."""
    n = int(n)
    starts = []
    if best_a is not None:
        _add_candidate(starts, _resample(best_a, n))
    if given is not None:
        _add_candidate(starts, _resample(given, n))
    tops = pools.get(n, [])
    for st in tops:
        _add_candidate(starts, st)
    if len(starts) < 3:
        _add_candidate(starts, _make_power(n, 0.50))
        _add_candidate(starts, _make_two_bump_power(n, 0.50, 1.0, 0.70, 0.25, 0.5, 0.90, 0.20))
        _add_candidate(starts, _make_smooth_random(_make_power(n, 0.50), rng, 0.45))
    return starts


def _refine_sequence(a, t_end, rng, label=""):
    """
    Main local optimizer: adaptive LSE smoothed projected gradient on simplex.
    Returns best (sequence, score) within t_end.
    """
    a = _normalize(a)
    n = a.size
    if n < 16:
        a = _make_power(128, 0.5)
        n = a.size
    if n > 16000:
        sc = _score(a)
        return a, sc

    conv_fft = _next_pow2(2 * n - 1)
    grad_fft = _next_pow2(3 * n - 2)

    b = _convolve(a, conv_fft)
    maxb = float(np.max(b)) if b.size else 0.0
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    target_beta = 6.0
    min_ne = max(8, int(0.04 * n))
    target_ne = max(min_ne, int(0.15 * n))
    alpha = 0.35
    fail = 0
    it = 0
    last_soft = 0
    exact_proj = n <= 8192

    while time.monotonic() < t_end:
        b = _convolve(a, conv_fft)
        if b.size == 0:
            break
        maxb = float(np.max(b))
        if not np.isfinite(maxb) or maxb <= 1e-300:
            break
        score = 2.0 * float(n) * maxb

        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score

        beta = max(1.0, target_beta / maxb)
        beta = min(1e7, beta)
        p, lse_old, neff = _softmax_weights(b, beta)

        if target_ne > min_ne and it % 100 == 0:
            target_ne = max(min_ne, int(target_ne * 0.97))

        if neff < 0.7 * float(target_ne):
            target_beta = max(1.0, target_beta * 0.85)
        elif neff > 1.6 * float(target_ne):
            target_beta = min(300.0, target_beta * 1.25)

        if it % 250 == 0:
            target_beta = min(300.0, target_beta * 1.05)

        g = _weighted_gradient(a, p, grad_fft)
        gc = g - float(np.mean(g))
        scale = float(np.max(np.abs(gc)))
        if scale < 1e-20:
            a = _perturb(best_a, rng, 0.25)
            b = _convolve(a, conv_fft)
            maxb = float(np.max(b)) if b.size else 0.0
            score = 2.0 * float(n) * maxb
            alpha = 0.35
            fail = 0
            it += 1
            continue

        accepted = False
        accepted_lse = lse_old
        step_attempts = 6
        for _try in range(step_attempts):
            v = a - alpha * gc
            cand = _project(v, exact_proj)
            b2 = _convolve(cand, conv_fft)
            maxb2 = float(np.max(b2)) if b2.size else 0.0
            score2 = 2.0 * float(n) * maxb2
            lse2 = _lse_value(b2, beta)

            if np.isfinite(score2) and lse2 < lse_old - 1e-15 and score2 < max(score, best_score) + 0.03:
                accepted = True
                accepted_lse = lse2
                a = cand
                score = score2
                break
            elif np.isfinite(score2) and score2 < best_score - 1e-14 and score2 < score:
                accepted = True
                accepted_lse = lse2
                a = cand
                score = score2
                break

            alpha *= 0.5

        if accepted:
            fail = 0
            last_soft = it
            if it % 1000 == 0:
                sym = _symmetrize(a)
                bs = _convolve(sym, conv_fft)
                maxs = float(np.max(bs)) if bs.size else 0.0
                scs = 2.0 * float(n) * maxs
                if scs < score:
                    a = sym
                    score = scs
                    if scs < best_score:
                        best_a = sym.copy()
                        best_score = scs
            if accepted_lse < lse_old - 1e-15:
                alpha = min(1.0, alpha * 1.12)
        else:
            alpha *= 0.5
            fail += 1
            if fail > 30:
                a = _perturb(best_a, rng, 0.25)
                b = _convolve(a, conv_fft)
                maxb = float(np.max(b)) if b.size else 0.0
                score = 2.0 * float(n) * maxb
                alpha = 0.35
                fail = 0
                target_beta = max(3.0, target_beta * 0.6)
            elif fail > 12 and alpha < 0.04:
                alpha = 0.30

        if alpha < 1e-5:
            alpha = 0.05

        if score > best_score * 1.05 + 0.02 and it > 10:
            a = best_a.copy()
            b = _convolve(a, conv_fft)
            maxb = float(np.max(b)) if b.size else 0.0
            score = 2.0 * float(n) * maxb
            alpha = 0.30
            fail = 0

        if it - last_soft > 1200 and it > 200:
            a = _perturb(best_a, rng, 0.40)
            b = _convolve(a, conv_fft)
            maxb = float(np.max(b)) if b.size else 0.0
            score = 2.0 * float(n) * maxb
            alpha = 0.30
            fail = 0
            target_beta = max(3.0, target_beta * 0.5)

        it += 1

    print(f"refine {label} len={best_a.size} score={best_score:.8f}", flush=True)
    return best_a, best_score


def _polish_topq(a, t_end, rng):
    """Exact top-peak projected-gradient polish for final squeezing."""
    a = _normalize(a)
    n = a.size
    if n < 16 or n > 16000:
        return a, _score(a)

    conv_fft = _next_pow2(2 * n - 1)
    grad_fft = _next_pow2(3 * n - 2)

    b = _convolve(a, conv_fft)
    maxb = float(np.max(b)) if b.size else 0.0
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    alpha = 0.25
    q_frac = 0.08
    fail = 0
    it = 0

    while time.monotonic() < t_end:
        b = _convolve(a, conv_fft)
        m_len = b.size
        if m_len == 0:
            break
        maxb = float(np.max(b))
        score = 2.0 * float(n) * maxb
        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score

        q = max(2, int(q_frac * float(m_len)))
        if q >= m_len:
            w = np.full(m_len, 1.0 / float(m_len), dtype=np.float64)
        else:
            idx = np.argpartition(b, -q)[-q:]
            w = np.zeros(m_len, dtype=np.float64)
            w[idx] = 1.0 / float(q)

        g = _weighted_gradient(a, w, grad_fft)
        gc = g - float(np.mean(g))
        scale = float(np.max(np.abs(gc)))
        if scale < 1e-20:
            a = _perturb(best_a, rng, 0.15)
            b = _convolve(a, conv_fft)
            maxb = float(np.max(b)) if b.size else 0.0
            score = 2.0 * float(n) * maxb
            alpha = 0.25
            fail = 0
            it += 1
            continue

        accepted = False
        for _try in range(4):
            cand = _project(a - alpha * gc, True)
            b2 = _convolve(cand, conv_fft)
            maxb2 = float(np.max(b2)) if b2.size else 0.0
            score2 = 2.0 * float(n) * maxb2
            if np.isfinite(score2) and score2 < score - 1e-15:
                accepted = True
                a = cand
                score = score2
                break
            alpha *= 0.6

        if accepted:
            fail = 0
            alpha = min(0.8, alpha * 1.05)
            if it % 150 == 0:
                q_frac = max(0.012, q_frac * 0.95)
        else:
            alpha *= 0.6
            fail += 1
            if fail > 25:
                a = _perturb(best_a, rng, 0.15)
                b = _convolve(a, conv_fft)
                maxb = float(np.max(b)) if b.size else 0.0
                score = 2.0 * float(n) * maxb
                alpha = 0.25
                fail = 0
                q_frac = max(0.02, q_frac * 1.2)

        it += 1

    return best_a, best_score


def _local_moving(a, t_end, rng):
    """
    Lightweight greedy mass-transfer search for small lengths.
    Moves a fraction from a high-gradient coordinate to a low-gradient one.
    """
    a = _normalize(a)
    n = a.size
    if n < 16:
        return a, _score(a)
    if n > 4096:
        return a, _score(a)

    conv_fft = _next_pow2(2 * n - 1)
    grad_fft = _next_pow2(3 * n - 2)

    b = _convolve(a, conv_fft)
    maxb = float(np.max(b)) if b.size else 0.0
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    fail = 0
    it = 0

    while time.monotonic() < t_end:
        b = _convolve(a, conv_fft)
        m_len = b.size
        if m_len == 0:
            break
        maxb = float(np.max(b))
        score = 2.0 * float(n) * maxb
        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score

        q = max(4, min(m_len, int(0.05 * float(m_len))))
        if q >= m_len:
            w = np.full(m_len, 1.0 / float(m_len), dtype=np.float64)
        else:
            idx = np.argpartition(b, -q)[-q:]
            w = np.zeros(m_len, dtype=np.float64)
            w[idx] = 1.0 / float(q)

        g = _weighted_gradient(a, w, grad_fft)
        high_count = min(12, n)
        low_count = min(12, n)
        high_order = np.argsort(g)[-high_count:]
        low_order = np.argsort(g)[:low_count]

        improved = False
        for _rep in range(3):
            p = int(high_order[int(rng.integers(0, high_count))])
            qidx = int(low_order[int(rng.integers(0, low_count))])
            if p == qidx:
                continue
            if a[p] <= 1e-18:
                continue

            accepted_move = False
            for frac in (0.25, 0.10, 0.05, 0.45, 0.02):
                delta = float(frac) * float(a[p])
                if delta <= 0.0:
                    continue
                a2 = a.copy()
                a2[p] -= delta
                a2[qidx] += delta
                a2 = np.maximum(a2, 0.0)
                s2 = float(a2.sum())
                if s2 <= 1e-300:
                    continue
                a2 = a2 / s2
                b2 = _convolve(a2, conv_fft)
                maxb2 = float(np.max(b2)) if b2.size else 0.0
                sc2 = 2.0 * float(n) * maxb2
                if np.isfinite(sc2) and sc2 < score - 1e-14:
                    a = a2
                    score = sc2
                    if score < best_score - 1e-14:
                        best_a = a.copy()
                        best_score = score
                    accepted_move = True
                    improved = True
                    break

        if not improved:
            fail += 1
            if fail > 40:
                a = _perturb(best_a, rng, 0.15)
                b = _convolve(a, conv_fft)
                maxb = float(np.max(b)) if b.size else 0.0
                score = 2.0 * float(n) * maxb
                fail = 0

        if it % 300 == 0:
            sym = _symmetrize(a)
            bs = _convolve(sym, conv_fft)
            maxs = float(np.max(bs)) if bs.size else 0.0
            scs = 2.0 * float(n) * maxs
            if scs < score:
                a = sym
                score = scs
                if scs < best_score:
                    best_a = sym.copy()
                    best_score = scs

        it += 1

    return best_a, best_score


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Entry point. Searches for at most budget_s seconds and returns a list of
    nonnegative floats minimizing the evaluator.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 5.0:
        budget = 5.0

    start = time.monotonic()
    reserve = min(3.0, 0.05 * budget)
    deadline = start + max(1.0, budget - reserve)
    rng = np.random.default_rng(seed)

    given = _load_given()

    if given is not None:
        best_a = _normalize(given)
        best_score = _score(best_a)
    else:
        best_a = _make_power(256, 0.5)
        best_score = _score(best_a)

    print(f"initial best len={best_a.size} score={best_score:.8f}", flush=True)

    if not np.isfinite(best_score) or best_score > 2.0:
        best_a = _make_power(256, 0.5)
        best_score = _score(best_a)
        print(f"fallback best len={best_a.size} score={best_score:.8f}", flush=True)

    # Candidate pool lengths. Keep to tractable sizes for 1 CPU.
    pool_lengths = [256, 512, 1024, 2048]
    if budget >= 1500:
        pool_lengths.append(4096)

    pools = {}
    if time.monotonic() < deadline - 80.0:
        pools = _build_pools(pool_lengths, given, best_a, rng, deadline)
        for n in pools:
            if pools[n]:
                sc = _score(pools[n][0])
                if sc < best_score:
                    best_a = pools[n][0].copy()
                    best_score = sc
                    print(f"pool improved len={best_a.size} score={best_score:.8f}", flush=True)

    # Choose work scale.
    scale = min(1.0, max(0.25, (budget - max(10.0, budget * 0.08) - 15.0) / 2400.0))
    if scale < 0.05:
        scale = 0.05

    reserve_final = max(15.0, min(220.0, 0.085 * budget))
    search_deadline = deadline - reserve_final
    if search_deadline < time.monotonic() + 5.0:
        search_deadline = deadline - 2.0

    if budget >= 1800:
        stages = [
            (256, 8, 135),
            (512, 5, 125),
            (1024, 3, 120),
            (2048, 2, 100),
            (4096, 1, 80),
        ]
    elif budget >= 900:
        stages = [
            (256, 6, 150),
            (512, 4, 140),
            (1024, 2, 120),
            (2048, 1, 90),
        ]
    else:
        stages = [
            (256, 3, 120),
            (512, 2, 90),
        ]

    # If we already have a very good best, still give 256 extra time.
    if best_score < 1.53 and time.monotonic() < search_deadline - 60.0:
        stages = [(256, 10, int(stages[0][1] * 0.9 + stages[0][2]))] + stages[1:]

    for n, count, sec in stages:
        n = int(n)
        if n < 16 or n > 8192:
            continue
        if time.monotonic() > search_deadline - 8.0:
            break

        sec = max(6, int(float(sec) * scale))
        t_start = time.monotonic()
        starts = _get_starts(n, pools, given, best_a, rng)

        for idx in range(min(int(count), max(1, len(starts)))):
            if time.monotonic() > search_deadline - 5.0:
                break
            start_a = starts[idx % len(starts)].copy()
            if idx >= len(starts):
                start_a = _resample(best_a, n)
                start_a = _perturb(start_a, rng, 0.35)

            # For repeated starts, deliberately diversify.
            if idx > 0 and (len(starts) <= count) and rng.random() < 0.55:
                start_a = _perturb(start_a, rng, 0.18 + 0.08 * min(4, idx))

            t_end = min(search_deadline - 1.0, time.monotonic() + float(sec))
            if time.monotonic() >= t_end:
                continue

            res_a, res_score = _refine_sequence(start_a, t_end, rng, label=f"n={n}#{idx}")
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                print(f"improved n={best_a.size} score={best_score:.8f} t={time.monotonic()-start:.1f}s", flush=True)

        if time.monotonic() > search_deadline - 3.0:
            break

    # Final reserved phase: deepen the current best, then polish/squeeze.
    remaining = deadline - time.monotonic()
    if remaining > 10.0 and best_a.size <= 8192:
        t_refine_end = min(deadline - max(10.0, remaining * 0.35), time.monotonic() + max(10.0, remaining * 0.55))
        if t_refine_end > time.monotonic():
            res_a, res_score = _refine_sequence(best_a.copy(), t_refine_end, rng, label="final-refine")
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                print(f"final refine improved len={best_a.size} score={best_score:.8f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 15.0 and best_a.size <= 2048:
        t_local_end = min(deadline - 8.0, time.monotonic() + max(5.0, remaining * 0.35))
        if t_local_end > time.monotonic():
            res_a, res_score = _local_moving(best_a.copy(), t_local_end, rng)
            if np.isfinite(res_score) and res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                print(f"local moving improved len={best_a.size} score={best_score:.8f}", flush=True)

    remaining = deadline - time.monotonic()
    if remaining > 8.0 and best_a.size <= 8192:
        res_a, res_score = _polish_topq(best_a.copy(), deadline - 1.0, rng)
        if np.isfinite(res_score) and res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)
            print(f"topq polish improved len={best_a.size} score={best_score:.8f}", flush=True)

    # Try one last symmetrization if it helps.
    sym = _symmetrize(best_a)
    sym_score = _score(sym)
    if np.isfinite(sym_score) and sym_score < best_score:
        best_a = sym
        best_score = sym_score
        print(f"final sym improved len={best_a.size} score={best_score:.8f}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        out = _make_uniform(256)
        s = float(out.sum())
    out = out / s

    result = [float(x) for x in out]

    # Optional final oracle check, if already available in the environment.
    eval_func = globals().get("evaluate_sequence", None)
    if callable(eval_func):
        try:
            oracle = float(eval_func(result))
            print(f"final internal={best_score:.8f} oracle={oracle:.8f} len={len(result)}", flush=True)
        except Exception:
            pass

    return result
```
