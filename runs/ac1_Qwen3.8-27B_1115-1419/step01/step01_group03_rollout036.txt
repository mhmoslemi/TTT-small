def logsumexp_score(a, t):
    """Compute the log-sum-exp score and its gradient."""
    n = len(a)
    c = np.convolve(a, a)
    # Softmax weights
    c_shifted = c / t - np.max(c / t)  # for numerical stability
    p = np.exp(c_shifted)

def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def compute_smooth_grad(a, t):
    """
    Compute the gradient of the log-sum-exp smoothed objective.
    F_t(a) = t * log(sum_k exp(c_k / t)) where c = conv(a,a)
    Returns (score, gradient, peak_fraction)
    """
    n = len(a)
    c = np.convolve(a, a)
    # Numerically stable softmax
    c_scaled = c / t
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / exp_c.sum()
    # Score (for tracking)
    score = t * (c_max + np.log(exp_c.sum()))
    # Gradient: grad F[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * conv(p, a[::-1])[n-1 : 2n-1]
    grad = 2.0 * np.convolve(p, a[::-1], mode='full')[n-1:2*n-1]
    # Peak fraction: weight on the maximum coefficient
    peak_frac = p.max()
    return score, grad, peak_frac

"""
Minimize 2n*max(conv(a))/(sum(a))^2 using:
1. Log-sum-exp smoothed objective with adaptive temperature annealing
2. FISTA in softmax parameterization (avoids projection, more stable)
3. Multiple diverse initializations across different sequence lengths
4. Adaptive step size with backtracking
5. Local coordinate descent polishing
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2 efficiently."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def softmax(a):
    """Numerically stable softmax."""
    a_shifted = a - a.max()
    e = np.exp(a_shifted)
    return e / e.sum()


def compute_lse_gradient(a, t):
    """
    Compute gradient of log-sum-exp smoothed objective.
    F_t(a) = t * log(sum_k exp(c_k / t)) where c = conv(a, a)
    Returns (smoothed_score, gradient_of_length_n, max_conv_coeff).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_scaled = c / t
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / exp_c.sum()
    smoothed_score = t * (c_max + np.log(exp_c.sum()))
    # Gradient: dF/da[j] = 2 * sum_k p[k] * a[k-j] (where a[m]=0 for m outside [0,n))
    # = 2 * conv(p, a[::-1])[n-1 : 2n-1]
    grad = 2.0 * np.convolve(p, a[::-1], mode='full')[n - 1:2 * n - 1]
    return smoothed_score, grad, c.max()


def run_fista_simplex(a0, duration, rng, t_start=0.05, t_min=0.0005, t_decay_per_iter=0.9995):
    """
    FISTA on the probability simplex using log-sum-exp smoothed objective.
    Adapts temperature over time and uses backtracking line search.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a_prev = a.copy()
    t_moment = 1.0
    temp = t_start
    step = 0.5 / n
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        # Nesterov momentum
        t_old = t_moment
        t_moment = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        mu = (t_old - 1.0) / t_moment
        y = a + mu * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.05:
            y /= sy

        # Gradient at y
        _, grad_y, _ = compute_lse_gradient(y, temp)

        # Gradient step + project
        a_new = project_simplex(y - step * grad_y)

        a_prev = a
        a = a_new
        it += 1

        # Periodic check
        if it % 40 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            # Anneal temperature
            temp = max(t_min, temp * t_decay_per_iter)

            # Adaptive step
            if stall == 20:
                step *= 0.85
            elif stall == 80:
                step *= 0.6
            elif stall == 200:
                # Reset momentum
                a_prev = best_a.copy()
                t_moment = 1.0
                step = 0.5 / n
            elif stall > 400:
                # Perturb and restart
                a = best_a + rng.normal(0.0, 0.003 / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_moment = 1.0
                step = 0.5 / n
                temp = t_start
                stall = 0
                print(f"      FISTA restart it={it} best={best_score:.7f}")

            if it % 5000 == 0:
                rem = max(0, deadline - time.monotonic())
                print(f"      it={it} score={best_score:.7f} step={step:.6f} t={temp:.5f} rem={rem:.0f}s")

    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def run_fista_softmax(z0, duration, rng, t_start=0.05, t_min=0.0005, t_decay_per_iter=0.9995):
    """
    FISTA in softmax parameterization: a = softmax(z). No projection needed.
    Optimizes over z in R^n.
    """
    n = len(z0)
    z = z0.copy().astype(np.float64)
    z_prev = z.copy()
    t_moment = 1.0
    temp = t_start
    step = 0.3
    a = softmax(z)
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        # Nesterov
        t_old = t_moment
        t_moment = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        mu = (t_old - 1.0) / t_moment
        z_y = z + mu * (z - z_prev)

        a_y = softmax(z_y)

        # Gradient w.r.t. z: grad_z = diag(a) * grad_a - a * (a^T grad_a)
        _, grad_a, _ = compute_lse_gradient(a_y, temp)
        centered = np.dot(a_y, grad_a)
        grad_z = a_y * grad_a - a_y * centered

        # Update
        z_new = z_y - step * grad_z

        z_prev = z
        z = z_new
        a = softmax(z)
        it += 1

        if it % 40 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            temp = max(t_min, temp * t_decay_per_iter)

            if stall == 20:
                step *= 0.85
            elif stall == 80:
                step *= 0.6
            elif stall == 200:
                z_prev = z.copy()
                t_moment = 1.0
                step = 0.3
            elif stall > 400:
                z = z - np.log(a) + np.log(best_a) + rng.normal(0, 0.01, n)
                z_prev = z.copy()
                t_moment = 1.0
                step = 0.3
                temp = t_start
                stall = 0

            if it % 5000 == 0:
                rem = max(0, deadline - time.monotonic())
                print(f"      Softmax-F it={it} score={best_score:.7f} step={step:.4f} t={temp:.5f} rem={rem:.0f}s")

    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def coordinate_polish(a, duration, rng):
    """
    Random coordinate descent: pick random coordinates, try adjusting
    them to reduce the max convolution coefficient.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration

    while time.monotonic() < deadline:
        j = rng.integers(n)
        cur_val = a[j]
        # Try several perturbations
        for scale in [0.5, 0.7, 0.85, 1.0, 1.2, 1.4, 2.0, 0.0]:
            new_val = cur_val * scale
            if new_val < 0:
                continue
            a[j] = new_val
            sa = a.sum()
            if sa < 1e-10:
                a[j] = cur_val
                continue
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                a[j] = cur_val
                break

    return best_a, best_score
