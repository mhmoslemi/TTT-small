```python
import numpy as np
import time
from scipy.optimize import linprog

def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, 1.0/n)
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1)
    return np.maximum(v - theta, 0.0)

def solve_lp(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Variables: b_0, ..., b_{n-1}
    Constraints: for each k in 0..2n-2: sum_i a_i * b_{k-i} <= M
                (where b_{k-i} = 0 if k-i not in [0, n-1])
    """
    n = len(a)
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub @ b <= b_ub
    # (a*b)[k] = sum_{i: 0<=k-i<=n-1} a[i] * b[k-i]
    # For a given k, the coefficient of b_j in (a*b)[k] is a[k-j] if 0<=k-j<=n-1, else 0
    # So row k of A_ub has: A_ub[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # This is a (2n-1) x n matrix. For n=4806, that's 9611 x 4806.
    # We can build it efficiently.
    
    # Actually, for the LP, we might not need all constraints. The most binding ones
    # are near the center (where convolution peaks tend to be).
    
    # Let's build the full matrix for now.
    # For efficiency, use the structure: row k is a shifted version of a.
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = k - np.arange(lo, hi)
        A_ub[k, lo:hi] = a[idx]
    
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='hihps')
    
    if res.success:
        return -res.fun, res.x
    else:
        return None, None
```

```python
def build_convolution_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    # For each k (0 to 2n-2), the non-zero entries are at j in [max(0,k-n+1), min(n-1,k)]
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        js = np.arange(lo, hi)
        vals_k = a[k - js]
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(vals_k.tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
def run_fista(a0, n, duration, best_a, best_score):
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = 0.3 / np.sqrt(n)  # initial step size
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    
    while time.monotonic() < deadline:
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.05:
            y = y / sy
        
        # Compute convolution and find top-k peaks
        c_y = np.convolve(y, y)
        k_max = c_y.argmax()
        
        # Subgradient from the max peak
        g = subgrad_k(y, int(k_max))
        
        # Gradient step
        a_new = y - step * g
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic check
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck > 100:
                step *= 0.9
            if stuck > 500:
                a = best_a + np.random.normal(0, 0.001, n)
                np.maximum(a, 0, out=a)
                a /= a.sum()
                a_prev = a.copy()
                t = 1.0
                step = 0.3 / np.sqrt(n)
                stuck = 0
            
            if it % 500 == 0:
                print(f"  FISTA it={it}, score={best_score:.8f}, step={step:.6f}")
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_a.copy(), best_score
```

def project_simplex(v):
    n = len(v)
    if n <= 1:
        return np.ones(n) * max(v.sum(), 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    w = np.arange(1, n + 1)
    cond = u * w > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
