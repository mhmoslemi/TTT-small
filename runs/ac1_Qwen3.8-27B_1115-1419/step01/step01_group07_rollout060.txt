"""
Search for nonnegative sequences minimizing

    2*n*max(convolve(a,a))/(sum(a))^2

using a multi-stage smooth log-sum-exp optimization in softmax coordinates,
followed by exact-peak mirror descent, a dual-safe Frank-Wolfe-style kick,
and several derived/secondary restarts.  All candidates are tracked with a
fast FFT scoring proxy and the top candidates are finally checked with the
provided evaluate_sequence().
"""

import numpy as np
from scipy.optimize import minimize

# Globals used by the top-level objective/gradient functions required by
# scipy.minimize.  They are reassigned before each minimization stage.
G_N = 1
G_L = 2
G_TAU = 1e-6
G_TOP = 0
G_DEADLINE = 0.0

# Stage lists: (tau_factor, top_k, maxfun)
# top_k = 0 means full log-sum-exp over all convolution coefficients.

def clean_sequence(seq, normalize=True):
    """Convert input to a finite nonnegative float array, optionally normalized."""
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        arr = np.ones(64, dtype=np.float64)

    if arr.size == 0:
        arr = np.ones(64, dtype=np.float64)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1000.0)

    if normalize:
        s = float(arr.sum())
        if s < 1e-15:
            arr = np.ones(arr.size, dtype=np.float64) / float(arr.size)
        else:
            arr = arr / s
    return arr


def resize_sequence(seq, n):
    """Resize a sequence to length n while preserving shape as much as possible."""
    arr = clean_sequence(seq, normalize=False)
    if arr.size == n:
        s = float(arr.sum())
        if s > 1e-15:
            arr = arr / s
        else:
            arr = np.ones(n, dtype=np.float64) / float(n)
        return arr

    if arr.size == 0 or n <= 0:
        return np.ones(max(1, int(n)), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, arr.size)
    x_new = np.linspace(0.0, 1.0, int(n))
    out = np.interp(x_new, x_old, arr)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        return np.ones(int(n), dtype=np.float64) / float(int(n))
    return out / s


def fft_len_for(n):
    """FFT length sufficient for convolution and required correlation."""
    need = max(1, 3 * int(n) - 2)
    L = 1
    while L < need:
        L <<= 1
    return L


def conv_full_fft(a, L):
    """Full linear convolution via real FFT."""
    n = len(a)
    if L < 2 * n - 1:
        L = fft_len_for(n)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = np.where(c > 0.0, c, 0.0)
    return c


def corr_p_w(p, w, L):
    """
    Return h[t] = sum_j p[j] * w[t + j] for t = 0..len(p)-1.
    This is the valid cross-correlation needed for gradients of convolution.
    """
    n = len(p)
    if L < 3 * n - 2:
        L = fft_len_for(n)
    P = np.fft.rfft(p, L)
    W = np.fft.rfft(w, L)
    h = np.fft.irfft(np.conj(P) * W, L)[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    return h


def score_fft_with_l(p, L):
    """Fast scoring proxy using FFT convolution."""
    n = len(p)
    if n == 0:
        return float("inf")
    s = float(p.sum())
    if s < 1e-15:
        return float("inf")
    c = conv_full_fft(p, L)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def softmax(x):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(x.shape[0], dtype=np.float64) / float(x.shape[0])
    return e / s


def logit_from_p(p, eps=1e-12):
    """Map a nonnegative simplex vector to log coordinates."""
    arr = np.asarray(p, dtype=np.float64).copy()
    arr = np.maximum(arr, eps)
    s = float(arr.sum())
    if s <= 1e-15:
        arr = np.ones_like(arr)
        s = float(arr.sum())
    arr = arr / s
    x = np.log(arr)
    x -= float(np.mean(x))
    return x


def top_lse_value_grad(x):
    """
    Value and gradient of scaled log-sum-exp smoothed max of a*a.
    Uses the current global parameters G_N, G_L, G_TAU, G_TOP.
    Returns raw (without the 2*n objective scale) value and gradient.
    """
    n = G_N
    L = G_L
    tau = G_TAU
    top = G_TOP

    s = softmax(x)
    c = conv_full_fft(s, L)
    m = len(c)

    if top is None or top <= 0 or top >= m:
        vals = c
        maxv = float(np.max(vals))
        e = np.exp((vals - maxv) / tau)
        e_sum = float(e.sum())
        if not np.isfinite(e_sum) or e_sum <= 0.0:
            e = np.ones(m, dtype=np.float64)
            e_sum = float(m)
        val = tau * math.log(e_sum) + maxv
        warn = e / e_sum
    else:
        K = int(min(max(1, top), m))
        idx = np.argpartition(c, m - K)[-K:]
        vals = c[idx]
        maxv = float(np.max(vals))
        e = np.exp((vals - maxv) / tau)
        e_sum = float(e.sum())
        if not np.isfinite(e_sum) or e_sum <= 0.0:
            e = np.ones(K, dtype=np.float64)
            e_sum = float(K)
        val = tau * math.log(e_sum) + maxv
        warn = np.zeros(m, dtype=np.float64)
        warn[idx] = e / e_sum

    g = corr_p_w(s, warn, L)
    grad_a = 2.0 * g
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)

    dot = float(np.dot(s, grad_a))
    grad_x = s * (grad_a - dot)
    grad_x = np.where(np.isfinite(grad_x), grad_x, 0.0)

    if not np.isfinite(val):
        val = maxv
    return float(val), grad_x


def obj_global(x):
    """Objective used by scipy.minimize."""
    val, _ = top_lse_value_grad(x)
    return float(2.0 * G_N * val)


def grad_global(x):
    """Gradient used by scipy.minimize."""
    _, g = top_lse_value_grad(x)
    return 2.0 * float(G_N) * g


def lbfgs_timeout_callback(xk):
    """Stop L-BFGS when the stage/global deadline is passed or x becomes invalid."""
    try:
        if not np.all(np.isfinite(xk)):
            return True
    except Exception:
        pass
    return time.monotonic() > G_DEADLINE


def run_lbfgs_from_p(p, n, L, max_time, stages=None):
    """Run staged L-BFGS smoothed optimization from a simplex vector p."""
    if max_time <= 0.1:
        p = clean_sequence(p, normalize=True)
        return p, score_fft_with_l(p, L), logit_from_p(p)

    if stages is None:
        stages = DEFAULT_STAGES

    deadline = time.monotonic() + float(max_time)
    p = clean_sequence(p, normalize=True)
    if p.size != n:
        p = resize_sequence(p, n)

    best = p.copy()
    best_score = score_fft_with_l(p, L)
    x = logit_from_p(p)

    G_N = n
    G_L = L
    stage_budget = max(20.0, float(max_time) / float(max(1, len(stages))))

    for tau_factor, top_k, maxfun in stages:
        tnow = time.monotonic()
        if tnow >= deadline:
            break

        G_TAU = max(1e-12, float(tau_factor) / float(n))
        G_TOP = int(top_k)
        G_DEADLINE = min(deadline, tnow + stage_budget)

        if not np.all(np.isfinite(x)):
            x = logit_from_p(best)
        x = x - float(np.mean(x))

        try:
            res = minimize(
                obj_global,
                x,
                jac=grad_global,
                method="L-BFGS-B",
                options={
                    "maxiter": 2000,
                    "maxfun": int(maxfun),
                    "maxls": 30,
                    "ftol": 1e-18,
                    "gtol": 1e-10,
                    "m": 30,
                },
                callback=lbfgs_timeout_callback,
            )
            if hasattr(res, "x") and res.x is not None and np.all(np.isfinite(res.x)):
                x = res.x
        except Exception as e:
            print("L-BFGS stage error:", e)

        p_new = softmax(x)
        s_new = score_fft_with_l(p_new, L)

        if np.isfinite(s_new) and s_new < best_score:
            best_score = s_new
            best = p_new.copy()
            print("    L-BFGS improved to", round(best_score, 8))
        elif np.isfinite(s_new) and s_new > best_score * (1.0 + 0.05):
            # Avoid drifting into a much worse region for later stages.
            x = logit_from_p(best)

    return best, best_score, x


def run_mirror_from_p(p, n, L, max_time, tau_factor=0.02, top_k=512, eta0=1.0):
    """Exact-peak mirror descent polish in log coordinates."""
    if max_time <= 0.1:
        p = clean_sequence(p, normalize=True)
        return p, score_fft_with_l(p, L), logit_from_p(p)

    deadline = time.monotonic() + float(max_time)
    p = clean_sequence(p, normalize=True)
    if p.size != n:
        p = resize_sequence(p, n)

    best = p.copy()
    best_score = score_fft_with_l(p, L)
    current = best_score
    x = logit_from_p(p)

    G_N = n
    G_L = L
    G_TAU = max(1e-12, float(tau_factor) / float(n))
    G_TOP = int(top_k)
    G_DEADLINE = deadline

    eta = float(eta0)
    stuck = 0
    total_reset = 0

    while time.monotonic() < deadline:
        if not np.all(np.isfinite(x)):
            break

        _, grad = top_lse_value_grad(x)
        grad = 2.0 * float(n) * grad
        if not np.all(np.isfinite(grad)):
            break

        if float(np.linalg.norm(grad)) < 1e-18:
            break

        alpha = eta
        accepted = False
        for _ in range(7):
            x_new = x - alpha * grad
            x_new = x_new - float(np.mean(x_new))
            p_new = softmax(x_new)
            s_new = score_fft_with_l(p_new, L)
            if np.isfinite(s_new) and s_new < current - 1e-15:
                x = x_new
                current = s_new
                best_score = s_new
                best = p_new.copy()
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            stuck = 0
            if eta < 25.0:
                eta *= 1.03
        else:
            stuck += 1
            eta *= 0.75
            if stuck > 20:
                stuck = 0
                G_TAU *= 2.0
                if G_TOP == 0:
                    G_TOP = 512
                else:
                    G_TOP = max(128, int(G_TOP) // 2)
                total_reset += 1
                if total_reset > 10:
                    break
            if eta < 1e-6:
                break

    return best, best_score, x


def run_dualsafe_from_p(p, n, L, max_time, top_k=256, tau_factor=0.2, eta0=2.0, alpha0=0.05):
    """
    Frank-Wolfe-style kick: weight current peaks, move mass to positions with
    low sensitivity to those peaks, and accept only exact improvements.
    """
    if max_time <= 0.1:
        p = clean_sequence(p, normalize=True)
        return p, score_fft_with_l(p, L)

    deadline = time.monotonic() + float(max_time)
    p = clean_sequence(p, normalize=True)
    if p.size != n:
        p = resize_sequence(p, n)

    best = p.copy()
    best_score = score_fft_with_l(p, L)
    current = best_score
    uniform = np.ones(n, dtype=np.float64) / float(n)

    tau = max(1e-12, float(tau_factor) / float(n))
    eta = float(eta0)
    alpha = float(alpha0)
    top = int(max(1, min(top_k, 2 * n - 1)))
    stuck = 0
    total_reset = 0

    while time.monotonic() < deadline:
        c = conv_full_fft(p, L)
        sc = float(2.0 * n * np.max(c))
        if not np.isfinite(sc):
            break
        current = sc
        if current < best_score:
            best_score = current
            best = p.copy()

        m = len(c)
        K = int(min(max(1, top), m))
        if K >= m:
            idx = np.arange(m)
            vals = c
        else:
            idx = np.argpartition(c, m - K)[-K:]
            vals = c[idx]

        vmax = float(np.max(vals))
        e = np.exp((vals - vmax) / tau)
        es = float(e.sum())
        if not np.isfinite(es) or es <= 0.0:
            e = np.ones(K, dtype=np.float64)
            es = float(K)

        warn = np.zeros(m, dtype=np.float64)
        warn[idx] = e / es

        h = corr_p_w(p, warn, L)
        hmin = float(np.min(h))
        hrange = max(float(np.max(h)) - hmin, 1e-15)
        q = softmax(-eta * (h - hmin) / hrange)
        q = 0.90 * q + 0.10 * uniform

        accepted = False
        for a in (alpha, 0.3 * alpha):
            p_new = (1.0 - a) * p + a * q
            s = float(p_new.sum())
            if s <= 1e-15:
                continue
            p_new = p_new / s
            s_new = score_fft_with_l(p_new, L)
            if np.isfinite(s_new) and s_new < current - 1e-16:
                p = p_new
                current = s_new
                if current < best_score:
                    best_score = current
                    best = p.copy()
                accepted = True
                break

        if accepted:
            stuck = 0
            if alpha < 0.20:
                alpha *= 1.10
        else:
            alpha *= 0.8
            if alpha < 1e-4:
                stuck += 1
                alpha = float(alpha0)
                tau *= 1.5
                eta *= 0.8
                top = min(int(top) * 2, 4096)
                total_reset += 1
                if total_reset > 15:
                    break

    return best, best_score


def downsample_avg(p, factor):
    """Average-resample a sequence to a shorter length."""
    arr = np.asarray(p, dtype=np.float64).ravel()
    n = arr.size
    m = n // max(1, int(factor))
    if m < 16:
        return arr.copy()

    if n % int(factor) == 0:
        q = arr.reshape(m, int(factor)).mean(axis=1)
    else:
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, m)
        q = np.interp(x_new, x_old, arr)

    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-15:
        return np.ones(m, dtype=np.float64) / float(m)
    return q / s


def try_trim_endpoints(p, max_time=20.0):
    """Greedy endpoint trimming if it reduces the exact FFT score."""
    arr = np.asarray(p, dtype=np.float64).copy()
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s > 1e-15:
        arr = arr / s
    else:
        return arr

    n = arr.size
    if n < 256:
        return arr

    L = fft_len_for(n)
    best = score_fft_with_l(arr, L)
    deadline = time.monotonic() + float(max_time)

    while n > 128 and time.monotonic() < deadline:
        if arr[0] <= arr[-1]:
            cand = arr[1:].copy()
        else:
            cand = arr[:-1].copy()

        if cand.size < 128:
            break
        cs = float(cand.sum())
        if cs <= 1e-15:
            break
        cand = cand / cs
        Lc = fft_len_for(cand.size)
        sc = score_fft_with_l(cand, Lc)

        if np.isfinite(sc) and sc < best - 1e-14:
            arr = cand
            n = cand.size
            L = Lc
            best = sc
        else:
            break

    return arr


def add_candidate(candidates, p, name, score=None):
    """Add a normalized candidate to the tracking list, capping list size."""
    try:
        arr = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return

    if arr.size < 1:
        return

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s < 1e-15:
        return
    arr = arr / s

    if score is None:
        L = fft_len_for(arr.size)
        score = score_fft_with_l(arr, L)

    if not np.isfinite(score):
        return

    candidates.append((float(score), len(candidates), str(name), arr.copy()))

    if len(candidates) > 200:
        worst = 0
        ws = candidates[0][0]
        for i in range(1, len(candidates)):
            if candidates[i][0] > ws:
                ws = candidates[i][0]
                worst = i
        del candidates[worst]


def add_seed(p, name, seeds, candidates, counter, L):
    """Create a normalized seed, add it to candidates and the seed ranking."""
    try:
        arr = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return counter

    if arr.size == 0:
        return counter

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-15:
        return counter

    arr = arr / s
    sc = score_fft_with_l(arr, L)
    if np.isfinite(sc):
        add_candidate(candidates, arr, name, sc)
        seeds.append((sc, counter, str(name), arr.copy()))
        counter += 1
    return counter


def exact_score_numpy(p):
    """Call the evaluator on a numpy array and guard exceptions."""
    try:
        seq = [float(v) for v in np.asarray(p, dtype=np.float64).ravel()]
        v = evaluate_sequence(seq)
        return float(v)
    except Exception:
        return float("inf")


def propose_candidate(budget_s=2400.0, **kwargs):
    """
    Main search entry point.

    Returns a list of nonnegative floats.  It tracks multiple candidates,
    optimizes the best candidate with several complementary methods, and
    finally returns the candidate with the lowest exact evaluate_sequence score.
    """
    global G_N, G_L, G_TAU, G_TOP, G_DEADLINE

    t0 = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 2400.0
    if total <= 120.0:
        total = 120.0
    deadline = t0 + total - 45.0

    rng = np.random.default_rng(12345)
    candidates = []
    seeds = []
    counter = 0

    # Primary length: prefer the supplied construction length, with safety caps.
    seq1 = globals().get("height_sequence_1", None)
    n_primary = 4806
    if seq1 is not None:
        try:
            arr0 = np.asarray(seq1, dtype=np.float64)
            if arr0.size > 0:
                n_primary = int(arr0.size)
        except Exception:
            pass
    if n_primary < 128:
        n_primary = 4806
    if n_primary > 9612:
        n_primary = 4806

    L = fft_len_for(n_primary)
    print(f"Search length: {n_primary}, FFT length: {L}")

    # Build diverse seeds.
    if seq1 is not None:
        p = resize_sequence(seq1, n_primary)
        counter = add_seed(p, "height", seeds, candidates, counter, L)

    counter = add_seed(np.ones(n_primary, dtype=np.float64) / float(n_primary),
                       "uniform", seeds, candidates, counter, L)

    idx = np.arange(n_primary, dtype=np.float64)

    # Noise / Dirichlet / random smooth seeds.
    for alpha in (0.2, 0.5, 0.8):
        noise = 1.0 + alpha * (rng.random(n_primary) - 0.5)
        counter = add_seed(noise, f"noise_{alpha}", seeds, candidates, counter, L)

    for alpha in (0.25, 0.6, 1.2, 2.5):
        d = rng.dirichlet(np.full(n_primary, alpha, dtype=np.float64))
        counter = add_seed(d, f"dirichlet_{alpha}", seeds, candidates, counter, L)

    r = np.exp(0.5 * rng.normal(0.0, 1.0, size=n_primary))
    counter = add_seed(r, "randexp", seeds, candidates, counter, L)

    # Low dimensional smooth modes.
    for freq in (1, 2, 3, 5):
        for phase in (0.0, np.pi / 3.0, 2.0 * np.pi / 3.0):
            for amp in (0.25, 0.5, 0.8):
                p = 1.0 + amp * np.cos(2.0 * np.pi * freq * idx / float(n_primary) + phase)
                counter = add_seed(p, f"cos_{freq}_{amp}_{phase:.2f}", seeds, candidates, counter, L)

    for period in (2, 3, 4, 5):
        for amp in (0.25, 0.55, 0.85):
            p = 1.0 + amp * np.sin(2.0 * np.pi * idx / float(period))
            counter = add_seed(p, f"period_{period}_{amp}", seeds, candidates, counter, L)

    # Edge emphasis.
    for amp in (0.4, 0.8, 1.2):
        p = 1.0 + amp * np.cos(np.pi * idx / float(max(1, n_primary - 1)))
        p = np.maximum(p, 1e-6)
        counter = add_seed(p, f"edge_{amp}", seeds, candidates, counter, L)

    if not seeds:
        p = np.ones(n_primary, dtype=np.float64) / float(n_primary)
        counter = add_seed(p, "fallback", seeds, candidates, counter, L)

    if not seeds:
        return [1.0]

    seeds.sort()
    best_p = np.asarray(seeds[0][3], dtype=np.float64).copy()
    best_score = float(seeds[0][0])
    print(f"Best seed score: {best_score:.8f} ({seeds[0][2]})")
    add_candidate(candidates, best_p, "seed_best", best_score)

    # Phase 1: main staged L-BFGS on the best seed.
    remaining = deadline - time.monotonic()
    phase1 = max(0.0, min(1000.0, remaining * 0.50))
    if phase1 > 30.0:
        p1, s1, x1 = run_lbfgs_from_p(best_p, n_primary, L, phase1, DEFAULT_STAGES)
        add_candidate(candidates, p1, "primary_lbfgs", s1)
        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_p = p1
            print("After primary L-BFGS:", round(best_score, 8))

    # Symmetrization and derived shorter candidates.
    p_sym = 0.5 * (best_p + best_p[::-1])
    L_sym = fft_len_for(p_sym.size)
    s_sym = score_fft_with_l(p_sym, L_sym)
    add_candidate(candidates, p_sym, "primary_sym", s_sym)
    if np.isfinite(s_sym) and s_sym < best_score:
        best_score = s_sym
        best_p = p_sym
        print("Symmetrization improved:", round(best_score, 8))

    for fac in (2, 3):
        p_ds = downsample_avg(best_p, fac)
        L_ds = fft_len_for(p_ds.size)
        s_ds = score_fft_with_l(p_ds, L_ds)
        add_candidate(candidates, p_ds, f"downsample_{fac}", s_ds)
        if np.isfinite(s_ds) and s_ds < best_score:
            best_score = s_ds
            best_p = p_ds
            print(f"Downsample {fac} improved:", round(best_score, 8))

    # Phase 2: dual-safe exact-peak move.
    remaining = deadline - time.monotonic()
    phase2 = max(0.0, min(220.0, remaining * 0.14))
    if phase2 > 20.0:
        n2 = best_p.size
        L2 = fft_len_for(n2)
        p2, s2 = run_dualsafe_from_p(best_p, n2, L2, phase2,
                                     top_k=256, tau_factor=0.2, eta0=2.0, alpha0=0.05)
        add_candidate(candidates, p2, "dualsafe", s2)
        if np.isfinite(s2) and s2 < best_score:
            best_score = s2
            best_p = p2
            print("After dual-safe:", round(best_score, 8))

    # Phase 3: low-temperature L-BFGS polish.
    remaining = deadline - time.monotonic()
    phase3 = max(0.0, min(520.0, remaining * 0.30))
    if phase3 > 30.0:
        n3 = best_p.size
        L3 = fft_len_for(n3)
        p3, s3, x3 = run_lbfgs_from_p(best_p, n3, L3, phase3, LOW_STAGES)
        add_candidate(candidates, p3, "low_lbfgs", s3)
        if np.isfinite(s3) and s3 < best_score:
            best_score = s3
            best_p = p3
            print("After low L-BFGS:", round(best_score, 8))

    # Phase 4: mirror descent polish.
    remaining = deadline - time.monotonic()
    phase4 = max(0.0, min(260.0, remaining * 0.22))
    if phase4 > 20.0:
        n4 = best_p.size
        L4 = fft_len_for(n4)
        p4, s4, x4 = run_mirror_from_p(best_p, n4, L4, phase4,
                                       tau_factor=0.02, top_k=512, eta0=1.0)
        add_candidate(candidates, p4, "mirror", s4)
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_p = p4
            print("After mirror polish:", round(best_score, 8))

    # Phase 5: secondary restarts from other strong initial seeds.
    for k in (1, 2, 3):
        remaining = deadline - time.monotonic()
        if remaining < 180.0 or k >= len(seeds):
            break
        seed_score, _, _name, sp = seeds[k]
        if seed_score > best_score + 0.25:
            continue
        budget_sec = min(220.0, remaining * 0.40)
        if budget_sec <= 30.0:
            break
        n5 = sp.size
        L5 = fft_len_for(n5)
        p5, s5, x5 = run_lbfgs_from_p(sp, n5, L5, budget_sec, SHORT_STAGES)
        add_candidate(candidates, p5, f"secondary_{k}", s5)
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_p = p5
            print(f"Secondary {k} improved:", round(best_score, 8))

    # Final small trim attempt on the current best.
    remaining = deadline - time.monotonic()
    if remaining > 25.0:
        p_trim = try_trim_endpoints(best_p, max_time=min(25.0, remaining - 15.0))
        L_tr = fft_len_for(p_trim.size)
        s_tr = score_fft_with_l(p_trim, L_tr)
        add_candidate(candidates, p_trim, "trimmed", s_tr)
        if np.isfinite(s_tr) and s_tr < best_score:
            best_score = s_tr
            best_p = p_trim
            print("Trimmed endpoint improved:", round(best_score, 8))

    # Final guard: ensure a valid best is available.
    if best_p is None or len(best_p) == 0 or not np.isfinite(best_score):
        best_p = clean_sequence(best_p if best_p is not None else [1.0], normalize=True)
        best_score = score_fft_with_l(best_p, fft_len_for(len(best_p)))
        add_candidate(candidates, best_p, "final_fallback", best_score)

    # Sort candidates by FFT proxy score, then query the real evaluator on top ones.
    candidates.sort()
    exact_best_score = float("inf")
    exact_best_seq = best_p

    limit = min(40, len(candidates))
    for i in range(limit):
        sc, _, _name, arr = candidates[i]
        if not np.isfinite(sc):
            continue
        es = exact_score_numpy(arr)
        if np.isfinite(es) and es < exact_best_score:
            exact_best_score = es
            exact_best_seq = arr.copy()
        if time.monotonic() > deadline - 5.0:
            break

    if exact_best_seq is None or len(exact_best_seq) == 0:
        exact_best_seq = best_p

    # Final clean normalized return, ensuring sum >= 0.01 and finite nonnegative.
    out = clean_sequence(exact_best_seq, normalize=True)
    if out.size == 0:
        out = np.ones(64, dtype=np.float64) / 64.0

    final_list = [float(v) for v in out]
    return final_list
