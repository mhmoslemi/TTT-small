def _conv(a):

def _top_k_grad(c, a, n, K):
    """Compute subgradient using top K convolution lags."""
    idx = np.argsort(c)[-K:]
    weights = c[idx]
    weights = weights / weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            ai = k - np.arange(js, je)
            g[js:je] += w * 2.0 * a[ai]
    return g

def _adam_step(state, g, lr, t):
    """Adam optimizer step."""
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = state['m']
    v = state['v']
    m = beta1 * m + (1-beta1) * g
    v = beta2 * v + (1-beta2) * g**2
    m_hat = m / (1 - beta1**(t+1))
    v_hat = v / (1 - beta2**(t+1))
    state['m'] = m
    state['v'] = v
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return step

    sa = a.sum()
    if sa < 1e-10:

def _freq_perturb(a, rng, strength):
    """Perturb in frequency domain: randomize phases, re-IFFT, clip, project."""

    return np.convolve(a, a)


def _proj(a, s):
    a = np.maximum(a, 0.0)
    t = a.sum()
    if t < 1e-15:
        return np.full(a.shape, s / a.size, dtype=np.float64)
    return a * (s / t)


def _grad_lag(a, k, n):
    g = np.zeros(n, dtype=np.float64)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        g[js:je] = 2.0 * a[k - np.arange(js, je)]
    return g


def _grad_topk(a, c, n, K):
    top = np.argsort(c)[-K:]
    w = c[top].astype(np.float64)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(K) / K
    else:
        w /= wsum
    g = np.zeros(n, dtype=np.float64)
    for wi, ki in zip(w, top):
        g += wi * _grad_lag(a, int(ki), n)
    g -= g.mean()
    return g


def _smooth_grad(a, c, tau, n, K):
    mx = c.max()
    ec = np.exp((c - mx) / tau)
    p = ec / ec.sum()
    # Use top-K of the softmax weights for efficiency
    top = np.argsort(p)[-K:]
    pk = p[top]
    pk /= pk.sum()
    g = np.zeros(n, dtype=np.float64)
    for wi, ki in zip(pk, top):
        g += wi * _grad_lag(a, int(ki), n)
    g -= g.mean()
    return 2.0 * tau * g / (n * c.max() + 1e-15)


def _freq_perturb(a, rng, strength):
    N = 1
    while N < 2 * len(a):
        N *= 2
    A = np.fft.rfft(a, N)
    mags = np.abs(A)
    phases = np.angle(A) + rng.normal(0.0, strength, size=A.size)
    A_new = mags * np.exp(1j * phases)
    a_new = np.fft.irfft(A_new, N)[:len(a)]
    return _proj(np.maximum(a_new, 0.0), float(len(a)))


def _block_perturb(a, rng, sigma, wf):
    n = len(a)
    b = a.copy()
    w = max(2, int(n * wf))
    s = int(rng.integers(0, max(1, n - w + 1)))
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _proj(b, float(n))


def _optimize_n(a_init, n, time_budget, rng, verbose_tag):
    """Optimize a single sequence of length n within time_budget seconds."""
    t0 = time.monotonic()
    deadline = t0 + time_budget
    a = a_init.copy()
    best_a = a.copy()
    best_mc = float(np.max(_conv(a)))
    S = float(n)  # target sum

    # --- Phase 1: Smooth gradient descent with momentum ---
    tau = n * 0.3
    beta = 0.85
    lr = 0.15
    mom = np.zeros(n, dtype=np.float64)
    K_smooth = 20
    p1_end = t0 + time_budget * 0.30
    it = 0
    stall = 0

    while time.monotonic() < p1_end:
        c = _conv(a)
        mc = float(c.max())
        if mc < best_mc - 1e-9:
            best_mc = mc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        g = _smooth_grad(a, c, tau, n, K_smooth)
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        g /= mg
        mom = beta * mom + (1.0 - beta) * g
        a = _proj(a - lr * mom, S)
        tau = max(tau * 0.99992, n * 0.005)
        if stall > 3000:
            lr = max(lr * 0.7, 0.01)
            stall = 0
        it += 1

    a = best_a
    if verbose_tag:
        print(f"  [{verbose_tag} P1] it={it} maxc={best_mc:.2f} obj={2*best_mc/n:.6f}")

    # --- Phase 2: Hard top-K descent with backtracking ---
    K = 10
    beta = 0.8
    lr = 0.06
    mom = np.zeros(n, dtype=np.float64)
    p2_end = t0 + time_budget * 0.60
    it = 0
    stall = 0

    while time.monotonic() < p2_end:
        c = _conv(a)
        mc = float(c.max())
        if mc < best_mc - 1e-9:
            best_mc = mc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        g = _grad_topk(a, c, n, K)
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        g /= mg
        mom = beta * mom + (1.0 - beta) * g
        gm = 2.0 * mom

        # Backtracking line search (max 4 trials)
        c0 = float(c.max())
        accepted = False
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_try = _proj(a - alpha * gm, S)
            c1 = float(np.max(_conv(a_try)))
            if c1 < c0 * 1.001:
                a = a_try
                accepted = True
                break
        if not accepted:
            a = _proj(a - lr * 0.1 * gm, S)

        if stall > 5000:
            lr = max(lr * 0.8, 0.005)
            stall = 0
        it += 1

    a = best_a
    if verbose_tag:
        print(f"  [{verbose_tag} P2] it={it} maxc={best_mc:.2f} obj={2*best_mc/n:.6f}")

    # --- Phase 3: Perturbation + re-optimize loop ---
    attempt = 0
    p3_end = deadline - 5.0
    while time.monotonic() < p3_end:
        attempt += 1
        # Choose perturbation type
        r = rng.random()
        if r < 0.4:
            sigma = float(rng.uniform(0.05, 0.35))
            wf = float(rng.choice([0.01, 0.02, 0.05, 0.10]))
            a_p = _block_perturb(a, rng, sigma, wf)
        elif r < 0.7:
            strength = float(rng.uniform(0.1, 0.8))
            a_p = _freq_perturb(a, rng, strength)
        else:
            # Larger structural change: random subset scaling
            a_p = a.copy()
            mask = rng.random(n) < 0.1
            a_p[mask] *= float(rng.uniform(0.3, 2.5, size=int(mask.sum())))
            a_p = _proj(a_p, S)

        # Quick refinement (limited time)
        sub_time = min(8.0, p3_end - time.monotonic() - 1.0)
        if sub_time < 2.0:
            break

        sub_deadline = time.monotonic() + sub_time
        sub_k = 8
        sub_lr = 0.04
        sub_beta = 0.8
        sub_mom = np.zeros(n, dtype=np.float64)
        sub_a = a_p.copy()
        sub_best_mc = float(np.max(_conv(sub_a)))
        sub_best_a = sub_a.copy()

        while time.monotonic() < sub_deadline:
            c = _conv(sub_a)
            mc = float(c.max())
            if mc < sub_best_mc:
                sub_best_mc = mc
                sub_best_a = sub_a.copy()
            g = _grad_topk(sub_a, c, n, sub_k)
            mg = float(np.max(np.abs(g)))
            if mg < 1e-14:
                break
            g /= mg
            sub_mom = sub_beta * sub_mom + (1.0 - sub_beta) * g
            a_try = _proj(sub_a - sub_lr * 2.0 * sub_mom, S)
            c_try = float(np.max(_conv(a_try)))
            if c_try < mc:
                sub_a = a_try
            else:
                sub_a = _proj(sub_a - sub_lr * 0.5 * 2.0 * sub_mom, S)

        if sub_best_mc < best_mc:
            best_mc = sub_best_mc
            best_a = sub_best_a
            a = best_a
            if verbose_tag and attempt <= 30:
                print(f"  [{verbose_tag} P3 #{attempt}] maxc={best_mc:.2f} obj={2*best_mc/n:.6f}")

    a = best_a
    final_obj = 2.0 * best_mc / n
    if verbose_tag:
        print(f"  [{verbose_tag} FINAL] maxc={best_mc:.2f} obj={final_obj:.6f}")
    return a, best_mc, final_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search: smooth gradient -> hard active-set -> perturb+refine loop.
    Returns best non-negative sequence found within budget_s seconds.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    hard_deadline = t0 + budget_s - 10.0

    # Determine which n values to try and how much time for each
    n_options = [3809, 3000, 5000]
    # Allocate time: 60% to primary, 20% to second, 20% to third
    time_alloc = [budget_s * 0.55, budget_s * 0.20, budget_s * 0.17]

    best_overall_a = None
    best_score = np.inf

    # Get parent for initialization
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        pass

    for idx, n in enumerate(n_options):
        alloc = time_alloc[idx]
        if time.monotonic() + alloc * 0.5 > hard_deadline:
            break

        # Initialize
        if parent is not None and parent.size == n:
            a_init = parent.copy()
        elif parent is not None:
            a_init = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        else:
            a_init = rng.uniform(0.5, 1.5, size=n).astype(np.float64)
        a_init = _proj(a_init, float(n))

        # Alternate: for non-primary n, sometimes use random init
        if idx > 0 and rng.random() < 0.5:
            a_init = rng.uniform(0.3, 2.0, size=n).astype(np.float64)
            a_init = _proj(a_init, float(n))

        tag = f"n{n}"
        print(f"Starting optimization: n={n}, time_budget={alloc:.0f}s")
        a_opt, best_mc, obj_val = _optimize_n(a_init, n, alloc, rng, verbose_tag=True)

        # Compute actual evaluator score
        sc = 2.0 * best_mc / n
        if sc < best_score:
            best_score = sc
            best_overall_a = a_opt.copy()
            print(f"  New best: obj={sc:.6f} at n={n}")

    # Final polish on the best sequence found
    if best_overall_a is not None and time.monotonic() < hard_deadline - 10:
        n_final = len(best_overall_a)
        remaining = hard_deadline - time.monotonic() - 5.0
        if remaining > 20:
            print(f"\nFinal polish: n={n_final}, time={remaining:.0f}s")
            best_overall_a, best_mc, obj_val = _optimize_n(
                best_overall_a, n_final, remaining, rng, verbose_tag="polish"
            )
            sc = 2.0 * best_mc / n_final
            if sc < best_score:
                best_score = sc
                print(f"  After polish: obj={sc:.6f}")

    # Validate and return
    if best_overall_a is None:
        best_overall_a = np.ones(3809, dtype=np.float64)

    result = [float(x) for x in best_overall_a]
    try:
        sc_eval = float(evaluate_sequence(result))
        print(f"\nFINAL: internal={best_score:.6f}  evaluated={sc_eval:.6f}  n={len(result)}")
    except Exception as e:
        print(f"Eval error: {e}")
        sc_eval = np.inf

    if not np.isfinite(sc_eval):
        result = [1.0] * 3809
        print("Falling back to uniform sequence")

    return result
