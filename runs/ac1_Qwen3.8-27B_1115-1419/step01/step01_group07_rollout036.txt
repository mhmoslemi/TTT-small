Optimize step function sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import minimize

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def compute_score(a):
    """Compute the objective value 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a ** 2)

def lse_objective(x, tau):
    """LSE-smoothed max of convolve(softmax(x), softmax(x))."""
    a = softmax(x)
    c = np.convolve(a, a)
    c_max = c.max()
    c_shifted = c - c_max
    vals = np.exp(c_shifted / tau)
    return tau * np.log(vals.sum()) + c_max

def lse_gradient(x, tau):
    """Gradient of LSE-smoothed max w.r.t. x."""
    a = softmax(x)
    c = np.convolve(a, a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(c_shifted / tau)
    w /= w.sum()
    # Gradient of c_k w.r.t. a_i: 2 * a_{k-i} (with valid range)
    # Weighted sum: grad_a_i = 2 * sum_k w_k * a_{k-i} = 2 * correlate(w, a)
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    # Chain rule through softmax
    dot = np.dot(a, grad_a)
    return a * (grad_a - dot)

def subgradient_step_a(a, n_active, step):
    """One subgradient step in a-space (not x-space)."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-30:
        return a.copy()
    # Find top peaks
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(active)
    # Step and project
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-30:
        a_new /= s
    return a_new
