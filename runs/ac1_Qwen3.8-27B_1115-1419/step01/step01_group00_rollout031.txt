Or better: **use the KKT condition to determine the optimal direction**.

At the current point a, the subgradient of max_k conv[k] is the set of convex combinations of the gradients of the active (maximizing) conv coefficients. The "best" direction to go is the one that minimizes the increase in the subgradient, which is the negative of the minimum-norm element of the subgradient.

The minimum-norm element of the subgradient is:
g_min = argmin_{λ >= 0, sum λ = 1} ||sum λ_k grad_k||^2

This is a quadratic program in λ. For a small active set (say K=5), this is a 5-variable QP that's easy to solve.

This is a much better direction than just the gradient of the argmax! Let me implement this.

```python
    Uses the top-K active positions."""
    max_c = np.max(c)
    # Find active set: positions where c[k] >= max_c - tol * max_c
    active = np.where(c >= max_c - tol * max_c)[0]
    # Take top K
    if len(active) > K:
        # Sort by c value, take top K
        active = active[np.argsort(c[active])[-K:]]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    s = len(active)

    # We want to minimize ||G^T λ||^2 subject to λ >= 0, sum λ = 1
    # where G^T λ = sum_k λ_k * grad_k = sum_k λ_k * 2*a[k-j (with bounds)]
    # This is equivalent to minimizing (G G^T) in terms of λ...
    # Actually: ||G^T λ||^2 = λ^T G G^T λ
    # where (G G^T)_{i,l} = <grad_{active[i]}, grad_{active[l]}> = 4 * sum_j a[active[i]-j] * a[active[l]-j] (with bounds)
    # = 4 * conv(a,a)[active[i] - active[l] + ... ]

    # <grad_i, grad_l> = sum_j (2*a[active[i]-j]) * (2*a[active[l]-j]) where both indices are in [0,n)
    # = 4 * sum_{j: 0<=active[i]-j<n, 0<=active[l]-j<n} a[active[i]-j] * a[active[l]-j]
    # Let m = active[i] - j, then j = active[i] - m, and the condition is 0 <= m < n and 0 <= active[l] - active[i] + m < n
    # = 4 * sum_{m in overlap} a[m] * a[m + active[l] - active[i]]
    # This is 4 * (cross-correlation of a with a at lag active[l] - active[i])
    # = 4 * conv_full[active[l] - active[i] + (n-1)] (using full conv indexing)
    # Actually, let's define: cross_corr[i,l] = sum_j a[j] * a[j + (active[l] - active[i])] (with bounds)
    # This is conv(a,a)[active[l] - active[i] + (n-1)] if we use the "full" convolution indexing
    # where full_conv[k] = sum_j a[j] * a[k-j] for k = 0..2n-2
    
    # So (G G^T)_{i,l} = 4 * full_conv[active[l] - active[i] + (n-1)]

    # The factor 4 doesn't matter for the minimizer.
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_minimize

def _conv(a):
    return np.convolve(a, a)

def _obj_from_conv(c, n):
    return 2.0 * float(np.max(c)) / n

def _project(a, n):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-14:
        return np.ones(n, dtype=np.float64)
    return a / s * n

def _smooth_obj(a, tau, n):
    """Smooth approximation of max(conv(a,a)) using log-sum-exp."""
    c = _conv(a)
    mx = np.max(c)
    if tau < 1e-10:
        return mx
    return mx + tau * np.log(np.sum(np.exp((c - mx) / tau)))

def _smooth_grad(a, tau, n):
    """Gradient of smooth max(conv) w.r.t. a."""
    c = _conv(a)
    mx = np.max(c)
    w = np.exp((c - mx) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # g = 2 * correlate(w, a, 'valid')
    # w has length 2n-1, a has length n
    # correlate(w, a, 'valid')[j] = sum_i w[j+i] * a[i] for i=0..n-1
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def _kkt_grad(a, c, n, K=5, rel_tol=0.02):
    """Compute KKT-optimal descent direction using top-K active set."""
    max_c = float(np.max(c))
    threshold = max_c * (1.0 - rel_tol)
    active_mask = c >= threshold
    active = np.where(active_mask)[0]
    
    if len(active) > K:
        # Keep only the top K
        top_idx = np.argsort(c[active])[-K:]
        active = active[top_idx]
    elif len(active) == 0:
        active = np.array([int(np.argmax(c))])
    
    s = len(active)
    
    if s == 1:
        k = active[0]
        g = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            idx_arr = k - np.arange(js, je)
            g[js:je] = 2.0 * a[idx_arr]
        return g
    
    # Build H matrix: H[i,l] = c_full[active[l] - active[i] + (n-1)]
    H = np.zeros((s, s))
    for i in range(s):
        for l in range(s):
            idx = int(active[l]) - int(active[i]) + (n - 1)
            if 0 <= idx < 2 * n - 1:
                H[i, l] = c[idx]
    
    # Solve QP: min λ^T H λ s.t. sum(λ) = 1, λ >= 0
    def qfun(x):
        return float(x @ H @ x)
    
    constraints = [{'type': 'eq', 'fun': lambda x: float(x.sum() - 1.0)}]
    bounds = [(0.0, 1.0)] * s
    x0 = np.ones(s) / s
    
    try:
        res = sp_minimize(qfun, x0, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 100, 'ftol': 1e-12})
        lam = res.x
    except Exception:
        lam = np.ones(s) / s
    
    lam = np.maximum(lam, 0.0)
    lam_sum = lam.sum()
    if lam_sum < 1e-15:
        lam = np.ones(s) / s
    else:
        lam /= lam_sum
    
    # Compute direction
    g = np.zeros(n, dtype=np.float64)
    for i in range(s):
        k = int(active[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            idx_arr = k - np.arange(js, je)
            g[js:je] += lam[i] * 2.0 * a[idx_arr]
    
    return g

def _line_search_smooth(a, g, n, tau, alpha_init, max_iter=20):
    """Backtracking line search on the smooth objective."""
    f0 = _smooth_obj(a, tau, n)
    # Simple directional derivative estimate
    alpha = alpha_init
    for _ in range(max_iter):
        a_new = _project(a - alpha * g, n)
        f_new = _smooth_obj(a_new, tau, n)
        if f_new < f0:
            return a_new, alpha
        alpha *= 0.5
    return a, 0.0

def _hard_step(a, c, n, alpha):
    """One step of hard gradient descent using KKT direction."""
    g = _kkt_grad(a, c, n, K=5, rel_tol=0.02)
    g -= g.mean()  # Preserve sum
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
```

Let me also think about whether the KKT direction is actually better than the simple argmax gradient. 

The argmax gradient direction is: -grad_{argmax} (i.e., reduce the a's that contribute most to the single peak).

The KKT direction is: -sum_k λ_k * grad_k (weighted combination of gradients of all active peaks).

The KKT direction is better because it accounts for the fact that reducing one peak might increase another. By using the weighted combination, we reduce all peaks simultaneously.

However, the KKT direction requires solving a QP at each step, which adds overhead. For n=3809 and K=5, the QP is a 5-variable problem, which scipy should solve in <1ms. The main cost is the convolution computation (O(n^2) ≈ 14.5M ops), which dominates.

So the QP solve is negligible. Let me use the KKT direction always.

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_minimize

def _conv(a):
    return np.convolve(a, a)

def _obj_from_conv(c, n):
    return 2.0 * float(np.max(c)) / n

def _project(a, n):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-14:
        return np.ones(n, dtype=np.float64)
    return a / s * n

def _smooth_grad(a, c, tau, n):
    """Gradient of entropy-smoothed max of c w.r.t. a. c = conv(a,a)."""
    mx = float(np.max(c))
    w = np.exp((c - mx) / tau)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def _kkt_grad(a, c, n, K=5, rel_tol=0.02):
    """KKT-optimal direction using top-K active set."""
    max_c = float(np.max(c))
    if max_c < 1e-15:
        return np.zeros(n)
    threshold = max_c * (1.0 - rel_tol)
    active = np.where(c >= threshold)[0]
    
    if len(active) > K:
        top_idx = np.argsort(c[active])[-K:]
        active = active[top_idx]
    elif len(active) == 0:
        active = np.array([int(np.argmax(c))])
    
    s = len(active)
    
    if s == 1:
        k = int(active[0])
        g = np.zeros(n, dtype=np.float64)
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            idx_arr = k - np.arange(js, je)
            g[js:je] = 2.0 * a[idx_arr]
        return g
    
    # Build Gram matrix
    H = np.zeros((s, s))
    for i in range(s):
        for l in range(s):
            idx = int(active[l]) - int(active[i]) + (n - 1)
            if 0 <= idx < 2 * n - 1:
                H[i, l] = c[idx]
    
    # Solve QP
    def qfun(x):
        return float(x @ H @ x)
    qfun_jac = lambda x: 2.0 * (H @ x)
    
    constraints = [{'type': 'eq', 'fun': lambda x: float(x.sum() - 1.0),
                   'jac': lambda x: np.ones(s)}]
    bounds = [(0.0, 1.0)] * s
    x0 = np.ones(s) / s
    
    try:
        res = sp_minimize(qfun, x0, jac=qfun_jac, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-14})
        lam = np.maximum(res.x, 0.0)
    except Exception:
        lam = np.ones(s) / s
    
    lam_sum = lam.sum()
    if lam_sum < 1e-14:
        lam = np.ones(s) / s
    else:
        lam /= lam_sum
    
    g = np.zeros(n, dtype=np.float64)
    for i in range(s):
        k = int(active[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            idx_arr = k - np.arange(js, je)
            g[js:je] += lam[i] * 2.0 * a[idx_arr]
    
    return g
```
