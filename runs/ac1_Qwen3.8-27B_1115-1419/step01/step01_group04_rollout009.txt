def fast_conv(a):

    n = len(a)

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    np.maximum(c, 0, out=c)

"""
Multi-resolution Adam optimization with LSE smooth max for step function coefficients.
"""

import numpy as np
import time
import math

def fast_conv(a):
    n = len(a)
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A*A, N)[:L]
    np.maximum(c, 0, out=c)
    return c

def lse_and_weights(c, T):
    cmax = np.max(c)
    shifted = T * (c - cmax)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    lse = cmax + np.log(total) / T
    w = exp_vals / total
    return lse, w

def compute_grad(a, w, T):
    """d((2/T)*LSE)/da = (2/T) * correlate(w, a)"""
    n = len(a)
    Lw = len(w)
    N = 1 << (Lw + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, N)
    Arev = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(W * Arev, N)[:Lw + n - 1]
    # g[m] = conv[n-1-m] for m=0,...,n-1
    g = conv[n-1:2*n-1][::-1]
    return (2.0 / T) * g

def optimize_sequence(a_init, n_iters, T_init=5.0, T_final=50.0, lr_init=0.05, lr_final=0.001, seed=42):
    """Adam in log-space with LSE smooth max."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()  # normalize to sum=1
    
    v = np.log(a)
    m = np.zeros(n)
    vhat = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fast_conv(a))  # score with sum=1
    
    rng = np.random.default_rng(seed)
    
    for it in range(n_iters):
        t += 1
        # Anneal T and lr
        frac = it / max(1, n_iters)
        T_curr = T_init + (T_final - T_init) * min(1, frac * 2)  # reach T_final by half
        lr_curr = lr_init + (lr_final - lr_init) * frac
        
        # Compute c and smooth score
        c = fast_conv(a)
        lse_val, w = lse_and_weights(c, T_curr)
        score = 2.0 * n * lse_val  # sum=1
        
        if not np.isfinite(score):
            break
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Gradient
        g_a = compute_grad(a, w, T_curr)  # This is (2n)*d(LSE)/da since d(2n*LSE)/da = 2n*(2/T)*correlate(w,a)
        # Wait: score = 2n * LSE, so d(score)/da = 2n * d(LSE)/da = 2n * (2/T) * correlate(w,a)
        # But compute_grad returns (2/T)*correlate(w,a). So:
        g_a = 2.0 * n * g_a
        
        # Project to v-space
        g_v = a * (g_a - np.dot(g_a, a))  # sum=1, so dot(g_a, a)/1 = dot(g_a, a)
        
        # Add small random perturbation to escape local minima
        if it % 500 == 499:
            g_v += 0.01 * a * rng.normal(0, 1, n)
        
        # Adam

        vhat = b2 * vhat + (1 - b2) * g_v**2
        m_hat = m / (1 - b1**t)
        vhat_hat = vhat / (1 - b2**t)
        v -= lr_curr * m_hat / (np.sqrt(vhat_hat) + eps)
        
        # Convert to a (with normalization)
        v_centered = v - np.max(v)
        a = np.exp(v_centered)
        a = a / a.sum()
        
    return best_a, best_score

"""
Multi-resolution Adam optimization with LSE smooth max for minimizing
2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.

Uses log-space parameterization, LSE-based differentiable max, and
multi-resolution strategy from n=256 up to n=2048.
"""

import numpy as np
import time
import math


def fast_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    np.maximum(c, 0, out=c)
    return c


def lse_and_weights(c, T):
    """Compute LSE smooth max and softmax weights."""
    cmax = np.max(c)
    shifted = T * (c - cmax)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    lse = cmax + np.log(total) / T
    w = exp_vals / total
    return float(lse), w


def correlate_wa(w, a):
    """Compute correlate(w, a)[i] = sum_j w[i+j]*a[j] for i=0..n-1."""
    n = len(a)
    Lw = len(w)
    N = 1 << (Lw + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, N)
    Alef = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(W * Alef, N)[:Lw + n - 1]
    return conv[n - 1:2 * n - 1][::-1]


def score_from_c(c, n, s):
    """Compute score from convolution."""
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * np.max(c) / (s * s))


def optimize_adam_lse(a_init, n_iters, T_start=5.0, T_end=50.0,
                      lr_start=0.1, lr_end=0.005, seed=42,
                      time_budget=None, t0=None):
    """
    Adam optimization in log-space using LSE smooth max.
    a is kept with sum=1.
    Returns (best_a, best_score) where score = 2*n*max(conv) (sum=1 normalization).
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    v = np.log(a)
    m_vhat = np.zeros(n)
    v_vhat = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = fast_conv(a)
    best_a = a.copy()
    best_score = float(2.0 * n * np.max(c))
    
    rng = np.random.default_rng(seed)
    
    for it in range(n_iters):
        # Time check
        if time_budget is not None:
            if time.monotonic() - t0 > time_budget:
                break
        
        t += 1
        frac = t / max(1, n_iters)
        
        # Anneal T (reach T_end by 1/3 of iterations)
        T_curr = T_start + (T_end - T_start) * min(1.0, 3.0 * frac)
        # Anneal learning rate
        lr_curr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute convolution
        c = fast_conv(a)
        
        # Compute LSE and weights
        lse_val, w = lse_and_weights(c, T_curr)
        score = 2.0 * n * lse_val
        
        if not np.isfinite(score):
            break
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Compute gradient: d(2n*LSE)/da = (4n/T) * correlate(w, a)
        corr = correlate_wa(w, a)
        g_a = (4.0 * n / T_curr) * corr
        
        # Project to v-space (sum constraint)
        # d(score)/dv_j = a_j * (g_a_j - dot(g_a, a))
        g_v = a * (g_a - np.dot(g_a, a))
        
        # Periodic perturbation to escape local minima
        if t % 1000 == 999:
            g_v = g_v + 0.05 * a * rng.normal(0, 1, n)
        
        # Adam update
        m_vhat = b1 * m_vhat + (1 - b1) * g_v
        v_vhat = b2 * v_vhat + (1 - b2) * g_v * g_v
        m_unbias = m_vhat / (1 - b1**t)
        v_unbias = v_vhat / (1 - b2**t)
        v = v - lr_curr * m_unbias / (np.sqrt(v_unbias) + eps)
        
        # Convert back to a
        v_shift = v - np.max(v)
        a = np.exp(v_shift)
        a_sum = np.sum(a)
        a = a / a_sum
        
    return best_a, best_score


def upsample(a, n_new):
    """Linearly interpolate sequence to new length, renormalize to sum=1."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0)
    s = np.sum(out)
    if s > 0:
        out = out / s
    else:
        out = np.ones(n_new) / n_new
    return out


def make_random_start(n, rng, style='smooth'):
    """Generate a random positive sequence with sum=1."""
    if style == 'uniform':
        a = np.ones(n)
    elif style == 'linear':
        a = np.linspace(0.1, 2.0, n)
    elif style == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * (1 - (2*t - 1)**2)
    elif style == 'sine':
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.0 * np.sin(np.pi * t)
    elif style == 'bathtub':
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * np.cos(np.pi * (2*t - 1))
        a = np.maximum(a, 0.01)
    elif style == 'random_smooth':
        t = np.linspace(0, 1, n)
        z = np.zeros(n)
        for m in range(1, 6):
            amp = rng.normal(0, 1.0 / m)
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * m * t + ph)
        a = np.exp(0.5 * z - 0.5 * np.mean(z))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def perturbation_refine(a, n_perturb, seed=0):
    """
    Try small random perturbations to individual elements or blocks.
    Accept only if score improves.
    """
    n = len(a)
    current = a.copy()
    current_score = 2.0 * n * np.max(fast_conv(current))
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng(seed)
    
    for _ in range(n_perturb):
        trial = current.copy()
        
        # Random perturbation type
        r = rng.random()
        if r < 0.5:
            # Perturb a single element
            i = rng.integers(0, n)
            factor = np.exp(rng.normal(0, 0.1))
            trial[i] *= factor
        elif r < 0.8:
            # Perturb a block
            width = rng.integers(2, max(3, n // 50))
            start = rng.integers(0, max(1, n - width))
            factor = np.exp(rng.normal(0, 0.3))
            trial[start:start+width] *= factor
        else:
            # Swap two random values
            i, j = rng.integers(0, n, 2)
            if i != j:
                trial[i], trial[j] = trial[j], trial[i]
        
        trial = np.maximum(trial, 0)
        s = np.sum(trial)
        if s < 0.01:
            continue
        trial = trial / s
        
        c = fast_conv(trial)
        sc = 2.0 * n * np.max(c)
        
        if sc < current_score - 1e-12:
            current = trial
            current_score = sc
            if sc < best_score:
                best_score = sc
                best = trial.copy()
        elif rng.random() < 0.01:
            # Occasional acceptance of slightly worse for exploration
            if sc < current_score * 1.01:
                current = trial
                current_score = sc
    
    return best, best_score
