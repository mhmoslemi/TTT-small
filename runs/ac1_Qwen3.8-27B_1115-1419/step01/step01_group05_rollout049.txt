"""
Search for low-peak nonnegative self-convolutions using LSE-smoothedgradient descent with symmetric and asymmetric variants.

Score: 2*n*max(a*a)/(sum a)^2, scale-invariant so normalize to sum=1.
For sum=1: score = 2*n*max(a*a).
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s

def _convolve(a, M=None):
    n = a.size
    if M is None:
        M = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def _grad_lse(a, beta, M=None, Mg=None):
    """Compute gradient of LSE-smoothed objective w.r.t. a.
    Returns (gradient, lse_value, weights)."""
    n = a.size
    if M is None:
        M = _next_pow2(2 * n - 1)
    if Mg is None:
        Mg = _next_pow2(2 * n - 1)
    b = _convolve(a, M)
    m = float(np.max(b))
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    se = float(e.sum())
    if se <= 0:
        e = np.ones(b.size)
        se = b.size
    w = e / se
    lse_val = m + float(np.log(se)) / beta
    
    # Gradient: 2 * (w * a_flip)_{n-1:2n-1}
    a_flip = a[::-1]
    W = np.fft.rfft(w, Mg)
    Af = np.fft.rfft(a_flip, Mg)
    conv = np.fft.irfft(W * Af, Mg)
    grad = 2.0 * conv[n-1:n-1+n]
    return grad, lse_val, w

def _grad_exact(a, M=None):
    """Compute gradient using the exact argmax (subgradient)."""
    n = a.size
    b = _convolve(a, M)
    k_star = int(np.argmax(b))
    # grad_i = 2 * a_{k_star - i} if 0 <= k_star - i < n else 0
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, b, k_star

def _make_exponential(n, alpha=2.0):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-alpha * x))

def _make_symmetric_exp(n, alpha=3.0):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-alpha * np.abs(x - 0.5) * 2))

def _make_gaussian(n, sigma=0.3):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-(x - 0.5)**2 / (2 * sigma**2)))

def _make_cosine(n):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.cos(np.pi * x / 2)**2)

def _make_mixture(n, alpha=2.0, ratio=1.0):
    a1 = _make_exponential(n, alpha)
    a2 = a1[::-1].copy()
    return _normalize(a1 + ratio * a2)

def _smooth_perturb(a, rng, amplitude=0.3):
    n = a.size
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 8)
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - f.mean()
    sd = f.std()
    if sd > 0:
        f = f / sd
    a2 = a * (1.0 + amplitude * f)
    return _normalize(np.maximum(a2, 1e-15))

def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.maximum(old, 0.0)
    s = old.sum()
    if s <= 1e-300 or old.size == 0:
        return _normalize(np.ones(new_n))
    old = old / s
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

def _score(a):
    a = _normalize(a)
    n = a.size
    M = _next_pow2(2 * n - 1)
    b = _convolve(a, M)
    return float(2.0 * n * np.max(b))

def _project(v):
    """Soft projection to simplex (clip + normalize)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 1e-300:
        return np.ones(v.size) / v.size
    return v / s

def _optimize(a, deadline, rng, use_symmetric=False):
    """LSE-smoothed gradient descent."""
    a = _normalize(a.copy())
    n = a.size
    M = _next_pow2(2 * n - 1)
    
    b = _convolve(a, M)
    maxb = float(np.max(b))
    score = 2.0 * n * maxb
    best_a = a.copy()
    best_score = score
    
    alpha = 0.5
    beta = 8.0
    it = 0
    fail = 0
    stagnant = 0
    
    while time.monotonic() < deadline:
        # Compute LSE gradient
        grad, lse_old, w = _grad_lse(a, beta, M, M)
        
        # Center the gradient (remove mean to stay near simplex)
        grad_c = grad - grad.mean()
        gnorm = float(np.max(np.abs(grad_c)))
        
        if gnorm < 1e-12:
            # No meaningful gradient, perturb
            a = _smooth_perturb(best_a, rng, 0.3)
            b = _convolve(a, M)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.5
            fail = 0
            it += 1
            continue
        
        # Gradient step
        v = a - alpha * grad_c / gnorm  # Normalize gradient for stability
        a = _project(v)
        
        if use_symmetric:
            a = 0.5 * (a + a[::-1])
            a = _normalize(a)
        
        # Evaluate
        b = _convolve(a, M)
        maxb = float(np.max(b))
        score = 2.0 * n * maxb
        lse_new = float(np.max(b)) + float(np.log(float(np.exp(beta * (b - maxb)).sum()))) / beta
        
        # Adaptive step size
        if lse_new < lse_old - 1e-12:
            alpha = min(5.0, alpha * 1.15)
            fail = 0
        else:
            alpha *= 0.5
            fail += 1
        
        if score < best_score - 1e-10:
            best_a = a.copy()
            best_score = score
            stagnant = 0
        else:
            stagnant += 1
        
        # Increase beta periodically
        if it % 50 == 0:
            beta = min(500.0, beta * 1.3)
        
        # Restart if stuck
        if (fail > 200 or stagnant > 500) and time.monotonic() < deadline - 5:
            if rng.random() < 0.7:
                a = _smooth_perturb(best_a, rng, 0.25)
            else:
                a = _smooth_perturb(best_a, rng, 0.5)
            b = _convolve(a, M)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            alpha = 0.5
            beta = max(8.0, beta * 0.3)
            fail = 0
            stagnant = 0
        
        it += 1
    
    return best_a, best_score

def _optimize_subgradient(a, deadline, rng):

    a = _normalize(a.copy())

    best_a = a.copy()

        grad = np.zeros(n)

- Alternate between: (1) minimize (p*p)_{k*} for fixed k*, (2) choose new k*.
- Step (1) is: minimize (p*p)_{k*} subject to p >= 0, sum p = 1.
  This is a QP: minimize p^T Q p s.t. p >= 0, 1^T p = 1, where Q has Q(i,j) = 1 if i+j=k*, else 0.
  This can be solved in O(n) using the KKT conditions (since Q is very sparse).

Actually, for a fixed k*, the objective is:
(p*p)_{k*} = sum_{i=0}^{n-1} p_i * p_{k*-i} (with p_j = 0 for j out of range)

Let S = {i : 0 <= i < n and 0 <= k*-i < n} = {i : max(0, k*-(n-1)) <= i <= min(n-1, k*)}
|S| = min(n-1, k*) - max(0, k*-(n-1)) + 1

For k* < n: S = {0, 1, ..., k*}, |S| = k*+1
For k* >= n: S = {k*-(n-1), ..., n-1}, |S| = 2n-1-k*

The objective becomes: sum_{i in S} p_i * p_{k*-i}

This is a quadratic form p^T Q p where Q(i,j) = 1 if i+j=k* and both in S, else 0.
Note that Q is symmetric and has a specific "anti-diagonal" structure.

The QP: min p^T Q p s.t. p >= 0, 1^T p = 1.

"""
Search for low-peak nonnegative self-convolutions using LSE-smoothed projected
gradient descent with multiple parametric initializations, adaptive sharpening,
stochastic subgradient alternatives, and multi-length warm-start strategy.

Score: 2*n*max(a*a)/(sum a)^2, scale-invariant → normalize to sum=1.
For sum=1: score = 2*n*max(a*a).
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _convolve(a, M=None):
    n = a.size
    if M is None:
        M = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def _grad_lse(a, beta, M=None):
    """Compute LSE-smoothed gradient and value."""
    n = a.size
    if M is None:
        M = _next_pow2(2 * n - 1)
    b = _convolve(a, M)
    m = float(np.max(b))
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    se = float(e.sum())
    if se <= 0:
        e = np.ones(b.size, dtype=np.float64)
        se = float(b.size)
    w = e / se
    lse_val = m + float(np.log(se)) / beta
    # Gradient via cross-correlation: 2 * (w * a_flip)_{n-1:2n-1}
    a_flip = a[::-1]
    W = np.fft.rfft(w, M)
    Af = np.fft.rfft(a_flip, M)
    conv = np.fft.irfft(W * Af, M)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, lse_val


def _make_exponential(n, alpha=2.0):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-alpha * x))


def _make_symmetric_exp(n, alpha=3.0):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-alpha * np.abs(x - 0.5) * 2.0))


def _make_gaussian(n, sigma=0.25):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.exp(-((x - 0.5) ** 2) / (2.0 * sigma * sigma)))


def _make_cosine(n):
    x = np.linspace(0.0, 1.0, n)
    return _normalize(np.cos(np.pi * x / 2.0) ** 2)


def _make_mixture(n, alpha=2.0, ratio=1.0):
    a1 = _make_exponential(n, alpha)
    a2 = a1[::-1].copy()
    return _normalize(a1 + ratio * a2)


def _make_epanechnikov(n):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    a = np.maximum(0.0, 1.0 - t * t)
    return _normalize(a)


def _make_quartic(n):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    a = np.maximum(0.0, 1.0 - t ** 4)
    return _normalize(a)


def _smooth_perturb(a, rng, amplitude=0.3):
    n = a.size
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 8)
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - f.mean()
    sd = f.std()
    if sd > 1e-15:
        f = f / sd
    a2 = a * (1.0 + amplitude * f)
    return _normalize(np.maximum(a2, 1e-15))


def _smooth_perturb_highfreq(a, rng, amplitude=0.2):
    n = a.size
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, n // 32)
    if cutoff < len(F):
        F[:cutoff] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - f.mean()
    sd = f.std()
    if sd > 1e-15:
        f = f / sd
    a2 = a * (1.0 + amplitude * f)
    return _normalize(np.maximum(a2, 1e-15))


def _block_perturb(a, rng, strength=0.2):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= np.exp(rng.normal(0.0, strength, size=stop - start))
    return _normalize(c)


def _resample(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _normalize(np.ones(new_n))
    s = float(old.sum())
    if s <= 1e-300:
        return _normalize(np.ones(new_n))
    old = old / s
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _score(a):
    a = _normalize(a)
    n = a.size
    M = _next_pow2(2 * n - 1)
    b = _convolve(a, M)
    return float(2.0 * n * np.max(b))


def _project(v):
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / v.size
    return v / s
