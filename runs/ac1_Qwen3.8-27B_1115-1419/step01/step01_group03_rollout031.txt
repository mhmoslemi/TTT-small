import numpy as np

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.full(n, 1.0/n)
    rho = np.where(cond)[0][-1] + 1
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)

def score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def subgradient(c, p, n):
    """Compute subgradient of max(c) w.r.t. p.
    c = convolve(p, p), so dc[k]/dp[j] = p[j] + p[k-j] if 0<=k-j<n.
    We average over all 'active' k values (those with c[k] close to max).

    max_c = c.max()

        lo = max(0, k - n + 1)

            vals = k - np.arange(lo, hi)

import numpy as np

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0/n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(p):
    n = len(p)
    s = p.sum()
    if s < 1e-15:
        return np.inf
    c = np.convolve(p, p)
    return 2.0 * n * c.max() / (s * s)

def compute_softmax_weights(c, beta):
    mc = c.max()
    w = np.exp(beta * (c - mc))
    w /= w.sum()
    return w

def subgradient_from_weights(p, w, n):
    """Compute subgradient using precomputed weights w on c."""

    pw = np.convolve(p_r, w)  # length 3n-2

    center = (n - 1) / 2
    for σ_frac in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        σ = σ_frac * n
        x = np.arange(n) - center
        g = np.exp(-x**2 / (2 * σ**2))
        candidates.append(g / g.sum())
    
    # Raised cosine
    x = np.linspace(-1, 1, n)
    rc = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 1e-10)
    candidates.append(rc / rc.sum())
    
    # Power law
    x = np.linspace(-1, 1, n)
    for alpha in [0.5, 1.0, 2.0]:
        pw = np.maximum(1 - np.abs(x), 1e-10) ** alpha
        candidates.append(pw / pw.sum())
    
    # Random
    for seed in [1, 2, 3]:
        r = np.random.default_rng(seed).exponential(1, n)
        candidates.append(r / r.sum())
    
    # Sparse
    sp = rng.exponential(1, n)
    mask = rng.random(n) > 0.5
    sp[~mask] = 0
    candidates.append(sp / sp.sum())
    
    # Evaluate all
    best_c = None
    best_sc = np.inf
    for i, c in enumerate(candidates):
        sc = compute_score(c)
        if sc < best_sc:
            best_sc = sc
            best_c = c.copy()
    
    print(f"Best initial: {best_sc:.8f}")
    
    # Run FISTA from best
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        best_p, best_sc = run_f
