def normalize_raw(a):
    """Return a finite non-negative array normalized to sum 1, or None."""
    try:
        arr = np.asarray(a, dtype=np.float64)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = arr.ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s):

            return None

        s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr = arr / s
    return arr

def fast_conv(a):
    """Fast linear convolution a*a via FFT. Clips tiny negative FFT noise."""
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def fast_max(a):
    """Fast max coefficient of a*a. Assumes non-negative input."""
    if a is None or a.size == 0:
        return 0.0
    return float(np.max(fast_conv(a)))


def fast_score(a):
    """Fast value of 2*n*max(a*a)/(sum a)^2."""
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    mx = fast_max(a)
    return float(2.0 * n * mx / (s * s))


def correlate_wa(w, a):
    """
    Compute h_i = sum_m w_{i+m} a_m for i=0..n-1,
    where w length is 2n-1 and a length is n.
    """
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([w[0] * a[0]], dtype=np.float64)
    L = next_fast_len(3 * n - 2)
    wr = w[::-1]
    conv = irfft(rfft(wr, L) * rfft(a, L), L)
    return conv[2 * n - 2:n - 2:-1].copy()


def softmax(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, -1e300)
    m = float(np.max(x))
    e = np.exp(x - m)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(x.size, dtype=np.float64) / max(1, x.size)
    return e / s


def smooth_objective(x, tau):
    a = softmax(x)
    c = fast_conv(a)
    cmax = float(np.max(c))
    if tau <= 0.0:
        return cmax
    z = (c - cmax) / tau
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return cmax
    return float(cmax + tau * (zmax + np.log(se)))


def smooth_gradient(x, tau):
    a = softmax(x)
    c = fast_conv(a)
    cmax = float(np.max(c))
    if tau <= 0.0 or cmax <= 0.0:
        return np.zeros_like(a)
    z = (c - cmax) / tau
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / se
    grad_a = 2.0 * correlate_wa(w, a)
    if not np.all(np.isfinite(grad_a)):
        grad_a = np.zeros_like(a)
    dot = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - dot)
    grad_x = np.where(np.isfinite(grad_x), grad_x, 0.0)
    return grad_x

def project_simplex(v):
    """Project Euclidean onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)

    vmin = float(np.min(v))
    vsum = float(np.sum(v))
    if vmin >= -1e-12 and abs(vsum - 1.0) < 1e-8:
        v = np.maximum(v, 0.0)
        s = float(v.sum())
        if s > 0.0:
            return v / s
        return np.ones(n, dtype=np.float64) / n

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv / ind)
    rho = int(np.count_nonzero(cond))
    if rho <= 0:
        return np.ones(n, dtype=np.float64) / n
    theta = float(cssv[rho - 1] / rho)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return w / s


def active_weights(a, k, temp_factor):
    """Convolution values and weights concentrated on active peaks."""
    c = fast_conv(a)
    m = int(c.size)
    if m == 0:
        return c, None
    k = int(min(k, m))
    if k <= 0:
        return c, None
    maxc = float(np.max(c))
    if k >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(m, dtype=np.float64)
    if maxc <= 1e-300:
        w[idx] = 1.0 / k
        return c, w
    if temp_factor is not None and temp_factor > 0.0:
        temp = max(float(temp_factor) * maxc, 1e-300)
        vals = np.exp((c[idx] - maxc) / temp)
        s = float(np.sum(vals))
        if s > 0.0 and np.isfinite(s):
            w[idx] = vals / s
        else:
            w[idx] = 1.0 / k
    else:
        w[idx] = 1.0 / k
    return c, w

def add_mode(modes, m):
    if modes is None or m is None:
        return
    m = np.asarray(m, dtype=np.float64)
    if m.size == 0:
        return
    m = np.where(np.isfinite(m), m, 0.0)
    m = m - float(np.mean(m))
    nrm = float(np.linalg.norm(m))
    if nrm < 1e-12:
        return
    m = m / nrm
    modes.append(m)


def random_smooth_mode(n, rng, max_cycles=4):
    n = int(max(2, n))
    vals = np.zeros(n // 2 + 1, dtype=complex)
    maxf = max(1, min(max_cycles, max(1, n // 4)))
    for f in range(1, maxf + 1):
        vals[f] = (rng.normal() + 1j * rng.normal()) / (float(f) ** 0.7)
    m = irfft(vals, n)
    m = np.asarray(m, dtype=np.float64)
    m = m - float(np.mean(m))
    nrm = float(np.linalg.norm(m))
    if nrm < 1e-12:
        return np.ones(n, dtype=np.float64)
    return m / nrm


def build_modes(n, rng, max_modes=40):
    modes = []
    n = int(max(2, n))
    t = np.linspace(0.0, 1.0, n, endpoint=False)

    max_f = max(1, min(12, n // 20))
    for f in range(1, max_f + 1):
        if len(modes) >= max_modes:
            break
        add_mode(modes, np.cos(2.0 * np.pi * f * t))
        if len(modes) >= max_modes:
            break
        add_mode(modes, np.sin(2.0 * np.pi * f * t))

    if len(modes) < max_modes:
        add_mode(modes, t)
    if len(modes) < max_modes:
        add_mode(modes, t * t)
    if len(modes) < max_modes:
        add_mode(modes, (t - 0.5) ** 3)

    for B in (8, 16, 32, 64, 128, 256):
        if len(modes) >= max_modes:
            break
        if B > n:
            continue
        block_count = max(1, n // B)
        bweights = rng.normal(size=block_count)
        m = np.repeat(bweights, B)[:n]
        add_mode(modes, m)

    if len(modes) < max_modes:
        for _ in range(8):
            if len(modes) >= max_modes:
                break
            add_mode(modes, random_smooth_mode(n, rng, max_cycles=6))

    if len(modes) < max_modes:
        x = np.arange(n, dtype=np.float64)
        for _ in range(6):
            if len(modes) >= max_modes:
                break
            center = rng.random() * n
            width = rng.uniform(0.01, 0.08) * n
            m = np.exp(-((x - center) ** 2) / (2.0 * width * width))
            add_mode(modes, m)

    if len(modes) > max_modes:
        modes = modes[:max_modes]
    return modes


def modulate(base, mode, theta):
    if base is None or mode is None:
        return None
    z = float(theta) * mode
    z = np.clip(z, -8.0, 8.0)
    b = base * np.exp(z)
    s = float(b.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return base.copy()
    b = b / s
    return np.maximum(b, 0.0)


def modulate_matrix(base, modes_matrix, theta):
    if base is None or modes_matrix is None:
        return base.copy() if base is not None else None
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0 or modes_matrix.size == 0:
        return base.copy()
    z = np.dot(theta, modes_matrix)
    z = np.clip(z, -8.0, 8.0)
    b = base * np.exp(z)
    s = float(b.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return base.copy()
    b = b / s
    return np.maximum(b, 0.0)

def search_one_mode(state, base, mode, deadline, base_score):
    global MOD_BASE, MOD_MODE, GLOBAL_OBJ_DEADLINE
    if base is None or mode is None:
        return False, base_score

    local_best_a = base
    local_best_score = float(base_score)
    best_theta = 0.0

    if USE_FAST_PREFILTER:
        base_fast = fast_score(base)
        best_fast = float(base_fast)
        best_fast_theta = 0.0
        for th in THETA_GRID:
            if time.monotonic() > deadline - 10.0:
                break
            a = modulate(base, mode, th)
            if a is None:
                continue
            sf = fast_score(a)
            if np.isfinite(sf) and sf < best_fast - 1e-10:
                best_fast = float(sf)
                best_fast_theta = th
        if best_fast_theta == 0.0:
            return False, base_score
        local_best_a = modulate(base, mode, best_fast_theta)
        if local_best_a is None:
            return False, base_score
        sc = evaluate_arr(local_best_a)
        if not np.isfinite(sc) or sc >= base_score - 1e-10:
            return False, base_score
        local_best_score = float(sc)
        best_theta = best_fast_theta
    else:
        for th in THETA_GRID:
            if time.monotonic() > deadline - 10.0:
                break
            a = modulate(base, mode, th)
            if a is None:
                continue
            sc = evaluate_arr(a)
            if np.isfinite(sc) and sc < local_best_score - 1e-10:
                local_best_score = float(sc)
                local_best_a = a
                best_theta = th

    if local_best_score >= base_score - 1e-10:
        return False, base_score

    if time.monotonic() < deadline - 30.0:
        MOD_BASE = base
        MOD_MODE = mode
        GLOBAL_OBJ_DEADLINE = deadline - 5.0
        try:
            res = minimize_scalar(
                mod_objective,
                bounds=(-2.0, 2.0),
                method="bounded",
                options={"maxiter": 18, "xatol": 1e-4},
            )
            if hasattr(res, "x"):
                th = float(res.x)
                a = modulate(base, mode, th)
                if a is not None:
                    sc = evaluate_arr(a)
                    if np.isfinite(sc) and sc < local_best_score - 1e-12:
                        local_best_score = float(sc)
                        local_best_a = a
        except Exception:
            pass

    if local_best_score < float(state.get("best_score", np.inf)) - 1e-12:
        update_best(state, local_best_a, local_best_score)
        improved = local_best_score < float(base_score) - 1e-12
        return improved, float(state["best_score"])
    return False, float(state.get("best_score", base_score))


def mode_coordinate_search(state, deadline, max_modes=36, max_rounds=2):
    good_ids = []
    base = state.get("best_a")
    if base is None:
        return good_ids, []
    n = int(base.size)
    rng = ensure_rng()
    modes = build_modes(n, rng, max_modes)
    if not modes:
        return good_ids, modes

    base_score = float(state.get("best_score", np.inf))
    for r in range(max_rounds):
        if time.monotonic() > deadline - 40.0:
            break
        pass_improved = False
        for idx in range(min(max_modes, len(modes))):
            if time.monotonic() > deadline - 40.0:
                break
            mode = modes[idx]
            improved, old_score = base_score, base_score
            improved, new_score = search_one_mode(state, base, mode, deadline, base_score)
            if improved and new_score < old_score - 1e-12:
                base = state.get("best_a", base)
                base_score = float(state.get("best_score", np.inf))
                pass_improved = True
                if idx not in good_ids:
                    good_ids.append(idx)
        if not pass_improved:
            break
    return good_ids, modes


def multi_search(state, modes, good_ids, deadline, rng, max_seconds=220):
    global MULTI_BASE, MULTI_MODES, GLOBAL_OBJ_DEADLINE
    base = state.get("best_a")
    if base is None or not modes:
        return
    start = time.monotonic()
    end = min(deadline - 5.0, start + float(max_seconds))
    if end <= start:
        return

    selected = []
    for idx in good_ids:
        if 0 <= idx < len(modes) and idx not in selected:
            selected.append(idx)
        if len(selected) >= 6:
            break
    for idx in range(len(modes)):
        if len(selected) >= 6:
            break
        if idx not in selected:
            selected.append(idx)
    if not selected:
        return

    M = np.vstack([modes[i] for i in selected]).astype(np.float64)
    MULTI_BASE = base.copy()
    MULTI_MODES = M
    d = int(M.shape[0])

    for rep in range(2):
        if time.monotonic() > end - 10.0:
            break
        GLOBAL_OBJ_DEADLINE = end
        if rep == 0:
            x0 = np.zeros(d, dtype=np.float64)
        else:
            x0 = np.asarray(rng.normal(0.0, 0.15, size=d), dtype=np.float64)
        try:
            res = minimize(
                multi_objective,
                x0,
                method="Nelder-Mead",
                options={"maxiter": 130, "fatol": 1e-8, "xatol": 1e-4},
            )
        except Exception:
            res = None
        if res is not None and hasattr(res, "x") and time.monotonic() < end - 1.0:
            theta = np.asarray(res.x, dtype=np.float64)
            a = modulate_matrix(base, M, theta)
            if a is not None:
                sc = evaluate_arr(a)
                if np.isfinite(sc):
                    update_best(state, a, sc)
            base = state.get("best_a", base)


def upsample_sequence(a, factor=2, max_len=12000):
    a = normalize_raw(a)
    if a is None:
        return None
    new_n = int(min(a.size * int(factor), max_len))
    if new_n <= a.size:
        return a
    x_old = np.linspace(0.0, 1.0, a.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    b = np.interp(x_new, x_old, a)
    return normalize_raw(b)


def decimate_sequence(a, target_len):
    a = normalize_raw(a)
    if a is None:
        return None
    n = int(a.size)
    target_len = int(max(2, target_len))
    if n <= target_len:
        return a
    block = max(1, n // target_len)
    m = (n // block)
    b = a[:m * block].reshape(m, block).mean(axis=1)
    return normalize_raw(b)


def make_heuristic(n, kind, rng):
    n = int(max(2, n))
    i = np.arange(n, dtype=np.float64)
    t = i / max(1, n - 1)
    if kind == 0:
        return np.ones(n, dtype=np.float64)
    if kind == 1:
        return rng.dirichlet(np.ones(n) * 0.5)
    if kind == 2:
        m = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        return np.maximum(m, 1e-6)
    if kind == 3:
        return np.linspace(2.0, 0.3, n, dtype=np.float64)
    if kind == 4:
        return np.exp(-np.linspace(0.0, 3.0, n, dtype=np.float64))
    if kind == 5:
        m = np.ones(n, dtype=np.float64)
        m[: n // 2] = 1.5
        m[n // 2:] = 0.7
        return m
    if kind == 6:
        m = np.where(rng.random(n) > 0.5, 1.0, 0.1).astype(np.float64)
        return m + 1e-3
    if kind == 7:
        m = 1.0 - np.abs(t - 0.5) * 2.0
        return np.maximum(m, 0.1)
    if kind == 8:
        m = 0.3 + np.abs(t - 0.5) * 2.0
        return np.maximum(m, 1e-3)
    if kind == 9:
        m = np.zeros(n, dtype=np.float64)
        m[n // 2] = 1.0
        return m
    if kind == 10:
        m = np.zeros(n, dtype=np.float64)
        m[0] = 1.0
        return m
    return np.ones(n, dtype=np.float64)


def finalize_sequence(state):
    a = state.get("best_a")
    if a is None:
        a = np.ones(10, dtype=np.float64)
    a = normalize_raw(a)
    if a is None:
        a = np.ones(10, dtype=np.float64)
    sc = evaluate_arr(a)
    if np.isfinite(sc):
        update_best(state, a, sc)
        a = state.get("best_a", a)
    a = normalize_raw(a)
    if a is None:
        a = np.ones(10, dtype=np.float64)
    return [float(v) for v in a.tolist()]


def propose_candidate(budget_s=2400, **kwargs):

    t0 = time.monotonic()
    deadline = t0 + budget - 45.0
    if deadline < t0 + 10.0:
        deadline = t0 + 5.0

    rng = ensure_rng(20240607)
    state = {}

    parent = None
    try:
        if "height_sequence_1" in globals():
            parent = normalize_raw(globals()["height_sequence_1"])
    except Exception:
        parent = None

    if parent is None:
        target_n = 4806
    else:
        target_n = int(parent.size)
        if target_n > 20000:
            parent = decimate_sequence(parent, 8000)
            target_n = int(parent.size) if parent is not None else 4806

    cands = []

        cands.append(parent)

    if target_n <= 8000:
        imp = np.zeros(target_n, dtype=np.float64)

        cands.append(imp)

    for k in (2, 3, 4, 5, 6, 7, 8):

    max_initial = 12 if target_n <= 8000 else 7
    for i, c in enumerate(cands):
        if i >= max_initial or time.monotonic() > deadline - 30.0:
            break
        arr = normalize_raw(c)
        if arr is None:
            continue
        sc = evaluate_arr(arr)
        if np.isfinite(sc):
            update_best(state, arr, sc)
        if state.get("best_score", np.inf) <= 1e-8:
            break

    if state.get("best_a") is None:
        fallback = np.ones(10, dtype=np.float64)
        sc = evaluate_arr(fallback)
        if np.isfinite(sc):
            update_best(state, fallback, sc)
        else:
            return [1.0] * 10

    print(
        "initial best score=%.12f len=%d" % (
            float(state.get("best_score", np.inf)),
            int(state.get("best_n", 0)),
        ),
        flush=True,
    )
    if float(state.get("best_score", np.inf)) <= 1e-7:
        return finalize_sequence(state)

    if int(state.get("best_n", 0)) < 8000:
        up = upsample_sequence(state["best_a"], 2, 12000)
        if up is not None and time.monotonic() < deadline - 30.0:
            sc = evaluate_arr(up)
            if np.isfinite(sc):
                update_best(state, up, sc)
            if float(state.get("best_score", np.inf)) <= 1e-7:
                return finalize_sequence(state)

    best = state.get("best_a")
    if best is not None:
        best_fast = fast_score(best)
        USE_FAST_PREFILTER = not (float(state.get("best_score", np.inf)) < float(best_fast) - 0.1)
        print(
            "fast_score_base=%.6f actual_base=%.6f prefilter=%s" % (
                best_fast, float(state.get("best_score", np.inf)), USE_FAST_PREFILTER
            ),
            flush=True,
        )
    else:
        USE_FAST_PREFILTER = False

    good_ids = []
    modes1 = []
    if time.monotonic() < deadline - 160.0:
        max_modes = 36 if int(state.get("best_n", 4096)) <= 8000 else 22
        good_ids, modes1 = mode_coordinate_search(
            state, deadline, max_modes=max_modes, max_rounds=2
        )
        if modes1 and time.monotonic() < deadline - 80.0:
            multi_search(state, modes1, good_ids, deadline, rng, max_seconds=180)
        good_ids = []
        modes1 = []

    best = state.get("best_a")
    if best is not None:
        best_fast = fast_score(best)
        USE_FAST_PREFILTER = not (float(state.get("best_score", np.inf)) < float(best_fast) - 0.1)

    if USE_FAST_PREFILTER and time.monotonic() < deadline - 180.0:
        remaining = max(60.0, (deadline - time.monotonic()) * 0.25)
        polish_subgrad(state, state["best_a"], deadline, max_seconds=min(500.0, remaining), actual_interval=80)

        if start_a is not None:

            m = random_smooth_mode(best.size, rng, max_cycles=3)
            cand = modulate(best, m, th)
            if cand is None:
                continue
            sc = evaluate_arr(cand)
            if np.isfinite(sc):
                update_best(state, cand, sc)

    return finalize_sequence(state)
