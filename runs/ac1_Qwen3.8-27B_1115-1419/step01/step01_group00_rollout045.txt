import time
import math

def _conv(a):
    return np.convolve(a, a)

def _obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * np.max(_conv(a)) / (s * s)

def _project(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _softmax_obj(a, tau):
    c = _conv(a)
    mx = c.max()
    e = np.exp((c - mx) / tau)
    return tau * np.log(e.sum()) + mx

def _softmax_grad(a, tau):
    c = _conv(a)
    mx = c.max()
    e = np.exp((c - mx) / tau)
    p = e / e.sum()
    n = len(a)
    g = 2.0 * np.correlate(p, a, 'valid')
    g -= g.mean()
    return g

def _adam_smooth(a, tau0, tau_min, lr, deadline, tag):
    """Adam-smoothed gradient descent."""
    n = len(a)
    tau = tau0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best = a.copy()
    best_o = _obj(a)
    t = 0
    while time.monotonic() < deadline:
        t += 1
        g = _softmax_grad(a, tau)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project(a_new, n)
        tau = max(tau * 0.9999, tau_min)
        o = _obj(a)
        if o < best_o:
            best_o = o
            best = a.copy()
        if t % 5000 == 0:
            print(f"  [{tag}] t={t} obj={best_o:.6f} tau={tau:.1f}")
    return best, best_o
