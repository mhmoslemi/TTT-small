def _fft_conv(a, b):
    """FFT-based linear convolution."""
    n_out = len(a) + len(b) - 1
    L = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, L) * fft(b, L)).real[:n_out]


def _lse_and_grad(x, alpha):
    """Compute LSE objective and gradient for normalized autocorrelation."""
    n = len(x)
    s = np.sum(x)
    if s < 1e-15:
        return 1e10, np.zeros(n)
    an = x / s
    c = _fft_conv(an, an)
    mc = np.max(c)
    shifted = alpha * (c - mc)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    lse_val = mc + np.log(sum_exp) / alpha
    
    # Gradient: g_an[j] = 2*alpha * sum_k w[k]*an[k-j]*1_{0<=k-j<n}
    # = 2*alpha * correlate(w, an, 'valid')[j]
    g_an = 2.0 * alpha * np.correlate(w, an, mode='valid')
    
    # Chain rule through normalization: an = x/s
    g_an_dot_an = np.dot(g_an, an)
    g_x = (g_an - g_an_dot_an * an) / s
    
    return lse_val, g_x


def _run_lbfgsb(x0, alpha, maxiter=50, ftol=1e-12, gtol=1e-10):
    """Run L-BFGS-B optimization with LSE surrogate at given alpha."""

    a = a.copy()

        c = _fft_conv(a, a)
        # Find top peaks
        top_idx = np.argsort(c)[-n_peaks:]
        peak_val = c[top_idx[0]]
        
        # For each peak, compute the subgradient direction
        # and take a small step to reduce it
        for k in top_idx:
            # Reduce c[k] by decreasing a[j] proportional to a[k-j]
            j_range = np.arange(max(0, k - len(a) + 1), min(k, len(a) - 1) + 1)
            contrib = a[np.clip(k - j_range, 0, len(a) - 1)]
            # a_new[j] -= eta * a[k-j] for valid j

                a[j] -= eta * a[int(k) - int(j)]
        
        a = np.maximum(a, 0)
        a = a / np.sum(a)
    
    return a

LSE + L-BFGS-B autocorrelation minimization with alpha annealing and peak shaving.
Uses log-sum-exp surrogate for smooth optimization of max autocorrelation,
L-BFGS-B quasi-Newton local optimization, and multi-start exploration.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """FFT-based linear convolution."""
    n_out = len(a) + len(b) - 1
    L = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, L) * fft(b, L)).real[:n_out]


def _lse_and_grad(x, alpha):
    """Compute LSE objective and gradient for normalized autocorrelation min-max problem."""
    n = len(x)
    s = np.sum(x)
    if s < 1e-15:
        return 1e10, np.zeros(n)
    an = x / s
    c = _fft_conv(an, an)
    mc = float(np.max(c))
    shifted = alpha * (c - mc)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    if sum_exp < 1e-300:
        return mc, np.zeros(n)
    w = exp_s / sum_exp
    lse_val = mc + np.log(sum_exp) / alpha

    # Gradient of LSE w.r.t. an: g_an[j] = 2*alpha * sum_k w[k]*an[k-j]*1_{0<=k-j<n}
    g_an = 2.0 * alpha * np.correlate(w, an, mode='valid')

    # Chain rule: an = x/s, so dLSE/dx[j] = (g_an[j] - sum_l g_an[l]*an[l]) / s
    g_an_dot_an = float(np.dot(g_an, an))

    return lse_val, g_x

    F = fft(best)
    F *= (1 + 0.05 * rng.standard_normal(len(F)))
    x = ifft(F).real
    candidates.append(_normalize(x))
    return candidates


def propose_candidate(seed=42, budget_s=2400):
    """
    Minimize 2n*max(autocorr)/(sum)^2 using LSE+L-BFGS-B with alpha annealing,
    peak shaving, and multi-start exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    n = 5582
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.sum() < 0.01:
                parent = np.ones(n)
            n = parent.size
    except (NameError, Exception):
        parent = np.ones(n)

    best = _normalize(parent)
    best_score = _score(best)
    print(f"Initial: n={n}, score={best_score:.6f}")

    # === Phase 1: Alpha annealing with L-BFGS-B ===
    print("\n=== Phase 1: LSE + L-BFGS-B annealing ===")
    alphas_coarse = [5, 10, 20, 40, 70, 100, 150, 200]
    for alpha in alphas_coarse:
        if time.monotonic() > deadline - 180:
            break
        t0 = time.monotonic()
        x = _run_lbfgsb(best, alpha, maxiter=80)
        x = _normalize(x)
        sc = _score(x)
        el = time.monotonic() - t0
        if sc < best_score - 1e-12:
            best = x.copy()
            best_score = sc
            print(f"  a={alpha:4d}: {best_score:.6f} ({el:.1f}s) *")
        else:
            print(f"  a={alpha:4d}: {sc:.6f} ({el:.1f}s)")

    # === Phase 2: Fine L-BFGS-B with high alpha ===
    print("\n=== Phase 2: Fine-tuning ===")
    alphas_fine = [300, 500, 800, 1200, 2000, 3000]
    for alpha in alphas_fine:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        x = _run_lbfgsb(best, alpha, maxiter=50, ftol=1e-14, gtol=1e-12)
        x = _normalize(x)
        sc = _score(x)
        el = time.monotonic() - t0
        if sc < best_score - 1e-12:
            best = x.copy()
            best_score = sc
            print(f"  a={alpha:4d}: {best_score:.6f} ({el:.1f}s) *")
        else:
            print(f"  a={alpha:4d}: {sc:.6f} ({el:.1f}s)")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak shaving (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 60:
        for n_steps in [100, 300]:
            x = _peak_shave(best.copy(), n_steps=n_steps, n_peaks=2)
            sc = _score(x)
            if sc < best_score:
                best = x.copy()
                best_score = sc
                print(f"  peak_shave({n_steps}): {best_score:.6f} *")
            else:
                print(f"  peak_shave({n_steps}): {sc:.6f}")

    if time.monotonic() < deadline - 60:
        for n_steps in [200]:
            x = _weight_shave(best.copy(), n_iters=n_steps)
            sc = _score(x)
            if sc < best_score:
                best = x.copy()
                best_score = sc
                print(f"  weight_shave({n_steps}): {best_score:.6f} *")
            else:
                print(f"  weight_shave({n_steps}): {sc:.6f}")

    # === Phase 4: L-BFGS-B after peak shaving ===
    print(f"\n=== Phase 4: Post-shave L-BFGS-B ===")
    if time.monotonic() < deadline - 60:
        for alpha in [500, 1000, 2000]:
            if time.monotonic() > deadline - 60:
                break
            x = _run_lbfgsb(best, alpha, maxiter=40)
            x = _normalize(x)
            sc = _score(x)
            if sc < best_score:
                best = x.copy()
                best_score = sc
                print(f"  post-shave a={alpha}: {best_score:.6f} *")

    # === Phase 5: Multi-start exploration ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 60:
        candidates = _escape_local(best, rng, n)
        for i, cand in enumerate(candidates):
            if time.monotonic() > deadline - 60:
                break
            for alpha in [300, 800]:
                if time.monotonic() > deadline - 60:
                    break
                x = _run_lbfgsb(cand, alpha, maxiter=30)
                x = _normalize(x)
                sc = _score(x)
                if sc < best_score:
                    best = x.copy()
                    best_score = sc
                    print(f"  start {i}, a={alpha}: {best_score:.6f} *")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 20:
        # Very high alpha L-BFGS-B
        x = _run_lbfgsb(best, 5000, maxiter=30, ftol=1e-15, gtol=1e-13)
        x = _normalize(x)
        sc = _score(x)
        if sc < best_score:
            best = x.copy()
            best_score = sc
            print(f"  high-alpha: {best_score:.6f} *")

        # One more shave + optimize cycle
        x = _peak_shave(best.copy(), n_steps=100, n_peaks=2)
        x = _run_lbfgsb(x, 2000, maxiter=30)
        x = _normalize(x)
        sc = _score(x)
        if sc < best_score:
            best = x.copy()
            best_score = sc
            print(f"  shave+opt: {best_score:.6f} *")

    # Final evaluation with actual evaluator
    final_list = [float(max(0.0, x)) for x in best]
    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = best_score
    print(f"\nFinal evaluate_sequence score: {final_score:.6f}")
    print(f"Tracked best score: {best_score:.6f}")

    # Return the better of tracked vs evaluated
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    # Sanity check
    result = [float(max(0.0, min(1000.0, x))) for x in best]
    return result
