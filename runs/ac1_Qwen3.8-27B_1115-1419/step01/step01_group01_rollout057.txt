import numpy as np

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _conv_self(a, m):
    n = a.size
    F = np.fft.rfft(a, m)
    b = np.fft.irfft(F * F, m)[:2 * n - 1]
    return np.maximum(b, 0.0)


def _score(a, m=None):
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-30:
        return np.inf
    if m is None:
        m = _next_pow2(2 * n - 1)
    b = _conv_self(a, m)
    bm = float(b.max())
    if not np.isfinite(bm) or bm <= 0:
        return np.inf
    return float(2.0 * n * bm / (S * S))


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(a.size) / a.size
    return a / s


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    vm = float(v.max())
    e = np.exp(v - vm)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones(v.shape, dtype=np.float64) / v.size
    return e / s


def _interp_to_length(a, n):

    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    if a.size == n:
        return a / s
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return out / ss

        # db[kmax]/da[j] = a[kmax-j] + (a[j] if kmax-j != j and 0<=kmax-j<n else 0)
        # Let me redo:
        # b[kmax] = sum_{j=lo}^{hi} a[j] * a[kmax-j]
        # db[kmax]/da[j] for j in [lo, hi]:
        #   = a[kmax-j] (from differentiating a[j] in the term a[j]*a[kmax-j])
        #   + a[kmax-j] (from differentiating a[kmax-j] in the term a[j]*a[kmax-j], IF kmax-j != j)
        #   Actually no. Let's be careful.
        # b[k] = sum_j a[j] * a[k-j]
        # db[k]/da[j] = d/d a[j] [sum_i a[i] * a[k-i]]
        # = a[k-j] (from the i=j term) + a[j] * d/d a[j] [a[k-j] where k-j=j, i.e., the term a[k-j]*a[j] with i=k-j]
        # = a[k-j] + a[j] (if k-j=j, i.e., k=2j)
        # = a[k-j] + (a[k-j] if k=2j else 0) ... wait no.
        # = a[k-j] + (a[j] if 0<=k-j<n and k-j != j, from the term where i = k-j)
        # Hmm, let me think again.
        # b[k] = sum_i a[i] * a[k-i]
        # The terms involving a[j] are:
        #   i = j: contributes a[j] * a[k-j], derivative w.r.t. a[j] is a[k-j]
        #   i = k-j: contributes a[k-j] * a[j], derivative w.r.t. a[j] is a[k-j] (if k-j != j)
        #   i = k-j = j (when k=2j): contributes a[j]*a[j] = a[j]^2, derivative is 2*a[j]
        # So: db[k]/da[j] = a[k-j] + a[k-j] (if k-j != j) = 2*a[k-j] (if k-j != j)
        #                       = 2*a[j] (if k = 2j)
        # In both cases it's 2*a[k-j] (since when k=2j, k-j=j, so 2*a[k-j] = 2*a[j]).
        # So db[k]/da[j] = 2*a[k-j] for all valid j (where 0<=k-j<n).

        # OK so the subgradient is simply 2*a[kmax-j] for j in valid range.
        grad = np.zeros(n, dtype=np.float64)
        for j in range(max(0, kmax - n + 1), min(kmax, n - 1) + 1):
            grad[j] = 2.0 * a[kmax - j]

        # Projected gradient step on the simplex {a >= 0, sum(a) = 1}
        lr = lr0 / (1.0 + 0.01 * it)
        a_new = a - lr * grad
        # Project onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s

        if symmetric:
            a_new = _make_symmetric(a_new)
            a_new = _normalize(a_new)

        sc = _score(a_new, m)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a - lr * 0.5 * grad
            a_try = np.maximum(a_try, 0.0)
            st = float(a_try.sum())
            if st > 1e-12:
                a_try = a_try / st
                if symmetric:
                    a_try = _normalize(_make_symmetric(a_try))
                sc_try = _score(a_try, m)
                if np.isfinite(sc_try) and sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a, float(best_score)


def _generate_inits(n, rng, prev_a=None, m=None):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    if prev_a is not None:
        inits.append(_interp_to_length(prev_a, n))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)

    # Exponentials
    for c in (0.5, 1.0, 2.0, 4.0):
        a = np.exp(-c * t)
        inits.append(_normalize(a))

    # Symmetric envelopes
    for c in (0.5, 1.0, 2.0):
        a = np.exp(-c * (t - 0.5) ** 2)
        inits.append(_normalize(a))

    # Power-law edges
    left = i_arr + 1.0
    right = n - i_arr
    for p in (0.25, 0.5, 1.0):
        a = (left * right) ** (-p)
        inits.append(_normalize(a))

    # Cosine shapes
    for k in (1, 2, 3):
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
        inits.append(_normalize(np.maximum(a, 0.01)))

    # Random smooth
    kc = min(10, n // 4)
    a = np.ones(n, dtype=np.float64)
    for k in range(1, kc + 1):
        amp = 0.3 / (k ** 0.8)
        ph = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + ph)
    inits.append(_normalize(np.maximum(a, 0.0)))

    return inits


def _process_n(n, deadline, prev_best_a, rng, total_budget):
    """Process a single sequence length n."""
    now = time.monotonic()
    if now >= deadline:
        return prev_best_a, (_score(prev_best_a) if prev_best_a is not None else np.inf)

    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1

    best_a = None
    best_score = np.inf

    if prev_best_a is not None:
        best_a = _interp_to_length(prev_best_a, n)
        best_score = _score(best_a, m)

    # Generate initializations
    inits = _generate_inits(n, rng, best_a, m)

    # Spectral initialization
    spec_time = min(80.0, 0.15 * (deadline - now))
    spec_dl = min(deadline - 10.0, time.monotonic() + max(5.0, spec_time))
    _spectral_init._deadline = spec_dl if hasattr(_spectral_init, '_deadline') else spec_dl

    # Flat target
    flat_target = np.ones(L, dtype=np.float64)
    # Slightly elevated center target
    center_boost = np.ones(L, dtype=np.float64)
    center_boost[n-1] = 1.5

    for tgt in (flat_target, center_boost):
        if time.monotonic() >= spec_dl:
            break
        for a_init in inits[:3]:
            if time.monotonic() >= spec_dl:
                break
            a_spec, sc_spec = _spectral_init(tgt, a_init, n, m, iters=50, do_symmetry=True)
            if np.isfinite(sc_spec) and sc_spec < best_score:
                best_score = sc_spec
                best_a = a_spec
            inits.append(a_spec)
            if time.monotonic() >= spec_dl:
                break
        if time.monotonic() >= spec_dl:
            break

    # Score all inits and pick top candidates
    scored = []
    seen = []
    for a in inits:
        a = _normalize(a)
        sc = _score(a, m)
        if not np.isfinite(sc):
            continue
        diverse = True
        for b in seen:
            d = float(np.sqrt(np.sum((a - b) ** 2)))
            if d < 0.03:
                diverse = False
                break
        if diverse:
            scored.append((sc, a))
            seen.append(a)
            if len(seen) >= 12:
                break

    scored.sort(key=lambda x: x[0])

    # Select top candidates
    cands = [x[1] for x in scored[:4]]
    if not cands:
        cands = [np.ones(n, dtype=np.float64) / n]

    # Run soft-max optimization on each candidate
    opt_time = max(10.0, (deadline - now) * 0.65)
    per_cand = opt_time / len(cands)

    for ci, a_cand in enumerate(cands):
        now2 = time.monotonic()
        if now2 >= deadline - 15.0:
            break

        v = np.log(np.maximum(a_cand, 1e-30))
        v = v - float(v.mean())

        alloc = min(per_cand * 0.7, deadline - now2 - 30.0)
        if alloc < 5.0:
            break

        n_iters = max(200, int(alloc * 800))  # Estimate iterations
        _optimize_soft._deadline = now2 + alloc

        a_opt, sc_opt = _optimize_soft(v, n, m, n_iters, symmetric=True,
                                       seed=int(rng.integers(0, 2**31)),
                                       tau_init=0.5, tau_final=0.003,
                                       lr_init=0.12, lr_final=0.005)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
        print("      n=%d cand=%d score=%.6f" % (n, ci, best_score))

        # Second pass with different params (less symmetric exploration)
        now3 = time.monotonic()
        alloc2 = min(per_cand * 0.3, deadline - now3 - 25.0)
        if alloc2 > 5.0:
            n_iters2 = max(100, int(alloc2 * 800))
            _optimize_soft._deadline = now3 + alloc2
            a_opt2, sc_opt2 = _optimize_soft(v, n, m, n_iters2, symmetric=False,
                                             seed=int(rng.integers(0, 2**31)),
                                             tau_init=0.3, tau_final=0.005,
                                             lr_init=0.10, lr_final=0.01)
            if np.isfinite(sc_opt2) and sc_opt2 < best_score:
                best_score = sc_opt2
                best_a = a_opt2
            print("      n=%d cand=%d asym score=%.6f" % (n, ci, best_score))

    # Subgradient polishing
    if best_a is not None:
        now4 = time.monotonic()
        polish_time = min(60.0, max(5.0, deadline - now4 - 10.0))
        if polish_time > 5.0:
            _subgrad_polish._deadline = now4 + polish_time
            a_pol, sc_pol = _subgrad_polish(best_a, n, m,
                                            max_iters=5000, symmetric=True,
                                            seed=int(rng.integers(0, 2**31)))
            if np.isfinite(sc_pol) and sc_pol < best_score:
                best_score = sc_pol
                best_a = a_pol
            print("      n=%d polished score=%.6f" % (n, best_score))

    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    seed = int(seed)
    budget_s = float(max(60.0, budget_s))
    start = time.monotonic()
    global_deadline = start + max(60.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    # Load previous best
    prev = None
    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 0.0:
                prev = arr / float(arr.sum())
    except Exception:
        prev = None

    best_a = prev
    best_score = _score(prev) if prev is not None else np.inf
    print("Initial previous score: %.6f (n=%d)" % (best_score, len(prev) if prev is not None else 0))

    # Stage definitions: (n, time_fraction)
    # Allocate more time to larger n
    stage_defs = [
        (2048, 0.08),
        (4096, 0.25),
        (8192, 0.45),
        (16384, 0.22),
    ]

    base_time = max(60.0, budget_s - 35.0)
    used_frac = 0.0

    for n, frac in stage_defs:
        now = time.monotonic()
        if now + 60.0 >= global_deadline:
            break

        stage_time = base_time * frac
        stage_deadline = min(global_deadline - 20.0, now + stage_time)

        # Check if this stage is feasible
        if stage_deadline - now < 30.0:
            continue

        print("\n=== Stage n=%d, budget=%.1fs, current_best=%.6f ===" %
              (n, stage_time, best_score))

        a_new, sc_new = _process_n(int(n), stage_deadline, best_a, rng, stage_time)

        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_a = a_new
            best_score = sc_new
            print("  -> New best: %.6f" % best_score)
        else:
            print("  -> No improvement (best=%.6f)" % best_score)

    # Final processing: if we have time, do one more round on the best n
    now = time.monotonic()
    if now < global_deadline - 60.0 and best_a is not None:
        n_best = len(best_a)
        final_time = min(120.0, global_deadline - now - 15.0)
        if final_time > 30.0:
            print("\n=== Final polish n=%d, budget=%.1fs ===" % (n_best, final_time))
            m = _next_pow2(2 * n_best - 1)
            v = np.log(np.maximum(best_a, 1e-30))
            v = v - float(v.mean())
            _optimize_soft._deadline = now + final_time * 0.7
            n_iters = max(500, int(final_time * 600))
            a_fin, sc_fin = _optimize_soft(v, n_best, m, n_iters, symmetric=True,
                                           seed=int(rng.integers(0, 2**31)),
                                           tau_init=0.2, tau_final=0.002,
                                           lr_init=0.08, lr_final=0.003)
            if np.isfinite(sc_fin) and sc_fin < best_score:
                best_score = sc_fin
                best_a = a_fin
                print("  -> Final: %.6f" % best_score)

    # Ensure valid output
    if best_a is None:
        best_a = np.ones(1024, dtype=np.float64) / 1024
    best_a = _normalize(best_a)

    # Verify with exact evaluation
    try:
        if len(best_a) <= 20000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("\nFinal: FFT score=%.6f, exact score=%.6f, n=%d" %
                  (best_score, exact, len(best_a)))
            if not np.isfinite(exact):
                print("  WARNING: exact evaluation failed, using uniform fallback")
                best_a = np.ones(1024, dtype=np.float64) / 1024
        else:
            print("\nFinal: FFT score=%.6f, n=%d (exact skipped)" % (best_score, len(best_a)))
    except Exception as e:
        print("Exact validation error: %s" % str(e))

    elapsed = time.monotonic() - start
    print("Total time: %.1fs" % elapsed)

    return [float(x) for x in best_a]
