    corr = np.fft.ifft(A * W_rev).real

def _next_pow2(x):
    x = max(1, int(x))
    p = 1
    while p < x:
        p <<= 1
    return p

def _score(a):
    """Compute objective via FFT."""
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return np.inf
    m = _next_pow2(2*n-1)
    F = np.fft.rfft(a, m)
    b = np.fft.irfft(F * F, m)[:2*n-1]
    bmax = b.max()
    if not np.isfinite(bmax):
        return np.inf
    return 2.0*n*bmax/(S*S)

def _interp(a, n_new):
    """Interpolate sequence to new length."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n_new) / n_new
    a = a / s
    x_old = np.linspace(0, 1, a.size)
    x_new = np.linspace(0, 1, n_new)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0)
    ss = out.sum()
    if ss < 1e-12:
        out[:] = 1.0/n_new
    else:
        out /= ss
    return out
