def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    N = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, N) * fft(b, N)).real[:n_out]

def _fft_grad(w, a):
    """Compute grad[i] = 2*T * sum_j w[i+j] * a[j] using FFT.
    Actually, compute h[i] = sum_j w[i+j] * a[j], then multiply by 2*T outside."""
    n = len(a)
    len_w = len(w)
    a_rev = a[::-1]
    N = 1 << (len_w + n - 1 - 1).bit_length()
    result = ifft(fft(w, N) * fft(a_rev, N)).real
    return result[n-1 : n-1+n]

def _score_fast(a):
    n = len(a)
    c = _fft_conv(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target_sum):
    S = float(np.sum(a))
    if S < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / S)

def _compute_softgrad(a, T):
    """Compute the gradient of the smooth approximation to max(c)."""
    n = len(a)
    c = _fft_conv(a, a)
    # Softmax with shift for stability
    c_shifted = c - np.max(c)
    w = np.exp(T * c_shifted)
    w /= np.sum(w)
    # Gradient: grad_i = 2*T * sum_j w[i+j] * a[j]
    h = _fft_grad(w, a)
    grad = 2.0 * T * h
    return grad, c, w

def _smoothed_grad_descent(a, deadline, T_init=1.0, T_max=50.0, T_rate=1.005,
                           lr_init=0.05, lr_decay=0.9999, momentum_beta=0.9,
                           eval_interval=100):

    n = len(a)
    S = np.sqrt(2.0 * n)
    a = _normalize(a.copy(), S)
    momentum = np.zeros(n)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_score = _score_fast(a)
    stuck_count = 0
    step = 0
    
    t0 = time.monotonic()
    while time.monotonic() < deadline:
        grad, c, w = _compute_softgrad(a, T)
        
        # Momentum update
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        a = _normalize(a, S)
        
        # Anneal temperature
        T = min(T * T_rate, T_max)
        lr = max(lr * lr_decay, 1e-8)
        step += 1
        
        if step % eval_interval == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                stuck_count = 0
            else:
                stuck_count += 1
            
            if stuck_count > 30:
                # Perturb and restart
                a = best_a + 0.1 * np.random.standard_normal(n) * (S / n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, S)
                momentum = np.zeros(n)
                T = T_init
                lr = lr_init
                stuck_count = 0

import numpy as np
import time
from scipy.fft import fft, ifft

def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    N = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, N) * fft(b, N)).real[:n_out]

def _fft_grad_raw(w, a):
    """Compute h[i] = sum_j w[i+j] * a[j] for i=0..n-1 using FFT."""
    n = len(a)
    len_w = len(w)
    a_rev = a[::-1]
    N = 1 << (len_w + n - 1 - 1).bit_length()
    result = ifft(fft(w, N) * fft(a_rev, N)).real
    return result[n-1:n-1+n]

def _score_fast(a):
    """Quick score using FFT convolution."""
    n = len(a)
    c = _fft_conv(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:

    if S < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / S)

    n = len(a)
    S = np.sqrt(2.0 * n)
    a = _normalize(a.copy(), S)
    momentum = np.zeros(n)
    T = T_init
    lr = lr_init

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
            n_parent = parent.size
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.ones(n_parent)

    n = n_parent
    S = np.sqrt(2.0 * n)
    parent_norm = _normalize(parent.copy(), S)

    # Evaluate parent with actual evaluator
    best = parent_norm.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Parent: n={n}, score={best_score:.6f}")
    print(f"Total time budget: {budget_s}s")

    # Phase 1: Smoothed descent from parent (main phase, ~40% of time)
    print("\n=== Phase 1: Smoothed GD from parent ===")
    phase1_end = time.monotonic() + (total_deadline - time.monotonic()) * 0.40
    a1, sc1 = _smoothed_descent(
        parent_norm, phase1_end,
        T_init=0.5, T_max=100.0, T_rate=1.004,
        lr_init=0.4, lr_decay=0.99997, beta=0.9,
        eval_interval=200, perturb_scale=0.08)
    print(f"  Phase 1 FFT score: {sc1:.6f}")
    if sc1 < best_score:
        actual = float(evaluate_sequence(a1.tolist()))
        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best = a1.copy()
            print(f"  Phase 1 actual: {best_score:.6f} (improved!)")
        else:
            print(f"  Phase 1 actual: {actual:.6f}")
    else:
        print(f"  Phase 1: no improvement over {best_score:.6f}")

    # Phase 2: Different hyperparameter configurations (~30% of time)
    print("\n=== Phase 2: Hyperparameter sweep ===")
    phase2_end = time.monotonic() + (total_deadline - time.monotonic()) * 0.30
    configs = [
        # (T_init, T_max, T_rate, lr_init, lr_decay, beta, perturb)
        (1.0, 200.0, 1.005, 0.2, 0.99993, 0.95, 0.12),
        (0.3, 150.0, 1.006, 0.5, 0.99995, 0.85, 0.06),
        (2.0, 300.0, 1.002, 0.15, 0.99997, 0.92, 0.15),
        (0.8, 120.0, 1.004, 0.35, 0.99994, 0.88, 0.10),
    ]

    per_cfg = (phase2_end - time.monotonic()) / max(1, len(configs))
    for idx, (ti, tm, tr, li, ld, bt, ps) in enumerate(configs):
        if time.monotonic() > phase2_end:
            break
        # Start from best with small perturbation
        a0 = best + 0.03 * (S / n) * rng.standard_normal(n)
        a0 = np.maximum(a0, 0.0)
        a0 = _normalize(a0, S)
        local_a, local_sc = _smoothed_descent(
            a0, time.monotonic() + per_cfg,
            T_init=ti, T_max=tm, T_rate=tr,
            lr_init=li, lr_decay=ld, beta=bt,
            eval_interval=150, perturb_scale=ps)
        status = ""
        if local_sc < best_score:
            actual = float(evaluate_sequence(local_a.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = local_a.copy()
                status = f" -> IMPROVED to {best_score:.6f}"
            else:
                status = f" (FFT {local_sc:.6f}, actual {actual:.6f})"
        else:
            status = f" ({local_sc:.6f}, no improvement)"
        print(f"  Config {idx}: T={ti}->{tm}, lr={li}{status}")

    # Phase 3: Fine-tuning with high temperature, low LR (~20% of time)
    print(f"\n=== Phase 3: Fine-tuning (best={best_score:.6f}) ===")
    if time.monotonic() < total_deadline - 20:
        a0 = best.copy()
        local_a, local_sc = _smoothed_descent(
            a0, total_deadline,
            T_init=5.0, T_max=500.0, T_rate=1.002,
            lr_init=0.1, lr_decay=0.99998, beta=0.95,
            eval_interval=300, perturb_scale=0.03,
            stuck_threshold=60)
        if local_sc < best_score:
            actual = float(evaluate_sequence(local_a.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = local_a.copy()
                print(f"  Fine-tuned to {best_score:.6f}")
            else:
                print(f"  Fine-tune: FFT {local_sc:.6f}, actual {actual:.6f}")
        else:
            print(f"  Fine-tune: {local_sc:.6f} (no improvement)")

    # Phase 4: Try a couple of different lengths if time permits
    if time.monotonic() < total_deadline - 60:
        print(f"\n=== Phase 4: Alternative lengths (best={best_score:.6f}) ===")
        remaining = total_deadline - time.monotonic()
        alt_lengths = [4096, 8192]
        per_len = remaining * 0.4 / 2
        for alt_n in alt_lengths:
            if time.monotonic() > total_deadline - 30:
                break
            if per_len < 30:
                break
            print(f"  Trying n={alt_n}...")
            alt_a, alt_sc = _try_length(alt_n, time.monotonic() + per_len,
                                        parent=None, seed=seed + alt_n)
            if alt_a is not None and alt_sc < best_score:
                actual = float(evaluate_sequence(alt_a.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best = alt_a.copy()
                    print(f"  n={alt_n}: IMPROVED to {best_score:.6f}")
                else:
                    print(f"  n={alt_n}: FFT {alt_sc:.6f}, actual {actual:.6f}")
            else:
                print(f"  n={alt_n}: best={alt_sc:.6f}, no improvement")

    # Final check
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"\n=== FINAL ===")
    print(f"Final evaluated score: {final_score:.6f}")
    print(f"Best tracked score: {best_score:.6f}")
    print(f"Sequence length: {len(best)}")

    if final_score < best_score:
        best_score = final_score

    return [float(x) for x in best]
