import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def _score_fast(a):
    """Compute the score using FFT."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _soft_max_obj_grad(params, temp):
    """Objective and gradient for soft-max smoothed version."""
    n = len(params)
    a = np.maximum(params, 0.0)
    s = float(np.sum(a))
    if s < 0.01:
        return 1e10, np.zeros(n)
    
    b = _fft_conv(a, a)
    b_scaled = b / temp
    b_max = np.max(b_scaled)
    exp_b = np.exp(b_scaled - b_max)
    sum_exp = np.sum(exp_b)
    soft_max_val = temp * (b_max + np.log(sum_exp))
    
    obj = 2.0 * n * soft_max_val / (s * s)
    
    # Weights for soft-max
    w = exp_b / sum_exp
    
    # Gradient via FFT cross-correlation
    N = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = ifft(W * np.conj(A)).real[:n]
    
    grad_soft = 2.0 * corr
    grad = (2.0 * n * grad_soft) / (s * s) - (4.0 * n * soft_max_val) / (s * s * s)
    
    # Zero gradient where a is 0 (for non-negativity constraint)
    grad = np.where(a > 1e-10, grad, 0.0)
    
    return obj, grad

def _lbfgs_optimize(a0, deadline, temp_init, temp_factor=0.75, max_iters=300):
    """Run L-BFGS-B optimization with temperature annealing."""
    n = len(a0)
    best_a = a0.copy()
    best_score = _score_fast(a0)
    temp = temp_init
    
    while time.monotonic() < deadline - 10:
        # Run L-BFGS-B for this temperature
        result = minimize(
            _soft_max_obj_grad,
            a0.copy(),
            args=(temp,),
            method='L-BFGS-B',
            jac=True,
            bounds=[(0, 1000)] * n,
            options={'maxiter': max_iters, 'maxfun': max_iters*2, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        
        a0 = np.maximum(result.x, 0.0)
        sc = _score_fast(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()

    return best_a, best_score

def _subgrad_step(a, lr, grad):
    """One subgradient step with non-negativity and normalization."""
    n = len(a)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = np.sum(a_new)
    s_old = np.sum(a)
    if s_new > 0:
        a_new = a_new * (s_old / s_new)
    return a_new

def _run_subgrad(a, deadline, lr_init=0.5, lr_decay=0.999):

    n = len(a)
    best_a = a.copy()
    best_score = _score_fast(a)

        k_lo = max(0, p - n + 1)
        k_hi = min(n - 1, p)
        mirror[k_lo:k_hi+1] = a[p - np.arange(k_lo, k_hi+1)]

def _make_inits(n, rng, parent):
    """Create diverse initializations."""
    inits = []
    
    # Parent
    if parent is not None and len(parent) == n:
        inits.append(("parent", parent.copy()))
    
    # Uniform
    inits.append(("uniform", np.ones(n)))
    
    # Random
    inits.append(("random", rng.random(n) * 10 + 1))
    
    # Linear ramp
    k = np.arange(n, dtype=float)
    inits.append(("ramp", 1.0 + k / n))
    inits.append(("inv_ramp", 1.0 + (n - 1 - k) / n))
    
    # Cosine
    for freq in [1, 3, 7]:
        inits.append((f"cos_{freq}", np.maximum(1.0 + 0.8 * np.cos(2 * np.pi * freq * k / n), 0.1)))
    
    # Anti-symmetric (large on one side)
    inits.append(("asym", np.maximum(0.1, 1.0 + 1.5 * (k / n - 0.5))))
    
    # Bimodal
    inits.append(("bimodal", 0.5 + 1.5 * np.exp(-((k - n*0.25)/n*5)**2) + 1.5 * np.exp(-((k - n*0.75)/n*5)**2)))
    
    # Exponential
    inits.append(("exp", np.exp(k / n * 3)))
    
    # Low-pass filtered random
    x = rng.random(n)
    x = np.convolve(x, np.ones(10)/10, mode='same')
    inits.append(("lpf_rand", x + 0.5))
    
    return inits
