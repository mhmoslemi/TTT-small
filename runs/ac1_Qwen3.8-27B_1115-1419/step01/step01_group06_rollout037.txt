"""
Hybrid subgradient + smoothed surrogate optimizer for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses exact subgradient at the max, adaptive softmax smoothing,
structured restarts, and multi-length search.
"""

import numpy as np
import time


def _fft_self_conv(a, N):
    """Compute self-convolution via FFT. Returns c of length 2*len(a)-1."""
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _objective_from_conv(c, n):
    """Compute 2*n*max(c) for a normalized sequence (sum=1)."""
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * mx


def _softmax_weights(c, beta):
    """Compute softmax weights exp(beta*(c-max(c))) normalized."""
    L = len(c)
    if L == 0:
        return np.ones(1)
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx <= 0:
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        return w
    x = -beta * (c - mx) / (mx + 1e-300)
    x = np.clip(x, -740.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _convolution_gradient(a, c_weights, N):
    """
    Compute gradient of sum_k w_k * c[k] w.r.t. a.
    c = convolve(a, a), so dc[k]/da[j] = 2*a[k-j] (if valid).
    In FFT domain: grad = 2 * ifft(FFT(w_padded) * conj(FFT(a))) shifted.
    """
    n = len(a)
    # Pad c_weights to full convolution length, then to FFT length
    cw_full = np.zeros(N)
    cw_full[:len(c_weights)] = c_weights
    W = np.fft.rfft(cw_full)
    A = np.fft.rfft(a, N)
    # grad[j] = sum_k w_k * 2*a[k-j] = 2 * (w * a)[j] in circular sense
    # In FFT: 2 * ifft(W * conj(A)) gives circular cross-correlation
    grad_circ = np.fft.irfft(W * np.conjugate(A), N)
    grad = 2.0 * grad_circ[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def _log_softmax(w):
    """Convert log-weights to probability vector."""
    w = np.asarray(w, dtype=np.float64)
    w = np.where(np.isfinite(w), w, -1e30)
    w = w - float(np.max(w))
    e = np.exp(w)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(len(w)) / len(w)
    return e / s


def _normalize(v):
    """Normalize non-negative vector to sum 1."""
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(len(v)) / len(v)
    return v / s


def _adam_step(grad, w, m, v, t, lr):
    """Adam step in log-space with projection to simplex."""
    t += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    step = m_hat / (np.sqrt(v_hat) + 1e-8)
    step = np.clip(step, -20.0, 20.0)
    w_new = w - lr * step
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    a_new = _log_softmax(w_new)
    return a_new, w_new, m, v, t


def _exact_score(a):
    """Exact score using np.convolve. a is assumed non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    c = np.convolve(a, a)
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s)


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _load_initial():
    """Try to load height_sequence_1."""
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 8:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0:
        return None
    return arr / s


def _structured_perturb(a, rng):
    """Apply a structured perturbation: reversal, shift, or block swap."""
    n = len(a)
    op = rng.integers(0, 5)
    if op == 0:
        # Circular shift
        k = int(rng.integers(1, max(2, n // 4)))
        return np.roll(a, k)
    elif op == 1:
        # Reverse
        return a[::-1].copy()
    elif op == 2:
        # Block swap
        block = max(2, n // int(rng.integers(4, 16)))
        i = int(rng.integers(0, n - 2 * block))
        j = int(rng.integers(0, n - 2 * block))
        tmp = a[i:i+block].copy()
        a[i:i+block] = a[j:j+block]
        a[j:j+block] = tmp
        return a
    elif op == 3:
        # Taper (smooth ends)
        k = max(1, n // 10)
        out = a.copy()
        t = np.linspace(0, 1, k)
        out[:k] *= t
        out[n-k:] *= t[::-1]
        return out
    else:
        # Gentle noise in log-space
        w = np.log(np.maximum(a, 1e-30))
        w += rng.normal(0, 0.1, size=n)
        return _log_softmax(w)


def _run_optimizer(a_init, budget, rng, verbose=True):
    """Main optimization loop. Returns best sequence (normalized to sum=1)."""
    start_time = time.monotonic()
    deadline = start_time + budget
    
    a = _normalize(a_init)
    n = len(a)
    N = _next_pow2(2 * n + 1)
    
    c = _fft_self_conv(a, N)
    score = _objective_from_conv(c, n)
    
    best_a = a.copy()
    best_score = score
    
    # Log-space parameterization
    w = np.log(np.maximum(a, 1e-30))
    w -= float(np.max(w))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Hyperparameters
    beta = 10.0
    lr = 0.1
    stall = 0
    iter_count = 0
    last_improve = start_time
    
    if verbose:
        print(f"[init] n={n}, score={score:.6f}")
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Compute convolution and identify max positions
        c = _fft_self_conv(a, N)
        c_score = _objective_from_conv(c, n)
        if not np.isfinite(c_score):
            # Recovery
            a = best_a.copy()
            c = _fft_self_conv(a, N)
            c_score = _objective_from_conv(c, n)
            w = np.log(np.maximum(a, 1e-30))
            w -= float(np.max(w))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        # Update best
        if c_score < best_score:
            best_score = c_score
            best_a = a.copy()
            last_improve = time.monotonic()
            stall = 0
            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.0f}s] iter={iter_count} score={best_score:.6f}")
        else:
            stall += 1
        
        # Compute subgradient / smoothed gradient
        # Use exact subgradient with some smoothing
        c_max = float(np.max(c))
        if c_max > 0:
            # Softmax weights (sharper = more like exact subgradient)
            cw = _softmax_weights(c, beta)
        else:
            cw = np.ones(2 * n - 1) / (2 * n - 1)
        
        grad = _convolution_gradient(a, cw, N)
        
        # Project gradient onto simplex tangent: grad_log = a * (grad - dot(a, grad))
        grad_tan = a * (grad - float(np.dot(a, grad)))
        grad_tan = np.where(np.isfinite(grad_tan), grad_tan, 0.0)
        
        # Adam step
        a, w, m, v, t = _adam_step(grad_tan, w, m, v, t, lr)
        
        # Adaptive learning rate
        if stall == 0:
            lr = min(0.3, lr * 1.02)
        elif stall > 5:
            lr = max(0.01, lr * 0.95)
        
        # Increase beta (sharper max approximation)
        if iter_count % 20 == 0:
            beta = min(1000.0, beta * 1.15 + 1.0)
        
        # Restart logic
        if stall > 50:
            if rng.random() < 0.5:
                # Return to best and perturb
                a = _normalize(best_a + rng.normal(0, 0.05, size=n))
            else:
                # Structured perturbation of best
                a = _structured_perturb(best_a.copy(), rng)
                a = _normalize(a)
            
            w = np.log(np.maximum(a, 1e-30))
            w -= float(np.max(w))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.1
            beta = max(5.0, beta * 0.5)
            stall = 0
        
        # Moderate restart for diversity
        elif stall > 25 and rng.random() < 0.05:
            a = _structured_perturb(best_a.copy(), rng)
            a = _normalize(a)
            w = np.log(np.maximum(a, 1e-30))
            w -= float(np.max(w))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
    
    if verbose:
        elapsed = time.monotonic() - start_time
        print(f"[done] n={n}, best_score={best_score:.6f}, iters={iter_count}, time={elapsed:.0f}s")
    
    return best_a


def _polish(a_init, budget, rng, verbose=True):
    """Fine-grained polish using very small steps and exact subgradient."""
    start_time = time.monotonic()
    deadline = start_time + budget
    
    a = _normalize(a_init)
    n = len(a)
    N = _next_pow2(2 * n + 1)
    
    c = _fft_self_conv(a, N)
    score = _objective_from_conv(c, n)
    best_a = a.copy()
    best_score = score
    
    w = np.log(np.maximum(a, 1e-30))
    w -= float(np.max(w))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    beta = 500.0  # Start very sharp (near-exact subgradient)
    lr = 0.02
    stall = 0
    iter_count = 0
    
    if verbose:
        print(f"[polish] n={n}, initial score={score:.6f}")
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        c = _fft_self_conv(a, N)
        c_score = _objective_from_conv(c, n)
        
        if not np.isfinite(c_score):
            a = best_a.copy()
            continue
        
        if c_score < best_score:
            best_score = c_score
            best_a = a.copy()
            stall = 0
            if verbose and iter_count % 100 == 0:
                elapsed = time.monotonic() - start_time
                print(f"[polish {elapsed:.0f}s] score={best_score:.6f}")
        else:
            stall += 1
        
        # Exact subgradient (or near-exact with high beta)
        c_max = float(np.max(c))
        if c_max > 0:
            cw = _softmax_weights(c, beta)
        else:
            cw = np.ones(2 * n - 1) / (2 * n - 1)
        
        grad = _convolution_gradient(a, cw, N)
        grad_tan = a * (grad - float(np.dot(a, grad)))
        grad_tan = np.where(np.isfinite(grad_tan), grad_tan, 0.0)
        
        a, w, m, v, t = _adam_step(grad_tan, w, m, v, t, lr)
        
        # Decay learning rate
        if stall == 0:
            lr = min(0.05, lr * 1.01)
        else:
            lr = max(0.003, lr * 0.99)
        
        # Periodic sharpening
        if iter_count % 50 == 0:
            beta = min(5000.0, beta * 1.1)
        
        # Small restart if stuck
        if stall > 80:
            a = _normalize(best_a + rng.normal(0, 0.02, size=n))
            w = np.log(np.maximum(a, 1e-30))
            w -= float(np.max(w))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.03
            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - start_time
        print(f"[polish done] score={best_score:.6f}, iters={iter_count}, time={elapsed:.0f}s")
    
    return best_a


def _try_length(a_best, n_target, budget, rng):
    """Try to optimize at a specific sequence length by resampling."""
    a_best = np.asarray(a_best, dtype=np.float64)
    n = len(a_best)
    
    if n_target == n:
        a = a_best.copy()
    elif n_target > n:
        # Upsample via interpolation + noise
        indices = np.linspace(0, n - 1, n_target)
        a_up = np.interp(indices, np.arange(n), a_best)
        a_up += rng.uniform(0, 0.1) * a_up
    else:
        # Downsample via block averaging
        blocks = n // n_target
        if blocks > 0:
            a = a_best[:blocks * n_target].reshape(n_target, blocks).mean(axis=1)
        else:
            a = a_best[:n_target]
    
    a = _normalize(a)
    return _run_optimizer(a, budget, rng, verbose=False)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point: searches for a sequence minimizing 2*n*max(convolve(a,a))/sum(a)^2.
    
    Strategy:
    1. Phase 1 (60% time): Coarse search from provided sequence with restarts.
    2. Phase 2 (25% time): Try alternative lengths and structured starts.
    3. Phase 3 (15% time): Fine polish on the best candidate found.
    4. Final: Exact evaluation and return.
    """
    try:
        budget = min(float(budget_s), 2400.0)
    except Exception:
        budget = 2400.0
    
    total_start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Load initial sequence
    a_init = _load_initial()
    if a_init is None:
        n = 4096
        a_init = np.ones(n) / n
        # Add structure: taper + noise
        t = np.linspace(0, 1, n)
        a_init = 0.5 + 0.5 * np.sin(np.pi * t)
        a_init += rng.normal(0, 0.1, size=n)
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()
        if verbose := True:
            print(f"[init] Generated random init, n={n}")
    
    n_init = len(a_init)
    if n_init > 16384:
        # Downsample for feasibility
        target = 8192
        blocks = n_init // target
        a_init = a_init[:target * blocks].reshape(target, blocks).mean(axis=1)
        a_init = _normalize(a_init)
        n_init = len(a_init)
    
    # Phase 1: Coarse search (60% of time)
    phase1_budget = 0.55 * budget - 5
    t0 = time.monotonic()
    if verbose := True:
        print(f"=== Phase 1: Coarse search (budget={phase1_budget:.0f}s) ===")
    best_a = _run_optimizer(a_init, phase1_budget, rng, verbose=True)
    phase1_time = time.monotonic() - t0
    
    # Get FFT score of best
    n = len(best_a)
    N = _next_pow2(2 * n + 1)
    c = _fft_self_conv(best_a, N)
    best_fft_score = _objective_from_conv(c, n)
    if verbose := True:
        print(f"[phase1] best FFT score: {best_fft_score:.6f}, time used: {phase1_time:.0f}s")
    
    # Phase 2: Try alternative approaches (25% of time)
    phase2_budget = 0.25 * budget - 5
    t0 = time.monotonic()
    if verbose := True:
        print(f"=== Phase 2: Alternative searches (budget={phase2_budget:.0f}s) ===")
    
    remaining_for_phase2 = max(1.0, phase2_budget)
    
    # Try different starting points
    candidates = [best_a]
    
    # Try reversal
    a_rev = best_a[::-1].copy()
    a_rev = _normalize(a_rev)
    sub_budget = remaining_for_phase2 / 4
    a_rev_opt = _run_optimizer(a_rev, sub_budget, np.random.default_rng(seed + 1), verbose=False)
    n2 = len(a_rev_opt)
    N2 = _next_pow2(2 * n2 + 1)
    c2 = _fft_self_conv(a_rev_opt, N2)
    s2 = _objective_from_conv(c2, n2)
    if verbose:
        print(f"[phase2] reversal: score={s2:.6f}")
    candidates.append(a_rev_opt)
    
    # Try a shifted version
    shift = max(1, n // 8)
    a_shift = np.roll(best_a, shift)
    a_shift = _normalize(a_shift)
    sub_budget = remaining_for_phase2 / 4
    a_shift_opt = _run_optimizer(a_shift, sub_budget, np.random.default_rng(seed + 2), verbose=False)
    n3 = len(a_shift_opt)
    N3 = _next_pow2(2 * n3 + 1)
    c3 = _fft_self_conv(a_shift_opt, N3)
    s3 = _objective_from_conv(c3, n3)
    if verbose:
        print(f"[phase2] shift: score={s3:.6f}")
    candidates.append(a_shift_opt)
    
    # Try a tapered version
    a_taper = best_a.copy()
    k = max(1, n // 6)
    tt = np.linspace(0.1, 1.0, k)
    a_taper[:k] *= tt
    a_taper[n-k:] *= tt[::-1]
    a_taper = _normalize(a_taper)
    sub_budget = remaining_for_phase2 / 4
    a_taper_opt = _run_optimizer(a_taper, sub_budget, np.random.default_rng(seed + 3), verbose=False)
    n4 = len(a_taper_opt)
    N4 = _next_pow2(2 * n4 + 1)
    c4 = _fft_self_conv(a_taper_opt, N4)
    s4 = _objective_from_conv(c4, n4)
    if verbose:
        print(f"[phase2] taper: score={s4:.6f}")
    candidates.append(a_taper_opt)
    
    # Try a different length (smaller, for potential better ratio)
    if n > 2048:
        n_small = 2048
        indices = np.linspace(0, n - 1, n_small)
        a_small = np.interp(indices, np.arange(n), best_a)
        a_small = _normalize(a_small)
        sub_budget = remaining_for_phase2 / 4
        a_small_opt = _run_optimizer(a_small, sub_budget, np.random.default_rng(seed + 4), verbose=False)
        n5 = len(a_small_opt)
        N5 = _next_pow2(2 * n5 + 1)
        c5 = _fft_self_conv(a_small_opt, N5)
        s5 = _objective_from_conv(c5, n5)
        if verbose:
            print(f"[phase2] small n={n5}: score={s5:.6f}")
        candidates.append(a_small_opt)
    
    # Select best from all candidates
    best_score = float('inf')
    best_final_a = best_a
    for cand in candidates:
        nc = len(cand)
        Nc = _next_pow2(2 * nc + 1)
        cc = _fft_self_conv(_normalize(cand), Nc)
        sc = _objective_from_conv(cc, nc)
        if sc < best_score:
            best_score = sc
            best_final_a = _normalize(cand)
    
    phase2_time = time.monotonic() - t0
    if verbose:
        print(f"[phase2] best FFT score: {best_score:.6f}, time used: {phase2_time:.0f}s")
    
    # Phase 3: Fine polish (remaining time)
    elapsed = time.monotonic() - total_start
    phase3_budget = max(1.0, budget - elapsed - 10)
    t0 = time.monotonic()
    if verbose:
        print(f"=== Phase 3: Polish (budget={phase3_budget:.0f}s) ===")
    
    best_final_a = _polish(best_final_a, phase3_budget, rng, verbose=True)
    
    phase3_time = time.monotonic() - t0
    if verbose:
        print(f"[phase3] time used: {phase3_time:.0f}s")
    
    # Final exact evaluation
    final_a = _normalize(best_final_a)
    n_f = len(final_a)
    N_f = _next_pow2(2 * n_f + 1)
    c_f = _fft_self_conv(final_a, N_f)
    fft_final = _objective_from_conv(c_f, n_f)
    if verbose:
        print(f"[final] FFT score: {fft_final:.6f}, n={n_f}")
    
    # Exact score (for logging)
    if n_f <= 16384:
        exact = _exact_score(final_a)
        if verbose:
            print(f"[final] exact score: {exact:.6f}")
    
    # Scale to sum = sqrt(2*n) for the final output
    scale = float(np.sqrt(2.0 * n_f))
    s = float(final_a.sum())
    result = final_a * scale / s
    result = np.clip(result, 0.0, 1000.0)
    
    # Verify sum is not too small
    s2 = float(result.sum())
    if s2 < 0.01:
        result = result * (0.01 / max(s2, 1e-30))
        result = np.clip(result, 0.0, 1000.0)
    
    return [float(x) for x in result]
