"""
Minimize 2*n*max(autocorrelation(a)) / sum(a)^2 for non-negative a.

def _next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def _softmax_stable(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - float(v.max())
    v = np.clip(v, -745.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / v.shape[0], dtype=np.float64)
    return e / s


def _autocorr_fft(a, m):
    n = len(a)
    F = np.fft.rfft(a, m)
    b = np.fft.irfft(F * F.conj(), m)[:2 * n - 1]
    return np.maximum(b, 0.0)


def _score_from_parts(n, b, S):
    if S < 1e-300:
        return np.inf
    bm = float(b.max())
    if not np.isfinite(bm) or bm <= 0.0:
        return np.inf
    return 2.0 * n * bm / (S * S)


def _compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    b = _autocorr_fft(a, m)
    return _score_from_parts(n, b, S)


def _grad_softmax_loss(a, b, tau, m_grad):
    n = len(a)
    bmax = float(b.max())
    if bmax < 1e-300:
        return np.zeros(n, dtype=np.float64)
    x = (b - bmax) / tau
    x = np.clip(x, -100.0, 100.0)
    e = np.exp(x)
    s = float(e.sum())
    if s < 1e-300:
        w = np.ones(len(b), dtype=np.float64) / len(b)
    else:
        w = e / s
    Wp = w.copy()
    Wp[n - 1] = 0.0
    Wf = np.fft.rfft(Wp, m_grad)
    Af = np.fft.rfft(a, m_grad)
    C = np.fft.irfft(Wf * Af, m_grad)
    grad = 2.0 * w[n - 1] * a + C[n - 1:n - 1 + n]
    return grad

    kmax = int(np.argmax(b))
    k = kmax - (n - 1)

    absk = abs(k)
    g = np.zeros(n, dtype=np.float64)
    j_start = max(0, absk)
    j_end = min(n, n - absk)
    g[j_start:j_end] += a[j_end - absk:j_start - absk + (j_end - j_start)] if absk < n else 0
    # More careful:
    g = np.zeros(n, dtype=np.float64)
    # a[j - k] for valid j

        if 0 <= idx1 < n:
            g[j] += a[idx1]
    lo2 = 0
    hi2 = min(n, n - absk)
    for j in range(lo2, hi2):
        idx2 = j + absk
        if 0 <= idx2 < n:
            g[j] += a[idx2]
    return g

def _spectral_peak_reduce(a, deadline, max_iters=200):

    m = _next_pow2(2 * n - 1)

def _adam_optimize(a_init, deadline, seed=0):
    n = len(a_init)
    m = _next_pow2(2 * n - 1)
    m_grad = _next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)

    a = np.maximum(a_init, 1e-15)
    S = float(a.sum())
    a = a / S
    v = np.log(a + 1e-300)
    v = v - v.mean()
    v = np.clip(v, -30.0, 30.0)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0

    best_a = a.copy()
    best_score = _compute_score(a)
    start_time = time.monotonic()
    total_budget = max(1.0, deadline - start_time)
    last_improve = start_time

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / total_budget)

        # Stuck detection
        if time.monotonic() - last_improve > min(60.0, 0.25 * total_budget):
            sigma = 0.05 + 0.15 * (1.0 - progress)
            v += rng.normal(0, sigma, n)
            v = v - v.mean()
            v = np.clip(v, -30.0, 30.0)
            last_improve = time.monotonic()
            continue

        a = _softmax_stable(v)
        b = _autocorr_fft(a, m)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax < 1e-300:
            break

        # Temperature schedule
        tau_frac = 0.12 * (1.0 - progress) + 0.015
        if rng.random() < 0.3:
            tau_frac *= rng.uniform(0.5, 2.0)
        tau = max(1e-15, bmax * tau_frac)

        # Compute gradient (alternate between soft-max and exact)
        if rng.random() < 0.2 and progress > 0.3:
            grad_a = _grad_exact_max_fast(a, b, m_grad)
        else:
            grad_a = _grad_softmax_loss(a, b, tau, m_grad)

        # Gradient in v-space
        grad_mean = float(np.dot(a, grad_a))
        grad_v = a * (grad_a - grad_mean)

        # Check gradient magnitude
        grms = float(np.sqrt(np.mean(grad_v ** 2)))
        if not np.isfinite(grms) or grms < 1e-18:
            sigma = 0.03 + 0.10 * (1.0 - progress)
            v += rng.normal(0, sigma, n)
            v = v - v.mean()
            v = np.clip(v, -30.0, 30.0)
            continue

        # GradNorm
        if grms > 1e-3:
            grad_v = grad_v / grms

        t += 1
        mu = beta1 * mu + (1 - beta1) * grad_v
        nu = beta2 * nu + (1 - beta2) * (grad_v * grad_v)
        b1 = beta1 ** t
        b2 = beta2 ** t
        m_hat = mu / max(1e-12, 1 - b1)
        v_hat = nu / max(1e-12, 1 - b2)

        lr = 0.08 * (1.0 - 0.5 * progress)
        step = m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -0.3, 0.3)
        v = v - lr * step
        v = v - v.mean()
        v = np.clip(v, -30.0, 30.0)

        # Score every K steps
        check_int = max(3, min(30, 4000 // max(1, n // 64)))
        if t % check_int == 0:
            a_check = _softmax_stable(v)
            b_check = _autocorr_fft(a_check, m)
            sc = _score_from_parts(n, b_check, float(a_check.sum()))
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best_score = sc
                best_a = a_check.copy()
                last_improve = time.monotonic()

    # Final evaluation
    a_final = _softmax_stable(v)
    sc_final = _compute_score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final

    return best_a, best_score


def _local_refine(a, deadline):
    n = len(a)
    m = _next_pow2(2 * n - 1)
    a = np.array(a, dtype=np.float64)
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_score = _compute_score(a)

    # Find top coordinates by contribution to peak autocorrelation
    b = _autocorr_fft(a, m)
    kmax = int(np.argmax(b))
    k = kmax - (n - 1)
    absk = abs(k)

    # Try perturbations on a subset of coordinates
    # Focus on coordinates near the peak lags
    if absk < n:
        important = np.zeros(n, dtype=np.float64)
        important[absk:] += a[:n - absk] ** 2
        important[:n - absk] += a[absk:] ** 2
    else:
        important = a ** 2

    top_idx = np.argsort(important)[::-1][:min(200, n)]

    fractions = [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]

    for j in top_idx:
        if time.monotonic() >= deadline:
            break
        aj = a[j]
        for frac in fractions:
            delta = frac * aj
            new_aj = aj + delta
            if new_aj < 0:
                continue
            a[j] = new_aj
            S_new = S + delta
            if S_new < 1e-300:
                a[j] = aj
                continue
            b_new = _autocorr_fft(a, m)
            sc = _score_from_parts(n, b_new, S_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                S = S_new
            else:
                a[j] = aj
    return best_a, best_score


def _make_inits(n, prev_a, rng):

        x_new = np.linspace(0, 1, n)

        interp = np.maximum(interp, 0)
        s = interp.sum()
        if s > 0:
            inits.append(interp / s)

    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)

    # Semicircle
    a = np.sqrt(np.maximum(1 - t ** 2, 1e-10))
    s = a.sum()
    if s > 0:
        inits.append(a / s)

    # Exponential
    for c in [1.0, 2.0, 0.5]:
        a = np.exp(-c * t)
        s = a.sum()
        if s > 0:
            inits.append(a / s)

    # Bell
    a = np.exp(-5 * (t - 0.5) ** 2)
    s = a.sum()
    if s > 0:
        inits.append(a / s)

    # 1/sqrt edges
    left = i + 1.0
    right = n - i
    a = 1.0 / np.sqrt(left * right)
    s = a.sum()
    if s > 0:
        inits.append(a / s)

    # Random smooth
    fc = min(10, n // 4)
    a = np.ones(n, dtype=np.float64)
    for k in range(fc):
        amp = 0.3 / (k + 1) ** 0.7
        ph = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * (k + 1) * t + ph)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        inits.append(a / s)

    return inits
