```
def gradient_top_weighted(a, c, K, lam_factor=8.0):
    n=a.size; L=c.size
    if L ==0: return zeros
    if K >= L:
       idx=np.arange(L)
    else:
       idx=np.argpartition(c, L-K)[-(K):]
    vals=c[idx]
    vmax=float(np.max(vals))
    vmin=float(np.min(vals))
    rngv = vmax - vmin
    if rngv < 1e-12:
       rngv = max(vmax*1e-12, 1e-12)
    wf = np.exp((lam_factor/rngv)*(vals - vmax))
    sw=wf.sum()
    if sw <=0: wf.fill(1.0/len(wf)) else wf/=sw
    grad=np.zeros(n)
    nm1=n-1
    for ii in range(idx.size):
       k=int(idx[ii]); w=float(wf[ii])
       if w <=0: continue
       start = k - nm1
       if start < 0: start=0
       stop = k + 1
       if stop > n: stop=n
       # slice of a: indices from k-(stop-1) to k-start inclusive
       s_start = k - (stop - 1)
       s_end = k - start + 1
       if s_end > s_start:
          grad[start:stop] += w * a[s_start:s_end][::-1]
    grad *= 2.0
    return grad
```

```python
"""
Searches for non-negative step-function coefficients minimizing

    2*n*max(convolve(a, a)) / (sum(a))^2.

The implementation uses:
- fast FFT-based convolution and a direct fallback for short sequences;
- coarse-to-fine multi-resolution search over length;
- diverse deterministic and random starts, including beta shapes and low-frequency
  smooth perturbations of the supplied previous construction;
- a linear-programming guided mixing step: for current f, maximize 1^T g subject to
  (f*g)_k <= max(f*f) and g >= 0, then mix f with normalized g;
- a robust multiplicative projected sub-gradient local search using exponentially
  weighted top convolution peaks;
- final exact validation with evaluate_sequence().
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

_INF = float("inf")


def normalize_sequence(a, target_sum, cap=1000.0):
    """Clip to [0, cap] and normalize to target_sum as much as possible."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)

    a = a.copy()
    a[~np.isfinite(a)] = 0.0
    a[a < 0.0] = 0.0
    a[a > cap] = cap

    t = float(target_sum)
    if not np.isfinite(t) or t <= 0.0:
        if float(a.sum()) < 0.01:
            a.fill(1.0)
        return a

    if t > cap * a.size:
        t = float(cap * a.size)

    for _ in range(12):
        s = float(a.sum())
        if s <= 1e-15:
            a.fill(t / float(a.size))
        else:
            a *= t / s
        a[a < 0.0] = 0.0
        a[a > cap] = cap
        s = float(a.sum())
        if abs(s - t) <= 1e-8 * max(1.0, t):
            break

    s = float(a.sum())
    deficit = t - s
    if deficit > 1e-7 * max(1.0, t):
        for _ in range(5):
            mask = a < cap
            cnt = int(mask.sum())
            if cnt == 0:
                a.fill(min(1.0, cap))
                break
            deficit = t - float(a.sum())
            if deficit <= 0.0:
                break
            a[mask] += deficit / float(cnt)
            a[a > cap] = cap

    if float(a.sum()) < 0.01:
        a.fill(1.0)
    return a


def fast_convolution(a):
    """Linear convolution a*a, using direct code for short arrays."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    if n <= 512:
        c = np.convolve(a, a, mode="full")
        c = np.asarray(c, dtype=np.float64)
        np.maximum(c, 0.0, out=c)
        return c

    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def score_from_c(c, sum_a, n):
    """Score from an already computed convolution."""
    if c.size == 0 or sum_a <= 0.01:
        return _INF
    return float(2.0 * n * float(np.max(c)) / (sum_a * sum_a))


def fast_score(a):
    """Fast proxy for evaluate_sequence(), using scale-invariant score."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return _INF

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return _INF

    c = fast_convolution(a)
    if c.size == 0:
        return _INF
    return score_from_c(c, s, n)


def gradient_top_weighted(a, c, K, lam_factor=8.0):
    """
    Weighted subgradient for the score with respect to a.

    For coefficient k, d c_k / d a_i = 2*a_{k-i} when valid.
    We choose weights on the largest K coefficients, emphasizing the largest.
    """
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    L = c.size
    grad = np.zeros(n, dtype=np.float64)

    if n == 0 or L == 0:
        return grad

    K = int(min(max(1, K), L))
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, L - K)[-(K):]

    vals = c[idx]
    vmax = float(np.max(vals))
    vmin = float(np.min(vals))
    rngv = vmax - vmin
    rngv = max(rngv, abs(vmax) * 1e-6, 1e-12)

    exponents = (float(lam_factor) / rngv) * (vals - vmax)
    exponents = np.clip(exponents, -50.0, 0.0)
    w = np.exp(exponents)
    sw = float(w.sum())
    if sw <= 0.0 or not np.isfinite(sw):
        w.fill(1.0 / float(w.size))
        sw = 1.0
    else:
        w /= sw

    nm1 = n - 1
    for ii in range(idx.size):
        k = int(idx[ii])
        ww = float(w[ii])
        if ww <= 0.0:
            continue

        start = k - nm1
        if start < 0:
            start = 0
        stop = k + 1
        if stop > n:
            stop = n

        s_start = k - (stop - 1)
        s_end = k - start + 1
        if s_end > s_start:
            grad[start:stop] += ww * a[s_start:s_end][::-1]

    grad *= 2.0
    return grad


def upsample_sequence(seq, n):
    """Linearly interpolate a sequence to length n and renormalize."""
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    if seq.size == 0:
        return np.ones(n, dtype=np.float64)

    if seq.size == 1:
        v = float(seq[0])
        if not np.isfinite(v) or v <= 0.0:
            v = 1.0
        return normalize_sequence(np.full(n, v, dtype=np.float64), float(n))

    if seq.size == n:
        return normalize_sequence(seq, float(n))

    xold = np.linspace(0.0, 1.0, seq.size)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_sequence(out, float(n))


def make_two_block(n, frac, ratio):
    """Two-block start."""
    n = int(n)
    m = int(round(frac * n))
    if m < 1:
        m = 1
    if m >= n:
        m = n - 1

    denom = float(m) * float(ratio) + float(n - m)
    if denom <= 0.0:
        B = 1.0
    else:
        B = float(n) / denom
    A = float(ratio) * B

    a = np.empty(n, dtype=np.float64)
    a[:m] = A
    a[m:] = B
    return normalize_sequence(a, float(n))


def make_beta(n, p, q):
    """Beta-like shape: t^p (1-t)^q."""
    n = int(n)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.ones(n, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
        if p != 0.0:
            a *= t ** p
        if q != 0.0:
            a *= (1.0 - t) ** q
    a = np.where(np.isfinite(a) & (a > 0.0), a, 1e-300)
    return normalize_sequence(a, float(n))


def make_harmonic(n, rng, max_freq=None, amp=2.4):
    """Random low-frequency cosine shape, exponentiated."""
    n = int(n)
    if n < 2:
        return np.ones(max(1, n), dtype=np.float64)
    if max_freq is None:
        max_freq = min(20, max(5, n // 16))

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, max_freq + 1):
        am = amp / ((m + 1.0) ** 0.75)
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        z += float(rng.uniform(-am, am)) * np.cos(2.0 * math.pi * m * t + ph)

    z -= float(np.max(z))
    out = np.exp(z)
    return normalize_sequence(out, float(n))


def make_block_sequence(n, rng, num_blocks=None):
    """Random block constant sequence."""
    n = int(n)
    if n < 6:
        return np.ones(max(1, n), dtype=np.float64)

    if num_blocks is None:
        num_blocks = int(rng.integers(4, min(28, n // 2) + 1))
    num_blocks = max(2, min(num_blocks, n // 2))

    edges = np.unique(np.linspace(0, n, num_blocks + 1).astype(int))
    if edges[0] != 0:
        edges[0] = 0
    if edges[-1] != n:
        edges = np.append(edges, n)

    lengths = np.diff(edges)
    vals = rng.uniform(0.05, 3.0, lengths.size)
    if rng.random() < 0.25 and vals.size > 0:
        vals[int(rng.integers(0, vals.size))] = 0.0

    out = np.repeat(vals, lengths)
    if out.size < n:
        out = np.concatenate([out, np.ones(n - out.size, dtype=np.float64)])
    elif out.size > n:
        out = out[:n]

    return normalize_sequence(out, float(n))


def add_start(starts, seq, n):
    """Evaluate one start and append if finite."""
    try:
        a = normalize_sequence(np.asarray(seq, dtype=np.float64).reshape(-1), float(n))
        sc = fast_score(a)
    except Exception:
        return _INF

    if np.isfinite(sc):
        starts.append((float(sc), len(starts), a))
    return sc


def generate_starts(n, rng, previous_best, random_count=20):
    """Generate a diverse pool of starts at length n."""
    n = int(n)
    starts = []

    add_start(starts, np.ones(n, dtype=np.float64), n)

    if previous_best is not None:
        add_start(starts, upsample_sequence(previous_best, n), n)

    parent1 = globals().get("height_sequence_1", None)
    if parent1 is not None:
        try:
            p1 = np.asarray(parent1, dtype=np.float64).reshape(-1)
            if p1.size > 0:
                add_start(starts, upsample_sequence(p1, n), n)
        except Exception:
            pass

    # Deterministic two-block starts.
    for frac in (0.20, 0.30, 0.40, 0.50):
        for ratio in (0.5, 1.6, 3.0):
            try:
                add_start(starts, make_two_block(n, frac, ratio), n)
            except Exception:
                pass

    # Deterministic beta shapes.
    beta_pairs = [
        (-0.5, 0.8),
        (-0.8, 0.4),
        (-0.3, 1.8),
        (0.0, -0.8),
        (0.0, 0.8),
        (0.8, -0.8),
        (1.2, 0.4),
        (-0.2, -0.4),
        (0.4, 1.2),
    ]
    for p, q in beta_pairs:
        try:
            add_start(starts, make_beta(n, float(p), float(q)), n)
        except Exception:
            pass

    # Random smooth / block starts.
    random_count = max(0, int(random_count))
    for _ in range(random_count):
        try:
            if rng.random() < 0.60:
                seq = make_harmonic(n, rng)
            else:
                seq = make_block_sequence(n, rng)
            add_start(starts, seq, n)
        except Exception:
            pass

    if not starts:
        starts.append((2.0, 0, np.ones(n, dtype=np.float64)))

    starts.sort()
    return starts


def perturb_smooth(a, rng, amp=0.35):
    """Smooth multiplicative perturbation; caller normalizes."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n <= 2 or amp <= 0.0:
        return a

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, 7):
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        z += (amp / (m + 1.0)) * np.cos(2.0 * math.pi * m * t + ph)
    a *= np.exp(z)

    if n > 16 and rng.random() < 0.55:
        width = int(n * float(rng.uniform(0.002, 0.06)))
        width = max(1, min(width, n))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width
        a[start:stop] *= np.exp(rng.normal(0.0, amp, stop - start))

    return a


def local_search(a, budget_s, seed=0):
    """
    Multiplicative projected sub-gradient search.

    At each step, a weighted average of the top convolution peaks is converted
    into a directional subgradient. The direction is centered to respect the
    normalization, momentum and adaptivity are used, and occasional smooth
    perturbations escape stagnation.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        return a, _INF

    target = float(n)
    a = normalize_sequence(a, target)
    sc0 = fast_score(a)
    if not np.isfinite(sc0):
        a = np.ones(n, dtype=np.float64)
        sc0 = fast_score(a)

    budget_s = float(max(0.0, budget_s))
    if budget_s < 0.3:
        return a, sc0

    t0 = time.monotonic()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    c = fast_convolution(a)
    s = float(a.sum())
    score = score_from_c(c, s, n)
    best = a.copy()
    best_score = score

    mom = np.zeros(n, dtype=np.float64)
    eta = 0.20
    beta = 0.35
    noise = 0.02
    accept_rel = 0.015
    stagnant = 0
    it = 0

    while time.monotonic() < deadline - 1e-3:
        it += 1

        if (not np.all(np.isfinite(a))) or s <= 0.0:
            a = best.copy()
            c = fast_convolution(a)
            s = float(a.sum())
            score = best_score
            mom = np.zeros(n, dtype=np.float64)

        L = 2 * n - 1
        if L <= 0:
            break

        frac = 0.18 * math.exp(-float(it) / 800.0)
        frac = max(0.006, min(0.22, frac))
        K = int(L * frac)
        K = max(4, min(600, K, L))
        lam = 6.0 + 8.0 * min(1.0, float(it) / 600.0)

        try:
            grad = gradient_top_weighted(a, c, K, lam)
        except Exception:
            break

        if not np.all(np.isfinite(grad)):
            break

        s = float(a.sum())
        if s <= 0.0:
            a = best.copy()
            c = fast_convolution(a)
            s = float(a.sum())
            score = best_score
            continue

        dot = float(np.dot(a, grad)) / s
        d = grad - dot
        maxd = float(np.max(np.abs(d)))
        if maxd <= 1e-14:
            stagnant += 1
            if stagnant > 40:
                a = normalize_sequence(perturb_smooth(best, rng, 0.40), target)
                c = fast_convolution(a)
                s = float(a.sum())
                score = score_from_c(c, s, n)
                mom = np.zeros(n, dtype=np.float64)
                eta = 0.30
                accept_rel = 0.04
                noise = 0.05
                stagnant = 0
                if not np.isfinite(score) or score > best_score * 1.10:
                    a = best.copy()
                    c = fast_convolution(a)
                    s = float(a.sum())
                    score = best_score
            continue

        d /= maxd
        mom = beta * mom + (1.0 - beta) * d
        if noise > 1e-5:
            mom = mom + noise * rng.normal(0.0, 1.0, n)

        logf = np.clip(-eta * mom, -3.5, 3.5)
        cand = a * np.exp(logf)
        cand = normalize_sequence(cand, target)

        c2 = fast_convolution(cand)
        s2 = float(cand.sum())
        sc2 = score_from_c(c2, s2, n)

        if not np.isfinite(sc2):
            eta = max(0.02, eta * 0.70)
            mom *= 0.5
            stagnant += 1
            if stagnant > 30:
                break
            continue

        if sc2 < best_score:
            best = cand.copy()
            best_score = sc2
            stagnant = 0

        if sc2 <= score * (1.0 + accept_rel) + 1e-12:
            old = score
            a = cand
            c = c2
            s = s2
            score = sc2
            mom *= 0.85

            if score < old * (1.0 - 1e-10):
                eta = min(1.2, eta * 1.07)
                accept_rel = max(0.001, accept_rel * 0.9985)
                noise = max(0.002, noise * 0.9985)
            else:
                eta = max(0.03, eta * 0.97)

            stagnant += 1
        else:
            eta = max(0.02, eta * 0.70)
            mom *= 0.5
            stagnant += 1

        if stagnant > 450:
            a = normalize_sequence(perturb_smooth(best, rng, 0.45), target)
            c = fast_convolution(a)
            s = float(a.sum())
            score = score_from_c(c, s, n)
            mom = np.zeros(n, dtype=np.float64)
            eta = 0.30
            accept_rel = 0.04
            noise = 0.05
            stagnant = 0
            if not np.isfinite(score) or score > best_score * 1.15:
                a = best.copy()
                c = fast_convolution(a)
                s = float(a.sum())
                score = best_score

    return best, best_score


def solve_lp(f, M, time_limit=5.0):
    """
    Solve:
        maximize 1^T g
        subject to (f*g)_k <= M for all k,
                   g >= 0.
    """
    if not _HAVE_SCIPY:
        return None

    f = np.asarray(f, dtype=np.float64).reshape(-1)
    n = int(f.size)
    if n < 8 or n > 1100:
        return None

    L = 2 * n - 1
    M = float(M)
    if not np.isfinite(M) or M <= 0.0:
        return None

    try:
        i = np.repeat(np.arange(n, dtype=np.int32), n)
        j = np.tile(np.arange(n, dtype=np.int32), n)
        rows = i + j
        cols = i
        data = np.tile(f, n)

        A = coo_matrix((data, (rows, cols)), shape=(L, n)).tocsr()
        b = np.full(L, M * (1.0 + 1e-9), dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        opts = {}
        if time_limit is not None:
            opts["time_limit"] = float(time_limit)
        if "presolve" in opts:
            pass
        else:
            opts["presolve"] = True

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="hihgs", options=opts)
        if res.status == 0 and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64)
            x = np.where(np.isfinite(x), x, 0.0)
            if x.sum() > 1e-12:
                return np.clip(x, 0.0, None)
    except Exception:
        return None

    return None


def lp_refine(a, budget_s, seed=0):
    """
    Iterate the LP-guided mixing step on a sequence.

    This can produce large global changes while staying close to the feasible
    region suggested by the current sequence f.
    """
    a = normalize_sequence(np.asarray(a, dtype=np.float64).reshape(-1), float(a.size))
    n = int(a.size)
    if n == 0:
        return a, _INF
    if n < 8 or n > 1024 or not _HAVE_SCIPY:
        return a, fast_score(a)

    budget_s = float(max(0.0, budget_s))
    if budget_s < 5.0:
        return a, fast_score(a)

    t0 = time.monotonic()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    c = fast_convolution(a)
    s = float(a.sum())
    score = score_from_c(c, s, n)
    best = a.copy()
    best_score = score

    max_iters = 24 if n <= 512 else 10
    stuck = 0

    for _ in range(max_iters):
        if time.monotonic() >= deadline - 2.0:
            break

        if (not np.all(np.isfinite(a))) or s <= 0.0:
            a = best.copy()
            c = fast_convolution(a)
            s = float(a.sum())
            score = best_score

        M = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(M) or M <= 0.0:
            break

        rem = max(0.0, deadline - time.monotonic())
        if rem < 3.0:
            break
        tlim = min(8.0 if n <= 512 else 14.0, max(2.0, rem * 0.25))

        g = solve_lp(a, M, time_limit=tlim)
        if g is None:
            break

        g = np.asarray(g, dtype=np.float64).reshape(-1)
        if g.size != n:
            break
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.clip(g, 0.0, None)
        if float(g.sum()) <= 1e-12:
            break
        g = normalize_sequence(g, float(n))

        best_mix = None
        best_mix_score = score_from_c(c, s, n) * 1.2
        improved_global = False

        bases = (a, best)
        alphas = (0.20, 0.40, 0.60, 0.80)

        for base in bases:
            for al in alphas:
                try:
                    cand = (1.0 - al) * base + al * g
                    cand = normalize_sequence(cand, float(n))
                    cc = fast_convolution(cand)
                    sc = score_from_c(cc, float(cand.sum()), n)
                except Exception:
                    continue
                if not np.isfinite(sc):
                    continue

                if sc < best_mix_score:
                    best_mix = cand.copy()
                    best_mix_score = sc

                if sc < best_score:
                    best = cand.copy()
                    best_score = sc
                    improved_global = True

        if improved_global:
            a = best.copy()
            c = fast_convolution(a)
            s = float(a.sum())
            score = best_score
            stuck = 0
        elif best_mix is not None:
            current_score = score_from_c(c, s, n)
            if best_mix_score < current_score * 1.02 + 1e-12 and best_mix_score < best_score * 1.05:
                a = best_mix
                c = fast_convolution(a)
                s = float(a.sum())
                score = best_mix_score
                stuck += 1
            else:
                stuck += 1
        else:
            stuck += 1

        if stuck >= 3:
            a = normalize_sequence(perturb_smooth(best, rng, 0.35), float(n))
            c = fast_convolution(a)
            s = float(a.sum())
            score = score_from_c(c, s, n)
            stuck = 0
            if not np.isfinite(score) or score > best_score * 1.20:
                a = best.copy()
                c = fast_convolution(a)
                s = float(a.sum())
                score = best_score

        if stuck > 14:
            break

    return best, best_score


def run_level(n, budget_s, previous_best, seed=0, allow_lp=False):
    """Refine the best sequence available at length n."""
    n = int(n)
    budget_s = float(max(0.0, budget_s))

    if budget_s < 8.0 or n < 2:
        if previous_best is not None:
            seq = normalize_sequence(upsample_sequence(previous_best, n), float(n))
        else:
            seq = np.ones(n, dtype=np.float64)
        return seq, fast_score(seq)

    t0 = time.monotonic()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    random_count = 12
    if budget_s > 240 and n <= 2048:
        random_count = 22
    elif budget_s > 500 and n <= 4096:
        random_count = 18

    starts = generate_starts(n, rng, previous_best, random_count=random_count)
    if not starts:
        seq = np.ones(n, dtype=np.float64)
        return seq, fast_score(seq)

    best_seq = np.array(starts[0][2], dtype=np.float64, copy=True)
    best_score = float(starts[0][0])

    # LP-guided moves are most useful at small/medium lengths.
    if allow_lp and n <= 1024 and budget_s > 70.0:
        lp_budget = min(budget_s * 0.35, 120.0 if n <= 512 else 80.0)
        try:
            seq, sc = lp_refine(best_seq, lp_budget, seed + 111)
            if seq is not None and np.isfinite(sc) and sc < best_score:
                best_seq = seq
                best_score = sc
        except Exception:
            pass

    remain = max(0.0, deadline - time.monotonic())
    if remain > 20.0 and best_seq is not None:
        # Quick refinement of a few promising starts.
        quick_budget = min(80.0, 0.22 * remain)
        if quick_budget > 15.0:
            candidates = [best_seq]
            for tup in starts[:3]:
                if len(candidates) < 4:
                    candidates.append(tup[2])
            k = max(1, min(len(candidates), int(quick_budget / 15.0) + 1, 4))
            step = quick_budget / float(k)
            for ii in range(min(k, len(candidates))):
                if time.monotonic() >= deadline - 5.0:
                    break
                time_this = max(1.0, min(step, deadline - time.monotonic() - 5.0))
                if time_this <= 0.0:
                    break
                try:
                    seq, sc = local_search(candidates[ii], time_this, seed + 200 + ii)
                    if seq is not None and np.isfinite(sc) and sc < best_score:
                        best_seq = seq
                        best_score = sc
                except Exception:
                    continue

    # Long refinement of the current best.
    remain = max(0.0, deadline - time.monotonic())
    if remain > 15.0 and best_seq is not None:
        time_this = max(5.0, remain - 1.0)
        try:
            seq, sc = local_search(best_seq, time_this, seed + 999)
            if seq is not None and np.isfinite(sc) and sc < best_score:
                best_seq = seq
                best_score = sc
        except Exception:
            pass

    return best_seq, best_score


def add_candidate(pool, seq, score):
    """Add a candidate to the pool and return its fast score."""
    if seq is None:
        return _INF
    try:
        seq = np.asarray(seq, dtype=np.float64).reshape(-1).copy()
        if seq.size == 0:
            return _INF
        seq = normalize_sequence(seq, float(seq.size))
        if np.isfinite(float(score)):
            sc = float(score)
        else:
            sc = fast_score(seq)
    except Exception:
        return _INF

    if not np.isfinite(sc) or sc < 0.0:
        return _INF

    pool.append((float(sc), len(pool), seq))
    pool.sort()
    if len(pool) > 24:
        pool.pop()
    return float(sc)


def sanitize_return(seq):
    """Convert a stored array to a valid Python list without changing scale."""
    try:
        seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    except Exception:
        return [1.0] * 512

    if seq.size == 0:
        return [1.0] * 512

    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 512

    return [float(x) for x in seq]


def exact_value(lst):
    """Call the external evaluator safely."""
    try:
        return float(evaluate_sequence(lst))
    except Exception:
        return _INF


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    for key in ("budget", "time_limit", "time_budget", "remaining_time", "runtime", "seconds"):
        if key in kwargs:
            try:
                val = float(kwargs[key])
                if np.isfinite(val) and val > 0.0:
                    budget = val
            except Exception:
                pass

    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = min(max(budget, 1.0), 2400.0)

    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except Exception:
        seed = 42

    t_start = time.monotonic()
    hard_deadline = t_start + budget
    validation_reserve = 90.0 if budget > 300.0 else max(20.0, 0.15 * budget)
    search_deadline = t_start + max(15.0, budget - validation_reserve)

    pool = []
    current_best_seq = None
    current_best_score = _INF

    def record_candidate(seq, sc):
        nonlocal current_best_seq, current_best_score
        if seq is None:
            return
        out_sc = add_candidate(pool, seq, sc)
        if pool and pool[0][0] < current_best_score:
            current_best_score = float(pool[0][0])
            current_best_seq = np.asarray(pool[0][2], dtype=np.float64).copy()

    # Parse the known previous construction, if present.
    parent = globals().get("height_sequence_1", None)
    parent_arr = None
    if parent is not None:
        try:
            arr = np.asarray(parent, dtype=np.float64).reshape(-1)
            if arr.size > 0:
                parent_arr = normalize_sequence(arr, float(arr.size))
                sc = fast_score(parent_arr)
                if np.isfinite(sc):
                    record_candidate(parent_arr, sc)
        except Exception:
            parent_arr = None

    if current_best_seq is None:
        fallback_seq = np.ones(512, dtype=np.float64)
        record_candidate(fallback_seq, fast_score(fallback_seq))

    # Main coarse-to-fine search.
    try:
        total_search = max(0.0, search_deadline - time.monotonic())
        if total_search > 1800.0:
            levels = [256, 512, 1024, 2048, 4096, 8192]
            weights = [0.020, 0.060, 0.120, 0.220, 0.320, 0.260]
        elif total_search > 1100.0:
            levels = [256, 512, 1024, 2048, 4096]
            weights = [0.035, 0.100, 0.180, 0.300, 0.385]
        elif total_search > 600.0:
            levels = [512, 1024, 2048, 4096]
            weights = [0.15, 0.25, 0.35, 0.25]
        else:
            levels = [512, 1024]
            weights = [0.35, 0.65]

        sum_w = float(sum(weights)) if weights else 1.0
        prev_best = current_best_seq

        for idx, n in enumerate(levels):
            if time.monotonic() >= search_deadline - 10.0:
                break

            rem = max(0.0, search_deadline - time.monotonic())
            if rem < 25.0:
                break

            bud = total_search * (float(weights[idx]) / sum_w)
            bud = max(15.0, min(bud, rem - 1.0))

            allow_lp = _HAVE_SCIPY and (n <= 1024) and (bud > 60.0)

            try:
                seq, sc = run_level(n, bud, prev_best, seed + 1000 + 17 * idx, allow_lp)
            except Exception:
                seq, sc = None, _INF

            if seq is not None and np.isfinite(sc):
                print(f"level n={n} fast_score={sc:.10f} time={time.monotonic() - t_start:.1f}")
                record_candidate(seq, sc)
                # Use a sequence at the current/similar length for the next upsample.
                if prev_best is None or seq.size >= prev_best.size or sc < current_best_score:
                    prev_best = seq

        # Final polishing on the globally best candidate, optionally upsampling one step.
        polish_rng = np.random.default_rng(seed + 5555)
        polish_iters = 0
        while polish_iters < 3 and time.monotonic() < search_deadline - 60.0:
            best_len = int(current_best_seq.size)
            rem = max(0.0, search_deadline - time.monotonic())
            if rem < 90.0:
                break

            if best_len < 4096 and rem > 260.0:
                target_n = min(8192, best_len * 2)
                start_seq = upsample_sequence(current_best_seq, target_n)
                time_this = min(rem - 10.0, 330.0 if target_n < 4096 else 420.0)
            else:
                target_n = best_len
                start_seq = perturb_smooth(current_best_seq, polish_rng, 0.25)
                start_seq = normalize_sequence(start_seq, float(best_len))
                time_this = min(rem - 10.0, 150.0 if target_n >= 4096 else 110.0)

            if time_this < 20.0:
                break

            try:
                seq, sc = local_search(start_seq, time_this, seed + 9000 + target_n + polish_iters)
                if seq is not None and np.isfinite(sc):
                    record_candidate(seq, sc)
            except Exception:
                pass

            polish_iters += 1

    except Exception:
        pass

    # Exact validation of the top fast-scored candidates.
    final_seq = current_best_seq
    final_score = current_best_score if current_best_seq is not None else _INF

    try:
        if final_seq is not None:
            lst = sanitize_return(final_seq)
            ex = exact_value(lst)
            if np.isfinite(ex):
                final_score = ex
            print(f"preexact global fast={final_score:.10f}")

        pool.sort()
        validated = 0
        for tup in pool:
            sc_fast, _, seq_c = tup
            if time.monotonic() >= hard_deadline - 3.0:
                break
            if not np.isfinite(sc_fast):
                continue
            if np.isfinite(final_score) and sc_fast > final_score * 1.20 + 1e-4:
                break
            if validated > 10:
                break

            lst = sanitize_return(seq_c)
            ex = exact_value(lst)
            validated += 1
            if np.isfinite(ex):
                if ex < final_score:
                    final_score = ex
                    final_seq = seq_c
                print(f"exact candidate fast={sc_fast:.10f} exact={ex:.10f} n={int(seq_c.size)}")
    except Exception:
        pass

    if final_seq is None:
        final_seq = np.ones(512, dtype=np.float64)

    try:
        out = sanitize_return(final_seq)
    except Exception:
        out = [1.0] * 512

    try:
        print(f"final exact_score={final_score:.10f} len={len(out)}")
    except Exception:
        pass

    return out
```
