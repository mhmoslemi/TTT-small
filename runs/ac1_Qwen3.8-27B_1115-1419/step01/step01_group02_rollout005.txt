- Near-peak weighting (dynamic threshold)

- Structured perturbations for diversity
- Multiple initialization strategies
- Coarse-to-fine optimization

import numpy as np

from scipy.fft import fft, ifft


def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _norm_to(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def _step(a, lr, mom, beta, pfrac):
    """One subgradient step. pfrac: use peaks above pfrac*max."""
    n = len(a)
    b = _fft_conv(a, a)
    S = a.sum()
    mx = float(b.max())
    
    thresh = pfrac * mx
    idx = np.where(b >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(b))])
    
    w = b[idx] - thresh + 1e-12
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        if hi >= lo:
            js = np.arange(lo, hi + 1)
            grad[lo:hi+1] += wi * 2.0 * a[ki - js]
    
    mom = beta * mom + (1.0 - beta) * grad
    a2 = a - lr * mom
    a2 = np.maximum(a2, 0.0)
    a2 = _norm_to(a2, S)
    return a2, mom, mx

    st = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        a, mom, _ = _step(a, lr, mom, bet, pf)
        lr = max(lr * dec, 1e-9)
        st += 1
        
        if st % ev_int == 0:
            sc = _score_fast(a)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = a.copy()
                stale = 0
            else:
                stale += 1
            
            if stale >= perturb_every:
                # Structured perturbation
                k = np.arange(n)
                freq = np.random.uniform(1, 20)
                amp = 0.01 * S / n
                perturb = amp * np.cos(2 * np.pi * freq * k / n + np.random.uniform(0, 2*np.pi))
                a = best + perturb
                a = np.maximum(a, 0)
                a = _norm_to(a, S)
                mom = np.zeros(n)
                lr = lr0 * 0.4
                stale = 0
