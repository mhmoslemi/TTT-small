def _score_weights(q, n, temp):
    if q.size == 0:
        return 0.0, 0.0, 0.0, None
    maxq = float(np.max(q))
    exact = 2.0 * float(n) * maxq
    tau = max(float(temp), 1e-4) / float(n)
    shifted = (q - maxq) / tau
    exps = np.exp(shifted)
    se = float(np.sum(exps))
    if se <= 1e-300 or not math.isfinite(se):
        w = np.full(q.size, 1.0/q.size, dtype=np.float64)
        fs = exact
    else:
        w = exps / se
        fs = 2.0 * float(n) * (maxq + tau * math.log(se))
    return maxq, exact, fs, w

def _conv_fftp(p, N, n):
    f = np.fft.rfft(p, N)
    qfull = np.fft.irfft(f * f, N)
    L = 2*n - 1
    q = qfull[:L].copy()
    if q.size:
        if float(np.min(q)) < 0.0:
            q = np.maximum(q, 0.0)
    return q, f

def _grad_from_info(p, f, w, N, n):
    if w is None or w.size != 2*n-1:
       return np.zeros(n)

    fwr = np.fft.rfft(wr, N)

    g *= float(n)

"""
Search for a non-negative step-function height sequence minimizing

    2*n*max(conv(a,a)) / sum(a)^2

using:
- fast FFT convolution and gradient information for the exact/smooth max;
- log-domain exponentiated-gradient descent with backtracking line search on
  a softmax-surrogate of the maximum convolution coefficient;
- multi-phase optimization: broad smoothing, sharpening, then perturbation
  restarts / crossover / symbolic-symmetry moves;
- two length tracks (shorter parent-like track and longer track), keeping the
  best sequence found.
"""

import numpy as np
import time
import math


def _next_pow2_ge(x):
    """Smallest power of two >= integer x (x >= 1)."""
    p = 1
    while p < x:
        p <<= 1
    return p


def _fft_len(n):
    """FFT length for self-convolution and weighted gradient correlation."""
    if n <= 0:
        return 1
    # We need no circular aliasing for cross-correlation of length-2n-1 weights
    # with length-n p, so use length >= 3n-2.
    return _next_pow2_ge(max(1, 3 * n - 2))


def _normalize_p(a, n):
    """Normalize a non-negative array to a probability vector of length n."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)

    if a.size != n:
        if a.size == 1:
            a = np.full(n, a[0], dtype=np.float64)
        else:
            old_x = np.linspace(0.0, 1.0, a.size)
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, a)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)
    return a / s


def _interp_p(p, n):
    """Interpolate a probability vector to length n and renormalize."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    if p.size == 0:
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)
    if p.size == n:
        return _normalize_p(p, n)
    if p.size == 1:
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)

    old_x = np.linspace(0.0, 1.0, p.size)
    new_x = np.linspace(0.0, 1.0, n)
    q = np.interp(new_x, old_x, p)
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / max(1, n), dtype=np.float64)
    return q / s


def _parent_p():
    """Try to load the provided previous-good sequence as a probability vector."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if arr.size > 0 and s > 1e-300 and math.isfinite(s):
            return arr / s
    except Exception:
        pass
    return None


def _log_pos(p):
    """Logarithm of a positive vector, clipped to a robust range."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p2 = np.maximum(p, 1e-300)
    x = np.log(p2)
    x = np.where(np.isfinite(x), x, 0.0)

    mx = float(np.max(x))
    mn = float(np.min(x))
    if not math.isfinite(mx) or not math.isfinite(mn):
        return np.zeros_like(x)

    # Keep logs in a range where softmax is stable while retaining dynamic range.
    if mx - mn > 80.0:
        x = np.clip(x, mx - 80.0, mx)

    x -= 0.5 * (float(np.max(x)) + float(np.min(x)))
    return x


def _softmax_x(x):
    """Stable softmax of log-domain variable x."""
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    x = np.where(np.isfinite(x), x, 0.0)
    z = x - float(np.max(x))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not math.isfinite(s):
        return np.full(x.size, 1.0 / max(1, x.size), dtype=np.float64)
    return e / s


def _score_weights(q, n, temp):
    """
    Given convolution coefficients q (p has sum 1), return
    maxq, exact_score, smooth_surrogate, and softmax weights.

    exact_score = 2*n*max(q)
    smooth_surrogate = 2*n*(maxq + tau*logsumexp((q-maxq)/tau)), tau=temp/n.
    """
    if q.size == 0:
        return 0.0, 0.0, 0.0, None

    maxq = float(np.max(q))
    exact = 2.0 * float(n) * maxq
    tau = max(float(temp), 1e-4) / float(n)

    shifted = (q - maxq) / tau
    exps = np.exp(shifted)
    se = float(np.sum(exps))

    if se <= 1e-300 or not math.isfinite(se):
        w = np.full(q.size, 1.0 / q.size, dtype=np.float64)
        fs = exact
    else:
        w = exps / se
        fs = 2.0 * float(n) * (maxq + tau * math.log(se))

    return maxq, exact, fs, w


def _conv_fftp(p, N, n):
    """FFT self-convolution of p and rfft(p) for gradient reuse."""
    f = np.fft.rfft(p, N)
    qfull = np.fft.irfft(f * f, N)
    L = 2 * n - 1
    q = qfull[:L].copy()
    if q.size:
        if float(np.min(q)) < 0.0:
            q = np.maximum(q, 0.0)
    return q, f


def _grad_from_info(p, f, w, N, n):
    """
    Gradient direction in p for the smooth objective.

    For q_k = sum_i p_i p_{k-i}, dz_k/dp_i = 2 p_{k-i}.
    The constant factor is irrelevant because we normalize the direction.
    """
    if w is None or w.size != 2 * n - 1:
        return np.zeros(n, dtype=np.float64)

    wr = w[::-1]
    fwr = np.fft.rfft(wr, N)
    h = np.fft.irfft(fwr * f, N)

    # H_i = sum_k w_k p_{k-i}, valid for i=0..n-1.
    seg = h[n - 1:2 * n - 1]
    g = seg[::-1].copy()
    # Scale to roughly O(1) values.
    g *= float(n)
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def _spectral_noise(n, rng, cutoff_frac):
    """Zero-mean, unit-variance low-pass random noise via inverse FFT."""
    n = int(max(1, n))
    m = n // 2 + 1
    c = rng.normal(size=m) + 1j * rng.normal(size=m)
    k = np.arange(m, dtype=np.float64)
    cutoff = max(1.0, float(n) * float(cutoff_frac))
    amp = np.exp(-(k / cutoff) ** 2)
    c = c * amp
    noise = np.fft.irfft(c, n)
    noise = np.asarray(noise, dtype=np.float64)
    noise -= float(np.mean(noise))
    std = float(np.std(noise))
    if std <= 1e-12 or not math.isfinite(std):
        return np.zeros(n, dtype=np.float64)
    return noise / std


def _random_smooth_p(n, rng, cutoff=0.08, sigma=0.3):
    """Random smooth positive probability vector."""
    n = int(max(1, n))
    noise = _spectral_noise(n, rng, cutoff)
    z = float(sigma) * noise
    z = z - float(np.max(z))
    p = np.exp(z)
    s = float(p.sum())
    if s <= 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def _perturb_x(x, n, rng, sigma, mode):
    """Create a perturbed log-domain variable from x."""
    x = np.array(x, dtype=np.float64, copy=True)
    n = int(max(1, n))
    x = np.where(np.isfinite(x), x, 0.0)
    m = int(mode) % 4

    if m == 0:
        # Global low-frequency log perturbation.
        cutoff = float(rng.uniform(0.03, 0.25))
        x += float(sigma) * _spectral_noise(n, rng, cutoff)
    elif m == 1:
        # Local block log perturbation.
        w = int(max(2, n * float(rng.uniform(0.01, 0.30))))
        if w >= n:
            w = max(1, n // 2)
        s = int(rng.integers(0, max(1, n - w)))
        x[s:s + w] += float(sigma) * rng.normal(0.0, 1.0, size=w)
    elif m == 2:
        # Reflective symmetrization plus small noise.
        x = 0.5 * (x + x[::-1])
        if n > 1:
            cutoff = float(rng.uniform(0.05, 0.15))
            x += 0.25 * float(sigma) * _spectral_noise(n, rng, cutoff)
    else:
        # Higher-frequency perturbation.
        cutoff = float(rng.uniform(0.20, 0.50))
        x += 0.5 * float(sigma) * _spectral_noise(n, rng, cutoff)

    x -= float(np.mean(x))
    mx = float(np.max(x))
    mn = float(np.min(x))
    if math.isfinite(mx - mn) and (mx - mn) > 120.0:
        x = np.clip(x, mx - 120.0, mx)
        x -= float(np.mean(x))
    x = np.where(np.isfinite(x), x, 0.0)
    return x


def _crossover_x(xa, xb, rng, sigma):
    """Geometric-mean crossover in log space with a small low-pass jitter."""
    n = int(max(1, len(xa)))
    x = 0.5 * xa + 0.5 * xb
    if n > 1:
        cutoff = float(rng.uniform(0.05, 0.15))
        x += 0.25 * float(sigma) * _spectral_noise(n, rng, cutoff)

    x = np.where(np.isfinite(x), x, 0.0)
    x -= float(np.mean(x))
    mx = float(np.max(x))
    mn = float(np.min(x))
    if math.isfinite(mx - mn) and (mx - mn) > 120.0:
        x = np.clip(x, mx - 120.0, mx)
        x -= float(np.mean(x))
    return x


def _local_optimize(p0, n, deadline, temp_start, temp_end, alpha_start, tag, alpha_max=5.0):
    """
    Local log-domain exponentiated-gradient optimizer.

    Returns (best_p, best_exact_score, best_x).
    """
    n = int(max(1, n))
    p = _normalize_p(p0, n)
    x = _log_pos(p)
    N = _fft_len(n)
    t0 = time.monotonic()

    if not math.isfinite(deadline):
        deadline = t0 + 1.0
    total = max(0.0, deadline - t0)

    q, f = _conv_fftp(p, N, n)
    _, best_score, _, _ = _score_weights(q, n, max(float(temp_start), float(temp_end)))
    best_p = p.copy()
    best_x = x.copy()

    if total <= 0.2:
        return best_p, best_score, best_x

    alpha = max(1e-4, float(alpha_start))
    min_alpha = 1e-4
    max_alpha = max(alpha, float(alpha_max))
    fail = 0
    it = 0
    last_print = t0
    factors = (1.0, 0.5, 0.25, 0.125)

    while time.monotonic() < deadline - 0.01:
        now = time.monotonic()
        elapsed = max(0.0, now - t0)
        frac = min(1.0, elapsed / max(total, 1e-6))
        temp = float(temp_end) + (float(temp_start) - float(temp_end)) * (1.0 - frac)
        if temp < 1e-4:
            temp = 1e-4

        _, exact, fs, w = _score_weights(q, n, temp)

        old_best = best_score
        if exact < old_best - 1e-10:
            best_score = exact
            best_p = p.copy()
            best_x = x.copy()
            if now - last_print > 20.0 or (old_best - best_score) > 1e-5:
                print(f"[{tag}] exact={best_score:.8f} temp={temp:.4f} alpha={alpha:.4f} it={it}")
                last_print = now

        # If the current point has drifted far above the tracked best, revert.
        if exact > best_score + 0.30 and it > 5:
            p = best_p.copy()
            x = best_x.copy()
            q, f = _conv_fftp(p, N, n)
            fail = 0
            last_print = now
            continue

        g = _grad_from_info(p, f, w, N, n)
        mean_g = float(np.dot(p, g))
        d = g - mean_g
        scale = float(np.max(np.abs(d))) if d.size else 0.0
        if not math.isfinite(scale) or scale < 1e-12:
            break

        v = d / scale
        accepted = False

        for fac in factors:
            a = alpha * float(fac)
            if a < min_alpha:
                a = min_alpha
            x_tr = x - a * v
            p_tr = _softmax_x(x_tr)

            q_tr, f_tr = _conv_fftp(p_tr, N, n)
            _, exact_tr, fs_tr, _ = _score_weights(q_tr, n, temp)

            # Accept if the smooth surrogate or the exact score improves.
            if (fs_tr < fs - 1e-10) or (exact_tr < exact - 1e-9):
                x = x_tr
                p = p_tr
                q = q_tr
                f = f_tr
                if fac == 1.0:
                    alpha = min(alpha * 1.04, max_alpha)
                else:
                    alpha = max(min_alpha, a * 1.08)
                fail = 0
                accepted = True
                break

        if not accepted:
            alpha = max(min_alpha, alpha * 0.65)
            fail += 1
            if fail > 8:
                break

        it += 1
        if it % 100 == 99:
            x = x - float(np.max(x))

    return best_p, best_score, best_x


def _perturb_loop(x_best, n, track_deadline, tag, rng, best_score):
    """
    Repeatedly perturb the current log solution and run short local refinements.
    Maintains a tiny population of good log-states for crossover.
    """
    n = int(max(1, n))
    p_best = _softmax_x(x_best)
    N = _fft_len(n)
    q, f = _conv_fftp(p_best, N, n)
    _, exact_b, _, _ = _score_weights(q, n, 0.05)

    if best_score is None or not math.isfinite(best_score):
        best_score = exact_b

    pop = [np.asarray(x_best, dtype=np.float64).copy()]
    attempts = 0
    stagn = 0

    while time.monotonic() < track_deadline - 8.0:
        remaining = max(0.0, track_deadline - time.monotonic())
        if remaining < 20.0:
            break

        attempts += 1
        if remaining > 200.0:
            dur = float(min(20.0, max(8.0, 0.15 * remaining)))
        else:
            dur = float(min(20.0, max(8.0, 0.40 * remaining)))
        sub_deadline = min(time.monotonic() + dur, track_deadline - 2.0)

        base = 0.15 + min(0.60, 0.008 * float(attempts))
        sigma = float(rng.uniform(0.6 * base, 1.6 * base))
        mode = int(rng.integers(0, 5))

        if mode < 4:
            x_try = _perturb_x(x_best, n, rng, sigma, mode)
        else:
            if len(pop) >= 2:
                other = pop[int(rng.integers(0, len(pop) - 1))]
            else:
                other = x_best
            x_try = _crossover_x(np.asarray(x_best, dtype=np.float64),
                                 np.asarray(other, dtype=np.float64),
                                 rng, sigma)

        p_try = _softmax_x(x_try)
        t0 = time.monotonic()
        p2, s2, x2 = _local_optimize(
            p_try, n, sub_deadline,
            temp_start=2.0, temp_end=0.05,
            alpha_start=1.2,
            tag=(tag + f"_p{attempts}"),
            alpha_max=4.0
        )
        dt = max(0.0, time.monotonic() - t0)

        if s2 < best_score - 1e-9:
            best_score = s2
            p_best = np.asarray(p2, dtype=np.float64).copy()
            x_best = np.asarray(x2, dtype=np.float64).copy()
            pop.insert(0, x_best.copy())
            if len(pop) > 8:
                pop.pop()
            stagn = 0
            print(f"[{tag}] perturbation {attempts} -> {best_score:.8f} ({dt:.1f}s, mode={mode})")
        else:
            stagn += 1

        # Occasional full random restart if perturbations stall.
        if stagn > 3:
            if remaining > 45.0:
                p_rand = _random_smooth_p(n, rng, 0.08, 0.5)
                rand_deadline = min(time.monotonic() + 30.0, track_deadline - 2.0)
                pr, sr, xr = _local_optimize(
                    p_rand, n, rand_deadline,
                    temp_start=2.5, temp_end=0.05,
                    alpha_start=1.5,
                    tag=(tag + "_rand"),
                    alpha_max=4.0
                )
                if sr < best_score - 1e-9:
                    best_score = sr
                    p_best = np.asarray(pr, dtype=np.float64).copy()
                    x_best = np.asarray(xr, dtype=np.float64).copy()
                    pop.insert(0, x_best.copy())
                    if len(pop) > 8:
                        pop.pop()
                    print(f"[{tag}] random restart -> {best_score:.8f}")
            stagn = 0

    # Final sharpening pass from the best point.
    if time.monotonic() < track_deadline - 5.0:
        p2, s2, x2 = _local_optimize(
            p_best, n, track_deadline - 1.0,
            temp_start=0.5, temp_end=0.02,
            alpha_start=0.6,
            tag=(tag + "_final"),
            alpha_max=2.5
        )
        if s2 < best_score - 1e-9:
            best_score = s2
            p_best = np.asarray(p2, dtype=np.float64).copy()
            x_best = np.asarray(x2, dtype=np.float64).copy()
            pop.insert(0, x_best.copy())
            if len(pop) > 8:
                pop.pop()

    return p_best, best_score, x_best


def _run_track(p0, n, track_deadline, tag, rng):
    """
    Run one length track:
    1) broad smooth optimization;
    2) sharper optimization;
    3) perturbation/refinement loop.
    """
    n = int(max(1, n))
    t0 = time.monotonic()
    rem = max(0.0, track_deadline - t0)

    if rem < 20.0:
        return _local_optimize(p0, n, track_deadline, 1.5, 0.05, 1.0, tag)
    if rem < 80.0:
        return _local_optimize(p0, n, track_deadline, 2.0, 0.05, 1.2, tag)

    t1 = t0 + 0.25 * rem
    t2 = t0 + 0.60 * rem

    p1, s1, x1 = _local_optimize(
        p0, n, t1,
        temp_start=3.0, temp_end=1.0,
        alpha_start=1.5,
        tag=(tag + "_M"),
        alpha_max=5.0
    )

    p2, s2, x2 = _local_optimize(
        p1, n, t2,
        temp_start=1.0, temp_end=0.05,
        alpha_start=0.8,
        tag=(tag + "_H"),
        alpha_max=4.0
    )

    if s2 < s1:
        p, s, x = p2, s2, x2
    else:
        p, s, x = p1, s1, x1

    p, s, x = _perturb_loop(x, n, track_deadline, tag, rng, s)
    return p, s, x


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Returns the best non-negative sequence found before the time budget
    (with a small safety margin for final validation/return).
    """
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if budget <= 0.0:
        return [1.0] * 1000

    # Leave time for final evaluation/serialization.
    deadline = t0 + budget - 30.0
    if deadline <= t0 + 30.0:
        deadline = t0 + max(30.0, budget - 5.0)

    rng = np.random.default_rng(seed)

    best_p = None
    best_n = 0
    best_score = float("inf")

    # Initialize from the provided previous construction, if available.
    par = _parent_p()
    if par is not None and par.size > 0:
        ppar = _normalize_p(par, par.size)
        Npar = _fft_len(ppar.size)
        q, _ = _conv_fftp(ppar, Npar, ppar.size)
        sp = 2.0 * float(ppar.size) * float(np.max(q))
        if math.isfinite(sp):
            best_p = ppar.copy()
            best_n = int(ppar.size)
            best_score = sp
            print(f"[init] parent length={best_n} exact={best_score:.8f}")
    else:
        ninit = 1024
        pinit = np.full(ninit, 1.0 / ninit, dtype=np.float64)
        Ninit = _fft_len(ninit)
        qinit, _ = _conv_fftp(pinit, Ninit, ninit)
        best_p = pinit.copy()
        best_n = ninit
        best_score = 2.0 * float(ninit) * float(np.max(qinit))
        print(f"[init] no parent, uniform length={best_n} exact={best_score:.8f}")

    remaining = max(30.0, deadline - t0)

    if budget > 1200.0:
        n1 = 4095
        n2 = 8191
        track1_time = max(60.0, min(600.0, 0.25 * remaining))
        track1_deadline = t0 + track1_time
        # Ensure enough time remains for the long track.
        if deadline - track1_deadline < 300.0:
            track1_deadline = max(t0 + 30.0, deadline - 300.0)
    else:
        n1 = 4095
        n2 = 4095
        track1_time = max(30.0, 0.45 * remaining)
        track1_deadline = t0 + track1_time
        if deadline - track1_deadline < 60.0:
            track1_deadline = max(t0 + 10.0, deadline - 60.0)

    # Track 1: shorter, parent-like length.
    p1 = _interp_p(best_p, n1) if best_p is not None else _random_smooth_p(n1, rng, 0.08, 0.3)
    pbest1, s1, x1 = _run_track(p1, n1, track1_deadline, "T1", rng)
    if s1 < best_score - 1e-12:
        best_p = np.asarray(pbest1, dtype=np.float64).copy()
        best_n = int(n1)
        best_score = s1
        print(f"[track1] best={best_score:.8f} n={best_n}")

    # Track 2: longer length for more attack surface.
    if time.monotonic() < deadline - 120.0:
        p2 = _interp_p(best_p, n2) if best_p is not None else _random_smooth_p(n2, rng, 0.08, 0.4)
        pbest2, s2, x2 = _run_track(p2, n2, deadline - 5.0, "T2", rng)
        if s2 < best_score - 1e-12:
            best_p = np.asarray(pbest2, dtype=np.float64).copy()
            best_n = int(n2)
            best_score = s2
            print(f"[track2] best={best_score:.8f} n={best_n}")

    # A couple of independent random restarts if time remains.
    if time.monotonic() < deadline - 30.0 and best_n > 0:
        for rr in range(2):
            rr_deadline = min(time.monotonic() + 40.0, deadline - 15.0)
            if time.monotonic() >= rr_deadline - 10.0:
                break
            p_rand = _random_smooth_p(best_n, rng, 0.10, 0.60)
            pr, sr, xr = _local_optimize(
                p_rand, best_n, rr_deadline,
                temp_start=2.5, temp_end=0.05,
                alpha_start=1.5,
                tag=f"RR{rr}",
                alpha_max=4.0
            )
            if sr < best_score - 1e-12:
                best_p = np.asarray(pr, dtype=np.float64).copy()
                best_score = sr
                print(f"[restart{rr}] best={best_score:.8f} n={best_n}")

    # Build final sequence. Scaling by sqrt(n) is objective-neutral and keeps values healthy.
    if best_p is None or best_n <= 0:
        result = [1.0] * 1000
    else:
        p = np.asarray(best_p, dtype=np.float64).reshape(-1)
        if p.size != best_n:
            p = _normalize_p(p, best_n)
        else:
            p = _normalize_p(p, best_n)

        scale = math.sqrt(float(best_n))
        result_arr = p * scale
        result_arr = np.where(np.isfinite(result_arr), result_arr, 0.0)
        result_arr = np.clip(result_arr, 0.0, 1000.0)

        if float(np.sum(result_arr)) < 0.01:
            result_arr = np.full(1000, 0.1, dtype=np.float64)

        result = [float(v) for v in result_arr]

    # Validate once against the provided evaluator.
    try:
        ev = evaluate_sequence(result)
        ev = float(ev)
        print(f"FINAL internal={best_score:.8f} eval={ev:.8f} len={len(result)}")
    except Exception as exc:
        ev = float("inf")
        print("Final validation failed:", exc)

    if not math.isfinite(ev):
        result = [1.0] * 1000

    return result
