Minimize 2n*max(self-convolution)/(sum)^2 using Adam optimizer with 
log-sum-exp smoothing, spectral peak compression, and adaptive restarts.

import numpy as np
import time
from scipy.fft import fft, ifft

def _next_pow2(n):
    return 1 << (int(n) - 1).bit_length()

def _convolve_self(a, N):
    """Full linear self-convolution of a using FFT of size N."""
    A = fft(a, N)
    return ifft(A * A).real

def _score_fast(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    c = _convolve_self(a, N)[:2*n-1]
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def _softmax_weights(c, T):
    """Compute softmax(T*c) in a numerically stable way."""
    c_max = np.max(c)
    e = np.exp(T * (c - c_max))
    e_sum = np.sum(e)
    if e_sum < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return e / e_sum

def _grad_lse(a, p, N):
    """
    Gradient of LSE(conv(a), T) w.r.t. a.
    g[j] = sum_{i=0}^{n-1} a[i] * p[j+i]
    Computed as cross-correlation via FFT.
    """
    n = len(a)
    # cross_corr(a, p)[j] = sum_i a[i] * p[i+j]
    # = ifft(conj(FFT(a)) * FFT(p)).real
    g = ifft(np.conj(fft(a, N)) * fft(p, N)).real[:n]
    return g

def _adam_update(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

def _project_nonnegative_normalize(a, target_sum):
    """Project to non-negative and normalize to target sum."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _spectral_compress(a, N, beta=0.7):
    """
    Spectral compression: reduce dynamic range of convolution.
    c -> mean + beta*(c - mean), then inverse transform.
    """
    n = len(a)
    c = _convolve_self(a, N)[:2*n-1]
    c_mean = np.mean(c)
    c_compressed = c_mean + beta * (c - c_mean)
    # Ensure non-negative (the convolution of a non-negative sequence should be non-negative,
    # but compression might introduce negatives at the tails)
    c_compressed = np.maximum(c_compressed, 0.0)
    
    # Get C_compressed in frequency domain
    C_compressed = fft(c_compressed, N)
    
    # A_new = sqrt(C_compressed) (complex square root)
    # We need A_new such that A_new^2 = C_compressed
    A_new = np.sqrt(C_compressed + 0j)
    
    # Recover time domain
    a_new = ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new * (np.sum(a) / s)
    else:
        a_new = a.copy()
    
    return a_new

Minimize 2n*max(self-convolution)/(sum)^2 using Adam optimizer with log-sum-exp
smoothing, temperature annealing, spectral peak compression, and adaptive restarts.
Key improvements over subgradient: smooth objective uses all peaks simultaneously,
Adam provides adaptive per-coordinate learning rates, and spectral compression

import numpy as np
import time
from scipy.fft import fft, ifft


def _next_pow2(n):
    return 1 << (int(n) - 1).bit_length()


def _convolve_self(a, N):
    Full = fft(a, N)

def _score_fast(a):

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best = a.copy()
    best_score = _score_fast(a)
    stuck = 0
    lr = lr_init

    step = 0
    while time.monotonic() < deadline:
        step += 1
        # Temperature schedule: increase T over time (sharper max approximation)
        progress = min(1.0, step / 50000.0)

        g = g - np.mean(g)

def _run_subgrad_refine(a, N, target_sum, deadline, lr_init=0.05, beta=0.9, top_k=3):
    n = len(a)
    momentum = np.zeros(n)
    lr = lr_init
    best = a.copy()
    best_score = _score_fast(a)
    stuck = 0
    step = 0

    while time.monotonic() < deadline:
        step += 1
        c = _convolve_self(a, N)[:2*n-1]
        # Top-k peaks
        top_idx = np.argsort(c)[-top_k:]
        weights = c[top_idx] - c[top_idx[0]] + 1e-10
        weights = weights / np.sum(weights)

        # Weighted mirror
        mirror = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:

        grad = mirror
        grad = grad - np.mean(grad)
        momentum = beta * momentum + (1.0 - beta) * grad
        a = a - lr * momentum
        a = _project_nonneg_normalize(a, target_sum)
        lr = max(lr * 0.9999, lr_init * 0.05)

            sc = _score_fast(a)

                a = _project_nonneg_normalize(a, target_sum)
                momentum = np.zeros(n)
                lr = lr_init * 0.5
                stuck = 0

    return best, best_score
