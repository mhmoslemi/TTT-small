    """
    n = len(a)
    # Build constraint matrix
    K = len(active_lags)
    A_rows = []
    A_cols = []

            A_cols.append(j)
            A_vals.append(a[i])

    c = -np.ones(n)

        g = np.maximum(g, 0)
        if g.sum() > 1e-10:

    return None

"""
Optimize step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based Frank-Wolfe (conditional gradient) with adaptive active set,
combined with momentum subgradient descent and line search.
Key difference from previous: direct LP-based direction finding, nonlinear step sizes,
and multi-scale active set updates instead of L-BFGS with LSE smoothing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 without calling evaluate_sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def build_lp_matrix(a, active_lags):
    """Build sparse constraint matrix for conv(a,g)[k] <= M.
    Returns (A_sparse, n_vars) where A has shape (len(active_lags), n)."""
    n = len(a)
    K = len(active_lags)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_lags):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        # a[k-j] for j in [j_lo, j_hi]
        segment = a[k - j_hi:k - j_lo + 1]
        rows.extend([idx] * len(segment))
        cols.extend(range(j_lo, j_hi + 1))
        vals.extend(segment)
    if len(vals) == 0:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    return A


def solve_lp_direction(a, M, active_lags):
    """Solve LP: max sum(g) s.t. g>=0, conv(a,g)[k]<=M for k in active_lags.
    Returns g (the LP solution) or None on failure."""
    n = len(a)
    A = build_lp_matrix(a, active_lags)
    if A is None:
        return None
    K = A.shape[0]
    if K == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = M * np.ones(K)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def compute_subgradient(a, top_idx):
    """Compute average subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for k in top_idx:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        # d/da[j] conv(a,a)[k] = 2*a[k-j] for j in [j_lo, j_hi]
        g[j_lo:j_hi + 1] += 2.0 * a[k - j_hi:k - j_lo + 1]
    g /= len(top_idx)
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def line_search_step(a, direction, t_max=1.0, n_evals=8):
    """Find good step size t by evaluating score at several t values."""
    best_t = t_max / 2
    best_score = compute_score_fast(a)
    for i in range(1, n_evals + 1):
        t = t_max * (1.0 - (1.0 - t_max) * i / (n_evals + 1))
        a_try = (1.0 - t) * a + t * direction
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = compute_score_fast(a_try)
        if sc < best_score:
            best_score = sc
            best_t = t
    return best_t, best_score


def frank_wolfe_phase(a, deadline, rng, verbose=True):
    """LP-based Frank-Wolfe optimization. Returns (best_a, best_score)."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_fast(a)
    K_start = 30
    K_end = 300
    max_iters = 500
    momentum = 0.0
    v = np.zeros(n)  # momentum velocity

    if verbose:
        print(f"FW Phase: n={n}, initial score={best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        K = int(K_start + (K_end - K_start) * it / max_iters)
        K = max(K, 20)
        top_idx = np.argsort(conv)[::-1][:K]
        top_idx = sorted(top_idx)

        # Solve LP for direction
        g = solve_lp_direction(a, M, top_idx)
        if g is not None and g.sum() > 1e-10:
            direction = g / g.sum()
        else:
            # Fallback: subgradient direction
            sg = compute_subgradient(a, top_idx[:min(50, K)])
            direction = project_simplex(1.0 - 0.01 * sg / (sg.max() + 1e-15))

        # Momentum update
        v = momentum * v + (1.0 - momentum) * (direction - a)
        v = np.maximum(v, -1e-10)
        vs = v.sum()
        if vs > 1e-12:
            v = v / vs

        # Line search
        t_current = 2.0 / (it + 2)
        t_try = min(t_max=1.0, t_current * 2)
        t, sc = line_search_step(a, a + v, t_max=t_try)
        direction = a + v
        a_new = (1.0 - t) * a + t * direction
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new / s
        sc_new = compute_score_fast(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            if verbose and it % 50 == 0:
                print(f"  FW iter {it}: score={best_score:.6f}, t={t:.4f}, K={K}")
        a = a_new
        momentum = min(0.5, 0.01 * it)

    return best_a, best_score


def subgradient_phase(a, deadline, rng, verbose=True):
    """Momentum subgradient with adaptive step size. Returns (best_a, best_score)."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_fast(a)
    v = np.zeros(n)
    beta = 0.7
    step = 0.05
    improve_count = 0
    patience = 200

    if verbose:
        print(f"SG Phase: n={n}, initial score={best_score:.6f}")

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        # Adaptive K: more lags as we converge
        K = min(200, 30 + it // 5)
        top_idx = np.argsort(conv)[::-1][:K]

        sg = compute_subgradient(a, top_idx)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        # Normalize subgradient
        sg_norm = sg / sg_max

        # Momentum
        v = beta * v - step * sg_norm

        # Update
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s

        sc = compute_score_fast(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            improve_count = 0
            a = a_new
            if verbose and it % 200 == 0:
                print(f"  SG iter {it}: score={best_score:.6f}, step={step:.5f}, K={K}")
        else:
            a = a_new  # Still move (momentum helps escape)
            improve_count += 1
            if improve_count > patience:
                step *= 0.8
                v *= 0.5
                improve_count = 0
                # Occasionally perturb
                if step < 0.001:
                    pert = rng.normal(0, 0.002, n)
                    a = np.maximum(a + pert, 0) / (np.maximum(a + pert, 0).sum() + 1e-15)
                    step = 0.01
                    patience = 100

        # Decay step
        step *= 0.9995

    return best_a, best_score


def try_alternative_sequence(n_new, deadline, rng):
    """Try a sequence of different length. Returns (a, score) or (None, inf)."""
    if n_new < 50 or n_new > 20000:
        return None, np.inf
    a = rng.dirichlet(np.ones(n_new) * 0.5)
    # Add some structure
    freq = rng.integers(2, 10)
    modulation = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n_new) / n_new)
    a = a * modulation
    a = np.maximum(a, 0)
    a = a / a.sum()
    sc = compute_score_fast(a)
    return a, sc


def polish_phase(a, deadline, rng, verbose=True):
    """Final fine polishing with very small steps and occasional restarts."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_fast(a)
    step = 0.005
    v = np.zeros(n)
    beta = 0.5

    if verbose:
        print(f"Polish Phase: initial score={best_score:.6f}")

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        top_idx = np.argsort(conv)[::-1][:min(100, len(conv))]
        sg = compute_subgradient(a, top_idx)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        v = beta * v - step * sg / sg_max
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new / s
        sc = compute_score_fast(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 500 == 0:
                print(f"  Polish iter {it}: score={best_score:.6f}")
        else:
            a = a_new

        step *= 0.999
        if step < 1e-5:
            # Random perturbation to escape
            if rng.random() < 0.1:
                pert = rng.normal(0, 0.003, n)
                a_p = np.maximum(a + pert, 0)
                if a_p.sum() > 1e-10:
                    a_p = a_p / a_p.sum()
                    sc_p = compute_score_fast(a_p)
                    if sc_p < best_score:
                        best_score = sc_p
                        best_a = a_p.copy()
                        a = a_p
                        print(f"  Perturbation helped: {best_score:.6f}")
            step = 0.003
            v *= 0.3

    return best_a, best_score
