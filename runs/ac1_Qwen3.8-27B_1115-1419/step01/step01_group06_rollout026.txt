def next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def normalize(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = v.sum()

        v = np.ones(v.size if v.size > 0 else 1)
        s = v.sum()
    return v / s

def softmax_weights(w, beta):
    """Compute softmax(beta * w) / sum(softmax(beta*w))"""
    w = np.asarray(w, dtype=np.float64)
    b = float(beta)
    x = b * w
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.zeros(w.size)
        out[w.argmax()] = 1.0
        return out
    return e / s

def compute_gradient(a, N_fft, beta):

    Compute gradient of smoothed max of self-convolution.
    Returns (grad, C, score) where score = 2*n*max(C).
    """
    n = len(a)
    A = np.fft.rfft(a, N_fft)
    C = np.fft.irfft(A * A, N_fft)
    C = C[:2*n-1]
    C = np.maximum(C, 0.0)
    
    p = softmax_weights(C, beta)
    
    # Cross-correlation: q[i] = sum_m p[m+i] * a[m]

    N2 = next_pow2(2*n - 1 + n - 1)  # = next_pow2(3n-2)

    A2 = np.fft.rfft(a, N2)

    grad = 2.0 * q - p[0:2*n-1:2] * a
    
    score = 2.0 * n * float(C.max())

def adam_step(grad_w, w, m, v, t, lr):

    return w_new, m, v, t

using FFT-based gradient descent with adaptive sharpening, peak deflation,
spectral equalization, and multi-start strategy.
"""

import numpy as np
import time


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 0.0 or not np.isfinite(s):
        v = np.ones(max(1, v.size), dtype=np.float64)
        s = float(v.sum())
    return v / s


def _softmax_w(w, beta):
    w = np.asarray(w, dtype=np.float64)
    if w.size == 0:
        return np.array([1.0])
    b = float(beta)
    x = b * w
    x = x - x.max()
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        out = np.zeros(w.size, dtype=np.float64)
        out[int(np.argmax(w))] = 1.0
        return out
    return e / s


def _fft_score(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    C = C[:2 * n - 1]
    C = np.maximum(C, 0.0)
    mx = float(C.max())
    return float(2.0 * n * mx) if np.isfinite(mx) else float('inf')


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    n = len(a)
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return float('inf')
    C = np.convolve(a, a)
    mx = float(C.max())
    return float(2.0 * n * mx / (s * s)) if np.isfinite(mx) else float('inf')


def _compute_grad_C(a, N, N2, beta):
    """Compute (grad, C, p, score) for smoothed-Linf objective."""
    n = len(a)
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    C = C[:2 * n - 1]
    C = np.maximum(C, 0.0)
    mx = float(C.max())
    score = float(2.0 * n * mx)
    p = _softmax_w(C, beta)
    # Cross-correlation q[i] = sum_m p[m+i] * a[m]
    P = np.fft.rfft(p, N2)
    A2 = np.fft.rfft(a, N2)
    q = np.fft.irfft(P * np.conj(A2), N2)[:n]
    grad = 2.0 * q - p[0:2 * n - 1:2] * a
    return grad, C, p, score


def _adam_log_grad(grad, a, w, m, v, t, lr):
    """Adam step in log-probability space given gradient w.r.t. a."""
    t += 1
    b1, b2, eps = 0.9, 0.999, 1e-8
    grad_w = a * (grad - float(np.dot(a, grad)))
    grad_w = np.where(np.isfinite(grad_w), grad_w, 0.0)
    m = b1 * m + (1.0 - b1) * grad_w
    v = b2 * v + (1.0 - b2) * grad_w * grad_w
    mhat = m / (1.0 - b1 ** t)
    vhat = v / (1.0 - b2 ** t)
    step = mhat / (np.sqrt(vhat) + eps)
    step = np.clip(step, -8.0, 8.0)
    w_new = w - lr * step
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= float(w_new.max())
    a_new = np.exp(w_new - w_new.max())
    s = float(a_new.sum())
    if s <= 0.0 or not np.isfinite(s):
        a_new = a.copy()
    else:
        a_new = a_new / s
    return a_new, w_new, m, v, t


def _deflation_step(a, C, p, eta):
    """Peak deflation: reduce a[i] where it contributes to the peak."""
    n = len(a)
    peak_pos = int(np.argmax(C))
    # For i contributing to C[peak_pos]: a[i]*a[peak_pos - i]
    # Reduce a[i] proportional to a[peak_pos - i]
    a_new = a.copy()
    lo = max(0, peak_pos - (n - 1))
    hi = min(n - 1, peak_pos)
    for i in range(lo, hi + 1):
        j = peak_pos - i
        if 0 <= j < n:
            a_new[i] = a[i] * (1.0 - eta * a[j])
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s > 0.0 and np.isfinite(s):
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def _spectral_equalize(a, N, strength):

    n = len(a)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    mag[0] = 1.0  # DC is fixed (sum = 1)

        target = float(non_dc.mean())

            mixed_mag[0] = 1.0
            phase = A / (np.abs(A) + 1e-300)
            A_new = mixed_mag * phase
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = float(a_new.sum())
            if s > 0.0 and np.isfinite(s):

    return a.copy()


def _log_weights(a):

def _from_log_weights(w):

        if iters % 50 == 0 and stall > 5:
            C_now = np.fft.irfft(np.fft.rfft(a, N) ** 2, N)[:2 * n - 1]
            C_now = np.maximum(C_now, 0.0)
            p_now = _softmax_w(C_now, beta)
            for eta in (0.01, 0.005, 0.002):
                a_defl = _deflation_step(a, C_now, p_now, eta)
                sc_defl = _fft_score(a_defl, N)
                if np.isfinite(sc_defl) and sc_defl < score:
                    a = a_defl
                    w = _log_weights(a)
                    m = np.zeros(n, dtype=np.float64)
                    v = np.zeros(n, dtype=np.float64)
                    t_adam = 0
                    break

        # Periodic spectral equalization
        if iters % 100 == 0 and stall > 10:
            a_spec = _spectral_equalize(a, N, 0.1)
            sc_spec = _fft_score(a_spec, N)
            if np.isfinite(sc_spec) and sc_spec < score:
                a = a_spec
                w = _log_weights(a)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                t_adam = 0
                stall = 0

        # Restart if stalled too long
        if stall > 100:
            # Try structured perturbation
            for amp in (0.05, 0.1, 0.2, 0.3):
                w_pert = w + rng.normal(0.0, amp, size=n)
                a_pert = _from_log_weights(w_pert)
                sc_pert = _fft_score(a_pert, N)
                if np.isfinite(sc_pert) and sc_pert < best_score:
                    a = a_pert
                    w = w_pert
                    m = np.zeros(n, dtype=np.float64)
                    v = np.zeros(n, dtype=np.float64)
                    t_adam = 0
                    score = sc_pert
                    break
                elif np.isfinite(sc_pert) and sc_pert < score + 0.05:
                    a = a_pert
                    w = w_pert
                    m = np.zeros(n, dtype=np.float64)
                    v = np.zeros(n, dtype=np.float64)
                    t_adam = 0
                    break
            else:
                # Reset to best
                a = best_a.copy()
                w = _log_weights(a)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                t_adam = 0
                score = best_score
            stall = 0
            beta = max(5.0, beta * 0.5)
            lr = 0.1

    return best_a, best_score


def _make_gaussian_init(n, center_frac=0.5, sigma_frac=0.15):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * n
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return _normalize(a)


def _make_raised_cosine_init(n):
    """Raised cosine window."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    return _normalize(a)


def _make_tapered_init(n, power=2.0):
    """Tapered (sin^power) initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n) ** power
    return _normalize(a)


def _finalize(a):
    """Scale to valid range and return as list."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    scale = float(np.sqrt(2.0 * n))
    if s > 0.0 and np.isfinite(s):
        a = a * scale / s
    else:
        a = np.ones(n, dtype=np.float64) * (scale / float(n))
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.02 / max(s2, 1e-300))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point: runs multi-start FFT-based optimization to find a sequence
    minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    """
    t_start = time.monotonic()
    total_budget = min(float(budget_s), 2390.0)

    # Load parent if available
    parent = None
    try:
        obj = globals().get("height_sequence_1", None)
        if obj is not None:
            arr = np.asarray(obj, dtype=np.float64).ravel()
            if arr.size >= 16:
                arr = np.where(np.isfinite(arr), arr, 0.0)
                arr = np.maximum(arr, 0.0)
                if arr.sum() > 0:
                    parent = arr / arr.sum()
    except Exception:
        pass

    # Define candidate configurations: (length, init_type, time_fraction)
    # We'll try the parent length and a few others
    if parent is not None:
        parent_len = len(parent)
        # Round to power of 2 for efficiency, but keep close to parent
        target_len = min(16384, max(2048, parent_len))
        # Use parent length directly (don't need power of 2)
        configs = [
            (parent_len, "parent", 0.35),
            (4096, "gaussian", 0.15),
            (8192, "gaussian", 0.20),
            (4096, "raised_cosine", 0.10),
            (2048, "gaussian", 0.10),
            (parent_len, "perturbed", 0.10),
        ]
    else:
        configs = [
            (4096, "gaussian", 0.30),
            (8192, "gaussian", 0.30),
            (4096, "raised_cosine", 0.20),
            (2048, "tapered", 0.20),
        ]

    best_overall = None
    best_overall_score = float('inf')

    rng = np.random.default_rng(seed)

    for (n, init_type, frac) in configs:
        elapsed = time.monotonic() - t_start
        remaining = total_budget - elapsed
        if remaining < 30.0:
            break

        # Create initialization
        if init_type == "parent" and parent is not None:
            if len(parent) == n:
                a_init = parent.copy()
            else:
                # Interpolate or truncate
                if len(parent) > n:
                    a_init = parent[:n]
                else:
                    # Pad with small values
                    pad = n - len(parent)
                    a_init = np.concatenate([parent, np.full(pad, 1e-6)])
                a_init = _normalize(a_init)
        elif init_type == "perturbed" and parent is not None:
            if len(parent) == n:
                base = parent.copy()
            else:
                if len(parent) > n:
                    base = parent[:n]
                else:
                    pad = n - len(parent)
                    base = np.concatenate([parent, np.full(pad, 1e-6)])
                base = _normalize(base)
            w = _log_weights(base)
            w_pert = w + rng.normal(0.0, 0.15, size=n)
            a_init = _from_log_weights(w_pert)
        elif init_type == "gaussian":
            a_init = _make_gaussian_init(n, center_frac=0.5, sigma_frac=0.12)
        elif init_type == "raised_cosine":
            a_init = _make_raised_cosine_init(n)
        elif init_type == "tapered":
            a_init = _make_tapered_init(n, power=2.0)
        else:
            a_init = _make_gaussian_init(n)

        # FFT sizes
        N = _next_pow2(2 * n - 1)
        N2 = _next_pow2(3 * n - 2)

        time_for_this = min(remaining * frac, remaining - 10.0)
        if time_for_this < 20.0:
            continue

        print(f"  Trying n={n}, init={init_type}, budget={time_for_this:.0f}s")

        result_a, result_score = _optimize_single(a_init, N, N2, time_for_this, seed + n)

        if result_score < best_overall_score:
            best_overall_score = result_score
            best_overall = result_a.copy()
            print(f"    -> FFT score: {result_score:.6f}")

    # If we still have time, do a final refinement on the best
    elapsed = time.monotonic() - t_start
    remaining = total_budget - elapsed
    if remaining > 60.0 and best_overall is not None:
        n = len(best_overall)
        N = _next_pow2(2 * n - 1)
        N2 = _next_pow2(3 * n - 2)
        print(f"  Final refinement on n={n}, budget={remaining - 15:.0f}s")
        result_a, result_score = _optimize_single(best_overall, N, N2, remaining - 15.0, seed + 999)
        if result_score < best_overall_score:
            best_overall_score = result_score
            best_overall = result_a.copy()
            print(f"    -> FFT score after refinement: {result_score:.6f}")

    # Exact validation of best candidate
    if best_overall is not None:
        exact_sc = _exact_score(best_overall)
        print(f"  Exact score of best candidate: {exact_sc:.6f}")
    else:
        # Fallback: uniform sequence
        best_overall = np.ones(4096, dtype=np.float64) / 4096.0
        print("  WARNING: using fallback uniform sequence")

    return _finalize(best_overall)
