"""
Search for low-peak nonnegative self-convolution sequences using LP-based
Frank-Wolfe (conditional gradient) optimization combined with log-sum-exp
gradient descent. The LP method finds optimal feasible directions at each step
by solving a linear program, providing a fundamentally different optimization
path from pure gradient methods and helping escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _normalize(a):
    """Normalize to nonnegative, sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _score_of(a):
    """Compute 2*n*max(a*a)/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(a.sum())
    if s <= 1e-300:
        return 1e100
    a = a / s
    b = np.convolve(a, a)
    return 2.0 * a.size * float(b.max())


def _build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f*g)[k] for k=0..2n-2."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        A[k, jmin:jmax + 1] = f[k - jmax:k - jmin + 1]
    return A


def _lp_solve(f, M, timeout=5.0):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = _build_conv_matrix(f)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.status == 0:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None


def _make_exponential(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_hann(n):
    x = np.arange(n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_cosine(n):
    x = np.arange(n)
    a = np.cos(np.pi * x / (2.0 * n)) ** 2
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_bell(n, center=0.3, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return _normalize(a)


def _make_poly(n, degree=2):
    x = np.linspace(0, 1, n)
    a = x ** degree
    return _normalize(a)


def _make_symmetric_exp(n, alpha=1.0, center=0.4):
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * np.abs(x - center))
    return _normalize(a)


def _make_two_peak(n, c1=0.2, c2=0.7, w=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
    return _normalize(a)


def _perturb(a, rng, strength=0.15):
    noise = np.exp(rng.normal(0, strength, size=a.size))
    return _normalize(a * noise)


def _block_perturb(a, rng, strength=0.25):
    n = a.size
    c = a.copy()
    width = max(2, n // 16)
    start = int(rng.integers(0, max(1, n - width + 1)))
    c[start:start + width] *= np.exp(rng.normal(0, strength))
    return _normalize(c)


def _resample(old, new_n):
    old = _normalize(np.asarray(old, dtype=np.float64).ravel())
    new_n = max(int(new_n), 2)
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0, 1, old.size)
    x_new = np.linspace(0, 1, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _fw_search(f_init, deadline, rng, verbose=False):
    """LP-based Frank-Wolfe search."""
    f = _normalize(f_init.copy())
    n = f.size
    b = np.convolve(f, f)
    M = float(b.max())
    best_f = f.copy()
    best_M = M
    it = 0
    stall = 0

    if verbose:
        print(f"    FW start: M={M:.10f}, score={2*n*M:.8f}", flush=True)

    t_list = [0.5, 0.3, 0.7, 1.0, 0.2, 0.4, 0.8, 0.9, 0.1]

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break

        to = min(8.0, max(1.0, remaining / 20.0))
        g = _lp_solve(f, M, timeout=to)

        if g is None or g.sum() <= 1e-12:
            stall += 1
            if stall > 3:
                f = _block_perturb(best_f, rng, 0.3)
                M = float(np.convolve(f, f).max())
                stall = 0
            it += 1
            continue

        g = _normalize(g)

        if np.abs(g - f).sum() < 1e-12:
            stall += 1
            if stall > 3:
                f = _perturb(best_f, rng, 0.25)
                M = float(np.convolve(f, f).max())
                stall = 0
            it += 1
            continue

        improved = False
        for t in t_list:
            f_new = (1.0 - t) * f + t * g
            peak = float(np.convolve(f_new, f_new).max())
            if peak < M - 1e-14:
                f = f_new
                M = peak
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 8:
                f = _perturb(best_f, rng, 0.2)
                M = max(float(np.convolve(f, f).max()), 1e-12)
                stall = 0

        it += 1
        if verbose and it % 200 == 0:
            print(f"    FW it={it}: M={M:.10f}, best_M={best_M:.10f}, "
                  f"score={2*n*best_M:.8f}", flush=True)

    if verbose:
        print(f"    FW done: it={it}, score={2*n*best_M:.8f}", flush=True)

    return best_f, 2.0 * n * best_M


def _lse_search(f_init, deadline, rng, verbose=False):
    """LSE-smoothed gradient descent on the simplex."""
    a = _normalize(f_init.copy())
    n = a.size

    fft_size = 1
    while fft_size < 3 * n - 1:
        fft_size <<= 1

    b = np.convolve(a, a)
    maxb = float(b.max())
    score = 2.0 * n * maxb
    best_a = a.copy()
    best_score = score

    alpha = 0.05
    beta = 15.0
    it = 0
    fail = 0
    last_improve = 0

    while time.monotonic() < deadline:
        if maxb <= 1e-300:
            maxb = 1e-300

        beta = min(800.0, beta * 1.005)
        eff_beta = max(1.0, beta / (maxb * n))

        m = float(b.max())
        arg = eff_beta * (b - m) / max(m, 1e-300)
        arg = np.maximum(arg, -700.0)
        e = np.exp(arg)
        p = e / float(e.sum())

        a_rev = a[::-1]
        A_r = np.fft.rfft(a_rev, fft_size)
        P_r = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(A_r * P_r, fft_size).real
        grad = 2.0 * conv[n - 1:2 * n - 1]
        grad = grad - float(grad.mean())

        scale = float(np.max(np.abs(grad)))
        if scale < 1e-300:
            a = _perturb(best_a, rng, 0.2)
            b = np.convolve(a, a)
            maxb = float(b.max())
            score = 2.0 * n * maxb
            alpha = 0.05
            fail = 0
            it += 1
            continue

        step = alpha * maxb / scale
        v = a - step * grad
        v = np.maximum(v, 0.0)
        s = float(v.sum())
        if s < 1e-300:
            a = _perturb(best_a, rng, 0.2)
        else:
            a = v / s

        b = np.convolve(a, a)
        maxb = float(b.max())
        score = 2.0 * n * maxb

        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score
            fail = 0
            last_improve = it
            alpha = min(0.3, alpha * 1.1)
        else:
            fail += 1
            alpha *= 0.92

        if alpha < 0.005:
            alpha = 0.05

        if fail > 100 or (it - last_improve > 500 and it > 100):
            a = _perturb(best_a, rng, 0.2)
            b = np.convolve(a, a)
            maxb = float(b.max())
            score = 2.0 * n * maxb
            alpha = 0.05
            beta = max(10.0, beta * 0.4)
            fail = 0

        it += 1

    if verbose:
        print(f"    LSE done: it={it}, score={best_score:.8f}", flush=True)

    return best_a, best_score


def _get_given():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or arr.sum() <= 1e-300:
        return None
    return arr / arr.sum()


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main search: LP-based Frank-Wolfe + LSE gradient descent.
    Returns best nonnegative coefficient list found within budget.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + budget - 25.0
    rng = np.random.default_rng(seed)
    given = _get_given()

    # Best tracking
    best_a = _make_exponential(256, 1.0)
    best_score = _score_of(best_a)
    print(f"Initial: n={best_a.size}, score={best_score:.8f}", flush=True)

    if given is not None:
        gs = _score_of(given)
        print(f"Given: n={given.size}, score={gs:.8f}", flush=True)
        if gs < best_score:
            best_a = given.copy()
            best_score = gs

    # ---- Phase 1: Frank-Wolfe LP on n=256 (target length) ----
    print("\n[Phase 1] FW-LP on n=256", flush=True)
    n1 = 256
    inits_256 = [
        _make_exponential(n1, 0.6),
        _make_exponential(n1, 1.2),
        _make_exponential(n1, 2.0),
        _make_hann(n1),
        _make_cosine(n1),
        _make_bell(n1, 0.25, 0.35),
        _make_bell(n1, 0.5, 0.25),
        _make_symmetric_exp(n1, 1.0, 0.35),
        _make_two_peak(n1, 0.15, 0.65, 0.12),
        _make_poly(n1, 1.5),
    ]
    if given is not None:
        inits_256.append(_resample(given, n1))
    if best_a.size != n1:
        inits_256.append(_resample(best_a, n1))
    else:
        inits_256.append(best_a.copy())

    # Deduplicate
    seen_sigs = []
    unique_inits = []
    for ic in inits_256:
        sig = tuple(np.round(ic[::8], 4))
        if sig not in seen_sigs:
            seen_sigs.append(sig)
            unique_inits.append(ic)

    # Allocate time proportionally
    num_starts = len(unique_inits)
    time_per_start = max(20, min(80, (deadline - time.monotonic() - 60) / (num_starts + 4)))

    for idx, init in enumerate(unique_inits):
        if time.monotonic() > deadline - 45:
            break
        t_end = min(deadline - 15, time.monotonic() + time_per_start)
        if time.monotonic() >= t_end:
            continue
        res_a, res_score = _fw_search(init.copy(), t_end, rng, verbose=(idx == 0))
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"  NEW BEST (FW n=256, init={idx}): score={best_score:.8f}", flush=True)
        else:
            print(f"  FW n=256 init={idx}: score={res_score:.8f} (best={best_score:.8f})", flush=True)

    # ---- Phase 2: LSE refinement on best ----
    print("\n[Phase 2] LSE refinement", flush=True)
    if time.monotonic() < deadline - 45:
        t_end = min(deadline - 15, time.monotonic() + max(30, min(120, (deadline - time.monotonic()) / 3)))
        res_a, res_score = _lse_search(best_a.copy(), t_end, rng, verbose=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"  LSE improved: score={best_score:.8f}", flush=True)

    # ---- Phase 3: More FW starts with perturbation of best ----
    print("\n[Phase 3] FW perturbation restarts", flush=True)
    for restart_idx in range(6):
        if time.monotonic() > deadline - 30:
            break
        perturbed = _perturb(best_a, rng, 0.15 + 0.05 * restart_idx)
        # Also try block perturbations
        perturbed = _block_perturb(perturbed, rng, 0.2)
        t_end = min(deadline - 10, time.monotonic() + max(15, min(50, (deadline - time.monotonic()) / 4)))
        res_a, res_score = _fw_search(perturbed.copy(), t_end, rng, verbose=False)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"  NEW BEST (restart {restart_idx}): score={best_score:.8f}", flush=True)
        else:
            print(f"  Restart {restart_idx}: score={res_score:.8f} (best={best_score:.8f})", flush=True)

    # ---- Phase 4: Try larger n ----
    print("\n[Phase 4] Larger n exploration", flush=True)
    for n_big in [512, 1024]:
        if time.monotonic() > deadline - 60:
            break
        init_big = _resample(best_a, n_big)
        init_big = _normalize(0.75 * init_big + 0.25 * _make_exponential(n_big, 0.8))
        t_end = min(deadline - 20, time.monotonic() + max(30, min(150, (deadline - time.monotonic()) / 2)))
        res_a, res_score = _fw_search(init_big.copy(), t_end, rng, verbose=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"  NEW BEST (n={n_big}): score={best_score:.8f}", flush=True)
        else:
            print(f"  n={n_big}: score={res_score:.8f} (best={best_score:.8f})", flush=True)

    # ---- Phase 5: Final combined refinement ----
    print("\n[Phase 5] Final refinement", flush=True)
    # Alternate LSE and FW for remaining time
    remaining = deadline - time.monotonic() - 5
    if remaining > 20:
        # LSE first
        t_end = time.monotonic() + max(10, remaining * 0.4)
        res_a, res_score = _lse_search(best_a.copy(), min(t_end, deadline - 10), rng, verbose=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
            print(f"  Final LSE: score={best_score:.8f}", flush=True)

        # Then FW
        if time.monotonic() < deadline - 15:
            t_end = min(deadline - 3, time.monotonic() + max(10, (deadline - time.monotonic() - 5) * 0.6))
            res_a, res_score = _fw_search(best_a.copy(), t_end, rng, verbose=True)
            if res_score < best_score:
                best_a = res_a.copy()
                best_score = res_score
                print(f"  Final FW: score={best_score:.8f}", flush=True)

    # Ensure we return at n=256 if that's better (the evaluator uses n from the sequence)
    # Actually, the score already accounts for n, so just return the best.

    out = _normalize(best_a)
    final_score = _score_of(out)
    print(f"\n=== FINAL: n={out.size}, score={final_score:.10f} ===", flush=True)

    return [float(x) for x in out]
