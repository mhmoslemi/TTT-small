        idx = k - np.arange(js, je)
        g[js:je] = 2.0 * a[idx]
    return g


def _run_smooth(a, n, deadline, tag, it_limit=10000000):
    """Momentum smooth gradient descent. Returns (best_a, best_obj)."""
    beta = 0.9
    eta = 0.08
    v = np.zeros(n, dtype=np.float64)
    best = a.copy()
    best_o = _obj_val(a)
    tau = n / 3.0
    stall = 0
    t0 = time.monotonic()

    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _fft_convolve_self(a)
        mc = float(np.max(c))
        o = 2.0 * mc / n  # since sum(a) = n

        if o < best_o - 1e-10:
            best_o = o
            best = a.copy()
            stall = 0
            if it % 20000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} tau={tau:.1f} eta={eta:.4f} t={time.monotonic()-t0:.0f}s")
        else:
            stall += 1

        # Softmax weighting
        mx = float(np.max(c))
        p = np.exp((c - mx) / tau)
        ps = float(p.sum())
        if ps < 1e-300:
            p = np.full_like(c, 1.0 / len(c))
        else:
            p /= ps

        g = _grad_smooth(a, p)
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break

        v = beta * v + (1.0 - beta) * (g / mg)
        a = _project(a - eta * v, n)

        tau = max(tau * 0.99996, n * 0.0003)
        if stall > 8000:
            eta = max(eta * 0.5, 1e-7)
            stall = 0

    return best, best_o


def _run_hard_topk(a, n, deadline, tag, k_top=1, it_limit=10000000):
    """Hard gradient descent targeting top-k peaks. Returns (best_a, best_obj)."""
    eta = 0.04
    beta = 0.85
    v = np.zeros(n, dtype=np.float64)
    best = a.copy()
    best_o = _obj_val(a)
    stall = 0
    t0 = time.monotonic()

    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _fft_convolve_self(a)
        mc = float(np.max(c))
        o = 2.0 * mc / n

        if o < best_o - 1e-10:
            best_o = o
            best = a.copy()
            stall = 0
            if it % 10000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} eta={eta:.5f} t={time.monotonic()-t0:.0f}s")
        else:
            stall += 1

        # Top-k peaks
        if k_top == 1:
            k = int(np.argmax(c))
            g = _grad_single(a, k)
        else:
            top_idx = np.argpartition(c, -k_top)[-k_top:]
            g = np.zeros(n, dtype=np.float64)
            for ki in top_idx:
                g += _grad_single(a, int(ki))

        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break

        v = beta * v + (1.0 - beta) * (g / mg)
        a = _project(a - eta * v, n)

        if stall > 4000:
            eta = max(eta * 0.6, 1e-8)
            stall = 0
        if it > 0 and it % 50000 == 49999:
            eta = max(eta * 0.97, 1e-9)

    return best, best_o


def _perturb_targeted(a, n, c, rng, strength=0.15):
    """Targeted perturbation: reduce peak contributors, increase others."""
    b = a.copy()
    k_peak = int(np.argmax(c))
    # Identify top contributors to peak
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak) + 1
    contrib = np.zeros(n)
    idx = k_peak - np.arange(js, je)
    contrib[js:je] = a[js:je] * a[idx]
    
    total_contrib = float(contrib.sum())
    if total_contrib < 1e-15:
        # Random perturbation
        b += rng.normal(0, strength / 10, size=n)
        return _project(b, n)
    
    # Reduce top contributors, increase random others
    top_mask = contrib > 0.3 * float(np.max(contrib))
    noise_top = rng.normal(0, strength * 0.3, size=n)
    noise_top[~top_mask] *= 3.0
    b += noise_top * a
    return _project(b, n)


def _perturb_random_block(a, n, rng, sigma=0.2, width_frac=0.05):
    """Perturb a random contiguous block."""
    b = a.copy()
    w = max(2, int(n * width_frac))
    s = int(rng.integers(0, max(1, n - w + 1)))
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _project(b, n)


def _perturb_swap(a, n, rng):
    """Swap two random blocks."""
    b = a.copy()
    w = max(2, n // 100)
    s1 = int(rng.integers(0, max(1, n - w)))
    s2 = int(rng.integers(0, max(1, n - w))
    if abs(s1 - s2) > w + 1:
        b[s1:s1+w], b[s2:s2+w] = b[s2:s2+w].copy(), b[s1:s1+w].copy()
    return _project(b, n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: smooth convergence -> hard top-k -> perturb+refine cycle.
    Tests multiple sequence lengths, uses FFT for speed, momentum for convergence.
    Returns best non-negative sequence found within budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    t_start = time.monotonic()

    # Load parent if available
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        pass

    # Try a few n values
    if parent is not None:
        n_primary = len(parent)
        n_list = [n_primary, max(2000, n_primary - 300), max(2000, n_primary + 300)]
    else:
        n_list = [3809, 3500, 4200]

    # Deduplicate and sort
    n_list = sorted(set(n_list))
    # Limit to 3 values to ensure enough time each
    n_list = n_list[:3]

    best_a_global = None
    best_o_global = float('inf')

    # Allocate time: give most to primary n
    weights = [1.0, 0.6, 0.6][:len(n_list)]
    total_w = sum(weights)
    avail_time = budget_s - 40.0

    for ni, n in enumerate(n_list):
        t_alloc = avail_time * weights[ni] / total_w
        n_deadline = min(time.monotonic() + t_alloc, deadline - 5.0)
        print(f"\n=== n={n}, time_alloc={t_alloc:.0f}s ===")

        # Initialize
        if parent is not None and len(parent) == n:
            a = _project(parent.copy(), n)
        elif parent is not None:
            # Interpolate parent to new length
            idx = np.linspace(0, len(parent) - 1, n)
            a = np.interp(idx, np.arange(len(parent)), parent).astype(np.float64)
            a = _project(a, n)
        else:
            a = _project(rng.uniform(0.8, 1.2, size=n), n)

        best_a = a.copy()
        best_o = _obj_val(a)
        print(f"  Init obj: {best_o:.6f}")

        if best_o < best_o_global:
            best_o_global = best_o
            best_a_global = best_a.copy()

        # Phase A: Smooth convergence (25% of n's budget)
        tA = min(time.monotonic() + 0.25 * t_alloc, n_deadline - 10.0)
        if time.monotonic() < tA:
            best_a, best_o = _run_smooth(best_a, n, tA, f"sm_{n}")
            if best_o < best_o_global:
                best_o_global = best_o
                best_a_global = best_a.copy()

        # Phase B: Hard top-1 (30% of n's budget)
        tB = min(time.monotonic() + 0.30 * t_alloc, n_deadline - 5.0)
        if time.monotonic() < tB:
            best_a, best_o = _run_hard_topk(best_a, n, tB, f"h1_{n}", k_top=1)
            if best_o < best_o_global:
                best_o_global = best_o
                best_a_global = best_a.copy()

        # Phase C: Perturb + hard refine loop (40% of n's budget)
        tC = n_deadline
        attempt = 0
        while time.monotonic() < tC - 3.0:
            attempt += 1
            c = _fft_convolve_self(best_a)

            # Choose perturbation type
            ptype = attempt % 4
            sigma = min(0.05 + 0.02 * attempt, 0.5)

            if ptype == 0:
                a_p = _perturb_targeted(best_a, n, c, rng, strength=sigma)
            elif ptype == 1:
                wf = float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.10]))
                a_p = _perturb_random_block(best_a, n, rng, sigma=sigma * 0.5, width_frac=wf)
            elif ptype == 2:
                a_p = _perturb_swap(best_a, n, rng)
            else:
                # Small Gaussian noise on all elements
                a_p = _project(best_a + rng.normal(0, sigma * 0.05, size=n), n)

            # Quick hard refinement (up to 8 seconds)
            sub_deadline = min(time.monotonic() + 8.0, tC - 2.0)
            a_q, o_q = _run_hard_topk(a_p, n, sub_deadline, f"pc{attempt}_{n}", k_top=1, it_limit=20000)

            if o_q < best_o:
                best_o = o_q
                best_a = a_q
                if best_o < best_o_global:
                    best_o_global = best_o
                    best_a_global = best_a.copy()
                    print(f"  [n={n}] attempt {attempt}: NEW GLOBAL BEST obj={best_o:.6f}")

            # Also try top-3 from best
            if time.monotonic() < tC - 4.0:
                sub2 = min(time.monotonic() + 6.0, tC - 2.0)
                a_q2, o_q2 = _run_hard_topk(best_a, n, sub2, f"p3_{attempt}_{n}", k_top=3, it_limit=15000)
                if o_q2 < best_o:
                    best_o = o_q2
                    best_a = a_q2
                    if best_o < best_o_global:
                        best_o_global = best_o
                        best_a_global = best_a.copy()
                        print(f"  [n={n}] attempt {attempt} top3: NEW GLOBAL BEST obj={best_o:.6f}")

        # Final polish on best for this n
        if time.monotonic() < deadline - 3.0:
            final_dl = min(time.monotonic() + 5.0, deadline - 2.0)
            best_a, best_o = _run_hard_topk(best_a, n, final_dl, f"fin_{n}", k_top=1, it_limit=50000)
            if best_o < best_o_global:
                best_o_global = best_o
                best_a_global = best_a.copy()

    # Final: one more polish on global best
    if best_a_global is not None and time.monotonic() < deadline - 2.0:
        nfilt = len(best_a_global)
        final_dl = deadline - 1.0
        best_a_global, best_o_global = _run_hard_topk(
            best_a_global, nfilt, final_dl, "FINAL", k_top=1, it_limit=100000)

    # Build result
    result = [float(max(0.0, min(1000.0, x))) for x in best_a_global]

    # Validate
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')

    if not np.isfinite(sc):
        # Fallback
        if parent is not None:
            result = [float(max(0.0, min(1000.0, x))) for x in parent]
        else:
            result = [1.0] * 3809

    el = time.monotonic() - t_start
    print(f"\nFINAL: internal_obj={best_o_global:.6f} evaluated={sc:.6f} n={len(result)} elapsed={el:.0f}s")
    return result
