    t0 = time.monotonic()

    for it in range(1000000):
        if time.monotonic() >= deadline:
            break
        c = _conv(a)
        f = float(np.max(c))
        if f < best_f - 1e-10:
            best_f = f
            best_a = a.copy()
            stall = 0
            if it % 5000 == 0:
                el = time.monotonic() - t0
                print(f"  [{tag}] it={it} best={best_f:.4f} obj={2*best_f/n:.6f} beta={beta:.1f} step={step:.4f} t={el:.0f}s")
        else:
            stall += 1

        val, g = _smooth_grad(a, beta)
        g -= g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break

        # Backtracking line search
        s = step
        a_new = a
        f_new = f
        for _ in range(40):
            a_new = _proj(a - s * g / gn, S)
            f_new = float(np.max(_conv(a_new)))
            if f_new < f - 1e-11 * max(1.0, abs(f)) or s < 1e-8:
                break
            s *= 0.5
        a = a_new
        step = s * 1.2
        step = min(step, 3.0)

        beta *= 0.99995
        if beta < beta_min:
            beta = beta_min

        if stall > 5000:
            step *= 0.5
            beta = max(beta * 0.998, 3.0)
            stall = 0
            if step < 1e-7:
                break

    return best_a, best_f


def _hard_descent(a, S, deadline, tag):
    """Hard max gradient descent with backtracking."""
    n = len(a)
    best_a = a.copy()
    best_f = float(np.max(_conv(a)))
    step = 0.5
    stall = 0
    t0 = time.monotonic()

    for it in range(1000000):
        if time.monotonic() >= deadline:
            break
        mc, g = _hard_max_info(a)
        if mc < best_f - 1e-10:
            best_f = mc
            best_a = a.copy()
            stall = 0
            if it % 10000 == 0:
                el = time.monotonic() - t0
                print(f"  [{tag}] it={it} best={best_f:.4f} obj={2*best_f/n:.6f} step={step:.4f} t={el:.0f}s")
        else:
            stall += 1

        g -= g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break

        s = step
        a_new = a
        for _ in range(40):
            a_new = _proj(a - s * g / gn, S)
            if float(np.max(_conv(a_new))) < mc - 1e-12 or s < 1e-8:
                break
            s *= 0.5
        a = a_new
        step = s * 1.1
        step = min(step, 2.0)

        if it % 20000 == 19999:
            step *= 0.97
        if stall > 10000:
            step *= 0.5
            stall = 0
            if step < 1e-7:
                break

    return best_a, best_f


def _coord_refine(a, S, deadline, tag, n_rounds=5, samples_per_coord=8):
    """Coordinate-wise refinement: try scaling each element."""
    n = len(a)
    best_a = a.copy()
    best_f = float(np.max(_conv(a)))
    rng = np.random.default_rng(99)
    order = rng.permutation(n)

    for rnd in range(n_rounds):
        for ii in order:
            if time.monotonic() >= deadline:
                return best_a, best_f
            orig = a[ii]
            if orig < 1e-10:
                continue
            # Try several multipliers
            mults = np.linspace(0.3, 2.0, samples_per_coord)
            best_m = 1.0
            best_fm = np.max(_conv(a))
            for m in mults:
                a[ii] = orig * m
                a_t = _proj(a, S)
                fm = float(np.max(_conv(a_t)))
                if fm < best_fm - 1e-10:
                    best_fm = fm
                    best_m = m
                    best_a_t = a_t.copy()
            a[ii] = orig * best_m
            a = _proj(a, S)
            f = float(np.max(_conv(a)))
            if f < best_f - 1e-10:
                best_f = f
                best_a = a.copy()

    return best_a, best_f


def _perturb_spectral(a, S, rng, strength):
    """Perturb in frequency domain: add small random spectral component."""
    n = len(a)
    N = 2 * n - 1
    A = fft(a, N)
    # Add small random perturbation to non-DC frequencies
    dA = np.zeros(N, dtype=np.complex128)
    dA[1:] += strength * (rng.normal(size=N - 1) + 1j * rng.normal(size=N - 1))
    # Keep DC fixed to preserve sum
    dA[0] = 0
    A += dA
    a_new = np.real(ifft(A))[:n]
    return _proj(a_new, S)


def _perturb_block(a, S, rng, sigma, width_frac):
    """Multiply a random contiguous block by exp(N(0,sigma))."""
    n = len(a)
    b = a.copy()
    w = max(2, int(n * width_frac))
    s = int(rng.integers(0, max(1, n - w + 1)))
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _proj(b, S)


def _perturb_swap(a, S, rng):
    """Swap two contiguous blocks."""
    n = len(a)
    b = a.copy()
    w = max(2, n // 80)
    s1 = int(rng.integers(0, max(1, n - 2 * w)))
    s2 = s1 + int(rng.integers(w + 1, n - w))
    b[s1:s1+w], b[s2:s2+w] = b[s2:s2+w].copy(), b[s1:s1+w].copy()
    return _proj(b, S)


def _get_init_n(n, rng):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n))
    # Cosine peak in middle
    j = np.arange(n)
    inits.append(0.5 + 0.5 * np.cos(np.pi * (j / (n - 1) - 0.5)))
    # Cosine peak at edges
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * j / (n - 1)))
    # Linear ramp
    inits.append(np.linspace(0.5, 1.5, n))
    # Random
    inits.append(rng.uniform(0.5, 1.5, size=n))
    # Half-high half-low
    a_h = np.ones(n)
    a_h[:n // 2] = 1.3
    a_h[n // 2:] = 0.7
    inits.append(a_h)
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase FFT-based optimizer. Returns best non-negative sequence.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0

    # Try same n as before
    n = 3809
    S = float(n)  # normalize sum to n

    # Get initial from height_sequence_1 if available
    a_parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.all(np.isfinite(parent)):
            a_parent = np.clip(parent, 0.0, 1000.0)
            a_parent = _proj(a_parent, S)
            print(f"Loaded parent: obj={2*float(np.max(_conv(a_parent)))/n:.6f}")
    except Exception:
        pass

    # Best overall
    best_a = None
    best_obj = float('inf')

    def _update_best(a, tag):
        nonlocal best_a, best_obj
        c = _conv(a)
        mc = float(np.max(c))
        sm = float(np.sum(a))
        if sm < 0.01:
            return
        obj = 2.0 * n * mc / (sm * sm)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"  ** NEW BEST [{tag}]: obj={obj:.6f} (max_c={mc:.2f}, sum={sm:.2f})")

    # Phase 1: Optimize from parent (or uniform) - 35% time
    print("=== Phase 1: Primary optimization ===")
    p1_deadline = t_start + 0.35 * (budget_s - 10)

    if a_parent is not None:
        a = a_parent.copy()
    else:
        a = np.ones(n)

    a = _proj(a, S)
    _update_best(a, "init")

    # Smooth descent
    a, _ = _smooth_descent(a, S, p1_deadline, "smooth1", beta_init=500.0, beta_min=20.0)
    _update_best(a, "smooth1")

    # Hard descent
    if time.monotonic() < deadline:
        a, _ = _hard_descent(a, S, p1_deadline, "hard1")
        _update_best(a, "hard1")

    # Coordinate refine
    if time.monotonic() < deadline:
        a, _ = _coord_refine(a, S, p1_deadline, "coord1", n_rounds=3)
        _update_best(a, "coord1")

    # Phase 2: Multi-start from diverse initializations - 30% time
    print("=== Phase 2: Multi-start ===")
    p2_deadline = t_start + 0.65 * (budget_s - 10)
    inits = _get_init_n(n, rng)

    for i_init, a_i in enumerate(inits):
        if time.monotonic() >= p2_deadline:
            break
        a_i = _proj(a_i, S)
        # Quick smooth
        t_sub = min(time.monotonic() + 60.0, p2_deadline - 5.0)
        a_q, _ = _smooth_descent(a_i, S, t_sub, f"ms_s{i_init}", beta_init=300.0, beta_min=30.0)
        _update_best(a_q, f"ms_s{i_init}")
        # Quick hard
        if time.monotonic() < p2_deadline - 5.0:
            t_sub2 = min(time.monotonic() + 40.0, p2_deadline - 3.0)
            a_q2, _ = _hard_descent(a_q, S, t_sub2, f"ms_h{i_init}")
            _update_best(a_q2, f"ms_h{i_init}")

    # Phase 3: Perturbation + refinement loop - 30% time
    print("=== Phase 3: Perturb + refine ===")
    p3_deadline = deadline - 30.0
    base_a = best_a if best_a is not None else np.ones(n)
    base_a = _proj(base_a, S)

    attempt = 0
    while time.monotonic() < p3_deadline:
        attempt += 1
        # Choose perturbation type
        ptype = rng.integers(0, 4)
        if ptype == 0:
            sigma = float(rng.uniform(0.05, 0.4))
            wf = float(rng.choice([0.01, 0.02, 0.05, 0.1, 0.15, 0.25]))
            a_p = _perturb_block(base_a, S, rng, sigma, wf)
        elif ptype == 1:
            a_p = _perturb_swap(base_a, S, rng)
        elif ptype == 2:
            strength = float(rng.uniform(0.1, 1.0))
            a_p = _perturb_spectral(base_a, S, rng, strength)
        else:
            # Random restart (25% of attempts)
            a_p = _proj(rng.uniform(0.3, 1.7, size=n), S)

        # Quick refinement: smooth then hard
        t_s = min(time.monotonic() + 20.0, p3_deadline - 15.0)
        a_q, _ = _smooth_descent(a_p, S, t_s, f"p3_s{attempt}", beta_init=200.0, beta_min=50.0)
        _update_best(a_q, f"p3_s{attempt}")

        if time.monotonic() < p3_deadline - 5.0:
            t_h = min(time.monotonic() + 15.0, p3_deadline - 5.0)
            a_q2, _ = _hard_descent(a_q, S, t_h, f"p3_h{attempt}")
            _update_best(a_q2, f"p3_h{attempt}")

        # Update base if improved
        c_q = _conv(a_q)
        if float(np.max(c_q)) < float(np.max(_conv(base_a))):
            base_a = a_q.copy()

        if attempt % 5 == 0:
            print(f"  Phase 3 attempt {attempt}: current best obj={best_obj:.6f}, elapsed={time.monotonic()-t_start:.0f}s")

    # Phase 4: Final polish with remaining time
    print("=== Phase 4: Final polish ===")
    if best_a is not None and time.monotonic() < deadline - 5.0:
        a_final = _proj(best_a, S)
        # Extended hard descent
        a_final, _ = _hard_descent(a_final, S, deadline - 5.0, "final_hard")
        _update_best(a_final, "final_hard")

        # Coordinate refine
        if time.monotonic() < deadline - 2.0:
            a_final, _ = _coord_refine(a_final, S, deadline - 2.0, "final_coord", n_rounds=2, samples_per_coord=12)
            _update_best(a_final, "final_coord")

    # Final answer
    if best_a is None:
        result = [1.0] * n
    else:
        result = [float(x) for x in best_a]

    # Validate
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    if not np.isfinite(sc):
        result = [1.0] * n
        sc = 2.0

    print(f"FINAL: internal_obj={best_obj:.6f} evaluated={sc:.6f} n={n} time={time.monotonic()-t_start:.1f}s")
    return result
