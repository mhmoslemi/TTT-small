def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def next_fast_len(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    """Return a finite nonnegative vector normalized to sum 1."""
    arr = np.asarray(v, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(max(1, arr.size), dtype=np.float64)
        s = float(arr.sum())
    return arr / s


def log_weights_from_sequence(a):
    """Log of a positive-dimensional affine-core sequence, centered."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, EPS)
    w = np.log(a)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= float(np.max(w))
    return w


def sequence_from_log_weights(w):
    """Stable softmax to a probability vector."""
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size == 0:
        return np.ones(1, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= float(np.max(w))
    e = np.exp(w)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    return e / s


def make_weights(c, beta):
    """Soft weights p_k proportional to exp(beta * (c_k - max(c)) / max(c))."""
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return np.ones(1, dtype=np.float64)
    L = float(np.max(c))
    if not np.isfinite(L) or L <= 0.0:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    x = (c - L) / L
    x = np.where(np.isfinite(x), x, -1e300)
    y = np.clip(float(beta) * x, -745.0, 0.0)
    e = np.exp(y)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def fft_score_only(a, N):
    """FFT-based objective value for a normalized nonnegative sequence."""
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[: 2 * n - 1]
    if c.size:
        if float(np.min(c)) < 0.0:
            c = np.maximum(c, 0.0)
        mx = float(np.max(c))
    else:
        mx = 0.0
    if not np.isfinite(mx):
        return INF
    return float(2.0 * n * mx)


def fft_score_A_c(a, N):
    """Return DFT of a, clipped full self-convolution, and score."""
    n = len(a)

def exact_score_array(a):
    """Exact O(n^2) score using numpy.convolve, for final/top candidates."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return INF
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return INF
    b = np.convolve(a, a)
    if b.size == 0:
        return INF
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return INF
    val = float(2.0 * n * mx / (s * s))
    return val if np.isfinite(val) else INF


def adam_log_step(grad_w, w, m, v, t, lr):
    """One projected-gradient Adam step in log-probability space."""
    t += 1
    beta1 = 0.9
    beta2 = 0.999
    grad_w = np.asarray(grad_w, dtype=np.float64)
    m = m * beta1 + (1.0 - beta1) * grad_w
    gw2 = grad_w * grad_w
    v = v * beta2 + (1.0 - beta2) * gw2
    mhat = m / (1.0 - beta1 ** t)
    vhat = v / (1.0 - beta2 ** t)
    dw = mhat / (np.sqrt(vhat) + 1e-12)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -10.0, 10.0)
    w_new = w - float(lr) * dw
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= float(np.max(w_new))
    a_new = sequence_from_log_weights(w_new)
    return a_new, w_new, m, v, t


def cross_inner_step(F, FR, b, w, m, v, t, beta, lr, N):
    """
    One convex-subproblem step for fixed f:
        minimize max_k (f*b)_k over b >= 0, sum(b)=1.
    """
    n = len(b)
    B = np.fft.rfft(b, N)
    h = np.fft.irfft(F * B, N)
    h = h[: 2 * n - 1]
    if h.size:
        if float(np.min(h)) < 0.0:
            h = np.maximum(h, 0.0)
        mx = float(np.max(h))
    else:
        mx = 0.0
    if not np.isfinite(mx) or mx <= 0.0:
        p = np.zeros(h.size if h.size else 1, dtype=np.float64)
        if p.size:
            p[0] = 1.0
    else:
        p = make_weights(h, beta)

    P = np.fft.rfft(p, N)
    grad = np.fft.irfft(P * FR, N)
    grad = grad[n - 1: 2 * n - 1]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    grad_w = b * (grad - float(np.dot(b, grad)))
    return adam_log_step(grad_w, w, m, v, t, lr)


def self_gradient_candidate(f, A, c, w, m, v, t, beta, lr, N, phase):
    """One smoothed-Linf self-convolution gradient step."""
    n = len(f)
    p = make_weights(c, beta)
    P = np.fft.rfft(p, N)
    FR = phase * np.conjugate(A)
    conv = np.fft.irfft(P * FR, N)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    grad_w = f * (grad - float(np.dot(f, grad)))
    return adam_log_step(grad_w, w, m, v, t, lr)


def add_top_candidate(top, a, score, max_items=16):
    """Keep a small list of best FFT-score candidates."""
    if not top:
        top.append((score, np.asarray(a, dtype=np.float64).copy()))
        return top

    worst_idx = 0
    worst_score = float(top[0][0])
    for i in range(1, len(top)):
        si = float(top[i][0])
        if si > worst_score:
            worst_score = si
            worst_idx = i

    if len(top) < max_items or score < worst_score:
        if len(top) < max_items:
            top.append((score, np.asarray(a, dtype=np.float64).copy()))
        else:
            top[worst_idx] = (score, np.asarray(a, dtype=np.float64).copy())
    return top

def finalize_sequence(a):
    """Scale to sum sqrt(2n), keep valid nonnegative floats, return a list."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        a = np.ones(64, dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    n = a.size
    s = float(a.sum())
    scale = float(np.sqrt(2.0 * n))

    if s > 0.0 and np.isfinite(s):
        a = (a * scale) / s
    else:
        a = np.ones(n, dtype=np.float64) * (scale / float(n))

    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.01 / max(s2, 1e-300))
        a = np.clip(a, 0.0, 1000.0)

    return [float(x) for x in a]
