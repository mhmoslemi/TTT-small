def _grad_hard(a, c):
    """Exact subgradient at the argmax of conv(a,a)."""
    n = len(a)
    k = int(np.argmax(c))
    g = np.zeros(n)
    # d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n and i != k-i
    #                      = a[i] if i == k-i (k = 2i)
    # For simplicity: g[i] = 2*a[k-i] for valid i, then fix the diagonal
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        idx = np.arange(js, je)
        g[idx] = 2.0 * a[k - idx]
        # Fix diagonal: if 2i == k for some i in [js, je), then g[i] should be a[i], not 2*a[i]
        diag = k // 2

            # Wait: d(a[diag]^2)/d(a[diag]) = 2*a[diag]. So g[diag] = 2*a[diag] is correct!
            # The formula g[i] = 2*a[k-i] gives g[diag] = 2*a[k-diag] = 2*a[diag]. Correct!
            # So no fix needed. The formula works as-is.
    return g


def _optimize_smooth(a, n, deadline, tag, it_limit=500000):
    """Entropy-smoothed gradient descent with momentum."""
    tau = n / 2.0
    eta = 0.5
    momentum = 0.9
    v = np.zeros(n)
    best_a = a.copy()
    best_o = _obj(a)
    stall = 0
    start = time.monotonic()
    
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _conv(a)
        o = 2.0 * c.max() / n  # assuming sum(a) = n
        if o < best_o - 1e-9:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it % 20000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} tau={tau:.1f} elapsed={time.monotonic()-start:.0f}s")
        else:
            stall += 1
        
        g = _grad_smooth(a, c, tau)
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        # Momentum update
        v = momentum * v - eta * g
        a_new = a + v
        a = _project_nonneg_sum(a_new, n)
        
        # Decay tau
        tau = max(tau * 0.9998, n * 0.005)
        
        # Adaptive eta
        if stall > 5000:
            eta = max(eta * 0.7, 0.01)
            momentum = max(momentum * 0.95, 0.5)
            stall = 0
    
    return best_a, best_o

"""
Multi-start active-set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses entropy-smoothed GD with momentum, transitions to exact subgradient,
then perturb+refine. Tries multiple n values and diverse initializations
including linear (monotone) sequences which exploit non-symmetry to reduce
the reversal correlation peak.
"""

import numpy as np
import time


def _conv(a):
    return np.convolve(a, a)


def _project(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a), dtype=np.float64)
    return a * (target_sum / s)


def _obj_val(a):
    """Objective assuming sum(a) is known; returns 2*max(conv)/sum."""
    n = len(a)
    c = _conv(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def _softmax(c, tau):
    mx = c.max()
    e = np.exp((c - mx) / tau)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return e / s


def _grad_smooth(a, c, tau):
    """Gradient of entropy-smoothed max of conv(a,a)."""
    p = _softmax(c, tau)
    # d(smooth_max)/d(a[i]) = 2 * sum_j p[i+j] * a[j] for j in [0,n-1]
    # This is correlate(p, a, 'valid') when p is longer
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g


def _grad_hard(a, c):
    """Exact subgradient at the argmax index of conv(a,a)."""
    n = len(a)
    k = int(np.argmax(c))
    g = np.zeros(n, dtype=np.float64)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        idx = np.arange(js, je)
        g[idx] = 2.0 * a[k - idx]
    return g


def _multiple_grad_hard(a, c, frac=0.01):
    """Subgradient averaging over near-max indices."""
    n = len(a)
    cmax = c.max()
    threshold = cmax * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        return _grad_hard(a, c)
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            idx = np.arange(js, je)
            g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    return g


def _optimize_smooth(a, n, deadline, tag, it_limit=1000000):
    """Entropy-smoothed gradient descent with momentum."""
    tau = n / 3.0
    eta = 1.0
    mom = 0.85
    v = np.zeros(n)
    best_a = a.copy()
    best_o = 2.0 * _conv(a).max() / n
    stall = 0
    t_start = time.monotonic()
    
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _conv(a)
        o = 2.0 * c.max() / n
        if o < best_o - 1e-10:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it % 30000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} tau={tau:.1f} eta={eta:.3f} t={time.monotonic()-t_start:.0f}s")
        else:
            stall += 1
        
        g = _grad_smooth(a, c, tau)
        # Center gradient (remove mean to preserve sum more naturally)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        v = mom * v - eta * g
        a_new = a + v
        a = _project(a_new, n)
        
        tau = max(tau * 0.9997, n * 0.003)
        
        if stall > 4000:
            eta = max(eta * 0.7, 0.05)
            mom = max(mom * 0.97, 0.5)
            stall = 0
        elif stall == 0 and eta < 2.0:
            eta = min(eta * 1.02, 2.0)
    
    return best_a, best_o


def _optimize_hard(a, n, deadline, tag, it_limit=1000000):
    """Exact subgradient descent with adaptive step and momentum."""
    eta = 0.3
    mom = 0.6
    v = np.zeros(n)
    best_a = a.copy()
    best_o = 2.0 * _conv(a).max() / n
    stall = 0
    t_start = time.monotonic()
    
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _conv(a)
        o = 2.0 * c.max() / n
        if o < best_o - 1e-10:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it % 20000 == 0:
                print(f"  [{tag}] it={it} obj={o:.6f} eta={eta:.4f} t={time.monotonic()-t_start:.0f}s")
        else:
            stall += 1
        
        # Use averaged subgradient when multiple peaks are close
        if it % 500 == 499:
            cmax = c.max()
            n_active = int((c >= cmax * 0.995).sum())
            if n_active > 3:
                g = _multiple_grad_hard(a, c, frac=0.005)
            else:
                g = _grad_hard(a, c)
        else:
            g = _grad_hard(a, c)
        
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        v = mom * v - eta * g
        a_new = a + v
        a = _project(a_new, n)
        
        if stall > 3000:
            eta = max(eta * 0.7, 0.01)
            mom = max(mom * 0.95, 0.3)
            stall = 0
        elif stall == 0 and eta < 1.0:
            eta = min(eta * 1.03, 1.0)
    
    return best_a, best_o


def _make_linear_init(n, rng):
    """Linearly increasing sequence with optional noise."""
    j = np.arange(n, dtype=np.float64)
    a = j + 1.0
    # Add small noise to break symmetry
    a += rng.normal(0, 0.1, size=n)
    return _project(a, n)


def _make_quadratic_init(n, rng):
    """Quadratically increasing sequence."""
    j = np.arange(n, dtype=np.float64)
    a = j**2 + 1.0
    a += rng.normal(0, 1.0, size=n)
    return _project(a, n)


def _make_oscillating_init(n, rng):
    """Linear + oscillation to spread spectral energy."""
    j = np.arange(n, dtype=np.float64)
    p = int(rng.integers(n//7, n//4))
    a = j + 1.0 + 0.3 * n * np.sin(2.0 * np.pi * p * j / n)
    a = np.maximum(a, 0.1)
    return _project(a, n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-start optimizer trying multiple n values and diverse initializations.
    Phase 1: Smooth GD from multiple starts.
    Phase 2: Hard subgradient refinement.
    Phase 3: Perturb + re-optimize loop.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 8.0
    
    global_best_a = None
    global_best_o = float('inf')
    
    # --- n values to try ---
    # Primary is 3809 (from previous work), try a few others
    n_list = [3809, 3200, 4500]
    
    # Estimate time per n (favor the primary)
    total_time = budget_s - 20.0
    weights = [0.55, 0.20, 0.25]
    n_times = [w * total_time for w in weights]
    
    for ni, n in enumerate(n_list):
        if time.monotonic() >= deadline - 5.0:
            break
        n_deadline = time.monotonic() + n_times[ni]
        
        print(f"\n=== n={n}, time_budget={n_times[ni]:.0f}s ===")
        
        # --- Build initializations ---
        inits = []
        tags = []
        
        # 1. Previous best (if matches n)
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            if parent.ndim == 1 and parent.size == n and np.all(np.isfinite(parent)):
                a0 = np.clip(parent, 0.0, 1000.0)
                if a0.sum() > 0.01:
                    inits.append(_project(a0, n))
                    tags.append("prev")
        except Exception:
            pass
        
        # 2. Linear increasing
        inits.append(_make_linear_init(n, rng))
        tags.append("linear")
        
        # 3. Linear increasing (different noise)
        inits.append(_make_linear_init(n, rng))
        tags.append("linear2")
        
        # 4. Quadratic
        inits.append(_make_quadratic_init(n, rng))
        tags.append("quad")
        
        # 5. Oscillating
        inits.append(_make_oscillating_init(n, rng))
        tags.append("osc")
        
        # 6. Random exponential
        inits.append(_project(rng.exponential(1.0, size=n), n))
        tags.append("rand")
        
        # 7. Uniform (baseline)
        inits.append(np.ones(n, dtype=np.float64))
        tags.append("uniform")
        
        # --- Phase 1: Quick smooth optimization from each init ---
        n_smooth = len(inits)
        smooth_time_each = min(25.0, (n_deadline - time.monotonic()) * 0.5 / n_smooth)
        local_best_a = None
        local_best_o = float('inf')
        
        for ii, (a0, tag) in enumerate(zip(inits, tags)):
            if time.monotonic() >= n_deadline - 10.0:
                break
            sub_dl = time.monotonic() + smooth_time_each
            if sub_dl > time.monotonic() + 3.0:
                a1, o1 = _optimize_smooth(a0, n, sub_dl, f"n{n}_{tag}_s", it_limit=200000)
                if o1 < local_best_o:
                    local_best_o = o1
                    local_best_a = a1.copy()
                    print(f"  n={n} {tag}: smooth obj={o1:.6f}")
                if o1 < global_best_o:
                    global_best_o = o1
                    global_best_a = a1.copy()
        
        # --- Phase 2: Hard optimization from local best ---
        if local_best_a is not None and time.monotonic() < n_deadline - 5.0:
            hard_time = n_deadline - time.monotonic() - 3.0
            if hard_time > 5.0:
                a2, o2 = _optimize_hard(local_best_a, n, time.monotonic() + hard_time, 
                                         f"n{n}_hard", it_limit=500000)
                if o2 < local_best_o:
                    local_best_o = o2
                    local_best_a = a2.copy()
                if o2 < global_best_o:
                    global_best_o = o2
                    global_best_a = a2.copy()
                    print(f"  n={n} hard: obj={o2:.6f}")
    
    # --- Phase 3: Perturb + refine the global best ---
    print(f"\n=== Phase 3: Perturb+refine (obj={global_best_o:.6f}) ===")
    if global_best_a is not None and time.monotonic() < deadline - 5.0:
        n_best = len(global_best_a)
        attempt = 0
        while time.monotonic() < deadline - 3.0:
            attempt += 1
            # Vary perturbation strength
            if attempt <= 20:
                sigma = float(rng.uniform(0.03, 0.15))
            elif attempt <= 50:
                sigma = float(rng.uniform(0.05, 0.25))
            else:
                sigma = float(rng.uniform(0.10, 0.40))
            
            # Multiplicative perturbation
            a_p = global_best_a * np.exp(rng.normal(0, sigma, size=n_best))
            a_p = _project(a_p, n_best)
            
            # Also try additive perturbation sometimes
            if attempt % 3 == 0:
                a_p = global_best_a + rng.normal(0, sigma * global_best_a.mean(), size=n_bottom if False else n_best)
                a_p = _project(a_p, n_best)
            
            # Quick hard refinement
            refine_time = min(12.0, max(3.0, (deadline - time.monotonic()) * 0.4))
            a_r, o_r = _optimize_hard(a_p, n_best, time.monotonic() + refine_time,
                                       f"p{attempt}", it_limit=50000)
            if o_r < global_best_o:
                global_best_o = o_r
                global_best_a = a_r.copy()
                print(f"  Perturb {attempt}: obj={o_r:.6f} sigma={sigma:.3f}")
            
            # Occasionally try smooth refinement too
            if attempt % 5 == 0 and time.monotonic() < deadline - 10.0:
                s_time = min(10.0, (deadline - time.monotonic()) * 0.3)
                a_s, o_s = _optimize_smooth(a_r, n_best, time.monotonic() + s_time,
                                             f"ps{attempt}", it_limit=30000)
                if o_s < global_best_o:
                    global_best_o = o_s
                    global_best_a = a_s.copy()
                    print(f"  Perturb+smooth {attempt}: obj={o_s:.6f}")
    
    # --- Validation and return ---
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_a]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    
    if not np.isfinite(sc) or sc > 2.0:
        # Fallback: linear sequence
        n_fb = 3809
        result = [float((i + 1) * 1000.0 / n_fb) for i in range(n_fb)]
        try:
            sc = float(evaluate_sequence(result))
        except Exception:
            result = [1.0] * n_fb
            sc = float(evaluate_sequence(result))
    
    print(f"\nFINAL: internal={global_best_o:.6f} evaluated={sc:.6f} n={len(result)}")
    return result
