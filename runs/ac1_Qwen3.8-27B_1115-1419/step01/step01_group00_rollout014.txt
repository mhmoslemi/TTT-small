"""
Search for low-peak self-convolution sequences.

Strategy summary:
- Work with probability vectors a (sum = 1), so the objective is
  F(a) = 2*n*max(a*a).
- Use FFT-based convolution and gradients for speed (n ~ 10000 is manageable).
- Primary optimizer: damped Adam on log-heights (softmax simplex), with an
  annealed log-sum-exp surrogate for the maximum convolution coefficient.
- Local polish: exact active-set gradient (top-T max coefficients) with
  backtracking mirror (log-space) line search.
- Global exploration: interpolate/refine the supplied previous best sequence,
  then mix with structured random initializations, low-pass Fourier noise,
  block perturbations, and active-coefficient antisymmetric perturbations.
- Keep the best few candidates and finally validate with evaluate_sequence.
"""

import numpy as np
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize_pos(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / s


def _softprob(theta):
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0:
        return np.ones(1, dtype=np.float64)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    theta = theta - float(np.mean(theta))
    theta = np.clip(theta, -30.0, 30.0)
    mx = float(np.max(theta))
    e = np.exp(theta - mx)
    s = float(e.sum())
    if s < 1e-300:
        return np.full(theta.shape, 1.0 / theta.size, dtype=np.float64)
    return e / s


def _theta_from_prob(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(a > 1e-280, a, 1e-280)
    t = np.log(a)
    t = t - float(np.mean(t))
    return np.clip(t, -30.0, 30.0)


def _conv_and_obj(a, n, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    F = 2.0 * float(n) * float(np.max(c))
    return F, c


def _exact_obj_only(a, n, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return 2.0 * float(n) * float(np.max(c))


def _grad_from_pa(p, a, A, N, n):
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(P * np.conjugate(A), N).real[:n]
    return 4.0 * float(n) * corr


def _smooth_p_from_u(u, tau):
    umax = float(np.max(u))
    ex = np.exp((u - umax) / tau)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full(u.shape, 1.0 / len(u), dtype=np.float64)
        return p, umax
    p = ex / s
    return p, umax + tau * float(np.log(s))


def _candidate_key(item):
    return item[0]


def _add_candidate(candidates, a, n, N, max_keep=30):
    F, c = _conv_and_obj(a, n, N)
    if not np.isfinite(F):
        return candidates, F
    item = (float(F), np.asarray(a, dtype=np.float64).copy(), int(n), int(N))
    candidates.append(item)
    if len(candidates) > max_keep:
        candidates.sort(key=_candidate_key)
        del candidates[max_keep - 1:]
    return candidates, float(F)


def _interp_sequence(src, n):
    src = np.asarray(src, dtype=np.float64)
    m = int(src.size)
    n = int(n)
    if m == n:
        return src.copy()
    if m == 1:
        return np.full(n, float(src[0]), dtype=np.float64)
    xo = np.linspace(0.0, float(m - 1), m)
    xn = np.linspace(0.0, float(m - 1), n)
    return np.interp(xn, xo, src)


def _init_lowpass(n, rng, amp=0.30):
    n = int(n)
    z = rng.normal(0.0, 1.0, size=n)
    if n > 20:
        Z = np.fft.rfft(z)
        cut = max(1, n // 25)
        Z[cut:] = 0.0
        z = np.fft.irfft(Z, n)
    z = z - float(np.mean(z))
    sd = float(np.std(z))
    if sd > 1e-15:
        z = z / sd
    a = 1.0 + amp * z
    a = np.maximum(a, 1e-8)
    return _normalize_pos(a)


def _init_harmonic(n, rng, num=4):
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    if n > 1:
        x = i / float(n)
        for m in range(1, int(num) + 1):
            amp = float(rng.uniform(0.05, 0.25))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * float(m) * x + phase)
    a = np.maximum(a, 1e-8)
    return _normalize_pos(a)


def _init_ramp(n, amp=0.70):
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        x = i / float(n - 1) - 0.5
        a = 1.0 + amp * x + 0.10 * np.sin(2.0 * np.pi * i / float(n))
    else:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 1e-8)
    return _normalize_pos(a)


def _init_parab(n, amp=0.45, center_low=False):
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = i / float(n - 1) - 0.5
        shape = 1.0 - 2.0 * t * t
        a = 1.0 - amp * shape if center_low else 1.0 + amp * shape
    else:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 1e-8)
    return _normalize_pos(a)


def _adam_run(theta, n, N, deadline, it_limit=2000000,
              tau_init=0.55, tau_min=0.035, lr_init=0.05,
              rng=None, tag=""):
    n = int(n)
    N = int(N)
    theta = np.asarray(theta, dtype=np.float64).copy()
    a = _softprob(theta)
    L = 2 * n - 1

    F, c = _conv_and_obj(a, n, N)
    bestF = float(F)
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    it = 0
    stall = 0
    resets = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    t0 = time.monotonic()
    duration = max(1e-9, deadline - t0)

    while it < it_limit:
        now = time.monotonic()
        if now >= deadline - 0.05:
            break
        rem = max(0.0, (deadline - now) / duration)

        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:L]
        F = 2.0 * float(n) * float(np.max(c))

        if F < bestF - 1e-12:
            bestF = F
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 10000 and rng is not None and resets < 6 and rem > 0.1:
            theta = _theta_from_prob(best_a) + rng.normal(0.0, 0.04, size=n)
            a = _softprob(theta)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            it = 0
            resets += 1
            stall = 0
            continue

        u = c * float(n)
        tau = max(tau_min, tau_min + (tau_init - tau_min) * (rem ** 0.7))
        p, _ = _smooth_p_from_u(u, tau)

        g = _grad_from_pa(p, a, A, N, n)
        gc = g - float(np.mean(g))
        gn = float(np.sqrt(np.mean(gc * gc)))

        if gn < 1e-14:
            if rng is not None and rem > 0.2 and resets < 6:
                theta = _theta_from_prob(best_a) + rng.normal(0.0, 0.06, size=n)
                a = _softprob(theta)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                it = 0
                resets += 1
                stall = 0
                continue
            break

        gd = gc / gn
        it += 1

        m = beta1 * m + (1.0 - beta1) * gd
        v = beta2 * v + (1.0 - beta2) * (gd * gd)
        mhat = m / (1.0 - beta1 ** it)
        vhat = v / (1.0 - beta2 ** it)

        lr = lr_init * (0.35 + 0.65 * rem)
        if stall > 4000:
            lr *= 0.7

        theta = theta - lr * mhat / (np.sqrt(vhat) + eps)
        theta = theta - float(np.mean(theta))
        theta = np.clip(theta, -30.0, 30.0)
        a = _softprob(theta)

        if it % 5000 == 0:
            print(f"  [{tag}] adam it={it} F={F:.6f} best={bestF:.6f} tau={tau:.4f}")

    return best_a, float(bestF)


def _polish_active(a_input, n, N, deadline, it_limit=20000,
                   base_lr=1e-3, active_frac=0.02, tau_soft=0.05,
                   rng=None, tag=""):
    n = int(n)
    N = int(N)
    L = 2 * n - 1
    a = _normalize_pos(a_input)
    theta = _theta_from_prob(a)

    F, c = _conv_and_obj(a, n, N)
    bestF = float(F)
    best_a = a.copy()

    lr = float(base_lr)
    it = 0
    no_accept = 0
    t0 = time.monotonic()
    duration = max(1e-9, deadline - t0)

    while it < it_limit:
        now = time.monotonic()
        if now >= deadline - 0.05:
            break
        rem = max(0.0, (deadline - now) / duration)

        frac = max(0.002, 0.01 + 0.015 * rem)
        T = max(1, min(L, int(L * frac)))

        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:L]
        F = 2.0 * float(n) * float(np.max(c))
        if F < bestF - 1e-13:
            bestF = F
            best_a = a.copy()

        u = c * float(n)

        if (it & 1) == 0:
            umax = float(np.max(u))
            ex = np.exp((u - umax) / tau_soft)
            s = float(ex.sum())
            if s > 0.0 and np.isfinite(s):
                p = ex / s
            else:
                p = np.full(L, 1.0 / L, dtype=np.float64)
        else:
            if T < L:
                idx = np.argpartition(-u, T)[:T]
            else:
                idx = np.arange(L)
            p = np.zeros(L, dtype=np.float64)
            p[idx] = 1.0 / float(len(idx))

        g = _grad_from_pa(p, a, A, N, n)
        gc = g - float(np.mean(g))
        scale = float(np.max(np.abs(gc)))

        if scale < 1e-10:
            if rng is not None and no_accept < 50:
                theta = _theta_from_prob(best_a) + rng.normal(0.0, 0.02, size=n)
                a = _softprob(theta)
                it += 1
                no_accept += 1
                continue
            break

        gdir = -gc / scale
        trial_lr = lr
        improved = False

        for _trial in range(6):
            theta_new = theta + trial_lr * gdir
            theta_new = theta_new - float(np.mean(theta_new))
            theta_new = np.clip(theta_new, -30.0, 30.0)
            a_new = _softprob(theta_new)
            F_new = _exact_obj_only(a_new, n, N)
            if F_new < bestF - 1e-13:
                theta = theta_new
                a = a_new
                bestF = F_new
                best_a = a_new.copy()
                lr = min(trial_lr * 1.2, 0.03)
                improved = True
                break
            trial_lr /= 2.5

        it += 1
        if improved:
            no_accept = 0
        else:
            lr = max(trial_lr / 1.1, 1e-8)
            no_accept += 1
            if no_accept > 80 or lr < 1e-8:
                break

        if it % 200 == 0:
            print(f"  [{tag}] polish it={it} best={bestF:.6f} lr={lr:.3e}")

    return best_a, float(bestF)


def _global_lowpass_perturb(theta, n, rng, sigma):
    n = int(n)
    z = rng.normal(0.0, 1.0, size=n)
    if n > 20:
        Z = np.fft.rfft(z)
        cut = max(1, n // 25)
        Z[cut:] = 0.0
        z = np.fft.irfft(Z, n)
    z = z - float(np.mean(z))
    sd = float(np.std(z))
    if sd > 1e-15:
        z = z / sd
    new_theta = theta + float(sigma) * z
    new_theta = new_theta - float(np.mean(new_theta))
    return np.clip(new_theta, -30.0, 30.0)


def _block_perturb_theta(theta, n, rng, sigma, width_frac=None):
    n = int(n)
    a = _softprob(theta)
    b = a.copy()
    if width_frac is None:
        w = max(2, int(n * 0.02))
    else:
        w = max(2, int(n * float(width_frac)))
    w = min(w, n)
    lo = max(1, n - w + 1)
    s = int(rng.integers(0, lo))
    b[s:s + w] *= np.exp(rng.normal(0.0, float(sigma), size=w))
    b = _normalize_pos(b)
    return _theta_from_prob(b)


def _antisym_add_perturb(a, n, N, rng, strength=0.15, num_active=3):
    n = int(n)
    a = np.asarray(a, dtype=np.float64)
    F, c = _conv_and_obj(a, n, N)
    L = 2 * n - 1
    kth = min(int(num_active), L) - 1
    if L > 1:
        idx = np.argpartition(-c, kth)[:kth + 1]
    else:
        idx = np.array([0])

    d = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        vs = max(0, k - n + 1)
        ve = min(n - 1, k)
        for j in range(vs, ve + 1):
            q = k - j
            if q < 0 or q >= n or q < j or j == q:
                continue
            amp = max(float(a[j]), float(a[q]))
            if amp > 1e-12:
                t = float(rng.normal(0.0, float(strength) * amp))
                d[j] += t
                d[q] -= t

    a_p = a + d
    a_p = np.where(a_p > 0.0, a_p, 0.0)
    a_p = _normalize_pos(a_p)
    return _theta_from_prob(a_p)

                theta_p = _theta_from_prob(a_alt)
                slice_len = min(60.0, max(15.0, 0.12 * remaining))
                tag = "alt%d" % attempt
                tau_init = 0.50
                tau_min = 0.04
                lr = 0.05
            else:
                sigma = 0.03 + 0.025 * min(attempt, 15)
                mode = int(rng.integers(3))
                if mode == 0:
                    theta_p = _global_lowpass_perturb(theta_best, n, rng, sigma)
                elif mode == 1:
                    wf = float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.10]))
                    theta_p = _block_perturb_theta(theta_best, n, rng, sigma, wf)
                else:
                    theta_p = _antisym_add_perturb(main_best_a, n, N, rng, strength=0.15, num_active=3)

                slice_len = min(45.0, max(12.0, 0.12 * remaining))
                tag = "r%d" % attempt
                tau_init = 0.45
                tau_min = 0.03
                lr = 0.05

        end = now + slice_len
        a1, F1 = _adam_run(theta_p, n, N, end, rng=rng,
                           tau_init=tau_init, tau_min=tau_min, lr_init=lr, tag=tag)
        candidates, F1 = _add_candidate(candidates, a1, n, N)
        if F1 < best_F:
            best_F = F1
            best_a = a1.copy()
        if F1 < main_best_F:
            main_best_F = F1
            main_best_a = a1.copy()
            theta_best = _theta_from_prob(main_best_a)

        e2 = min(time.monotonic() + min(10.0, 0.25 * slice_len), deadline - 25.0)
        if e2 > time.monotonic() + 1.5:
            a2, F2 = _polish_active(a1, n, N, e2, rng=rng, tag=tag + "p", base_lr=1e-4)
            candidates, F2 = _add_candidate(candidates, a2, n, N)
            if F2 < best_F:
                best_F = F2
                best_a = a2.copy()
            if F2 < main_best_F:
                main_best_F = F2
                main_best_a = a2.copy()
                theta_best = _theta_from_prob(main_best_a)

        if attempt % 10 == 0:
            print("Attempt %d: main_best=%.6f global_best=%.6f" % (attempt, main_best_F, best_F))

    # ---- final polish of two best candidates ----
    candidates.sort(key=_candidate_key)
    for item in candidates[:2]:
        now = time.monotonic()
        rem = deadline - now - 18.0
        if rem < 5.0:
            break
        a, ni, Ni = item[1], item[2], item[3]
        e = now + min(12.0, 0.5 * rem)
        a2, F2 = _polish_active(a, ni, Ni, e, rng=rng, base_lr=5e-5, tag="final")
        candidates, F2 = _add_candidate(candidates, a2, ni, Ni)
        if F2 < best_F:
            best_F = F2
            best_a = a2.copy()

    candidates.sort(key=_candidate_key)

    # ---- validate with exact evaluator and return best finite score ----
    best_eval = None
    best_sc = np.inf

    for item in candidates[:2]:
        now = time.monotonic()
        if now > deadline - 8.0:
            break
        a = item[1]
        res = [float(x) for x in a]
        try:
            sc = float(evaluate_sequence(res))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_eval = res
        print("Eval candidate internal=%.6f exact=%.6f" % (item[0], sc))

    if best_eval is None:
        best_eval = [1.0] * 10

    print("FINAL returned evaluated score=%.6f" % best_sc)
    return best_eval
