import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n_fft = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_fft) * fft(b, n_fft)).real[:n_out]

def _score(a):
    """Compute the score 2n*max_b/sum_a^2."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _softmax(x):
    """Stable softmax."""
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    return exp_x / np.sum(exp_x)

def _smooth_max_and_grad(a, tau):
    """
    Compute the smooth max approximation of conv(a,a) and its gradient w.r.t. a.
    smooth_max = (1/tau) * log(sum_k exp(tau * b_k))
    grad_i = sum_k w_k * a[k-i] (cross-correlation, first n elements)
    """
    n = len(a)
    b = _fft_conv(a, a)  # length 2n-1
    
    # Smooth max
    b_max = np.max(b)
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    smooth_val = b_max + np.log(sum_exp) / tau
    
    # Weights
    w = exp_shifted / sum_exp
    
    # Gradient via cross-correlation
    # grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    N = 1 << (len(w) + n - 1).bit_length()
    corr = ifft(fft(w, N) * np.conj(fft(a, N))).real[:n]
    
    return smooth_val, corr

def _obj_and_grad(x, tau):
    """Objective and gradient in softmax parameter space."""
    a = _softmax(x)
    smooth_val, grad_a = _smooth_max_and_grad(a, tau)
    
    # Chain rule through softmax
    # da_i/dx_j = a_i * (delta_ij - a_j)
    # d_obj/dx_j = sum_i grad_a[i] * a_i * (delta_ij - a_j)
    #            = a_j * (grad_a[j] - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return smooth_val, grad_x

            return (np.inf, np.zeros_like(x))
        val, grad = _obj_and_grad(x, tau)
        return val, grad
    
    result = minimize(fun, x0, method='L-BFGS-B', jac=True,
                      options={'maxiter': maxiter, 'maxfun': 100000})
    return result.x, result.fun

def _subgrad_refine(a, n_steps=10000, lr_init=0.01, lr_decay=0.9999):
    """Refine using exact subgradient descent (using argmax of conv)."""
    n = len(a)
    sum_a = np.sum(a)

        k_star = int(np.argmax(b))

        # i.e., g = the "mirror" of a at position k_star
        g = np.zeros(n)

```python
def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.full(n, S / n)
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cv - S) / rho > 0
    if not np.any(cond):
        return np.full(n, S / n)
    rho_star = np.max(np.where(cond)[0]) + 1
    theta = (cv[rho_star - 1] - S) / rho_star
    return np.maximum(v - theta, 0)

def _fista(a0, S, n_iter=10000, time_limit=60.0):
    """FISTA for minimizing max_conv(a) subject to a in simplex(S)."""
    n = len(a0)
    a = a0.copy()
    a_old = a0.copy()
    t = 1.0
    step = 1.0  # initial step size
    
    start_time = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Compute gradient of smooth max (or subgradient of exact max)
        b = _fft_conv(a_old, a_old)
        k_star = int(np.argmax(b))
        
        # Subgradient
        g = np.zeros(n)
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        g[k_lo:k_hi+1] = a_old[k_star - np.arange(k_lo, k_hi + 1)]
        
        # Fista update
        a_new = _project_simplex(a_old - step * g, S)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        a_old = a_new + ((t - 1) / t_new) * (a_new - a)
        a = a_new
        t = t_new
    
    return a
```

            for j in js:
                quad += a[j] * a[k - j]

            # Actually, the above already handles it correctly since
            # a[j] * a[k-j] when j = k-j is a[j]^2
            constraints.append(quad <= t)

    prob = cp.Problem(obj, constraints)
    # prob.solve(solver=cp.ECOS)
    # prob.solve(solver=cp.CLARABEL)

```python
"""
Minimize 2n*max(autocorr)/(sum)^2 using smooth gradient optimization
with softmax parameterization, L-BFGS-B, continuation method on temperature,
and exact subgradient refinement. Multiple restarts with diverse initializations
and multiple sequence lengths.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n_fft = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_fft) * fft(b, n_fft)).real[:n_out]


def _score(a):
    """Compute the score 2n*max_b/sum_a^2."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _softmax(x):
    """Stable softmax."""
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    return exp_x / np.sum(exp_x)


def _smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Uses log-sum-exp for numerical stability.
    """
    n = len(a)
    b = _fft_conv(a, a)  # length 2n-1
    
    # Smooth max via log-sum-exp
    b_max = float(np.max(b))
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    smooth_val = b_max + np.log(sum_exp) / tau
    
    # Weights for the soft-argmax
    w = exp_shifted / sum_exp
    
    # Gradient: grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the cross-correlation of w and a, first n elements
    N = 1 << (len(w) + n - 1).bit_length()
    corr = ifft(fft(w, N) * np.conj(fft(a, N))).real[:n]
    
    return smooth_val, corr


def _obj_and_grad(x, tau):
    """Objective and gradient in softmax parameter space."""
    a = _softmax(x)
    smooth_val, grad_a = _smooth_max_and_grad(a, tau)
    
    # Chain rule: d_obj/dx_j = a_j * (grad_a[j] - dot(grad_a, a))
    dot = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot)
    
    return smooth_val, grad_x


def _lgbfgs_optimize(x0, tau, maxiter=500, time_limit=30.0):
    """Run L-BFGS-B optimization."""
    start_time = time.monotonic()
    n = len(x0)
    
    def fun_grad(x):
        if time.monotonic() - start_time > time_limit:
            return (np.inf, np.zeros(n))
        val, grad = _obj_and_grad(x, tau)
        # Check for numerical issues
        if not np.isfinite(val):
            return (np.inf, np.zeros(n))
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        return val, grad
    
    result = minimize(fun_grad, x0, method='L-BFGS-B', jac=True,
                      options={'maxiter': maxiter, 'maxfun': 50000,
                               'ftol': 1e-12, 'gtol': 1e-8})
    return result.x


def _subgrad_step(a):
    """One exact subgradient step (with signal averaging for stability)."""
    n = len(a)
    b = _fft_conv(a, a)
    k_star = int(np.argmax(b))
    
    # Subgradient direction
    k_lo = max(0, k_star - n + 1)
    k_hi = min(n - 1, k_star)
    g = np.zeros(n)
    g[k_lo:k_hi + 1] = a[k_star - np.arange(k_lo, k_hi + 1)]
    
    return g, k_star, float(b[k_star])


def _subgrad_refine(a, n_steps=5000, lr_init=0.005, lr_decay=0.99995, 
                    use_signal_avg=False, avg_window=5):
    """
    Refine using exact subgradient descent.
    If use_signal_avg, average the subgradient over recent peak positions.
    """
    n = len(a)
    sum_a = float(np.sum(a))
    lr = lr_init
    
    # For signal averaging, keep a list of recent peaks
    recent_peaks = []
    
    for step in range(n_steps):
        if use_signal_avg:
            # Use a random lag to explore the landscape
            if step % 10 == 0:
                b = _fft_conv(a, a)
                # Find top few peaks
                top_k = min(3, len(b))
                top_indices = np.argsort(b)[-top_k:]
                recent_peaks = [int(i) for i in top_indices]
            
            if recent_peaks:
                k_star = recent_peaks[step % len(recent_peaks)]
            else:
                b = _fft_conv(a, a)
                k_star = int(np.argmax(b))
        else:
            b = _fft_conv(a, a)
            k_star = int(np.argmax(b))
        
        # Subgradient
        k_lo = max(0, k_star - n + 1)
        k_hi = min(n - 1, k_star)
        g = np.zeros(n)
        g[k_lo:k_hi + 1] = a[k_star - np.arange(k_lo, k_hi + 1)]
        
        # Project onto tangent space of sum(a) = const
        g = g - np.mean(g)
        
        # Update
        a = a - lr * g
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-15:
            a = np.full(n, sum_a / n)
        else:
            a = a * (sum_a / s)
        
        lr *= lr_decay
    
    return a


def _make_initialization(n, init_type, rng, parent=None):
    """Create an initialization array of length n."""
    k = np.arange(n)
    
    if init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'random':
        a = rng.random(n) + 0.1
        return a / np.sum(a)
    elif init_type == 'parent':
        if parent is not None and len(parent) == n:
            a = np.maximum(parent, 1e-10)
            return a / np.sum(a)
        elif parent is not None:
            # Interpolate to length n
            idx = np.linspace(0, len(parent) - 1, n)
            a = np.interp(idx, np.arange(len(parent)), np.maximum(parent, 0))
            a = np.maximum(a, 1e-10)
            return a / np.sum(a)
        else:
            return np.ones(n) / n
    elif init_type == 'bipolar':
        a = 1.0 + 0.5 * np.cos(2 * np.pi * k / n)
        return a / np.sum(a)
    elif init_type == 'triple_harmonic':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * k / n) + 0.2 * np.cos(4 * np.pi * k / n) - 0.2 * np.cos(6 * np.pi * k / n)
        return a / np.sum(a)
    elif init_type == 'windowed_noise':
        window = 0.5 * (1 + np.cos(2 * np.pi * k / n))
        a = window * (rng.random(n) + 0.5)
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif init_type == 'spectrum_shaping':
        # Create a sequence with a specific power spectrum
        spectrum = rng.random(n // 2 + 1)
        # Emphasize certain frequencies
        freqs = np.arange(n // 2 + 1)
        mask = (freqs % 3 != 0).astype(float)  # Suppress every 3rd frequency
        spectrum = spectrum * mask + 0.1
        # Build real signal via inverse FFT
        full_spec = np.zeros(n, dtype=complex)
        full_spec[:n // 2 + 1] = spectrum
        if n % 2 == 0:
            full_spec[n // 2 + 1:] = np.conj(full_spec[1:n // 2][::-1])
        else:
            full_spec[(n // 2 + 1):] = np.conj(full_spec[1:(n // 2 + 1)][::-1])
        a = np.abs(ifft(full_spec))
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    else:
        return np.ones(n) / n

def _fista_solve(a0, n_iter=20000, time_limit=60.0):
    """FISTA for minimizing max conv(a,a) on the simplex."""
    n = len(a0)
    S = float(np.sum(a0))
    a = a0.copy()
    a_tilde = a0.copy()
    t = 1.0
    step = 1.0 / (2.0 * np.max(a0) + 1e-6)  # Initial step size
    
    start_time = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break
        
        # Compute subgradient at a_tilde
        b = _fft_conv(a_tilde, a_tilde)
        k_star = int(np.argmax(b))
        
        g = np.zeros(n)
        if 0 <= k_star < 2*n-1:
            # g[i] = 2 * a[k_star - i] for valid i
            i_lo = max(0, k_star - n + 1)
            i_hi = min(n - 1, k_star)
            g[i_lo:i_hi+1] = 2.0 * a_tilde[k_star - np.arange(i_lo, i_hi + 1)]
        
        # FISTA update
        a_new = _project_simplex(a_tilde - step * g, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        a_tilde = a_new + ((t - 1.0) / t_new) * (a_new - a)
        # Clip to ensure feasibility
        a_tilde = _project_simplex(a_tilde, S)
        a = a_new
        t = t_new
    
    return a
```

- Softmax parameterization (a_i = exp(x_i)/sum(exp(x_j)))
- L-BFGS-B on a log-sum-exp smooth approximation of the max convolution value

- Exact subgradient refinement with adaptive learning rate
- Multiple restarts with diverse initializations and multiple sequence lengths

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n_fft = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_fft) * fft(b, n_fft)).real[:n_out]


def _score(a):
    """Compute the score 2n*max_b/sum_a^2."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _softmax(x):
    """Stable softmax."""
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    return exp_x / np.sum(exp_x)


def _smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    smooth_max = (1/tau) * log(sum_k exp(tau * b_k))
    grad[i] = sum_{j} w[i+j] * a[j]  (cross-correlation)
    """
    n = len(a)
    b = _fft_conv(a, a)  # length 2n-1

    b_max = float(np.max(b))
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    smooth_val = b_max + np.log(sum_exp) / tau

    w = exp_shifted / sum_exp

    # Gradient via cross-correlation
    N = 1 << (len(w) + n - 1).bit_length()
    corr = ifft(fft(w, N) * np.conj(fft(a, N))).real[:n]

    return smooth_val, corr


def _obj_and_grad(x, tau):
    """Objective and gradient in softmax parameter space."""
    a = _softmax(x)
    smooth_val, grad_a = _smooth_max_and_grad(a, tau)
    dot = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot)
    return smooth_val, grad_x


def _lgbfgs(x0, tau, maxiter=500, time_limit=30.0):
    """Run L-BFGS-B optimization with time limit."""
    start_time = time.monotonic()
    n = len(x0)
    aborted = [False]

    def fun_grad(x):
        if aborted[0] or time.monotonic() - start_time > time_limit:
            aborted[0] = True
            return (np.inf, np.zeros(n))
        val, grad = _obj_and_grad(x, tau)
        if not np.isfinite(val):
            return (np.inf, np.zeros(n))
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        return val, grad

    result = minimize(fun_grad, x0, method='L-BFGS-B', jac=True,
                      options={'maxiter': maxiter, 'maxfun': 50000,
                               'ftol': 1e-14, 'gtol': 1e-10})
    return result.x
