"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses adaptive soft-max gradient with Adam optimizer, spectral factorization
initialization, and multi-length search strategy.
"""
import numpy as np
import time
import math


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj_nonneg_sum(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a.fill(1.0 / max(1, a.size))
    else:
        a /= s
    return a


def score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _convolve_self(a, m):
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * len(a) - 1]
    b = np.maximum(b, 0.0)
    return b, F


def _interp_to(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s <= 0.0:
        return np.ones(int(max(1, n)), dtype=np.float64) / float(max(1, n))
    if a.size == int(n):
        return a / s
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, int(n))
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 0.0:
        out.fill(1.0 / n)
    else:
        out /= ss
    return out


def _spectral_factor_init(target, a_start, n, m):
    """Compute spectral factor of target autocorrelation profile."""
    L = 2 * n - 1
    target = np.asarray(target, dtype=np.float64)
    if target.size < L:
        padded = np.zeros(L, dtype=np.float64)
        padded[:target.size] = target
        target = padded
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    tmax = float(target.max())
    if tmax > 0.0 and np.isfinite(tmax):
        target = target / tmax
    else:
        target = np.ones(L, dtype=np.float64)

    T = np.fft.fft(target[:min(L, m)], m)
    # Phase-unwrapped square root
    T_abs = np.abs(T)
    T_phase = np.angle(T)
    sqrt_T = np.sqrt(T_abs) * np.exp(0.5j * T_phase)
    a = np.fft.ifft(sqrt_T).real[:n]
    a = np.maximum(a, 0.0)
    amax = float(a.max())
    if amax <= 1e-300:
        a = a_start.copy() if a_start is not None else np.ones(n) / n
    else:
        a = a / amax
    if a_start is not None:
        a = 0.5 * a + 0.5 * a_start
    a = np.maximum(a, 1e-12)
    return _proj_nonneg_sum(a)


def _adam_optimize(a_init, n, budget, seed, tau_scale=0.08):
    """Adam optimizer with adaptive soft-max temperature."""
    n = int(max(4, n))
    rng = np.random.default_rng(seed)
    m = _next_pow2(2 * n - 1)
    m_grad = _next_pow2(3 * n)  # For cross-correlation

    a = _proj_nonneg_sum(a_init)
    # Parameterize: v = log(a) with centering
    v = np.log(np.maximum(a, 1e-20))
    v = v - float(np.mean(v))
    v = np.clip(v, -20.0, 20.0)

    def _soft(v):
        v_shift = v - float(np.max(v))
        v_shift = np.clip(v_shift, -50.0, 50.0)
        e = np.exp(v_shift)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            return np.full(v.shape, 1.0 / v.size, dtype=np.float64)
        return e / s

    best_a = a.copy()
    best_score = score_fft(a)
    if not np.isfinite(best_score):
        best_a = np.ones(n) / n
        v = np.zeros(n)
        best_score = 2.0

    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t_step = 0

    start = time.monotonic()
    deadline = start + max(1.0, budget)
    last_improve = start
    best_tau = bmax_init = None

    # Initial bmax estimate
    a_cur = _soft(v)
    b_init, _ = _convolve_self(a_cur, m)
    bmax_init = float(b_init.max())

    while time.monotonic() < deadline:
        elapsed = max(1e-9, time.monotonic() - start)
        progress = min(1.0, elapsed / max(1.0, budget))

        # Stuck detection
        stuck_limit = max(15.0, min(60.0, 0.15 * budget))
        if time.monotonic() - last_improve > stuck_limit:
            # Re-randomize with shape preservation
            sigma = 0.15 + 0.4 * (1.0 - progress)
            v = v + rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -20.0, 20.0)
            # Also try a different tau
            last_improve = time.monotonic()
            continue

        a = _soft(v)
        b, F = _convolve_self(a, m)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 1e-300:
            break

        # Adaptive temperature
        # Target: effective number of active lags ~ sqrt(n)
        # K_eff = 1 / sum(w^2) for normalized w
        tau = bmax * tau_scale * (1.0 + 1.5 * (1.0 - progress))
        if rng.random() < 0.3:
            tau *= rng.uniform(0.5, 2.0)
        tau = max(tau, bmax * 0.01)

        # Soft-max weights
        x = (b - bmax) / tau
        x = np.clip(x, -50.0, 50.0)
        x = x - float(x.max())
        w = np.exp(x)
        ws = float(w.sum())
        if ws <= 0.0 or not np.isfinite(ws):
            break
        w = w / ws

        # Top-k emphasis
        blen = len(b)
        top_k = max(1, int(0.08 * blen))
        if top_k < blen:
            idx = np.argpartition(b, -top_k)[-top_k:]
            w[idx] += 0.4
            ws = float(w.sum())
            w = w / ws

        # Gradient: 2 * cross_corr(w, a)
        Fw = np.fft.fft(w, m_grad)
        Fa = np.fft.fft(a, m_grad)
        grad = np.fft.ifft(Fw * np.conj(Fa)).real[:n]
        grad *= 2.0

        # Normalize gradient by its RMS for stable Adam
        grms = float(np.sqrt(np.mean(grad * grad)))
        if not np.isfinite(grms) or grms < 1e-14:
            sigma = 0.1 + 0.3 * (1.0 - progress)
            v = v + rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -20.0, 20.0)
            continue
        grad = grad / grms

        # Adam update
        t_step += 1
        mu = beta1 * mu + (1.0 - beta1) * grad
        nu = beta2 * nu + (1.0 - beta2) * (grad * grad)
        b1 = beta1 ** t_step
        b2 = beta2 ** t_step
        mhat = mu / max(1e-12, 1.0 - b1)
        vhat = nu / max(1e-12, 1.0 - b2)

        lr = 0.15 * (1.0 - 0.5 * progress)
        if progress > 0.8:
            lr *= 0.3
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.6, 0.6)
        v = v - lr * step
        v = v - float(np.mean(v))
        v = np.clip(v, -20.0, 20.0)

        # Periodic score check
        check_every = max(3, min(30, 600 // max(1, n // 256)))
        if t_step % check_every == 0:
            a_check = _soft(v)
            sc = score_fft(a_check)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best_score = sc
                best_a = a_check.copy()
                last_improve = time.monotonic()

    # Final check
    a_final = _soft(v)
    sc = score_fft(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final

    return _proj_nonneg_sum(best_a), float(best_score)


def _spectral_refine(a, n, m, iters=40, blend=0.5):
    """Iterative spectral refinement: alternate between time and frequency domain."""
    L = 2 * n - 1
    a = np.asarray(a, dtype=np.float64).copy()
    a = _proj_nonneg_sum(a)
    best_a = a.copy()
    best_sc = score_fft(a)

    for it in range(iters):
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:L]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if bmax <= 0:
            break

        # Target: flatten the top of b
        # Create a smoothed target
        target = b.copy()
        target[target > 0.9 * bmax] = 0.9 * bmax
        target = np.maximum(target, 0.0)
        tmax = float(target.max())
        if tmax > 0:
            target = target / tmax * bmax

        # Spectral update: modify F to better match target
        T = np.fft.fft(target[:min(L, m)], m)
        F_abs = np.abs(F)
        T_abs = np.abs(T)
        # Blend the magnitudes
        blend_it = blend * (1.0 - 0.5 * it / max(1, iters))
        F_new_abs = (1.0 - blend_it) * F_abs + blend_it * T_abs
        F_new_abs = np.maximum(F_new_abs, 1e-10)
        F_new = F_new_abs * np.exp(1j * np.angle(F))

        anew = np.fft.ifft(F_new).real[:n]
        pos = np.maximum(anew, 0.0)
        psum = float(pos.sum())
        if psum < 1e-6:
            pos = a.copy()
        else:
            pos = pos / psum

        a = (1.0 - 0.6) * a + 0.6 * pos
        a = np.maximum(a, 1e-12)
        a = _proj_nonneg_sum(a)

        sc = score_fft(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return _proj_nonneg_sum(best_a), float(best_sc)


def _generate_initializations(n, rng, prev_best):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    # From previous best
    if prev_best is not None:
        inits.append(_proj_nonneg_sum(prev_best))

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponentials
    for c in (0.5, 1.0, 2.0, 4.0):
        inits.append(_proj_nonneg_sum(np.exp(-c * t)))

    # Inverse sqrt edges
    left = (i + 1.0)
    right = (n - i)
    inits.append(_proj_nonneg_sum(1.0 / np.sqrt(left * right + 1.0)))

    # Cosine window variants
    for alpha in (0.5, 1.0, 2.0):
        a = np.cos(np.pi * (t - 0.5) * alpha) ** 2
        inits.append(_proj_nonneg_sum(a))

    # Random Fourier
    kc = min(20, n)
    a = 1.0
    for k in range(1, kc + 1):
        amp = 0.3 / (float(k) ** 0.7)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t + ph)
    inits.append(_proj_nonneg_sum(a))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found within budget."""
    seed = int(seed)
    budget_s = float(max(120.0, budget_s))
    start = time.monotonic()
    global_deadline = start + max(120.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    # Load previous best
    prev = None
    try:
        cand = height_sequence_1
        if isinstance(cand, list):
            arr = np.asarray(cand, dtype=np.float64)
        else:
            arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0 and np.isfinite(arr).all():
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 0.0:
                prev = arr / float(arr.sum())
    except Exception:
        prev = None

    best_a = prev
    best_score = score_fft(prev) if prev is not None else np.inf
    print("Initial score: %.6f (n=%d)" % (best_score, len(prev) if prev is not None else 0))

    # Stage definitions: (length, time_fraction)
    stage_defs = [
        (1024, 0.04),
        (2048, 0.32),
        (4096, 0.40),
        (8192, 0.20),
    ]

    base_time = max(120.0, budget_s - 30.0)

    for n, frac in stage_defs:
        now = time.monotonic()
        if now + 60.0 >= global_deadline:
            print("Time running out, stopping at n=%d" % n)
            break

        stage_time = base_time * frac
        stage_deadline = min(global_deadline - 10.0, now + stage_time)
        if stage_deadline <= now:
            continue

        print("\n=== Stage n=%d, budget=%.0fs, current_best=%.6f ===" % (n, stage_time, best_score))

        m = _next_pow2(2 * n - 1)

        # Generate initializations
        prev_interp = _interp_to(best_a, n) if best_a is not None else np.ones(n) / n
        inits = _generate_initializations(n, rng, prev_interp)

        # Add spectral factor inits
        L = 2 * n - 1
        tL = np.linspace(0.0, 1.0, L)
        edge = np.abs(tL - 0.5) * 2.0
        for eb in (0.0, 0.5, 1.0, 2.0):
            target = 1.0 + eb * (edge ** 1.5)
            target = np.maximum(target, 0.1)
            sf = _spectral_factor_init(target, prev_interp, n, m)
            inits.append(sf)

        # Score and select top initializations
        scored = []
        for init in inits:
            init = _proj_nonneg_sum(init)
            sc = score_fft(init)
            if np.isfinite(sc):
                scored.append((sc, init))

        scored.sort(key=lambda x: x[0])
        print("  Top init: %s" % ("%.6f" % scored[0][0] if scored else "N/A"))

        # Select top candidates (diverse)
        cands = []
        seen = []
        for sc, a in scored:
            diverse = True
            for b in seen:
                d = float(np.sqrt(np.sum((a - b) ** 2)))
                if d < 0.03:
                    diverse = False
                    break
            if diverse:
                cands.append(a)
                seen.append(a)
                if len(cands) >= 5:
                    break

        if not cands:
            cands.append(np.ones(n) / n)

        # Run spectral refinement on best init
        if cands and len(cands) > 0:
            sr_deadline = min(stage_deadline, now + max(5.0, stage_time * 0.15))
            a_sr, sc_sr = _spectral_refine(cands[0], n, m, iters=50, blend=0.4)
            if np.isfinite(sc_sr) and sc_sr < best_score:
                best_score = sc_sr
                best_a = a_sr
                print("  Spectral refine: %.6f" % sc_sr)
            cands.insert(0, a_sr)

        # Run Adam optimization on top candidates
        remaining = stage_deadline - time.monotonic()
        n_cands = min(len(cands), max(1, int(remaining / max(10.0, stage_time / 5))))
        weights = [0.4, 0.25, 0.15, 0.1, 0.1][:n_cands]
        wsum = sum(weights)
        weights = [w / wsum for w in weights]

        for ci in range(n_cands):
            now2 = time.monotonic()
            if now2 >= stage_deadline - 5.0:
                break
            alloc = max(10.0, (stage_deadline - now2 - 5.0) * weights[ci] * 1.5)
            if len(cands) > ci + 1:
                alloc = max(10.0, alloc, (stage_deadline - now2) / (n_cands - ci))
            alloc = min(alloc, stage_deadline - now2 - 5.0)
            if alloc < 5.0:
                break

            a_opt, sc_opt = _adam_optimize(cands[ci], n, alloc, int(rng.integers(0, 2**31 - 1)))
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt
                print("    Adam cand %d: %.6f (improved!)" % (ci, sc_opt))
            else:
                print("    Adam cand %d: %.6f" % (ci, sc_opt))

            # Also try spectral refine on the Adam result
            if np.isfinite(sc_opt):
                rem2 = stage_deadline - time.monotonic()
                if rem2 > 10.0:
                    a_sr2, sc_sr2 = _spectral_refine(a_opt, n, m, iters=30, blend=0.3)
                    if np.isfinite(sc_sr2) and sc_sr2 < best_score:
                        best_score = sc_sr2
                        best_a = a_sr2
                        print("    Spectral refine after Adam: %.6f (improved!)" % sc_sr2)

        print("  Stage n=%d done. Best=%.6f" % (n, best_score))

    # Final polish on the best sequence found
    now = time.monotonic()
    rem = global_deadline - now
    if best_a is not None and rem > 30.0:
        n_best = len(best_a)
        m_best = _next_pow2(2 * n_best - 1)
        print("\n=== Final polish (n=%d, %.0fs remaining) ===" % (n_best, rem))
        a_pol, sc_pol = _adam_optimize(best_a, n_best, rem * 0.6, int(rng.integers(0, 2**31 - 1)))
        if np.isfinite(sc_pol) and sc_pol < best_score:
            best_score = sc_pol
            best_a = a_pol
            print("  Polish: %.6f" % sc_pol)

        rem2 = global_deadline - time.monotonic()
        if rem2 > 15.0:
            a_pol2, sc_pol2 = _spectral_refine(best_a, n_best, m_best, iters=int(rem2 * 5), blend=0.3)
            if np.isfinite(sc_pol2) and sc_pol2 < best_score:
                best_score = sc_pol2
                best_a = a_pol2
                print("  Spectral polish: %.6f" % sc_pol2)

    # Final validation
    if best_a is None:
        best_a = np.ones(512) / 512
    best_a = _proj_nonneg_sum(best_a)

    # Ensure values are in valid range for evaluate_sequence
    best_a = np.clip(best_a, 0.0, 1000.0)
    if float(best_a.sum()) < 0.01:
        best_a = np.ones(512) / 512

    final_score = score_fft(best_a)
    print("\n=== FINAL: score=%.6f, n=%d ===" % (final_score, len(best_a)))

    # Try exact evaluation if sequence is small enough
    try:
        if len(best_a) <= 5000:
            exact = evaluate_sequence([float(x) for x in best_a])
            print("Exact evaluation: %.6f" % exact)
            if not np.isfinite(exact):
                print("WARNING: exact eval failed, using fallback")
                best_a = np.ones(512) / 512
    except Exception as e:
        print("Exact eval skipped: %s" % e)

    return [float(x) for x in best_a]
