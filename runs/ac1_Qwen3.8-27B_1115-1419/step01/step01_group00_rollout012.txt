def _build_lp_matrix(a, rows, n):
    if rows.size==0: return None
    starts = np.maximum(0, rows - (n - 1))
    ends = np.minimum(n - 1, rows) + 1
    lengths = (ends - starts).astype(np.int64)
    total = int(np.sum(lengths))
    if total <=0: return None
    row_reps = np.repeat(np.arange(rows.size, dtype=np.int64), lengths)
    local = np.arange(total, dtype=np.int64) - np.repeat(starts, lengths)
    cols = starts + local
    kvals = rows[row_reps]
    vals = a[kvals - cols]
    rows2 = row_reps
    cols_tau = np.full(len(rows2), n, dtype=np.int64)
    vals_tau = np.full(len(rows2), -1.0, dtype=np.float64)
    row_all = np.concatenate((rows2, rows2))
    col_all = np.concatenate((cols, cols_tau))
    val_all = np.concatenate((vals, vals_tau))
    if hasattr(sparse,'coo_array'):
       A = sparse.coo_array((val_all, (row_all, col_all)), shape=(rows.size, n+1))
    else: A=sparse.coo_matrix(...)
    A.sum_duplicates()
    return A.tocsr()

"""
Search for non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The optimizer combines:
  * FFT-based exact self-convolution evaluation,
  * entropy/smoothed-max multiplicative (mirror-descent) gradient steps,
  * active-lag high-sharpness descent,
  * an LP/surrogate game direction for fixed a:
        min_b  max_k (a*b)_k  subject to b >= 0, sum(b)=1,
    implemented partially by multiplicative weights and partly by
    sparse cutting-plane LP calls when scipy is available,
  * random smooth/block perturbations and length sweeps.
"""

import numpy as np
import time
import math

np.seterr(over="ignore", under="ignore", divide="ignore", invalid="ignore")

try:
    from scipy.optimize import linprog
    _HAS_SCIPY_OPT = True
except Exception:

    _HAS_SCIPY_OPT = False

def _norm1(x):
    """Normalize a 1-D array to non-negative values with sum 1."""
    x = np.atleast_1d(np.asarray(x, dtype=np.float64))
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if not math.isfinite(s) or s <= 1e-300:
        if x.size == 0:
            return np.ones(1, dtype=np.float64)
        return np.full(x.shape, 1.0 / x.size, dtype=np.float64)
    return x / s


def _get_parent():
    """Read the optional global height_sequence_1 safely."""
    try:
        h = height_sequence_1
    except Exception:
        return None
    try:
        p = np.asarray(h, dtype=np.float64)
        if p.ndim != 1 or p.size == 0:
            return None
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if float(np.sum(p)) <= 0.0:
            return None
        return _norm1(p)
    except Exception:
        return None


def _fft_len(n):
    """FFT length large enough for linear convolution/correlation of
    two length-n sequences and the required middle correlation slice."""
    m = 2 * int(n)
    p = 1
    while p < m:
        p <<= 1
    return max(4, p)


def _max_conv(a, n, L):
    """max_k (a*a)_k for length-n sequence a, using FFT."""
    a = np.asarray(a, dtype=np.float64)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, n=L)[:2 * n - 1]
    return float(np.max(c))


def _conv_self(a, n, L):
    """Full nonnegative self-convolution, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, n=L)[:2 * n - 1].astype(np.float64, copy=False)
    np.maximum(c, 0.0, out=c)
    return c


def _conv_cross(a, b, n, L):
    """Full cross-convolution a*b, length 2n-1, clipped nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    Fa = np.fft.rfft(a, L)
    Fb = np.fft.rfft(b, L)
    c = np.fft.irfft(Fa * Fb, n=L)[:2 * n - 1].astype(np.float64, copy=False)
    np.maximum(c, 0.0, out=c)
    return c


def _softmax_c(c, gamma):
    """
    Softmax-like probability on nonnegative array c.
    gamma <= 0 gives uniform; larger gamma sharpens toward the maximum.
    """
    c = np.asarray(c, dtype=np.float64)
    c = np.maximum(c, 0.0)
    N = c.size
    if N == 0:
        return np.array([], dtype=np.float64)
    mx = float(np.max(c))
    if not math.isfinite(mx) or mx <= 1e-300 or gamma <= 1e-12:
        return np.full(N, 1.0 / N, dtype=np.float64)
    temp = mx / float(gamma)
    z = (c - mx) / temp
    p = np.exp(np.clip(z, -700.0, 0.0))
    s = float(np.sum(p))
    if s <= 1e-300:
        p = np.zeros(N, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= s
    return p


def _smooth_noise(n, rng, sigma):
    """Smooth, zero-mean Gaussian-like noise of standard deviation sigma."""
    if sigma == 0.0:
        return np.zeros(n, dtype=np.float64)
    if rng is None:
        rng = np.random.default_rng(0)
    w = rng.normal(0.0, 1.0, size=n).astype(np.float64)
    k = np.ones(5, dtype=np.float64) / 5.0
    w = np.convolve(w, k, mode="same")
    w = np.convolve(w, k, mode="same")
    m = float(np.mean(w))
    s = float(np.std(w))
    if s > 1e-12:
        w = (w - m) / s * sigma
    else:
        w = np.zeros(n, dtype=np.float64)
    return w


def _smooth_base(n, mode, rng):
    """Deterministic/random smooth positive initial patterns."""
    i = np.arange(n, dtype=np.float64)
    center = 0.5 * (float(n) - 1.0)

    if mode == 0:
        a = np.ones(n, dtype=np.float64)
    elif mode == 1:
        a = (
            1.0
            + 0.4 * np.cos(2.0 * np.pi * i / n)
            + 0.25 * np.cos(4.0 * np.pi * i / n + 0.6)
        )
    elif mode == 2:
        ph1 = 0.0 if rng is None else float(rng.normal(0.0, 1.0))
        ph2 = 0.0 if rng is None else float(rng.normal(0.0, 1.0))
        a = (
            1.3
            + 0.6 * np.cos(2.0 * np.pi * i / n + ph1)
            + 0.45 * np.cos(5.0 * np.pi * i / n + ph2)
        )
    elif mode == 3:
        # Center-peaked.
        a = (i + 1.0) * (float(n) - i) + 1.0
    elif mode == 4:
        # Edge-peaked.
        x = (i + 0.5) * (max(n - 1, 1) - i + 0.5) + 1.0
        a = 1.0 / np.sqrt(x) + 0.1
    else:
        a = 1.0 + _smooth_noise(n, rng, 0.5)

    a = np.maximum(a, 0.05)
    return _norm1(a)


def _resample_parent(parent, n):
    """Resample a normalized parent sequence to length n."""
    p = _norm1(parent)
    m = p.size
    if m == n:
        return p.copy()
    if m <= 1:
        return np.full(n, 1.0 / n, dtype=np.float64)
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    q = np.interp(new, old, p)
    return _norm1(q)


def _initial_sequence(parent, n, rng):
    """Build a starting sequence for a given length."""
    if parent is not None and parent.size > 0:
        if parent.size == n:
            a = parent.copy()
        else:
            a = _resample_parent(parent, n)
            noise = _smooth_noise(n, rng, 0.07)
            a = np.maximum(a * (1.0 + noise), 1e-20)
        a = _norm1(a)
        # If symmetrization is locally better, use it.
        try:
            L = _fft_len(n)
            a_sym = _norm1(0.5 * (a + a[::-1]))
            if _max_conv(a_sym, n, L) < _max_conv(a, n, L) - 1e-12:
                a = a_sym
        except Exception:
            pass
        return a
    mode = int(rng.integers(0, 5))
    return _smooth_base(n, mode, rng)


def _perturb(a, n, rng, sigma, mode):
    """Random smooth/block/mixing perturbation of a positive sequence."""
    a = _norm1(a)
    m = int(mode)
    if m % 3 == 0:
        noise = _smooth_noise(n, rng, sigma)
        b = np.maximum(a * (1.0 + noise), 1e-20)
    elif m % 3 == 1:
        width = max(2, n // 40)
        width = min(width, max(2, n // 2))
        s = int(rng.integers(0, max(1, n - width + 1)))
        b = a.copy()
        b[s:s + width] *= np.exp(rng.normal(0.0, sigma, size=width))
        b = np.maximum(b, 1e-20)
    else:
        pat = _smooth_base(n, int(rng.integers(0, 5)), rng)
        w = min(0.6, max(0.05, 0.5 * sigma))
        b = (1.0 - w) * a + w * pat
    return _norm1(b)


def _mix_candidates(a, b, n, L, alphas, cur_mc=None):
    """
    Try candidates (1-alpha)*a + alpha*b and return the best seen.
    """
    a = _norm1(a)
    b = _norm1(b)
    if cur_mc is None:
        cur_mc = _max_conv(a, n, L)
    best_a = a.copy()
    best_mc = float(cur_mc)

    if not np.all(np.isfinite(b)) or float(np.max(np.abs(a - b))) < 1e-7:
        return best_a, best_mc

    for alpha in alphas:
        alpha = float(alpha)
        if not math.isfinite(alpha) or alpha <= 0.0:
            continue
        if alpha >= 1.0:
            cand = b.copy()
        else:
            cand = (1.0 - alpha) * a + alpha * b
        cand = _norm1(cand)
        mc = _max_conv(cand, n, L)
        if math.isfinite(mc) and mc < best_mc - 1e-12:
            best_mc = mc
            best_a = cand.copy()
    return best_a, best_mc


def _gradient_descent(
    a,
    n,
    L,
    deadline,
    tag,
    gamma=1.5,
    eta=2.0,
    gamma_max=80.0,
    gamma_inc=1.0015,
    eta_min=1e-4,
    eta_max=20.0,
    do_print=True,
):
    """
    Multiplicative smooting-descent on the LSE/smoothed max of conv(a,a).
    Maintains a >= 0, sum(a)=1.
    """
    a = _norm1(np.asarray(a, dtype=np.float64).copy())
    best_a = a.copy()
    cur_mc = _max_conv(a, n, L)
    best_mc = cur_mc

    eta = float(min(max(eta, eta_min), eta_max))
    gamma = float(max(gamma, 1e-6))
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, n=L)[:2 * n - 1].astype(np.float64, copy=False)
        np.maximum(c, 0.0, out=c)
        cur_mc = float(np.max(c))

        if cur_mc < best_mc - 1e-12:
            best_mc = cur_mc
            best_a = a.copy()
            stall = 0
            if do_print and it % 2000 == 0:
                print(
                    f"  [{tag}] it={it} J={2.0 * n * cur_mc:.6f} "
                    f"gamma={gamma:.2f} eta={eta:.3f}"
                )

        p = _softmax_c(c, gamma)
        Fp = np.fft.rfft(p, L)

        # Gradient of F(a) wrt a_i is 2 * sum_j a_j p_{i+j}.
        g = 2.0 * np.fft.irfft(Fp * np.conj(F), n=L)[:n].real
        g = np.maximum(g, 0.0)

        h = a * g
        h = np.maximum(h, 0.0)
        h = h - float(np.min(h))
        sc = float(np.max(h))
        if sc < 1e-14:
            break
        h /= sc

        trial_etas = [eta]
        if eta > eta_min * 2.0:
            trial_etas.append(0.5 * eta)
        if eta > 1.0:
            trial_etas.append(2.0 * eta)

        best_trial = None
        best_trial_mc = cur_mc

        for te in trial_etas:
            if not math.isfinite(te) or te <= eta_min * 0.25:
                continue
            cand = a * np.exp(-te * h)
            cand = np.maximum(cand, 1e-14)
            cand = _norm1(cand)
            mc2 = _max_conv(cand, n, L)
            if math.isfinite(mc2) and mc2 < best_trial_mc - 1e-12:
                best_trial_mc = mc2
                best_trial = cand

        if best_trial is not None:
            a = best_trial
            cur_mc = best_trial_mc
            if cur_mc < best_mc - 1e-12:
                best_mc = cur_mc
                best_a = a.copy()
            eta = min(eta * 1.01, eta_max)
        else:
            eta = max(eta * 0.6, eta_min)
            stall += 1
            if stall > 1000:
                eta = max(eta * 0.5, eta_min)
                gamma = min(gamma * 1.6, gamma_max)
                stall = 0

        gamma = min(gamma * gamma_inc, gamma_max)

        if eta <= eta_min * 1.2 and best_trial is None:
            break

    if cur_mc < best_mc:
        best_a = a.copy()
        best_mc = cur_mc
    return best_a, best_mc


def _game_direction(a, n, L, deadline, gamma_r=8.0, gamma_c=5.0, a_weight=0.35):
    """
    Approximate the LP direction
        min_b max_k (a*b)_k, b>=0, sum b=1
    by an alternating multiplicative-weights / game iteration.
    """
    a = _norm1(np.asarray(a, dtype=np.float64).copy())
    aw = min(1.0, max(0.0, float(a_weight)))

    Ff = np.fft.rfft(a, L)
    c0 = _conv_self(a, n, L)
    b = a.copy()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        Fb = np.fft.rfft(b, L)
        cross = np.fft.irfft(Ff * Fb, n=L)[:2 * n - 1].astype(np.float64, copy=False)
        np.maximum(cross, 0.0, out=cross)
        if cross.size == 0:
            break

        gr = min(gamma_r * (1.0 + 0.015 * it), 60.0)
        gc = min(gamma_c * (1.0 + 0.015 * it), 25.0)

        r = _softmax_c(cross, gr)
        Fr = np.fft.rfft(r, L)
        # column payoff h_j = sum_i a_i r_{i+j}
        h = np.fft.irfft(Fr * np.conj(Ff), n=L)[:n].real
        h = np.maximum(h, 0.0)
        hmax = float(np.max(h))
        hmin = float(np.min(h))
        if hmax - hmin < 1e-14:
            break

        t = (h - hmin) / (hmax - hmin)
        low_payoff = np.exp(-gc * t)
        low_payoff = _norm1(low_payoff)
        b_new = _norm1(aw * a + (1.0 - aw) * low_payoff)

        if not np.all(np.isfinite(b_new)):
            break

        if it > 20 and float(np.max(np.abs(b_new - a))) < 1e-7:
            break
        b = b_new

    return _norm1(b)


def _build_lp_matrix(a, rows, n):
    """
    Build A for constraints
        sum_j a_{k-j} b_j - tau <= 0
    for selected lags rows. Variables are [b_0..b_{n-1}, tau].
    """
    rows = np.asarray(rows, dtype=np.int64)
    R = rows.size
    if R == 0:
        return None

    starts = np.maximum(0, rows - (n - 1)).astype(np.int64)
    ends = np.minimum(n - 1, rows) + 1
    lengths = (ends - starts).astype(np.int64)
    total = int(np.sum(lengths))
    if total <= 0:
        return None

    offsets = np.arange(total, dtype=np.int64)
    cum = np.concatenate((np.array([0], dtype=np.int64), np.cumsum(lengths)[:-1]))
    block_start = np.repeat(cum, lengths)
    local = offsets - block_start

    start_rep = np.repeat(starts, lengths)
    row_k = np.repeat(rows, lengths)
    row_idx = np.repeat(np.arange(R, dtype=np.int64), lengths)

    cols = start_rep + local
    coeff_idx = row_k - cols
    vals = a[coeff_idx]

    row_all = np.concatenate((row_idx, row_idx))
    col_all = np.concatenate((cols, np.full(row_idx.size, n, dtype=np.int64)))
    val_all = np.concatenate((vals, np.full(row_idx.size, -1.0, dtype=np.float64)))

    try:
        if hasattr(_sparse, "coo_array"):
            A = _sparse.coo_array((val_all, (row_all, col_all)), shape=(R, n + 1))
        else:
            A = _sparse.coo_matrix((val_all, (row_all, col_all)), shape=(R, n + 1))
        A.sum_duplicates()
        return A.tocsr()
    except Exception:
        return None


def _call_linprog(cvec, A, b_ub, Aeq, b_eq, bounds, tlimit):
    """Call scipy linprog with graceful fallbacks."""
    if linprog is None:
        return None
    try:
        return linprog(
            cvec,
            A_ub=A,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": max(1.0, float(tlimit))},
        )
    except Exception:
        pass
    try:
        return linprog(
            cvec,
            A_ub=A,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        pass
    try:
        return linprog(
            cvec,
            A_ub=A,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
        )
    except Exception:
        return None


def _lp_direction(a, n, L, remaining, max_rows=300):
    """
    Sparse cutting-plane LP for
        min_tau  tau
        s.t.    (a*b)_k <= tau for selected k,
                b >= 0, sum(b)=1.
    Returns b direction (normalized nonnegative) or None.
    """
    if not (_HAS_SCIPY_OPT and _HAS_SPARSE and linprog is not None):
        return None
    if remaining <= 5.0:
        return None

    a = _norm1(a)
    F_a = np.fft.rfft(a, L)
    c0 = _conv_self(a, n, L)
    Lc = 2 * n - 1

    if n > 7000:
        max_rows = min(int(max_rows), 160)
    elif n > 4500:
        max_rows = min(int(max_rows), 220)
    else:
        max_rows = min(int(max_rows), 300)

    init_rows = int(min(60, Lc))
    active_set = set(int(k) for k in np.argsort(-c0)[:init_rows])
    raw_b = a.copy()
    tau = float(np.max(c0))

    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[n] = 1.0
    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    lp_deadline = time.monotonic() + max(5.0, remaining)
    batch = 8

    for _ in range(30):
        if time.monotonic() > lp_deadline - 2.0:
            break
        if len(active_set) >= max_rows:
            break

        rows = np.asarray(sorted(active_set), dtype=np.int64)
        A = _build_lp_matrix(a, rows, n)
        if A is None:
            break

        tlimit = max(1.0, min(8.0, lp_deadline - time.monotonic() - 1.0))
        res = _call_linprog(cvec, A, np.zeros(rows.size, dtype=float), Aeq, b_eq, bounds, tlimit)
        x = getattr(res, "x", None) if res is not None else None
        if x is None or len(x) != n + 1:
            break
        x = np.asarray(x, dtype=np.float64)
        if not np.all(np.isfinite(x)):
            break

        raw_b = np.maximum(x[:n], 0.0)
        s = float(np.sum(raw_b))
        if s <= 1e-300:
            break
        raw_b /= s
        tau = max(0.0, float(x[n]))

        cross = np.fft.irfft(F_a * np.fft.rfft(raw_b, L), n=L)[:Lc].astype(np.float64, copy=False)
        np.maximum(cross, 0.0, out=cross)
        viol = cross - tau
        max_viol = float(np.max(viol))

        tok = 1e-7 + 1e-5 * max(1.0, abs(tau))
        if max_viol <= tok:
            return _norm1(0.999 * raw_b + 0.001 * a)

        add_count = 0
        for k in np.argsort(-viol)[:batch]:
            k = int(k)
            if k not in active_set:
                active_set.add(k)
                add_count += 1
        if add_count == 0:
            if max_viol < 1e-4:
                return _norm1(0.999 * raw_b + 0.001 * a)
            break

    # Fallback: return raw_b if approximately feasible.
    try:
        cross = np.fft.irfft(F_a * np.fft.rfft(raw_b, L), n=L)[:Lc].astype(np.float64, copy=False)
        np.maximum(cross, 0.0, out=cross)
        if float(np.max(cross)) <= 1.05 * (tau + 1e-12) + 1e-5:
            return _norm1(0.999 * raw_b + 0.001 * a)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Returns a Python list of nonnegative floats.
    """
    try:
        start = time.monotonic()
        budget_s = max(60.0, float(budget_s))
        deadline = start + budget_s - 8.0
        rng = np.random.default_rng(seed)
        parent = _get_parent()

        # Candidate sequence lengths.
        lengths = []
        if parent is not None and parent.size >= 100:
            lengths.append(int(parent.size))
        for cand in (3809, 4703, 5609, 7001, 8903):
            if cand not in lengths:
                lengths.append(cand)
        if parent is None and len(lengths) > 4:
            lengths = lengths[:4]

        best_a = None
        best_n = 0
        best_obj = float("inf")

        # ---------- Short length sweep ----------
        screen_total = min(0.22 * budget_s, 340.0)
        if budget_s < 700:
            screen_total = min(screen_total, 0.18 * budget_s)
        per_screen = max(8.0, screen_total / max(1, len(lengths)))

        print(f"[screen] lengths={lengths} per={per_screen:.1f}s")
        for n in lengths:
            if time.monotonic() > deadline - 240.0:
                break
            n = int(n)
            a = _initial_sequence(parent, n, rng)
            L = _fft_len(n)
            end_screen = min(time.monotonic() + per_screen, deadline - 200.0)
            if end_screen <= time.monotonic() + 2.0:
                break

            a2, mc2 = _gradient_descent(
                a, n, L, end_screen,
                tag=f"scr{n}",
                gamma=1.5, eta=2.0, gamma_max=50.0, do_print=False,
            )
            obj = 2.0 * float(n) * mc2
            if math.isfinite(obj) and obj < best_obj:
                best_obj = obj
                best_n = n
                best_a = a2.copy()
                print(f"[screen] n={n} J={best_obj:.6f}")

        if best_a is None:
            best_n = int(parent.size if parent is not None else 3809)
            best_a = _norm1(np.ones(best_n, dtype=np.float64))
            best_obj = 2.0 * float(best_n) * _max_conv(best_a, best_n, _fft_len(best_n))
            print(f"[fallback] initialized n={best_n} J={best_obj:.6f}")

        best_n = int(best_n)
        L = _fft_len(best_n)
        best_obj = float(best_obj)
        print(f"[chosen] n={best_n} J={best_obj:.6f}")

        last_improve = time.monotonic()
        long_deadline = deadline - 25.0
        attempt = 0

        # ---------- Main refinement loop ----------
        while time.monotonic() < long_deadline:
            attempt += 1
            rem = long_deadline - time.monotonic()
            if rem <= 20.0:
                break

            # 1. Broad smoothed descent.
            gb = min(25.0, 0.25 * rem)
            if gb > 6.0:
                a2, mc2 = _gradient_descent(
                    best_a, best_n, L, time.monotonic() + gb,
                    tag="broad",
                    gamma=1.5, eta=2.0, gamma_max=60.0,
                    do_print=False,
                )
                obj = 2.0 * float(best_n) * mc2
                if math.isfinite(obj) and obj < best_obj - 1e-8:
                    best_obj = obj
                    best_a = a2.copy()
                    last_improve = time.monotonic()
                    print(f"[broad] J={best_obj:.6f} attempt={attempt}")

            # 2. LP/game direction from current best.
            if time.monotonic() + 8.0 < long_deadline:
                gbud = min(12.0, 0.12 * (long_deadline - time.monotonic()))
                if gbud > 4.0:
                    try:
                        b = _game_direction(
                            best_a, best_n, L,
                            time.monotonic() + gbud,
                            gamma_r=8.0, gamma_c=5.0, a_weight=0.35,
                        )
                        a_mix, mc_mix = _mix_candidates(
                            best_a, b, best_n, L,
                            [0.1, 0.25, 0.5, 0.8],
                            cur_mc=best_obj / (2.0 * float(best_n)),
                        )
                        obj = 2.0 * float(best_n) * mc_mix
                        if math.isfinite(obj) and obj < best_obj - 1e-8:
                            best_obj = obj
                            best_a = a_mix.copy()
                            last_improve = time.monotonic()
                            print(f"[game] J={best_obj:.6f} attempt={attempt}")
                        else:
                            # Even if not immediately better, try short polish from the mix.
                            ref_start = a_mix
                            ref_dead = min(time.monotonic() + 10.0, long_deadline - 4.0)
                            if ref_dead > time.monotonic() + 2.0:
                                a_pol, mc_pol = _gradient_descent(
                                    ref_start, best_n, L, ref_dead,
                                    tag="gameref",
                                    gamma=3.0, eta=2.0, gamma_max=50.0,
                                    do_print=False,
                                )
                                obj_pol = 2.0 * float(best_n) * mc_pol
                                if math.isfinite(obj_pol) and obj_pol < best_obj - 1e-8:
                                    best_obj = obj_pol
                                    best_a = a_pol.copy()
                                    last_improve = time.monotonic()
                                    print(f"[gameref] J={best_obj:.6f} attempt={attempt}")
                    except Exception:
                        pass

            # 3. Active/high-sharpness descent.
            if time.monotonic() + 8.0 < long_deadline:
                abud = min(12.0, 0.18 * (long_deadline - time.monotonic()))
                if abud > 4.0:
                    a3, mc3 = _gradient_descent(
                        best_a, best_n, L, time.monotonic() + abud,
                        tag="active",
                        gamma=30.0, eta=1.2, gamma_max=140.0,
                        gamma_inc=1.004, do_print=False,
                    )
                    obj = 2.0 * float(best_n) * mc3
                    if math.isfinite(obj) and obj < best_obj - 1e-8:
                        best_obj = obj
                        best_a = a3.copy()
                        last_improve = time.monotonic()
                        print(f"[active] J={best_obj:.6f} attempt={attempt}")

            # 4. Sparse LP cutting-plane direction occasionally.
            if attempt % 5 == 3 and time.monotonic() + 25.0 < long_deadline:
                lbud = min(25.0, 0.10 * (long_deadline - time.monotonic()))
                if lbud > 6.0:
                    try:
                        b = _lp_direction(best_a, best_n, L, lbud, max_rows=300)
                        if b is not None:
                            a_lp, mc_lp = _mix_candidates(
                                best_a, b, best_n, L,
                                [0.1, 0.25, 0.5, 0.8],
                                cur_mc=best_obj / (2.0 * float(best_n)),
                            )
                            obj = 2.0 * float(best_n) * mc_lp
                            if math.isfinite(obj) and obj < best_obj - 1e-8:
                                best_obj = obj
                                best_a = a_lp.copy()
                                last_improve = time.monotonic()
                                print(f"[lp] J={best_obj:.6f} attempt={attempt}")
                    except Exception:
                        pass

            # 5. Perturb and re-polish if stuck.
            if time.monotonic() - last_improve > 100.0 and time.monotonic() + 20.0 < long_deadline:
                sigma = min(0.7, 0.15 + 0.03 * attempt)
                ap = _perturb(best_a, best_n, rng, sigma, mode=attempt)
                ref_dead = min(time.monotonic() + 18.0, long_deadline - 4.0)
                if ref_dead > time.monotonic() + 3.0:
                    a_p, mc_p = _gradient_descent(
                        ap, best_n, L, ref_dead,
                        tag="pert",
                        gamma=2.5, eta=2.0, gamma_max=60.0,
                        do_print=False,
                    )
                    obj = 2.0 * float(best_n) * mc_p
                    if math.isfinite(obj) and obj < best_obj - 1e-8:
                        best_obj = obj
                        best_a = a_p.copy()
                        last_improve = time.monotonic()
                        print(f"[perturb] J={best_obj:.6f} attempt={attempt}")

        # ---------- Final polish ----------
        final_deadline = deadline - 2.0
        if time.monotonic() < final_deadline - 5.0:
            a_f, mc_f = _gradient_descent(
                best_a, best_n, L, final_deadline,
                tag="final",
                gamma=50.0, eta=0.9, gamma_max=160.0,
                gamma_inc=1.002, do_print=False,
            )
            obj = 2.0 * float(best_n) * mc_f
            if math.isfinite(obj) and obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_f.copy()
                print(f"[final-polish] J={best_obj:.6f}")

        # Final symmetrization check.
        try:
            sym = _norm1(0.5 * (best_a + best_a[::-1]))
            mc_sym = _max_conv(sym, best_n, L)
            obj_sym = 2.0 * float(best_n) * mc_sym
            if math.isfinite(obj_sym) and obj_sym < best_obj - 1e-10:
                best_obj = obj_sym
                best_a = sym.copy()
                print(f"[sym] J={best_obj:.6f}")
        except Exception:
            pass

        result = _norm1(best_a)
        print(f"[FINAL] n={best_n} own-J={2.0 * float(best_n) * _max_conv(result, best_n, L):.6f}")

        # Optional quick validation for moderate lengths.
        if best_n <= 5000 and time.monotonic() < deadline - 15.0:
            try:
                ev = float(evaluate_sequence([float(x) for x in result]))
                if math.isfinite(ev):
                    print(f"[evaluate_sequence] {ev:.6f}")
            except Exception:
                pass

        # Sanity: force finite, non-negative, acceptable sum.
        result = np.asarray(result, dtype=np.float64)
        result = np.where(np.isfinite(result), result, 0.0)
        result = np.maximum(result, 0.0)
        if float(np.sum(result)) < 0.01:
            m = max(1000, best_n)
            result = np.ones(m, dtype=np.float64)
        result = _norm1(result)

        return [float(x) for x in result]

    except Exception:
        # Absolute fallback: safe parent resample or ones.
        try:
            parent = _get_parent()
            if parent is not None:
                seq = _norm1(parent)
            else:
                seq = _norm1(np.ones(3000, dtype=np.float64))
            return [float(x) for x in seq]
        except Exception:
            return [1.0 / 1000.0] * 1000
