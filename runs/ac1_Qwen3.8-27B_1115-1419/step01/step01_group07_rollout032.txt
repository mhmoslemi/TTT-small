```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    fa = np.fft.rfft(a, n)
    fb = np.fft.rfft(b, n)
    return np.fft.irfft(fa * fb, n)

def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    c = fft_convolve(a, a)
    max_c = c.max()
    sum_a = a.sum()
    return 2.0 * n * max_c / (sum_a ** 2), c, max_c

def solve_lp(a_t, M, n):
    """Solve the LP: max 1'b s.t. T(a_t)*b <= M*1, b >= 0."""
    # Build sparse constraint matrix A (2n-1 x n)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a_t[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize 1'b => minimize -1'b
    c_obj = -np.ones(n)
    b_rhs = M * np.ones(2*n - 1)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_rhs, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, result.fun  # result.x is b*, result.fun is -sum(b*)
    else:
        return None, None
```

Optimize step function sequence to minimize 2*n*max(a*a)/(sum(a))^2.
Uses FFT-accelerated LSE smoothing, multiple tau scales with L-BFGS warm-starting,
diverse asymmetric initializations, and exact subgradient polishing.
Key: FFT gives O(n log n) convolutions enabling 1000x more iterations than O(n^2).

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    fa = np.fft.rfft(a, n)
    fb = np.fft.rfft(b, n)
    return np.fft.irfft(fa * fb, n)


def compute_convolution(a):
    """Compute a * a using FFT."""
    n = 2 * len(a) - 1
    fa = np.fft.rfft(a, n)
    return np.fft.irfft(fa * fa, n)


def lse_objective(x, tau):
    """LSE-smoothed max of convolve(a,a) where a = softmax(x)."""
    a = softmax(x)
    c = compute_convolution(a)
    c_max = c.max()
    shifted = c - c_max
    vals = np.exp(shifted / tau)
    return c_max + tau * np.log(vals.sum())


def lse_gradient(x, tau):
    """Gradient of LSE-smoothed max w.r.t. x (softmax parameterization)."""
    a = softmax(x)
    c = compute_convolution(a)
    c_max = c.max()
    shifted = c - c_max
    w = np.exp(shifted / tau)
    w /= w.sum()
    # Gradient of smoothed max w.r.t. a: 2 * correlate(w, a)
    # np.correlate(w, a, 'valid') gives sum_k w[k+i]*a[k] for i=0..n-1
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    dot = np.dot(a, grad_a)
    return a * (grad_a - dot)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def exact_subgradient(a, n_active=80, step=0.01):
    """Exact subgradient step on the simplex."""
    n = len(a)
    c = compute_convolution(a)
    max_c = c.max()
    if max_c < 1e-15:
        return a
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    # Compute subgradient: for each active k, the subgradient contribution is
    # 2 * a[k-hi:k-lo+1][::-1] added to positions lo..hi
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

    best_val = lse_objective(x, tau)

        return lse_objective(x_, tau)

    inits.append(x / x.sum())

    num_inits = len(inits)
    time_per_init = (budget_s - 120) / num_inits  # Reserve 120s for polishing
    print(f"Phase 1: {num_inits} initializations, ~{time_per_init:.0f}s each")

    for idx, init_a in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break

        init_a = np.maximum(init_a, 1e-15)
        init_a = init_a / init_a.sum()
        x = np.log(init_a)

        init_deadline = min(time.monotonic() + time_per_init, deadline - 120)

        # Multi-scale tau: from coarse to fine
        taus = [0.1 / n, 0.05 / n, 0.02 / n, 0.01 / n, 0.005 / n, 0.002 / n]

        for ti, tau in enumerate(taus):
            if time.monotonic() > init_deadline:
                break
            remaining = init_deadline - time.monotonic()
            max_iter = max(30, int(remaining * 50))

            x, val = run_lbfgs_tau(x, tau, max_iter=max_iter)

            a_check = softmax(x)
            sc = score_of_a(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                print(f"  Init {idx}, tau={tau*n:.3f}: score={best_score:.6f}")

        # Evaluate final
        a_check = softmax(x)
        sc = score_of_a(a_check)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_check.copy()
            print(f"  Init {idx} final: score={best_score:.6f}")

    print(f"\nAfter Phase 1: best_score={best_score:.6f}")

    # Phase 2: Fine-tuning from best with very small tau
    a = best_a.copy()
    x = np.log(np.maximum(a, 1e-15))

    fine_taus = [0.001 / n, 0.0005 / n, 0.0002 / n]
    fine_deadline = min(time.monotonic() + 300, deadline - 60)

    for tau in fine_taus:
        if time.monotonic() > fine_deadline:
            break
        remaining = fine_deadline - time.monotonic()
        max_iter = max(50, int(remaining * 30))
        x, val = run_lbfgs_tau(x, tau, max_iter=max_iter)
        a_check = softmax(x)
        sc = score_of_a(a_check)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_check.copy()
            print(f"  Fine tau={tau*n:.4f}: score={best_score:.6f}")

    print(f"\nAfter Phase 2: best_score={best_score:.6f}")

    # Phase 3: Exact subgradient polishing
    a = best_a.copy()
    best_score = score_of_a(a)
    print(f"\nPhase 3: Subgradient polishing from score={best_score:.6f}")

    # Adaptive step size with momentum-like behavior
    step = 0.05
    no_improve_count = 0
    polish_deadline = deadline - 10

    while time.monotonic() < polish_deadline:
        # Batch of subgradient steps
        for _ in range(50):
            if time.monotonic() > polish_deadline:
                break
            a = exact_subgradient(a, n_active=100, step=step)

        sc = score_of_a(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
            if int((budget_s - (deadline - time.monotonic())) % 30) < 0.1:
                print(f"  Polish: score={best_score:.6f}, step={step:.6f}")
        else:
            no_improve_count += 1

        # Decay step size
        step *= 0.98
        if step < 0.001:
            step = 0.02
            # Perturb to escape
            a_pert = a + rng.normal(0, 0.003, n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            sc_p = score_of_a(a_pert)
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_pert.copy()
                a = a_pert.copy()
                print(f"  Perturbation improved: {best_score:.6f}")

    # Final evaluation
    final_score = evaluate_sequence([float(val) for val in best_a])
    print(f"\nFinal score: {final_score:.6f}")

    return [float(val) for val in best_a]
