from scipy.sparse import csr_matrix
import time

def build_conv_matrix(a, n):
    """Build the matrix A where A @ g = conv(a, g)."""
    # A has shape (2n-1, n)
    # A[j, i] = a[i] if j-i in [0, n-1], else 0
    # i.e., for column i, rows i to i+n-1 have value a[i]
    
    rows = []

        if a[i] > 0:

    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

"""
LP-based Frank-Wolfe optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Linear Programming to find descent directions, combined with gradient polish.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _conv_max(a):
    """Compute max of self-convolution."""
    return float(np.max(np.convolve(a, a)))


def _objective(a, n):
    """Compute the objective function."""
    s = float(a.sum())
    if s < 1e-12:
        return 1e10
    return 2.0 * n * _conv_max(a) / (s * s)


def _build_A_vals(a, n, all_rows, all_cols):
    """Build sparse convolution matrix values."""
    # Column i has value a[i] in n consecutive rows
    vals = np.repeat(a, n)
    return csr_matrix((vals, (all_rows, all_cols)), shape=(2*n-1, n))


def _lp_step(a, n, all_rows, all_cols):
    """Solve the LP subproblem. Returns g (normalized to sum=1) or None."""
    c = np.convolve(a, a)
    M = float(c.max())
    
    A = _build_A_vals(a, n, all_rows, all_cols)
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, 2.0)] * n  # g[i] in [0, 2]
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception as e:
        print(f"  LP error: {e}")
        return None
    
    if res.status not in (0, 3):  # 0=optimal, 3=time limit
        return None
    if res.x is None or res.x.sum() < 1e-12:
        return None
    
    g = np.maximum(res.x, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _gradient_step(a, n, lr=0.01):
    """One gradient step on the smoothed max."""
    c = np.convolve(a, a)
    k = int(np.argmax(c))
    # Gradient of c[k] w.r.t. a
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:

        g[js:je] = 2.0 * a[ai]
    # Subtract mean to preserve sum
    g -= g.mean()
    a = np.maximum(a - lr * g, 0.0)
    a = a / a.sum()
    return a

"""
LP-based Frank-Wolfe optimizer with gradient polish for minimizing
2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Linear Programming to find globally-informed descent directions,
combined with hard-max gradient steps and adaptive perturbations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _obj(a, n):
    """Objective: 2*n*max(conv(a,a))/sum(a)^2."""
    s = float(a.sum())
    if s < 1e-12:
        return 1e10
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _grad_hard(a, k, n):
    """Gradient of conv(a,a)[k] w.r.t. a, mean-subtracted."""
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        g[js:je] = 2.0 * a[k - np.arange(js, je)]
    g -= g.mean()
    norm = np.max(np.abs(g))
    if norm > 1e-15:
        g /= norm
    return g


def _smooth_obj(a, n, tau):
    """Entropy-smoothed max of convolution."""
    c = np.convolve(a, a)
    mx = c.max()
    p = np.exp((c - mx) / tau)
    return mx + tau * np.log(p.sum())


def _smooth_grad(a, n, tau):
    """Gradient of entropy-smoothed max."""
    c = np.convolve(a, a)
    mx = c.max()
    p = np.exp((c - mx) / tau)
    p /= p.sum()
    g = 2.0 * np.correlate(p, a, mode='valid')
    g -= g.mean()
    return g

            stall = 0

                stall = 0
        if it % 5000 == 0 and it > 0:
            print(f"    [{tag}] it={it} obj={best_o:.6f} eta={eta:.5f}")
    return best, best_o


def _lp_solve(a, n, all_rows, all_cols, time_limit=25):
    """Solve one LP subproblem. Returns normalized g or None."""
    c = np.convolve(a, a)
    M = float(c.max())
    
    # Build sparse constraint matrix: A[g] = conv(a, g)
    vals = np.repeat(a, n)
    A = csr_matrix((vals, (all_rows, all_cols)), shape=(2 * n - 1, n))
    
    # LP: max sum(g) s.t. A@g <= M, 0 <= g <= 2
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 2.0)] * n
    
    try:

            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

            best_t = t
