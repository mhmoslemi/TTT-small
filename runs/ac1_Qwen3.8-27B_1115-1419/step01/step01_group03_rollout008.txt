def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def compute_smooth_gradient(a, beta=100.0):
    """Compute smooth gradient of max(a*a) using softmax weighting."""
    n = len(a)
    c = np.convolve(a, a)
    # Softmax weights (numerically stable)
    c_max = c.max()
    weights = np.exp(beta * (c - c_max))
    weights /= weights.sum()
    
    # Gradient: for each k, d(a*a)[k]/da[j] = a[k-j] (with bounds)
    grad = np.zeros(n)
    # For efficiency, only use top few weights
    top_ids = np.where(weights > 1e-6)[0]
    for k in top_ids:
        k = int(k)
        # (a*a)[k] = sum_{j} a[j]*a[k-j] where 0<=j<n and 0<=k-j<n
        # d/d a[j] = a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += weights[k] * a[idx]
    return grad

"""
Minimize 2n*max(a*a)/(sum(a))^2 using improved FISTA on the simplex with:
- Adaptive softmax gradient (increasing beta for sharper peak focusing)
- Growing top-k active peaks
- Backtracking step size
- Multiple restarts with perturbation
- Mirror descent (multiplicative updates) as complementary method
- Local polishing phase
"""

import numpy as np
import time


def fast_score(a):
    """Compute score without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho_idx = np.where(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_gradient(a, beta, top_k=30):
    """
    Compute smooth approximate gradient of max(a*a) using softmax weighting.
    Returns (gradient, max_conv_value).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    if top_k >= c.size:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    vals = c[top_idx] - c_max
    w = np.exp(beta * vals)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= wsum
    grad = np.zeros(n, dtype=np.float64)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        j_range = np.arange(lo, hi)
        a_idx = k - j_range
        grad[j_range] += w[i] * a[a_idx]
    return grad, c_max


def run_fista(a0, duration, beta_init=30.0, beta_max=500.0,
              top_k_init=10, top_k_max=80, step_init=0.8,
              rng=None, verbose=False):
    """
    Run FISTA to minimize max(a*a) on the simplex.
    Returns (best_a, best_score_on_simplex).
    """
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a0)
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = step_init
    beta = beta_init
    top_k = top_k_init
    best_a = a0.copy()
    best_maxc = np.convolve(a0, a0).max()
    deadline = time.monotonic() + duration
    it = 0
    stall = 0
    improve_period = 50

    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_new
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.05:
            y /= sy

        # Compute gradient
        grad, _ = softmax_gradient(y, beta, top_k)

        # Gradient step
        a_new = y - step * grad
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1

        # Periodic score check and adaptation
        if it % improve_period == 0:
            c_new = np.convolve(a, a)
            cur_maxc = c_new.max()
            if cur_maxc < best_maxc * (1.0 - 1e-9):
                best_maxc = cur_maxc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            # Adapt beta (increase sharpness)
            if it % 2000 == 0 and beta < beta_max:
                beta *= 1.5

            # Adapt top_k
            if it % 3000 == 0 and top_k < top_k_max:
                top_k = min(top_k_max, top_k + 10)

            # Backtracking step size
            if stall >= 100:
                step *= 0.7
                if stall >= 250:
                    # Reset momentum
                    a_prev = best_a.copy()
                    t = 1.0
                    step = max(0.1, step * 0.8)
                if stall >= 500:
                    # Restart with perturbation
                    a = best_a.copy()
                    perturb = rng.normal(0.0, 0.5 / np.sqrt(n), n)
                    a = a + perturb
                    np.maximum(a, 0.0, out=a)
                    sa = a.sum()
                    if sa > 1e-12:
                        a /= sa
                    else:
                        a = best_a.copy()
                    a_prev = a.copy()
                    t = 1.0
                    step = step_init
                    beta = max(30.0, beta * 0.5)
                    top_k = max(10, top_k // 2)
                    stall = 0
                    if verbose:
                        print(f"    Restart at it={it}, maxc={best_maxc:.10f}")

            if verbose and it % 10000 == 0:
                cur_score = 2.0 * n * best_maxc
                print(f"    It={it}, maxc={best_maxc:.10f}, score={cur_score:.6f}, "
                      f"step={step:.3f}, beta={beta:.1f}, tk={top_k}")

    # Final check
    c_final = np.convolve(a, a)
    if c_final.max() < best_maxc:
        best_maxc = c_final.max()
        best_a = a.copy()

    return best_a, best_maxc


def run_mirror_descent(a0, duration, beta=50.0, top_k=30,
                       eta_init=0.5, rng=None, verbose=False):
    """
    Run mirror descent (multiplicative weight updates) on the simplex.
    """
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a0)
    a = a0.copy()
    best_a = a0.copy()
    best_maxc = np.convolve(a0, a0).max()
    eta = eta_init
    deadline = time.monotonic() + duration
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        grad, _ = softmax_gradient(a, beta, top_k)
        # Multiplicative update (mirror descent with entropy)
        a_new = a * np.exp(-eta * grad)
        sm = a_new.sum()
        if sm < 1e-15:
            a_new = a.copy()
        else:
            a_new /= sm
        a = a_new
        it += 1

        if it % 100 == 0:
            c_cur = np.convolve(a, a)
            cur_maxc = c_cur.max()
            if cur_maxc < best_maxc * (1.0 - 1e-9):
                best_maxc = cur_maxc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if stall >= 50:
                eta *= 0.7
            if stall >= 200:
                # Restart
                a = best_a + rng.normal(0.0, 0.3 / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                eta = eta_init
                stall = 0
            if verbose and it % 5000 == 0:
                cur_score = 2.0 * n * best_maxc
                print(f"    MD it={it}, maxc={best_maxc:.10f}, score={cur_score:.6f}, eta={eta:.4f}")

    c_final = np.convolve(a, a)
    if c_final.max() < best_maxc:
        best_maxc = c_final.max()
        best_a = a.copy()
    return best_a, best_maxc


def local_polish(a, duration, rng=None, verbose=False):
    """
    Local polishing: subgradient with very small step and many active peaks.
    """
    if rng is None:
        rng = np.random.default_rng(999)
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_maxc = np.convolve(a, a).max()
    step = 0.1
    deadline = time.monotonic() + duration
    it = 0

    while time.monotonic() < deadline:
        grad, _ = softmax_gradient(a, beta=200.0, top_k=100)
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        a = a_new
        it += 1
        if it % 100 == 0:
            c_cur = np.convolve(a, a)
            cur_maxc = c_cur.max()
            if cur_maxc < best_maxc * (1.0 - 1e-10):
                best_maxc = cur_maxc
                best_a = a.copy()
                step = min(0.3, step * 1.1)
            else:
                step *= 0.95
    return best_a, best_maxc
