"""
Search for low-peak nonnegative self-convolutions using a hybrid approach:
- LSE (log-sum-exp) surrogate gradient descent with adaptive beta
- Momentum-based acceleration
- Peak-shaving via sensitivity-weighted updates
- Coordinate descent refinement
- Simulated annealing diversification
- Multi-scale search (optimize at small n, resample to larger n)
"""

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _lse_and_p(b, beta):
    if not b.size:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _weighted_gradient(a, weights, fft_size, n):
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    """Proper projection onto the probability simplex (sum=1, a>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = u > cssv / np.arange(1, n + 1)
    if not rho.any():
        return np.ones(n) / n
    theta = cssv[rho][-1] / float(rho.sum())
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return result / s


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_mixture(n, alpha, gamma):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x) + float(gamma) * np.exp(-float(alpha) * (1.0 - x))
    a = a + 0.15
    return _normalize(a)


def _make_bell(n, center, width):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
    return _normalize(a)


def _make_smooth_random(n, rng, base=None, amplitude=0.4):
    n = max(int(n), 2)
    if base is None:
        base = _make_uniform(n)
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 16 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_uniform(n):
    return np.ones(max(int(n), 1), dtype=np.float64) / max(int(n), 1)


def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _smoothing_perturb(a, rng, scale=0.15):
    """Add smooth noise to a."""
    n = a.size
    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    cutoff = max(2, min(len(F), n // 8 + 2))
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = (f - f.mean()) / (f.std() + 1e-300)
    a2 = a + scale * f / n
    return _normalize(a2)


def _block_perturb(a, rng, strength=0.3):
    """Perturb a random block."""
    n = a.size
    c = a.copy()
    width = max(2, n // 16)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    return _normalize(c)


def _peak_shave(a, b, rng, n_shave=5, strength=0.1):
    """Directly reduce elements that contribute most to the peak."""
    c = a.copy()
    peak_idx = int(np.argmax(b))
    # Find which a_i contribute most to b[peak_idx]
    # b[peak_idx] = sum_j a_j * a[peak_idx - j]
    contribs = []
    for j in range(len(a)):
        k = peak_idx - j
        if 0 <= k < len(a):
            contribs.append((j, a[j] * a[k]))
    contribs.sort(key=lambda x: -x[1])
    # Reduce the top contributors
    for j, _ in contribs[:n_shave]:
        c[j] *= (1.0 - strength * rng.random())
    return _normalize(c)


def _coordinate_descent(a, max_sweeps=5, deadline=None):
    """
    Simple coordinate descent: for each i, try to find a better value of a_i.
    Uses ternary search on [0, 4*a_i] to minimize the peak of b.
    """
    a = _normalize(a).copy()
    n = a.size
    b_fft = _next_pow2(2 * n - 1)
    
    best_score = _score_of(a)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and (i % 64 == 0) and time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting it to a small positive value
                candidate = a.copy()
                candidate[i] = max(0.01 / n, 1e-8)
                a = _normalize(candidate)
                b_check = _autocorr(a, b_fft)
                s = 2.0 * n * float(np.max(b_check))
                if s < best_score - 1e-15:
                    best_score = s
                    improved = True
                else:
                    a[i] = old_val
                continue
            
            # Ternary search for optimal a[i] in [0.1*old_val, 3*old_val]
            lo = 0.05 * old_val
            hi = 3.0 * old_val
            for _ in range(12):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                a[i] = m1
                a = _normalize(a)
                b1 = _autocorr(a, b_fft)
                s1 = float(np.max(b1))
                
                a[i] = m2
                a = _normalize(a)
                b2 = _autocorr(a, b_fft)
                s2 = float(np.max(b2))
                
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
            
            a[i] = (lo + hi) / 2.0
            a = _normalize(a)
            b_new = _autocorr(a, b_fft)
            s_new = 2.0 * n * float(np.max(b_new))
            if s_new < best_score - 1e-15:
                best_score = s_new
                improved = True
            else:
                a[i] = old_val
                a = _normalize(a)
        
        if not improved:
            break
    
    return a, best_score


def _optimize_lse(a, deadline, rng, beta_start=8.0, beta_end=200.0, use_momentum=True):
    """
    LSE-based gradient optimization with momentum and adaptive beta.
    """
    a = _normalize(a)
    n = a.size
    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(2 * (2 * n - 1))
    
    b = _autocorr(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score
    
    alpha = 0.15
    beta = beta_start
    it = 0
    fail = 0
    
    # Momentum
    if use_momentum:
        momentum = np.zeros(n, dtype=np.float64)
        mu = 0.7
    
    while time.monotonic() < deadline:
        if maxb <= 1e-300:
            maxb = 1e-300
        # Adaptive beta: target is beta * maxb ~ O(10-50)
        target_beta = max(5.0, min(beta_end, 30.0 / maxb))
        beta = max(beta * 1.005, target_beta)
        
        p, smooth_old = _lse_and_p(b, beta)
        grad = _weighted_gradient(a, p, g_fft, n)
        
        # Center the gradient
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))
        
        if scale <= 1e-300:
            # Stuck - perturb
            a = _smoothing_perturb(best_a, rng, 0.2)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.15
            fail = 0
            if use_momentum:
                momentum = np.zeros(n, dtype=np.float64)
            it += 1
            continue
        
        # Step
        if use_momentum:
            momentum = mu * momentum + (1.0 - mu) * grad_c
            step = momentum
        else:
            step = grad_c
        
        v = a - alpha * step / scale * 0.1  # Normalize step
        
        # Line search: try several step sizes
        best_trial = None
        best_trial_score = smooth_old
        for cand_alpha in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25]:
            v_trial = a - cand_alpha * step / scale * 0.1
            v_trial = np.maximum(v_trial, 0.0)
            s_trial = v_trial.sum()
            if s_trial < 1e-300:
                continue
            v_trial = v_trial / s_trial
            b_trial = _autocorr(v_trial, b_fft)
            trial_score = _lse_and_p(b_trial, beta)[1]
            if trial_score < best_trial_score:
                best_trial_score = trial_score
                best_trial = v_trial
        
        if best_trial is not None:
            a = best_trial
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
        else:
            a = _project_simplex(v)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
        
        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score
            fail = 0
        else:
            fail += 1
        
        # Periodic diversification
        if fail > 500 and time.monotonic() < deadline - 5.0:
            if rng.random() < 0.5:
                a = _smoothing_perturb(best_a, rng, 0.25)
            else:
                a = _block_perturb(best_a, rng, 0.35)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = 0.15
            fail = 0
            beta = max(5.0, beta * 0.3)
            if use_momentum:
                momentum = np.zeros(n, dtype=np.float64)
        
        if fail > 1500:
            a = best_a.copy()
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            beta = beta_start
            alpha = 0.1
            fail = 0
            if use_momentum:
                momentum = np.zeros(n, dtype=np.float64)
        
        it += 1
    
    return best_a, best_score

4. result_i = max(v_i - theta, 0)

"""

import numpy as np

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _lse_and_p(b, beta):
    if not b.size:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _grad_lse(a, b, p, beta, fft_size):
    """Gradient of LSE(b, beta) w.r.t. a. = 2 * convolve(a[::-1], p)[:n] * beta / sum(exp(...))"""
    n = a.size
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    """Exact Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = v.size
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    if not np.all(np.isfinite(v)):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rng_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = u > cssv / rng_arr
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1]) + 1
    theta = cssv[rho - 1] / float(rho)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return result


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _make_exponential(n, alpha):

def _make_mixture(n, alpha, gamma):

def _make_smooth_random(n, rng, base=None, amplitude=0.4):

        base = np.ones(n) / n

    f = np.fft.irfft(F, n)
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if sd > 0.0:
        f = f / sd
    a = base * (1.0 + float(amplitude) * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _make_uniform(n):

def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return _make_uniform(new_n)
    s = float(old.sum())
    if s <= 1e-300:
        return _make_uniform(new_n)
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _smoothing_perturb(a, rng, scale=0.2):
    n = a.size
    z = rng.normal(size=n)

        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f = (f - f.mean()) / (f.std() + 1e-300)
    a2 = a + scale * f / np.sqrt(float(n))
    return _normalize(a2)


def _block_perturb(a, rng, strength=0.3):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    return _normalize(c)


def _peak_shave(a, b, rng, top_k=8, strength=0.08):
    """Reduce elements that contribute most to the peak of b."""
    c = a.copy()
    n = a.size
    peak_idx = int(np.argmax(b))
    contribs = []
    for j in range(n):
        k = peak_idx - j
        if 0 <= k < n:
            contribs.append((j, a[j] * a[k]))
    contribs.sort(key=lambda x: -x[1])
    for j, _ in contribs[:top_k]:
        c[j] *= (1.0 - strength * (0.5 + 0.5 * rng.random()))
    # Also shave the second peak
    b_copy = b.copy()
    b_copy[peak_idx] = -np.inf
    if b_copy.max() > 0:
        peak_idx2 = int(np.argmax(b_copy))
        contribs2 = []
        for j in range(n):
            k = peak_idx2 - j
            if 0 <= k < n:
                contribs2.append((j, c[j] * c[k]))
        contribs2.sort(key=lambda x: -x[1])
        for j, _ in contribs2[:top_k // 2]:
            c[j] *= (1.0 - strength * 0.7 * rng.random())
    return _normalize(c)

def _coordinate_descent(a, deadline, max_sweeps=5):
    """
    Coordinate descent: for each element, use golden-section search to find optimal value.
    Efficient for n <= 512.
    """
    a = _normalize(a).copy()
    n = a.size
    if n > 1024:
        return a, _score_of(a)

    b_fft = _next_pow2(2 * n - 1)
    best_score = _score_of(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved_count = 0
        for i in range(n):
            if time.monotonic() > deadline:
                break
            if i % 64 == 0 and improved_count == 0 and sweep > 0:
                break

            old_val = a[i]
            if old_val < 1e-12:
                # Try a small positive value
                cand = a.copy()
                cand[i] = 1e-4 / n
                cand = _normalize(cand)
                b_c = _autocorr(cand, b_fft)
                s_c = 2.0 * n * float(np.max(b_c))
                if s_c < best_score - 1e-14:
                    a = cand
                    best_score = s_c
                    best_a = a.copy()
                    improved_count += 1
                continue

            # Golden section search in [0.05*old, 3*old]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.02 * old_val
            hi = 3.5 * old_val

            # Quick check: is current value already good?
            b_cur = _autocorr(a, b_fft)
            cur_peak = float(np.max(b_cur))

            for _ in range(10):
                m1 = lo + gr * (hi - lo)
                m2 = hi - gr * (hi - lo)

                a[i] = m1
                a = _normalize(a)
                b1 = _autocorr(a, b_fft)
                p1 = float(np.max(b1))

                a[i] = m2
                a = _normalize(a)
                b2 = _autocorr(a, b_fft)
                p2 = float(np.max(b2))

                if p1 < p2:
                    hi = m2
                else:
                    lo = m1

            # Evaluate at midpoint
            a[i] = (lo + hi) / 2.0
            a = _normalize(a)
            b_new = _autocorr(a, b_fft)
            s_new = 2.0 * n * float(np.max(b_new))

            if s_new < best_score - 1e-14:
                best_score = s_new
                best_a = a.copy()
                improved_count += 1
            else:
                # Revert
                a = best_a.copy()

        if improved_count == 0:
            break

    return best_a, best_score

    t_phase1_end = time.monotonic() + min(200, budget * 0.12)

    inits_1 = [
        _make_exponential(n1, 0.8),
        _make_exponential(n1, 1.5),
        _make_symmetric_exp(n1, 1.0),
        _make_mixture(n1, 1.0, 0.6),
        _make_bell(n1, 0.5, 0.2),
        _make_bell(n1, 0.3, 0.15),
        _make_u_shaped(n1, 1.5),
        _make_smooth_random(n1, rng, _make_exponential(n1, 1.0), 0.5),
        _make_smooth_random(n1, rng, _make_symmetric_exp(n1, 0.8), 0.4),
    ]
    if given is not None and given.size > 4:
        inits_1.append(_resample_sequence(given, n1))
        inits_1.append(_resample_sequence(given, n1)[::-1])

    # Also try the current best
    inits_1.append(_resample_sequence(best_a, n1))

    for idx, init in enumerate(inits_1):
        if time.monotonic() > t_phase1_end:
            break
        t_end = min(t_phase1_end, time.monotonic() + 22)
        res_a, res_score = _optimize_main(init.copy(), t_end, rng, beta_start=5, beta_max=200)
        print(f"P1 n={n1} init={idx} score={res_score:.8f} best={best_score:.8f} t={time.monotonic() - (deadline - budget + 25):.0f}s", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

    # ===== Phase 2: Focused optimization of best at n=256 (~400s) =====
    if time.monotonic() < deadline - 200:
        t_p2_end = min(deadline - 200, time.monotonic() + 400)
        res_a, res_score = _optimize_main(best_a.copy(), t_p2_end, rng, beta_start=8, beta_max=400)
        print(f"P2 focused n={best_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

    # ===== Phase 3: Coordinate descent refinement at n=256 (~120s) =====
    if time.monotonic() < deadline - 150 and best_a.size <= 512:
        t_p3_end = min(deadline - 150, time.monotonic() + 120)
        res_a, res_score = _coordinate_descent(best_a.copy(), t_p3_end, max_sweeps=4)
        print(f"P3 coord n={res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

    # ===== Phase 4: Try symmetric version =====
    if time.monotonic() < deadline - 150:
        sym_a = _symmetrize(best_a)
        sym_score = _score_of(sym_a)
        if sym_score < best_score:
            best_a = sym_a
            best_score = sym_score
            print(f"P4 symmetric: score={best_score:.8f}", flush=True)
        # Optimize the symmetric version
        if time.monotonic() < deadline - 100:
            t_p4_end = min(deadline - 100, time.monotonic() + 60)
            res_a, res_score = _optimize_main(sym_a.copy(), t_p4_end, rng, beta_start=10, beta_max=300)
            print(f"P4 sym-opt score={res_score:.8f} best={best_score:.8f}", flush=True)
            if res_score < best_score:
                best_a = res_a.copy()
                best_score = res_score

    # ===== Phase 5: Upscale to n=512 (~200s) =====
    if time.monotonic() < deadline - 100:
        n2 = 512
        init_2 = _resample_sequence(best_a, n2)
        # Mix with a few shapes for diversity
        init_2 = _normalize(0.85 * init_2 + 0.15 * _make_symmetric_exp(n2, 1.0))
        t_p5_end = min(deadline - 50, time.monotonic() + 200)
        res_a, res_score = _optimize_main(init_2, t_p5_end, rng, beta_start=6, beta_max=300)
        print(f"P5 upscale n={res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

        # Coordinate descent on n=512 if time allows
        if time.monotonic() < deadline - 30 and best_a.size == n2:
            t_p5b_end = min(deadline - 10, time.monotonic() + 60)
            res_a, res_score = _coordinate_descent(best_a.copy(), t_p5b_end, max_sweeps=2)
            print(f"P5b coord n=512 score={res_score:.8f} best={best_score:.8f}", flush=True)
            if res_score < best_score:
                best_a = res_a.copy()
                best_score = res_score

    # ===== Phase 6: Final deep refinement of overall best =====
    if time.monotonic() < deadline - 10:
        t_p6_end = deadline - 3
        res_a, res_score = _optimize_main(best_a.copy(), t_p6_end, rng, beta_start=12, beta_max=500)
        print(f"P6 final n={res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

    # ===== Phase 7: Final coordinate descent =====
    if time.monotonic() < deadline - 3 and best_a.size <= 512:
        res_a, res_score = _coordinate_descent(best_a.copy(), deadline - 1, max_sweeps=2)
        print(f"P7 coord n={res_a.size} score={res_score:.8f} best={best_score:.8f}", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

    print(f"FINAL: n={best_a.size} score={best_score:.8f}", flush=True)

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)
    return [float(x) for x in out]
