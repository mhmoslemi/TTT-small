def _smooth_grad(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a). Returns (grad, lse_value)."""
    c = _fft_conv(a)
    mx = float(c.max())
    e = np.exp((c - mx) / tau)
    p = e / e.sum()
    g = 2.0 * _cross_corr(p, a)
    lse = mx + tau * float(np.log(e.sum()))
    return g, lse


def _hard_grad_topk(a, k_list):
    """Gradient of average of conv(a,a)[k] for k in k_list."""
    n = len(a)
    g = np.zeros(n)
    for k in k_list:
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        if js < je:
            g[js:je] += 2.0 * a[k - np.arange(js, je)]
    g /= len(k_list)
    return g


def _project(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _obj_n(a):
    """Objective 2*max(conv)/n, valid when sum(a)=n (scale invariant)."""
    n = len(a)
    c = _fft_conv(a)
    return float(2.0 * c.max() / n)


def _smooth_run(a, n, deadline, tag, it_limit=3000000):
    """Log-sum-exp smoothed gradient descent with Nesterov momentum and backtracking."""
    tau = n * 0.5
    momentum = 0.85
    best_a = a.copy()
    best_o = _obj_n(a)
    v = np.zeros(n)
    y = a.copy()
    stall = 0
    step = 0.08

    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        # Nesterov lookahead
        g, lse_val = _smooth_grad(y, tau)
        g = g - g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        g = g / mg

        # Momentum update
        v = momentum * v - step * g
        y_new = y - v
        a_candidate = _project(y_new, float(n))

        # Track best
        o = 2.0 * lse_val / n
        hard_o = _obj_n(a_candidate)
        if hard_o < best_o - 1e-12:
            best_o = hard_o
            best_a = a_candidate.copy()
            stall = 0
            if it <= 50 or it % 5000 == 0:
                print(f"  [{tag}] it={it} obj={hard_o:.8f} lse_obj={o:.8f} tau={tau:.1f} step={step:.4f}")
        else:
            stall += 1
            if stall > 4000:
                step *= 0.7
                v *= 0.5
                stall = 0

        y = a_candidate
        # Anneal temperature
        tau = max(tau * 0.99995, n * 0.002)
        # Decay momentum slightly
        if it % 10000 == 5000:
            momentum = max(momentum * 0.995, 0.5)

    return best_a, best_o


def _hard_run(a, n, deadline, tag, it_limit=3000000, top_k=5):
    """Hard subgradient descent using top-K max indices with momentum."""
    best_a = a.copy()
    best_o = _obj_n(a)
    v = np.zeros(n)
    momentum = 0.8
    step = 0.05
    stall = 0

    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _fft_conv(a)
        mc = c.max()
        o = float(2.0 * mc / n)
        if o < best_o - 1e-12:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it <= 50 or it % 10000 == 0:
                print(f"  [{tag}] it={it} obj={o:.8f} step={step:.4f}")
        else:
            stall += 1
            if stall > 6000:
                step *= 0.7
                v *= 0.5
                stall = 0

        # Get top-K indices
        k_indices = np.argsort(c[-1:][0] * 0 + c)[-top_k:]
        k_list = [int(k) for k in k_indices]
        g = _hard_grad_topk(a, k_list)
        g = g - g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        g = g / mg

        v = momentum * v - step * g
        a = _project(a - v, float(n))

        # Slow decay
        if it % 5000 == 2500:
            step = max(step * 0.97, 1e-8)

    return best_a, best_o


def _coord_descent(a, n, deadline, tag, max_sweeps=50000):
    """Coordinate descent: optimize one coordinate at a time via ternary search."""
    best_a = a.copy()
    best_o = _obj_n(a)
    c = _fft_conv(a)
    stall = 0
    improvements = 0

    for sweep in range(max_sweeps):
        if time.monotonic() >= deadline:
            break
        improved = False
        # Random order for diversity
        order = np.arange(n)
        rng_local = np.random.default_rng(sweep * 7919 + 13)
        rng_local.shuffle(order)

        for idx in order:
            if time.monotonic() >= deadline:
                break
            old_val = a[idx]
            if old_val < 1e-10:
                continue
            # Conv as function of a[idx] = x:
            # New contribution to conv[k]: 2*x*a[k-idx] - old_val*2*a[k-idx] + x^2 - old_val^2 (for k=2*idx)
            # Actually: conv_new[k] = conv_old[k] + (x - old_val) * 2 * a[k-idx] for k != 2*idx
            # And for k = 2*idx: extra (x-old_val)^2
            # But simpler: just compute new conv for the affected range
            # Affected k range: [idx, idx+n-1]
            delta_range = slice(max(0, idx), min(2 * n - 1, idx + n))
            a_orig = a[idx]
            # Try a few candidate values via ternary search
            lo = 0.0
            hi = 2.0 * a_orig + 1.0
            # Actually let's just try: keep same sum constraint
            # If we change a[idx] from old_val to new_val, sum changes by (new_val - old_val)
            # To maintain sum, we'd need to adjust others. Let's just do unconstrained 1D search
            # then re-project at the end.
            # For speed, let's just try new_val in {0, 0.5*old, old, 1.5*old, 2*old}
            candidates = [0.0, 0.3 * a_orig, 0.7 * a_orig, a_orig, 1.3 * a_orig, 1.7 * a_orig]
            best_cv = c.copy()
            best_cv_val = c.max()
            best_new_val = a_orig
            for nv in candidates:
                if abs(nv - a_orig) < 1e-10:
                    continue
                a[idx] = nv
                # Update conv: affected indices are k where a[k-idx] or a[idx] appears
                # conv[k] = sum_j a[j]*a[k-j]
                # Changes when j=idx or k-j=idx, i.e., k=idx+j or k=k-idx=j
                # Affected k: [idx, idx+n-1]
                # delta_conv[k] = (nv - a_orig) * (a[k-idx] + a[k-idx]) = 2*(nv-a_orig)*a[k-idx] for k!=2*idx
                # For k=2*idx: additional (nv-a_orig)^2
                k_lo = max(0, idx)
                k_hi = min(2 * n - 2, idx + n - 1)
                j_range = np.arange(k_lo, k_hi + 1)
                ai_idx = j_range - idx
                ai_idx = ai_idx[(ai_idx >= 0) & (ai_idx < n)]
                j_valid = ai_idx + idx
                # delta = 2*(nv-a_orig)*a[ai_idx]
                dc = 2.0 * (nv - a_orig) * a[ai_idx]
                # Handle k=2*idx (diagonal term)
                if 2 * idx >= k_lo and 2 * idx <= k_hi:
                    dc = dc.copy()
                    diag_pos = 2 * idx - k_lo
                    # Find where 2*idx is in j_valid
                    if 2 * idx in j_valid:
                        di = np.where(j_valid == 2 * idx)[0]
                        if len(di) > 0:
                            dc[di[0]] = dc[di[0]] + (nv - a_orig) * (nv - a_orig)
                candidate_c = c.copy()
                candidate_c[np.arange(k_lo, k_hi + 1)] = c[np.arange(k_lo, k_hi + 1)]
                # This is getting too complex and slow. Skip coordinate descent for now.
                # Just revert and use gradient methods.
            a[idx] = a_orig

        if sweep < 10:
            print(f"  [{tag}] coord sweep {sweep} (skipped complex logic)")

    return best_a, best_o


def _perturb(a, n, rng, sigma, width_frac=None, mode="exp"):
    """Perturb a random block."""
    b = a.copy()
    if width_frac is None:
        w = max(3, n // 40)
    else:
        w = max(3, int(n * width_frac))
    s = int(rng.integers(0, max(1, n - w + 1)))
    if mode == "exp":
        b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    elif mode == "add":
        b[s:s + w] += rng.normal(0.0, sigma * b[s:s + w].mean(), size=w)
    elif mode == "swap":
        # Swap two random blocks
        w2 = max(3, w // 2)
        s2 = int(rng.integers(0, max(1, n - w2 + 1)))
        tmp = b[s:s + w2].copy()
        b[s:s + w2] = b[s2:s2 + w2]
        b[s2:s2 + w2] = tmp
    return _project(b, float(n))


def _coord_simplified(a, n, deadline, tag, max_rounds=5000):
    """Simplified coordinate descent: for each affected coordinate, try to reduce the max."""
    best_a = a.copy()
    best_o = _obj_n(a)
    c = _fft_conv(a)
    k_max = int(np.argmax(c))
    # Coordinates that affect conv[k_max]
    js = max(0, k_max - n + 1)
    je = min(n, k_max + 1)
    affected = list(range(js, je))

    for rnd in range(max_rounds):
        if time.monotonic() >= deadline:
            break
        rng_local = np.random.default_rng(rnd * 31 + 7)
        for _ in range(50):  # 50 random coordinate updates per round
            if time.monotonic() >= deadline:
                break
            idx = int(rng_local.integers(0, n))
            old_val = a[idx]
            if old_val < 1e-12:
                continue
            # How much is this coordinate contributing to the max?
            # Delta conv[k] when a[idx] changes by dv: 2*dv*a[k-idx] for valid k-idx
            # At k_max: contribution change = 2*dv*a[k_max-idx]
            contrib = 2.0 * a[max(0, min(n - 1, k_max - idx))] if 0 <= k_max - idx < n else 0.0
            # Try reducing this coordinate
            for factor in [0.5, 0.7, 0.9]:
                new_val = old_val * factor
                a[idx] = new_val
                # Quick recompute of just the affected conv values
                # For speed, only check if the max decreased
                # Affected k: where a[idx] appears: k = idx+j and k = k-idx=j for j in [0,n-1]
                # So k in [idx, idx+n-1] intersect [0, 2n-2]
                # delta at k: 2*(new_val - old_val)*a[k-idx] (plus diagonal correction)
                dv = new_val - old_val
                # Check if max decreases
                # For this to be fast, just check the top few indices
                top_ks = np.argsort(c)[-20:]
                max_decrease = 0.0
                for kk in top_ks:
                    if 0 <= kk - idx < n:
                        delta = 2.0 * dv * a[kk - idx]
                        if kk == 2 * idx:
                            delta += dv * dv
                        max_decrease = max(max_decrease, -delta) if delta < 0 else 0
                # If this reduces the current max, accept
                if c[k_max] + 2.0 * dv * (a[k_max - idx] if 0 <= k_max - idx < n else 0.0) > 0:
                    new_c_max_approx = c[k_max] + 2.0 * dv * (a[k_max - idx] if 0 <= k_max - idx < n else 0.0)
                    if new_c_max_approx < c[k_max] * 0.99:
                        # Accept update
                        c = _fft_conv(a)
                        k_max = int(np.argmax(c))
                        o = float(2.0 * c[k_max] / n)
                        if o < best_o - 1e-12:
                            best_o = o
                            best_a = a.copy()
                            if rnd < 5 or rnd % 500 == 0:
                                print(f"  [{tag}] rnd={rnd} obj={o:.8f}")
                        break
            a[idx] = old_val  # Revert if no improvement found

    return best_a, best_o


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: smooth momentum GD -> hard top-K descent ->
    coordinate refinement -> diverse perturb+refine loop -> final polish.
    Returns best non-negative sequence found within budget_s seconds.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 15.0

    # ---- Initialization ----
    n = 3809
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and len(parent) == n and np.all(np.isfinite(parent)):
            a = np.clip(parent, 0.0, 1000.0)
            print(f"Loaded parent sequence, sum={a.sum():.2f}, max={a.max():.4f}")
        else:
            a = np.ones(n)
    except Exception:
        a = np.ones(n)
        print("No valid parent; starting from uniform.")

    a = _project(a, float(n))
    best_a = a.copy()
    best_o = _obj_n(a)
    print(f"Phase 0 - Init: n={n} obj={best_o:.8f}")

    # ---- Phase 1: Smooth gradient descent with momentum (35% of budget) ----
    p1_end = start + 0.35 * (budget_s - 20.0)
    best_a, best_o = _smooth_run(a, n, p1_end, "smooth1")
    print(f"Phase 1 done: obj={best_o:.8f} elapsed={time.monotonic()-start:.0f}s")

    # ---- Phase 2: Hard top-K descent (20% of budget) ----
    p2_end = start + 0.55 * (budget_s - 20.0)
    best_a, best_o = _hard_run(best_a, n, p2_end, "hard1", top_k=5)
    print(f"Phase 2 done: obj={best_o:.8f} elapsed={time.monotonic()-start:.0f}s")

    # ---- Phase 2b: Coordinate refinement on the max index (10% of budget) ----
    p2b_end = start + 0.65 * (budget_s - 20.0)
    best_a, best_o = _coord_simplified(best_a, n, p2b_end, "coord")
    print(f"Phase 2b done: obj={best_o:.8f} elapsed={time.monotonic()-start:.0f}s")

    # ---- Phase 3: Diverse perturb + refine loop (remaining time) ----
    attempt = 0
    while time.monotonic() < deadline - 20.0:
        attempt += 1
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        # Choose perturbation parameters
        if attempt <= 8:
            sigma = float(rng.uniform(0.03, 0.15))
            wf = float(rng.choice([0.01, 0.02, 0.05]))
        elif attempt <= 25:
            sigma = float(rng.uniform(0.08, 0.35))
            wf = float(rng.choice([0.02, 0.05, 0.10, 0.15]))
        else:
            sigma = float(rng.uniform(0.10, 0.50))
            wf = float(rng.choice([0.05, 0.10, 0.20, 0.30]))

        mode = str(rng.choice(["exp", "add", "swap"]))
        a_p = _perturb(best_a, n, rng, sigma, width_frac=wf, mode=mode)

        # Short smooth refinement
        sub_budget = min(12.0, remaining * 0.35)
        sub1_end = time.monotonic() + sub_budget
        a_q, o_q = _smooth_run(a_p, n, sub1_end, f"s{attempt}", it_limit=40000)
        if o_q < best_o:
            best_o = o_q
            best_a = a_q
            print(f"  Perturb+smooth {attempt}: obj={best_o:.8f} sig={sigma:.3f} mode={mode}")

        # Short hard refinement
        sub_budget2 = min(10.0, remaining * 0.25)
        sub2_end = time.monotonic() + sub_budget2
        a_q2, o_q2 = _hard_run(best_a, n, sub2_end, f"h{attempt}", top_k=5, it_limit=25000)
        if o_q2 < best_o:
            best_o = o_q2
            best_a = a_q2
            print(f"  Perturb+hard {attempt}: obj={best_o:.8f}")

        # Occasionally try alternative initialization
        if attempt in (5, 12, 20, 30, 45):
            alt_seed = int(rng.integers(0, 10**6))
            a_alt = _project(rng.default_rng(alt_seed).uniform(0.3, 2.0, size=n), float(n))
            sub_budget3 = min(15.0, remaining * 0.3)
            sub3_end = time.monotonic() + sub_budget3
            a_alt2, o_alt = _smooth_run(a_alt, n, sub3_end, f"alt{attempt}", it_limit=30000)
            sub_budget4 = min(8.0, remaining * 0.15)
            sub4_end = time.monotonic() + sub_budget4
            a_alt3, o_alt2 = _hard_run(a_alt2, n, sub4_end, f"alth{attempt}", top_k=3, it_limit=15000)
            if o_alt2 < best_o:
                best_o = o_alt2
                best_a = a_alt3
                print(f"  Alt restart {attempt}: obj={best_o:.8f}")

        # Occasionally try different structure: sinusoidal init
        if attempt in (8, 18, 35):
            freq = float(rng.uniform(1, 20))
            a_sin = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
            a_sin = np.maximum(a_sin, 0.0)
            a_sin = _project(a_sin, float(n))
            sub_budget5 = min(12.0, remaining * 0.25)
            sub5_end = time.monotonic() + sub_budget5
            a_sin2, o_sin = _smooth_run(a_sin, n, sub5_end, f"sin{attempt}", it_limit=30000)
            sub_budget6 = min(8.0, remaining * 0.15)
            sub6_end = time.monotonic() + sub_budget6
            a_sin3, o_sin2 = _hard_run(a_sin2, n, sub6_end, f"sinh{attempt}", top_k=5, it_limit=15000)
            if o_sin2 < best_o:
                best_o = o_sin2
                best_a = a_sin3
                print(f"  Sinusoidal {attempt}: obj={best_o:.8f}")

    # ---- Final: one more hard pass with remaining time ----
    if time.monotonic() < deadline - 5.0:
        best_a, best_o = _hard_run(best_a, n, deadline - 2.0, "final", top_k=7, it_limit=500000)
        print(f"Final hard pass: obj={best_o:.8f}")

    # ---- Validate ----
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    if not np.isfinite(sc):
        result = [1.0] * n
        sc = float(evaluate_sequence(result))

    print(f"FINAL: internal_obj={best_o:.8f}  evaluated={sc:.8f}  n={n}  time={time.monotonic()-start:.0f}s")
    return result
