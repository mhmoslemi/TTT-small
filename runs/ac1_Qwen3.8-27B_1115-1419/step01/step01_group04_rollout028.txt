Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using log-sum-exp smoothed
objective with Adam optimizer in log-space parameterization.
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0, out=c)
    return c

def score_of(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (S * S)

def smooth_grad(x, tau_K):
    """
    x: log-space variables
    tau_K: number of "active" coefficients (tau = max(c)/tau_K)
    Returns: (objective, gradient_wrt_x, a, c)
    """
    n = len(x)
    # Convert to probability
    x_centered = x - x.max()
    a_raw = np.exp(x_centered)
    S = a_raw.sum()
    a = a_raw / S
    
    c = fast_autocorr(a)
    c_max = c.max()
    
    if c_max < 1e-30:
        return 0.0, np.zeros(n), a, c
    
    tau = c_max / tau_K
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    # Objective
    obj = c_max + tau * np.log(w_sum)  # = c_max/tau * tau + tau * log(sum(exp(shifted)))
    # Actually: lse = c_max/tau + log(sum(exp((c-c_max)/tau)))
    # = (c_max + tau * log(sum(exp((c-c_max)/tau)))) / tau
    # The raw lse is c_max/tau + log(w_sum) [before normalization]
    # Let me just track the unnormalized version for monotonicity
    
    # Gradient of sum_k w_k * c_k w.r.t. a
    L = 2 * n - 1
    N2 = 1 << (L - 1).bit_length()
    W = np.fft.rfft(w, N2)
    Ar = np.fft.rfft(a[::-1], N2)
    conv = np.fft.irfft(W * Ar, N2)[:L]
    g_a = 2.0 * conv[:n][::-1]
    
    # Gradient w.r.t. x
    dot_ga_a = np.dot(g_a, a)
    g_x = a * (g_a - dot_ga_a)
    
    return obj, g_x, a, c

def adam_step(x, g, m, v, t, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1 - b1) * g
    v = b2 * v + (1 - b2) * g * g
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v

def optimize_n(n, budget_s, seed):
    """Optimize for a given n within the time budget."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    # Generate starts
    starts = []
    
    # Uniform
    starts.append(np.zeros(n))
    
    # Random gaussian starts
    for sigma in [0.1, 0.3, 0.5, 1.0]:
        starts.append(rng.normal(0, sigma, n))
    
    # Smooth (low freq) starts
    for _ in range(3):
        x = np.zeros(n)
        t = np.arange(n) / n
        for m in range(1, 6):
            amp = rng.uniform(0, 1.0) / m
            ph = rng.uniform(0, 2 * np.pi)
            x += amp * np.cos(2 * np.pi * m * t + ph)
        starts.append(x)
    
    # Block starts
    for _ in range(3):
        nblocks = rng.integers(3, 15)
        edges = np.sort(rng.choice(range(1, n), nblocks - 1, replace=False))
        boundaries = np.concatenate([[0], edges, [n]])
        x = np.zeros(n)
        for i in range(nblocks):
            x[boundaries[i]:boundaries[i+1]] = rng.normal(0, 0.5)
        starts.append(x)
    
    best_score = float('inf')
    best_a = None
    
    n_starts = len(starts)
    time_per_start = budget_s / n_starts
    
    for si in range(n_starts):
        t_start = time.monotonic()
        t_end = t_start + time_per_start
        
        x = starts[si].copy()
        x = x - x.mean()  # center for numerical stability
        
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        
        lr = 0.1
        tau_K = 50.0  # start with ~50 active coefficients
        tau_K_min = 3.0
        
        local_best_score = float('inf')
        local_best_a = None
        local_best_x = x.copy()
        
        no_improve = 0
        eval_count = 0
        
        while time.monotonic() < t_end and no_improve < 2000:
            t += 1
            obj, g_x, a, c = smooth_grad(x, tau_K)
            
            if not np.all(np.isfinite(g_x)):
                no_improve += 100
                continue
            
            # Compute actual score
            score = 2.0 * n * c.max()  # since sum(a) = 1
            eval_count += 1
            
            if score < local_best_score:
                local_best_score = score
                local_best_a = a.copy()
                local_best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adam update
            x, m, v = adam_step(x, g_x, m, v, t, lr)
            
            # Decay tau_K
            tau_K = max(tau_K_min, tau_K * 0.9995)
            
            # Occasionally reset Adam state
            if t % 5000 == 0:
                m *= 0.5
                v *= 0.5
            
            # Adaptive learning rate
            if no_improve > 500:
                lr = max(0.001, lr * 0.9)
            elif no_improve == 0:
                lr = min(1.0, lr * 1.01)
        
        if local_best_score < best_score:
            best_score = local_best_score
            best_a = local_best_a
        
        if time.monotonic() > t0 + budget_s:
            break
    
    return best_a, best_score

"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using log-sum-exp smoothed
objective with Adam optimizer in log-space parameterization.

Key ideas:
1. Parameterize a = softmax(x) to ensure non-negativity automatically.
2. Use log-sum-exp as a smooth approximation of the max autocorrelation coefficient.
3. Adam optimizer with adaptive learning rate and temperature annealing.
4. Multiple diverse restarts (uniform, random, smooth, block, parent-based).
5. Try multiple sequence lengths and pick the best.
6. Final polish phase using hard-max subgradient for last-mile improvement.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute aperiodic autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.empty(0)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def score_from_autocorr(a, c):
    """Compute 2n*max(c)/(sum(a))^2 given precomputed autocorrelation."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def compute_score(a):
    """Full score computation."""
    c = fast_autocorr(a)
    return score_from_autocorr(a, c)


def smooth_objective_grad(x, tau_K):
    """
    Compute smooth log-sum-exp objective and gradient w.r.t. x.
    a = softmax(x), i.e., a_i = exp(x_i) / sum(exp(x_j)).
    tau = max(c) / tau_K (adaptive temperature).
    
    Returns: (objective, grad_x, a, c)
    """
    n = len(x)
    # Numerically stable softmax
    xc = x - x.max()
    ex = np.exp(xc)
    S = ex.sum()
    a = ex / S  # probability vector, sum(a) = 1
    
    c = fast_autocorr(a)
    c_max = c.max()
    
    if c_max < 1e-30:
        return 0.0, np.zeros(n), a, c
    
    tau = c_max / tau_K
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -60.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Objective: c_max + tau * log(w_sum_original) where w_sum_original = sum(exp(shifted))
    # We already normalized w, so w_sum_original = w_sum * (post-normalization sum = 1)
    # Actually: sum(exp(shifted)) = w_sum (before normalization)
    # lse = c_max/tau + log(sum(exp(shifted)))
    # For tracking purposes, just use c_max as a proxy (monotone relationship)
    obj = c_max  # simplified; the actual smooth value would be c_max/tau + log(sum exp)
    
    # Gradient: g_a[i] = 2 * sum_j w[i+j] * a[j]
    # Compute via FFT: g_a[i] = 2 * (w * a_reversed)[n-1+i]
    L = 2 * n - 1
    conv_len = L + n - 1  # = 3n - 2
    N_fft = 1 << (conv_len - 1).bit_length()
    
    W = np.fft.rfft(w, N_fft)
    Ar = np.fft.rfft(a[::-1], N_fft)
    h = np.fft.irfft(W * Ar, N_fft)
    g_a = 2.0 * h[n - 1: 2 * n - 1]
    
    # Gradient w.r.t. x: g_x[i] = a[i] * (g_a[i] - dot(g_a, a))
    dot_ga_a = np.dot(g_a, a)
    g_x = a * (g_a - dot_ga_a)
    
    return obj, g_x, a, c


def adam_update(x, g, m, v, t, lr):
    """In-place Adam update. Returns new x, m, v."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1.0 - b1) * g
    v = b2 * v + (1.0 - b2) * g * g
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x, m, v


def topK_gradient(a, c, K):
    """
    Subgradient of max(c) using top-K coefficients.
    Returns gradient w.r.t. a (before log-space transformation).
    """
    n = len(a)
    L = 2 * n - 1
    K = max(1, min(K, L))
    
    if K == 1:
        idx = np.array([np.argmax(c)])
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    # For each active k, dc_k/da_i = 2*a_{k-i} for valid i
    # g_a[i] = 2/K * sum_{k in active} a[k-i] (for valid k-i)
    # This is (2/K) * cross-correlation of indicator(active) with a
    
    # Build sparse weight vector
    w = np.zeros(L)
    w[idx] = 1.0 / K
    
    conv_len = L + n - 1
    N_fft = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, N_fft)
    Ar = np.fft.rfft(a[::-1], N_fft)
    h = np.fft.irfft(W * Ar, N_fft)
    g_a = 2.0 * h[n - 1: 2 * n - 1]
    
    return g_a


def polish_hardmax(a, budget_s, seed):
    """
    Final polish using hard-max subgradient in log-space.
    a is a probability vector (sum=1).
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    x = np.log(a + 1e-30)
    x = x - x.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.05
    best_a = a.copy()
    best_score = compute_score(a)
    stagnant = 0
    
    while time.monotonic() - t0 < budget_s:
        t += 1
        a = np.exp(x - x.max())
        a = a / a.sum()
        
        c = fast_autocorr(a)
        score = 2.0 * n * c.max()  # sum(a)=1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 100:
            break
        
        # Hard max gradient (top 5)
        K = min(5, 2 * n - 1)
        g_a = topK_gradient(a, c, K)
        
        # Transform to log-space gradient
        dot_ga_a = np.dot(g_a, a)
        g_x = a * (g_a - dot_ga_a)
        
        if not np.all(np.isfinite(g_x)):
            break
        
        x, m, v = adam_update(x, g_x, m, v, t, lr)
        
        if stagnant > 50:
            lr = max(0.001, lr * 0.9)
    
    return best_a, best_score


def generate_starts(n, rng, parent_seq=None):
    """Generate diverse starting points in log-space."""
    starts = []
    
    # Uniform
    starts.append(np.zeros(n))
    
    # Random Gaussian
    for sigma in [0.2, 0.5, 1.0, 2.0]:
        starts.append(rng.normal(0, sigma, n))
    
    # Smooth low-frequency
    for _ in range(4):
        x = np.zeros(n)
        t = np.arange(n) / n
        for k in range(1, 8):
            amp = rng.uniform(0, 1.5) / (k ** 0.5)
            ph = rng.uniform(0, 2 * np.pi)
            x += amp * np.cos(2 * np.pi * k * t + ph)
        starts.append(x)
    
    # Block structure
    for nblocks in [3, 5, 8, 12, 20]:
        if nblocks > n // 2:
            continue
        edges = np.sort(rng.choice(range(1, n), min(nblocks - 1, n - 1), replace=False))
        boundaries = np.concatenate([[0], edges.astype(int), [n]])
        x = np.zeros(n)
        for i in range(len(boundaries) - 1):
            x[boundaries[i]:boundaries[i + 1]] = rng.normal(0, 0.7)
        starts.append(x)
    
    # Ramp / quadratic shapes
    t = np.linspace(0, 1, n)
    for alpha in [-2, -1, 1, 2]:
        starts.append(alpha * (2 * t - 1))
        starts.append(alpha * (2 * t - 1) ** 2)
    
    # From parent
    if parent_seq is not None and len(parent_seq) > 0:
        p = np.asarray(parent_seq, dtype=np.float64)
        p = np.maximum(p, 1e-10)
        # Upsample/resize to length n
        if len(p) != n:
            x_old = np.linspace(0, 1, len(p))
            x_new = np.linspace(0, 1, n)
            p_resized = np.interp(x_new, x_old, p)
        else:
            p_resized = p
        p_resized = np.maximum(p_resized, 1e-10)
        x_parent = np.log(p_resized)
        x_parent = x_parent - x_parent.mean()
        starts.append(x_parent)
        # Perturbed parent
        starts.append(x_parent + rng.normal(0, 0.1, n))
        starts.append(x_parent + rng.normal(0, 0.3, n))
    
    return starts

    # Ensure sum is reasonable
    s = seq.sum()
    if s < 0.01:
        seq = np.ones(512)
    
    result = [float(x) for x in seq]
    print(f"Returning sequence of length {len(result)}, fast score ~{global_best_score:.6f}")
    
    return result
