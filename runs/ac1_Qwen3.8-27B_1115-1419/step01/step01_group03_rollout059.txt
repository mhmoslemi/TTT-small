```python
def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def compute_gradient(a, w):
    """Compute gradient of sum_k w[k]*c[k] where c = convolve(a, a)."""
    n = len(a)
    conv_wa = fftconvolve(w, a, mode='full')
    g2 = conv_wa[:n].copy()
    
    w_rev = w[::-1]
    conv_wrev_a = fftconvolve(w_rev, a, mode='full')
    g1 = conv_wrev_a[n-1:2*n-1][::-1].copy()
    
    return g1 + g2

def mirror_descent(a0, duration, rng, beta_start=5.0, beta_end=500.0, eta_initial=0.1, score_interval=50):
    """Run mirror descent to minimize max convolution coefficient."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration
    
    eta = eta_initial
    it = 0
    no_improve = 0
    stagnant = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        c = np.convolve(a, a)
        
        # Adaptive beta
        beta = beta_start + (beta_end - beta_start) * (1.0 - np.exp(-it / 50000.0))
        
        # Softmax weights
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient
        g = compute_gradient(a, w)
        
        # Mirror descent update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-300)
        s = a_new.sum()
        if s > 1e-300:
            a_new /= s
        a = a_new
        it += 1
        
        # Score check
        if it % score_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                stagnant = 0
            else:
                no_improve += 1
            
            # Adaptive step size
            if no_improve % 100 == 0:
                eta *= 0.95
                if eta < 0.001:
                    eta = 0.001
            
            # Restart if very stagnant
            if no_improve > 1000:
                stagnant += 1
                a = best_a + rng.normal(0, 0.005 / np.sqrt(n), n)
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                no_improve = 0
                eta = eta_initial / (stagnant + 1)
            
            if it % 10000 == 0:
                print(f"    MD it={it}, best_score={best_score:.7f}, beta={beta:.1f}, eta={eta:.5f}, no_improve={no_improve}")
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score
```

```python
def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    fft_size = 1
    while fft_size < n:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:n]
```

```python
from scipy.sparse import lil_matrix

def build_lp_matrix(a, n):
    """Build the constraint matrix A for the LP.
    A has shape (2n-1 + 1, n + 1) where the last row is for z.
    Actually, let me separate: the convolution constraints form a matrix of shape (2n-1, n).
    """
    # For constraint k, row k of the matrix has a[k-i] at column i (for valid i)
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            A[k, i] = a[k - i]
    return A.tocsr()
```

def fast_score(a):
    """Compute the score without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def fft_conv2(a, n):
    """Compute convolve(a, a) using FFT."""
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_grad_fft(a, w, n):
    """Compute gradient of sum_k w[k]*c[k] where c = conv(a,a), using FFT."""
    size = 1
    while size < 3 * n - 2:
        size *= 2
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    
    # g2[i] = sum_k w[k]*a[i-k] = convolve(w, a)[i]
    g2 = np.fft.irfft(fw * fa, size)[:n]
    
    # g1[i] = sum_k w[k]*a[k-i] = cross-correlation
    # Using: correlate(w, a)[i] = convolve(flip(w), a)[2n-2-i]
    fw_rev = np.fft.rfft(w[::-1], size)
    g1_full = np.fft.irfft(fw_rev * fa, size)
    g1 = g1_full[n-1:2*n-1][::-1]
    
    return g1 + g2
