"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines L-BFGS with LSE smoothing, exact subgradient methods,
adaptive step sizing, and multi-start diversity.
"""

import numpy as np
import time
from scipy.optimize import minimize


def compute_conv(a):
    """Compute a * a (full convolution)."""
    return np.convolve(a, a)


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = compute_conv(a)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def lse_smoothing(conv, tau):
    """Compute log-sum-exp smoothed max."""
    c_max = conv.max()
    if c_max < 1e-300:
        return c_max, np.zeros(len(conv))
    shifted = (conv - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    smooth_max = c_max + tau * np.log(total)
    weights = exp_shifted / total
    return smooth_max, weights


def lse_gradient(weights, a):
    """Gradient of LSE-smoothed max w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def subgradient_topk(a, k=20):
    """Compute averaged subgradient using top-k active lags."""
    n = len(a)
    conv = compute_conv(a)
    max_val = conv.max()
    if max_val < 1e-300:
        return np.zeros(n), max_val
    # Find top-k lags
    top_idx = np.argsort(conv)[::-1][:k]
    g = np.zeros(n)
    for idx in top_idx:
        k_val = idx
        lo = max(0, k_val - n + 1)

        lo = max(0, int(idx) - n + 1)
        hi = min(n - 1, int(idx))
        g[lo:hi + 1] += 2.0 * a[lo:hi + 1][::-1]
    g /= len(top_idx)
    return g, max_val


def proximal_step(a, g, step):
    """Subgradient step + projection to {x >= 0, sum preserved}."""
    target_sum = a.sum()
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-300:
        a_new *= target_sum / s
    else:
        a_new = a.copy()
    return a_new


def multiplicative_step(a, g, step):
    """Mirror descent multiplicative step."""
    target_sum = a.sum()
    g_clipped = np.clip(g, -3.0 / step, 3.0 / step)
    a_new = a * np.exp(-step * g_clipped)
    a_new = np.maximum(a_new, 1e-300)
    a_new *= target_sum / a_new.sum()
    return a_new


def run_lbfgs_phase(a_init, time_budget):
    """L-BFGS with geometric-decay LSE temperature schedule."""
    n = len(a_init)
    x = np.log(np.maximum(a_init, 1e-15))
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    deadline = time.monotonic() + time_budget

    tau_start = 1.0 / n
    tau_end = 0.0005 / n
    n_levels = 12
    taus = tau_start * (tau_end / tau_start) ** (np.arange(n_levels) / max(1, n_levels - 1))

    for tau in taus:
        if time.monotonic() > deadline:
            break

        def obj(x_):
            a_ = np.exp(x_ - x_.max())
            a_ /= a_.sum()
            conv = fast_convolve(a_, a_)
            sm, _ = lse_smooth_max(conv, tau)
            return sm

        def grad(x_):
            a_ = np.exp(x_ - x_.max())
            a_ /= a_.sum()
            conv = fast_convolve(a_, a_)
            _, w = lse_smooth_max(conv, tau)
            g_a = lse_grad_from_weights(w, a_)
            dot = np.dot(a_, g_a)
            return a_ * (g_a - dot)

        try:
            result = minimize(
                obj, x, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 400}
            )
            x = result.x
        except Exception:
            break

        a = np.exp(x - x.max())
        a /= a.sum()
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def adaptive_subgradient(a_init, time_budget, rng, use_mult=False, annealing=False):
    """Adaptive subgradient with optional multiplicative updates and annealing."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + time_budget

    step = 2.0 / n
    step_min = 1e-10 / n
    step_max = 20.0 / n
    patience = 0
    max_patience = 500
    temperature = 0.01 / n
    k_active = 30

    while time.monotonic() < deadline:
        if patience % 500 == 0:
            k_active = min(100, 30 + patience // 100)

        g, _ = subgradient_topk(a, k=k_active)

        if use_mult:
            a_new = multiplicative_step(a, g, step)
        else:
            a_new = proximal_step(a, g, step)

        score_new = compute_score(a_new)

        accept = False
        if score_new < best_score:
            accept = True
        elif annealing and temperature > 1e-8:
            delta = score_new - best_score
            if delta < 0.1:
                prob = np.exp(-delta / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
            patience = 0
            step = min(step * 1.02, step_max)
        else:
            patience += 1
            step = max(step * 0.93, step_min)
            temperature *= 0.995

            if patience >= max_patience:
                noise_scale = (0.002 + 0.01 * (patience // max_patience)) / n
                a_pert = a + rng.normal(0, noise_scale, n)
                a_pert = np.maximum(a_pert, 0)
                a_pert *= a.sum() / a_pert.sum()
                a = a_pert
                patience = 0
                step = 3.0 / n
                temperature = max(temperature, 0.005 / n)

    return best_a, best_score


def try_length_extension(a_init, budget, rng):
    """Try extending the sequence by a few elements and optimizing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    deadline = time.monotonic() + budget

    for ext in [5, 10, 20]:
        if time.monotonic() > deadline - 60:
            break
        n_new = n + ext
        a_new = np.zeros(n_new)
        a_new[:n] = a_init
        # Initialize new elements with small values based on neighbors
        for i in range(n, n_new):
            idx = min(i, n - 1)
            a_new[i] = a_init[idx] * 0.1
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Quick subgradient polish
        time_left = deadline - time.monotonic()
        if time_left > 30:
            a_pol, s_pol = adaptive_subgradient(a_new, min(60, time_left * 0.5), rng, use_mult=True)
            if s_pol < best_score:
                best_score = s_pol
                best_a = a_pol.copy()

    return best_a, best_score


def diverse_initialization(n, rng):
    """Generate a diverse random initialization."""
    choices = rng.integers(0, 8)
    if choices == 0:
        # Random Dirichlet
        return rng.dirichlet(np.ones(n) * 0.3)
    elif choices == 1:
        # Cosine-modulated
        x = 1.0 + 0.5 * np.cos(2.0 * np.pi * rng.integers(2, 8) * np.arange(n) / n)
        x = np.maximum(x, 0.01)
        return x / x.sum()
    elif choices == 2:
        # Linear ramp
        x = np.linspace(rng.uniform(0.1, 2.0), rng.uniform(0.1, 2.0), n)
        return np.abs(x) / np.abs(x).sum()
    elif choices == 3:
        # Two-block
        x = np.ones(n)
        split = rng.integers(n // 3, 2 * n // 3)
        x[:split] = rng.uniform(1.0, 3.0)
        x[split:] = rng.uniform(0.2, 1.0)
        return x / x.sum()
    elif choices == 4:
        # Sparse random
        x = np.where(rng.random(n) > 0.5, 1.0, 0.05)
        return x / x.sum()
    elif choices == 5:
        # Gaussian bump
        center = rng.uniform(n * 0.2, n * 0.8)
        width = rng.uniform(n * 0.05, n * 0.3)
        x = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        x[np.arange(n) < 0] = 0
        return x / x.sum()
    elif choices == 6:
        # Multi-frequency
        x = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            x *= 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
        x = np.maximum(x, 0.01)
        return x / x.sum()
    else:
        # Power law
        x = (np.arange(n) + 1) ** (-rng.uniform(0.1, 1.0))
        return x / x.sum()


def propose_candidate(budget_s=2400, **kwargs):
    """
    Multi-strategy optimizer: L-BFGS smoothing, adaptive subgradient with
    annealing, multiplicative updates, length extension, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 40  # Safety buffer

    rng = np.random.default_rng(2024)

    # Get initial sequence from parent
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        a0 = np.ones(4096, dtype=np.float64)

    n = len(a0)
    a0 = np.maximum(a0, 0)
    s0 = a0.sum()
    if s0 < 0.01:
        a0 = np.ones(n)
        s0 = n
    a0 = a0 / s0

    best_score = compute_score(a0)
    best_a = a0.copy()
    print(f"Initial: n={n}, score={best_score:.6f}")

    # Phase 1: L-BFGS with LSE smoothing on the parent
    t1_budget = min(500, (deadline - time.monotonic()) * 0.25)
    print(f"\n=== Phase 1: L-BFGS smoothing. Budget: {t1_budget:.0f}s ===")
    a1, s1 = run_lbfgs_phase(a0, t1_budget)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
        print(f"  L-BFGS improved: {best_score:.6f}")
    else:
        print(f"  L-BFGS: {s1:.6f} (no improvement)")

    # Phase 2: Adaptive subgradient (proximal) with annealing
    t2_budget = min(400, (deadline - time.monotonic()) * 0.2)
    print(f"\n=== Phase 2: Subgradient proximal + annealing. Budget: {t2_budget:.0f}s ===")
    a2, s2 = adaptive_subgradient(best_a.copy(), t2_budget, rng, use_mult=False, annealing=True)
    if s2 < best_score:
        best_score = s2
        best_a = a2.copy()
        print(f"  Proximal+anneal improved: {best_score:.6f}")
    else:
        print(f"  Proximal+anneal: {s2:.6f} (no improvement)")

    # Phase 3: Multiplicative subgradient
    t3_budget = min(400, (deadline - time.monotonic()) * 0.2)
    print(f"\n=== Phase 3: Multiplicative subgradient. Budget: {t3_budget:.0f}s ===")
    a3, s3 = adaptive_subgradient(best_a.copy(), t3_budget, rng, use_mult=True, annealing=True)
    if s3 < best_score:
        best_score = s3
        best_a = a3.copy()
        print(f"  Multiplicative improved: {best_score:.6f}")
    else:
        print(f"  Multiplicative: {s3:.6f} (no improvement)")

    # Phase 4: L-BFGS on the current best (second pass with refined init)
    t4_budget = min(300, (deadline - time.monotonic()) * 0.2)
    if t4_budget > 30:
        print(f"\n=== Phase 4: L-BFGS second pass. Budget: {t4_budget:.0f}s ===")
        a4, s4 = run_lbfgs_phase(best_a.copy(), t4_budget)
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()
            print(f"  Second L-BFGS improved: {best_score:.6f}")
        else:
            print(f"  Second L-BFGS: {s4:.6f} (no improvement)")

    # Phase 5: Try length extension
    t5_budget = min(200, (deadline - time.monotonic()) * 0.15)
    if t5_budget > 30:
        print(f"\n=== Phase 5: Length extension. Budget: {t5_budget:.0f}s ===")
        a5, s5 = try_length_extension(best_a.copy(), t5_budget, rng)
        if s5 < best_score:
            best_score = s5
            best_a = a5.copy()
            print(f"  Extension improved: {best_score:.6f} (n={len(best_a)})")
        else:
            print(f"  Extension: {s5:.6f} (no improvement)")

    # Phase 6: Diverse restarts
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n=== Phase 6: Diverse restarts. Budget: {remaining:.0f}s ===")
        n_restarts = min(5, int(remaining / 60))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Create diverse init by mixing best with random
            diverse = diverse_initialization(len(best_a), rng)
            mix = 0.5 * best_a + 0.5 * diverse
            mix = np.maximum(mix, 0)
            mix /= mix.sum()

            # Quick L-BFGS
            t_r = min(120, (deadline - time.monotonic()) * 0.4)
            a_r, s_r = run_lbfgs_phase(mix, t_r)
            if s_r < best_score:
                best_score = s_r
                best_a = a_r.copy()
                print(f"  Restart {r+1} improved: {best_score:.6f}")

            # Quick subgradient polish
            t_sr = min(60, (deadline - time.monotonic()) * 0.3)
            if t_sr > 10:
                a_sr, s_sr = adaptive_subgradient(a_r, t_sr, rng, use_mult=(r % 2 == 0), annealing=True)
                if s_sr < best_score:
                    best_score = s_sr
                    best_a = a_sr.copy()
                    print(f"  Restart {r+1} polish improved: {best_score:.6f}")

    # Phase 7: Final fine polishing
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Final polish. Budget: {remaining:.0f}s ===")
        a_final, s_final = adaptive_subgradient(best_a.copy(), remaining - 10, rng,
                                                use_mult=False, annealing=False)
        if s_final < best_score:
            best_score = s_final
            best_a = a_final.copy()

        # One more multiplicative pass
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_final2, s_final2 = adaptive_subgradient(best_a.copy(), remaining2 - 5, rng,
                                                      use_mult=True, annealing=False)
            if s_final2 < best_score:
                best_score = s_final2
                best_a = a_final2.copy()

    # Final random perturbation search
    for scale in [0.005, 0.002, 0.001, 0.0005]:
        if time.monotonic() > deadline - 5:
            break
        for _ in range(20):
            if time.monotonic() > deadline - 5:
                break
            a_pert = best_a + rng.normal(0, scale / len(best_a), len(best_a))
            a_pert = np.maximum(a_pert, 0)
            a_pert *= best_a.sum() / a_pert.sum()
            s_pert = compute_score(a_pert)
            if s_pert < best_score:
                best_score = s_pert
                best_a = a_pert.copy()

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL RESULT: score={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
    print(f"{'='*50}")

    return [float(val) for val in best_a]
