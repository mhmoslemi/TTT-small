def _softmax_weights(c, tau):
    cs = c - c.max()

def _grad_at_k(a, k, n):
    g = np.zeros(n, dtype=np.float64)
    js = max(0, k - n + 1)

def _grad_topk(a, c, n, k_top, tau):

    # Get top-k indices
    if k_top >= len(c):

    top_vals = c[top_indices]
    # Softmax weights over top-k

    tw = tw / tw.sum()
    # Weighted gradient

        je = min(n - 1, idx) + 1
        if js < je:

            g[js:je] += w * a[ai]
    g *= 2.0
    return g
