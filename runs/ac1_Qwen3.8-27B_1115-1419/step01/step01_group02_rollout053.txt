                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 15:
                noise = 0.04 * np.random.standard_normal(n)
                a = best_a + noise
                a = np.maximum(a, 0.0)
                a = _normalize(a, float(np.sum(best_a)))
                momentum[:] = 0
                lr = lr_init * 0.6
                stuck = 0
                c = _fft_conv(a, a)

    return best_a, best_score


def _lp_refine(f, deadline, max_iters=50):
    try:
        import cvxpy as cp
    except ImportError:
        return f, _score_fast(f)

    n = len(f)
    M0 = float(np.max(_fft_conv(f, f)))
    M = M0
    g = f.copy()
    best_g = f.copy()
    best_M = _score_fast(f)

    c_max = n - 1
    srs = list(range(c_max + 1)) + list(range(c_max, 2 * n - 2))
    scs = list(range(n)) + list(range(n - 1, -1, -2))
    data = (f[scs] ** 2).astype(np.float64)
    A_mat = sp.coo_matrix(
        (data, (srs, scs)), shape=(2 * n - 1, n)).tocsr()

    var = cp.Variable(n, nonneg=True)
    try:
        opt = 'MOSEK'
    except Exception:
        opt = 'CLARABEL'

    t = 0.12
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        con = [A_mat @ var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(var)), con)
        try:
            prob.solve(solver=opt)
        except Exception:
            try:
                prob.solve(solver='GLPK')
            except Exception:
                break
        if var.value is None:
            break
        g = np.maximum(var.value, 0.0)
        g = _normalize(g, float(np.sum(f)))
        new_f = (1 - t) * f + t * g
        new_M = float(np.max(_fft_conv(new_f, new_f)))
        if new_M < M + 1e-9:
            f = new_f
            M = new_M
            sc = _score_fast(f)
            if sc < best_M:
                best_M = sc
                best_g = f.copy()
            t = max(t * 0.95, 0.03)
        else:
            t *= 0.5
            if t < 0.01:
                break

    return best_g, best_M


def _make_init(kind, n, rng):
    if kind == 'uniform':
        return np.ones(n)
    if kind == 'random':
        return rng.random(n) + 0.05
    if kind == 'gaussian':
        x = np.linspace(-3, 3, n)
        return np.exp(-0.5 * x ** 2) + 0.05
    if kind == 'bimodal':
        x = np.linspace(0, 1, n)
        return 0.3 + np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    if kind == 'spectral':
        phases = rng.uniform(0, 2 * np.pi, n)
        sig = ifft(np.exp(1j * phases)).real
        return np.abs(sig) + 0.05
    if kind == 'cosine':
        x = np.arange(n)
        return 1.0 + 0.5 * np.cos(2 * np.pi * x / (n // 7 + 1))
    if kind == 'ramp':
        return 0.5 + 0.8 * np.linspace(0, 1, n)
    if kind == 'sawtooth':
        x = np.arange(n)
        return 1.0 + 0.3 * (2.0 * ((x % 17) / 17.0 - 0.5))
    return np.ones(n)


def propose_candidate(budget_s=2400):
    global _DEADLINE
    rng = np.random.default_rng(2025)
    _DEADLINE = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    n_default = 5582
    n = n_default
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            n = parent.size
            parent = _normalize(parent, np.sqrt(2.0 * n))
        else:
            parent = np.ones(n)
    except Exception:
        parent = np.ones(n_default)
        n = n_default

    target_sum = np.sqrt(2.0 * n)
    parent = _normalize(np.maximum(parent, 0), target_sum)

    best = parent.copy()
    best_score = _score_fast(best)
    print(f"Parent: score={best_score:.6f}, n={n}")

    kinds = ['uniform', 'random', 'gaussian', 'bimodal', 'spectral',
             'cosine', 'ramp', 'sawtooth']
    phase1_budget = min(40.0, (_DEADLINE - time.monotonic()) * 0.15)

    print(f"\nPhase 1: {len(kinds)} inits, ~{phase1_budget:.0f}s each")
    for kind in kinds:
        if time.monotonic() > _DEADLINE - 120:
            break
        init = _make_init(kind, n, rng)
        init = _normalize(np.maximum(init, 0), target_sum)
        lb, ls = _run_subgrad(init, time.monotonic() + phase1_budget,
                              lr_init=0.10, lr_decay=0.9998, beta=0.9,
                              top_k=3, eval_every=100, smooth_every=2000)
        if ls < best_score - 1e-8:
            best_score = ls
            best = lb.copy()
            print(f"  {kind}: {best_score:.6f} *")
        else:
            print(f"  {kind}: {ls:.6f}")

    print(f"\nPhase 2: Hyperparam sweep, best={best_score:.6f}")
    remaining = _DEADLINE - time.monotonic()
    configs = [
        (0.06, 0.92, 5), (0.03, 0.95, 8),
        (0.08, 0.88, 3), (0.04, 0.93, 2),
    ]
    per = min(55.0, max(10.0, (remaining - 60) / len(configs)))
    for ci, (lri, be, tk) in enumerate(configs):
        if time.monotonic() > _DEADLINE - 60:
            break
        a0 = best + 0.02 * np.random.standard_normal(n)
        a0 = np.maximum(a0, 0)
        a0 = _normalize(a0, target_sum)
        lb, ls = _run_subgrad(a0, time.monotonic() + per,
                              lr_init=lri, lr_decay=0.9999, beta=be,
                              top_k=tk, eval_every=120, smooth_every=2500)
        if ls < best_score - 1e-8:
            best_score = ls
            best = lb.copy()
            print(f"  cfg{ci}: {best_score:.6f} *")
        else:
            print(f"  cfg{ci}: {ls:.6f}")

    print(f"\nPhase 3: Refinement, best={best_score:.6f}")
    if time.monotonic() < _DEADLINE - 30:
        lb, ls = _run_subgrad(best, _DEADLINE - 30,
                              lr_init=0.02, lr_decay=0.99995, beta=0.95,
                              top_k=2, eval_every=80, smooth_every=4000)
        if ls < best_score - 1e-8:
            best_score = ls
            best = lb.copy()
            print(f"  refined: {best_score:.6f}")
        else:
            print(f"  refined: {ls:.6f}")

    if time.monotonic() < _DEADLINE - 10 and n <= 8000:

                                    lr_init=0.015, lr_decay=0.99995, beta=0.93,
                                    top_k=3, eval_every=100, smooth_every=3000)

                print(f"  post-LP: {best_score:.6f} *")

    if time.monotonic() < _DEADLINE - 50:

            ts = np.sqrt(2.0 * n_alt)

            if n_alt > n:

                sig[:n] = best
            else:
                sig = best[:n_alt]
            a_alt = _normalize(np.maximum(sig, 0), ts)
            bt = min(_DEADLINE - 50, time.monotonic() + 100)

            else:

    b = np.clip(best, 0.0, 1000.0)
    b = [float(v) for v in b]
    final = evaluate_sequence(b)
    print(f"\nFinal: evaluate_sequence={final:.6f}, tracked={best_score:.6f}, n={n}")

    if not np.isfinite(final) or final > 10:

        final = evaluate_sequence(b)
        print(f"  Fallback parent: {final:.6f}")

    return b
