    a = np.asarray(a, dtype=np.float64)

    fa = np.fft.rfft(a, N)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative step-function heights.
Primary optimizer: LP-based water-filling (solving max sum(b) s.t. conv(a,b)<=M, b>=0).
Supplemented by multiplicative gradient refinement for larger sequence lengths.
Multi-resolution: optimize at n=512, 1024, 2048, then gradient-refine at best length.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """FFT-based linear convolution of a and b."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    L = n + m - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:L]
    return np.maximum(c, 0.0)


def fft_autoconv(a):
    """FFT-based autocorrelation conv(a,a)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(c, 0.0)


def compute_score(a):
    """Fast score: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fft_autoconv(a)
    if c.size == 0:
        return float('inf')
    return float(2.0 * n * float(np.max(c)) / (s * s))


def normalize_seq(a, target_sum):
    """Clip to non-negative, normalize to target_sum."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        a = np.full(a.shape, target_sum / max(1, a.size))
    else:
        a *= target_sum / s
    return a


def upsample_to(a, n_new):
    """Linearly interpolate sequence to new length, renormalize."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    n_new = int(n_new)
    if m == 0:
        return np.ones(n_new)
    if m == n_new:
        return normalize_seq(a, float(n_new))
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    out = np.interp(x_new, x_old, a)
    return normalize_seq(out, float(n_new))


def build_lp_matrix(a):
    """Build constraint matrix A for LP: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            A[k, idx] = a[k - idx]
    return A


def lp_waterfill_step(a, M, time_limit=60.0):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0.
    Returns optimal b or None on failure.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1

    t_start = time.monotonic()

    # Build constraint matrix
    A = build_lp_matrix(a)

    # LP: minimize -sum(b) s.t. A@b <= M*ones, b >= 0
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(L, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': max(5.0, time_limit - (time.monotonic() - t_start)),
                               'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            return b
    except Exception as e:
        print(f"  LP failed: {e}")

    return None


def lp_waterfill(a, time_budget, verbose=True):
    """
    Iterative LP water-filling: repeatedly solve LP and blend.
    Returns (best_seq, best_score).
    """
    a = normalize_seq(np.asarray(a, dtype=np.float64), float(a.size))
    n = a.size
    target_sum = float(a.sum())

    t0 = time.monotonic()
    deadline = t0 + time_budget

    c = fft_autoconv(a)
    score = float(2.0 * n * np.max(c) / (target_sum * target_sum))
    best = a.copy()
    best_score = score

    it = 0
    t_step = 0.5  # blending factor
    stagnant = 0

    while time.monotonic() < deadline - 5.0:
        it += 1
        M = float(np.max(fft_autoconv(a)))

        # Check time for LP
        elapsed_lp = time.monotonic() - t0
        remaining = deadline - time.monotonic()
        if remaining < 10.0:
            break

        # Time limit for LP scales with n
        lp_time = min(remaining * 0.8, max(10.0, 2.0 * n / 100.0))

        b = lp_waterfill_step(a, M, time_limit=lp_time)
        if b is None:
            # Try with slightly larger M
            b = lp_waterfill_step(a, M * 1.01, time_limit=lp_time)
            if b is None:
                stagnant += 1
                if stagnant > 5:
                    break
                continue

        sum_b = float(b.sum())
        if sum_b <= target_sum * (1.0 + 1e-8):
            stagnant += 1
            if stagnant > 5:
                break
            # Try gradient step instead
            a = gradient_single_step(a, 0.15, K=16)
            continue

        # Normalize and blend
        b_norm = b * target_sum / sum_b
        new_a = (1.0 - t_step) * a + t_step * b_norm
        new_a = np.maximum(new_a, 0.0)
        new_a = normalize_seq(new_a, target_sum)

        new_c = fft_autoconv(new_a)
        new_score = float(2.0 * n * np.max(new_c) / (target_sum * target_sum))

        if new_score < best_score:
            best = new_a.copy()
            best_score = new_score
            stagnant = 0
            if verbose and it % 10 == 0:
                print(f"    LP it {it}: score={new_score:.6f} (best={best_score:.6f})")
        else:
            stagnant += 1

        a = new_a

        # Adapt step size
        if stagnant < 3:
            t_step = min(0.8, t_step * 1.05)
        elif stagnant > 5:
            t_step = max(0.2, t_step * 0.8)

        if stagnant > 20:
            break

    return best, best_score


def gradient_single_step(a, step_size, K=16):
    """Single multiplicative gradient step using top-K peaks."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    target_sum = float(a.sum())
    c = fft_autoconv(a)
    L = 2 * n - 1

    K = max(1, min(K, L, max(1, n // 4)))
    top_k = np.argpartition(c, -K)[-K:]

    grad = np.zeros(n, dtype=np.float64)
    INV = 1.0 / float(K)
    for k in top_k:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += INV * a[k - idx]

    grad *= 2.0
    grad -= np.mean(grad)
    scale = np.max(np.abs(grad))
    if scale < 1e-14:
        return a
    grad /= scale

    log_update = -step_size * grad
    log_update = np.clip(log_update, -2.0, 2.0)
    new_a = a * np.exp(log_update)
    new_a = np.maximum(new_a, 0.0)
    new_a = normalize_seq(new_a, target_sum)
    return new_a


def gradient_refine(a, time_budget, seed=0):
    """
    Multiplicative gradient refinement with adaptive step, momentum, noise.
    """
    a = normalize_seq(np.asarray(a, dtype=np.float64), float(a.size))
    n = a.size
    target_sum = float(a.sum())

    best_score = score

    beta = 0.4

    stagnant = 0
    it = 0

    while time.monotonic() < deadline - 2.0:
        it += 1
        c = fft_autoconv(a)
        L = 2 * n - 1
        K = max(4, min(64, int(L * 0.01), n // 4))

        top_k = np.argpartition(c, -K)[-K:]
        grad = np.zeros(n, dtype=np.float64)
        INV = 1.0 / float(K)
        for k in top_k:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                idx = np.arange(j_lo, j_hi + 1)
                grad[idx] += INV * a[k - idx]
        grad *= 2.0

        g = grad - np.mean(grad)
        scale = np.max(np.abs(g))
        if scale < 1e-14:
            stagnant += 1
            if stagnant > 100:
                break
            a = smooth_perturb(best, rng, 0.3)
            c = fft_autoconv(a)
            score = float(2.0 * n * np.max(c) / (target_sum * target_sum))
            continue

        g /= scale
        mom = beta * mom + (1.0 - beta) * g

        log_f = -eta * mom
        if noise > 1e-5:
            log_f += rng.normal(0.0, noise, n)
        log_f = np.clip(log_f, -3.0, 3.0)

        cand = a * np.exp(log_f)
        cand = np.maximum(cand, 0.0)
        cand = normalize_seq(cand, target_sum)

        cand_c = fft_autoconv(cand)
        cand_score = float(2.0 * n * np.max(cand_c) / (target_sum * target_sum))

        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            stagnant = 0
            if it % 200 == 0:
                print(f"    Grad it {it}: score={cand_score:.6f} (best={best_score:.6f})")

        if cand_score <= score * 1.02 + 1e-12:
            a = cand
            score = cand_score
            mom *= 0.85
            if cand_score < score * 0.999:
                eta = min(1.0, eta * 1.05)
                noise = max(0.003, noise * 0.997)
            else:
                eta = max(0.05, eta * 0.97)
            stagnant += 1
        else:
            eta = max(0.03, eta * 0.75)
            mom *= 0.5
            stagnant += 1

        if stagnant > 500:
            a = smooth_perturb(best, rng, 0.4)
            c = fft_autoconv(a)
            score = float(2.0 * n * np.max(c) / (target_sum * target_sum))
            eta = 0.35
            noise = 0.04
            mom = np.zeros(n, dtype=np.float64)
            stagnant = 0

    return best, best_score


def smooth_perturb(a, rng, strength=0.3):
    """Apply smooth multiplicative perturbation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n < 3:
        return a
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, 5):
        amp = strength / (0.5 + 0.5 * m)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    a *= np.exp(z)
    return normalize_seq(a, float(a.size))


def make_two_block(n, frac, ratio):
    """Two-block sequence."""
    m = int(round(frac * n))
    m = max(1, min(n - 1, m))
    denom = m * ratio + (n - m)
    B = float(n) / denom
    A = ratio * B
    a = np.empty(n)
    a[:m] = A
    a[m:] = B
    return normalize_seq(a, float(n))


def make_harmonic(n, rng, max_freq=5):
    """Random low-frequency cosine shape, exponentiated."""
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n)
    for m in range(1, max_freq + 1):
        amp = rng.uniform(-1.5, 1.5) / (m + 1.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    z -= np.max(z)
    a = np.exp(z)
    return normalize_seq(a, float(n))


def generate_starts(n, rng, prev_best, count=10):
    """Generate diverse starting sequences."""
    starts = []
    t = 0.0

    # Uniform
    a = np.ones(n)
    starts.append(normalize_seq(a, float(n)))

    # Two-block
    for frac in (0.2, 0.35, 0.5):
        for ratio in (0.5, 1.5, 3.0):
            starts.append(make_two_block(n, frac, ratio))

    # Harmonic
    for _ in range(count):
        starts.append(make_harmonic(n, rng))

    # Up-sampled previous best
    if prev_best is not None and len(prev_best) > 0:
        starts.append(upsample_to(prev_best, n))
        starts.append(smooth_perturb(prev_best, rng, 0.2))
        starts.append(smooth_perturb(prev_best, rng, 0.5))

    # Evaluate and sort
    scored = []
    for a in starts:
        sc = compute_score(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda x: x[0])
    return scored


def sanitize_return(seq):
    """Ensure valid output: list of non-negative finite floats."""
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 512
    scale = float(seq.size) / s
    if seq.max() * scale <= 1000.0:
        seq *= scale
    seq = np.clip(seq, 0.0, 1000.0)
    return [float(x) for x in seq]
