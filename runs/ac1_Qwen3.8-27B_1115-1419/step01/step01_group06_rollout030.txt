    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)

        n = v.size if v.size > 0 else 1
        return np.ones(n) / n
    return v / s

    c = np.fft.irfft(A * A, N)
    return c[:2*len(a)-1]

    Compute subgradient of max(c) w.r.t. f using top-K active positions.
    c = convolve(f, f), length 2n-1.
    N2 = FFT length for the cross-correlation (should be >= 3n-2).

    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 0.0 or not np.isfinite(s):
        n = v.size if v.size > 0 else 1
        return np.ones(n, dtype=np.float64) / float(n)
    return v / s


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[: 2 * len(a) - 1]


def _softmax_weights(c, beta):
    """Compute softmax weights over the convolution values."""
    c_max = np.max(c)
    x = beta * (c - c_max)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        return p
    return e / s


def _compute_gradient(f, c, p, N2):
    """
    Compute gradient of sum_k p_k * c_k with respect to f.
    c_k = sum_i f[i]*f[k-i]
    d(sum_k p_k c_k)/d(f[j]) = 2*sum_k p[k]*f[k-j] - p[2j]*f[j]
    """
    n = len(f)
    # Cross-correlation: corr[j] = sum_k p[k]*f[k-j]
    P = np.fft.rfft(p, N2)
    F = np.fft.rfft(f, N2)
    corr = np.fft.irfft(np.conj(P) * F, N2)
    corr = corr[:n]

    # Diagonal correction: p[2j]*f[j]
    diag = np.zeros(n, dtype=np.float64)
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    diag[valid] = p[idx[valid]] * f[np.arange(n)[valid]]

    grad = 2.0 * corr - diag
    return grad


def _adam_log_step(grad_w, log_f, m, v, t, lr):
    """One Adam step in log-probability space."""
    t += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1.0 - b1) * grad_w
    v = b2 * v + (1.0 - b2) * grad_w * grad_w
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    step = m_hat / (np.sqrt(v_hat) + 1e-12)
    step = np.clip(step, -20.0, 20.0)
    step = np.where(np.isfinite(step), step, 0.0)
    log_f = log_f - lr * step
    log_f = np.where(np.isfinite(log_f), log_f, 0.0)
    log_f -= np.max(log_f)
    exp_f = np.exp(log_f)
    f_new = exp_f / float(exp_f.sum())
    return f_new, log_f, m, v, t


def _convex_subproblem_step(f, b, beta, lr, N, N2, inner_iters):
    """
    Optimize b to minimize max_k (f*b)_k.
    Returns updated b and log_b state.
    """
    n = len(b)
    log_b = np.log(np.maximum(b, 1e-300))
    log_b -= np.max(log_b)
    m_b = np.zeros(n)
    v_b = np.zeros(n)
    t_b = 0

    F_fft = np.fft.rfft(f, N)

    for _ in range(inner_iters):
        B = np.fft.rfft(b, N)
        h = np.fft.irfft(F_fft * B, N)
        h = h[: 2 * n - 1]
        h = np.maximum(h, 0.0)

        mx = float(np.max(h))
        if mx <= 0.0:
            break

        # Softmax weights on h
        p_h = _softmax_weights(h, beta)

        # Gradient of sum_k p_h[k]*(f*b)[k] w.r.t. b[j]
        # (f*b)[k] = sum_i f[i]*b[k-i]
        # d/d b[j] = sum_k p_h[k] * f[k-j]
        P_h = np.fft.rfft(p_h, N2)
        F_full = np.fft.rfft(f, N2)
        gb = np.fft.irfft(P_h * F_full, N2)
        gb = gb[:n]
        gb = np.where(np.isfinite(gb), gb, 0.0)

        # Project to tangent space
        gb_proj = gb - float(np.dot(b, gb))

        t_b += 1
        m_b = 0.9 * m_b + 0.1 * gb_proj
        v_b = 0.999 * v_b + 0.001 * gb_proj * gb_proj
        m_hat = m_b / (1.0 - 0.9 ** t_b)
        v_hat = v_b / (1.0 - 0.999 ** t_b)
        step = m_hat / (np.sqrt(v_hat) + 1e-12)
        step = np.clip(step, -10.0, 10.0)
        log_b = log_b - lr * step
        log_b = np.where(np.isfinite(log_b), log_b, 0.0)
        log_b -= np.max(log_b)
        exp_b = np.exp(log_b)
        b = exp_b / float(exp_b.sum())

    return b


def _exact_score(a):
    """Exact objective using np.convolve."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0:
        return float("inf")
    c = np.convolve(a, a)
    mx = float(np.max(c))
    val = 2.0 * n * mx / (s * s)
    return val if np.isfinite(val) else float("inf")


def _load_parent():
    """Load parent sequence if available."""
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 8:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return arr / s


def _run_single_length(f_init, budget_s, seed=42, verbose=True):
    """Run optimization for a fixed sequence length."""
    start_time = time.monotonic()
    deadline = start_time + budget_s

    f = _normalize(f_init)
    n = len(f)
    N = _next_fast_len(2 * n - 1)
    N2 = _next_fast_len(3 * n - 2)

    c = _fft_conv(f, N)
    current_score = 2.0 * n * float(np.max(c))

    best_f = f.copy()
    best_score = current_score

    # Log-domain state
    log_f = np.log(np.maximum(f, 1e-300))
    log_f -= np.max(log_f)
    m_f = np.zeros(n)
    v_f = np.zeros(n)
    t_f = 0

    # Hyperparameters
    lr = 0.15
    beta = 20.0
    stall = 0
    outer = 0
    rng = np.random.default_rng(seed)

    # Convex subproblem state
    b = f.copy()
    lr_cvx = 0.1
    beta_cvx = 20.0

    # Phase tracking
    phase = 0  # 0: explore, 1: exploit, 2: polish
    phase_thresholds = [0.6, 0.9]  # fractions of budget

    while time.monotonic() < deadline:
        elapsed_frac = (time.monotonic() - start_time) / max(budget_s, 1.0)

        # Determine phase
        if elapsed_frac < phase_thresholds[0]:
            phase = 0
        elif elapsed_frac < phase_thresholds[1]:
            phase = 1
        else:
            phase = 2

        outer += 1

        # Phase-based hyperparameter adjustment
        if phase == 0:
            lr_use = lr
            beta_use = beta
            inner_cvx = 5
        elif phase == 1:
            lr_use = lr * 0.7
            beta_use = beta * 1.5
            inner_cvx = 10
        else:
            lr_use = lr * 0.3
            beta_use = min(beta * 3.0, 500.0)
            inner_cvx = 20

        # --- Step 1: Self-gradient (softmax-weighted subgradient) ---
        p = _softmax_weights(c, beta_use)
        grad = _compute_gradient(f, c, p, N2)
        grad_proj = grad - float(np.dot(f, grad))

        f_new, log_f, m_f, v_f, t_f = _adam_log_step(
            grad_proj, log_f, m_f, v_f, t_f, lr_use
        )

        c_new = _fft_conv(f_new, N)
        score_new = 2.0 * n * float(np.max(c_new))

        improved = False
        if np.isfinite(score_new) and score_new < current_score:
            f = f_new
            c = c_new
            current_score = score_new
            improved = True
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
                if verbose and outer % 100 == 0:
                    print(f"  n={n} iter={outer} score={current_score:.6f} elapsed={time.monotonic()-start_time:.0f}s")

        # --- Step 2: Convex subproblem (every few iterations) ---
        if outer % 4 == 0 or not improved:
            b = _convex_subproblem_step(f, b, beta_cvx, lr_cvx, N, N2, inner_cvx)

            # Try convex combinations
            best_combo_score = current_score
            best_combo = None
            if phase == 0:
                t_vals = [0.3, 0.5, 0.7, 0.9, 1.0]
            elif phase == 1:
                t_vals = [0.3, 0.5, 0.7, 0.9, 1.0]
            else:
                t_vals = [0.5, 0.8, 1.0]

            for t_val in t_vals:
                cand = (1.0 - t_val) * f + t_val * b
                cand = _normalize(cand)
                c_cand = _fft_conv(cand, N)
                sc = 2.0 * n * float(np.max(c_cand))
                if np.isfinite(sc) and sc < best_combo_score:
                    best_combo_score = sc
                    best_combo = cand

            if best_combo is not None:
                f = best_combo.copy()
                c = _fft_conv(f, N)
                current_score = best_combo_score
                # Reset log state
                log_f = np.log(np.maximum(f, 1e-300))
                log_f -= np.max(log_f)
                m_f = np.zeros(n)
                v_f = np.zeros(n)
                t_f = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                    if verbose and outer % 100 == 0:
                        print(f"  n={n} iter={outer} score={current_score:.6f} (combo) elapsed={time.monotonic()-start_time:.0f}s")
                # Update b to be close to new f
                b = 0.5 * b + 0.5 * f
                b = _normalize(b)
                stall = 0

        if not improved and best_combo is None:
            stall += 1
        elif not improved:
            stall = 0

        # --- Adaptive learning rate ---
        if stall > 10:
            lr = max(0.01, lr * 0.92)
            lr_cvx = max(0.02, lr_cvx * 0.92)
        elif stall == 0:
            lr = min(0.4, lr * 1.01)
            lr_cvx = min(0.3, lr_cvx * 1.01)

        # --- Increase beta over time ---
        beta = min(200.0, beta * 1.005 + 0.1)
        beta_cvx = min(200.0, beta_cvx * 1.005 + 0.1)

        # --- Restart logic ---
        if stall > 50:
            strategy = rng.random()
            if strategy < 0.4:
                # Perturb best
                sig = float(rng.choice([0.03, 0.05, 0.1, 0.2]))
                w_pert = np.log(np.maximum(best_f, 1e-300)) + rng.normal(0, sig, n)
                f = _normalize(np.exp(w_pert - np.max(w_pert)))
            elif strategy < 0.7:
                # Return to best
                f = best_f.copy()
            else:
                # Fresh random restart (still in log domain near best)
                sig = float(rng.choice([0.3, 0.5, 1.0]))
                w_pert = np.log(np.maximum(best_f, 1e-300)) + rng.normal(0, sig, n)
                f = _normalize(np.exp(w_pert - np.max(w_pert)))

            c = _fft_conv(f, N)
            current_score = 2.0 * n * float(np.max(c))
            log_f = np.log(np.maximum(f, 1e-300))
            log_f -= np.max(log_f)
            m_f = np.zeros(n)
            v_f = np.zeros(n)
            t_f = 0
            b = f.copy()
            stall = 0
            lr = 0.15
            lr_cvx = 0.1
            beta = 20.0
            beta_cvx = 20.0

        # --- Print progress ---
        if verbose and outer % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"  n={n} iter={outer} elapsed={elapsed:.1f}s cur={current_score:.6f} best={best_score:.6f} lr={lr:.4f} beta={beta:.1f} stall={stall} phase={phase}")

    return best_f, best_score


def _try_multiple_lengths(parent, budget_s, seed=42):
    """
    Try multiple sequence lengths, allocating time to each.
    Returns the best sequence found.
    """
    if parent is None:
        return _run_single_length(np.ones(4096), budget_s, seed=seed)

    n_parent = len(parent)

    # Candidate lengths to try
    lengths = []
    # Always include parent length (or nearest power of 2)
    lengths.append(n_parent)
    # Try a couple of other lengths
    for cand in [2048, 4096, 8192, 16384]:
        if cand not in lengths and abs(cand - n_parent) / max(n_parent, 1) > 0.3:
            lengths.append(cand)

    # Limit to at most 3 lengths
    lengths = lengths[:3]

    # If only parent length, just use it
    if len(lengths) == 1:
        f, sc = _run_single_length(parent, budget_s * 0.95, seed=seed)
        return f, sc

    # Allocate time: give more to parent length
    total = len(lengths)
    weights = []
    for L in lengths:
        if L == n_parent:
            weights.append(0.5)
        else:
            weights.append(0.5 / (total - 1))
    w_sum = sum(weights)
    weights = [w / w_sum * 0.92 * budget_s for w in weights]

    best_f = None
    best_score = float("inf")
    time_start = time.monotonic()

    for i, L in enumerate(lengths):
        remaining = deadline_val = budget_s - (time.monotonic() - time_start) - 2.0
        alloc = min(weights[i], remaining)
        if alloc < 10.0:
            break

        print(f"Trying length {L}, allocated {alloc:.0f}s")

        # Create initial sequence for this length
        if L == n_parent:
            f_init = parent.copy()
        else:
            # Downsample or upsample parent
            if L < n_parent:
                # Block average
                blocks = n_parent // L
                if blocks > 1:
                    use = L * blocks
                    f_init = np.mean(parent[:use].reshape(L, blocks), axis=1)
                else:
                    f_init = parent[:L]
            else:
                # Upsample via interpolation
                x_old = np.linspace(0, 1, n_parent)
                x_new = np.linspace(0, 1, L)
                f_init = np.interp(x_new, x_old, parent)

        f_init = _normalize(f_init)
        seed_i = seed + i * 137

        f, sc = _run_single_length(f_init, alloc, seed=seed_i)

        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            print(f"  Updated best: length={L} score={best_score:.6f}")

    if best_f is None:
        best_f, best_score = _run_single_length(parent, budget_s * 0.9, seed=seed)

    return best_f, best_score


def _polish(f, budget_s, seed=42):
    """Final polishing phase with very high beta and small steps."""
    start_time = time.monotonic()
    deadline = start_time + budget_s

    n = len(f)
    N = _next_fast_len(2 * n - 1)
    N2 = _next_fast_len(3 * n - 2)

    c = _fft_conv(f, N)
    current_score = 2.0 * n * float(np.max(c))
    best_f = f.copy()
    best_score = current_score

    log_f = np.log(np.maximum(f, 1e-300))
    log_f -= np.max(log_f)
    m_f = np.zeros(n)
    v_f = np.zeros(n)
    t_f = 0

    lr = 0.03
    beta = 100.0
    outer = 0

    while time.monotonic() < deadline:
        outer += 1

        p = _softmax_weights(c, beta)
        grad = _compute_gradient(f, c, p, N2)
        grad_proj = grad - float(np.dot(f, grad))

        f_new, log_f, m_f, v_f, t_f = _adam_log_step(
            grad_proj, log_f, m_f, v_f, t_f, lr
        )

        c_new = _fft_conv(f_new, N)
        score_new = 2.0 * n * float(np.max(c_new))

        if np.isfinite(score_new) and score_new < current_score:
            f = f_new
            c = c_new
            current_score = score_new
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()

        # Very slowly increase beta
        beta = min(1000.0, beta * 1.002)
        # Very slowly decrease lr
        lr = max(0.005, lr * 0.9995)

    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point: search for optimal non-negative coefficient sequence.
    Uses hybrid softmax-subgradient + convex subproblem optimization
    with adaptive scheduling and multi-length exploration.
    """
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 2400.0
    budget = min(max(budget, 100.0), 2390.0)

    print(f"Total budget: {budget:.0f}s")

    # Load parent
    parent = _load_parent()
    if parent is not None:
        print(f"Loaded parent: length={len(parent)}")
        n_parent = len(parent)
        # Quick exact score of parent
        if n_parent <= 20000:
            p_score = _exact_score(parent)
            print(f"Parent exact score: {p_score:.6f}")
    else:
        parent = np.ones(4096, dtype=np.float64)
        print("No parent loaded, using uniform 4096")

    # Allocate time: main search + polish
    main_budget = budget * 0.85
    polish_budget = budget * 0.13
    reserve = budget * 0.02

    # Main search (with multi-length if beneficial)
    n_parent = len(parent)
    if n_parent > 20000:
        # Too long for exact validation, truncate
        parent = parent[:16384]
        parent = _normalize(parent)
        print(f"Truncated parent to length {len(parent)}")

    best_f, best_score = _try_multiple_lengths(parent, main_budget, seed=seed)
    print(f"Main search done: best_score={best_score:.6f} (FFT approx)")

    # Polish phase
    if polish_budget > 30:
        print(f"Starting polish phase ({polish_budget:.0f}s)")
        best_f, polished_score = _polish(best_f, polish_budget, seed=seed + 999)
        if polished_score < best_score:
            best_score = polished_score
        print(f"Polish done: best_score={polished_score:.6f}")

    # Final exact validation
    n = len(best_f)
    if n <= 20000:
        exact_sc = _exact_score(best_f)
        print(f"Final exact score: {exact_sc:.6f}")
        if not np.isfinite(exact_sc):
            # Fallback
            best_f = np.ones(4096, dtype=np.float64) / 4096.0
            exact_sc = _exact_score(best_f)
            print(f"Fallback exact score: {exact_sc:.6f}")

    # Scale for output (scale-invariant, just ensure sum >= 0.01)
    s = float(best_f.sum())
    if s < 1.0:
        best_f = best_f * (1.0 / max(s, 1e-300))
    # Keep in valid range
    best_f = np.clip(best_f, 0.0, 1000.0)
    s2 = float(best_f.sum())
    if s2 < 0.01:
        best_f = best_f * (1.0 / max(s2, 1e-300))

    return [float(x) for x in best_f]
