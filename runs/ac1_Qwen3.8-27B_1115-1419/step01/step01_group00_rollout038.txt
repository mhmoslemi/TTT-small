```python
import numpy as np
import time

def _fft_conv(a, fft_size=None):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1
        while fft_size < size:
            fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fc = np.fft.irfft(fa * fa, fft_size)
    return fc[:size]

def _project(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _obj(a, n):
    """Objective value."""
    c = _fft_conv(a)
    return 2.0 * n * np.max(c) / (np.sum(a) ** 2)

def _smooth_step(a, n, tau, eta):
    """One step of entropy-smoothed gradient descent."""
    c = _fft_conv(a)
    mx = np.max(c)
    p = np.exp((c - mx) / tau)
    ps = p.sum()
    if ps < 1e-300:
        return a, 2.0 * n * mx / (np.sum(a)**2)
    p = p / ps
    g = 2.0 * np.correlate(p, a, mode='valid')
    # Remove mean to maintain sum constraint
    g = g - g.mean()
    a_new = a - eta * g / (np.max(np.abs(g)) + 1e-30)
    a_new = _project(a_new, np.sum(a))
    return a_new, 2.0 * n * mx / (np.sum(a)**2)

def _hard_step(a, n, eta, top_k=1):
    """One step of hard active-set descent."""
    c = _fft_conv(a)
    # Find top-k indices
    if top_k == 1:
        k = int(np.argmax(c))
        g = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        g[js:je] = 2.0 * a[k - np.arange(js, je)]
    else:
        # Weighted combination of top-k gradients
        idx = np.argsort(c)[-top_k:]
        w = c[idx]
        w = w / w.sum()
        g = np.zeros(n)
        for k, wk in zip(idx, w):
            js = max(0, k - n + 1)
            je = min(n - 1, k) + 1
            g[js:je] += wk * 2.0 * a[k - np.arange(js, je)]
    g = g - g.mean()
    mg = np.max(np.abs(g))
    if mg < 1e-15:
        return a, 2.0 * n * np.max(c) / (np.sum(a)**2)
    a_new = a - eta * g / mg
    a_new = _project(a_new, np.sum(a))
    return a_new, 2.0 * n * np.max(c) / (np.sum(a)**2)
```

```python
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based convolution, multi-start initialization, top-K hard descent,
and simulated annealing with structured perturbations.
"""

import numpy as np
import time


def _fft_conv(a, fft_size):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = np.fft.rfft(a, fft_size)
    fc = np.fft.irfft(fa * fa, fft_size)
    return fc[:size]


def _next_pow2(m):
    """Next power of 2 >= m."""
    p = 1
    while p < m:
        p *= 2
    return p


def _project(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _obj_from_conv(c, a, n, S):
    """Compute objective from precomputed convolution."""
    return 2.0 * n * float(np.max(c)) / (S * S)


def _smooth_phase(a, n, S, fft_size, deadline, tag, it_limit=500000):
    """Entropy-smoothed gradient descent."""
    tau = n / 4.0
    tau_min = n * 0.002
    eta = 0.05
    best = a.copy()
    c = _fft_conv(a, fft_size)
    best_o = float(2.0 * n * np.max(c) / (S * S))
    stall = 0
    
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _fft_conv(a, fft_size)
        mx = float(np.max(c))
        o = 2.0 * mx / (S / n) ** 2  # = 2*n*mx/S^2
        # Simpler: o = 2.0 * n * mx / (S * S)
        o = 2.0 * n * mx / (S * S)
        if o < best_o - 1e-12:
            best_o = o
            best = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Softmax weights
        p = np.exp((c - mx) / tau)
        ps = p.sum()
        if ps < 1e-300:
            break
        p = p / ps
        
        # Gradient
        g = 2.0 * np.correlate(p, a, mode='valid')
        g = g - g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-15:
            break
        
        a = _project(a - eta * g / mg, S)
        
        # Annealing
        tau = max(tau * 0.99998, tau_min)
        if stall > 5000:
            eta = max(eta * 0.7, 1e-6)
            stall = 0
            # Reheat
            tau = min(tau * 2.0, n / 4.0)
    
    return best, best_o


def _hard_phase(a, n, S, fft_size, deadline, tag, it_limit=500000, top_k=1):
    """Hard active-set descent on top-K max convolution coefficients."""
    best = a.copy()
    c = _fft_conv(a, fft_size)
    best_o = float(2.0 * n * np.max(c) / (S * S))
    eta = 0.03
    
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        c = _fft_conv(a, fft_size)
        mc = float(np.max(c))
        o = 2.0 * n * mc / (S * S)
        if o < best_o - 1e-12:
            best_o = o
            best = a.copy()
        
        if top_k == 1:
            k = int(np.argmax(c))
            g = np.zeros(n, dtype=np.float64)
            js = max(0, k - n + 1)
            je = min(n - 1, k) + 1
            g[js:je] = 2.0 * a[k - np.arange(js, je)]
        else:
            # Top-K with weighting
            idx = np.argpartition(c, -top_k)[-top_k:]
            w = c[idx].copy()
            w = w / w.sum()
            g = np.zeros(n, dtype=np.float64)
            for kk, wk in zip(idx, w):
                js = max(0, kk - n + 1)
                je = min(n - 1, kk) + 1
                g[js:je] += wk * 2.0 * a[kk - np.arange(js, je)]
        
        g = g - g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-15:
            break
        a = _project(a - eta * g / mg, S)
        
        # Decay learning rate
        if it > 0 and it % 50000 == 49999:
            eta = max(eta * 0.85, 1e-7)
    
    return best, best_o


def _perturb_gauss(a, n, S, rng, sigma):
    """Add Gaussian noise to entire sequence, then project."""
    noise = rng.normal(0, sigma, size=n)
    return _project(a + noise, S)


def _perturb_block(a, n, S, rng, sigma, width_frac=None):
    """Multiply a random contiguous block by exp(N(0, sigma))."""
    b = a.copy()
    if width_frac is None:
        w = max(2, n // 50)
    else:
        w = max(2, int(n * width_frac))
    s = int(rng.integers(0, max(1, n - w + 1)))
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _project(b, S)


def _perturb_swap(a, n, S, rng):
    """Swap two random blocks."""
    b = a.copy()
    w1 = max(2, int(n * rng.uniform(0.01, 0.1)))
    w2 = max(2, int(n * rng.uniform(0.01, 0.1)))
    s1 = int(rng.integers(0, max(1, n - w1 + 1)))
    s2 = int(rng.integers(0, max(1, n - w2 + 1)))
    # Ensure blocks don't overlap
    if s2 < s1 + w1 and s1 + w1 <= s2 + w2:
        s2 = (s1 + w1 + int(rng.integers(0, n - s1 - w1))) % max(1, n - w2)
    b[s1:s1+w1], b[s2:s2+w2] = b[s2:s2+w2].copy(), b[s1:s1+w1].copy()
    return _project(b, S)


def _perturb_sparse(a, n, S, rng, frac=0.05, sigma=0.5):
    """Perturb a random subset of elements."""
    b = a.copy()
    m = max(1, int(n * frac))
    idx = rng.choice(n, size=m, replace=False)
    b[idx] += rng.normal(0, sigma * np.max(a), size=m)
    return _project(b, S)
```

```python
    eta = 0.05
    best = a.copy()
    c = _fft_conv(a, fsz)
    best_o = _obj_val(c, n, S)
    stall = 0
    for it in range(max_it):
        if time.monotonic() >= end_t:
            break
        c = _fft_conv(a, fsz)
        mx = float(np.max(c))
        o = 2.0 * n * mx / (S * S)
        if o < best_o - 1e-13:
            best_o = o
            best = a.copy()
            stall = 0
        else:
            stall += 1
        p = np.exp((c - mx) / tau)
        ps = p.sum()
        if ps < 1e-300:
            break
        p /= ps
        g = 2.0 * np.correlate(p, a, mode='valid')
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        a = _proj(a - eta * g / mg, S)
        tau = max(tau * 0.99998, tau_min)
        if stall > 3000:
            eta = max(eta * 0.6, 1e-7)
            tau = min(tau * 3.0, n * 0.25)
            stall = 0
    return best, best_o


def _hard_run(a, n, S, fsz, end_t, tag, max_it, topk):
    best = a.copy()
    c = _fft_conv(a, fsz)
    best_o = _obj_val(c, n, S)
    eta = 0.03
    for it in range(max_it):
        if time.monotonic() >= end_t:
            break
        c = _fft_conv(a, fsz)
        mx = float(np.max(c))
        o = 2.0 * n * mx / (S * S)
        if o < best_o - 1e-13:
            best_o = o
            best = a.copy()
        if topk <= 1:
            k = int(np.argmax(c))
            g = np.zeros(n, dtype=np.float64)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k) + 1
            g[lo:hi] = 2.0 * a[k - np.arange(lo, hi)]
        else:
            idx = np.argpartition(c, -topk)[-topk:]
            w = c[idx]
            w = w / (w.sum() + 1e-30)
            g = np.zeros(n, dtype=np.float64)
            for ki, wi in zip(idx, w):
                lo = max(0, ki - n + 1)
                hi = min(n - 1, ki) + 1
                g[lo:hi] += wi * 2.0 * a[ki - np.arange(lo, hi)]
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        a = _proj(a - eta * g / mg, S)
        if it > 0 and it % 40000 == 39999:
            eta = max(eta * 0.87, 1e-8)
    return best, best_o


def _make_thue_morse(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(n):
        cnt = 0
        x = i
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = cnt % 2
    return tm


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 8.0

    n = 3809
    fsz = _next_pow2(2 * n - 1)
    S = float(n)

    best_a = None
    best_o = np.inf

    # --- Build initializations ---
    inits = []

    # Parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size == n and np.all(np.isfinite(p)):
            inits.append(_proj(np.maximum(p, 0), S))
    except Exception:
        pass

    # Uniform
    inits.append(np.full(n, S / n, dtype=np.float64))

    # Random
    inits.append(_proj(rng.uniform(0.3, 2.0, n), S))

    # Binary
    b = np.zeros(n, dtype=np.float64)
    b[rng.choice(n, n // 2, replace=False)] = 1.0
    inits.append(_proj(b, S))

    # Thue-Morse
    tm = _make_thue_morse(n)
    inits.append(_proj(0.3 + 0.7 * tm, S))

    # Periodic (period 5)
    pat5 = np.array([1.0, 0.4, 0.9, 0.2, 0.7])
    inits.append(_proj(np.tile(pat5, n // 5 + 1)[:n], S))

    # Periodic (period 11)
    pat11 = np.array([1.0, 0.3, 0.8, 0.1, 0.6, 1.0, 0.2, 0.9, 0.4, 0.7, 0.5])
    inits.append(_proj(np.tile(pat11, n // 11 + 1)[:n], S))

    # Smooth random (integrated noise)
    t_arr = np.cumsum(rng.normal(0, 1, n))
    t_arr = (t_arr - t_arr.min()) / (t_arr.max() - t_arr.min() + 1e-10)
    inits.append(_proj(0.3 + 0.7 * t_arr, S))

    # Bi-modal (two levels)
    bim = np.zeros(n, dtype=np.float64)
    bim[rng.choice(n, int(0.4 * n), replace=False)] = 1.0
    bim[rng.choice(n, int(0.3 * n), replace=False)] = 0.5
    inits.append(_proj(bim, S))

    print(f"n={n}, fsz={fsz}, {len(inits)} initializations")

    # --- Phase 1: Multi-start screening ---
    p1_end = t0 + 0.20 * budget_s
    per_alloc = max(8.0, (p1_end - t0) / len(inits))
    candidates = []

    for i_init, a0 in enumerate(inits):
        if time.monotonic() >= p1_end - 5:
            break
        end1 = min(time.monotonic() + per_alloc * 0.6, p1_end - 8)
        a_s, o_s = _smooth_run(a0, n, S, fsz, end1, f"m{i_init}", 30000)
        end2 = min(time.monotonic() + per_alloc * 0.35, p1_end - 3)
        a_h, o_h = _hard_run(a_s, n, S, fsz, end2, f"mh{i_init}", 20000, 3)
        o = min(o_s, o_h)
        a = a_h if o_h <= o_s else a_s
        candidates.append((o, a))
        if o < best_o:
            best_o = o
            best_a = a.copy()
            print(f"  init[{i_init}]: obj={o:.6f} (t={time.monotonic()-t0:.0f}s)")

    # Sort candidates and do a second pass on top 3
    candidates.sort(key=lambda x: x[0])
    print(f"Multi-start done: best={best_o:.6f} (t={time.monotonic()-t0:.0f}s)")

    # --- Phase 2: Deep optimization ---
    p2_end = t0 + 0.50 * budget_s
    rem2 = p2_end - time.monotonic()
    if rem2 > 10:
        best_a, best_o = _smooth_run(best_a, n, S, fsz,
                                      time.monotonic() + rem2 * 0.55, "deep_s", 500000)
    rem2b = p2_end - time.monotonic()
    if rem2b > 5:
        best_a, best_o = _hard_run(best_a, n, S, fsz,
                                    time.monotonic() + rem2b * 0.8, "deep_h", 500000, 3)
    print(f"Deep opt done: best={best_o:.6f} (t={time.monotonic()-t0:.0f}s)")

    # --- Phase 3: Perturbation + re-optimization loop ---
    p3_end = deadline - 25.0
    attempt = 0
    sa_temp = 1e-4

    while time.monotonic() < p3_end:
        attempt += 1
        rem = p3_end - time.monotonic()
        if rem < 10.0:
            break

        r = rng.random()
        a_p = best_a.copy()

        if r < 0.25:
            # Gaussian noise
            sig = float(rng.uniform(0.02, 0.12))
            a_p = _proj(best_a + rng.normal(0, sig * np.std(best_a), n), S)
        elif r < 0.50:
            # Block multiplication
            sig = float(rng.uniform(0.05, 0.35))
            w = max(3, int(n * float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.1, 0.2]))))
            s = int(rng.integers(0, max(1, n - w)))
            a_p[s:s + w] *= np.exp(rng.normal(0, sig, w))
            a_p = _proj(a_p, S)
        elif r < 0.70:
            # Block swap
            w1 = max(3, int(n * float(rng.uniform(0.01, 0.08))))
            w2 = max(3, int(n * float(rng.uniform(0.01, 0.08))))
            s1 = int(rng.integers(0, max(1, n - w1)))
            s2 = int(rng.integers(0, max(1, n - w2)))
            trial = 0
            while s1 < s2 + w2 and s2 < s1 + w1 and trial < 20:
                s2 = int(rng.integers(0, max(1, n - w2)))
                trial += 1
            tmp = a_p[s1:s1 + w1].copy()
            a_p[s1:s1 + w1] = a_p[s2:s2 + w2]
            a_p[s2:s2 + w2] = tmp
            a_p = _proj(a_p, S)
        elif r < 0.85:
            # Sparse perturbation
            m = max(2, int(n * float(rng.uniform(0.02, 0.10))))
            idx = rng.choice(n, m, replace=False)
            a_p[idx] += rng.normal(0, 0.3 * float(np.max(best_a)), m)
            a_p = _proj(a_p, S)
        else:
            # Sinusoidal modulation
            w = max(5, int(n * float(rng.uniform(0.03, 0.15))))
            s = int(rng.integers(0, max(1, n - w)))
            freq = float(rng.uniform(1.0, 10.0))
            amp = float(rng.uniform(0.1, 0.6))
            mod = 1.0 + amp * np.sin(2.0 * np.pi * freq * np.arange(w) / w)
            a_p[s:s + w] *= np.maximum(mod, 0.0)
            a_p = _proj(a_p, S)

        # Quick smooth optimization
        sub_t = min(10.0, rem * 0.3)
        end_s = min(time.monotonic() + sub_t, p3_end - 8)
        a_r, o_r = _smooth_run(a_p, n, S, fsz, end_s, f"p{attempt}", 40000)

        if o_r < best_o:
            best_o = o_r
            best_a = a_r.copy()
            print(f"  P{attempt}: obj={best_o:.6f} (t={time.monotonic()-t0:.0f}s)")
            # Follow up with hard
            if time.monotonic() < p3_end - 6:
                end_h = min(time.monotonic() + min(5.0, (p3_end - time.monotonic()) * 0.2), p3_end - 3)
                a_r2, o_r2 = _hard_run(best_a, n, S, fsz, end_h, f"ph{attempt}", 20000, 3)
                if o_r2 < best_o:
                    best_o = o_r2
                    best_a = a_r2.copy()
                    print(f"  P{attempt}+h: obj={best_o:.6f}")

        # Occasional restart from scratch
        if attempt in (15, 40, 70, 100, 150) and time.monotonic() < p3_end - 25:
            a_alt = _proj(rng.uniform(0.1, 2.5, n), S)
            end_alt = min(time.monotonic() + 12.0, p3_end - 12)
            a_a2, o_a2 = _smooth_run(a_alt, n, S, fsz, end_alt, f"rs{attempt}", 30000)
            end_ah = min(time.monotonic() + 5.0, p3_end - 5)
            a_a3, o_a3 = _hard_run(a_a2, n, S, fsz, end_ah, f"rsh{attempt}", 15000, 3)
            if o_a3 < best_o:
                best_o = o_a3
                best_a = a_a3.copy()
                print(f"  Restart{attempt}: obj={best_o:.6f}")

        # Periodic top-K hard polish
        if attempt % 15 == 0 and time.monotonic() < p3_end - 8:
            end_k = min(time.monotonic() + min(6.0, (p3_end - time.monotonic()) * 0.15), p3_end - 3)
            a_k, o_k = _hard_run(best_a, n, S, fsz, end_k, f"tk{attempt}", 25000, 5)
            if o_k < best_o:
                best_o = o_k
                best_a = a_k.copy()
                print(f"  TopK{attempt}: obj={best_o:.6f}")

        sa_temp *= 0.995

    # --- Phase 4: Final hard polish ---
    if time.monotonic() < deadline - 3.0:
        best_a, best_o = _hard_run(best_a, n, S, fsz, deadline - 1.0, "final", 2000000, 1)
        print(f"Final hard: obj={best_o:.6f}")

    # One more smooth pass if time allows
    if time.monotonic() < deadline - 2.0:
        best_a, best_o2 = _smooth_run(best_a, n, S, fsz, deadline - 1.0, "final_s", 100000)
        if best_o2 < best_o:
            best_o = best_o2

    # Validate
    result = [float(x) for x in best_a]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    if not np.isfinite(sc):
        result = [1.0] * n
        sc = 2.0
    print(f"DONE: internal={best_o:.6f} evaluated={sc:.6f} n={n} time={time.monotonic()-t0:.0f}s")
    return result
```
