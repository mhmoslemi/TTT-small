def resize_seq(a,m):

"""
Search for a non-negative step sequence minimizing

    2*n*max(full_conv(a,a)) / sum(a)^2

where full_conv is numpy's full linear convolution.

Strategy:
- Normalize sequences to sum 1 (scale invariant).
- Use a numerically stable LSE-smoothed maximum of the convolution.
- Use an exact dense gradient for the LSE objective:
    if w_k = softmax(c_k/tau), then
    grad_a_j = 2*sum_m w_{j+m} a_m,
  computed with one FFT convolution w * reversed(a).
- Optimize in log/softmax parameterization with L-BFGS-B, using a tau schedule.
- Use lower-resolution global searches, then upsample and polish.
- Finish with exact active-set / small-tau projected-gradient polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def clean_parent(seq):
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-15:
        return None
    return arr / s


def safe_eval(a):
    try:
        vals = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(vals))
    except Exception:
        return np.inf


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def square_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.zeros(0)
    if n <= 512:
        c = np.convolve(a, a)
        c = np.where(np.isfinite(c), c, 0.0)
        np.maximum(c, 0.0, out=c)
        return c
    length = 2 * n - 1
    nfft = next_pow2(length)
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)[:length]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def fft_convolution(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if len(x) == 0 or len(y) == 0:
        return np.zeros(0)
    if len(x) * len(y) <= 1000000:
        out = np.convolve(x, y)
        out = np.where(np.isfinite(out), out, 0.0)
        np.maximum(out, 0.0, out=out)
        return out
    length = len(x) + len(y) - 1
    nfft = next_pow2(length)
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    out = np.fft.irfft(fx * fy, n=nfft)[:length]
    out = np.where(np.isfinite(out), out, 0.0)
    np.maximum(out, 0.0, out=out)
    return out


def softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, -60.0)
    x = x - x.max()
    e = np.exp(x)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(len(x), 1.0 / float(len(x)), dtype=np.float64)
    return e / s


def softmax_weights(c, tau):
    c = np.asarray(c, dtype=np.float64)
    cmax = float(c.max())
    L = len(c)
    if L == 0:
        return np.zeros(0)
    if cmax <= 1e-300:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    if tau <= 1e-15:
        K = min(64, L)
        w = np.zeros(L, dtype=np.float64)
        if K == 1:
            w[int(np.argmax(c))] = 1.0
        else:
            idx = np.argpartition(c, -K)[-K:]
            w[idx] = 1.0 / float(K)
        return w
    z = (c - cmax) / tau
    z = np.minimum(z, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        K = min(64, L)
        w = np.zeros(L, dtype=np.float64)
        if K == 1:
            w[int(np.argmax(c))] = 1.0
        else:
            idx = np.argpartition(c, -K)[-K:]
            w[idx] = 1.0 / float(K)
        return w
    return e / s


def top_weights(c, K):
    c = np.asarray(c, dtype=np.float64)
    L = len(c)
    if L == 0:
        return np.zeros(0)
    if K <= 0 or K >= L:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    w = np.zeros(L, dtype=np.float64)
    if K == 1:
        w[int(np.argmax(c))] = 1.0
    else:
        idx = np.argpartition(c, -K)[-K:]
        w[idx] = 1.0 / float(K)
    return w


def gradient_for_weights(weights, a):
    n = len(a)
    h = fft_convolution(weights, a[::-1])
    g = 2.0 * h[n - 1:n - 1 + n]
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def smooth_objective(x, tau):
    a = softmax(x)
    n = len(a)
    c = square_conv(a)
    cmax = float(c.max())
    if tau <= 1e-300:
        return 2.0 * n * cmax
    z = (c - cmax) / tau
    z = np.minimum(z, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return 2.0 * n * cmax
    return 2.0 * n * (cmax + tau * np.log(s))


def smooth_gradient(x, tau):
    a = softmax(x)
    n = len(a)
    c = square_conv(a)
    w = softmax_weights(c, tau)
    g = gradient_for_weights(w, a)
    # Multiply by 2*n to match the actual objective scale.
    g *= 2.0 * float(n)
    mean = float(np.dot(a, g))
    grad = a * (g - mean)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def surrogate_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    b = a / s
    c = square_conv(b)
    return float(2.0 * len(b) * c.max())


def resize_seq(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(max(1, m))
    if a.size == 0:
        return np.ones(m, dtype=np.float64)
    if a.size == m:
        return normalize_seq(a)
    x_old = np.linspace(0.0, 1.0, a.size, endpoint=True)
    x_new = np.linspace(0.0, 1.0, m, endpoint=True)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-15:
        b = np.ones(m, dtype=np.float64)
    else:
        b = b / s
    return b


def default_initial(n):
    n = int(max(1, n))
    t = np.linspace(0.0, 1.0, n)
    x = np.exp(-1.2 * t) + 0.1 * np.cos(2.0 * np.pi * 3.0 * t)
    x = np.maximum(x, 1e-6)
    return x / x.sum()


def make_low_inits(n, rng, parent):
    n = int(max(2, n))
    inits = []
    if parent is not None:
        p = resize_seq(parent, n)
        inits.append(normalize_seq(p))
        inits.append(normalize_seq(p[::-1]))

    inits.append(np.ones(n, dtype=np.float64) / float(n))

    t = np.linspace(0.0, 1.0, n)
    for lam in (1.0, 4.0):
        x = np.exp(-lam * t)
        inits.append(x / x.sum())
        x = np.exp(-lam * (1.0 - t))
        inits.append(x / x.sum())

    for f, amp, ph in ((3.0, 0.5, 0.0), (7.0, 0.5, 1.0)):
        x = 1.0 + amp * np.cos(2.0 * np.pi * f * t + ph)
        x = np.maximum(x, 1e-6)
        inits.append(x / x.sum())

    x = np.ones(n, dtype=np.float64)
    x[:n // 2] = 2.0
    x[n // 2:] = 0.3
    inits.append(x / x.sum())

    x = np.abs(2.0 * t - 1.0)
    x = np.maximum(x, 0.1)
    inits.append(x / x.sum())

    x = rng.dirichlet(np.full(n, 0.5))
    x = np.maximum(x, 0.0)
    inits.append(normalize_seq(x))

    return inits


def run_lbfgs_improve(a, deadline, time_budget, tau_factors, tag="", use_actual=True):

    if a.size == 0:

    best_score = safe_eval(a) if use_actual else surrogate_score(a)

    if time_budget < 5.0 or time.monotonic() >= deadline:
        return best_a, best_score

    x = np.log(a + 1e-300)

    n = len(a)

            budget_weights.append(2.0)
        else:
            budget_weights.append(1.0)
    total_w = float(sum(budget_weights))
    if total_w <= 0:
        return best_a, best_score

    start = time.monotonic()
    print(f"[{tag}] start n={n} best={best_score:.8f}")

    for factor, bw in zip(tau_factors, budget_weights):

                    else:
                        stall += 1
            else:

    else:

def active_polish(a, deadline, time_budget, tag="polish"):
    a = normalize_seq(a)
    if a.size == 0 or time_budget < 20.0 or time.monotonic() >= deadline:
        return a, safe_eval(a)

    best_a = a.copy()
    best_score = safe_eval(a)
    if not np.isfinite(best_score):
        return best_a, best_score

    n = len(a)
    maxdelta = 0.02 * (1.0 / float(n))
    start = time.monotonic()
    end = min(deadline, start + time_budget)
    stall = 0
    rnd = 0
    rng = np.random.default_rng(987654321)

    while time.monotonic() < end:
        rnd += 1
        c = square_conv(a)
        cmax = float(c.max())
        if cmax <= 1e-300:
            break

        if rnd % 2 == 0:
            w = softmax_weights(c, 1e-3 * cmax)
        else:
            K = 32 if n > 1000 else 16
            w = top_weights(c, K)

        g = gradient_for_weights(w, a)
        g -= np.mean(g)
        mag = float(np.mean(np.abs(g)))
        if mag <= 1e-18:
            break

        base = maxdelta / (mag + 1e-30)
        best_cand = None
        best_sur = np.inf

        for rel in (0.5, 1.0, 2.0, 4.0):
            cand = a - base * rel * g
            cand = np.maximum(cand, 0.0)
            s = float(cand.sum())
            if s <= 1e-15:
                continue
            cand = cand / s
            sur = surrogate_score(cand)
            if np.isfinite(sur) and sur < best_sur:
                best_sur = sur
                best_cand = cand

        if best_cand is None:
            break

        score = safe_eval(best_cand)
        if np.isfinite(score) and score < best_score - 1e-10:
            a = best_cand.copy()
            best_score = score
            best_a = a.copy()
            maxdelta = min(1.0 / n, maxdelta * 1.2)
            stall = 0
            if rnd % 10 == 0:
                print(f"[{tag}] round={rnd} score={best_score:.8f}")
        else:
            maxdelta = max(1e-8 / n, maxdelta * 0.7)
            stall += 1
            if stall > 25:
                scale = 0.1 * (1.0 / float(n))
                cand = best_a + rng.normal(0.0, scale, n)
                cand = normalize_seq(cand)
                sp = safe_eval(cand)
                if np.isfinite(sp) and sp < best_score * 1.03:
                    a = cand
                else:
                    a = best_a.copy()
                maxdelta = 0.02 * (1.0 / float(n))
                stall = 0

    return best_a, best_score


def random_perturb_search(a, best_score, deadline, time_budget, tag="perturb"):

    base = a.copy()
    best = a.copy()
    best_sc = best_score

    alpha = 0.05
    start = time.monotonic()
    end = min(deadline, start + time_budget)
    attempts = 12

    for attempt in range(attempts):
        now = time.monotonic()
        if now >= end - 60.0:
            break
        rem = end - now
        per_time = min(90.0, rem - 30.0)
        if per_time < 25.0:
            break

        mult = 1.0 + rng.normal(0.0, alpha, n)
        noise = rng.normal(0.0, 0.1 * alpha / float(n), n)
        cand = base * mult + noise
        cand = normalize_seq(cand)

        sc0 = safe_eval(cand)
        if not np.isfinite(sc0) or sc0 > best_sc * 1.05:
            cand = base.copy()
            sc0 = best_sc

        cand, sc = run_lbfgs_improve(
            cand,
            deadline,
            per_time,
            (0.03, 0.012, 0.004),
            tag=f"{tag}{attempt}",
            use_actual=True,
        )
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()
            base = cand.copy()
            alpha = max(0.01, alpha * 0.75)
            print(f"[{tag}] attempt={attempt} new best={best_sc:.8f} alpha={alpha:.3f}")
        else:
            alpha = max(0.01, min(0.20, alpha * 1.25))

    return best, best_sc

                        (0.02, 0.008, 0.003),
                        tag=f"{tag}{m}",
                        use_actual=True,
                    )
                    if np.isfinite(sc2) and sc2 < best_sc:
                        best_sc = sc2
                        best = ar.copy()
                        print(f"[{tag}] length={m} optimized best={best_sc:.8f}")

    return best, best_sc


def propose_candidate(budget_s=2400, **kwargs):
    if "budget_s" in kwargs:
        budget_s = kwargs["budget_s"]
    elif "time_budget" in kwargs:
        budget_s = kwargs["time_budget"]

    start = time.monotonic()
    deadline = start + max(60.0, float(budget_s) - 75.0)

    parent = None
    try:
        if "height_sequence_1" in globals():
            parent = clean_parent(globals()["height_sequence_1"])
    except Exception:
        parent = None

    if parent is None:
        parent = default_initial(4806)

    best_a = normalize_seq(parent)
    best_s = safe_eval(best_a)
    print(f"initial parent: n={len(best_a)} score={best_s:.8f}")

    # Lower-resolution global exploration.
    if deadline - time.monotonic() > 450.0:
        low_a, low_sur = low_res_search(best_a, deadline, 180.0)
        if low_a is not None:
            low_hi = resize_seq(low_a, len(best_a))
            sc = safe_eval(low_hi)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_a = low_hi.copy()
                print(f"low-res upsample improved: {best_s:.8f}")
            elif np.isfinite(sc) and sc < best_s * 1.01 and deadline - time.monotonic() > 240.0:
                cand, sc2 = run_lbfgs_improve(
                    low_hi,
                    deadline,
                    min(120.0, deadline - time.monotonic() - 240.0),
                    (0.05, 0.02, 0.008),
                    tag="lowup",
                    use_actual=True,
                )
                if np.isfinite(sc2) and sc2 < best_s:
                    best_s = sc2
                    best_a = cand.copy()
                    print(f"low-up optimized improved: {best_s:.8f}")

    # Main high-resolution smooth optimization.
    if deadline - time.monotonic() > 360.0:
        main_budget = deadline - time.monotonic() - 330.0
        main_budget = min(1600.0, max(60.0, main_budget))
        best_a, best_s = run_lbfgs_improve(
            best_a,
            deadline,
            main_budget,
            (0.20, 0.10, 0.04, 0.015, 0.006, 0.002),
            tag="main",
            use_actual=True,
        )

    # Active-set / small-tau polishing.
    if deadline - time.monotonic() > 160.0:
        pol_budget = min(360.0, deadline - time.monotonic() - 150.0)
        if pol_budget > 60.0:
            best_a, best_s = active_polish(best_a, deadline, pol_budget, tag="polish")

    # Escape local minima with small perturbations + short L-BFGS.
    if deadline - time.monotonic() > 180.0:
        pert_budget = min(240.0, deadline - time.monotonic() - 90.0)
        if pert_budget > 80.0:
            best_a, best_s = random_perturb_search(
                best_a, best_s, deadline, pert_budget, tag="perturb"
            )

    # Try only a few nearby lengths.
    if deadline - time.monotonic() > 80.0:
        len_budget = min(90.0, deadline - time.monotonic() - 50.0)
        if len_budget > 40.0:
            best_a, best_s = length_search(best_a, best_s, deadline, len_budget, tag="len")

    # Final short polish.
    if deadline - time.monotonic() > 40.0:
        final_budget = min(50.0, deadline - time.monotonic() - 20.0)
        if final_budget > 20.0:
            best_a, best_s = active_polish(best_a, deadline, final_budget, tag="final")

    best_a = normalize_seq(best_a)
    final_s = safe_eval(best_a)
    if not np.isfinite(final_s):
        # Fallback: if something unexpectedly invalid slipped through, return a valid flat sequence.
        final_fallback = normalize_seq(np.ones(len(best_a)))
        fs = safe_eval(final_fallback)
        if np.isfinite(fs):
            best_a = final_fallback
            final_s = fs
    print(f"final best: n={len(best_a)} score={final_s:.8f}")

    out = []
    for v in best_a:
        f = float(v)
        if not np.isfinite(f) or f < 0.0:
            f = 0.0
        if f > 1000.0:
            f = 1000.0
        out.append(f)
    return out
