```
def _project_simplex(v):
    n=v.size
    if n==0: return np.array([1.0])
    v=np.asarray(v,float)
    u=np.sort(v)[::-1]
    cssum=np.cumsum(u)
    rho=np.where(u > (cssum - 1)/np.arange(1,n+1))[0]
    if len(rho)==0: return np.ones(n)/n
    rho=rho[-1]+1
    theta=(cssum[rho-1]-1)/rho
    w=np.maximum(v-theta,0)
    s=w.sum()
    if s<=0: return ones/n
    return w/s
```

```python
"""
Search for nonnegative step-function coefficients with low peak self-convolution.

The score is
    S(a) = 2 n max_k (a*a)_k / (sum a)^2,
which is scale invariant, so all sequences are internally normalized to sum 1.

This implementation combines:
  * diversified parametric/random initializers,
  * linear-programming direction generators inspired by the SOTA LP step
    (max-sum polar direction and min-cross Chebyshev direction),
  * multiplicative / projected-subgradient descent on a sharpened log-sum-exp
    surrogate of the max autocorrelation,
  * staged resampling to larger lengths, and
  * final exact convolution scoring.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_LP = True
except Exception:
    linprog = None
    HAVE_LP = False

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _normalize(a):
    """Clip to nonnegative finite values and normalize to sum 1."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return np.ones(1, dtype=np.float64)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _next_pow2(m):
    """Smallest power of two >= m."""
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _conv_full(a):
    """Linear convolution a*a by zero-padded FFT, with tiny negatives clipped."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    b = np.fft.irfft(A * A, size)
    b = b[:2 * n - 1]
    b = np.where(b > 0.0, b, 0.0)
    return b


def _score_internal(a):
    """Internal score using FFT convolution. Sequence is normalized internally."""
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b = _conv_full(a)
    if b.size == 0:
        return 1e100
    m = float(np.max(b))
    if not np.isfinite(m):
        return 1e100
    return float(2.0 * float(n) * m)


def _score_direct(a):
    """Exact score using np.convolve, matching the evaluator except list typing."""
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b = np.convolve(a, a)
    b = np.where(b > 0.0, b, 0.0)
    m = float(np.max(b))
    if not np.isfinite(m):
        return 1e100
    return float(2.0 * float(n) * m)


def _grad_linear(a, weights):
    """
    Gradient vector of sum_k weights_k (a*a)_k w.r.t a_i.

    For b = a*a,
        d/d a_i sum_k w_k b_k = 2 * sum_j w_{i+j} a_j.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    w = np.asarray(weights, dtype=np.float64).ravel()
    size = _next_pow2(3 * n - 2)
    ar = a[::-1]
    A = np.fft.rfft(ar, size)
    W = np.fft.rfft(w, size)
    conv = np.fft.irfft(A * W, size).real
    h = conv[n - 1:2 * n - 1]
    h = np.where(h > 0.0, h, 0.0)
    return 2.0 * h


def _softmax_weights(b, rel_sharp):
    """Softmax weights proportional to exp(rel_sharp * (b - max(b)) / max(b))."""
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    m = float(np.max(b))
    if m <= 1e-300 or rel_sharp <= 0.0 or not np.isfinite(rel_sharp):
        return np.ones(n, dtype=np.float64) / float(n)
    arg = float(rel_sharp) * (b - m) / m
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _top_weights(b, m=None, rel=1e-4):
    """Uniform weights over coefficients close to the maximum."""
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    if m is None:
        m = float(np.max(b))
    if m <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    threshold = m - max(1e-300, float(rel) * m)
    w = (b >= threshold).astype(np.float64)
    s = float(w.sum())
    if s <= 0.0:
        w = np.zeros(n, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        s = 1.0
    return w / s


def _mix_weights(b, rel_sharp, top_frac):
    """Mixture of smooth softmax weights and exact near-max weights."""
    b = np.asarray(b, dtype=np.float64).ravel()
    if b.size == 0:
        return np.array([1.0], dtype=np.float64)
    soft = _softmax_weights(b, rel_sharp)
    if top_frac <= 0.0:
        return soft
    if top_frac >= 1.0:
        return _top_weights(b, None, rel=1e-4)
    top = _top_weights(b, None, rel=1e-4)
    w = (1.0 - float(top_frac)) * soft + float(top_frac) * top
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return soft
    return w / s


def _mult_step(p, g, eta):
    """
    Multiplicative projected update:
        p_i * exp(-eta * centered_norm_grad_i)
    then clip and normalize.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    if g.size != p.size:
        return p.copy()
    gc = g - float(np.mean(g))
    scale = float(np.max(np.abs(gc))) if gc.size else 0.0
    if not np.isfinite(scale) or scale <= 1e-300:
        return p.copy()
    z = gc / scale
    z = np.clip(z, -1.0, 1.0)
    factor = np.exp(-float(eta) * z)
    q = p * factor
    q = np.maximum(q, 1e-15)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return p.copy()
    return q / s


def _project_simplex(v):
    """Euclidean projection onto the nonnegative simplex (sum = 1)."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    u = np.sort(v)[::-1]
    idx = np.arange(1, n + 1, dtype=np.float64)
    cssum = np.cumsum(u)
    cond = u > (cssum - 1.0) / idx
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1]) + 1
    theta = (float(cssum[rho - 1]) - 1.0) / float(rho)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def _additive_step(p, g, alpha):
    """Additive projected step for small alphas."""
    p = np.asarray(p, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    if g.size != p.size:
        return p.copy()
    gc = g - float(np.mean(g))
    scale = float(np.max(np.abs(gc))) if gc.size else 0.0
    if not np.isfinite(scale) or scale <= 1e-300:
        return p.copy()
    v = p - float(alpha) * (gc / scale)
    return _project_simplex(v)


def _smooth_perturb(a, rng, amp=0.2, cutoff=8):
    """Low-frequency multiplicative perturbation."""
    a = _normalize(a)
    n = a.size
    if n < 3 or amp <= 0.0 or rng is None:
        return a.copy()
    z = rng.standard_normal(n)
    F = np.fft.rfft(z)
    keep = max(2, min(len(F), 1 + n // max(1, int(cutoff))))
    if keep < len(F):
        F[keep:] = 0.0
    f = np.fft.irfft(F, n).real
    f = f - float(np.mean(f))
    sd = float(np.std(f))
    if not np.isfinite(sd) or sd <= 1e-300:
        return a.copy()
    f = f / sd
    q = a * (1.0 + float(amp) * f)
    q = np.maximum(q, 1e-14)
    return _normalize(q)


def _block_perturb(a, rng, amp=0.2):
    """Random block multiplicative perturbation."""
    a = _normalize(a)
    n = a.size
    if n < 4 or rng is None or amp <= 0.0:
        return a.copy()
    c = a.copy()
    width = max(2, n // 20)
    upper = max(1, n - width + 1)
    start = int(rng.integers(0, upper))
    stop = min(n, start + width)
    c[start:stop] = c[start:stop] * np.exp(
        rng.normal(0.0, float(amp), size=stop - start)
    )
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return _normalize(c)


def _resample(a, n):
    """Resample a normalized sequence to length n by linear interpolation."""
    old = _normalize(a)
    m = old.size
    n = int(max(1, n))
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if m == n:
        return old.copy()
    if m == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

    """Sort helper for (score, array) pairs."""
    return item[0]

def _param_inits(n, rng, given, best):
    """Generate a diverse set of positive initial sequences of length n."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    inits = []

    if best is not None and np.asarray(best).size > 0:
        inits.append(_resample(best, n))
    if given is not None and np.asarray(given).size > 0:
        inits.append(_resample(given, n))

    # Exponential and mixture profiles.
    for alpha in (0.6, 1.0, 1.5, 2.2):
        inits.append(_normalize(np.exp(-alpha * x)))
        inits.append(_normalize(np.exp(-alpha * (1.0 - x))))
        inits.append(_normalize(np.exp(-alpha * np.square(x - 0.5))))
        mix = np.exp(-alpha * x) + 0.6 * np.exp(-alpha * (1.0 - x)) + 1e-3
        inits.append(_normalize(mix))

    # Periodic cosine profiles.
    cos_sets = (
        (0.3, 0.0, 0.0),
        (0.4, 0.2, 0.0),
        (0.3, -0.3, 0.0),
        (0.0, 0.4, 0.0),
        (0.5, 0.2, 0.1),
        (0.2, -0.3, 0.3),
        (0.6, 0.0, -0.2),
    )
    for A1, A2, A3 in cos_sets:
        f = (
            1.0
            + A1 * np.cos(2.0 * np.pi * x)
            + A2 * np.cos(4.0 * np.pi * x)
            + A3 * np.cos(6.0 * np.pi * x)
        )
        f = np.maximum(f, 1e-4)
        inits.append(_normalize(f))

    if rng is not None:
        # Low-degree polynomial profiles.
        for _ in range(5):
            coeff = rng.normal(size=4) * 0.35
            y = 2.0 * x - 1.0
            f = 1.0 + coeff[0] * y + coeff[1] * (y * y) + coeff[2] * (y * y * y) + coeff[3] * (y * y * y * y)
            f = np.maximum(f, 1e-4)
            inits.append(_normalize(f))

        # Random low-frequency Fourier profiles.
        for _ in range(4):
            f = np.ones(n, dtype=np.float64)
            nmodes = max(1, min(8, n // 32))
            for mmode in range(1, nmodes + 1):
                amp = float(rng.uniform(0.0, 0.5) * np.exp(-0.5 * mmode))
                ph = float(rng.uniform(0.0, 2.0 * np.pi))
                f = f + amp * np.cos(2.0 * np.pi * float(mmode) * x + ph)
            f = np.maximum(f, 1e-4)
            inits.append(_normalize(f))

        if best is not None and np.asarray(best).size > 4:
            b = _resample(best, n)
            inits.append(_smooth_perturb(b, rng, 0.25))
            inits.append(_block_perturb(b, rng, 0.25))
            br = b[::-1]
            inits.append(_smooth_perturb(br, rng, 0.25))

        if given is not None and np.asarray(given).size > 4:
            b = _resample(given, n)
            inits.append(_smooth_perturb(b, rng, 0.25))

    # Remove approximate duplicates.
    seen = set()
    out = []
    for ic in inits:
        arr = np.asarray(ic, dtype=np.float64).ravel()
        if arr.size < 1:
            continue
        step = max(1, arr.size // 32)
        key = tuple(np.round(arr[::step], 6))
        if key not in seen:
            seen.add(key)
            out.append(_normalize(arr))

    if not out:
        out.append(_normalize(np.exp(-1.0 * x)))
    return out


def _fill_conv_matrix(p):
    """
    Build the matrix B such that B @ g = p * g (linear convolution),
    where p and g are length n.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    B = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        B[j:j + n, j] = p
    return B


def _run_linprog(c, A_ub, b_ub, A_eq, b_eq, bounds, time_limit):
    """Run HiGHS through scipy if available."""
    if not HAVE_LP:
        return None
    try:
        return linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(max(0.0, time_limit))},
        )
    except Exception:
        pass
    try:
        return linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None


def _lp_min_cross(p, rng=None, obj_noise=0.0, time_limit=1.0):
    """
    For fixed p, solve
        minimize t
        subject to p*q_k <= t, sum(q)=1, q>=0.
    Returns (q, t).
    """
    if not HAVE_LP:
        return None, None
    p = _normalize(p)
    n = p.size
    if n < 2:
        return None, None
    B = _fill_conv_matrix(p)
    m = 2 * n - 1

    A_ub = np.empty((m, n + 1), dtype=np.float64)
    A_ub[:, :n] = B
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m, dtype=np.float64)

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    if rng is not None and obj_noise > 0.0:
        noise = rng.standard_normal(n)
        noise -= noise.mean()
        c[:n] += float(obj_noise) * noise

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]

    res = _run_linprog(c, A_ub, b_ub, A_eq, b_eq, bounds, time_limit)
    if res is None or getattr(res, "x", None) is None:
        return None, None

    x = np.asarray(res.x, dtype=np.float64)
    q = x[:n]
    t = float(x[-1])
    q = np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if not np.isfinite(t) or s <= 1e-300:
        return None, None
    return q / s, t


def _lp_max_sum(p, rng=None, obj_noise=0.0, time_limit=1.0):
    """
    SOTA polar direction:
        maximize sum(g)
        subject to p*g <= 1, g >= 0,
    then return g / sum(g).
    """
    if not HAVE_LP:
        return None
    p = _normalize(p)
    n = p.size
    if n < 2:
        return None
    B = _fill_conv_matrix(p)
    c = np.ones(n, dtype=np.float64)
    if rng is not None and obj_noise > 0.0:
        noise = rng.standard_normal(n)
        noise -= noise.mean()
        c = c + float(obj_noise) * noise

    b_ub = np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = _run_linprog(-c, B, b_ub, None, None, bounds, time_limit)
    if res is None or getattr(res, "x", None) is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-300:
        return None
    return g / s


def _line_search_mix(base, q, base_score, alphas):
    """Line search along base -> normalized mixed sequence."""
    base = _normalize(base)
    q = _normalize(q)
    best = base.copy()
    best_s = float(base_score)
    for a in alphas:
        a = float(a)
        if a <= 0.0:
            p = base
        elif a >= 1.0:
            p = q
        else:
            p = (1.0 - a) * base + a * q
        p = _normalize(np.maximum(p, 0.0))
        s = _score_internal(p)
        if np.isfinite(s) and s < best_s - 1e-14:
            best = p
            best_s = s
    return best, best_s


def _lp_refine(a, deadline, rng, use_min_cross=True, use_max_sum=True, time_limit=0.7):
    """
    Iterate LP directions with exact line search.
    The max-sum direction is the SOTA move; min-cross is an auxiliary move.
    """
    a = _normalize(a)
    n = a.size
    if n < 4 or not HAVE_LP:
        return a, _score_internal(a)

    best = a.copy()
    best_s = _score_internal(a)
    current = a.copy()
    cur_s = best_s

    noimp = 0
    it = 0
    alphas = (0.005, 0.02, 0.05, 0.1, 0.2, 0.35, 0.7)

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline - 1.5:
            break

        cur_s = _score_internal(current)
        if np.isfinite(cur_s) and cur_s < best_s - 1e-13:
            best = current.copy()
            best_s = float(cur_s)

        if cur_s > best_s + 0.05:
            current = best.copy()
            cur_s = best_s

        noise = 0.0 if noimp < 2 else 1e-6 * min(5.0, float(noimp))

        q = None
        if noimp < 3:
            q = _lp_max_sum(current, rng, obj_noise=noise, time_limit=time_limit)

        if q is None or noimp >= 3:
            if use_min_cross and (it % 2 == 1 or noimp >= 3):
                qmin, _ = _lp_min_cross(
                    current, rng, obj_noise=noise, time_limit=time_limit
                )
                if qmin is not None:
                    if q is None:
                        q = qmin
                    else:
                        q = _normalize(0.5 * q + 0.5 * qmin)

        if q is None and use_max_sum:
            q = _lp_max_sum(
                current, rng, obj_noise=noise * 2.0, time_limit=time_limit
            )

        if q is None:
            noimp += 2
            if noimp > 10:
                break
            current = _smooth_perturb(best, rng, 0.12 + 0.03 * min(5.0, float(noimp)))
            continue

        q = _normalize(q)

        best2, s2 = _line_search_mix(best, q, best_s, alphas)

        if noimp > 0 or cur_s > best_s + 1e-12:
            cur2, cs2 = _line_search_mix(current, q, cur_s, alphas)
            if cs2 < s2:
                best2, s2 = cur2, cs2

        if s2 < best_s - 1e-13:
            best = best2
            best_s = float(s2)
            current = best.copy()
            cur_s = best_s
            noimp = 0
        elif s2 < cur_s - 1e-13:
            current = best2
            cur_s = float(s2)
            if noimp > 0:
                noimp = max(0, noimp - 1)
        else:
            noimp += 1
            if noimp == 1:
                current = _smooth_perturb(best, rng, 0.05)
            elif noimp == 2:
                current = _smooth_perturb(best, rng, 0.12)
            elif noimp == 4:
                current = _block_perturb(best, rng, 0.18)
            elif noimp >= 10:
                break

        it += 1

    final_s = _score_internal(best)
    if np.isfinite(final_s) and final_s < best_s:
        best_s = float(final_s)
    return best, best_s


def _gradient_refine(a, deadline, rng, start_sharp=None, max_sharp=None, top_frac_start=None):
    """
    Multiplicative / projected gradient descent on sharpened max-conv surrogate.
    Uses exact score for acceptance.
    """
    a = _normalize(a)
    n = a.size
    if deadline <= time.monotonic():
        return a, _score_internal(a)
    if n == 0:
        return a, 1e100

    if start_sharp is None:
        start_sharp = 8.0 if n <= 512 else 3.0
    if max_sharp is None:
        max_sharp = 12000.0
    if top_frac_start is None:
        top_frac_start = 0.15 if n <= 512 else 0.05

    best = a.copy()
    best_s = _score_internal(a)
    current = a.copy()
    cur_s = best_s

    sharp = float(start_sharp)
    top_frac = float(top_frac_start)
    eta = 0.30 if n <= 512 else 0.14
    fail = 0
    it = 0
    last_good = 0
    add_limit = 2048

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline - 0.2:
            break

        b = _conv_full(current)
        if b.size == 0:
            cur_s = 1e100
        else:
            m = float(np.max(b))
            cur_s = float(2.0 * float(n) * m)

        if not np.isfinite(cur_s):
            current = best.copy()
            cur_s = best_s
            it += 1
            continue

        if cur_s < best_s - 1e-14:
            best = current.copy()
            best_s = float(cur_s)
            last_good = it
            fail = 0
        elif cur_s > best_s + 0.03 * max(1.0, best_s):
            current = best.copy()
            cur_s = best_s
            fail = 0

        w = _mix_weights(b, sharp, top_frac)
        g = _grad_linear(current, w)
        if g.size == n:
            gc = g - float(np.mean(g))
            scale = float(np.max(np.abs(gc)))
        else:
            scale = 0.0

        if not np.isfinite(scale) or scale <= 1e-300:
            current = _smooth_perturb(best, rng, 0.15)
            cur_s = _score_internal(current)
            sharp = max(float(start_sharp), sharp * 0.5)
            eta = 0.30 if n <= 512 else 0.14
            fail = 0
            it += 1
            continue

        if n <= 1024:
            e_list = (eta, eta * 0.5, eta * 0.25, eta * 0.125)
        else:
            e_list = (eta, eta * 0.5)

        accepted = False
        cand = None
        cand_s = cur_s

        for e in e_list:
            q = _mult_step(current, g, e)
            s = _score_internal(q)
            if np.isfinite(s) and s < cand_s - 1e-14:
                cand = q
                cand_s = s
                accepted = True
                break

        if (not accepted) and n <= add_limit:
            for ae in (2.0 / float(n), 1.0 / float(n), 0.5 / float(n)):
                q = _additive_step(current, g, ae)
                s = _score_internal(q)
                if np.isfinite(s) and s < cand_s - 1e-14:
                    cand = q
                    cand_s = s
                    accepted = True
                    break

        if accepted:
            current = cand
            cur_s = cand_s
            eta = min(1.3, eta * 1.08)
            fail = 0
        else:
            eta *= 0.65
            fail += 1
            if fail > 40:
                amp = 0.15 + 0.20 * min(1.0, float(fail) / 200.0)
                current = _smooth_perturb(best, rng, amp)
                if rng is not None and rng.random() < 0.35:
                    current = _block_perturb(current, rng, 0.25)
                cur_s = _score_internal(current)
                sharp = max(float(start_sharp), sharp * 0.4)
                eta = 0.30 if n <= 512 else 0.14
                top_frac = max(0.05, top_frac * 0.7)
                fail = 0

        if it % 30 == 0:
            sharp = min(float(max_sharp), sharp * 1.10)
            top_frac = min(0.80, top_frac + 0.02)

        if it - last_good > 350 and it > 50:
            current = _smooth_perturb(best, rng, 0.30)
            if rng is not None and rng.random() < 0.4:
                current = _block_perturb(current, rng, 0.30)
            cur_s = _score_internal(current)
            sharp = max(float(start_sharp), sharp * 0.5)
            eta = 0.35 if n <= 512 else 0.16
            fail = 0

        it += 1

    final_s = _score_internal(best)
    if np.isfinite(final_s) and final_s < best_s:
        best_s = float(final_s)
    return best, best_s


def _run_stage(n, stage_end, rng, given, best_a, best_score, use_lp):
    """Run a stage of search at length n."""
    n = int(max(2, n))

    if best_a is None:
        best_a = _normalize(np.exp(-1.0 * np.linspace(0.0, 1.0, n, dtype=np.float64)))
        best_score = _score_internal(best_a)

    if stage_end - time.monotonic() < 25.0:
        return best_a, best_score

    inits = _param_inits(n, rng, given, best_a)

    scored = []
    for arr in inits:
        if time.monotonic() >= stage_end - 20.0:
            break
        a = _normalize(arr)
        if a.size != n:
            a = _resample(a, n)
        s = _score_internal(a)
        if np.isfinite(s):
            scored.append((float(s), a))

    if not scored:
        a = _normalize(np.exp(-1.0 * np.linspace(0.0, 1.0, n, dtype=np.float64)))
        scored.append((float(_score_internal(a)), a))

    scored.sort(key=_score_key)

    for count in range(len(scored)):
        if time.monotonic() >= stage_end - 25.0:
            break
        s_init, a_init = scored[count]

        rem = float(stage_end - time.monotonic())
        if count == 0:
            sub_deadline = min(stage_end - 10.0, time.monotonic() + max(30.0, 0.55 * rem))
        else:
            if count > 8:
                break
            sub_deadline = min(stage_end - 25.0, time.monotonic() + max(15.0, 0.10 * rem))

        a0 = np.asarray(a_init, dtype=np.float64).copy()

        if use_lp and n <= 512 and sub_deadline - time.monotonic() > 50.0:
            sub_len = float(sub_deadline - time.monotonic())
            lp_deadline = min(sub_deadline - 5.0, time.monotonic() + max(15.0, 0.5 * sub_len))
            a1, _ = _lp_refine(
                a0,
                lp_deadline,
                rng,
                use_min_cross=True,
                use_max_sum=True,
                time_limit=0.6 if n <= 384 else 0.35,
            )
            a2, s2 = _gradient_refine(a1, sub_deadline - 2.0, rng)
        else:
            a2, s2 = _gradient_refine(a0, sub_deadline - 2.0, rng)

        if np.isfinite(s2) and s2 < best_score:
            best_score = float(s2)
            best_a = np.asarray(a2, dtype=np.float64).copy()
            print(f"stage n={n} start={count} score={best_score:.10f}", flush=True)

    return best_a, best_score


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for a low-score nonnegative coefficient list
    for the given time budget and returns the best list found.
    """
    start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0

    hard_end = start + max(60.0, budget)
    reserve = 40.0 if budget >= 600.0 else 10.0
    end = hard_end - reserve
    end = max(start + 30.0, end)

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 0
    rng = np.random.default_rng((seed_int ^ 0x9E3779B9) & 0xFFFFFFFFFFFFFFFF)

    given = _get_given_sequence()

    best_a = None
    best_score = 1e100

    if given is not None:
        best_a = _normalize(given)
        if best_a.size > 8192:
            best_a = _resample(best_a, 4096)
        s = _score_internal(best_a)
        if np.isfinite(s):
            best_score = float(s)

    if best_a is None or not np.isfinite(best_score):
        n0 = 256
        best_a = _normalize(np.exp(-1.0 * np.linspace(0.0, 1.0, n0, dtype=np.float64)))
        best_score = _score_internal(best_a)

    print(
        f"initial best len={best_a.size} score={best_score:.10f} lp_available={HAVE_LP}",
        flush=True,
    )

    scale = max(0.05, min(1.0, budget / 2400.0))

    try:
        stages = (
            (128, 40),
            (256, 720),
            (512, 380),
            (1024, 380),
            (2048, 460),
            (4096, 220),
        )
        for n, alloc in stages:
            if time.monotonic() >= end - 30.0:
                break
            remaining_to_end = end - time.monotonic()
            if remaining_to_end <= 12.0:
                break
            sec = int(alloc * scale)
            sec = min(sec, int(remaining_to_end))
            if sec < 15:
                continue
            stage_end = time.monotonic() + float(sec)
            use_lp = (n <= 256) or (n <= 512 and sec > 120)
            best_a, best_score = _run_stage(
                int(n), stage_end, rng, given, best_a, best_score, use_lp
            )
            print(
                f"after stage n={n} best_len={best_a.size} best_score={best_score:.10f}",
                flush=True,
            )
    except Exception as exc:
        print("stage error:", exc, flush=True)

    # Final polishing of the best sequence.
    try:
        if best_a is not None and time.monotonic() < end - 10.0:
            final_end = min(end, hard_end - 12.0)
            if final_end > time.monotonic() + 10.0:
                n = int(best_a.size)
                if n > 4096:
                    best_a = _resample(best_a, 4096)
                    best_score = _score_internal(best_a)
                    n = 4096

                if n <= 512 and HAVE_LP and final_end - time.monotonic() > 30.0:
                    lp_deadline = min(
                        final_end - 20.0,
                        time.monotonic() + min(120.0, 0.35 * (final_end - time.monotonic())),
                    )
                    a_tmp, _ = _lp_refine(
                        best_a,
                        lp_deadline,
                        rng,
                        use_min_cross=True,
                        use_max_sum=True,
                        time_limit=0.5,
                    )
                    a_fin, s_fin = _gradient_refine(a_tmp, final_end - 3.0, rng)
                    if np.isfinite(s_fin) and s_fin < best_score:
                        best_a = np.asarray(a_fin, dtype=np.float64).copy()
                        best_score = float(s_fin)
                else:
                    a_fin, s_fin = _gradient_refine(best_a, final_end - 3.0, rng)
                    if np.isfinite(s_fin) and s_fin < best_score:
                        best_a = np.asarray(a_fin, dtype=np.float64).copy()
                        best_score = float(s_fin)
    except Exception as exc:
        print("final polish error:", exc, flush=True)

    out = _normalize(best_a)
    out = np.maximum(out, 0.0)

    # Final exact score check.
    if out.size <= 8192:
        final_score = _score_direct(out)
    else:
        final_score = _score_internal(out)
    if np.isfinite(final_score):
        best_score = float(final_score)

    seq = [float(x) for x in out]
    print(f"final len={len(seq)} direct_score={best_score:.10f}", flush=True)
    return seq
```
