import numpy as np
import time
from scipy.fft import fft, ifft

def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def _score(a):
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _normalize(a, target_sum):
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _mirror_step(a, lr, top_k=3, smooth_tau=0.1):
    """One mirror descent step with top-k smoothing."""
    n = len(a)
    S = float(np.sum(a))
    b = _fft_conv(a, a)
    
    # Top-k peaks with exponential weighting (smoothed max)
    sorted_idx = np.argsort(b)[::-1]
    top_vals = b[sorted_idx[:top_k]]
    # Weight by exp(val/tau) for smooth max
    weights = np.exp(top_vals / (smooth_tau * S))
    weights /= np.sum(weights)
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, sorted_idx[:top_k]):
        k = int(k)
        # (a*a)_k = sum_j a_j * a_{k-j}

        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi+1):
            grad[j] += w * a[k - j]

Minimize 2n*max(autocorr)/(sum)^2 using:
1. Multi-length exploration (parent length + longer sequences)
2. Mirror descent (multiplicative weight updates) with smoothed max objective
3. Structured local search with mirror-pair balancing and Fourier moves
4. Adaptive learning rate with backtracking

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    """Fast score using FFT (for internal tracking, may differ slightly from evaluator)."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    """Normalize array to target sum, preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _mirror_step(a, lr, top_k, tau):
    """
    One mirror descent step with top-k exponential smoothing.
    Uses multiplicative updates: a_j *= exp(-lr * g_j / S)
    """
    n = len(a)
    S = float(np.sum(a))
    b = _fft_conv(a, a)

    # Get top-k peak indices
    if top_k >= n:
        top_k = n
    top_indices = np.argsort(b)[::-1][:top_k]
    top_values = b[top_indices]

    # Exponential weights (smoothed max)
    scale = max(tau * S, 1e-10)
    weights = np.exp(top_values / scale - top_values[0] / scale)
    weights /= np.sum(weights)

    # Compute weighted subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):
        k = int(k)
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            grad[j_lo:j_hi + 1] += w * a[k - np.arange(j_lo, j_hi + 1)]

    # Multiplicative update (mirror descent with entropy mirror)
    a_new = a * np.exp(-lr * grad / S)
    a_new = np.maximum(a_new, 1e-15)
    a_new = _normalize(a_new, S)
    return a_new, float(b[int(np.argmax(b))])


def _backtrack_mirror(a, lr, top_k, tau, current_score, max_iter=5):
    """Mirror step with backtracking line search."""
    best_a = a
    best_sc = current_score
    cur_lr = lr
    for _ in range(max_iter):
        a_new, _ = _mirror_step(a, cur_lr, top_k, tau)
        sc = _score_fast(a_new)
        if sc < current_score:
            return a_new, sc, cur_lr
        cur_lr *= 0.5
    return best_a, best_sc, lr


def _fourier_perturb(a, rng, n_terms=5):
    """Add a small random Fourier perturbation (preserves mean approximately)."""
    n = len(a)
    S = float(np.sum(a))
    k = np.arange(n)
    pert = np.zeros(n)
    for _ in range(n_terms):
        freq = rng.integers(1, max(2, n // 2))
        amp = rng.uniform(0.01, 0.05) * S / n
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(2 * np.pi * freq * k / n + phase)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    a_new = _normalize(a_new, S)
    return a_new


def _mirror_pair_local_search(a, n_trials, rng):
    """
    Local search using mirror-pair balancing moves.
    For peak lag k*, transfer between a_i and a_{k*-i} to reduce the peak.
    """
    n = len(a)
    S = float(np.sum(a))
    best = a.copy()
    best_score = _score_fast(best)

    for _ in range(n_trials):
        b = _fft_conv(best, best)
        k_star = int(np.argmax(b))
        # Find the pair that contributes most to the peak
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        products = best[j_lo:j_hi + 1] * best[k_star - np.arange(j_lo, j_hi + 1)]
        if len(products) == 0:
            continue
        top_pairs = np.argsort(products)[::-1][:20]

        for pi in top_pairs[:5]:
            j = int(j_lo + top_pairs[pi])
            partner = k_star - j
            if partner < 0 or partner >= n or partner == j:
                continue
            # Try transferring from larger to smaller
            a_i = best[j]
            a_p = best[partner]
            larger = j if a_i > a_p else partner
            smaller = partner if a_i > a_p else j
            for frac in [0.1, 0.05, 0.02, 0.01]:
                delta = frac * best[larger]
                if delta < 1e-15:
                    continue
                a_new = best.copy()
                a_new[larger] -= delta
                a_new[smaller] += delta
                if a_new[larger] < 0:
                    continue
                sc = _score_fast(a_new)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best = a_new
                    break
    return best, best_score

    """Local search with random coordinate swaps."""
    n = len(a)
    S = float(np.sum(a))
    best = a.copy()
    best_score = _score_fast(best)
    eps_base = S / n

    for _ in range(n_trials):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        for frac in [0.01, 0.005, 0.001]:
            delta = frac * eps_base
            for sign in [1, -1]:
                a_new = best.copy()
                a_new[i] += sign * delta
                a_new[j] -= sign * delta
                if a_new[i] < 0 or a_new[j] < 0:
                    continue
                sc = _score_fast(a_new)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best = a_new
                    break
            else:
                continue
            break
    return best, best_score


def _run_optimization(a, deadline, rng, label=""):
    """
    Run the full optimization pipeline on a sequence:
    1. Mirror descent with adaptive lr
    2. Mirror-pair local search
    3. Fourier perturbation + restart
    4. Random swap local search
    """
    n = len(a)
    S = float(np.sum(a))
    best = a.copy()
    best_score = _score_fast(a)
    print(f"  [{label}] Starting: score={best_score:.6f}")

    # Phase A: Mirror descent
    lr = 0.5
    top_k = 3
    tau = 0.5
    stuck = 0
    it = 0
    deadline_a = deadline - max(60, (deadline - time.monotonic()) * 0.3)

    while time.monotonic() < deadline_a:
        a_new, max_b = _mirror_step(a, lr, top_k, tau)
        sc = _score_fast(a_new)
        it += 1

        if sc < best_score - 1e-9:
            if sc < _score_fast(a) * 0.999:
                # Good improvement, accept
                a = a_new
                best_score = sc
                best = a.copy()
                stuck = 0
                lr = min(lr * 1.05, 2.0)
            else:
                a = a_new
                stuck = max(0, stuck - 1)
        else:
            stuck += 1
            if stuck > 30:
                # Restart from best with perturbation
                a = _fourier_perturb(best, rng, n_terms=3)
                lr = 0.3
                stuck = 0
            elif stuck > 10:
                lr *= 0.9
                top_k = min(top_k + 1, 7)
                tau = max(tau * 0.8, 0.1)

        if it % 200 == 0:
            print(f"    [{label}] mirror iter {it}: best={best_score:.6f}, lr={lr:.3f}")

    print(f"  [{label}] After mirror descent: {best_score:.6f}")

    # Phase B: Mirror-pair local search
    remaining_b = deadline - time.monotonic()
    if remaining_b > 30:
        n_trials_b = int(remaining_b * 0.4 / max(0.01, (time.monotonic() - (deadline - remaining_b)) + 1e-10))
        # Estimate trials: each trial does ~1 FFT conv (~0.2ms for n=5000)
        trials_est = int(remaining_b * 0.35 / (0.002 * (1 + n / 5000)))
        trials_est = max(100, min(trials_est, 50000))
        best, best_score = _mirror_pair_local_search(best, trials_est, rng)
        print(f"  [{label}] After mirror-pair LS: {best_score:.6f}")

    # Phase C: More mirror descent from improved point
    remaining_c = deadline - time.monotonic()
    if remaining_c > 60:
        a = best.copy()
        lr = 0.2
        top_k = 5
        tau = 0.3
        stuck = 0
        it = 0
        deadline_c = deadline - max(30, remaining_c * 0.3)

        while time.monotonic() < deadline_c:
            a_new, _ = _mirror_step(a, lr, top_k, tau)
            sc = _score_fast(a_new)
            it += 1
            if sc < best_score - 1e-9:
                a = a_new
                best_score = sc
                best = a.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 20:
                    a = _fourier_perturb(best, rng, n_terms=2)
                    lr = 0.1
                    stuck = 0
                elif stuck > 5:
                    lr *= 0.92

        print(f"  [{label}] After 2nd mirror: {best_score:.6f}")

    # Phase D: Random swap local search
    remaining_d = deadline - time.monotonic()
    if remaining_d > 20:
        trials_d = max(100, int(remaining_d / 0.003))
        best, best_score = _random_swap_local_search(best, min(trials_d, 30000), rng)
        print(f"  [{label}] After random swap LS: {best_score:.6f}")

    return best, best_score


def _make_structured_init(n, rng, target_sum):
    """Create diverse structured initializations."""
    inits = []
    k = np.arange(n)

    # Cosine modulations
    for freq in [1, 2, 3, 5, 7, 12]:
        sig = np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * k / n), 0.01)
        inits.append(_normalize(sig, target_sum))

    # Multi-frequency
    inits.append(_normalize(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * k / n) +
                                        0.2 * np.cos(2 * np.pi * 8 * k / n + 0.5), 0.01), target_sum))
    inits.append(_normalize(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 5 * k / n) +
                                        0.2 * np.cos(2 * np.pi * 11 * k / n), 0.01), target_sum))

    # Bimodal
    inits.append(_normalize(np.concatenate([np.ones(n // 2) * 1.5, np.ones(n - n // 2) * 0.5]), target_sum))

    # V-shape
    inits.append(_normalize(1.0 + 0.5 * np.abs(2 * k / n - 1), target_sum))

    # Gaussian-like
    inits.append(_normalize(np.exp(-2 * ((k / n) - 0.5) ** 2) + 0.1, target_sum))

    # Random
    inits.append(_normalize(rng.random(n) + 0.1, target_sum))

    # Pentatonic-like (irrational frequency)
    for freq in [n * 0.618, n * 0.382]:
        inits.append(_normalize(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * k / n), 0.01), target_sum))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2n*max(autocorr)/(sum)^2 using mirror descent with multiplicative
    updates, structured local search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Get parent sequence
    n_parent = 5582
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
            n_parent = parent.size
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.ones(n_parent)
        n_parent = parent.size

    print(f"Parent length: {n_parent}")

    # Determine candidate lengths to try
    # Parent length is priority; longer lengths may help
    candidate_lengths = [n_parent]
    if n_parent < 15000:
        candidate_lengths.append(10000)
    if n_parent < 30000:
        candidate_lengths.append(20000)
    # Remove duplicates, sort by priority (parent first)
    candidate_lengths = list(dict.fromkeys(candidate_lengths))

    best_overall = None
    best_overall_score = np.inf

    # Allocate time: give parent length the most time
    n_lengths = len(candidate_lengths)
    time_alloc = []
    for i, L in enumerate(candidate_lengths):
        if i == 0:
            time_alloc.append(0.50)  # 50% for parent
        elif i == 1:
            time_alloc.append(0.30)  # 30% for second
        else:
            time_alloc.append(0.20)  # 20% for rest
    # Normalize
    total_frac = sum(time_alloc)
    time_alloc = [f / total_frac * (total_deadline - time.monotonic() - 60) for f in time_alloc]

    for idx, n in enumerate(candidate_lengths):
        phase_deadline = time.monotonic() + time_alloc[idx]
        print(f"\n{'='*60}")
        print(f"Processing n={n} (time budget: {time_alloc[idx]:.0f}s)")
        print(f"{'='*60}")

        if time.monotonic() > phase_deadline - 30:
            break

        S = np.sqrt(2.0 * n)

        # Build initializations
        inits = []
        if n == n_parent and parent is not None:
            inits.append(_normalize(parent.copy(), S))

        # Add structured inits
        struct_inits = _make_structured_init(n, rng, S)
        inits.extend(struct_inits[:5])  # Limit to 5 structured

        # Quick screen: pick the best 2-3 initializations
        init_scores = []
        for i, sig in enumerate(inits):
            sc = _score_fast(sig)
            init_scores.append((sc, i, sig))
        init_scores.sort(key=lambda x: x[0])

        # Run optimization on top initializations
        best_local = None
        best_local_score = np.inf
        n_init_to_try = min(3, len(inits))
        per_init_time = (phase_deadline - time.monotonic() - 10) / n_init_to_try

        for trial_idx in range(n_init_to_try):
            if time.monotonic() > phase_deadline - 10:
                break
            sc, i, sig = init_scores[trial_idx]
            init_deadline = time.monotonic() + per_init_time
            label = f"n={n}, init{trial_idx}"
            print(f"\n  Trying init {trial_idx} (score={sc:.6f}), budget={per_init_time:.0f}s")

            local_best, local_score = _run_optimization(sig, init_deadline, rng, label)

            if local_score < best_local_score:
                best_local_score = local_score
                best_local = local_best

        # Verify best with actual evaluator
        if best_local is not None:
            actual = float(evaluate_sequence(best_local.tolist()))
            print(f"  n={n}: FFT score={best_local_score:.6f}, actual={actual:.6f}")
            if np.isfinite(actual) and actual < best_overall_score:
                best_overall_score = actual
                best_overall = best_local.copy()
                print(f"  ** New best: {best_overall_score:.6f} **")

    # Final: if we have time, do one more refinement on the best
    remaining = total_deadline - time.monotonic()
    if best_overall is not None and remaining > 60:
        print(f"\nFinal refinement with {remaining:.0f}s remaining")
        n = len(best_overall)
        S = np.sqrt(2.0 * n)
        a = _normalize(best_overall.copy(), S)
        refined, ref_score = _run_optimization(a, total_deadline, rng, "final")

        actual = float(evaluate_sequence(refined.tolist()))
        print(f"  Final: FFT={ref_score:.6f}, actual={actual:.6f}")
        if np.isfinite(actual) and actual < best_overall_score:
            best_overall_score = actual
            best_overall = refined.copy()

    # Ensure we return the best verified
    if best_overall is None:
        # Fallback: return parent
        S = np.sqrt(2.0 * n_parent)
        best_overall = _normalize(parent, S)
        best_overall_score = float(evaluate_sequence(best_overall.tolist()))

    final_check = float(evaluate_sequence(best_overall.tolist()))
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {final_check:.6f}")
    print(f"Sequence length: {len(best_overall)}")
    print(f"{'='*60}")

    return [float(max(0.0, min(1000.0, x))) for x in best_overall]
