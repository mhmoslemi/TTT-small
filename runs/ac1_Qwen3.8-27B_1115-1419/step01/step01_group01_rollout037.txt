    b = np.fft.ifft(F * F).real[:2*len(a)-1]

    s = e.sum()
    if s <= 0:

        nc = min(8, n//8)
        phases = rng.uniform(0, 2*np.pi, nc)
        amps = 0.3 / (np.arange(1, nc+1)**0.7)
        a = 1.0
        for k in range(nc):
            a += amps[k] * np.cos(2*np.pi*(k+1)*t + phases[k])
        a = np.maximum(a, 0)
        if a.sum() > 0:
            inits.append(a / a.sum())
    
    # Spectral targets
    L = 2*n - 1
    tc = np.linspace(0, 1, L)
    for taper in [0, 0.5, 1.0, 2.0]:
        target = np.ones(L)
        if taper > 0:
            edge = np.maximum(tc - 0.5, 0) * 2  # 0 at center, 1 at edges
            target *= (1 - taper * edge**2)
        target = np.maximum(target, 0.1)
        if inits:
            a_sp, _ = _spectral_sqrt(target, inits[0], n, iters=25)
            inits.append(a_sp)
        else:
            a_sp, _ = _spectral_sqrt(target, np.ones(n)/n, n, iters=25)
            inits.append(a_sp)
    
    return inits
