    """Compute score, return inf if invalid."""
    if len(a) == 0:
        return float('inf')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2 * len(a) * c.max() / (s**2))

def lse_smooth(c, tau):
    """Log-sum-exp smoothing of max."""
    c_max = c.max()
    return c_max + tau * np.log(np.exp((c - c_max) / tau).sum())

def lse_smooth_grad(c, a, tau):
    """Gradient of LSE-smoothed max autocorrelation w.r.t. a."""
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    return 2.0 * np.correlate(w, a, mode='valid')

def simplex_project(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.maximum(v, 0)
    v_sum = v.sum()
    if v_sum > 0:
        v = v * s / v_sum
    else:
        v = np.full(len(v), s / len(v))
    return v

def adam_step(params, grads, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step. params and grads are lists of arrays."""
    m_new = []
    v_new = []
    p_new = []
    for p, g in zip(params, grads):
        m_t = beta1 * m + (1 - beta1) * g
        v_t = beta2 * v + (1 - beta2) * g**2
        m_hat = m_t / (1 - beta1**t)
        v_hat = v_t / (1 - beta2**t)
        p_t = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        m_new.append(m_t)
        v_new.append(v_t)
        p_new.append(p_t)
    return p_new, m_new, v_new

def optimize_smooth(a, n, tau_schedule, lr, n_iterations, deadline, rng):
    """Optimize using Adam on LSE-smoothed objective. Returns (best_a, best_score)."""
    best_score = safe_score(a)
    best_a = a.copy()
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        
        m = [np.zeros(n)]
        v = [np.zeros(n)]
        t = 0
        
        for _ in range(n_iterations):
            if time.monotonic() > deadline:
                break
            t += 1
            
            a_current = simplex_project(a, 1.0)
            c = np.convolve(a_current, a_current)
            val = lse_smooth(c, tau)
            grad = lse_smooth_grad(c, a_current, tau)
            
            # Negative gradient (we minimize)
            a, m, v = adam_step([a], [-grad], m, v, t, lr)
            
            # Every 50 steps, check score
            if t % 50 == 0:
                a_check = simplex_project(a, 1.0)
                s = safe_score(a_check)
                if s < best_score:
                    best_score = s
                    best_a = a_check.copy()
    
    return best_a, best_score

def multi_peak_subgradient(a, n_peaks=20, step_size=0.01, deadline=None, decay=0.999):
    """Multi-peak subgradient descent. Returns best (a, score)."""
    best_score = safe_score(a)
    best_a = a.copy()
    
    while time.monotonic() < deadline:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s  # normalize to sum 1
        
        c = np.convolve(a, a)
        c_max = c.max()
        
        # Get top peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:min(n_peaks, len(c))]
        
        # Compute subgradient (average of top peaks)
        g = np.zeros(len(a))
        for k in active:
            lo = max(0, k - len(a) + 1)
            hi = min(len(a) - 1, k)
            g[lo:hi+1] += 2.0 * a[k - hi : k - lo + 1][::-1]
        g /= len(active)
        
        # Step
        a = a - step_size * g
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        s_val = safe_score(a)
        if s_val < best_score:
            best_score = s_val
            best_a = a.copy()
        
        step_size *= decay
    
    return best_a, best_score

def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramps (various slopes)
    for alpha in [-0.8, -0.5, -0.3, 0.3, 0.5, 0.8]:
        x = 1.0 + alpha * np.linspace(0, 1, n)
        x = np.maximum(x, 0.01)
        inits.append(x / x.sum())
    
    # Gaussian profiles at various positions
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            x = np.exp(-((np.arange(n) - center * n)**2) / (2 * (width * n)**2))
            inits.append(x / x.sum())
    
    # Exponential decay (various rates)
    for rate in [1, 2, 3, 5]:
        x = np.exp(-rate * np.arange(n) / n)
        inits.append(x / x.sum())
    
    # Quadratic profiles
    for a_coeff in [-2, -1, 1, 2]:
        x = 1.0 + a_coeff * ((np.arange(n) / (n-1)) - 0.5)**2
        x = np.maximum(x, 0.01)
        inits.append(x / x.sum())
    
    # Random smooth (low-frequency Fourier)
    for _ in range(3):
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2*np.pi, size=4)
        amps = rng.uniform(0.1, 0.5, size=4)
        x = np.ones(n)
        for f, p, a in zip(freqs, phases, amps):
            x += a * np.cos(2 * np.pi * f * np.arange(n) / n + p)
        x = np.maximum(x, 0.01)
        inits.append(x / x.sum())
    
    # Bimodal
    for gap in [0.3, 0.5]:
        x1 = np.exp(-((np.arange(n) - 0.25*n)**2) / (2*(0.1*n)**2))
        x2 = np.exp(-((np.arange(n) - (0.25+gap)*n)**2) / (2*(0.1*n)**2))
        x = x1 + x2
        inits.append(x / x.sum())
    
    return inits

- $k=0$: only $i=0$ is valid. $(a*a)[0] = a[0]^2$.

$a[0]^2 \approx 2a[0]a[1] \approx 2a[0]a[2]+a[1]^2 \approx ... \approx (a*a)[n-1]$

For coordinate $j$, the objective (with $\sum a = 1$ and others fixed) is:
$\max_k (a*a)[k]$ where $(a*a)[k]$ depends on $a[j]$ through the terms where $i=j$ or $k-i=j$.

Specifically, $(a*a)[k]$ gets contributions from $a[j]a[k-j]$ (when $i=j$) and $a[k-j]a[j]$ (when $k-i=j$, i.e., $i=k-j$).

So $\partial (a*a)[k] / \partial a[j] = 2a[k-j]$ (for valid $k-j$), as before.

The gradient of the max w.r.t. $a[j]$ is $2 \sum_{k \in A} w_k a[k-j]$ where $A$ is the active set.

1. Pick a coordinate $j$.
2. Compute the optimal value $a[j]^*$ that minimizes $\max_k (a*a)[k]$ (with other coordinates fixed).
3. Update $a[j] = a[j]^*$ and renormalize.

                a_trial[j] = max(0.0, a[j] - frac * 0.01 * g_j / max(g_j, 1e-10))
                a_trial[j] = max(0.0, a[j] - frac * 0.01)
                a_trial = a_trial / a_trial.sum()
                s_trial = safe_score(a_trial)
                if s_trial < best_s * 0.9999:
                    a = a_trial
                    best_s = s_trial
                    best_a = a.copy()
                    break
    
    return best_a, best_s

def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / max(n - 1, 1)
    
    # Linear ramps
    for alpha in np.linspace(-0.9, 0.9, 13):
        v = 1.0 + alpha * x
        v = np.maximum(v, 0.01)
        inits.append(v / v.sum())
    
    # Gaussian profiles
    for center in [0.15, 0.25, 0.35, 0.5, 0.65, 0.75, 0.85]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            v = np.exp(-0.5 * ((x * n - center * n) / (width * n)) ** 2)
            inits.append(v / v.sum())
    
    # Exponential profiles
    for rate in [1, 2, 3, 5, 8]:
        v = np.exp(-rate * x)
        inits.append(v / v.sum())
        v = np.exp(-rate * (1 - x))
        inits.append(v / v.sum())
    
    # Power-law profiles
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        v = (x + 0.01) ** p
        inits.append(v / v.sum())
        v = ((1 - x) + 0.01) ** p
        inits.append(v / v.sum())
    
    #Smooth asymmetric (skewed Gaussian)
    for skew in [1, 2, 3]:
        v1 = np.exp(-0.5 * ((x - 0.3) * 4) ** 2)
        v2 = np.exp(-0.5 * ((x - 0.7) / (1.0 / skew)) ** 2)
        v = v1 + v2 * 0.5
        inits.append(v / v.sum())
    
    # Random smooth
    for _ in range(5):
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.1, 0.6, size=3)
        v = np.ones(n)
        for f, p, a in zip(freqs, phases, amps):
            v += a * np.cos(2 * np.pi * f * x + p)
        v = np.maximum(v, 0.01)
        inits.append(v / v.sum())
    
    return inits

def propose_candidate(budget_s=2400, **kwargs):
    """Main search: multi-phase optimization to minimize the evaluation score."""
    global rng_normal
    deadline = time.monotonic() + budget_s - 30
    rng = np.random.default_rng(42)
    
    best_a = None
    best_score = float('inf')
    
    # Load previous best if available
    if 'height_sequence_1' in globals():
        prev = np.array(height_sequence_1, dtype=np.float64)
        prev = np.maximum(prev, 0)
        s_prev = prev.sum()
        if s_prev > 0.01:
            prev = prev / s_prev
            sc = safe_score(prev)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = prev.copy()
            print(f"Previous best: n={len(prev)}, score={sc:.6f}")
    
    # Phase 1: Explore different lengths with diverse initializations
    print("\n=== Phase 1: Length exploration ===")
    lengths = [1024, 2048, 4096, 6144, 8192, 12288]
    
    time_per_length = min(180.0, (budget_s - 600) / len(lengths))
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        
        print(f"\n  Length n={n}, time_budget={time_per_length:.0f}s")
        
        inits = generate_inits(n, rng)
        
        # Include interpolation of previous best
        if best_a is not None:
            prev_n = len(best_a)
            if prev_n != n:
                x_old = np.linspace(0, 1, prev_n)
                x_new = np.linspace(0, 1, n)
                from scipy.interpolate import interp1d
                f_interp = interp1d(x_old, best_a, kind='cubic', fill_value='extrapolate')
                a_interp = np.maximum(f_interp(x_new), 0)
                if a_interp.sum() > 0:
                    inits.append(a_interp / a_interp.sum())
        
        # Try a subset of inits
        n_to_try = min(len(inits), max(4, int(time_per_length / 30)))
        
        tau_base = 5.0 / n
        tau_sched = [tau_base, tau_base * 0.5, tau_base * 0.25, tau_base * 0.1]
        lr = 0.1
        
        for idx in range(n_to_try):
            if time.monotonic() > deadline - 400:
                break
            
            init = inits[idx]
            t_alloc = min(45.0, time_per_length - 10)
            
            a_opt, s_opt = adam_optimize(
                init, n, tau_sched, lr,
                max_iters=300,
                deadline=time.monotonic() + t_alloc
            )
            
            if s_opt < best_score:
                best_score = s_opt
                best_a = a_opt.copy()
                print(f"    n={n}, init[{idx}]: {s_opt:.6f} *** NEW BEST ***")
        
        # Quick subgradient polish on the best from this length
        if best_a is not None and len(best_a) == n:
            a_pol, s_pol = subgradient_polish(
                best_a, deadline=time.monotonic() + min(30, time_per_length - 10),
                initial_step=0.02, n_peaks=20, decay=0.998
            )
            if s_pol < best_score:
                best_score = s_pol
                best_a = a_pol.copy()
                print(f"    n={n}, after polish: {s_pol:.6f} *** NEW BEST ***")
    
    # Phase 2: Extended optimization on best length
    print(f"\n=== Phase 2: Extended optimization (n={len(best_a)}, score={best_score:.6f}) ===")
    
    n_best = len(best_a)
    a = best_a.copy()
    a = a / a.sum()
    
    # Multiple rounds of refinement with different strategies
    phase2_deadline = deadline - 120
    round_num = 0
    
    while time.monotonic() < phase2_deadline:
        round_num += 1
        print(f"\n  Round {round_num}: current best={best_score:.6f}")
        
        # Strategy A: LSE-smoothed gradient with very small tau (near-exact)
        tau = 1.0 / n_best
        a_cur = a.copy()
        m = np.zeros(n_best)
        v = np.zeros(n_best)
        t = 0
        lr = 0.05
        
        max_iters = 500
        t_start = time.monotonic()
        t_limit = min(120, phase2_deadline - time.monotonic())
        
        for _ in range(max_iters):
            if time.monotonic() > time.monotonic() + t_limit:
                break
            if time.monotonic() > phase2_deadline:
                break
            t += 1
            grad, _ = lse_gradient(a_cur, tau)
            m = 0.9 * m + 0.1 * (-grad)
            v = 0.999 * v + 0.001 * (-grad) ** 2
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            a_cur = a_cur - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_cur = np.maximum(a_cur, 0)
            a_cur = a_cur / a_cur.sum()
            
            if t % 200 == 0:
                s = safe_score(a_cur)
                if s < best_score:
                    best_score = s
                    best_a = a_cur.copy()
        
        # Strategy B: Subgradient polish
        a_pol, s_pol = subgradient_polish(
            a_cur, deadline=time.monotonic() + min(60, phase2_deadline - time.monotonic() - 5),
            initial_step=0.015, n_peaks=40, decay=0.999
        )
        if s_pol < best_score:
            best_score = s_pol
            best_a = a_pol.copy()
            print(f"    Subgradient improved to {s_pol:.6f}")
        
        # Strategy C: Perturbation to escape local minimum
        a_pert = best_a + rng.normal(0, 0.02, n_best)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        s_pert = safe_score(a_pert)
        
        if s_pert < best_score:
            best_score = s_pert
            best_a = a_pert.copy()
            print(f"    Perturbation improved to {s_pert:.6f}")
        else:
            # Try optimization from perturbed point
            a_pert_opt, s_pert_opt = adam_optimize(
                a_pert, n_best,
                [5.0/n_best, 2.0/n_best, 1.0/n_best],
                0.08, 200,
                deadline=time.monotonic() + min(30, phase2_deadline - time.monotonic() - 5)
            )
            if s_pert_opt < best_score:
                best_score = s_pert_opt
                best_a = a_pert_opt.copy()
                print(f"    Perturbed optimization improved to {s_pert_opt:.6f}")
        
        a = best_a.copy()
    
    # Phase 3: Final fine polishing
    print(f"\n=== Phase 3: Final polishing (score={best_score:.6f}) ===")
    a = best_a.copy()
    a = a / a.sum()
    
    # Very fine subgradient steps
    step = 0.005
    n_peaks = 50
    while time.monotonic() < deadline:
        g = multi_peak_subgrad(a, n_peaks)
        a = a - step * g
        a = np.maximum(a, 0)
        s_sum = a.sum()
        if s_sum < 1e-15:
            break
        a = a / s_sum
        
        s_val = safe_score(a)
        if s_val < best_score:
            best_score = s_val
            best_a = a.copy()
        
        step *= 0.9995
        if step < 1e-5:
            step = 0.003
    
    # Try a few different peak counts
    for np_ in [10, 20, 50, 100]:
        if time.monotonic() > deadline - 5:
            break
        a_try = best_a.copy()
        a_try = a_try / a_try.sum()
        step = 0.01
        sub_deadline = min(time.monotonic() + 20, deadline)
        while time.monotonic() < sub_deadline:
            g = multi_peak_subgrad(a_try, np_)
            a_try = a_try - step * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            s_try = safe_score(a_try)
            if s_try < best_score:
                best_score = s_try
                best_a = a_try.copy()
            step *= 0.999
    
    result = best_a.copy()
    print(f"\nFinal result: n={len(result)}, score={best_score:.6f}")
    return [float(x) for x in result]
