"""
Optimize step function to minimize 2*n*max(autocorrelation)/sum^2.
Uses Adam optimizer with FFT-based convolution, multiple diverse initializations,
peak-tracking subgradients, periodic restarts, and coordinate descent polish.
Key insight: non-symmetric (monotone or shaped) sequences can beat uniform (score=2)
by reducing the convolution peak relative to the sum.
"""

import numpy as np
import time
import math
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    result = ifft(fft(a, n) * fft(b, n)).real[:n_out]
    return np.maximum(result, 0.0)  # numerical safety


def _score_fast(a):
    """Fast score computation using FFT. Returns (score, max_b, argmax_b)."""
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    argmax_b = int(np.argmax(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf, max_b, argmax_b
    return 2.0 * n * max_b / (sum_a * sum_a), max_b, argmax_b


def _normalize(a, target_sum):
    """Normalize array to target sum, preserving non-negativity."""
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _adam_subgrad_step(a, momentum, velocity, t, lr, beta1, beta2, eps, top_k=1):
    """
    One Adam step using subgradient of 2n*max(b)/sum^2.
    Uses top-k peaks for subgradient to handle non-differentiable max.
    Returns (new_a, momentum, velocity).
    """
    n = len(a)
    b = _fft_conv(a, a)
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return a, momentum, velocity

    # Get top-k peak positions
    if top_k >= n:
        top_k = n
    # Use partial sort for efficiency
    if top_k <= 8:
        top_indices = np.argsort(b)[-top_k:][::-1]
    else:
        top_indices = np.argpartition(b, -top_k)[-top_k:][::-1]

    top_values = b[top_indices]
    # Weight by prominence (exponential decay)
    max_val = top_values[0] + 1e-15
    weights = np.exp(-0.5 * (max_val - top_values) / (max_val * 0.1 + 1e-15))
    weights /= np.sum(weights)

    # Compute subgradient: weighted sum of 2*a[m-i] for each peak m
    # grad[i] = (4n/sum^2) * sum_j w_j * a[m_j - i] - 4n*max_b/sum^3
    grad = np.zeros(n)
    weighted_mirror = np.zeros(n)
    for w, m in zip(weights, top_indices):
        m = int(m)
        # a[m-i] for i in range(n): this is a reversed/shifted version of a
        # a[m-i] is non-zero when 0 <= m-i < n, i.e., m-n < i <= m
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if hi >= lo:
            # For i in [lo, hi], a[m-i] = a[m-hi], a[m-hi+1], ..., a[m-lo]
            weighted_mirror[lo:hi + 1] += w * a[m - hi:m - lo + 1][::-1]

    weighted_peak = float(np.dot(weights, top_values))
    grad = 4.0 * n * weighted_mirror / (sum_a * sum_a) - 4.0 * n * weighted_peak / (sum_a ** 3)

    # Adam update
    bias1 = 1.0 - beta1 ** t
    bias2 = 1.0 - beta2 ** t
    momentum = beta1 * momentum + (1.0 - beta1) * grad
    velocity = beta2 * velocity + (1.0 - beta2) * grad * grad
    m_hat = momentum / bias1
    v_hat = velocity / bias2

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = np.minimum(a_new, 1000.0)
    # Renormalize to keep sum stable
    a_new = _normalize(a_new, sum_a)

    return a_new, momentum, velocity


def _run_adam(a, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8,
              top_k=3, eval_interval=500, restart_patience=30):
    """Run Adam optimization until deadline. Returns best sequence and score."""
    n = len(a)
    momentum = np.zeros(n)
    velocity = np.zeros(n)
    t = 0
    best_local = a.copy()
    best_score = _score_fast(a)[0]
    stuck = 0
    lr_current = lr
    lr_decay = 0.99995
    min_lr = 1e-5

    start_time = time.monotonic()
    while time.monotonic() < deadline:
        a, momentum, velocity = _adam_subgrad_step(
            a, momentum, velocity, t + 1, lr_current, beta1, beta2, eps, top_k)
        t += 1
        lr_current = max(lr_current * lr_decay, min_lr)

        if t % eval_interval == 0:
            sc, _, _ = _score_fast(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_local = a.copy()
                stuck = 0
            else:
                stuck += 1

            # Restart if stuck
            if stuck >= restart_patience:
                # Significant perturbation
                noise_scale = 0.1 * np.max(best_local) + 0.01
                a = best_local + noise_scale * np.random.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, float(np.sum(best_local)))
                momentum = np.zeros(n)
                velocity = np.zeros(n)
                t = 0
                lr_current = lr * 0.5
                stuck = 0

    return best_local, best_score


def _coordinate_descent_polish(a, deadline, n_iter=20):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed. Uses golden section search within [0, max_val].
    Cycles through coordinates multiple times.
    """
    n = len(a)
    best = a.copy()
    best_score = _score_fast(best)[0]
    target_sum = float(np.sum(best))

    for iteration in range(n_iter):
        if time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in random order
        order = np.random.permutation(n)
        for idx in order:
            if time.monotonic() > deadline - 5:
                break
            # Try perturbing a[idx] while keeping sum constant
            a_save = a[idx]
            # The score depends on a[idx] through b (convolution) and sum
            # We'll try a few candidates and pick the best
            candidates = [
                max(0.0, a_save * 0.5),
                max(0.0, a_save * 0.75),
                a_save,
                a_save * 1.25,
                a_save * 1.5,
                max(0.0, a_save - 0.1 * np.mean(a)),
                a_save + 0.1 * np.mean(a),
            ]
            best_cand = a_save
            best_cand_score = best_score
            for cand in candidates:
                cand = min(cand, 1000.0)
                if cand == a_save:
                    continue
                a[idx] = cand
                # Adjust one other element to maintain sum (simple: adjust a[0])
                diff = target_sum - float(np.sum(a))
                if diff != 0:
                    # Put the difference on a[0] if possible
                    if 0 <= a[0] + diff <= 1000:
                        a[0] += diff
                    else:
                        a[idx] = a_save
                        continue
                sc, _, _ = _score_fast(a)
                if sc < best_cand_score:
                    best_cand_score = sc
                    best_cand = cand
            a[idx] = best_cand
            if best_cand != a_save:
                improved = True
                # Renormalize
                a = _normalize(a, target_sum)
                sc, _, _ = _score_fast(a)
                if sc < best_score:
                    best_score = sc
                    best = a.copy()

    return best, best_score


def _make_power_law(n, p, c=0.0):
    """Create a power-law shaped sequence."""
    i = np.arange(n, dtype=np.float64)
    if p == 0:
        return np.ones(n) + c
    return (i + 1.0) ** p + c


def _make_exponential(n, alpha):
    """Create an exponential shaped sequence."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(alpha * i / max(n - 1, 1))


def _make_gaussian_shifted(n, center_frac, width):
    """Gaussian centered at center_frac * n with given width."""
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    return np.exp(-0.5 * ((i - center) / width) ** 2)


def _make_cosine_modulated(n, freq, amplitude=0.5):
    """Cosine-modulated sequence."""
    i = np.arange(n, dtype=np.float64)
    return 1.0 + amplitude * np.cos(2 * np.pi * freq * i / n)


def _make_ramp_mixture(n, frac=0.5, left_amp=0.3, right_amp=1.0):
    """Ramp that's low on the left, high on the right with a mix."""
    i = np.arange(n, dtype=np.float64)
    t = i / max(n - 1, 1)
    return left_amp + (right_amp - left_amp) * t


def _try_sequence(seq, deadline, budget_s, lr=0.1, top_k=3):
    """Try optimizing from a given initialization. Returns (best_seq, best_score)."""
    n = len(seq)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(seq, 0.0)
    a = _normalize(a, target_sum)
    a = np.minimum(a, 1000.0)

    initial_score = _score_fast(a)[0]

    if budget_s < 5:
        return a, initial_score

    best, best_score = _run_adam(
        a, time.monotonic() + min(budget_s * 0.7, deadline - time.monotonic()),
        lr=lr, top_k=top_k, eval_interval=300, restart_patience=25)

    # Coordinate descent polish with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_score < initial_score:
        best, best_score = _coordinate_descent_polish(best, deadline, n_iter=5)

    return best, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2*n*max(autocorrelation)/sum^2 using Adam-based subgradient descent
    with diverse shape initializations, multiple sequence lengths, peak-weighted
    subgradients, and coordinate descent polish.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Get parent sequence if available
    parent_seq = None
    parent_n = None
    try:
        if 'height_sequence_1' in globals():
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
                parent_seq = np.clip(p, 0.0, 1000.0)
                parent_n = p.size
    except Exception:
        pass

    print(f"Budget: {budget_s}s, deadline: {(total_deadline - time.monotonic()):.0f}s remaining")
    if parent_seq is not None:
        p_score = _score_fast(parent_seq)[0]
        print(f"Parent: n={parent_n}, score={p_score:.6f}")

    # Phase 1: Quick exploration of shapes and lengths
    print("\n=== Phase 1: Shape/Length Exploration ===")
    phase1_deadline = time.monotonic() + 120  # 2 minutes for exploration

    # Define candidate (n, shape_function) pairs
    n_values = [2048, 4096, 8192]
    if parent_n is not None:
        n_values.insert(0, parent_n)

    candidates = []  # (score, n, sequence_name)

    for n in n_values:
        if time.monotonic() > phase1_deadline:
            break
        shapes = {
            'power_1': _make_power_law(n, 1.0),
            'power_2': _make_power_law(n, 2.0),
            'power_0p5': _make_power_law(n, 0.5),
            'power_3': _make_power_law(n, 3.0),
            'exp_2': _make_exponential(n, 2.0),
            'exp_3': _make_exponential(n, 3.0),
            'exp_1': _make_exponential(n, 1.0),
            'exp_1p5': _make_exponential(n, 1.5),
            'exp_4': _make_exponential(n, 4.0),
            'ramp': _make_ramp_mixture(n, 0.3, 1.0),
            'ramp_steep': _make_ramp_mixture(n, 0.05, 1.0),
            'cos_1': _make_cosine_modulated(n, 1, 0.3),
            'cos_3': _make_cosine_modulated(n, 3, 0.3),
        }
        for name, seq in shapes.items():
            seq = np.maximum(seq, 0.0)
            ts = np.sqrt(2.0 * n)
            seq = _normalize(seq, ts)
            seq = np.minimum(seq, 1000.0)
            sc = _score_fast(seq)[0]
            candidates.append((sc, n, name, seq.copy()))
            if len(candidates) % 5 == 0:
                print(f"  n={n}: explored {len([c for c in candidates if c[1]==n])} shapes, best so far: {min(c[0] for c in candidates):.6f}")

    # Add parent if available
    if parent_seq is not None:
        p_norm = _normalize(parent_seq, np.sqrt(2.0 * parent_n))
        p_score = _score_fast(p_norm)[0]
        candidates.append((p_score, parent_n, 'parent', p_norm.copy()))
        print(f"  Parent score: {p_score:.6f}")

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5 initializations:")
    for i, (sc, n, name, _) in enumerate(candidates[:5]):
        print(f"  {i+1}. n={n}, shape={name}, score={sc:.6f}")

    best_overall = candidates[0][3].copy()
    best_overall_score = candidates[0][0]
    best_n = candidates[0][1]

    # Phase 2: Main optimization on top candidates
    print(f"\n=== Phase 2: Main Optimization ===")
    phase2_deadline = total_deadline - 300  # Keep 5 min for phase 3

    # Optimize from top-3 candidates (or fewer if time is tight)
    num_to_try = 3
    time_per_try = (phase2_deadline - time.monotonic()) / num_to_try

    for i in range(min(num_to_try, len(candidates))):
        if time.monotonic() > phase2_deadline:
            break
        sc_init, n, name, seq = candidates[i]
        remaining_for_this = min(time_per_try, phase2_deadline - time.monotonic())
        print(f"\n  Optimizing candidate {i+1}: n={n}, shape={name}, init_score={sc_init:.6f}, budget={remaining_for_this:.0f}s")

        if remaining_for_this < 10:
            break

        local_best, local_score = _try_sequence(
            seq, phase2_deadline, remaining_for_this,
            lr=0.1, top_k=3)

        if local_score < best_overall_score:
            # Verify with actual evaluator
            actual = float(evaluate_sequence(local_best.tolist()))
            if np.isfinite(actual) and actual < best_overall_score:
                best_overall_score = actual
                best_overall = local_best.copy()
                best_n = n
                print(f"    IMPROVED! score={actual:.6f}")
            else:
                print(f"    FFT score={local_score:.6f}, actual={actual:.6f} (no verified improvement)")
        else:
            print(f"    best={local_score:.6f} (no improvement over {best_overall_score:.6f})")

    # Phase 3: Extended refinement from the best found
    print(f"\n=== Phase 3: Extended Refinement ===")
    print(f"  Current best: {best_overall_score:.6f} (n={best_n})")

    remaining = total_deadline - time.monotonic()
    if remaining > 60:
        # Multiple refined passes with different hyperparameters
        n_best = len(best_overall)
        passes = [
            {'lr': 0.05, 'top_k': 5, 'label': 'slow_wide'},
            {'lr': 0.03, 'top_k': 2, 'label': 'slow_narrow'},
            {'lr': 0.08, 'top_k': 1, 'label': 'fast_peak'},
            {'lr': 0.02, 'top_k': 10, 'label': 'very_slow_broad'},
        ]

        time_per_pass = remaining / (len(passes) + 1)  # +1 for final CD

        for pass_info in passes:
            if time.monotonic() > total_deadline - 30:
                break
            # Perturb the best
            noise = 0.03 * np.std(best_overall)
            a_perturbed = best_overall + noise * rng.standard_normal(n_best)
            a_perturbed = np.maximum(a_perturbed, 0.0)
            a_perturbed = _normalize(a_perturbed, float(np.sum(best_overall)))
            a_perturbed = np.minimum(a_perturbed, 1000.0)

            local_best, local_score = _run_adam(
                a_perturbed, time.monotonic() + time_per_pass,
                lr=pass_info['lr'], top_k=pass_info['top_k'],
                eval_interval=200, restart_patience=30)

            if local_score < best_overall_score:
                actual = float(evaluate_sequence(local_best.tolist()))
                if np.isfinite(actual) and actual < best_overall_score:
                    best_overall_score = actual
                    best_overall = local_best.copy()
                    print(f"  Pass {pass_info['label']}: IMPROVED to {actual:.6f}")
                else:
                    print(f"  Pass {pass_info['label']}: FFT {local_score:.6f}, actual {actual:.6f}")
            else:
                print(f"  Pass {pass_info['label']}: {local_score:.6f} (no improvement)")

    # Phase 4: Coordinate descent polish
    if time.monotonic() < total_deadline - 10:
        print(f"\n=== Phase 4: Coordinate Descent Polish ===")
        cd_deadline = min(total_deadline, time.monotonic() + min(120, total_deadline - time.monotonic() - 10))
        best_overall, best_overall_score = _coordinate_descent_polish(
            best_overall, cd_deadline, n_iter=10)
        print(f"  After CD: {best_overall_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_overall.tolist()))
    print(f"\n=== FINAL ===")
    print(f"Final FFT score: {best_overall_score:.6f}")
    print(f"Final actual score: {final_score:.6f}")
    print(f"Sequence length: {len(best_overall)}")

    return [float(x) for x in best_overall]
