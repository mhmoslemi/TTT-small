LP-based iterative refinement for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses active-constraint LP to find descent directions at each step.
"""

from scipy.sparse import csr_matrix


def _full_conv(a):
    return np.convolve(a, a)


def _obj(a, n):
    c = _full_conv(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * np.max(c) / (s * s)


def _project(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s

    cols = []
    vals = []

            cols.append(j)
            vals.append(a[j])

    s = a.sum()

def _project(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)


def _build_lp_matrix(a, k_indices, n):
    """Build sparse constraint matrix for given k indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))

                  bounds=[(0, None)] * n, method='highs')

    return None

        # Add -t term
        rows.append(idx)
        cols.append(n)  # t is the (n+1)-th variable
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K_actual, n + 1))
    b_ub = np.zeros(K_actual)
    
    # Equality constraint: sum(g) = S
    A_eq = csr_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n+1))
    b_eq = np.array([S])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds = [(0, None)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]  # g, t
    return None, None

"""
LP-based iterative refinement for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses a min-max LP (fix a, minimize max of conv(a,g) subject to sum constraint)
to find descent directions. Combined with smooth gradient polishing and
hard active-set refinement phases.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _full_conv(a):
    return np.convolve(a, a)


def _project(a, target_sum):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.full(len(a), target_sum / len(a), dtype=np.float64)
    return a * (target_sum / s)


def _build_conv_rows(a, k_indices, n, t_col=None):
    """Build (rows, cols, vals) for constraint matrix rows at given k indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[j])
        if t_col is not None:
            rows.append(idx)
            cols.append(t_col)
            vals.append(-1.0)
    return rows, cols, vals


def _solve_lp_minmax(a, K, n, S):
    """LP: min t s.t. conv(a,g)[k] <= t (for top-K k), sum(g)=S, g>=0.
    Returns (g, t) or (None, None).
    """
    c_full = _full_conv(a)
    top_k = np.argsort(c_full)[-K:]
    K_act = len(top_k)

    t_col = n
    rows, cols, vals = _build_conv_rows(a, top_k, n, t_col=t_col)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(K_act, n + 1))
    b_ub = np.zeros(K_act)

    # Equality: sum(g) = S
    eq_data = np.ones(n)
    A_eq = csr_matrix((eq_data, (np.arange(n), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([S])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success and res.x is not None:
        return res.x[:n].copy(), float(res.x[n])
    return None, None


def _smooth_gd(a, n, target_sum, deadline, eta=0.05):
    """Entropy-smoothed gradient descent. Returns (best_a, best_obj)."""
    tau = n / 4.0
    best_a = a.copy()
    best_o = float(np.max(_full_conv(a)))
    stall = 0

    while time.monotonic() < deadline:
        c = _full_conv(a)
        mc = float(np.max(c))
        if mc < best_o:
            best_o = mc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        mx = float(np.max(c))
        p = np.exp((c - mx) / tau)
        ps = float(p.sum())
        if ps < 1e-300:
            p = np.ones_like(c) / len(c)
        else:
            p = p / ps

        g = 2.0 * np.correlate(p, a, mode='valid')
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        a = _project(a - eta * g / mg, target_sum)
        tau = max(tau * 0.99995, n * 0.001)

        if stall > 5000:
            eta = max(eta * 0.5, 1e-7)
            stall = 0

    return best_a, best_o


def _hard_gd(a, n, target_sum, deadline, eta=0.02):
    """Hard active-set gradient descent. Returns (best_a, best_obj)."""
    best_a = a.copy()
    best_o = float(np.max(_full_conv(a)))

    while time.monotonic() < deadline:
        c = _full_conv(a)
        k = int(np.argmax(c))
        mc = float(c[k])
        if mc < best_o:
            best_o = mc
            best_a = a.copy()

        g = np.zeros(n, dtype=np.float64)
        js = max(0, k - n + 1)
        je = min(n - 1, k) + 1
        if js < je:
            ai = k - np.arange(js, je)
            g[js:je] = 2.0 * a[ai]
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-14:
            break
        a = _project(a - eta * g / mg, target_sum)

    return best_a, best_o


def _perturb(a, n, rng, sigma, w_frac):
    """Perturb a random block multiplicatively."""
    b = a.copy()
    w = max(2, int(n * w_frac))
    s = int(rng.integers(0, max(1, n - w + 1)))
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return b
