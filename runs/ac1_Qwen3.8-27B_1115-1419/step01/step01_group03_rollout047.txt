import numpy as np
import time

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def grad_f(a, c, eps=1e-6):
    """Compute subgradient of max_k c[k] where c[k] = (a*a)[k].
    
    Uses average of gradients over active set {k : c[k] >= c[max] - eps}.
    
    grad of c[k] w.r.t. a[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    max_c = c.max()
    active = np.where(c >= max_c - eps)[0]
    
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        # grad[k] w.r.t. a[j] = 2*a[k-j] for j where 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            j_vals = np.arange(lo, hi)
            a_vals = 2.0 * a[k - j_vals]
            g[lo:hi] += a_vals
    g /= len(active)
    return g

def fista_optimize(a0, budget, rng, verbose=True):
    """Run FISTA to minimize max conv coefficient on the simplex."""
    n = len(a0)
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = 1.0  # Will be adjusted by backtracking
    # Lipschitz constant estimate: L = 2 (from analysis above)
    # So initial step = 1/L = 0.5... but let's use backtracking
    step = 0.4
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + budget
    it = 0
    stuck = 0
    eps = 1e-4  # Tolerance for active set
    
    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif sy > 0:
            y /= sy
        
        # Compute convolution at y
        c_y = np.convolve(y, y)
        
        # Gradient from active set
        g = grad_f(y, c_y, eps=eps)
        
        # Gradient step + project onto simplex
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            # Adapt step size and eps
            if stuck > 100:
                step *= 0.95
                eps *= 0.8
            if stuck > 500:
                # Restart with perturbation
                a = best_a + rng.normal(0, 0.005/np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-15:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t = 1.0
                step = 0.4
                eps = 1e-4
                stuck = 0
        
        # Gradually tighten eps (more active constraints)
        if it % 1000 == 0:
            eps = max(1e-6, eps * 0.5)
    
    # Final
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

Minimize 2n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated FISTA with
exact subgradient, multiple restarts, and basin hopping.
Fixes gradient computation errors and leverages FFT for speed.

import time

def project_simplex(v):
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v, kind='stable')[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def get_fft_size(n):
    # Need size >= 3n-2 for cross-correlation
    target = 3 * n - 2
    # Find next power of 2 (simple and fast)
    size = 1
    while size < target:
        size *= 2
    return size

def compute_subgrad(a, active_mask, n_active, fft_size):
    """Compute subgradient using FFT cross-correlation."""
    n = len(a)
    I = np.zeros(fft_size)
    I[:len(active_mask)] = active_mask.astype(np.float64)
    
    fi = np.fft.rfft(I, fft_size)
    fa = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(fi) * fa, fft_size)
    g = (2.0 / n_active) * cross[:n]
    return g

    a_prev = a0.copy()

    stall = 0

        print(f"  {label}: start score={best_score:.8f}")

"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated FISTA with
exact subgradient (fixing gradient factor-of-2 bug from prior art),
multiple diverse initializations, targeted basin hopping, and
adaptive active-set management. Uses O(n log n) per iteration via FFT.

import numpy as np
import time


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v, kind='mergesort')[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def score_normalized(a):
    """Score assuming sum(a) = 1: just 2n * max(conv)."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max()


def compute_subgrad_fft(a, c, tol, fft_size):
    """Compute exact subgradient of max_k c[k] using FFT cross-correlation.
    
    Subgradient = average over active set of grad(c[k])/da[j] = 2*a[k-j].
    Uses: g = (2/|A|) * ifft(conj(fft(I_A)) * fft(a))
    """
    n = len(a)
    max_c = c.max()
    threshold = max_c - max(tol, max_c * 1e-4)
    active = c >= threshold
    n_active = active.sum()
    if n_active == 0:
        return np.zeros(n, dtype=np.float64)
    
    # Build indicator
    I = np.zeros(fft_size, dtype=np.float64)
    I[:len(c)] = active.astype(np.float64)
    
    # Cross-correlation via FFT
    fi = np.fft.rfft(I)
    fa = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(fi) * fa, fft_size)
    g = (2.0 / float(n_active)) * cross[:n]
    return g


def compute_subgrad_direct(a, c, tol):
    """Direct subgradient (for verification or small active sets)."""
    n = len(a)
    max_c = c.max()
    threshold = max_c - max(tol, max_c * 1e-4)
    active = np.where(c >= threshold)[0]
    g = np.zeros(n, dtype=np.float64)
    for k in active:

        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * a[k - np.arange(lo, hi)]
    g /= len(active)
    return g


def fista_optimize(a0, budget, rng, verbose=True, label="", step_init=0.45, tol_init=1e-3):
    """Run FISTA to minimize max conv coefficient on probability simplex.
    
    Uses FFT-based exact subgradient, adaptive step size, and momentum reset.
    """
    n = len(a0)
    # FFT size: need >= 3n-2 for linear cross-correlation
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size *= 2
    
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = step_init
    best_a = a0.copy()
    best_score = score_normalized(a0)
    deadline = time.monotonic() + budget
    it = 0
    stall = 0
    tol = tol_init
    
    if verbose:
        print(f"    {label}: init score={best_score:.8f}, n={n}, fft_size={fft_size}")
    
    while time.monotonic() < deadline:
        # FISTA momentum update
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        mom = (t_old - 1.0) / t
        y = a + mom * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        
        # Compute convolution at y
        c_y = np.convolve(y, y)
        max_c = c_y.max()
        
        # Compute subgradient
        g = compute_subgrad_fft(y, c_y, tol, fft_size)
        
        # Gradient step + project
        a_new = project_simplex(y - step * g)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation and adaptation

            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:

            elif stall == 150:

            elif stall == 350:

                t = 1.0
                step *= 0.8
            elif stall > 600:

    if sc < best_score:
        best_score = sc
        best_a = a.copy()

def targeted_perturb(a, c, rng, n_move=50, frac=0.02):
    """Perturb by moving mass from positions contributing most to peaks
    to positions contributing least."""
    n = len(a)
    max_c = c.max()
    active = np.where(c >= max_c - max_c * 0.05)[0]
    
    # Compute contribution of each position to active peaks
    contrib = np.zeros(n)
    for k in active[:min(20, len(active))]:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            contrib[lo:hi] += a[k - np.arange(lo, hi)]
    
    # Positions with high contribution (source) and low contribution (target)
    above_med = contrib > np.median(contrib[contrib > 0]) if (contrib > 0).any() else np.ones(n, dtype=bool)
    sources = np.where((above_med) & (a > 0))[0]
    targets = np.where((~above_med) | (a == 0))[0]
    
    if len(sources) < 5 or len(targets) < 5:
        return a.copy()
    
    # Move mass
    n_s = min(n_move, len(sources))
    n_t = min(n_move, len(targets))
    src_idx = rng.choice(sources, n_s, replace=False)
    tgt_idx = rng.choice(targets, n_t, replace=False)
    
    a_new = a.copy()
    for i, j in zip(src_idx, tgt_idx):
        amt = a_new[i] * frac
        a_new[i] -= amt
        a_new[j] += amt
    
    np.maximum(a_new, 0.0, out=a_new)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new
