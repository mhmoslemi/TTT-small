"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FISTA with all-active-peaks subgradient (peaks within adaptive tolerance of max)
2. Log-sum-exp smoothed gradient descent with continuation (decreasing tau)
3. Coordinate descent polishing
Tries multiple sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.signal import fftconvolve


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * float(c.max()) / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def get_subgradient_active(a, tol_frac=0.001):
    """
    Compute subgradient of max(conv(a,a)) using all peaks within tol_frac of max.
    Returns (gradient, max_value, num_active).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = float(c.max())
    threshold = c_max * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * a[k - np.arange(lo, hi, dtype=np.int64)]
    g /= float(len(active))
    return g, c_max, len(active)


def smoothed_objective_and_grad(a, tau):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    f(a) = 2n * lse(c(a)) / (sum(a))^2
    where lse(c) = max(c) + tau * log(sum(exp((c - max(c))/tau)))
    """
    n = len(a)
    c = np.convolve(a, a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf, np.zeros(n)
    
    c_max = float(c.max())
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    lse_val = c_max + tau * log_sum
    
    # Softmax weights
    p = exp_shifted / np.sum(exp_shifted)
    
    # Gradient of lse w.r.t. a:
    # d(lse)/da_m = sum_k p_k * (dc_k/dm) = sum_k p_k * 2*a[k-m] (for valid k-m)
    # This is cross-correlation of p and a
    # g_c[m] = 2 * sum_j p[m+j] * a[j]
    # Use FFT: ifft(FFT(p) * conj(FFT(a)))
    N = 1
    while N < 2 * n:
        N *= 2
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    g_c = 2.0 * np.fft.irfft(P * np.conj(A), N)[:n]
    
    # f = 2n * lse / s^2
    # df/dm = 2n * (g_c[m] * s^2 - lse * 2*s) / s^4
    #       = 2n * (s * g_c[m] - 2*lse) / s^3
    g = 2.0 * n * (s * g_c - 2.0 * lse_val) / (s ** 3)
    
    obj = 2.0 * n * lse_val / (s * s)
    return obj, g


def run_fista_active(a0, duration, rng, verbose=True):
    """
    FISTA with subgradient from all active peaks (within tol_frac of max).
    Adaptive step size and periodic restarts.
    """
    n = len(a0)
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = 0.5
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    stale = 0
    tol_frac = 0.005
    
    if verbose:
        print(f"    FISTA start: score={best_score:.8f}, n={n}")
    
    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.05:
            y /= sy
        
        # Subgradient from active peaks
        g, c_max, n_active = get_subgradient_active(y, tol_frac)
        
        # Gradient step + project
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation
        if it % 40 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stale = 0
            else:
                stale += 1
            
            # Adapt step size
            if stale == 30:
                step *= 0.92
            elif stale == 100:
                step *= 0.8
                a_prev = best_a.copy()
                t = 1.0
            elif stale > 300:
                # Restart with perturbation
                sigma = 0.01 / np.sqrt(n)
                a = best_a + rng.normal(0.0, sigma, n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t = 1.0
                step = 0.5
                stale = 0
                
                if verbose:
                    print(f"      FISTA restart at it={it}, score={best_score:.8f}")
            
            # Gradually tighten tolerance as we converge
            if it % 2000 == 0 and tol_frac > 0.0005:
                tol_frac = max(0.0005, tol_frac * 0.5)
            
            if verbose and it % 10000 == 0:
                print(f"      FISTA it={it}, score={best_score:.8f}, step={step:.4f}, n_active={n_active}, tol={tol_frac:.5f}")
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    if verbose:
        print(f"    FISTA done: score={best_score:.8f}, iters={it}")
    
    return best_a, best_score


def run_smoothed_polish(a0, duration, rng, verbose=True):
    """
    Polish using log-sum-exp smoothed objective with continuation.
    Start with large tau, decrease geometrically.
    Use gradient descent with adaptive step size.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    
    if verbose:
        print(f"    Smoothed polish start: score={best_score:.8f}")
    
    # Determine initial tau
    c = np.convolve(a, a)
    c_spread = c.max() - c.min()
    tau = max(c_spread * 0.1, 1e-6)
    tau_min = max(c_spread * 1e-5, 1e-10)
    
    obj, g = smoothed_objective_and_grad(a, tau)
    step = 1.0  # Will be adapted
    
    it = 0
    total_tau_decreases = 0
    tau_decrease_count = 0
    
    while time.monotonic() < deadline:
        # Compute smoothed gradient
        obj, g = smoothed_objective_and_grad(a, tau)
        if not np.isfinite(obj):
            break
        
        # Gradient step with simplex projection
        a_new = a - step * g
        a_new = project_simplex(a_new)
        
        # Check improvement
        obj_new, _ = smoothed_objective_and_grad(a_new, tau)
        
        if obj_new < obj * (1.0 + 1e-12):
            a = a_new
            step = min(step * 1.05, 10.0)  # Can increase step slightly
        else:
            # Try smaller step
            step *= 0.7
            if step < 1e-10:
                # Stuck, decrease tau
                tau *= 0.5
                step = 1.0
                tau_decrease_count += 1
                total_tau_decreases += 1
        
        it += 1
        
        # Periodic true score check
        if it % 20 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            # Decrease tau periodically
            if it % 200 == 0 and tau > tau_min:
                tau *= 0.8
        
        # If tau is at minimum, stay there
        if tau <= tau_min:
            pass  # Just continue with gradient descent at minimum tau
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    if verbose:
        print(f"    Smoothed polish done: score={best_score:.8f}, iters={it}, tau_decreases={total_tau_decreases}")
    
    return best_a, best_score


def run_coord_descent(a0, duration, rng, verbose=True):
    """
    Coordinate descent polishing.
    For each coordinate, find the optimal value given others fixed.
    Cycles through coordinates multiple times.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    
    if verbose:
        print(f"    Coord descent start: score={best_score:.8f}")
    
    # Precompute: for each coordinate m, which convolution indices include it
    # c_k includes a_m when k-m is in [0, n-1], i.e., k in [m, m+n-1]
    # dc_k/dm = 2*a[k-m] for those k
    
    it = 0
    while time.monotonic() < deadline:
        # One full pass through coordinates
        for m in range(n):
            # Current contribution to convolution from a_m
            # c_k has term a_m * a[k-m] for k in [max(0,m), min(2n-2, m+n-1)]
            # And term a[k-m] * a_m (same thing, already counted)
            # Actually c_k = sum_i a_i * a[k-i], so a_m appears as:
            #   term i=m: a_m * a[k-m]
            #   term k-i=m, i=k-m: a[k-m] * a_m
            # So a_m contributes 2*a_m*a[k-m] to c_k (for valid k-m), unless k=2m where it's a_m^2.
            
            # For fixed other coordinates, c_k as function of a_m:
            # c_k = base_k + a_m * (2*a[k-m]) for k != 2m
            # c_{2m} = base_{2m} + a_m^2 + a_m * (2*a[k-m] - a_m) ... 
            # Actually let me just compute:
            # c_k = sum_i a_i * a[k-i]
            # = (sum_{i != m} a_i * a[k-i]) + a_m * a[k-m]  [if k-m valid]
            # + (if 2m == k: a_m * a_m already included above as a_m * a[m])
            # Hmm this is getting complicated. Let me just do:
            # c_k = base_k + a_m * coeff_k where coeff_k = 2*a[k-m] (for k != 2m) or a[k-m] + a_m (for k=2m)
            # Actually simpler: c_k is linear in a_m except at k=2m where it's quadratic.
            
            # Let me just try a few values of a_m and pick the best
            # Current value
            am_old = a[m]
            
            # Compute base convolution (without a_m's contribution)
            # This is expensive if done every iteration. Let me approximate:
            # Just try a few perturbed values
            candidates = [0.0, am_old * 0.5, am_old, am_old * 1.5, am_old * 2.0]
            # Add some values around the current one
            for frac in [0.8, 0.9, 1.1, 1.2]:
                candidates.append(am_old * frac)
            
            # Remove duplicates and filter
            candidates = [max(0.0, x) for x in candidates]
            
            best_val = None
            best_am = am_old
            for trial in candidates:
                a[m] = trial
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    continue
                a_norm = a / s
                sc = fast_score(a_norm)
                if best_val is None or sc < best_val:
                    best_val = sc
                    best_am = trial
            
            a[m] = best_am
            s = a.sum()
            if s > 1e-15:
                a /= s
        
        it += 1
        # Check score after full pass
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 10 == 0:
            print(f"      Coord descent pass {it}: score={best_score:.8f}")
    
    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    cands = []
    names = []
    
    # Uniform
    cands.append(np.ones(n, dtype=np.float64) / n)
    names.append("uniform")
    
    # Gaussian bump
    x = np.linspace(-3, 3, n)
    g = np.exp(-0.5 * x * x)
    g /= g.sum()
    cands.append(g.copy())
    names.append("gaussian")
    
    # Two peaks
    two = np.zeros(n)
    half = max(1, n // 10)
    two[:half] = 1.0
    two[n-half:] = 1.0
    two /= two.sum()
    cands.append(two.copy())
    names.append("two_peaks")
    
    # Raised cosine
    x = np.linspace(-1, 1, n)
    rc = 0.5 * (1 + np.cos(np.pi * x))
    rc = np.maximum(rc, 0)
    rc /= rc.sum()
    cands.append(rc.copy())
    names.append("raised_cos")
    
    # Exponential random
    er = rng.exponential(1.0, n)
    er /= er.sum()
    cands.append(er.copy())
    names.append("exp_rand")
    
    # Sparse random
    sp = rng.exponential(1.0, n)
    mask = rng.random(n) > 0.7
    sp[~mask] = 0.0
    if sp.sum() > 1e-10:
        sp /= sp.sum()
        cands.append(sp.copy())
        names.append("sparse_rand")
    
    # Laplacian (heavier tails than Gaussian)
    lap = np.exp(-np.abs(np.linspace(-4, 4, n)))
    lap /= lap.sum()
    cands.append(lap.copy())
    names.append("laplacian")
    
    # Triangular
    tri = np.zeros(n)
    mid = n // 2
    tri[:mid] = np.linspace(1, 0, mid)
    tri[mid:] = np.linspace(0, 1, n - mid)
    tri /= tri.sum()
    cands.append(tri.copy())
    names.append("triangular")
    
    return cands, names


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a different length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    return a_interp
