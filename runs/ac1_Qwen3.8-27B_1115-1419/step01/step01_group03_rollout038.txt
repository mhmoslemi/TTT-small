```python
"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using smoothed FISTA with FFT-based
gradient computation, adaptive softmax temperature, and LP-based refinement.
Hybrid approach: smoothed gradient descent + periodic LP direction finding.
"""

import numpy as np
import time
from scipy import optimize

def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho_idx = np.where(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_full(a, b=None):
    """Full convolution using FFT."""
    if b is None:
        b = a
    n_a = len(a)
    n_b = len(b)
    n_out = n_a + n_b - 1
    L = 1
    while L < n_out:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n_out]

def cross_corr(a, p):
    """
    Compute h[i] = sum_j a[j] * p[i+j] for i = 0, ..., len(a)-1.
    Uses FFT.
    """
    n = len(a)
    m = len(p)
    # h[i] = sum_j a[j] * p[i+j]
    # This is the cross-correlation. Use the identity:
    # h = conv(a[::-1], p)[n-1 : n-1+n]
    a_rev = a[::-1]
    n_out = n + m - 1
    L = 1
    while L < n_out:
        L *= 2
    fa = np.fft.rfft(a_rev, L)
    fp = np.fft.rfft(p, L)
    result = np.fft.irfft(fa * fp, L)[:n_out]
    return result[n-1 : n-1+n]

def smooth_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*c_k)) where c = conv(a,a).
    Uses FFT-based computation.
    """
    n = len(a)
    c = conv_full(a)  # length 2n-1
    # Softmax
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()
    # Gradient: g[i] = sum_j a[j] * p[i+j] + p[2*i]*a[i]
    h = cross_corr(a, p)  # h[i] = sum_j a[j] * p[i+j]
    # Diagonal correction
    diag = p[0:2*n-1:2]  # p[0], p[2], p[4], ..., p[2n-2], length n
    g = h + a * diag
    return g

def run_smoothed_fista(a0, n, duration, rng, tau_init=50.0, tau_max=5000.0):
    """
    Run FISTA on the smoothed objective with adaptive tau.
    """
    a = a0.copy()
    a_prev = a0.copy()
    t_k = 1.0
    step = None  # Will be set by line search
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    tau = tau_init
    
    # Precompute FFT size
    n_out = 3 * n - 2  # for cross-corr
    L = 1
    while L < n_out:
        L *= 2
    
    while time.monotonic() < deadline:
        # FISTA update
        t_old = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_k
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy  # Renormalize to simplex
        
        # Compute gradient at y
        g = smooth_gradient(y, tau)
        
        # Adaptive step size (initial guess)
        if step is None:
            # Estimate Lipschitz constant
            # L ~ tau * ||A^T A|| / n roughly
            step = 1.0 / (tau * y.max() * n)
            step = min(step, 0.5)
        
        # Gradient step + project
        a_new = y - step * g
        a_new = project_simplex(a_new)
        
        # Simple line search (reduce step if objective increases)
        # (Skip for speed; rely on FISTA convergence)
        
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            # Adapt tau (increase to sharpen)
            if it % 200 == 0:
                tau = min(tau * 1.05, tau_max)
            
            # Restart if stuck
            if stuck > 100:
                # Perturb and restart
                a = best_a + rng.normal(0, 0.001/np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_k = 1.0
                step = None
                stuck = 0
                tau = max(tau_init, tau * 0.5)
        
        if it % 1000 == 0:
            print(f"  SmoothedFISTA it={it}, score={best_score:.7f}, tau={tau:.1f}, step={step:.6f}")
    
    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score
```

```python
    s = a.sum()
    if s < 0.01:

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Binary search for theta
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(100):

        s = np.maximum(v - theta, 0).sum()
        if s > S:
            lo = theta
        else:
            hi = theta
    theta = 0.5 * (lo + hi)
```

```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1] + 1
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    return S * project_simplex(v / S)
```

The first term is: 2 * sum_{k: c_k > t} (c_k - t) * 2 * Q_k = 4 * sum_{k: c_k > t} (c_k - t) * Q_k

Since ||Q_k|| = 1, ||H|| <= 4 * sum_{k: c_k > t} (c_k - t) = 2 * sum active w_k (where w_k = 2*(c_k-t)*active_k).

So L = 2 * sum(w[active]). With w_k = 2*(c_k - t) for active k:
L = 2 * sum_active 2*(c_k - t) = 4 * sum_active (c_k - t)

With threshold = M * (1-frac), we have c_k - t <= M*frac for active k. And k_active is the number of active constraints.
L <= 4 * k_active * M * frac

L <= 4 * 50 * 1.5 * 0.1 = 30
