def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def normalize_sequence(a, target_sum, max_value=1000.0):
    """Clip, normalize to target sum, and keep values in [0, max_value]."""
    a = np.array(a, dtype=np.float64, copy=True)
    if a.size == 0:
        return a

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, max_value)

def fast_convolution(a):
    """FFT-based linear convolution a * a, truncated to length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def fast_score_array(a):
    """Fast FFT-based analogue of evaluate_sequence()."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")

        return float("inf")

    c = fast_convolution(a)
    if c.size == 0:
        return float("inf")

    return float(2.0 * n * float(np.max(c)) / (s * s))


def gradient_topK(a, c, K):
    """
    Subgradient of the average maximum convolution coefficient.

    For active coefficient k, d c_k / d a_i = 2*a_{k-i} when valid.
    This returns 2 * average over top K coefficients.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    K = max(1, min(int(K), L))
    d = np.zeros(n, dtype=np.float64)

    if K == 1:
        k = int(np.argmax(c))
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] = a[s_start:s_end][::-1]
        return 2.0 * d

    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]

    inv = 1.0 / float(K)
    for k in idx:
        k = int(k)
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s_start = k - (stop - 1)
        s_end = k - start + 1
        d[start:stop] += inv * a[s_start:s_end][::-1]

    return 2.0 * d


def add_start(starts, a, n):

    try:

def make_two_block(n, frac, ratio):
    """Two-block sequence with left length frac*n and level ratio A/B."""
    n = int(n)
    m = int(round(frac * n))
    if m < 1:
        m = 1
    if m >= n:
        m = n - 1

    denom = m * float(ratio) + (n - m)
    if denom <= 0.0:
        B = 1.0
    else:
        B = float(n) / denom
    A = float(ratio) * B

    a = np.empty(n, dtype=np.float64)
    a[:m] = A
    a[m:] = B
    return normalize_sequence(a, float(n))


def make_harmonic_sequence(n, rng, max_freq=6, max_amp=2.0):
    """Random low-freq cosine/sine shape, exponentiated to stay positive."""
    n = int(n)
    if n < 2:
        return normalize_sequence(np.ones(n), float(n))

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)

    for m in range(1, max_freq + 1):
        amp = rng.uniform(-max_amp, max_amp) / ((m + 1.0) ** 0.7)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)

    amp = rng.uniform(-max_amp, max_amp)
    z += amp * np.cos(np.pi * (2.0 * t - 1.0))

    z -= float(np.max(z))
    a = np.exp(z)
    return normalize_sequence(a, float(n))


def make_block_sequence(n, rng, num_blocks=8):
    """Random piecewise-constant sequence."""
    n = int(n)
    num_blocks = int(max(2, min(num_blocks, n // 2)))
    if n < 4:
        return normalize_sequence(np.ones(n), float(n))

    edges = np.round(np.linspace(0.0, n, num_blocks + 1)).astype(int)
    edges[0] = 0
    edges[-1] = n
    lengths = np.diff(edges)

    if np.any(lengths <= 0):
        edges = np.linspace(0.0, n, num_blocks + 1).astype(int)
        edges[0] = 0
        edges[-1] = n
        lengths = np.diff(edges)

    vals = rng.uniform(0.1, 3.0, num_blocks)
    if rng.random() < 0.25:
        vals[int(rng.integers(0, num_blocks))] = 0.0

    a = np.repeat(vals, lengths)
    return normalize_sequence(a, float(n))


def upsample_sequence(seq, n):
    """Linearly interpolate a sequence to length n and renormalize."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    n = int(n)

    if m == 0:
        return normalize_sequence(np.ones(n), float(n))
    if m == n:
        return normalize_sequence(seq, float(n))

    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_sequence(out, float(n))


def smooth_perturbation(a, rng, strength=0.35):
    """Multiplicative smooth perturbation used for restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n > 2:
        t = np.linspace(0.0, 1.0, n)
        z = np.zeros(n, dtype=np.float64)
        for m in range(1, 4):
            amp = strength / (0.5 + 0.5 * m)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            z += amp * np.cos(2.0 * np.pi * m * t + ph)
        a *= np.exp(z)

    if n > 4 and rng.random() < 0.35:
        width = int(rng.integers(max(1, n // 200), max(2, n // 10 + 1)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        a[start:stop] *= np.exp(rng.normal(0.0, strength, stop - start))

    return normalize_sequence(a, float(n))


def generate_starts(n, rng, previous_best, random_count=30):
    """Create a diverse pool of start sequences for a given length."""
    n = int(n)
    starts = []

    add_start(starts, np.ones(n), n)

    # Deterministic two-block starts.
    for frac in (0.10, 0.20, 0.30, 0.40, 0.45):
        for ratio in (0.5, 0.8, 1.2, 2.0, 4.0):
            add_start(starts, make_two_block(n, frac, ratio), n)

    # Simple global shapes: ramps, quadratics, half-waves.
    if n > 1:
        t = np.linspace(0.0, 1.0, n)
        for alpha in (-1.5, -1.0, -0.5, 0.5, 1.0, 1.5):
            add_start(starts, 1.0 + alpha * (2.0 * t - 1.0), n)
            add_start(starts, 1.0 + alpha * ((2.0 * t - 1.0) ** 2 - 1.0 / 3.0), n)
            add_start(starts, 1.0 + alpha * np.cos(np.pi * (2.0 * t - 1.0)), n)
            add_start(starts, 1.0 + alpha * np.sin(np.pi * t), n)

    # Random smooth/block starts.
    for _ in range(max(0, int(random_count))):
        if rng.random() < 0.5:
            nb = int(rng.integers(4, 17))
            a = make_block_sequence(n, rng, nb)
        else:
            a = make_harmonic_sequence(n, rng)
        add_start(starts, a, n)

    if previous_best is not None:
        add_start(starts, upsample_sequence(previous_best, n), n)

    try:
        parent = globals().get("height_sequence_1", None)
        if parent is not None and len(parent) > 0:
            add_start(starts, upsample_sequence(parent, n), n)
    except Exception:
        pass

    if not starts:
        starts.append((2.0, normalize_sequence(np.ones(n), float(n))))

    starts.sort(key=_pair_score)
    return starts


def local_search(a, budget_s, seed=0, label=""):
    """
    Multiplicative projected-gradient local search.

    The sequence is kept nonnegative and normalized to sum = n.
    The step direction is the top-peak subgradient, centered to preserve
    the sum constraint.  Momentum, adaptive step, noise, and occasional
    smooth restarts help escape flat/stagnant regions.
    """
    a = normalize_sequence(np.asarray(a, dtype=np.float64), float(len(a)))
    n = int(a.size)
    target = float(n)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))

    c = fast_convolution(a)
    s = float(a.sum())
    score = float(2.0 * n * float(np.max(c)) / (s * s)) if s > 0 else float("inf")

    best = a.copy()
    best_score = score

    eta = 0.25
    beta = 0.35
    mom = np.zeros(n, dtype=np.float64)
    accept_rel = 0.03
    noise = 0.02
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        if not np.all(np.isfinite(a)) or float(a.sum()) <= 0.0:
            a = best.copy()
            c = fast_convolution(a)
            score = best_score
            mom = np.zeros(n, dtype=np.float64)

        L = 2 * n - 1
        frac = max(0.002, 0.04 / (1.0 + it / 2000.0))
        K = int(L * frac)
        K = max(1, min(K, min(512, max(1, n // 8))))

        grad = gradient_topK(a, c, K)
        if not np.all(np.isfinite(grad)):
            break

        g = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(g)))
        if scale < 1e-14:
            stagnant += 1
            if stagnant > 80:
                break
            continue

        g /= scale
        mom = beta * mom + (1.0 - beta) * g

        logf = -eta * mom
        if noise > 1e-6:
            logf += rng.normal(0.0, noise, n)
        logf = np.clip(logf, -3.0, 3.0)

        cand = a * np.exp(logf)
        cand = np.maximum(cand, 0.0)
        cand = normalize_sequence(cand, target)

        if float(cand.max()) > 1000.0:
            cand = 0.5 * cand + 0.5 * best
            cand = normalize_sequence(cand, target)

        c2 = fast_convolution(cand)
        s2 = float(cand.sum())
        cand_score = float(2.0 * n * float(np.max(c2)) / (s2 * s2)) if s2 > 0 else float("inf")

        if not np.isfinite(cand_score):
            eta = max(0.01, eta * 0.7)
            mom *= 0.5
            stagnant += 1
            continue

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            stagnant = 0

        if cand_score <= score * (1.0 + accept_rel) + 1e-14:
            old = score
            a = cand
            c = c2
            score = cand_score
            mom *= 0.8

            if cand_score < old * (1.0 - 1e-10):
                eta = min(1.0, eta * 1.07)
                accept_rel = max(0.001, accept_rel * 0.998)
                noise = max(0.002, noise * 0.998)
            else:
                eta = max(0.03, eta * 0.96)

            stagnant += 1
        else:
            eta = max(0.01, eta * 0.7)
            mom *= 0.5
            stagnant += 1

        if stagnant > 800:
            a = smooth_perturbation(best, rng, 0.35)
            c = fast_convolution(a)
            s = float(a.sum())
            score = float(2.0 * n * float(np.max(c)) / (s * s)) if s > 0 else float("inf")
            if (not np.isfinite(score)) or score > best_score * 1.05:
                a = best.copy()
                c = fast_convolution(a)
                score = best_score
            eta = 0.3
            accept_rel = 0.05
            noise = 0.05
            mom = np.zeros(n, dtype=np.float64)
            stagnant = 0

    return best, best_score


def run_level(n, budget_s, previous_best, seed=0, label=""):
    """Generate starts for a length and locally refine the best ones."""
    n = int(n)
    budget_s = max(0.0, float(budget_s))
    if budget_s <= 5.0:
        seq = upsample_sequence(previous_best, n) if previous_best is not None else np.ones(n)
        seq = normalize_sequence(seq, float(n))
        return seq, fast_score_array(seq)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s

    starts = generate_starts(n, rng, previous_best, random_count=35)

    remain = max(0.0, deadline - time.monotonic())
    k = max(1, min(len(starts), int(remain / 35.0) + 1, 5))
    time_each = remain / float(k)

    best_seq = None
    best_score = float("inf")

    for i in range(k):
        if time.monotonic() >= deadline - 0.5:
            break
        sc, sq = starts[i]
        if sc < best_score:
            best_seq = sq.copy()
            best_score = sc

        time_this = max(1.0, min(time_each, deadline - time.monotonic() - 0.5))
        if time_this <= 0.0:
            break

        try:
            seq, score = local_search(sq, time_this, seed=seed * 100 + i, label=label)
            if np.isfinite(score) and score < best_score:
                best_seq = seq
                best_score = score
        except Exception:
            continue

    if best_seq is None:
        best_seq = normalize_sequence(np.ones(n), float(n))
        best_score = fast_score_array(best_seq)

    return best_seq, best_score


def add_candidate_pool(pool, seq, score):
    """Maintain a small pool of promising sequences."""
    if seq is None or score is None:
        return
    try:
        sc = float(score)
    except Exception:
        return
    if not np.isfinite(sc):
        return

    pool.append((sc, np.asarray(seq, dtype=np.float64).copy()))
    pool.sort(key=_pair_score)
    if len(pool) > 12:
        pool.pop()


def sanitize_for_return(seq):
    """Return a valid Python list of non-negative finite floats."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return [1.0]

    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)

    s = float(seq.sum())
    if s < 0.01:
        return [1.0] * 1024

    # Rescale to mean 1 if this does not violate the 1000 clipping cap.
    scale = float(seq.size) / s
    if seq.max() * scale <= 1000.0:
        seq = seq * scale

    seq = np.clip(seq, 0.0, 1000.0)
    if float(seq.sum()) < 0.01:
        return [1.0] * 1024

    return [float(x) for x in seq]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point.

    Uses a coarse-to-fine plan:
    1. Optimize at progressively larger lengths.
    2. Refine the best final-length candidate for the remaining time.
    3. Exact-evaluate a few top candidates and return the best valid one.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))

    if budget < 25.0:
        return [1.0] * 1024

    # Leave time for exact evaluation and final cleanup.
    search_deadline = t0 + max(10.0, budget - 45.0)

    if budget < 300.0:
        final_n = 1024
        small_levels = []
    elif budget < 900.0:
        final_n = 2048
        small_levels = [256, 512, 1024]
    else:
        final_n = 6733
        small_levels = [256, 512, 1024, 2048, 4096]

    rng = np.random.default_rng(seed)
    candidates = []
    global_best_seq = None
    global_best_score = float("inf")

    tiny_fallback = np.ones(1024, dtype=np.float64)
    add_candidate_pool(candidates, tiny_fallback, 2.0)

    # If budget is modest, skip small levels and go directly to final length.
    if small_levels and (search_deadline - time.monotonic()) > 180.0:
        weights = [1.0, 1.4, 2.2, 3.5, 5.5]
        small_budget = min(600.0, 0.25 * max(0.0, search_deadline - time.monotonic()))
        if small_levels[-1] < final_n:
            small_budget = min(small_budget, 0.25 * max(0.0, search_deadline - time.monotonic()))
        else:
            small_budget = 0.0

        scale = small_budget / float(sum(weights[:len(small_levels)])) if small_levels else 0.0
        prev_best = None

        for idx, level_n in enumerate(small_levels):
            b = scale * weights[idx]
            if b < 12.0 or time.monotonic() >= search_deadline - 30.0:
                break
            try:
                seq, sc = run_level(
                    level_n,
                    b,
                    prev_best,
                    seed=int(seed + 1000 + idx),
                    label="small"
                )
            except Exception:
                seq, sc = None, float("inf")

            if seq is not None and np.isfinite(sc) and sc < global_best_score:
                global_best_seq = seq
                global_best_score = sc
                add_candidate_pool(candidates, seq, sc)
                prev_best = seq

    # Final length start pool.
    try:
        final_starts = generate_starts(final_n, rng, global_best_seq, random_count=45)
    except Exception:
        final_starts = [(2.0, normalize_sequence(np.ones(final_n), float(final_n)))]

    for sc, sq in final_starts[:5]:
        if sc < global_best_score:
            global_best_seq = sq.copy()
            global_best_score = sc
            add_candidate_pool(candidates, sq, sc)

    # Quick refinement of a few final-length starts.
    remain = max(0.0, search_deadline - time.monotonic())
    quick_budget = min(420.0, 0.25 * remain)
    if quick_budget > 30.0 and final_starts:
        k = max(1, min(4, len(final_starts), int(quick_budget / 80.0) + 1))
        time_each = quick_budget / float(k)
        for i in range(k):
            if time.monotonic() >= search_deadline - 45.0:
                break
            sc, sq = final_starts[i]
            time_this = max(10.0, min(time_each, search_deadline - time.monotonic() - 45.0))
            try:
                seq, score = local_search(
                    sq,
                    time_this,
                    seed=int(seed + 7000 + i),
                    label="final_quick"
                )
                if seq is not None and score < global_best_score:
                    global_best_score = score
                    global_best_seq = seq
                    add_candidate_pool(candidates, seq, score)
            except Exception:
                continue

    # Long refinement on the current global best.
    long_budget = max(0.0, search_deadline - time.monotonic())
    if long_budget > 30.0 and global_best_seq is not None:
        try:
            seq, score = local_search(
                global_best_seq,
                long_budget,
                seed=int(seed + 12345),
                label="final_long"
            )
            if seq is not None and score < global_best_score:
                global_best_score = score
                global_best_seq = seq
                add_candidate_pool(candidates, seq, score)
        except Exception:
            pass

    # Ensure there is at least something valid.
