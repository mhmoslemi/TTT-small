        if rng.random() < 0.3:
            alpha *= rng.uniform(0.5, 2.0)
        lr = alpha * M / (gnorm * gnorm)

        # Safety: don't make any element too negative
        pos_idx = grad > 1e-15
        if np.any(pos_idx):
            safe_lr = 0.85 * np.min(a[pos_idx] / grad[pos_idx])
            lr = min(lr, max(safe_lr, 1e-20))

        # Momentum
        momentum = beta * momentum + grad / gnorm
        mnorm = np.sqrt(np.sum(momentum * momentum))
        if mnorm > 1e-15:
            momentum /= mnorm

        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = float(a_new.sum())
        if a_new_sum < 1e-15:
            a_new = a
        else:
            a_new /= a_new_sum

        a = a_new
        iter_count += 1

        # Score check (every few iterations for large n)
        check_every = max(1, n // 256)
        if iter_count % check_every == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                last_improve = time.monotonic()

    # Final score
    sc = _score(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()

    if verbose:
        print("  Subgrad: %d iterations, score=%.8f" % (iter_count, best_score))
    return best_a, best_score


def _spectral_reconstruct(target_b, a0, n, m, iters=200, verbose=False):
    """
    Iterative spectral reconstruction:
    Given target convolution shape target_b, find a >= 0 with conv(a) ~ target_b.
    Uses Gerchberg-Saxton style iteration.
    """
    a = _normalize(a0.astype(np.float64).copy())
    L = 2 * n - 1
    target = np.asarray(target_b, dtype=np.float64)
    if target.size < L:
        tmp = np.zeros(L)
        tmp[:target.size] = target
        target = tmp
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    tmax = float(target.max())
    if tmax <= 0 or not np.isfinite(tmax):
        target = np.ones(L)
        tmax = 1.0
    target = target / tmax

    T = np.fft.fft(target, m)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(iters):
        F = np.fft.fft(a, m)
        if not np.all(np.isfinite(F)):
            break
        F_abs = np.abs(F)
        F_abs_max = float(F_abs.max())
        if F_abs_max < 1e-30:
            break
        eps = 1e-10 * F_abs_max
        F_new = F / (F_abs + eps) * np.abs(T)
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        ss = float(a_new.sum())
        if ss < 1e-15:
            break
        a_new /= ss
        # Blend to avoid oscillation
        a = 0.4 * a + 0.6 * a_new

        if (it + 1) % 20 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = _score(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def _local_perturb(a, n, max_seconds, seed, verbose=False):
    """
    Local search: perturb small groups of elements and accept improvements.
    """
    rng = np.random.default_rng(seed)
    a = _normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = _score(a)
    start = time.monotonic()
    deadline = start + max_seconds
    group_size = max(1, n // 200)
    temp = 0.5 * best_sc  # "Temperature" for simulated annealing
    last_improve = start

    while time.monotonic() < deadline:
        if time.monotonic() - last_improve > min(30.0, 0.15 * max_seconds):
            break

        # Pick a random group of indices
        start_idx = rng.integers(0, max(1, n - group_size))
        idx = np.arange(start_idx, min(n, start_idx + group_size))

        # Perturb
        delta = rng.normal(0, 0.05 * np.mean(a), len(idx))
        a_new = a.copy()
        a_new[idx] = np.maximum(a_new[idx] + delta, 0.0)
        a_new = _normalize(a_new)

        sc_new = _score(a_new)
        if np.isfinite(sc_new):
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                last_improve = time.monotonic()
                a = a_new
            elif rng.random() < np.exp(-(sc_new - best_sc) / (temp + 1e-10)):
                a = a_new
        temp *= 0.999

    if verbose:
        print("  Local: score=%.8f" % best_sc)
    return best_a, best_sc


def _flat_target(n):
    """Generate a target convolution shape that's flatter than triangular."""
    L = 2 * n - 1
    t = np.linspace(0, 1, L)
    # Blend between triangular and flat
    tri = np.minimum(np.minimum(t * 2, (1 - t) * 2), 1.0)
    for blend in (0.3, 0.5, 0.7):
        target = (1 - blend) * tri + blend * np.ones(L)
        yield target


def _process_length(n, deadline, prev_best, rng, verbose=True):
    """Run all optimization methods for a given length n."""
    best_a = None
    best_score = np.inf

    if prev_best is not None:
        best_a = _interp(prev_best, n)
        best_score = _score(best_a)

    if verbose:
        print("  n=%d: interp score=%.8f" % (n, best_score))

    # Generate inits
    inits = []
    if best_a is not None:
        inits.append(best_a)
    inits.extend(_make_inits(n, rng))

    # Score and select diverse candidates
    scored = []
    seen = []
    for a in inits:
        a = _normalize(a)
        sc = _score(a)
        if not np.isfinite(sc):
            continue
        diverse = True
        for b in seen:
            d = float(np.sqrt(np.sum((a - b) ** 2)))
            if d < 0.03:
                diverse = False
                break
        if diverse:
            scored.append((sc, a))
            seen.append(a)
            if len(seen) >= 12:
                break
    scored.sort(key=lambda x: x[0])

    rem = max(0, deadline - time.monotonic())
    if rem <= 1:
        return best_a, best_score

    # Allocate time: 60% subgradient, 20% spectral, 20% local
    sg_time = rem * 0.58
    spec_time = rem * 0.18
    local_time = rem * 0.18
    buffer_time = rem * 0.06

    # Run subgradient on top candidates
    m = _next_pow2(2 * n - 1)
    n_cands = 3 if n <= 1024 else 2
    sg_per = max(5.0, sg_time / n_cands)
    for ci in range(min(n_cands, len(scored))):
        if time.monotonic() >= deadline - buffer_time:
            break
        sc0, a0 = scored[ci]
        seed = int(rng.integers(0, 2**31 - 1))
        a_opt, sc_opt = _subgradient_optimize(a0, n, sg_per, seed, verbose=False)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            if verbose:
                print("  Subgrad cand %d: %.8f" % (ci, best_score))

    # Spectral reconstruction
    if time.monotonic() < deadline - buffer_time:
        spec_rem = min(spec_time, deadline - time.monotonic() - buffer_time)
        if spec_rem > 3.0 and best_a is not None:
            spec_iters = max(50, int(spec_rem * 30 / max(1, n / 512)))
            for ti, target in enumerate(_flat_target(n)):
                if time.monotonic() >= deadline - buffer_time:
                    break
                a_spec, sc_spec = _spectral_reconstruct(target, best_a, n, m,
                                                        iters=spec_iters)
                if np.isfinite(sc_spec) and sc_spec < best_score:
                    best_score = sc_spec
                    best_a = a_spec
                    if verbose:
                        print("  Spectral target %d: %.8f" % (ti, best_score))
                # Also run subgradient from spectral result
                if time.monotonic() < deadline - buffer_time - 5:
                    sg_rem = min(30.0, deadline - time.monotonic() - buffer_time)
                    a_sg, sc_sg = _subgradient_optimize(a_spec, n, sg_rem,
                                                        int(rng.integers(0, 2**31)))
                    if np.isfinite(sc_sg) and sc_sg < best_score:
                        best_score = sc_sg
                        best_a = a_sg
                        if verbose:
                            print("  Post-spectral subgrad: %.8f" % best_score)

    # Local perturbation
    if time.monotonic() < deadline - buffer_time and best_a is not None:
        local_rem = min(local_time, deadline - time.monotonic() - buffer_time)
        if local_rem > 3.0:
            a_loc, sc_loc = _local_perturb(best_a, n, local_rem,
                                           int(rng.integers(0, 2**31)))
            if np.isfinite(sc_loc) and sc_loc < best_score:
                best_score = sc_loc
                best_a = a_loc
                if verbose:
                    print("  Local: %.8f" % best_score)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for the best non-negative sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    """
    seed = int(seed)
    budget_s = float(max(300, budget_s))
    start_total = time.monotonic()
    global_deadline = start_total + max(300, budget_s - 30)
    rng = np.random.default_rng(seed)

    # Try to get previous sequence
    prev_best = None
    try:
        cand = height_sequence_1
        if isinstance(cand, list):
            arr = np.asarray(cand, dtype=np.float64).ravel()
        else:
            arr = np.asarray(cand, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.maximum(arr, 0.0)
            if arr.sum() > 0:
                prev_best = arr / arr.sum()
    except Exception:
        prev_best = None

    if prev_best is not None:
        prev_sc = _score(prev_best)
        print("Previous sequence: n=%d, score=%.8f" % (len(prev_best), prev_sc))
    else:
        print("No previous sequence available.")

    best_a = prev_best
    best_score = _score(prev_best) if prev_best is not None else np.inf

    # Progressive lengths
    # Time allocation: weight larger n more since they have better scores
    # n=256: fast trial, n=512, n=1024, n=2048, n=4096, n=8192
    base_time = max(300, budget_s - 40)
    length_defs = [
        (256, 0.03),
        (512, 0.06),
        (1024, 0.12),
        (2048, 0.28),
        (4096, 0.32),
        (8192, 0.19),
    ]

    for n, frac in length_defs:
        now = time.monotonic()
        if now + 20 >= global_deadline:
            break
        stage_time = base_time * frac
        stage_deadline = min(global_deadline - 10, now + stage_time)
        print("\nStage n=%d, budget=%.1fs, current_best=%.8f" %
              (n, stage_time, best_score))

        # Ensure prev_best is at this length
        if prev_best is None:
            stage_prev = None
        else:
            stage_prev = _interp(prev_best, n)

        a_new, sc_new = _process_length(n, stage_deadline, stage_prev, rng,
                                         verbose=True)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new
            print("  ** New best: n=%d, score=%.8f" % (n, best_score))

        if time.monotonic() >= global_deadline:
            break

    # Final subgradient polish on the best sequence
    if best_a is not None and time.monotonic() < global_deadline - 15:
        rem = max(5.0, global_deadline - time.monotonic() - 10)
        print("\nFinal polish: remaining=%.1fs" % rem)
        a_polish, sc_polish = _subgradient_optimize(
            best_a, len(best_a), rem, int(rng.integers(0, 2**31)))
        if np.isfinite(sc_polish) and sc_polish < best_score:
            best_score = sc_polish
            best_a = a_polish
            print("  Polished: %.8f" % best_score)

    # Local perturbation polish
    if best_a is not None and time.monotonic() < global_deadline - 10:
        rem = max(3.0, global_deadline - time.monotonic() - 8)
        if rem > 5:
            a_loc, sc_loc = _local_perturb(best_a, len(best_a), rem * 0.5,
                                           int(rng.integers(0, 2**31)))
            if np.isfinite(sc_loc) and sc_loc < best_score:
                best_score = sc_loc
                best_a = a_loc
                print("  Local polish: %.8f" % best_score)

    # Final validation
    if best_a is None:
        best_a = np.ones(256, dtype=np.float64)

    best_a = _normalize(best_a)
    fft_score = _score(best_a)
    print("\nFinal FFT score: %.8f, n=%d" % (fft_score, len(best_a)))

    # Validate with evaluate_sequence if size allows
    try:
        if len(best_a) <= 8192:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Exact evaluate_sequence score: %.8f" % exact)
            if not np.isfinite(exact):
                print("WARNING: evaluation returned inf, using fallback")
                best_a = np.ones(256, dtype=np.float64)
    except Exception as e:
        print("Validation error: %s" % str(e))

    elapsed = time.monotonic() - start_total
    print("Total time: %.1fs" % elapsed)
    return [float(x) for x in best_a]
