Multi-phase approach: LSE-smoothed L-BFGS in log-space, projected subgradient
with Nesterov momentum on the simplex, and local coordinate refinement.

import numpy as np
import time
from scipy.optimize import minimize


def project_simplex(v, target_sum=1.0):
    """Project vector v onto {x: x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Scale v so that sum = target_sum approximately, then project
    # Standard algorithm: find threshold such that sum(max(v_i - threshold, 0)) = target_sum
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n+1) > (cv - target_sum))[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_conv_max(a, n_active=50):
    """Compute subgradient of max(conv(a,a)) using active set."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros(n), max_c
    # Find active set
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    # grad[i] = sum_{k in active} 2*a[k-i] / len(active) (where valid)
    # Using the formula: grad[i] = 2 * sum_j a[j] * w[i+j] / |active|
    # where w = 1/|active| * sum_{k in active} e_k
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    w_rev = w[::-1]
    conv_aw = np.convolve(a, w_rev)
    grad = 2.0 * conv_aw[n-1:2*n-1][::-1]
    return grad, max_c


def lse_objective_and_grad(x, tau):
    """LSE-smoothed max of conv(softmax(x), softmax(x)) and its gradient w.r.t. x."""
    # Softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        a = np.ones(len(x)) / len(x)
    else:
        a = a / a_sum
    
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum
    
    obj = c_max + tau * np.log(w_sum)
    
    # Gradient w.r.t. a: 2 * conv_aw[...]
    w_rev = w[::-1]
    conv_aw = np.convolve(a, w_rev)
    grad_a = 2.0 * conv_aw[len(a)-1:len(a)-1+len(a)][::-1]
    
    # Chain rule through softmax: grad_x = a * (grad_a - a @ grad_a)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

"""
Optimize step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Multi-phase approach: LSE-smoothed L-BFGS in log-space, projected subgradient
with Nesterov momentum on the simplex, spectral flattening, and local refinement.
Starts from previous best and explores structured perturbations with adaptive tau.
"""

import numpy as np
import time
from scipy.optimize import minimize


def project_simplex(v, target_sum=1.0):
    """Project vector v onto {x: x>=0, sum(x)=target_sum} using efficient algorithm."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_and_score(a):
    """Compute convolution and score."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    s = a.sum()
    if s < 1e-15:
        return c, max_c, np.inf
    score = float(2 * n * max_c / (s ** 2))
    return c, max_c, score


def subgradient_active(a, n_active=100):
    """Compute subgradient of max(conv(a,a)) using top active peaks."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros(n), max_c
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    w_rev = w[::-1]
    conv_aw = np.convolve(a, w_rev)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    return grad, max_c


def lse_obj_grad_x(x, tau):
    """LSE-smoothed objective and gradient w.r.t. log-parameterization x."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        a = np.ones(len(x)) / len(x)
    else:
        a = e / s
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    e2 = np.exp(shifted)
    w_sum = e2.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = e2 / w_sum
    obj = c_max + tau * np.log(w_sum)
    w_rev = w[::-1]
    conv_aw = np.convolve(a, w_rev)
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x


def run_lse_lbfgs(x0, n, deadline, verbose=True):
    """Run L-BFGS with geometrically decreasing tau schedule."""
    best_a = np.exp(x0 - x0.max())
    best_a = np.maximum(best_a, 0)
    best_a = best_a / best_a.sum()
    best_score = compute_conv_and_score(best_a)[2]
    if verbose:
        print(f"  LSE start: score={best_score:.6f}")

    # Geometric tau schedule
    tau_start = 0.5 / n
    tau_end = 0.001 / n
    n_taus = 8
    taus = np.geomspace(tau_start, tau_end, n_taus)

    x = x0.copy()
    for ti, tau in enumerate(taus):
        if time.monotonic() > deadline:
            break
        tau_time = min(120.0, (deadline - time.monotonic()) * 0.3)

        def obj_func(x_):
            return lse_obj_grad_x(x_, tau)[0]

        def grad_func(x_):
            return lse_obj_grad_x(x_, tau)[1]

        try:
            res = minimize(
                obj_func, x, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxfun': 2000, 'maxls': 40,
                         'iprint': -1}
            )
            x = res.x
        except Exception:
            break

        a = np.exp(x - x.max())
        a = np.maximum(a, 0)
        a = a / a.sum()
        score = compute_conv_and_score(a)[2]
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and (ti % 2 == 0 or ti == len(taus) - 1):
            print(f"  LSE tau={tau:.6f}: score={best_score:.6f}")

    return best_a, best_score


def run_momentum_subgradient(a0, deadline, n_active=100, verbose=True):
    """Projected subgradient with Nesterov momentum on the simplex."""
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)
    beta = 0.9
    best_a = a.copy()
    best_score = compute_conv_and_score(a)[2]
    T = 0
    alpha = 1.0
    patience = 0
    best_for_patience = best_score

    while time.monotonic() < deadline:
        T += 1
        # Nesterov lookahead
        y = a + beta * (a - (a - v))  # simplified: a + beta*v (since v is momentum)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        grad, max_c = subgradient_active(y, n_active)
        # Normalize gradient by the current max for better scaling
        if max_c > 1e-15:
            grad = grad / max_c

        # Adaptive step size
        alpha = 1.0 / np.sqrt(T + 1.0)

        # Update momentum
        v = beta * v + (1 - beta) * grad

        # Step and project
        a_new = project_simplex(a - alpha * v, target_sum=1.0)

        score = compute_conv_and_score(a_new)[2]
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        # Periodic report
        if T % 5000 == 0 and verbose:
            print(f"  Momentum T={T}, alpha={alpha:.6f}, score={best_score:.6f}")

        # If stuck, try a larger step or restart momentum
        if patience > 2000:
            patience = 0
            v = np.zeros(n)
            beta = min(beta * 0.95, 0.99)
            # Small perturbation
            a_pert = a + np.random.default_rng(T).normal(0, 0.01, n)
            a = project_simplex(a_pert, target_sum=1.0)

    return best_a, best_score


def run_spectral_flatten(a0, deadline, n_iter_max=500, verbose=True):
    """Spectral flattening: iteratively flatten |FFT(a)|^2 to reduce peak autocorrelation."""
    n = len(a0)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    N = N * 2  # extra zero-padding

    a = a0.copy()
    best_a = a.copy()
    best_score = compute_conv_and_score(a)[2]

    rng = np.random.default_rng(42)
    alpha_seq = np.concatenate([
        np.full(100, 0.3),
        np.full(200, 0.5),
        np.full(200, 0.7)
    ])

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        if it >= len(alpha_seq):
            alpha_seq_next = 0.7 + 0.3 * (1 - 1.0 / (it + 1))
            alpha = min(alpha_seq_next, 0.99)
        else:
            alpha = alpha_seq[it]

        # Zero-pad and FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # Flatten magnitude
        mag = np.abs(A)
        mag_mean = mag.mean()
        mag_target = alpha * mag + (1 - alpha) * mag_mean

        # Preserve phase, update magnitude
        phase = np.angle(A)
        A_new = mag_target * np.exp(1j * phase)

        # Inverse FFT (real)
        a_new = np.fft.irfft(A_new, N)
        a_new = a_new[:n]

        # Project to simplex
        a_new = np.maximum(a_new, 0)
        if a_new.sum() > 1e-15:
            a_new = a_new / a_new.sum()
        else:
            a_new = np.ones(n) / n

        score = compute_conv_and_score(a_new)[2]
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 100 == 0:
            print(f"  Spectral it={it}, alpha={alpha:.2f}, score={best_score:.6f}")

    return best_a, best_score


def run_local_refinement(a0, deadline, verbose=True):
    """Local coordinate refinement: try small perturbations of individual coordinates."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_conv_and_score(a)[2]
    rng = np.random.default_rng(777)

    # Identify active peaks for gradient guidance
    c, max_c, _ = compute_conv_and_score(a)
    grad, _ = subgradient_active(a, n_active=50)

    # Try random coordinate moves
    n_trials = 0
    while time.monotonic() < deadline:
        # Get a few random coordinates to try
        coords = rng.choice(n, size=min(50, n), replace=False)
        for i in coords:
            if time.monotonic() > deadline:
                break
            n_trials += 1

            # Try moving mass from i to a random other coordinate
            j = rng.integers(n)
            if i == j:
                continue
            # Try different transfer amounts
            for frac in [0.01, 0.05, 0.1, 0.2]:
                a_trial = a.copy()
                transfer = a[i] * frac
                a_trial[i] -= transfer
                a_trial[j] += transfer
                if a_trial[i] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = compute_conv_and_score(a_trial)[2]
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    # Recompute gradient
                    grad, _ = subgradient_active(a, n_active=50)
                    break

        # Also try gradient-guided moves
        if n_trials % 100 == 0:
            # Find coordinate to decrease most (highest gradient)
            i_dec = np.argmax(grad)
            # Find coordinate to increase most (lowest gradient)
            i_inc = np.argmin(grad)
            if i_dec != i_inc:
                for frac in [0.01, 0.05, 0.1]:
                    a_trial = a.copy()
                    transfer = a[i_dec] * frac
                    a_trial[i_dec] -= transfer
                    a_trial[i_inc] += transfer
                    if a_trial[i_dec] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = compute_conv_and_score(a_trial)[2]
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        grad, _ = subgradient_active(a, n_active=50)
                        break

        if verbose and n_trials % 5000 == 0:
            print(f"  Local refinement trials={n_trials}, score={best_score:.6f}")

    return best_a, best_score


def make_diverse_inits(a_base, n, rng):
    """Create diverse initializations based on a base sequence."""
    inits = [a_base.copy()]

    # Cosine modulation
    for freq in [2, 3, 5, 7]:
        x = a_base.copy()
        for i in range(n):
            x[i] = a_base[i] * (1.0 + 0.3 * np.cos(2.0 * np.pi * freq * i / n))
        x = np.maximum(x, 1e-10)
        inits.append(x / x.sum())

    # Linear ramp perturbation
    ramp = np.linspace(0.5, 1.5, n)
    x = a_base * ramp
    x = np.maximum(x, 1e-10)
    inits.append(x / x.sum())

    # Reverse ramp
    ramp = np.linspace(1.5, 0.5, n)
    x = a_base * ramp
    x = np.maximum(x, 1e-10)
    inits.append(x / x.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Block structure
    n_blocks = 4
    block_size = n // n_blocks
    x = np.ones(n)
    for b in range(n_blocks):
        x[b * block_size:(b + 1) * block_size] = 0.5 + rng.random()
    x = np.maximum(x, 0.01)
    inits.append(x / x.sum())

    # Gaussian bump
    center = n // 2
    sigma = n / 6
    x = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    x = np.maximum(x, 0.01)
    inits.append(x / x.sum())

    return inits
