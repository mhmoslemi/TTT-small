def _next_pow2(x):
    x = int(max(1, x))

    while p < x:

def _conv_self_fft(a, m):
    n = len(a)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    return np.maximum(b, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    m = _next_pow2(2 * n - 1)
    b = _conv_self_fft(a, m)
    bmax = float(b.max())
    if bmax < 1e-300 or not np.isfinite(bmax):
        return np.inf
    return float(2.0 * n * bmax / (s * s))


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def _interp_to_length(a, n):
    a = np.asarray(a, dtype=np.float64)
    if len(a) == n:
        return a.copy()
    if len(a) == 0:
        return np.ones(n, dtype=np.float64) / n
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a)


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    v = v - float(v.max())
    v = np.clip(v, -700.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / len(v), dtype=np.float64)
    return e / s


def _log_safe(a):
    a = np.asarray(a, dtype=np.float64)
    return np.log(np.maximum(a, 1e-300))


def _lse_score_and_grad(v, tau, m_conv, m_xcorr, top_k_boost=0.0, top_k_frac=0.0):
    """
    Compute LSE surrogate objective and gradient w.r.t. v (log-space params).
    a = softmax(v), b = conv(a,a), LSE = tau*log(sum(exp(b/tau)))
    """
    n = len(v)
    a = _softmax(v)
    F = np.fft.fft(a, m_conv)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if bmax < 1e-300:
        return np.inf, np.zeros(n), a, b

    # LSE computation
    b_scaled = b / tau
    b_shifted = b_scaled - float(b_scaled.max())
    e = np.exp(b_shifted)
    es = float(e.sum())
    if not np.isfinite(es) or es < 1e-300:
        return np.inf, np.zeros(n), a, b
    w = e / es

    # Top-k boosting
    if top_k_boost > 0.0 and top_k_frac > 0.0:
        k = max(1, int(top_k_frac * len(b)))
        if k < len(b):
            idx = np.argpartition(b, -k)[-k:]
            w[idx] += top_k_boost
            w = w / float(w.sum())

    # Gradient: dLSE/da[i] = 2 * (w cross-corr a)[i]
    Fw = np.fft.fft(w, m_xcorr)
    Fa = np.fft.fft(a, m_xcorr)
    grad_a = 2.0 * np.fft.ifft(Fw * np.conj(Fa)).real[:n]

    # Gradient w.r.t. v: a[i] * (grad_a[i] - dot(a, grad_a))
    mean_ga = float(np.dot(a, grad_a))
    grad_v = a * (grad_a - mean_ga)

    obj = float(tau * (float(b_scaled.max()) + np.log(es)))
    return obj, grad_v, a, b


def _make_initializations(n, rng, prev_a):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    # Previous best (interpolated)
    if prev_a is not None and len(prev_a) > 0:
        inits.append(_normalize(_interp_to_length(prev_a, n)))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)

    # Exponential decays (various rates, both directions)
    for c in (0.5, 1.0, 2.0, 4.0, 8.0):
        inits.append(_normalize(np.exp(-c * t)))
        inits.append(_normalize(np.exp(-c * (1.0 - t))))

    # Symmetric bell shapes
    for c in (1.0, 2.0, 4.0):
        inits.append(_normalize(np.exp(-c * (t - 0.5) ** 2)))

    # Power-law edges
    for p in (0.5, 1.0, 2.0):
        inits.append(_normalize((t + 1e-6) ** p))
        inits.append(_normalize(((1.0 - t) + 1e-6) ** p))

    # Cosine bumps
    inits.append(_normalize(0.5 + 0.5 * np.cos(2.0 * np.pi * t)))
    inits.append(_normalize(0.3 + 0.7 * np.cos(2.0 * np.pi * (t - 0.3)) ** 2))

    # Asymmetric biases (key for beating symmetric solutions)
    inits.append(_normalize(np.exp(-1.5 * (t - 0.4) ** 2)))
    inits.append(_normalize(np.exp(-1.5 * (t - 0.6) ** 2)))
    inits.append(_normalize(1.0 + 0.5 * np.sin(2.0 * np.pi * t)))

    # Fourier-based (random phases)
    kc = min(20, n)
    a_f = 1.0
    for k in range(1, kc + 1):
        amp = 0.3 / (float(k) ** 0.7)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a_f += amp * np.cos(2.0 * np.pi * k * t + ph)
    inits.append(_normalize(a_f))

    a_f2 = 1.0
    for k in range(1, kc + 1):
        amp = 0.25 / (float(k) ** 0.8)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a_f2 += amp * np.sin(2.0 * np.pi * k * t + ph)
    inits.append(_normalize(a_f2))

    return inits


def _optimize_lse(a_init, n, budget_s, seed, verbose=True):
    """
    Main LSE-based optimizer. Returns (best_a, best_score).
    """
    rng = np.random.default_rng(seed)
    a_init = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a_init)

    # FFT sizes
    m_conv = _next_pow2(2 * n - 1)
    m_xcorr = _next_pow2(3 * n - 2)

    # Initialize in log-space
    v = _log_safe(a_init).copy()
    v = v - float(v.mean())
    v = np.clip(v, -30.0, 30.0)

    # Get initial score
    a0 = _softmax(v)
    best_a = a0.copy()
    best_score = _score(a0)
    if not np.isfinite(best_score):
        best_a = a0.copy()
        best_score = 1e10

    # Adam state
    beta1 = 0.85
    beta2 = 0.999
    eps_adam = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t_step = 0

    start = time.monotonic()
    deadline = start + budget_s
    last_improve = start
    check_every = max(1, n // 1024)

    # Temperature schedule parameters
    tau_init_frac = 0.4  # initial tau as fraction of bmax
    tau_min_frac = 0.005  # minimum tau as fraction of bmax
    # We'll use a dynamic schedule: tau = bmax * frac, where frac decreases

    # Top-k parameters
    top_k_boost = 0.5
    top_k_frac = 0.05

    iteration = 0
    while time.monotonic() < deadline:
        # Compute score and gradient
        # First get current bmax for adaptive tau
        a = _softmax(v)
        F = np.fft.fft(a, m_conv)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax < 1e-300:
            break

        # Adaptive temperature
        elapsed_frac = (time.monotonic() - start) / max(1e-9, budget_s)
        # Geometric decay of tau fraction
        tau_frac = tau_init_frac * (tau_min_frac / tau_init_fract) ** elapsed_frac if False else \
                   tau_init_frac * (tau_min_frac / tau_init_frac) ** elapsed_frac
        tau_frac = max(tau_frac, tau_min_frac)
        tau = max(1e-15, bmax * tau_frac)

        # Top-k boost: decrease as we converge
        boost = top_k_boost * (1.0 - 0.5 * elapsed_frac)
        tkf = top_k_frac

        # Compute gradient
        obj, grad_v, a, b = _lse_score_and_grad(v, tau, m_conv, m_xcorr, boost, tkf)
        if not np.isfinite(obj):
            break

        # Normalize gradient
        gnorm = float(np.linalg.norm(grad_v))
        if gnorm < 1e-14:
            # Stuck: add noise
            v += rng.normal(0.0, 0.1, n)
            v = v - float(v.mean())
            v = np.clip(v, -30.0, 30.0)
            last_improve = time.monotonic()
            continue

        grad_v = grad_v / (gnorm + 1e-30)

        # Adam update
        t_step += 1
        mu = beta1 * mu + (1.0 - beta1) * grad_v
        nu = beta2 * nu + (1.0 - beta2) * (grad_v * grad_v)
        mu_hat = mu / (1.0 - beta1 ** t_step)
        nu_hat = nu / (1.0 - beta2 ** t_step)

        # Learning rate schedule: start high, decay
        lr = 0.15 * (1.0 - 0.7 * elapsed_frac) ** 1.5
        step = lr * mu_hat / (np.sqrt(nu_hat) + eps_adam)
        step = np.clip(step, -0.3, 0.3)

        v = v - step
        v = v - float(v.mean())
        v = np.clip(v, -30.0, 30.0)

        # Periodic score check
        if iteration % check_every == 0:
            a_check = _softmax(v)
            sc = _score(a_check)
            if np.isfinite(sc) and sc < best_score - 1e-9:
                best_score = sc
                best_a = a_check.copy()
                last_improve = time.monotonic()
                if verbose:
                    pass  # too verbose

        # Stuck detection
        if time.monotonic() - last_improve > min(60.0, 0.2 * budget_s):
            # Add perturbation
            sigma = 0.05 + 0.2 * (1.0 - elapsed_frac)
            v += rng.normal(0.0, sigma, n)
            v = v - float(v.mean())
            v = np.clip(v, -30.0, 30.0)
            last_improve = time.monotonic()

        iteration += 1

    # Final check
    a_final = _softmax(v)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return _normalize(best_a), float(best_score)


def _spectral_init(n, target_b, a_start):
    """
    Spectral method: given target convolution b_target, find a whose
    self-convolution approximates it. Uses iterative FFT refinement.
    """
    n = int(n)
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1

    target_b = np.asarray(target_b, dtype=np.float64)
    if len(target_b) < L:
        padded = np.zeros(L)
        padded[:len(target_b)] = target_b
        target_b = padded
    else:
        target_b = target_b[:L].copy()
    target_b = np.maximum(target_b, 0.0)
    tm = float(target_b.max())
    if tm > 0:
        target_b /= tm
    else:
        target_b = np.ones(L) / L

    T = np.fft.fft(target_b, m)

    a = np.asarray(a_start, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    am = float(a.max())
    if am > 0:
        a /= am

    # Initial spectral guess
    sqrt_T = np.fft.ifft(np.sqrt(np.maximum(T, 0))).real[:n]
    sp = np.maximum(sqrt_T, 0.0)
    sm = float(sp.max())
    if sm > 0 and np.isfinite(sm):
        sp /= sm
    a = 0.5 * (a / float(a.max())) + 0.5 * sp
    a = np.maximum(a, 1e-12)

    # Iterate: a <- blend(a, max(ifft((F+T/F)/2), 0))
    for it in range(50):
        F = np.fft.fft(a, m)
        eps = 1e-10 * max(1.0, float(np.max(np.abs(F))))
        F_new = 0.5 * (F + T / (F + eps))
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        blend = max(0.3, 0.7 * (1.0 - it / 50.0))
        a = (1.0 - blend) * a + blend * a_new
        a = np.maximum(a, 1e-12)

    return _normalize(a)


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search entry point. Returns best sequence found within budget.
    """
    start = time.monotonic()
    deadline = start + max(60.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(42)

    # Load previous best
    prev_a = None
    try:
        cand = height_sequence_1
        if isinstance(cand, list):
            arr = np.asarray(cand, dtype=np.float64)
        else:
            arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0 and np.isfinite(arr).all():
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 0.0:
                prev_a = arr / float(arr.sum())
    except Exception:
        prev_a = None

    if prev_a is not None:
        print("Loaded previous sequence, length=%d, score=%.6f" % (len(prev_a), _score(prev_a)))
    else:
        print("No previous sequence, starting from scratch")

    best_a = None
    best_score = np.inf

    if prev_a is not None:
        best_a = prev_a.copy()
        best_score = _score(prev_a)

    # Stage 1: Optimize at the previous length (or 2048)
    n_prev = len(prev_a) if prev_a is not None else 2048
    n_prev = max(512, min(n_prev, 4096))

    # Determine lengths to try
    lengths_to_try = []
    if n_prev >= 1024:
        lengths_to_try.append(n_prev)
    lengths_to_try.extend([2048, 4096, 8192])
    # Remove duplicates, keep unique
    seen_len = set()
    unique_lengths = []
    for nl in lengths_to_try:
        if nl not in seen_len and 512 <= nl <= 16384:
            seen_len.add(nl)
            unique_lengths.append(nl)

    # Time allocation: give more time to medium lengths
    time_frac = {}
    total_frac = 0.0
    for nl in unique_lengths:
        if nl <= 2048:
            time_frac[nl] = 0.25
        elif nl <= 4096:
            time_frac[nl] = 0.45
        else:
            time_frac[nl] = 0.30
        total_frac += time_frac[nl]
    for nl in unique_lengths:
        time_frac[nl] /= total_frac

    for n in unique_lengths:
        elapsed = time.monotonic() - start
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        alloc = min(remaining * 0.95, (budget_s - 25.0) * time_frac.get(n, 0.2))
        alloc = max(10.0, alloc)

        print("\n=== Stage n=%d, budget=%.0fs, current_best=%.6f ===" % (n, alloc, best_score))
        stage_start = time.monotonic()
        stage_deadline = stage_start + alloc

        # Generate initializations
        inits = _make_initializations(n, rng, best_a)

        # Also try spectral initialization with flat target
        flat_target = np.ones(2 * n - 1)
        # Slightly tapered target (better for non-negative sequences)
        tL = np.linspace(0.0, 1.0, 2 * n - 1)
        edge_factor = 1.0 - 0.3 * (2.0 * tL - 1.0) ** 2
        tapered_target = np.maximum(edge_factor, 0.1)
        inits.append(_spectral_init(n, flat_target, inits[0]))
        inits.append(_spectral_init(n, tapered_target, inits[0]))

        # Score all inits and pick best ones
        scored = []
        seen_vecs = []
        for a in inits:
            a = _normalize(a)
            sc = _score(a)
            if not np.isfinite(sc):
                continue
            # Diversity check
            diverse = True
            for bv in seen_vecs:
                d = float(np.linalg.norm(a - bv))
                if d < 0.03:
                    diverse = False
                    break
            if diverse:
                scored.append((sc, a))
                seen_vecs.append(a.copy())
                if len(seen_vecs) >= 10:
                    break

        scored.sort(key=lambda x: x[0])
        print("  Top init scores: " + ", ".join("%.6f" % s for s, _ in scored[:5]))

        # Optimize from top candidates
        n_cands = min(4, len(scored))
        per_cand_time = alloc / max(1, n_cands)
        per_cand_time = max(5.0, per_cand_time - 2.0)

        for ci in range(n_cands):
            if time.monotonic() >= stage_deadline:
                break
            remaining_c = stage_deadline - time.monotonic()
            if remaining_c < 5.0:
                break
            this_budget = min(per_cand_time, remaining_c - 1.0)
            if this_budget < 3.0:
                break

            sc_init, a_init = scored[ci]
            seed = int(rng.integers(0, 2**31 - 1))
            a_opt, sc_opt = _optimize_lse(a_init, n, this_budget, seed)

            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print("  Stage n=%d cand %d: improved to %.6f" % (n, ci, best_score))
            else:
                print("  Stage n=%d cand %d: score=%.6f (not improved)" % (n, ci, sc_opt))

        # If time remains, do another round from best found
        remaining = stage_deadline - time.monotonic()
        if remaining > 15.0:
            seed = int(rng.integers(0, 2**31 - 1))
            a_opt, sc_opt = _optimize_lse(best_a, n, remaining - 2.0, seed)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print("  Stage n=%d polish: improved to %.6f" % (n, best_score))

    # Final: validate with exact evaluation
    if best_a is None:
        best_a = np.ones(2048, dtype=np.float64) / 2048.0

    best_a = _normalize(best_a)
    final_fft_score = _score(best_a)
    print("\nFinal FFT score: %.6f, length: %d" % (final_fft_score, len(best_a)))

    # Validate with exact evaluator if feasible
    try:
        if len(best_a) <= 10000:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Exact score: %.6f" % exact)
            if not np.isfinite(exact):
                best_a = np.ones(2048, dtype=np.float64) / 2048.0
                exact = float(evaluate_sequence([float(x) for x in best_a]))
                print("Fallback exact score: %.6f" % exact)
    except Exception as e:
        print("Validation error: %s" % str(e))

    return [float(x) for x in best_a]
