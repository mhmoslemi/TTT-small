def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    ffc_size = 1
    while ffc_size < size:
        ffc_size *= 2
    fa = np.fft.rfft(a, ffc_size)
    c = np.fft.irfft(fa * fa, ffc_size)[:size]
    return c

def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def grad_topk(a, top_k=5):
    """Compute gradient of max(conv) using top-k peaks."""
    n = len(a)
    c = fast_convolve(a)
    if top_k >= c.size:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            idx = k - np.arange(lo, hi)
            g[lo:hi] += a[idx]
        if k % 2 == 0:
            j = k // 2
            if 0 <= j < n:
                g[j] += a[j]
    g /= len(top_idx)
    return g, c

def spectral_init(n, rng):
    """Initialize using spectral method: random phase, square magnitude, IFFT, clip."""
    size = 2 * n
    # Random spectrum with random phase
    mag = rng.exponential(1.0, size // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    spectrum = mag * np.exp(1j * phase)
    spectrum[0] = mag[0] + 0j  # Keep DC real
    # Make conjugate symmetric
    full_spectrum = np.zeros(size, dtype=complex)
    full_spectrum[:size//2+1] = spectrum
    for i in range(1, size//2+1):
        if i < size - i:
            full_spectrum[size-i] = np.conj(spectrum[i])
    a = np.fft.irfft(full_spectrum, size)
    a = a[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

    c = fast_convolve(a)
    top_idx = np.argpartition(c, -top_k)[-top_k:]

        a_try[j] += delta
