```python
"""
Search for non-negative step-function heights a_0,...,a_{n-1} minimizing

    2*n*max_k conv(a,a)_k / (sum a)^2.

The scale is fixed during optimization by using the simplex sum(a)=1.  The
core is a non-convex continuous optimization of the maximum coefficient of the
full convolution a*a, using:

  * stable log-sum-exp smoothing of the maximum,
  * exact FFT-based first derivatives in both self-optimization and an
    alternating convex subproblem that optimizes g for fixed f in max(f*g),
  * L-BFGS with wall-clock interruption,
  * upsampled continuation from the previous best sequence,
  * spectral square-root initializations and low-pass random restarts,
  * mixture blends of the old sequence and the alternating g-sequence.

The program returns the best finite sequence found before a safety margin in
budget_s seconds.
"""

import numpy as np
import time

try:
    from scipy import optimize as _sp_opt
except Exception:
    _sp_opt = None

_TIME_LIMIT = [None]
_LAST_X = [None]


class _TimeLimit(Exception):
    pass


def _timeout_callback(xk):
    _LAST_X[0] = np.array(xk, dtype=np.float64, copy=True)
    dl = _TIME_LIMIT[0]
    if dl is not None and time.monotonic() >= dl:
        raise _TimeLimit()
    return 0


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score_fft(a):
    a = _proj(a)
    n = int(a.size)
    if n == 0:
        return np.inf, None
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:L]
    b = np.maximum(b, 0.0)
    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf, b
    return float(2.0 * n * bmax), b


def _softmax(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.ones(0, dtype=np.float64)
    vm = float(np.max(v))
    if not np.isfinite(vm):
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    e = np.exp(np.clip(v - vm, -745.0, 0.0))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return e / s


def _log_init(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-300)
    v = np.log(a)
    v = v - float(np.mean(v))
    return np.clip(v, -25.0, 25.0)


def _smooth_max(b, tau):
    b = np.asarray(b, dtype=np.float64)
    L = int(b.size)
    bmax = float(np.max(b))
    if not np.isfinite(bmax):
        return np.inf, np.zeros(L, dtype=np.float64)
    if tau <= 0.0:
        tol = 1e-14 * max(1.0, abs(bmax))
        active = b >= (bmax - tol)
        cnt = float(np.count_nonzero(active))
        if cnt <= 0.0:
            active = (b == bmax)
            cnt = float(np.count_nonzero(active))
        w = np.zeros(L, dtype=np.float64)
        if cnt > 0.0:
            w[active] = 1.0 / cnt
        else:
            w.fill(1.0 / L)
        return bmax, w
    u = (b - bmax) / tau
    u = np.clip(u, -700.0, 0.0)
    e = np.exp(u)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        w = np.full(L, 1.0 / L, dtype=np.float64)
        return bmax, w
    val = bmax + tau * float(np.log(s))
    w = e / s
    return float(val), w


def _self_smooth_obj(x, n, m, tau, scale):
    v = np.asarray(x, dtype=np.float64)
    if v.size != n or not np.all(np.isfinite(v)):
        return np.inf, np.zeros(n, dtype=np.float64)
    a = _softmax(v)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    val, w = _smooth_max(b, tau)
    if not np.isfinite(val):
        return np.inf, np.zeros(n, dtype=np.float64)
    P = np.fft.fft(w, m)
    c = np.fft.ifft(P * np.conj(F)).real[:n]
    grad_a = 2.0 * c
    gm = float(np.dot(a, grad_a))
    grad = a * (grad_a - gm)
    grad = grad - float(np.mean(grad))
    if not np.all(np.isfinite(grad)):
        return np.inf, np.zeros(n, dtype=np.float64)
    return float(scale * val), grad * float(scale)


def _lin_smooth_obj(x, F_f, n, m, tau, scale):
    v = np.asarray(x, dtype=np.float64)
    if v.size != n or not np.all(np.isfinite(v)):
        return np.inf, np.zeros(n, dtype=np.float64)
    g = _softmax(v)
    Fg = np.fft.fft(g, m)
    b = np.fft.ifft(F_f * Fg).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    val, w = _smooth_max(b, tau)
    if not np.isfinite(val):
        return np.inf, np.zeros(n, dtype=np.float64)
    P = np.fft.fft(w, m)
    c = np.fft.ifft(P * np.conj(F_f)).real[:n]
    gm = float(np.dot(g, c))
    grad = g * (c - gm)
    grad = grad - float(np.mean(grad))
    if not np.all(np.isfinite(grad)):
        return np.inf, np.zeros(n, dtype=np.float64)
    return float(scale * val), grad * float(scale)


def _self_lbgfs_phase(v, n, m, tau, deadline):
    v = np.asarray(v, dtype=np.float64).copy()
    if _sp_opt is None:
        return v
    _TIME_LIMIT[0] = deadline
    _LAST_X[0] = v.copy()
    args = (int(n), int(m), float(tau), float(2.0 * n))
    opts = {
        "maxiter": 20000,
        "ftol": 1e-18,
        "gtol": 1e-12,
        "maxls": 20,
        "disp": False,
    }
    try:
        res = _sp_opt.optimize.minimize(
            _self_smooth_obj,
            v,
            args=args,
            jac=True,
            method="L-BFGS-B",
            callback=_timeout_callback,
            options=opts,
        )
        newv = res.x
    except _TimeLimit:
        newv = _LAST_X[0]
    except Exception as e:
        print("    self L-BFGS phase failed:", e)
        newv = v
    finally:
        _TIME_LIMIT[0] = None

    if newv is None:
        newv = v
    newv = np.asarray(newv, dtype=np.float64)
    if newv.size != n or not np.all(np.isfinite(newv)):
        newv = v
    newv = newv - float(np.mean(newv))
    return np.clip(newv, -30.0, 30.0)

def _simple_adam(a0, budget_s):
    a = _proj(a0)
    n = int(a.size)
    if n == 0 or budget_s <= 0.0:
        return a, _score_fft(a)[0]
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    v = _log_init(a)
    best_a = a.copy()
    best_sc, _ = _score_fft(a)
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-9
    rng = np.random.default_rng(987654321)
    start = time.monotonic()
    end = start + max(1.0, float(budget_s))
    last_improve = start
    check_interval = max(4, min(40, 400 // max(1, n)))

    while time.monotonic() < end:
        progress = min(1.0, max(0.0, (time.monotonic() - start) / max(1e-9, end - start)))
        if time.monotonic() - last_improve > min(60.0, 0.35 * max(1.0, budget_s)):
            sig = 0.08 + 0.25 * (1.0 - progress)
            v += rng.normal(0.0, sig, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            last_improve = time.monotonic()
            continue

        a = _softmax(v)
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:L]
        b = np.maximum(b, 0.0)
        bmax = float(np.max(b))
        if not np.isfinite(bmax) or bmax <= 0.0:
            break

        tau = max(1e-14, bmax * (0.09 * (1.0 - progress) + 0.005))
        u = (b - bmax) / tau
        u = np.clip(u, -700.0, 0.0)
        w = np.exp(u)
        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= 0.0:
            break
        w /= ws

        top_k = max(1, int(0.04 * L))
        if top_k < L:
            idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            idx = np.arange(L)
        w[idx] += 0.4
        w /= float(np.sum(w))

        P = np.fft.fft(w, m)
        c = np.fft.ifft(P * np.conj(F)).real[:n]
        grad_a = 2.0 * c
        gm = float(np.dot(a, grad_a))
        grad = a * (grad_a - gm)
        grad -= float(np.mean(grad))
        grms = float(np.sqrt(np.mean(grad * grad)))
        if not np.isfinite(grms) or grms < 1e-16:
            sig = 0.05 + 0.15 * (1.0 - progress)
            v += rng.normal(0.0, sig, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -25.0, 25.0)
            continue

        grad /= grms
        t += 1
        mu = beta1 * mu + (1.0 - beta1) * grad
        nu = beta2 * nu + (1.0 - beta2) * (grad * grad)
        b1 = beta1 ** t
        b2 = beta2 ** t
        mhat = mu / max(1e-12, 1.0 - b1)
        vhat = nu / max(1e-12, 1.0 - b2)
        lr = 0.22 * (1.0 - 0.60 * progress)
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -1.0, 1.0)
        v = v - lr * step
        v = v - float(np.mean(v))
        v = np.clip(v, -25.0, 25.0)

        if t % check_interval == 0:
            an = _softmax(v)
            sc, _ = _score_fft(an)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_a = an.copy()
                last_improve = time.monotonic()

    return best_a, float(best_sc)


def _optimize_self(a0, budget_s):
    a = _proj(a0)
    n = int(a.size)
    sc0, _ = _score_fft(a)
    if not np.isfinite(sc0):
        return a, sc0
    if budget_s <= 2.0 or n == 0:
        return a, sc0

    if _sp_opt is None:
        return _simple_adam(a, budget_s)

    m = _next_pow2(2 * n - 1)
    v = _log_init(a)
    best_a = a.copy()
    best_sc = sc0
    best_v = v.copy()

    start = time.monotonic()
    end = start + max(2.0, float(budget_s))

    if budget_s <= 25.0:
        nph = 2
    elif budget_s <= 100.0:
        nph = 4
    elif budget_s <= 300.0:
        nph = 6
    elif budget_s <= 1000.0:
        nph = 8
    else:
        nph = 10

    tau_rels = [
        0.12, 0.06, 0.03, 0.016, 0.008,
        0.004, 0.002, 0.001, 0.0005, 0.0002,
    ]
    fracs = np.asarray(
        [0.10, 0.12, 0.14, 0.16, 0.16, 0.14, 0.10, 0.08, 0.06, 0.04],
        dtype=np.float64,
    )[:nph]
    fracs = fracs / float(np.sum(fracs))

    cur_v = v.copy()
    for i in range(nph):
        now = time.monotonic()
        if (end - now) < 2.0:
            break

        if i > 0:
            a_cur = _softmax(cur_v)
            sc_cur, _ = _score_fft(a_cur)
            if (not np.isfinite(sc_cur)) or (np.isfinite(best_sc) and sc_cur > 1.04 * best_sc):
                cur_v = best_v.copy()

        a_cur = _softmax(cur_v)
        sc_cur, _ = _score_fft(a_cur)
        if not np.isfinite(sc_cur):
            sc_cur = best_sc

        bmax = sc_cur / (2.0 * n)
        if not np.isfinite(bmax) or bmax <= 0.0:
            bmax = max(1e-9, float(np.max(np.square(a_cur))))

        tau = max(tau_rels[i] * bmax, 1e-6 * bmax, 1e-14)
        phase_time = float(budget_s * fracs[i])
        deadline = min(end, now + max(1.0, phase_time))

        cur_v = _self_lbgfs_phase(cur_v, n, m, tau, deadline)
        a_new = _softmax(cur_v)
        sc_new, _ = _score_fft(a_new)

        if np.isfinite(sc_new) and sc_new < best_sc - 1e-12:
            best_sc = float(sc_new)
            best_a = a_new.copy()
            best_v = cur_v.copy()
            print("    self phase %d improved score=%.7f" % (i, best_sc))

        if not np.isfinite(sc_new):
            cur_v = best_v.copy()
        elif np.isfinite(best_sc) and sc_new > 1.04 * best_sc and i > 0:
            cur_v = best_v.copy()

    return best_a, float(best_sc)


def _optimize_linear(f, budget_s):
    f = _proj(f)
    n = int(f.size)
    if n == 0 or budget_s <= 2.0 or _sp_opt is None:
        return f
    m = _next_pow2(2 * n - 1)
    L = 2 * n - 1
    Ff = np.fft.fft(f, m)
    self_sc, _ = _score_fft(f)
    best_lin = self_sc / (2.0 * n)
    v = _log_init(f)
    best_g = f.copy()

    start = time.monotonic()
    end = start + max(2.0, float(budget_s))

    if budget_s <= 30.0:
        nph = 3
    elif budget_s <= 120.0:
        nph = 4
    else:
        nph = 5

    tau_rels = [0.12, 0.06, 0.03, 0.015, 0.008]
    fracs = np.asarray([0.20, 0.30, 0.30, 0.12, 0.08], dtype=np.float64)[:nph]
    fracs = fracs / float(np.sum(fracs))

    for i in range(nph):
        now = time.monotonic()
        if (end - now) < 2.0:
            break

        g = _softmax(v)
        Fg = np.fft.fft(g, m)
        b = np.fft.ifft(Ff * Fg).real[:L]
        b = np.maximum(b, 0.0)
        lin = float(np.max(b)) if b.size else np.inf
        if np.isfinite(lin) and lin < best_lin:
            best_lin = lin
            best_g = g.copy()
            print("    linear phase %d f*g max=%.9f" % (i, best_lin))

        bmax = max(1e-12, best_lin)
        tau = max(tau_rels[i] * bmax, 1e-6 * bmax, 1e-14)
        phase_time = float(budget_s * fracs[i])
        deadline = min(end, now + max(1.0, phase_time))
        v = _lin_lbgfs_phase(v, Ff, n, m, tau, deadline)

    g = _softmax(v)
    Fg = np.fft.fft(g, m)
    b = np.fft.ifft(Ff * Fg).real[:L]
    b = np.maximum(b, 0.0)
    lin = float(np.max(b)) if b.size else np.inf
    if np.isfinite(lin) and lin < best_lin:
        best_g = g.copy()

    return best_g


def _blend_search(a, g):
    a = _proj(a)
    g = _proj(g)
    if a.size != g.size:
        return a, _score_fft(a)[0]

    best_a = a
    best_sc, _ = _score_fft(a)

    for alpha in (1.0, 0.95, 0.85, 0.75, 0.65, 0.55, 0.45, 0.35, 0.25, 0.15, 0.08, 0.0):
        c = alpha * a + (1.0 - alpha) * g
        c = _proj(c)
        sc, _ = _score_fft(c)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = float(sc)
            best_a = c
            print("    blend alpha=%.2f score=%.7f" % (alpha, best_sc))

    return best_a, float(best_sc)


def _refine_candidate(a, budget_s):
    a = _proj(a)
    budget_s = max(0.0, float(budget_s))
    if budget_s <= 30.0:
        return _optimize_self(a, budget_s)

    start = time.monotonic()
    end = start + budget_s

    first = min(0.45 * budget_s, max(2.0, end - start - 5.0))
    a, best_sc = _optimize_self(a, first)

    rem = end - time.monotonic()
    if rem < 30.0:
        return a, float(best_sc)

    lin_b = min(0.25 * budget_s, 0.55 * rem)
    if lin_b > 15.0:
        g = _optimize_linear(a, lin_b)
        rem = end - time.monotonic()
        if rem > 10.0:
            a, best_sc = _blend_search(a, g)

    rem = end - time.monotonic()
    if rem > 60.0 and a.size <= 16384:
        lin2_b = min(0.14 * budget_s, 0.45 * rem)
        if lin2_b > 25.0:
            g2 = _optimize_linear(a, lin2_b)
            rem = end - time.monotonic()
            if rem > 10.0:
                a, best_sc = _blend_search(a, g2)

    rem = end - time.monotonic()
    fin = min(0.35 * budget_s, max(0.0, rem - 2.0))
    if fin > 15.0:
        a, best_sc = _optimize_self(a, fin)

    return a, float(best_sc)


def _upsample_linear(a, n):
    a = _proj(a)
    n = int(n)
    if a.size == n:
        return a
    x_old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(x_new, x_old, a)
    return _proj(out)


def _upsample_hold(a, n):
    a = _proj(a)
    n = int(n)
    if a.size == n:
        return a
    if n % a.size == 0:
        r = int(n // a.size)
        out = np.repeat(a, r)
    else:
        out = _upsample_linear(a, n)
    return _proj(out)


def _perturb_lowpass(a, rng, band, sigma):
    a = _proj(a)
    n = int(a.size)
    v = _log_init(a)
    i = np.arange(n, dtype=np.float64)
    two_pi = 2.0 * np.pi
    for k in range(1, int(band) + 1):
        amp = float(rng.normal(0.0, sigma / (float(k) ** 0.5)))
        ph = float(rng.uniform(0.0, two_pi))
        if amp != 0.0:
            v += amp * np.cos(two_pi * float(k) * i / float(n) + ph)
    v = np.clip(v - float(np.mean(v)), -25.0, 25.0)
    return _softmax(v)


def _norm_target(t):
    t = np.asarray(t, dtype=np.float64).ravel()
    t = np.maximum(t, 0.0)
    m = float(np.max(t))
    if not np.isfinite(m) or m <= 0.0:
        return np.ones_like(t, dtype=np.float64)
    return t / m


def _make_targets(n, b_norm, rng):
    L = 2 * int(n) - 1
    x = np.linspace(0.0, 1.0, L, dtype=np.float64)
    edge = np.abs(x - 0.5) * 2.0
    flat = np.ones(L, dtype=np.float64)
    targets = []

    if b_norm is not None:
        b_norm = _norm_target(b_norm)
        if b_norm.size == L:
            targets.append(_norm_target(0.70 * b_norm + 0.30 * flat))
            targets.append(_norm_target(0.85 * b_norm + 0.15 * (1.0 + edge)))

    for e in (0.35, 0.75, 1.10):
        targets.append(_norm_target(1.0 + e * (edge ** 1.5)))

    targets.append(_norm_target(np.maximum(1.0 - 0.25 * (edge ** 1.5), 0.15)))
    targets.append(_norm_target(np.maximum(1.0 - 0.35 * np.cos(np.pi * x), 0.10)))
    for c1, c2 in ((-0.25, 0.15), (-0.45, 0.25), (0.0, 0.35)):
        targets.append(_norm_target(1.0 + c1 * np.cos(np.pi * x) + c2 * np.cos(2.0 * np.pi * x)))

    if rng is not None:
        for _ in range(2):
            c1 = float(rng.uniform(-0.50, 0.15))
            c2 = float(rng.uniform(0.05, 0.45))
            c3 = float(rng.uniform(0.0, 0.25))
            targets.append(
                _norm_target(
                    1.0 + c1 * np.cos(np.pi * x) + c2 * np.cos(2.0 * np.pi * x)
                    + c3 * np.sin(2.0 * np.pi * x)
                )
            )

    return targets


def _spectral_candidate(target, a0, n, m, iters=25, blend=0.55):
    n = int(n)
    L = 2 * n - 1
    target = np.asarray(target, dtype=np.float64).ravel()
    if target.size < L:
        padded = np.zeros(L, dtype=np.float64)
        padded[: target.size] = target
        target = padded
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    tm = float(np.max(target))
    if not np.isfinite(tm) or tm <= 0.0:
        target = np.ones(L, dtype=np.float64)
    else:
        target /= tm

    a = _proj(a0)
    if a.size != n:
        a = _upsample_linear(a, n)
    best_a = a.copy()
    best_sc, _ = _score_fft(a)

    T = np.fft.fft(target, m)
    Tnorm = float(np.max(np.abs(T)))
    if not np.isfinite(Tnorm) or Tnorm <= 0.0:
        Tnorm = 1.0

    direct = np.fft.ifft(np.sqrt(T)).real[:n]
    pos = np.maximum(direct, 0.0)
    if float(np.sum(pos)) > 1e-10:
        a = _proj(0.45 * a + 0.55 * pos)

    for it in range(max(1, int(iters))):
        F = np.fft.fft(a, m)
        Fmax = float(np.max(np.abs(F)))
        if not np.isfinite(Fmax) or Fmax <= 0.0:
            break
        eps = 1e-8 * max(1.0, Fmax) + 1e-12 * Tnorm
        Fnew = 0.5 * (F + T / (F + eps))
        if not np.all(np.isfinite(Fnew)):
            break
        anew = np.fft.ifft(Fnew).real[:n]
        if not np.all(np.isfinite(anew)):
            break
        pos = np.maximum(anew, 0.0)
        if float(np.sum(pos)) <= 1e-10:
            pos = a
        bld = float(blend)
        if it > 0.7 * iters:
            bld = 0.75
        a = _proj((1.0 - bld) * a + bld * pos)
        if (it + 1) % 8 == 0 or it == iters - 1:
            sc, _ = _score_fft(a)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = float(sc)
                best_a = a.copy()

    return best_a, float(best_sc)


def _add_candidate(cand_list, a, max_score=2.35, thresh=0.055):
    a = _proj(a)
    if a.size == 0:
        return
    sc, _ = _score_fft(a)
    if not np.isfinite(sc):
        return
    if sc > max_score and cand_list:
        return
    if sc >= 3.5 and not cand_list:
        return
    for _, _, old in cand_list:
        if old.size != a.size:
            continue
        d = float(np.sqrt(np.sum(np.square(a - old))))
        if d < thresh:
            return
    cand_list.append((float(sc), len(cand_list) + 1, a))
    if len(cand_list) > 10:
        cand_list.sort()
        del cand_list[6:]


def _make_candidates(base_a, N, rng, stage_budget, do_spectral):
    N = int(N)
    base_a = _proj(base_a)
    cand = []

    lin = _upsample_linear(base_a, N)
    _add_candidate(cand, lin, max_score=3.0)
    hold = _upsample_hold(base_a, N)
    _add_candidate(cand, hold, max_score=3.0)

    if not cand:
        _add_candidate(cand, np.ones(N, dtype=np.float64) / float(N), max_score=1e9)

    cand.sort()
    best_a = cand[0][2]
    best_sc = cand[0][0]

    for band, sig in ((4, 0.10), (12, 0.22), (32, 0.45), (80, 0.75)):
        p = _perturb_lowpass(best_a, rng, band, sig)
        _add_candidate(cand, p, max_score=max(1.90, 1.08 * best_sc + 0.05), thresh=0.045)

    if do_spectral:
        m = _next_pow2(2 * N - 1)
        best_sc2, bbest = _score_fft(best_a)
        bnorm = None
        if bbest is not None and bbest.size and np.isfinite(best_sc2):
            bnorm = _norm_target(bbest)
        targets = _make_targets(N, bnorm, rng)
        for t in targets[:4]:
            t = _norm_target(t)
            sp, _ = _spectral_candidate(t, best_a, N, m, iters=22, blend=0.58)
            _add_candidate(cand, sp, max_score=max(1.90, 1.06 * best_sc + 0.08), thresh=0.05)

    cand.sort()
    return cand[:4]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Returns a list of non-negative floats.  It is optimized to minimize
    evaluate_sequence() while staying under the supplied time budget.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
        if not np.isfinite(budget_s) or budget_s <= 0.0:
            budget_s = 2400.0
    except Exception:
        budget_s = 2400.0

    start = time.monotonic()
    deadline = start + max(120.0, budget_s - 90.0)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 0 and np.isfinite(arr).all():
            parent = _proj(arr)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = np.ones(2048, dtype=np.float64) / 2048.0

    best_a = parent.copy()
    best_sc, _ = _score_fft(best_a)
    print("Initial parent length=%d score=%.7f" % (best_a.size, best_sc))

    total_work = max(120.0, budget_s - 90.0)
    if total_work > 1700.0:
        max_len = 16384
    elif total_work > 900.0:
        max_len = 8192
    elif total_work > 400.0:
        max_len = 4096
    else:
        max_len = 2048

    base_len = max(1024, int(best_a.size))
    if base_len > max_len:
        max_len = base_len

    lengths = []
    p = 1024
    while p <= max_len:
        if p >= base_len or len(lengths) > 0:
            lengths.append(int(p))
        p <<= 1
    if not lengths:
        lengths.append(int(max(base_len, 1024)))
    lengths = lengths[-4:]

    if len(lengths) == 1:
        weights = [1.0]
    elif len(lengths) == 2:
        weights = [0.16, 0.84]
    elif len(lengths) == 3:
        weights = [0.05, 0.22, 0.73]
    else:
        weights = [0.03, 0.08, 0.59, 0.30]

    history = []

    for stage_idx, (N, w) in enumerate(zip(lengths, weights)):
        now = time.monotonic()
        rem = deadline - now - 30.0
        if rem <= 20.0:
            break
        stage_budget = min(total_work * float(w), max(20.0, rem * 0.98))
        if stage_budget < 25.0:
            continue

        print("Stage N=%d budget=%.1fs current_best=%.7f" % (N, stage_budget, best_sc))

        gen_start = time.monotonic()
        do_spectral = (
            (N <= 4096 and stage_budget > 150.0)
            or (N == 8192 and stage_budget > 500.0)
            or (N > 8192 and stage_budget > 900.0)
        )
        cand = _make_candidates(best_a, N, rng, stage_budget, do_spectral)
        gen_time = time.monotonic() - gen_start
        stage_deadline = min(deadline, gen_start + stage_budget)
        stage_time = max(0.0, stage_deadline - time.monotonic())

        if stage_time < 25.0 or not cand:
            continue

        chosen = None
        chosen_sc = np.inf

        if len(cand) > 1 and stage_time > 450.0:
            b0 = min(0.68 * stage_time, max(25.0, stage_deadline - time.monotonic() - 160.0))
            if b0 > 25.0:
                a0, sc0 = _refine_candidate(cand[0][2], b0)
                chosen = a0
                chosen_sc = sc0

            rem2 = stage_deadline - time.monotonic() - 30.0
            if rem2 > 50.0:
                b1 = min(0.26 * stage_time, max(25.0, rem2))
                a1, sc1 = _refine_candidate(cand[1][2], b1)
                if np.isfinite(sc1) and sc1 < chosen_sc:
                    chosen = a1
                    chosen_sc = sc1
        else:
            chosen, chosen_sc = _refine_candidate(cand[0][2], max(5.0, stage_deadline - time.monotonic()))

        if chosen is None:
            continue
        if np.isfinite(chosen_sc) and chosen_sc < best_sc - 1e-12:
            best_a = chosen
            best_sc = float(chosen_sc)
            print("After stage N=%d best=%.7f" % (N, best_sc))
        else:
            print("After stage N=%d no improvement, keeping score=%.7f" % (N, best_sc))
        history.append((best_sc, best_a.copy()))

    rem = deadline - time.monotonic() - 20.0
    if rem > 30.0 and best_a is not None:
        pol = _refine_candidate(best_a, min(rem, max(30.0, 0.15 * budget_s)))
        if np.isfinite(pol[1]) and pol[1] < best_sc - 1e-12:
            best_a, best_sc = pol
            print("Final polish score=%.7f" % best_sc)

    best_a = _proj(best_a)
    best_a = np.asarray(best_a, dtype=np.float64)
    if best_a.size == 0 or not np.isfinite(best_a).all():
        best_a = np.ones(2048, dtype=np.float64) / 2048.0
        best_sc = _score_fft(best_a)[0]

    # Remove extremely tiny values only for cleanliness; this is negligible.
    tiny = best_a < 1e-15
    if np.any(tiny):
        best_a[tiny] = 0.0
        best_a = _proj(best_a)

    def _fallback_sequence():
        for _, a in reversed(history):
            if a is not None and a.size <= 8192:
                return _proj(a)
        return np.ones(2048, dtype=np.float64) / 2048.0

    try:
        exact = float(evaluate_sequence([float(x) for x in best_a]))
        print("Final FFT score=%.7f exact score=%.7f length=%d" % (best_sc, exact, len(best_a)))
        if not np.isfinite(exact):
            best_a = _fallback_sequence()
    except Exception as e:
        print("Exact validation failed, using fallback: %s" % e)
        best_a = _fallback_sequence()
        if best_a.size <= 8192:
            try:
                exact = float(evaluate_sequence([float(x) for x in best_a]))
                print("Fallback exact score=%.7f length=%d" % (exact, len(best_a)))
            except Exception as e2:
                print("Fallback validation failed: %s" % e2)

    return [float(x) for x in best_a]
```
