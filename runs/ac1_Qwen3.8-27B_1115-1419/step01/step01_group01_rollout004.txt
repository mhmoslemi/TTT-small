    # cross_corr = ifft(fft(a, M) * conj(fft(w, M))) ... 
    # Wait, I need to check the convention.
    
    # np.fft-based cross-correlation:
    # corr = ifft(conj(fft(a, M)) * fft(w, M)) gives circular cross-correlation
    # For linear, we need M >= n + (2n-1) - 1 = 3n - 2
    
    # Actually, let me just use:
    # g[i] = 2 * ifft(fft(a, M) * conj(fft(w_rev, M))).real[i]
    # where w_rev = w[::-1] and M >= n + len(w) - 1 = n + 2n-2 = 3n-2
    
    # Hmm this is getting complicated. Let me just use a simpler FFT:
    # Note that g[i] = 2 * sum_j a[j]*w[i+j]
    # Let's define v[j] = a[j] (length n) and u[k] = w[k] (length 2n-1)
    # g[i] = sum_j v[j]*u[i+j] = (v ⋆ u)[i]
    # = conv(v, u_rev)[n-1+i] where u_rev = u[::-1]
    
    # Using FFT:
    # M >= n + (2n-1) - 1 = 3n-2
    # g = ifft(fft(v, M) * fft(u_rev, M)).real[n-1 : n-1+n]
    
    # Wait, conv(v, u_rev)[k] = sum_j v[j]*u_rev[k-j] = sum_j v[j]*u_rev[k-j]
    # u_rev[m] = u[2n-2-m]
    # conv(v, u_rev)[k] = sum_j v[j] * u[2n-2-(k-j)] = sum_j v[j]*u[2n-2-k+j]
    
    # I want g[i] = sum_j v[j]*u[i+j]
    # So I need 2n-2-k+j = i+j, i.e., k = 2n-2-i
    # conv(v, u_rev)[2n-2-i] = g[i]
    
    # So g = conv(v, u_rev)[2n-2 : n-1 : -1] ... i.e., indices 2n-2, 2n-3, ..., n-1
    # In numpy: conv_result[2n-2 : n-1 : -1]
    
    # This is getting too complicated. Let me just use the simpler approach:
    # g[i] = 2 * ifft(fft(w, M) * conj(fft(a, M))).real[i] (circular, with M large enough)
    # Actually for M >= 3n-2, the circular and linear correlation match for the range we need.
    
    M = _next_pow2(3 * n)  # enough for linear cross-correlation
    W = np.fft.fft(w, M)
    A = np.fft.fft(a, M)
    c = np.fft.ifft(W * np.conj(A)).real[:n]  # cross-correlation
    return 2.0 * c
