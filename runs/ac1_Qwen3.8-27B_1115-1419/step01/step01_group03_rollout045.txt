```python
def compute_logsumexp_grad(a, beta):
    """Compute gradient of log-sum-exp of convolution coefficients."""
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    
    # Numerically stable log-sum-exp
    max_c = c.max()
    shifted = beta * (c - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    weights = exp_shifted / sum_exp  # softmax weights
    
    # Gradient: g[j] = 2 * sum_k weights[k] * a[k-j]
    # This is 2 * (weights * a_reversed)[n-1+j]
    a_rev = a[::-1].copy()
    conv = np.convolve(weights, a_rev, mode='full')  # length (2n-1) + n - 1 = 3n-2
    # We need positions n-1+j for j=0,...,n-1, i.e., positions n-1 to 2n-2
    g = 2.0 * conv[n-1:2*n-1]
    
    # Value of log-sum-exp
    val = max_c + np.log(sum_exp) / beta
    
    return g, val
```

```python
import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    r = np.where(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1)
    return np.maximum(v - theta, 0.0)

def fast_score(a):
    """Compute the score quickly."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def run_fista_logsumexp(a0, duration, rng, beta_start=5.0, beta_end=50.0):
    """
    FISTA with log-sum-exp smoothing.
    """
    n = len(a0)
    a = a0.copy()
    a_prev = a0.copy()
    t = 1.0
    step = 0.5  # Initial step size
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    
    # Schedule for beta: start small, increase over time
    total_time_est = duration
    beta = beta_start
    
    while time.monotonic() < deadline:
        # Update beta (annealing schedule)
        progress = min(1.0, (duration - (deadline - time.monotonic())) / duration)
        beta = beta_start + (beta_end - beta_start) * progress * progress  # quadratic schedule
        
        # FISTA momentum
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.1:
            y /= sy
        
        # Compute convolution and gradient
        c_y = np.convolve(y, y)
        max_c = c_y.max()
        shifted = beta * (c_y - max_c)
        exp_shifted = np.exp(shifted)
        sum_exp = exp_shifted.sum()
        weights = exp_shifted / sum_exp
        
        # Gradient via convolution
        a_rev = y[::-1]
        conv = np.convolve(weights, a_rev, mode='full')
        g = 2.0 * conv[n-1:2*n-1]
        
        # Gradient step + project
        a_new = y - step * g
        a_new = project_simplex(a_new)
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic score check
        if it % 20 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck == 30:
                step *= 0.9
            elif stuck == 100:
                step *= 0.8
                a_prev = best_a.copy()
                t = 1.0
            elif stuck > 300:
                a = best_a + rng.normal(0.0, 0.02 / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t = 1.0
                step = 0.5
                stuck = 0
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = np.where(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def grad_logsumexp(y, beta):
    """Compute gradient of log-sum-exp of conv(y,y) w.r.t. y."""
    n = len(y)
    c = np.convolve(y, y)
    max_c = c.max()
    shifted = beta * (c - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)

    if sum_exp < 1e-300:

    # = 2 * (weights * y_rev)[n-1+j]
    y_rev = y[::-1]
    conv = np.convolve(weights, y_rev, mode='full')
    g = 2.0 * conv[n-1:2*n-1]
    return g
```

```python
        improved = False

        for idx in indices:
            if time.monotonic() >= deadline:
                break
            orig = a[idx]
            # Try multiplicative perturbations
            for factor in [0.5, 0.75, 1.25, 1.5, 0.9, 1.1]:
                a[idx] = orig * factor
                if a[idx] < 0:
                    a[idx] = 0
                s = a.sum()
                if s < 0.01:
                    a[idx] = orig
                    continue
                a_norm = a / s
                sc = fast_score(a_norm)
                if sc < best_score - 1e-10:
                    best_score = sc
                    a = a_norm.copy()
                    improved = True
                    break
            a[idx] = orig

        if improved:
            stuck = 0
        else:
            stuck += 1
        it += 1

        if it % 50 == 0:
            print(f"    Coord polish it={it}, score={best_score:.8f}")

    return a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search function. Uses FISTA with log-sum-exp smoothing (FFT-accelerated),
    multiple diverse initializations, beta annealing, symmetrization, and
    coordinate-descent polishing to minimize 2n*max(a*a)/(sum(a))^2.
    """
    rng = np.random.default_rng(12345)
    total_deadline = time.monotonic() + float(budget_s) - 25.0

    # Get initial sequence
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    if n < 2:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    # Normalize
    S = arr.sum()
    if S < 1e-10:
        arr = np.ones(n)
        S = n
    arr = arr / S

    # Generate diverse initial candidates
    candidates = []
    names = []

    # 1. Given sequence
    candidates.append(arr.copy())
    names.append("given")

    # 2. Symmetrized given
    sym = 0.5 * (arr + arr[::-1])
    sym = np.maximum(sym, 0.0)
    ss = sym.sum()
    if ss > 1e-12:
        sym /= ss
    candidates.append(sym)
    names.append("sym_given")

    # 3. Uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)
    names.append("uniform")

    # 4. Random exponential (several seeds)
    for seed in range(4):
        r = rng.exponential(1.0, n)
        sr = r.sum()
        if sr > 1e-12:
            r /= sr
            candidates.append(r)
            names.append(f"exp_{seed}")

    # 5. Bimodal structure
    x = np.linspace(0.0, 1.0, n)
    for center, width in [(0.3, 0.15), (0.5, 0.2), (0.7, 0.15)]:
        bump = np.exp(-((x - center) / width) ** 2)
        sb = bump.sum()
        if sb > 1e-12:
            bump /= sb
            candidates.append(bump.copy())
            names.append(f"gauss_{center}_{width}")

    # 6. Raised cosine
    bump = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    sb = bump.sum()
    if sb > 1e-12:
        bump /= sb
        candidates.append(bump.copy())
        names.append("raised_cos")

    # 7. Block structure
    block_size = max(1, n // 80)
    n_blocks = (n + block_size - 1) // block_size
    block_vals = rng.exponential(1.0, n_blocks)
    block_seq = np.repeat(block_vals, block_size)[:n]
    bs = block_seq.sum()
    if bs > 1e-12:
        block_seq /= bs
        candidates.append(block_seq)
        names.append("block")

    # 8. Sparse random
    sparse = rng.exponential(1.0, n)
    mask = rng.random(n) > 0.5
    sparse[~mask] = 0.0
    ss = sparse.sum()
    if ss > 1e-12:
        sparse /= ss
        candidates.append(sparse)
        names.append("sparse")

    # Evaluate all candidates
    print(f"=== n = {n} ===")
    best_cand = None
    best_cand_score = np.inf
    cand_scores = []
    for c, nm in zip(candidates, names):
        sc = fast_score(c)
        cand_scores.append(sc)
        print(f"  {nm}: {sc:.8f}")
        if sc < best_cand_score:
            best_cand_score = sc
            best_cand = c.copy()

    print(f"Best initial: {best_cand_score:.8f}")

    # Sort candidates by score
    order = np.argsort(cand_scores)

    # Phase 1: FISTA-LSE from best candidate (largest time allocation)
    time_left = total_deadline - time.monotonic()
    phase1_time = min(time_left * 0.5, 1200.0)
    if phase1_time > 20:
        print(f"Phase 1: FISTA-LSE best ({phase1_time:.0f}s)")
        best_a, best_score = run_fista_lse(
            best_cand, phase1_time, rng,
            beta_start=1.0, beta_end=80.0, step_scale=1.0
        )
        print(f"  After phase 1: {best_score:.8f}")
    else:
        best_a = best_cand.copy()
        best_score = best_cand_score

    # Phase 2: FISTA-LSE from second-best candidate
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        # Pick a diverse second candidate
        for idx in order:
            if np.allclose(candidates[idx], best_cand, atol=0.01):
                continue
            phase2_time = min(time_left * 0.3, 600.0)
            if phase2_time < 20:
                break
            print(f"Phase 2: FISTA-LSE '{names[idx]}' ({phase2_time:.0f}s)")
            a2, s2 = run_fista_lse(
                candidates[idx], phase2_time, rng,
                beta_start=2.0, beta_end=100.0, step_scale=1.0
            )
            if s2 < best_score:
                best_a, best_score = a2, s2
                print(f"  Phase 2 improved: {best_score:.8f}")
            else:
                print(f"  Phase 2 no improvement: {s2:.8f}")
            break

    # Phase 3: Continue FISTA-LSE on current best with high beta
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        phase3_time = min(time_left * 0.4, 600.0)
        if phase3_time > 15:
            print(f"Phase 3: High-beta polish ({phase3_time:.0f}s)")
            best_a, best_score = run_fista_lse(
                best_a, phase3_time, rng,
                beta_start=50.0, beta_end=300.0, step_scale=0.5
            )
            print(f"  After phase 3: {best_score:.8f}")

    # Phase 4: Another diverse start if time allows
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        for idx in order:
            if np.allclose(candidates[idx], best_cand, atol=0.01):
                continue
            if np.allclose(candidates[idx], best_a, atol=0.01):
                continue
            phase4_time = min(time_left * 0.3, 300.0)
            if phase4_time < 20:
                break
            print(f"Phase 4: FISTA-LSE '{names[idx]}' ({phase4_time:.0f}s)")
            a4, s4 = run_fista_lse(
                candidates[idx], phase4_time, rng,
                beta_start=3.0, beta_end=120.0, step_scale=1.0
            )
            if s4 < best_score:
                best_a, best_score = a4, s4
                print(f"  Phase 4 improved: {best_score:.8f}")
            else:
                print(f"  Phase 4 no improvement: {s4:.8f}")
            break

    # Phase 5: Coordinate descent polishing
    time_left = total_deadline - time.monotonic()
    if time_left > 15:
        polish_time = min(time_left * 0.7, 120.0)
        if polish_time > 5:
            print(f"Phase 5: Coordinate polish ({polish_time:.0f}s)")
            best_a, best_score = coordinate_polish(best_a, polish_time, rng)
            print(f"  After polish: {best_score:.8f}")

    # Final symmetrization and verification
    best_a = 0.5 * (best_a + best_a[::-1])
    best_a = np.maximum(best_a, 0.0)
    bs = best_a.sum()
    if bs > 1e-12:
        best_a /= bs

    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.8f}")

    # Fallback check
    if not np.isfinite(final_score) or final_score > best_cand_score:
        print(f"Falling back to best initial: {best_cand_score:.8f}")
        final_list = [float(x) for x in best_cand]
        final_score = best_cand_score

    print(f"=== FINAL: {final_score:.8f} (n={n}) ===")
    return final_list
```
