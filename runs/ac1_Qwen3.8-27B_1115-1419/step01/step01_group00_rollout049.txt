"""
Discrete low-peak self-convolution optimizer.

The objective is
    R(a) = 2*n*max_k (a*a)_k / (sum_j a_j)^2
for a non-negative sequence a of length n.  The code works by
 - generating diverse analytic/interpolated initial candidates (including
   the known (1-z)^(-1/2) truncated square-root profile and its modulations),
 - optimizing in log-space with a smoothed active-subgradient for the maximum
   convolution coefficient while maximizing the total mass under the
   implicit max-convolution constraint,
 - periodically perturbing the current best basin to escape local minima,
 - returning the best finite-length sequence found before the time budget.
"""

import time
import numpy as np
from operator import itemgetter


def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _conv_only(a, L):
    n = a.size
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_and_fa(a, L):
    n = a.size
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c, fa


def _evaluate_candidate(a, n, L):
    if a is None:
        return float("inf"), None
    a = np.asarray(a, dtype=np.float64)
    if a.size != n:
        return float("inf"), None
    if not np.all(np.isfinite(a)):
        return float("inf"), None
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return float("inf"), None
    amax = float(a.max())
    if amax > 1e100:
        a = a / amax
        s = float(a.sum())
        if s < 1e-12:
            return float("inf"), None
    c, _ = _conv_and_fa(a, L)
    m = float(c.max())
    if m <= 0.0 or not np.isfinite(m):
        return float("inf"), None
    a_norm = a / np.sqrt(m)
    f = 2.0 * float(n) * m / (s * s)
    if not np.isfinite(f):
        return float("inf"), None
    return f, a_norm


def _add_candidate(pool, a, n, L):
    f, a_norm = _evaluate_candidate(a, n, L)
    if np.isfinite(f) and a_norm is not None:
        pool.append((f, a_norm))


def _update_pool(pool, item, max_size=7):
    f = item[0]
    if not np.isfinite(f):
        return pool
    if len(pool) < max_size:
        pool.append(item)
        pool.sort(key=itemgetter(0))
        return pool
    if f < pool[-1][0]:
        pool[-1] = item
        pool.sort(key=itemgetter(0))
    return pool


def _central_binomial(n):
    a = np.empty(n, dtype=np.float64)
    if n <= 0:
        return a
    a[0] = 1.0
    val = 1.0
    for k in range(1, n):
        val *= (2.0 * k - 1.0) / (2.0 * k)
        a[k] = val
    return a


def _random_profile(n, rng):
    idx = np.arange(n, dtype=np.float64)
    t = idx / float(n - 1)
    alpha = float(rng.uniform(0.0, 0.85))
    beta = float(rng.uniform(0.0, 0.65))
    gamma = float(rng.uniform(-2.0, 3.0))
    kappa = float(rng.uniform(-1.0, 1.0))
    logw = (
        -alpha * np.log1p(idx)
        - beta * np.log(n - idx)
        + gamma * t
        + kappa * (t - 0.5) ** 2
    )
    logw -= float(logw.max())
    return np.exp(logw)


def _initial_candidates(n, L, rng, parent_array):
    cands = []
    idx = np.arange(n, dtype=np.float64)
    t = idx / float(n - 1)

    simple = _central_binomial(n)
    rev = simple[::-1].copy()

    _add_candidate(cands, simple, n, L)
    _add_candidate(cands, rev, n, L)

    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        blend = alpha * simple + (1.0 - alpha) * rev
        _add_candidate(cands, blend, n, L)

    # Shapes built by multiplicatively modulating the known left/right profiles.
    log_simple = np.log(np.maximum(simple, 1e-300))
    log_rev = np.log(np.maximum(rev, 1e-300))

    for base_log in (log_simple, log_rev):
        for g1 in (-1.0, -0.4, 0.0, 0.4, 1.0, 2.0):
            for g2 in (-1.0, 0.0, 1.0):
                logw = base_log + g1 * t + g2 * (t - 0.5) ** 2
                logw -= float(logw.max())
                _add_candidate(cands, np.exp(logw), n, L)

        # Tail / head localized enhancements.
        for p in (2.0, 5.0, 11.0):
            tp = t ** p
            for g in (0.5, 1.5, 3.0):
                logw = base_log + g * tp
                logw -= float(logw.max())
                _add_candidate(cands, np.exp(logw), n, L)

                if t[-1] > 0.0:
                    tp2 = (1.0 - t) ** p
                    logw2 = base_log + g * tp2
                    logw2 -= float(logw2.max())
                    _add_candidate(cands, np.exp(logw2), n, L)

    # General power-law profiles.
    for alpha in (0.0, 0.2, 0.4, 0.6):
        for beta in (0.0, 0.2, 0.5):
            for gamma in (-1.0, 0.0, 1.0, 2.0):
                for kappa in (-1.0, 1.0):
                    logw = (
                        -alpha * np.log1p(idx)
                        - beta * np.log(n - idx)
                        + gamma * t
                        + kappa * (t - 0.5) ** 2
                    )
                    logw -= float(logw.max())
                    _add_candidate(cands, np.exp(logw), n, L)

    # A few fully random profiles.
    for _ in range(8):
        _add_candidate(cands, _random_profile(n, rng), n, L)

    # Parent construction, if available, directly at the same length.
    if parent_array is not None and np.all(np.isfinite(parent_array)):
        _add_candidate(cands, np.maximum(parent_array, 0.0), n, L)

    cands.sort(key=itemgetter(0))
    return cands


def _perturb_x(x, rng, std, mode="full"):
    y = x.copy()
    if mode == "block":
        w = int(max(4, np.arange(4, 17, 3)[int(rng.integers(0, 4))] * (x.size // 32 + 1)))
        if x.size > w:
            s = int(rng.integers(0, x.size - w + 1))
            y[s:s + w] += rng.normal(0.0, max(0.02, 2.0 * std), size=w)
    # Always a small global component to keep the whole shape alive.
    y += rng.normal(0.0, max(0.005, 0.25 * std), size=x.size)
    return y


def _optimize_log(a0, n, L, deadline, rng, tag="", tau_start=0.25, lr_start=0.18,
                  max_iter=2000000):
    """
    Optimize in log-space.  The sequence is kept normalized so that
    max_k (a*a)_k is 1; then minimizing R is equivalent to maximizing
    sum(a)^2.
    """
    t0 = time.monotonic()
    a = np.asarray(a0, dtype=np.float64).copy()
    if a.size != n:
        return a, float("inf"), np.log(np.maximum(a, 1e-12))
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-12)

    c, _ = _conv_and_fa(a, L)
    m = float(c.max())
    if m <= 0.0 or not np.isfinite(m):
        return a, float("inf"), np.log(a)

    a = a / np.sqrt(m)
    x = np.log(a)
    min_log = -30.0
    max_log = 5.0
    x = np.clip(x, min_log, max_log)

    s = float(a.sum())
    f = 2.0 * float(n) / (s * s)
    best_a = a.copy()
    best_f = f
    best_x = x.copy()

    m_mom = np.zeros(n, dtype=np.float64)
    v_mom = np.zeros(n, dtype=np.float64)
    b1 = 0.90
    b2 = 0.999
    b1_pow = 1.0
    b2_pow = 1.0

    steps = 0
    lr = float(lr_start)
    tau = float(tau_start)
    stall = 0
    last_print = t0
    last_f = best_f

    while time.monotonic() < deadline:
        if steps >= max_iter:
            break
        steps += 1
        b1_pow *= b1
        b2_pow *= b2

        a = np.exp(np.clip(x, min_log, max_log))
        fa = np.fft.rfft(a, L)
        c = np.fft.irfft(fa * fa, L)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        m = float(c.max())
        s = float(a.sum())
        if m <= 0.0 or s < 1e-12 or not np.isfinite(m) or not np.isfinite(s):
            break

        f = 2.0 * float(n) * m / (s * s)
        if f < best_f - 1e-11:
            best_f = f
            best_a = a / np.sqrt(m)
            best_x = x - 0.5 * np.log(m)
            stall = 0
            now = time.monotonic()
            if now - last_print > 5.0 or f < last_f - 5e-5:
                print(f"  [{tag}] obj={f:.6f} tau={tau:.4f} lr={lr:.3f} stall={stall}")
                last_print = now
                last_f = best_f
        else:
            stall += 1

        if stall > 9000:
            x = x + rng.normal(0.0, 0.05, size=n)
            a_tmp = np.exp(np.clip(x, min_log, max_log))
            c_tmp, _ = _conv_and_fa(a_tmp, L)
            m_tmp = float(c_tmp.max())
            if m_tmp > 0.0 and np.isfinite(m_tmp):
                x = x - 0.5 * np.log(m_tmp)
            stall = 0
            lr = max(0.02, 0.5 * lr)
            tau = max(0.05, 0.6 * tau)
            continue

        scores = (c - m) / max(1e-8, tau)
        scores = np.minimum(scores, 0.0)
        p = np.exp(scores)
        psum = float(p.sum())
        if psum <= 1e-300:
            p = np.zeros(2 * n - 1, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            p /= psum

        fp = np.fft.rfft(p, L)
        h = 2.0 * np.fft.irfft(fp * np.conj(fa), L)[:n]
        dot = float(np.dot(a, h))
        if not np.isfinite(dot) or dot <= 1e-12:
            break

        h *= (2.0 * m / dot)
        g = a * (h / m - 2.0 / s)
        g -= float(g.mean())
        rms = float(np.sqrt(np.mean(g * g)))
        if rms < 1e-14:
            break

        g = g / rms
        m_mom = b1 * m_mom + (1.0 - b1) * g
        v_mom = b2 * v_mom + (1.0 - b2) * (g * g)
        m_hat = m_mom / (1.0 - b1_pow)
        v_hat = v_mom / (1.0 - b2_pow)
        dr = m_hat / (np.sqrt(v_hat) + 1e-8)
        drms = float(np.sqrt(np.mean(dr * dr)))
        if drms < 1e-8:
            break

        current_f = f
        accepted = False
        for alpha in (lr, 0.5 * lr, 0.25 * lr, 0.125 * lr):
            if not np.isfinite(alpha) or alpha <= 0.0:
                continue
            x_try = x - alpha * dr
            x_try = np.clip(x_try, min_log, max_log)
            a_try = np.exp(x_try)
            c_try = _conv_only(a_try, L)
            m_try = float(c_try.max())
            s_try = float(a_try.sum())
            if m_try <= 0.0 or s_try < 1e-12:
                continue
            f_try = 2.0 * float(n) * m_try / (s_try * s_try)
            if f_try < current_f - 1e-12:
                x = x_try - 0.5 * np.log(m_try)
                a_norm = a_try / np.sqrt(m_try)
                if f_try < best_f - 1e-11:
                    best_f = f_try
                    best_a = a_norm.copy()
                    best_x = x.copy()
                    stall = 0
                    now = time.monotonic()
                    if now - last_print > 5.0 or f_try < last_f - 5e-5:
                        print(f"  [{tag}] step obj={f_try:.6f}")
                        last_print = now
                        last_f = best_f
                accepted = True
                break

        if accepted:
            lr = min(1.0, lr * 1.04)
            stall = 0
            if tau > 0.02:
                tau = max(0.02, tau * 0.999)
        else:
            lr = max(0.02, lr * 0.70)
            stall += 1
            if stall > 2500:
                x = x + rng.normal(0.0, 0.025, size=n)
                a_tmp = np.exp(np.clip(x, min_log, max_log))
                c_tmp, _ = _conv_and_fa(a_tmp, L)
                m_tmp = float(c_tmp.max())
                if m_tmp > 0.0 and np.isfinite(m_tmp):
                    x = x - 0.5 * np.log(m_tmp)
                stall = 0

        if steps % 700 == 0:
            if stall < 300:
                tau = max(0.015, tau * 0.985)
            else:
                tau = min(0.45, tau * 1.15)

    return best_a, best_f, best_x


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    deadline = start + max(60.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    # Prefer continuing with the parent construction's length if it is sane.
    parent_array = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):

        parent_array = None

        n = parent_array.size
    else:
        n = 4096

    L = _next_pow2(2 * n - 1)

    cands = _initial_candidates(n, L, rng, parent_array)
    if not cands:
        a = _central_binomial(n)
        f, a2 = _evaluate_candidate(a, n, L)
        if not np.isfinite(f) or a2 is None:
            a2 = a / np.sqrt(max(1e-300, float(np.max(_conv_only(a, L)))))
            f = 2.0 * n / float(a2.sum()) ** 2
        cands = [(f, a2)]

    cands.sort(key=itemgetter(0))
    best_f, best_a = cands[0]
    pool = [(f, a) for f, a in cands[:6]]

    print(f"n={n} L={L} initial best={best_f:.6f} top={len(cands)}")

    # Main long polish of the best initial candidate.
    total = deadline - start
    main_seconds = min(1200.0, max(300.0, 0.55 * total))
    main_deadline = min(deadline - 350.0, start + main_seconds)
    best_a, best_f, _ = _optimize_log(
        best_a, n, L, main_deadline, rng,
        tag="main", tau_start=0.25, lr_start=0.18
    )
    pool = _update_pool(pool, (best_f, best_a))
    print(f"after main: best={best_f:.6f}")

    # Perturbation + re-polish loop.
    attempt = 0
    while time.monotonic() < deadline - 130.0:
        attempt += 1

        if attempt % 5 == 0 and len(pool) > 1:
            src_f, src_a = pool[attempt % len(pool)]
        else:
            src_f, src_a = pool[0]

        if attempt <= 4:
            std = 0.035
        elif attempt <= 12:
            std = 0.06
        elif attempt <= 28:
            std = 0.10
        else:
            std = 0.16

        if attempt % 4 == 0:
            mode = "block"
        else:
            mode = "full"

        src_x = np.log(np.maximum(src_a, 1e-12))
        x_p = _perturb_x(src_x, rng, std, mode)

        # Occasionally inject a fresh random basin.
        if attempt in (8, 16, 27, 40, 55):
            a_rand = _random_profile(n, rng)
            f_rand, a_rand_n = _evaluate_candidate(a_rand, n, L)
            if np.isfinite(f_rand) and f_rand < best_f:
                best_f = f_rand
                best_a = a_rand_n
                pool = _update_pool(pool, (best_f, best_a))
                print(f"new random basin improves: best={best_f:.6f}")
            if a_rand_n is not None:
                x_p = np.log(np.maximum(a_rand_n, 1e-12))

        subtime = 30.0
        if attempt > 8:
            subtime = 45.0
        if attempt > 20:
            subtime = 65.0
        sub_deadline = min(time.monotonic() + subtime, deadline - 130.0)
        if sub_deadline - time.monotonic() < 12.0:
            break

        a_p = np.exp(np.clip(x_p, -30.0, 5.0))
        a_new, f_new, _ = _optimize_log(
            a_p, n, L, sub_deadline, rng,
            tag=f"p{attempt}", tau_start=0.18, lr_start=0.15
        )

        if np.isfinite(f_new):
            pool = _update_pool(pool, (f_new, a_new))
            if f_new < best_f:
                best_f = f_new
                best_a = a_new
                print(f"perturb {attempt} improves: best={best_f:.6f}")

    # Final refinishing with all remaining time.
    final_deadline = deadline - 5.0
    if final_deadline - time.monotonic() > 15.0:
        best_a, best_f, _ = _optimize_log(
            best_a, n, L, final_deadline, rng,
            tag="final", tau_start=0.10, lr_start=0.13
        )

    # Sanitize and validate.
    result_arr = np.asarray(best_a, dtype=np.float64)

        result_arr = np.ones(n, dtype=np.float64)
    result_arr = np.clip(result_arr, 0.0, 1000.0)
    result = [float(v) for v in result_arr]

    try:
        val = float(evaluate_sequence(result))
    except Exception:
        val = float("inf")

    if not np.isfinite(val):
        # Fallback: known safe truncated (1-z)^(-1/2) profile.
        a_fb = _central_binomial(n)

            result = [float(max(0.0, min(1000.0, v))) for v in a_fb_norm]
        else:
            result = [1.0] * n
        try:
            val = float(evaluate_sequence(result))
        except Exception:
            val = float("inf")

    print(f"FINAL: internal={best_f:.6f} evaluator={val:.6f} len={len(result)}")
    return result
