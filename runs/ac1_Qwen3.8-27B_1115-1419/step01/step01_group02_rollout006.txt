from scipy.fft import fft, ifft

def _fft_conv(a, a2):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(a2) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(a2, n)).real[:n_out]

def _autocorr(a):
    """Compute linear autocorrelation (conv with itself)."""
    return _fft_conv(a, a)

def _score(a):
    """Compute score."""
    n = len(a)
    b = _autocorr(a)
    max_b = float(np.max(b))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

"""
Minimize 2n*max(autocorr)/(sum)^2 using mirror descent with smooth-max subgradient,
Adam per-coordinate adaptation, multi-peak weighting, and diverse restarts.
Uses FFT-based convolution for O(n log n) subgradient computation.
Combines global mirror descent with local coordinate refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _score_fast(a):
    n = len(a)
    b = _fft_conv(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _normalize(a, target_sum):
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, max(target_sum / len(a), 1e-6))
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full_like(a, max(target_sum / len(a), 1e-6))
    return a * (target_sum / s)


def _compute_subgrad(b, a, alpha=30.0, top_k_frac=0.05):
    """
    Compute subgradient of smooth max of b (autocorrelation) w.r.t. a.
    Uses FFT to compute g[i] = 2 * sum_j a[j] * w[i+j] efficiently.
    """
    n = len(a)
    b_max = float(np.max(b))
    
    # Compute weights with top-k emphasis
    n_pear = max(top_k_frac * len(b), 5)
    if n_pear >= len(b):
        w = np.exp(alpha * (b - b_max))
        w /= np.sum(w)
    else:
        # Top-k with exponential decay in full range
        w_full = np.exp(alpha * (b - b_max))
        w_full /= np.sum(w_full)
        # Emphasize top peaks
        top_idx = np.argsort(b)[-int(n_pear):]
        w_boosted = w_full.copy()
        w_boosted[top_idx] *= 3.0
        w_boosted /= np.sum(w_boosted)
        w = w_boosted
    
    # Compute g[i] = 2 * sum_j a[j] * w[i+j]
    # This equals conv(a, w_rev)[2n-2-i] where w_rev = w[::-1]
    # conv length = n + (2n-1) - 1 = 3n-2
    w_rev = w[::-1]
    n_conv = n + (2 * n - 1) - 1  # 3n-2
    n_fft = 1 << (n_conv - 1).bit_length()
    A = fft(a, n_fft)
    W = fft(w_rev, n_fft)
    conv_result = ifft(A * W).real[:n_conv]
    
    # g[i] = conv_result[2n-2-i] for i=0..n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    g = conv_result[n-1:2*n-1][::-1] * 2.0
    
    return g, b_max


def _mirror_step(a, g, m, v, lr, t, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-style mirror descent step."""
    m = beta1 * m + (1.0 - beta1) * g
    v = beta2 * v + (1.0 - beta2) * g * g
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a * np.exp(-step)
    a_new = np.maximum(a_new, 1e-10)
    a_new = np.minimum(a_new, 1000.0)
    return a_new, m, v


def _local_refine(a, b, n_peaks=5, n_rounds=200):
    """Local coordinate descent on elements contributing most to peaks."""
    n = len(a)
    a = a.copy()
    b = b.copy()
    S = float(np.sum(a))
    target_sum = S
    
    for rnd in range(n_rounds):
        # Find peak
        k_max = int(np.argmax(b))
        max_val = b[k_max]
        
        # Find top contributing pairs to b[k_max]
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        if hi < lo:
            break
        indices = np.arange(lo, hi + 1)
        products = a[indices] * a[k_max - indices]
        
        # Get top contributors
        n_top = min(n_peaks, len(indices))
        top_local = np.argsort(products)[-n_top:]
        top_indices = indices[top_local]
        
        # Try reducing the larger element of each top pair
        improved = False
        for idx in top_indices:
            partner = k_max - idx
            if partner < 0 or partner >= n:
                continue
            # Try reducing a[idx] slightly
            factor_candidates = [0.85, 0.7, 0.9]
            for factor in factor_candidates:
                a_trial = a.copy()
                a_trial[idx] *= factor
                a_trial = np.maximum(a_trial, 0.0)
                S_t = float(np.sum(a_trial))
                if S_t < 0.01:
                    continue
                # Quick score check (recompute only affected b values)
                b_new = _fft_conv(a_trial, a_trial)
                score_new = 2.0 * n * float(np.max(b_new)) / (S_t * S_t)
                score_cur = 2.0 * n * float(np.max(b)) / (S * S)
                if score_new < score_cur - 1e-10:
                    a = a_trial
                    b = b_new
                    S = S_t
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    a = _normalize(a, target_sum)
    return a


def _run_mirror_descent(a0, deadline, lr=0.5, alpha=30.0, top_k_frac=0.05,
                        eval_interval=200, stuck_patience=30, name=""):

    a = a0.copy().astype(np.float64)

    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = _score_fast(a)
    stuck = 0
    lr_cur = lr
    
    while time.monotonic() < deadline:
        b = _fft_conv(a, a)
        b_max = float(np.max(b))
        S = float(np.sum(a))
        if S < 0.01:
            break
        
        g, _ = _compute_subgrad(b, a, alpha=alpha, top_k_frac=top_k_frac)
        
        # Subtract mean to handle the 1/S^2 normalization term
        g = g - np.mean(g)
        
        t += 1
        a, m, v = _mirror_step(a, g, m, v, lr_cur, t)
        a = _normalize(a, target_sum)
        
        # Adaptive learning rate
        lr_cur = max(lr * 0.99995, 0.01)
        
        if t % eval_interval == 0:
            sc = _score_fast(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1

                a = best_a + noise_scale * np.random.standard_normal(n)

                a = _normalize(a, target_sum)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr_cur = lr * 0.7
                stuck = 0
    
    return best_a, best_score


def _structured_init(n, rng, variant=0):
    """Generate structured initializations."""
    k = np.arange(n, dtype=np.float64)
    t = k / n
    
    if variant == 0:
        return np.ones(n)
    elif variant == 1:
        return 1.0 + 0.5 * np.cos(2 * np.pi * 3 * t)
    elif variant == 2:
        return 1.0 + 0.4 * np.sin(2 * np.pi * 7 * t) * np.cos(2 * np.pi * 2 * t)
    elif variant == 3:
        # Bimodal
        return np.exp(-8 * (t - 0.25)**2) + np.exp(-8 * (t - 0.75)**2) + 0.2
    elif variant == 4:
        # Chirp-like (phase varies quadratically, amplitude)
        return 1.0 + 0.3 * np.cos(2 * np.pi * (5 * t + 10 * t**2))
    elif variant == 5:
        # Random with smoothing
        raw = rng.random(n)
        # Smooth
        kernel = np.ones(10) / 10.0
        smooth = np.convolve(raw, kernel, mode='same')
        return 0.5 + smooth
    elif variant == 6:
        # Power law
        return 1.0 / (1.0 + 10 * np.abs(t - 0.5)**1.5) + 0.1
    elif variant == 7:
        # Multi-frequency
