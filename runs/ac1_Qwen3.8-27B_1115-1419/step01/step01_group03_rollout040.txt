```python
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def grad_smoothed(a, tau):

    Compute gradient of the log-sum-exp smoothed objective w.r.t. a.
```

```python
    # (w * a_flipped)[k] = sum_m w[m+k-n+1] * a[m]
    # We want sum_m w[i+m] * a[m], so set m+k-n+1 = i+m, i.e., k-n+1 = i, i.e., k = i+n-1
    # So result[i] = (w * a_flipped)[i+n-1]
    
    a_flipped = a[::-1].copy()
    # Compute convolution using FFT
    N = 1
    while N < len(w) + len(a_flipped) - 1:
        N *= 2
    W = np.fft.rfft(w, N)
    Af = np.fft.rfft(a_flipped, N)
    conv = np.fft.irfft(W * Af, N)
    
    # result[i] = conv[i + n - 1] for i = 0..n-1
    return conv[n-1:2*n-1]
```

Let me verify: `conv[k] = sum_j w[j] * a_flipped[k-j] = sum_j w[j] * a[n-1-(k-j)] = sum_j w[j] * a[n-1-k+j]`

For `k = i+n-1`: `conv[i+n-1] = sum_j w[j] * a[n-1-(i+n-1)+j] = sum_j w[j] * a[j-i]`

But we want `sum_m w[i+m] * a[m]`. Let's substitute `j = i+m`: `sum_m w[i+m] * a[m]`. ✓ (since `a[j-i] = a[i+m-i] = a[m]` when `j = i+m`)

Wait, I need to be careful: `sum_j w[j] * a[j-i]` where `j` ranges over valid indices. If `j < i`, then `j-i < 0` and `a[j-i] = 0` (since `a` is only defined for indices 0 to n-1). If `j >= i+n`, then `w[j] = 0` (since `w` has length 2n-1 and max valid index is 2n-2).

So effectively, `j` ranges from `i` to `i+n-1` (intersected with `[0, 2n-2]`). Since `i < n` and `i+n-1 < 2n-1`, the range is exactly `[i, i+n-1]`.

Letting `m = j - i`: `sum_{m=0}^{n-1} w[i+m] * a[m]`. ✓

Great, so the formula is correct.

Now, let me put it all together. I'll use the FFT-based cross-correlation for the gradient.

Actually, for n=4806, the FFT size would be the next power of 2 >= (2n-1) + n - 1 = 3n-2 ≈ 14416, so N=16384. The FFT of size 16384 takes about 0.1ms. Actually, let me think about one more approach: **proximal gradient with the indicator of the constraint set**.

The problem is: minimize $f(a) = 2n \cdot \tau \cdot \text{logsumexp}(c(a)/\tau)$ subject to $a \in \Delta$ (probability simplex).

The proximal gradient method (FISTA) with the simplex projection is what the previous code did, but with the wrong gradient. With the correct gradient, it should work much better.

The optimal step size for gradient descent on the log-sum-exp objective is $1/L$ where $L$ is the Lipschitz constant of the gradient.

The gradient is $2 \cdot \text{crosscorr}(w, a)$ where $w$ is the softmax weights.
The Lipschitz constant of the gradient (i.e., the spectral norm of the Hessian) is bounded by:
$L \leq \frac{2}{\tau} \cdot \max_i \sum_k |w_k| \cdot |\partial c_k / \partial a_i| \cdot ...$

This is hard to bound tightly. Let me just use a backtracking line search.

```python
"""
Minimize 2n*max(a*a)/(sum(a))^2 using L-BFGS with log-sum-exp smoothing
and softmax parameterization. Key fix: correct gradient (factor of 2 in
cross-correlation). Uses annealing of smoothing parameter, multiple
restarts, and FFT-based gradient for O(n log n) per evaluation.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def stable_softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return e / s


def crosscorr_valid(w, a):
    """
    Compute result[i] = sum_{m=0}^{n-1} w[i+m] * a[m] for i = 0..n-1
    where w has length 2n-1 and a has length n. Uses FFT.
    """
    n = len(a)
    a_flipped = a[::-1].copy()
    N = 1
    while N < len(w) + len(a_flipped) - 1:
        N *= 2
    W = np.fft.rfft(w, N)
    Af = np.fft.rfft(a_flipped, N)
    conv = np.fft.irfft(W * Af, N)
    return conv[n - 1:2 * n - 1]


def grad_smoothed_wrt_x(x, tau, n):
    """
    Compute objective and gradient w.r.t. x where a = softmax(x).
    Returns (obj, grad_x).
    """
    a = stable_softmax(x)
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    sum_exp = exp_c.sum()
    lse = c_max + np.log(sum_exp)
    obj = 2.0 * n * tau * lse

    weights = exp_c / sum_exp
    # Correct gradient: 2 * crosscorrelation(weights, a)
    grad_a = 2.0 * crosscorr_valid(weights, a)

    # Chain rule through softmax: grad_x = a * (grad_a - sum(a * grad_a))
    mean_g = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_g)
    return obj, grad_x


def run_lbfgs_phase(a0, n, tau, max_iter=5000):
    """Run L-BFGS on the softmax-parameterized smoothed problem."""
    x0 = np.log(np.clip(a0, 1e-15, None))

    def func(x):
        return grad_smoothed_wrt_x(x, tau, n)

    result = minimize(
        func, x0, method='L-BFGS-B', jac=True,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                 'maxls': 40, 'maxfun': 50000, 'iprint': 0}
    )

    a_opt = stable_softmax(result.x)
    a_opt = np.clip(a_opt, 0, None)
    s = a_opt.sum()
    if s > 1e-10:
        a_opt /= s
    else:
        a_opt = a0.copy()
    return a_opt, result


def fista_corrected(a0, n, tau, duration, top_k=5):
    """
    Corrected FISTA on the simplex with proper gradient and backtracking.
    Uses top-k active constraint averaging.
    """
    a = a0.copy()
    a_prev = a0.copy()
    t_mom = 1.0
    step = 1.0 / (4.0 / tau)  # initial step estimate
    best_a = a0.copy()
    best_score = fast_score(a0)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0

    while time.monotonic() < deadline:
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_mom
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy

        c_y = np.convolve(y, y)
        c_max = c_y.max()

        # Top-k weights
        if top_k >= c_y.size:
            top_idx = np.arange(c_y.size)
        else:
            top_idx = np.argpartition(c_y, -top_k)[-top_k:]

        # Use softmax weights over top-k for smooth gradient
        c_top = c_y[top_idx]
        w_top = np.exp((c_top - c_top.max()) / (tau * 0.5))
        w_top /= w_top.sum()

        # Gradient = 2 * crosscorr of weighted top-k
        weights = np.zeros(2 * n - 1)
        for j, idx in enumerate(top_idx):
            weights[int(idx)] = w_top[j]

        g = 2.0 * crosscorr_valid(weights, y)

        # Backtracking line search
        a_new = y - step * g
        a_new = project_simplex(a_new)

        # Check if step is good (simple: just accept if score doesn't increase too much)
        if it % 50 == 0:
            sc = fast_score(a_new)
            if sc > best_score * 1.01 and step > 0.001:
                step *= 0.5
                a_new = y - step * g
                a_new = project_simplex(a_new)

        a_prev = a
        a = a_new
        it += 1

        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck == 30:
                step *= 0.8
            elif stuck == 100:
                step *= 0.5
                a_prev = best_a.copy()
                t_mom = 1.0
            elif stuck > 300:
                a = best_a + np.random.normal(0, 0.005 / np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-10:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_mom = 1.0
                step = 1.0 / (4.0 / tau)
                stuck = 0

    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```
