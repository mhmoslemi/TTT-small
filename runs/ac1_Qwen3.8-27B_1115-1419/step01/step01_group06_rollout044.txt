```python
LP-based alternating minimization for flat autocorrelation sequences.
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using exact inner LP solves.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def solve_inner_lp(f, n):
    """Solve: min t s.t. (f*b)[k] <= t for all k=0..2n-2, b>=0, sum(b)=1.
    Returns opt b and optimal t."""
    # Build A_ub sparsely
    hop = 2 * n - 1
    # A_ub shape: (hop, n+1), last column is for t
    row_idx = []
    col_idx = []
    vals = []
    
    for k in range(hop):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            if f[j] != 0.0:
                row_idx.append(k)
                col_idx.append(j)
                vals.append(f[j])
        row_idx.append(k)
        col_idx.append(n)  # t column
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(hop, n + 1))
    b_ub = np.zeros(hop)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None, None
    
    b_opt = result.x[:n]
    t_opt = result.x[n]
    return b_opt, t_opt
```

```python
def compute_score_exact(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)

        return float('inf')
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return float('inf')
    b = np.convolve(a, a)
    mx = float(np.max(b))
    return 2.0 * n * mx / (s * s)

def build_lp_sparse(n):
    """Build sparsity pattern for the convolution LP."""
    hop = 2 * n - 1
    # For each lag k, the non-zero entries in column j are f[j] if 0<=k-j<n
    # Plus the t column (index n) with value -1
    row_idx = []
    col_idx = []
    for k in range(hop):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        cnt = j_hi - j_lo + 1
        row_idx.extend([k] * (cnt + 1))
        col_idx.extend(list(range(j_lo, j_hi + 1)) + [n])
    return np.array(row_idx, dtype=np.int32), np.array(col_idx, dtype=np.int32)

def solve_lp(f, row_idx, col_idx, n):
    """Solve inner LP: min t s.t. F*b - t*1 <= 0, 1^T*b = 1, b >= 0, t >= 0."""
    hop = 2 * n - 1

                  b_eq=np.array([1.0]),
```

```python
def build_lp_template(n):
    """Build the CSR matrix template. Returns (data_template, A_structure)."""
    hop = 2 * n - 1
    row_idx = []
    col_idx = []
    for k in range(hop):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        cnt = j_hi - j_lo + 1
        row_idx.extend([k] * (cnt + 1))
        col_idx.extend(list(range(j_lo, j_hi + 1)) + [n])
    
    nnz = len(row_idx)
    row_idx = np.array(row_idx, dtype=np.int32)
    col_idx = np.array(col_idx, dtype=np.int32)
    
    # Build CSR directly
    # For CSR, we need indptr (row pointers) and indices (column indices)
    # Since rows are in order (0, 0, ..., 1, 1, ..., 2, 2, ...), we can build indptr easily
    indptr = np.zeros(hop + 1, dtype=np.int32)
    for k in range(hop):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        indptr[k + 1] = indptr[k] + (j_hi - j_lo + 1 + 1)
    
    indices = np.array(col_idx, dtype=np.int32)
    
    return hop, indptr, indices, n

def create_lp_matrix(f, hop, indptr, indices, n):
    """Create the A_ub CSR matrix for the LP given f."""
    b_mask = indices < n
    data = np.empty(len(indices), dtype=np.float64)
    data[b_mask] = f[indices[b_mask]]
    data[~b_mask] = -1.0
    A = csr_matrix((data, indices, indptr), shape=(hop, n + 1))
    return A
```

```python
"""
LP-based alternating minimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses exact inner LP solves via HiGHS with optimized sparse matrix construction,
running-average tracking, and multiple restarts at different sequence lengths.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_score_exact(a):
    """Compute the objective: 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return float('inf')
    b = np.convolve(a, a)
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float('inf')
    return 2.0 * n * mx / (s * s)


def build_lp_template(n):
    """Pre-build CSR structure for the inner LP. Returns (hop, indptr, indices)."""
    hop = 2 * n - 1
    indptr = np.zeros(hop + 1, dtype=np.int64)
    cols = []
    for k in range(hop):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        cols.extend(range(j_lo, j_hi + 1))
        cols.append(n)  # t column
        indptr[k + 1] = len(cols)
    indices = np.array(cols, dtype=np.int64)
    return hop, indptr, indices


def solve_inner_lp(f, hop, indptr, indices, n):
    """Solve: min t s.t. (f*b)[k] <= t, sum(b)=1, b>=0, t>=0."""
    b_mask = indices < n
    data = np.empty(len(indices), dtype=np.float64)
    data[b_mask] = f[indices[b_mask]]
    data[~b_mask] = -1.0
    A = csr_matrix((data, indices, indptr), shape=(hop, n + 1))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    a_eq = np.zeros((1, n + 1))
    a_eq[0, :n] = 1.0
    bounds = [(0.0, None)] * (n + 1)
    res = linprog(c_obj, A_ub=A, b_ub=np.zeros(hop),
                  A_eq=a_eq, b_eq=np.array([1.0]),
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 15.0})
    if not res.success or res.x is None:
        return None
    return np.maximum(res.x[:n], 0.0)


def lp_alternating(f_init, n, budget_s, hop, indptr, indices, label=""):
    """Run LP-based alternating minimization with running average."""
    t0 = time.monotonic()
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-10:
        f = np.ones(n) / n
        s = 1.0
    f = f / s

    best_f = f.copy()
    best_score = compute_score_exact(f)
    print(f"  [{label}] Start: {best_score:.6f}")

    avg_f = f.copy()
    avg_count = np.float64(1.0)
    avg_best_f = f.copy()
    avg_best_score = best_score

    alpha = 1.0
    stall = 0
    it = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < budget_s:
        it += 1
        b_opt = solve_inner_lp(f, hop, indptr, indices, n)
        if b_opt is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            continue
        lp_fail_count = 0

        s_b = float(b_opt.sum())
        if s_b < 1e-10:
            b_opt = np.ones(n) / n
        else:
            b_opt = b_opt / s_b

        f_new = (1.0 - alpha) * f + alpha * b_opt
        f_new = np.maximum(f_new, 0.0)
        s_new = float(f_new.sum())
        if s_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s_new

        avg_f = (avg_f * avg_count + f_new) / (avg_count + 1.0)
        avg_count += 1.0

        score = compute_score_exact(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            stall = 0
            if it % 100 == 1 or best_score < 1.5:
                print(f"  [{label}] it={it}: best={score:.6f} alpha={alpha:.3f} t={time.monotonic()-t0:.0f}s")
        else:
            stall += 1

        if it % 50 == 0:
            avg_score = compute_score_exact(avg_f)
            if avg_score < avg_best_score:
                avg_best_score = avg_score
                avg_best_f = avg_f.copy()

        if stall >= 30:
            alpha = max(0.05, alpha * 0.75)
            stall = 0
        elif stall == 0 and alpha < 1.0:
            alpha = min(1.0, alpha * 1.03)

        f = f_new

    # Final avg check
    avg_score = compute_score_exact(avg_f)
    if avg_score < avg_best_score:
        avg_best_score = avg_score
        avg_best_f = avg_f.copy()

    print(f"  [{label}] Done: it={it} best={best_score:.6f} avg={avg_best_score:.6f} time={time.monotonic()-t0:.0f}s")

    if avg_best_score < best_score:
        return avg_best_f, avg_best_score
    return best_f, best_score


def finalize_sequence(a):
    """Scale sequence to sum ~sqrt(2n) for the evaluator."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        a = np.ones(64)
        n = 64
    s = float(a.sum())
    scale = float(np.sqrt(2.0 * n))
    if s > 1e-15:
        a = a * scale / s
    else:
        a = np.ones(n) * scale / n
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.01 / max(s2, 1e-30))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def get_parent():
    """Load parent sequence if available."""
    obj = globals().get('height_sequence_1', None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        if arr.size < 8 or arr.sum() < 1e-10:
            return None
        return arr
    except Exception:
        return None

    for n in n_list:

        if n not in templates:

    # Phase allocation: give more time to larger n
    weights = {}

        weights[n] = w
        total_w += w
    phase_budgets = {}
    for n in n_list:
        phase_budgets[n] = total_budget * weights[n] / total_w * 0.55

    # Starting points per n
    for n in n_list:
        if time.monotonic() - t_global > total_budget - 30:
            break
        remaining = total_budget - (time.monotonic() - t_global)
        pb = min(phase_budgets[n], remaining * 0.7)
        if pb < 30:
            break

        hop, indptr, indices = templates[n]
        starts = []

        if parent is not None:

                f0 = parent[:n].astype(np.float64).copy()

                reps = n // parent.size + 1
                f0[:] = np.tile(parent, reps)[:n]
            f0 = np.maximum(f0, 0.0)
            if f0.sum() > 1e-10:
                starts.append(f0 / f0.sum())

        # Uniform
        starts.append(np.ones(n) / n)

        # Cosine-modulated
        j = np.arange(n, dtype=np.float64)

            a_cos = np.maximum(a_cos, 0.01)
            starts.append(a_cos / a_cos.sum())

        # Number of starting points to try

        for si, f_init in enumerate(starts):

    else:

    print(f"Total time: {total_time:.1f}s")
```
