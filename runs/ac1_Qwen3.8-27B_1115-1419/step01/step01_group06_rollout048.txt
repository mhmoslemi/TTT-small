def _next_pow2(m):
    if m <= 1:

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

def _fft_self_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)

def _obj_fft(a, N):
    n = len(a)
    c = _fft_self_conv(a, N)

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.

Algorithm combines:
- Smoothed-Linf gradient descent in log-space with Adam optimizer
- Hot-lag surgical corrections targeting worst convolution lags
- Multi-length exploration with diverse structured starts
- Adaptive beta (smoothing) schedule
- Periodic restarts with diversity
- Frequency-domain iterative refinement (Gerchberg-Saxton style)

import numpy as np
import time


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _fft_self_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * len(a) - 1]


def _obj_fft(a, N):
    n = len(a)
    c = _fft_self_conv(a, N)
    c = np.maximum(c, 0.0)
    return 2.0 * n * float(np.max(c))


def _exact_obj(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    return float(2.0 * len(a) * float(np.max(np.convolve(a, a))) / (s * s))


def _softmax_w(c, beta):
    cmax = float(np.max(c))
    if cmax <= 0 or not np.isfinite(cmax):
        w = np.zeros_like(c)
        w[0] = 1.0
        return w
    x = beta * (c - cmax)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _log_weights_from_a(a):
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w -= w.max()
    return w


def _a_from_log_weights(w):
    w = np.asarray(w, dtype=np.float64)
    w = np.where(np.isfinite(w), w, -100.0)
    w -= w.max()
    e = np.exp(w)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(w) / len(w)
    return e / s


def _adam_step(grad_w, w, m, v, t, lr):
    t += 1
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1.0 - b1) * grad_w
    v = b2 * v + (1.0 - b2) * grad_w * grad_w
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    dw = m_hat / (np.sqrt(v_hat) + eps)
    dw = np.clip(dw, -5.0, 5.0)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    w_new = w - lr * dw
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    a_new = _a_from_log_weights(w_new)
    return a_new, w_new, m, v, t


def _compute_grad_log(a, N, beta, phase, A):
    n = len(a)
    c = _fft_self_conv(a, N)
    c = np.maximum(c, 0.0)
    w = _softmax_w(c, beta)
    W = np.fft.rfft(w, N)
    FR = phase * np.conjugate(A)
    conv = np.fft.irfft(W * FR, N)
    grad_a = 2.0 * conv[n - 1:2 * n - 1]
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)
    grad_w = a * (grad_a - float(np.dot(a, grad_a)))
    return grad_w, c


def _hot_lag_correction(a, c, n_top=5, reduce_factor=0.97):
    """Identify top hot lags and reduce their top contributors."""
    a = a.copy()
    n = len(a)
    # Find top K max lags
    top_k = np.argsort(c)[-n_top:][::-1]
    for k_max in top_k:
        # Find top contributors to c[k_max]
        products = []
        for j in range(n):
            k2 = k_max - j
            if 0 <= k2 < n:
                products.append((a[j] * a[k2], j, k2))
        products.sort(reverse=True)
        # Reduce top few contributors
        count = 0
        for prod, j, k2 in products:
            if count >= 3:
                break
            if prod < 1e-15:
                break
            a[j] *= reduce_factor
            if k2 != j:
                a[k2] *= reduce_factor
            count += 1
    return _normalize(a)


def _freq_refine(a, n_iters=50, alpha=0.5):
    """
    Frequency-domain iterative refinement (Gerchberg-Saxton style).
    Tries to make |A|^2 more concentrated at DC while keeping a >= 0.
    """
    n = len(a)
    N = _next_pow2(3 * n - 2)

    for _ in range(n_iters):
        A = np.fft.rfft(a, N)
        # Compute current convolution
        c = _fft_self_conv(a, N)
        c = np.maximum(c, 0.0)
        cmax = float(np.max(c))
        if cmax <= 0:
            break
        # Target: make c as flat as possible
        # In frequency: want |A|^2 to be proportional to FT of flat sequence
        # FT of flat sequence of length 2n-1 is a Dirichlet kernel
        # Approximate: want |A[m]|^2 to be more uniform (less peaked)
        # Actually we want c flat, which means C[m] should be a Dirichlet kernel
        # C[m] = A[m]^2, so we want A[m]^2 to look like a Dirichlet kernel
        # Simpler: just try to make |A| more uniform across frequencies
        mag = np.abs(A)
        mag_mean = float(mag.mean())
        if mag_mean > 0:
            # Blend current magnitude with uniform magnitude
            new_mag = (1 - alpha) * mag + alpha * mag_mean
            # Preserve phase
            phase = A / (mag + 1e-300)
            A_new = new_mag * phase
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 0:
                a = a_new / s
            else:
                break
    return a

        a_sota = np.ones(main_n) / main_n

    # === Phase 1: Primary optimization from SOTA (45% time) ===
    p1_budget = 0.45 * total_budget
    p1_deadline = t0 + p1_budget
    print(f"\n--- Phase 1: Primary optimization (budget={p1_budget:.0f}s) ---")

    if a_sota is not None and time.monotonic() < p1_deadline - 30:
        a_opt, obj = _optimize(
            a_sota, p1_budget - 10, seed=42,
            beta_init=15.0, lr_init=0.08,
            verbose=True, use_hot_lag=True, use_freq=False
        )
        exact = _exact_obj(a_opt)
        print(f"  Phase 1 result: fft_obj={obj:.6f}, exact_obj={exact:.6f}")
        results.append((exact, a_opt.copy()))

    # === Phase 2: Fine-tuning with higher beta (20% time) ===
    p2_budget = 0.20 * total_budget
    if results and time.monotonic() < deadline - 60:
        results.sort(key=lambda x: x[0])
        _, best_seq = results[0]
        print(f"\n--- Phase 2: Fine-tuning (budget={p2_budget:.0f}s) ---")
        a_fine, obj = _optimize(
            best_seq, p2_budget - 5, seed=99,
            beta_init=40.0, lr_init=0.02,
            verbose=True, use_hot_lag=True, use_freq=False
        )
        exact = _exact_obj(a_fine)
        print(f"  Phase 2 result: exact_obj={exact:.6f}")
        results.append((exact, a_fine.copy()))

    # === Phase 3: Try alternative lengths (20% time) ===
    p3_budget = 0.20 * total_budget
    remaining = deadline - time.monotonic()
    p3_budget = min(p3_budget, remaining)

    if p3_budget > 100:
        print(f"\n--- Phase 3: Alternative lengths (budget={p3_budget:.0f}s) ---")
        alt_lengths = [4096, 8192, 2048, 6144]
        # Remove lengths close to main_n
        tried_n = [len(r[1]) for r in results]
        alt_lengths = [l for l in alt_lengths
                       if all(abs(l - tl) > l * 0.3 for tl in tried_n)]
        alt_lengths = alt_lengths[:3]

        if alt_lengths:
            per_len = p3_budget / len(alt_lengths)
            for length in alt_lengths:
                if time.monotonic() >= deadline - 60:
                    break
                print(f"  Trying n={length}, budget={per_len:.0f}s")
                best_local = None
                best_local_obj = float('inf')
                per_start = per_len / 3
                for kind in ['sine', 'gaussian', 'raised_cos']:
                    if time.monotonic() >= deadline - 30:
                        break
                    a_init = _structured_start(length, kind)
                    a_init = a_init / a_init.sum()
                    a_try, obj_try = _optimize(
                        a_init, per_start, seed=777 + length,
                        beta_init=10.0, lr_init=0.1,
                        verbose=False, use_hot_lag=True, use_freq=False
                    )
                    if obj_try < best_local_obj:
                        best_local_obj = obj_try
                        best_local = a_try
                if best_local is not None:
                    exact = _exact_obj(best_local)
                    print(f"    n={length}: exact_obj={exact:.6f}")
                    results.append((exact, best_local.copy()))

    # === Phase 4: Final polish (15% time) ===
    remaining = deadline - time.monotonic()
    if results and remaining > 40:
        results.sort(key=lambda x: x[0])
        _, final_best = results[0]
        p4_budget = min(0.15 * total_budget, remaining - 15)
        print(f"\n--- Phase 4: Final polish (budget={p4_budget:.0f}s) ---")

        a_pol, obj_pol = _optimize(
            final_best, p4_budget - 5, seed=2024,
            beta_init=60.0, lr_init=0.015,
            verbose=True, use_hot_lag=True, use_freq=True
        )
        exact = _exact_obj(a_pol)
        print(f"  Phase 4 result: exact_obj={exact:.6f}")
        if exact < results[0][0]:
            results.insert(0, (exact, a_pol.copy()))
        else:
            results.append((exact, a_pol.copy()))

    # === Phase 5: Hot-lag burst on best (remaining time) ===
    remaining = deadline - time.monotonic()
    if results and remaining > 20:
        results.sort(key=lambda x: x[0])
        _, burst_seq = results[0]
        burst_budget = remaining - 10
        print(f"\n--- Phase 5: Hot-lag burst (budget={burst_budget:.0f}s) ---")
        a_burst, obj_burst = _optimize(
            burst_seq, burst_budget, seed=555,
            beta_init=80.0, lr_init=0.01,
            verbose=False, use_hot_lag=True, use_freq=False
        )
        exact = _exact_obj(a_burst)
        print(f"  Phase 5 result: exact_obj={exact:.6f}")
        if exact < results[0][0]:
            results.insert(0, (exact, a_burst.copy()))
        else:
            results.append((exact, a_burst.copy()))

    # === Select best and finalize ===
    if not results:
        print("WARNING: No valid results, using fallback")
        best_seq = np.ones(4096)
    else:
        best_seq = min(results, key=lambda x: x[0])[1]

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.maximum(best_seq, 0.0)
    n = len(best_seq)
    s = float(best_seq.sum())

    # Scale to sum sqrt(2n) for the evaluator
    if s > 0 and np.isfinite(s):
        scale = float(np.sqrt(2.0 * n)) / s
        best_seq = best_seq * scale
    else:
        best_seq = np.ones(n) * (np.sqrt(2.0 * n) / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Ensure sum > 0.01
    final_sum = float(best_seq.sum())
    if final_sum < 0.01:
        best_seq = best_seq * (0.01 / max(final_sum, 1e-300))
        best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact check
    final_obj = _exact_obj(best_seq)
    print(f"\n=== FINAL: n={n}, exact_obj={final_obj:.6f} ===")
    print(f"Total time: {time.monotonic() - t0:.1f}s")

    return [float(x) for x in best_seq]
