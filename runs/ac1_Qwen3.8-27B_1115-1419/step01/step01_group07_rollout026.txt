```python
def lp_direction(f, M, solver='ECOS'):
    """Solve the LP to find the best direction.
    max sum(b) s.t. b >= 0, (f*b)_k <= M for all k
    """
    import cvxpy as cp
    n = len(f)
    b = cp.Variable(n)
    constraints = [b >= 0]
    # (f*b)_k = sum_i f_i * b_{k-i} for valid i, k-i
    # We need this for k = 0, ..., 2n-2
    # (f*b)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f_i * b_{k-i}
    # Let's build the constraint matrix A where (Ab)_k = (f*b)_k
    # A is (2n-1) x n
    # A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[k-j] if k-j >= 0 and k-j < n
    
    # Actually, (f*b)_k = sum_j f[k-j] * b_j where f[m] = 0 if m not in [0,n-1]
    # So A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    
    # For efficiency, let's build A as a sparse matrix
    from scipy.sparse import sparse

    # A[k, j] = f[k-j] for max(0, k-n+1) <= j <= min(n-1, k)
    # For each j, A[k, j] = f[k-j] for max(0, j) <= k <= min(2n-2, j+n-1)
    # i.e., for each j, the non-zero entries in column j are at rows k = j, j+1, ..., j+n-1
    # (clamped to [0, 2n-2])
    
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(max(0, j), min(2*n-1, j+n)):
            idx = k - j  # This is the index into f
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    constraints.append(A @ b <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    if prob.status in ['optimal', 'optimal_inaccurate']:
        return b.value
    return None
```

"""
Optimize step function to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses LP-based direction finding (solving max sum(b) s.t. (f*b)_k <= M),
FISTA with subgradient, adaptive truncation, and multi-size exploration.
Different from previous L-BFGS + LSE approach.

import numpy as np
import time
from scipy.sparse import csr_matrix as sparse_csr
from scipy.optimize import minimize


def simplex_project(v):
    """Project vector onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_score(a):
    """Compute 2*n*max(a*a)/(sum(a))^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def build_convolve_matrix(n, f):
    """Build sparse matrix A where (A @ b)_k = (f*b)_k for k=0..2n-2."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2 * n - 1, j + n)
        for k in range(k_start, k_end):
            idx = k - j
            if 0 <= idx < n and f[idx] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    A = sparse_csr((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k.
    Returns optimal b or None if infeasible."""
    import cvxpy as cp
    n = len(f)
    A = build_convolve_matrix(n, f)
    b = cp.Variable(n)
    constraints = [b >= 0, A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    # Try solvers in order of speed for this problem size
    for solver in ['MOSEK', 'ECOS', 'SCS', 'GLPK']:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if b.value is not None:
                    return np.maximum(b.value, 0)
        except Exception:
            continue
    return None


def subgradient_step(a, n_active=100, step_size=0.01):
    """One subgradient step with projection onto simplex."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    if max_c < 1e-15:
        return a
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(active)
    a_new = a - step_size * grad
    a_new = simplex_project(a_new)
    return a_new


def fista_optimize(a, max_time=100, n_subgrad=50):
    """FISTA optimization with decreasing subgradient step size."""
    n = len(a)
    a = a.copy()
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    deadline = time.monotonic() + max_time
    
    step = 1.0 / n
    t = 1.0
    y = a.copy()
    momentum = 0.0
    
    while time.monotonic() < deadline:
        # Compute subgradient at y
        c = np.convolve(y, y)
        max_c = c.max()
        if max_c < 1e-15:
            break
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:min(n_subgrad, len(c))]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * y[k - hi:k - lo + 1][::-1]
        grad /= len(active)
        
        # Gradient step
        z = y - step * grad
        z = simplex_project(z)
        
        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = z + ((t - 1.0) / t_new) * (z - a)
        y = simplex_project(y)
        a = z
        t = t_new
        
        # Evaluate
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decrease step size
        step *= 0.999
        if step < 1e-6:
            step = 1e-6
    
    return best_a, best_score


def truncate_sequence(a, min_score_improvement=0.001):
    """Remove small edge elements to reduce n without increasing peak much."""
    a = a.copy()
    n = len(a)
    improved = True
    while improved and n > 16:
        improved = False
        old_score = compute_score(a)
        # Try removing from both ends
        for trim in range(1, min(n // 4, 20)):
            a_trim = a[trim:n - trim] if n - 2 * trim >= 16 else a[trim:n]
            if len(a_trim) < 16:
                continue
            a_trim = a_trim / a_trim.sum()
            new_score = compute_score(a_trim)
            if new_score < old_score - min_score_improvement:
                a = a_trim
                n = len(a)
                old_score = new_score
                improved = True
                break
    return a


def lp_optimize(a, max_time=600):
    """Main LP-based optimization loop."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = time.monotonic() + max_time
    iteration = 0
    
    while time.monotonic() < deadline - 5:
        iteration += 1
        # Compute current peak
        c = np.convolve(a, a)
        M = c.max()
        
        # Solve LP
        t0 = time.monotonic()
        b = solve_lp_direction(a, M)
        lp_time = time.monotonic() - t0
        
        if b is None or b.sum() < 1e-10:
            # LP failed, use subgradient step instead
            a = subgradient_step(a, n_active=100, step_size=1.0 / n)
            continue
        
        # Rescale b to have sum = 1
        b = b / b.sum()
        
        # Line search: try t = 1.0, 0.5, 0.25, 0.1
        best_t = 0.0
        best_trial_score = best_score
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08]:
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = compute_score(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = a.copy()
                if iteration % 5 == 0:
                    print(f"    LP iter {iteration}: score={best_score:.6f} "
                          f"(t={best_t:.2f}, lp_time={lp_time:.2f}s)")
        else:
            # LP direction didn't help, try subgradient
            a = subgradient_step(a, n_active=80, step_size=0.5 / n)
        
        # Periodic truncation
        if iteration % 20 == 0:
            a_trunc = truncate_sequence(a)
            if len(a_trunc) < len(a):
                sc_t = compute_score(a_trunc)
                if sc_t < best_score:
                    best_score = sc_t
                    best_a = a_trunc.copy()
                    a = a_trunc.copy()
                    n = len(a)
                    print(f"    Truncated to n={n}: score={best_score:.6f}")
                a = a / a.sum()
                n = len(a)

def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    x = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2
    inits.append(x / x.sum())
    
    # Gaussian window
    mid = (n - 1) / 2.0
    x = np.exp(-((np.arange(n) - mid) ** 2) / (2 * (n / 4) ** 2))
    inits.append(x / x.sum())
    
    # Linear ramp
    x = np.linspace(2.0, 0.3, n)
    inits.append(x / x.sum())
    
    # Exponential
    x = np.exp(-3 * np.arange(n) / n)
    inits.append(x / x.sum())
    
    # Two-peak
    x = np.zeros(n)
    x[n // 4] = 1.0
    x[3 * n // 4] = 1.0
    # Smooth around peaks
    for i in range(n):
        x[i] += 0.5 * np.exp(-((i - n // 4) ** 2) / (2 * (n // 16) ** 2))
        x[i] += 0.5 * np.exp(-((i - 3 * n // 4) ** 2) / (2 * (n // 16) ** 2))
    inits.append(x / x.sum())
    
    # Random Dirichlet (multiple)
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    # Sine-modulated
    for freq in [1, 2, 3]:
        x = 1.0 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        x = np.maximum(x, 0.01)
        inits.append(x / x.sum())
    
    # Random binary-like
    for _ in range(3):
        x = np.where(rng.random(n) > 0.5, 1.0, 0.1)
        inits.append(x / x.sum())
    
    return inits

    if len(best_a_global) < 2048 and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 20
        a_lp, score_lp = lp_optimize(best_a_global, max_time=remaining)
        if score_lp < best_score_global:
            best_score_global = score_lp
            best_a_global = a_lp.copy()
            print(f"  Final LP on truncated: n={len(a_lp)}, score={score_lp:.6f}")
    
    # Verify with evaluate_sequence
    final_score = evaluate_sequence(best_a_global.tolist())
    if np.isfinite(final_score) and final_score < best_score_global:
        best_score_global = final_score
    
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_a_global)}, score={best_score_global:.6f}")
    print(f"Time used: {time.monotonic()-start_time:.0f}s / {budget_s}s")
    print(f"{'='*50}")
    
    return [float(val) for val in best_a_global]
