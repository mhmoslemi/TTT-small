import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def solve_blocking_lp(a, M):
    """Solve: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k"""
    n = len(a)
    # Build constraint matrix sparsely
    # Row k: sum_j A[k,j] * b[j] <= M
    # A[k,j] = a[k-j] if 0 <= k-j < n and 0 <= j < n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    result = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(2*n-1, M),
        bounds=[(0, None)] * n,
        method='highs'
    )
    
    if result.success:
        return result.x
    else:
        return None

provides global direction changes to escape local plateaus.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_c_k(a, k, n):
    """Gradient of (a*a)_k w.r.t. a."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        idx = k - np.arange(lo, hi)
        grad[lo:hi] = a[idx]
    if k % 2 == 0:
        j = k // 2
        if 0 <= j < n:
            grad[j] += a[j]
    return grad


def top_k_gradient(a, top_k=5):
    """Compute averaged gradient from top-k convolution peaks."""
    n = len(a)
    c = np.convolve(a, a)
    if top_k >= c.size:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    g = np.zeros(n)
    for mk in top_idx:
        g += grad_c_k(a, int(mk), n)
    g /= float(len(top_idx))
    return g, c.max()


def solve_reduced_lp(a, M, num_constraints=200):
    """Solve reduced LP: max sum(b) s.t. b>=0, (a*b)_k <= M for top constraints."""
    n = len(a)
    c = np.convolve(a, a)
    # Get top constraints
    if num_constraints >= c.size:
        top_k = np.arange(c.size)
    else:
        top_k = np.argpartition(c, -num_constraints)[-num_constraints:]
    top_k = sorted(top_k.tolist())
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(top_k):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[i])
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_k), n))
    
    result = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(len(top_k), M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 5}
    )
    
    if result.success:
        return result.x
    return None


def run_hybrid_optimize(a0, n, duration, rng, verbose=True):
    """
    Hybrid FISTA + LP-jump optimization.
    """
    a = a0.copy()
    a = project_simplex(a)
    a_prev = a.copy()
    t_mom = 1.0
    step = 1.0 / np.sqrt(n)  # initial step size
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0
    lp_interval = 200  # solve LP every 200 iterations
    
    if verbose:
        print(f"  Starting: score={best_score:.8f}, step={step:.6f}")
    
    while time.monotonic() < deadline:
        # FISTA momentum
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_mom
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        elif abs(sy - 1.0) > 0.1:
            y /= sy
        
        # Top-k gradient
        top_k = 5 if it < 5000 else 10
        g, cmax = top_k_gradient(y, top_k)
        
        # Gradient step + project
        a_new = y - step * g
        a_new = project_simplex(a_new)
        
        # Check if improvement
        if it % 20 == 0:
            sc = fast_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                stuck = 0
            else:
                stuck += 1
        
        a_prev = a
        a = a_new
        it += 1
        
        # Adaptive step size
        if stuck == 30:
            step *= 0.9
        elif stuck == 100:
            step *= 0.8
            # Reset momentum
            a_prev = best_a.copy()
            t_mom = 1.0
        elif stuck == 300:
            # Try LP jump
            a_lp = solve_reduced_lp(best_a, cmax, num_constraints=200)
            if a_lp is not None:
                sa = a_lp.sum()
                if sa > 1e-10:
                    a_lp = a_lp / sa
                    # Blend
                    blend = 0.5
                    a_new = (1 - blend) * best_a + blend * a_lp
                    a_new = project_simplex(a_new)
                    sc = fast_score(a_new)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        a = a_new
                        a_prev = a
                        t_mom = 1.0
                        step = 1.0 / np.sqrt(n)
                        stuck = 0
                        if verbose:
                            print(f"    LP jump at it={it}: score={best_score:.8f}")
                    else:
                        # Try smaller blend
                        for b in [0.3, 0.2, 0.1]:
                            a_try = (1 - b) * best_a + b * a_lp
                            a_try = project_simplex(a_try)
                            sc_try = fast_score(a_try)
                            if sc_try < best_score:
                                best_score = sc_try
                                best_a = a_try.copy()
                                a = a_try
                                a_prev = a
                                t_mom = 1.0
                                step = 1.0 / np.sqrt(n)
                                stuck = 0
                                if verbose:
                                    print(f"    LP blend={b} at it={it}: score={best_score:.8f}")
                                break
            
            # If LP didn't help, do a random perturbation
            if stuck >= 300:
                a = best_a + rng.normal(0, 0.005 / np.sqrt(n), n)
                np.maximum(a, 0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                else:
                    a = best_a.copy()
                a_prev = a.copy()
                t_mom = 1.0
                step = 1.0 / np.sqrt(n)
                stuck = 0
                if verbose:
                    print(f"    Random restart at it={it}: score={best_score:.8f}")
        
        # Periodic reporting
        if verbose and it % 500 == 0:
            time_left = deadline - time.monotonic()
            print(f"    it={it}, score={best_score:.8f}, step={step:.6f}, stuck={stuck}, time_left={time_left:.0f}s")
    
    # Final check
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

"""
Minimize 2n*max(a*a)/(sum(a))^2 using FISTA with symmetry enforcement,
top-k gradient averaging, adaptive backtracking step size, and periodic
LP-based direction jumps. Key improvements over prior art:
1. Enforce symmetry a[i]=a[n-1-i] (free improvement, halves search space)
2. Backtracking line search for step size
3. LP-based direction changes using reduced constraint set
4. Weighted top-k gradient (weights by proximity to max)
5. Multiple diverse initializations with structured designs
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    return (a + a[::-1].copy()) / 2.0


def weighted_top_k_gradient(a, top_k=10, temperature=1.0):
    """
    Compute gradient as weighted average over top-k convolution peaks.
    Weights are softmax over the top-k values (higher weight for larger peaks).
    Returns (gradient, max_conv_value).
    """
    n = len(a)
    c = np.convolve(a, a)
    cmax = c.max()
    
    if top_k >= c.size:
        top_idx = np.arange(c.size)
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_idx]
        # Softmax weights
        shifted = top_vals - top_vals.max()
        weights = np.exp(shifted / temperature)
        weights /= weights.sum()
    
    g = np.zeros(n)
    # Vectorized gradient computation
    for mk, w in zip(top_idx, weights):
        k = int(mk)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            idx = k - np.arange(lo, hi)
            g[lo:hi] += 2.0 * w * a[idx]
    
    return g, cmax


def solve_reduced_lp(a, M, num_constraints=300):
    """
    Solve reduced LP: max sum(b) s.t. b>=0, (a*b)_k <= M for top constraints.
    Returns normalized direction or None.
    """
    n = len(a)
    c = np.convolve(a, a)
    
    if num_constraints >= c.size:
        top_k = np.arange(c.size)
    else:
        top_k = np.argpartition(c, -num_constraints)[-num_constraints:]
    top_k_sorted = np.sort(top_k)
    
    # Build sparse rows for constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    
    for idx, k in enumerate(top_k_sorted):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            is_ = k - js
            mask = (is_ >= 0) & (is_ < n)
            js = js[mask]
            is_ = is_[mask]
            rows_list.extend([idx] * len(js))
            cols_list.extend(js.tolist())
            vals_list.extend(a[is_].tolist())
    
    if not rows_list:
        return None
    
    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(len(top_k_sorted), n))
    
    try:
        result = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(len(top_k_sorted), M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': 10}
        )
        if result.success and result.x is not None:
            b = result.x
            b = np.maximum(b, 0)
            sb = b.sum()
            if sb > 1e-10:
                return b / sb
    except Exception:
        pass
    return None
