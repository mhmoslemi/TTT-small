        tau = init_tau
    if init_eta is None:
        eta = 0.06
    else:
        eta = init_eta

    best_a = a.copy()
    best_obj = _obj_hard(a, n, L)
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < it_cap:
        c = _fft_conv_self(a, L)
        mc = float(np.max(c))
        obj_cur = 2.0 * mc / n

        if obj_cur < best_obj - 1e-12:
            best_obj = obj_cur
            best_a = a.copy()
            stall = 0
            if it % 10000 == 0:
                print(f"  [{tag}] it={it} obj={obj_cur:.6f} tau={tau:.1f} eta={eta:.4f}")
        else:
            stall += 1

        # Softmax with numerical stability
        cs = c - mc
        p = np.exp(cs / tau)
        ps = p.sum()
        if ps < 1e-300:
            p = np.ones_like(c) / len(c)
        else:
            p /= ps

        # FFT gradient
        g = _fft_grad_smooth(p, a, L)
        gmag = float(np.max(np.abs(g)))
        if gmag < 1e-14:
            break
        g /= gmag
        g -= g.mean()

        a = _project(a - eta * g, n)

        # Anneal: slowly cool, occasionally kick
        tau = max(tau * 0.99995, 0.3)
        if stall > 4000:
            eta = max(eta * 0.7, 1e-7)
            stall = 0
        if stall > 12000:
            # Reheating: jump temperature back up
            tau = n * 0.03
            eta = max(eta * 1.5, 0.02)
            stall = 0

        it += 1

    return best_a, best_obj


def _hard_run(a, n, L, deadline, tag, init_eta=None, K=8, it_cap=5000000):
    """Hard top-K active-set descent using FFT gradient."""
    if init_eta is None:
        eta = 0.04
    else:
        eta = init_eta

    best_a = a.copy()
    best_obj = _obj_hard(a, n, L)
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < it_cap:
        c = _fft_conv_self(a, L)
        mc = float(np.max(c))
        obj_cur = 2.0 * mc / n

        if obj_cur < best_obj - 1e-12:
            best_obj = obj_cur
            best_a = a.copy()
            stall = 0
            if it % 20000 == 0:
                print(f"  [{tag}] it={it} obj={obj_cur:.6f} eta={eta:.5f}")
        else:
            stall += 1

        # Build a "pseudodistribution" concentrated on top-K peaks
        order = np.argsort(c)[::-1]
        topK = order[:K]
        w = c[topK] - mc + 1.0  # shift so max gets weight 1
        w = np.exp(w / (mc * 0.005 + 1e-6))
        w /= w.sum()

        # Build sparse p: only top-K positions get weight
        p = np.zeros(len(c), dtype=np.float64)
        p[topK] = w

        g = _fft_grad_smooth(p, a, L)
        gmag = float(np.max(np.abs(g)))
        if gmag < 1e-14:
            break
        g /= gmag
        g -= g.mean()

        a = _project(a - eta * g, n)

        if stall > 6000:
            eta = max(eta * 0.85, 1e-8)
            stall = 0
        if it % 30000 == 29999:
            eta = max(eta * 0.93, 1e-9)

        it += 1

    return best_a, best_obj


def _perturb_lag_target(a, n, L, rng, alpha):
    """Perturb elements that contribute most to the max lag."""
    c = _fft_conv_self(a, L)
    k_max = int(np.argmax(c))
    b = a.copy()
    # Gradient direction at k_max: g[i] = 2*a[k_max - i]
    js = max(0, k_max - n + 1)
    je = min(n - 1, k_max) + 1
    if js < je:
        contrib = a[k_max - np.arange(js, je)]
        # Increase where contrib is small, decrease where large
        noise = rng.normal(0, alpha, size=je - js)
        b[js:je] -= contrib / (contrib.max() + 1e-10) * noise
    # Also perturb a random secondary block
    w = max(4, n // 80)
    s = int(rng.integers(0, n - w))
    b[s:s + w] *= np.exp(rng.normal(0, alpha * 0.5, size=w))
    return _project(b, n)


def _perturb_mirror(a, n, rng, alpha):
    """Mirror-symmetric perturbation: nudge a[i] up, a[n-1-i] down."""
    b = a.copy()
    n_pert = max(10, n // 20)
    for _ in range(n_pert):
        i = int(rng.integers(0, n))
        j = n - 1 - i
        d = rng.normal(0, alpha)
        b[i] += d
        b[j] -= d
    return _project(b, n)


def _perturb_block(a, n, rng, sigma, wf=None):
    """Standard random block multiplicative perturbation."""
    b = a.copy()
    if wf is None:
        w = max(4, n // 60)
    else:
        w = max(4, int(n * wf))
    s = int(rng.integers(0, max(1, n - w + 1)))
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _project(b, n)


def _make_inits(n, rng):
    """Generate diverse initial sequences, each normalized to sum=n."""
    inits = []
    # 1: uniform + noise
    a = np.ones(n) * (1 + rng.normal(0, 0.1, n))
    inits.append(_project(a, n))
    # 2: half-sine (high in middle)
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2 + 0.01
    inits.append(_project(a, n))
    # 3: inverted half-sine (high at edges)
    a = 1.0 - 0.8 * np.sin(x) ** 2 + 0.01
    inits.append(_project(a, n))
    # 4: sparse comb (alternating blocks)
    a = np.ones(n)
    a[n // 4:3 * n // 4] *= 0.1
    inits.append(_project(a, n))
    # 5: random with broader distribution
    a = rng.exponential(1.0, size=n) + 0.01
    inits.append(_project(a, n))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase FFT-accelerated search: multi-start -> smooth -> hard ->
    structured perturb + refine -> final polish.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 8.0

    # --- Choose n and FFT length ---
    n = 3809
    L = 2 ** int(np.ceil(np.log2(3 * n + 1)))  # 16384

    # --- Get parent if available ---
    parent = None
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)
        if par.ndim == 1 and par.size == n and np.all(np.isfinite(par)):
            parent = np.clip(par, 0.0, 1000.0)
    except Exception:
        pass

    # --- Phase 1: Multi-start quick optimization (20% budget) ---
    p1_budget = 0.18 * (budget_s - 10)
    p1_end = t0 + p1_budget
    inits = _make_inits(n, rng)
    if parent is not None:
        p = _project(parent.copy(), n)
        inits.insert(0, p)
        print(f"Using parent: sum={parent.sum():.1f}")

    best_a = inits[0].copy()
    best_o = _obj_hard(best_a, n, L)
    print(f"Phase 1: Multi-start ({len(inits)} inits, {p1_budget:.0f}s budget)")

    per_init = max(1.0, (p1_end - time.monotonic()) / len(inits))
    for idx, a_init in enumerate(inits):
        if time.monotonic() > p1_end:
            break
        sub_end = min(time.monotonic() + per_init * 1.2, p1_end)
        o_init = _obj_hard(a_init, n, L)
        print(f"  Init {idx}: obj={o_init:.6f}")
        a_q, o_q = _smooth_run(a_init, n, L, sub_end, f"m{idx}", init_tau=n * 0.1, init_eta=0.08, it_cap=100000)
        if o_q < best_o:
            best_o = o_q
            best_a = a_q
            print(f"  -> New best from init {idx}: {best_o:.6f}")

    print(f"After Phase 1: obj={best_o:.6f}  elapsed={time.monotonic()-t0:.0f}s")

    # --- Phase 2: Extended smooth optimization on best (25% budget) ---
    p2_end = min(time.monotonic() + 0.25 * (budget_s - 10), deadline - 5)
    a2, o2 = _smooth_run(best_a, n, L, p2_end, "smooth2", init_tau=n * 0.04, init_eta=0.05)
    if o2 < best_o:
        best_o, best_a = o2, a2
    print(f"After Phase 2: obj={best_o:.6f}  elapsed={time.monotonic()-t0:.0f}s")

    # --- Phase 3: Hard top-K descent (20% budget) ---
    p3_end = min(time.monotonic() + 0.20 * (budget_s - 10), deadline - 5)
    a3, o3 = _hard_run(best_a, n, L, p3_end, "hard3", init_eta=0.04, K=10)
    if o3 < best_o:
        best_o, best_a = o3, a3
    print(f"After Phase 3: obj={best_o:.6f}  elapsed={time.monotonic()-t0:.0f}s")

    # --- Phase 4: Structured perturb + refine loop (25% budget) ---
    p4_end = min(time.monotonic() + 0.25 * (budget_s - 10), deadline - 10)
    attempt = 0
    while time.monotonic() < p4_end:
        attempt += 1
        # Choose perturbation type
        r = rng.random()
        if r < 0.35:
            # Lag-targeted
            a_p = _perturb_lag_target(best_a, n, L, rng, alpha=rng.uniform(0.3, 1.0))
        elif r < 0.60:
            # Mirror
            a_p = _perturb_mirror(best_a, n, rng, alpha=rng.uniform(0.3, 0.8))
        elif r < 0.85:
            # Random block
            wf = float(rng.choice([0.02, 0.05, 0.10, 0.20]))
            a_p = _perturb_block(best_a, n, rng, rng.uniform(0.1, 0.5), wf)
        else:
            # Large-scale: mix with a fresh random init
            a_rand = _project(rng.uniform(0.1, 2.0, size=n), n)
            a_p = _project(0.7 * best_a + 0.3 * a_rand, n)

        # Quick smooth refine
        sub_s = min(time.monotonic() + 8.0, p4_end - 2)
        a_q, o_q = _smooth_run(a_p, n, L, sub_s, f"ps{attempt}", init_tau=n * 0.03, init_eta=0.06, it_cap=50000)

        # Quick hard refine
        sub_h = min(time.monotonic() + 6.0, p4_end - 2)
        a_q2, o_q2 = _hard_run(a_q, n, L, sub_h, f"ph{attempt}", init_eta=0.03, K=6, it_cap=30000)

        o_final = min(o_q, o_q2)
        a_final = a_q if o_q < o_q2 else a_q2
        if o_final < best_o:
            best_o = o_final
            best_a = a_final
            print(f"  Perturb {attempt}: obj={best_o:.6f} (type={'LT' if r<0.35 else 'M' if r<0.6 else 'RB' if r<0.85 else 'MX'})")

    print(f"After Phase 4: obj={best_o:.6f}  elapsed={time.monotonic()-t0:.0f}s")

    # --- Phase 5: Final hard polish with remaining time ---
    if time.monotonic() < deadline - 3:
        a5, o5 = _hard_run(best_a, n, L, deadline - 1.0, "final", init_eta=0.02, K=5, it_cap=1000000)
        if o5 < best_o:
            best_o, best_a = o5, a5
        print(f"After Phase 5 (final): obj={best_o:.6f}  elapsed={time.monotonic()-t0:.0f}s")

    # --- Validate and return ---
    result = [float(x) for x in best_a]
    # Ensure all values are valid
    result = [max(0.0, min(1000.0, x)) for x in result]

    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')

    if not np.isfinite(sc):
        # Fallback: return a valid uniform sequence
        result = [1.0] * n
        sc = 2.0
        print("WARNING: Fallback to uniform!")

    print(f"FINAL: internal_obj={best_o:.6f}  evaluated={sc:.6f}  n={n}")
    return result
