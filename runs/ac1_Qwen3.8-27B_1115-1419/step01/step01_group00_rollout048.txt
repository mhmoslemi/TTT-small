```python
"""
Multi-method optimizer for minimizing

    2 n max_k conv(a, a)_k / (sum a)^2,

with a nonnegative.  The main ingredients are:

1. FFT-based self-convolution and objective evaluation.
2. A spectral Newton / projection method that tries to make the
   convolution histogram close to a water-filled target histogram with a
   prescribed maximum coefficient.
3. Log-softmax Adam-style smooth-max descent.
4. A top-active-set projected gradient polish.
5. Multiple diverse lengths / initializations with brief screening,
   followed by longer refinement of the best candidates.
"""

import numpy as np
import time


def _norm(x):
    """Make x a finite nonnegative probability vector, or None if impossible."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return np.full(x.shape, 1.0 / x.size, dtype=np.float64)
    return x / s


def _fft_len(n):
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    return L


def _objective(b, n):
    return 2.0 * float(n) * float(np.max(b))


def _fft_transform(a, n, L):
    """Return A=rfft(a,L) and b=a*a (linear conv), normalized to sum 1."""
    m = 2 * n - 1
    A = np.fft.rfft(a, L)
    B = A * A
    b = np.fft.irfft(B, L)[:m]
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-300:
        b = np.full(m, 1.0 / m, dtype=np.float64)
    elif abs(s - 1.0) > 1e-12:
        b = b / s
    return A, b


def _fast_obj(a, n, L):
    if a is None:
        return float('inf')
    _, b = _fft_transform(a, n, L)
    return _objective(b, n)


def _get_parent():
    try:
        obj = globals().get("height_sequence_1", None)
        if obj is None:
            return None
        arr = np.asarray(obj, dtype=np.float64).ravel()
        if arr.size == 0:
            return None
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if s <= 0.0:
            return None
        return arr / s
    except Exception:
        return None


def _upsample(parent, n):
    m = parent.size
    if m == n:
        return _norm(parent)
    x0 = np.linspace(0.0, 1.0, m)
    x1 = np.linspace(0.0, 1.0, n)
    y = np.log(np.maximum(parent, 0.0) + 1e-15)
    a = np.exp(np.interp(x1, x0, y))
    return _norm(a)


def _initials_for_n(n, parent):
    """A few deterministic / random-free initial shapes."""
    out = []

    x = np.arange(n, dtype=np.float64) + 0.5

    # Left-sided square-root singular shape.
    a0 = 1.0 / np.sqrt(x)
    a0 = _norm(a0)
    if a0 is not None:
        out.append(a0)
        out.append(a0[::-1].copy())

    # Mild power-law shape.
    a1 = np.power(x, -0.3)
    a1 = _norm(a1)
    if a1 is not None:
        out.append(a1)

    # Load and upsample the previous best if it exists.
    if parent is not None and parent.size > 0:
        au = _upsample(parent, n)
        if au is not None:
            out.append(au)

    # Ensure at least one valid initialization.
    valid = [u for u in out if u is not None]
    if not valid:
        valid.append(np.full(n, 1.0 / n, dtype=np.float64))
    return valid


def _perturb(a, n, rng, sigma=None, width_frac=None):
    b = a.copy()
    if sigma is None:
        sigma = float(rng.uniform(0.05, 0.25))
    if width_frac is None:
        width_frac = float(rng.choice([0.01, 0.02, 0.05, 0.10, 0.20]))
    w = int(np.clip(n * width_frac, 3, max(3, n // 2)))
    if n <= w:
        w = max(1, n // 2)
    s = int(rng.integers(0, n - w + 1)) if n > w else 0
    b[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return _norm(b)


def _make_target_fast(b, T):
    """
    Build a nonnegative target histogram y with:
      sum(y) = 1,
      max(y) close to T,
    and y close to current b.  Uses capping plus a small water-fill.
    Returns None if T appears infeasible.
    """
    if T <= 0.0:
        return None
    maxb = float(np.max(b))
    if maxb <= T:
        return b
    m = b.size
    y = np.minimum(b, T)
    s = float(y.sum())
    rem = 1.0 - s
    if rem <= 1e-12:
        return y

    # First/add inactive entries until sum is restored as much as possible.
    for _ in range(3):
        active = y < T
        cnt = int(np.count_nonzero(active))
        if cnt == 0:
            break
        y[active] += rem / cnt
        y = np.minimum(y, T)
        s = float(y.sum())
        rem = 1.0 - s
        if rem <= 1e-12:
            break

    s = float(y.sum())
    if s < 1.0 - 1e-8:
        return None
    y = y / s
    if float(np.max(y)) > T * 1.05:
        # Do not return an implausible target; caller will relax T.
        return None
    return y


def _spectral_update(A, b_target, gamma, n, L):
    """
    Given current spectrum A = FFT(a/L), damp toward a square root of
    FFT(b_target), project back to nonnegative probabilities.
    """
    Bt = np.fft.rfft(b_target, L)
    up = A.astype(np.complex128, copy=True)
    absA = np.abs(A)
    small = absA < 1e-14
    if np.any(~small):
        big = ~small
        with np.errstate(divide="ignore", invalid="ignore"):
            cand = 0.5 * (A[big] + Bt[big] / A[big])
        bad = ~np.isfinite(cand)
        old_abs = np.abs(A[big])
        new_abs = np.abs(cand)
        bad |= new_abs > (1e4 * old_abs + 1e-8)
        if np.any(bad):
            cand[bad] = A[big][bad]
        up[big] = (1.0 - gamma) * A[big] + gamma * cand
    if np.any(small):
        with np.errstate(invalid="ignore"):
            up[small] = np.sqrt(Bt[small])
        up[small] = np.where(np.isfinite(up[small]), up[small], 1e-12)
    bad = ~np.isfinite(up)
    if np.any(bad):
        up[bad] = A[bad]
    a = np.fft.irfft(up, L)[:n]
    return _norm(a)


def _spectral_loop(a, n, L, end_time, tag, rng=None):
    """Spectral target-tracking optimization."""
    a = _norm(a)
    if a is None:
        return None, float('inf')

    _, b0 = _fft_transform(a, n, L)
    best = a.copy()
    best_o = _objective(b0, n)

    T_min = 0.75 / n  # corresponds to objective 1.5
    maxb0 = float(np.max(b0))
    T_cur = max(T_min, maxb0)
    gamma = 0.6
    stall = 0
    fail_count = 0
    it = 0

    while time.monotonic() < end_time:
        A, b = _fft_transform(a, n, L)
        maxb = float(np.max(b))
        o = _objective(b, n)

        if o < best_o - 1e-12:
            best = a.copy()
            best_o = o
            stall = 0
            gamma = min(0.9, gamma * 1.02)
            if it % 20000 == 0:
                print(f"  [{tag}] spectral it={it} obj={o:.6f} best={best_o:.6f}")
        else:
            stall += 1
            if stall > 8000 or o > 2.5 * best_o:
                if rng is not None:
                    a = _perturb(best, n, rng)
                else:
                    a = best.copy()
                stall = 0
                fail_count = 0

        best_max = best_o / (2.0 * n)
        T = max(T_min, min(T_cur, 0.999 * best_max))
        # Gentle annealing toward the asymptotic target.
        T_cur = max(T_min, T_cur * (1.0 - 1e-6))

        b_target = _make_target_fast(b, T)
        if b_target is None:
            b_target = b
            fail_count += 1
            if fail_count > 80:
                # Relax the target if it is being too aggressive.
                T_cur = min(best_max * 1.05, max(T_min, T_cur * 1.02))
                fail_count = 0
        else:
            fail_count = 0

        a2 = _spectral_update(A, b_target, gamma, n, L)
        if a2 is None:
            gamma = max(0.1, gamma * 0.6)
            a2 = a
        else:
            if o < best_o * 0.999:
                gamma = min(0.9, gamma * 1.01)
            elif o > 1.5 * best_o and it > 200:
                gamma = max(0.1, gamma * 0.8)
        a = a2
        it += 1

    return best, best_o


def _top_gradient(a, b, n, K):
    """Average gradient of the top K convolution coefficients."""
    m = b.size
    if K >= m:
        idx = np.arange(m)
    else:
        kth = m - K
        idx = np.argpartition(b, kth)[kth:]

    g = np.zeros(n, dtype=np.float64)
    invK = 1.0 / len(idx)
    for k in idx:
        s = max(0, int(k) - n + 1)
        e = min(n - 1, int(k)) + 1
        if s < e:
            g[s:e] += 2.0 * invK * a[k - np.arange(s, e)]
    return g


def _hard_loop(a, n, L, end_time, tag, rng=None, K_init=5, K_final=1):
    """Polish with exact / top-active-set projected gradient steps."""
    a = _norm(a)
    if a is None:
        return None, float('inf')
    _, b0 = _fft_transform(a, n, L)
    best = a.copy()
    best_o = _objective(b0, n)

    eta = 0.20
    stall = 0
    it = 0
    start = time.monotonic()
    total_t = max(1e-9, end_time - start)

    while time.monotonic() < end_time:
        _, b = _fft_transform(a, n, L)
        o = _objective(b, n)
        improved = o < best_o - 1e-12
        if improved:
            best = a.copy()
            best_o = o
            stall = 0
            eta = min(0.35, eta * 1.01)
        else:
            stall += 1
            if stall > 5000:
                eta = max(0.03, eta * 0.7)
                if rng is not None:
                    a = _perturb(best, n, rng, sigma=0.12, width_frac=0.03)
                stall = 0

        progress = min(1.0, (time.monotonic() - start) / total_t)
        K = int(round(K_init - (K_init - K_final) * progress))
        K = max(K_final, min(K_init, K))

        g = _top_gradient(a, b, n, K)
        g -= g.mean()
        mg = float(np.max(np.abs(g)))
        if mg < 1e-15:
            break
        a = _norm(np.maximum(a - eta * (g / mg), 0.0))
        it += 1
        if it % 20000 == 0:
            print(f"  [{tag}] hard it={it} obj={o:.6f} best={best_o:.6f}")

    return best, best_o


def _adam_log(a, n, L, end_time, tag, rng=None,
              temp_factor_start=20.0, temp_factor_end=2.0, lr=0.15):
    """Log-softmax Adam on a smooth-max approximation of the objective."""
    a = _norm(a)
    if a is None:
        return None, float('inf')
    _, b0 = _fft_transform(a, n, L)
    best = a.copy()
    best_o = _objective(b0, n)

    v = np.log(np.maximum(a, 1e-300))
    v -= v.mean()
    m1 = np.zeros(n, dtype=np.float64)
    m2 = np.zeros(n, dtype=np.float64)
    bias1 = 0.0
    bias2 = 0.0
    temp_factor = temp_factor_start
    stall = 0
    zero_count = 0
    max_stall = max(4000, n // 2)

    while time.monotonic() < end_time:
        if not np.all(np.isfinite(v)):
            v = np.log(np.maximum(best, 1e-300))
            v -= v.mean()
            m1[:] = 0.0
            m2[:] = 0.0
            bias1 = bias2 = 0.0
            continue

        mx = float(np.max(v))
        ex = np.exp(v - mx)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0.0:
            a = best.copy()
            v = np.log(np.maximum(a, 1e-300))
            v -= v.mean()
            m1[:] = 0.0
            m2[:] = 0.0
            bias1 = bias2 = 0.0
            continue
        a = ex / s

        A, b = _fft_transform(a, n, L)
        maxb = float(np.max(b))
        o = _objective(b, n)

        if o < best_o - 1e-12:
            best = a.copy()
            best_o = o
            stall = 0
            if it > 0 and it % 20000 == 0:
                print(f"  [{tag}] adam obj={o:.6f} best={best_o:.6f} temp={temp_factor:.3f}")
        else:
            stall += 1
            if stall > max_stall:
                if rng is not None:
                    a = _perturb(best, n, rng, sigma=0.12, width_frac=0.03)
                else:
                    a = best.copy()
                v = np.log(np.maximum(a, 1e-300))
                v -= v.mean()
                m1[:] = 0.0
                m2[:] = 0.0
                bias1 = bias2 = 0.0
                temp_factor = max(temp_factor, 10.0)
                stall = 0
                zero_count = 0
                continue

        if temp_factor > temp_factor_end:
            temp_factor = max(temp_factor_end, temp_factor * 0.99992)

        temp = maxb / max(1.0, temp_factor)
        if temp <= 1e-300:
            temp = 1e-300

        z = (b - maxb) / temp
        p = np.exp(z)
        ps = float(p.sum())
        if not np.isfinite(ps) or ps <= 0.0:
            p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
            ps = float(b.size)
        p = p / ps

        P = np.fft.rfft(p, L)
        corr = np.fft.irfft(P * np.conj(A), L)[:n]
        g = 2.0 * corr
        g = np.where(np.isfinite(g), g, 0.0)

        dot = float(np.dot(a, g))
        gv = a * (g - dot)
        gv = np.where(np.isfinite(gv), gv, 0.0)
        gs = float(np.max(np.abs(gv)))

        if gs < 1e-15:
            zero_count += 1
            if zero_count > 2000:
                if rng is not None:
                    a = _perturb(best, n, rng, sigma=0.08, width_frac=0.02)
                else:
                    a = best.copy()
                v = np.log(np.maximum(a, 1e-300))
                v -= v.mean()
                m1[:] = 0.0
                m2[:] = 0.0
                bias1 = bias2 = 0.0
                zero_count = 0
            continue
        zero_count = 0

        gv = gv / gs
        m1 = 0.9 * m1 + 0.1 * gv
        m2 = 0.999 * m2 + 0.001 * (gv * gv)

        bias1 = 1.0 - 0.9 * (1.0 - bias1)
        bias2 = 1.0 - 0.999 * (1.0 - bias2)
        c1 = max(bias1, 1e-8)
        c2 = max(bias2, 1e-8)

        v = v + lr * (m1 / c1) / (np.sqrt(m2 / c2) + 1e-4)
        v = np.where(np.isfinite(v), v, 0.0)
        v -= v.mean()

        vmin = float(np.min(v))
        vmax = float(np.max(v))
        if vmax - vmin > 60.0:
            v = v * (60.0 / (vmax - vmin + 1e-12))

    return best, best_o


def _refine_candidate(a, n, L, end_time, tag, rng):
    a = _norm(a)
    now = time.monotonic()
    if now >= end_time or a is None:
        return a, float('inf')

    total = max(1.0, end_time - now)
    spec_end = now + 0.50 * total
    adam_end = now + 0.80 * total

    a2, o2 = _spectral_loop(a, n, L, spec_end, tag + ":spec", rng)
    if a2 is not None:
        a, best_o = a2, o2

    if time.monotonic() < end_time:
        a3, o3 = _adam_log(a, n, L, adam_end, tag + ":adam1", rng,
                           temp_factor_start=20.0, temp_factor_end=4.0, lr=0.15)
        if a3 is not None:
            a, best_o = a3, o3

    if time.monotonic() < end_time:
        a4, o4 = _hard_loop(a, n, L, end_time, tag + ":hard", rng,
                            K_init=5, K_final=1)
        if a4 is not None:
            a, best_o = a4, o4

    return a, best_o


def propose_candidate(seed=42, budget_s=2400):
    """Return the best nonnegative sequence found within the time budget."""
    start = time.monotonic()
    deadline = start + max(60.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    parent = _get_parent()

    # Candidate lengths.  Parent length if available, then a few modern sizes.
    n_list = []
    if parent is not None and parent.size >= 200:
        n_list.append(int(parent.size))
    for nn in (8192, 16384, 32768):
        if nn not in n_list:
            n_list.append(nn)
    n_list = list(dict.fromkeys(n_list))[:4]

    # Build pre-screen candidates.
    precands = []
    for n in n_list:
        L = _fft_len(n)
        for a0 in _initials_for_n(n, parent):
            if a0 is None:
                continue
            precands.append((n, L, a0))

    # Brief screen: run a short spectral loop on each candidate.
    screen_budget = min(120.0, max(30.0, 0.05 * (deadline - start)))
    screen_start = time.monotonic()
    if precands and time.monotonic() < deadline - screen_budget:
        per = min(15.0, screen_budget / max(1, len(precands)))
        screen_total_start = screen_start
        candidates = []
        for i, (n, L, a0) in enumerate(precands):
            end = min(deadline - 10.0, screen_total_start + per * (i + 1))
            if time.monotonic() >= end:
                end = time.monotonic() + 0.5
            a2, o2 = _spectral_loop(a0, n, L, end, f"screen_n{n}_i{i}", rng)
            if a2 is not None and np.isfinite(o2):
                candidates.append({"a": a2, "n": n, "L": L, "o": o2})
        candidates.sort(key=lambda x: x["o"])
    else:
        candidates = []

    # Fallback if screening produced nothing.
    if not candidates:
        if parent is not None:
            n = int(parent.size)
            a0 = _norm(parent)
        else:
            n = 8192
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        L = _fft_len(n)
        o0 = _fast_obj(a0, n, L)
        candidates.append({"a": a0, "n": n, "L": L, "o": o0})

    global_best = candidates[0]
    print(f"Screen best: n={global_best['n']} obj={global_best['o']:.6f} "
          f"elapsed={time.monotonic() - start:.1f}s")

    # Main refinement of the top three screened candidates.
    main_start = max(time.monotonic(), screen_start)
    main_end = deadline - 75.0
    if main_end > main_start + 20.0:
        total_main = main_end - main_start
        fractions = [0.50, 0.30, 0.20]
        cum = 0.0
        for i, cand in enumerate(candidates[:3]):
            frac = fractions[i]
            end = main_start + (cum + frac) * total_main
            cum += frac
            if time.monotonic() >= end:
                break
            print(f"Refine candidate {i}: n={cand['n']} obj={cand['o']:.6f} "
                  f"until {(end - time.monotonic()):.1f}s")
            a2, o2 = _refine_candidate(cand["a"], cand["n"], cand["L"], end,
                                       f"main_i{i}", rng)
            if a2 is not None and o2 < global_best["o"]:
                global_best = {"a": a2, "n": cand["n"], "L": cand["L"], "o": o2}
                print(f"  new global best: n={cand['n']} obj={o2:.6f}")

    # Final short polish on the global best.
    if time.monotonic() < deadline - 30.0 and global_best is not None:
        final_end = deadline - 15.0
        a2, o2 = _refine_candidate(global_best["a"], global_best["n"],
                                   global_best["L"], final_end, "final", rng)
        if a2 is not None and o2 < global_best["o"]:
            global_best = {"a": a2, "n": global_best["n"],
                           "L": global_best["L"], "o": o2}

    # Final validation and return.
    if global_best is None:
        fallback = parent if parent is not None else np.ones(4096, dtype=np.float64) / 4096
        result = [float(x) for x in _norm(fallback)]
    else:
        result = [float(x) for x in np.asarray(global_best["a"], dtype=np.float64)]

    # Ensure security constraints accepted by evaluate_sequence.
    try:
        result = [min(1000.0, max(0.0, float(x))) for x in result]
    except Exception:
        result = [1.0 / 4096] * 4096

    try:
        direct = float(evaluate_sequence(result))
    except Exception:
        direct = float('inf')

    print(f"FINAL internal obj={global_best['o'] if global_best is not None else float('inf'):.6f} "
          f"direct={direct:.6f} n={len(result)} elapsed={time.monotonic() - start:.1f}s")

    if not np.isfinite(direct):
        fallback = parent if parent is not None else np.ones(4096, dtype=np.float64) / 4096
        result = [float(x) for x in _norm(fallback)]
        try:
            direct = float(evaluate_sequence(result))
            print(f"FALLBACK direct={direct:.6f}")
        except Exception:
            pass

    return result
```
