"""
Search for non-negative step-function heights minimizing

    2 * n * max(convolve(a, a)) / sum(a)**2

The main changes from the previous program are:
  * exact gradients for the LINEAR convolution objective
    (the earlier conjugate-correlation gradient can drift),
  * annealed active-set / smooth-max weights,
  * Adam-style optimization in log-simplicial coordinates,
  * warm true lbfgs polishing on a smooth log-sum-exp surrogate,
  * multiple constructive starting points, including the provided global,
  * final 8192-point upscaling if enough time remains.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _pair_first(pair):
    return pair[0]


def _proj(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
    else:
        a /= s
    return a


def _norm_nonneg(x):
    try:
        arr = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return arr / s


def _interp(a, n):
    a = _proj(a)
    n = int(max(2, n))
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return _proj(out)


def _soft(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v = v - float(np.max(v))
    v = np.clip(v, -745.0, 700.0)
    e = np.exp(v)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(v.shape, 1.0 / float(v.size), dtype=np.float64)
    return e / s


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    a = a / s
    n = int(a.size)
    m = _next_pow2(2 * n - 1)
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    B = float(b.max())
    if not np.isfinite(B) or B <= 0.0:
        return np.inf
    return float(2.0 * n * B)


def _softmax_weights_logsum(b, B, tau):
    L = int(b.size)
    if L <= 0:
        return np.empty(0), -np.inf
    if (not np.isfinite(B)) or B <= 1e-300 or (not np.isfinite(tau)) or tau <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), -np.log(float(L))
    x = (b - B) / float(tau)
    x = np.clip(x, -745.0, 700.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64), -np.log(float(L))
    w = e / s
    logsum = float(np.log(s) - np.log(float(L)))
    return w, logsum


def _grad_from_FA(a, w, n, m, FA, WW=None):
    """
    Gradient of sum_k w_k * (a*a)_k with respect to a.

    For linear convolution,
        (a*a)_k = sum_{i+j=k} a_i a_j,
    so d/d a_i = 2 * sum_j w_{i+j} a_j.

    This is a Hankel matrix-vector product, computed by one convolution.
    """
    L = 2 * n - 1
    if WW is not None:
        WW[:L] = w[::-1]
    else:
        WW = np.zeros(m, dtype=np.float64)
        WW[:L] = w[::-1]

    FW = np.fft.fft(WW, m)
    res = np.fft.ifft(FW * FA).real
    idx = L - 1 - np.arange(n, dtype=np.int64)
    h = res[idx]
    return 2.0 * h


def _smooth_lbfgs_fun(x, tau, n, m, scale):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        k = min(n, x.size)
        tmp[:k] = x[:k]
        x = tmp

    a = _soft(x)
    L = 2 * n - 1
    FA = np.fft.fft(a, m)
    b = np.fft.ifft(FA * FA).real[:L]
    b = np.maximum(b, 0.0)
    B = float(b.max())

    if not np.isfinite(B) or B <= 1e-300:
        return 1e100, np.zeros(n, dtype=np.float64)

    tau = float(max(1e-15, tau))
    w, logsum = _softmax_weights_logsum(b, B, tau)
    ga = _grad_from_FA(a, w, n, m, FA, None)
    gdot = float(np.dot(ga, a))
    gv = a * (ga - gdot)

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(n, dtype=np.float64)
    else:
        gv -= float(np.mean(gv))

    val = B + tau * logsum
    val *= float(scale)
    gv *= float(scale)

    if not np.isfinite(val):
        val = 1e100
        gv = np.zeros(n, dtype=np.float64)

    return float(val), gv


def _run_lbfgs_once(a_init, tau_coeff, iters):
    a = _proj(a_init)
    n = int(a.size)
    sc0 = _score_fft(a)
    if not np.isfinite(sc0) or minimize is None or iters <= 0:
        return a, sc0

    v = np.log(np.maximum(a, 1e-12))
    v = v - float(np.mean(v))
    v = np.clip(v, -30.0, 30.0)

    m = _next_pow2(3 * n - 2)
    scale = 100.0 * float(n)
    tau = float(tau_coeff) / float(n)

    best_a = a.copy()
    best_sc = float(sc0)

    try:
        opts = {
            "maxiter": int(max(1, iters)),
            "ftol": 0.0,
            "gtol": 1e-12,
            "maxls": 40,
            "disp": 0,
        }
        res = minimize(
            _smooth_lbfgs_fun,
            v,
            args=(tau, n, m, scale),
            method="L-BFGS-B",
            jac=True,
            options=opts,
        )
        if res is not None and res.x is not None:
            v2 = np.asarray(res.x, dtype=np.float64).ravel()
            if v2.size != n:
                tmp = np.zeros(n, dtype=np.float64)
                k = min(n, v2.size)
                tmp[:k] = v2[:k]
                v2 = tmp
            v2 = v2 - float(np.mean(v2))
            v2 = np.clip(v2, -30.0, 30.0)
            a2 = _soft(v2)
            sc2 = _score_fft(a2)
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = sc2
                best_a = a2.copy()
    except Exception:
        pass

    return _proj(best_a), float(best_sc)


def _run_lbfgs_sequence(a_init, tau_coeffs, iters_per):
    a = _proj(a_init)
    best_a = a.copy()
    best_sc = _score_fft(a)
    if minimize is None or len(tau_coeffs) == 0:
        return best_a, best_sc

    for tc in tau_coeffs:
        a_new, sc_new = _run_lbfgs_once(a, float(tc), int(iters_per))
        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

    return _proj(best_a), float(best_sc)


def optimize_adam(a_init, seconds, seed=0):
    """
    Adam-like optimization on the simple exploding-exponential parameterization
    with exact linear-convolution gradients and annealed active-set weights.
    """
    seconds = float(seconds)
    a = _proj(a_init)
    n = int(a.size)
    init_sc = _score_fft(a)

    if not np.isfinite(init_sc):
        a = np.ones(n, dtype=np.float64) / float(n)
        init_sc = 2.0

    if seconds <= 1.0:
        return a, float(init_sc)

    rng = np.random.default_rng(int(seed))
    best_a = a.copy()
    best_sc = float(init_sc)
    best_B = best_sc / (2.0 * float(n))
    B_scale = max(best_B, 1.0 / float(n))

    v = np.log(np.maximum(a, 1e-12))
    v = v - float(np.mean(v))
    v = np.clip(v, -30.0, 30.0)

    L = 2 * n - 1
    m = _next_pow2(3 * n - 2)
    WW = np.zeros(m, dtype=np.float64)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)
    t = 0
    it = 0

    start = time.monotonic()
    deadline = start + max(1.0, seconds)
    last_improve = start

    active_idx = None
    active_k = -1

    while time.monotonic() < deadline:
        now = time.monotonic()
        progress = min(1.0, (now - start) / max(1e-9, seconds))

        a = _soft(v)
        FA = np.fft.fft(a, m)
        b = np.fft.ifft(FA * FA).real[:L]
        b = np.maximum(b, 0.0)
        B = float(b.max())
        it += 1

        if (not np.isfinite(B)) or B <= 1e-300:
            sigma = 0.10 + 0.25 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -30.0, 30.0)
            last_improve = now
            continue

        if B < best_B - max(1e-12, best_B * 1e-10):
            best_B = B
            best_sc = float(2.0 * n * B)
            best_a = a.copy()
            B_scale = max(best_B, 1.0 / float(n))
            last_improve = now

        stuck_limit = min(90.0, 0.25 * seconds)
        if now - last_improve > stuck_limit:
            sigma = 0.08 + 0.25 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -30.0, 30.0)
            last_improve = now
            continue

        tau_frac = (0.55 * (1.0 - progress) + 0.04) * float(rng.uniform(0.80, 1.35))
        tau = max(1e-15, B_scale * tau_frac)

        top_frac = 0.25 * (1.0 - progress) + 0.03
        top_k = int(L * top_frac)
        if top_k < 3:
            top_k = 3
        if top_k > L:
            top_k = L

        if top_k >= L:
            active_idx = None
            active_k = -1
        else:
            if active_idx is None or active_k != top_k or (it % 4 == 0):
                active_idx = np.argpartition(b, -top_k)[-top_k:]
                active_k = top_k

        w, _ = _softmax_weights_logsum(b, B, tau)
        if active_idx is not None and active_idx.size > 0:
            top_weight = 0.45 + 0.35 * progress
            w = w * (1.0 - float(top_weight))
            w[active_idx] += float(top_weight) / float(active_idx.size)
            ws = float(w.sum())
            if not np.isfinite(ws) or ws <= 0.0:
                w, _ = _softmax_weights_logsum(b, B, tau)
            else:
                w /= ws

        ga = _grad_from_FA(a, w, n, m, FA, WW)
        gdot = float(np.dot(ga, a))
        gv = a * (ga - gdot)
        gv -= float(np.mean(gv))
        grms = float(np.sqrt(np.mean(gv * gv)))

        if (not np.isfinite(grms)) or grms < 1e-16:
            sigma = 0.08 + 0.20 * (1.0 - progress)
            v += rng.normal(0.0, sigma, n)
            v = v - float(np.mean(v))
            v = np.clip(v, -30.0, 30.0)
            last_improve = now
            continue

        gv /= grms
        t += 1
        mu = beta1 * mu + (1.0 - beta1) * gv
        nu = beta2 * nu + (1.0 - beta2) * (gv * gv)

        if t > 2000:
            b1 = 0.0
            b2 = 0.0
        else:
            b1 = beta1 ** t
            b2 = beta2 ** t

        den1 = max(1e-12, 1.0 - b1)
        den2 = max(1e-12, 1.0 - b2)
        mhat = mu / den1
        vhat = nu / den2

        lr = (0.06 * (1.0 - progress) + 0.012) * float(rng.uniform(0.85, 1.15))
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.35, 0.35)

        v = v - lr * step
        v = v - float(np.mean(v))
        v = np.clip(v, -30.0, 30.0)

    a_final = _soft(v)
    sc_final = _score_fft(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return _proj(best_a), float(best_sc)


def _analytic_inits(n, rng):
    out = []
    n = int(max(3, n))
    t = np.linspace(0.0, 1.0, n)

    out.append(_proj(np.ones(n, dtype=np.float64)))

    for p in (0.25, 0.5, 1.0, 2.0, 3.0):
        out.append(_proj(np.power(np.maximum(t, 1e-12), p)))
        out.append(_proj(np.power(np.maximum(1.0 - t, 1e-12), p)))

    tp = np.maximum(t, 1e-6)
    tq = np.maximum(1.0 - t, 1e-6)
    pairs = (
        (1.0, 0.0), (0.0, 1.0), (2.0, 0.0), (0.0, 2.0),
        (1.0, 1.0), (2.0, 1.0), (1.0, 2.0), (3.0, 1.0),
        (1.0, 3.0), (0.5, 2.0), (2.0, 0.5), (1.5, 1.0), (1.0, 1.5),
    )
    for p, q in pairs:
        out.append(_proj(np.power(tp, p) * np.power(tq, q)))

    for c in (0.5, 1.0, 2.0, 4.0):
        out.append(_proj(np.exp(c * t)))
        out.append(_proj(np.exp(c * (1.0 - t))))

    for ph in (0.0, 0.1, 0.25, 0.5, 0.75):
        a = (
            1.0
            + 0.5 * np.cos(2.0 * np.pi * (t + ph))
            + 0.25 * np.cos(4.0 * np.pi * (t + 2.0 * ph))
            + 0.10 * np.cos(6.0 * np.pi * (t + 3.0 * ph))
        )
        out.append(_proj(a))

    for _r in range(5):
        a = np.ones(n, dtype=np.float64)
        for k in range(1, 13):
            amp = 0.18 / (float(k) ** 0.70)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            ph2 = rng.uniform(0.0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t + ph)
            a += 0.5 * amp * np.sin(2.0 * np.pi * k * t + ph2)
        out.append(_proj(a))

    return out


def _diverse_select(a_list, max_count=8, thresh=0.03):
    res = []
    seen = []
    for a in a_list:
        try:
            a = _proj(a)
        except Exception:
            continue
        sc = _score_fft(a)
        if not np.isfinite(sc):
            continue

        ok = True
        for b in seen:
            if b.size != a.size:
                aa = _interp(a, b.size)
            else:
                aa = a
            d = aa - b
            if float(np.sqrt(np.dot(d, d))) <= float(thresh):
                ok = False
                break
        if ok:
            res.append((sc, a))
            seen.append(a)
            if len(res) >= max_count:
                break

    res.sort(key=_pair_first)
    return res


def _load_parent():
    try:
        seq = height_sequence_1
    except Exception:
        seq = None
    return _norm_nonneg(seq)


def _try_exact(a):
    try:
        val = evaluate_sequence([float(x) for x in a])
        val = float(val)
    except Exception:
        return np.inf
    if np.isfinite(val):
        return val
    return np.inf


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point.  Searches for the best sequence within the given budget
    and returns a Python list of non-negative floats.
    """
    start = time.monotonic()
    budget_s = float(max(60.0, float(budget_s)))
    deadline = start + max(60.0, budget_s - 35.0)
    seed = int(seed)
    rng = np.random.default_rng(seed)

    parent = _load_parent()

    N_SMALL = 4096
    N_LARGE = 8192

    if parent is not None:
        a0 = _interp(parent, N_SMALL)
    else:
        a0 = np.ones(N_SMALL, dtype=np.float64) / float(N_SMALL)

    best_a = a0.copy()
    best_score = _score_fft(a0)
    best_small_a = a0.copy()
    best_small_score = float(best_score)
    best_large_a = None
    best_large_score = np.inf

    print("parent_available=%s small_n=%d initial_score=%.8f" %
          (parent is not None, N_SMALL, best_score))

    inits = []
    if parent is not None:
        inits.append(a0)
        inits.append(_interp(parent[::-1], N_SMALL))
        p_noisy = a0 + rng.normal(0.0, 0.06 / np.sqrt(float(N_SMALL)), N_SMALL)
        inits.append(p_noisy)

    inits.extend(_analytic_inits(N_SMALL, rng))

    top = []
    for sc, a in _diverse_select(inits, max_count=8, thresh=0.035):
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_a = a.copy()
            if a.size == N_SMALL:
                best_small_score = float(sc)
                best_small_a = a.copy()
        top.append(a)
        if len(top) >= 3:
            break

    if not top:
        top.append(a0)

    print("after init selection best=%.8f" % best_score)

    # Quick true-L-BFGS polish from a couple of good basins.
    if minimize is not None and (deadline - time.monotonic()) > 220.0:
        for a_seed in top[:2]:
            if (deadline - time.monotonic()) < 90.0:
                break
            a_lb, sc_lb = _run_lbfgs_sequence(a_seed, (1.0, 0.4), 18)
            if np.isfinite(sc_lb) and sc_lb < best_score:
                best_score = float(sc_lb)
                best_a = a_lb.copy()
                if a_lb.size == N_SMALL:
                    best_small_score = float(sc_lb)
                    best_small_a = a_lb.copy()
            print("LBFGS quick improved_to %.8f" % best_score)

    # Main Adam search at 4096.
    now = time.monotonic()
    rem = deadline - now
    if rem > 120.0:
        sec = max(40.0, min(1100.0, rem * 0.45))
        a_ad, sc_ad = optimize_adam(best_a, sec, seed=seed + 1)
        if np.isfinite(sc_ad) and sc_ad < best_score:
            best_score = float(sc_ad)
            best_a = a_ad.copy()
            if a_ad.size == N_SMALL:
                best_small_score = float(sc_ad)
                best_small_a = a_ad.copy()
        print("after main 4096 Adam best=%.8f" % best_score)

    # Try a second basin if worthwhile.
    if len(top) > 1 and (deadline - time.monotonic()) > 300.0:
        a2 = top[1]
        if a2.size != best_a.size:
            a2 = _interp(a2, best_a.size)
        sec = max(40.0, min(350.0, (deadline - time.monotonic()) * 0.18))
        a_ad2, sc_ad2 = optimize_adam(a2, sec, seed=seed + 123)
        if np.isfinite(sc_ad2) and sc_ad2 < best_score:
            best_score = float(sc_ad2)
            best_a = a_ad2.copy()
            if a_ad2.size == N_SMALL:
                best_small_score = float(sc_ad2)
                best_small_a = a_ad2.copy()
        print("after second basin best=%.8f" % best_score)

    # Small-tau L-BFGS polishing near the end of the 4096-phase.
    if minimize is not None and (deadline - time.monotonic()) > 220.0:
        for tc in (0.14, 0.06, 0.025):
            if (deadline - time.monotonic()) < 90.0:
                break
            a_lb, sc_lb = _run_lbfgs_once(best_a, tc, 24)
            if np.isfinite(sc_lb) and sc_lb < best_score:
                best_score = float(sc_lb)
                best_a = a_lb.copy()
                if a_lb.size == N_SMALL:
                    best_small_score = float(sc_lb)
                    best_small_a = a_lb.copy()
        print("after small-tau LBFGS best=%.8f" % best_score)

    # Optional 8192 upscaling and refinement.
    if (deadline - time.monotonic()) > 350.0:
        a_up = _interp(best_a, N_LARGE)
        sc_up = _score_fft(a_up)
        if np.isfinite(sc_up) and sc_up < best_score + 0.004:
            sec_up = max(70.0, min(900.0, (deadline - time.monotonic()) - 220.0))
            a_up2, sc_up2 = optimize_adam(a_up, sec_up, seed=seed + 456)
            if np.isfinite(sc_up2):
                if sc_up2 < best_score:
                    best_score = float(sc_up2)
                    best_a = a_up2.copy()
                if sc_up2 < best_large_score:
                    best_large_score = float(sc_up2)
                    best_large_a = a_up2.copy()
            else:
                if np.isfinite(sc_up) and sc_up < best_large_score:
                    best_large_score = float(sc_up)
                    best_large_a = a_up.copy()
        print("after possible 8192 phase best=%.8f" % best_score)

    candidates = []
    if best_a is not None:
        candidates.append(best_a)
    if best_small_a is not None:
        candidates.append(best_small_a)
    if best_large_a is not None:
        candidates.append(best_large_a)
    if parent is not None:
        cand_parent = _interp(parent, min(parent.size, N_LARGE))
        if cand_parent.size <= 12000:
            candidates.append(cand_parent)

    # Choose the smallest exact evaluated score; exact is the true metric.
    final_a = best_a if best_a is not None else np.ones(N_SMALL, dtype=np.float64)
    final_exact = np.inf
    for cand in candidates:
        if cand is None:
            continue
        cand = _proj(cand)
        if cand.size > 12000:
            continue
        ex = _try_exact(cand)
        if np.isfinite(ex) and ex < final_exact:
            final_exact = ex
            final_a = cand.copy()

    # Ensure final sequence is strictly non-negative, finite, and has
    # sum comfortably above the evaluator's floor.
    final_a = _proj(final_a)
    final_a = np.maximum(final_a, 1e-14)
    s = float(final_a.sum())
    if not np.isfinite(s) or s <= 0.0:
        final_a = np.ones(N_SMALL, dtype=np.float64) / float(N_SMALL)
        s = 1.0
    else:
        final_a /= s

    # One last exact check is safe for our lengths.
    if final_a.size <= 12000:
        ex = _try_exact(final_a)
        if not np.isfinite(ex):
            final_a = _proj(np.ones(N_SMALL, dtype=np.float64))

    return [float(x) for x in final_a]
