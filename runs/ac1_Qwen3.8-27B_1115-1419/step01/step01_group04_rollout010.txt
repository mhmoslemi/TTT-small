```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a):
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:L]
    np.maximum(c, 0, out=c)
    return c

def score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))

def build_constraint_matrix(a):
    """Build the convolution constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        # i.e., for k from j to j+n-1, a[k-j] = a[0], a[1], ..., a[n-1]
        A[j:j+n, j] = a
    return A

def solve_lp(a, M):
    """Solve the LP to find b maximizing sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    L = 2 * n - 1
    
    c_obj = -np.ones(n)
    A = build_constraint_matrix(a)
    b_ub = np.full(L, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    
    if result.success:
        return result.x
    return None

def normalize(a, target=None):
    if target is None:
        target = float(len(a))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full(len(a), target / len(a))
    return a * (target / s)

def lp_iteration(a, M, n, t):
    b = solve_lp(a, M)
    if b is None:
        return a, M
    b = normalize(b, n)
    a_new = (1 - t) * a + t * b
    a_new = normalize(a_new, n)
    c = fft_convolve(a_new)
    M_new = float(np.max(c))
    return a_new, M_new
```

```python
    Minimize M s.t. (a*b)_k <= M for all k, sum(b) = n, b >= 0.
    Returns (b_opt, M_opt).

    c_obj[n] = 1  # minimize M

    A_ineq[:, n] = -1  # subtract M

    A_eq[0, n] = 0
    b_eq = np.array([n])
    
    bounds = [(0, None)] * n + [(0, None)]  # b_i >= 0, M >= 0
```

```python
import numpy as np
```

```python
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:L]
    np.maximum(c, 0, out=c)
    return c

def build_A(a):
    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n))
    for j in range(n):
        A[j:j+n-1, j] = a
    return A

def solver_lp_min_peak(a, n):
    L = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = build_A(a)
    A_ub[:, n] = -1.0
    b_ub = np.zeros(L)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(n)])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x[:n], float(res.x[n])
    return None, float('inf')

def fast_score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))

def lp_optimize(a, n, deadline, verbose=True):
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a *= n / s
    
    conv = fft_conv(a)
    M = float(np.max(conv))
    best_a = a.copy()
    best_M = M
    t = 0.1
    stagnant = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        b, M_opt = solver_lp_min_peak(a, n)
        if b is None:
            break
        
        if M_opt >= M * (1 - 1e-10):
            stagnant += 1
            if stagnant > 10:
                break
            continue
        
        # Try the update
        for trial_t in [t, t * 0.5, t * 2.0]:
            trial_t = max(0.01, min(0.8, trial_t))
            a_new = (1 - trial_t) * a + trial_t * b
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new *= n / sn
            M_new = float(np.max(fft_conv(a_new)))
            
            if M_new < best_M:
                best_a = a_new.copy()
                best_M = M_new
            
            if M_new < M:
                a = a_new
                M = M_new
                stagnant = 0
                t = min(0.5, t * 1.05)
                break
        else:
            t = max(0.01, t * 0.5)
            stagnant += 1
            if stagnant > 30:
                break
        
        if verbose and it % 100 == 0:
            sc = 2.0 * M / n
            print(f"  LP iter {it}: M={M:.4f}, score≈{sc:.6f}")
    
    return best_a, best_M
```

```python
"""
Optimizes non-negative step-function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding, L-BFGS with log-sum-exp smoothing, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft


def fft_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def build_conv_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A


def fast_score(a):
    """Fast score computation (scale-invariant part)."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    if c.size == 0:
        return float('inf')
    return float(2.0 * n * float(np.max(c)) / (s * s))


def normalize(a, target=None):
    """Clip to [0, inf] and scale to target sum."""
    a = np.asarray(a, dtype=np.float64)
    if target is None:
        target = float(a.size)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.full(a.size, target / a.size, dtype=np.float64)
    return a * (target / s)


def solve_lp(a, n):
    """
    Solve: min M s.t. (a*b)_k <= M for all k, sum(b)=n, b>=0.
    Returns (b_opt, M_opt).
    """
    L = 2 * n - 1
    A_base = build_conv_matrix(a)
    
    # Variables: b_0,...,b_{n-1}, M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = A_base
    A_ub[:, n] = -1.0
    b_ub = np.zeros(L)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(n)])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9})
    
    if res.success and res.x is not None:
        return np.maximum(res.x[:n], 0.0), float(max(res.x[n], 0.0))
    return None, float('inf')


def lp_optimize(a, n, deadline, verbose=True):
    """Iterate LP-based updates to reduce the convolution peak."""
    a = normalize(a.copy(), n)
    conv = fft_conv(a)
    M = float(np.max(conv))
    best_a = a.copy()
    best_M = M
    t = 0.1
    stagnant = 0
    it = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        it += 1
        b, M_opt = solve_lp(a, n)
        if b is None:
            break
        
        if M_opt >= M * (1 - 1e-9):
            stagnant += 1
            if stagnant > 15:
                break
            continue
        
        b = normalize(b, n)
        
        # Compute safe step size
        conv_b = fft_conv(b)
        M_bb = float(np.max(conv_b))
        
        # Find max t such that upper bound on new peak < M
        # (1-t)^2*M + 2t(1-t)*M_opt + t^2*M_bb < M
        # t^2*(M - 2*M_opt + M_bb) + t*(2*M_opt - 2*M) < 0
        # For t>0: t*(M - 2*M_opt + M_bb) < 2*(M - M_opt)
        
        num = 2.0 * (M - M_opt)
        den = M - 2.0 * M_opt + M_bb
        
        if den > 1e-10:
            t_max = num / den
        else:
            t_max = 1.0
        
        t_try = min(t, max(0.01, 0.7 * t_max))
        
        a_new = (1 - t_try) * a + t_try * b
        a_new = normalize(a_new, n)
        M_new = float(np.max(fft_conv(a_new)))
        
        if M_new < M:
            a = a_new
            M = M_new
            if M_new < best_M:
                best_a = a.copy()
                best_M = M_new
            stagnant = 0
            t = min(0.5, t * 1.03)
        else:
            t = max(0.01, t * 0.5)
            stagnant += 1
            if stagnant > 40:
                break
        
        if verbose and it % 200 == 0:
            sc = 2.0 * best_M / n
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it} ({elapsed:.0f}s): best_M={best_M:.4f}, score≈{sc:.6f}")
    
    return best_a, best_M
```

```python
def smooth_score_and_grad(x, beta):
    """Compute smooth score and gradient with respect to log-parameters x."""
    a = np.exp(x - np.max(x))  # for numerical stability
    n = len(a)
    s = float(a.sum())
    c = fft_conv(a)
    M = float(np.max(c))
    
    if M < 1e-30 or s < 1e-30:
        return 0.0, np.zeros(n)
    
    # Score
    score = 2.0 * n * M / (s * s)
    
    # Smooth max weights
    c_shifted = c - M
    max_exp = np.max(beta * c_shifted)
    if max_exp > 50:  # numerical stability
        weights = np.exp(beta * c_shifted - max_exp)
        weights /= weights.sum()
    else:
        weights = np.exp(beta * c_shifted)
        weights /= weights.sum()
    
    # Gradient of smooth max: grad_S[i] = 2 * sum_k weights[k] * a[k-i]
    # Using matrix trick
    # grad_S[i] = 2 * dot(weights[i:i+n], a)
    if weights.size >= n:
        # Build the view
        strides = (weights.strides[0], weights.strides[0])
        W_mat = np.lib.stride_tricks.as_strided(weights, shape=(n, n), strides=strides)
        grad_S = 2.0 * (W_mat @ a)
    else:
        grad_S = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                grad_S[j] += wk * a[idx]
        grad_S *= 2.0
    
    # d(score)/d(a_i) = 2n*grad_S_i/s^2 - 2n*M/s^3
    d_score_da = 2.0 * n * grad_S / (s * s) - 2.0 * n * M / (s * s * s)
    
    # d(score)/d(x_i) = d_score_da_i * a_i
    d_score_dx = d_score_da * a
    
    return score, d_score_dx


def lbfgs_optimize(a0, deadline, verbose=True):
    """Optimize using L-BFGS on the smooth max with log-parameters."""
    a0 = normalize(a0, float(a0.size))
    n = len(a0)
    x0 = np.log(np.maximum(a0, 1e-10))
    best_a = a0.copy()
    best_score = fast_score(a0)
    
    t0 = time.monotonic()
    
    # Continuation on beta
    betas = [5, 10, 20, 50, 100, 200, 500]
    x = x0.copy()
    
    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        
        # Run L-BFGS for this beta
        max_iter = 50
        x, func, info = minimize(
            lambda xx: smooth_score_and_grad(xx, beta),
            x, jac=True, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-6,
                     'maxls': 50}
        )
        
        a = np.exp(x - x.max())
        a = normalize(a, n)
        sc = fast_score(a)
        
        if sc < best_score:
            best_a = a.copy()
            best_score = sc
            x = np.log(np.maximum(a, 1e-10))
        
        if verbose:
            print(f"    L-BFGS beta={beta}: score={best_score:.6f}")
        
        # Reset x for next beta (with small perturbation)
        if time.monotonic() > deadline - 30:
            break
    
    return best_a, best_score
```

def coordinate_descent(a, budget_s, verbose=True):
    """Coordinate descent: try small multiplicative changes to each element."""
    a = normalize(a.copy(), float(a.size))
    n = len(a)
    best_score = fast_score(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    delta = 0.05
    
    pass_count = 0
    while time.monotonic() < t0 + budget_s and delta > 0.001:
        pass_count += 1
        improved_any = False
        
        for i in range(n):
            if time.monotonic() > t0 + budget_s:
                break
            
            # Try increasing
            a[i] *= (1 + delta)
            a[i] = max(0, a[i])
            a = normalize(a, n)
            sc = fast_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                improved_any = True

                a = normalize(a, n)
            
            # Try decreasing
