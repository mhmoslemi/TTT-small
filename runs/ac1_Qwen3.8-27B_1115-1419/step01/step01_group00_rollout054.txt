    # Full softmax smooth first
    a_work, o = _smooth_run(a_work, n_work, min(t_start + 260.0, p2_end), "smooth_full",
                            tau_init=n_work * 0.3, tau_min=n_work * 0.005,
                            eta_init=0.1, top_frac=1.0, tau_decay=0.99997)
    _track(a_work, n_work, "smooth_full")

    # Multi-peak smooth
    a_work, o = _smooth_run(a_work, n_work, min(t_start + 400.0, p2_end), "smooth_peak50",
                            tau_init=n_work * 0.1, tau_min=n_work * 0.003,
                            eta_init=0.08, top_frac=0.5, tau_decay=0.99998)
    _track(a_work, n_work, "smooth_peak50")

    # Multi-peak (top 10%)
    a_work, o = _smooth_run(a_work, n_work, p2_end, "smooth_peak10",
                            tau_init=n_work * 0.05, tau_min=n_work * 0.001,
                            eta_init=0.06, top_frac=0.1, tau_decay=0.99999)
    _track(a_work, n_work, "smooth_peak10")

    # === Phase 3: Explore alternative n values (200s) ===
    p3_end = min(t_start + 660.0, deadline - 400.0)
    print(f"Phase 3: Alternative n exploration")
    for n_alt in [5000, 7000, 4500]:
        if time.monotonic() > p3_end:
            break
        a_alt = _structured_init(n_alt, seed + 200 + n_alt // 100, 4)
        sub_end = min(time.monotonic() + 40.0, p3_end)
        a_alt, o_alt = _smooth_run(a_alt, n_alt, sub_end, f"alt{n_alt}",
                                   tau_init=n_alt * 0.2, eta_init=0.08)
        sub_end2 = min(time.monotonic() + 20.0, p3_end)
        a_alt, o_alt = _hard_run(a_alt, n_alt, sub_end2, f"altH{n_alt}")
        _track(a_alt, n_alt, f"n={n_alt}")

    # === Phase 4: Spectral perturbation + refine loop (remaining bulk) ===
    p4_end = deadline - 120.0
    print(f"Phase 4: Spectral perturb + refine loop")
    a_main = best_a.copy()
    n_main = best_n
    attempt = 0

    while time.monotonic() < p4_end:
        attempt += 1
        if attempt % 10 == 0:
            print(f"  Loop iter {attempt}, best={best_obj:.6f}, t={time.monotonic()-t_start:.0f}s")

        # Choose perturbation parameters
        if attempt <= 10:
            n_freqs = rng.integers(3, 8)
            amp = rng.uniform(0.02, 0.08)
        elif attempt <= 30:
            n_freqs = rng.integers(5, 15)
            amp = rng.uniform(0.05, 0.15)
        else:
            n_freqs = rng.integers(10, 25)
            amp = rng.uniform(0.1, 0.3)

        a_pert = _spectral_perturb(a_main, n_main, rng, n_freqs, amp)

        # Quick smooth refinement
        sub_time = min(25.0, (p4_end - time.monotonic()) * 0.4)
        a_ref, o_ref = _smooth_run(a_pert, n_main, time.monotonic() + sub_time,
                                   f"sp{attempt}", tau_init=n_main * 0.1,
                                   eta_init=0.06, top_frac=0.3,
                                   tau_decay=0.99998)
        if o_ref < best_obj:
            _track(a_ref, n_main, f"perturb{attempt}")
            a_main = a_ref.copy()
        else:
            # Keep main as is, but update if close
            if o_ref < best_obj + 0.001:
                a_main = a_ref.copy()

        # Occasional hard refinement
        if attempt % 3 == 0 and time.monotonic() < p4_end - 30:
            sub_end_h = min(time.monotonic() + 15.0, p4_end - 5.0)
            a_h, o_h = _hard_run(a_main, n_main, sub_end_h, f"hard{attempt}",
                                 eta_init=0.01, top_k=2)
            if o_h < best_obj:
                _track(a_h, n_main, f"hard{attempt}")
                a_main = a_h.copy()

        # Occasional full restart with different structure
        if attempt in (20, 50, 80, 120):
            a_new = _structured_init(n_main, seed + 500 + attempt, rng.integers(0, 7))
            sub_end_r = min(time.monotonic() + 20.0, p4_end - 10.0)
            a_new, o_new = _smooth_run(a_new, n_main, sub_end_r, f"restart{attempt}",
                                       tau_init=n_main * 0.25, eta_init=0.1)
            _track(a_new, n_main, f"restart{attempt}")

    # === Phase 5: Final hard polish (120s) ===
    print(f"Phase 5: Final hard polish")
    final_end = deadline
    a_final = best_a.copy()
    n_final = best_n

    # Hard on main peak
    a_f1, o_f1 = _hard_run(a_final, n_final, min(time.monotonic() + 50.0, final_end),
                           "finalH1", eta_init=0.008, top_k=1)
    if o_f1 < best_obj:
        _track(a_f1, n_final, "finalH1")
        a_final = a_f1.copy()

    # Multi-peak smooth (very small tau)
    a_f2, o_f2 = _smooth_run(a_final, n_final, min(time.monotonic() + 40.0, final_end),
                             "finalS", tau_init=n_final * 0.01, tau_min=n_final * 0.0005,
                             eta_init=0.03, top_frac=0.05, tau_decay=0.99999)
    if o_f2 < best_obj:
        _track(a_f2, n_final, "finalS")
        a_final = a_f2.copy()

    # Final hard
    a_f3, o_f3 = _hard_run(a_final, n_final, final_end - 0.5, "finalH3",
                           eta_init=0.005, top_k=1)
    if o_f3 < best_obj:
        _track(a_f3, n_final, "finalH3")

    # === Validate and return ===
    result = [float(max(0.0, x)) for x in a_final]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    if not np.isfinite(sc):
        result = [1.0] * n_final
        sc = float(evaluate_sequence(result))
    elapsed = time.monotonic() - t_start
    print(f"FINAL: obj={best_obj:.6f} evaluated={sc:.6f} n={n_final} t={elapsed:.0f}s")
    return result
