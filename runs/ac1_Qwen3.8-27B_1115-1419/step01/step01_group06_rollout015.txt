```
def self_gradient(p, A, weights, N, phase, n):
    Q = np.fft.rfft(weights, N)
    Ar = phase * np.conjugate(A)
    g = 2.0 * np.fft.irfft(Q * Ar, N)
    start = n - 1
    stop = 2*n - 1
    if g.size >= stop:
       g = g[start:stop]
    else:
       # pad fallback
       tmp=np.zeros(max(stop,n), dtype=np.float64)
       tmp[:min(g.size,stop)] = g[:min(g.size,stop)]
       g=tmp[start:stop] if tmp.size>=stop else tmp
    g=np.where(np.isfinite(g),g,0.0)
    if g.size < n:
       g=np.concatenate((g, np.zeros(n-g.size)))
    return g[:n]
```

```python
"""
Search for nonnegative sequences minimizing

    2 * n * max(np.convolve(a, a)) / sum(a)**2

by working in probability normalization sum(a)=1.

Main ideas:
- Use relative-temperature log-sum-exp gradients for the maximum convolution
  coefficient.  This is much stronger than absolute-beta smoothing for large n.
- Optimize in log-probability coordinates with softmax parameterization, mirror
  gradient descent, Adam direction, and short line searches.
- Periodically solve/approximate the fixed-factor response problem
      min_{g} max_k (f*g)_k
  and blend f with the response.  This is the convex subproblem used by the
  previous LP-based state of the art, implemented here with more time allocation
  and sharper relative-beta weights.
- Start from multiple basins: the provided parent sequence, one-sided endpoint
  beta-like power laws, cosine/mixing starts, and smooth random log-weights.
- Run a secondary length if time permits, then exact-validate the best FFT
  candidates with numpy.convolve before returning.
"""

import numpy as np
import time

MAX_LEN = 8192
EPS = 1e-300


# ---------------------------------------------------------------------------
# Basic probability/log-space utilities
# ---------------------------------------------------------------------------

def clean_prob(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(arr.size, dtype=np.float64)
        s = float(arr.size)
    return arr / s


def log_prob(a):
    a = clean_prob(a)
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w = np.where(np.isfinite(w), w, -700.0)
    mx = float(np.max(w)) if w.size else 0.0
    if np.isfinite(mx):
        w -= mx
    else:
        w = np.zeros_like(w)
    return w


def prob_from_log(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size == 0:
        return np.ones(1, dtype=np.float64)
    w = np.where(np.isfinite(w), w, -700.0)
    mx = float(np.max(w)) if w.size else 0.0
    if np.isfinite(mx):
        w -= mx
    else:
        w = np.zeros_like(w)
    e = np.exp(np.clip(w, -745.0, 0.0))
    s = float(e.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones(w.size, dtype=np.float64)
        s = float(w.size)
    return e / s


def clean_log(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    w = np.where(np.isfinite(w), w, -700.0)
    mx = float(np.max(w)) if w.size else 0.0
    if np.isfinite(mx):
        w -= mx
    else:
        w = np.zeros_like(w)
    return w


def next_fast_len(m):
    m = int(max(1, m))
    p = 1
    while p < m:
        p <<= 1
    return p


# ---------------------------------------------------------------------------
# FFT-based objective and gradients
# ---------------------------------------------------------------------------

def fft_score_and_b(p, N):
    p = clean_prob(p)
    n = int(p.size)
    A = np.fft.rfft(p, N)
    b = np.fft.irfft(A * A, N)
    b = b[: 2 * n - 1]
    b = np.maximum(b, 0.0)
    b = np.where(np.isfinite(b), b, 0.0)
    if b.size == 0:
        return A, b, float("inf"), 0.0
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return A, b, float("inf"), 0.0
    score = 2.0 * float(n) * mx
    return A, b, score, mx


def fft_score_only(p, N):
    _, _, sc, _ = fft_score_and_b(p, N)
    return sc


def smoothed_weights(vals, beta):
    """
    Relative-temperature weights:
        q_s proportional exp(beta * (vals_s - max(vals)) / max(vals)).
    """
    vals = np.asarray(vals, dtype=np.float64).ravel()
    L = vals.size
    if L == 0:
        return np.ones(1, dtype=np.float64)
    if beta <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)

    mx = float(np.max(vals)) if vals.size else 0.0
    if not np.isfinite(mx) or mx <= EPS:
        return np.full(L, 1.0 / float(L), dtype=np.float64)

    z = float(beta) * (vals - mx) / mx
    z = np.where(np.isfinite(z), z, -745.0)
    z = np.minimum(z, 0.0)
    e = np.exp(np.clip(z, -745.0, 0.0))
    s = float(e.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= EPS:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    return e / s


def _extract_slice(deg, n):
    start = n - 1
    stop = 2 * n - 1
    if deg.size >= stop:
        deg = deg[start:stop]
    else:
        tmp = np.zeros(max(stop, n), dtype=np.float64)
        if deg.size:
            take = min(deg.size, stop)
            tmp[:take] = deg[:take]
        deg = tmp[start:stop] if tmp.size >= stop else tmp
    deg = np.where(np.isfinite(deg), deg, 0.0)
    if deg.size < n:
        deg = np.concatenate((deg, np.zeros(n - deg.size, dtype=np.float64)))
    return deg[:n]


def self_gradient(p, A, weights, N, phase, n):
    Q = np.fft.rfft(weights, N)
    Ar = phase * np.conjugate(A)
    g = 2.0 * np.fft.irfft(Q * Ar, N)
    return _extract_slice(g, n)


def response_gradient(weights, F_rev, N, n):
    Q = np.fft.rfft(weights, N)
    g = 2.0 * np.fft.irfft(Q * F_rev, N)
    return _extract_slice(g, n)


def normalize_gradient(gl, rng):
    gl = np.asarray(gl, dtype=np.float64).ravel()
    gl = np.where(np.isfinite(gl), gl, 0.0)
    gl = gl - float(np.mean(gl))
    rms = float(np.sqrt(np.mean(gl * gl)))
    if not np.isfinite(rms) or rms < 1e-12:
        gl = rng.normal(0.0, 1.0, size=gl.size)
        gl = gl - float(np.mean(gl))
        rms = float(np.sqrt(np.mean(gl * gl)))
        if not np.isfinite(rms) or rms < 1e-12:
            return gl
    return gl / rms


def adam_update(gl, m, v, t, lr):
    b1 = 0.9
    b2 = 0.999
    m = b1 * m + (1.0 - b1) * gl
    v = b2 * v + (1.0 - b2) * (gl * gl)
    t += 1
    mhat = m / (1.0 - (b1 ** t))
    vhat = v / (1.0 - (b2 ** t))
    upd = mhat / (np.sqrt(vhat) + 1e-12)
    upd = np.where(np.isfinite(upd), upd, 0.0)
    upd = np.clip(upd, -10.0, 10.0)
    delta = -float(lr) * upd
    return delta, m, v, t


# ---------------------------------------------------------------------------
# Exact score for final validation
# ---------------------------------------------------------------------------

def exact_score_array(p):
    p = clean_prob(p)
    n = int(p.size)
    if n == 0:
        return float("inf")
    b = np.convolve(p, p, mode="full")
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(p.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= EPS:
        return float("inf")
    val = float(2.0 * n * float(np.max(b)) / (s * s))
    return val if np.isfinite(val) else float("inf")

# Resampling / parent loading
# ---------------------------------------------------------------------------

def cdf_resample(a, m):
    p = clean_prob(a)
    L = int(p.size)
    m = int(max(1, m))
    if L == m:
        return p
    c = np.cumsum(p)
    s = float(c[-1])
    if not np.isfinite(s) or s <= EPS:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    c = c / s
    c_ext = np.empty(L + 1, dtype=np.float64)
    c_ext[0] = 0.0
    c_ext[1:] = c
    c_ext[-1] = 1.0
    x = np.arange(L + 1, dtype=np.float64)
    u = np.linspace(0.0, 1.0, m + 1, dtype=np.float64)
    y = np.interp(u, c_ext, x, left=0.0, right=float(L))
    masses = np.diff(y)
    masses = np.where(np.isfinite(masses), masses, 0.0)
    masses = np.maximum(masses, 0.0)
    sm = float(masses.sum(dtype=np.float64))
    if not np.isfinite(sm) or sm <= EPS:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    return masses / sm


def load_parent_sequence():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 16:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= EPS:
        return None
    arr = arr / s
    if arr.size > MAX_LEN:
        arr = cdf_resample(arr, MAX_LEN)
    return arr


def choose_main_length(parent_p):
    if parent_p is not None:
        n = int(parent_p.size)
        if 2048 <= n <= MAX_LEN:
            return n
        if n > MAX_LEN:
            return MAX_LEN
        if n < 2048:
            return 4096
    return 4096


def choose_secondary_length(n_main):
    n_main = int(n_main)
    if n_main >= 7000:
        return 4096
    return 8192

def make_power(n, left_exp, right_exp, eps=0.5, reverse=False, zero_first=False):
    n = int(max(1, n))
    i = np.arange(n, dtype=np.float64)
    left_pos = i + float(eps)
    right_pos = float(n) - i - float(eps)
    if left_exp != 0.0:
        left = np.power(np.maximum(left_pos, EPS), -float(left_exp))
    else:
        left = np.ones(n, dtype=np.float64)
    if right_exp != 0.0:
        right = np.power(np.maximum(right_pos, EPS), -float(right_exp))
    else:
        right = np.ones(n, dtype=np.float64)
    if reverse:
        left, right = right, left
    p = left * right
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    if zero_first and p.size > 0:
        p[0] = 0.0
    return clean_prob(p)


def mix_prob(a, b, w):
    a = clean_prob(a)
    b = clean_prob(b)
    if a.size != b.size:
        b = cdf_resample(b, a.size)
    out = (1.0 - float(w)) * a + float(w) * b
    return clean_prob(out)


def make_cos(n, k, r, phi):
    n = int(max(16, int(n)))
    t = np.arange(n, dtype=np.float64) / float(n)
    p = 1.0 + float(r) * np.cos(2.0 * np.pi * float(k) * t + float(phi))
    p = np.maximum(p, 1e-6)
    return clean_prob(p)


def make_random_smooth(n, rng, sigma=0.35):
    n = int(max(128, int(n)))
    L = next_fast_len(n)
    w = rng.normal(0.0, 1.0, size=n)
    W = np.fft.rfft(w, L)
    freq = np.arange(W.size, dtype=np.float64) / float(L)
    decay = np.exp(-(freq * freq) / (2.0 * float(sigma) * float(sigma)))
    W = W * decay
    w2 = np.fft.irfft(W, L)[:n]
    w2 = w2 + 0.05 * rng.normal(0.0, 1.0, size=n)
    return prob_from_log(w2)


def probs_similar(a, b):
    a = clean_prob(a)
    b = clean_prob(b)
    if a.size != b.size:
        return False
    d = a - b
    rms = float(np.sqrt(np.mean(d * d)))
    return np.isfinite(rms) and rms < 1e-4


def sort_by_score(cands):
    src = list(cands)
    res = []
    while src:
        best_i = 0
        best_sc = float("inf")
        for i in range(len(src)):
            sc = float(src[i][0])
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_i = i
        res.append(src.pop(best_i))
    return res


def pool_top(cands, k):
    src = list(cands)
    if len(src) <= k:
        return src
    res = []
    while src and len(res) < k:
        best_i = 0
        best_sc = float("inf")
        for i in range(len(src)):
            sc = float(src[i][0])
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_i = i
        res.append(src.pop(best_i))
    return res


def build_initial_pool(parent_p, n, rng, N, max_pool=36):
    pool = []
    n = int(max(16, int(n)))

    if parent_p is not None:
        p = clean_prob(parent_p)
        sc = fft_score_only(p, N)
        if np.isfinite(sc):
            pool.append((sc, p.copy()))

    uniform = np.full(n, 1.0 / float(n), dtype=np.float64)

    specs = []
    specs.append((0.5, 0.0, 0.5, False, False))   # one-sided singular
    specs.append((0.5, 0.0, 0.5, True, False))    # reversed one-sided
    specs.append((0.5, 0.0, 0.5, False, True))    # omit first bin
    specs.append((0.5, 0.0, 0.5, True, True))
    specs.append((0.25, 0.0, 0.5, False, False))
    specs.append((0.75, 0.0, 0.5, False, False))
    specs.append((0.5, 0.25, 0.5, False, False))
    specs.append((0.25, 0.5, 0.5, False, False))
    specs.append((0.5, 0.5, 0.5, False, False))
    specs.append((0.0, 0.5, 0.5, False, False))
    specs.append((0.35, 0.15, 0.5, False, False))
    specs.append((0.15, 0.35, 0.5, False, False))

    for le, re, eps, rev, zf in specs:
        p = make_power(n, le, re, eps, rev, zf)
        sc = fft_score_only(p, N)
        if np.isfinite(sc):
            pool.append((sc, p.copy()))
        for w in (0.15, 0.35):
            pm = mix_prob(p, uniform, w)
            sc = fft_score_only(pm, N)
            if np.isfinite(sc):
                pool.append((sc, pm.copy()))

    # Low-dimensional oscillatory starts.
    for k in (1, 2, 3, 5, 8):
        for r in (0.4, 0.7):
            phi = float(rng.uniform(0.0, 2.0 * np.pi))
            p = make_cos(n, k, r, phi)
            sc = fft_score_only(p, N)
            if np.isfinite(sc):
                pool.append((sc, p.copy()))
            pm = mix_prob(p, uniform, 0.25)
            sc = fft_score_only(pm, N)
            if np.isfinite(sc):
                pool.append((sc, pm.copy()))

    # Smooth random log-weight starts.
    for _ in range(4):
        p = make_random_smooth(n, rng, 0.35)
        sc = fft_score_only(p, N)
        if np.isfinite(sc):
            pool.append((sc, p.copy()))

    # Blend the two best distinct starts.
    small = pool_top(pool, 8)
    if len(small) >= 2:
        pa = small[0][1]
        pb = small[1][1]
        try:
            if not probs_similar(pa, pb):
                pm = mix_prob(pa, pb, 0.35)
                sc = fft_score_only(pm, N)
                if np.isfinite(sc):
                    pool.append((sc, pm.copy()))
        except Exception:
            pass

    if len(pool) > max_pool:
        pool = pool_top(pool, max_pool)
    return pool


def select_starts(pool, parent_p, limit=3):
    starts = []
    if parent_p is not None:
        starts.append((float("inf"), clean_prob(parent_p)))

    pool_sorted = sort_by_score(pool)
    for sc, p in pool_sorted:
        if len(starts) >= limit:
            break
        if parent_p is not None:
            try:
                if probs_similar(p, parent_p):
                    continue
            except Exception:
                pass
        starts.append((float(sc) if np.isfinite(sc) else float("inf"), clean_prob(p)))

    while len(starts) < limit:
        n = int(parent_p.size) if parent_p is not None else 4096
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
        starts.append((float("inf"), p))
    return starts

    if p is None or not np.isfinite(score):
        return list(top)
    top = list(top)
    pa = np.asarray(p, dtype=np.float64).copy()
    if len(top) < max_items:
        top.append((float(score), pa))
        return top
    worst_i = 0
    worst_sc = float(top[0][0])
    for i in range(1, len(top)):
        sc = float(top[i][0])
        if sc > worst_sc:
            worst_sc = sc
            worst_i = i
    if float(score) < worst_sc:
        top[worst_i] = (float(score), pa)
    return top


def add_candidate(candidates, p, score, max_items=80):
    if p is None or not np.isfinite(score):
        return list(candidates)
    candidates = list(candidates)
    candidates.append((float(score), np.asarray(p, dtype=np.float64).copy()))
    if len(candidates) <= max_items:
        return candidates
    worst_i = 0
    worst_sc = float(candidates[0][0])
    for i in range(1, len(candidates)):
        sc = float(candidates[i][0])
        if sc > worst_sc:
            worst_sc = sc
            worst_i = i
    candidates.pop(worst_i)
    return candidates


# ---------------------------------------------------------------------------
# Response subproblem
# ---------------------------------------------------------------------------

def try_response(p, A, N, phase, rng, beta, max_time_s, base_score):
    n = int(p.size)
    F_rev = phase * np.conjugate(A)
    g = p.copy()
    w_g = log_prob(g)
    m_g = np.zeros(n, dtype=np.float64)
    v_g = np.zeros(n, dtype=np.float64)
    t_g = 0
    lr = 0.08
    stall = 0

    best_cand = p.copy()
    best_sc = float(base_score) if np.isfinite(base_score) else float("inf")

    start = time.monotonic()
    deadline = start + max(0.0, float(max_time_s))
    inner = 0

    while time.monotonic() < deadline:
        inner += 1
        G = np.fft.rfft(g, N)
        h = np.fft.irfft(A * G, N)[: 2 * n - 1]
        h = np.maximum(h, 0.0)
        h = np.where(np.isfinite(h), h, 0.0)
        maxh = float(np.max(h)) if h.size else 0.0
        if not np.isfinite(maxh):
            break

        q = smoothed_weights(h, beta)
        grad = response_gradient(q, F_rev, N, n)
        gl = g * (grad - float(np.dot(g, grad)))
        gl = normalize_gradient(gl, rng)

        delta, m_g, v_g, t_g = adam_update(gl, m_g, v_g, t_g, lr)
        acc = False

        for scale in (1.0, 0.5, 0.25):
            w2 = clean_log(w_g + scale * delta)
            p2 = prob_from_log(w2)
            G2 = np.fft.rfft(p2, N)
            h2 = np.fft.irfft(A * G2, N)[: 2 * n - 1]
            h2 = np.maximum(h2, 0.0)
            h2 = np.where(np.isfinite(h2), h2, 0.0)
            sc2 = float(np.max(h2)) if h2.size else 0.0
            if np.isfinite(sc2) and sc2 < maxh - 1e-14:
                g = p2
                w_g = w2
                h = h2
                maxh = sc2
                acc = True
                break

        if acc:
            stall = 0
            lr = min(0.25, lr * 1.02)
            beta = min(300.0, beta * 1.05)
        else:
            stall += 1
            lr = max(0.01, lr * 0.85)
            if stall >= 20:
                break

        if inner % 3 == 0 or inner == 1:
            for tv in (0.25, 0.5, 0.75, 1.0):
                cand = (1.0 - tv) * p + tv * g
                cand = clean_prob(cand)
                _, _, csc, _ = fft_score_and_b(cand, N)
                if np.isfinite(csc) and csc < best_sc:
                    best_sc = csc
                    best_cand = cand.copy()

    return best_cand, best_sc


# ---------------------------------------------------------------------------
# Main local optimizer
# ---------------------------------------------------------------------------

def local_optimize(p0, N, phase, rng, budget_s, name="local"):
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s))
    if total_budget <= 1.0:
        p = clean_prob(p0)
        _, _, sc, _ = fft_score_and_b(p, N)
        top = add_to_top([], p, sc, 8)
        return p, sc, top

    deadline = start + total_budget - 1.5

    p = clean_prob(p0)
    n = int(p.size)
    if n < 128:
        p = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
        n = 1024

    A, b, score, _ = fft_score_and_b(p, N)
    if not np.isfinite(score):
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
        A, b, score, _ = fft_score_and_b(p, N)

    best_p = p.copy()
    best_score = float(score)
    top = add_to_top([], best_p, best_score, 8)

    w = log_prob(p)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    beta = 6.0
    beta_max = 300.0
    lr = 0.08
    lr_start = 0.08
    stall = 0
    outer = 0
    last_response = start
    response_cd = min(60.0, max(15.0, 0.01 * total_budget))

    while time.monotonic() < deadline:
        outer += 1

        # Reset if numerical state broke.
        if (not np.isfinite(score) or
                not np.all(np.isfinite(p)) or
                not np.all(np.isfinite(A)) or
                not np.all(np.isfinite(b))):
            p = best_p.copy()
            n = int(p.size)
            A, b, score, _ = fft_score_and_b(p, N)
            if not np.isfinite(score):
                p = np.full(n, 1.0 / float(n), dtype=np.float64)
                A, b, score, _ = fft_score_and_b(p, N)
            w = log_prob(p)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t = 0
            stall = 0
            continue

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = float(score)
            best_p = p.copy()
            top = add_to_top(top, best_p, best_score, 8)

        weights = smoothed_weights(b, beta)

        # Adapt relative temperature to keep an effective active set size.
        ssq = float(np.dot(weights, weights))
        if ssq > EPS:
            ess = 1.0 / ssq
            target_ess = max(10.0, 0.01 * float(n))
            if beta > 1.0:
                if ess < 0.25 * target_ess:
                    beta = max(1.0, beta * 0.7)
                elif outer % 40 == 0:
                    beta = min(beta_max, beta * 1.08 + 0.1)

        g = self_gradient(p, A, weights, N, phase, n)
        gl = p * (g - float(np.dot(p, g)))
        gl = normalize_gradient(gl, rng)

        delta, m, v, t = adam_update(gl, m, v, t, lr)
        accepted = False

        for scale in (1.0, 0.65, 0.4, 0.22):
            w2 = clean_log(w + scale * delta)
            p2 = prob_from_log(w2)
            A2, b2, sc2, _ = fft_score_and_b(p2, N)
            if np.isfinite(sc2) and sc2 < score - 1e-12:
                p = p2
                A = A2
                b = b2
                score = sc2
                w = w2
                accepted = True
                break

        if accepted:
            stall = 0
            if score < best_score - 1e-12:
                best_score = float(score)
                best_p = p.copy()
                top = add_to_top(top, best_p, best_score, 8)
            lr = min(0.25, lr * 1.02)
        else:
            stall += 1
            lr = max(0.005, lr * 0.87)
            if stall % 6 == 0:
                beta = max(1.0, beta * 0.75)
            if stall >= 12:
                m = m * 0.5
                v = v * 0.5

        # Periodic fixed-factor response move.
        if stall >= 18 and (time.monotonic() - last_response) >= response_cd:
            last_response = time.monotonic()
            rem = deadline - time.monotonic()
            if rem > 30.0:
                resp_time = min(
                    35.0,
                    max(5.0, rem * 0.02),
                    total_budget * 0.02
                )
                if resp_time > 4.0:
                    resp_beta = min(250.0, max(10.0, beta))
                    rc, rs = try_response(p, A, N, phase, rng, resp_beta, resp_time, score)
                    if rc is not None and np.isfinite(rs):
                        if rs < score - 1e-12 or rs < best_score - 1e-12:
                            p = clean_prob(rc)
                            n = int(p.size)
                            A, b, score, _ = fft_score_and_b(p, N)
                            w = log_prob(p)
                            m = np.zeros(n, dtype=np.float64)
                            v = np.zeros(n, dtype=np.float64)
                            t = 0
                            lr = max(0.02, min(0.15, lr_start * 0.9))
                            beta = min(80.0, max(2.0, beta * 0.5))
                            stall = 0
                            if np.isfinite(score) and score < best_score - 1e-12:
                                best_score = float(score)
                                best_p = p.copy()
                            top = add_to_top(top, p, score, 8)

        # Reheat from best when stuck.
        if stall >= 55:
            if rng.random() < 0.7:
                w = log_prob(best_p) + rng.normal(0.0, 0.12, size=n)
            else:
                w = log_prob(best_p)
            p = prob_from_log(w)
            A, b, score, _ = fft_score_and_b(p, N)
            if not np.isfinite(score):
                p = best_p.copy()
                n = int(p.size)
                A, b, score, _ = fft_score_and_b(p, N)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t = 0
            lr = max(0.015, lr_start * 0.5)
            beta = min(50.0, max(2.0, beta * 0.5))
            stall = 0

        if outer % 2000 == 0:
            print(
                f"{name}: elapsed={time.monotonic()-start:.1f}s "
                f"best={best_score:.6f} current={score:.6f} beta={beta:.1f}"
            )

    if np.isfinite(score) and score < best_score:
        best_score = float(score)
        best_p = p.copy()

    best_p = clean_prob(best_p)
    _, _, final_sc, _ = fft_score_and_b(best_p, N)
    if np.isfinite(final_sc):
        best_score = float(final_sc)
        best_p = best_p.copy()

    top = add_to_top(top, best_p, best_score, 8)
    print(f"{name}: done best={best_score:.6f} len={best_p.size}")
    return best_p, best_score, top


# ---------------------------------------------------------------------------
# Chunk runner
# ---------------------------------------------------------------------------

def run_chunk(p, rng, budget_s, candidates):
    if budget_s <= 5.0 or p is None:
        return candidates

    p = clean_prob(p)
    n = int(p.size)
    if n < 128:
        p = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
        n = 1024
    if n > MAX_LEN:
        p = cdf_resample(p, MAX_LEN)
        n = MAX_LEN

    N = next_fast_len(3 * n - 2)
    phase = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / float(N))

    best_p, best_score, top = local_optimize(p, N, phase, rng, budget_s, name="chunk")
    candidates = add_candidate(candidates, best_p, best_score)
    for sc, pa in top:
        candidates = add_candidate(candidates, pa, sc)
    return candidates


def make_chunk_plan(starts, has_parent, total_budget):
    chunks = []
    if not starts:
        return chunks
    if has_parent:
        shares = [0.42, 0.32, 0.26]
    else:
        shares = [0.45, 0.33, 0.22]

    count = min(len(starts), len(shares))
    for i in range(count):
        p = starts[i][1]
        chunks.append((p, shares[i] * max(0.0, float(total_budget))))

    used = 0.0
    for _, b in chunks:
        used += float(b)
    rem = max(0.0, float(total_budget) - used)
    if rem > 0.0 and chunks:
        add_each = rem / float(len(chunks))
        new_chunks = []
        for p, b in chunks:
            new_chunks.append((p, float(b) + add_each))
        chunks = new_chunks
    return chunks


# ---------------------------------------------------------------------------
# Final exact validation and formatting
# ---------------------------------------------------------------------------

def validate_candidates(candidates, time_budget_s):
    if not candidates:
        return None, float("inf")

    cands = sort_by_score(candidates)
    best_p = None
    best_sc = float("inf")
    start = time.monotonic()
    checked_long = 0

    for sc, p in cands:
        if p is None:
            continue
        p = clean_prob(p)
        if p.size == 0:
            continue

        rem = float(time_budget_s) - (time.monotonic() - start)
        L = int(p.size)

        if L <= 5000:
            min_rem = 3.0
        elif L <= 7000:
            min_rem = 5.0
            checked_long += 1
            if checked_long > 4:
                continue
        else:
            if checked_long >= 2:
                continue
            checked_long += 1
            min_rem = 9.0

        if rem < min_rem:
            if best_p is not None:
                break
            continue

        try:
            t0 = time.monotonic()
            val = exact_score_array(p)
            dt = time.monotonic() - t0
        except Exception:
            continue

        if np.isfinite(val) and val < best_sc:
            best_sc = float(val)
            best_p = p.copy()
            print(f"exact candidate score={best_sc:.7f} len={L} time={dt:.2f}s")

        if best_sc < 1.45 and checked_long >= 2:
            break

    return best_p, best_sc


def finalize_sequence(p):
    p = clean_prob(p)
    if p.size == 0:
        p = np.ones(256, dtype=np.float64)
    if p.size > MAX_LEN:
        p = cdf_resample(p, MAX_LEN)
    p = clean_prob(p)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    p = np.minimum(p, 1000.0)
    s = float(p.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones(p.size, dtype=np.float64)
        s = float(p.size)
    p = p / s
    return [float(x) for x in p]

# Entry point

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        total_time = float(budget_s)
    except Exception:
        total_time = 2400.0
    if not np.isfinite(total_time) or total_time <= 0.0:
        return [1.0]

    total_time = min(total_time, 2398.0)
    start_total = time.monotonic()

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    seed_int = (seed_int & 0xffffffff) or 42
    rng = np.random.default_rng(seed_int)

    parent = load_parent_sequence()
    n_main = choose_main_length(parent)
    if parent is not None and int(parent.size) != n_main:
        if parent.size > n_main:
            parent = cdf_resample(parent, n_main)
        else:
            parent = cdf_resample(parent, n_main)

    N_main = next_fast_len(3 * n_main - 2)
    phase_main = np.exp(-2j * np.pi * np.arange(N_main // 2 + 1) * (n_main - 1) / float(N_main))

    pool = build_initial_pool(parent, n_main, rng, N_main, max_pool=36)
    starts = select_starts(pool, parent, limit=3)

    reserve = 18.0 if total_time > 200 else 5.0
    search_budget = max(0.0, total_time - reserve - 3.0)
    main_fraction = 0.82 if search_budget > 1500.0 else 1.0
    main_budget = main_fraction * search_budget
    sec_budget = (1.0 - main_fraction) * search_budget
    if sec_budget < 80.0:
        main_budget = search_budget
        sec_budget = 0.0

    candidates = []
    chunk_plan = make_chunk_plan(starts, parent is not None, main_budget)

    for i, (p, b) in enumerate(chunk_plan):
        if time.monotonic() > start_total + total_time - reserve - 2.0:
            break
        names = ["main", "main_alt1", "main_alt2", "main_alt3"]
        nm = names[i] if i < len(names) else "main_extra"
        print(f"=== running chunk {nm} len={int(clean_prob(p).size)} budget={float(b):.1f}s ===")
        candidates = run_chunk(p, rng, b, candidates)

    # Secondary length, if time remains.
    if sec_budget > 80.0 and time.monotonic() < start_total + total_time - reserve - 5.0:
        n2 = choose_secondary_length(n_main)
        if n2 != n_main:
            N2 = next_fast_len(3 * n2 - 2)
            print(f"=== secondary length {n2}, N={N2}, budget={sec_budget:.1f}s ===")

            # First try resampling the best main candidate to the secondary length.
            best_main_p = None
            if candidates:
                sorted_cands = sort_by_score(candidates)
                if sorted_cands:
                    best_main_p = sorted_cands[0][1]

            used_sec = 0.0
            if best_main_p is not None:
                b1 = min(0.6 * sec_budget, sec_budget)
                p2 = cdf_resample(best_main_p, n2)
                candidates = run_chunk(p2, rng, b1, candidates)
                used_sec += b1

            rem_sec = sec_budget - used_sec
            if rem_sec > 30.0:
                p2 = make_power(n2, 0.5, 0.0, 0.5, False, False)
                candidates = run_chunk(p2, rng, rem_sec, candidates)

    if not candidates:
        p = np.full(n_main, 1.0 / float(n_main), dtype=np.float64)
        candidates = add_candidate(candidates, p, 2.0)

    # Exact validation with remaining budget.
    exact_budget = max(5.0, total_time - (time.monotonic() - start_total) - 2.0)
    best_exact, exact_score = validate_candidates(candidates, exact_budget)

    if best_exact is not None:
        ans = finalize_sequence(best_exact)
        print(f"final returning exact-validated score={exact_score:.7f} len={len(ans)}")
        return ans

    # Fallback: best FFT candidate.
    sorted_cands = sort_by_score(candidates)
    fallback_p = sorted_cands[0][1] if sorted_cands else np.full(n_main, 1.0 / float(n_main), dtype=np.float64)
    ans = finalize_sequence(fallback_p)
    print(f"final returning FFT candidate len={len(ans)}")
    return ans
```
