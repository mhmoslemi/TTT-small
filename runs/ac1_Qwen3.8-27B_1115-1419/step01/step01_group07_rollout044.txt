import time


def fft_convolve_square(a):
    """Compute a * a (linear autocorrelation via convolution) using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Fast score computation (same as evaluate_sequence but inlined)."""

    max_c = c.max()
    s = a.sum()

    momentum = momentum - momentum.mean()
    a_new = a - step * momentum
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new, momentum, c


def mwu_step(a, n_active, eta):
    """Multiplicative weight update step."""
    n = len(a)
    c = fft_convolve_square(a)
    top_k = np.argsort(c)[::-1][:n_active]
    T = np.zeros(n)
    for k in top_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        T[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    T /= n_active
    T = T - T.mean()
    a_new = a * np.exp(-eta * T)
    a_new = a_new / a_new.sum()
    return a_new

def fourier_init(n, rng):
    """Initialize using Fourier domain: try to make |DFT| flat."""
    # Target: |A(k)| = 1 for all k
    # Choose random phases
    phases = rng.uniform(0, 2*np.pi, n // 2 + 1)
    phases[0] = 0  # DC component must be real and positive
    # Conjugate symmetry
    A = np.zeros(n // 2 + 1, dtype=complex)
    A[0] = 1.0 + 0j  # DC = sum(a) = 1
    for k in range(1, n // 2 + 1):
        A[k] = np.exp(1j * phases[k])
    # Inverse DFT
    a_complex = np.fft.irfft(A, n)
    # Make non-negative
    a = np.maximum(a_complex, 0)
    a = a / a.sum()
    return a

def fourier_init_ig(n, rng, n_iter=1000):
    """Initialize using iterative Gerchberg-Saxton method."""
    a = np.ones(n) / n
    for _ in range(n_iter):
        # Forward DFT
        A = np.fft.rfft(a)
        # Project to target magnitude (flat)
        target_mag = np.ones(len(A))
        A = target_mag * A / (np.abs(A) + 1e-10)
        # Inverse DFT
        a = np.fft.irfft(A, n)
        # Project to non-negativity
        a = np.maximum(a, 0)
        a = a / a.sum()  # normalize
    return a
