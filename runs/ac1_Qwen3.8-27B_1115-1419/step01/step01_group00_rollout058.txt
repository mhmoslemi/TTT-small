"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines multi-peak smoothed gradient descent with Nesterov momentum,
LP-guided direction finding, structured perturbations, and coordinate descent polish.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def _full_conv(a):
    return np.convolve(a, a)


def _obj_val(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(_full_conv(a))) / (s * s)


def _proj_sum1(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s


def _smooth_max_grad(a, c, tau, top_k):
    """
    Compute gradient of the smoothed max of convolution.
    Uses softmax weights over top-k peaks, combined with full softmax as fallback.
    """
    n = len(a)
    # Get top-k peak indices
    if top_k >= len(c):
        peak_indices = np.arange(len(c))
    else:
        peak_indices = np.argsort(c)[::-1][:top_k]
    
    peak_vals = c[peak_indices]
    # Weights: softmax over peak values with adaptive temperature
    if len(peak_vals) > 1:
        spread = peak_vals[0] - peak_vals[-1]
        if spread < 1e-12:
            w_k = np.ones(len(peak_vals)) / len(peak_vals)
        else:
            w_k = np.exp((peak_vals - peak_vals[0]) / (spread * 0.5))
            w_k /= w_k.sum()
    else:
        w_k = np.array([1.0])
    
    g = np.zeros(n)
    for idx, pk in enumerate(peak_indices):
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        if lo <= hi:
            g[lo:hi + 1] += w_k[idx] * 2.0 * a[pk - np.arange(lo, hi + 1)]
    return g


def _hard_grad_at(a, k, n):
    """Exact gradient of conv(a,a)[k] w.r.t. a."""
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - np.arange(lo, hi + 1)]
    return g


def _smooth_phase(a, n, deadline, top_k=20, tau_init=None, tau_min_frac=0.002,

    if tau_init is None:

    tau = tau_init
    eta = eta_init
    mu = momentum_init
    best_a = a.copy()
    best_o = _obj_val(a)
    v = np.zeros(n)  # momentum velocity
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        c = _full_conv(a)
        mc = float(np.max(c))
        s = float(np.sum(a))
        o = 2.0 * n * mc / (s * s)

            best_o = o
            best_a = a.copy()

        g = _smooth_max_grad(a, c, tau, top_k)
        # Subtract mean to preserve sum
        g -= g.mean()

        if mg < 1e-14:
            break
        g = g / mg  # normalize
        
        v = mu * v + g
        a_new = a + eta * v

        it += 1
    
    return best_a, best_o

def _lp_direction(a, n, top_k_constraints=50):
    """
    Solve LP to find optimal first-order direction.
    Minimize t s.t. conv(d,a)[k] <= t - conv(a,a)[k] for top-k constraints,
    d >= -a, sum(d)=0, |d| <= 0.1.
    Returns direction d or None if LP fails.
    """
    c = _full_conv(a)
    # Select top-k constraints
    k_sel = np.argsort(c)[::-1][:top_k_constraints]
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        d_var >= -a,
        d_var <= 0.1,
        cp.sum(d_var) == 0,
        t_var >= 0,
    ]
    
    for k in k_sel:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # conv(d,a)[k] = sum_{j=lo}^{hi} d[j] * a[k-j]
            coef = a[k - np.arange(lo, hi + 1)]
            constraints.append(coef @ d_var[lo:hi + 1] <= t_var - c[k])
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                return None
    
    if d_var.value is None:
        return None
    
    d = np.array(d_var.value).flatten()
    if not np.all(np.isfinite(d)):
        return None
    return d


def _take_lp_step(a, n, d, best_o):
    """Take a step in LP direction with line search."""
    if d is None:
        return a, best_o
    d = d - d.mean()  # ensure zero-sum
    d = np.maximum(d + a, 0) - a  # ensure non-negativity of a+d
    
    for alpha in [0.5, 0.1, 0.05, 0.01]:
        a_try = _proj_sum1(a + alpha * d)
        o_try = _obj_val(a_try)
        if o_try < best_o:
            return a_try, o_try
    return a, best_o

def _perturb_structured(a, n, rng, mode):
    """Structured perturbation: add a smooth random function."""
    b = a.copy()
    if mode == 0:
        # Gaussian bump
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.02, n * 0.2)
        amp = rng.uniform(-0.1, 0.1)
        t = np.arange(n)
        bump = amp * np.exp(-0.5 * ((t - center) / width) ** 2)
        b += bump
    elif mode == 1:
        # Low-frequency sinusoid
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.1)
        t = np.arange(n)
        b += amp * np.sin(2 * np.pi * freq * t / n + phase)
    elif mode == 2:
        # Random block scaling
        w = max(2, int(n * rng.uniform(0.01, 0.1)))
        s = int(rng.integers(0, n - w))
        b[s:s + w] *= np.exp(rng.normal(0, 0.2))
    else:
        # Small uniform noise
        b += rng.normal(0, 0.01, n)
    
    return _proj_sum1(b)

    # ---- Phase 4: Coordinate descent polish ----
    p4_deadline = min(t_start + budget - 50.0, deadline - 30.0)
    if time.monotonic() < p4_deadline - 60.0:
        best_a, best_o = _coordinate_polish(best_a, n, p4_deadline, tag="coord")
        print(f"After Phase 4 (coord): obj={best_o:.8f}")
    
    # ---- Phase 5: Final hard descent with remaining time ----
    p5_deadline = deadline - 10.0
    if time.monotonic() < p5_deadline - 5.0:
        best_a, best_o = _hard_phase(best_a, n, p5_deadline, top_k=3,
                                      eta_init=0.01, tag="final")
        print(f"After Phase 5 (final): obj={best_o:.8f}")
    
    # ---- Validate and return ----
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        sc = float(evaluate_sequence(result))
    except Exception:
        sc = float('inf')
    
    if not np.isfinite(sc):
        # Fallback: uniform
        result = [1.0] * n
        sc = 2.0
    
    print(f"FINAL: internal_obj={best_o:.8f} evaluated={sc:.8f} n={n}")
    print(f"Total time: {time.monotonic()-t_start:.1f}s")
    return result
