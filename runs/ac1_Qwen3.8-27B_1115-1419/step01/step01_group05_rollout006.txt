"""
Multi-phase optimizer for low-peak nonnegative self-convolutions.
Combines LSE/power-sum gradient descent, LP-based refinement (HiGHS),
peak shaving, and multi-length search with intelligent upscaling.
"""

import numpy as np
import time
from scipy.optimize import linprog

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(max(a.size, 1), dtype=np.float64) / max(a.size, 1)
    return a / s


def _full_conv(a):
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    full = np.fft.irfft(A * A, size)
    return full[:2 * n - 1]


def _score(a):
    a = _normalize(a)
    n = a.size
    b = _full_conv(a)
    mb = float(np.max(b))
    if not np.isfinite(mb) or mb < 0:
        return 1e100
    return 2.0 * float(n) * mb


def _tf_gradient(a, p, fft_size):
    """
    Compute 2 * (p star a)[:n] where (p star a)[k] = sum_j p[j+k]*a[j].
    Uses FFT cross-correlation.
    """
    n = a.size
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size).real
    return 2.0 * corr[:n]


def _power_sum_gradient(a, b, p_power, fft_size):
    """
    Gradient of (sum b_k^p)^(1/p) w.r.t. a.
    = (sum b^p)^(1-1/p) * sum_k b_k^(p-1) * db_k/da_i
    """
    b_pos = np.maximum(b, 1e-30)
    weights = b_pos ** (p_power - 1)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(b.size) / b.size
    else:
        weights = weights / wsum
    # (sum b^p)^(1/p - 1) * sum_k weights_k * 2*a_{k-i}
    # = (sum b^p)^(1-1/p) * (p gradient with weights)
    factor = (b_pos ** p_power).sum() ** (1.0 - 1.0 / p_power)
    grad = _tf_gradient(a, weights, fft_size)
    return factor * grad


def _lse_weighted(a, p, fft_size):
    """LSE-style gradient: 2 * sum_k p_k * a_{k-i}"""
    return _tf_gradient(a, p, fft_size)


def _softmax_weights(b, beta):
    m = float(np.max(b))
    if beta <= 1.0:
        return np.ones(b.size) / b.size
    arg = beta * (b - m)
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    s = e.sum()
    if s < 1e-300:
        return np.ones(b.size) / b.size
    return e / s


def _make_exponential(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * x)
    return _normalize(a)


def _make_cosine(n, p=1.0):
    x = np.linspace(0.0, np.pi, n)
    a = np.maximum(np.cos(x / 2.0), 0.0) ** p
    s = a.sum()
    if s < 1e-30:
        return _make_exponential(n, 1.0)
    return a / s


def _make_gaussian(n, sigma_frac=0.3):
    x = np.linspace(-1.0, 1.0, n)
    a = np.exp(-0.5 * (x / sigma_frac) ** 2)
    return _normalize(a)


def _make_u_shaped(n, eps=0.5):
    x = np.linspace(0.0, 2.0 * np.pi, n)
    a = 1.0 + eps * np.cos(x)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _make_symmetric_exponential(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    x_center = x - 0.5
    a = np.exp(-alpha * np.abs(x_center) * 4.0)
    return _normalize(a)


def _make_smooth_random(n, rng, base=None, amplitude=0.4):
    if base is None:
        base = _make_exponential(n, 1.0)
    z = rng.normal(size=n)
    S = _next_pow2(n)
    F = np.fft.rfft(z, S)
    cutoff = max(2, n // 16 + 2)
    if cutoff < len(F):
        F[cutoff:] *= 0.0
    f = np.fft.irfft(F, n)
    f -= f.mean()
    sd = f.std()
    if sd > 0:
        f /= sd
    a = base * (1.0 + amplitude * f)
    a = np.maximum(a, 1e-12)
    return _normalize(a)


def _resample_sequence(old, new_n):
    old = _normalize(old)
    new_n = max(int(new_n), 2)
    if old.size == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb(a, rng, strength=0.2):
    noise = np.exp(rng.normal(0.0, strength, size=a.size))
    a2 = a * noise
    return _normalize(a2)


def _block_perturb(a, rng, strength=0.15):
    n = a.size
    c = a.copy()
    width = max(2, n // 20)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= np.exp(rng.normal(0.0, strength))
    return _normalize(c)


def _lp_refine(a, max_iters=3):
    """
    LP-based refinement: solve max sum(g) s.t. (a*g)_k <= M, g >= 0.
    Returns normalized g.
    """
    a = _normalize(a)
    n = a.size
    if n < 16 or n > 2048:
        return a.copy()
    
    b = _full_conv(a)
    M = float(np.max(b))
    if M < 1e-30:
        return a.copy()
    
    # Build constraint matrix A_ub[k,j] = a[k-j] if 0<=k-j<n else 0
    A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    # Use toeplitz-like structure
    for j in range(n):
        k_start = 0
        k_end = min(2 * n - 1, j + n)
        A_ub[k_start:k_end, j] = a[:k_end - k_start]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
                return g
    except Exception:
        pass
    return a.copy()


def _peak_shave(a, rng, num_iters=50):
    """Targeted peak reduction: reduce contributions to top convolution peaks."""
    a = _normalize(a)
    n = a.size
    b = _full_conv(a)
    mb = float(np.max(b))
    best_a = a.copy()
    best_mb = mb
    
    for it in range(num_iters):
        # Find top 5 peaks
        top_k = np.argsort(b)[-5:]
        # For each peak, reduce the larger contributing a_i
        changed = False
        for k in top_k:
            # b_k = sum_j a[j]*a[k-j] for valid j
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min > j_max:
                continue
            a_left = a[j_min:j_max + 1]
            a_right = a[k - j_max:k - j_min + 1][::-1]
            # Find the pair with largest product
            prods = a_left * a_right
            if prods.size == 0:
                continue
            idx = int(np.argmax(prods))
            gi = j_min + idx
            # Reduce a[gi] slightly
            a[gi] *= 0.92
            changed = True
        a = _normalize(a)
        b = _full_conv(a)
        mb = float(np.max(b))
        if mb < best_mb - 1e-12:
            best_mb = mb
            best_a = a.copy()
        if not changed:
            break
    return best_a


def _optimize_sequence(a, deadline, rng, use_lp=True, p_power=30.0):
    """
    Main optimization: LSE + power-sum gradient descent with LP refinement.
    """
    a = _normalize(a)
    n = a.size
    if n < 8:
        return a, _score(a)
    
    fft_conv = _next_pow2(2 * n - 1)
    fft_grad = _next_pow2(3 * n - 1)
    
    b = _full_conv(a)
    maxb = float(np.max(b))
    score = 2.0 * n * maxb
    best_a = a.copy()
    best_score = score
    
    alpha = 1.0 / max(n, 64)
    beta = 2.0 / max(maxb, 1e-10)
    beta = float(min(beta, 5000.0))
    
    it = 0
    fail_count = 0
    last_improve_it = 0
    lp_count = 0
    last_lp_it = 0
    
    while time.monotonic() < deadline - 2.0:
        it += 1
        maxb = float(np.max(b))
        if maxb < 1e-30:
            maxb = 1e-30
        
        # Adaptive beta
        beta = float(min(50000.0, max(1.0, 2.0 / maxb)))
        
        # Choose gradient method
        if it % 60 < 40:
            # LSE gradient
            p = _softmax_weights(b, beta)
            grad = _lse_weighted(a, p, fft_grad)
        else:
            # Power-sum gradient
            grad = _power_sum_gradient(a, b, p_power, fft_grad)
        
        # Center the gradient (removes mean, helps maintain normalization)
        grad_c = grad - grad.mean()
        grad_norm = np.linalg.norm(grad_c)
        if grad_norm < 1e-30:
            break
        
        # Adaptive step size
        v = a - alpha * (grad_c / grad_norm) * np.sqrt(n)
        a_new = _normalize(v)
        
        b_new = _full_conv(a_new)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * n * maxb_new
        
        # Smooth proxy for step acceptance
        p_old = _softmax_weights(b, beta)
        proxy_old = float(np.dot(p_old, b)) + 0.1 * float(np.max(b))
        p_new = _softmax_weights(b_new, beta)
        proxy_new = float(np.dot(p_new, b_new)) + 0.1 * float(np.max(b_new))
        
        if proxy_new < proxy_old - 1e-15:
            a = a_new
            b = b_new
            maxb = maxb_new
            score = score_new
            alpha = min(alpha * 1.08, 5.0 / max(n, 64))
            fail_count = 0
            last_improve_it = it
        else:
            alpha = max(alpha * 0.5, 1e-6 / max(n, 64))
            fail_count += 1
        
        if score < best_score - 1e-14:
            best_a = a.copy()
            best_score = score
        
        # Periodic LP refinement
        if (use_lp and n <= 2048 and it - last_lp_it > 80 and
                it > 20 and time.monotonic() < deadline - 10.0 and
                lp_count < 15 and fail_count > 30):
            last_lp_it = it
            lp_count += 1
            g = _lp_refine(best_a, max_iters=1)
            if g is not None:
                # Blend
                t = 0.3
                blend = _normalize((1.0 - t) * best_a + t * g)
                b_blend = _full_conv(blend)
                score_blend = 2.0 * n * float(np.max(b_blend))
                if score_blend < best_score:
                    best_a = blend
                    best_score = score_blend
                    a = blend
                    b = b_blend
        
        # Restart if stuck
        if it - last_improve_it > 500 and it > 100:
            if time.monotonic() < deadline - 10.0:
                # Mix with perturbed best
                a = _normalize(0.7 * best_a + 0.3 * _perturb(best_a, rng, 0.25))
                b = _full_conv(a)
                maxb = float(np.max(b))
                score = 2.0 * n * maxb
                alpha = 1.0 / max(n, 64)
                beta = 2.0 / max(maxb, 1e-10)
                fail_count = 0
                last_improve_it = it
        
        if fail_count > 2000:
            break
    
    # Final LP polish
    if use_lp and n <= 2048 and time.monotonic() < deadline - 5.0:
        g = _lp_refine(best_a, max_iters=2)
        if g is not None:
            for t in [0.5, 0.3, 0.2]:
                blend = _normalize((1.0 - t) * best_a + t * g)
                s_bl = 2.0 * n * float(np.max(_full_conv(blend)))
                if s_bl < best_score:
                    best_a = blend
                    best_score = s_bl
            g = _lp_refine(best_a, max_iters=1)
            if g is not None:
                blend = _normalize(0.6 * best_a + 0.4 * g)
                s_bl = 2.0 * n * float(np.max(_full_conv(blend)))
                if s_bl < best_score:
                    best_a = blend
                    best_score = s_bl
    
    # Peak shaving
    if time.monotonic() < deadline - 3.0:
        ps_a = _peak_shave(best_a, rng, num_iters=30)
        ps_score = _score(ps_a)
        if ps_score < best_score:
            best_a = ps_a
            best_score = ps_score
    
    return best_a, best_score


def _gen_inits(n, rng, given=None, best=None):
    """Generate diverse initializations for length n."""
    inits = []
    
    inits.append(_make_exponential(n, 1.0))
    inits.append(_make_cosine(n, 1.0))
    inits.append(_make_gaussian(n, 0.25))
    inits.append(_make_u_shaped(n, 0.6))
    inits.append(_make_symmetric_exponential(n, 1.5))
    
    base = _make_exponential(n, 0.8)
    inits.append(_make_smooth_random(n, rng, base, 0.5))
    inits.append(_make_mixture(n, rng))
    
    if best is not None and best.size >= 8:
        inits.append(_resample_sequence(best, n))
    
    if given is not None and given.size >= 8:
        inits.append(_resample_sequence(given, n))
    
    # Deduplicate
    seen = []
    out = []
    for ic in inits:
        sig = tuple(np.round(ic[::max(1, ic.size // 16)], 4))
        if sig not in seen:
            seen.append(sig)
            out.append(ic)
    return out


def _make_mixture(n, rng):
    alpha1 = rng.uniform(0.5, 2.0)
    alpha2 = rng.uniform(0.5, 2.0)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha1 * x) + rng.uniform(0.3, 1.0) * np.exp(-alpha2 * (1.0 - x))
    return _normalize(a)


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for low-peak nonnegative self-convolution sequences.
    Returns the best sequence found within the time budget.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0
    
    deadline = time.monotonic() + max(budget, 60.0) - 30.0
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Check for given sequence
    given = None
    seq_raw = globals().get("height_sequence_1", None)
    if seq_raw is not None:
        try:
            arr = np.asarray(seq_raw, dtype=np.float64).ravel()
            arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
            arr = np.maximum(arr, 0.0)
            if arr.size >= 8 and arr.sum() > 1e-10:
                given = arr / arr.sum()
        except Exception:
            pass
    
    # Initial best: try a few quick starts
    best_a = _make_exponential(256, 1.0)
    best_score = _score(best_a)
    print(f"Initial: len={best_a.size} score={best_score:.8f}", flush=True)
    
    # Also try cosine and symmetric
    for init_fn in [_make_cosine, _make_symmetric_exponential, _make_gaussian]:
        trial = init_fn(256)
        s = _score(trial)
        if s < best_score:
            best_a = trial
            best_score = s
    print(f"After quick inits: len={best_a.size} score={best_score:.8f}", flush=True)
    
    # If we have a given sequence, try it
    if given is not None and given.size >= 64:
        trial = _normalize(given)
        s = _score(trial)
        print(f"Given sequence: len={trial.size} score={s:.8f}", flush=True)
        if s < best_score:
            best_a = trial
            best_score = s
    
    # === Main search phases ===
    
    # Phase 1: Small lengths for fast shape discovery
    for n in [128, 256]:
        if time.monotonic() > deadline - 120.0:
            break
        inits = _gen_inits(n, rng, given, best_a)
        num_starts = min(len(inits), 3)
        sec_per = min(40, max(15, int(40.0 * (budget / 2400.0))))
        
        for idx in range(num_starts):
            if time.monotonic() > deadline - 60.0:
                break
            t_end = min(deadline - 10.0, time.monotonic() + sec_per)
            init = inits[idx].copy()
            res_a, res_score = _optimize_sequence(init, t_end, rng, use_lp=True, p_power=30.0)
            elapsed = time.monotonic() - t_start
            print(f"n={n} start={idx} score={res_score:.8f} best={best_score:.8f} t={elapsed:.0f}s", flush=True)
            if res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
    
    # Phase 2: Medium lengths (main optimization)
    for n in [512, 1024]:
        if time.monotonic() > deadline - 180.0:
            break
        inits = _gen_inits(n, rng, given, best_a)
        num_starts = min(len(inits), 4)
        sec_per = min(80, max(25, int(80.0 * (budget / 2400.0))))
        
        for idx in range(num_starts):
            if time.monotonic() > deadline - 60.0:
                break
            t_end = min(deadline - 10.0, time.monotonic() + sec_per)
            init = inits[idx].copy()
            use_lp = (n <= 1024)
            res_a, res_score = _optimize_sequence(init, t_end, rng, use_lp=use_lp, p_power=40.0)
            elapsed = time.monotonic() - t_start
            print(f"n={n} start={idx} score={res_score:.8f} best={best_score:.8f} t={elapsed:.0f}s", flush=True)
            if res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
    
    # Phase 3: Larger lengths if time permits
    for n in [2048]:
        if time.monotonic() > deadline - 200.0:
            break
        inits = _gen_inits(n, rng, given, best_a)
        num_starts = min(len(inits), 3)
        sec_per = min(120, max(30, int(120.0 * (budget / 2400.0))))
        
        for idx in range(num_starts):
            if time.monotonic() > deadline - 60.0:
                break
            t_end = min(deadline - 10.0, time.monotonic() + sec_per)
            init = inits[idx].copy()
            res_a, res_score = _optimize_sequence(init, t_end, rng, use_lp=True, p_power=50.0)
            elapsed = time.monotonic() - t_start
            print(f"n={n} start={idx} score={res_score:.8f} best={best_score:.8f} t={elapsed:.0f}s", flush=True)
            if res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
    
    # Phase 4: Upscaling if best score is still high
    if best_score > 1.52 and time.monotonic() < deadline - 100.0:
        n_cur = best_a.size
        for factor in [2, 2, 2]:
            if time.monotonic() > deadline - 50.0:
                break
            n_new = min(4096, n_cur * factor)
            if n_new <= n_cur:
                break
            init = _resample_sequence(best_a, n_new)
            # Blend with exponential to add structure
            init = _normalize(0.8 * init + 0.2 * _make_exponential(n_new, 1.0))
            t_end = min(deadline - 30.0, time.monotonic() + 90.0)
            res_a, res_score = _optimize_sequence(init, t_end, rng,
                                                  use_lp=(n_new <= 2048), p_power=40.0)
            elapsed = time.monotonic() - t_start
            print(f"Upscale {n_cur}->{n_new}: score={res_score:.8f} best={best_score:.8f} t={elapsed:.0f}s", flush=True)
            if res_score < best_score:
                best_a = np.asarray(res_a, dtype=np.float64).copy()
                best_score = float(res_score)
                n_cur = n_new
            else:
                break
    
    # Phase 5: Final deep refinement on the best
    if time.monotonic() < deadline - 5.0:
        t_end = deadline - 2.0
        n_best = best_a.size
        res_a, res_score = _optimize_sequence(best_a.copy(), t_end, rng,
                                             use_lp=(n_best <= 2048), p_power=60.0)
        if res_score < best_score:
            best_a = np.asarray(res_a, dtype=np.float64).copy()
            best_score = float(res_score)
            print(f"Final refine: score={res_score:.8f}", flush=True)
        else:
            print(f"Final refine: no improvement, best={best_score:.8f}", flush=True)
    
    # Phase 6: Peak shaving as last resort
    if time.monotonic() < deadline - 1.0:
        ps_a = _peak_shave(best_a, rng, num_iters=50)
        ps_s = _score(ps_a)
        if ps_s < best_score:
            best_a = ps_a
            best_score = ps_s
            print(f"Peak shave: score={ps_s:.8f}", flush=True)
    
    # Also try: verify with evaluate_sequence if available
    try:
        verify_score = evaluate_sequence([float(x) for x in best_a])
        print(f"Verification: evaluate_sequence gives {verify_score:.8f}", flush=True)
        if not np.isfinite(verify_score):
            # Fallback to a safe sequence
            best_a = _make_exponential(256, 1.0)
            best_score = _score(best_a)
    except NameError:
        pass
    except Exception as e:
        print(f"Verification error: {e}", flush=True)
    
    # Final output preparation
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)
    
    elapsed_total = time.monotonic() - t_start
    print(f"DONE: len={out.size} score={best_score:.8f} time={elapsed_total:.1f}s", flush=True)
    
    return [float(x) for x in out]
