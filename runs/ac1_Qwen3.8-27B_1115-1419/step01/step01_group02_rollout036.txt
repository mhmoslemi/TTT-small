import numpy as np
import time
from scipy.fft import fft, ifft

def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def _softmax(x, tau):
    """Numerically stable softmax with temperature."""
    x_shifted = x - np.max(x)
    x_scaled = x_shifted / tau
    exp_x = np.exp(x_scaled)
    return exp_x / np.sum(exp_x)

def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _score(a):
    """Compute the actual score."""
    n = len(a)
    c = _fft_conv(a, a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)

def _smoothed_grad(a, tau):
    """Compute gradient of smoothed objective (up to constant)."""
    n = len(a)
    c = _fft_conv(a, a)
    p = _softmax(c, tau)
    # grad[i] = conv(p, reverse(a))[i + n - 1] for i in range(n)
    a_rev = a[::-1]
    pa = _fft_conv(p, a_rev)
    # The cross-correlation sum_j p[i+j]*a[j] = pa[i + n - 1]
    grad = pa[n-1:n-1+n]
    return grad - np.mean(grad), c

def _optimize_from(a, deadline, tau_init, tau_min, tau_decay, lr_init, lr_decay):
    """Run smoothed gradient descent from initialization a."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = _project_simplex(a.astype(np.float64), S)
    
    tau = tau_init
    lr = lr_init
    best_score = _score(a)
    best_a = a.copy()
    stuck = 0
    
    while time.monotonic() < deadline:
        grad, c = _smoothed_grad(a, tau)
        max_c = np.max(c)
        
        a = a - lr * grad
        a = _project_simplex(a, S)
        
        lr = max(lr * lr_decay, 1e-10)
        tau = max(tau * tau_decay, tau_min)
        
        # Check score every 100 iterations
        # (We already have c from the gradient computation)
        sum_a = np.sum(a)
        current_score = 2.0 * n * max_c / (sum_a * sum_a)
        
        if current_score < best_score - 1e-8:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 500:
            # Reheat
            tau = tau_init * 0.5
            lr = lr_init * 0.3
            a = best_a + 0.01 * np.random.standard_normal(n)
            a = _project_simplex(a, S)
            stuck = 0
    
    return best_a, best_score

import time
from scipy.fft import fft, ifft

def _fft_conv(a, b):
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def _softmax(x, tau):
    x_max = np.max(x)
    x_scaled = (x - x_max) / tau
    exp_x = np.exp(x_scaled)
    return exp_x / np.sum(exp_x)

def _project_simplex(v, S):
    n = len(v)
    u, _ = np.sort(v), None
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest rho such that u[rho] > (cssv[rho] - S) / (rho+1)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:

Let me also think about the simplex projection more carefully. The standard algorithm is:
1. Sort v in decreasing order: v_(1) >= v_(2) >= ... >= v_(n)
2. Compute cumulative sums: S_k = v_(1) + ... + v_(k)
3. Find the largest k such that v_(k) > (S_k - S) / k
4. Set theta = (S_k - S) / k
5. Result: max(v - theta, 0)

This is O(n log n) due to sorting. For n = 5582, this is fast.

One optimization: if the gradient step is small (small lr), the projection is a small perturbation of the previous point, and we might be able to use a faster "warm start" for the sorting. But for simplicity, I'll just sort each time.

"""
Minimize 2n*max(convolve(a,a))/(sum(a))^2 using smoothed (log-sum-exp) gradient descent
with simplex projection, multiple temperature schedules, and diverse restarts.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]


def _softmax(x, tau):
    """Numerically stable softmax with temperature."""
    x
    x_max = np.max(x)
    exp_x = np.exp((x - x_max) / tau)
    return exp_x / np.sum(exp_x)


def _project_simplex(v, S):
    """Project v onto the slice {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_candidates = np.nonzero(u * idx > (cssv - S))[0]
    if len(rho_candidates) == 0:
        return np.full(n, S / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _true_score(a):
    """Compute the true evaluation score."""
    n = len(a)
    c = _fft_conv(a, a)
    max_c = float(np.max(c))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)


def _run_lse_optimization(a_init, deadline, tau_start, tau_end, lr_start, lr_end,
                          total_iters=None):
    """
    Run LSE-smoothed gradient descent with simplex projection.
    Uses exponential decay for tau and lr.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = _project_simplex(a_init.astype(np.float64).copy(), S)
    
    if total_iters is None:
        # Estimate based on time
        total_iters = 200000
    
    best_score = _true_score(a)
    best_a = a.copy()
    iters_since_improve = 0
    
    # Precompute FFT size
    n_out = 2 * n - 1
    n_fft = 1 << (n_out - 1).bit_length()
    
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        
        # Interpolate tau and lr
        frac = it / max(1, total_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute convolution
        A = fft(a, n_fft)
        C = ifft(A * A).real[:n_out]
        
        # Compute softmax weights
        p = _softmax(C, tau)
        
        # Compute gradient: grad[i] = conv(p, reverse(a))[n-1+i]
        P = fft(p, n_fft)
        A_rev = fft(a[::-1], n_fft)
        PA = ifft(P * A_rev).real[:len(p) + n - 1]
        grad = PA[n-1:n-1+n]
        grad -= np.mean(grad)
        
        # Update and project
        a = a - lr * grad
        a = _project_simplex(a, S)
        
        # Periodic evaluation
        if (it + 1) % 200 == 0:
            # C is slightly stale (computed before update), but good enough for tracking
            # Actually let's use the C from before the update as an approximation
            sum_a = float(np.sum(a))

            A2 = fft(a, n_fft)
            C2 = ifft(A2 * A2).real[:n_out]
            max_c = float(np.max(C2))
            score = 2.0 * n * max_c / (sum_a * sum_a)

                best_a = a.copy()

            iters_since_improve = 0
            # Reset the schedule effectively by adjusting remaining iters
            # (simple approach: just continue with current schedule)
    
    return best_a, best_score

    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    # i.e., u[k-1] > (cssv[k-1] - S) / k
