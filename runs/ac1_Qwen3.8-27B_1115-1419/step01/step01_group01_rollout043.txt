Now let me fill in the helper functions:

```python
def _waterfall(n):
    """Generate the waterfall sequence a[k] = C(2k,k)/4^k."""
    a = np.ones(n, dtype=np.float64)
    # C(2k,k)/4^k can be computed recursively
    # C(2k,k) = C(2k-2,k-1) * (2k)(2k-1)/k^2
    # C(2k,k)/4^k = C(2k-2,k-1)/4^(k-1) * (2k)(2k-1)/(4*k^2)
    # = prev * (k-0.5)/k
    for k in range(1, n):
        a[k] = a[k-1] * (k - 0.5) / k
    return a

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a.fill(1.0 / len(a))
    else:
        a /= s
    return a

def _score(a):
    """Compute 2*n*max(conv)/sum^2 using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 1e-300:
        return np.inf
    m = 1
    while m < 2*n - 1:
        m <<= 1
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2*n-1]
    b = np.maximum(b, 0.0)
    bmax = b.max()
    if not np.isfinite(bmax) or bmax <= 0:
        return np.inf
    return 2.0 * n * bmax / (S * S)

def _interp(a, n):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones(n) / n
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss <= 0:
        return np.ones(n) / n
    return out / ss
```
