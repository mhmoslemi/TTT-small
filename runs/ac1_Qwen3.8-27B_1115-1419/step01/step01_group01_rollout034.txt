    n = a.size
    best_a = a.copy()
    best_score = score_fft(a)
    
    for it in range(max_iters):
        F = np.fft.fft(a, m)
        b = np.fft.ifft(F * F).real[:2*n-1]
        b = np.maximum(b, 0.0)
        bmax = b.max()

        tau = max(tau_min, tau_init * (tau_min/tau_init)**(it/max(1,max_iters)))
        x = (b - bmax) / tau

        w = np.exp(x)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using iterative multiplicative peak-shaving,
spectral factorization with target shaping, and gradient polishing.
Combines NMF-inspired updates with FFT-based acceleration.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _proj_nonneg_sum(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        a.fill(1.0 / max(1, a.size))
    else:
        a /= s
    return a


def _conv_b(a, m):
    """Compute full self-convolution using FFT."""
    n = a.size
    F = np.fft.fft(a, m)
    b = np.fft.ifft(F * F).real[:2 * n - 1]
    b = np.maximum(b, 0.0)
    return b


def score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-300:
        return np.inf
    m = _next_pow2(2 * n - 1)
    b = _conv_b(a, m)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.inf
    return float(2.0 * n * bmax / (S * S))


def _get_weights(b, bmax, tau, top_frac=0.03):
    """Compute soft-max weights emphasizing the peak."""
    x = (b - bmax) / max(tau, 1e-15)
    x = np.clip(x, -50.0, 0.0)
    w = np.exp(x)
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return w
    w /= ws
    blen = b.size
    top_k = max(1, int(top_frac * blen))
    if top_k < blen:
        idx = np.argpartition(b, -top_k)[-top_k:]
        w[idx] += 0.5
        w /= w.sum()
    return w


def _cross_corr_wa(w, a, m):
    """Compute g[i] = sum_k w[k]*a[k-i] via FFT cross-correlation."""
    n = a.size
    Fa = np.fft.fft(a, m)
    Fw = np.fft.fft(w, m)
    g = np.fft.ifft(np.conj(Fa) * Fw).real[:n]
    return g


def _peak_shave(a, max_iters, deadline, tau_init=0.15, tau_min=0.02,
                damp=0.6, smooth_sigma=0.0, top_frac=0.03, verbose=False):
    """Iterative multiplicative peak-shaving (NMF-inspired)."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = _proj_nonneg_sum(a)
    m = _next_pow2(3 * n - 1)  # Large enough for cross-correlation without aliasing

    best_a = a.copy()
    best_score = score_fft(a)
    last_improve = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = _conv_b(a, m)
        bmax = float(b.max())
        if bmax <= 1e-300:
            break

        progress = min(1.0, it / max(1, max_iters))
        tau = tau_init * (tau_min / tau_init) ** progress
        tau = max(tau, tau_min)

        w = _get_weights(b, bmax, tau, top_frac)
        g = _cross_corr_wa(w, a, m)
        g = np.maximum(g, 1e-15)

        # Multiplicative update: target is proportional to 1/g
        a_target = 1.0 / g
        tsum = float(a_target.sum())
        if tsum <= 0 or not np.isfinite(tsum):
            break
        a_target /= tsum

        # Damped update
        a_new = (1.0 - damp) * a + damp * a_target

        # Optional smoothing
        if smooth_sigma > 0:
            kernel_size = max(3, int(4 * smooth_sigma * n) | 1)
            kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (smooth_sigma * n) ** 2)
            kernel /= kernel.sum()
            a_new = np.convolve(a_new, kernel, mode='same')

        a_new = np.maximum(a_new, 0.0)
        ss = float(a_new.sum())
        if ss <= 1e-300:
            break
        a_new /= ss

        sc = score_fft(a_new)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            last_improve = time.monotonic()
            if verbose and it % 100 == 0:
                print("      peak_shave it=%d score=%.6f" % (it, best_score))

        a = a_new

        # Early stop if stuck
        if time.monotonic() - last_improve > 60.0 and it > 200:
            break

    return best_a, best_score


def _spectral_refine(a, n, m, target_shape, iters=40, blend=0.5, deadline=None):
    """Spectral factorization targeting a given convolution shape."""
    L = 2 * n - 1
    target = np.asarray(target_shape, dtype=np.float64).copy()
    if target.size < L:
        padded = np.zeros(L)
        padded[:target.size] = target
        target = padded
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    tm = float(target.max())
    if tm <= 0:
        target = np.ones(L)
        tm = 1.0
    target /= tm

    T = np.fft.fft(target, m)

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    amax = float(a.max())
    if amax <= 0:
        a = np.ones(n)
        amax = 1.0
    a /= amax

    best_a = a.copy()
    best_sc = np.inf

    for it in range(iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        F = np.fft.fft(a, m)
        F_safe = F + 1e-12 * np.ones_like(F)
        F_new = 0.5 * (F + T / F_safe)
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        psum = float(a_new.sum())
        asum = float(a.sum())
        if psum < 1e-8 * max(asum, 1e-12):
            a_new = a.copy()
        a_new = (1.0 - blend) * a + blend * a_new
        a_new = np.maximum(a_new, 1e-15)
        a = a_new

        if (it + 1) % 10 == 0:
            sc = score_fft(a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = score_fft(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    best_a = _proj_nonneg_sum(best_a)
    sc = score_fft(best_a)
    return best_a, float(sc)


def _make_targets(n, best_a=None):
    """Generate various target convolution shapes."""
    L = 2 * n - 1
    t = np.linspace(0.0, 1.0, L)
    edge = np.abs(t - 0.5) * 2.0

    targets = []
    # Flat
    targets.append(np.ones(L))
    # Slightly edge-decayed (account for boundary effects)
    for alpha in (0.1, 0.3, 0.5):
        targets.append(1.0 - alpha * edge ** 2)
    # Center-peaked (to push energy to edges)
    for alpha in (0.1, 0.3):
        targets.append(1.0 - alpha * (1.0 - edge) ** 2)

    # Based on current convolution (normalized)
    if best_a is not None:
        m = _next_pow2(2 * n - 1)
        b = _conv_b(best_a, m)
        bm = float(b.max())
        if bm > 0:
            b_norm = b / bm
            for mix in (0.3, 0.5, 0.7):
                targets.append(mix * b_norm + (1 - mix) * np.ones(L))

    return targets


def _gradient_polish(a, deadline, steps_per_check=200):
    """Adam-based gradient polishing in log-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = _proj_nonneg_sum(a)

    # Log-space parameterization
    v = np.log(np.maximum(a, 1e-300))
    v = v - v.mean()

    m = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = score_fft(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    mu = np.zeros(n)
    nu = np.zeros(n)
    t = 0
    last_improve = time.monotonic()
    start = time.monotonic()

    while time.monotonic() < deadline:
        elapsed = max(1e-9, time.monotonic() - start)
        progress = min(1.0, elapsed / max(1.0, deadline - start))

        # Current a in original space
        v_max = v.max()
        a_curr = np.exp(v - v_max)
        a_curr = a_curr / a_curr.sum()

        F = np.fft.fft(a_curr, m)
        b = np.fft.ifft(F * F).real[:2 * n - 1]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if bmax <= 1e-300:
            break

        # Soft weights
        tau_frac = 0.12 * (1.0 - 0.5 * progress) + 0.02
        tau = bmax * max(tau_frac, 0.01)
        x = (b - bmax) / tau
        x = np.clip(x, -50, 0)
        w = np.exp(x)
        ws = w.sum()
        if ws <= 0:
            break
        w /= ws

        top_k = max(1, int(0.04 * b.size))
        if top_k < b.size:
            idx = np.argpartition(b, -top_k)[-top_k:]
            w[idx] += 0.5
            w /= w.sum()

        # Gradient via cross-correlation
        Fw = np.fft.fft(w, m)
        g = np.fft.ifft(np.conj(F) * Fw).real[:n]
        g = np.maximum(g, 1e-15)

        # Gradient of soft objective w.r.t. v (log-space)
        # d(f)/d(v_j) = a_j * g_j - a_j * sum(a_i * g_i) / sum(a_i)
        # Since a is normalized: = a_j * (g_j - mean(a*g))
        ag_mean = float(np.sum(a_curr * g))
        grad_v = a_curr * (g - ag_mean)

        grms = float(np.sqrt(np.mean(grad_v ** 2)))
        if grms < 1e-14:
            v += np.random.normal(0, 0.1, n)
            v = v - v.mean()
            continue

        grad_v /= grms
        t += 1
        mu = beta1 * mu + (1 - beta1) * grad_v
        nu = beta2 * nu + (1 - beta2) * grad_v ** 2
        mhat = mu / (1 - beta1 ** t)
        vhat = nu / (1 - beta2 ** t)

        lr = 0.08 * (1.0 - 0.5 * progress)
        step = mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -0.3, 0.3)
        v = v - lr * step
        v = v - v.mean()
        v = np.clip(v, -20, 20)

        if t % steps_per_check == 0:
            an = np.exp(v - v.max())
            an = an / an.sum()
            sc = score_fft(an)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best_score = sc
                best_a = an.copy()
                last_improve = time.monotonic()

        if time.monotonic() - last_improve > 90 and t > steps_per_check * 10:
            v += np.random.normal(0, 0.15 * (1 - progress), n)
            v = v - v.mean()
            last_improve = time.monotonic()

    an = np.exp(v - v.max())
    an = an / an.sum()
    sc = score_fft(an)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = an

    return _proj_nonneg_sum(best_a), float(best_score)


def _interp_to_length(a, n):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if a.size == 0 or s <= 0:
        return np.ones(n) / n
    if a.size == n:
        return a / s
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 0:
        out.fill(1.0 / n)
    else:
        out /= ss
    return out


def _initial_candidates(n, rng, prev_a=None):
    """Generate diverse initial candidates."""
    cands = []
    if prev_a is not None:
        cands.append(_interp_to_length(prev_a, n))

    t = np.linspace(0, 1, n)
    i = np.arange(n, dtype=np.float64)

    # Exponentials
    for c in (0.5, 1.0, 2.0, 4.0, 8.0):
        cands.append(np.exp(-c * t))
    # Symmetric exponentials
    for c in (1.0, 3.0, 8.0):
        cands.append(np.exp(-c * np.abs(2 * t - 1)))
    # Inverse sqrt endpoints
    cands.append(1.0 / np.sqrt((i + 1.0) * (n - i)))
    # Cosine window
    cands.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    # Raised cosine variants
    for beta in (1.5, 2.0, 3.0):
        cands.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * (2 * t - 1) / beta), 0))
    # Polynomial tapers
    for p in (1, 2, 4, 8):
        edge_p = np.minimum(t, 1 - t) ** (1.0 / p)
        cands.append(edge_p)
    # Random Fourier
    for _ in range(3):
        a = np.ones(n)
        kc = min(20, n - 1)
        for k in range(1, kc + 1):
            amp = rng.uniform(0.1, 0.5) / (k ** 0.6)
            ph = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + ph)
        cands.append(a)

    # Project all
    result = []
    seen = []
    for c in cands:
        c = _proj_nonneg_sum(c)
        diverse = True
        for s in seen:
            d = float(np.sqrt(np.sum((c - s) ** 2)))
            if d < 0.03:
                diverse = False
                break
        if diverse:
            result.append(c)
            seen.append(c)
            if len(result) >= 12:
                break
    return result


def _optimize_length(n, deadline, prev_best_a, rng):
    """Full optimization pipeline for a given length."""
    if time.monotonic() >= deadline:
        return prev_best_a, score_fft(prev_best_a) if prev_best_a is not None else np.inf

    best_a = None
    best_score = np.inf

    if prev_best_a is not None:
        best_a = _interp_to_length(prev_best_a, n)
        best_score = score_fft(best_a)
        print("    Start score (n=%d): %.6f" % (n, best_score))

    cands = _initial_candidates(n, rng, prev_best_a)
    if best_a is not None:
        cands.insert(0, best_a)

    # Sort candidates by score
    scored = []
    for c in cands:
        sc = score_fft(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=lambda x: x[0])

    # Pick top candidates
    top = min(4, len(scored))
    top_cands = [scored[i][1] for i in range(top)]
    if not top_cands:
        top_cands = [np.ones(n) / n]

    m = _next_pow2(3 * n - 1)
    targets = _make_targets(n, top_cands[0] if top_cands else None)

    # Phase 1: Peak shaving on top candidates
    phase1_budget = 0.40
    phase2_budget = 0.30
    phase3_budget = 0.30

    total_budget = max(1.0, deadline - time.monotonic())
    p1_deadline = time.monotonic() + total_budget * phase1_budget
    p2_deadline = time.monotonic() + total_budget * (phase1_budget + phase2_budget)
    p3_deadline = deadline

    # Peak shaving
    for i, c in enumerate(top_cands):
        if time.monotonic() >= p1_deadline:
            break
        rem = max(5.0, p1_deadline - time.monotonic())
        a_new, sc = _peak_shave(c, max_iters=5000, deadline=time.monotonic() + rem,
                                 tau_init=0.2, tau_min=0.03, damp=0.5,
                                 smooth_sigma=0.002 if n > 1000 else 0.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new
            print("    peak_shave cand %d: %.6f" % (i, best_score))

    # Spectral refinement on best
    if time.monotonic() < p2_deadline and best_a is not None:
        m_spec = _next_pow2(2 * n - 1)
        for ti, target in enumerate(targets[:4]):
            if time.monotonic() >= p2_deadline:
                break
            rem = max(5.0, p2_deadline - time.monotonic())
            iters = min(60, max(15, int(rem / 0.5)))
            a_new, sc = _spectral_refine(best_a, n, m_spec, target, iters=iters,
                                         blend=0.4, deadline=time.monotonic() + rem)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new
                print("    spectral target %d: %.6f" % (ti, best_score))

    # Gradient polish
    if time.monotonic() < p3_deadline and best_a is not None:
        rem = max(10.0, p3_deadline - time.monotonic() - 5.0)
        a_new, sc = _gradient_polish(best_a, deadline=time.monotonic() + rem)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new
            print("    gradient polish: %.6f" % best_score)

        # One more peak-shave pass
        if time.monotonic() < p3_deadline - 10:
            rem2 = max(5.0, p3_deadline - time.monotonic() - 5.0)
            a_new, sc = _peak_shave(best_a, max_iters=3000, deadline=time.monotonic() + rem2,
                                     tau_init=0.08, tau_min=0.01, damp=0.4,
                                     smooth_sigma=0.0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new
                print("    final peak_shave: %.6f" % best_score)

    if best_a is None:
        best_a = np.ones(n) / n
        best_score = 2.0

    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main entry point: search for best non-negative sequence minimizing the objective."""
    seed = int(seed)
    budget_s = float(max(60.0, budget_s))
    start = time.monotonic()
    global_deadline = start + budget_s - 30.0  # Safety margin
    rng = np.random.default_rng(seed)

    # Load previous best if available
    prev_a = None
    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64)
        if arr.ndim != 1:
            arr = arr.ravel()
        if arr.size > 0 and np.isfinite(arr).all():
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 0.0:
                prev_a = arr / float(arr.sum())
    except Exception:
        prev_a = None

    best_a = prev_a
    best_score = score_fft(prev_a) if prev_a is not None else np.inf
    print("Previous best score: %.6f (n=%s)" % (best_score, len(prev_a) if prev_a is not None else 0))

    # Progressive length strategy
    # Start with the length of previous best, then try longer
    if prev_a is not None:
        base_n = len(prev_a)
    else:
        base_n = 2048

    # Determine which lengths to try based on budget
    # Each length needs roughly: 20s for n<=1024, 40s for n=2048, 80s for n=4096, 160s for n=8192
    lengths_to_try = []
    remaining = budget_s - 30.0

    # Always try the base length
    lengths_to_try.append((base_n, remaining * 0.30))
    remaining *= 0.70

    # Try longer lengths if time permits
    longer = [l for l in [4096, 8192, 16384] if l > base_n]
    if budget_s > 600:
        lengths_to_try.append((min(4096, 4096), remaining * 0.35))
        remaining *= 0.65
    if budget_s > 1200:
        lengths_to_try.append((8192, remaining * 0.40))
        remaining *= 0.60
    if budget_s > 2000:
        lengths_to_try.append((16384, remaining * 0.50))

    # Also try a shorter length for quick wins
    short_n = max(512, base_n // 2)
    if budget_s > 300:
        lengths_to_try.insert(0, (short_n, 30.0))

    print("Planning lengths: %s" % [(l, round(f, 1)) for l, f in lengths_to_try])

    for n, frac in lengths_to_try:
        now = time.monotonic()
        if now >= global_deadline - 15:
            break
        stage_budget = min(frac, global_deadline - now - 10)
        if stage_budget < 20:
            continue
        stage_deadline = now + stage_budget
        print("Stage n=%d, budget=%.0fs, current best=%.6f" % (n, stage_budget, best_score))

        a_new, sc_new = _optimize_length(int(n), stage_deadline, best_a, rng)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_a = a_new
            best_score = sc_new
            print("  Improved! New best: %.6f (n=%d)" % (best_score, n))
        else:
            print("  No improvement (best=%.6f)" % best_score)

    # Final validation
    if best_a is None:
        best_a = np.ones(1024)
    best_a = _proj_nonneg_sum(best_a)

    n_final = len(best_a)
    fft_score = score_fft(best_a)
    print("Final: n=%d, FFT score=%.6f" % (n_final, fft_score))

    # Validate with exact evaluation if feasible
    try:
        if n_final <= 5000:
            exact = evaluate_sequence([float(x) for x in best_a])
            print("Exact evaluation: %.6f" % exact)
            if not np.isfinite(exact):
                print("  Validation failed, using fallback")
                best_a = np.ones(1024)
                fft_score = 2.0
    except Exception as e:
        print("Validation error: %s" % e)

    print("Returning sequence of length %d with score %.6f" % (len(best_a), fft_score))
    return [float(x) for x in best_a]
