3. For positions where c_k > target, we want to reduce c_k.
4. For positions where c_k < target, we can afford to increase c_k.
5. Find a direction g that reduces the high c_k's and increases the low c_k's.

This is essentially what the LP approach does, but we can approximate it with a simpler method.

A simple version:
- Compute the "excess" at each position: `e_k = max(0, c_k - target)`.
- The gradient of the excess is: for each position k with excess, the subgradient points in the direction that increases c_k. To reduce c_k, we move in the opposite direction.
- So the direction is: `g = -sum_{k: e_k > 0} (e_k / sum(e)) * subgrad(c_k)`.

This is a weighted subgradient where the weights are proportional to the excess. It automatically focuses on the most problematic peaks.

Let me implement this.

```python
def excess_subgradient(a, c, target, alpha=1.0):
    """Subgradient weighted by excess above target."""
    n = a.size
    L = 2 * n - 1
    excess = np.maximum(c - target, 0)
    total_excess = excess.sum()
    if total_excess < 1e-15:
        return np.zeros(n)
    
    # Weight each peak by its excess
    weights = excess / total_excess
    active = np.where(excess > 1e-10)[0]
    
    g = np.zeros(n)
    for k in active:
        k = int(k)
        w = weights[k]
        start = max(0, k - n + 1)
        stop = min(n - 1, k)
        g[start:stop+1] += w * a[k-stop : k-start+1][::-1]
    
    g *= 2.0
    return g
```

This is O(n * |active|) per step, which might be slow if many peaks are active. But in practice, only a few peaks should be significantly above the target.

def fast_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fast_convolution(a)
    return float(2.0 * n * float(c.max()) / (s * s))

    a = np.maximum(a, 0.0)

        return np.full(a.shape, target_sum / a.size)
    a = a * (target_sum / s)
    return np.minimum(a, 1000.0)

def gaussian_smooth(a, sigma):
    n = a.size
    if sigma < 0.5:
        return a.copy()
    ksize = int(math.ceil(6 * sigma)) | 1
    if ksize < 3:
        ksize = 3
    x = np.arange(ksize) - ksize // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    # Use FFT for efficiency
    N = 1 << (n + ksize - 1).bit_length()
    fa = np.fft.fft(a, N)
    fk = np.fft.fft(kernel, N)
    result = np.fft.ifft(fa * fk).real[:n]
    return np.maximum(result, 0.0)

Searches for non-negative step-function heights minimizing
2*n*max(convolve(a,a))/(sum(a))^2.

Strategy: Adaptive excess-weighted subgradient descent with periodic
Gaussian smoothing, multi-length exploration, and diverse initializations.
Uses the previous best sequence (height_sequence_1) as a strong starting point.
Key improvements over previous approach:
- Excess-weighted gradients focus optimization on the worst peaks.
- Periodic Gaussian smoothing reduces sharp features that create peaks.
- Multi-length search explores the trade-off between resolution and score.
- Adaptive step size with backtracking.

def fast_convolution(a):
    """FFT-based linear convolution a * a, result has length 2n-1."""
    n = a.size
    if n == 0:
        return np.empty(0)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def compute_score(a):
    """Fast computation of the objective (matches evaluate_sequence up to normalization)."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fast_convolution(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def normalize(a, target_sum):
    """Clip to [0, 1000] and rescale to target_sum."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    for _ in range(3):
        s = float(a.sum())
        if s < 1e-15:
            a = np.full(a.shape, target_sum / max(1, a.size))
            break
        a = a * (target_sum / s)
        a = np.clip(a, 0.0, 1000.0)
    return a


def subgradient_at_peak(a, k):
    """Subgradient contribution from a single peak at position k."""
    n = a.size
    start = max(0, k - n + 1)
    stop = min(n - 1, k)
    if start > stop:
        return np.zeros(n)
    seg = a[k - stop : k - start + 1][::-1]
    g = np.zeros(n)
    g[start:stop+1] = seg
    return g


def weighted_subgradient(a, c, K=50, use_excess_weights=True):
    """
    Weighted subgradient of max(c) using top-K peaks.
    If use_excess_weights, weight each peak by (c_k - mean(c))_+.
    """
    n = a.size
    L = c.size
    K = max(1, min(K, L))
    
    if K == L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:].astype(int)
    
    if use_excess_weights:
        c_mean = c.mean()
        excess = np.maximum(c[idx] - c_mean, 0.0)
        total = excess.sum()
        if total < 1e-15:
            weights = np.ones(len(idx)) / len(idx)
        else:
            weights = excess / total
    else:
        weights = np.ones(len(idx)) / len(idx)
    
    g = np.zeros(n)
    for i in range(len(idx)):
        if weights[i] < 1e-14:
            continue
        k = int(idx[i])
        start = max(0, k - n + 1)
        stop = min(n - 1, k)
        if start <= stop:
            g[start:stop+1] += weights[i] * a[k - stop : k - start + 1][::-1]
    
    g *= 2.0
    return g


def gaussian_smooth(a, sigma):
    """Apply FFT-based Gaussian smoothing."""
    n = a.size
    if sigma < 0.5 or n < 5:
        return a.copy()
    half = int(math.ceil(4 * sigma))
    ksize = 2 * half + 1
    x = np.arange(ksize) - half
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    N = 1 << (n + ksize - 1).bit_length()
    fa = np.fft.fft(a, N)
    fk = np.fft.fft(kernel, N)
    result = np.fft.ifft(fa * fk).real[:n]
    return np.maximum(result, 0.0)
