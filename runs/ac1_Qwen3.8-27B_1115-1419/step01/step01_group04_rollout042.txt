Uses FFT-based convolution, top-K subgradient with adaptive multiplicative

import numpy as np
import time

def fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return c

def score_of(a):
    """Compute the objective value."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)

def normalize(a, target_sum=None):
    """Normalize a to have sum = target_sum (default: len(a))."""
    n = len(a)
    target = target_sum if target_sum is not None else float(n)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(n, target / n)
    a = a * (target / s)
    return np.clip(a, 0.0, 1000.0)

def topk_gradient(a, c, K, temperature=0.0):
    """
    Compute gradient of max(c) (or soft-max if temperature > 0).
    
    For hard top-K: equal weight on top K peaks.
    For soft: weight proportional to exp(c[k]/temperature).
    """
    n = len(a)
    L = len(c)
    grad = np.zeros(n)
    
    if temperature > 1e-10:
        # Soft weights
        c_max = np.max(c)
        if temperature < 1e-10:
            temperature = 1e-10
        weights = np.exp((c - c_max) / temperature)
        # Only use top K for efficiency
        if K < L:
            idx = np.argpartition(c, -K)[-K:]
            # Zero out non-top-K weights
            mask = np.zeros(L, dtype=bool)
            mask[idx] = True
            weights *= mask
        total = np.sum(weights)
        if total < 1e-30:
            weights = np.zeros(L); weights[np.argmax(c)] = 1.0
            total = 1.0
        weights /= total
    else:
        # Hard top-K
        K = min(K, L)
        if K == L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, -K)[-K:]
        weights = np.zeros(L)
        weights[idx] = 1.0 / K
    
    # Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (valid indices)
    # Only iterate over non-zero weights
    nonzero = np.where(weights > 1e-15)[0]
    for k in nonzero:
        k = int(k)
        w = weights[k]
        if w < 1e-15:
            continue
        # Valid i: max(0, k-n+1) <= i <= min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        # j = k - i, so j ranges from k-i_hi to k-i_lo
        j_lo = k - i_hi
        j_hi = k - i_lo
        grad[i_lo:i_hi+1] += 2.0 * w * a[j_lo:j_hi+1][::-1]
    
    return grad

Uses FFT-based convolution, top-K subgradient with adaptive multiplicative
updates, multiple diverse starts (including U-shaped and structural priors),

import numpy as np

def _fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.empty(0)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    # Remove tiny negative values from FFT roundoff
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Compute objective: 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * np.max(c) / (s * s))


def _normalize(a, target_sum=None):
    """Normalize to non-negative, sum = target_sum (default len(a))."""
    n = len(a)
    target = float(target_sum) if target_sum is not None else float(n)
    a = np.array(a, dtype=np.float64, copy=True)
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, None, out=a)
    s = np.sum(a)
    if s < 1e-15:
        a.fill(target / n)
        return a
    a *= target / s
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _topk_grad(a, c, K, temp=0.0):
    """
    Gradient of soft/hard max of convolution.
    Returns grad[i] = 2 * sum_k w[k] * a[k-i] (valid indices).
    """
    n = len(a)
    L = len(c)
    K = max(1, min(K, L))
    grad = np.zeros(n, dtype=np.float64)

    if temp > 1e-8:
        # Softmax weights (only over top K for efficiency)
        if K < L:
            idx = np.argpartition(c, -K)[-K:]
        else:
            idx = np.arange(L)
        vals = c[idx]
        v_max = np.max(vals)
        w = np.exp((vals - v_max) / temp)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros_like(w)
            w[np.argmax(vals)] = 1.0
            w_sum = 1.0
        w /= w_sum
    else:
        if K < L:
            idx = np.argpartition(c, -K)[-K:]
        else:
            idx = np.arange(L)
        w = np.full(K, 1.0 / K, dtype=np.float64)

    for j in range(len(idx)):
        k = int(idx[j])
        wj = w[j]
        if wj < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        j_lo = k - i_hi
        j_hi = k - i_lo
        grad[i_lo:i_hi + 1] += 2.0 * wj * a[j_lo:j_hi + 1][::-1]

    return grad

            a[m:] = 1.0

            # Also try the "U" variant: high at both edges

            a2[m:n - m] = 1.0
            a2[n - m:] = ratio
            add(a2)

def _upsample(seq, n):
    """Interpolate sequence to length n."""
    seq = np.array(seq, dtype=np.float64)
    if len(seq) == n:
        return _normalize(seq)
    if len(seq) == 0:
        return _normalize(np.ones(n))
    xold = np.linspace(0, 1, len(seq))
    xnew = np.linspace(0, 1, n)
    return _normalize(np.interp(xnew, xold, seq))

    t_start = time.time()
    deadline = t_start + max(10.0, budget_s - 30.0)

    print(f"Starting search with budget {budget_s}s")

    # Length plan
    if budget_s < 300:
        lengths = [1024]
    elif budget_s < 900:
        lengths = [1024, 2048]
    elif budget_s < 1500:
        lengths = [1024, 2048, 4096]
    else:
        lengths = [1024, 2048, 4096, 8192]

    global_best = None
    global_score = np.inf
    seed_base = 42

    for li, n in enumerate(lengths):
        if time.time() > deadline - 60.0:
            break

        print(f"\n=== Length n={n} ===")
        prev = global_best

        starts = _make_starts(n, prev, seed=seed_base + li * 7)
        print(f"  Generated {len(starts)} starts, best start score = {starts[0][0]:.6f}")

        # Allocate time
        remain = max(10.0, deadline - time.time())
        # Budget weights: more time for later (longer) lengths
        is_last = (li == len(lengths) - 1)
        n_starts = min(len(starts), 4 if is_last else 3)
        time_each = remain / (n_starts + 1)  # +1 for final polish

        for si in range(n_starts):
            if time.time() > deadline - 30.0:
                break
            sc, seq = starts[si]
            t_this = max(5.0, min(time_each, deadline - time.time() - 30.0))
            if t_this < 3.0:
                break

            print(f"  Start {si}: score={sc:.6f}, budget={t_this:.1f}s")
            try:
                best_seq, best_sc = _local_search(seq, t_this, seed=seed_base + li * 100 + si, label=f"n{n}_s{si}")
            except Exception as e:
                print(f"  Error: {e}")
                continue

            if np.isfinite(best_sc) and best_sc < global_score:
                global_score = best_sc
                global_best = best_seq
                print(f"  *** New global best: {best_sc:.6f} ***")

        # Final polish at this length
        if time.time() < deadline - 30.0 and global_best is not None and len(global_best) == n:
            t_polish = max(10.0, min(120.0, deadline - time.time() - 30.0))
            print(f"  Polish phase: {t_polish:.1f}s")
            try:
                best_seq, best_sc = _local_search(global_best, t_polish, seed=seed_base + 900 + li, label=f"n{n}_polish")
                if np.isfinite(best_sc) and best_sc < global_score:
                    global_score = best_sc
                    global_best = best_seq
                    print(f"  *** New global best after polish: {best_sc:.6f} ***")
            except Exception:
                pass

    # Final: try a few more short runs at the best length with different seeds
    if time.time() < deadline - 30.0 and global_best is not None:
        n_best = len(global_best)
        remain = deadline - time.time() - 30.0
        if remain > 30.0:
            n_extra = max(1, int(remain / 60.0))
            n_extra = min(n_extra, 4)
            for ei in range(n_extra):
                if time.time() > deadline - 30.0:
                    break
                t_this = max(5.0, (deadline - time.time() - 30.0) / (n_extra - ei))
                # Perturb best and res
                rng_e = np.random.default_rng(seed_base + 5000 + ei)
                a_pert = global_best * np.exp(rng_e.normal(0, 0.15, n_best))
                a_pert = _normalize(a_pert)
                sc_pert = _score(a_pert)
                print(f"  Extra run {ei}: perturbed score={sc_pert:.6f}")
                try:
                    best_seq, best_sc = _local_search(a_pert, t_this, seed=seed_base + 5000 + ei, label=f"extra{ei}")
                    if np.isfinite(best_sc) and best_sc < global_score:
                        global_score = best_sc
                        global_best = best_seq
                        print(f"  *** New global best: {best_sc:.6f} ***")
                except Exception:
                    pass

    # Exact evaluation
    print(f"\n=== Final exact evaluation ===")
    if global_best is None:
        global_best = np.ones(1024)

    # Sanitize
    global_best = np.where(np.isfinite(global_best), global_best, 0.0)
    np.clip(global_best, 0.0, 1000.0, out=global_best)
    s = float(np.sum(global_best))
    if s < 0.01:
        global_best = np.ones(1024)

    # Try a few normalizations
    candidates = [global_best]
    # Also try rescaled versions
    for scale in (0.5, 1.0, 2.0):
        c = global_best * scale
        if np.max(c) <= 1000.0 and np.sum(c) >= 0.01:
            candidates.append(c)

    best_exact = global_best
    best_exact_score = np.inf

    for cand in candidates:
        try:
            lst = [float(x) for x in cand]
            ex = float(evaluate_sequence(lst))
            if np.isfinite(ex) and ex < best_exact_score:
                best_exact_score = ex
                best_exact = cand
        except Exception:
            pass

    print(f"Best exact score: {best_exact_score:.6f}")
    print(f"Sequence length: {len(best_exact)}")

    # Return as list
    result = [float(x) for x in best_exact]
    return result
