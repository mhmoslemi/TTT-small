import math


def fft_convolve(a):
    """Compute full convolution a*a using FFT for speed."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    np.maximum(result, 0.0, out=result)

def project_simplex(v):
    """Project v onto the standard probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute 2n*max(conv(a))/(sum(a))^2. Returns (score, conv)."""
    n = len(a)

        # Compute gradient
        g = compute_avg_gradient(y, active)

        # Gradient step + project
        a_new = project_simplex(y - step * g)

        # Score check
        if it % score_every == 0:
            sc_new, _ = compute_score(a_new)
            if sc_new < best_score:
                improvement = best_score - sc_new
                best_score = sc_new
                best_a = a_new.copy()
                if improvement > 1e-12:
                    stall = 0
                # Increase step if making good progress
                if improvement > 1e-6:
                    step = min(step * 1.02, 0.5)
            else:
                stall += 1

            # Step adaptation
            if stall > 50:
                step *= 0.92
            elif stall > 200:
                step *= 0.8
                # Reset momentum
                a_prev = best_a.copy()
                t = 1.0
                stall = 100
            elif stall > 600:
                # Smooth restart
                sigma = max(2.0, float(n) / 400.0)
                a = gaussian_smooth(best_a, sigma)
                a = normalize_nonneg(a)
                a_prev = a.copy()
                t = 1.0
                step = max(0.01, step * 0.5)
                stall = 0
            elif stall > 1500:
                # Perturbation restart
                noise_scale = 0.02 / np.sqrt(n)
                a = best_a + rng.normal(0.0, noise_scale, n)
                a = normalize_nonneg(a)
                a_prev = a.copy()
                t = 1.0
                step = 0.02
                stall = 0
                if verbose:
                    print(f"    Perturb restart it={it} score={best_score:.7f}")

        a_prev = a
        a = a_new
        it += 1

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic()
            remaining = max(0, deadline - elapsed)
            print(f"    it={it} score={best_score:.7f} step={step:.5f} |act|={len(active)} stall={stall} rem={remaining:.0f}s")

    # Final eval
    sc, _ = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def local_polish(a_init, duration, rng, verbose=True):
    """
    Local polishing: try small random perturbations and keep improvements.
    Uses coordinate-style updates and multiplicative adjustments.
    """
    n = len(a_init)
    a = normalize_nonneg(a_init.copy())
    best_a = a.copy()
    best_score, _ = compute_score(a)
    deadline = time.monotonic() + duration

    if verbose:
        print(f"    Polish start: score={best_score:.7f}")

    while time.monotonic() < deadline:
        # Try random multiplicative perturbation
        scale = rng.uniform(0.9, 1.1, n)
        a_new = a * scale
        a_new = normalize_nonneg(a_new)

        # Or try a coordinate swap
        if rng.random() < 0.3:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-0.01, 0.01)
            a_new = a.copy()
            a_new[i] += delta
            a_new[j] -= delta
            a_new = normalize_nonneg(a_new)

        sc_new, _ = compute_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Sometimes accept neutral steps (simulated annealing lite)
            if rng.random() < 0.01:
                a = a_new.copy()

    return best_a, best_score


def propose_candidate(budget_s=2400):
    """
    Main search function. Uses accelerated subgradient with adaptive active set,
    Nesterov momentum, backtracking step, periodic smoothing, and local polishing.
    Multiple initializations and restarts for robustness.
    Returns the best list of non-negative floats found.
    """
    rng = np.random.default_rng(42)
    total_deadline = time.monotonic() + float(budget_s) - 30.0

    # --- Get initial sequence ---
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    if n < 2:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    print(f"=== Sequence length: {n} ===")

    # --- Build candidates ---
    candidates = []
    names = []

    # Primary: given sequence
    S = arr.sum()
    if S > 1e-10:
        candidates.append(arr / S)
        names.append("given")
    else:
        candidates.append(np.ones(n) / n)
        names.append("given_flat")

    # Additional candidates
    extra_cands, extra_names = make_initializations(n, rng)
    candidates.extend(extra_cands)
    names.extend(extra_names)

    # Evaluate all
    best_cand = None
    best_cand_score = np.inf
    for i, (c, nm) in enumerate(zip(candidates, names)):
        sc, _ = compute_score(c)
        print(f"  Candidate '{nm}': score = {sc:.8f}")
        if sc < best_cand_score:
            best_cand_score = sc
            best_cand = c.copy()

    print(f"Best initial: {best_cand_score:.8f}")

    # --- Main optimization phase ---
    time_left = total_deadline - time.monotonic()
    # Allocate 70% to main FISTA, 20% to polish, 10% to alternatives
    main_budget = time_left * 0.65
    polish_budget = time_left * 0.20
    alt_budget = time_left * 0.15

    print(f"Phase 1: Main subgradient ({main_budget:.0f}s budget)")
    best_a, best_score = run_subgradient(
        best_cand, main_budget, rng, verbose=True
    )
    print(f"Phase 1 done: {best_score:.8f}")

    # --- Polish phase ---
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        print(f"Phase 2: Local polish ({polish_budget:.0f}s budget)")
        polish_a, polish_score = local_polish(
            best_a, min(polish_budget, time_left - 10), rng, verbose=True
        )
        if polish_score < best_score:
            best_a = polish_a
            best_score = polish_score
            print(f"Polish improved: {best_score:.8f}")
        else:
            print(f"Polish no improvement (best was {best_score:.8f})")

    # --- Alternative starts ---
    time_left = total_deadline - time.monotonic()
    if time_left > 60 and best_score > 1.503:
        print(f"Phase 3: Alternative start ({time_left - 10:.0f}s)")
        # Try from the second-best candidate
        sorted_cands = sorted(
            zip(candidates, names, [compute_score(c)[0] for c in candidates]),
            key=lambda x: x[2]
        )
        alt_cand = sorted_cands[1][0] if len(sorted_cands) > 1 else candidates[0]
        if np.sum(np.abs(alt_cand - best_a)) > 0.01:
            alt_a, alt_score = run_subgradient(
                alt_cand, min(time_left - 10, 600), rng, verbose=True
            )
            if alt_score < best_score:
                best_a = alt_a
                best_score = alt_score
                print(f"Alt improved: {best_score:.8f}")

    # --- Final polish with remaining time ---
    time_left = total_deadline - time.monotonic()
    if time_left > 20 and best_score > 1.503:
        print(f"Phase 4: Final polish ({time_left - 5:.0f}s)")
        final_a, final_score = local_polish(
            best_a, min(time_left - 5, 120), rng, verbose=True
        )
        if final_score < best_score:
            best_a = final_a
            best_score = final_score

    # --- Final verification ---
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.8f}")

    if not np.isfinite(final_score):
        print("Score is inf, falling back to best candidate")
        final_list = [float(x) for x in best_cand]
        final_score = float(evaluate_sequence(final_list))

    print(f"=== FINAL: {final_score:.8f} (n={n}) ===")
    return final_list
