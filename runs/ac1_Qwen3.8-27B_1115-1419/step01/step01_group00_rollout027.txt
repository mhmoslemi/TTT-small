import numpy as np
import time


def _fft_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size=size)
    c = np.fft.irfft(fa * fa, size=size)
    return c[:2*n-1]  # Full convolution


def _grad_at_max(a, k, n):
    """Gradient of conv(a,a)[k] w.r.t. a. Returns array of length n."""
    g = np.zeros(n)
    # conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * a[k-i]
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    indices = np.arange(i_lo, i_hi + 1)
    # d/d a[i] of conv[k] = a[k-i] (appears twice unless i == k-i)
    # Actually: conv[k] = sum_j a[j]*a[k-j], so d/da[i] = a[k-i] + a[k-i] = 2*a[k-i]
    # But if i == k-i (i.e., 2i == k), then it's just a[i] (derivative of a[i]^2 is 2a[i])
    # Wait no: if 2i == k, the term is a[i]*a[i] = a[i]^2, derivative is 2*a[i]
    # If 2i != k, the terms are a[i]*a[k-i] and a[k-i]*a[i], each contributing a[k-i] and a[i]
    # So total derivative = a[k-i] + a[k-i] = 2*a[k-i] (when i != k-i)
    # And when i == k-i (i.e., k is even and i = k/2), derivative = 2*a[i]
    # In both cases: d/da[i] = 2*a[k-i] (with a[k-i] = a[i] when 2i=k)
    g[indices] = 2.0 * a[k - indices]
    return g


def _project_simplex(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * np.max(c) / (s * s)


def _smooth_grad(a, tau, n):
    """Compute gradient of entropy-smoothed max of conv(a,a)."""
    c = _fft_conv(a)
    # Softmax weights
    mx = c.max()
    p = np.exp((c - mx) / tau)
    p /= p.sum()
    # Gradient: 2 * cross-correlation of p with a
    # (p * a) in the cross-correlation sense
    # For our purposes: g[i] = 2 * sum_j p[j] * a[j-i] (valid range)
    # This is a correlation: g = 2 * correlate(p_padded, a)
    # But p has length 2n-1 and a has length n.
    # Actually, the gradient of sum_j p_j * conv(a,a)[j] w.r.t. a_i is:
    # 2 * sum_j p_j * (d conv[j]/d a_i) = 2 * sum_j p_j * a[j-i] (valid ones)
    # = 2 * (p * a)[i] in the appropriate cross-correlation
    # Let's compute this via FFT for speed
    # cross-correlation of p (len 2n-1) with a (len n)
    size = 4 * n
    P = np.fft.rfft(p, size=size)
    A = np.fft.rfft(a, size=size)
    # Correlation: ifft(conj(P) * A) gives cross-correlation
    g = 2.0 * np.fft.irfft(np.conj(P) * A, size=size)[:n]
    return g, c


def _momentum_smooth(a, n, deadline, tag, it_limit, tau_init, tau_min, eta_init, momentum):
    """Smooth gradient descent with Nesterov momentum."""
    target_sum = a.sum()
    tau = tau_init

    v = np.zeros(n)

        g, c = _smooth_grad(a, tau, n)

        o = 2.0 * n * mc / (target_sum * target_sum)
        
        if o < best_o - 1e-10:
            best_o = o
            best_a = a.copy()
            stall = 0

        else:
            stall += 1
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * g
        # Normalize gradient
        mg = np.max(np.abs(v))
        if mg < 1e-14:

        if stall > 5000:
            eta = max(eta * 0.5, 1e-8)

def _topk_smooth_grad(a, k, n, top_k):
    """Gradient using top-k soft weighting of convolution values."""
    c = _fft_conv(a)
    # Find top-k values
   _THRESH = np.partition(c, -top_k)[-top_k]
    # Soft weight: exponential decay from threshold
    weights = np.where(c >= _THRESH, np.exp((c - _THRESH) / (c.max() - c.min() + 1e-10)), 0.0)
    weights /= weights.sum() + 1e-30
    
    # Compute gradient via FFT
    size = 4 * n
    P = np.fft.rfft(weights, size=size)
    A = np.fft.rfft(a, size=size)
    g = 2.0 * np.fft.irfft(np.conj(P) * A, size=size)[:n]
    return g, c

Uses FFT-accelerated convolution, Nesterov momentum, top-k active constraint

import numpy as np
import time


def _fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size=size)
    c = np.fft.irfft(fa * fa, size=size)
    return c[:2 * n - 1]


def _grad_smooth_fft(a, p):
    """Gradient of sum_j p[j]*conv(a,a)[j] w.r.t. a, via FFT cross-correlation.
    p has length 2n-1, a has length n. Returns array of length n."""
    n = len(a)
    size = 2 * (2 * n - 1)  # enough for correlation
    P = np.fft.rfft(p, size=size)
    A = np.fft.rfft(a, size=size)
    # cross-correlation: ifft(conj(P) * A)
    g = 2.0 * np.fft.irfft(np.conj(P) * A, size=size)[:n]
    return g


def _grad_hard(a, k, n):
    """Exact gradient of conv(a,a)[k] w.r.t. a."""
    g = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    if i_lo <= i_hi:
        indices = np.arange(i_lo, i_hi + 1)
        g[indices] = 2.0 * a[k - indices]
    return g


def _project(a, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)


def _obj_from_conv(c, n, sum_a):
    """Objective from precomputed convolution."""
    return 2.0 * n * float(np.max(c)) / (sum_a * sum_a)

        a_new = _project(a - eta * g, target_sum)
        c_new = _fft_conv(a_new)
        o_new = 2.0 * n * float(np.max(c_new)) / (target_sum * target_sum)
        
        if o_new > o + 1e-10:
            eta = max(eta * 0.7, 1e-10)
        elif o_new < o - 1e-12:
            eta = min(eta * 1.03, 0.3)
        a = a_new
        
        if stall > 5000:
            eta = max(eta * 0.4, 1e-10)
            stall = 0
    
    return best_a, best_o


def _hard_run(a, n, deadline, tag, it_limit):
    """Hard active-set descent on exact max with backtracking."""
    target_sum = float(a.sum())
    best_a = a.copy()
    best_o = _obj_from_conv(_fft_conv(a), n, target_sum)
    eta = 0.02
    stall = 0
    t0 = time.monotonic()
    
    for it in range(it_limit):
        if time.monotonic() >= deadline:
            break
        
        c = _fft_conv(a)
        k = int(np.argmax(c))
        mc = float(c[k])
        o = 2.0 * n * mc / (target_sum * target_sum)
        
        if o < best_o - 1e-11:
            best_o = o
            best_a = a.copy()
            stall = 0
            if it % 5000 == 0:
                print(f"  [{tag}] it={it} obj={o:.7f} eta={eta:.5f} t={time.monotonic()-t0:.0f}s")
        else:
            stall += 1
        
        g = _grad_hard(a, k, n)
        g -= g.mean()
        mg = np.max(np.abs(g))
        if mg < 1e-15:
            break
        g /= mg
        
        a_new = _project(a - eta * g, target_sum)
        c_new = _fft_conv(a_new)
        o_new = 2.0 * n * float(np.max(c_new)) / (target_sum * target_sum)
        
        if o_new > o + 1e-12:
            eta = max(eta * 0.6, 1e-11)
        else:
            eta = min(eta * 1.04, 0.15)
        a = a_new
        
        if stall > 4000:
            eta = max(eta * 0.3, 1e-11)
            stall = 0
    
    return best_a, best_o


def _targeted_perturb(a, n, rng):
    """Perturb specific to the active constraint (max lag)."""
    c = _fft_conv(a)
    k = int(np.argmax(c))
    target_sum = float(a.sum())
    b = a.copy()
    
    # Identify indices contributing to conv[k]
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    indices = np.arange(i_lo, i_hi + 1)
    
    # Perturb these indices: reduce contributions
    sigma = float(rng.uniform(0.1, 0.5))
    b[indices] *= np.exp(-sigma * rng.uniform(0.5, 1.5, size=len(indices)))
    
    # Also perturb a random block for diversity
    w = int(rng.integers(n // 50, n // 5))
    s = int(rng.integers(0, max(1, n - w)))
    b[s:s + w] *= np.exp(rng.normal(0, float(rng.uniform(0.05, 0.3)), size=w))
    
    return _project(b, target_sum)


def _random_perturb(a, n, rng, strength):
    """General random perturbation."""
    b = a.copy()
    target_sum = float(a.sum())
    
    mode = int(rng.integers(0, 4))
    if mode == 0:
        w = int(rng.integers(n // 20, n // 3))
        s = int(rng.integers(0, max(1, n - w)))
        b[s:s + w] *= np.exp(rng.normal(0, strength, size=w))
    elif mode == 1:
        # Multi-scale: perturb at 3 different scales
        for _ in range(3):
            w = int(rng.integers(max(2, n // 200), n // 10))
            s = int(rng.integers(0, max(1, n - w)))
            b[s:s + w] *= np.exp(rng.normal(0, strength * 0.5, size=w))
    elif mode == 2:
        # Periodic modulation
        period = int(rng.integers(7, 100))
        phase = float(rng.uniform(0, 2 * np.pi))
        mod = 1.0 + strength * 0.5 * np.sin(2 * np.pi * np.arange(n) / period + phase)
        b *= np.maximum(mod, 0.1)
    else:
        # Edge boost
        center = n // 2
        sig = n / float(rng.integers(3, 15))
        x = np.arange(n)
        boost = 1.0 + strength * 0.3 * np.exp(-(x - center) ** 2 / (2 * sig * sig))
        b *= boost
    
    return _project(b, target_sum)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search combining momentum smooth descent, top-k targeting,
    hard active-set, and structured perturbations with adaptive scheduling.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    # Try parent sequence
    a_init = None
    n = 3809
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and len(parent) == n and np.all(np.isfinite(parent)):
            a_init = np.clip(parent, 0.0, 1000.0)
            if a_init.sum() < 0.01:
                a_init = None
    except Exception:
        pass
    
    if a_init is None:
        a_init = np.ones(n)
    
    a = _project(a_init.copy(), float(a_init.sum()))
    best_a = a.copy()
    best_o = _obj_from_conv(_fft_conv(a), n, float(a.sum()))
    print(f"Init: n={n} obj={best_o:.7f}")
    
    # ---- Phase 1: Momentum smooth (30% time) ----
    p1_end = t_start + 0.30 * (budget_s - 30.0)
    print(f"Phase 1: momentum smooth, {p1_end - t_start:.0f}s budget")
    best_a, best_o = _momentum_smooth(best_a, n, p1_end, "mom1",
                                       it_limit=500000, tau_init=n / 3.0,
                                       tau_min=n * 0.003, eta_init=0.05, mu=0.75)
    print(f"After phase 1: obj={best_o:.7f}")
    
    # ---- Phase 2: Top-k smooth (15% time) ----
    p2_end = t_start + 0.45 * (budget_s - 30.0)
    print(f"Phase 2: top-k smooth, {p2_end - t_start:.0f}s budget")
    a2, o2 = _topk_run(best_a, n, p2_end, "topk50", it_limit=200000, top_k=50)
    if o2 < best_o:
        best_a, best_o = a2, o2
        print(f"  topk50: obj={best_o:.7f}")
    
    # Narrow to top-10
    p2b_end = min(t_start + 0.53 * (budget_s - 30.0), deadline - 20)
    if time.monotonic() < p2b_end:
        a2b, o2b = _topk_run(best_a, n, p2b_end, "topk10", it_limit=100000, top_k=10)
        if o2b < best_o:
            best_a, best_o = a2b, o2b
            print(f"  topk10: obj={best_o:.7f}")
    
    # ---- Phase 3: Hard refine (10% time) ----
    p3_end = t_start + 0.63 * (budget_s - 30.0)
    print(f"Phase 3: hard refine, {p3_end - t_start:.0f}s budget")
    a3, o3 = _hard_run(best_a, n, p3_end, "hard1", it_limit=200000)
    if o3 < best_o:
        best_a, best_o = a3, o3
        print(f"After phase 3: obj={best_o:.7f}")
    
    # ---- Phase 4: Perturb + refine loop (37% time) ----
    print(f"Phase 4: perturb+refine loop, {deadline - time.monotonic():.0f}s remaining")
    attempt = 0
    best_streak = 0
    
    while time.monotonic() < deadline - 30:
        attempt += 1
        improved = False
        
        # Choose perturbation type
        if attempt <= 15:
            strength = float(rng.uniform(0.05, 0.25))
            use_targeted = (attempt % 3 == 0)
        elif attempt <= 40:
            strength = float(rng.uniform(0.10, 0.40))
            use_targeted = (attempt % 3 == 0)
        else:
            strength = float(rng.uniform(0.15, 0.60))
            use_targeted = (attempt % 2 == 0)
        
        if use_targeted:
            a_p = _targeted_perturb(best_a, n, rng)
        else:
            a_p = _random_perturb(best_a, n, rng, strength)
        
        # Quick top-k smooth (20-30s)
        d1 = min(time.monotonic() + 25.0, deadline - 20.0)
        a_q, o_q = _topk_run(a_p, n, d1, f"tk{attempt}", it_limit=10000, top_k=30)
        if o_q < best_o:
            best_a, best_o = a_q, o_q
            improved = True
            best_streak = 0
            print(f"  P{attempt} topk: obj={best_o:.7f} (targeted={use_targeted})")
        
        # Quick hard (10-15s)
        d2 = min(time.monotonic() + 12.0, deadline - 15.0)
        a_q2, o_q2 = _hard_run(best_a, n, d2, f"h{attempt}", it_limit=30000)
        if o_q2 < best_o:
            best_a, best_o = a_q2, o_q2
            improved = True
            best_streak = 0
            print(f"  P{attempt} hard: obj={best_o:.7f}")
        
        if not improved:
            best_streak += 1
        else:
            best_streak = 0
        
        # Every 8th attempt: try momentum smooth from perturbed point
        if attempt % 8 == 0 and time.monotonic() < deadline - 40:
            d3 = min(time.monotonic() + 20.0, deadline - 35.0)
            a_q3, o_q3 = _momentum_smooth(a_p, n, d3, f"m{attempt}",
                                           it_limit=5000, tau_init=n / 5.0,
                                           tau_min=n * 0.01, eta_init=0.03, mu=0.7)
            if o_q3 < best_o:
                best_a, best_o = a_q3, o_q3
                print(f"  P{attempt} momentum: obj={best_o:.7f}")
        
        # Every 12th attempt: try different n
        if attempt % 12 == 6 and time.monotonic() < deadline - 50:
            n_alt = int(rng.choice([3500, 4200, 5000, 4500]))
            a_alt = _project(np.ones(n_alt) * rng.uniform(0.8, 1.2, size=n_alt), n_alt)
            d4 = min(time.monotonic() + 25.0, deadline - 40.0)
            a_alt2, o_alt = _momentum_smooth(a_alt, n_alt, d4, f"alt{attempt}",
                                             it_limit=5000, tau_init=n_alt / 4.0,
                                             tau_min=n_alt * 0.01, eta_init=0.04, mu=0.7)
            d5 = min(time.monotonic() + 10.0, deadline - 30.0)
            a_alt3, o_alt2 = _hard_run(a_alt2, n_alt, d5, f"altd{attempt}", it_limit=10000)
            if o_alt2 < best_o:
                best_a, best_o = a_alt3, o_alt2
                n = n_alt
                print(f"  Alt n={n} P{attempt}: obj={best_o:.7f}")
    
    # ---- Final: last hard refinement ----
    if time.monotonic() < deadline - 5:
        best_a, best_o = _hard_run(best_a, n, deadline - 2.0, "FINAL", it_limit=100000)
        print(f"After final hard: obj={best_o:.7f}")
    
    # Build result
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    
    # Validate
    try:
        sc = float(evaluate_sequence(result))
        print(f"FINAL: n={len(result)} obj_computed={best_o:.7f} evaluated={sc:.7f}")
    except Exception:
        sc = float('inf')
        print(f"FINAL: n={len(result)} obj_computed={best_o:.7f} eval FAILED")
    
    if not np.isfinite(sc):
        print("Fallback to uniform!")
        result = [1.0] * 3809
    
    return result
