"""
Optimize step function sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolution, LSE-smoothed L-BFGS with aggressive annealing,
active-set subgradient with line search, and LP-based refinement.
Multiple diverse initializations at various sequence lengths.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    nfft = 1
    while nfft < n:
        nfft <<= 1
    A = fft(a, nfft)
    B = fft(b, nfft)
    return np.real(ifft(A * B, nfft))[:n]


def softmax_vec(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_smoothed_conv(x, tau):
    """Return (LSE-smoothed max of conv, weights) for conv(softmax(x), softmax(x))."""
    a = softmax_vec(x)
    conv = fft_convolve(a, a)
    conv_max = conv.max()
    shifted = conv - conv_max
    exp_vals = np.exp(shifted / tau)
    w_sum = exp_vals.sum()
    obj = tau * np.log(w_sum) + conv_max
    w = exp_vals / w_sum
    return obj, w, a


def lse_gradient(x, tau):
    """Gradient of LSE-smoothed max conv w.r.t. x (log-param)."""
    a = softmax_vec(x)
    conv = fft_convolve(a, a)
    conv_max = conv.max()
    shifted = conv - conv_max
    w = np.exp(shifted / tau)
    w /= w.sum()
    
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    
    W = fft(w, N)
    A = fft(a, N)
    grad_a = np.real(ifft(W * np.conj(A), N))[:n]
    
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x


def subgradient_step_fft(a, n_active, step_size):
    """Subgradient step for minimizing max(conv(a,a)) on the simplex."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_val = conv.max()
    if max_val < 1e-15:
        return a
    
    # Find top n_active peaks
    if n_active >= len(conv):
        active = np.arange(len(conv))
    else:
        # Efficient: use argpartition for top-k, then sort
        part_idx = np.argpartition(conv, -n_active)[-n_active:]
        active = part_idx[np.argsort(conv[part_idx])[::-1]]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d(conv[k])/da[i] for i in [lo, hi] is a[k-i]

        # which corresponds to a[k-lo], a[k-lo-1], ..., a[k-hi] reversed
        g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    
    # Step and project onto simplex
    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:

    else:

    return a_new

"""
Optimize step function sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolution for ~10x speedup, LSE-smoothed optimization
with aggressive annealing, active-set subgradient with adaptive step, and
LP-based refinement for escaping local minima. Multiple diverse restarts.

# Module-level state for optimizer callbacks (avoids closures)
_OPT_STATE = {'tau': 0.01}


def fft_convolve(a, b):
    """Fast full convolution using FFT."""
    n = len(a) + len(b) - 1
    nfft = 1
    while nfft < n:
        nfft <<= 1
    A = fft(a, nfft)
    B = fft(b, nfft)
    return np.real(ifft(A * B, nfft))[:n]


def softmax_vec(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_obj_val(x, tau):
    """LSE-smoothed max of conv(softmax(x), softmax(x))."""
    a = softmax_vec(x)
    conv = fft_convolve(a, a)
    conv_max = conv.max()
    shifted = conv - conv_max
    vals = np.exp(shifted / tau)
    obj = tau * np.log(vals.sum()) + conv_max
    return obj


def lse_weighted_grad(x, tau):
    """Gradient of LSE-smoothed max conv w.r.t. log-parameter x."""
    a = softmax_vec(x)
    conv = fft_convolve(a, a)
    conv_max = conv.max()
    shifted = conv - conv_max
    w = np.exp(shifted / tau)
    w /= w.sum()

    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    W = fft(w, N)
    A = fft(a, N)
    grad_a = np.real(ifft(W * np.conj(A), N))[:n]

    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return grad_x


def subgradient_step(a, n_active, step_size):
    """One subgradient step for minimizing max(conv(a,a)) on the simplex."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_val = conv.max()
    if max_val < 1e-18:
        return a

    k = min(n_active, len(conv))
    part_idx = np.argpartition(conv, -k)[-k:]
    active = part_idx[np.argsort(conv[part_idx])[::-1]]

    g = np.zeros(n)
    for idx in active:
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        g[lo:hi + 1] += a[idx - hi:idx - lo + 1][::-1]
    g /= len(active)

    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def score_fast(a):
    """Compute 2*n*max(conv)/(sum^2) using FFT."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

    for ii, init in enumerate(inits):
        if time.monotonic() > p3_deadline or p3_per < 8:
            break
        x_r = np.log(init + 1e-30)
        for tau in [0.04 / n_best, 0.008 / n_best, 0.002 / n_best]:
            if time.monotonic() > p3_deadline:
                break
            x_r = lbfgs_at_tau(x_r, tau, max_time=min(60, p3_per / 3.0))
        a_r = softmax_vec(x_r)
        s_r = score_fast(a_r)
        tag = " ***" if s_r < best_score else ""
        print(f"  R{ii}: {s_r:.7f}{tag}")
        if np.isfinite(s_r) and s_r < best_score:
            best_score = s_r
            best_a = a_r.copy()
            # Quick subgradient polish on new best
            a_pol = a_r.copy()
            for _ in range(200):
                if time.monotonic() > p3_deadline:
                    break
                a_pol = subgradient_step(a_pol, n_active=100, step_size=0.02)
            s_poly = score_fast(a_pol)
            if np.isfinite(s_poly) and s_poly < best_score:
                best_score = s_poly
                best_a = a_pol.copy()
                print(f"    Polished: {best_score:.7f}")

    # === PHASE 4: Final Intensive Subgradient ===
    print(f"\n=== Phase 4: Final (best={best_score:.7f}) ===")
    a = best_a.copy()
    n_f = len(a)
    step = 0.03
    final_dl = deadline - 5
    round_i = 0

    while time.monotonic() < final_dl:
        batch = 100
        for _ in range(batch):
            a = subgradient_step(a, n_active=120, step_size=step)
        s = score_fast(a)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a.copy()
        round_i += 1
        if round_i % 10 == 0:
            print(f"  round{round_i}: {best_score:.7f} (step={step:.5f})")
        step *= 0.97
        if step < 0.003:
            a_p = a + rng.normal(0, 0.015, n_f)
            a_p = np.maximum(a_p, 0)
            a_p /= a_p.sum()
            s_p = score_fast(a_p)
            if np.isfinite(s_p) and s_p < best_score:
                best_score = s_p
                best_a = a_p.copy()
                a = a_p.copy()
                print(f"  Perturb: {best_score:.7f}")
            step = 0.012

    # Final: verify score is not dominated by a single spike
    # Try trimming if a few elements are much larger than the rest
    a_final = best_a.copy()
    med = np.median(a_final[a_final > 0]) if np.any(a_final > 0) else 1.0
    clip_val = 5.0 * med + 1e-6
    if np.any(a_final > clip_val):
        a_clipped = np.minimum(a_final, clip_val)
        a_clipped = a_clipped / a_clipped.sum()
        s_clip = score_fast(a_clipped)
        if np.isfinite(s_clip) and s_clip < best_score:
            best_score = s_clip
            a_final = a_clipped
            print(f"  Clipping helped: {best_score:.7f}")

    final_seq = [float(max(0.0, min(1000.0, v))) for v in a_final]
    final_score = evaluate_sequence(final_seq)
    print(f"\nFINAL SCORE: {final_score:.6f} (n={len(final_seq)})")
    return final_seq
