```
def fft_linear_convolve(a,b=None):
    a=np.asarray(a,dtype=np.float64)
    if b is None: b=a
    b=np.asarray(b,dtype=np.float64)

    return c
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / (sum(a))^2.

The implementation uses:
  - FFT-based convolution for fast exact objective evaluation,
  - softmax-parameterized L-BFGS refinement of a smoothed max objective,
  - a multiplicative mirror-descent local search,
  - a linear-programming best-response step (for moderate lengths),
  - coarse-to-fine length refinement.

The returned sequence is scaled to unit sum, which is valid because the
objective is scale-invariant and avoids the 1000 clipping cap.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize, linprog
    from scipy.sparse import csr_matrix
    HAS_SCIPY_OPT = True
except Exception:
    HAS_SCIPY_OPT = False


def _pair_score(pair):
    return pair[0]


def normalize_sum1(a):
    """Clip to non-negative finite values and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = int(a.size)
    if s <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s


def fft_linear_convolve(a, b=None):
    """Linear convolution via FFT. Inputs are assumed non-negative."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    b = np.asarray(b, dtype=np.float64)
    n = int(a.size)
    m = int(b.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    L = n + m - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:L]
    if c.size:
        np.maximum(c, 0.0, out=c)
    return c


def score_sum1(a, n=None):
    """Fast score for a sum-normalized sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = int(n if n is not None else a.size)
    if a.size == 0 or n == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return float("inf")
    a = a / s
    c = fft_linear_convolve(a)
    if c.size == 0:
        return float("inf")
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return float("inf")
    return float(2.0 * n * maxc)


def cross_corr_conv(p, a):
    """
    d_i = sum_k p_k a_{k-i}, valid indices only.
    Computed as a slice of cross-correlation using FFT.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    rev_a = a[::-1]
    conv = fft_linear_convolve(p, rev_a)
    start = n - 1
    stop = start + n
    if conv.size < stop:
        pad = np.zeros(max(0, stop - conv.size), dtype=np.float64)
        conv = np.concatenate((conv, pad))
    return conv[start:stop].copy()


def softmax_from_y(y):
    """Stable softmax, sum = 1."""
    y = np.asarray(y, dtype=np.float64)
    n = int(y.size)
    if n == 0:
        return np.ones(0, dtype=np.float64)
    maxy = float(np.max(y))
    if not np.isfinite(maxy):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    z = y - maxy
    a = np.exp(z)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s


def smooth_fun_grad_holder(x, tau, n, holder=None):
    """
    Objective and gradient wrt y[1:], where y = (0, x).
    Objective is log-sum-exp(score_unit / tau).
    holder[0], holder[1] are updated with the best exact score and y.
    """
    n = int(n)
    if n <= 1:
        return (1e100, np.zeros(0, dtype=np.float64))

    x = np.asarray(x, dtype=np.float64)
    if x.size != n - 1:
        x = np.resize(x, max(0, n - 1))
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.0)

    y = np.empty(n, dtype=np.float64)
    y[0] = 0.0
    y[1:] = x

    a = softmax_from_y(y)
    c = fft_linear_convolve(a)
    if c.size == 0:
        return (1e100, np.zeros(n - 1, dtype=np.float64))

    u = 2.0 * float(n) * c
    M = float(np.max(u))
    if not np.isfinite(M):
        return (1e100, np.zeros(n - 1, dtype=np.float64))

    if holder is not None and len(holder) >= 2 and M < holder[0]:
        holder[0] = float(M)
        holder[1] = y.copy()

    tau = float(tau)
    if tau <= 1e-12:
        p = np.zeros_like(u)
        p[int(np.argmax(u))] = 1.0
        logz = 0.0
    else:
        alpha = (u - M) / tau
        alpha = np.clip(alpha, -745.0, 0.0)
        e = np.exp(alpha)
        zsum = float(e.sum())
        if zsum <= 1e-300 or not np.isfinite(zsum):
            p = np.zeros_like(u)
            p[int(np.argmax(u))] = 1.0
            logz = 0.0
        else:
            p = e / zsum
            logz = float(np.log(zsum))

    d = cross_corr_conv(p, a)
    G = 4.0 * float(n) * d
    offset = float(np.dot(a, G))
    gy = a * (G - offset)
    g = gy[1:]
    g = np.where(np.isfinite(g), g, 0.0)

    obj = M + tau * logz
    if not np.isfinite(obj):
        obj = 1e100
    return (float(obj), g)


def run_lbfgs_refine(a, n, time_budget, seed=0):
    """Smoothed L-BFGS refinement in softmax y-parameters."""
    a = normalize_sum1(a)
    n = int(a.size)
    if n < 4 or time_budget < 10.0 or not HAS_SCIPY_OPT:
        return a, score_sum1(a, n)

    a_safe = np.maximum(a, 1e-12)
    s = float(a_safe.sum())
    if s <= 0:
        a_safe = np.full(n, 1.0 / float(n))
    else:
        a_safe = a_safe / s
    y = np.log(a_safe)
    y -= float(y[0])
    if not np.all(np.isfinite(y)):
        y = np.zeros(n, dtype=np.float64)

    best_y = y.copy()
    best_score = score_sum1(softmax_from_y(best_y), n)

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_budget))

    base = max(0.03, 0.055 * best_score)
    taus = []
    tau = base
    while tau > 0.00035 and len(taus) < 10:
        taus.append(float(tau))
        tau *= 0.62
    if not taus:
        taus.append(0.01)
    if taus[-1] > 0.0012:
        taus.append(0.0012)

    if n <= 256:
        max_iter_per_tau = 320
    elif n <= 512:
        max_iter_per_tau = 260
    elif n <= 1024:
        max_iter_per_tau = 190
    elif n <= 2048:
        max_iter_per_tau = 120
    else:
        max_iter_per_tau = 70

    for ti, tau in enumerate(taus):
        if time.monotonic() >= deadline - 10.0:
            break

        holder = [best_score, best_y.copy()]
        x0 = best_y[1:].copy()
        remaining = max(0.0, deadline - time.monotonic())
        iters_left = max(1, len(taus) - ti)
        suggested = int(remaining / (iters_left * 0.07))
        iters = max(20, min(max_iter_per_tau, suggested if suggested > 0 else max_iter_per_tau))

        options = {
            "maxiter": int(iters),
            "maxfun": int(iters * 18),
            "ftol": 1e-12,
            "gtol": 1e-7,
            "maxls": 30,
        }
        try:
            res = minimize(
                smooth_fun_grad_holder,
                x0,
                args=(tau, n, holder),
                jac=True,
                method="L-BFGS-B",
                options=options,
            )
        except Exception:
            res = None

        if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
            yfull = np.empty(n, dtype=np.float64)
            yfull[0] = 0.0
            yfull[1:] = np.asarray(res.x, dtype=np.float64)
            a_try = softmax_from_y(yfull)
            sc_try = score_sum1(a_try, n)
            if np.isfinite(sc_try) and sc_try < best_score:
                best_score = float(sc_try)
                best_y = yfull.copy()

        if np.isfinite(holder[0]) and holder[0] < best_score:
            best_score = float(holder[0])
            best_y = np.asarray(holder[1], dtype=np.float64).copy()

        a_best = softmax_from_y(best_y)
        sc_best = score_sum1(a_best, n)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = float(sc_best)
            best_y = np.log(np.maximum(a_best, 1e-300))
            best_y -= float(best_y[0])

        print(
            f"LBFGS n={n} tau={tau:.6f} exact={best_score:.9f} "
            f"elapsed={time.monotonic()-t0:.1f}s",
            flush=True
        )

    a_out = softmax_from_y(best_y)
    a_out = normalize_sum1(a_out)
    sc_out = score_sum1(a_out, n)
    return a_out, float(sc_out)


def lp_best_response(a):
    """
    For fixed a (sum arbitrary positive), solve
        minimize M
        s.t. (a*g)_k <= M for all k, sum(g)=sum(a), g>=0.
    Returns (g, M*) or (None, None).
    """
    if not HAS_SCIPY_OPT:
        return None, None
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n < 4:
        return None, None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return None, None

    i = np.tile(np.arange(n, dtype=np.int32), n)
    j = np.repeat(np.arange(n, dtype=np.int32), n)
    rows = (i + j)
    cols = j
    vals = np.tile(a, n)

    mask = vals != 0.0
    rows = rows[mask]
    cols = cols[mask]
    vals = vals[mask]

    rows_m = np.arange(2 * n - 1, dtype=np.int32)
    cols_m = np.zeros(2 * n - 1, dtype=np.int32)
    data_m = -np.ones(2 * n - 1, dtype=np.float64)

    all_rows = np.concatenate((rows_m, rows))
    all_cols = np.concatenate((cols_m, cols))
    all_data = np.concatenate((data_m, vals.astype(np.float64)))

    A_ub = csr_matrix((all_data, (all_rows, all_cols)), shape=(2 * n - 1, n + 1))

    eq_rows = np.zeros(n, dtype=np.int32)
    eq_cols = np.arange(1, n + 1, dtype=np.int32)
    eq_data = np.ones(n, dtype=np.float64)
    A_eq = csr_matrix((eq_data, (eq_rows, eq_cols)), shape=(1, n + 1))

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[0] = 1.0
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)
    b_eq = np.array([s], dtype=np.float64)
    bounds = [(0.0, None)] + [(0.0, 1000.0)] * n

    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None, None

    if res is not None and bool(getattr(res, "success", False)):
        x = np.asarray(res.x, dtype=np.float64)
        M = float(x[0])
        g = x[1:]
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs > 1e-12:
            g = g * (s / gs)
        if np.all(np.isfinite(g)):
            return g, M
    return None, None


def smooth_perturbation(a, rng, strength=0.3):
    """Smooth multiplicative perturbation, preserving sum 1."""
    a = normalize_sum1(a).astype(np.float64, copy=True)
    n = int(a.size)
    if n <= 1:
        return a

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, 5):
        amp = strength / (1.0 + 0.45 * m)
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
    a *= np.exp(z)

    if n > 6 and float(rng.random()) < 0.45:
        width = int(rng.integers(max(1, n // 200), max(2, n // 12 + 1)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        a[start:stop] *= np.exp(float(rng.normal(0.0, strength)))

    return normalize_sum1(a)


def run_lp_search(a, budget, seed=0, max_calls=None):
    """Alternating LP best-response local search."""
    a = normalize_sum1(a)
    n = int(a.size)
    if n < 4 or budget < 10.0 or not HAS_SCIPY_OPT or n > 1100:
        return a, score_sum1(a, n)

    best = a.copy()
    best_score = score_sum1(best, n)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget))

    if max_calls is None:
        est = 0.6 + 0.45 * (float(n) / 512.0)
        max_calls = int(min(45, max(3, float(budget) / (2.5 * est))))

    no_improve = 0
    calls = 0
    cur_score = best_score

    while calls < max_calls and time.monotonic() < deadline - 5.0:
        t_call = time.monotonic()
        a_scaled = a * float(n)
        g_scaled, Mstar = lp_best_response(a_scaled)
        call_time = time.monotonic() - t_call

        if g_scaled is None:
            break

        g = normalize_sum1(g_scaled)
        sc_g = score_sum1(g, n)

        improved = False
        cur_score = score_sum1(a, n)

        mix_ts = (1.0, 0.75, 0.55, 0.4, 0.25, 0.15, 0.08, 0.04)
        for t in mix_ts:
            cand = (1.0 - t) * a + t * g
            cand = normalize_sum1(cand)
            sc = score_sum1(cand, n)
            if not np.isfinite(sc):
                continue
            if sc < best_score:
                best = cand.copy()
                best_score = float(sc)
            if sc < cur_score and not improved:
                a = cand
                cur_score = sc
                improved = True
                break

        if not improved:
            if np.isfinite(sc_g) and sc_g < best_score:
                best = g.copy()
                best_score = float(sc_g)
            if np.isfinite(sc_g) and sc_g < cur_score:
                a = g.copy()
                cur_score = float(sc_g)
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 2:
                a = smooth_perturbation(best, rng, 0.25)
                cur_score = score_sum1(a, n)
                no_improve = 0
            elif best_score < cur_score:
                a = normalize_sum1(0.7 * best + 0.3 * a)
                cur_score = score_sum1(a, n)

        calls += 1
        print(
            f"LP n={n} call={calls} Mstar={float(Mstar):.5f} "
            f"cur={cur_score:.9f} best={best_score:.9f} "
            f"lp_elapsed={time.monotonic()-t0:.1f}s",
            flush=True
        )

        if call_time > max(3.0, 0.20 * float(budget)) and calls > 1:
            break

    sc_a = score_sum1(a, n)
    if best_score <= sc_a:
        return best, float(best_score)
    return a, float(sc_a)


def mirror_refine(a, n, time_budget, seed=0):
    """
    Multiplicative mirror descent using smooth double-peak averaging.
    """
    a = normalize_sum1(a)
    n = int(a.size)
    if n < 4 or time_budget < 5.0:
        return a, score_sum1(a, n)

    c = fft_linear_convolve(a)
    score = float(2.0 * n * float(np.max(c)))
    best = a.copy()
    best_score = score

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_budget))

    eta = 0.22
    beta = 0.35
    mom = np.zeros(n, dtype=np.float64)
    noise = 0.02
    accept_rel = 0.025
    tau = 0.045
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        if not np.all(np.isfinite(a)) or float(a.sum()) <= 1e-12:
            a = best.copy()
            c = fft_linear_convolve(a)
            score = float(2.0 * n * float(np.max(c)))
            mom = np.zeros(n, dtype=np.float64)

        u = 2.0 * float(n) * c
        M = float(np.max(u))

        alpha = (u - M) / max(tau, 1e-12)
        alpha = np.clip(alpha, -745.0, 0.0)
        e = np.exp(alpha)
        zsum = float(e.sum())
        if zsum <= 1e-300 or not np.isfinite(zsum):
            p = np.zeros_like(u)
            p[int(np.argmax(u))] = 1.0
        else:
            p = e / zsum

        if tau < 0.008:
            L = int(u.size)
            K = max(1, L // 100)
            if K < L:
                idx = np.argpartition(u, -K)[-K:]
            else:
                idx = np.arange(L)
            pt = np.zeros_like(u)
            pt[idx] = 1.0 / float(K)
            p = 0.5 * p + 0.5 * pt

        d = cross_corr_conv(p, a)
        G = 4.0 * float(n) * d
        G = G - float(np.dot(a, G))
        scale = float(np.max(np.abs(G)))
        if not np.isfinite(scale) or scale < 1e-13:
            stagnant += 1
            if stagnant > 250:
                a = smooth_perturbation(best, rng, 0.30)
                c = fft_linear_convolve(a)
                score = float(2.0 * n * float(np.max(c)))
                mom = np.zeros(n, dtype=np.float64)
                eta = 0.25
                accept_rel = 0.03
                noise = 0.03
                tau = 0.045
                stagnant = 0
            continue

        mom = beta * mom + (1.0 - beta) * (G / scale)
        logf = -eta * mom
        if noise > 1e-6:
            logf = logf + rng.normal(0.0, noise, n)
        logf = np.clip(logf, -4.0, 4.0)

        cand = a * np.exp(logf)
        cand = normalize_sum1(cand)
        c2 = fft_linear_convolve(cand)
        sc2 = float(2.0 * n * float(np.max(c2)))

        if np.isfinite(sc2) and sc2 < best_score:
            best = cand.copy()
            best_score = sc2

        accepted = False
        if np.isfinite(sc2) and sc2 <= score * (1.0 + accept_rel) + 1e-12:
            old_score = score
            a = cand
            c = c2
            score = sc2
            mom *= 0.82
            stagnant += 1
            accepted = True
            if sc2 < old_score * (1.0 - 1e-10):
                eta = min(1.10, eta * 1.035)
                accept_rel = max(0.0012, accept_rel * 0.998)
                noise = max(0.0015, noise * 0.997)
                tau = max(0.00065, tau * 0.995)
            else:
                eta = max(0.05, eta * 0.965)
                tau = max(0.00065, tau * 0.997)
        else:
            eta = max(0.05, eta * 0.72)
            mom *= 0.50
            stagnant += 1
            tau = max(0.00065, tau * 0.985)

        if stagnant > 320:
            a = smooth_perturbation(best, rng, 0.32)
            c = fft_linear_convolve(a)
            score = float(2.0 * n * float(np.max(c)))
            eta = 0.25
            accept_rel = 0.035
            noise = 0.035
            tau = 0.05
            mom = np.zeros(n, dtype=np.float64)
            stagnant = 0

        if it % 700 == 0 and accepted:
            print(
                f"MIRROR n={n} it={it} cur={score:.9f} best={best_score:.9f} "
                f"tau={tau:.6f} elapsed={time.monotonic()-t0:.1f}s",
                flush=True
            )

    sc_best = score_sum1(best, n)
    if np.isfinite(sc_best) and sc_best < best_score:
        best_score = float(sc_best)
    return best, float(best_score)


def upsample_sequence(seq, n):
    """Linear interpolation to length n, then sum-1 normalize."""
    seq = np.asarray(seq, dtype=np.float64)
    n = int(n)
    if seq.size == 0:
        return normalize_sum1(np.ones(n))
    if seq.size == 1:
        return normalize_sum1(np.full(n, seq[0]))
    m = int(seq.size)
    xold = np.linspace(0.0, 1.0, m)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_sum1(out)


def make_geometric(n, alpha):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    z = float(alpha) * (t - 0.5)
    a = np.exp(np.clip(z, -40.0, 40.0))
    return normalize_sum1(a)


def make_power(n, p, side=0):
    n = int(n)
    t = np.linspace(1e-9, 1.0 - 1e-9, n)
    x = t if side == 0 else (1.0 - t)
    with np.errstate(divide="ignore", over="ignore", invalid="ignore"):
        a = np.power(np.maximum(x, 1e-12), float(p))
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1e6)
    return normalize_sum1(a)


def make_two_level(n, frac, ratio):
    n = int(n)
    m = int(round(float(frac) * n))
    m = max(1, min(n - 1, m))
    a = np.ones(n, dtype=np.float64)
    a[:m] = float(ratio)
    return normalize_sum1(a)


def make_block_sequence(n, rng, num_blocks=8):
    n = int(n)
    num_blocks = int(max(2, min(num_blocks, max(2, n // 2))))
    if n < 4:
        return normalize_sum1(np.ones(n))

    edges = np.round(np.linspace(0.0, n, num_blocks + 1)).astype(int)
    edges[0] = 0
    edges[-1] = n
    lengths = np.diff(edges)
    if np.any(lengths <= 0):
        edges = np.linspace(0.0, n, num_blocks + 1).astype(int)
        edges[0] = 0
        edges[-1] = n
        lengths = np.diff(edges)

    vals = rng.uniform(0.05, 3.0, num_blocks)
    if float(rng.random()) < 0.30:
        vals[int(rng.integers(0, num_blocks))] = 0.0
    a = np.repeat(vals, lengths)
    return normalize_sum1(a)


def make_fourier_sequence(n, rng, max_freq=8):
    n = int(n)
    if n < 2:
        return normalize_sum1(np.ones(n))
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, max_freq + 1):
        amp = rng.uniform(-1.6, 1.6) / (m ** 0.65)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)
        amp2 = rng.uniform(-1.2, 1.2) / (m ** 0.80)
        ph2 = rng.uniform(0.0, 2.0 * np.pi)
        z += amp2 * np.sin(2.0 * np.pi * m * t + ph2)
    a = np.exp(np.clip(z, -30.0, 30.0))
    return normalize_sum1(a)


def make_spline_sequence(n, rng, m=8):
    n = int(n)
    if n < 4:
        return normalize_sum1(np.ones(n))
    m = int(max(2, min(m, max(2, n // 4))))
    if m == 2:
        inner = np.array([], dtype=np.float64)
    else:
        inner = np.sort(rng.uniform(0.0, 1.0, m - 2))
    knots = np.concatenate(([0.0], inner, [1.0])).astype(np.float64)
    vals = rng.normal(0.0, 1.0, knots.size)
    drift = float(rng.uniform(-2.0, 2.0))
    vals += drift * (np.linspace(-1.0, 1.0, knots.size) - 0.5)
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(np.clip(np.interp(t, knots, vals), -30.0, 30.0))
    return normalize_sum1(a)


def add_scored_start(starts, seq):
    seq = normalize_sum1(seq)
    sc = score_sum1(seq, seq.size)
    if np.isfinite(sc) and sc < 8.0:
        starts.append((float(sc), seq))


def generate_starts(n, rng, prev_best, random_count=22):
    n = int(n)
    if n < 2:
        n = 2
    starts = []

    if prev_best is not None:
        add_scored_start(starts, upsample_sequence(prev_best, n))

    add_scored_start(starts, np.ones(n))

    for alpha in (-6.0, -4.0, -2.5, -1.5, -1.0, -0.6, 0.6, 1.0, 1.5, 2.5, 4.0, 6.0):
        add_scored_start(starts, make_geometric(n, alpha))

    for p in (-3.0, -2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 3.0):
        add_scored_start(starts, make_power(n, p, 0))
        add_scored_start(starts, make_power(n, p, 1))

    for frac in (0.05, 0.10, 0.18, 0.25, 0.33, 0.42, 0.50):
        for ratio in (0.0, 0.2, 0.5, 2.0, 4.0, 8.0):
            add_scored_start(starts, make_two_level(n, frac, ratio))

    rc = int(random_count)
    if n > 1024:
        rc = max(8, rc // 2)

    for i in range(rc):
        if i % 3 == 0:
            nb = int(rng.integers(3, max(4, 17)))
            add_scored_start(starts, make_block_sequence(n, rng, nb))
        elif i % 3 == 1:
            add_scored_start(starts, make_fourier_sequence(n, rng, max_freq=9))
        else:
            add_scored_start(starts, make_spline_sequence(n, rng, m=int(rng.integers(5, max(6, 15)))))

    if not starts:
        starts.append((2.0, normalize_sum1(np.ones(n))))

    starts.sort(key=_pair_score)
    return starts


def refine_candidate(a, n, budget, seed=0, allow_lp=True):
    """Run LP (optional), L-BFGS, then mirror on one candidate."""
    a = normalize_sum1(a)
    n = int(len(a))
    best = a.copy()
    best_score = score_sum1(best, n)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget))

    if budget < 8.0:
        return best, float(best_score)

    if allow_lp and n <= 1024:
        if n <= 512:
            lp_budget = min(0.38 * float(budget), 700.0)
        else:
            lp_budget = min(0.28 * float(budget), 350.0)
        lp_budget = min(lp_budget, max(0.0, deadline - time.monotonic() - 60.0))
        if lp_budget > 15.0:
            try:
                lab, scb = run_lp_search(best, lp_budget, seed=seed)
                if np.isfinite(scb) and scb < best_score:
                    best = lab
                    best_score = float(scb)
            except Exception:
                pass

    if best_score > score_sum1(a, n):
        a = best.copy()

    rem = max(0.0, deadline - time.monotonic())
    lb_budget = min(0.34 * float(budget), rem)
    lb_budget = min(lb_budget, max(0.0, deadline - time.monotonic() - 30.0))
    if lb_budget > 15.0:
        try:
            lab, scb = run_lbfgs_refine(a, n, lb_budget, seed=seed + 101)
            if np.isfinite(scb) and scb < best_score:
                best = lab
                best_score = float(scb)
        except Exception:
            pass

    if best_score > score_sum1(a, n):
        a = best.copy()

    rem = max(0.0, deadline - time.monotonic() - 1.0)
    if rem > 8.0:
        try:
            mam, scm = mirror_refine(a, n, rem, seed=seed + 202)
            if np.isfinite(scm) and scm < best_score:
                best = mam
                best_score = float(scm)
        except Exception:
            pass

    sc_a = score_sum1(a, n)
    if np.isfinite(sc_a) and sc_a < best_score:
        best = a
        best_score = float(sc_a)

    return normalize_sum1(best), float(best_score)


def run_level(n, budget, prev_best, seed=0):
    """Generate starts at length n, refine the best few."""
    n = int(n)
    budget = max(0.0, float(budget))
    if budget < 10.0 or n < 2:
        if prev_best is not None:
            seq = upsample_sequence(prev_best, n)
        else:
            seq = normalize_sum1(np.ones(n))
        return seq, score_sum1(seq, n)

    rng = np.random.default_rng(seed)
    random_count = 26 if n <= 512 else 16 if n <= 1024 else 10
    starts = generate_starts(n, rng, prev_best, random_count=random_count)
    if not starts:
        seq = normalize_sum1(np.ones(n))
        return seq, score_sum1(seq, n)

    top = starts[:3]
    best_seq = None
    best_score = float("inf")
    for sc, seq in top:
        if sc < best_score:
            best_seq = seq.copy()
            best_score = float(sc)

    t0 = time.monotonic()
    deadline = t0 + budget
    fracs = (0.55, 0.30, 0.15)
    for i in range(len(top)):
        time_left = max(0.0, deadline - time.monotonic())
        if time_left < 15.0:
            break
        time_this = fracs[i] * budget
        time_this = min(time_this, max(8.0, time_left - 5.0))
        allow_lp = (i == 0) or (n <= 512)
        sc0, seq0 = top[i]
        try:
            seq1, sc1 = refine_candidate(seq0, n, time_this, seed=int(seed + 700 * i), allow_lp=allow_lp)
        except Exception:
            seq1, sc1 = seq0, sc0
        if seq1 is None:
            seq1, sc1 = seq0, sc0
        if np.isfinite(sc1) and sc1 < best_score:
            best_seq = seq1.copy()
            best_score = float(sc1)
        if sc0 < best_score:
            best_seq = seq0.copy()
            best_score = float(sc0)

    if best_seq is None:
        best_seq = normalize_sum1(np.ones(n))
        best_score = score_sum1(best_seq, n)
    return best_seq, float(best_score)


def add_candidate(pool, seq, score):
    if seq is None:
        return
    score = float(score)
    if not np.isfinite(score):
        return
    seq = normalize_sum1(seq)
    pool.append((score, seq.copy()))
    pool.sort(key=_pair_score)
    if len(pool) > 25:
        pool.pop()


def sanitize_for_return(seq):
    """Return a plain Python list of finite non-negative floats."""
    a = np.asarray(seq, dtype=np.float64)
    if a.size == 0:
        return [1.0]
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    mx = float(a.max())

    if s < 0.01:
        if mx > 0.0:
            scale = max(1.0, 1.0 / s)
            if scale * mx <= 1000.0:
                a = a * scale
                s = float(a.sum())
                mx = float(a.max())
        if s < 0.01:
            n = int(a.size)
            return [1.0 / float(n) if n > 0 else 1.0] * max(1, n)

    if mx > 1000.0 and mx > 0.0:
        a = a * (1000.0 / mx)
        s = float(a.sum())
        if s < 0.01:
            n = int(a.size)
            return [1.0 / float(n) if n > 0 else 1.0] * max(1, n)

    return [float(x) for x in a]


def eval_exact(seq):
    try:
        return float(evaluate_sequence([float(x) for x in np.asarray(seq, dtype=np.float64).tolist()]))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Main search entry point. Runs coarse-to-fine optimization and returns
    the sequence with best exact (or fast) score found.
    """
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    pool = []

    parent = None
    try:
        p = globals().get("height_sequence_1", None)
        if p is not None:
            parent = np.asarray(p, dtype=np.float64)
            if parent.size > 0:
                parent = normalize_sum1(parent)
                ps = score_sum1(parent, parent.size)
                add_candidate(pool, parent, ps)
    except Exception:
        parent = None

    if parent is None:
        parent = normalize_sum1(np.ones(512))
        add_candidate(pool, parent, score_sum1(parent, 512))

    global_best_seq = parent.copy()
    global_best_score = score_sum1(parent, parent.size)

    if budget < 60.0:
        return sanitize_for_return(global_best_seq)

    reserve = 90.0
    total_time = max(30.0, budget - reserve)

    if budget < 450.0:
        levels = [(512, 0.55), (1024, 0.45)]
    elif budget < 1100.0:
        levels = [(512, 0.30), (1024, 0.70)]
    else:
        levels = [(512, 0.14), (1024, 0.40), (2048, 0.46)]

    for idx, (n, frac) in enumerate(levels):
        if time.monotonic() > t0 + total_time - 30.0:
            break
        b = frac * total_time
        if b < 25.0:
            continue
        n = int(n)
        try:
            seq, sc = run_level(n, b, global_best_seq, seed=int(seed + 1000 + 137 * idx))
        except Exception:
            seq, sc = None, float("inf")

        if seq is not None and np.isfinite(sc):
            add_candidate(pool, seq, sc)
            if sc < global_best_score:
                global_best_score = float(sc)
                global_best_seq = seq.copy()

    # If time remains and best length is still small, try one longer length.
    if time.monotonic() < t0 + total_time - 120.0 and global_best_seq.size <= 1024:
        try:
            n2 = int(min(2048, max(2, 2 * global_best_seq.size)))
            b2 = max(0.0, min(420.0, t0 + total_time - 90.0 - time.monotonic()))
            if b2 > 80.0 and n2 != int(global_best_seq.size):
                seq, sc = run_level(n2, b2, global_best_seq, seed=int(seed + 91000))
                if seq is not None and np.isfinite(sc):
                    add_candidate(pool, seq, sc)
                    if sc < global_best_score:
                        global_best_score = float(sc)
                        global_best_seq = seq.copy()
        except Exception:
            pass

    # Exact validation of best fast-score candidates.
    pool.sort(key=_pair_score)
    best_exact_score = float("inf")
    best_exact_seq = None

    # Ensure the current global best is tested even if not in top pool.
    candidates = []
    seen = 0
    for sc, seq in pool[:14]:
        if seen > 16:
            break
        arr = np.asarray(seq, dtype=np.float64)
        sig = (int(arr.size), float(np.sum(arr[:5])))
        if all(abs(arr[:6] - s[1]) > 1e-6 or arr.size != s[0] for s in candidates):
            pass
        ex = eval_exact(arr)
        if np.isfinite(ex) and ex < best_exact_score:
            best_exact_score = ex
            best_exact_seq = arr.copy()
        seen += 1

    ex = eval_exact(global_best_seq)
    if np.isfinite(ex) and (best_exact_seq is None or ex < best_exact_score):
        best_exact_score = ex
        best_exact_seq = global_best_seq.copy()

    if best_exact_seq is None or not np.isfinite(best_exact_score):
        best_exact_seq = global_best_seq

    result = sanitize_for_return(best_exact_seq)
    try:
        final_exact = eval_exact(result)
        if np.isfinite(final_exact):
            print(f"FINAL proposed n={len(result)} exact_score={final_exact:.9f}", flush=True)
        else:
            fast_sc = score_sum1(result, len(result))
            print(f"FINAL proposed n={len(result)} fast_score={fast_sc:.9f} exact=inf", flush=True)
    except Exception:
        pass

    return result
```
