import time

    n = len(a)
    size = 2 * n - 1
    N = 1

        N <<= 1
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:size].copy()

    x = x / tau
    x -= x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(x, 1.0 / len(x))
    return e / s

    p: length 2n-1 (softmax weights over convolution coefficients)

def _hard_phase(a, n, deadline, tag, it_limit=1000000):

    best_a = a.copy()

    beta = 0.9

        bw = b[idx] - b[idx[0]] + 0.1
        bw /= bw.sum()

            g += w * _grad_hard(a, ki)

def _perturb(a, n, rng, mode=0):
    """Apply a perturbation to a and project back to simplex."""
    b = a.copy()
    if mode == 0:
        # Block multiplicative
        w = max(2, n // 30)
        s = int(rng.integers(0, n - w + 1))
        b[s:s+w] *= np.exp(rng.normal(0, 0.15, size=w))
    elif mode == 1:
        # Additive Gaussian
        b += rng.normal(0, 0.01, size=n)
    elif mode == 2:
        # Frequency domain
        f = np.fft.fft(a)
        mask = np.ones(n, dtype=bool)
        n_mask = max(1, n // 20)
        idx = rng.choice(n, n_mask, replace=False)
        f[idx] *= np.exp(1j * rng.normal(0, 0.3, size=n_mask))
        b = np.fft.ifft(f).real
    elif mode == 3:
        # Anti-peak: reduce mass where it contributes most to the max
        conv = _fft_conv(a)
        k_max = int(np.argmax(conv))
        # Gradient of b_k_max
        g = _grad_hard(a, k_max)
        g -= g.mean()
        b -= 0.05 * g
    elif mode == 4:
        # Swap two random blocks
        w = max(2, n // 20)
        s1 = int(rng.integers(0, n - w))
        s2 = int(rng.integers(0, n - w))
        if abs(s1 - s2) > w:
            b[s1:s1+w], b[s2:s2+w] = b[s2:s2+w].copy(), b[s1:s1+w].copy()
    return _project(b)
