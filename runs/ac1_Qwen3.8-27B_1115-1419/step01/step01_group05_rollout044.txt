```python
"""
Multi-stage optimization for low-peak nonnegative self-convolutions.
Combines LP-based direction finding with LSE gradient descent.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s

def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]

def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb

def _lse_and_p(b, beta):
    if not b.size:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val

def _weighted_gradient(a, weights, fft_size, n):
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    h = np.where(h > 0.0, h, 0.0)
    return 2.0 * h

def _make_exponential(n, alpha=1.0):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)

def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    s = float(old.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)

def _build_lp_matrix(f, n):
    """Build the constraint matrix for the LP.
    A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    Shape: (2n-1, n)
    """
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # f[j] contributes to rows k = j, j+1, ..., j+n-1
        # i.e., row k = j + i where i = 0, ..., n-1 (and k < 2n-1)
        for i in range(n):
            k = j + i
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _lp_direction(f, M, time_limit=10.0):
    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0.
    Returns the optimal g (unnormalized) or None if failed."""
    n = f.size
    A = _build_lp_matrix(f, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
    
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, 
                        method='highs', options={'time_limit': time_limit, 'disp': False})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def _lse_gradient_descent(a, deadline, rng, alpha_init=0.1, beta_init=10.0):

    beta = beta_init

        if s <= 1e-300:

            fail += 1
            it += 1

            b = b_new
```

```python
"""
Multi-stage optimization for low-peak nonnegative self-convolutions.
Combines diverse parametric initialization, LSE gradient descent,
LP-based direction finding, and length scaling to minimize the score
2*n*max(a*a)/(sum a)^2 for non-negative sequences.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
```

"""
Multi-stage optimization for low-peak nonnegative self-convolutions.
Combines diverse parametric initialization, LSE gradient descent,
LP-based direction finding, and length scaling to minimize the score
2*n*max(a*a)/(sum a)^2 for non-negative sequences.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if n == 0 or s <= 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return a / s


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _score_of(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(a, b_fft)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _lse_and_p(b, beta):
    if not b.size:
        return np.ones(1), 0.0
    m = float(np.max(b))
    if beta <= 1.0:
        p = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return p, m
    arg = beta * (b - m)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        e = np.ones(b.size, dtype=np.float64)
        s = float(b.size)
    p = e / s
    val = m + float(np.log(s)) / beta
    return p, val


def _weighted_gradient(a, weights, fft_size, n):
    ar = a[::-1]
    A = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(A * W, fft_size).real
    h = conv[n - 1:2 * n - 1]
    h = np.where(h > 0.0, h, 0.0)
    return 2.0 * h


def _make_exponential(n, alpha=1.0):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_gaussian(n, sigma=0.3):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - 0.5) / float(sigma)) ** 2)
    return _normalize(a)


def _make_raised_cosine(n):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    return _normalize(a)


def _make_power_law(n, p=1.5):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (x + 0.01) ** float(p)
    return _normalize(a)


def _make_two_sided_exp(n, alpha=1.0):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * (x - 0.5)) + 0.01
    return _normalize(a)


def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    s = float(old.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _build_lp_matrix(f, n):
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_direction(f, M, time_limit=8.0):
    n = f.size
    if n > 2048:
        return None
    try:
        A = _build_lp_matrix(f, n)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                         method='highs', options={'time_limit': time_limit, 'disp': False})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def _perturb(a, rng, strength=0.3):
    noise = np.exp(rng.normal(0.0, float(strength), size=a.size))
    a2 = a * noise
    return _normalize(a2)


def _lse_gradient_descent(a, deadline, rng, alpha_init=0.15, beta_init=12.0, max_beta=400.0):
    a = _normalize(a)
    n = a.size
    b_fft = _next_pow2(2 * n - 1)
    g_fft = _next_pow2(3 * n - 2)

    b = _autocorr(a, b_fft)
    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    best_a = a.copy()
    best_score = score

    alpha = alpha_init
    beta = beta_init
    it = 0
    fail = 0
    last_improve = 0

    while time.monotonic() < deadline:
        p, smooth_old = _lse_and_p(b, beta)
        grad = _weighted_gradient(a, p, g_fft, n)
        grad_c = grad - float(np.mean(grad))
        gnorm = float(np.sqrt(np.sum(grad_c ** 2)))

        if gnorm <= 1e-300:
            break

        step = alpha / max(gnorm, 1e-10)
        v = a - step * grad_c
        a_new = np.maximum(v, 0.0)
        s = float(a_new.sum())
        if s <= 1e-300:
            alpha *= 0.5
            fail += 1
            it += 1
            continue
        a_new = a_new / s

        b_new = _autocorr(a_new, b_fft)
        maxb_new = float(np.max(b_new))
        score_new = 2.0 * float(n) * maxb_new
        _, smooth_new = _lse_and_p(b_new, beta)

        if smooth_new < smooth_old - 1e-14:
            a = a_new
            b = b_new
            maxb = maxb_new
            score = score_new
            alpha = min(2.0, alpha * 1.08)
            fail = 0
            last_improve = it
            if score < best_score - 1e-15:
                best_a = a.copy()
                best_score = score
        else:
            alpha *= 0.5
            fail += 1

        if it % 150 == 0 and it > 0:
            beta = min(max_beta, beta * 1.3)

        if fail > 100:
            a = _perturb(best_a, rng, 0.25)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = alpha_init
            beta = min(max(20.0, beta * 0.5), 80.0)
            fail = 0

        if it - last_improve > 2000 and it > 500:
            a = _perturb(best_a, rng, 0.4)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * float(n) * maxb
            alpha = alpha_init
            beta = 20.0
            fail = 0
            last_improve = it

        it += 1

    return best_a, best_score


def _lp_refine(f, deadline, rng, max_iters=30):
    f = _normalize(f)
    n = f.size
    if n > 1500:
        return f, _score_of(f)
    b_fft = _next_pow2(2 * n - 1)
    b = _autocorr(f, b_fft)
    M = float(np.max(b))
    current_score = 2.0 * float(n) * M
    best_a = f.copy()
    best_score = current_score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        f = _normalize(f)
        b = _autocorr(f, b_fft)
        M = float(np.max(b))
        current_score = 2.0 * float(n) * M

        g = _lp_direction(f, M, time_limit=6.0)
        if g is None:
            break

        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum

        best_t = None
        best_new_score = current_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
            new_f = (1.0 - t) * f + t * g_norm
            new_f = _normalize(new_f)
            new_b = _autocorr(new_f, b_fft)
            new_M = float(np.max(new_b))
            new_score = 2.0 * float(n) * new_M
            if new_score < best_new_score - 1e-12:
                best_new_score = new_score
                best_t = t

        if best_t is None:
            break

        f = (1.0 - best_t) * f + best_t * g_norm
        f = _normalize(f)
        if best_new_score < best_score:
            best_a = f.copy()
            best_score = best_new_score
            if it % 3 == 0:
                print(f"  LP iter {it}: score={best_new_score:.8f} (t={best_t:.2f})", flush=True)
        else:
            break

    return best_a, best_score
