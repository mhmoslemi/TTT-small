```python
def lp_direction(a, active_set, max_step=1.0):
    """Find the LP direction that minimizes the first-order increase in max convolution."""
    n = len(a)
    from scipy.optimize import linprog
    
    # Variables: d_0, ..., d_{n-1}, t
    # c = objective coefficients: [0, 0, ..., 0, 1] (minimize t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # (a*d)_k <= t for k in active_set
    # => sum_i a_i * d_{k-i} - t <= 0
    # d_i >= -a_i => -d_i <= a_i
    # Optionally: d_i <= a_i => d_i <= a_i
    
    m = len(active_set)
    A_ub = np.zeros((m + 2*n, n + 1))
    b_ub = np.zeros(m + 2*n)
    
    for row, k in enumerate(active_set):
        # (a*d)_k = sum_i a_i * d_{k-i} where valid i
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[row, i] = a[j]  # a_{k-i} * d_i
        A_ub[row, -1] = -1.0  # -t <= 0
        b_ub[row] = 0.0
    
    # -d_i <= a_i (i.e., d_i >= -a_i)
    for i in range(n):
        A_ub[m + i, i] = -1.0
        b_ub[m + i] = a[i]
    
    # d_i <= a_i * max_step (optional upper bound)
    for i in range(n):
        A_ub[m + n + i, i] = 1.0
        b_ub[m + n + i] = a[i] * max_step
    
    # Equality constraint: sum d_i = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    if result.success:
        return result.x[:n], result.x[-1]
    return None, None
```

```python
def single_constraint_direction(a, k):
    """
    Find d that minimizes (a*d)_k subject to sum(d)=0, -a <= d <= a.
    This is: minimize sum_i a[k-i] * d_i s.t. sum d_i = 0, -a_i <= d_i <= a_i.
    
    Solution: d_i = -a_i * sign(a[k-i] - threshold)
    where threshold is chosen so that sum(d) = 0.
    """
    n = len(a)
    # The "cost" of d_i is a[k-i] (where 0<=k-i<n)
    # We want to put d_i as negative as possible where cost is high,
    # and as positive as possible where cost is low.
    
    costs = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            costs[i] = a[j]
        else:
            costs[i] = 0  # d_i doesn't affect (a*d)_k
    
    # Now: minimize sum costs[i] * d[i] s.t. sum d = 0, -a[i] <= d[i] <= a[i]
    # Optimal: d[i] = -a[i] if costs[i] > threshold, d[i] = a[i] if costs[i] < threshold
    # Find threshold by sorting costs and finding the median-like value.
    
    # More precisely: we want to find t such that:
    # sum_i (a[i] * 1[costs[i] < t] - a[i] * 1[costs[i] > t]) = 0
    # (ignoring the case where costs[i] = t)
    
    # Sort by cost
    order = np.argsort(costs)
    # Try threshold at each sorted cost value
    # d[i] = -a[i] if costs[i] > t, +a[i] if costs[i] < t
    # sum d = sum_{costs < t} a[i] - sum_{costs > t} a[i] = 0
    
    # This means: sum_{costs <= t} a[i] = sum_{costs >= t} a[i] = total/2
    
    sorted_costs = costs[order]
    sorted_a = a[order]
    cumsum = np.cumsum(sorted_a)
    total = cumsum[-1]
    
    # Find the split point where cumsum crosses total/2
    half = total / 2
    idx = np.searchsorted(cumsum, half)
    idx = min(idx, n - 1)
    
    threshold = sorted_costs[idx]
    
    d = np.zeros(n)
    d[costs > threshold] = -a[costs > threshold]
    d[costs < threshold] = a[costs < threshold]
    # For costs == threshold, split to make sum exactly 0
    eq_mask = costs == threshold
    if np.any(eq_mask):
        current_sum = d.sum()
        eq_indices = np.where(eq_mask)[0]
        # We need to distribute -current_sum among the equal-cost positions
        # Each can contribute between -a[i] and a[i]
        remaining = -current_sum
        for i in eq_indices:
            max_contrib = a[i]
            if remaining >= 0:
                d[i] = min(max_contrib, remaining)
                remaining -= d[i]
            else:
                d[i] = max(-max_contrib, remaining)
                remaining -= d[i]
            if abs(remaining) < 1e-15:
                break
    
    # Verify: (a*d)_k should be negative (descent)
    # (a*d)_k = sum_i a[k-i] * d[i]
    val = np.dot(costs, d)
    
    return d, val
```

```python
def project_simplex(v):
    """Project v onto probability simplex {x : x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_obj_grad(a, tau):
    """LSE-smoothed max of a*a and its gradient."""
    c = np.convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum())
    obj = c_max + tau * log_sum
    w = exp_vals / exp_vals.sum()
    n = len(a)
    # grad[j] = 2 * sum_m w[j+m] * a[m]
    grad = 2.0 * np.correlate(w, a, mode='full')[:n]
    # Project gradient to tangent space of simplex (remove mean)
    grad = grad - np.mean(grad)
    return obj, grad

        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def run_fista_lse(a, deadline, initial_tau=None, min_tau=None):
    """FISTA on LSE-smoothed objective."""
    n = len(a)
    if initial_tau is None:
        initial_tau = 0.1 / n
    if min_tau is None:
        min_tau = 0.002 / n
    
    best_a = a.copy()
    best_score = safe_evaluate(a)
    
    tau = initial_tau
    x = a.copy()
    z = a.copy()
    t_k = 1.0
    it = 0
    
    while time.monotonic() < deadline:
        # Decrease tau
        if (it + 1) % 100 == 0:
            tau = max(min_tau, tau * 0.8)
        
        obj, grad = lse_obj_grad(z, tau)
        # Step size (conservative)
        lip = 8.0 * np.max(a) * (2 * n) / tau + 1e-10
        step = 1.0 / lip
        step = min(step, 0.5 * np.max(a) / (np.max(np.abs(grad)) + 1e-15))
        
        x_new = z - step * grad
        x_new = project_simplex(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z = project_simplex(z)
        x = x_new
        t_k = t_new
        it += 1
        
        if it % 200 == 0:
            score = safe_evaluate(x)
            if score < best_score:
                best_score = score
                best_a = x.copy()
            raw_max = np.convolve(x, x).max()
            print(f"  FISTA it {it}: tau={tau:.6f}, lse={obj:.8f}, raw_max={raw_max:.8f}, score={score:.6f}")
    
    # Final evaluation
    score = safe_evaluate(x)
    if score < best_score:
        best_score = score
        best_a = x.copy()
    
    return best_a, best_score


def run_subgradient(a, deadline, momentum=0.9):
    """Accelerated subgradient with exact max."""
    n = len(a)
    best_a = a.copy()
    best_score = safe_evaluate(a)
    
    v = np.zeros(n)  # momentum
    step = 0.05
    it = 0
    rng = np.random.default_rng(99)
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        c_max = c.max()
        
        # Active set: top K where c is within 5% of max
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        K = min(80, len(active))
        active = active[np.argsort(c[active])[::-1][:K]]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= len(active)
        
        # Project to tangent space
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        # Check improvement
        new_max = np.convolve(a_new, a_new).max()
        if new_max < c_max:
            a = a_new
        else:
            # Revert and reduce step
            step *= 0.95
        
        # Decay step
        step = max(0.001, step * 0.9995)
        
        if it % 50 == 0:
            score = safe_evaluate(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if it % 200 == 0:
                print(f"  SG it {it}: max={np.convolve(a,a).max():.8f}, score={score:.6f}, step={step:.6f}")
        
        # Occasional perturbation
        if it % 500 == 0 and step < 0.01:
            a_pert = a + rng.normal(0, 0.003 * np.max(a), n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            if np.convolve(a_pert, a_pert).max() < np.convolve(a, a).max():
                a = a_pert
                step = 0.02
    
    score = safe_evaluate(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def run_lbfgs_lse(a, deadline, n_taus=5):
    """L-BFGS on LSE-smoothed objective with log parameterization."""
    n = len(a)
    best_a = a.copy()
    best_score = safe_evaluate(a)
    
    x0 = np.log(a + 1e-30)
    taus = np.logspace(np.log10(0.05 / n), np.log10(0.003 / n), n_taus)
    
    for tau in taus:
        if time.monotonic() > deadline:
            break
        
        def obj(x_):
            a_t = np.exp(x_ - x_.max())
            a_t = a_t / a_t.sum()
            c = np.convolve(a_t, a_t)
            c_max = c.max()
            shifted = (c - c_max) / tau
            shifted = np.clip(shifted, -50, 50)
            return c_max + tau * np.log(np.exp(shifted).sum())
        
        def grad(x_):
            a_t = np.exp(x_ - x_.max())
            a_t = a_t / a_t.sum()
            c = np.convolve(a_t, a_t)
            c_max = c.max()
            shifted = (c - c_max) / tau
            shifted = np.clip(shifted, -50, 50)
            w = np.exp(shifted)
            w /= w.sum()
            g = 2.0 * np.correlate(w, a_t, mode='full')[:n]
            # Chain rule through softmax-like parameterization
            # d/dx exp(x - max) / sum(exp(x-max))
            # = a_t * (1 - a_t) * sign-like
            # Simplified: gradient wrt x is a_t * (g - dot(a_t, g))
            dot = np.dot(a_t, g)
            return a_t * (g - dot)
        
        try:
            res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-12,
                                   'maxfun': 500, 'maxls': 30})
            x0 = res.x
        except Exception:
            break
        
        a_t = np.exp(x0 - x0.max())
        a_t = a_t / a_t.sum()
        score = safe_evaluate(a_t)
        if score < best_score:
            best_score = score
            best_a = a_t.copy()
    
    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search function. Combines multiple optimization strategies:
    1. L-BFGS with LSE for smooth initial optimization
    2. Water-filling steepest descent (fast O(n log n) steps)
    3. FISTA on LSE-smoothed objective
    4. Accelerated subgradient polishing
    
    Returns the best sequence found.
    """
    deadline = time.monotonic() + budget_s - 30
    start_time = time.monotonic()
    rng = np.random.default_rng(2024)
    
    # Initialize
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0)
        if a.sum() > 0.01:
            a = a / a.sum()
        else:
            a = np.ones(256, dtype=np.float64) / 256
    else:
        a = np.ones(256, dtype=np.float64) / 256
    
    n = len(a)
    best_score = safe_evaluate(a)
    best_a = a.copy()
    print(f"Initial: n={n}, score={best_score:.6f}")
    
    # Phase 1: L-BFGS with LSE (120s)
    print("\n=== Phase 1: L-BFGS with LSE ===")
    phase1_deadline = min(time.monotonic() + 120, deadline - 1800)
    a1, score1 = run_lbfgs_lse(a, phase1_deadline, n_taus=4)
    if score1 < best_score:
        best_score = score1
        best_a = a1.copy()
        print(f"  After L-BFGS: {best_score:.6f}")
    else:
        a1 = a.copy()
    
    # Try a couple of alternative initializations briefly
    alt_inits = []
    # Asymmetric
    x = np.linspace(1.5, 0.5, n)
    alt_inits.append(x / x.sum())
    # Two-block
    x = np.ones(n)
    x[:n // 2] = 1.3
    x[n // 2:] = 0.7
    alt_inits.append(x / x.sum())
    
    for idx, init in enumerate(alt_inits):
        if time.monotonic() > phase1_deadline:
            break
        small_deadline = min(time.monotonic() + 15, phase1_deadline)
        ai, si = run_lbfgs_lse(init, small_deadline, n_taus=2)
        if si < best_score:
            best_score = si
            best_a = ai.copy()
            a1 = ai.copy()
            print(f"  Alt init {idx} better: {best_score:.6f}")
    
    # Phase 2: Water-filling steepest descent (900s)
    print(f"\n=== Phase 2: Water-filling descent (start score={safe_evaluate(best_a):.6f}) ===")
    a_wf = best_a.copy()
    phase2_deadline = min(time.monotonic() + 900, deadline - 900)
    a2, score2 = run_water_filling(a_wf, phase2_deadline, K_lp=25, lp_interval=30)
    if score2 < best_score:
        best_score = score2
        best_a = a2.copy()
        print(f"  After water-filling: {best_score:.6f}")
    else:
        a2 = best_a.copy()
    
    # Phase 3: FISTA on LSE (600s)
    print(f"\n=== Phase 3: FISTA LSE (start score={safe_evaluate(best_a):.6f}) ===")
    a_fista = best_a.copy()
    phase3_deadline = min(time.monotonic() + 600, deadline - 300)
    a3, score3 = run_fista_lse(a_fista, phase3_deadline)
    if score3 < best_score:
        best_score = score3
        best_a = a3.copy()
        print(f"  After FISTA: {best_score:.6f}")
    else:
        a3 = best_a.copy()
    
    # Phase 4: Subgradient polishing (remaining time)
    print(f"\n=== Phase 4: Subgradient polish (start score={safe_evaluate(best_a):.6f}) ===")
    a_sg = best_a.copy()
    a4, score4 = run_subgradient(a_sg, deadline - 10, momentum=0.85)
    if score4 < best_score:
        best_score = score4
        best_a = a4.copy()
        print(f"  After subgradient: {best_score:.6f}")
    
    # Final check
    final_score = safe_evaluate(best_a)
    if final_score < best_score:
        best_score = final_score
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s ===")
    
    return [float(val) for val in best_a]
```
