"""
Hybrid subgradient search for minimal autocorrelation peak non-negative sequences.
Uses FFT-accelerated evaluation, multiple parallel candidates with diverse
hyperparameters, spectral equalization steps, and adaptive restarts.
Starts from provided SOTA if available, explores different lengths.

import numpy as np
import time

_INF = float('inf')
_EPS = 1e-300


def _normalize_nonneg(v):
    arr = np.asarray(v, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(4, dtype=np.float64) * 0.25
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * len(a) - 1]


def _score_fft(a, N):
    c = _fft_conv(a, N)
    c = np.maximum(c, 0.0)
    mx = float(c.max())
    return float(2.0 * len(a) * mx), int(c.argmax())


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return _INF
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return _INF
    c = np.convolve(a, a)
    mx = float(c.max())
    return float(2.0 * n * mx / (s * s))


def _log_softmax(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - float(w.max())
    e = np.exp(w)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(w.size, dtype=np.float64) / float(w.size)
    return e / s


def _log_of(a):
    a = np.maximum(a, _EPS)
    w = np.log(a)
    w = w - float(w.max())
    return w


def _subgrad_direction(a, c, m_idx, n):
    """Compute subgradient of max autocorrelation at index m_idx."""
    S = float(a.sum())
    M = float(c[m_idx])
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, m_idx - n + 1)
    hi = min(n - 1, m_idx)
    if hi >= lo:
        grad[lo:hi + 1] = a[m_idx - lo:m_idx - hi]
    dF = 2.0 * n * (grad - 2.0 * M) / (S * S)
    proj = float(np.dot(a, dF)) / (S * S)
    dF = dF - proj
    return dF


def _step_logspace(a, w, dF, eta, m_state, v_state, t):
    """Adam step in log-softmax space."""
    n = a.size
    grad_w = a * dF
    grad_w = grad_w - float(np.dot(a, grad_w))
    t += 1
    b1, b2 = 0.9, 0.999
    m_state = b1 * m_state + (1.0 - b1) * grad_w
    v_state = b2 * v_state + (1.0 - b2) * grad_w * grad_w
    mhat = m_state / (1.0 - b1 ** t)
    vhat = v_state / (1.0 - b2 ** t)
    dw = mhat / (np.sqrt(vhat) + 1e-8)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -20.0, 20.0)
    w = w - eta * dw
    w = np.where(np.isfinite(w), w, 0.0)
    w = w - float(w.max())
    a_new = _log_softmax(w)
    return a_new, w, m_state, v_state, t


def _spectral_equalize(a, N, strength=0.5):
    """Apply one spectral equalization step to flatten |FFT(a)|^2."""
    n = a.size
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = A / np.maximum(mag, _EPS)
    mag2 = mag * mag
    target = float(np.mean(mag2))
    # Soft equalization: push magnitudes toward target
    # New mag: geometric mean between current and target
    new_mag = np.sqrt(mag * np.sqrt(target))
    new_mag = new_mag ** strength + mag ** (1.0 - strength)
    # Reconstruct
    A_new = phase * new_mag
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s > 0 and np.isfinite(s):
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def _make_hann_init(n):
    j = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / max(n - 1, 1)))
    return h / float(h.sum())


def _make_super_gauss_init(n, p=4.0, sigma_frac=0.4):
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    g = np.exp(-((j - center) / (sigma_frac * n)) ** p)
    return g / float(g.sum())


def _make_blackman_init(n):
    j = np.arange(n, dtype=np.float64)
    b = 0.42 + 0.5 * np.cos(2.0 * np.pi * j / max(n - 1, 1)) + 0.08 * np.cos(4.0 * np.pi * j / max(n - 1, 1))
    b = np.maximum(b, 0.0)
    return b / float(b.sum())


def _load_sota():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 8:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0 or not np.isfinite(s):
        return None
    return arr / s


def _optimize_candidate(a, N, deadline, rng, lr0=0.05, beta_max=200.0,
                        eq_interval=50, eq_strength=0.3, restart_every=80,
                        tag=""):
    """Optimize a single candidate. Returns best found."""
    n = a.size
    a = _normalize_nonneg(a)
    w = _log_of(a)
    m_state = np.zeros(n, dtype=np.float64)
    v_state = np.zeros(n, dtype=np.float64)
    t = 0
    best_a = a.copy()
    best_score, _ = _score_fft(a, N)
    eta = lr0
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        score, m_idx = _score_fft(a, N)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        dF = _subgrad_direction(a, _fft_conv(a, N), m_idx, n)
        a, w, m_state, v_state, t = _step_logspace(a, w, dF, eta, m_state, v_state, t)

        # Spectral equalization every eq_interval steps
        if eq_interval > 0 and it % eq_interval == 0:
            a_eq = _spectral_equalize(a, N, eq_strength)
            s_eq, _ = _score_fft(a_eq, N)
            if s_eq < score:
                a = a_eq
                w = _log_of(a)

        # Adaptive learning rate
        if stall < 5:
            eta = min(0.15, eta * 1.02)
        elif stall > 20:
            eta = max(0.005, eta * 0.93)

        # Restart
        if restart_every > 0 and stall > restart_every:
            sig = float(rng.choice([0.05, 0.1, 0.2, 0.3]))
            w_pert = w + rng.normal(0.0, sig, size=n)
            a = _log_softmax(w_pert)
            w = w_pert
            m_state = np.zeros(n, dtype=np.float64)
            v_state = np.zeros(n, dtype=np.float64)
            t = 0
            eta = lr0
            stall = 0

        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _try_length(a_parent, target_n, rng):
    """Resample parent to target length (interpolation)."""
    from numpy import interp
    x_old = np.linspace(0, 1, len(a_parent))
    x_new = np.linspace(0, 1, target_n)
    a_new = interp(x_new, x_old, a_parent)
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s > 0:
        a_new = a_new / s
    return a_new
