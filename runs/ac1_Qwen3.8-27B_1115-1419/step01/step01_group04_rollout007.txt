"""
LP-based and gradient search for minimizing peak autocorrelation of
non-negative step functions. Score: 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find peak-reducing update directions,
combined with multiplicative gradient polish.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog

# --- Core utilities ---

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2*n - 1
    N = 1 << (L-1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return float(2.0 * n * c.max() / (s*s))

def norm_seq(a, target):
    a = np.array(a, dtype=np.float64, copy=True)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        a.fill(target / a.size)
    else:
        a *= target / s
    return np.clip(a, 0.0, 1000.0)

# --- LP solver ---

def lp_peak_reduction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector."""
    n = len(f)
    L = 2*n - 1
    
    # Build convolution matrix T (L x n)
    T = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        js = max(0, k - n + 1)
        je = min(k, n-1) + 1
        T[k, js:je] = f[js:je]
    
    # LP: minimize -sum(g) s.t. T@g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(L, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 300})
    
    if res.status != 0:
        return None
    
    g = np.clip(res.x, 0.0, None)
    return g

def lp_iteration(f, t=0.1, time_limit=60):
    """One LP-based update step. Returns (new_f, new_score)."""
    n = len(f)
    target = math.sqrt(2.0 * n)
    f = norm_seq(f, target)
    
    c = fft_conv(f)
    M = float(c.max())
    old_score = 2.0 * n * M / (target * target)
    
    t0 = time.monotonic()
    g = lp_peak_reduction(f, M)
    if g is None or g.sum() < 1e-15:
        return f, old_score
    
    g = g / g.sum() * target
    g = np.clip(g, 0.0, 1000.0)
    
    f_new = (1.0 - t) * f + t * g
    f_new = np.clip(f_new, 0.0, 1000.0)
    f_new = norm_seq(f_new, target)
    
    new_score = score(f_new)
    if not np.isfinite(new_score) or new_score > old_score * 1.1:
        return f, old_score
    
    return f_new, new_score

# --- Gradient-based polish ---

def topk_subgradient(a, c, K):
    n = len(a)
    L = 2*n - 1
    K = max(1, min(K, L))
    d = np.zeros(n)
    
    if K == 1:
        k = int(np.argmax(c))
        js = max(0, k - n + 1)
        je = min(k, n-1) + 1
        d[js:je] = a[k - (je-1) : k - js + 1][::-1]
        return 2.0 * d
    
    idx = np.argpartition(c, -K)[-K:]
    inv = 1.0 / K
    for k in idx:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n-1) + 1
        d[js:je] += inv * a[k - (je-1) : k - js + 1][::-1]
    return 2.0 * d

def gradient_polish(a, budget_s, seed=0):
    n = len(a)
    target = math.sqrt(2.0 * n)
    a = norm_seq(a, target)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    c = fft_conv(a)
    sc = score(a)
    best = a.copy()
    best_sc = sc
    
    eta = 0.2
    beta = 0.3
    mom = np.zeros(n)
    noise = 0.02
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < budget_s:
        it += 1
        L = 2*n - 1
        frac = max(0.005, 0.05 / (1.0 + it/500.0))
        K = max(1, min(int(L*frac), 256, n//4))
        
        grad = topk_subgradient(a, c, K)
        g = grad - grad.mean()
        sc_ = g.max() - g.min()
        if sc_ < 1e-14:
            stagnant += 1
            if stagnant > 200:
                break
            continue
        g /= sc_
        mom = beta*mom + (1-beta)*g
        
        log_step = -eta * mom + rng.normal(0, noise, n)
        log_step = np.clip(log_step, -2, 2)
        cand = a * np.exp(log_step)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = norm_seq(cand, target)
        
        c2 = fft_conv(cand)
        sc2 = score(cand)
        
        if not np.isfinite(sc2):
            eta *= 0.7
            mom *= 0.5
            stagnant += 1
            continue
        
        if sc2 < best_sc:
            best = cand.copy()
            best_sc = sc2
            stagnant = 0
        
        if sc2 <= sc * 1.03:
            a = cand
            c = c2
            sc = sc2
            mom *= 0.8
            if sc2 < best_sc:
                eta = min(0.8, eta*1.05)
            else:
                eta = max(0.05, eta*0.97)
            stagnant += 1
        else:
            eta = max(0.02, eta*0.7)
            mom *= 0.5
            stagnant += 1
        
        if stagnant > 500:
            # Restart
            a = best.copy() * (1 + 0.3*rng.normal(0, 1, n))
            a = norm_seq(a, target)
            c = fft_conv(a)
            sc = score(a)
            mom = np.zeros(n)
            eta = 0.25
            noise = 0.05
            stagnant = 0
    
    return best, best_sc

def upsample(seq, n):
    seq = np.asarray(seq, dtype=np.float64)
    if len(seq) == n:
        return seq.copy()
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, seq)

LP-guided search for low-peak-autocorrelation non-negative sequences.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP to find peak-reducing blend directions
2. Multiplicative gradient polish
3. Multi-resolution with diverse starts

import numpy as np
import time
import math
from scipy.optimize import linprog


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def obj(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def nsq(a, tgt):

    a = np.where(np.isfinite(a), a, 0.0)

        return seq.copy()
    return np.interp(np.linspace(0, 1, n), np.linspace(0, 1, m), seq)


def build_T(f):
    n = len(f)
    L = 2 * n - 1
    T = np.zeros((L, n))
    f = np.ascontiguousarray(f)
    for k in range(L):
        js = max(0, k - n + 1)
        je = min(k, n - 1) + 1
        T[k, js:je] = f[js:je]
    return T


def solve_lp(f, M, t_lim=120):
    n = len(f)
    T = build_T(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bnds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bnds,
                  method='highs', options={'time_limit': t_lim, 'presolve': True})
    if res.status != 0:
        return None
    g = np.maximum(res.x, 0.0)
    if g.sum() < 1e-12:
        return None
    return g

        return f, s0, False
