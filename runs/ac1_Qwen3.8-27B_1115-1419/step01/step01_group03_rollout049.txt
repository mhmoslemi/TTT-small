def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def soft_subgrad(a, beta):

    c_max = c.max()

    weights = np.exp(beta * (c - c_max) / max(abs(c_max), 1e-15))

Minimize 2n*max(conv(a))/(sum(a))^2 using FISTA with soft subgradient,
multiple diverse starts, adaptive hyperparameters, and simulated annealing restarts.

import numpy as np

def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_subgrad(a, beta):
    """Compute soft subgradient of max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n), c_max
    # Softmax weights (numerically stable)
    shifted = beta * (c - c_max) / c_max
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return 2.0 * a[::-1][:n] if n >= 1 else np.zeros(n), c_max
    weights /= w_sum
    # Only use non-negligible weights
    threshold = 1e-6 / len(c)
    active = np.where(weights > threshold)[0]
    g = np.zeros(n)
    for k in active:
        wk = weights[k]
        if wk < 1e-10:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            g[idx] += 2.0 * wk * a[k - idx]
    return g, c_max
