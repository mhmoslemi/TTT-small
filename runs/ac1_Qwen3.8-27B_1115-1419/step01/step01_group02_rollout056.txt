```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n) * fft(b, n)).real[:n_out]

def score(a):
    """Compute the score."""
    n = len(a)
    c = fft_conv(a, a)
    max_c = np.max(c)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)

def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix T where (T@g)[k] = (f*g)[k]."""
    # Column j has f[0:n] at rows j:j+n
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.tile(f, n)
    T = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return T

def lp_step(f, n):
    """Solve the LP: max sum(g) s.t. T_f @ g <= M, g >= 0."""
    M = np.max(fft_conv(f, f))
    T = build_conv_matrix(f, n)
    b_ub = np.full(2*n-1, M)
    res = linprog(
        c=-np.ones(n),  # minimize -sum(g) = maximize sum(g)
        A_ub=T, b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )
    if res.status == 0:
        return res.x
    return None

def line_search(f, g, n):
    """Find best t in [0,1] for f_new = (1-t)*f + t*g."""
    # Binary search / ternary search on t
    ff = fft_conv(f, f)
    fg = fft_conv(f, g)
    gg = fft_conv(g, g)
    sum_f = np.sum(f)
    
    def score_t(t):
        c = (1-t)**2 * ff + 2*t*(1-t) * fg + t**2 * gg
        max_c = np.max(c)
        s = sum_f  # sum is preserved
        return 2.0 * n * max_c / (s * s)
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if score_t(m1) < score_t(m2):
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2
    return t, score_t(t)
```

- Return (b - lambda)_+
