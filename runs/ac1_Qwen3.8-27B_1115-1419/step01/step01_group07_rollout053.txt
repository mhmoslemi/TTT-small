from scipy.optimize import minimize

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def smooth_obj_grad(x, tau):
    """Returns (objective, gradient) for LSE-smoothed max of conv(a,a)."""
    a = softmax(x)
    c = fft_convolve(a, a)
    c_max = c.max()
    c_shifted = c - c_max
    ex = np.exp(c_shifted / tau)
    w = ex / ex.sum()
    obj = c_max + tau * np.log(ex.sum())
    
    # Gradient
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    # Verify length
    if len(grad_a) != len(a):
        # Shouldn't happen, but just in case
        grad_a = np.correlate(w, a, mode='full')[:len(a)]
    
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x

def exact_subgradient(a, n_active=50, step=0.01):
    """One exact subgradient step."""
    n = len(a)
    c = fft_convolve(a, a)
    max_c = c.max()

        return a
    
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(n_active, len(c))]

        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

Optimize step function to minimize 2*n*max(a*a)/(sum(a))^2 using FFT-based

import numpy as np
import time
from scipy.optimize import minimize


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    size = 1 << (max(n - 1, 1)).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def compute_score(a):
    """Compute 2n*max(a*a)/(sum a)^2 directly."""
    c = fft_convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * c.max() / (s * s)


def smooth_obj_grad(x, tau):
    """LSE-smoothed max of conv(a,a) where a=softmax(x). Returns (obj, grad)."""
    a = softmax(x)
    c = fft_convolve(a, a)
    c_max = c.max()
    c_shifted = c - c_max
    ex = np.exp(c_shifted / tau)
    logsum = np.log(ex.sum())
    obj = c_max + tau * logsum
    w = ex / ex.sum()
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    dot = np.dot(a, grad_a)
    grad = a * (grad_a - dot)
    return obj, grad


def subgradient_step(a, n_active, step):
    """Exact subgradient step on the true max of a*a."""
    n = len(a)

            a = best_a + rng.normal(0, 0.003, n)
            a = np.maximum(a, 0)
            a /= a.sum()
            step = base_step * 0.5
            patience = 0

    return best_a, best_score


def diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Ramp (asymmetric!)
    x = np.linspace(1.0, 0.1, n)
    inits.append(x / x.sum())

    # Reverse ramp
    x = np.linspace(0.1, 1.0, n)
    inits.append(x / x.sum())

    # Sine bump
    x = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    inits.append(x / x.sum())

    # Cosine window (Hann)
    x = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    x = np.maximum(x, 0.01)
    inits.append(x / x.sum())

    # Exponential decay
    x = np.exp(-2.0 * np.arange(n) / n)
    inits.append(x / x.sum())

    # Two-sided exponential
    mid = n / 2
    x = np.exp(-1.5 * np.abs(np.arange(n) - mid) / (n * 0.3))
    inits.append(x / x.sum())

    # Parabolic
    t = np.arange(n) / (n - 1)
    x = t * (1 - t) + 0.05
    inits.append(x / x.sum())

    # Random
    x = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(x)

    # Two-block asymmetric
    x = np.ones(n)
    x[:n // 3] = 1.8
    x[n // 3:n * 2 // 3] = 1.0
    x[n * 2 // 3:] = 0.4
    inits.append(x / x.sum())

    # Sine-modulated
    x = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    x = np.maximum(x, 0.05)
    inits.append(x / x.sum())

    return inits


def propose_candidate(budget_s=2400, **kwargs):
    """Main search: minimizes 2n*max(a*a)/(sum a)^2 over non-negative sequences.
    
    Strategy: FFT-accelerated LSE-smooth L-BFGS + exact subgradient polish,
    with diverse restarts and multi-length exploration.
    """
    deadline = time.monotonic() + budget_s - 30  # 30s safety margin
    rng = np.random.default_rng(42)

    # Load parent if available
    parent_a = None
    if 'height_sequence_1' in globals():
        try:
            parent_a = np.array(height_sequence_1, dtype=np.float64)
            parent_a = np.maximum(parent_a, 0)
            if parent_a.sum() > 0.01:
                parent_a /= parent_a.sum()
                print(f"Parent loaded: n={len(parent_a)}, score={compute_score(parent_a):.6f}")
        except Exception as e:
            print(f"Could not load parent: {e}")

    best_score = float('inf')
    best_a = None

    # Try multiple lengths
    lengths = [2048, 4096, 4806, 8192]
    if parent_a is not None:
        lengths = [len(parent_a)] + [l for l in lengths if l != len(parent_a)]

    # Phase 1: Quick exploration (20% time)
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.2
    quick_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n=== Length n={n} ===")
        inits = diverse_inits(n, rng)

        if parent_a is not None and len(parent_a) == n:
            inits.insert(0, parent_a.copy())

        for idx, init in enumerate(inits[:5]):  # Limit to 5 inits per length
            if time.monotonic() > phase1_deadline:
                break
            x = np.log(np.maximum(init, 1e-15))
            tau = 0.02 / n
            a_res, score, x_res = run_lbfgs_taus(x, [tau, tau * 0.5], phase1_deadline)
            quick_results.append((score, a_res, n))
            if score < best_score:
                best_score = score
                best_a = a_res.copy()
                print(f"  New best: {best_score:.8f} (n={n}, init={idx})")

    quick_results.sort(key=lambda t: t[0])
    print(f"\nPhase 1 done. Best: {best_score:.8f}")

    # Phase 2: Extended L-BFGS on top candidates (40% time)
    phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.4
    top_k = min(4, len(quick_results))

    print(f"\n=== Phase 2: Extended optimization (top {top_k}) ===")
    for idx in range(top_k):
        if time.monotonic() > phase2_deadline:
            break
        score_cand, a_cand, n_cand = quick_results[idx]
        print(f"\n  Candidate {idx}: score={score_cand:.8f}, n={n_cand}")

        x = np.log(np.maximum(a_cand, 1e-15))
        n = n_cand
        tau_schedule = [0.015 / n, 0.008 / n, 0.004 / n, 0.002 / n, 0.001 / n]
        a_res, score, x_res = run_lbfgs_taus(x, tau_schedule, phase2_deadline)

        if score < best_score:
            best_score = score
            best_a = a_res.copy()
            print(f"  New best: {best_score:.8f}")

        # Quick subgradient polish
        a_res = a_res.copy()
        for _ in range(200):
            if time.monotonic() > phase2_deadline:
                break
            a_res = subgradient_step(a_res, n_active=50, step=0.015)
        score = compute_score(a_res)
        if score < best_score:
            best_score = score
            best_a = a_res.copy()
            print(f"  After quick polish: {best_score:.8f}")

    # Phase 3: Intensive polishing of the best (40% time)
    phase3_deadline = deadline
    print(f"\n=== Phase 3: Intensive polishing ===")
    print(f"  Starting score: {best_score:.8f}")

    a = best_a.copy()
    n = len(a)

    # Alternating: subgradient + occasional L-BFGS re-optimization
    step = 0.015
    round_num = 0
    while time.monotonic() < phase3_deadline:
        round_num += 1

        # Subgradient phase
        sg_deadline = min(time.monotonic() + 30, phase3_deadline - 60)
        for _ in range(500):
            if time.monotonic() > sg_deadline:
                break
            a = subgradient_step(a, n_active=50, step=step)

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if round_num % 5 == 0:
                print(f"  Round {round_num}: {best_score:.8f} (step={step:.6f})")

        # If we have time, do a short L-BFGS re-optimization
        if time.monotonic() < phase3_deadline - 120:
            x = np.log(np.maximum(a, 1e-15))
            tau = 0.003 / n
            def obj3(x_):
                return smooth_obj_grad(x_, tau)[0]
            def grad3(x_):
                return smooth_obj_grad(x_, tau)[1]
            try:
                result = minimize(
                    obj3, x, jac=grad3,
                    method='L-BFGS-B',
                    options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-16,
                             'maxfun': 300, 'maxls': 40}
                )
                a_new = softmax(result.x)
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    print(f"  L-BFGS boost: {best_score:.8f}")
            except Exception:
                pass

        step *= 0.97
        if step < 0.001:
            step = 0.005

        # Occasional random perturbation
        if round_num % 10 == 0:
            a_pert = best_a + rng.normal(0, 0.002, n)
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
            sc_p = compute_score(a_pert)
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_pert.copy()
                a = a_pert.copy()
                print(f"  Perturbation helped: {best_score:.8f}")

    # Final validation
    final_score = evaluate_sequence([float(v) for v in best_a])
    if not np.isfinite(final_score) or final_score > best_score * 1.01:
        # Fallback: use a safe sequence
        final_score = best_score
        final_a = best_a
    else:
        final_a = best_a
        if final_score < best_score:
            best_score = final_score

    print(f"\n=== FINAL: score={best_score:.8f}, n={len(best_a)} ===")
    return [float(v) for v in best_a]
