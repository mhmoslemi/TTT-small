Minimizes 2*n*max(convolve(a,a))/sum(a)^2 for non-negative a.
Uses peak-shaving gradient descent in softmax/log-space with FFT-accelerated
convolutions, adaptive learning rates (Adam), and cosine-annealed soft-max
temperature. Multi-scale refinement from 512 to 4096.

import numpy as np

def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p

def _score_fft(a):
    """Compute score using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 1e-300:
        return np.inf
    m = 2*n - 1
    L = _next_pow2(m)
    F = np.fft.fft(a, L)
    b = np.fft.ifft(F*F).real[:m]
    b = np.maximum(b, 0)
    bmax = b.max()
    if bmax < 1e-300:
        return np.inf
    return 2.0 * n * bmax / (S*S)

def _interpolate(a, n_new):
    """Interpolate sequence a to length n_new."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(n_new) / n_new
    if a.size == n_new:
        return a / s
    x_old = np.linspace(0, 1, a.size)
    x_new = np.linspace(0, 1, n_new)
    out = np.interp(x_new, x_old, a / s)
    out = np.maximum(out, 0)
    ss = out.sum()
    if ss < 1e-300:
        out.fill(1.0/n_new)
    else:
        out /= ss
    return out

def _softmax(v):
    v = v - v.max()
    v = np.clip(v, -700, 700)
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(v) / v.size
    return e / s

def _rand_inits(n, rng, n_init=10):

    for sigma in (0.15, 0.25, 0.35, 0.5):

        c = np.maximum(c, 0.01)
        c /= c.sum()
        inits.append(c)

    rng = np.random.default_rng(seed)
    n = int(max(4, n))

    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    v = np.log(a)

    L = _next_pow2(3*n)  # for FFT of convolutions
    m = 2*n - 1
    
    # Adam
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    mu = np.zeros(n)
    nu = np.zeros(n)
    step_count = 0
    
    best_a = a.copy()
    best_score = _score_fft(a)
    
    deadline = time.monotonic() + seconds
    last_improve = time.monotonic()
    no_improve_count = 0
    max_no_improve = 200
    
    while time.monotonic() < deadline:
        elapsed_ratio = min(1.0, (time.monotonic() - (deadline - seconds)) / max(1, seconds))
        
        # Compute b = a*a
        F = np.fft.fft(a, L)
        b = np.fft.ifft(F*F).real[:m]
        b = np.maximum(b, 0)
        bmax = b.max()
        if bmax < 1e-300:
            break
        
        # Soft-max temperature (annealing)
        tau_frac = 0.4 * (1 - elapsed_ratio) + 0.02  # from 0.4 to 0.02
        tau = bmax * tau_frac
        tau = max(tau, bmax * 0.01)
        
        # Soft weights
        x = (b - bmax) / tau
        x = x - x.max()
        w = np.exp(x)
        ws = w.sum()
        if ws < 1e-300 or not np.isfinite(ws):

        else:
            w /= ws

        a_rev = a[::-1]
        grad = np.fft.ifft(np.fft.fft(a_rev, L) * np.fft.fft(w, L)).real

            # Gradient of smoothness penalty in v-space
            grad_smooth = a * (lap - np.dot(lap, a))
            grad_v = grad_v + lam_smooth * grad_smooth

        mu = beta1*mu + (1-beta1)*grad_v
        nu = beta2*nu + (1-beta2)*grad_v**2

        nu_hat = nu / (1 - beta2**step_count)

                best_score = sc
                best_a = a.copy()
                last_improve = time.monotonic()
                no_improve_count = 0
            else:
                no_improve_count += 1

        if no_improve_count >= max_no_improve or (time.monotonic() - last_improve > 60):

            v += rng.normal(0, noise_scale, n)
            v = v - v.mean()
            v = np.clip(v, -30, 30)
            no_improve_count = 0
