def _resample_to_length(a, new_n):

"""
Search for nonnegative step-function heights minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

Method:
- Normalize any candidate to sum(a) = sqrt(2n); then the objective equals max conv(a,a).
- Primary optimization: smooth max-gradient multiplicative (mirror) descent on the simplex,
  with momentum and adaptive log-steps.
- Soft-peak weights are computed by a tail-temperature softmax over the convolution.
- Occasional linearized trust-region LP directions over the dominant convolution peaks.
- Diverse initializers: parent, reversals, shifts, modulations, upsampling/downsampling,
  and short small-length explorations resampled to the target length.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _score_key(item):
    return item[1]


def _fft_len_conv(n):
    m = 2 * int(n) - 1
    return 1 << max(0, (m - 1).bit_length())


def _fft_len_grad(n):
    m = 3 * int(n) - 2
    return 1 << max(0, (m - 1).bit_length())


def _max_conv_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    N = _fft_len_conv(n)
    fa = rfft(a, N)
    b = irfft(fa * fa, N).real[:2 * n - 1]
    m = float(np.max(b))
    if not np.isfinite(m):
        return np.inf
    return m


def _softmax_weights(b, m, tau):
    b = np.asarray(b, dtype=np.float64)
    m = float(m)
    if tau <= 1e-8:
        tol = max(1e-10, 1e-7 * abs(m))
        mask = (m - b) <= tol
        if not np.any(mask):
            mask = np.zeros_like(b, dtype=bool)
            mask[int(np.argmax(b))] = True
        w = mask.astype(np.float64)
        s = w.sum()
        if s <= 0.0:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= s
        return w

    z = (b - m) / tau
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    return w


def _compute_state(a, tau):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    N = _fft_len_grad(n)
    fa = rfft(a, N)
    L = 2 * n - 1
    b = irfft(fa * fa, N).real[:L]
    m = float(np.max(b))

    w = _softmax_weights(b, m, tau)
    fw = rfft(w, N)
    c = irfft(np.conj(fa) * fw, N).real[:n]
    c = np.where(c > 0.0, c, 0.0)
    return b, m, w, c


def _scale_to_target(a, target):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.shape, target / a.size, dtype=np.float64)
    a *= target / s
    return a


def _normalize_safe(a, target):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return _scale_to_target(a, target)


def _clean_normalize(arr, target):
    a = np.asarray(arr, dtype=np.float64)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    max_allowed = max(1.0, target * 1.0e6)
    a = np.minimum(a, max_allowed)
    return _scale_to_target(a, target)


def _resample_to_length(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    new_n = int(new_n)
    if m == 0:
        return np.ones(new_n, dtype=np.float64)
    if new_n == m:
        return a.copy()
    if m == 1:
        return np.full(new_n, float(a[0]), dtype=np.float64)
    x_old = np.arange(m, dtype=np.float64)
    x_new = np.linspace(0.0, float(m - 1), new_n)
    return np.interp(x_new, x_old, a)


def _shift_mix(a, shift, alpha):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n <= 1:
        return a
    shift = int(shift) % n
    r = np.roll(a, shift)
    alpha = float(alpha)
    return (1.0 - alpha) * a + alpha * r


def _modulate(a, freq, amp, phase):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n <= 1:
        return a
    x = np.arange(n, dtype=np.float64) / float(n - 1)
    return a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))


def _shape_mix(a, mode, strength):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n <= 1:
        return a
    x = np.linspace(0.0, 1.0, n)
    strength = float(strength)
    if mode == "end":
        mult = 1.0 + strength * (np.exp(-20.0 * x * x) + np.exp(-20.0 * (x - 1.0) * (x - 1.0)))
    elif mode == "center":
        t = (x - 0.5) / 0.15
        mult = 1.0 + strength * np.exp(-t * t)
    else:
        mult = 1.0
    return a * mult


def _perturb(a, ref, rng, strength, target):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n == 0:
        return a
    if ref is not None and np.asarray(ref).size == n:
        ref = np.asarray(ref, dtype=np.float64)
        ref = np.where(np.isfinite(ref), ref, 0.0)
        ref = np.maximum(ref, 0.0)
        a = 0.75 * a + 0.25 * ref
    mean = float(a.mean())
    if not np.isfinite(mean) or mean <= 1e-300:
        mean = target / max(1, n)
        a = np.full(n, mean, dtype=np.float64)
    a += strength * mean * rng.standard_normal(n)
    return _normalize_safe(a, target)


def _eval_candidate_into_list(cands, arr, tag):
    if arr is None:
        return
    a0 = np.asarray(arr)
    n = int(a0.size) if hasattr(a0, "size") else len(a0)
    if n <= 0:
        return
    a = _clean_normalize(a0, np.sqrt(2.0 * float(n)))
    sc = _max_conv_fast(a)
    if np.isfinite(sc):
        cands.append((a, float(sc), tag))


def _unique_candidates(cands, max_per_n=2, max_total=12):
    if not cands:
        return []
    cands = list(cands)
    cands.sort(key=_score_key)
    counts = {}
    out = []
    for a, sc, tag in cands:
        n = int(a.size)
        c = counts.get(n, 0)
        if c >= max_per_n:
            continue
        out.append((a, sc, tag))
        counts[n] = c + 1
        if len(out) >= max_total:
            break
    return out


def _lp_direction_from_b(a, b, k):
    if linprog is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    L = b.size
    if n < 5 or L < 3:
        return None

    k = int(max(1, k))
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return None

    tol = max(1e-6, 0.02 * m)
    near = np.nonzero(b >= m - tol)[0]
    near_count = int(near.size)
    cap = 400
    rows = min(cap, max(k, min(near_count, max(1, 3 * k))))
    if rows <= 0:
        return None

    if rows >= L:
        idx = np.arange(L)
    elif near_count <= rows:
        idx = near
    else:
        idx = np.argpartition(b, -rows)[-rows:]
    idx = idx.astype(np.int64)

    mrows = int(idx.size)
    A = np.zeros((mrows, n + 1), dtype=np.float64)
    bu = np.empty(mrows, dtype=np.float64)

    max_a = float(np.max(a)) if n else 1.0
    mean_a = float(np.mean(a)) if n else 1.0
    U = max(1e-12, 0.25 * max_a, 0.10 * mean_a)

    for r, t in enumerate(idx):
        t = int(t)
        lo = max(0, t - n + 1)
        hi = min(n - 1, t)
        if hi >= lo:
            A[r, lo:hi + 1] = 2.0 * a[t - np.arange(lo, hi + 1)]
        A[r, n] = -1.0
        bu[r] = -b[t]

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0], dtype=np.float64)

    lower = -0.75 * a
    upper = np.full(n, U, dtype=np.float64)
    bounds = np.empty((n + 1, 2), dtype=np.float64)
    bounds[:n, 0] = lower
    bounds[:n, 1] = upper
    bounds[n, 0] = 0.0
    bounds[n, 1] = m

    try:
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=bu,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=bu,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
    except Exception:
        return None

    if res is None:
        return None
    if not getattr(res, "success", False):
        return None
    if getattr(res, "x", None) is None:
        return None

    delta = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(delta)):
        return None
    delta = delta - float(np.mean(delta))
    maxabs = float(np.max(np.abs(delta))) if n else 0.0
    if maxabs < 1e-12:
        return None

    y_est = float(getattr(res, "fun", np.inf))
    if not np.isfinite(y_est) or y_est > m - 1e-8:
        return None

    return delta


def _apply_lp_direction(a, delta, target, current_m):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if n == 0:
        return a, False, current_m
    best_m = float(current_m)
    best_a = None
    for alpha in (1.0, 0.6, 0.35, 0.20, 0.10, 0.05):
        a_try = a + alpha * delta
        a_try = _normalize_safe(a_try, target)
        m_try = _max_conv_fast(a_try)
        if np.isfinite(m_try) and m_try + 1e-12 < best_m:
            best_m = float(m_try)
            best_a = a_try.copy()
    if best_a is None:
        return a.copy(), False, current_m
    return best_a, True, best_m


def _run_mirror_lp(
    a,
    deadline,
    gamma0=0.20,
    tau0=0.50,
    beta=0.90,
    lp_after=80,
    lp_k=60,
    max_lp=10,
    seed=0,
    verbose=False,
):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n == 0:
        return a, np.inf

    target = np.sqrt(2.0 * float(n))
    a = _normalize_safe(a, target)
    a = a + (1e-12 * target / max(1, n))
    a = _scale_to_target(a, target)

    rng = np.random.default_rng(int(seed))
    b, m, w, c = _compute_state(a, tau0)
    best_a = a.copy()
    best_score = float(m)

    tau = float(tau0)
    gamma = float(gamma0)
    gamma0_ref = float(gamma0)
    momentum = np.zeros(n, dtype=np.float64)

    it = 0
    no_improve = 0
    rejected = 0
    lp_calls = 0
    use_lp = max_lp > 0 and linprog is not None
    floor = 1e-12 * target / max(1, n)

    while time.monotonic() < deadline - 3:
        if np.isfinite(m) and m + 1e-12 < best_score:
            best_score = float(m)
            best_a = a.copy()
            no_improve = 0
            gamma = min(0.8, gamma * 1.03)
        else:
            no_improve += 1

        if not np.isfinite(m):
            a = best_a.copy()
            b, m, w, c = _compute_state(a, tau)
            momentum.fill(0.0)
            no_improve = 0
            continue

        if best_score > 0.0 and m > best_score * 1.08 + 1e-8:
            a = best_a.copy()
            momentum.fill(0.0)
            tau = min(5.0, tau * 1.2)
            gamma = max(0.05, gamma * 0.7)
            no_improve = 0
            b, m, w, c = _compute_state(a, tau)
            continue

        if no_improve > max(20, int(lp_after)):
            no_improve = 0
            improved_lp = False

            if use_lp and lp_calls < max_lp and time.monotonic() < deadline - 60:
                delta = _lp_direction_from_b(a, b, lp_k)
                if delta is not None:
                    a_try, acc, m_try = _apply_lp_direction(a, delta, target, m)
                    lp_calls += 1
                    if acc and m_try + 1e-10 < m:
                        a = a_try
                        if m_try < best_score - 1e-10:
                            best_score = float(m_try)
                            best_a = a.copy()
                        momentum.fill(0.0)
                        gamma = max(0.05, min(gamma0_ref, gamma * 1.1))
                        tau = max(0.02, tau * 0.8)
                        improved_lp = True

            if not improved_lp:
                a = _perturb(a, best_a, rng, 0.03, target)
                momentum.fill(0.0)
                tau = min(5.0, tau * 1.2)
                gamma = max(0.05, min(gamma0_ref * 0.8, gamma * 0.5))
                b, m, w, c = _compute_state(a, tau)
            continue

        if no_improve > 40:
            if tau > 0.12:
                tau *= 0.8
            else:
                tau = min(3.0, tau * 1.4)

        gc = c - float(np.mean(c))
        momentum = beta * momentum + (1.0 - beta) * gc
        momentum -= float(np.mean(momentum))
        maxabs = float(np.max(np.abs(momentum))) if n else 0.0

        if maxabs < 1e-14:
            a = a * (1.0 + 1e-6 * rng.standard_normal(n))
            a += floor
            a = _scale_to_target(a, target)
            momentum.fill(0.0)
            no_improve = max(no_improve, 1)
        else:
            eff = min(1000.0, gamma / maxabs)
            a_try = a * np.exp(-eff * momentum)
            a_try += floor
            a_try = _scale_to_target(a_try, target)

            m_try = _max_conv_fast(a_try)
            tol = 1e-9 * max(1.0, best_score)

            if np.isfinite(m_try) and m_try <= m + tol:
                a = a_try
                if m_try < m - tol:
                    no_improve = 0
                    gamma = min(0.8, gamma * 1.03)
                else:
                    gamma = max(0.005, gamma * 0.999)
            else:
                rejected += 1
                gamma = max(0.005, gamma * 0.7)
                momentum *= 0.5
                if rejected > 8:
                    a = _perturb(a, best_a, rng, 0.02, target)
                    momentum.fill(0.0)
                    rejected = 0
                    b, m, w, c = _compute_state(a, tau)

        it += 1
        if verbose and it % 200 == 0:
            print(
                f"    mirror it={it} best={best_score:.6f} cur={m:.6f} "
                f"gamma={gamma:.4f} tau={tau:.4f}"
            )

    m_final = _max_conv_fast(a)
    if np.isfinite(m_final) and m_final < best_score:
        best_score = float(m_final)
        best_a = a.copy()
    return best_a, best_score


def _small_exploration(parent, parent_n, rng, deadline, cands):
    sizes = (64, 256, 1024)
    remaining = deadline - time.monotonic()
    total_time = min(45.0, max(0.0, remaining * 0.03))
    per = total_time / max(1, len(sizes) * 2)
    if per < 1.0:
        return

    for sn in sizes:
        if time.monotonic() >= deadline - 30:
            break

        target_small = np.sqrt(2.0 * float(sn))

        a1 = (0.5 + rng.random(sn)) * np.ones(sn, dtype=np.float64)
        a1 = _clean_normalize(a1, target_small)
        a_local, _ = _run_mirror_lp(
            a1,
            time.monotonic() + per,
            gamma0=0.25,
            tau0=0.5,
            beta=0.90,
            lp_after=10**9,
            lp_k=20,
            max_lp=0,
            seed=int(rng.integers(0, 2**31 - 1)),
        )
        _eval_candidate_into_list(cands, _resample_to_length(a_local, parent_n), f"small_rand_{sn}")

        if parent is not None and parent.size > 0 and per > 0.6:
            a2 = _resample_to_length(parent, sn)
            a2 = _clean_normalize(a2, target_small)
            a_local2, _ = _run_mirror_lp(
                a2,
                time.monotonic() + 0.5 * per,
                gamma0=0.15,
                tau0=0.3,
                beta=0.90,
                lp_after=10**9,
                lp_k=20,
                max_lp=0,
                seed=int(rng.integers(0, 2**31 - 1)),
            )
            _eval_candidate_into_list(cands, _resample_to_length(a_local2, parent_n), f"small_parent_{sn}")


def _safe_evaluate(seq):
    try:
        vals = []
        for x in seq:
            y = float(x)
            if not np.isfinite(y):
                return np.inf
            vals.append(y)
        val = evaluate_sequence(vals)
        if val is None:
            return np.inf
        val = float(val)
        if not np.isfinite(val):
            return np.inf
        return val
    except Exception:
        return np.inf


def _finalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    target = np.sqrt(2.0 * float(n))
    a = _clean_normalize(a, target)
    if a.size and float(np.max(a)) > 900.0:
        a *= 900.0 / float(np.max(a))
    out = []
    for x in a:
        y = float(x)
        if not np.isfinite(y) or y < 0.0:
            y = 0.0
        if y > 999.0:
            y = 999.0
        out.append(y)
    s = sum(out)
    if s <= 0.01:
        out = [1.0] * max(1, len(out))
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if budget <= 0.0:
        return [1.0]
    deadline = t_start + max(1.0, budget - 60.0)

    rng = np.random.default_rng(int(seed))

    parent = None
    try:
        raw = height_sequence_1
        arr = np.asarray(raw, dtype=np.float64).ravel()
        if arr.size > 0:
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            if float(arr.sum()) > 1e-300:
                parent = arr
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(5582, dtype=np.float64)
    parent_n = int(parent.size)

    cands = []

    _eval_candidate_into_list(cands, parent, "parent")
    _eval_candidate_into_list(cands, 0.5 * (parent + parent[::-1]), "sym")

    if parent_n > 1:
        shifts = [
            1,
            2,
            5,
            max(1, parent_n // 7),
            max(1, parent_n // 3),
            int(0.618 * parent_n),
            int(0.75 * parent_n),
        ]
        for sh in shifts:
            if 0 < sh < parent_n:
                _eval_candidate_into_list(cands, _shift_mix(parent, sh, 0.25), f"shift{sh}_a")
                _eval_candidate_into_list(cands, _shift_mix(parent, sh, 0.45), f"shift{sh}_b")

        for freq in (1, 2, 3, 5, 8):
            for amp, ph in ((0.15, 0.0), (0.25, 1.0), (0.35, 2.0)):
                _eval_candidate_into_list(
                    cands,
                    _modulate(parent, freq, amp, ph),
                    f"cos_f{freq}_a{amp}_p{ph}",
                )

        for mode, strength in (("end", 0.5), ("end", 1.0), ("center", 0.5), ("center", 1.0)):
            _eval_candidate_into_list(cands, _shape_mix(parent, mode, strength), f"shape_{mode}_{strength}")

        for st in (0.05, 0.12, 0.2):
            mean = float(np.mean(parent))
            if mean <= 0.0:
                mean = 1.0
            a = parent + st * mean * rng.standard_normal(parent_n)
            _eval_candidate_into_list(cands, a, f"rand_{st}")

        for factor in (0.70, 0.85, 1.15, 1.35, 1.60):
            nn = int(parent_n * factor)
            if 128 <= nn <= 12000:
                ar = _resample_to_length(parent, nn)
                _eval_candidate_into_list(cands, ar, f"len_{nn}")
                _eval_candidate_into_list(cands, 0.5 * (ar + ar[::-1]), f"symlen_{nn}")

    if time.monotonic() < deadline - 120:
        _small_exploration(parent, parent_n, rng, deadline, cands)

    pool = _unique_candidates(cands, max_per_n=2, max_total=14)
    if not pool:
        a = _clean_normalize(parent, np.sqrt(2.0 * float(parent_n)))
        sc = _max_conv_fast(a)
        pool = [(a.copy(), float(sc), "fallback")]

    global_best_a = pool[0][0].copy()
    global_best_score = float(pool[0][1])

    print(f"Initial best fast score: {global_best_score:.6f} (n={global_best_a.size})")

    configs = [
        (0.18, 1.00, 0.90, 100, 70),
        (0.28, 0.50, 0.90, 80, 90),
        (0.12, 0.20, 0.95, 70, 120),
        (0.38, 0.80, 0.85, 120, 90),
        (0.22, 0.30, 0.92, 60, 140),
        (0.55, 1.50, 0.80, 150, 120),
    ]

    runs = 0
    print("Starting main optimization loop.")

    while time.monotonic() < deadline - 90:
        run_start = time.monotonic()

        if runs % 3 == 0:
            target = np.sqrt(2.0 * float(global_best_a.size))
            strength = 0.012 + 0.012 * (runs % 2)
            start_a = _perturb(global_best_a.copy(), None, rng, strength, target)
            cfg = configs[runs % len(configs)]
            run_budget = min(900.0, max(30.0, (deadline - time.monotonic()) * 0.45))
            start_tag = "best"
        else:
            pc = pool[runs % len(pool)]
            start_a = pc[0].copy()
            target = np.sqrt(2.0 * float(start_a.size))
            start_a = _perturb(start_a, None, rng, 0.05, target)
            cfg = configs[(runs + 3) % len(configs)]
            run_budget = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.25))
            start_tag = pc[2]

        run_deadline = time.monotonic() + min(run_budget, deadline - 90)
        if run_deadline <= time.monotonic() + 5:
            run_deadline = deadline - 30

        local_a, local_score = _run_mirror_lp(
            start_a,
            run_deadline,
            gamma0=cfg[0],
            tau0=cfg[1],
            beta=cfg[2],
            lp_after=cfg[3],
            lp_k=cfg[4],
            max_lp=10,
            seed=int(rng.integers(0, 2**31 - 1)),
            verbose=False,
        )

        if np.isfinite(local_score) and local_score + 1e-10 < global_best_score:
            global_best_a = local_a.copy()
            global_best_score = float(local_score)
            if len(pool) < 24:
                pool.append((local_a.copy(), local_score, "improved"))
            print(
                f"  [{time.monotonic() - t_start:6.1f}s] run {runs:02d} start={start_tag}: "
                f"new best {global_best_score:.6f} (n={global_best_a.size})"
            )
        else:
            print(
                f"  [{time.monotonic() - t_start:6.1f}s] run {runs:02d} start={start_tag}: "
                f"best {global_best_score:.6f}, local {float('inf') if not np.isfinite(local_score) else local_score:.6f}"
            )

        runs += 1

    if time.monotonic() < deadline - 25:
        finish_cfgs = [
            (0.05, 0.10, 0.95, 40, 160),
            (0.10, 0.20, 0.92, 50, 140),
            (0.16, 0.35, 0.90, 60, 120),
        ]
        for pc in finish_cfgs:
            if time.monotonic() >= deadline - 15:
                break
            local_a, local_score = _run_mirror_lp(
                global_best_a,
                deadline - 12,
                gamma0=pc[0],
                tau0=pc[1],
                beta=pc[2],
                lp_after=pc[3],
                lp_k=pc[4],
                max_lp=8,
                seed=int(rng.integers(0, 2**31 - 1)),
                verbose=False,
            )
            if np.isfinite(local_score) and local_score + 1e-10 < global_best_score:
                global_best_a = local_a.copy()
                global_best_score = float(local_score)
                print(f"Final polish improved to {global_best_score:.6f}")

    final_arrays = [global_best_a.copy()]
    seen_n = set()
    for a, sc, tag in pool[1:]:
        n = int(a.size)
        if n not in seen_n:
            final_arrays.append(a)
            seen_n.add(n)
        if len(final_arrays) >= (3 if global_best_a.size > 8000 else 6):
            break

    best_seq = None
    best_actual = np.inf

    for arr in final_arrays:
        seq = _finalize_sequence(arr)
        val = _safe_evaluate(seq)
        if np.isfinite(val) and val < best_actual:
            best_actual = val
            best_seq = seq
            print(f"Candidate n={len(seq)} actual={val:.8f}")

    if best_seq is None:
        best_seq = _finalize_sequence(global_best_a)
        best_actual = _safe_evaluate(best_seq)

    print(f"Final returned score estimate: {best_actual}")
    print(f"Final length: {len(best_seq)}")
    return best_seq
