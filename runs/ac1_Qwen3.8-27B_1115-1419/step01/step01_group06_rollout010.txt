def _fft_len(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        v = np.ones(max(v.size, 1), dtype=np.float64)
        s = v.sum()
    return v / s

def _softmax_stable(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    w = np.where(np.isfinite(w), w, 0.0)
    w -= w.max()
    e = np.exp(w)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(w.size, dtype=np.float64) / max(w.size, 1)
    return e / s

def _log_weights(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w -= w.max()
    return w

def _conv_score(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:2*n-1]
    c = np.maximum(c, 0)
    mx = float(c.max())
    score = 2.0 * n * mx
    return A, c, score

def _smooth_max_weights(c, beta):
    c_max = c.max()
    exponents = beta * (c - c_max)
    exponents = np.clip(exponents, -700, 0)
    e = np.exp(exponents)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        p = np.zeros(c.size)
        p[c.argmax()] = 1.0
    else:
        p = e / s
    return p

def _self_gradient(a, c, beta, N, phase):
    n = len(a)
    p = _smooth_max_weights(c, beta)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    FR = phase * np.conjugate(A)
    conv = np.fft.irfft(P * FR, N)
    grad = 2.0 * conv[n-1:2*n-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad

def _cross_gradient(f, b, beta, N, phase, F, FR):
    n = len(b)
    B = np.fft.rfft(b, N)
    h = np.fft.irfft(F * B, N)

    h = np.maximum(h, 0)

    P = np.fft.rfft(p, N)
    grad = np.fft.irfft(P * FR, N)
    grad = grad[n-1:2*n-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad

def _adam_step(grad_w, w, m, v, t, lr):
    t += 1
    b1, b2 = 0.9, 0.999
    grad_w = np.asarray(grad_w, dtype=np.float64)
    m = m * b1 + (1 - b1) * grad_w
    v = v * b2 + (1 - b2) * grad_w * grad_w
    mhat = m / (1 - b1**t)
    vhat = v / (1 - b2**t)
    dw = mhat / (np.sqrt(vhat) + 1e-12)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -20.0, 20.0)

    w_new -= w_new.max()
    a_new = _softmax_stable(w_new)

def _exact_score(a):

    mx = float(b.max())

def _make_raised_cosine(n, power=1.0, offset=0.0):

def _make_gaussian(n, center_frac=0.5, width_frac=0.3):

def _make_trapezoid(n, rise_frac=0.2, fall_frac=0.2):
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1)
    a = np.ones(n)

        elif x[i] < 1 - fall_frac:
            a[i] = 1.0
        else:
            a[i] = (1 - x[i]) / fall_frac
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

def _block_refine(a, block_size, n_passes, budget, seed):
    """Refine using block coordinate descent with scipy."""
    rng = np.random.default_rng(seed)
    n = len(a)
    f = _normalize(a)
    N = _fft_len(2 * n)
    best_score = 2.0 * n * np.max(np.fft.irfft(np.fft.rfft(f, N)**2, N)[:2*n-1])
    best_f = f.copy()
    start = time.monotonic()

    for _ in range(n_passes):
        if time.monotonic() > budget:
            break
        # Pick random blocks
        n_blocks = max(1, n // block_size)
        for _ in range(min(n_blocks, 8)):
            if time.monotonic() > budget:
                break
            i0 = int(rng.integers(0, max(1, n - block_size)))
            i1 = min(i0 + block_size, n)
            block_idx = np.arange(i0, i1)
            
            block_val = f[i0:i1].copy()
            if block_val.sum() < 1e-15:
                continue

            def block_obj(x):
                x = np.maximum(x, 0)
                trial = f.copy()
                trial[i0:i1] = x
                s = trial.sum()
                if s <= 0:
                    return 1e6
                trial = trial / s
                c = np.fft.irfft(np.fft.rfft(trial, N)**2, N)[:2*n-1]
                c = np.maximum(c, 0)
                return c.max()

            res = scipy_minimize(block_obj, block_val, method='L-BFGS-B',
                                  options={'maxiter': 30, 'gtol': 1e-8})
            if res.success and np.all(np.isfinite(res.x)):
                new_block = np.maximum(res.x, 0)
                trial = f.copy()
                trial[i0:i1] = new_block
                s = trial.sum()
                if s > 0:
                    trial = trial / s
                    c = np.fft.irfft(np.fft.rfft(trial, N)**2, N)[:2*n-1]
                    c = np.maximum(c, 0)
                    sc = c.max()
                    if sc < best_score:
                        best_score = sc
                        best_f = trial.copy()

    return best_f, 2.0 * n * best_score

def _finalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    scale = math.sqrt(2.0 * n)
    if s > 0 and np.isfinite(s):
        a = a * scale / s
    else:
        a = np.ones(n) * (scale / n)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.01 / max(s2, 1e-300))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]

    # Filter to keep total work manageable
    # Allocate time per length
    n_lengths = len(lengths)
    time_per_length = budget / (n_lengths + 1)  # +1 for final refinement
    time_per_length = min(time_per_length, 600.0)

    global_best = None
    global_best_score = float('inf')

    # Build list of initializations
    initializations = []
    if sota is not None:
        initializations.append(("sota", sota.copy()))
    
    for n in lengths:
        initializations.append(("uniform", np.ones(n) / n))
        initializations.append(("raised_cos1", _make_raised_cosine(n, 1.0)))
        initializations.append(("raised_cos2", _make_raised_cosine(n, 2.0)))
        initializations.append(("gauss", _make_gaussian(n, 0.5, 0.35)))
        initializations.append(("double_peak", _make_double_peak(n, 0.3, 0.7, 0.12)))
        initializations.append(("trapezoid", _make_trapezoid(n, 0.15, 0.15)))
        # Asymmetric
        j = np.arange(n, dtype=np.float64)
        a_asym = np.exp(-0.5 * ((j - 0.35*(n-1)) / (0.25*n))**2)
        a_asym += np.exp(-0.5 * ((j - 0.8*(n-1)) / (0.15*n))**2) * 0.5
        a_asym = _normalize(a_asym)
        initializations.append(("asym_gauss", a_asym))
        # Ramped
        a_ramp = (j + 1) / (n + 1)
        a_ramp = _normalize(a_ramp)
        initializations.append(("ramp", a_ramp))
        # Cosine squared
        a_cs = np.cos(np.pi * j / (n-1+1e-10))**2
        a_cs = np.maximum(a_cs, 0)
        a_cs = _normalize(a_cs)
        initializations.append(("cos_sq", a_cs))

    # SOTA first (highest quality start), then others
    if sota is not None:
        initializations.sort(key=lambda x: -1 if x[0] == "sota" else 0)

    # Shuffle non-sota for variety
    sota_init = [x for x in initializations if x[0] == "sota"]
    other_init = [x for x in initializations if x[0] != "sota"]
    rng.shuffle(other_init)
    initializations = sota_init + other_init

    elapsed = time.monotonic() - start_time
    remaining = budget - elapsed

    for name, init_a in initializations:
        if time.monotonic() > start_time + budget - 30:
            break
        
        n = len(init_a)
        N = _fft_len(3 * n - 2)
        phase = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / float(N))

        # Time allocation: SOTA gets more time
        if name == "sota":
            t_alloc = min(remaining * 0.35, 900.0)
        elif name == "uniform":
            t_alloc = min(remaining * 0.1, 120.0)
        else:
            t_alloc = min(remaining * 0.08, 180.0)
        
        t_alloc = max(t_alloc, 15.0)
        t_end = time.monotonic() + t_alloc

        try:
            best_a, best_sc = _run_subgradient(init_a, N, phase, t_end, seed, n)
        except Exception:
            continue

        if np.isfinite(best_sc) and best_sc < global_best_score:
            global_best_score = best_sc
            global_best = best_a.copy()
            print(f"[{name}] n={n} FFT score={best_sc:.6f} (t={time.monotonic()-start_time:.1f}s)")

        remaining = budget - (time.monotonic() - start_time)
        if remaining < 60:
            break

    # If no valid result, use SOTA or uniform
    if global_best is None:
        if sota is not None:
            global_best = sota.copy()
            global_best_score = _exact_score(sota) if len(sota) <= 12000 else 2.0
        else:
            global_best = np.ones(4096) / 4096
            global_best_score = 2.0

    print(f"Best after main search: score={global_best_score:.6f}, n={len(global_best)}")

    # Final refinement phase
    if time.monotonic() - start_time < budget - 60:
        n = len(global_best)
        remaining = budget - (time.monotonic() - start_time) - 20
        remaining = max(remaining, 30.0)
        
        # Subgradient polish
        N = _fft_len(3 * n - 2)
        phase = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * (n - 1) / float(N))
        
        try:
            polished, pol_sc = _run_subgradient(global_best, N, phase,
                                                 time.monotonic() + remaining * 0.5,
                                                 seed + 1000, n)
            if np.isfinite(pol_sc) and pol_sc < global_best_score:
                global_best_score = pol_sc
                global_best = polished.copy()
                print(f"After polish: score={global_best_score:.6f}")
        except Exception:
            pass

        # Block refinement (only for manageable sizes)
        if n <= 8192 and time.monotonic() - start_time < budget - 60:
            try:
                block_a, block_sc = _block_refine(
                    global_best, block_size=32, n_passes=3,
                    budget=time.monotonic() + (budget - (time.monotonic()-start_time) - 30) * 0.4,
                    seed=seed + 2000
                )
                if np.isfinite(block_sc) and block_sc < global_best_score:
                    global_best_score = block_sc
                    global_best = block_a.copy()
                    print(f"After block refine: score={global_best_score:.6f}")
            except Exception:
                pass

        # Final exact validation
        if n <= 12000:
            try:
                exact_sc = _exact_score(global_best)
                if np.isfinite(exact_sc):
                    print(f"Exact score: {exact_sc:.6f}")
            except Exception:
                pass

    # Also try a few random perturbations of the best for final squeeze
    if time.monotonic() - start_time < budget - 10:
        n = len(global_best)
        N = _fft_len(3 * n - 2)
        for pert_seed in range(5):
            if time.monotonic() - start_time > budget - 5:
                break
            sig = float(rng.choice((0.02, 0.05, 0.1)))
            w_pert = _log_weights(global_best) + rng.normal(0, sig, size=n)
            cand = _softmax_stable(w_pert)
            c = np.fft.irfft(np.fft.rfft(cand, N)**2, N)[:2*n-1]
            c = np.maximum(c, 0)
            sc = 2.0 * n * c.max()
            if np.isfinite(sc) and sc < global_best_score:
                global_best_score = sc
                global_best = cand.copy()
                print(f"Perturbation {pert_seed}: score={global_best_score:.6f}")

    print(f"Final best FFT score: {global_best_score:.6f}, n={len(global_best)}")
    return _finalize(global_best)
