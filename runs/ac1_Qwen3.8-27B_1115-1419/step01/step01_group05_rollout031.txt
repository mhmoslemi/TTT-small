Search for low-peak nonnegative self-convolutions using top-k LSE optimization
with momentum, adaptive step size, and coordinate descent polishing.

The objective is score(a) = 2 * n * max_k (a*a)_k / (sum a)^2, a_i >= 0.
It is scale invariant, so sequences are normalized to sum 1 during search.

Algorithm:
  * Load provided sequence or generate good parametric starts.
  * Optimize a top-k mean / LSE surrogate of max(a*a) using momentum
    gradient descent on the positive simplex with backtracking line search.
  * Periodically sharpen (reduce k), perturb to escape local minima,
    and run coordinate descent polishing sweeps.
  * Try multiple sequence lengths via resampling.

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _next_pow2(size):
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def _autocorr(a, fft_size):
    n = a.size
    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)
    return b[:2 * n - 1]


def _score(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return 1e100
    n = a.size
    fft_size = _next_pow2(2 * n - 1)
    b = _autocorr(a, fft_size)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * n * mb / (s * s)


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _topk_mean_and_weights(b, k):
    """Return top-k mean value and weight vector (1/k for top-k, 0 else)."""
    m = b.size
    k = max(1, min(k, m))
    if k >= m:
        return float(np.mean(b)), np.full(m, 1.0 / m, dtype=np.float64)
    # Find top-k indices
    if k == 1:
        topk_idx = np.argmax(b)
        val = float(b[topk_idx])
        w = np.zeros(m, dtype=np.float64)
        w[topk_idx] = 1.0
        return val, w
    # Use partition for efficiency
    threshold = np.partition(b, m - k)[m - k]
    topk_mask = b >= threshold
    # Handle ties: if more than k pass, take exactly k
    cnt = int(topk_mask.sum())
    if cnt > k:
        # Need to pick exactly k from the tied group
        tied_idx = np.where(topk_mask)[0]
        tied_vals = b[tied_idx]
        # Take the largest k among tied
        order = np.argsort(-tied_vals)[:k]
        w = np.zeros(m, dtype=np.float64)
        w[tied_idx[order]] = 1.0 / k
        val = float(np.mean(b[tied_idx[order]]))
        return val, w
    elif cnt < k:
        # Shouldn't happen, but fallback
        topk_idx = np.argsort(-b)[:k]
    else:
        topk_idx = np.where(topk_mask)[0]
    w = np.zeros(m, dtype=np.float64)
    w[topk_idx] = 1.0 / k
    val = float(np.sum(b[topk_idx])) / k
    return val, w


def _weighted_grad(a, weights, g_fft, n):
    """Gradient of sum_k weights_k * (a*a)_k w.r.t. a.
    grad_i = 2 * sum_k weights_k * a_{k-i} (valid indices)."""
    ar = a[::-1]
    A = np.fft.rfft(ar, g_fft)
    W = np.fft.rfft(weights, g_fft)
    conv = np.fft.irfft(A * W, g_fft)
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _lse_smooth_and_grad(b, beta):
    """Compute LSE value and weight vector for smoothing max(b)."""
    m = b.size
    if m == 0:
        return 0.0, np.zeros(0)
    mx = float(np.max(b))
    if beta <= 1.0:
        p = np.full(m, 1.0 / m, dtype=np.float64)
        return mx, p
    arg = beta * (b - mx)
    arg = np.maximum(arg, -745.0)
    e = np.exp(arg)
    s = float(e.sum())
    if s <= 0.0:
        p = np.full(m, 1.0 / m, dtype=np.float64)
        return mx, p
    p = e / s
    val = mx + float(np.log(s)) / beta
    return val, p


def _project_simplex(v):
    """Clip to nonnegative and normalize to sum 1."""
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(v, 0.0)
    s = float(w.sum())
    n = w.size
    if n == 0 or s < 1e-300:
        return np.ones(max(n, 1), dtype=np.float64) / max(n, 1)
    return w / s


def _make_exponential(n, alpha):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x)
    return _normalize(a)


def _make_mixture(n, alpha, gamma):
    n = max(int(n), 2)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-float(alpha) * x) + float(gamma) * np.exp(-float(alpha) * (1.0 - x))
    a = a + 0.1
    return _normalize(a)


def _resample_sequence(old, new_n):
    old = np.asarray(old, dtype=np.float64).ravel()
    old = np.nan_to_num(old, nan=0.0, posinf=0.0, neginf=0.0)
    old = np.maximum(old, 0.0)
    new_n = max(int(new_n), 2)
    if old.size == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    s = float(old.sum())
    if s < 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    old = old / s
    if old.size == new_n:
        return old.copy()
    if old.size == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _perturb(a, rng, strength=0.3):
    """Random multiplicative perturbation."""
    n = a.size
    noise = np.exp(rng.normal(0.0, strength, size=n))
    a2 = a * noise
    a2 = np.maximum(a2, 1e-12)
    return _normalize(a2)


def _perturb_block(a, rng, strength=0.25):
    """Perturb a random block of the sequence."""
    n = a.size
    c = a.copy()
    width = max(2, n // 16)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.2:

    return _normalize(np.maximum(c, 1e-12))


def _coord_descent_sweep(a, max_sweeps=3):
    """Coordinate descent: for each coordinate, try to reduce the peak."""
    n = a.size
    fft_size = _next_pow2(2 * n - 1)
    best_score = _score(a)
    
    for sweep in range(max_sweeps):
        b = _autocorr(a, fft_size)
        max_b = float(np.max(b))
        for i in range(n):
            # Current value of a[i]
            ai = a[i]
            #Try a few candidate values
            candidates = [0.0, 0.5 * ai, 1.5 * ai, 2.0 * ai, 0.1 * ai, 5.0 * ai]
            # Also try values based on the current peak structure
            best_t = ai
            best_local = max_b
            for t in candidates:
                if t < 0:
                    continue
                a[i] = t
                b_try = _autocorr(a, fft_size)
                mb = float(np.max(b_try))
                if mb < best_local:
                    best_local = mb
                    best_t = t
            a[i] = best_t
        # Renormalize
        a = _normalize(a)
        new_score = _score(a)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            break
    return a, best_score

            if n <= 512:
                a_cd, score_cd = _coord_descent_sweep(a, max_sweeps=2)
                if score_cd < best_score:
                    best_a = a_cd.copy()
                    best_score = score_cd
                    best_it = it
                    a = a_cd.copy()
                    b = _autocorr(a, b_fft)
                    maxb = float(np.max(b))
                velocity = np.zeros(n, dtype=np.float64)
                fail = 0

        # Periodic perturbation to escape local minima
        if fail > 300 and time.monotonic() < deadline - 15:
            a = _perturb_block(best_a, rng, 0.35)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            velocity = np.zeros(n, dtype=np.float64)
            alpha = 0.4
            fail = 0
            k_top = max(min_k, k_top * 2)
            use_lse = False

        # If stuck for very long, more aggressive restart
        if it - best_it > 3000 and time.monotonic() < deadline - 20:
            a = _perturb(best_a, rng, 0.5)
            b = _autocorr(a, b_fft)
            maxb = float(np.max(b))
            score = 2.0 * n * maxb
            velocity = np.zeros(n, dtype=np.float64)
            alpha = 0.5
            fail = 0
            k_top = max(4, n // 8)
            use_lse = False

        it += 1

    return best_a, best_score


def _get_given_sequence():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or not np.isfinite(arr).all():
        return None
    s = float(arr.sum())
    if s < 1e-300:
        return None
    return arr / s


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point. Runs for at most budget_s seconds and returns the best
    nonnegative coefficient list found.
    """
    budget = float(budget_s) if np.isfinite(budget_s) and budget_s > 0 else 2400.0
    deadline = time.monotonic() + max(budget, 60.0) - 25.0
    rng = np.random.default_rng(seed)

    given = _get_given_sequence()

    # Initial best
    if given is not None and given.size >= 16:
        best_a = given.copy()
        best_score = _score(best_a)
    else:
        best_a = _make_exponential(256, 1.0)
        best_score = _score(best_a)

    print(f"Initial: n={best_a.size}, score={best_score:.8f}", flush=True)

    # Phase 1: Refine the given/best sequence at its native length
    if time.monotonic() < deadline - 60:
        t_end = min(deadline - 30, time.monotonic() + budget * 0.3)
        res_a, res_score = _refine(best_a.copy(), t_end, rng)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
        print(f"Phase 1: n={best_a.size}, score={best_score:.8f}", flush=True)

    # Phase 2: Try different lengths
    orig_n = best_a.size
    trial_lengths = [128, 192, 256, 320, 384, 512, 768, 1024]
    # Remove the original length (already tried) and sort by closeness to original
    trial_lengths = [n for n in trial_lengths if n != orig_n]
    trial_lengths.sort(key=lambda x: abs(x - orig_n))

    # Time budget for phase 2
    phase2_budget = budget * 0.35
    per_trial = max(15, int(phase2_budget / (len(trial_lengths) + 1)))

    for n_trial in trial_lengths[:5]:  # Try up to 5 different lengths
        if time.monotonic() > deadline - 60:
            break
        t_start = time.monotonic()
        t_end = min(deadline - 30, t_start + per_trial)
        if t_start >= t_end:
            continue

        # Resample best to new length
        init = _resample_sequence(best_a, n_trial)
        # Mix with exponential for diversity
        init = _normalize(0.8 * init + 0.2 * _make_exponential(n_trial, 1.0))

        res_a, res_score = _refine(init.copy(), t_end, rng)
        print(f"Phase 2: n={n_trial}, score={res_score:.8f}, best={best_score:.8f}", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

    # Phase 3: Multi-start at the best length
    best_n = best_a.size
    phase3_budget = budget * 0.2
    num_starts = max(2, int(phase3_budget / 60))
    per_start = max(20, int(phase3_budget / max(num_starts, 1)))

    for s in range(num_starts):
        if time.monotonic() > deadline - 30:
            break
        t_start = time.monotonic()
        t_end = min(deadline - 20, t_start + per_start)
        if t_start >= t_end:
            continue

        # Generate diverse initial candidates
        r = rng.random()
        if r < 0.3:
            init = _perturb(best_a, rng, 0.4)
        elif r < 0.5:
            init = _perturb_block(best_a, rng, 0.5)
        elif r < 0.7:
            init = _resample_sequence(best_a, best_n)
            init = _normalize(0.7 * init + 0.3 * _make_mixture(best_n, 1.2, 0.4))
        elif r < 0.85:
            init = _make_mixture(best_n, rng.uniform(0.5, 2.0), rng.uniform(0.2, 0.8))
            init = _normalize(0.5 * init + 0.5 * best_a)
        else:
            # Smooth random perturbation
            z = rng.normal(size=best_n)
            F = np.fft.rfft(z)
            cutoff = max(2, best_n // 8)
            if cutoff < len(F):
                F[cutoff:] *= 0.0
            f = np.fft.irfft(F, best_n)
            f = f / (np.std(f) + 1e-10)
            init = _normalize(best_a * (1.0 + 0.3 * f))

        res_a, res_score = _refine(init.copy(), t_end, rng)
        print(f"Phase 3 start {s+1}/{num_starts}: n={best_n}, score={res_score:.8f}, best={best_score:.8f}", flush=True)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score

    # Phase 4: Final deep refinement + coordinate descent
    if time.monotonic() < deadline - 10:
        t_end = min(deadline - 5, time.monotonic() + budget * 0.1)
        res_a, res_score = _refine(best_a.copy(), t_end, rng)
        if res_score < best_score:
            best_a = res_a.copy()
            best_score = res_score
        print(f"Phase 4 refine: score={best_score:.8f}", flush=True)

        # Final coordinate descent polishing
        if time.monotonic() < deadline - 3 and best_a.size <= 512:
            a_cd, score_cd = _coord_descent_sweep(best_a.copy(), max_sweeps=3)
            if score_cd < best_score:
                best_a = a_cd.copy()
                best_score = score_cd
                print(f"Phase 4 CD: score={best_score:.8f}", flush=True)

    # Final verification
    final_score = _score(best_a)
    print(f"Final: n={best_a.size}, score={final_score:.8f}", flush=True)

    if final_score < best_score:
        best_score = final_score

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    out = _normalize(out)

    return [float(x) for x in out]
