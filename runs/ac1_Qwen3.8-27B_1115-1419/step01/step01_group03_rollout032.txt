Therefore: $h[j] = (a_{\text{rev}} * w)[j + n - 1]$ for $j = 0, ..., n-1$.

So I compute the convolution of $a_{\text{rev}}$ (length $n$) with $w$ (length $2n-1$), which has length $3n-2$, and extract elements at indices $n-1, n, ..., 2n-2$.

"""
Minimize 2n*max(a*a)/(sum(a))^2 using a multi-phase FISTA approach with
FFT-based convolution and gradient computation. Uses smoothed max (softmax)
for warm-up, then switches to all-active-peaks subgradient for refinement.
Includes adaptive step size, momentum, and periodic restarts.

import numpy as np
import time


def fft_convolve_same(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    out_len = 2 * n - 1
    N = 1
    while N < out_len:
        N *= 2
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(A * A, N)
    return result[:out_len]


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(a):
    """Compute 2n*max(conv(a))/(sum(a))^2 without calling evaluate_sequence."""
    n = len(a)
    c = fft_convolve_same(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def compute_grad_active(a, c, tol=1e-10):
    """
    Compute subgradient from all active peaks (within tol of max).
    Returns gradient of the averaged active quadratic forms.
    """
    n = len(a)
    c_max = c.max()
    # Active set: all k where c[k] >= c_max - tol
    active = c >= (c_max - tol)
    k_active = np.where(active)[0]
    
    if len(k_active) == 0:
        k_active = np.array([np.argmax(c)])
    
    # For each j: grad[j] = 2 * sum_{k in active} a[k-j] * 1[0<=k-j<n]
    # = 2 * sum_{m=0}^{n-1} a[m] * 1[j+m in active]
    # This is a sliding dot product of a with indicator of active set
    active_float = active.astype(np.float64)
    
    # Compute sliding dot product using FFT
    # h[j] = sum_m a[m] * active[j+m] for j=0..n-1
    # h[j] = (a_rev * active)[j + n - 1]
    N = 1
    while N < 3 * n - 2:
        N *= 2
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(active_float, N)
    conv = np.fft.irfft(A * W, N)
    h = conv[n - 1:2 * n - 1]
    
    grad = 2.0 * h / len(k_active)
    return grad, len(k_active)


def compute_grad_softmax(a, c, epsilon):
    """Compute gradient of smoothed max using softmax weights."""
    n = len(a)
    c_max = c.max()
    shifted = (c - c_max) / epsilon
    shifted = np.clip(shifted, -50.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Sliding dot product: h[j] = sum_m a[m] * w[j+m]
    N = 1
    while N < 3 * n - 2:
        N *= 2
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(A * W, N)
    h = conv[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad


def compute_grad_topk(a, c, top_k=5):
    """Compute gradient from top-k peaks (averaged)."""
    n = len(a)
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    # Build indicator of top peaks
    active = np.zeros(len(c))
    active[top_idx] = 1.0
    active_float = active.astype(np.float64)
    
    N = 1
    while N < 3 * n - 2:
        N *= 2
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(active_float, N)
    conv = np.fft.irfft(A * W, N)
    h = conv[n - 1:2 * n - 1]
    
    grad = 2.0 * h / len(top_idx)
    return grad


def run_fista_smoothed(a0, duration, rng, phase_name=""):
    """
    Run FISTA with smoothed max objective for warm-up phase.

    a_prev = a0.copy()
    t_mom = 1.0
    step = 0.5  # Will be adapted
    best_a = a0.copy()
    best_score = fast_score(a0)
    
    # Initial epsilon based on scale of c
    c_init = fft_convolve_same(a)
    c_scale = c_init.max()
    epsilon = c_scale / 5.0
    epsilon_min = c_scale / 5000.0
    
    deadline = time.monotonic() + duration
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Decay epsilon
        epsilon = max(epsilon_min, epsilon * 0.999)
        
        # FISTA momentum
        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_mom
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        c_y = fft_convolve_same(y)
        grad = compute_grad_softmax(y, c_y, epsilon)
        
        # Gradient step + project
        a_new = y - step * grad
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        it += 1
        
        # Periodic evaluation
        if it % 100 == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve == 50:
                step *= 0.8
            elif no_improve == 200:

                # Reset momentum
                a_prev = a.copy()
                t_mom = 1.0
            elif no_improve > 500:

        t_old = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t_mom
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        c_y = fft_convolve_same(y)
        grad = compute_grad_topk(y, c_y, top_k)
        
        a_new = y - step * grad
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        it += 1
        
        if it % 50 == 0:
            sc = fast_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve == 30:
                step *= 0.85
            elif no_improve == 150:
                step *= 0.7
                a_prev = best_a.copy()
                t_mom = 1.0
            elif no_improve > 400:
                a = best_a + rng.normal(0.0, 0.003 / np.sqrt(n), n)
                np.maximum(a, 0.0, out=a)
                sa = a.sum()
                if sa > 1e-12:
                    a /= sa
                a_prev = a.copy()
                t_mom = 1.0
                step = 0.1
                no_improve = 0
            
            if it % 5000 == 0:
                print(f"  {phase_name} it={it}, score={best_score:.8f}, top_k={top_k}, step={step:.4f}")
    
    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def make_u_shape(n, alpha=1.0):
    """Create a U-shaped (bowl) distribution."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_semicircle(n):
    """Create a semicircle (Wigner) distribution."""
    x = np.linspace(-1.0, 1.0, n)
    a = np.sqrt(np.maximum(1.0 - x * x, 0.0))
    a = a / a.sum()
    return a


def make_two_bump(n, width=0.1):
    """Create a two-bump distribution at the edges."""
    x = np.linspace(0, 1, n)
    center_dist = np.minimum(x, 1.0 - x)
    a = np.exp(-center_dist ** 2 / (2.0 * width ** 2))
    a = a / a.sum()
    return a


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search function. Uses multi-phase FISTA:
    Phase 1: Smoothed max warm-up (softmax gradient)
    Phase 2: All-active-peaks refinement
    Phase 3: Top-k polishing
    multiple initializations.
    """
    rng = np.random.default_rng(42)
    total_deadline = time.monotonic() + float(budget_s) - 30.0
    
    # --- Get/initialize sequence ---
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.ones(n, dtype=np.float64)
    
    if n < 2:
        n = 256
        arr = np.ones(n, dtype=np.float64)
    
    print(f"Sequence length: {n}")
    
    # --- Build candidate initializations ---
    candidates = []
    names = []
    
    # 1. Given sequence normalized
    S = arr.sum()
    if S > 1e-10:
        candidates.append(arr / S)
        names.append("given")
    
    # 2. Uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)
    names.append("uniform")
    
    # 3. U-shaped
    candidates.append(make_u_shape(n, alpha=0.5))
    names.append("u_shape_0.5")
    candidates.append(make_u_shape(n, alpha=1.0))
    names.append("u_shape_1.0")
    candidates.append(make_u_shape(n, alpha=2.0))
    names.append("u_shape_2.0")
    
    # 4. Semicircle
    candidates.append(make_semicircle(n))
    names.append("semicircle")
    
    # 5. Two-bump at edges
    candidates.append(make_two_bump(n, width=0.08))
    names.append("two_bump_0.08")
    candidates.append(make_two_bump(n, width=0.15))
    names.append("two_bump_0.15")
    
    # 6. Sine-based (more mass in middle)
    x = np.linspace(0, 1, n)
    sine_a = np.sin(np.pi * x)
    sine_a = sine_a / sine_a.sum()
    candidates.append(sine_a)
    names.append("sine")
    
    # 7. Random
    rx = rng.exponential(1.0, n)
    candidates.append(rx / rx.sum())
    names.append("exp_random")
    
    # --- Evaluate all candidates ---
    best_cand = None
    best_cand_score = np.inf
    for c, nm in zip(candidates, names):
        sc = fast_score(c)
        print(f"  Init '{nm}': score = {sc:.8f}")
        if sc < best_cand_score:
            best_cand_score = sc
            best_cand = c.copy()
    
    print(f"Best initial score: {best_cand_score:.8f}")
    
    best_a = best_cand.copy()
    best_score = best_cand_score
    
    # --- Phase 1: Smoothed FISTA warm-up ---
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        p1_budget = min(time_left * 0.4, 600.0)
        print(f"\nPhase 1: Smoothed FISTA ({p1_budget:.0f}s)")
        a1, s1 = run_fista_smoothed(best_cand, p1_budget, rng, phase_name="P1")
        if s1 < best_score:
            best_a = a1
            best_score = s1
            print(f"  Phase 1 improved to: {best_score:.8f}")
    
    # --- Phase 2: All-active-peaks refinement ---
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        p2_budget = min(time_left * 0.35, 600.0)
        print(f"\nPhase 2: Active-peaks FISTA ({p2_budget:.0f}s)")
        a2, s2 = run_fista_active(best_a, p2_budget, rng, tol=1e-14, phase_name="P2")
        if s2 < best_score:
            best_a = a2
            best_score = s2
            print(f"  Phase 2 improved to: {best_score:.8f}")
    
    # --- Phase 3: Top-k polishing ---
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        p3_budget = min(time_left * 0.2, 300.0)
        print(f"\nPhase 3: Top-k polishing ({p3_budget:.0f}s)")
        a3, s3 = run_fista_topk(best_a, p3_budget, rng, top_k=20, phase_name="P3")
        if s3 < best_score:
            best_a = a3
            best_score = s3
            print(f"  Phase 3 improved to: {best_score:.8f}")
    
    # --- Phase 4: Final active-peaks with very tight tolerance ---
    time_left = total_deadline - time.monotonic()
    if time_left > 20:
        p4_budget = time_left - 10.0
        print(f"\nPhase 4: Final refinement ({p4_budget:.0f}s)")
        a4, s4 = run_fista_active(best_a, p4_budget, rng, tol=1e-16, phase_name="P4")
        if s4 < best_score:
            best_a = a4
            best_score = s4
            print(f"  Phase 4 improved to: {best_score:.8f}")
    
    # --- Final verification ---
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.8f}")
    
    if not np.isfinite(final_score) or final_score > best_cand_score:
        print(f"Falling back to best candidate: {best_cand_score:.8f}")
        final_list = [float(x) for x in best_cand]
        final_score = best_cand_score
    
    print(f"=== FINAL: {final_score:.8f} (n={n}) ===")
    return final_list
