def fft_convolve(a, b):

    """Numerically stable softmax."""
    x_shifted = x - x.max()

def lse_value_grad(x, tau):
    """
    LSE-smoothed max of conv(softmax(x), softmax(x)).
    Returns (value, gradient_wrt_x).

    # Chain rule through softmax
    dot = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - dot)
    
    return value, grad_x
