Optimize non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient descent in log-space with adaptive temperature,
multiple exploration strategies, and periodic exact validation.
"""

import numpy as np
import time

def next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p

def log_softmax(w):
    w = w - np.max(w)
    return w  # actually just return the shifted w

def softmax(w):
    w = w - np.max(w)
    e = np.exp(w)
    s = e.sum()
    if s == 0 or not np.isfinite(s):
        return np.ones(len(w)) / len(w)
    return e / s

def fft_conv(a, N):
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*len(a)-1]

def fft_score(a, N):
    n = len(a)
    c = fft_conv(a, N)
    mx = c.max()
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2*n*mx/(s*s))

def exact_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = np.convolve(a, a)
    return float(2*n*c.max()/(s*s))

def compute_grad(a, c, beta, N):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    bc = beta * c
    bc -= bc.max()
    p = np.exp(bc)
    p /= p.sum()
    # Cross-correlation: g[m] = 2*sum_j a[j]*p[m+j]

    A = np.fft.rfft(a, N)

    g = 2.0 * np.fft.irfft(np.conj(A) * P, N)[:n]
    return g

    m_adam = np.zeros(n)

"""
Optimize non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient descent in log-space with adaptive smoothed-max,
population-based exploration, structured restarts, and periodic exact validation.
Key differences from previous: proper cross-correlation gradient, multi-population
search, adaptive beta with hysteresis, structured perturbations, and cosine
annealing of learning rate.

import numpy as np
import time


def next_pow2(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def softmax_stable(w):
    """Numerically stable softmax."""
    w = np.asarray(w, dtype=np.float64)
    w = w - w.max()
    e = np.exp(w)
    s = e.sum()
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(w.shape, dtype=np.float64) / w.size
    return e / s


def fft_full_conv(a, N):
    """Full linear convolution via FFT. Returns array of length 2*len(a)-1."""
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[: 2 * len(a) - 1]


def fft_score_unit(a, N):
    """Score for unit-sum sequence: 2*n*max(conv)."""
    n = len(a)
    c = fft_full_conv(a, N)
    mx = float(c.max())
    return 2.0 * n * mx


def exact_score_unit(a):
    """Exact score for unit-sum sequence: 2*n*max(conv) using np.convolve."""
    n = len(a)
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())


def smooth_max_loss(c, beta):
    """Smoothed max: (1/beta)*log(sum(exp(beta*c)))."""
    bc = beta * c
    bc -= bc.max()
    return float(1.0 / beta * (np.log(np.sum(np.exp(bc))) + bc.max() * 0))


def compute_gradient_fft(a, c, beta, N):
    """
    Gradient of smoothed max loss w.r.t. a.
    g[m] = 2 * sum_k p[k] * a[k-m] for valid (k-m) in [0, n-1]
    = 2 * cross-correlation of a with p (the softmax weights).
    Computed via: 2 * IFFT(conj(FFT(a,N)) * FFT(p,N))[:n]
    """
    n = len(a)
    bc = beta * c
    bc -= bc.max()
    p = np.exp(bc)
    p_sum = p.sum()
    if not np.isfinite(p_sum) or p_sum < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / p_sum

    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    g = 2.0 * np.fft.irfft(np.conj(A) * P, N)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def adam_step(w, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """One Adam update step."""
    t += 1
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)
    return w_new, m, v, t


def log_weights(a):
    """Convert probability vector to log-weights (centered)."""
    a = np.maximum(a, 1e-300)
    w = np.log(a)
    w -= w.max()
    return w


class OptMember:
    """A single optimization member in the population."""
    __slots__ = ['w', 'a', 'c', 'score', 'm', 'v', 't_adam', 'beta',
                 'lr', 'stall', 'total_stall', 'name']

    def __init__(self, a, name=""):
        self.a = np.asarray(a, dtype=np.float64).copy()
        s = self.a.sum()
        if s < 1e-15:
            self.a = np.ones(len(a), dtype=np.float64) / len(a)
            s = 1.0
        self.a = self.a / s
        n = len(self.a)
        self.w = log_weights(self.a)
        self.m = np.zeros(n, dtype=np.float64)
        self.v = np.zeros(n, dtype=np.float64)
        self.t_adam = 0
        self.beta = 5.0
        self.lr = 0.05
        self.stall = 0
        self.total_stall = 0
        self.name = name
        self.c = None
        self.score = float('inf')


def reset_momentum(mem):
    mem.m = np.zeros(len(mem.a), dtype=np.float64)
    mem.v = np.zeros(len(mem.a), dtype=np.float64)
    mem.t_adam = 0


def do_gradient_step(mem, N):
    """One gradient descent step for a member. Returns new score."""
    a = mem.a
    c = mem.c
    n = len(a)

    g = compute_gradient_fft(a, c, mem.beta, N)
    gw = a * (g - np.dot(a, g))
    gw = np.where(np.isfinite(gw), gw, 0.0)
    gw = np.clip(gw, -50.0, 50.0)

    mem.w, mem.m, mem.v, mem.t_adam = adam_step(
        mem.w, gw, mem.m, mem.v, mem.t_adam, mem.lr
    )
    mem.a = softmax_stable(mem.w)
    mem.c = fft_full_conv(mem.a, N)
    return fft_score_unit(mem.a, N)


def structured_perturb(mem, rng, n, N):
    """Apply a structured perturbation to a member."""
    w = mem.w.copy()
    strategy = rng.integers(0, 6)

    if strategy == 0:
        # Gaussian noise
        sig = float(rng.choice([0.05, 0.1, 0.2, 0.4, 0.8]))
        w += rng.normal(0, sig, size=n)
    elif strategy == 1:
        # Add sinusoidal component
        k = int(rng.integers(1, n // 2))
        amp = float(rng.choice([0.1, 0.3, 0.5, 1.0]))
        phase = float(rng.uniform(0, 2 * np.pi))
        idx = np.arange(n)
        w += amp * np.sin(2 * np.pi * k * idx / n + phase)
    elif strategy == 2:
        # Scale a random block
        block_start = int(rng.integers(0, n - 10))
        block_len = int(rng.integers(10, max(11, n // 4)))
        block_len = min(block_len, n - block_start)
        scale = float(rng.uniform(0.3, 3.0))
        w[block_start:block_start + block_len] += np.log(scale)
    elif strategy == 3:
        # Reflection averaging (symmetrize)
        a_sym = (mem.a + mem.a[::-1]) / 2.0
        s = a_sym.sum()
        if s > 0:
            a_sym = a_sym / s
            w = log_weights(a_sym)
            w += rng.normal(0, 0.05, size=n)
    elif strategy == 4:
        # Add low-frequency ramp
        idx = np.arange(n, dtype=np.float64)
        w += float(rng.choice([-0.5, 0.5])) * idx / n
    else:
        # Multiply by raised cosine
        idx = np.arange(n, dtype=np.float64)
        rc = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
        w += np.log(np.maximum(rc, 1e-10) * n / 2)

    mem.w = w
    mem.a = softmax_stable(mem.w)
    reset_momentum(mem)
    mem.c = fft_full_conv(mem.a, N)
    mem.score = fft_score_unit(mem.a, N)
    mem.stall = 0
    mem.beta = 5.0
    mem.lr = 0.05

def finalize_sequence(a):
    """Scale to sum sqrt(2n), ensure valid non-negative floats."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    scale = float(np.sqrt(2.0 * n))
    if s > 1e-15:
        a = a * scale / s
    else:
        a = np.ones(n, dtype=np.float64) * (scale / n)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = a * (0.01 / max(s2, 1e-300))
        a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]

        start = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
        start += 0.1 * np.sin(4 * np.pi * idx / (n - 1))
        start = np.maximum(start, 0.01)
        start = start / start.sum()

    seq = run_optimization(start, budget_s=budget - 5.0, seed=seed)
    return finalize_sequence(seq)
