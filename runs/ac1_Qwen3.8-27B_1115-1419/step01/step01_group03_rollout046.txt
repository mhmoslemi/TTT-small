def fast_score(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n <= 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():

    return best_a, best_score


def run_soft_fista(a0, duration, rng, tau_start=0.05, tau_end=0.001, step_init=0.4):
    """
    FISTA with soft (exponential) weighting of all convolution coefficients.
    w_k = exp(c_k / tau) / sum(exp(c/tau)) — smoothly transitions from uniform to peak-focused.
    Uses O(n log n) gradient via convolution.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    a_prev = a.copy()
    t = 1.0
    step = step_init
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration
    start_time = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / duration)
        tau = tau_start * (tau_end / tau_start) ** progress

        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        c_y = np.convolve(y, y)

        # Softmax weights (numerically stable)
        c_max = c_y.max()
        shifted = (c_y - c_max) / tau
        np.clip(shifted, -500, 500, out=shifted)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(c_y)) / len(c_y)
        else:
            w /= w_sum

        grad = grad_weighted(y, w)
        a_new = project_simplex(y - step * grad)

        a_prev = a
        a = a_new
        it += 1

        if it % 40 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
                step = min(step * 1.005, 1.5)
            else:
                stuck += 1
                if stuck == 50:
                    step *= 0.85
                elif stuck == 200:
                    step *= 0.7
                    a_prev = best_a.copy()
                    t = 1.0
                elif stuck > 600:
                    a = best_a.copy()
                    mask = rng.random(n) < 0.05
                    a[mask] += rng.normal(0, 0.01 / np.sqrt(n), mask.sum())
                    np.maximum(a, 0, out=a)
                    sa = a.sum()
                    if sa > 1e-12:
                        a /= sa
                    a_prev = a.copy()
                    t = 1.0
                    step = step_init
                    stuck = 0

            if it % 3000 == 0:
                print(f"      SoftFISTA: it={it}, score={best_score:.7f}, tau={tau:.5f}, step={step:.4f}")

    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def run_polish(a0, duration, rng, top_k=3):
    """
    Final polishing phase with small steps and exact active-set gradient.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    a_prev = a.copy()
    t = 1.0
    step = 0.2
    best_a = a.copy()
    best_score = fast_score(a)
    deadline = time.monotonic() + duration
    it = 0
    stuck = 0

    while time.monotonic() < deadline:
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        momentum = (t_old - 1.0) / t
        y = a + momentum * (a - a_prev)
        np.maximum(y, 0.0, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        c_y = np.convolve(y, y)
        M = c_y.max()

        # Use all "nearly active" peaks (within 1% of max)
        threshold = M * 0.99
        active = c_y >= threshold
        if not active.any():
            active = np.zeros(len(c_y), dtype=bool)
            active[np.argmax(c_y)] = True

        w = np.zeros(len(c_y))
        w[active] = 1.0
        w /= w.sum()

        grad = grad_weighted(y, w)
        a_new = project_simplex(y - step * grad)

        a_prev = a
        a = a_new
        it += 1

        if it % 30 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 200:
                    step *= 0.8
                elif stuck > 800:
                    a = best_a.copy()
                    a_prev = a.copy()
                    t = 1.0
                    stuck = 0

            if it % 2000 == 0:
                print(f"      Polish: it={it}, score={best_score:.7f}, step={step:.5f}, n_active={active.sum()}")

    sc = fast_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization:
    1. Quick FISTA warmup with top-5 (2 min)
    2. Soft-weighted FISTA with annealing (28 min)
    3. Local polishing with active-set gradient (6 min)
    Returns best non-negative float list found.
    """
    rng = np.random.default_rng(42)
    total_deadline = time.monotonic() + float(budget_s) - 15.0

    # Get initial sequence
    try:
        arr = np.array(height_sequence_1, dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        n = len(arr)
    except Exception:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    if n < 2:
        n = 4806
        arr = np.ones(n, dtype=np.float64)

    print(f"=== Sequence length: {n} ===")

    # Normalize
    S = arr.sum()
    if S < 1e-10:
        arr = np.ones(n) / n
    else:
        arr = arr / S

    # Evaluate initial
    init_score = fast_score(arr)
    print(f"Initial score: {init_score:.8f}")

    # Also try a few alternative initializations
    candidates = [arr.copy()]
    candidate_scores = [init_score]

    # Slight perturbation
    pert = arr + rng.normal(0, 0.01 / np.sqrt(n), n)
    np.maximum(pert, 0, out=pert)
    pert /= pert.sum()
    candidates.append(pert.copy())
    candidate_scores.append(fast_score(pert))

    # Sparsify: zero out 20% of smallest
    sp = arr.copy()
    thresh = np.sort(sp)[int(0.2 * n)]
    sp[sp < thresh] = 0
    if sp.sum() > 1e-10:
        sp /= sp.sum()
        candidates.append(sp.copy())
        candidate_scores.append(fast_score(sp))

    # Pick best candidate
    best_init = candidates[np.argmin(candidate_scores)].copy()
    best_init_score = min(candidate_scores)
    print(f"Best init score: {best_init_score:.8f}")

    best_a = best_init.copy()
    best_score = best_init_score

    # Phase 1: Quick FISTA warmup (2 min)
    time_left = total_deadline - time.monotonic()
    if time_left > 120:
        p1_budget = min(120.0, time_left * 0.08)
        print(f"Phase 1: FISTA top-5 ({p1_budget:.0f}s)")
        a1, s1 = run_fista_topk(best_init, p1_budget, rng, top_k=5, step_init=0.4)
        if s1 < best_score:
            best_a, best_score = a1, s1
        print(f"Phase 1 result: {best_score:.8f}")

    # Phase 2: Soft-weighted FISTA with annealing (28 min)
    time_left = total_deadline - time.monotonic()
    if time_left > 60:
        p2_budget = min(time_left * 0.8, 1700.0)
        print(f"Phase 2: Soft FISTA ({p2_budget:.0f}s)")
        a2, s2 = run_soft_fista(best_a, p2_budget, rng, tau_start=0.03, tau_end=0.0005, step_init=0.4)
        if s2 < best_score:
            best_a, best_score = a2, s2
        print(f"Phase 2 result: {best_score:.8f}")

    # Phase 2b: If improved, run another soft pass from the new point with different tau
    time_left = total_deadline - time.monotonic()
    if time_left > 120:
        p2b_budget = min(time_left * 0.4, 400.0)
        print(f"Phase 2b: Soft FISTA round 2 ({p2b_budget:.0f}s)")
        a2b, s2b = run_soft_fista(best_a, p2b_budget, rng, tau_start=0.01, tau_end=0.0002, step_init=0.3)
        if s2b < best_score:
            best_a, best_score = a2b, s2b
        print(f"Phase 2b result: {best_score:.8f}")

    # Phase 3: Polishing (remaining time, at least 6 min if possible)
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        p3_budget = max(30.0, time_left - 5.0)
        print(f"Phase 3: Polishing ({p3_budget:.0f}s)")
        a3, s3 = run_polish(best_a, p3_budget, rng, top_k=3)
        if s3 < best_score:
            best_a, best_score = a3, s3
        print(f"Phase 3 result: {best_score:.8f}")

    # Phase 4: Final micro-polish if time remains
    time_left = total_deadline - time.monotonic()
    if time_left > 30:
        p4_budget = max(20.0, time_left - 3.0)
        print(f"Phase 4: Micro-polish ({p4_budget:.0f}s)")
        a4, s4 = run_fista_topk(best_a, p4_budget, rng, top_k=3, step_init=0.1)
        if s4 < best_score:
            best_a, best_score = a4, s4
        print(f"Phase 4 result: {best_score:.8f}")

    # Final verification with evaluate_sequence
    final_list = [float(max(0.0, x)) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.8f}")

    if not np.isfinite(final_score) or final_score > best_score:
        print(f"Falling back to best internal score: {best_score:.8f}")
        # Re-verify best internal
        fl = [float(max(0.0, x)) for x in best_a]
        fs = float(evaluate_sequence(fl))
        if np.isfinite(fs):
            final_list = fl
            final_score = fs
        else:
            final_list = [float(max(0.0, x)) for x in best_init]
            final_score = init_score

    print(f"=== FINAL: {final_score:.8f} (n={n}) ===")
    return final_list
