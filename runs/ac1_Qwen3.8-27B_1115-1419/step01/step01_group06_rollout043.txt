"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses symmetric parameterization, adaptive smoothed-Linf gradient descent with
FFT convolution, multi-candidate population, and aggressive restart strategies.
"""

import numpy as np
import time


def next_fast_len(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(v):
    arr = np.asarray(v, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(4)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(arr.size, dtype=np.float64) / arr.size
    return arr / s


def symmetrize(a):
    n = len(a)
    return (a + a[::-1]) / 2.0


def log_param(a):
    a = np.maximum(a, 1e-300)
    w = np.log(a).astype(np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= float(np.max(w))
    return w


def exp_param(w):
    w = np.asarray(w, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w -= float(np.max(w))
    e = np.exp(w)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(w.size, dtype=np.float64) / w.size
    return e / s


def fft_convolve(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def get_score_and_conv(a, N):
    n = len(a)
    c = fft_convolve(a, N)
    mx = float(np.max(c))
    return 2.0 * n * mx, c


def softmax_peaks(c, beta):
    L = float(np.max(c))
    if not np.isfinite(L) or L <= 0.0:
        p = np.zeros(len(c), dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    x = float(beta) * (c - L) / L
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(len(c), dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def compute_gradient(a, p, N, n, phase):
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    FR = phase * np.conjugate(A)
    corr = np.fft.irfft(P * FR, N)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    grad = grad - float(np.dot(a, grad))
    return grad


def adam_step(grad_w, w, m, v, t, lr):
    t += 1
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1.0 - b1) * grad_w
    v = b2 * v + (1.0 - b2) * (grad_w * grad_w)
    mhat = m / (1.0 - b1 ** t)
    vhat = v / (1.0 - b2 ** t)
    dw = mhat / (np.sqrt(vhat) + 1e-12)
    dw = np.where(np.isfinite(dw), dw, 0.0)
    dw = np.clip(dw, -20.0, 20.0)
    w_new = w - float(lr) * dw
    w_new = np.where(np.isfinite(w_new), w_new, 0.0)
    w_new -= float(np.max(w_new))
    a_new = exp_param(w_new)
    return a_new, w_new, m, v, t


def load_parent():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 100:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return arr / s


def exact_evaluate(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float('inf')
    return float(2.0 * n * mx / (s * s))


def finalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    scale = float(np.sqrt(2.0 * n))
    if s > 0.0 and np.isfinite(s):
        a = a * scale / s
    else:
        a = np.ones(n, dtype=np.float64) * (scale / n)
    a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def run_optimize(a_init, N, phase, budget_s, seed, use_sym=True):
    """Run gradient descent optimization on a single candidate."""
    start_t = time.monotonic()
    deadline = start_t + budget_s
    n = len(a_init)
    a = normalize_nonneg(a_init)
    if use_sym:
        a = symmetrize(a)
        a = normalize_nonneg(a)

    score, c = get_score_and_conv(a, N)
    best_a = a.copy()
    best_score = score

    w = log_param(a)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0

    beta = 8.0
    lr = 0.1
    stall = 0
    iters = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        iters += 1

        # Compute gradient
        p = softmax_peaks(c, beta)
        grad = compute_gradient(a, p, N, n, phase)
        grad_w = a * grad

        # Adam step
        a_new, w, m, v, t_adam = adam_step(grad_w, w, m, v, t_adam, lr)

        if use_sym:
            a_new = symmetrize(a_new)
            a_new = normalize_nonneg(a_new)

        new_score, new_c = get_score_and_conv(a_new, N)

        if np.isfinite(new_score) and new_score < score:
            a = a_new
            c = new_c
            score = new_score
            stall = 0
            lr = min(0.3, lr * 1.02)

            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            stall += 1
            lr = max(0.01, lr * 0.85)

            if stall > 20:
                # Restart with perturbation
                sig = float(rng.choice([0.1, 0.2, 0.4, 0.8]))
                w_try = log_param(best_a) + rng.normal(0.0, sig, size=n)
                a = exp_param(w_try)
                if use_sym:
                    a = symmetrize(a)
                    a = normalize_nonneg(a)
                score, c = get_score_and_conv(a, N)
                w = log_param(a)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                t_adam = 0
                stall = 0
                lr = 0.08
                beta = 8.0

        # Gradually increase beta (sharper peaks)
        if iters % 50 == 0:
            beta = min(300.0, beta * 1.12 + 0.3)

        # Periodic large perturbation for exploration
        if iters % 500 == 0 and rng.random() < 0.3:
            sig = float(rng.choice([0.3, 0.5, 1.0]))
            w_try = log_param(a) + rng.normal(0.0, sig, size=n)
            a_try = exp_param(w_try)
            if use_sym:
                a_try = symmetrize(a_try)
                a_try = normalize_nonneg(a_try)
            sc_try, c_try = get_score_and_conv(a_try, N)
            if np.isfinite(sc_try) and sc_try < score * 1.05:
                a = a_try
                c = c_try
                score = sc_try
                w = log_param(a)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                t_adam = 0

    return best_a, best_score, iters


def run_cross_optimize(a_fixed, N, phase, budget_s, seed):
    """Optimize b to minimize max(f*b) for fixed f, then blend."""
    start_t = time.monotonic()
    deadline = start_t + budget_s
    n = len(a_fixed)
    F = np.fft.rfft(a_fixed, N)
    FR = phase * np.conjugate(F)

    b = normalize_nonneg(a_fixed)
    score_f, c_f = get_score_and_conv(a_fixed, N)

    w_b = log_param(b)
    m_b = np.zeros(n, dtype=np.float64)
    v_b = np.zeros(n, dtype=np.float64)
    t_b = 0

    beta_b = 10.0
    lr_b = 0.1
    best_blend_a = a_fixed.copy()
    best_blend_score = score_f
    rng = np.random.default_rng(seed + 1)

    while time.monotonic() < deadline:
        # Compute h = f*b (cross-correlation-like)
        B = np.fft.rfft(b, N)
        h = np.fft.irfft(F * B, N)[:2 * n - 1]
        h = np.maximum(h, 0.0)
        mx_h = float(np.max(h))

        if not np.isfinite(mx_h) or mx_h <= 0:
            break

        p = softmax_peaks(h, beta_b)
        P = np.fft.rfft(p, N)
        grad = np.fft.irfft(P * FR, N)[:2 * n - 1]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        grad = grad - float(np.dot(b, grad))

        b_new, w_b, m_b, v_b, t_b = adam_step(b * grad, w_b, m_b, v_b, t_b, lr_b)
        b = b_new

        # Try blending with fixed a
        for tv in (0.3, 0.5, 0.7, 0.9, 1.0):
            cand = (1.0 - tv) * a_fixed + tv * b
            cs = float(cand.sum())
            if cs <= 0 or not np.isfinite(cs):
                continue
            cand = cand / cs
            sc, _ = get_score_and_conv(cand, N)
            if np.isfinite(sc) and sc < best_blend_score:
                best_blend_score = sc
                best_blend_a = cand.copy()

        beta_b = min(200.0, beta_b * 1.05 + 0.2)

    return best_blend_a, best_blend_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    n = len(parent)

        target = 10000
        blocks = n // target
        use = target * blocks
        parent = np.mean(parent[:use].reshape(target, blocks), axis=1)
        n = parent.size
    if n < 100:
        parent = np.ones(2000, dtype=np.float64) / 2000
        n = 2000

    N = next_fast_len(3 * n - 2)

    parent = normalize_nonneg(parent)

    # Initial evaluation
    init_score, _ = get_score_and_conv(parent, N)

    # Phase 1: Optimized gradient descent (symmetric)
    t1 = time.monotonic()
    time_for_p1 = min(1200.0, budget * 0.5)
    best_a, best_score, iters1 = run_optimize(

    )
    t_elapsed = time.monotonic() - t1

    # The exact_evaluate uses the same formula as the optimizer (sum=1)

    return finalize_sequence(best_a)
