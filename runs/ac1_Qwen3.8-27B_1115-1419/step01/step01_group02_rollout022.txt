```python
import numpy as np
import time
from scipy.fft import fft, ifft

"""
Minimizes 2n*max(convolve(a,a))/(sum(a))^2 using smoothed mirror descent on the simplex.
Uses log-sum-exp smoothing of the max, FFT-based convolution and gradient computation,
exponentiated gradient updates with Nesterov momentum, and temperature annealing.
"""

def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n_fft = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_fft) * fft(b, n_fft)).real[:n_out]

def _true_score(a):
    """Compute the true objective (for tracking, not for gradient)."""
    n = len(a)
    c = _fft_conv(a, a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (sum_a * sum_a)

def _smoothed_max_and_weights(c, t):
    """Compute smoothed maximum and softmax weights of c with temperature t."""
    c_max = np.max(c)
    x = t * (c - c_max)
    # Clamp for numerical stability
    x = np.maximum(x, -500)
    exp_x = np.exp(x)
    Z = np.sum(exp_x)
    smoothed = c_max + np.log(Z) / t
    w = exp_x / Z
    return smoothed, w

def _gradient_smoothed(w, a):
    """
    Compute gradient of (1/t)*log(sum exp(t*c[k])) where c=conv(a,a).
    
    g_j = sum_{d=1}^{n-1} a[d]*(w[j+d]+w[j-d]) + 2*a[0]*w[j]
    with w[k]=0 for k<0 or k>=2n-1.
    """
    n = len(a)
    if n <= 1:
        return np.zeros(n)
    
    f = a[1:]  # length n-1
    g = w      # length 2n-1
    N = 1 << (len(f) + len(g)).bit_length()
    
    F = fft(f, N)
    G = fft(g, N)
    
    # U[j] = (f ⋆ g)[j+1] = ifft(conj(F)*G)[j+1]
    U_full = ifft(np.conj(F) * G).real
    # V[j] = (f * g)[j-1] = ifft(F*G)[j-1]
    V_full = ifft(F * G).real
    
    U = U_full[1:n+1]  # U[0] to U[n-1]
    V = np.zeros(n)
    V[1:] = V_full[0:n-1]  # V[1] to V[n-1]
    
    grad = U + V + 2.0 * a[0] * w[:n]
    return grad

def _mirror_descent_step(p, grad, lr, nesterov_mom=None):
    """
    One step of (Nesterov) mirror descent on the simplex.
    p >= 0, sum(p) = 1.
    """
    if nesterov_mom is not None:
        # Nesterov: evaluate gradient at y = p + mu*(p - p_prev)
        # Here we approximate: grad is already computed at the lookahead point
        # So we just do the exponential update
        pass
    
    # Exponentiated gradient: p_new[j] = p[j] * exp(-lr * grad[j]) / Z
    # For numerical stability: shift grad by its max
    shifted = -lr * grad
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    p_new = p * weights
    p_new /= np.sum(p_new)
    return p_new

def _run_smoothed_opt(p, t_start, t_end, max_steps, lr_decay=0.99995, 
                      lr_init=0.5, nesterov_beta=0.9, deadline=None,
                      eval_interval=200, name=""):
    """
    Run smoothed mirror descent with annealing.
    p: initial probability vector (sum=1, p>=0)
    Returns best p and best score.
    """
    n = len(p)
    p = p.copy()
    p /= np.sum(p)
    p = np.maximum(p, 1e-15)
    p /= np.sum(p)
    
    momentum = np.zeros(n)
    p_prev = p.copy()
    lr = lr_init
    best_p = p.copy()
    best_score = _true_score(p)
    
    t = t_start
    t_ratio = (t_end - t_start) / max(1, max_steps)
    
    for step in range(max_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal temperature
        t = t_start + t_ratio * step
        t = max(t, 1e-3)
        
        # Nesterov lookahead
        y = p + nesterov_beta * (p - p_prev)
        y = np.maximum(y, 1e-15)
        y /= np.sum(y)
        
        # Compute convolution and gradient at y
        c = _fft_conv(y, y)
        _, w = _smoothed_max_and_weights(c, t)
        grad = _gradient_smoothed(w, y)
        
        # Momentum
        momentum = nesterov_beta * momentum + (1.0 - nesterov_beta) * grad
        
        # Mirror descent step
        shifted = -lr * momentum
        shifted -= np.max(shifted)
        p_new = p * np.exp(shifted)
        p_new = np.maximum(p_new, 1e-15)
        p_new /= np.sum(p_new)
        
        p_prev = p.copy()
        p = p_new
        lr *= lr_decay
        
        # Periodic evaluation
        if step % eval_interval == 0:
            sc = _true_score(p)
            if sc < best_score:
                best_score = sc
                best_p = p.copy()
    
    # Final evaluation
    sc = _true_score(p)
    if sc < best_score:
        best_score = sc
        best_p = p.copy()
    
    return best_p, best_score
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft

"""
Minimizes 2n*max(convolve(a,a))/(sum(a))^2 using smoothed mirror descent on the
probability simplex. Uses log-sum-exp smoothing of the max for a differentiable
objective, FFT-based convolution and gradient, exponentiated gradient updates
with momentum, and multi-phase annealing with diverse restarts.
"""


def _fft_conv(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    n_fft = 1 << (n_out - 1).bit_length()
    return ifft(fft(a, n_fft) * fft(b, n_fft)).real[:n_out]


def _true_score(a):
    """Compute the true objective: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def _smoothed_max_w(c, t):
    """Log-sum-exp smoothing of max: returns (smoothed_value, softmax_weights)."""
    c_max = np.max(c)
    x = t * (c - c_max)
    x = np.clip(x, -500.0, 500.0)
    exp_x = np.exp(x)
    Z = np.sum(exp_x)
    val = c_max + np.log(Z) / t
    w = exp_x / Z
    return val, w


def _gradient_fft(w, a):
    """
    Gradient of (1/t)*log(sum_k exp(t*c[k])) w.r.t. a, computed via FFT.
    c = convolve(a,a) (linear, length 2n-1).
    
    g_j = sum_{d=1}^{n-1} a[d]*(w[j+d] + w[j-d]) + 2*a[0]*w[j]
    where w[k]=0 for k<0 or k>=2n-1.
    """
    n = len(a)
    if n <= 1:
        return np.zeros(n)
    f = a[1:]  # length n-1
    g = w      # length 2n-1
    N = 1 << (len(f) + len(g) - 1).bit_length()
    F = fft(f, N)
    G = fft(g, N)
    # U[j] = (f ⋆ g)[j+1]: cross-correlation shifted
    U_full = ifft(np.conj(F) * G).real
    # V[j] = (f * g)[j-1]: convolution shifted
    V_full = ifft(F * G).real
    U = U_full[1:n + 1]
    V = np.zeros(n)
    V[1:] = V_full[0:n - 1]
    grad = U + V + 2.0 * a[0] * w[:n]
    return grad


def _topk_subgrad(c, a, top_k=3):
    """Subgradient at top-k peaks of c=conv(a,a). O(n*top_k)."""
    n = len(a)
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for m in top_idx:
        m = int(m)
        # For each j: dc[m]/da[j] = a[m-j]*(m-j in [0,n)) + a[j-m]*(j-m in [0,n))
        # Vectorized:
        # a[m-j] for j where 0<=m-j<n => m-n+1 <= j <= m
        # a[j-m] for j where 0<=j-m<n => m <= j <= m+n-1
        lo1 = max(0, m - n + 1)
        hi1 = min(n - 1, m)
        if hi1 >= lo1:
            js = np.arange(lo1, hi1 + 1)
            idx = m - js
            if m == 0 and 0 in js:
                # at j=m=0, the two terms are the same a[0]
                grad[lo1:hi1 + 1] += a[idx]
            else:
                grad[lo1:hi1 + 1] += a[idx]
        lo2 = max(0, m)
        hi2 = min(n - 1, m + n - 1)
        if hi2 >= lo2:
            js = np.arange(lo2, hi2 + 1)
            idx = js - m
            if m == 0 and 0 in js:
                # avoid double-counting at j=m
                mask = js != m
                grad[js[mask]] += a[idx[mask]]
            else:
                grad[lo2:hi2 + 1] += a[idx]
    return grad


def _optimize(p, t_start, t_end, total_time, lr, beta,
              deadline, eval_iv=500, use_topk_fallback=False, topk=1):
    """
    Smoothed mirror descent with momentum and annealing.
    p: probability vector (sum=1, p>=0).
    Returns (best_p, best_score).
    """
    n = len(p)
    p = np.maximum(p, 1e-15).astype(np.float64)
    p /= np.sum(p)
    mom = np.zeros(n)
    best_p = p.copy()
    best_sc = _true_score(p)
    t = t_start
    steps = 0
    start_t = time.monotonic()
    stuck = 0
    last_improve_sc = best_sc

    while time.monotonic() < deadline:
        t = min(t_start + (t_end - t_start) * (steps / max(1, int(total_time * 800))), t_end)
        t = max(t, 0.5)

        c = _fft_conv(p, p)
        _, w = _smoothed_max_w(c, t)
        grad = _gradient_fft(w, p)

        if use_topk_fallback and steps % 1000 == 500:
            g_top = _topk_subgrad(c, p, topk)
            grad = 0.5 * grad + 0.5 * g_top

        mom = beta * mom + (1.0 - beta) * grad

        # Exponentiated gradient (mirror descent on simplex)
        scaled = -lr * mom
        scaled -= np.max(scaled)
        p_new = p * np.exp(np.clip(scaled, -50, 50))
        p_new = np.maximum(p_new, 1e-15)
        s_new = np.sum(p_new)
        if s_new < 1e-10:
            p_new = np.ones(n) / n
        else:
            p_new /= s_new

        p = p_new
        lr *= 0.99998
        steps += 1

        if steps % eval_iv == 0:
            sc = _true_score(p)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_p = p.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 30:
                # Perturbation restart
                p = best_p + 0.02 * np.random.standard_normal(n)
                p = np.maximum(p, 0.0)
                p = p / np.sum(p)
                mom = np.zeros(n)
                lr = 0.3 * n
                stuck = 0

    sc = _true_score(p)
    if sc < best_sc:
        best_sc = sc
        best_p = p.copy()
    return best_p, best_sc

    print(f"  Elapsed: {time.monotonic() - t0_global:.0f}s")

    if final_actual < final_fft:
        best_score = final_actual

    # Return as list of floats
    result = [float(max(0.0, min(1000.0, x))) for x in best_p]
    # Verify it's valid
    check = evaluate_sequence(result)
    if not np.isfinite(check):
        # Fallback to parent
        result = [float(max(0.0, min(1000.0, x))) for x in p_par]
        check = evaluate_sequence(result)
    print(f"  Final returned score: {check:.6f}")
    return result
```
