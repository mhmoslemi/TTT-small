import time

def _fft_convolve(a, p):
    """Compute g[i] = 2 * sum_j a[j] * p[i+j] using FFT."""
    n = len(a)
    a_rev = a[::-1]
    L = 1
    while L < 2 * (n + len(p) - 1):
        L *= 2
    A = np.fft.rfft(a_rev, L)
    P = np.fft.rfft(p, L)
    C = np.fft.irfft(A * P, L)
    return 2.0 * C[n-1 : 2*n-1]

def _smooth_objective_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient using log-sum-exp."""
    n = len(a)
    # Compute convolution via FFT
    L = 1
    while L < 2*n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    c = C[:2*n-1]
    
    # Log-sum-exp for stable computation
    mc = np.max(c)
    logsumexp = mc + np.log(np.sum(np.exp(beta*(c - mc))))
    smooth_obj = logsumexp / beta
    
    # Softmax weights
    p = np.exp(beta * (c - mc))
    p /= np.sum(p)
    
    # Gradient
    g = _fft_correlate_grad(a, p)
    
    return smooth_obj, g, c

def _fft_correlate_grad(a, p):

import time

def _next_pow2(x):
    """Next power of 2 >= x."""
    L = 1
    while L < x:
        L <<= 1
    return L

def _fft_self_conv(a):
    n = len(a)
    L = _next_pow2(2*n - 1)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2*n-1]

def _fft_grad(a, p):
    n = len(a)
    a_rev = a[::-1].copy()
    L = _next_pow2(3*n - 2)
    A = np.fft.rfft(a_rev, L)
    P = np.fft.rfft(p, L)
    C = np.fft.irfft(A * P, L)

def _project(a, n):
    """Project to {a >= 0, sum(a) = n}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64)
    return a * (n / s)
