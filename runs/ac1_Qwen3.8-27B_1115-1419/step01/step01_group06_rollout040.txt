**Idea: Use the structure of the problem more directly.**

The key observation is that we want `convolve(a, a)` to be as flat as possible. The convolution has 2n-1 entries, and their sum is 1 (when sum(a)=1). So we want each entry to be approximately `1/(2n-1)`.

The "excess" at position k is `c_k - 1/(2n-1)`. We want to minimize the maximum excess.

**Idea: Direct peak reduction.**

At each step:
1. Find the position k* where c is maximized.
2. c_{k*} = sum_i a_i * a_{k*-i}.
3. To reduce c_{k*}, we need to reduce the product a_i * a_{k*-i} for the pairs that contribute most.
4. The "heaviest" pairs are those with large a_i * a_{k*-i}.
5. Move mass from one side of each heavy pair to somewhere else.

This is essentially a "peak shaving" approach. It's similar to what the gradient method does, but potentially more focused.

**Idea: Complementary approach - reduce the "energy" in the high-convolution region.**

Think of it this way: the convolution at position k is `sum_i a_i * a_{k-i}`. If we think of "a" as being "placed" on the number line, then c_k is the "overlap energy" between a and a shifted by k. We want to minimize the maximum overlap energy over all shifts.

This is related to the "minimum peak sidelobe" problem in radar/communications, where one designs a sequence to minimize the maximum sidelobe of its autocorrelation. The techniques used there (like the "Roy" algorithm or "Hadamard" based methods) might be adaptable.

"""
Optimize non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2
using multiplicative gradient descent (mirror descent on the simplex) with
adaptive smoothing and step size. Starts from the parent sequence and applies
iterative peak-reduction via softmax-weighted subgradients of the self-convolution.
"""

import numpy as np
import time


def _next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(v):
    arr = np.asarray(v, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(max(1, arr.size), dtype=np.float64) / max(1, arr.size)
    return arr / s


def _fft_score(a, N):
    n = len(a)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mx = float(c.max())
    if not np.isfinite(mx):
        return float('inf'), c
    return float(2.0 * n * mx), c


def _exact_score_list(alist):
    a = np.array(alist, dtype=np.float64)
    n = len(a)
    b = np.convolve(a, a)
    mx = float(b.max())
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * mx / (s * s))


def _multiplicative_step(a, beta, eta, N_conv, N_grad):
    """One multiplicative gradient descent step. Returns (a_new, score, c)."""
    n = len(a)
    A = np.fft.rfft(a, N_conv)
    c = np.fft.irfft(A * A, N_conv)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    max_c = float(c.max())
    if not np.isfinite(max_c) or max_c <= 0.0:
        return a, float('inf'), c

    # Softmax weights over convolution entries
    x = beta * c / max_c
    x -= x.max()
    w = np.exp(x)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ws

    # Gradient: g_i = sum_j w_{i+j} * a_j = convolve(w, flip(a))[n-1 : 2n-1]
    flip_a = a[::-1]
    W = np.fft.rfft(w, N_grad)
    FA = np.fft.rfft(flip_a, N_grad)
    g_full = np.fft.irfft(W * FA, N_grad)
    g = g_full[n - 1: 2 * n - 1].copy()
    g = np.where(np.isfinite(g), g, 0.0)

    # Center for simplex constraint
    g_centered = g - float(np.dot(a, g))

    # Adaptive effective step size
    max_gc = float(np.max(np.abs(g_centered)))
    if max_gc > 1e-15:
        eta_eff = min(eta, 0.9 / max_gc)
    else:
        eta_eff = eta

    # Multiplicative update
    log_u = -eta_eff * g_centered
    log_u = np.clip(log_u, -40.0, 40.0)
    a_new = a * np.exp(log_u)
    s = float(a_new.sum())
    if not np.isfinite(s) or s <= 0.0:
        return a, float('inf'), c
    a_new = a_new / s
    return a_new, 2.0 * n * max_c, c


def _try_altscores(a, n, N_conv, N_grad, rng):
    """Try a few alternative perturbations, return best (a, score)."""
    best_a = a.copy()
    best_sc, _ = _fft_score(a, N_conv)
    for sig in (0.05, 0.1, 0.2):
        log_a = np.log(np.maximum(a, 1e-300))
        log_a = log_a + rng.normal(0.0, sig, size=n)
        cand = np.exp(log_a - log_a.max())
        cand = cand / cand.sum()
        sc, _ = _fft_score(cand, N_conv)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    return best_a, best_sc

def _try_length(a_parent, target_n, budget, seed):

    n_parent = len(a_parent)
    if target_n == n_parent:
        return a_parent, _fft_score(a_parent, _next_pow2(2 * n_parent - 1))[0]

    if target_n < n_parent:
        # Truncate: take a chunk and renormalize

    else:

        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, target_n)
        a = interp(x_new, x_old, a_parent)
        a = _normalize(a)
