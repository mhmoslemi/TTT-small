class LSEObjective:
    def __init__(self, n, tau, reg):
        self.n = int(n)
        self.tau = float(tau)
        self.reg = float(reg)
        self.last_z = None
        self.last_val = None
        self.last_grad = None
    def _compute(self, z):
        z = np.asarray(z, dtype=np.float64)
        if z.size != self.n: z = z.reshape(self.n)
        if not np.all(np.isfinite(z)):
            z = np.where(np.isfinite(z), z, 0.0)
        zm = z - float(np.max(z))
        exp_z = np.exp(zm)
        s = float(np.sum(exp_z))
        if not np.isfinite(s) or s <= 0:
            return None
        a = (float(self.n) / s) * exp_z
        c = fast_convolution(a)
        cmax = float(np.max(c))
        x = (c - cmax) / self.tau
        ex = np.exp(x)
        se = float(np.sum(ex))
        if not np.isfinite(se) or se <= 0:
            return None
        lse = cmax + self.tau * float(np.log(se))
        val = (2.0 * lse / float(self.n)) + self.reg * float(np.sum(z*z))
        weights = ex / se
        a_rev = a[::-1]
        L = 2*self.n-1
        conv_len = self.n + L - 1  # 3n-2
        N = 1 << (conv_len - 1).bit_length()
        fa = np.fft.rfft(a_rev, N)
        fw = np.fft.rfft(weights, N)
        conv = np.fft.irfft(fa * fw, N)
        grad_a = 2.0 * conv[self.n-1:self.n-1+self.n]
        if grad_a.size != self.n: grad_a = np.resize(grad_a, self.n)
        mean_grad = float(np.dot(a, grad_a)) / float(self.n)
        gz = a * (grad_a - mean_grad)
        gz = gz - float(np.mean(gz))
        gz += 2.0 * self.reg * z
        gz = gz - float(np.mean(gz))
        if not np.all(np.isfinite(gz)): gz = np.zeros(self.n)
        return val, gz, a
    def fun(self,z):
        if self.last_z is not None and len(z)==self.n and np.array_equal(np.asarray(z), self.last_z):
            return self.last_val
        res = self._compute(z)
        if res is None: return float('inf')
        self.last_z = np.asarray(z).copy()
        self.last_val = float(res[0])
        self.last_grad = res[1]
        return self.last_val
    def jac(self,z):
        if self.last_z is not None and len(z)==self.n and np.array_equal(np.asarray(z), self.last_z):
            return self.last_grad
        res = self._compute(z)
        if res is None: return np.zeros(self.n)
        self.last_z = np.asarray(z).copy()
        self.last_val = float(res[0])
        self.last_grad = res[1]
        return self.last_grad

def normalize_fast(a, target):
    """Fast non-negative normalization to a fixed sum."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    if a.size == 0:
        return a

    s = float(np.sum(a))
    if s <= 1e-15:
        a.fill(float(target) / float(a.size))
        return a

    a *= float(target) / s
    np.maximum(a, 0.0, out=a)
    return a


def normalize_to_sum(a, target, max_value=1000.0):
    """Robust non-negative normalization to a fixed sum with clipping."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, max_value)

    if float(a.sum()) <= 1e-15:
        a.fill(float(target) / float(a.size))
        return a

    for _ in range(6):
        s = float(a.sum())
        if s <= 1e-15:
            a.fill(float(target) / float(a.size))
            break

        if s > target:
            a *= target / s
            a = np.clip(a, 0.0, max_value)
        else:
            deficit = target - s
            if deficit <= 0:
                break
            room = max_value - a
            total_room = float(np.sum(room))
            if total_room <= 1e-15:
                a.fill(max_value)
                s = float(a.sum())
                if s > target:
                    a *= target / s
                break
            a += room * (deficit / total_room)
            a = np.clip(a, 0.0, max_value)

        if abs(float(a.sum()) - target) <= 1e-7 * max(1.0, abs(target)):
            break

    return a


def fast_convolution(a):
    """FFT-based linear convolution a * a, truncated to length 2*n - 1."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)

    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def fast_score(a):
    """Fast analogue of evaluate_sequence(), with clipping to [0, 1000]."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")

    c = fast_convolution(a)
    if c.size == 0:
        return float("inf")

    return float(2.0 * n * float(np.max(c)) / (s * s))


def upsample_sequence(seq, n, method="linear"):
    """Resample a sequence to length n."""
    seq = np.asarray(seq, dtype=np.float64)
    n = int(n)
    m = int(seq.size)

    if m == 0:
        return normalize_to_sum(np.ones(n, dtype=np.float64), float(n))
    if m == n:
        return normalize_to_sum(seq, float(n))

    if m == 1:
        out = np.full(n, float(seq[0]), dtype=np.float64)
    elif method == "nearest":
        idx = np.floor(np.linspace(0.0, float(m), n, endpoint=False)).astype(int)
        idx = np.clip(idx, 0, m - 1)
        out = seq[idx].astype(np.float64)
    else:
        old = np.linspace(0.0, 1.0, m)
        new = np.linspace(0.0, 1.0, n)
        out = np.interp(new, old, seq).astype(np.float64)

    return normalize_to_sum(out, float(n))

def make_two_block(n, frac, ratio):
    n = int(n)
    m = int(round(frac * n))
    m = max(1, min(n - 1, m))

    denom = m * float(ratio) + float(n - m)
    if denom <= 0.0:
        b = 1.0
    else:
        b = float(n) / denom
    a = b * np.ones(n, dtype=np.float64)
    a[:m] = b * float(ratio)
    return normalize_to_sum(a, float(n))


def make_harmonic_sequence(n, rng, max_freq=8, max_amp=2.5):
    n = int(n)
    if n <= 1:
        return normalize_to_sum(np.ones(n, dtype=np.float64), float(n))

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)

    for m in range(1, max_freq + 1):
        amp = rng.uniform(-max_amp, max_amp) / (float(m) ** 0.6)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)

    amp = rng.uniform(-max_amp, max_amp)
    z += amp * np.cos(np.pi * (2.0 * t - 1.0))

    z = z - float(np.max(z))
    a = np.exp(z) + 1e-8
    return normalize_to_sum(a, float(n))


def make_block_sequence(n, rng, num_blocks=16):
    n = int(n)
    if n <= 1:
        return normalize_to_sum(np.ones(n, dtype=np.float64), float(n))

    num_blocks = int(max(2, min(num_blocks, n // 2)))
    if num_blocks <= 1:
        return normalize_to_sum(np.ones(n, dtype=np.float64), float(n))

    edges = np.sort(rng.integers(1, n, num_blocks - 1))
    edges = np.concatenate(([0], edges, [n]))
    lengths = np.diff(edges)

    if np.any(lengths <= 0):
        edges = np.linspace(0, n, num_blocks + 1).astype(int)
        edges[0] = 0
        edges[-1] = n
        lengths = np.diff(edges)
        if np.any(lengths <= 0):
            return normalize_to_sum(np.ones(n, dtype=np.float64), float(n))

    vals = rng.lognormal(mean=0.0, sigma=1.0, size=num_blocks)
    if rng.random() < 0.20:
        vals[int(rng.integers(0, num_blocks))] = 0.0

    a = np.repeat(vals, lengths)
    return normalize_to_sum(a, float(n))


def make_thue_morse(n, amplitude=0.5):
    n = int(n)
    seq = np.array([0.0], dtype=np.float64)
    while seq.size < n:
        seq = np.concatenate((seq, 1.0 - seq))
    seq = seq[:n]
    a = 1.0 + amplitude * (2.0 * seq - 1.0)
    return normalize_to_sum(a, float(n))


def smooth_perturbation(a, rng, strength=0.4):
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    target = float(n)
    if n <= 1:
        return normalize_fast(a, target)

    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)

    for m in range(1, 5):
        amp = strength / (0.5 + 0.5 * m)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * m * t + ph)

    a *= np.exp(z)

    if n > 6 and rng.random() < 0.55:
        lo = max(1, n // 500)
        hi = max(lo + 1, n // 5 + 1)
        width = int(rng.integers(lo, hi + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        a[start:stop] *= np.exp(rng.normal(0.0, strength, stop - start))

    if rng.random() < 0.18:
        idx = int(rng.integers(0, n))
        a[idx] *= rng.uniform(0.25, 2.5)

    return normalize_fast(a, target)


def add_start(starts, a, n):

        a = np.asarray(a, dtype=np.float64)

        a = normalize_to_sum(a, float(n))
        sc = fast_score(a)
        if np.isfinite(sc):

    except Exception:
        pass

def topk_indices(c, K):
    L = int(c.size)
    K = int(K)
    if K <= 1:
        return np.array([int(np.argmax(c))], dtype=int)
    if K >= L:
        return np.arange(L, dtype=int)
    idx = np.argpartition(c, -K)[-K:]
    return idx.astype(int, copy=False)


def weighted_top_gradient(a, c, K, tau, uniform=False):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = int(a.size)
    L = int(c.size)

    if K <= 1:
        idx = topk_indices(c, 1)
        k = int(idx[0])
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s0 = k - (stop - 1)
        s1 = k - start + 1
        seg = a[s0:s1][::-1]
        return 2.0 * seg

    idx = topk_indices(c, K)
    d = np.zeros(n, dtype=np.float64)
    vals = c[idx]

    if uniform or tau <= 0.0:
        w = 1.0 / float(idx.size)
        for k in idx:
            k = int(k)
            start = max(0, k - (n - 1))
            stop = min(n, k + 1)
            s0 = k - (stop - 1)
            s1 = k - start + 1
            seg = a[s0:s1][::-1]
            d[start:stop] += w * seg
        return 2.0 * d

    vmax = float(np.max(vals))
    x = (vals - vmax) / float(tau)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        # fallback: active set is just the largest coefficient
        idx = topk_indices(c, 1)
        k = int(idx[0])
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s0 = k - (stop - 1)
        s1 = k - start + 1
        return 2.0 * a[s0:s1][::-1]

    weights = ex / s

    for pos, k in enumerate(idx):
        w = float(weights[pos])
        if w < 1e-13:
            continue
        k = int(k)
        start = max(0, k - (n - 1))
        stop = min(n, k + 1)
        s0 = k - (stop - 1)
        s1 = k - start + 1
        seg = a[s0:s1][::-1]
        d[start:stop] += w * seg

    return 2.0 * d


def refine_simple(a, iterations=5):
    """Cheap deterministic improvements: symmetrization, small-value zeroing, spike capping."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    target = float(n)
    if n <= 1:
        return normalize_to_sum(a, target), fast_score(a)

    best = normalize_to_sum(a, target)
    best_s = fast_score(best)

    for _ in range(max(1, int(iterations))):
        improved = False
        mean = float(np.mean(best))

        cand = 0.5 * (best + best[::-1])
        cand = normalize_to_sum(cand, target)
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best = cand
            best_s = sc
            improved = True

        if mean > 0.0:
            for frac in (0.002, 0.005, 0.01, 0.02, 0.04):
                cand = best.copy()
                cand[cand < frac * mean] = 0.0
                cand = normalize_to_sum(cand, target)
                sc = fast_score(cand)
                if np.isfinite(sc) and sc < best_s - 1e-12:
                    best = cand
                    best_s = sc
                    improved = True

            cand = np.minimum(best, 5.0 * mean)
            cand = normalize_to_sum(cand, target)
            sc = fast_score(cand)
            if np.isfinite(sc) and sc < best_s - 1e-12:
                best = cand
                best_s = sc
                improved = True

        if not improved:
            break

    return best, best_s


def local_search(a0, budget_s, seed=0, label="", alt_starts=None):
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), float(len(a0)))
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64), 2.0

    target = float(n)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))

    c = fast_convolution(a)
    s = float(a.sum())
    score = float(2.0 * n * float(np.max(c)) / (s * s)) if s > 0 else float("inf")

    best = a.copy()
    best_score = score
    loga = np.log(np.maximum(a, 1e-12))
    mom = np.zeros(n, dtype=np.float64)

    eta = 0.30
    beta = 0.40
    noise = 0.025
    accept_rel = 0.03
    sa_temp = 0.006

    stagnant = 0
    it = 0

    K_init = max(32, min(256, n // 8))
    K_min = max(4, n // 256)
    restart_threshold = max(1500, min(8000, 4 * n))

    alt_list = []
    if alt_starts is not None:
        for x in alt_starts:
            if x is None:
                continue
            arr = np.asarray(x, dtype=np.float64)
            if arr.size > 0:
                alt_list.append(normalize_to_sum(arr, target))

    while time.monotonic() < deadline:
        it += 1

        if not np.all(np.isfinite(a)) or float(a.sum()) <= 0.0:
            a = best.copy()
            c = fast_convolution(a)
            score = best_score
            loga = np.log(np.maximum(a, 1e-12))
            mom = np.zeros(n, dtype=np.float64)

        L = 2 * n - 1
        prog = min(1.0, float(it) / 5000.0)
        K = int(round(K_init - (K_init - K_min) * prog))
        K = max(1, min(K, L))

        cmax = float(np.max(c))
        cmean = float(np.sum(c)) / float(L)
        spread = max(1e-6, cmax - cmean)

        if stagnant > 250:
            K = min(L, int(K * 1.4) + 1)
            tau = max(0.03 * spread, (0.22 + 0.25 * min(1.0, float(stagnant) / 1000.0)) * spread)
        else:
            tau = max(0.02 * spread, (0.25 - 0.22 * prog) * spread)

        grad = weighted_top_gradient(a, c, K, tau, uniform=False)
        if not np.all(np.isfinite(grad)):
            break

        g = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(g)))
        if scale < 1e-14:
            stagnant += 1
            if stagnant > 200:
                stagnant = restart_threshold
            continue

        g /= scale
        mom = beta * mom + (1.0 - beta) * g

        step = -eta * mom
        if noise > 1e-7:
            step = step + rng.normal(0.0, noise, n)
        step = np.clip(step, -3.0, 3.0)

        loga = loga + step
        loga = np.clip(loga, -25.0, 25.0)
        raw = np.exp(loga)
        s_raw = float(np.sum(raw))
        if not np.isfinite(s_raw) or s_raw <= 0.0:
            raw = best.copy()
            s_raw = float(np.sum(raw))

        cand = raw * (target / s_raw)
        if float(np.max(cand)) > 100.0:
            cand = np.minimum(cand, 100.0)
            cand = normalize_fast(cand, target)

        c2 = fast_convolution(cand)
        s2 = float(cand.sum())
        score2 = float(2.0 * n * float(np.max(c2)) / (s2 * s2)) if s2 > 0 else float("inf")

        if not np.isfinite(score2):
            eta = max(0.02, eta * 0.7)
            mom *= 0.5
            stagnant += 1
            continue

        improved_best = score2 < best_score
        if improved_best:
            best = cand.copy()
            best_score = score2
            stagnant = 0

        accept = False
        if score2 <= score * (1.0 + accept_rel) + 1e-12:
            accept = True
        elif score2 < score:
            accept = True
        elif score2 < score + 0.05:
            diff = score2 - score
            arg = -diff / max(1e-8, sa_temp)
            prob = math.exp(arg) if arg > -50.0 else 0.0
            accept = float(rng.random()) < prob

        if accept:
            a = cand
            c = c2
            score = score2
            loga = np.log(np.maximum(a, 1e-12))

            if improved_best:
                eta = min(0.9, eta * 1.05)
                accept_rel = max(0.005, accept_rel * 0.98)
                noise = max(0.005, noise * 0.98)
                sa_temp = max(0.0008, sa_temp * 0.98)
            else:
                eta = max(0.05, eta * 0.97)
                stagnant += 1
        else:
            eta = max(0.02, eta * 0.7)
            mom *= 0.5
            stagnant += 1

        if label and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  [{label}] it={it} elapsed={elapsed:.1f}s current={score:.6f} best={best_score:.6f}")

        if stagnant > restart_threshold:
            strength = min(1.0, 0.25 + 0.06 * float(stagnant // 1000))

            if alt_list and rng.random() < 0.35:
                alt = alt_list[int(rng.integers(0, len(alt_list)))]
                mix = 0.5 + 0.3 * float(rng.random())
                base = mix * best + (1.0 - mix) * alt
            else:
                base = smooth_perturbation(best, rng, strength)

            base = normalize_to_sum(base, target)
            cb = fast_convolution(base)
            sb = float(base.sum())
            scoreb = float(2.0 * n * float(np.max(cb)) / (sb * sb)) if sb > 0 else float("inf")

            if not np.isfinite(scoreb) or scoreb > best_score * 1.20:
                base = smooth_perturbation(best, rng, 0.15)
                base = normalize_to_sum(base, target)
                cb = fast_convolution(base)
                sb = float(base.sum())
                scoreb = float(2.0 * n * float(np.max(cb)) / (sb * sb)) if sb > 0 else float("inf")

            if not np.isfinite(scoreb) or scoreb > best_score * 1.20:
                base = best.copy()
                cb = fast_convolution(best)
                scoreb = best_score

            a = base
            c = cb
            score = scoreb
            loga = np.log(np.maximum(a, 1e-12))
            mom = np.zeros(n, dtype=np.float64)
            eta = 0.35
            accept_rel = 0.05
            noise = 0.05
            sa_temp = 0.01
            stagnant = 0

    best, best_score = refine_simple(best, 3)
    return best, best_score


# ----------------------------------------------------------------------------
# Smooth log-sum-exp polish using L-BFGS-B
# ----------------------------------------------------------------------------

class LSEObjective:
    def __init__(self, n, tau, reg):
        self.n = int(n)
        self.tau = float(tau)
        self.reg = float(reg)
        self.last_z = None
        self.last_val = None
        self.last_grad = None

    def _compute(self, z):
        z = np.asarray(z, dtype=np.float64)
        if z.size != self.n:
            z = z.reshape((self.n,))
        if not np.all(np.isfinite(z)):
            z = np.where(np.isfinite(z), z, 0.0)

        zm = z - float(np.max(z))
        exp_z = np.exp(zm)
        s = float(np.sum(exp_z))
        if not np.isfinite(s) or s <= 0.0:
            return None

        a = (float(self.n) / s) * exp_z
        c = fast_convolution(a)
        cmax = float(np.max(c))
        x = (c - cmax) / self.tau
        ex = np.exp(x)
        se = float(np.sum(ex))
        if not np.isfinite(se) or se <= 0.0:
            return None

        lse = cmax + self.tau * float(np.log(se))
        val = (2.0 * lse / float(self.n)) + self.reg * float(np.sum(z * z))

        limit = 100.0
        over = np.maximum(a - limit, 0.0)
        grad_a = None

        try:
            weights = ex / se
            a_rev = a[::-1]
            L = 2 * self.n - 1
            conv_len = self.n + L - 1
            N = 1 << (conv_len - 1).bit_length()
            fa = np.fft.rfft(a_rev, N)
            fw = np.fft.rfft(weights, N)
            conv = np.fft.irfft(fa * fw, N)
            grad_a = 2.0 * conv[self.n - 1:2 * self.n - 1]
            if grad_a.size != self.n:
                grad_a = np.resize(grad_a, self.n)
        except Exception:
            grad_a = np.zeros(self.n, dtype=np.float64)

        if not np.all(np.isfinite(grad_a)):
            grad_a = np.zeros(self.n, dtype=np.float64)

        mean_grad = float(np.dot(a, grad_a)) / float(self.n)
        gz = a * (grad_a - mean_grad)
        gz = gz - float(np.mean(gz))

        if np.any(over):
            pen_a = 2.0e-3 * over
            val += 1.0e-3 * float(np.sum(over * over))
            mz = float(np.dot(a, pen_a)) / float(self.n)
            gz += a * (pen_a - mz)

        gz += 2.0 * self.reg * z
        gz = gz - float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            gz = np.zeros(self.n, dtype=np.float64)

        return float(val), gz


def _lse_fun(z):
    if _LSE_OBJ is None:
        return float("inf")
    return _LSE_OBJ.fun(z)


def polish_lse(a0, budget_s, seed=0, label=""):
    global _LSE_OBJ

    try:
        from scipy.optimize import minimize
    except Exception:
        return a0, fast_score(a0)

    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), float(len(a0)))
    n = int(a.size)
    if n <= 1:
        return a, fast_score(a)

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))

    a0forbest = a.copy()
    best_score0 = fast_score(a0forbest)

    ap = np.maximum(a, 1e-6)
    z = np.log(ap)
    z = z - float(np.mean(z))
    z = np.clip(z, -8.0, 8.0)

    reg = 1e-8
    factors = (0.25, 0.13, 0.07, 0.035, 0.018, 0.009, 0.0045)
    bounds = [(-9.0, 9.0)] * n

    for phase, fac in enumerate(factors):
        if time.monotonic() >= deadline - 5.0:
            break

        c_best = fast_convolution(a0forbest)
        spread = max(1.0, float(c_best.max() - c_best.mean()))
        tau = max(0.5, fac * spread)

        _LSE_OBJ = LSEObjective(n, tau, reg)
        maxiter = 120
        if n > 1024:
            maxiter = 80

        opts = {
            "maxiter": maxiter,
            "maxfun": max(200, 2 * maxiter),
            "ftol": 1e-10,
            "gtol": 1e-4,
            "maxls": 20,
        }

        try:
            res = minimize(_lse_fun, z, jac=None, method="L-BFGS-B", bounds=bounds, options=opts)
            if hasattr(res, "x") and res.x is not None:
                znew = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(znew)):
                    z = znew - float(np.mean(znew))
                    z = np.clip(z, -9.0, 9.0)
        except Exception:
            pass

        zm = z - float(np.max(z))
        p = np.exp(zm)
        sp = float(np.sum(p))
        if np.isfinite(sp) and sp > 0.0:
            a_new = (float(n) / sp) * p
            a_new = normalize_to_sum(a_new, float(n))
            sc = fast_score(a_new)
            if np.isfinite(sc) and sc < best_score0:
                a0forbest = a_new
                best_score0 = sc

        elapsed = time.monotonic() - t0
        if label:
            print(f"  [{label} LSE] phase={phase} tau={tau:.3f} elapsed={elapsed:.1f}s best={best_score0:.6f}")

    _LSE_OBJ = None
    a0forbest, best_score0 = refine_simple(a0forbest, 3)
    return a0forbest, best_score0

def add_candidate_pool(pool, seq, score):
    try:
        sc = float(score)
    except Exception:
        return
    if not np.isfinite(sc):
        return
    arr = np.asarray(seq, dtype=np.float64).copy()
    if arr.size == 0:
        return
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if float(np.sum(arr)) < 0.01:
        return
    pool.append((sc, arr))
    pool.sort(key=_pair_score)
    if len(pool) > 18:
        pool.pop()


def run_level(n, budget_s, parent_seq, seed=0, label=""):
    n = int(n)
    budget = max(0.0, float(budget_s))
    if budget <= 12.0:
        if parent_seq is not None and np.asarray(parent_seq).size > 0:
            seq = upsample_sequence(parent_seq, n)
        else:
            seq = np.ones(n, dtype=np.float64)
        seq = normalize_to_sum(seq, float(n))
        return seq, fast_score(seq)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget
    print(f"[run_level] n={n} budget={budget:.1f}s label={label}")

    starts = generate_starts(n, rng, parent_seq, random_count=42)
    pool = []

    for sc, sq in starts[:8]:
        add_candidate_pool(pool, sq, sc)

    quick_budget = min(0.35 * budget, 420.0)
    if quick_budget > 30.0 and starts:
        k = max(1, min(4, len(starts), int(quick_budget / 90.0) + 1))
        time_each = quick_budget / float(k)
        for i in range(k):
            if time.monotonic() >= deadline - 40.0:
                break
            sc, sq = starts[i]
            add_candidate_pool(pool, sq, sc)
            remaining_quick = max(0.0, deadline - 40.0 - time.monotonic())
            time_this = max(8.0, min(time_each, remaining_quick))
            alt = [s[1] for s in starts[:max(2, i + 2)]]
            try:
                seq, score = local_search(sq, time_this, seed=seed * 101 + i, label=f"{label}_q{i}", alt_starts=alt)
                add_candidate_pool(pool, seq, score)
            except Exception:
                continue

    pool.sort(key=_pair_score)
    remain = max(0.0, deadline - time.monotonic())

    polish_budget = 0.0
    if remain > 300.0 and n <= 1024:
        polish_budget = min(0.18 * budget, 0.35 * remain, 320.0)

    search_deadline = t0 + (budget - polish_budget)
    if search_deadline > time.monotonic() + 20.0 and pool:
        best_seq = pool[0][1]
        alt = [c[1] for c in pool[:8]]
        time_this = max(0.0, search_deadline - time.monotonic())
        try:
            seq, score = local_search(best_seq, time_this, seed=seed * 777 + 5, label=f"{label}_long", alt_starts=alt)
            add_candidate_pool(pool, seq, score)
        except Exception:
            pass

    pool.sort(key=_pair_score)

    if polish_budget > 20.0 and pool:
        try:
            seq, score = polish_lse(pool[0][1], polish_budget, seed=seed * 31 + 7, label=label)
            add_candidate_pool(pool, seq, score)
        except Exception:
            pass

    pool.sort(key=_pair_score)
    if pool:
        best_seq = pool[0][1]
        best_seq, best_score = refine_simple(best_seq, 6)
        add_candidate_pool(pool, best_seq, best_score)
        pool.sort(key=_pair_score)
        return pool[0][1], pool[0][0]

    seq = normalize_to_sum(np.ones(n, dtype=np.float64), float(n))
    return seq, fast_score(seq)

def sanitize_for_return(seq):
    """Return a valid Python list of non-negative finite floats."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return [1.0] * 1024

    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)

    mx = float(np.max(seq))
    if mx <= 0.0:
        n = int(seq.size)
        return [1.0] * n

    seq = seq / mx
    s = float(np.sum(seq))

    if s < 0.01:
        # Scale up until the evaluator's sum threshold is met, while keeping max <= 1000.
        need = 0.01 / s
        if need <= 1000.0:
            seq *= need
        else:
            n = int(seq.size)
            return [1.0] * n

    mx = float(np.max(seq))
    if mx > 1000.0:
        seq *= 1000.0 / mx

    if float(np.sum(seq)) < 0.01:
        n = int(seq.size)
        return [1.0] * n

    return [float(x) for x in seq]


def validate_candidates(pool, t0, time_budget):
    deadline = t0 + max(0.0, float(time_budget))

    fallback = [1.0] * 512
    try:
        best_exact = float(evaluate_sequence(fallback))
    except Exception:
        best_exact = float("inf")
    best_list = fallback

    if not pool:
        return best_list, best_exact

    pool = list(pool)
    pool.sort(key=_pair_score)

    for i in range(len(pool)):
        if time.monotonic() > deadline - 1.0:
            break
        _, seq = pool[i]
        try:
            lst = sanitize_for_return(seq)
            val = float(evaluate_sequence(lst))
        except Exception:
            val = float("inf")

        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_list = lst
            print(f"[validate] candidate {i}: exact score={val:.6f}")

    return best_list, best_exact

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))

    if budget < 80.0:
        parent_raw = globals().get("height_sequence_1", None)
        if parent_raw is not None:
            try:
                parent = np.asarray(parent_raw, dtype=np.float64)
                parent = np.where(np.isfinite(parent), parent, 0.0)
                parent = np.clip(parent, 0.0, 1000.0)
                if parent.size > 0 and float(np.sum(parent)) > 0.0:
                    return sanitize_for_return(parent)
            except Exception:
                pass
        return [1.0] * 1024

    reserve = max(45.0, min(120.0, 0.05 * budget))
    search_deadline = t0 + max(0.0, budget - reserve)

    parents = []
    parent_raw = globals().get("height_sequence_1", None)
    if parent_raw is not None:
        try:
            p = np.asarray(parent_raw, dtype=np.float64)
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 0 and float(np.sum(p)) > 0.0:
                pnorm = normalize_to_sum(p, float(p.size))
                parents.append(pnorm)
        except Exception:
            pass

    pool = []
    for p in parents:
        add_candidate_pool(pool, p, fast_score(p))

    search_total = max(0.0, budget - reserve)

    if budget < 900.0:
        levels = [(512, 1.0)]
    elif budget < 1800.0:
        levels = [(512, 0.35), (1024, 0.65)]
    else:
        levels = [(512, 0.22), (1024, 0.55), (2048, 0.23)]

    prev_best = parents[0] if parents else None
    current_best_seq = prev_best
    current_best_score = float("inf")

    for idx, (n, w) in enumerate(levels):
        elapsed = time.monotonic() - t0
        if elapsed > budget - reserve - 80.0:
            break

        level_budget = min(w * search_total, search_deadline - time.monotonic() - 80.0)
        if level_budget < 80.0:
            break

        try:
            seq, sc = run_level(n, level_budget, prev_best, seed=int(seed + 1000 * idx + 1), label=f"level_{n}")
        except Exception:
            seq = None
            sc = float("inf")

        if seq is not None and np.isfinite(sc):
            add_candidate_pool(pool, seq, sc)
            if sc < current_best_score:
                current_best_score = sc
                current_best_seq = seq
            prev_best = seq

    # Use any remaining time for direct refinement of the current best length.
    if current_best_seq is not None:
        remaining = search_deadline - time.monotonic()
        if remaining > 60.0:
            try:
                alt = [c[1] for c in pool[:8] if c[1].size == current_best_seq.size]
                seq, sc = local_search(current_best_seq, remaining, seed=int(seed + 90001), label="final_direct", alt_starts=alt)
                add_candidate_pool(pool, seq, sc)
                if sc < current_best_score:
                    current_best_score = sc
                    current_best_seq = seq
            except Exception:
                pass

    if current_best_seq is None and pool:
        current_best_seq = pool[0][1]
        current_best_score = pool[0][0]

    final_seq, final_score = validate_candidates(pool, time.monotonic(), max(10.0, budget - (time.monotonic() - t0) + reserve + 1.0))

    print(f"[final] best exact score={final_score:.6f}, length={len(final_seq)}, elapsed={time.monotonic() - t0:.1f}s")
    return final_seq
