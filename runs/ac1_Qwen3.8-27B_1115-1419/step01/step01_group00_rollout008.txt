LP-driven optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines iterative LP "best response" moves with gradient descent
fine-tuning, adaptive step sizes, and periodic random perturbations.
"""

import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as sco


def _full_conv(a):
    return np.convolve(a, a)


def _obj(a, n):
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10
    return 2.0 * n * float(np.max(_full_conv(a))) / (s * s)


def _project(a, n):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n, dtype=np.float64)
    return a / s * n


def _grad_smoothed_max(a, tau, n):
    """Gradient of log-sum-exp smoothing of max(conv(a,a)), scaled by 1/tau."""
    c = _full_conv(a)
    mx = float(c.max())
    e = np.exp((c - mx) / tau)
    p = e / e.sum()
    # g[i] = (2/tau) * sum_k p[k] * a[k-i] (valid indices)
    corr = np.correlate(p.astype(np.float64), a, mode='full')
    g = (2.0 / tau) * corr[n - 1:2 * n - 1]
    return g


def _build_conv_A(a, n):
    """Build sparse (2n-1) x n matrix A where (A @ g)[k] = conv(a,g)[k]."""
    total_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(total_rows):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        cnt = i_max - i_min + 1
        if cnt > 0:
            idx = np.arange(i_min, i_max + 1)
            rows.extend([k] * cnt)
            cols.extend(idx.tolist())
            vals.extend((a[k - idx]).tolist())
    A = sp.csr_matrix((vals, (rows, cols)), shape=(total_rows, n))
    return A

        if res.success:
            return res.x
    except Exception:
        pass

    return np.ones(n) * M / n

```python
"""
LP-driven + gradient optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key techniques:
  - Iterative LP "best response" using sparse Toeplitz convolution matrix
  - Log-sum-exp smoothed gradient descent with adaptive temperature
  - Hard active-set gradient on the exact max convolution coefficient
  - Adaptive blending, periodic perturbations, and structured restarts
"""

import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as sco


def _full_conv(a):
    return np.convolve(a, a)


def _obj_val(a, n):
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10
    return 2.0 * n * float(np.max(_full_conv(a))) / (s * s)


def _project(a, n):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n, dtype=np.float64)
    return a / s * n


def _grad_lse(a, tau, n):
    """Gradient of (1/tau)*log(sum exp(tau*conv(a,a))) wrt a."""
    c = _full_conv(a)
    tc = tau * c
    mx = float(tc.max())
    e = np.exp(tc - mx)
    p = e / float(e.sum())
    corr = np.correlate(p, a, mode='full')
    return 2.0 * corr[n - 1:2 * n - 1]


def _grad_max(a, k, n):
    """Gradient of conv(a,a)[k] wrt a."""
    g = np.zeros(n, dtype=np.float644)
    js = max(0, k - n + 1)
    je = min(n - 1, k) + 1
    if js < je:
        g[js:je] = 2.0 * a[k - np.arange(js, je)]
    return g


def _toeplitz_conv(A_mat, a, n):
    """Build sparse Toeplitz convolution matrix for sequence a."""
    c = np.zeros(2 * n - 1)
    c[:n] = a
    r = np.zeros(n)
    r[0] = a[0]
    return sp.toeplitz(c, r).tocsr()


def _solve_lp(a, M, n):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0. Return rescaled g (sum=n)."""
    Amat = _toeplitz_conv(a, n)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, 1000.0)] * n
    try:
        res = sco.linprog(c_obj, A_ub=Amat, b_ub=b, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s < 1e-10:
                return np.ones(n) * n / n
            return g / s * n
    except Exception as ex:
        print(f"  LP error: {ex}")
    return np.ones(n)  # fallback
```
