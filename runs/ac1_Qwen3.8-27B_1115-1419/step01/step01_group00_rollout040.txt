```python
import numpy as np

def _simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

Wait, the standard simplex projection for `sum(a) = 1` (not `sum(a) = n`) is:
- Find `theta` such that `sum(max(v - theta, 0)) = 1`
- The algorithm: sort `v` in decreasing order, find the largest `rho` such that `u[0] + ... + u[rho] - rho * theta > 1`, where `theta = (u[0] + ... + u[rho] - 1) / (rho + 1)`.

Let me verify: if `v = [2, 1, 0]`, then `u = [2, 1, 0]`, `cssv = [2, 3, 3]`.
- `rho = 0`: `u[0] * 1 = 2 > cssv[0] - 1 = 1`? Yes.
- `rho = 1`: `u[1] * 2 = 2 > cssv[1] - 1 = 2`? No (equal, not greater).
- `rho = 2`: `u[2] * 3 = 0 > cssv[2] - 1 = 2`? No.

So `rho = 0`, `theta = (2 - 1) / 1 = 1`. Result: `max([2-1, 1-1, 0-1], 0) = [1, 0, 0]`. Sum = 1. Correct!

Now, for the Lipschitz constant `L`: The gradient is `2 * correlate(p, a)`. The Hessian involves second derivatives. Since `p` depends on `a` (through the softmax), the Hessian has additional terms. This makes it hard to bound `L` analytically.

A practical approach: use backtracking. At each iteration, check if the descent condition is satisfied:
`f(a_new) <= f(y) + grad(y)·(a_new - y) + L/2 * ||a_new - y||^2`

If not, increase `L` and retry.

```python
Convex optimization approach using FISTA with log-sum-exp smoothing and annealing.
Exploits the convexity of max(conv(a,a)) on the non-negative simplex.
Uses Nesterov acceleration, adaptive step size, and multi-scale search over n.

import numpy as np
import time


def _simplex_project(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_smooth(a, tau):
    """Compute gradient of tau * log(sum exp((a*a)[k]/tau))."""
    c = np.convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / tau)
    p = exp_c / exp_c.sum()
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad


def _fista_smooth(a, tau, n_iter, L_init=None):
    """Run FISTA to minimize log-sum-exp smoothing on simplex."""
    n = len(a)
    if L_init is None:
        L = 4.0 / tau + 4.0
    else:
        L = L_init
    
    a_prev = a.copy()
    y = a.copy()
    t = 1.0
    
    for k in range(n_iter):
        grad = _compute_grad_smooth(y, tau)
        a_new = _simplex_project(y - (1.0 / L) * grad)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        # Soft-project y to keep it in a reasonable range
        if np.any(y < -1.0):
            y = _simplex_project(y)
        a_prev = a_new
        a = a_new
        t = t_new
        
        # Occasionally increase L for safety
        if k > 0 and k % 10000 == 0:
            L *= 1.01
    
    return a


def _hard_subgrad(a, n_iter, eta_init=0.1):
    """Subgradient method on the exact max (no smoothing)."""
    n = len(a)
    a_avg = np.zeros(n)
    eta = eta_init
    t = 1.0
    a_prev = a.copy()
    y = a.copy()
    
    for k in range(n_iter):
        c = np.convolve(a, a)
        k_max = int(np.argmax(c))
        # Subgradient: 2 * a[k_max - m] for valid m
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        for j in range(js, je + 1):
            m = k_max - j
            grad[m] = 2.0 * a[j]
        
        # FISTA-style update
        a_new = _simplex_project(y - (1.0 / eta) * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        a_avg += a_new
        a_prev = a_new
        a = a_new
        t = t_new
        eta *= 0.999
    
    a_avg /= n_iter
    return a_avg


def _objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf')
    mc = float(np.max(np.convolve(a, a)))
    return 2.0 * n * mc / (s * s)


def _fista_anneal(a, n, deadline):
    """Full annealing schedule for FISTA."""
    # Initial tau
    c = np.convolve(a, a)
    tau = float(np.max(c))
    
    while time.monotonic() < deadline:
        # Run FISTA at current tau
        n_iter = max(100, min(5000, int((deadline - time.monotonic()) * 50)))
        a = _fista_smooth(a, tau, n_iter)
        
        if time.monotonic() >= deadline:
            break
        
        # Check convergence of smoothed problem
        c = np.convolve(a, a)
        mc = float(np.max(c))
        
        if tau < 0.005 * mc:
            break
        
        tau *= 0.85
    
    return a
```

What about a "bimodal" distribution? `a[i] = c` for `i` in some set `S` and 0 otherwise?

If `S = {0, 1, ..., n-1}` (all positions), we get the uniform case with objective 2.

If `S` is a "Sidon set" or "Bose-Chowla set" (where all pairwise sums are distinct), then each `b[k]` is either 0 or `c^2` (at most one pair sums to `k`). The max would be `c^2 = 1/|S|^2`, and the objective would be `2n/|S|^2`. For this to be less than 2, we need `|S|^2 > n`, i.e., `|S| > sqrt(n)`.

But for non-negative sequences, a Sidon set means that each sum `i+j` for `i, j in S` appears at most once. The size of a Sidon set in `{0, ..., n-1}` is at most about `sqrt(n)`. So `|S| <= sqrt(n) + O(1)`, giving `|S|^2 <= n + O(sqrt(n))`. The objective would be `2n/|S|^2 >= 2n/(n + O(sqrt(n))) ≈ 2 - O(1/sqrt(n))`.

So with a Sidon set, we can get close to 2 but not below it. Not helpful.

What about a "dense" set with a specific structure? For example, the support of `a` could be all of `{0, ..., n-1}` but with non-uniform weights.

`sum_i a[i]*a[k-i] <= T` for all `k`

For `k = 1`: `2*a[0]*a[1] <= T`

So `a[0] <= sqrt(T)`, `a[1] <= sqrt(T/2) + ...`, etc.

In the worst case (uniform), `T = 1/n`. Can we do better?

If we could make the convolution perfectly flat, `T = 1/(2n-1)`. The ratio is `n/(2n-1) ≈ 1/2`. So the theoretical best is about half of the uniform value, giving an objective of about 1 instead of 2.

"""
Convex optimization via accelerated FISTA with log-sum-exp annealing.
Exploits the convexity of max(self-convolution) over the non-negative simplex.
Uses FFT-based convolutions, Nesterov momentum, and multi-scale search.
"""

import numpy as np
import time


def _fft_conv(a, b):
    """FFT-based full linear convolution."""
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def _self_conv(a):
    """Self-convolution a * a using FFT."""
    return _fft_conv(a, a)


def _cross_corr_valid(p, a):
    """out[m] = sum_j p[m+j]*a[j] for m=0..len(a)-1, via FFT."""
    n = len(a)
    conv = _fft_conv(p, a[::-1])
    return conv[n - 1: 2 * n - 1]


def _simplex_project(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smooth(a, tau):
    """Gradient of tau*log(sum exp(conv(a,a)/tau)) w.r.t. a."""
    c = _self_conv(a)
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / tau)
    p = exp_c / exp_c.sum()
    return 2.0 * _cross_corr_valid(p, a)


def _grad_hard(a, k_star):
    """Exact subgradient of max conv at index k_star."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for j in range(js, je + 1):
        grad[k_star - j] += 2.0 * a[j]
    return grad


def _objective_normalized(a):
    """2*n*max(conv(a,a)) when sum(a)=1."""
    n = len(a)
    mc = float(np.max(_self_conv(a)))
    return 2.0 * n * mc


def _fista_phase(a, tau, n_iter, L=None):
    """One phase of FISTA at fixed tau."""
    n = len(a)
    if L is None:
        L = max(4.0 / tau + 4.0, 2.0 * n)
    
    a_prev = a.copy()
    y = a.copy()
    t = 1.0
    
    for k in range(n_iter):
        grad = _grad_smooth(y, tau)
        a_new = _simplex_project(y - (1.0 / L) * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = a_new + momentum * (a_new - a_prev)
        # Keep y in bounded range for numerical stability
        if y.min() < -n * 0.1:
            y = _simplex_project(y)
        a_prev = a_new
        a = a_new
        t = t_new
    
    return a


def _anneal_fista(a, deadline):
    """Full annealing schedule."""
    n = len(a)
    c = _self_conv(a)
    tau = float(np.max(c))
    
    obj = _objective_normalized(a)
    best_a = a.copy()
    best_obj = obj
    
    while time.monotonic() < deadline:
        phase_time = min(10.0, (deadline - time.monotonic()) * 0.3)
        n_iter = max(500, int(phase_time * 2000))  # ~2000 iters/sec target
        
        a = _fista_phase(a, tau, n_iter)
        
        obj = _objective_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        mc = float(np.max(_self_conv(a)))
        if tau < 0.005 * mc:
            break
        
        tau *= 0.80
    
    # Try to improve with a very last phase at smaller tau
    if time.monotonic() < deadline - 2:
        tau = max(tau * 0.5, 1e-6)
        n_iter = max(1000, int((deadline - time.monotonic()) * 10000))
        a = _fista_phase(a, tau, n_iter)
        obj = _objective_normalized(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj
