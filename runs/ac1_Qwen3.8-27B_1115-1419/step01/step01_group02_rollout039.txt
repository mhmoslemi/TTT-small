                                   lambda x: _obj_grad_lse(x, tau),
                                   maxiter=100, ftol=1e-12, gtol=1e-8)
        if sc < best_score:
            best_score = sc
            best_a = current.copy()
            print(f"    LSE tau={tau}: {sc:.6f}")
        current = best_a.copy()

    # Stage 2: Top-k objectives (increasingly sharp)
    for k in [5, 3, 2, 1]:
        if time.monotonic() > deadline - 20:
            break
        stage_deadline = min(time.monotonic() + 90, deadline - 20)
        current, sc = _run_lbfgsb(current, stage_deadline,
                                   lambda x, kk=k: _obj_grad_toppk(x, kk),
                                   maxiter=150, ftol=1e-13, gtol=1e-9)
        if sc < best_score:
            best_score = sc
            best_a = current.copy()
            print(f"    Top-{k}: {sc:.6f}")
        current = best_a.copy()

    # Stage 3: Direct (hard max)
    if time.monotonic() < deadline - 10:
        stage_deadline = deadline - 5
        current, sc = _run_lbfgsb(current, stage_deadline,
                                   _obj_grad_direct,
                                   maxiter=200, ftol=1e-13, gtol=1e-10)
        if sc < best_score:
            best_score = sc
            best_a = current.copy()
            print(f"    Direct: {sc:.6f}")

    return best_a, best_score


def _perturb_restart(a, rng, magnitude=0.05):
    """Create a perturbed version of a for restart."""
    n = len(a)
    target_sum = float(np.sum(a))
    perturbation = magnitude * float(np.max(a) + 1e-10)
    a_new = a + perturbation * rng.standard_normal(n)
    return _normalize_to_sum(np.maximum(a_new, 0), target_sum)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with adaptive
    smoothing (log-sum-exp, top-k), multiple restarts, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Get parent sequence
    n = 5582
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            n = parent.size
    except (NameError, Exception):
        parent = np.ones(n)

    target_sum = np.sqrt(2.0 * n)
    parent_norm = _normalize_to_sum(parent, target_sum)

    # Evaluate parent
    best = parent_norm.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Parent score: {best_score:.6f} (n={n})")

    # Phase 1: Refinement from parent
    print(f"\nPhase 1: Refinement from parent")
    phase1_deadline = min(time.monotonic() + 400, total_deadline - 100)
    a1, s1 = _run_refinement_cycle(parent_norm, phase1_deadline)
    if s1 < best_score:
        actual = float(evaluate_sequence(a1.tolist()))
        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best = a1.copy()
            print(f"Phase 1 result: {best_score:.6f}")
    print(f"After Phase 1: best = {best_score:.6f}")

    # Phase 2: Multiple restarts from best
    print(f"\nPhase 2: Multiple restarts")
    n_restarts = 6
    restart_time = 180.0
    for r in range(n_restarts):
        if time.monotonic() > total_deadline - 60:
            break
        # Vary perturbation magnitude
        mag = 0.03 + 0.02 * r
        a_pert = _perturb_restart(best, rng, magnitude=mag)
        restart_deadline = min(time.monotonic() + restart_time, total_deadline - 60)
        a_r, s_r = _run_refinement_cycle(a_pert, restart_deadline)
        if s_r < best_score:
            actual = float(evaluate_sequence(a_r.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = a_r.copy()
                print(f"  Restart {r}: improved to {best_score:.6f}")
        else:
            print(f"  Restart {r}: {s_r:.6f} (no improvement)")

    # Phase 3: Try different initializations
    print(f"\nPhase 3: Diverse initializations")
    k = np.arange(n)
    inits = [
        ("symmetric_u", 1.0 + 0.3 * np.cos(np.pi * k / n)),
        ("edges_high", 0.5 + 0.8 * np.abs(2 * k / n - 1)),
        ("center_low", 1.5 - 0.5 * np.exp(-((k - n/2) / (n/4))**2)),
        ("duo_peak", 0.5 + 1.0 * np.exp(-((k - n*0.25) / (n/8))**2) + 1.0 * np.exp(-((k - n*0.75) / (n/8))**2)),
        ("cos_3", 1.0 + 0.4 * np.cos(2 * np.pi * 3 * k / n)),
        ("cos_7", 1.0 + 0.3 * np.cos(2 * np.pi * 7 * k / n)),
        ("spline", 0.5 + 0.5 * np.sin(np.pi * k / n) ** 2),
    ]

    init_time = 120.0
    for name, init in inits:
        if time.monotonic() > total_deadline - 60:
            break
        a_init = _normalize_to_sum(np.maximum(init, 0), target_sum)
        init_deadline = min(time.monotonic() + init_time, total_deadline - 60)
        a_i, s_i = _run_refinement_cycle(a_init, init_deadline)
        if s_i < best_score:
            actual = float(evaluate_sequence(a_i.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = a_i.copy()
                print(f"  {name}: improved to {best_score:.6f}")
        else:
            print(f"  {name}: {s_i:.6f}")

    # Phase 4: Extended fine-tuning from best
    print(f"\nPhase 4: Fine-tuning, best = {best_score:.6f}")
    if time.monotonic() < total_deadline - 20:
        # Multiple passes with decreasing perturbation
        for pass_mag in [0.01, 0.005, 0.002]:
            if time.monotonic() > total_deadline - 15:
                break
            a_ft = _perturb_restart(best, rng, magnitude=pass_mag)
            ft_deadline = min(time.monotonic() + 100, total_deadline - 15)
            a_f, s_f = _run_refinement_cycle(a_ft, ft_deadline)
            if s_f < best_score:
                actual = float(evaluate_sequence(a_f.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best = a_f.copy()
                    print(f"  Fine-tune (mag={pass_mag}): {best_score:.6f}")

    # Phase 5: Final direct optimization
    print(f"\nPhase 5: Final direct L-BFGS-B")
    if time.monotonic() < total_deadline - 5:
        a_final, s_final = _run_lbfgsb(best, total_deadline - 3,
                                        _obj_grad_direct,
                                        maxiter=300, ftol=1e-14, gtol=1e-11)
        if s_final < best_score:
            actual = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = a_final.copy()
                print(f"  Final: {best_score:.6f}")

    # Final evaluation
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"\nFinal score: {final_score:.6f}")
    print(f"Best tracked: {best_score:.6f}")

    if final_score < best_score:
        best_score = final_score

    return [float(x) for x in best]
