```python
    Given fixed a, find g >= 0 maximizing sum(g) subject to (a*g)[k] <= M for all k.
    Returns the optimal g (or None if infeasible).

    # This is a banded matrix (bandwidth n)

    if n > 2000:

        from scipy.sparse import csr_matrix, diags

        # This is a lower-Hessenberg-like structure with bandwidth n
        
        # Actually, let's just build it densely if n is not too large
        # For n up to 4096, K is 8191 x 4096 = 33M elements = 264MB. Too much.
        # Use sparse.
        row_idx = []
        col_idx = []
        vals = []
        for k in range(2*n-1):
            j_lo = max(0, k-n+1)
            j_hi = min(n-1, k)
            for j in range(j_lo, j_hi+1):
                row_idx.append(k)
                col_idx.append(j)
                vals.append(a[j])

    For each coordinate i, find the optimal a[i] given others fixed.
    """

    b = _convolve(a)

            lo_i = max(0, i-(n-1))  # k-i >= 0 => k >= i

                s[k] = 2 * a[k-i]

            # Use golden section or ternary search.

            # At optimum, 0 is in the subdifferential, meaning some s[k*] <= 0 <= others
            # (among the active constraints)
            
            # Simpler: ternary search on f(delta)
            def f(delta):
                return np.max(b + delta * s)
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            f_c = f(c_gs)
            f_d = f(d_gs)

                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    f_c = f(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    f_d = f(d_gs)
            
            opt_delta = (a_gs + b_gs) / 2
            new_a_i = a_i + opt_delta
            
            if new_a_i < 0:
                new_a_i = 0
            
            # Check if this is an improvement

                b = _convolve(a)  # recompute
```

```python
def _lp_min_peak(a):
    """
    Find g >= 0, sum(g) = 1, minimizing max_k (a*g)[k].
    Returns (g*, M*).
    """
    n = len(a)
    # Variables: g_0, ..., g_{n-1}, M (total n+1 variables)
    # Minimize M
    # Constraints:
    #   (a*g)[k] - M <= 0 for k = 0,...,2n-2
    #   -g_j <= 0 for j = 0,...,n-1
    #   sum(g) = 1 (equality constraint)
    
    # c = [0, 0, ..., 0, 1] (minimize M, which is the last variable)
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # A_ub: (2n-1 + n) rows, (n+1) columns
    # First 2n-1 rows: (a*g)[k] - M <= 0
    # K[k,j] = a[k-j] if valid, and last column is -1
    A_ub = np.zeros((2*n-1+n, n+1))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k-j < n:
                A_ub[k, j] = a[k-j]
        A_ub[k, n] = -1.0  # -M term
    # Next n rows: -g_j <= 0
    for j in range(n):
        A_ub[2*n-1+j, j] = -1.0
    b_ub = np.zeros(2*n-1+n)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    
    if not result.success:
        return None, None
    
    g = result.x[:n]
    M_star = result.x[n]
    g = np.maximum(g, 0)
    g = g / g.sum()
    return g, float(M_star)
```

```python
"""
Search for low-peak nonnegative self-convolutions using LP-based minimax optimization.

Key insight: The LP min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0 finds the 'flattest'
cross-convolution partner g for a given a. Iterating a <- (1-t)*a + t*g drives a*a
toward flatness. Combined with coordinate descent and multiple initializations,
this explores both symmetric and asymmetric shapes effectively.

Score = 2n * max(convolve(a,a)) / (sum a)^2, which is scale-invariant.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _normalize(a):
    """Normalize to sum 1, ensuring non-negativity and finiteness."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _convolve_full(a):
    """Full linear convolution a*a."""
    return np.convolve(a, a)


def _score_of(a):
    """Compute the evaluation score (scale-invariant)."""
    a = _normalize(a)
    n = a.size
    if n == 0:
        return 1e100
    b = _convolve_full(a)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return 1e100
    return 2.0 * float(n) * mb


def _resample(old, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old = _normalize(old)
    new_n = max(int(new_n), 2)
    if old.size == new_n:
        return old.copy()
    if old.size <= 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    x_old = np.linspace(0, 1, old.size)
    x_new = np.linspace(0, 1, new_n)
    new = np.interp(x_new, x_old, old)
    return _normalize(new)


def _make_linear(n, direction=1, eps=0.01):
    """Linear ramp: f(x) = x^direction + eps."""
    x = np.linspace(0, 1, n)
    if direction == 1:
        a = x + eps
    else:
        a = (1.0 - x) + eps
    return _normalize(a)


def _make_poly(n, p, side=1, eps=0.01):
    """Polynomial: f(x) = (x or 1-x)^p + eps."""
    x = np.linspace(0, 1, n)
    if side == 1:
        a = x**p + eps
    else:
        a = (1.0 - x)**p + eps
    return _normalize(a)


def _make_gaussian(n, center=0.5, width=0.3):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
    return _normalize(a)


def _make_sine(n):
    """Sine arch."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) + 0.01
    return _normalize(a)


def _make_mixed_gaussian(n, c1, c2, w1, w2, ratio):
    """Sum of two Gaussians."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) ** 2) / (2.0 * w1 ** 2)) + ratio * np.exp(-((x - c2) ** 2) / (2.0 * w2 ** 2))
    return _normalize(a)


def _generate_inits(n, given_seq, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Linear ramps
    inits.append(_make_linear(n, 1))
    inits.append(_make_linear(n, -1))

    # Polynomials
    inits.append(_make_poly(n, 1, 1))
    inits.append(_make_poly(n, 2, 1))
    inits.append(_make_poly(n, 1, -1))
    inits.append(_make_poly(n, 2, -1))
    inits.append(_make_poly(n, 3, 1))

    # Gaussians (asymmetric and symmetric)
    inits.append(_make_gaussian(n, 0.5, 0.25))
    inits.append(_make_gaussian(n, 0.3, 0.2))
    inits.append(_make_gaussian(n, 0.7, 0.2))
    inits.append(_make_gaussian(n, 0.5, 0.4))
    inits.append(_make_gaussian(n, 0.5, 0.15))

    # Sine
    inits.append(_make_sine(n))

    # Mixed Gaussians
    inits.append(_make_mixed_gaussian(n, 0.2, 0.8, 0.15, 0.15, 1.0))
    inits.append(_make_mixed_gaussian(n, 0.25, 0.75, 0.2, 0.2, 0.5))
    inits.append(_make_mixed_gaussian(n, 0.1, 0.9, 0.12, 0.12, 1.0))

    # Given sequence (resampled)
    if given_seq is not None and len(given_seq) > 4:
        inits.append(_resample(given_seq, n))
        # Also try flipped
        inits.append(_normalize(given_seq[::-1]))

    # Random smooth perturbations of best few
    for base in inits[:3]:
        noise = rng.normal(0, 0.15, n)
        # Smooth via FFT lowpass
        F = np.fft.rfft(noise)
        cutoff = max(2, n // 8)
        if cutoff < len(F):
            F[cutoff:] *= 0.0
        f = np.fft.irfft(F, n)
        a = base + f
        inits.append(_normalize(a))

    # Deduplicate
    seen = set()
    out = []
    for ic in inits:
        sig = tuple((ic * 1000).astype(int)[::max(1, ic.size // 16)])
        if sig not in seen:
            seen.add(sig)
            out.append(ic)
    return out


def _build_lp_matrix(a, n):
    """
    Build the (2n-1) x n matrix K where K[k,j] = a[k-j] if 0<=k-j<n else 0.
    (a*g)[k] = K[k,:] @ g
    """
    K = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k) + 1
        if j_hi > j_lo:
            # K[k, j] = a[k-j] for j in [j_lo, j_hi-1]
            # a[k-j] for j = j_lo,...,j_hi-1 gives a[k-j_lo], a[k-j_lo-1], ..., a[k-(j_hi-1)]
            idx = np.arange(k - j_hi + 1, k - j_lo + 1)
            K[k, j_lo:j_hi] = a[idx]
    return K


def _lp_min_peak_cross(a, n=None):
    """
    Solve: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g*, M*) or (None, None) on failure.
    """
    if n is None:
        n = len(a)
    a = np.asarray(a, dtype=np.float64)
    a = _normalize(a)

    # Variables: [g_0, ..., g_{n-1}, M] (n+1 variables)
    # Objective: minimize M -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality constraints: A_ub @ x <= b_ub
    # (2n-1) rows for (a*g)[k] - M <= 0
    # n rows for -g_j <= 0
    num_ineq = (2 * n - 1) + n
    A_ub = np.zeros((num_ineq, n + 1), dtype=np.float64)
    b_ub = np.zeros(num_ineq, dtype=np.float64)

    # Build convolution constraint rows
    K = _build_lp_matrix(a, n)
    A_ub[:2 * n - 1, :n] = K
    A_ub[:2 * n - 1, n] = -1.0  # -M term

    # Non-negativity: -g_j <= 0
    for j in range(n):
        A_ub[2 * n - 1 + j, j] = -1.0

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         method='highs', options={'maxiter': 20000})
    except Exception:
        return None, None

    if not result.success:
        return None, None

    g = result.x[:n]
    M_star = float(result.x[n])
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-300:
        return None, None
    g = g / s
    return g, M_star


def _lp_optimize(a_init, deadline, rng, n_improvement_calls=5000):
    """
    Iterative LP-based optimization.
    a <- (1-t)*a + t*g where g minimizes max(a*g).
    Returns (best_a, best_score).
    """
    a = _normalize(a_init.copy())
    n = a.size

    b = _convolve_full(a)
    score = 2.0 * float(n) * float(np.max(b))
    best_a = a.copy()
    best_score = score

    t_mix = 0.4
    stall_count = 0
    call_count = 0

    while time.monotonic() < deadline and call_count < n_improvement_calls:
        g, M_star = _lp_min_peak_cross(a)
        call_count += 1

        if g is None:
            # Perturb and retry
            a = _normalize(a + rng.normal(0, 0.05, n))
            stall_count += 1
            if stall_count > 20:
                break
            continue

        # Compute current score of g
        g_score = _score_of(g)

        # Mix
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = _normalize(a_new)

        new_score = _score_of(a_new)

        if new_score < score - 1e-12:
            a = a_new
            score = new_score
            stall_count = 0
            # Slowly reduce mixing
            t_mix = max(0.15, t_mix * 0.995)
        else:
            # Try smaller mixing
            t_mix = max(0.05, t_mix * 0.7)
            a_new2 = (1.0 - t_mix) * a + t_mix * g
            a_new2 = _normalize(a_new2)
            new_score2 = _score_of(a_new2)
            if new_score2 < score - 1e-12:
                a = a_new2
                score = new_score2
                stall_count = 0
            else:
                stall_count += 1
                t_mix = min(0.8, t_mix * 1.3 + 0.05)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Periodic perturbation to escape local minima
        if stall_count > 30 and time.monotonic() < deadline - 5:
            a = _normalize(a + rng.normal(0, 0.08, n))
            a = np.maximum(a, 0)
            a = _normalize(a)
            b = _convolve_full(a)
            score = 2.0 * float(n) * float(np.max(b))
            stall_count = 0
            t_mix = 0.3

        # Reset mixing if it got too small
        if t_mix < 0.15:
            t_mix = 0.25

    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_sweeps=200):
    """
    Coordinate descent minimizing max(a*a).
    For each coordinate i, find optimal value via golden section search.
    Uses incremental convolution updates.
    """
    a = _normalize(a_init.copy())
    n = a.size
    b = _convolve_full(a)
    best_score = 2.0 * float(n) * float(np.max(b))
    best_b_max = float(np.max(b))
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        # Randomize coordinate order for diversity
        order = np.arange(n)

        for i in order:
            a_i = a[i]
            if a_i < 1e-15:
                # Try adding a small amount
                delta = best_b_max * 0.01
                # Check if adding helps
                s = np.zeros(2 * n - 1)
                for k in range(i, min(2 * n - 1, i + n)):
                    s[k] = 2.0 * a[k - i]
                new_max = np.max(b + delta * s)
                if new_max < best_b_max - 1e-12:
                    a[i] = delta
                    b += delta * s
                    improved = True

            if a[i] < 1e-15:
                continue

            # Sensitivity: db[k]/da[i] = 2*a[k-i] for k where 0<=k-i<n
            s = np.zeros(2 * n - 1, dtype=np.float64)
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            s[k_lo:k_hi + 1] = 2.0 * a[0:n - (k_hi - i)]

            max_s = float(np.max(s))
            if max_s < 1e-300:
                continue

            # Golden section search for delta in [-a_i, upper]
            lo = -a_i
            hi = float(np.max(b - b[i % (2*n-1)] if False else 0)) # placeholder

            # Upper bound: don't let any b[k] exceed 2*current_max
            cur_max = float(np.max(b))
            hi = cur_max / max_s

            if hi <= lo:
                continue

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)

            # f(delta) = max(k) (b[k] + delta * s[k])
            f_c = float(np.max(b + c_gs * s))
            f_d = float(np.max(b + d_gs * s))

            for _ in range(25):
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    f_c = float(np.max(b + c_gs * s))
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    f_d = float(np.max(b + d_gs * s))

            opt_delta = (a_gs + b_gs) / 2.0
            opt_f = float(np.max(b + opt_delta * s))

            if opt_f < float(np.max(b)) - 1e-12:
                a[i] = max(0.0, a_i + opt_delta)
                b += opt_delta * s
                improved = True

                new_max = float(np.max(b))
                new_score = 2.0 * float(n) * new_max
                if new_score < best_score:
                    best_score = new_score
                    best_b_max = new_max
                    best_a = a.copy()

        if not improved:
            break

    a = _normalize(best_a)
    return a, _score_of(a)


def _smooth_gradient_descent(a_init, deadline, rng):
    """
    Log-sum-exp smoothed gradient descent on the simplex.
    Used as a complementary method to LP and coordinate descent.
    """
    a = _normalize(a_init.copy())
    n = a.size

    # Use FFT for convolution
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2

    def fast_conv(x):
        X = np.fft.rfft(x, fft_size)
        return np.fft.irfft(X * X, fft_size)[:2 * n - 1]

    b = fast_conv(a)
    score = 2.0 * float(n) * float(np.max(b))
    best_a = a.copy()
    best_score = score

    beta = 20.0
    alpha = 0.1
    fail = 0
    it = 0

    def lse_val(b_vec, beta_v):
        m = float(np.max(b_vec))
        arg = beta_v * (b_vec - m)
        arg = np.clip(arg, -700, 700)
        e = np.exp(arg)
        s = float(e.sum())
        if s < 1e-300:
            return m, np.ones_like(b_vec) / b_vec.size
        p = e / s
        return m + np.log(s) / beta_v, p

    def grad_weighted(w, b_vec):
        # Gradient of sum_k w[k] * b[k] w.r.t. a[i]
        # = 2 * sum_k w[k] * a[k-i] for valid k-i
        # Use FFT: correlate a with w
        ar = a[::-1]
        A = np.fft.rfft(ar, fft_size)
        W = np.fft.rfft(w, fft_size)
        conv = np.fft.irfft(A * W, fft_size).real
        return 2.0 * conv[n - 1:2 * n - 1]

    while time.monotonic() < deadline:
        smooth_old, p = lse_val(b, beta)
        grad = grad_weighted(p, b)
        grad_c = grad - float(np.mean(grad))
        scale = float(np.max(np.abs(grad_c)))

        if scale < 1e-300:
            a = _normalize(a + rng.normal(0, 0.05, n))
            b = fast_conv(a)
            score = 2.0 * float(n) * float(np.max(b))
            fail = 0
            it += 1
            continue

        v = a - alpha * grad_c / scale * (beta * float(np.max(b)) / n)
        a = _normalize(v)
        b = fast_conv(a)
        maxb = float(np.max(b))
        score = 2.0 * float(n) * maxb
        smooth_new, _ = lse_val(b, beta)

        if smooth_new < smooth_old - 1e-14:
            alpha = min(1.0, alpha * 1.1)
            fail = 0
        else:
            alpha *= 0.6
            fail += 1

        if score < best_score:
            best_score = score
            best_a = a.copy()

        if it % 50 == 0:
            beta = min(200.0, beta * 1.05)

        if fail > 100:
            a = _normalize(best_a + rng.normal(0, 0.06, n))
            b = fast_conv(a)
            score = 2.0 * float(n) * float(np.max(b))
            alpha = 0.1
            beta = max(10.0, beta * 0.5)
            fail = 0

        it += 1

    return best_a, best_score


def _get_given_sequence():
    """Retrieve the provided sequence from global scope."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0 or float(arr.sum()) < 1e-300:
        return None
    return _normalize(arr)


def propose_candidate(seed=0, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for the best nonnegative coefficient sequence
    minimizing 2n*max(convolve(a,a))/(sum(a))^2 within the time budget.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 2400.0

    deadline = time.monotonic() + budget - 15.0
    rng = np.random.default_rng(seed)

    given = _get_given_sequence()

    # Initial best: use given sequence or a good default
    if given is not None:
        best_a = given.copy()
        best_score = _score_of(best_a)
    else:
        best_a = _make_linear(256, 1)
        best_score = _score_of(best_a)

    print(f"Initial best: len={best_a.size}, score={best_score:.8f}", flush=True)

    # Evaluate with the actual evaluate_sequence if available
    try:
        eval_score = evaluate_sequence([float(x) for x in best_a])
        if np.isfinite(eval_score) and eval_score < best_score:
            best_score = eval_score
            print(f"Eval-confirmed score: {best_score:.8f}", flush=True)
    except Exception:
        pass

    # Phase 1: Try multiple initializations at the given length
    n_given = best_a.size if given is not None else 256
    n_phase1 = max(128, min(1024, n_given))

    inits = _generate_inits(n_phase1, given, rng)
    print(f"Phase 1: {len(inits)} inits at n={n_phase1}", flush=True)

    # Time budget per init: spend 60% of total on Phase 1
    phase1_budget = 0.6 * (deadline - time.monotonic())
    time_per_init = phase1_budget / len(inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > deadline - 60:
            break
        t_end = min(deadline - 60, time.monotonic() + max(5, time_per_init))

        # LP optimization
        a_lp, score_lp = _lp_optimize(init, t_end, rng)
        if score_lp < best_score:
            best_score = score_lp
            best_a = a_lp.copy()
            print(f"  Init {idx}: LP score={score_lp:.8f} (improved!) t={time.monotonic()-(deadline-budget+15):.0f}s", flush=True)

        # Coordinate descent refinement
        if time.monotonic() < t_end - 2:
            a_cd, score_cd = _coordinate_descent(a_lp, min(t_end, time.monotonic() + (t_end - time.monotonic()) * 0.4), max_sweeps=50)
            if score_cd < best_score:
                best_score = score_cd
                best_a = a_cd.copy()
                print(f"  Init {idx}: CD score={score_cd:.8f} (improved!) t={time.monotonic()-(deadline-budget+15):.0f}s", flush=True)

    print(f"After Phase 1: best_score={best_score:.8f}, len={best_a.size}", flush=True)

    # Phase 2: Refine the best found so far with more iterations
    if time.monotonic() < deadline - 30:
        t_end = min(deadline - 10, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        a_r, score_r = _lp_optimize(best_a.copy(), t_end, rng)
        if score_r < best_score:
            best_score = score_r
            best_a = a_r.copy()
            print(f"Phase 2 LP: score={score_r:.8f}", flush=True)

    # Phase 3: Try different lengths by resampling
    if time.monotonic() < deadline - 30:
        for n_try in [256, 512, 1024, 2048]:
            if time.monotonic() > deadline - 30:
                break
            if n_try <= best_a.size and best_score < 1.52:
                continue

            a_res = _resample(best_a, n_try)
            # Mix with a fresh init for diversity
            fresh = _make_linear(n_try, 1)
            a_res = _normalize(0.7 * a_res + 0.3 * fresh)

            t_end = min(deadline - 20, time.monotonic() + min(60, 0.08 * (deadline - time.monotonic())))
            a_lp, score_lp = _lp_optimize(a_res, t_end, rng, n_improvement_calls=2000)

            if score_lp < best_score:
                best_score = score_lp
                best_a = a_lp.copy()
                print(f"Phase 3 n={n_try}: LP score={score_lp:.8f} (improved!)", flush=True)

            # CD refinement
            if score_lp < best_score + 0.05 and time.monotonic() < deadline - 15:
                t_end2 = min(deadline - 10, time.monotonic() + min(30, 0.04 * (deadline - time.monotonic())))
                a_cd, score_cd = _coordinate_descent(a_lp, t_end2, max_sweeps=30)
                if score_cd < best_score:
                    best_score = score_cd
                    best_a = a_cd.copy()
                    print(f"Phase 3 n={n_try}: CD score={score_cd:.8f} (improved!)", flush=True)

    # Phase 4: Smooth gradient descent on the best
    if time.monotonic() < deadline - 10:
        t_end = min(deadline - 5, time.monotonic() + min(30, 0.05 * (deadline - time.monotonic())))
        a_sg, score_sg = _smooth_gradient_descent(best_a.copy(), t_end, rng)
        if score_sg < best_score:
            best_score = score_sg
            best_a = a_sg.copy()
            print(f"Phase 4 smooth-grad: score={score_sg:.8f} (improved!)", flush=True)

    # Final: one more LP refinement pass
    if time.monotonic() < deadline - 3:
        a_final, score_final = _lp_optimize(best_a.copy(), deadline - 1.0, rng, n_improvement_calls=500)
        if score_final < best_score:
            best_score = score_final
            best_a = a_final.copy()

    print(f"FINAL: score={best_score:.8f}, len={best_a.size}", flush=True)

    # Validate and return
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 0.01:
        # Fallback to avoid the sum_a < 0.01 rejection
        out = np.ones(256, dtype=np.float64)
        s = 256.0
    # Scale so sum is well above 0.01 (the evaluation clips to [0,1000] then checks sum>=0.01)
    # Since score is scale-invariant, any positive scaling works
    out = out / s  # normalize to sum 1
    # Ensure sum_a in evaluation will be >= 0.01 (it is, since sum=1)

    return [float(x) for x in out]
```
