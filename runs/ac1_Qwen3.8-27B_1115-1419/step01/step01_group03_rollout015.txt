"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

The main search keeps lengths in the few-thousand range and optimizes the
sequence in log-variables with multiplicative/Fisher-style updates.  The
gradient direction is produced by measuring how far the linear convolution
distribution is from a uniform target (log-KL / variance / peak weights).
FFT is used for both the convolution and the cross-correlation needed by
the update.  Several initial densities and random logits are tried, with
periodic restarts/noise.
"""

import numpy as np
import time


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _setup_fft(n):
    N = _next_pow2(3 * n - 2)
    N = max(N, 16)
    Lw = 2 * n - 1
    return N, Lw


def _normalize(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return np.ones(1, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=float) / float(a.size)
    return a / s


def _center_clip_s(s, smax=40.0):
    s = np.asarray(s, dtype=float)
    if s.size == 0:
        return s
    s = np.where(np.isfinite(s), s, -smax)
    s = s - np.median(s)
    return np.clip(s, -smax, smax)


def _softmax_s(s):
    s = _center_clip_s(s, 40.0)
    if s.size == 0:
        return np.ones(1, dtype=float)
    m = float(np.max(s))
    e = np.exp(s - m)
    z = e.sum()
    if not np.isfinite(z) or z <= 0.0:
        return np.ones(s.size, dtype=float) / float(s.size)
    return e / z


def _s_from_p(p, floor=1e-12):
    p = _normalize(p)
    p = np.maximum(p, floor)
    p = _normalize(p)
    s = np.log(p)
    return _center_clip_s(s, 40.0)


def _rfft_convolution(p, N, n, Lw):
    p = np.asarray(p, dtype=float)
    if p.size != n:
        p = _normalize(p)[:n] if p.size >= n else np.concatenate([p, np.zeros(n - p.size)])
    Fp = np.fft.rfft(p, N)
    q = np.fft.irfft(Fp * Fp, n=N)
    q = q[:Lw]
    q = np.maximum(q, 0.0)
    return Fp, q


def _score_norm(p, N, n, Lw):
    p = _normalize(p)
    try:
        Fp, q = _rfft_convolution(p, N, n, Lw)
        val = 2.0 * float(n) * float(np.max(q))
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def _resample_nonnegative(arr, m):
    arr = np.asarray(arr, dtype=float).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size == 0:
        return np.ones(m, dtype=float) / float(m)
    s = arr.sum()
    if s <= 1e-15:
        return np.ones(m, dtype=float) / float(m)
    arr = arr / s
    old_x = np.arange(arr.size, dtype=float)
    new_x = (arr.size - 1) * np.linspace(0.0, 1.0, m, dtype=float)
    out = np.interp(new_x, old_x, arr)
    out = np.maximum(out, 0.0)
    return _normalize(out)


def _make_candidates(n, hs_global, aux_p, rng):
    out = []

    if hs_global is not None and hs_global.size > 0:
        if hs_global.size == n:
            out.append(_normalize(hs_global))
        else:
            out.append(_resample_nonnegative(hs_global, n))

        psm = _resample_nonnegative(hs_global, n)
        kernel = np.ones(7, dtype=float) / 7.0
        psm2 = np.convolve(psm, kernel, mode="same")
        out.append(_normalize(psm2))

    if aux_p is not None and aux_p.size > 0:
        out.append(_resample_nonnegative(aux_p, n))

    x = (np.arange(n, dtype=float) + 0.5) / float(n)

    out.append(np.ones(n, dtype=float))

    for alpha in (0.20, 0.35, 0.50, 0.65, 0.80):
        left = np.power(x, -alpha)
        right = np.power(1.0 - x, -alpha)
        out.append(_normalize(left))
        out.append(_normalize(right))

    for lam in (0.5, 1.0, 2.0, 4.0):
        out.append(_normalize(np.exp(-lam * x)))
        out.append(_normalize(np.exp(-lam * (1.0 - x))))

    for tilt in (0.5, 1.0, 1.5, 2.0):
        out.append(_normalize(np.maximum(0.0, 1.0 + tilt * (x - 0.5))))
        out.append(_normalize(np.maximum(0.0, 1.0 - tilt * (x - 0.5))))

    beta_pairs = (
        (1.0, 1.0),
        (1.0, 2.0),
        (2.0, 1.0),
        (0.5, 1.0),
        (1.0, 0.5),
        (2.0, 3.0),
        (3.0, 2.0),
    )
    for a, b in beta_pairs:
        p = np.power(x, a) * np.power(1.0 - x, b)
        out.append(_normalize(np.maximum(p, 0.0)))

    for temp in (0.2, 0.5, 1.0, 2.0):
        s = rng.normal(0.0, temp, n)
        out.append(_softmax_s(s))

    cleaned = []
    for p in out:
        p = _normalize(p)
        if p.size == n and np.all(np.isfinite(p)):
            cleaned.append(p)
    return cleaned


def _make_weights(qn, n, mode, wclip, floor_ratio, beta, var_scale):
    Lw = 2 * n - 1
    r = 1.0 / float(Lw)
    q = np.asarray(qn, dtype=float)
    if q.size != Lw:
        q = np.resize(q, Lw)
    if q.size == 0:
        return np.zeros(Lw, dtype=float)

    if mode == "kl":
        eps = max(r * float(floor_ratio), 1e-300)
        qsafe = np.maximum(q, eps)
        raw = np.log(qsafe / r)
        w = np.clip(raw, -float(wclip), float(wclip))
    elif mode == "var":
        denom = max(float(var_scale), 1e-6)
        raw = (q / r - 1.0) / denom
        w = np.clip(raw, -float(wclip), float(wclip))
    elif mode == "peak":
        qmax = float(np.max(q))
        if qmax <= 0.0:
            w = np.ones(Lw, dtype=float) / float(Lw)
        else:
            ratio = q / qmax
            w = np.power(ratio, float(beta))
            ws = w.sum()
            if ws > 0.0:
                w = w / ws
            else:
                w = np.ones(Lw, dtype=float) / float(Lw)
    else:
        w = np.zeros(Lw, dtype=float)

    w = np.where(np.isfinite(w), w, 0.0)
    return w


def _kl_run(s0, n, duration, rng, mode="kl", lr=0.05, wclip=4.0,
            floor_ratio=1e-8, beta=3.0, var_scale=0.5,
            smax=40.0, hclip=10.0, tag=""):
    if duration <= 0.0:
        s0 = _center_clip_s(s0, smax)
        best_p = _softmax_s(s0)
        return s0, best_p, np.inf

    N, Lw = _setup_fft(n)
    s = _center_clip_s(np.asarray(s0, dtype=float), smax)
    p = _softmax_s(s)
    Fp, q = _rfft_convolution(p, N, n, Lw)
    current_score = 2.0 * float(n) * float(np.max(q))

    best_s = s.copy()
    best_p = p.copy()
    best_score = current_score

    lr_cur = float(lr)
    min_lr = 1e-4
    no_improve = 0
    noise = 1.0 / float(np.sqrt(n))
    it = 0
    deadline = time.monotonic() + float(duration)
    last_print = time.monotonic()
    w_rev = np.empty(N, dtype=float)

    print(f"{tag} start score={best_score:.8f} mode={mode} lr={lr_cur:.4f}")

    while time.monotonic() < deadline:
        qsum = float(q.sum())
        if qsum > 0.0 and np.isfinite(qsum):
            qn = q / qsum
        else:
            qn = q.copy()

        w = _make_weights(qn, n, mode, wclip, floor_ratio, beta, var_scale)
        w_rev.fill(0.0)
        w_rev[:Lw] = w[::-1]
        Fw = np.fft.rfft(w_rev, N)
        conv = np.fft.irfft(Fp * Fw, n=N)
        h = conv[2 * n - 2:n - 2:-1]

        if h.size != n:
            h = np.zeros(n, dtype=float)

        h = h - float(np.mean(h))
        hmax = float(np.max(np.abs(h)))

        use_step = np.isfinite(hmax) and hmax >= 1e-12
        if not use_step:
            no_improve = max(no_improve, 1000)
        else:
            h = np.clip(h, -float(hclip), float(hclip))

        accepted = False
        lr_try = lr_cur
        for _attempt in range(3):
            s_try = s - lr_try * h
            s_try = _center_clip_s(s_try, smax)
            p_try = _softmax_s(s_try)
            Fp_try, q_try = _rfft_convolution(p_try, N, n, Lw)
            sc_try = 2.0 * float(n) * float(np.max(q_try))

            if np.isfinite(sc_try):
                ref = current_score if np.isfinite(current_score) else best_score
                acceptable = (sc_try < ref * 1.002) and (sc_try < best_score * 1.002)
                if acceptable:
                    s = s_try
                    p = p_try
                    Fp = Fp_try
                    q = q_try
                    current_score = sc_try
                    accepted = True
                    if sc_try < best_score:
                        best_score = sc_try
                        best_s = s.copy()
                        best_p = p.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    break
            lr_try *= 0.5
            if lr_try < 0.5 * min_lr:
                break

        if not accepted:
            if (not np.isfinite(current_score)) or current_score > best_score * 1.01:
                s = best_s.copy()
                p = _softmax_s(s)
                Fp, q = _rfft_convolution(p, N, n, Lw)
                current_score = 2.0 * float(n) * float(np.max(q))
            lr_cur = max(min_lr, lr_cur * 0.6)
            no_improve += 1

        if no_improve > 800:
            s_probe = best_s + rng.normal(0.0, noise, n)
            s_probe = _center_clip_s(s_probe, smax)
            p_probe = _softmax_s(s_probe)
            Fp_probe, q_probe = _rfft_convolution(p_probe, N, n, Lw)
            sc_probe = 2.0 * float(n) * float(np.max(q_probe))

            if np.isfinite(sc_probe) and sc_probe < best_score * 1.02:
                s = s_probe
                p = p_probe
                Fp = Fp_probe
                q = q_probe
                current_score = sc_probe
                if sc_probe < best_score:
                    best_score = sc_probe
                    best_s = s.copy()
                    best_p = p.copy()
                noise = min(8.0, noise * 1.25)
                lr_cur = max(lr_cur, 0.02)
            else:
                s = best_s.copy()
                p = _softmax_s(s)
                Fp, q = _rfft_convolution(p, N, n, Lw)
                current_score = best_score
                noise = max(0.2 / float(np.sqrt(n)), noise * 0.6)
            no_improve = 0

        it += 1
        now = time.monotonic()
        if now - last_print > 25.0:
            print(f"{tag} it={it} best={best_score:.8f} current={current_score:.8f} lr={lr_cur:.4f}")
            last_print = now

    print(f"{tag} end it={it} best={best_score:.8f}")
    return best_s, best_p, best_score


def _optimize_length(n, hs_global, aux_p, duration, rng, tag):
    if duration <= 0.0 or n < 2:
        return None, None, np.inf

    N, Lw = _setup_fft(n)
    cands = _make_candidates(n, hs_global, aux_p, rng)

    scored = []
    for p in cands:
        sc = _score_norm(p, N, n, Lw)
        scored.append((sc, p))

    if not scored:
        p = np.ones(n, dtype=float) / float(n)
        sc = _score_norm(p, N, n, Lw)
        s = _s_from_p(p)
        return s, p, sc

    scores_arr = np.array([sc for sc, _ in scored], dtype=float)
    order = np.argsort(scores_arr)
    sorted_cands = [scored[i][1] for i in order]
    sorted_scores = [scored[i][0] for i in order]

    best_p = sorted_cands[0].copy()
    best_score = float(sorted_scores[0])
    best_s = _s_from_p(best_p)

    print(f"{tag} best initial candidate score={best_score:.8f}")

    if duration < 60.0:
        return best_s, best_p, best_score

    if duration < 400.0 or n > 10000:
        variations = [
            # (start_idx, mode, lr, wclip, floor_ratio, beta, var_scale, fraction)
            (-1, "kl", 0.05, 4.0, 1e-8, 1.0, 0.5, 0.70),
            (-1, "kl", 0.02, 3.0, 1e-6, 1.0, 0.5, 0.30),
        ]
    else:
        variations = [
            (-1, "kl", 0.05, 4.0, 1e-8, 1.0, 0.5, 0.30),
            (-1, "kl", 0.08, 6.0, 1e-6, 1.0, 0.5, 0.20),
            (1, "kl", 0.06, 4.0, 1e-8, 1.0, 0.5, 0.15),
            (-1, "peak", 0.10, 1.0, 0.0, 3.0, 0.0, 0.10),
            (-1, "var", 0.04, 4.0, 0.0, 1.0, 0.3, 0.05),

    t0 = time.monotonic()

        idx = int(var[0])
        if idx != -1 and idx >= len(sorted_cands):

        ti = float(var[7]) * float(duration)

            continue

        if idx == -1:
            start_s = best_s.copy()
        else:
            start_s = _s_from_p(sorted_cands[idx])

        res_s, res_p, res_sc = _kl_run(
            start_s, n, ti, rng,
            mode=var[1], lr=var[2], wclip=var[3],
            floor_ratio=var[4], beta=var[5], var_scale=var[6],
            tag=f"{tag}:{var[1]}:{idx}"
        )
        if np.isfinite(res_sc) and res_sc < best_score:
            best_score = res_sc
            best_s = res_s
            best_p = res_p
            print(f"{tag} improved to {best_score:.8f}")

    # Final short polish from the best point of this length.

            best_s, n, remaining, rng,

        )
        if np.isfinite(res_sc) and res_sc < best_score:
            best_score = res_sc
            best_s = res_s
            best_p = res_p

    return best_s, best_p, best_score


def _to_output_list(p):
    p = np.asarray(p, dtype=float)
    if p.size == 0:
        return [1.0]
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if not np.isfinite(s) or s <= 1e-15:
        p = np.ones(p.size, dtype=float) / float(p.size)
    else:
        p = p / s
    p = np.minimum(p, 1000.0)
    return [float(x) for x in p]

    remaining = total_deadline - time.monotonic()
    if best_all_p is not None and remaining > 40.0:
        dur = max(20.0, remaining - 5.0)
        res_s, res_p, res_sc = _kl_run(
            best_all_s, best_all_n, dur, rng,
            mode="kl", lr=0.012, wclip=3.0,
            floor_ratio=1e-6, beta=1.0, var_scale=0.5,
            tag="final-global-polish"
        )
        if res_p is not None and np.isfinite(res_sc) and res_sc < best_all_score:
            best_all_s, best_all_p, best_all_score, best_all_n = res_s, res_p, res_sc, best_all_n
            print(f"final polish improved internal score = {best_all_score:.8f}")

    if best_all_p is None or not np.isfinite(best_all_score):
        final_p = hs_fallback
    else:
        fb_score = _score_norm(hs_fallback, _setup_fft(hs_fallback.size)[0], hs_fallback.size, _setup_fft(hs_fallback.size)[1])
        if best_all_score <= fb_score + 1e-9:
            final_p = best_all_p
        else:
            final_p = hs_fallback

    final_list = _to_output_list(final_p)

    try:
        final_score = float(evaluate_sequence(final_list))  # noqa: F821
    except Exception:
        final_score = np.inf

    if not np.isfinite(final_score):
        final_list = _to_output_list(hs_fallback)

    try:
        print(f"FINAL internal-score-style score={best_all_score:.8f} verified={final_score:.8f} n={len(final_list)}")
    except Exception:
        pass

    return final_list
