"""
Search for nonnegative step-function heights minimizing
    2 * n * max(np.convolve(a, a)) / sum(a)^2.

The code uses:
  - FFT-based convex-looking activation gradients with exact nonnegative sequence
    parameterization in log-softmax space.
  - Line-search backtracking on the max-convolution objective.
  - Analytic starts from binomial/sqrt(1/(1-z)) shapes, power-law envelopes,
    and spectral-factor attempts toward flatter target convolutions.
  - Optional sparse linear-programming directions inspired by the active-set
    state of the art.
  - Progressive length search: 512, 1024, 2048, 4096, 8192, 16384.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    _HAS_LINPROG = True
except Exception:
    _HAS_LINPROG = False


def _pair_score(pair):
    return pair[0]


def _next_pow2(x):
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


def _project(a):
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(arr.size, 1.0 / arr.size, dtype=np.float64)
    arr = arr / s
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    s2 = float(arr.sum())
    if not np.isfinite(s2) or s2 <= 0.0:
        return np.full(arr.size, 1.0 / arr.size, dtype=np.float64)
    arr = arr / s2
    return arr


def _soft(v):
    arr = np.asarray(v, dtype=np.float64).ravel().copy()
    n = arr.size
    if n == 0:
        return np.array([], dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=20.0, neginf=-20.0)
    arr = np.clip(arr, -20.0, 20.0)
    arr = arr - float(np.max(arr))
    e = np.exp(arr)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def _score_array(a, n, m):
    a = _project(a)
    if a is None or a.size != n:
        return np.inf
    F = np.fft.fft(a, m)
    B = np.fft.ifft(F * F).real[:2 * n - 1]
    B = np.maximum(B, 0.0)
    bm = float(B.max())
    if not np.isfinite(bm) or bm <= 0.0:
        return np.inf
    return float(2.0 * n * bm)


def _score_and_B_array(a, n, m):
    a = _project(a)
    if a is None or a.size != n:
        return None, None, np.inf
    F = np.fft.fft(a, m)
    B = np.fft.ifft(F * F).real[:2 * n - 1]
    B = np.maximum(B, 0.0)
    bm = float(B.max())
    if not np.isfinite(bm) or bm <= 0.0:
        return a, B, np.inf
    return a, B, float(2.0 * n * bm)


def _score_vector(v, n, m):
    a = _soft(v)
    F = np.fft.fft(a, m)
    B = np.fft.ifft(F * F).real[:2 * n - 1]
    B = np.maximum(B, 0.0)
    bm = float(B.max())
    if not np.isfinite(bm) or bm <= 0.0:
        return np.inf
    return float(2.0 * n * bm)


def _score_and_B_vector(v, n, m):
    a = _soft(v)
    F = np.fft.fft(a, m)
    B = np.fft.ifft(F * F).real[:2 * n - 1]
    B = np.maximum(B, 0.0)
    bm = float(B.max())
    if not np.isfinite(bm) or bm <= 0.0:

def _normalize_dir(g):
    g = np.asarray(g, dtype=np.float64)
    if g.size == 0:
        return None
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g = g - float(np.mean(g))
    grms = float(np.sqrt(np.mean(g * g)))
    if not np.isfinite(grms) or grms < 1e-12:
        return None
    return g / grms


def _correlation_weights(a, w, n, m3):
    L = 2 * n - 1
    wr = w[::-1]
    Fa = np.fft.fft(a, m3)
    Fw = np.fft.fft(wr, m3)
    conv = np.fft.ifft(Fa * Fw).real[:n + L - 1]
    return conv[n - 1:2 * n - 1][::-1]


def _active_direction(v, n, m, m3, p, top_frac, rng=None):
    sc, a, B, M = _score_and_B_vector(v, n, m)
    if not np.isfinite(sc) or M <= 0.0:
        return sc, a, None, M
    r = np.clip(B / M, 0.0, 1.0)
    if p < 1.0:
        w = r.copy()
    else:
        w = np.power(r, p)
    w = w + 1e-10

    K = int(top_frac * B.size)
    K = max(1, min(B.size, K))
    if K >= B.size:
        idx = np.arange(B.size)
    else:
        idx = np.argpartition(B, -K)[-K:]
    w[idx] += 0.5

    if rng is not None:
        knob = max(1, min(B.size, int(0.10 * B.size)))
        if knob < B.size:
            idx2 = np.argpartition(B, -knob)[-knob:]
            w[idx2] += 0.25 * rng.random(knob)

    w = w + 1e-8
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.full(B.size, 1.0 / B.size, dtype=np.float64)
    else:
        w = w / sw

    h = _correlation_weights(a, w, n, m3)
    gv = a * (h - float(np.dot(a, h)))
    gv = _normalize_dir(gv)
    return sc, a, gv, M


def _try_steps(v, g, lr, n, m, cur_sc):
    g = _normalize_dir(g)
    if g is None:
        return v, cur_sc, 0.0

    best_v = np.asarray(v, dtype=np.float64).copy()
    best_sc = float(cur_sc)
    best_s = 0.0

    trials = []
    for mult in (1.0, 2.0, 0.5, 0.25, 4.0, 0.125, 8.0):
        s = float(lr * mult)
        if 0.02 <= s <= 2.5:
            same = False
            for t in trials:
                if abs(s - t) < 1e-3:
                    same = True
                    break
            if not same:
                trials.append(s)

    for s in trials:
        v2 = v - s * g
        v2 = v2 - float(np.mean(v2))
        v2 = np.clip(v2, -20.0, 20.0)
        sc2 = _score_vector(v2, n, m)
        if np.isfinite(sc2) and sc2 < best_sc - 1e-13:
            best_sc = sc2
            best_v = v2.copy()
            best_s = s
    return best_v, best_sc, best_s


def _smooth_noise(n, rng, K, amp):
    noise = np.zeros(n, dtype=np.float64)
    K = int(min(max(1, K), max(1, n - 1)))
    x = np.arange(n, dtype=np.float64)
    for k in range(1, K + 1):
        ak = float(amp) / (float(k) ** 0.7)
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        noise += ak * np.cos(2.0 * math.pi * float(k) * x / float(n) + ph)
    return noise


def _optimize_active(a0, n, seconds, seed=0, lr0=0.55, p_start=2.0, p_end=30.0,
                     momentum=0.70):
    n = int(max(3, n))
    seconds = float(max(0.0, seconds))
    if seconds <= 0.0:
        a = _project(a0)
        if a is None:
            a = np.ones(n) / n
        m = _next_pow2(2 * n - 1)
        return a, _score_array(a, n, m)

    start = time.monotonic()
    deadline = start + seconds
    rng = np.random.default_rng(int(seed))

    a = _project(a0)
    if a is None:
        a = np.ones(n) / n
    v = np.log(a + 1e-300)
    v = v - float(np.mean(v))
    v = np.clip(v, -20.0, 20.0)

    m = _next_pow2(2 * n - 1)
    m3 = _next_pow2(3 * n - 2)

    cur_sc = _score_vector(v, n, m)
    if not np.isfinite(cur_sc):
        a = np.ones(n) / n
        v = np.zeros(n, dtype=np.float64)
        cur_sc = _score_vector(v, n, m)

    best_v = v.copy()
    best_sc = cur_sc
    best_a = _soft(v)
    last_improve = start
    lr = lr0
    mg = None
    stuck_limit = max(12.0, min(120.0, 0.22 * seconds))
    tf_choices = (0.08, 0.20, 0.45, 0.80, 1.00)

    while time.monotonic() < deadline - 0.2:
        now = time.monotonic()
        progress = min(1.0, max(0.0, (now - start) / max(1e-9, seconds)))

        if not np.isfinite(cur_sc) or cur_sc > best_sc + 0.05:
            if rng.random() < 0.35:
                v = best_v.copy()
                cur_sc = best_sc
                mg = None
                lr = lr0

        base_p = p_start + (p_end - p_start) * progress
        p = max(1.0, float(base_p * rng.uniform(0.45, 2.0)))
        if p > 150.0:
            p = 150.0

        top_frac = float(tf_choices[int(rng.integers(0, len(tf_choices)))])
        top_frac = max(0.03, min(1.0, top_frac * rng.uniform(0.5, 2.0)))

        sc_now, a_now, gv, M = _active_direction(v, n, m, m3, p, top_frac, rng)
        if np.isfinite(sc_now):
            cur_sc = sc_now
            if cur_sc < best_sc - 1e-12:
                best_sc = cur_sc
                best_v = v.copy()
                best_a = _soft(v)
                last_improve = time.monotonic()

        if gv is None:
            break

        if mg is None:
            mg = gv.copy()
        else:
            mag = momentum * mg + (1.0 - momentum) * gv
            mg2 = _normalize_dir(mag)
            if mg2 is None:
                mg = gv.copy()
            else:
                mg = mg2

        improved = False
        for d in (mg, gv):
            v2, sc2, s2 = _try_steps(v, d, lr, n, m, cur_sc)
            if s2 > 0.0:
                v = v2
                cur_sc = sc2
                lr = min(2.0, max(0.05, float(s2)))
                improved = True
                if cur_sc < best_sc - 1e-12:
                    best_sc = cur_sc
                    best_v = v.copy()
                    best_a = _soft(v)
                    last_improve = time.monotonic()
                break

        if not improved:
            lr = max(0.03, lr * 0.6)
            if time.monotonic() - last_improve > stuck_limit:
                sigma = 0.08 + 0.35 * (1.0 - progress)
                v = best_v.copy()
                Knoise = int(min(80, max(8, n // 24)))
                noise = _smooth_noise(n, rng, Knoise, sigma)
                v = v + noise
                v = v - float(np.mean(v))
                v = np.clip(v, -20.0, 20.0)
                mg = None
                lr = lr0
                cur_sc = _score_vector(v, n, m)
                if not np.isfinite(cur_sc):
                    v = best_v.copy()
                    cur_sc = best_sc
            else:
                v = v + rng.normal(0.0, 0.03, n)
                v = v - float(np.mean(v))
                v = np.clip(v, -20.0, 20.0)
                mg = None
                cur_sc = _score_vector(v, n, m)
                if not np.isfinite(cur_sc):
                    v = best_v.copy()
                    cur_sc = best_sc

    final_sc = _score_vector(v, n, m)
    if np.isfinite(final_sc) and final_sc < best_sc:
        best_sc = final_sc
        best_v = v.copy()
    best_a = _project(_soft(best_v))
    fin = _score_array(best_a, n, m)
    if np.isfinite(fin) and fin < best_sc:
        best_sc = fin
    return best_a, float(best_sc)


def _binom_sequence(n):
    n = int(max(1, n))
    C = np.ones(n, dtype=np.float64)
    for i in range(1, n):
        C[i] = C[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return C


def _analytic_inits(n, rng):
    n = int(max(3, n))
    res = []
    t = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)

    C = _binom_sequence(n)
    a = _project(C)
    if a is not None:
        res.append(a)

    for q in (0.60, 0.80, 1.20, 1.50):
        a = _project(np.power(C, q))
        if a is not None:
            res.append(a)

    Crev = C[::-1].copy()
    for r in (0.02, 0.05, 0.10, 0.18):
        a = _project((1.0 - r) * C + r * Crev)
        if a is not None:
            res.append(a)

    for alpha in (0.45, 0.65, 0.90, 1.15):
        a = _project(np.power(i + 1.0, -alpha))
        if a is not None:
            res.append(a)

    for c in (0.25, 0.6, 1.0, 2.0):
        a = _project(np.exp(-c * t))
        if a is not None:
            res.append(a)

    v = np.log(C + 1e-300)
    v = v - float(np.mean(v))
    noise = _smooth_noise(n, rng, 22, 0.14)
    a = _project(_soft(v + noise))
    if a is not None:
        res.append(a)

    v = np.log(C + 1e-300)
    v = v - float(np.mean(v))
    v = v + 0.18 * np.cos(2.0 * math.pi * 3.0 * t + rng.uniform(0.0, 2.0 * math.pi))
    a = _project(_soft(v))
    if a is not None:
        res.append(a)

    return res


def _interp_log(a_old, n):
    old = _project(a_old)
    if old is None or old.size == 0:
        return None
    if old.size == n:
        return old
    v = np.log(old + 1e-300)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, n)
    v_new = np.interp(x_new, x_old, v)
    v_new = v_new - float(np.mean(v_new))
    return _project(_soft(v_new))


def _add_candidate(scored, seen, a, n, m, threshold=0.05, cap=50):
    if len(seen) >= cap:
        return False
    a = _project(a)
    if a is None or a.size != n:
        return False
    sc = _score_array(a, n, m)
    if not np.isfinite(sc):
        return False
    d2thr = threshold * threshold
    for b in seen:
        d = float(np.dot(a - b, a - b))
        if d < d2thr:
            return False
    scored.append((sc, a))
    seen.append(a)
    return True


def _make_targets(B, n):
    L = 2 * n - 1
    B = np.asarray(B, dtype=np.float64).ravel()
    if B.size != L:
        B = np.resize(B, L)
    B = np.maximum(B, 0.0)
    s = float(B.sum())
    if not np.isfinite(s) or s <= 0.0:
        return [np.ones(L, dtype=np.float64) / float(L)]
    B = B / s
    flat = np.ones(L, dtype=np.float64) / float(L)

    out = [B.copy()]
    for p in (0.5, 0.75):
        t = np.power(B, p)
        t = np.maximum(t, 0.0)
        ts = float(t.sum())
        if np.isfinite(ts) and ts > 0.0:
            out.append(t / ts)
    for beta in (0.25, 0.45, 0.65):
        t = (1.0 - beta) * B + beta * flat
        ts = float(t.sum())
        if np.isfinite(ts) and ts > 0.0:
            out.append(t / ts)
    return out


def _spectral_factor(target, a_start, n, iters, blend=0.30):
    n = int(max(3, n))
    L = 2 * n - 1
    m = _next_pow2(L)
    target = np.asarray(target, dtype=np.float64).ravel()
    if target.size < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:target.size] = target
        target = tmp
    else:
        target = target[:L].copy()
    target = np.maximum(target, 0.0)
    ts = float(target.sum())
    if not np.isfinite(ts) or ts <= 0.0:
        target = np.ones(L, dtype=np.float64) / float(L)
    else:
        target = target / ts
    T = np.fft.fft(target, m)

    a = _project(a_start)
    if a is None:
        a = np.ones(n) / float(n)
    best = a.copy()
    best_sc = _score_array(a, n, m)
    iters = int(max(1, iters))

    for it in range(iters):
        a = np.maximum(a, 1e-300)
        sa = float(a.sum())
        if sa <= 0.0 or not np.isfinite(sa):
            a = np.ones(n) / float(n)
        else:
            a = a / sa

        F = np.fft.fft(a, m)
        fmax = float(np.max(np.abs(F)))
        eps = 1e-10 * max(1.0, fmax)
        Fnew = 0.5 * (F + T / (F + eps))
        anew = np.fft.ifft(Fnew).real[:n]
        pos = np.maximum(anew, 0.0)
        ps = float(pos.sum())
        if ps > 1e-12:
            pos = pos / ps
        else:
            pos = a.copy()

        b = max(0.12, min(0.85, blend * (1.0 - 0.45 * (float(it) / float(iters)))))
        cand = (1.0 - b) * a + b * pos
        cand = _project(cand)
        if cand is None:
            cand = a
        sc = _score_array(cand, n, m)
        if np.isfinite(sc) and sc < best_sc:
            best = cand.copy()
            best_sc = sc
        a = cand

    return best, float(best_sc)


def _lp_candidates(a, B, M, active_frac, max_cons):
    if not _HAS_LINPROG:
        return []
    a = _project(a)
    if a is None:
        return []
    n = a.size
    L = B.size
    M = float(M)
    if not np.isfinite(M) or M <= 0.0:
        return []

    N = int(min(L, max_cons, max(50, int(active_frac * L))))
    if N <= 0:
        return []

    if N >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(B, -N)[-N:]
    idx.sort()

    A = np.zeros((N, n), dtype=np.float64)
    for r in range(N):
        x = int(idx[r])
        lo = x - n + 1
        if lo < 0:
            lo = 0
        hi = x
        if hi >= n:
            hi = n - 1
        if hi >= lo:
            seg_lo = x - hi
            seg_hi = x - lo + 1
            if seg_hi > seg_lo:
                A[r, lo:hi + 1] = a[seg_lo:seg_hi][::-1]

    c = np.ones(n, dtype=np.float64)
    b = np.full(N, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        sol = linprog(-c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:
        return []
    if sol is None or sol.x is None:
        return []
    if int(getattr(sol, "status", -1)) != 0:
        return []

    g = np.asarray(sol.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if not np.isfinite(gs) or gs <= 0.0:
        return []
    g = g / gs
    g = _project(g)
    if g is None:
        return []

    out = [g]
    for tt in (0.25, 0.50, 0.80):
        mix = (1.0 - tt) * a + tt * g
        p = _project(mix)
        if p is not None:
            out.append(p)
    return out


def _process_length(n, deadline, prev_best, rng):
    n = int(max(3, n))
    now = time.monotonic()
    if now >= deadline:
        return prev_best, None
    L = 2 * n - 1
    m = _next_pow2(L)

    scored = []
    seen = []
    thresh = 0.045

    a_in = _interp_log(prev_best, n)
    if a_in is not None:
        _add_candidate(scored, seen, a_in, n, m, threshold=thresh)

    for ain in _analytic_inits(n, rng):
        _add_candidate(scored, seen, ain, n, m, threshold=thresh)

    if not scored:
        u = np.ones(n, dtype=np.float64) / float(n)
        scored.append((_score_array(u, n, m), u))
        seen.append(u)

    scored.sort(key=_pair_score)
    local_a = scored[0][1]
    local_sc = scored[0][0]

    now = time.monotonic()
    rem = deadline - now
    if rem > 70.0 and n <= 8192:
        a_best, B_best, sc_best = _score_and_B_array(local_a, n, m)
        if a_best is not None:
            local_a = a_best
            local_sc = sc_best

        targets = _make_targets(B_best, n) if B_best is not None else []

        C = _project(_binom_sequence(n))
        if C is not None:
            _, B_C, _ = _score_and_B_array(C, n, m)
            if B_C is not None:
                targets.extend(_make_targets(B_C, n))

        if targets:
            spec_eta = max(20.0, min(110.0, 0.10 * rem))
            spec_deadline = min(deadline - 30.0, time.monotonic() + spec_eta)
            max_runs = 7 if n <= 2048 else (5 if n <= 4096 else 4)
            iters = 28 if n <= 1024 else (20 if n <= 2500 else 12)
            runs = 0
            local_best_init = local_a

            # Take targets in a useful order: raw best, powered/flatten, one-sided mixes.
            # We have stored that order; just iterate up to max_runs.
            for T in targets:
                if time.monotonic() >= spec_deadline or runs >= max_runs:
                    break
                asp, scp = _spectral_factor(T, local_best_init, n, iters, blend=0.28)
                if asp is not None and _add_candidate(scored, seen, asp, n, m,
                                                       threshold=0.04):
                    runs += 1
                    if np.isfinite(scp) and scp < local_sc:
                        local_sc = scp
                        local_a = asp
                        local_best_init = asp

    if _HAS_LINPROG and n <= 2500 and (deadline - time.monotonic()) > 140.0:
        scored.sort(key=_pair_score)
        for sc, ain in scored[:2]:
            a2, B2, sc2 = _score_and_B_array(ain, n, m)
            if B2 is None:
                continue
            M2 = float(B2.max())
            for frac in (0.15, 0.40):
                for gc in _lp_candidates(a2, B2, M2, frac, 320):
                    _add_candidate(scored, seen, gc, n, m, threshold=0.04)
        scored.sort(key=_pair_score)
        if scored and scored[0][0] < local_sc:
            local_sc = scored[0][0]
            local_a = scored[0][1]

    scored.sort(key=_pair_score)
    top = []
    topseen = []
    for sc, ain in scored:
        diverse = True
        for b in topseen:
            if float(np.dot(ain - b, ain - b)) < 0.05 * 0.05:
                diverse = False
                break
        if diverse:
            top.append((sc, ain))
            topseen.append(ain)
        if len(top) >= 4:
            break

    if not top:
        u = np.ones(n, dtype=np.float64) / float(n)
        top.append((_score_array(u, n, m), u))

    weights = (0.55, 0.30, 0.15)
    for ii in range(min(3, len(top))):
        now = time.monotonic()
        rem = deadline - now - 8.0
        if rem < 25.0:
            break
        alloc = float(rem) * weights[ii]
        if ii == 0:
            alloc = max(alloc, min(rem, 60.0))
        alloc = max(12.0, min(alloc, rem * 0.9))
        seed = int(rng.integers(0, 2 ** 31 - 1))
        aopt, scopt = _optimize_active(top[ii][1], n, alloc, seed=seed)
        if aopt is not None and np.isfinite(scopt):
            if scopt < local_sc:
                local_sc = scopt
                local_a = aopt

    now = time.monotonic()
    rem = deadline - now - 6.0
    if rem > 60.0:
        seed = int(rng.integers(0, 2 ** 31 - 1))
        aopt, scopt = _optimize_active(local_a, n, rem * 0.9, seed=seed)
        if aopt is not None and np.isfinite(scopt) and scopt < local_sc:
            local_sc = scopt
            local_a = aopt

    local_a = _project(local_a)
    if local_a is None:
        local_a = np.ones(n, dtype=np.float64) / float(n)
        local_sc = _score_array(local_a, n, m)
    local_sc = _score_array(local_a, n, m)
    if not np.isfinite(local_sc):
        local_a = np.ones(n, dtype=np.float64) / float(n)
        local_sc = 2.0
    return local_a, float(local_sc)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    seed = int(seed)
    budget_s = float(max(120.0, budget_s))
    start = time.monotonic()
    horizon = max(180.0, budget_s - 45.0)
    global_deadline = start + horizon
    work_deadline = min(global_deadline, start + max(60.0, budget_s - 110.0))
    rng = np.random.default_rng(seed)

    best_a = None
    best_sc = np.inf

    try:
        cand = height_sequence_1
        arr = np.asarray(cand, dtype=np.float64).ravel()
        arr = _project(arr)
        if arr is not None and arr.size > 0:
            n0 = arr.size
            m0 = _next_pow2(2 * n0 - 1)
            best_a = arr
            best_sc = _score_array(best_a, n0, m0)
            print("Initial previous score: %.6f length=%d" % (best_sc, n0))
    except Exception:
        best_a = None

    if best_a is None:
        n0 = 512
        best_a = np.ones(n0, dtype=np.float64) / float(n0)
        m0 = _next_pow2(2 * n0 - 1)
        best_sc = _score_array(best_a, n0, m0)
        print("No previous sequence; uniform score=%.6f" % best_sc)

    stage_defs = [
        (512, 0.02),
        (1024, 0.03),
        (2048, 0.34),
        (4096, 0.28),
        (8192, 0.24),
        (16384, 0.09),
    ]

    total_span = max(1.0, work_deadline - start)
    cum = 0.0
    for nn, frac in stage_defs:
        now = time.monotonic()
        if now + 25.0 >= work_deadline:
            break
        cum += frac
        raw_end = start + total_span * min(1.0, cum)
        stage_deadline = min(work_deadline - 5.0, raw_end)
        if stage_deadline <= now + 15.0:
            continue
        stage_time = max(1.0, stage_deadline - now)
        print("Stage n=%d budget=%.1fs current_best=%.6f" %
              (nn, stage_time, best_sc))
        a_new, sc_new = _process_length(int(nn), stage_deadline, best_a, rng)
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_sc - 1e-12:
            best_a = a_new
            best_sc = float(sc_new)
        print("After stage n=%d best=%.6f" % (nn, best_sc))

    now = time.monotonic()
    rem = global_deadline - now
    if rem > 80.0 and best_a is not None:
        nbest = int(best_a.size)
        mbest = _next_pow2(2 * nbest - 1)
        if _score_array(best_a, nbest, mbest) < np.inf:
            seed_polish = int(rng.integers(0, 2 ** 31 - 1))
            a_pol, sc_pol = _optimize_active(best_a, nbest, rem * 0.7,
                                             seed=seed_polish)
            if a_pol is not None and np.isfinite(sc_pol) and sc_pol < best_sc - 1e-12:
                best_a = a_pol
                best_sc = float(sc_pol)
            print("After final polish best=%.6f length=%d" %
                  (best_sc, int(best_a.size)))

    best_a = _project(best_a)
    if best_a is None or best_a.size == 0:
        best_a = np.ones(512, dtype=np.float64) / 512.0

    nfin = int(best_a.size)
    mfin = _next_pow2(2 * nfin - 1)
    fft_sc = _score_array(best_a, nfin, mfin)
    print("Final FFT score=%.6f length=%d" % (fft_sc, nfin))

    # Verify for moderately sized outputs; avoid direct O(n^2) for huge lengths.
    try:
        if nfin <= 10000 and time.monotonic() < global_deadline - 5.0:
            exact = float(evaluate_sequence([float(x) for x in best_a]))
            print("Final exact score=%.6f length=%d" % (exact, nfin))
            if not np.isfinite(exact):
                fallback = np.ones(512, dtype=np.float64) / 512.0
                best_a = fallback
                print("Exact invalid; fallback uniform.")
    except Exception as e:
        print("Exact validation skipped: %s" % e)

    return [float(x) for x in best_a]
