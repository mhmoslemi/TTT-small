def make_inits(n, parent, rng):

```python
"""
Search for a non-negative step-function height sequence minimizing

    score = 2 n max_k (a * a)_k / (sum(a))^2

where * is the linear convolution.  The implementation combines:

  * FFT-based fast internal evaluation,
  * a squared-spectrum / autocorrelation water-filling projection
    (project the convolution into the capped-speed domain, then lift it
    back to the time domain through a spectral square-root update),
  * multiplicative mirror-descent-style gradient steps with soft peak
    and variance weights,
  * diverse initializations and multi-length exploration.

The routine returns the best list of non-negative floats found before
the deadline.
"""

import time
import numpy as np

try:
    from scipy.fft import fft, ifft
except Exception:
    from numpy.fft import fft, ifft


def _next_pow2(m):
    """Smallest power of two >= m."""
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _clean_array(a, max_val=1000.0):
    """Convert to float array and enforce finite values / range."""
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=max_val, neginf=0.0)
    a = np.clip(a, 0.0, max_val)
    return a


def _normalize_to_sum(a, target_sum, max_val=1000.0):
    """Normalize non-negative array to target_sum, with mild clipping."""
    a = _clean_array(a, max_val)
    n = len(a)
    if n == 0:
        return a
    target_sum = float(target_sum)
    if target_sum <= 0.0:
        return np.zeros(n, dtype=np.float64)

    s = float(np.sum(a))
    if s <= 1e-15:
        a = np.full(n, target_sum / float(n), dtype=np.float64)
        a = np.clip(a, 0.0, max_val)
        return a

    a = a * (target_sum / s)
    a = np.clip(a, 0.0, max_val)

    s2 = float(np.sum(a))
    if s2 <= 1e-15:
        a = np.full(n, target_sum / float(n), dtype=np.float64)
        a = np.clip(a, 0.0, max_val)
    else:
        a = a * (target_sum / s2)
        a = np.clip(a, 0.0, max_val)
    return a


def _score_fast(a):
    """Fast FFT-based score. Uses the same quantity as evaluate_sequence."""
    a = _clean_array(a)
    n = len(a)
    if n == 0:
        return np.inf
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = fft(a, N)
    b = ifft(fa * fa).real[:L]
    b = np.maximum(b, 0.0)
    max_b = float(np.max(b))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _upsample_array(a, n):
    """Linear interpolation between normalized positions [0,1]."""
    a = _clean_array(a)
    if len(a) == 0:
        return np.zeros(n, dtype=np.float64)
    if len(a) == n:
        return a.copy()
    if len(a) == 1:
        return np.full(n, float(a[0]), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, len(a), dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return np.interp(x_new, x_old, a)


def _make_inits(n, base, rng):
    """Several deterministic and randomized initializations."""
    S = float(np.sqrt(2.0 * n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    arts = []

    # stable baseline
    arts.append(np.ones(n, dtype=np.float64))

    # parent / current best resampled
    if base is not None and len(base) > 0:
        arts.append(_upsample_array(base, n))

    # one-sided power functions
    for p in (0.35, 0.7, 1.0, 1.7, 3.0):
        xp = np.power(np.maximum(x, 1e-12), p)
        xr = np.power(np.maximum(1.0 - x, 1e-12), p)
        arts.append(0.15 + xp)
        arts.append(0.15 + xr)
        arts.append(0.6 + 0.5 * xp + 0.5 * xr)

    # cosine / sine shapes
    arts.append(1.0 + 0.55 * np.cos(np.pi * x))
    arts.append(1.0 + 0.45 * np.cos(2.0 * np.pi * x))
    arts.append(1.0 + 0.50 * np.sin(np.pi * x))
    arts.append(1.0 + 0.35 * np.sin(3.0 * np.pi * x))

    # stochastic positive shapes
    arts.append(rng.exponential(1.0, size=n) + 0.05)
    arts.append(1.0 + 0.45 * rng.standard_normal(n))
    arts.append(np.abs(rng.standard_normal(n)) + 0.2)

    # two-level random
    breakpt = float(rng.uniform(0.2, 0.8))
    l1 = float(rng.uniform(0.75, 1.65))
    l2 = float(rng.uniform(0.75, 1.65))
    arts.append(np.where(x < breakpt, l1, l2))

    # three-level random
    b1 = float(rng.uniform(0.2, 0.45))
    b2 = float(rng.uniform(0.55, 0.80))
    l31 = float(rng.uniform(0.8, 1.7))
    l32 = float(rng.uniform(0.8, 1.7))
    l33 = float(rng.uniform(0.8, 1.7))
    arts.append(np.select([x < b1, (x >= b1) & (x < b2), x >= b2],
                          [l31, l32, l33]))

    out = []
    for art in arts:
        art = _clean_array(art)
        out.append(_normalize_to_sum(art, S))
    return out


def _pick_lowest(scored, k):
    """Return the k lowest (score, array) pairs, without lambdas."""
    items = []
    for sc, arr in scored:
        try:
            sc = float(sc)
        except Exception:
            continue
        if np.isfinite(sc) and arr is not None:
            items.append((sc, arr))

    m = len(items)
    for i in range(m):
        for j in range(i + 1, m):
            if items[j][0] < items[i][0]:
                items[i], items[j] = items[j], items[i]

    if k <= 0:
        return []
    return items[:k]


def _consider_candidate(arr, sc, best_arr, best_score, keep, keep_size=5):
    """Track global best and a small pool of promising alternatives."""
    if arr is None:
        return best_arr, best_score, keep
    try:
        sc = float(sc)
    except Exception:
        return best_arr, best_score, keep
    if not np.isfinite(sc):
        return best_arr, best_score, keep

    if sc < best_score - 1e-12:
        if best_arr is not None and len(keep) < keep_size:
            keep.append((best_score, best_arr.copy()))
        best_arr = np.array(arr, dtype=np.float64, copy=True)
        best_score = sc
    else:
        if sc < best_score + 0.08:
            if len(keep) < keep_size:
                keep.append((sc, arr.copy()))
            else:
                worst_idx = 0
                for i in range(1, len(keep)):
                    if keep[i][0] > keep[worst_idx][0]:
                        worst_idx = i
                if sc < keep[worst_idx][0] - 1e-12:
                    keep[worst_idx] = (sc, arr.copy())

    return best_arr, best_score, keep


def _waterfill_cap(b, T, target_sum):
    """
    Project a non-negative convolution vector into
    { b' >= 0, sum b' = target_sum, max b' <= T }
    using an L2-like water-filling clip.
    """
    b = np.asarray(b, dtype=np.float64).ravel()
    L = len(b)
    if L == 0:
        return b

    b = np.nan_to_num(b, nan=0.0, posinf=max(float(T), 1.0), neginf=0.0)
    b = np.maximum(b, 0.0)
    b = 0.5 * (b + b[::-1])

    s = float(np.sum(b))
    if s <= 1e-15:
        return np.full(L, target_sum / float(L), dtype=np.float64)

    b = b * (target_sum / s)
    avg = target_sum / float(L)
    T = max(float(T), avg + 1e-12)

    if float(np.max(b)) <= T * (1.0 + 1e-10):
        return b

    # Choose shift >= 0 such that sum(clip(b + shift, 0, T)) = target_sum.
    # For shift=0 after clipping peaks, sum <= target_sum.
    # For shift=T-min(b), all entries become T, sum = L*T >= target_sum.
    lo = 0.0
    hi = max(float(T) - float(np.min(b)), 1e-9)

    for _ in range(40):
        mid = 0.5 * (lo + hi)
        cand = np.clip(b + mid, 0.0, T)
        cs = float(np.sum(cand))
        if cs < target_sum:
            lo = mid
        else:
            hi = mid

    shift = 0.5 * (lo + hi)
    out = np.clip(b + shift, 0.0, T)
    os = float(np.sum(out))
    if os <= 1e-15:
        out = np.full(L, target_sum / float(L), dtype=np.float64)
    else:
        out = out * (target_sum / os)
        out = np.maximum(out, 0.0)
        out = 0.5 * (out + out[::-1])

    # One repair if rounding pushed peaks slightly above T.
    if float(np.max(out)) > T * 1.001:
        out = np.clip(out, 0.0, T * 1.005)
        os = float(np.sum(out))
        if os > 0:
            out = out * (target_sum / os)

    return out


def _run_spectral(a0, time_s, target_score, seed, alpha=0.5):
    """
    Alternating projection style optimizer:
      1) move convolution into capped/water-filled domain,
      2) lift back by spectral square-root updates,
      3) project to nonnegative, first-half support, fixed sum.
    """
    time_s = float(time_s)
    if time_s <= 0.0:
        return np.array(a0, dtype=np.float64, copy=True), np.inf

    t0 = time.monotonic()
    deadline = t0 + time_s

    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf

    S = float(np.sqrt(2.0 * n))
    S2 = S * S
    L = 2 * n - 1
    avg = S2 / float(L)
    lower = avg + 1e-9

    a = _normalize_to_sum(a, S)
    best = a.copy()
    best_score = _score_fast(a)

    target_score = max(float(target_score), lower)
    cap = max(lower, min(float(target_score), max(lower, best_score * 0.995)))
    if cap >= best_score:
        cap = max(lower, best_score * 0.99)

    N = _next_pow2(L)
    rng = np.random.default_rng(seed)
    floor = max(1e-14, 1e-12 * S / float(n))

    alpha_cur = float(np.clip(alpha, 0.05, 0.95))
    bad_run = 0
    last_improve = 0
    it = 0

    while time.monotonic() < deadline:
        fa = fft(a, N)
        fa2 = fa * fa
        b = ifft(fa2).real[:L]
        b = np.maximum(b, 0.0)
        b = 0.5 * (b + b[::-1])
        sb = float(np.sum(b))
        if not np.isfinite(sb) or sb <= 0.0:
            break
        b = b * (S2 / sb)

        max_b = float(np.max(b))
        sa = float(np.sum(a))
        sc = float(2.0 * n * max_b / (sa * sa)) if sa > 0.0 else np.inf

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best = a.copy()
            last_improve = it
            new_cap = max(lower, best_score - 0.002)
            cap = min(cap, new_cap)

        if np.isfinite(sc) and sc > best_score + 0.05:
            bad_run += 1
        else:
            bad_run = max(0, bad_run - 1)

        b_tar = _waterfill_cap(b, cap, S2)
        btar = fft(b_tar, N)

        ratio = np.ones(N, dtype=np.complex128)
        abs_fa2 = np.abs(fa2)
        mask = abs_fa2 > 1e-10
        ratio[mask] = btar[mask] / fa2[mask]

        rad = np.abs(ratio)
        finite = np.isfinite(rad)
        rad_clip = np.ones(N, dtype=np.float64)
        rad_clip[finite] = np.clip(rad[finite], 1e-6, 1e4)

        ang = np.angle(ratio)
        ang[~finite] = 0.0

        # Multiply current spectrum by a controlled square-root correction.
        factor_mag = np.exp(0.5 * np.log(rad_clip))
        half_ang = 0.5 * ang
        sqrt_factor = factor_mag * (np.cos(half_ang) + 1j * np.sin(half_ang))
        sqrt_factor[~finite] = 1.0

        sf_mag = np.abs(sqrt_factor)
        badmag = (sf_mag > 50.0) | (sf_mag < 0.02) | (sf_mag <= 0.0)
        if np.any(badmag):
            inv = np.zeros_like(sf_mag)
            inv[badmag] = 1.0 / np.maximum(sf_mag[badmag], 1e-30)
            sqrt_factor[badmag] = sqrt_factor[badmag] * inv[badmag]

        fa_new = fa * sqrt_factor
        a_inv = ifft(fa_new).real
        a_proj = a_inv[:n]
        a_proj = np.nan_to_num(a_proj, nan=0.0, posinf=1000.0, neginf=0.0)
        a_proj = np.maximum(a_proj, 0.0)

        sp = float(np.sum(a_proj))
        if not np.isfinite(sp) or sp <= 1e-12:
            a_proj = a.copy()
        else:
            a_proj = a_proj * (S / sp)
            a_proj = np.clip(a_proj, floor, 1000.0)
            sp2 = float(np.sum(a_proj))
            if sp2 <= 1e-12:
                a_proj = a.copy()
            else:
                a_proj = a_proj * (S / sp2)
                a_proj = np.clip(a_proj, 0.0, 1000.0)

        a_new = (1.0 - alpha_cur) * a + alpha_cur * a_proj
        a_new = np.nan_to_num(a_new, nan=0.0, posinf=1000.0, neginf=0.0)
        a_new = np.maximum(a_new, floor)
        sn = float(np.sum(a_new))
        if not np.isfinite(sn) or sn <= 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new * (S / sn)
            a_new = np.clip(a_new, floor, 1000.0)
            sn2 = float(np.sum(a_new))
            if sn2 <= 1e-12:
                a_new = a.copy()
            else:
                a_new = a_new * (S / sn2)
                a_new = np.clip(a_new, 0.0, 1000.0)

        a = a_new
        it += 1

        # If stuck, add multiplicative noise around the best point and loosen.
        if it - last_improve > 600:
            noise = rng.normal(0.0, 0.25, size=n)
            a = best * np.exp(np.clip(noise, -2.0, 2.0))
            a = _normalize_to_sum(a, S)
            alpha_cur = 0.5
            cap = max(cap, max(lower, best_score * 0.995))
            last_improve = it
            bad_run = 0

        if bad_run >= 6:
            alpha_cur = max(0.2, 0.65 * alpha_cur)
            bad_run = 0
            # Sometimes revert to the best configuration to recover.
            if rng.random() < 0.5:
                a = best.copy()

    final_sc = _score_fast(a)
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc
        best = a.copy()

    return best, best_score


def _run_gradient(a0, time_s, seed, mode, beta_soft, var_mix, power,
                  eta, beta_mom, floor_frac, eval_interval=18):
    """
    Multiplicative mirror-descent step for a weighted sum of convolution
    peaks.

    mode=0: variance weights w = b - mean(b)
    mode=1: softmax peak weights
    mode=2: mixture of variance and softmax peak weights
    """
    time_s = float(time_s)
    if time_s <= 0.0:
        return np.array(a0, dtype=np.float64, copy=True), np.inf

    t0 = time.monotonic()
    deadline = t0 + time_s

    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf

    S = float(np.sqrt(2.0 * n))
    S2 = S * S
    L = 2 * n - 1
    floor = max(1e-14, float(floor_frac) * S / float(n))

    a = _normalize_to_sum(a, S)
    best = a.copy()
    best_score = _score_fast(a)

    # Need length for cross-correlation h_i = sum_j w_{i+j} a_j.
    N = _next_pow2(3 * n - 2)
    rng = np.random.default_rng(seed)
    mom = np.zeros(n, dtype=np.float64)

    eta_cur = float(eta)
    beta_mom = float(np.clip(beta_mom, 0.0, 0.99))
    var_mix = float(var_mix)
    power = float(power)
    beta_soft = float(beta_soft)
    last_improve = 0
    bad_run = 0
    it = 0

    while time.monotonic() < deadline:
        fa = fft(a, N)
        b = ifft(fa * fa).real[:L]
        b = np.maximum(b, 0.0)
        b = 0.5 * (b + b[::-1])

        sb = float(np.sum(b))
        if not np.isfinite(sb) or sb <= 0.0:
            break

        max_b = float(np.max(b))
        sa = float(np.sum(a))
        sc = float(2.0 * n * max_b / (sa * sa)) if sa > 0.0 else np.inf

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best = a.copy()
            last_improve = it

        if np.isfinite(sc) and sc > best_score + 0.04:
            bad_run += 1
        else:
            bad_run = max(0, bad_run - 1)

        avg = sb / float(L)
        w_var = b - avg
        if abs(power - 1.0) > 1e-12:
            w_var = np.sign(w_var) * np.power(np.abs(w_var), power)

        if int(mode) == 0:
            w = w_var
        elif int(mode) == 1:
            z = beta_soft * (b - max(max_b, 1e-12))
            z = z - float(np.max(z))
            sw = np.exp(z)
            sws = float(np.sum(sw))
            if not np.isfinite(sws) or sws <= 0.0:
                sw = np.full(L, 1.0 / float(L))
            else:
                sw = sw / sws
            w = sw - 1.0 / float(L)
        else:
            z = beta_soft * (b - max(max_b, 1e-12))
            z = z - float(np.max(z))
            sw = np.exp(z)
            sws = float(np.sum(sw))
            if not np.isfinite(sws) or sws <= 0.0:
                sw = np.full(L, 1.0 / float(L))
            else:
                sw = sw / sws
            w_soft = sw - 1.0 / float(L)
            w = var_mix * w_var + (1.0 - var_mix) * w_soft

        w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
        wrms = float(np.sqrt(np.mean(w * w)))
        if not np.isfinite(wrms) or wrms <= 1e-14:
            w = np.zeros_like(w)
        else:
            w = w / wrms

        fa_rev = fft(a[::-1], N)
        fw = fft(w, N)
        h = ifft(fa_rev * fw).real[n - 1:n + n]
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)

        sa = float(np.sum(a))
        if sa <= 0.0:
            break
        c = float(np.dot(a, h) / sa)
        dirv = c - h
        dscale = float(np.sqrt(np.mean(dirv * dirv)))
        if not np.isfinite(dscale) or dscale <= 1e-12:
            dscale = 1.0
        dirv = dirv / dscale

        mom = beta_mom * mom + (1.0 - beta_mom) * dirv
        step = eta_cur * mom
        step = np.nan_to_num(step, nan=0.0, posinf=5.0, neginf=-5.0)
        step = np.clip(step, -5.0, 5.0)

        a_new = a * np.exp(step)
        a_new = np.nan_to_num(a_new, nan=0.0, posinf=1000.0, neginf=0.0)
        a_new = np.maximum(a_new, floor)
        a_new = np.minimum(a_new, 1000.0)

        sn = float(np.sum(a_new))
        if not np.isfinite(sn) or sn <= 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new * (S / sn)
            a_new = np.clip(a_new, floor, 1000.0)
            sn2 = float(np.sum(a_new))
            if sn2 <= 1e-12:
                a_new = a.copy()
            else:
                a_new = a_new * (S / sn2)
                a_new = np.clip(a_new, 0.0, 1000.0)

        a = a_new
        it += 1

        if it - last_improve > max(1500, 80 * int(eval_interval)):
            noise = rng.normal(0.0, 0.30, size=n)
            a = best * np.exp(np.clip(noise, -2.0, 2.0))
            a = _normalize_to_sum(a, S)
            mom = np.zeros(n, dtype=np.float64)
            eta_cur = max(0.05, 0.65 * eta_cur)
            last_improve = it
            bad_run = 0

        if bad_run >= 6:
            eta_cur = max(0.04, 0.55 * eta_cur)
            bad_run = 0
            if rng.random() < 0.35:
                a = best.copy()
                mom = np.zeros(n, dtype=np.float64)

        if it % max(1, int(eval_interval)) == 0:
            # minimal bookkeeping; avoids extra convolutions
            pass

    final_sc = _score_fast(a)
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc
        best = a.copy()

    return best, best_score


def _perfect_random_seed(base_seed):
    """Small deterministic scrambler for varied sub-runs."""
    return int((base_seed * 1000003 + 101) % (10**9))


def propose_candidate(seed=42, budget_s=2400):
    """
    Entry point invoked by the harness.
    Searches for a high-quality nonnegative sequence within the time budget.
    """
    t_start = time.monotonic()
    try:
        total_time = max(30.0, float(budget_s) - 25.0)
    except Exception:
        total_time = 2400.0
    deadline = t_start + total_time

    # ------------------------------------------------------------------
    # Parent construction, if present
    # ------------------------------------------------------------------
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and parent.ndim == 1 and parent.size > 0:
        parent = np.nan_to_num(parent, nan=0.0, posinf=1000.0, neginf=0.0)
        parent = np.clip(parent, 0.0, 1000.0)
        if np.sum(parent) <= 1e-12:
            parent = None

    if parent is None:
        parent = np.ones(5582, dtype=np.float64)

    # Avoid an excessively long final sequence.
    if parent.size > 20000:
        parent = _upsample_array(parent, 16000)
        parent = _clean_array(parent)

    parent_n = int(parent.size)
    S_parent = float(np.sqrt(2.0 * parent_n))
    best_arr = _normalize_to_sum(parent, S_parent)
    best_score = _score_fast(best_arr)
    keep_pool = []

    print(f"Parent n={parent_n}, initial score={best_score:.7f}")

    rng = np.random.default_rng(int(seed))

    # ------------------------------------------------------------------
    # Phase 1: aggressive refinement of the parent / current best
    # ------------------------------------------------------------------
    phase1 = max(30.0, min(320.0, 0.11 * total_time))
    rem = deadline - time.monotonic()

    if rem > 90:
        ts = phase1 * 0.45
        ts = min(ts, rem - 40)
        if ts > 20:
            target = max(1.0005, min(best_score - 0.016, 1.55))
            arr, sc = _run_spectral(best_arr, ts, target, seed + 1, alpha=0.45)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)
            print(f"Phase1 spectral target~{target:.4f}: {best_score:.7f}")

    rem = deadline - time.monotonic()
    if rem > 80:
        ts = min(phase1 * 0.25, rem - 35)
        if ts > 20:
            arr, sc = _run_gradient(
                best_arr, ts, seed + 11,
                mode=2, beta_soft=6.0, var_mix=0.45, power=1.0,
                eta=0.8, beta_mom=0.85, floor_frac=1e-12)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)
            print(f"Phase1 gradient mix: {best_score:.7f}")

    rem = deadline - time.monotonic()
    if rem > 80:
        ts = min(phase1 * 0.30, rem - 30)
        if ts > 15:
            target = max(1.0005, min(best_score - 0.004, 1.51))
            arr, sc = _run_spectral(best_arr, ts, target, seed + 22, alpha=0.55)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)
            print(f"Phase1 refinement: {best_score:.7f}")

    # ------------------------------------------------------------------
    # Phase 2: low-resolution shape exploration
    # ------------------------------------------------------------------
    low_patterns = []
    left = max(0.0, deadline - time.monotonic() - 10.0)
    low_budget = max(0.0, min(420.0, 0.18 * left))
    low_ns = [64, 128, 256, 512, 1024]
    if low_ns:
        per_low = low_budget / float(len(low_ns))

    for li, nl in enumerate(low_ns):
        rem = deadline - time.monotonic()
        if rem < 70 or low_budget <= 0.0:
            break

        ts_low = max(8.0, min(per_low, rem - 35))
        inits = _make_inits(nl, best_arr, rng)
        scored = []
        for init in inits:
            scor = _score_fast(init)
            scored.append((scor, init))

        top = _pick_lowest(scored, 3)
        if not top:
            continue

        per_run = ts_low / float(len(top)) + 4.0
        for idx, (sc_init, init) in enumerate(top):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            run_t = max(4.0, min(per_run, rem - 30))
            target_low = max(1.05, min(1.38, max(1.1, sc_init - 0.08)))
            arr, sc = _run_spectral(init, run_t, target_low, seed + 1000 + li * 17 + idx, alpha=0.55)
            if np.isfinite(sc) and sc < 1.85:
                if len(low_patterns) < 6:
                    low_patterns.append((sc, arr))
                else:
                    worst_idx = 0
                    for k in range(1, len(low_patterns)):
                        if low_patterns[k][0] > low_patterns[worst_idx][0]:
                            worst_idx = k
                    if sc < low_patterns[worst_idx][0]:
                        low_patterns[worst_idx] = (sc, arr)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)

    # Sort low_patterns quickly
    lp_n = len(low_patterns)
    for i in range(lp_n):
        for j in range(i + 1, lp_n):
            if low_patterns[j][0] < low_patterns[i][0]:
                low_patterns[i], low_patterns[j] = low_patterns[j], low_patterns[i]

    if low_patterns:
        print(f"Low-res best pattern score: {low_patterns[0][0]:.7f}")

    # ------------------------------------------------------------------
    # Phase 3: multi-length, diverse initialization exploitation
    # ------------------------------------------------------------------
    base_n = parent_n
    cand_ns = set()
    for mult in (0.7, 0.85, 1.0, 1.25, 1.6, 2.0, 2.4):
        nn = int(round(base_n * float(mult)))
        if 1024 <= nn <= 12500:
            cand_ns.add(nn)
    for nn in (4096, 8192, 10240, 11264):
        if 1024 <= nn <= 12500:
            cand_ns.add(nn)

    cand_ns = sorted(list(cand_ns))
    if len(cand_ns) > 5:
        # keep a balanced set around parent_n
        keep = set()
        for nn in cand_ns:
            if base_n not in keep:
                keep.add(base_n)
        for nn in cand_ns:
            if len(keep) >= 5:
                break
            if nn not in keep:
                keep.add(nn)
        cand_ns = sorted(list(keep))

    left = max(0.0, deadline - time.monotonic() - 30.0)
    multi_budget = max(0.0, min(1000.0, 0.36 * left))
    if cand_ns and multi_budget > 0:
        per_n_time = multi_budget / float(len(cand_ns))

    for ni, n in enumerate(cand_ns):
        rem = deadline - time.monotonic()
        if rem < 120:
            break

        per_n_time = max(20.0, min(320.0, multi_budget / max(1, len(cand_ns) - ni)))
        ts_budget = min(per_n_time, rem - 50)
        if ts_budget < 40:
            continue

        S = float(np.sqrt(2.0 * n))
        inits = _make_inits(n, best_arr, rng)

        # include current best and good low-res shapes
        inits.append(_normalize_to_sum(_upsample_array(best_arr, n), S))
        if low_patterns:
            inits.append(_normalize_to_sum(_upsample_array(low_patterns[0][1], n), S))
            if len(low_patterns) > 1:
                inits.append(_normalize_to_sum(_upsample_array(low_patterns[1][1], n), S))

        scored = []
        for init in inits:
            scor = _score_fast(init)
            scored.append((scor, init))

        top = _pick_lowest(scored, 3)
        if not top:
            continue

        per_run = max(12.0, ts_budget / float(len(top)))

        for ki, (sc_init, init) in enumerate(top):
            rem = deadline - time.monotonic()
            if rem < 70 or best_score > 1.45 and sc_init > best_score + 0.25:
                # If we already beat target and this init is poor, skip.
                if sc_init > best_score + 0.12:
                    continue
            target = max(1.0005, min(best_score - 0.011, 1.525))
            run_t = min(per_run, rem - 40)
            if run_t < 12:
                continue

            arr, sc = _run_spectral(
                init, run_t, target, seed + 40000 + ni * 101 + ki, alpha=0.5)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)

            rem = deadline - time.monotonic()
            if rem > 80:
                grad_t = max(8.0, min(run_t * 0.35, rem - 40))
                arr2, sc2 = _run_gradient(
                    arr, grad_t, seed + 60000 + ni * 101 + ki,
                    mode=2, beta_soft=10.0, var_mix=0.30, power=1.1,
                    eta=0.6, beta_mom=0.85, floor_frac=1e-12)
                best_arr, best_score, keep_pool = _consider_candidate(
                    arr2, sc2, best_arr, best_score, keep_pool)

        print(f"After length {n}: best={best_score:.7f}")

    # ------------------------------------------------------------------
    # Phase 4: final focus / polishing on global best and alternatives
    # ------------------------------------------------------------------
    focus_start = time.monotonic()
    while time.monotonic() < deadline - 80:
        rem = deadline - 80 - time.monotonic()
        if rem < 90:
            break

        task = max(60.0, min(260.0, rem / 3.0))

        # 1) tight spectral cap
        target = max(1.0005, best_score - 0.006)
        run_t = task * 0.55
        run_t = min(run_t, rem - 45)
        if run_t > 10:
            arr, sc = _run_spectral(best_arr, run_t, target, seed + 90001, alpha=0.42)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)

        rem = deadline - 80 - time.monotonic()
        if rem < 70:
            break

        # 2) gradient with strong peak weights
        run_t = task * 0.45
        run_t = min(run_t, rem - 40)
        if run_t > 10:
            arr, sc = _run_gradient(
                best_arr, run_t, seed + 90002,
                mode=2, beta_soft=22.0, var_mix=0.18, power=1.3,
                eta=0.45, beta_mom=0.80, floor_frac=1e-12)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)

        rem = deadline - 80 - time.monotonic()
        if rem < 70:
            break

        # 3) slightly looser spectral round
        target = max(1.0005, min(best_score - 0.020, 1.500))
        run_t = task * 0.40
        run_t = min(run_t, rem - 35)
        if run_t > 10:
            arr, sc = _run_spectral(best_arr, run_t, target, seed + 90003, alpha=0.58)
            best_arr, best_score, keep_pool = _consider_candidate(
                arr, sc, best_arr, best_score, keep_pool)

        # 4) if alternative pool exists, polish the best alternative once
        if keep_pool and len(keep_pool) > 0:
            # choose worst-free alternative: lowest score in pool
            alt_sc = np.inf
            alt_arr = None
            for sc_k, arr_k in keep_pool:
                if np.isfinite(sc_k) and sc_k < alt_sc:
                    alt_sc = sc_k
                    alt_arr = arr_k
            if alt_arr is not None and len(alt_arr) != len(best_arr):
                rem = deadline - 80 - time.monotonic()
                run_t = max(20.0, min(task * 0.30, rem - 30))
                target_alt = max(1.0005, min(alt_sc - 0.012, 1.520))
                arr, sc = _run_spectral(alt_arr, run_t, target_alt,
                                        seed + 90004, alpha=0.5)
                best_arr, best_score, keep_pool = _consider_candidate(
                    arr, sc, best_arr, best_score, keep_pool)

        # Avoid infinite micro-cycles
        if time.monotonic() - focus_start > 120:
            focus_start = time.monotonic()

    # ------------------------------------------------------------------
    # Final selection and return
    # ------------------------------------------------------------------
    candidates = [(best_score, best_arr)]
    for sc_k, arr_k in keep_pool:
        if arr_k is not None:
            candidates.append((sc_k, arr_k))

    final_score = np.inf
    final_arr = best_arr
    for sc_c, arr_c in candidates:
        arr_c = _clean_array(arr_c)
        if len(arr_c) == 0:
            continue
        sc_fast = _score_fast(arr_c)
        if np.isfinite(sc_fast) and sc_fast < final_score:
            final_score = sc_fast
            final_arr = arr_c

    # Avoid an FFT-free judge eval on a very long vector.
    if len(final_arr) > 18000:
        final_arr = _upsample_array(final_arr, 14000)
        final_arr = _normalize_to_sum(final_arr, float(np.sqrt(2.0 * len(final_arr))))

    final_arr = _clean_array(final_arr)
    final_arr = _normalize_to_sum(
        final_arr, float(np.sqrt(2.0 * len(final_arr)))
    )

    # One really late micro-polish if there is time, but do not break contract.
    rem = deadline - time.monotonic()
    if rem > 120:
        target_final = max(1.0005, _score_fast(final_arr) - 0.003)
        arr, sc = _run_spectral(final_arr, min(100.0, rem - 30),
                                target_final, seed + 99991, alpha=0.35)
        final_arr, final_score, _ = _consider_candidate(
            arr, sc, final_arr, final_score, [])

    final_list = [float(x) for x in np.clip(final_arr, 0.0, 1000.0)]

    try:
        actual = float(evaluate_sequence(final_list))
        print(f"Final fast score={final_score:.7f}, actual evaluate={actual:.7f}")
    except Exception as e:
        print(f"Final fast score={final_score:.7f}, actual eval unavailable: {e}")

    return final_list
```
