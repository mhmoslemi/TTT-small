def topk_subgradient(a, conv, K=30):
    """Compute subgradient using top K peaks of convolution."""
    n = len(a)
    conv_len = len(conv)
    # Get indices of top K peaks
    if K >= conv_len:

        hi = min(n - 1, k)

            grad[lo : hi+1] += 2.0 * segment[::-1]

strong starting point and carefully explores its neighborhood.
"""

import numpy as np
import time
import sys
from scipy.optimize import minimize


def compute_conv(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a)


def compute_score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * conv.max()


def lse_smooth_max(conv, tau):
    """Log-sum-exp smoothed max of conv."""
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    return cmax + tau * np.log(np.exp(shifted).sum() + 1e-300)


def lse_smooth_gradient(a, tau):
    """Gradient of LSE-smoothed max(conv(a,a))/(sum a)^2 w.r.t. a (rho=1 normalization)."""
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient of smoothed max w.r.t. a: sum_k w_k * grad_k
    grad = np.zeros(n)
    # For each lag k, grad_j(C_k) = 2*a[k-j] if 0<=k-j<n
    for k in range(len(conv)):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo < hi + 1:
            segment = a[k-hi:k-lo+1][::-1]
            grad[lo:hi+1] += 2.0 * w[k] * segment
    
    # Project gradient: we're optimizing on simplex (sum=1), so remove mean component
    # Actually for the objective 2*n*max(conv)/(sum a)^2 with rho fixed=1:
    # d/d a_j [2n*smooth_max] = 2n * grad_smooth[j]
    # But we constraint sum=1, so effective gradient is grad[j] - mean(grad)
    grad -= grad.mean()
    return grad


def topk_subgradient(a, conv, K=30):
    """Compute subgradient of max(conv) using top K peaks."""
    n = len(a)
    conv_len = len(conv)
    K = min(K, conv_len)
    
    if K >= conv_len:
        top_idx = np.arange(conv_len)
    else:
        top_idx = np.argpartition(conv, -K)[-K:]
    
    grad = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo < hi + 1:
            segment = a[k-hi:k-lo+1][::-1]
            grad[lo:hi+1] += 2.0 * segment
    grad /= K
    return grad


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 1e-30:
        return v * target / s
    return np.ones(len(v)) * target / len(v)


def run_lse_lbfgs(a_init, tau_schedule, max_time_per_tau, seed=0):
    """Run L-BFGS with LSE smoothing at decreasing temperatures."""
    n = len(a_init)
    a = project_simplex(a_init.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score_norm(a)
    deadline = time.monotonic() + max_time_per_tau * len(tau_schedule)
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        tau_deadline = time.monotonic() + max_time_per_tau
        
        def objective(x):
            ax = project_simplex(x, 1.0)
            conv = compute_conv(ax)
            return lse_smooth_max(conv, tau)
        
        def gradient(x):
            ax = project_simplex(x, 1.0)
            return lse_smooth_gradient(ax, tau)
        
        try:
            result = minimize(
                objective, a, jac=gradient,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': 600, 'maxls': 30}
            )
            a = project_simplex(result.x, 1.0)
        except Exception as e:
            print(f"  L-BFGS error at tau={tau}: {e}")
            break
        
        score = compute_score_norm(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  LSE(tau={tau:.2e}): score={score:.6f}")
    
    return best_a, best_score


def run_subgradient(a_init, num_iters, K=40, step0=0.01, seed=0, 
                    perturb_every=3000, perturb_scale=0.004):
    """Subgradient method with periodic perturbations."""
    n = len(a_init)
    a = project_simplex(a_init.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score_norm(a)
    
    rng = np.random.default_rng(seed)
    step = step0
    prev_grad = None
    prev_a = None
    stalled = 0
    
    for it in range(num_iters):
        conv = compute_conv(a)
        score = 2.0 * n * conv.max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stalled = 0
        else:
            stalled += 1
        
        grad = topk_subgradient(a, conv, K)
        
        # BB step size
        if prev_grad is not None:
            dx = a - prev_a
            dg = grad - prev_grad
            num = np.dot(dx, dx)
            den = np.dot(dx, dg)
            if den > 1e-18:
                bb = num / den
                step = 0.8 * step + 0.2 * np.clip(bb, 1e-5, 0.02)
        
        prev_grad = grad.copy()
        prev_a = a.copy()
        
        a_new = a - step * grad
        a_new = project_simplex(a_new, 1.0)
        
        # Periodic perturbation to escape local minima
        if it % perturb_every == perturb_every - 1:
            noise = rng.normal(0, perturb_scale, n)
            a_new = a_new + noise
            a_new = project_simplex(a_new, 1.0)
        
        # Simulated annealing: occasionally accept worse solutions
        if stalled > 5000 and rng.random() < 0.1:
            direction = rng.normal(0, 1, n)
            a_new = a + 0.01 * direction
            a_new = project_simplex(a_new, 1.0)
            score_new = compute_score_norm(a_new)
            if score_new < best_score * 1.05:  # Accept if within 5%
                a = a_new
            else:
                a = a_new
        else:
            a = a_new
        
        step *= 0.9997
    
    return best_a, best_score


def coordinate_descent_polish(a_init, max_iters, step=0.002, seed=0):
    """Coordinate descent: adjust one element at a time."""
    n = len(a_init)
    a = project_simplex(a_init.copy(), 1.0)
    best_a = a.copy()
    best_score = compute_score_norm(a)
    rng = np.random.default_rng(seed)
    
    for it in range(max_iters):
        # Pick a random coordinate
        j = rng.integers(0, n)
        
        # Try small increase and decrease
        for delta in [step, -step, 2*step, -2*step]:
            a_trial = a.copy()
            a_trial[j] = max(0.0, a_trial[j] + delta)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial /= s
            score = compute_score_norm(a_trial)
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_score


def propose_candidate(budget_s=2400, **kwargs):
    """Main search: minimize evaluate_sequence over non-negative sequences.
    
    Multi-phase approach:
    1. LSE-smoothed L-BFGS from parent + diverse inits
    2. Exact subgradient polishing with adaptive steps
    3. Simulated annealing perturbations
    4. Coordinate descent final polish
    """
    deadline = time.monotonic() + budget_s - 30
    
    # Get parent
    if 'height_sequence_1' in globals():
        parent = np.array(height_sequence_1, dtype=np.float64)
    else:
        parent = np.ones(4806)
    
    n = len(parent)
    parent = np.maximum(parent, 0)
    parent_sum = parent.sum()
    if parent_sum < 0.01:
        parent = np.ones(n)
        parent_sum = n
    parent_norm = parent / parent_sum
    
    best_a = parent_norm.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"n={n}, Parent score: {best_score:.6f}")
    sys.stdout.flush()
    
    rng = np.random.default_rng(777)
    
    # Phase 1: LSE-smoothed optimization from parent and diverse inits
    # Time budget: 40% for phase 1
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.40
    
    tau_schedule = [0.3/n, 0.15/n, 0.07/n, 0.03/n, 0.01/n]
    time_per_tau = 15.0
    
    # Phase 1a: LSE from parent
    print("\nPhase 1a: LSE from parent...")
    time_left = phase1_deadline - time.monotonic()
    a1, s1 = run_lse_lbfgs(parent_norm, tau_schedule, max_time_per_tau=time_per_tau, seed=1)
    actual_s1 = evaluate_sequence(a1.tolist())
    if actual_s1 < best_score:
        best_score = actual_s1
        best_a = a1.copy()
        print(f"  Parent LSE: {best_score:.6f}")
    sys.stdout.flush()
    
    # Phase 1b: LSE from diverse inits
    if time.monotonic() < phase1_deadline - 30:
        print("\nPhase 1b: LSE from diverse inits...")
        inits = []
        
        # Asymmetric splits
        for ratio in [1.5, 2.0, 3.0]:
            x = np.ones(n)
            x[:n//3] = ratio
            x[n//3:] = 1.0/ratio
            inits.append(x / x.sum())
        
        # Cosine modulated
        for freq in [1, 2, 3, 5]:
            x = 1.0 + 0.5 * np.cos(2*np.pi*freq*np.arange(n)/n)
            x = np.maximum(x, 0.05)
            inits.append(x / x.sum())
        
        # Dyadic
        x = np.ones(n)
        q = n // 4
        x[0:q] = 3.0
        x[q:2*q] = 1.5
        x[2*q:3*q] = 0.8
        x[3*q:n] = 0.4
        inits.append(x / x.sum())
        
        # Ramp
        x = np.linspace(4.0, 0.2, n)
        inits.append(x / x.sum())
        
        # Random smooth
        for _ in range(2):
            noise = rng.normal(0, 1, n)
            kernel = np.ones(max(3, n//50)) / max(3, n//50)
            smooth = np.convolve(noise, kernel, mode='same')
            x = np.maximum(1.0 + 0.6 * smooth, 0.05)
            inits.append(x / x.sum())
        
        # Best so far (warm start)
        inits.append(project_simplex(best_a, 1.0))
        
        for idx, init in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            time_left = phase1_deadline - time.monotonic()
            if time_left < 20:
                break
            # Use shorter tau schedule for non-parent inits
            short_tau = tau_schedule[2:] if time_left < 120 else tau_schedule
            a_try, s_try = run_lse_lbfgs(init, short_tau, max_time_per_tau=min(time_per_tau, time_left/len(short_tau)), seed=100+idx)
            actual = evaluate_sequence(a_try.tolist())
            if actual < best_score:
                best_score = actual
                best_a = a_try.copy()
                print(f"  Init {idx}: {best_score:.6f}")
            sys.stdout.flush()
    
    # Phase 2: Subgradient polishing (50% of remaining time)
    remaining = deadline - time.monotonic()
    phase2_deadline = time.monotonic() + remaining * 0.50
    print(f"\nPhase 2: Subgradient polishing. Best: {best_score:.6f}")
    sys.stdout.flush()
    
    # Multiple subgradient rounds with different K and step settings
    subgrad_configs = [
        {'K': 20, 'step0': 0.02, 'perturb_every': 2000, 'perturb_scale': 0.003},
        {'K': 40, 'step0': 0.01, 'perturb_every': 3000, 'perturb_scale': 0.004},
        {'K': 60, 'step0': 0.008, 'perturb_every': 4000, 'perturb_scale': 0.005},
        {'K': 100, 'step0': 0.005, 'perturb_every': 5000, 'perturb_scale': 0.006},
    ]
    
    for cfg_idx, cfg in enumerate(subgrad_configs):
        if time.monotonic() > phase2_deadline:
            break
        time_left = phase2_deadline - time.monotonic()
        # Estimate iterations: ~2ms per iter for n=4806
        num_iters = int(time_left / 0.003)
        num_iters = max(5000, min(num_iters, 200000))
        
        print(f"  Subgrad config {cfg_idx}: K={cfg['K']}, iters={num_iters}")
        a2, s2 = run_subgradient(best_a, num_iters, seed=200+cfg_idx, **cfg)
        actual = evaluate_sequence(a2.tolist())
        if actual < best_score:
            best_score = actual
            best_a = a2.copy()
            print(f"  Subgrad {cfg_idx}: NEW BEST {best_score:.6f}")
        else:
            print(f"  Subgrad {cfg_idx}: {actual:.6f}")
        sys.stdout.flush()
    
    # Phase 3: Simulated annealing + subgradient (remaining time)
    phase3_deadline = deadline - 20
    print(f"\nPhase 3: Annealing + polish. Best: {best_score:.6f}")
    sys.stdout.flush()
    
    a = project_simplex(best_a.copy(), 1.0)
    temperature = 0.01
    step = 0.005
    K = 50
    iter_count = 0
    last_improve = 0
    
    while time.monotonic() < phase3_deadline:
        conv = compute_conv(a)
        score = 2.0 * n * conv.max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            last_improve = iter_count
            if iter_count % 10000 == 0:
                print(f"  Phase3 iter {iter_count}: {best_score:.6f}")
                sys.stdout.flush()
        
        grad = topk_subgradient(a, conv, K)
        a = a - step * grad
        a = project_simplex(a, 1.0)
        
        # Annealing perturbation
        if iter_count % 2000 == 1999:
            noise_scale = temperature * (1.0 + 0.5 * iter_count / 100000)
            a = a + rng.normal(0, noise_scale, n)
            a = project_simplex(a, 1.0)
            temperature *= 0.97
            step = max(0.001, step * 0.98)
        
        # If stalled, try a bigger jump
        if iter_count - last_improve > 30000 and rng.random() < 0.05:
            jump = rng.normal(0, 0.02, n)
            a_candidate = project_simplex(a + jump, 1.0)
            s_candidate = compute_score_norm(a_candidate)
            if s_candidate < best_score * 1.02:
                a = a_candidate
                # Reset stall counter partially
                last_improve = iter_count - 15000
        
        iter_count += 1
    
    # Phase 4: Final coordinate descent polish
    if time.monotonic() < deadline - 10:
        print("\nPhase 4: Coordinate descent...")
        cd_deadline = deadline - 10
        cd_iters = int((cd_deadline - time.monotonic()) / 0.001)
        cd_iters = max(10000, min(cd_iters, 100000))
        a4, s4 = coordinate_descent_polish(best_a, cd_iters, step=0.001, seed=999)
        actual = evaluate_sequence(a4.tolist())
        if actual < best_score:
            best_score = actual
            best_a = a4.copy()
            print(f"  CD: NEW BEST {best_score:.6f}")
        else:
            print(f"  CD: {actual:.6f}")
        sys.stdout.flush()
    
    # Final validation
    final_score = evaluate_sequence(best_a.tolist())
    if not np.isfinite(final_score):
        # Fallback to uniform
        best_a = np.ones(n)
        final_score = evaluate_sequence(best_a.tolist())
    
    print(f"\n=== FINAL SCORE: {final_score:.6f} ===")
    sys.stdout.flush()
    
    result = np.clip(best_a, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(n) * 0.01 / n
    
    return [float(v) for v in result]
